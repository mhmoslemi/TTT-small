import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

_gn = 47
_gt = 23.5

def _obj(h):
    h = np.clip(h, 0.0, 1.0)
    return compute_c5(h)

def _con_eq(h):
    return np.sum(h) - _gt

def _round_to_near_binary(h, n):
    """Round h to near-binary: values at 0, 0.5, or 1."""
    h_r = np.where(h > 0.75, 1.0, np.where(h < 0.25, 0.0, 0.5))
    return make_feasible(h_r, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _gn, _gt

    cons = [{'type': 'eq', 'fun': _con_eq}]

        except Exception:
