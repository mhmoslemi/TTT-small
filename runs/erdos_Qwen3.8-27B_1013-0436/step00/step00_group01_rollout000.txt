            if c5_new < best_c5:

                best_h = h_new
        except:

    if time_left() > 60:

Let me also add a "local search" phase where I try all possible single-coordinate swaps to see if any improve.

Actually, for the binary case, a good local search is: try swapping each pair (i, j) where h[i]=1 and h[j]=0, and see if it improves. This is O(n²) per iteration but very effective for binary problems.

def binary_local_search(h, target_sum, max_iter=100):
    """Local search for binary sequences"""
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    
    for _ in range(max_iter):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                new_c5 = compute_c5(h)
                if new_c5 < cur:
                    cur = new_c5
                    improved = True
                    break
                h[i], h[j] = h[j], h[i]
            if improved:
                break
    
    return h, cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def make_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-10:
        return np.full(len(h), target_sum / len(h))
    h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    return h

def sa_optimize(h0, target_sum, budget_s, rng):
    start = time.time()
    h = h0.copy()
    cur = compute_c5(h)
    best = h.copy()
    best_c5 = cur
    T = 0.003
    T_min = 1e-7
    n = len(h)
    
    while time.time() - start < budget_s:
        h_new = h.copy()
        r = rng.random()
        
        if r < 0.35:
            i = rng.integers(n)
            h_new[i] += rng.uniform(-0.3, 0.3)
        elif r < 0.65:
            i, j = rng.choice(n, 2, replace=False)
            d = rng.uniform(-0.3, 0.3)
            h_new[i] += d
            h_new[j] -= d
        else:
            i = rng.integers(n)
            h_new[i] = 1.0 if h[i] >= 0.5 else 0.0
        
        h_new = make_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < cur or (c5_new - cur) / T < np.log(rng.random()):
            h = h_new
            cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h.copy()
        
        T = max(T_min, T * 0.99999)
    
    return best, best_c5

def binary_local_search(h, budget_s, rng):
    n = len(h)
    h = h.copy()
    cur = compute_c5(h)
    start = time.time()
    
    while time.time() - start < budget_s:
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                new_c5 = compute_c5(h)
                if new_c5 < cur:
                    cur = new_c5
                    improved = True
                    break
                h[i], h[j] = 1.0, 0.0
            if improved:
                break
        if not improved:
            break
    
    return h, cur

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(47, 0.5)
    
    n0 = len(h_init)
    ts0 = n0 / 2.0
    h_init = make_feasible(h_init, ts0)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Phase 1: SLSQP
    if time_left() > 120:

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts0}]
        try:
            res = minimize(compute_c5, h_init, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 300, 'ftol': 1e-14})
            h_new = make_feasible(res.x, ts0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
    
    # Phase 2: SA on initial n
    if time_left() > 90:

        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    return best_h, best_c5, len(best_h)
