    """Project x onto {0 <= h <= 1, sum(h) = target_sum}"""

    lo, hi = -1, 1  # bounds on lambda
    for _ in range(100):
        mid = (lo + hi) / 2
        p = np.clip(x - mid, 0, 1)
        s = np.sum(p)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    p = np.clip(x - (lo + hi) / 2, 0, 1)

        mask = (p > 0) & (p < 1)
        if mask.sum() > 0:
            p[mask] += diff / mask.sum()
            p = np.clip(p, 0, 1)
    return p

If j ∉ A: -|A| * ∂λ/∂z_j = 0, so ∂λ/∂z_j = 0 (assuming h_j is at a bound and stays there)

J[i,j] = 1 - 1/|A| if i, j ∈ A
J[i,j] = 0 otherwise

This is a rank-1 perturbation of the identity on the active set!

So the gradient of F w.r.t. z is:
∇z F = J^T * ∇h F

where ∇h F is the gradient of F w.r.t. h (which I already computed).

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project(x, target):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(x - (lo + hi) * 0.5, 0, 1)
    d = target - p.sum()
    if abs(d) > 1e-12:
        m = (p > 1e-9) & (p < 1.0 - 1e-9)
        if m.any():
            p[m] += d / m.sum()
            p = np.clip(p, 0, 1)
    return p

def smax_grad(h, n, tau):
    g = 1.0 - h
    dx = 2.0 / n
    c = np.correlate(h, g, mode='full')
    f = c * dx
    mf = f.max()
    ef = np.exp(tau * (f - mf))
    es = ef.sum()
    sig = ef / es
    F = (float(mf) + float(np.log(es))) / tau
    grad = np.zeros(n)
    for m in range(len(f)):
        w = sig[m] * dx
        if w < 1e-13:
            continue
        lag = m - (n - 1)
        a = max(0, lag)
        b = min(n, n + lag)
        c2 = max(0, -lag)
        d2 = min(n, n - lag)
        if b > a:
            grad[a:b] += w * (1.0 - h[np.arange(a, b) - lag])
        if d2 > c2:
            grad[c2:d2] -= w * h[np.arange(c2, d2) + lag]
    return F, grad

def pgd(h0, n, tau, tlimit, lr0=None):
    tgt = n * 0.5
    h = h0.copy()
    bh = h.copy()
    bc5 = compute_c5(h, n)
    lr = lr0 if lr0 else 2.0 / (tau * n)
    mom = np.zeros(n)
    beta = 0.6
    t0 = time.time()
    it = 0
    while time.time() - t0 < tlimit and it < 50000:
        it += 1
        F, grad = smax_grad(h, n, tau)
        mom = beta * mom + (1.0 - beta) * grad
        a = lr
        ht = project(h - a * mom, tgt)
        for _ in range(15):
            Ft, _ = smax_grad(ht, n, tau)
            sd = np.dot(mom, h - ht)
            if sd <= 1e-15 or Ft <= F - 1e-4 * a * sd:
                break
            a *= 0.5
            ht = project(h - a * mom, tgt)
        h = ht
        c5 = compute_c5(h, n)
        if c5 < bc5:
            bc5 = c5
            bh = h.copy()
        if a > lr * 0.2 and it % 100 == 0:
            lr = min(lr * 1.05, 20.0 / (tau * n))
    return bh, bc5

    for tau in [5, 10, 25, 60, 150, 400]:
        el = time.time() - T0

            break
        tl = (budget_s * 0.45 - el)
        if tl < 2:
            break
        h, c5 = pgd(bh, n, tau, tl * 0.4)
        if c5 < bc5:
            bc5 = c5
            bh = h

    # Phase 2: Random restarts
    for _ in range(40):
        el = time.time() - T0
        if el > budget_s * 0.7:
            break
        tl = budget_s * 0.7 - el
        if tl < 1:
            break
        hr = project(rs.uniform(0, 1, n), n * 0.5)
        h, c5 = pgd(hr, n, 50, tl * 0.025)
        if c5 < bc5:
            bc5 = c5
            bh = h

    # Phase 3: Try slightly different n
    for nt in [44, 46, 60, 90]:
        el = time.time() - T0
        if el > budget_s * 0.92:
            break
        tl = budget_s * 0.92 - el
        if tl < 10:
            break
        xo = np.linspace(0, 2, bn, endpoint=False)
        xn = np.linspace(0, 2, nt, endpoint=False)
        ht = np.interp(xn, xo, bh)
        ht = project(ht, nt * 0.5)
        h, c5 = pgd(ht, nt, 30, tl * 0.25)
        if c5 < bc5:
            bc5 = c5
            bh = h
            bn = nt
        h2, c52 = pgd(ht, nt, 100, tl * 0.2)
        if c52 < bc5:
            bc5 = c52
            bh = h2
            bn = nt

    return bh, bc5, bn
