def smooth_c5(h, T=200.0):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corrs.max()
    # Shift for numerical stability
    return m + np.log(np.sum(np.exp(T * (corrs - m)))) / T
