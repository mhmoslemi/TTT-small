from scipy.optimize import minimize, Bounds

def run_packing():
    # Initial grid-based placement
    n_circles = 26
    grid = np.indices((5, 5)).reshape((25, 2))
    centers = np.vstack((grid, np.array([0.5, 0.5]))).reshape((26, 2))
    radii = np.full(n_circles, 0.05)

    def objective(radii_input):
        return -np.sum(radii_input)  # Maximize sum of radii by minimizing negative sum

    def constraints_function(radii_input):
        # All circles must be within the unit square
        within_bounds = np.all((centers[:, 0] - radii_input >= -1e-12) & 
                              (centers[:, 0] + radii_input <= 1 + 1e-12) & 
                              (centers[:, 1] - radii_input >= -1e-12) & 
                              (centers[:, 1] + radii_input <= 1 + 1e-12))
        # No overlap between circles
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii_input[i] + radii_input[j] - 1e-12:
                    return False
        return within_bounds

    # Define the optimization problem
    bounds = [(0.0, 1.0) for _ in range(n_circles)]

    # Create an optimization problem with bounds and constraints
    result = minimize(

        radii,
        bounds=bounds,
        constraints=[{"type": "ineq", "fun": constraints_function}],
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    )
