def run_packing():

    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Compute horizontal and vertical spacing
    horizontal_spacing = 1.0 / (cols_per_row - 1)

    for i in range(num_rows):

            centers[i * cols_per_row + j] = [x, y]

    # We'll compute the maximum possible radius such that the circles don't exceed the unit square or overlap
    # Following a standard hexagonal packing, each circle has radius r
    # The horizontal extent is horizontal_spacing = 2r => r = horizontal_spacing / 2
    # The vertical spacing should be vertical_spacing = sqrt(3) * r
    # We ensure that the circles fit within the square
