import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist
from scipy.optimize import LinearConstraint, Bounds
import matplotlib.pyplot as plt

def create_initial_packing():
    # Initial guess: 16 circles in a grid and 10 circles in a hexagonal pattern
    # First 16 circles in a 4x4 grid
    grid = np.array([[
        [i / 3, j / 3] for j in range(4)
    ] for i in range(4)])
    grid = np.reshape(grid, (16, 2))
    
    # Then 10 circles in a hexagonal pattern
    hex_grid = np.array([[
        [i / 4, j / 4] for j in range(4)
    ] for i in range(4)])
    hex_grid = np.reshape(hex_grid, (16, 2))
    hex_grid[::2] = hex_grid[::2] + np.array([0.1, 0.1])  # Offset to simulate hexagonal packing
    
    # Combine all
    centers = np.vstack([grid, hex_grid[:10]])
    radii = np.zeros(26)
    return centers, radii

def run_packing():
    # Initial guess
    centers, radii = create_initial_packing()
    
    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    # Constraints
    def constraint_func(radii):
        # Ensure all circles are within the unit square
        for i in range(26):
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12:
                return np.inf
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12:
                return np.inf
        # Ensure circles don't overlap
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return 0
    
    # Set up optimization problem
    bounds = [(0, 1) for _ in range(26)]
    
    # Use Sequential Least Squares Programming (SLSQP)
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_func}],
        method='SLSQP'
    )
    
    # Extract results
    optimized_radii = result.x
    sum_radii = np.sum(optimized_radii)
    
    return centers, optimized_radii, sum_radii
