import cvxpy as cp

def run_packing():

    # 10 central cells and 16 corner/edge cells (for a total of 26 circles)

    # Parameters for initial spacing
    spacing = 0.2
    n = 26

    # Initial center positions
    centers = np.zeros((n, 2))

    # Place 16 circles near corners and edges (quarter grid)
    idx = 0
    for i in range(4):
        for j in range(4):
            if i == 0 or i == 3 or j == 0 or j == 3:
                centers[idx] = np.array([i * spacing, j * spacing])
                idx += 1

    # Place 10 circles in a hexagonal pattern in the center

        r = 0.2 + 0.05 * k
        angle = 2 * np.pi * k / 10
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)
        centers[idx] = np.array([x, y])
        idx += 1

    # Initial radii (simple estimate)
    radii = np.zeros(n)
    for i in range(n):
        if i < 16:
            radii[i] = 0.15
        else:
            radii[i] = 0.1

    # Define optimization variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Constraints
    constraints = []
    for i in range(n):
        constraints.append(cp.Minimize(r[i]))
        constraints.append(x[i, 0] >= r[i])
        constraints.append(x[i, 0] <= 1 - r[i])
        constraints.append(x[i, 1] >= r[i])
        constraints.append(x[i, 1] <= 1 - r[i])

    for i in range(n):

    # Objective: maximize total radius
    obj = cp.Maximize(cp.sum(r))

    # Solve the problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver='SCS', eps=1e-6)
