from scipy.optimize import minimize
from scipy.optimize import Bounds

def compute_sum_of_radii(radii):

def compute_circle_positions(n):

        y = (i // 5) * 0.3 + (i % 5) * 0.1
        grid.append((x, y))
    return np.array(grid)

def dist(x1, y1, x2, y2):
    return np.sqrt((x1 - x2)**2 + (y1 - y2)**2)

    n = centers.shape[0]
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -tol or x + r > 1 + tol or y - r < -tol or y + r > 1 + tol:
            return False
    for i in range(n):
        for j in range(i + 1, n):
            d = dist(centers[i, 0], centers[i, 1], centers[j, 0], centers[j, 1])
            if d < radii[i] + radii[j] - tol:
                return False
    return True

def optimize_circle_positions_and_radii(n):
