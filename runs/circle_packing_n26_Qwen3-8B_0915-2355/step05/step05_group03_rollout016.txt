import scipy.optimize as optimize

def run_packing():

    # We start with a slightly denser arrangement with repulsion between all circles

    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])

    # Initial guess for the radii, with a wide range for flexibility
    radii = np.full(26, 0.1)

    # Objective function to maximize the sum of the radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize negative sum for maximization

    # Distance between centers
    def dist(i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx * dx + dy * dy)

    # Constraint for non-overlapping
    def constraint_overlap(i, j):
        def constraint(r):
            return dist(i, j) - (r[i] + r[j])
        return constraint

    # Constraint for circles to stay inside the unit square
    def constraint_boundary(i):
        x, y = centers[i]
        def constraint(r):
            return np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])
        return constraint

    # Collect constraints
    constraints = []
    for i in range(26):
        for j in range(i + 1, 26):
            constraints.append({
                'type': 'ineq',
                'fun': constraint_overlap(i, j)
            })
    for i in range(26):
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary(i)
        })

    # Bounds for radii
    bounds = [(0, 0.5) for _ in range(26)]

    # Run the optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-10
    )

    # Extract results
    optimum_radii = result.x
    total_sum = np.sum(optimum_radii)

    return (centers, optimum_radii, total_sum)
