import scipy.optimize as optimize

        for row in range(n):
            for col in range(n):
                x = col / (n - 1)
                y = row / (n - 1)
                grid.append((x, y))
        # Remove the middle if needed
        grid = np.array(grid)
        # Shift to make the grid fit within [0, 1]^2
        grid = grid / np.max(grid)
        return grid

    grid = generate_hexagonal_grid(7)
    grid = grid[:26]

    # Ensure the grid is within [0, 1]^2
    grid = np.clip(grid, 0, 1)

    def distance_to_edge(x, y):
        return np.min([x, 1 - x, y, 1 - y])

    initial_radii = np.array([distance_to_edge(x, y) for x, y in centers])

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlapping circles
    def constraint_overlap(i, j):
        def fun(r):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            return np.sqrt(dx * dx + dy * dy) - (r[i] + r[j])
        return fun

    # Constraint for boundary conditions
    def constraint_boundary(i):
        def fun(r):
            x, y = centers[i]
            return np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])
        return fun

    # Define constraints
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):

            'fun': constraint_boundary(i)

    result = optimize.minimize(
