import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Use a hexagonal grid inside the unit square with more points
    num_rows = 7
    num_cols = 7
    grid_x = np.linspace(0.1, 0.9, num_cols)
    grid_y = np.linspace(0.1, 0.9, num_rows)
    grid = np.array([[x, y] for y in grid_y for x in grid_x])
    
    # Add corners as fixable positions
    corner_positions = np.array([
        [0.05, 0.05],
        [0.95, 0.05],
        [0.95, 0.95],
        [0.05, 0.95]
    ])
    grid = np.vstack([grid, corner_positions])
    
    # Trim to 26 positions
    grid = grid[:26]
    
    # Initialize radii based on minimal distance to boundaries
    def distance_to_edge(x, y):
        return np.min([x, 1 - x, y, 1 - y])
    
    initial_radii = np.array([distance_to_edge(x, y) for x, y in grid])
    
    centers = np.array(grid)
    
    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization
    
    # Define constraints as functions
    def constraint_overlap(i, j):
        def fun(r):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            return np.sqrt(dx**2 + dy**2) - (r[i] + r[j])
        return fun
    
    def constraint_boundary(i):
        def fun(r):
            x, y = centers[i]
            return np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])
        return fun
    
    # Create constraints
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})
    
    for i in range(len(centers)):
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})
    
    # Define bounds for radii: all between 0 and 0.5
    bounds = [(0, 0.5)] * len(centers)
    
    # Run optimization with updated settings for faster convergence
    result = optimize.minimize(
        fun=objective,
        x0=initial_radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-12,
        options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    
    # Return centers, radii, and total sum
    optimum_radii = result.x
    total_sum = np.sum(optimum_radii)
    
    return (centers, optimum_radii, total_sum)
