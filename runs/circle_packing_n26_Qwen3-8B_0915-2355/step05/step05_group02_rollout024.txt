import scipy.optimize as optimize

def run_packing():

            grid.append([x, y])

        [0.95, 0.05],
        [0.95, 0.95],
        [0.05, 0.95]

    additional_centers = [

        [0.8, 0.2],
        [0.8, 0.8],
        [0.2, 0.8],
        [0.5, 0.1],
        [0.5, 0.9],
        [0.1, 0.5],
        [0.9, 0.5]
    ]
    grid += additional_centers
    
    # Ensure only 26 centers are used
    grid = grid[:26]
    centers = np.array(grid)
    
    # Initial radii based on distance to nearest edge
    def distance_to_edge(x, y):
        return np.min([x, 1 - x, y, 1 - y])
    
    initial_radii = np.array([distance_to_edge(x, y) for x, y in centers])
    
    # Function to compute distance between two points
    def dist(i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx * dx + dy * dy)
    
    # Objective function: maximize total sum of radii
    def objective(x):
        return -np.sum(x)
    
    # Constraint: distance between centers >= sum of radii
    def constraint_overlap(i, j):
        def fun(r):
            return dist(i, j) - (r[i] + r[j])
        return fun
    
    # Constraint: circle stays within [0,1]x[0,1]
    def constraint_boundary(i):
        def fun(r):
            x, y = centers[i]
            return np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])
        return fun
    
    # Create constraints
    constraints = []

        })
