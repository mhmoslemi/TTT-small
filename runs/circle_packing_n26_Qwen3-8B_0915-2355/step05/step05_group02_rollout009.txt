import scipy.optimize as optimize

    grid = np.zeros((num_rows * num_cols, 2))
    
    for i in range(num_rows):
        for j in range(num_cols):
            x = cols[j]
            y = rows[i]
            grid[i * num_cols + j] = [x, y]

    corners = np.array([
        [0.05, 0.05],
        [0.95, 0.05],
        [0.95, 0.95],
        [0.05, 0.95]
    ])

    # Ensure we have exactly 26 positions
    grid = grid[:26]
    
    # Initial radii based on distance to nearest edge and neighbor
    def distance_to_edge(x, y):
        return np.min([x, 1 - x, y, 1 - y])
    
    def distance_to_neighbor(x, y, grid):
        min_dist = np.inf
        for i, (gx, gy) in enumerate(grid):
            if i != np.argmin(np.hypot(x - gx, y - gy)):
                dist = np.hypot(x - gx, y - gy)
                if dist < min_dist:
                    min_dist = dist
        return min_dist

    initial_radii = np.array([np.min([distance_to_edge(x, y), distance_to_neighbor(x, y, grid)]) for x, y in grid])

    def dist(i, j):
        dx = centers[i, 0] - centers[j, 0]

            return dist(i, j) - (r[i] + r[j])
        return fun
