import scipy.optimize as optimize

def run_packing():

    num_rows = 5
    num_cols = 5
    # Generate positions in a more efficient hexagonal packing
    positions = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = j / (num_cols - 1)
            y = i / (num_rows - 1)
            positions.append([x, y])
    # Add corner circles to fill in potential gaps
    corners = [
        [0.05, 0.05],
        [0.95, 0.05],
        [0.95, 0.95],
        [0.05, 0.95]
    ]
    positions += corners
    # Trim to exactly 26 positions
    positions = positions[:26]
    centers = np.array(positions)
    
    # Define a function to calculate Euclidean distance
    def dist(i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx * dx + dy * dy)
    
    # Initial radii based on distance to boundaries
    def distance_to_boundary(x, y):
        return np.min([x, 1 - x, y, 1 - y])
    
    initial_radii = np.array([distance_to_boundary(x, y) for x, y in centers])
    
    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization
    
    # Constraint for non-overlapping circles: distance >= r_i + r_j
    def constraint_overlap(i, j):
        def fun(r):
            return dist(i, j) - (r[i] + r[j])
        return fun
    
    # Create constraints for non-overlapping circles (i < j)
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            constraints.append({

                'fun': constraint_overlap(i, j)
            })

    def constraint_boundary(i):
        def fun(r):
            x, y = centers[i]
            return np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])
        return fun

    result = optimize.minimize(
