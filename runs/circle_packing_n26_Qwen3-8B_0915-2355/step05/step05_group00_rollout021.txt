             (0.05, 0.45), (0.95, 0.45), (0.05, 0.55), (0.95, 0.55)]
    positions += corners + edges

    positions = list(set(positions))
    positions = [tuple(p) for p in positions]
    # Select first 26 positions
    if len(positions) > 26:
        positions = positions[:26]

    centers = np.array(positions)

    # Define the objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization

    # Define constraints for non-overlap and boundary inspection
    def constraint_overlap(i, j):
        def fun(r):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            return dist - (r[i] + r[j])
        return fun

            right = 1 - x - r[i]
            bottom = y - r[i]
            top = 1 - y - r[i]
            return np.min([left, right, bottom, top])
        return fun

    # Create constraints for non-overlapping circles (i < j)
    constraints = []
    for i in range(len(centers)):
