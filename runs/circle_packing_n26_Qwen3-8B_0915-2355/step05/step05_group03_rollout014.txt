import numpy as np
import scipy.optimize as optimize

def run_packing():

    # Scaled to fit within [0, 1]x[1]
    num_circles = 26
    centers = []
    for i in range(13):
        for j in range(2):
            x = i / 12.0 + 0.05
            y = (j * (1 + 1 / 12.0)) + 0.05
            centers.append([x, y])
    centers = np.array(centers)

    # Initial guess for radii
    radii = np.full(num_circles, 0.05)

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize radii

    # Distance between points
    def dist(i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx * dx + dy * dy)

    # Constraint for non-overlapping circles
    def constraint_overlap(i, j):
        return lambda r: dist(i, j) - (r[i] + r[j])

    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j: constraint_overlap(i, j)(r)
            })

    # Constraint for boundary conditions (keeping circles within unit square)
    def constraint_boundary(i):
        x, y = centers[i]
        return lambda r: np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])

    for i in range(num_circles):

            'fun': lambda r, i=i: constraint_boundary(i)(r)
        })

    # Run the optimization
    result = optimize.minimize(

        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
