import scipy.optimize as optimize

        [0.1, 0.3],        [0.3, 0.3],        [0.5, 0.3],        [0.7, 0.3],        [0.9, 0.3],

        [0.3, 0.35]        # A slightly repositioned circle for corner optimization
    ])

    # Initial guess for the radii
    radii = np.full(26, 0.05)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization

    # Distance between points
    def dist(i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx * dx + dy * dy)

    # Non-overlapping constraints
    def constraint_overlap(i, j):
        return lambda r: dist(i, j) - (r[i] + r[j])

    constraints = []
    for i in range(26):
        for j in range(i + 1, 26):
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j: constraint_overlap(i, j)(r)
            })

    # Boundary constraints
    def constraint_boundary(i):
        x, y = centers[i]
        return lambda r: np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])

    for i in range(26):
        constraints.append({

            'fun': lambda r, i=i: constraint_boundary(i)(r)
        })

    bounds = [(0, 0.5)] * 26

    # Run optimization using COBYLA for robustness and constraint satisfaction
    result = optimize.minimize(

        method='COBYLA',
