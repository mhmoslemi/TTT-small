import scipy.optimize as optimize

def run_packing():

    ]

    inner_grid = []
    for x in np.linspace(0.1, 0.9, 5):
        for y in np.linspace(0.1, 0.9, 5):
            inner_grid.append((x, y))
    # Add hexagonal pattern points around the inner grid
    for x in np.linspace(0.2, 0.8, 4):
        for y in np.linspace(0.2, 0.8, 4):
            inner_grid.append((x, y))
    # Combine all positions
    positions = corners + edges + inner_grid
    # Ensure we only have 26 positions
    if len(positions) > 26:
        positions = positions[:26]
    centers = np.array(positions)

    # Define a function to compute distance between two points
    def dist(p1, p2):
        return np.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

    # Initial guess for radii using distance to boundaries
    def initial_radii_guess(centers):
        radii = []
        for x, y in centers:
            left = x
            right = 1 - x
            bottom = y
            top = 1 - y
            min_dist = min(left, right, bottom, top)
            radii.append(min_dist)
        return np.array(radii)

    initial_radii = initial_radii_guess(centers)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlapping and within square
    def constraint_overlap(i, j):
        return lambda r: dist(centers[i], centers[j]) - (r[i] + r[j])

    def constraint_boundary(i):
        return lambda r: min(centers[i, 0] - r[i], 1 - centers[i, 0] - r[i], centers[i, 1] - r[i], 1 - centers[i, 1] - r[i])

    # Build constraints
    constraints = []
    for i in range(len(centers)):

            })
