            positions.append((x, y))

    # Define corner and edge circles
    corners = [(0.05, 0.05), (0.95, 0.05), (0.95, 0.95), (0.05, 0.95)]
    edges = [(0.5, 0.05), (0.5, 0.95), (0.05, 0.5), (0.95, 0.5)]
    positions += corners + edges

    if len(positions) > 26:
        positions = positions[:26]
    centers = np.array(positions)

    # Define the objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization

    # Define constraints for non-overlap and boundary inspection
    def constraint_overlap(i, j):
        return lambda r: np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2) - (r[i] + r[j])

    def constraint_boundary(i):
        return lambda r: np.min([centers[i, 0] - r[i], 1 - centers[i, 0] - r[i], centers[i, 1] - r[i], 1 - centers[i, 1] - r[i]])

    # Create constraints for non-overlapping circles (i < j)
    constraints = []
    for i in range(len(centers)):

                'fun': lambda r, i=i, j=j: constraint_overlap(i, j)(r)
            })

        })
