import scipy.optimize as optimize

    def generate_hex_lattice(n, max_x, max_y):
        grid = []

        num_cols = int(np.ceil(n / num_rows))
        for i in range(num_rows):
            for j in range(num_cols):
                x = j + 0.5 * i
                y = i + 0.5 * j
                if x < max_x and y < max_y:
                    grid.append([x, y])

        if len(grid) > 26:

        [0.3, 0.3],
        [0.7, 0.3],
        [0.3, 0.7],
        [0.7, 0.7]
    ])
    free_positions = free_positions[:26 - len(centers)]
    centers = np.vstack([centers, free_positions])

    def distance_to_edge(x, y):
        return np.min([x, 1 - x, y, 1 - y])
    
    initial_radii = np.array([np.min([distance_to_edge(x, y), 0.5]) for x, y in centers])

    def dist(i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx * dx + dy * dy)

        return -np.sum(radii)  # Minimize negative sum for maximization

            return dist(i, j) - (r[i] + r[j])
        return fun
