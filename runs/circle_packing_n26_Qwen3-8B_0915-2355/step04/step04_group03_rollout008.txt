import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Baseline number of circles
    n_circles = 26

    # Grid parameters
    grid_rows = 5
    grid_cols = 5
    row_spacing = 0.5
    col_spacing = 0.5
    
    # Generate a hexagonal grid with offset rows
    centers = []
    for i in range(grid_rows):
        row_offset = 0.5 * (i % 2)
        row = []
        for j in range(grid_cols):
            x = j * col_spacing
            y = i * row_spacing + row_offset
            row.append(np.array([x, y]))
        centers.append(row)
    
    # Flatten the centers into a single array for the 26 circles
    centers = np.array([point for row in centers for point in row])

    # Generate the radius configuration based on proximity to boundaries
    radii = np.full(n_circles, 0.0)
    
    # Define the contact with boundaries
    for i in range(n_circles):
        x, y = centers[i]
        if x - 0.1 < 0:
            radii[i] = 0.1
        elif x + 0.1 > 1:
            radii[i] = 0.1
        elif y - 0.1 < 0:
            radii[i] = 0.1
        elif y + 0.1 > 1:
            radii[i] = 0.1
        else:
            # Use a radius based on proximity to a boundary
            min_dist = np.min([x, 1 - x, y, 1 - y])
            radii[i] = min_dist / 2.0

    # Define Euclidean distance
    def distance(a, b):
        dx = a[0] - b[0]
        dy = a[1] - b[1]
        return np.sqrt(dx * dx + dy * dy)
    
    # Objective function (maximize total radius)
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlapping circles
    def circle_overlap(i, j):
        return lambda r: distance(centers[i], centers[j]) - (r[i] + r[j])

    # Constraints for non-overlapping circles
    constraints = []
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j: circle_overlap(i, j)(r)
            })

    # Constraint for keeping circles within the unit square
    def circle_boundary(i):
        x, y = centers[i]
        return lambda r: np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])

    for i in range(n_circles):
        constraints.append({
            'type': 'ineq',
            'fun': lambda r, i=i: circle_boundary(i)(r)
        })

    # Set bounds for radii [0, 0.5]
    bounds = [(0, 0.5)] * n_circles

    # Run the optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-12
    )

    # Extract results
    optimum_radii = result.x
    total_sum = np.sum(optimum_radii)

    return (centers, optimum_radii, total_sum)
