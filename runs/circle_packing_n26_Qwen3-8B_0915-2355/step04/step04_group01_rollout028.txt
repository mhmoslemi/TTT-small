        x = r * np.cos(angles)
        y = r * np.sin(angles)
        return np.column_stack((x, y))

        dy = seeds[i, 1] - seeds[j, 1]
        return np.sqrt(dx*dx + dy*dy)

        return np.min([x, 1 - x, y, 1 - y])
