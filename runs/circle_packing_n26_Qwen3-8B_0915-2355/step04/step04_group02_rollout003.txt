import scipy.optimize as optimize

def run_packing():

    centers = []

    rows = 7
    cols = 4

        for j in range(cols):
            x = j * step + 0.05 * (j + 1)
            y = i * step + 0.05 * (i + 1)
            # Alternate the y-coordinate to form the hexagonal pattern
            if i % 2 == 1:
                y += step / 2
            centers.append((x, y))

    # Add corners with adjusted radii to avoid overlap
    corner_positions = [(0.05, 0.05), (0.95, 0.05), (0.95, 0.95), (0.05, 0.95)]
    centers += corner_positions

    # Add more points near the edges in the second layer
    edge_positions = [
        (0.5, 0.1), (0.5, 0.9), (0.1, 0.5), (0.9, 0.5),
        (0.5, 0.3), (0.5, 0.7), (0.3, 0.5), (0.7, 0.5)
    ]
    centers += edge_positions

    # Keep exactly 26 points
    centers = centers[:26]
    centers = np.array(centers)

    # Initial radii based on minimum distance to boundaries
    def min_boundary_dist(x, y):
        return min(x, 1 - x, y, 1 - y)

    radii = np.array([min_boundary_dist(x, y) for x, y in centers])

    # Function to calculate Euclidean distance between two points
    def dist(i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx * dx + dy * dy)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization

    # Constraint for non-overlapping circles: distance >= r_i + r_j
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):

                d = dist(i, j)
                return d - (r[i] + r[j])
            constraints.append({'type': 'ineq', 'fun': constraint_func})

            x, y = centers[i]

    result = optimize.minimize(
