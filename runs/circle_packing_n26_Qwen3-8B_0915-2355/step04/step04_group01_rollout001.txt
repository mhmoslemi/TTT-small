import scipy.optimize as optimize

    # and boundary-aware placement

    centers = []
    # Bottom-left corner
    centers.append([0.05, 0.05])
    # Top-left corner
    centers.append([0.05, 0.95])
    # Bottom-right corner
    centers.append([0.95, 0.05])
    # Top-right corner
    centers.append([0.95, 0.95])
    # Arrange 22 circles along hexagonal grid in the center
    for i in range(11):
        y = 0.5 + i * 0.15
        for j in range(11):
            x = 0.5 - j * 0.15
            if 0.1 < x < 0.9 and 0.1 < y < 0.9:
                centers.append([x, y])
    # Remove extra circles to get exactly 26
    centers = centers[:26]
    centers = np.array(centers)

    # Calculate initial radii based on distance to boundaries
    def distance_to_boundary(x, y):
        return np.min([x, 1 - x, y, 1 - y])

    initial_radii = [distance_to_boundary(x, y) for x, y in centers]

    # Objective function for maximization (negative for minimization)
    def objective(radii):
        return -np.sum(radii)

    # Distance between two points
    def dist(i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx**2 + dy**2)

    # Constraint for non-overlapping
    def constraint_overlap(i, j):
        def fun(r):
            return dist(i, j) - (r[i] + r[j])
        return fun

    # Constraint for staying within the unit square
    def constraint_boundary(i):
        def fun(r):
            x, y = centers[i]
            return np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])
        return fun

    # Build optimization bounds
    bounds = [(0, 0.5)] * n

    # Build optimization constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({
                'type': 'ineq',
                'fun': constraint_overlap(i, j)
            })

    # Run optimization
    result = optimize.minimize(

        method='SLSQP',
