import scipy.optimize as optimize

    num_cols = 6

    # Create a hexagonal grid of 25 circles

        for j in range(num_cols):
            x = (j + 0.5) / num_cols
            y = (i + 0.5) / num_rows
            grid.append([x, y])
    # At the end, add 4 corner circles to make total 26

    grid.append([0.95, 0.05])
    grid.append([0.95, 0.95])

    centers = np.array(grid)

    def objective(radii):

            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            return dist - (r[i] + r[j])
        return fun

    def constraint_boundary(i):

            return np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])
        return fun

    constraints = []

    result = optimize.minimize(
