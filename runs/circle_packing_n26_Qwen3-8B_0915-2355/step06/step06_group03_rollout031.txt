import scipy.optimize as optimize
import random

def run_packing():

    centers = []
    for _ in range(26):
        x = np.random.uniform(0.1, 0.9)
        y = np.random.uniform(0.1, 0.9)
        centers.append([x, y])
    centers = np.array(centers)

    # Initial radii: based on minimum of distance to nearest boundary
    def distance_to_boundary(x, y):
        return np.min([x, 1 - x, y, 1 - y])

    def objective(radii):

            dist = np.sqrt(dx * dx + dy * dy)
            return dist - (r[i] + r[j])
        return fun

    # Create constraints for non-overlapping circles
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):

            })

    def constraint_boundary(i):
