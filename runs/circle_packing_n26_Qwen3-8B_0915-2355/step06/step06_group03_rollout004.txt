    grid.append([0.625, 0.375])  # bottom-right
    grid.append([0.375, 0.625])  # top-left
    grid.append([0.625, 0.625])  # top-right

    grid = grid[:26]

        return np.min([x, 1 - x, y, 1 - y])

    def objective(radii):
        return -np.sum(radii)
    
    # Constraint for non-overlapping circles: distance >= r_i + r_j
    def constraint_overlap(i, j):

            dist = np.sqrt(dx * dx + dy * dy)
            return dist - (r[i] + r[j])
        return fun
