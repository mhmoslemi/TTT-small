import scipy.optimize as optimize

    num_cols = 5

    spacing_y = 0.2
    centers = []

        for j in range(num_cols):
            x = j * spacing_x
            y = i * spacing_y
            if i % 2 == 0:
                x += spacing_x / 2
            centers.append((x, y))

        centers = centers[:26]
    centers = np.array(centers)

            1 - centers[i, 0] - r[i],

            1 - centers[i, 1] - r[i]

    constraints = []
