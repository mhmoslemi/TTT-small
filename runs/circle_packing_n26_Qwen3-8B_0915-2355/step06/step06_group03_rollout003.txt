import scipy.optimize as optimize

def run_packing():
    # Create a staggered hexagonal grid in the unit square
    # Start with 26 circles arranged in a staggered hexagonal grid
    # Rows are staggered alternately
    num_rows = 6
    num_cols = 4
    centers = []
    
    # Generate staggered hexagonal grid
    for i in range(num_rows):
        for j in range(num_cols):
            x = (j + 0.5) / (num_cols - 1)
            y = (i + 0.5) / (num_rows - 1)
            if i % 2 == 1:
                x += 0.25 / (num_cols - 1)
            centers.append([x, y])
    
    # Add corner circles for aggressive boundary use
    centers.append([0.05, 0.05])
    centers.append([0.95, 0.05])
    centers.append([0.95, 0.95])
    centers.append([0.05, 0.95])

    centers = centers[:26]
    centers = np.array(centers)
    
    # Initial radii: based on distance to nearest boundary
    def distance_to_boundary(x, y):
        return np.min([x, 1 - x, y, 1 - y])
    
    initial_radii = np.array([distance_to_boundary(x, y) for x, y in centers])
    
    # Objective function (minimize negative sum of radii)
    def objective(radii):
        return -np.sum(radii)
    
    # Constraint for non-overlapping circles: distance >= r_i + r_j
    def constraint_overlap(i, j):
        def fun(r):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            return dist - (r[i] + r[j])
        return fun
    
    # Create constraints for non-overlapping circles
    constraints = []
    for i in range(len(centers)):

                'fun': constraint_overlap(i, j)
            })

    def constraint_boundary(i):
        def fun(r):
            x, y = centers[i]
            return np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])
        return fun
    
    for i in range(len(centers)):

        })

    bounds = [(0, 0.5)] * len(centers)

    result = optimize.minimize(

        method='SLSQP',
        bounds=bounds,
