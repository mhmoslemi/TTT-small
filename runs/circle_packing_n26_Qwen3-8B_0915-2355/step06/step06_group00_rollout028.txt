import scipy.optimize as optimize

def run_packing():

    centers = []

    # Corner circles with larger initial radii
    corner_positions = [(0.1, 0.1), (0.9, 0.1), (0.9, 0.9), (0.1, 0.9)]
    for x, y in corner_positions:
        centers.append((x, y))

    # Edge circles with tuned placement
    edge_positions = [
        (0.5, 0.15), (0.5, 0.85),
        (0.15, 0.5), (0.85, 0.5)
    ]
    for x, y in edge_positions:
        centers.append((x, y))

    # Additional hexagonal grid positions based on heuristically guided grid
    grid = np.linspace(0.2, 0.8, 6)
    for x in grid:
        for y in grid:
            if x < 0.4 or y < 0.4 or x > 0.6 or y > 0.6:
                continue
            centers.append((x, y))

    # Ensure we have exactly 26 positions
    if len(centers) > 26:
        centers = centers[:26]
    centers = np.array(centers)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization

    # Constraints for non-overlap and boundary
    def constraint_overlap(i, j):
        x1, y1 = centers[i]
        x2, y2 = centers[j]
        return lambda r: np.sqrt((x1 - x2)**2 + (y1 - y2)**2) - (r[i] + r[j])

    def constraint_boundary(i):
        x, y = centers[i]
        return lambda r: np.array([
            x - r[i],
            1 - x - r[i],
            y - r[i],
            1 - y - r[i]
        ])

    # Build constraints for all pairs of circles
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            constraints.append({
                'type': 'ineq',
                'fun': constraint_overlap(i, j)
            })
    for i in range(len(centers)):
        constraints.append({
            'type': 'ineq',
            'fun': constraint_boundary(i)
        })

    # Bounds for radii: all between 0 and 0.5
    bounds = [(0, 0.5)] * len(centers)

    # Start with equal radii for homogeneous distribution
    initial_radii = np.full(len(centers), 0.1)

    result = optimize.minimize(

            'gtol': 1e-10,

            'iprint': 0
        }

    return (centers, optimum_radii, total_sum)
