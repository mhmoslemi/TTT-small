import scipy.optimize as optimize

def run_packing():
    # Define positions for corner and edge circles
    corners = [
        (0.0, 0.0), (0.0, 1.0), (1.0, 1.0), (1.0, 0.0)
    ]
    edges = [
        (0.0, 0.5), (0.5, 0.0), (1.0, 0.5), (0.5, 1.0)
    ]
    # Additional edge points will be placed in between main edges
    edge_points = []
    for x in np.linspace(0.1, 0.9, 5):
        edge_points.append((x, 0.5))
        edge_points.append((x, 0.5))
    for y in np.linspace(0.1, 0.9, 5):
        edge_points.append((0.5, y))
        edge_points.append((0.5, y))
    # Remove duplicates and ensure exact count
    edge_points = np.unique(edge_points, axis=0).tolist()[:12]
    positions = corners + edges + edge_points

    # Remove duplicates and remainders
    positions = np.array(list(set(positions)))
    if len(positions) > 26:
        positions = positions[:26]
    centers = positions

    # Define the objective function
    def objective(radii):
        return -np.sum(radii)

    # Define constraints
    def constraint_overlap(i, j):
        return lambda r: np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2) - (r[i] + r[j])

    def constraint_boundary(i):
        return lambda r: np.min([centers[i, 0] - r[i], 1 - centers[i, 0] - r[i], centers[i, 1] - r[i], 1 - centers[i, 1] - r[i]])

    # Generate equality constraint for circle radius (optional, for dynamic non-overlap)
    def constraint_radius(i):
        return lambda r: r[i]

    # Set constraints
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            # Non-overlap constraint
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j: constraint_overlap(i, j)(r)
            })
    for i in range(len(centers)):
        constraints.append({
            'type': 'ineq',
            'fun': lambda r, i=i: constraint_boundary(i)(r)
        })

    # Bounds for radii: all radii between 0 and 0.5
    bounds = [(0, 0.5)] * len(centers)
    # Initial guess: start with uniform radii
    initial_radii = np.full(len(centers), 0.1)

    # Run optimization
    result = optimize.minimize(
        fun=objective,
        x0=initial_radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-9,
        options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9, 'eps': 1e-8}
    )

    # Return centers, radii, and sum of radii
    return (centers, result.x, np.sum(result.x))
