import scipy.optimize as optimize

        for y in grid_y:
            positions.append((x, y))

    hex_x = np.linspace(0.3, 0.7, 5)
    hex_y = np.linspace(0.3, 0.7, 5)

        for y in hex_y:
            positions.append((x, y))

    # Phase 3: Edge and corner placements
    edges = [(0.05, 0.5), (0.5, 0.05), (0.95, 0.5), (0.5, 0.95)]
    corners = [(0.05, 0.05), (0.95, 0.05), (0.95, 0.95), (0.05, 0.95)]
    for edge in edges:
        positions.append(edge)
    for corner in corners:
        positions.append(corner)

    # Ensure we have exactly 26 positions
    positions = np.array(positions)
    if len(positions) > 26:
        positions = positions[:26]
    centers = positions

    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization

    # Define constraint function for non-overlapping circles
    def constraint_overlap(i, j):
        return lambda r: np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2) - (r[i] + r[j])

    # Define constraint function for boundary constraints
    def constraint_boundary(i):
        return lambda r: np.min([centers[i, 0] - r[i], 1 - centers[i, 0] - r[i], centers[i, 1] - r[i], 1 - centers[i, 1] - r[i]])

    # Create constraints for non-overlapping circles (i < j)
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j: constraint_overlap(i, j)(r)
            })
    for i in range(len(centers)):

            'type': 'ineq',
            'fun': lambda r, i=i: constraint_boundary(i)(r)
        })

    # Bounds for radii: all between 0 and 0.5
    bounds = [(0, 0.5)] * len(centers)

    # Create initial guess with Voronoi-inspired radii

        dist_to_right = 1 - x
        dist_to_bottom = y
        dist_to_top = 1 - y
        distances_to_boundary.append(min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top))

    def optimize_with_method(method):

            result = optimize.minimize(

                x0=initial_radii,

    radii2 = optimize_with_method('Nelder-Mead')
    radii3 = optimize_with_method('L-BFGS-B')

    # Use the best of the three
    radii = np.mean([radii1, radii2, radii3], axis=0)

    total_sum = np.sum(radii)

    return (centers, radii, total_sum)
