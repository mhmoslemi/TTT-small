import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing():
    # Generate 7x7 grid with spacing
    spacing = 1.0 / 6.0
    positions = []
    for i in range(7):
        for j in range(7):
            x = i * spacing
            y = j * spacing
            positions.append((x, y))
    
    # Add special circles near corners and edges
    special_points = [
        (0.0, 0.0), (1.0, 0.0), (1.0, 1.0), (0.0, 1.0),
        (0.0, 0.5), (0.5, 0.0), (0.5, 1.0), (1.0, 0.5),
        (0.25, 0.25), (0.75, 0.25), (0.25, 0.75), (0.75, 0.75)
    ]
    
    # Add special points to positions
    for pt in special_points:
        positions.append(pt)
    
    # Ensure we have exactly 26 positions
    if len(positions) > 26:
        positions = positions[:26]
    centers = np.array(positions)
    
    # Define initial radius (based on spacing)
    initial_radius = 0.1
    radii = np.full(len(positions), initial_radius)
    
    # Function to compute Euclidean distance
    def dist(i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx*dx + dy*dy)
    
    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for maximization
    
    # Constraint for no overlapping circles
    def constraint_overlap(i, j, radii):
        return dist(i, j) - (radii[i] + radii[j])
    
    # Constraint for circle within unit square
    def constraint_boundary(i, radii):

        return min(x, 1 - x, y, 1 - y) - radii[i]

    constraints = []

        bounds=[(0, 1)] * len(positions),
