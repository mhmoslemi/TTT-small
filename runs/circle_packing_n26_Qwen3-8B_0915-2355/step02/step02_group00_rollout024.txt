import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define an optimized layout with 26 circles based on a hexagonal packing pattern
    # This layout will use staggered rows to better utilize space
    # Initial placement is slightly adjusted to allow for optimization
    
    # Create a staggered hexagonal layout
    # We'll use a hexagonal grid with varying row offsets to allow optimization
    centers = []
    for i in range(13):
        for j in range(13):
            if i % 2 == 0:
                x = j / 12.5
                y = i / 12.5
            else:
                x = (j + 0.5) / 12.5
                y = i / 12.5
            centers.append((x, y))
    
    # Trim to exactly 26 circles
    centers = centers[:26]
    centers = np.array(centers)
    
    # Initial radii: small value to allow optimization
    radii = np.full(len(centers), 0.1)
    
    # Function to calculate Euclidean distance between two points
    def dist(i, j):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        return np.sqrt(dx * dx + dy * dy)
    
    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization
    
    # Constraint for non-overlapping circles
    def constraint_overlap(i, j):
        return lambda r: dist(i, j) - (r[i] + r[j])
    
    # Create constraints for non-overlapping circles (i < j)
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, j=j: constraint_overlap(i, j)(r)
            })
    
    # Constraint for keeping circles within the unit square
    def constraint_boundary(i):
        x, y = centers[i]
        return lambda r: np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])
    
    for i in range(len(centers)):
        constraints.append({
            'type': 'ineq',
            'fun': lambda r, i=i: constraint_boundary(i)(r)
        })
    
    # Bounds for radii: all between 0 and 1 (we'll use a tighter bound in the optimization)
    bounds = [(0, 0.5)] * len(centers)
    
    # Run optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8
    )
    
    optimum_radii = result.x
    total_sum = np.sum(optimum_radii)
    
    return (centers, optimum_radii, total_sum)
