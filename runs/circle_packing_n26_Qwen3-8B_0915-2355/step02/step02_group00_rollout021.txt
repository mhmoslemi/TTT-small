import scipy.optimize as optimize

def run_packing():

    # This layout includes a central circle surrounded by rings of increasing circle count.

    # Create a central, medium, and outer ring of circles
    centers = []
    radii = []

    # Define positions for central cluster
    central_pos = [(0.5, 0.5)]
    centers.append(central_pos[0])
    radii.append(0.1)

    # Medium ring: 12 circles arranged in a hexagonal pattern
    for i in range(12):
        angle = 2 * np.pi * i / 12
        x = 0.5 + np.cos(angle) * 0.3
        y = 0.5 + np.sin(angle) * 0.3
        centers.append((x, y))
        radii.append(0.08)

    # Outer ring: 10 circles arranged in a hexagonal pattern
    for i in range(10):
        angle = 2 * np.pi * i / 10
        x = 0.5 + np.cos(angle) * 0.2
        y = 0.5 + np.sin(angle) * 0
        centers.append((x, y))
        radii.append(0.05)

    # Check if we have exactly 26 circles
    if len(centers) > 26:
        centers = centers[:26]
        radii = radii[:26]

    # Function to compute Euclidean distance
    def distance(i, j):
        dx = centers[i][0] - centers[j][0]
        dy = centers[i][1] - centers[j][1]
        return np.sqrt(dx * dx + dy * dy)

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint for non-overlapping circles
    def constraint_overlap(i, j):
        return lambda r: distance(i, j) - (r[i] + r[j])

    # Constraint for boundary condition
    def constraint_boundary(i):
        x, y = centers[i]
        return lambda r: np.min([x - r[i], 1 - x - r[i], y - r[i], 1 - y - r[i]])

    # Create the constraints
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            constraints.append({

        })

    # Bounds for radii (all radii must be between 0 and 0.5)
    bounds = [(0, 0.5)] * len(centers)

    # Run the optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8
    )

    optimum_radii = result.x
