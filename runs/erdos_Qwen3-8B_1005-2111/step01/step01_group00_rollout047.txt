            shift = max_index - (n - 1)
            h = np.roll(h, shift)
        
        # Optimization
        bounds = [(0, 1)] * n
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
        
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        # SLSQP
        res_slsqp = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res_slsqp.success:
            current_h = res_slsqp.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        # COBYLA
        res_cobyla = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res_cobyla.success:
            current_h = res_cobyla.x
            current_c5 = obj(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    return (best_h, best_c5, len(best_h))
