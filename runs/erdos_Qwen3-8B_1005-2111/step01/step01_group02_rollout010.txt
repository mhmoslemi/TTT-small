            scale = (n_points / 2) / total if total != 0 else 1.0
            h = h * scale
            h = np.clip(h, 0, 1)
            initial_pop[i] = h

    population = initial_pop
    best_h = initial_pop[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
