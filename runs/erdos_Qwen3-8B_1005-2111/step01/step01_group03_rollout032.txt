import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n = 376
    pop_size = 20
    gens = 100
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    def repair(h):
        s = np.sum(h)
        target = n / 2
        if abs(s - target) < 1e-6:
            return h
        scale = target / s
        h = h * scale
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        delta = target - s
        if delta != 0:
            h[-1] += delta
            h = np.clip(h, 0, 1)
        return h
    
    def obj(h):
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def local_opt(h):
        bounds = [(0, 1)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}]
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return res.x if res.success else h
    
    if initial_h is not None:
        pop = [initial_h]
    else:
        pop = []
    for _ in range(pop_size - 1):
        h = np.random.rand(n)
        s = np.sum(h)
        delta = n/2 - s
        h[-1] += delta
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        delta = n/2 - s
        if delta:
            h[0] += delta
            h = np.clip(h, 0, 1)
        pop.append(h)
    
    for _ in range(gens):
        if time.time() - start_time > budget_s:
            break
        fit = [obj(h) for h in pop]
        best_idx = np.argmin(fit)
        best_c5 = fit[best_idx]
        best_h = pop[best_idx].copy()
        parents = []
        for _ in range(pop_size):
            a, b = np.random.choice(range(pop_size), 2, replace=False)
            parents.append(pop[np.argmin([fit[a], fit[b]])])
        new_pop = []
        for i in range(0, pop_size, 2):
            if i+1 >= pop_size:
                break
            p1, p2 = parents[i], parents[i+1]
            c1, c2 = np.random.choice([p1, p2], size=n), np.random.choice([p1, p2], size=n)
            new_pop.extend([c1, c2])
        for i in range(len(new_pop)):
            if np.random.rand() < 0.1:
                idx = np.random.randint(n)
                new_pop[i][idx] += np.random.uniform(-0.1, 0.1)
                new_pop[i] = repair(new_pop[i])
        pop = new_pop
        best_h = local_opt(best_h)
    
    return (best_h, best_c5, n)
