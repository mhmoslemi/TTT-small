        h_values = initial_h
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        c5_bound = np.max(corr) * (2.0 / len(h_values))
        n_points = len(h_values)
        return (h_values, c5_bound, n_points)

    n_fine = 376
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, result_coarse.x)
    total_fine = np.sum(h_fine)
    scale = (n_fine / 2) / total_fine
    h_fine *= scale

    bounds_fine = [(0.0, 1.0) for _ in range(n_fine)]
    def objective_fine(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_fine)

    cons_fine = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_fine / 2})
    result_fine = minimize(objective_fine, h_fine, method='SLSQP', bounds=bounds_fine, constraints=cons_fine)

    h_coarse = result_coarse.x
    corr_coarse = np.correlate(h_coarse, 1 - h_coarse, mode='full')
    c5_coarse = np.max(corr_coarse) * (2.0 / n_coarse)

    h_fine_final = result_fine.x
    corr_fine = np.correlate(h_fine_final, 1 - h_fine_final, mode='full')
    c5_fine = np.max(corr_fine) * (2.0 / n_fine)

    if c5_fine < c5_coarse:
        best_h = h_fine_final
        best_c5 = c5_fine
    else:
        best_h = h_coarse
        best_c5 = c5_coarse

    elapsed_time_fine = time.time() - start_time
    if elapsed_time_fine > budget_s:
        return (best_h, best_c5, len(best_h))

    return (best_h, best_c5, len(best_h))
