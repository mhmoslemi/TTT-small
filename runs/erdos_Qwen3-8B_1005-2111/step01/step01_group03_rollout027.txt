            scale = (n_points / 2) / s

            h = np.clip(h, 0, 1)
        return h
    
    def evaluate(h):

        corr = np.correlate(h, h_1_minus, mode='full')
        return np.max(corr) * dx
    
    population = generate_initial_population(n_points)
    for _ in range(max_generations):

            break

        # Tournament selection

                        child1[j], child2[j] = child2[j], child1[j]
