import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import differential_evolution, minimize

    n_coarse = 188
    n_fine = 376
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = n_coarse

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_coarse)
        total = np.sum(initial_h)
        scale = (n_coarse / 2) / total
        initial_h *= scale

    # Coarse grid optimization
    bounds = [(0.0, 1.0) for _ in range(n_coarse)]
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * (2.0 / n_coarse)
    result = differential_evolution(obj, bounds, strategy='best1bin', popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, maxiter=100, seed=seed)
    if result.success:
        best_h = result.x
        best_c5 = result.fun
        best_n = n_coarse

    # Interpolate and refine
    if time.time() - start_time < budget_s:
        x_coarse = np.linspace(0, 2, n_coarse)
        x_fine = np.linspace(0, 2, n_fine)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        total = np.sum(h_fine) * (2.0 / n_fine)
        scale = (n_fine / 2) / total
        h_fine *= scale
        h_fine = np.clip(h_fine, 0, 1)

        bounds_fine = [(0.0, 1.0) for _ in range(n_fine)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_fine / 2})
        def obj_fine(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * (2.0 / n_fine)
        result_fine = minimize(obj_fine, h_fine, method='SLSQP', bounds=bounds_fine, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if result_fine.success:
            current_c5 = result_fine.fun
            if current_c5 < best_c5:
                best_h = result_fine.x
                best_c5 = current_c5
                best_n = n_fine

    return best_h, best_c5, best_n
