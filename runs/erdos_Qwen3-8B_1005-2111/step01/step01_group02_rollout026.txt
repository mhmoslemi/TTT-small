import numpy as np
import time

from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import minimize

    n_points = 188
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h)
    scale = (n_points / 2) / total if total != 0 else 1.0
    initial_h = np.clip(initial_h * scale, 0, 1)
    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * dx

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        h = np.clip(h * scale, 0, 1)
        return h

    def obj(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    pop_size = 100
    pop = [repair(np.random.rand(n_points)) for _ in range(pop_size)]
    pop = [h for h in pop if np.abs(np.sum(h) - n_points / 2) < 1e-6]
    pop = pop[:pop_size] + [best_h] * (pop_size - len(pop))

    def tournament_selection():
        idxs = np.random.choice(len(pop), size=5, replace=False)
        return min(idxs, key=lambda i: obj(pop[i]))

    def sbx_crossover(a, b):
        alpha = norm.rvs()
        alpha = np.clip(alpha, 0, 1)
        child = (1 + alpha) * a + (1 - alpha) * b
        child = np.clip(child, 0, 1)
        return child

    def polynomial_mutation(h):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < 0.1:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.1), 0, 1)
        return h

    start_time = time.time()
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        parents = [tournament_selection() for _ in range(pop_size // 2)]
        new_pop = []
        for i in range(0, len(parents), 2):
            a, b = pop[parents[i]], pop[parents[i+1]]
            child1 = sbx_crossover(a, b)
            child2 = sbx_crossover(a, b)
            new_pop.append(repair(polynomial_mutation(child1)))
            new_pop.append(repair(polynomial_mutation(child2)))
        pop = new_pop
        for h in pop:
            if np.abs(np.sum(h) - n_points / 2) < 1e-6:
                current_c5 = obj(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
    return best_h, best_c5, n_points
