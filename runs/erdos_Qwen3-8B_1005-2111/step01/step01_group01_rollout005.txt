def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import differential_evolution, minimize

    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_points = 188

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        n = n_points
        initial_h = np.random.rand(n)
        total = np.sum(initial_h)
        scale = (n / 2) / total
        initial_h *= scale
        initial_h = np.clip(initial_h, 0, 1)

    def obj(x, n):
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n)

    def cons(x):
        return np.sum(x) - n_points / 2

    def obj_with_penalty(x, n):
        corr = np.correlate(x, 1 - x, mode='full')
        max_corr = np.max(corr)
        penalty = 1e5 * abs(np.sum(x) - n / 2)
        return max_corr * (2.0 / n) + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Global optimization
    if time.time() - start_time < budget_s:
        result = differential_evolution(obj, bounds, args=(n_points,), strategy='best1bin', popsize=15, maxiter=100, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
        if result.success:
            best_h = result.x
            best_c5 = obj(best_h, n_points)
        else:
            best_h = initial_h
            best_c5 = obj(best_h, n_points)

    # Local optimization
    if time.time() - start_time < budget_s:
        result = minimize(obj_with_penalty, best_h, method='L-BFGS-B', bounds=bounds, args=(n_points,), tol=1e-6, options={'maxiter': 1000})
        if result.success:
