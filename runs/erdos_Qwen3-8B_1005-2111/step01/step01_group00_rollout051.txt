                return np.max(corr) * dx
            
            res = minimize(obj, start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = np.clip(res.x, 0, 1)
                if np.sum(h) < n/2 - 1e-5:
                    h = h * (n/2) / np.sum(h)
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            res = minimize(obj, start, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = np.clip(res.x, 0, 1)
                if np.sum(h) < n/2 - 1e-5:
                    h = h * (n/2) / np.sum(h)
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
        
        if best_h is not None:
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            shift_index = np.argmax(corr)
            shift = shift_index - (n - 1)
            best_h_shifted = np.roll(best_h, shift)
            start = best_h_shifted
            start = np.array(start)
            if abs(np.sum(start) - n/2) > 1e-5:
                continue
            bounds = [(0, 1)] * n
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
            
            res = minimize(obj, start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = np.clip(res.x, 0, 1)
                if np.sum(h) < n/2 - 1e-5:
                    h = h * (n/2) / np.sum(h)
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            res = minimize(obj, start, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = np.clip(res.x, 0, 1)
                if np.sum(h) < n/2 - 1e-5:
                    h = h * (n/2) / np.sum(h)
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
    
    return (best_h, best_c5, len(best_h))
