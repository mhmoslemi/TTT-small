import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    start_time = time.time()

    for n in grid_sizes:

        start_points = []

                start_points.append(h)
            else:
                h_sym = np.zeros(n)
                h_sym[:m] = h[:m]
                h_sym[m:] = h[:m][::-1]

                delta = n/4 - sum_first
                h_sym[:m] = np.clip(h_sym[:m] + delta/m, 0, 1)
                start_points.append(h_sym)
        else:
            for _ in range(5):
                h = np.random.rand(m)
                sum_h = np.sum(h)
                delta = n/4 - sum_h
                h[-1] += delta
                h = np.clip(h, 0, 1)
                sum_h = np.sum(h)
                delta = n/4 - sum_h
                if delta != 0:
                    h[0] += delta
                    h = np.clip(h, 0, 1)
                start_points.append(h)
        
        for start in start_points:
            def objective(x):
                h = np.zeros(n)
                h[:m] = x
                h[m:] = x[::-1]
                dx = 2.0 / n
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * dx
            
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/4}]

            if res.success:

                    best_h = res.x
                    best_n = n
                    best_full = np.zeros(n)
                    best_full[:m] = best_h
                    best_full[m:] = best_h[::-1]
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return (best_full, best_c5, best_n)
