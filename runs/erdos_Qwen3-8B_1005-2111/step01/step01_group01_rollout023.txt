    from scipy.optimize import differential_evolution, minimize
    import time

    n = 376
    dx = 2.0 / n
    bounds = [(0.0, 1.0) for _ in range(n)]

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        old_n = len(initial_h)
        x_old = np.linspace(0, 2, old_n)
        x_new = np.linspace(0, 2, n)
        h_new = np.interp(x_new, x_old, initial_h)
        sum_new = np.sum(h_new)
        scale = (n / 2) / sum_new
        h_new *= scale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n / 2) / np.sum(h_new)
    else:
        initial_h = np.random.rand(n)
        total = np.sum(initial_h)
        scale = (n / 2) / total
        h_new = initial_h * scale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n / 2) / np.sum(h_new)

    def objective_DE(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        penalty = 1e6 * (np.sum(h) - n/2)**2
        return c5 + penalty

    start_time = time.time()
    result_DE = differential_evolution(objective_DE, bounds, strategy='best1bin', popsize=15, maxiter=100, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h = h_new
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        return (h, c5, n)

    best_h_DE = result_DE.x
    best_h_DE = np.clip(best_h_DE, 0, 1)
    best_h_DE = best_h_DE * (n / 2) / np.sum(best_h_DE)

    def objective_SLSQP(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n/2})
    result_SLSQP = minimize(objective_SLSQP, best_h_DE, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    elapsed_SLSQP = time.time() - start_time - elapsed
    if elapsed_SLSQP > budget_s - elapsed:
        h = best_h_DE
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        return (h, c5, n)

    if result_SLSQP.success:
        best_h = result_SLSQP.x
        c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    else:
        best_h = best_h_DE
        c5 = np.max(np.correlate(best_h_DE, 1 - best_h_DE, mode='full')) * dx

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n / 2) / np.sum(best_h)
    return (best_h, c5, n)
