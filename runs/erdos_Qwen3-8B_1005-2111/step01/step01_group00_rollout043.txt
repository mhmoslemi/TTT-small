            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x
                c5 = obj(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
            res = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h_opt = res.x
                c5 = obj(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
    
    return (best_h, best_c5, len(best_h))
