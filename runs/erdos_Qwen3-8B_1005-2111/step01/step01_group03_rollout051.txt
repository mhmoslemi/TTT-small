        population.append(h)
    for _ in range(population_size - 1):
        h = np.random.rand(n)
        h = adjust_sum(h)
        population.append(h)

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.9

    for gen in range(generations):
        fitness = []
        for h in population:
            h_1_minus = 1.0 - h
            corr = np.correlate(h, h_1_minus, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        best_idx = np.argmin(fitness)
        best_c5_current = fitness[best_idx]
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = population[best_idx]
        parents = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, size=3, replace=False)
            best_tournament_idx = np.argmin([fitness[i] for i in tournament])
            parents.append(population[tournament[best_tournament_idx]])
        offspring = []
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_rate:
                parent1 = parents[i]
                parent2 = parents[i+1]
                child1 = np.where(np.random.rand(n) < 0.5, parent1, parent2)
                child2 = np.where(np.random.rand(n) < 0.5, parent2, parent1)
                offspring.extend([child1, child2])
            else:
                offspring.append(parents[i])
                offspring.append(parents[i+1])
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                perturb = np.random.uniform(-0.1, 0.1, size=n)
                offspring[i] += perturb
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = repair_h(offspring[i])
        population = offspring
        current_best = population[np.argmin(fitness)]
        def objective(x):
            h = x
            h_1_minus = 1.0 - h
            corr = np.correlate(h, h_1_minus, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1) for _ in range(n)]
        res_slsqp = minimize(objective, current_best, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res_slsqp.success:
            c5 = res_slsqp.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = res_slsqp.x
        res_cobyla = minimize(objective, current_best, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if res_cobyla.success:
            c5 = res_cobyla.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = res_cobyla.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return (best_h, best_c5, n)
