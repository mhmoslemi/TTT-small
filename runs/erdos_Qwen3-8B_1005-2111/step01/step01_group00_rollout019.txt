        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        dx = 2.0 / n
        # Generate initial h for this grid size
        if n == 94:
            h = prev_h.copy()
        else:
            prev_n = grid_sizes[grid_sizes.index(n) - 1]
            h = np.tile(prev_h, n // prev_n)
        # Ensure sum is correct
        if abs(np.sum(h) - n/2) > 1e-5:
            continue
        # Perform cyclic rearrangement
        for _ in range(50):
            corr = np.correlate(h, 1 - h, mode='full')
            i = np.argmax(corr)
            shift_idx = (i - n + 1) % n
            h = np.roll(h, shift_idx)
        # Local optimization
        bounds = [(0, 1)] * n
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
        
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        # SLSQP
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            c5 = obj(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        # COBYLA
        res = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            h = res.x
            c5 = obj(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        # Update prev_h for next grid size
        prev_h = h
    
    return (best_h, best_c5, len(best_h))
