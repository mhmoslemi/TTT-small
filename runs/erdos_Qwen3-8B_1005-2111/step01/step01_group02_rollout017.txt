import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 188
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if np.isclose(s, target_sum):
            return h
        scale = target_sum / s if s != 0 else 1.0
        return h * scale

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def sbx_crossover(p1, p2, eta=20):
        u = np.random.rand(len(p1))
        beta = np.random.rand(len(p1)) * eta
        beta = np.where(u < 0.5, 1.0 / (beta + 1e-10), 1.0 / (1.0 / beta + 1e-10))
        c1 = 0.5 * (p1 + p2) - beta * (p1 - p2)
        c2 = 0.5 * (p1 + p2) + beta * (p1 - p2)
        return np.clip(c1, 0, 1), np.clip(c2, 0, 1)

    def polynomial_mutation(ind, eta=20):
        ind = np.copy(ind)
        for i in range(len(ind)):
            if np.random.rand() < 0.5:
                r = np.random.rand()
                delta = (1.0 + eta) * (r ** (1.0 / (eta + 1.0))) - 1.0
            else:
                r = np.random.rand()
                delta = 1.0 - (1.0 + eta) * (1.0 - r) ** (1.0 / (eta + 1.0))
            ind[i] += delta
            ind[i] = np.clip(ind[i], 0, 1)
        return ind

    if 'initial_h_values' in globals():
        initial_pop = [np.array(initial_h_values, dtype=np.float64)]
    else:
        initial_pop = [np.random.rand(n_points) for _ in range(100)]
        for i in range(len(initial_pop)):
            h = initial_pop[i]
            h = np.clip(h, 0, 1)
            s = np.sum(h)
            scale = target_sum / s if s != 0 else 1.0
            initial_pop[i] = h * scale
            initial_pop[i] = np.clip(initial_pop[i], 0, 1)

    best_h = initial_pop[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()
    pop_size = 100
    tournament_size = 5
    crossover_prob = 0.9
    mutation_prob = 0.1
    max_evals = 10000

    while time.time() - start_time < budget_s:
        fitness = np.array([evaluate(h) for h in initial_pop])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = initial_pop[best_idx]
            best_c5 = fitness[best_idx]
        new_pop = []
        for _ in range(pop_size):
            t = random.sample(range(pop_size), tournament_size)
            t_fitness = fitness[t]
            winner = initial_pop[t[np.argmin(t_fitness)]]
            if random.random() < crossover_prob:
                other = initial_pop[random.choice([i for i in range(pop_size) if i != t[np.argmin(t_fitness)]])]
                c1, c2 = sbx_crossover(winner, other)
                c1 = repair(c1)
                c2 = repair(c2)
                new_pop.extend([c1, c2])
            else:
                child = np.copy(winner)
                new_pop.append(child)
            if random.random() < mutation_prob:
                child = polynomial_mutation(new_pop[-1])
                new_pop[-1] = repair(child)
        initial_pop = new_pop[:pop_size]

    fitness = np.array([evaluate(h) for h in initial_pop])
    best_idx = np.argmin(fitness)
    best_h = initial_pop[best_idx]
    best_c5 = fitness[best_idx]
    return best_h, best_c5, n_points
