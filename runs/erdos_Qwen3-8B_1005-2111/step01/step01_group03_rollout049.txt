                    delta = np.random.uniform(-0.1, 0.1)
                    child[j] += delta
                    child[j] = np.clip(child[j], 0, 1)
            # Repair
            current_sum = np.sum(child)
            delta = target - current_sum
            child[-1] += delta
            child = np.clip(child, 0, 1)
            current_sum = np.sum(child)
            delta = target - current_sum
            if delta != 0:
                child[0] += delta
                child = np.clip(child, 0, 1)
        # Replace worst individuals
        sorted_indices = np.argsort(fitness)
        for i in range(len(population) // 2):
            population[sorted_indices[i]] = offspring[i]
        # Local optimization on the best candidate
        h_local = np.copy(current_best)
        def objective(x):
            h = x
            dx = 2.0 / n
            h_1_minus = 1.0 - h
            corr = np.correlate(h, h_1_minus, mode='full')
            return np.max(corr) * dx
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}]
        bounds = [(0, 1) for _ in range(n)]
        res_slsqp = minimize(objective, h_local, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res_slsqp.success:
            c5 = res_slsqp.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = res_slsqp.x
        res_cobyla = minimize(objective, h_local, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if res_cobyla.success:
            c5 = res_cobyla.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = res_cobyla.x
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return (best_h, best_c5, n)
