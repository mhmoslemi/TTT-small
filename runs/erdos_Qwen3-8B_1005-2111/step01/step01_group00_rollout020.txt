            prev_h = best_h if best_h is not None else initial_h_values
            x_prev = np.linspace(0, 2, prev_n)
            x_new = np.linspace(0, 2, n)
            h_new = np.interp(x_new, x_prev, prev_h)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n/2) / np.sum(h_new)
            starts.append(h_new)
            for _ in range(4):
                h = np.random.rand(n)
                s = np.sum(h)
                if s == 0:
                    continue
                h = h * (n/2) / s
                h = np.clip(h, 0, 1)
                starts.append(h)
        
        for start in starts:
            start = np.array(start)
            if abs(np.sum(start) - n/2) > 1e-5:
                continue
            bounds = [(0, 1)] * n
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
            
            h = start.copy()
            for _ in range(50):
                corr = np.correlate(h, 1 - h, mode='full')
                max_index = np.argmax(corr)
                shift = (max_index - (n - 1)) % n
                h = np.roll(h, shift)
            
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            res = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
    
    return (best_h, best_c5, len(best_h))
