        starts = []
        if n == 94:
            starts.append(initial_h_values)
        else:
            repeat = n // 94
            upsampled = np.tile(initial_h_values, repeat)
            upsampled = np.clip(upsampled, 0, 1)
            starts.append(upsampled)
            for _ in range(4):
                h = np.random.rand(n)
                s = np.sum(h)
                if s == 0:
                    continue
                h = h * (n/2) / s
                h = np.clip(h, 0, 1)
                starts.append(h)
        
        for start in starts:
            h = np.array(start)
            if abs(np.sum(h) - n/2) > 1e-5:
                continue
            for _ in range(50):
                corr = np.correlate(h, 1 - h, mode='full')
                m = np.argmax(corr)
                k = m - n + 1
                h = np.roll(h, k)
            
            bounds = [(0, 1)] * n
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
            
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                current_h = res.x
                current_c5 = obj(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
            res = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                current_h = res.x
                current_c5 = obj(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
    
    return (best_h, best_c5, len(best_h))
