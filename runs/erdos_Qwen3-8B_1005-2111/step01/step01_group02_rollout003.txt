from scipy.optimize import cma
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 188
    dx = 2.0 / n_points
    
    def generate_random_feasible(n):
        h = np.random.rand(n)
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            h = np.ones(n) * 0.5
        else:
            scale = (n/2) / total
            h = h * scale
        h = np.clip(h, 0, 1)
        return h
    
    def repair(h):
        h_clip = np.clip(h, 0, 1)
        total_clip = np.sum(h_clip)
        if total_clip == 0:
            return generate_random_feasible(n_points)
        scale = (n_points/2) / total_clip
        h_repaired = h_clip * scale
        h_repaired = np.clip(h_repaired, 0, 1)
        return h_repaired
    
    def objective(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    initial_mean = kwargs.get('initial_h_values', generate_random_feasible(n_points))
    
    options = {
        'maxeval': 110000,
        'ftol': 1e-10,
        'gtol': 1e-10,
        'bounds': [0, 1]
    }
    
    start_time = time.time()

    best_h = result['x']

    corr = np.correlate(best_h, 1 - best_h, mode='full')
