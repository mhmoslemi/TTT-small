import numpy as np
import time
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import differential_evolution, minimize

    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        n = 188
        initial_h = np.random.rand(n)
        total = np.sum(initial_h)
        scale = (n / 2) / total
        initial_h *= scale

    def obj(x, n_p):
        dx = 2.0 / n_p
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def cons(x, n_p):
        return np.sum(x) - n_p / 2

    bounds = [(0.0, 1.0) for _ in range(188)]
    start_time = time.time()

    # Global search with DE
    result_de = differential_evolution(lambda x: obj(x, 188) + 1e5 * (cons(x, 188))**2, bounds, popsize=20, maxiter=100, tol=1e-6, seed=seed)
    best_h = result_de.x
    best_c5 = obj(best_h, 188)

    # Local refinement with L-BFGS-B
    cons_lbgfs = ({'type': 'eq', 'fun': lambda x: cons(x, 188)})
    result_lbgfs = minimize(lambda x: obj(x, 188), best_h, method='L-BFGS-B', bounds=bounds, constraints=cons_lbgfs, tol=1e-6)
    best_h = result_lbgfs.x
    best_c5 = obj(best_h, 188)

    # Check time and upsample if possible
    if time.time() - start_time < budget_s:
        n_p = 376
        h_up = np.interp(np.linspace(0, 2, n_p), np.linspace(0, 2, 188), best_h)
        total = np.sum(h_up)
        scale = (n_p / 2) / total
        h_up *= scale
        best_c5_up = obj(h_up, n_p)
        if best_c5_up < best_c5:
            best_h = h_up
            best_c5 = best_c5_up

    return best_h, best_c5, len(best_h)
