    best_c5_coarse = np.max(np.correlate(best_h_coarse, 1 - best_h_coarse, mode='full')) * dx_coarse

    # Local phase: Interpolate to fine grid (n=376) and optimize
    n_fine = 376
    dx_fine = 2.0 / n_fine
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, best_h_coarse)

    def objective_fine(x):
        h = x
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx_fine

    cons_fine = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_fine / 2})
    bounds_fine = [(0.0, 1.0) for _ in range(n_fine)]

    start_local = time.time()
    result_slsqp = minimize(
        objective_fine, h_fine,
        method='SLSQP', bounds=bounds_fine,
        constraints=cons_fine, tol=1e-6, options={'maxiter': 100}
    )
    elapsed_local = time.time() - start_local
    total_time = elapsed_global + elapsed_local

    if total_time > budget_s:
        return best_h_coarse, best_c5_coarse, n_coarse

    best_h_fine = result_slsqp.x
    best_c5_fine = np.max(np.correlate(best_h_fine, 1 - best_h_fine, mode='full')) * dx_fine

    if best_c5_fine < best_c5_coarse:
        return best_h_fine, best_c5_fine, n_fine
    else:
        return best_h_coarse, best_c5_coarse, n_coarse
