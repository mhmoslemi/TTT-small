import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import differential_evolution
    import time

    n = 188
    dx = 2.0 / n

    def repair_feasible(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        total = np.sum(h_clipped)
        if total == 0:
            return np.zeros_like(h)
        scale = (n / 2) / total
        return h_clipped * scale

    def generate_random_feasible(n_points):
        h = np.random.rand(n_points)
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    def objective(h):
        repaired = repair_feasible(h)
        corr = np.correlate(repaired, 1 - repaired, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    start_time = time.time()
    if 'initial_h_values' in globals():
        initial_guess = initial_h_values
    else:
        initial_guess = generate_random_feasible(n)

    bounds = [(0.0, 1.0) for _ in range(n)]

    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   popsize=100, tol=1e-6, mutation=(0.5, 1),
                                   recombination=0.9, seed=seed,
                                   maxiter=1000, disp=False,
                                   polish=True, vectorized=True)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_final = initial_guess
        c5_final = objective(h_final)
    else:
        h_final = result.x
        c5_final = result.fun

    h_final = repair_feasible(h_final)
    c5_final = objective(h_final)

    return h_final, c5_final, n
