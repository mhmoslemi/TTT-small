import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    best_h = None
    best_c5 = float('inf')
    grid_sizes = [94, 188, 376]
    start_time = time.time()
    
    for n in grid_sizes:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        dx = 2.0 / n
        starts = []
        if n == 94:
            starts.append(initial_h_values)
        else:
            if best_h is not None:
                starts.append(np.tile(best_h, 2))
            for _ in range(4):
                h = np.random.rand(n)
                s = np.sum(h)
                if s == 0:
                    continue
                h = h * (n/2) / s
                h = np.clip(h, 0, 1)
                starts.append(h)
        
        for start in starts:
            start = np.array(start)
            if abs(np.sum(start) - n/2) > 1e-5:
                continue
            bounds = [(0, 1)] * n
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
            
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            
            res = minimize(obj, start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            res = minimize(obj, start, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            
            if best_h is not None:
                corr = np.correlate(best_h, 1 - best_h, mode='full')
                max_idx = np.argmax(corr)
                k = max_idx - (n - 1)
                h_shifted = np.roll(best_h, k)
                res = minimize(obj, h_shifted, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
                if res.success:
                    h = res.x
                    c5 = obj(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h
                res = minimize(obj, h_shifted, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
                if res.success:
                    h = res.x
                    c5 = obj(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h
    
    if best_h is None:
        return (initial_h_values, 0.382847, 94)
    return (best_h, best_c5, len(best_h))
