import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    start_time = time.time()

        target_sum_first_half = n / 4

                current_sum = np.sum(h)
                delta = target_sum_first_half - current_sum

                current_sum = np.sum(h)
                delta = target_sum_first_half - current_sum

                start_points.append(h)
