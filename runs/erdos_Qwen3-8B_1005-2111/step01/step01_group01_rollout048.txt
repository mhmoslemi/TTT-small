                best_h = current_h_up.copy()
                best_c5 = current_c5_up

    # Final adjustments
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    scale = (len(best_h) / 2) / sum_h
    best_h *= scale
    best_h = np.clip(best_h, 0, 1)

    return best_h, best_c5, len(best_h)
