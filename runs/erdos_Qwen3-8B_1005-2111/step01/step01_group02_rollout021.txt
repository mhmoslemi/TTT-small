import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import differential_evolution
    import time

    n = 188
    dx = 2.0 / n

    def repair(h):
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scale = (n / 2) / sum_h
        return h * scale

    def objective(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    start_time = time.time()
    if 'initial_h_values' in globals():
        initial_guess = initial_h_values
    else:
        h = np.random.rand(n)
        sum_h = np.sum(h)
        scale = (n / 2) / sum_h
        initial_guess = h * scale

    bounds = [(0.0, 1.0) for _ in range(n)]

    result = differential_evolution(
        objective, bounds,
        strategy='best1bin',
        popsize=50,
        mutation=(0.5, 1),
        crossover=0.9,
        maxiter=10000,
        tol=1e-6,
        polish=True,
        disp=False
    )

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        h_final = initial_guess
        c5 = objective(h_final)
    else:
        h_final = result.x
        c5 = result.fun

    h_final_repaired = repair(h_final)
    c5 = objective(h_final_repaired)

    return h_final_repaired, c5, n
