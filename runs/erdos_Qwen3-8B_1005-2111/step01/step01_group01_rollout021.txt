        c5_slsqp = compute_c5(x_slsqp, n_points)
        if c5_slsqp < best_c5:
            best_h = x_slsqp.copy()
            best_c5 = c5_slsqp

    if time.time() - start_time > budget_s:
        return best_h, best_c5, n_points

    n_points_up = 376
    x_interp = np.interp(np.linspace(0, 2, n_points_up), np.linspace(0, 2, n_points), best_h)
    sum_interp = np.sum(x_interp)
    scale_interp = (n_points_up / 2) / sum_interp
    x_interp_scaled = np.clip(x_interp * scale_interp, 0, 1)
    c5_up = compute_c5(x_interp_scaled, n_points_up)
    if c5_up < best_c5:
        best_h = x_interp_scaled.copy()
        best_c5 = c5_up
        n_points = n_points_up

    bounds_up = [(0.0, 1.0) for _ in range(n_points_up)]
    def objective_de_up(x):
        c5 = compute_c5(x, n_points_up)
        constraint = abs(np.sum(x) - n_points_up / 2)
        return c5 + 1000 * constraint
    result_de_up = differential_evolution(objective_de_up, bounds_up, strategy='best1bin', maxiter=50, popsize=15, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
    x_de_up = result_de_up.x
    sum_de_up = np.sum(x_de_up)
    scale_up = (n_points_up / 2) / sum_de_up
    x_de_up_scaled = np.clip(x_de_up * scale_up, 0, 1)
    c5_de_up = compute_c5(x_de_up_scaled, n_points_up)
    if c5_de_up < best_c5:
        best_h = x_de_up_scaled.copy()
        best_c5 = c5_de_up

    cons_up = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_up / 2})
    result_slsqp_up = minimize(objective_slsqp, x_de_up_scaled, method='SLSQP', bounds=bounds_up, constraints=cons_up)
    if result_slsqp_up.success:
        x_slsqp_up = result_slsqp_up.x
        c5_slsqp_up = compute_c5(x_slsqp_up, n_points_up)
        if c5_slsqp_up < best_c5:
            best_h = x_slsqp_up.copy()
            best_c5 = c5_slsqp_up

    if time.time() - start_time > budget_s:

    return best_h, best_c5, n_points
