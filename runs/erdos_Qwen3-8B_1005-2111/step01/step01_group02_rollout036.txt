import numpy as np
import random
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 188
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h)
    scale = (n_points / 2) / total if total != 0 else 1.0
    initial_h = initial_h * scale
    initial_h = np.clip(initial_h, 0, 1)

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        return h * scale

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def tournament_selection(pop, size=5):
        return min(random.sample(pop, size), key=objective)

    def crossover(a, b):
        alpha = np.random.rand()
        return alpha * a + (1 - alpha) * b

    def mutate(ind, rate=0.1, eta=20):
        for i in range(len(ind)):
            if np.random.rand() < rate:
                u = np.random.rand()
                if u <= 0.5:
                    delta = (2 * u) ** (1/(eta+1)) - 1
                else:
                    delta = 1 - (2 * (1 - u)) ** (1/(eta+1))
                ind[i] += delta * (ind[i] - 0.5)
                ind[i] = np.clip(ind[i], 0, 1)
        return ind

    pop_size = 100
    population = [initial_h.copy()]
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        new_pop = []
        for _ in range(pop_size):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child.copy())
            child = repair(child)
            new_pop.append(child)
        population = new_pop
        for ind in population:
            curr = objective(ind)
            if curr < best_c5:
                best_h = ind.copy()
                best_c5 = curr
    return best_h, best_c5, n_points
