                h = res.x
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            res = minimize(obj, start, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            corr = np.correlate(h, 1 - h, mode='full')
            max_index = np.argmax(corr)
            shift_index = max_index - n + 1
            shift = shift_index % n
            h_shifted = np.roll(h, shift)
            res_shifted = minimize(obj, h_shifted, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res_shifted.success:

                    best_h = h_shifted_opt
            res_shifted = minimize(obj, h_shifted, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res_shifted.success:

                c5_shifted = obj(h_shifted_opt)
                if c5_shifted < best_c5:
                    best_c5 = c5_shifted
                    best_h = h_shifted_opt
        prev_h = best_h
    return (best_h, best_c5, len(best_h))
