from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    best_h = None
    best_c5 = float('inf')
    grid_sizes = [94, 188, 376]
    start_time = time.time()
    np.random.seed(seed)
    
    def upsample(h_old, n_old, n_new):
        dx_old = 2.0 / n_old
        dx_new = 2.0 / n_new
        x_old = np.linspace(0, 2, n_old)
        x_new = np.linspace(0, 2, n_new)
        h_new = np.interp(x_new, x_old, h_old)
        h_new = np.clip(h_new, 0, 1)
        total = np.sum(h_new) * dx_new
        scale = (n_new / 2) / total if total != 0 else 1.0
        h_new *= scale
        return h_new
    
    for n in grid_sizes:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        dx = 2.0 / n
        starts = []
        if n == 94:
            starts.append(initial_h_values)
        else:
            prev_h = best_h
            if prev_h is not None:
                starts.append(upsample(prev_h, len(prev_h), n))
            for _ in range(4):
                h = np.random.rand(n)
                s = np.sum(h)
                if s == 0:
                    continue
                h = h * (n/2) / s
                h = np.clip(h, 0, 1)
                starts.append(h)
        
        for start in starts:
            start = np.array(start)
            if abs(np.sum(start) - n/2) > 1e-5:
                continue
            bounds = [(0, 1)] * n
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
            
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            
            res = minimize(obj, start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
            if res.success:
                h = res.x

                    best_c5 = c5

            res = minimize(obj, start, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
