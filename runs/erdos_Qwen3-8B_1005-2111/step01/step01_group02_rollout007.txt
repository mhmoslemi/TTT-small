import numpy as np
import time
from scipy.optimize import minimize
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    n_points = 188
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h = h * scale
        h = np.clip(h, 0, 1)
    h = h.astype(np.float64)

    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scale = (n_points / 2) / total
        h = h * scale
        h = np.clip(h, 0, 1)
        return h

    def objective(h):
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def tournament_selection(pop, k=5):
        idxs = np.random.randint(len(pop), size=k)
        return pop[idxs[np.argmin([objective(pop[i]) for i in idxs])]]

    def sbx_crossover(parent1, parent2, eta=15):
        alpha = np.random.rand()
        if alpha < 0.5:
            alpha = 1.0 / (1.0 + np.exp(-eta * (parent1 - parent2)))
        else:
            alpha = 1.0 / (1.0 + np.exp(-eta * (parent2 - parent1)))
        child = (1.0 - alpha) * parent1 + alpha * parent2
        return np.clip(child, 0, 1)

    def polynomial_mutation(h, eta=20, mutation_rate=0.1):
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.normal(0, 1 / np.sqrt(len(h)))
                h[i] = np.clip(h[i], 0, 1)
        return h

    pop_size = 50
    pop = [h + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    pop = [repair(p) for p in pop]
    best_h = pop[np.argmin([objective(p) for p in pop])]
    best_c5 = objective(best_h)
    start_time = time.time()

    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        new_pop = []
        for _ in range(pop_size // 2):
            p1 = tournament_selection(pop)
            p2 = tournament_selection(pop)
            child1 = sbx_crossover(p1, p2)
            child2 = sbx_crossover(p2, p1)
            child1 = polynomial_mutation(child1)
            child2 = polynomial_mutation(child2)
            new_pop.extend([child1, child2])
        pop = new_pop
        current_best = min(pop, key=objective)
        if objective(current_best) < best_c5:
            best_h = current_best
            best_c5 = objective(best_h)
    return best_h, best_c5, n_points
