    for n in grid_sizes:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        dx = 2.0 / n
        starts = []
        
        if n == 94:
            starts.append(initial_h_values)
            for _ in range(4):
                h = np.random.rand(n)
                s = np.sum(h)
                if s == 0:
                    continue
                h = h * (n/2) / s
                h = np.clip(h, 0, 1)
                starts.append(h)
        else:
            if best_h is not None:
                prev_n = grid_sizes[grid_sizes.index(n) - 1]
                upsampled = np.tile(best_h, n // prev_n)
                starts.append(upsampled)
            for _ in range(4):
                h = np.random.rand(n)
                s = np.sum(h)
                if s == 0:
                    continue
                h = h * (n/2) / s
                h = np.clip(h, 0, 1)
                starts.append(h)
        
        for start in starts:
            start = np.array(start)
            if abs(np.sum(start) - n/2) > 1e-5:
                continue
            bounds = [(0, 1)] * n
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
            
            def obj(x):
                corr = np.correlate(x, 1 - x, mode='full')
                return np.max(corr) * dx
            
            res = minimize(obj, start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            res = minimize(obj, start, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
            if res.success:
                h = res.x
                c5 = obj(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
            
            if best_h is not None:
                corr = np.correlate(best_h, 1 - best_h, mode='full')
                max_corr_idx = np.argmax(corr)
                shift = max_corr_idx - (n - 1)
                if shift > 0:
                    shifted = np.roll(best_h, shift)
                elif shift < 0:
                    shifted = np.roll(best_h, shift)
                else:
                    shifted = best_h
                res = minimize(obj, shifted, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
                if res.success:
                    h = res.x
                    c5 = obj(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h
                res = minimize(obj, shifted, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
                if res.success:
                    h = res.x
                    c5 = obj(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h
    
    return (best_h, best_c5, len(best_h))
