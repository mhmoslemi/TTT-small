        initial_h = initial_h_values
    else:
        n = 188
        initial_h = np.random.rand(n)
        total = np.sum(initial_h)
        scale = (n / 2) / total
        initial_h *= scale

    best_h = initial_h.copy()
    best_c5 = np.max(np.correlate(initial_h, 1 - initial_h, mode='full')) * 2.0 / len(initial_h)
    n_points = len(best_h)
    start_time = time.time()

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * 2.0 / len(h)

    def objective_de(x):
        h = x
        penalty = 1e10 * abs(np.sum(h) - n_points / 2)
        return compute_c5(h) + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result_de = differential_evolution(objective_de, bounds, strategy='best1bin', popsize=15, maxiter=100, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False)
    if result_de.success:
        current_h = result_de.x
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    def objective_lbfgs(x):
        return compute_c5(x)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    result_lbfgs = minimize(objective_lbfgs, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, maxiter=1000)
    if result_lbfgs.success:
        current_h = result_lbfgs.x
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    if time.time() - start_time < budget_s:
        n_upsample = 376
        x_original = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_upsample)
        h_upsampled = np.interp(x_new, x_original, best_h)
        sum_h = np.sum(h_upsampled)
        scale_factor = (n_upsample / 2) / sum_h
        h_upsampled = np.clip(h_upsampled * scale_factor, 0, 1)

        def objective_de_up(x):
            h = x
            penalty = 1e10 * abs(np.sum(h) - n_upsample / 2)
            return compute_c5(h) + penalty

        bounds_up = [(0.0, 1.0) for _ in range(n_upsample)]
        result_de_up = differential_evolution(objective_de_up, bounds_up, strategy='best1bin', popsize=15, maxiter=100, tol=1e-6, mutation=(0.5, 1), recombination=0.8, disp=False)
        if result_de_up.success:
            current_h_up = result_de_up.x
            current_c5_up = compute_c5(current_h_up)
            if current_c5_up < best_c5:
                best_h = current_h_up.copy()
                best_c5 = current_c5_up

        def objective_lbfgs_up(x):
            return compute_c5(x)

        cons_up = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_upsample / 2})
        result_lbfgs_up = minimize(objective_lbfgs_up, h_upsampled, method='L-BFGS-B', bounds=bounds_up, constraints=cons_up, tol=1e-6, maxiter=1000)
        if result_lbfgs_up.success:
            current_h_up = result_lbfgs_up.x
            current_c5_up = compute_c5(current_h_up)
            if current_c5_up < best_c5:
                best_h = current_h_up.copy()
                best_c5 = current_c5_up

    return best_h, best_c5, len(best_h)
