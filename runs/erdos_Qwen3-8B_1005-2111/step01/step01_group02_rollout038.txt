    import numpy as np

    from scipy.optimize import differential_evolution

    dx = 2.0 / n_points

        scale = (n_points / 2) / total if total != 0 else 1.0

        h = np.clip(h, 0, 1)

        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)

            scale = (n_points / 2) / total

        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
