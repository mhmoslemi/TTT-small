import numpy as np
from scipy.optimize import differential_evolution, minimize

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    from scipy.optimize import differential_evolution, minimize
    import time

    # Initialize with parent solution if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values

        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * (2.0 / n)

    def penalty(x):
        return 1e10 * np.abs(np.sum(x) - n / 2)

    def global_objective(x):
        return objective(x) + penalty(x)

    bounds = [(0.0, 1.0) for _ in range(n)]
    start_time = time.time()

    # Global optimization with DE
    result_de = differential_evolution(
        global_objective,
        bounds,
