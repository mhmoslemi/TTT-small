            prev_x = np.linspace(0, 2, prev_n)
            new_x = np.linspace(0, 2, n)
            h = np.interp(new_x, prev_x, best_h)
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            scale = (n / 2) / current_sum
            h = h * scale
        # Cyclic rearrangement
        for _ in range(50):
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            shift = max_idx - (n - 1)
            h = np.roll(h, shift)
        # Local optimization
        bounds = [(0, 1)] * n
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n/2}
        
        def obj(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        res_slsqp = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if res_slsqp.success:
            h_candidate = res_slsqp.x
            c5_candidate = obj(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
        res_cobyla = minimize(obj, h, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
        if res_cobyla.success:
            h_candidate = res_cobyla.x
            c5_candidate = obj(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
    
    return (best_h, best_c5, len(best_h))
