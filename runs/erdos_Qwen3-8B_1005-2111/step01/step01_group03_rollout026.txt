    elapsed = 0.0
    start_time = time.time()
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair(h):
        total = np.sum(h)
        if np.isclose(total, n_points / 2):
            return h
        scale = (n_points / 2) / total
        h = np.clip(h * scale, 0, 1)
        return h
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return repair(h)
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(len(child)):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return repair(child)
    
    population = [initial_h.copy()]
    for _ in range(200):
        if elapsed >= budget_s - 10:
            break

            population += [np.random.uniform(0, 1, n_points) for _ in range(20 - len(population))]
        population = [repair(p) for p in population]
        fitness = [evaluate(p) for p in population]
        best_idx = np.argmin(fitness)

            offspring.append(crossover(p1, p2))
