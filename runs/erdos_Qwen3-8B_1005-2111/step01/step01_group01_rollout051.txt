        best_c5_fine = np.max(np.correlate(best_h_fine, 1 - best_h_fine, mode='full')) * dx_fine

    if best_c5_coarse < best_c5_fine:
        best_h = best_h_coarse
        best_c5 = best_c5_coarse
        n_points = n_coarse
    else:
        best_h = best_h_fine
        best_c5 = best_c5_fine
        n_points = n_fine

    return best_h, best_c5, n_points
