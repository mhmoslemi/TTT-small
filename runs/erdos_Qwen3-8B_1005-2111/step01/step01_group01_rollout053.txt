    bounds_slsqp = [(0, 1)]*n_fine
    cons_slsqp = [{'type': 'eq', 'fun': lambda x: np.sum(x) - (n_fine - 1)/2}]
    result_slsqp = minimize(objective_slsqp, new_h, method='SLSQP', bounds=bounds_slsqp, constraints=cons_slsqp, 
                            options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8})

    elapsed_slsqp = time.time() - start_time - elapsed
    if elapsed + elapsed_slsqp > budget_s:
        return best_h, best_c5, n_initial

    final_h = result_slsqp.x
    final_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * (2 / (n_fine - 1))
    if final_c5 < best_c5:
        best_h, best_c5 = final_h, final_c5

    return best_h, best_c5, n_fine
