import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    from scipy.optimize import differential_evolution

    n_points = 188
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    if 'initial_h_values' in globals():
        initial_guess = initial_h_values
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        total = np.sum(h)

        h = h * scale
        h = np.clip(h, 0, 1)
        initial_guess = h

    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        scale = (n_points / 2) / total if total != 0 else 1.0
        h_scaled = h_clipped * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        dx = 2.0 / n_points
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

                                   disp=False, seed=seed)

    if result.success:

        current_c5 = objective(current_h)
        if current_c5 < best_c5:

            best_c5 = current_c5
    else:
        current_h = initial_guess
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
