def run(seed=42, budget_s=1100, **kwargs):

    n_points = 188
    initial_h = initial_h_values if 'initial_h_values' in globals() else None

            return np.full(n_points, 0.5)

        scale = target / s
        h = h * scale
        h = np.clip(h, 0, 1)

        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def sbx_crossover(parent1, parent2, eta):
        u = np.random.rand(n_points)
        beta = np.zeros(n_points)
        for i in range(n_points):
            if u[i] < 0.5:
                beta[i] = (2 * u[i]) ** (1 / (eta + 1))
            else:
                beta[i] = (2 * (1 - u[i])) ** (1 / (eta + 1))
        child = parent1 + beta * (parent1 - parent2)
        child = np.clip(child, 0, 1)
        return child

    def polynomial_mutation(individual, mutation_prob, eta):
