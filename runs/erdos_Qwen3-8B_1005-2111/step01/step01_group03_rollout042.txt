            delta = target - sum_h

                h[0] += delta
                h = np.clip(h, 0, 1)
            population.append(h)
    else:
        population = [np.random.rand(n_points) for _ in range(population_size)]

            h = population[i]
            sum_h = np.sum(h)
            target = n_points / 2
