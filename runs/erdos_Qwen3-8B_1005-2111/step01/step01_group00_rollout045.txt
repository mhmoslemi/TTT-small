        res_slsqp = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res_slsqp.success:
            h_slsqp = res_slsqp.x
            c5_slsqp = obj(h_slsqp)
            if c5_slsqp < best_c5:
                best_c5 = c5_slsqp
                best_h = h_slsqp
        
        # Run COBYLA
        res_cobyla = minimize(obj, h_start, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
        if res_cobyla.success:
            h_cobyla = res_cobyla.x
            c5_cobyla = obj(h_cobyla)
            if c5_cobyla < best_c5:
                best_c5 = c5_cobyla
                best_h = h_cobyla
        
        return best_h, best_c5
    
    # Process each grid size
    for n in grid_sizes:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Upsample if needed
        if n == 94:
            h_current = h
        else:
            # Upsample from previous grid
            if n == 188:
                h_current = upsample(h, n)
            elif n == 376:
                h_current = upsample(h, n)
            else:
                raise ValueError("Unsupported grid size")
        
        # Perform cyclic rearrangement
        h_current = cyclic_rearrangement(h_current, n)
        
        # Generate starts: initial and random
        starts = [h_current]
        # Generate random starts
        for _ in range(4):
            h_rand = np.random.rand(n)
            s = np.sum(h_rand)
            if s == 0:
                continue
            h_rand = h_rand * (n / 2) / s
            h_rand = np.clip(h_rand, 0, 1)
            starts.append(h_rand)
        
        # Optimize each start
        for start in starts:
            # Check if the start is feasible
            if np.abs(np.sum(start) - n / 2) > 1e-5:
                continue
            # Run optimization
            h_opt, c5_opt = optimize(start, n)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    return (best_h, best_c5, len(best_h))
