                full_h_opt = np.concatenate([h_opt, h_opt[::-1]])
                dx = 2.0 / n
                h_1_minus = 1.0 - full_h_opt
                corr = np.correlate(full_h_opt, h_1_minus, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = full_h_opt
                    best_n = n
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    return (best_h, best_c5, best_n)
