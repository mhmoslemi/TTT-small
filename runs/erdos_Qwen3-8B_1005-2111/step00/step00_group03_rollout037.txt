        initial_h = np.random.rand(n_points)
        initial_h = initial_h / initial_h.sum() * (n_points / 2)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - (n_points / 2)})
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max()
    
    def gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = corr.argmax()
        k = max_index - (n_points - 1)
        grad = np.zeros(n_points)
        for j in range(n_points):
            term1 = 1 - h[j + k] if 0 <= j + k < n_points else 0.0
            term2 = h[j - k] if 0 <= j - k < n_points else 0.0
            grad[j] = term1 - term2
        return grad
    
    result = minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=constraints, jac=gradient, options={'maxiter': 1000})
    
    h_values = result.x
    h_values = h_values / h_values.sum() * (n_points / 2)
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    c5_bound = corr.max()
    
    return (h_values, c5_bound, n_points)
