        optimized_h = result.x
        # Compute C5 for the optimized solution
        current_c5 = compute_c5(optimized_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5
        
        # Update the elapsed time
        elapsed_time = time.time() - start_time
        # print(f"n_points={n_points}, best_c5={best_c5}, elapsed={elapsed_time}")
    
    return best_h, best_c5, len(best_h)

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
