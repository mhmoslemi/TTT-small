    start_time = time.time()
    result = differential_evolution(objective, bounds, strategy='best1bin', 
                                   popsize=15, mutation=(0.5, 1), recombination=0.8, 
                                   maxiter=100000, tol=1e-6, disp=False, 
                                   polish=True, seed=seed)

    elapsed = time.time() - start_time
    if elapsed > budget_s:
        if 'initial_h_values' in globals():
            h_values = initial_h_values
            c5_bound = objective(h_values)
        else:
            h_values = np.random.rand(n_points)
            h_values = repair(h_values)
            c5_bound = objective(h_values)
    else:
        h_values = result.x
        c5_bound = objective(h_values)

    return (h_values, c5_bound, n_points)
