    population_size = 100
    max_generations = 5000
    mutation_rate = 0.1
    eta_crossover = 10
    eta_mutation = 10
    elitism_rate = 0.02
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(corr) * (2.0 / n_points)
            fitness_val = 1.0 / (1.0 + max_overlap)
            fitness.append((fitness_val, h))
        
        # Sort by fitness
        fitness.sort(reverse=True)
        # Select elites
        elite_count = int(population_size * elitism_rate)
        elites = fitness[:elite_count]
        
        # Select parents
        parents = []
        for _ in range(population_size - elite_count):
            a = random.randint(0, len(fitness)-1)
            b = random.randint(0, len(fitness)-1)
            if fitness[a][0] > fitness[b][0]:
                parents.append(fitness[a][1])
            else:
                parents.append(fitness[b][1])
        
        # Crossover
        new_population = []
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                break
            p1 = parents[i]
            p2 = parents[i+1]
            c1, c2 = sbx_crossover(p1, p2, eta_crossover)
            new_population.append(c1)
            new_population.append(c2)
        
        # Mutate
        for i in range(len(new_population)):
            new_population[i] = polynomial_mutation(new_population[i], mutation_rate, eta_mutation)
        
        # Repair
        for i in range(len(new_population)):
            new_population[i] = repair(new_population[i], n_points)
        
        # Combine
        population = elites + new_population
        
        # Update best
        for fit, h in fitness:
            current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
    
    return (best_h, best_c5, n_points)

def sbx_crossover(parent1, parent2, eta):
    u = np.random.rand(len(parent1))
    child1 = np.zeros_like(parent1)
    child2 = np.zeros_like(parent2)
    for i in range(len(parent1)):
        if u[i] <= 0.5:
            beta = (1 + 2 * (u[i] ** (eta + 1))) ** (-1 / (eta + 1))
            child1[i] = parent1[i] + beta * (parent2[i] - parent1[i])
            child2[i] = parent2[i] + beta * (parent1[i] - parent2[i])
        else:
            beta = (1 + 2 * (1 - u[i]) ** (eta + 1)) ** (-1 / (eta + 1))
            child1[i] = parent1[i] - beta * (parent1[i] - parent2[i])
            child2[i] = parent2[i] - beta * (parent2[i] - parent1[i])
        child1[i] = np.clip(child1[i], 0, 1)
        child2[i] = np.clip(child2[i], 0, 1)
    return child1, child2

def polynomial_mutation(child, mutation_rate, eta):
    for i in range(len(child)):
        if np.random.rand() < mutation_rate:
            u = np.random.rand()
            if u <= 0.5:
                delta = (1 - child[i]) * (1 - u ** (eta + 1)) ** (1 / (eta + 1))
                child[i] += delta
            else:
                delta = child[i] * (1 - u ** (eta + 1)) ** (1 / (eta + 1))
                child[i] -= delta
            child[i] = np.clip(child[i], 0, 1)
    return child

def repair(h, n_points):
    T = n_points / 2
    S = np.sum(h)
    if abs(S - T) < 1e-6:
        return h
    scale = T / S
    h_scaled = h * scale
    h_clipped = np.clip(h_scaled, 0, 1)
    return h_clipped
