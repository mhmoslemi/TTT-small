import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values')
    start_time = time.time()
    best_solution = None
    best_c5 = float('inf')
    n_list = [45, 90, 180, 360]
    dx = 2.0

    def compute_smooth_objective(h, t=1000):
        n_val = len(h)
        g = 1 - h
        h_pad = np.fft.rfft(h)
        g_pad = np.fft.rfft(g)
        corr_fft = np.fft.irfft(h_pad * g_pad)
        max_val = np.max(corr_fft)
        if max_val < 0:
            shift = -max_val + 1e-10
            corr_shifted = corr_fft + shift
            smooth_max_val = (1/t) * np.log(np.sum(np.exp(t * (corr_shifted + 1e-10))))
        else:
            smooth_max_val = (1/t) * np.log(np.sum(np.exp(t * (corr_fft + 1e-10))))
        return smooth_max_val * dx

    def compute_smooth_gradient(h, t=1000):
        n_val = len(h)
        g = 1 - h
        h_pad = np.fft.rfft(h)
        g_pad = np.fft.rfft(g)
        corr_fft = np.fft.irfft(h_pad * g_pad)
        grad = np.zeros(n_val)
        exp_corr = np.exp(t * (corr_fft + 1e-10))
        sum_exp = np.sum(exp_corr)
        for k in range(n_val):
            i_max = n_val - 1 - k
            if i_max >= 0:
                term = np.sum(exp_corr[:i_max+1] * g[:i_max+1+k])
            else:
                term = 0
            grad[k] = dx * (1/t) * (1/sum_exp) * term
        return grad

    def repair_solution(h, n, dx_val):
        h_clip = np.clip(h, 0, 1)
        total_clip = np.sum(h_clip) * dx_val
        if total_clip < 1 - 1e-5:
            scale = (1 / total_clip) * dx_val
            h_clip = np.minimum(1, h_clip * scale)
        elif total_clip > 1 + 1e-5:
            scale = (1 / total_clip) * dx_val
            h_clip = h_clip * scale
        h_clip = np.clip(h_clip, 0, 1)
        total_clip = np.sum(h_clip) * dx_val
        if abs(total_clip - 1) > 1e-5:
            scale = (1 / total_clip) * dx_val
            h_clip = h_clip * scale
        return h_clip

        total = np.sum(h) * (2.0 / n)
        h = h * (1 / total)
        return h

                initial_guesses = [generate_random_feasible(n) for _ in range(6)]
