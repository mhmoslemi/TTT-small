    for n in n_list:
        if time.time() - start_time > budget_s:
            break
        if n == 47 and 'initial_h_values' in globals():
            initial_guess = initial_h_values
        else:
            if best_h is None:
                initial_guess = generate_random_feasible(n)
            else:
                initial_guess = map_solution(best_h, len(best_h), n)
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2})
        result = minimize(objective, initial_guess, args=(n,), method='SLSQP', bounds=bounds, constraints=cons)
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h, n)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        else:
            current_h = initial_guess
            current_c5 = objective(current_h, n)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    return best_h, best_c5, len(best_h)
