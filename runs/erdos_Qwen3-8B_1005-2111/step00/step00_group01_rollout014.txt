        # Define objective function
        def objective(h):
            return compute_c5(h, n)
        
        # Constraints
        cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2})
        bounds = [(0.0, 1.0) for _ in range(n)]
        
        # Run optimization
        result = minimize(objective, current_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        else:
            current_c5 = compute_c5(current_h, n)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    return best_h, best_c5, n_list[-1]
