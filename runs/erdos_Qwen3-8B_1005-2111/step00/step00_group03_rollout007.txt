        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_h = h.copy()
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        new_h_list = []
        for i in ones:
            for j in zeros:
                new_h = current_h.copy()
                new_h[i] = 0
                new_h[j] = 1
                new_h_list.append(new_h)
        
        for new_h in new_h_list:
            current_c5 = compute_c5(new_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = new_h.copy()
        
        current_h = best_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
