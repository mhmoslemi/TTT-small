import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_list = [47, 94, 188, 376]
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():

    def objective(h, n_points):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Feasibility check
    def is_feasible(h, n_points):
        integral = np.sum(h) * (2.0 / n_points)
        return np.abs(integral - 1.0) < 1e-5 and np.all((h >= 0) & (h <= 1))
    
    # Solution mapping
    def map_solution(parent_h, n_parent, n_child):
        x_parent = np.linspace(0, 2, n_parent)
        x_child = np.linspace(0, 2, n_child)
        return np.interp(x_child, x_parent, parent_h)
    
    # Optimization loop
    for n in n_list:
        if time.time() - start_time > budget_s:
            break
        # Generate initial guess
        if n == 47:
            h_initial = initial_h
        else:

                h_initial = map_solution(best_h, len(best_h), n)

                h_initial = h_initial / np.sum(h_initial) * (n / 2)
