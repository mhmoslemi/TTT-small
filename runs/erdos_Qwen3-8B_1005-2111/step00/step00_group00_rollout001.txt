import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 45
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)

    def repair(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return np.full(n_points, 0.5)
        return h * (n_points / 2) / s

    def objective(h):
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    result = differential_evolution(
        objective, bounds, strategy='best1bin', popsize=15, mutation=(0.5, 1), recombination=0.9,
        tol=1e-6, maxiter=1000, seed=seed, polish=True, disp=False
    )
    best_h = repair(result.x)
    return (best_h, result.fun, n_points)
