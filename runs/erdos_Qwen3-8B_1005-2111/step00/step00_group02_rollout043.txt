        result = minimize(objective, initial_guess, method='Nelder-Mead', bounds=bounds, tol=1e-6)
        h_opt = result.x
        sum_h = np.sum(h_opt)
        if abs(sum_h - n_points / 2) > 1e-6:
            h_opt = h_opt / sum_h * (n_points / 2)
        h_array = np.array(h_opt)
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        current_c5 = max_corr * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt
    return best_h, best_c5, len(best_h)
