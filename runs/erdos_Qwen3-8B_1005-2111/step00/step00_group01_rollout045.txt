        for i in range(n_child):
            j = i // 2
            child_h[i] = parent_h[j]
        return child_h
    
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def objective(h, n_points):
        return compute_c5(h, n_points)
    
    def is_feasible(h, n_points):
        integral = np.sum(h) * (2.0 / n_points)
        return np.abs(integral - 1.0) < 1e-5 and np.all((h >= 0) & (h <= 1))
    
    initial_h = initial_h_values if 'initial_h_values' in globals() else generate_feasible(47)
    current_h = initial_h.copy()
    
    for n in n_list:
        if time.time() - start_time > budget_s:
            break
        if n == 47:
            h_initial = initial_h.copy()
        else:
            if best_h is None:
                h_initial = generate_feasible(n)
            else:
                h_initial = map_solution(best_h, n // 2, n)
        
        h_initial = np.clip(h_initial, 0, 1)
        total = np.sum(h_initial) * (2.0 / n)
        if np.abs(total - 1.0) > 1e-5:
            scale = 1.0 / total if total != 0 else 1.0
            h_initial *= scale
        
        def optimize_func(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            total = np.sum(h) * (2.0 / n)
            if np.abs(total - 1.0) > 1e-5:
                scale = 1.0 / total if total != 0 else 1.0
                h *= scale
            return objective(h, n)
        
        result = minimize(optimize_func, x0=h_initial, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n)], options={'maxiter': 100})
        best_h_candidate = result.x
        c5_candidate = compute_c5(best_h_candidate, n)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = best_h_candidate
    
    return best_h, best_c5, len(best_h)
