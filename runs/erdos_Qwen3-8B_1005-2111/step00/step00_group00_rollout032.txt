        return max_corr * dx

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
    
    if result.success:
        h_opt = result.x
        c5_bound = result.fun
    else:
        h_opt = h_initial
        c5_bound = objective(h_initial)
    
    return (h_opt, c5_bound, n_points)
