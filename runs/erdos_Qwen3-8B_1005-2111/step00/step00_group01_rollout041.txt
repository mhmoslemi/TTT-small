                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x.copy()
            else:
                result = minimize(objective, h_start, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
                if np.allclose(np.sum(result.x), n_points / 2, atol=1e-5) and np.all(0 <= result.x) and np.all(result.x <= 1):
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x.copy()

            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break

    return (best_h, best_c5, len(best_h))
