import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    # Use provided initial_h_values directly
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(x):
        # Compute overlap integrals
        corr = correlate(x, 1 - x, mode='full') * dx
        max_corr = np.max(corr)
        # Penalize sum constraint
        penalty = 1e4 * (np.sum(x) - target_sum) ** 2
        return max_corr + penalty

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
    h_opt = result.x
    c5 = np.max(correlate(h_opt, 1 - h_opt, mode='full') * dx)
    return (h_opt, c5, n_points)
