import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_start = len(initial_h)
    best_c5 = float('inf')
    best_h = np.zeros(n_start)
    n_points_list = [n_start, 2 * n_start, 4 * n_start, 8 * n_start]
    time_used = 0.0

    def objective(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - len(h) / 2

    def feasible(h):
        return np.all(0 <= h) and np.all(h <= 1) and np.isclose(np.sum(h), len(h)/2, atol=1e-5)

    def map_grid(old_h, new_n):
        if new_n % len(old_h) != 0:
            return None
        new_h = np.zeros(new_n)
        for i in range(len(old_h)):
            new_h[i::len(old_h)] = old_h[i]
        return new_h

    for n_points in n_points_list:
        if time_used >= budget_s:
            break
        if n_points < n_start:
            continue
        if n_points == n_start:
            start_h = initial_h.copy()
        else:
            mapped = map_grid(best_h, n_points)
            if mapped is not None:
                start_h = mapped
            else:
                start_h = np.random.rand(n_points)
                while not feasible(start_h):
                    start_h = np.random.rand(n_points)
        start_h = np.clip(start_h, 0, 1)
        while feasible(start_h):
            try:
                res = opt.minimize(objective, start_h, method='SLSQP',
                                   bounds=[(0,1)]*n_points,
                                   constraints={'type': 'eq', 'fun': constraint_sum})
                if res.success:
                    h_candidate = res.x
                    if feasible(h_candidate):
                        current_c5 = objective(h_candidate)
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_h = h_candidate.copy()
                            time_used += time.time() - time_used
                            break
                time_used += 0.1
            except:
                pass
            start_h = np.random.rand(n_points)
            while not feasible(start_h):
                start_h = np.random.rand(n_points)
            time_used += 0.1
    return (best_h, best_c5, len(best_h))
