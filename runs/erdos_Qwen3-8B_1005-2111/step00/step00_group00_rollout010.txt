        row[-1] = -dx * np.sum(h)  # -dx * sum_h
        for i in range(n_points):
            j = (i + k) % n_points
            row[i] = dx * h[i]  # Approximate linear term
        A_ineq.append(row)
        b_ineq.append(0.0)

    A_ineq = np.array(A_ineq)
    b_ineq = np.array(b_ineq)

    # Solve LP
    c = c.reshape(-1, 1)
    A_eq = A_eq.reshape(1, -1)
    b_eq = b_eq.reshape(-1, 1)
    A_ineq = A_ineq.reshape(-1, n_points + 1)
    b_ineq = b_ineq.reshape(-1, 1)

    # Use scipy's linprog
    res = linprog(c, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq, bounds=[(0, 1)]*n_points + [(None, None)], method='highs')

    if res.success:
        h_values = res.x[:n_points]
        t = res.x[-1]
        c5_bound = t * dx
    else:
        h_values = h
        c5_bound = np.max([np.correlate(h, 1 - h, mode='full') * dx])

    return (h_values, c5_bound, n_points)
