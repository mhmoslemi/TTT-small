            for idx in idxs:
                if delta_sum > 0:
                    add = min(delta_sum, 1.0 - h_clip[idx])
                    h_clip[idx] += add
                    delta_sum -= add
                else:
                    sub = min(-delta_sum, h_clip[idx])
                    h_clip[idx] -= sub
                    delta_sum += sub
        h = h_clip
        corr_new = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(corr_new) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        alpha *= 0.9995
    
    if time.time() - start_time < budget_s:
        h140 = np.repeat(h, 2)
        sum_h140 = np.sum(h140)
        scale = (n_points / 2) / sum_h140
        h140 = h140 * scale
        corr = np.correlate(h140, 1 - h140, mode='full')
        c5_new = np.max(corr) * (2.0 / 140)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h140.copy()
            n_points = 140
    
    return best_h, best_c5, n_points
