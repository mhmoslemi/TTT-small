import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    # Assume initial_h_values is provided
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values

    # Objective function
    def objective(h):
        h_array = np.array(h)
        corr = correlate(h_array, 1 - h_array, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Constraints
    def constraint(h):
        return np.sum(h) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimization
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 200})
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
