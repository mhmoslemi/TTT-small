        max_sub = h[indices]
        total_possible = np.sum(max_sub)
        if total_possible < -deficit:
            sub = total_possible / len(indices)
            h[indices] -= sub
            h = np.clip(h, 0, 1)
        else:
            sub = -deficit / len(indices)
            h[indices] -= sub
            h = np.clip(h, 0, 1)
    return h

def objective(h):
    n_points = len(h)
    h_repaired = repair(h, n_points)
    corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    return max_corr * dx

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        h_initial = np.zeros(45)
        h_initial[:22] = 1
        h_initial[22] = 0.5
    n_points = len(h_initial)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = differential_evolution(objective, bounds, strategy='best1bin',
                                   maxiter=1000, popsize=15, tol=1e-6,
                                   mutation=(0.5, 1), recombination=0.8,
                                   seed=seed, disp=False)
    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
