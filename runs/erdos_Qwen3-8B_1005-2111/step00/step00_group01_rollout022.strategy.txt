            # Objective function
            def objective(h):
                dx = 2.0 / n
                corr = np.correlate(h, 1-h, mode='full')
                max_corr = np.max(corr)
                return max_corr * dx

            # Neighborhood function: given h, returns a candidate h_candidate by flipping two bits (if possible)
            # We assume h is binary (0/1) and has sum(n/2)

            # We will use a fixed number of iterations or time-based.
            # We'll use a cooling schedule.

            T = 1.0
            alpha = 0.995
            best_h = h_start.copy()
            best_obj = objective(best_h)

            current_h = best_h.copy()
            current_obj = best_obj

            start_time_sa = time.time()
            while time.time() - start_time_sa < time_limit:
                # Generate a candidate
