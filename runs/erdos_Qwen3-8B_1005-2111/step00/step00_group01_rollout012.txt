            # Define equality constraint: sum(h) = n / 2
            cons = LinearConstraint(np.ones(n), [n/2], [n/2])

            # Run optimization
            result = minimize(
                objective,
                h_start,
                method='COBYLA',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100}
            )

            # Check if the result is feasible
            if not result.success:
                continue

            # Compute the C5 value
            h_opt = result.x
            corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n
            c5 = max_corr * dx

            # Update best solution if better
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

            # Check if time is exceeded
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                return (best_h, best_c5, n)

    # After processing all grid sizes, return the best solution
    return (best_h, best_c5, len(best_h))
