        return h_scaled
    
    def objective(h):
        h_repaired = repair(h)
        one_minus_h = 1 - h_repaired
        corr = np.correlate(h_repaired, one_minus_h, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        return max_corr * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = differential_evolution(
        objective, bounds, strategy='best1bin',
        popsize=15, mutation=(0.5, 1), recombination=0.9,
        tol=1e-6, maxiter=10000, polish=True, seed=seed
    )
    elapsed_time = time.time() - start_time
    
    if elapsed_time > budget_s:
        return (initial_h, objective(initial_h), n_points)
    
    best_h = result.x
    best_h_repaired = repair(best_h)
    one_minus_h = 1 - best_h_repaired
    corr = np.correlate(best_h_repaired, one_minus_h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n_points
    final_c5 = max_corr * dx
    
    return (best_h_repaired, final_c5, n_points)
