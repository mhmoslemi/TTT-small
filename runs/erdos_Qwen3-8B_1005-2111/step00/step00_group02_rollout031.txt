import numpy as np
from scipy.optimize import minimize
import time

def compute_smooth_objective(h, t=1000):
    n = len(h)
    g = 1 - h
    h_pad = np.fft.rfft(h)
    g_pad = np.fft.rfft(g)
    corr_fft = np.fft.irfft(h_pad * g_pad)
    max_val = np.max(corr_fft)
    shift = -max_val + 1e-10
    corr_shifted = corr_fft + shift
    exp_corr = np.exp(t * (corr_shifted + 1e-10))
    sum_exp = np.sum(exp_corr)
    return (np.log(sum_exp) / t) * 2.0 / n

def compute_smooth_gradient(h, t=1000):
    n = len(h)
    g = 1 - h
    h_pad = np.fft.rfft(h)
    g_pad = np.fft.rfft(g)
    corr_fft = np.fft.irfft(h_pad * g_pad)
    exp_corr = np.exp(t * (corr_fft + 1e-10))
    sum_exp = np.sum(exp_corr)
    grad = np.zeros(n)
    for k in range(n):
        i_max = n - 1 - k
        if i_max >= 0:
            i_range = np.arange(0, i_max + 1)
            term = np.sum(exp_corr[i_range] * g[i_range + k])
        else:
            term = 0
        grad[k] = (1 / t) * (term / sum_exp) * 2.0 / n
    return grad

def repair_solution(h, n, dx):
    h_clip = np.clip(h, 0, 1)
    total = np.sum(h_clip) * dx
    if total < 1 - 1e-5:
        scale = (1 / total) * dx
        h_clip = np.minimum(1, h_clip * scale)
    elif total > 1 + 1e-5:
        scale = (1 / total) * dx
        h_clip = h_clip * scale
    h_clip = np.clip(h_clip, 0, 1)
    total = np.sum(h_clip) * dx
    if abs(total - 1) > 1e-5:
        scale = (1 / total) * dx
        h_clip = h_clip * scale
    return h_clip

def generate_random_feasible(n):
