from scipy.optimize import minimize
import time

        corr = np.fft.irfft(h_pad * g_pad)

        else:
            return (1/t) * np.log(np.sum(np.exp(t * (corr + 1e-10)))) * dx

    def compute_smooth_grad(h, t=1000):
        n = len(h)
        g = 1 - h
        h_pad = np.fft.rfft(h)

                term = 0

        return grad

            h = np.clip(h, 0, 1)
