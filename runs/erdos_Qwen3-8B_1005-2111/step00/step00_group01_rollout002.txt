import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    n_list = [47, 94, 188, 376]
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

            h = np.copy(initial)

        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1
        h *= scale
        return h
    
    def map_solution(parent_h, n_parent, n_child):
        child_h = np.zeros(n_child)
        for i in range(n_child):
            x = i * 2.0 / (n_child - 1)
            j = int(np.floor(x * (n_parent - 1) / 2.0))
            child_h[i] = parent_h[min(j, n_parent - 1)]
        return child_h
    
    for n in n_list:
        if time.time() - start_time > budget_s:
