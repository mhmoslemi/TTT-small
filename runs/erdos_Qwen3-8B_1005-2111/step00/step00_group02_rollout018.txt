            current_h = np.clip(result.x, 0, 1)
            sum_current = np.sum(current_h)
            if not np.isclose(sum_current, n_points / 2):
                scale_factor = (n_points / 2) / sum_current
                current_h = current_h * scale_factor
            c5 = compute_c5(current_h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h
                best_n = n_points
    return (best_h, best_c5, best_n)
