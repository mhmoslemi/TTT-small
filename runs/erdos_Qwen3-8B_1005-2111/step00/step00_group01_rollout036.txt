import numpy as np
from scipy.optimize import minimize
from scipy.stats import dirichlet
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    grid_sizes = [47, 94, 188, 376]
    initial_h = initial_h_values if 'initial_h_values' in globals() else None

    for n in grid_sizes:
        if n == 47:
            starts = [initial_h] if initial_h is not None else []
        else:
            starts = []
        if initial_h is not None and n == 47:
            starts.append(initial_h)
        if n > 47:
            if best_h is not None:
                starts.append(np.repeat(best_h, 2))
        for _ in range(4):
            samples = dirichlet.rvs([1.0]*n, size=1)
            scaled = samples * (n / 2)
            starts.append(scaled[0])
        for start in starts:
            h0 = np.array(start)
            def obj(h):
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * (2.0 / n)
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2}]
            bounds = [(0, 1) for _ in range(n)]
            res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons)
            if res.success:
                h_opt = res.x
                if np.all(0 <= h_opt <= 1) and np.isclose(np.sum(h_opt), n / 2):
                    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
                    c5 = np.max(corr) * (2.0 / n)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt
                        best_n = n
            res_cobyla = minimize(obj, h0, method='COBYLA', bounds=bounds, constraints=cons)
            if res_cobyla.success:
                h_opt_cobyla = res_cobyla.x
                if np.all(0 <= h_opt_cobyla <= 1) and np.isclose(np.sum(h_opt_cobyla), n / 2):
                    corr_cobyla = np.correlate(h_opt_cobyla, 1 - h_opt_cobyla, mode='full')

                    if c5_cobyla < best_c5:

                        best_h = h_opt_cobyla
                        best_n = n
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
