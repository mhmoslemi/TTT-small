        if total_time >= budget_s:
            break

        h_opt = np.clip(result.x, 0, 1)
        sum_h = np.sum(h_opt)
        target_sum = n / 2
        delta = sum_h - target_sum
        if delta != 0:
            h_opt[-1] += delta
            h_opt = np.clip(h_opt, 0, 1)

        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        max_corr = np.max(corr)
        current_c5 = max_corr * (2.0 / n)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt

    return (best_h, best_c5, len(best_h) if best_h is not None else 45)
