import time
from scipy.optimize import minimize

    n_list = [47, 94, 188, 376]
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

                h = np.random.rand(n)

        def objective_func(x):
            h = x
            h_feasible = feasible_h(h, n)
