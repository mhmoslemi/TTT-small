            new_h = new_h / sum_new * 1
            new_h = np.clip(new_h, 0, 1)
            # Adjust sum again
            new_h = new_h / np.sum(new_h) * (n_points / 2)
            new_h = np.clip(new_h, 0, 1)
        # Now, perform local search
        # Perturb the new_h and evaluate
        # For simplicity, use a random search with a limited number of iterations
        for _ in range(100):
            # Perturb the new_h
            perturb = np.random.rand(n_points) * 0.1 - 0.05
            candidate = new_h + perturb
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate / np.sum(candidate) * (n_points / 2)
            # Evaluate C5
            current_c5 = evaluate_c5(candidate, n_points)
            if current_c5 < best_c5:
                best_h = candidate
                best_c5 = current_c5
        # Update elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        current_grid_size = n_points

    return (best_h, best_c5, len(best_h))
