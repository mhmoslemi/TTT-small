                res_cobyla = minimize(objective, start, method='COBYLA', bounds=bounds, constraints=cons, tol=1e-6)
                if res_cobyla.success:
                    current_h = res_cobyla.x
                    if np.all(current_h >= 0) and np.all(current_h <= 1) and np.isclose(np.sum(current_h), n_points / 2):
                        corr = np.correlate(current_h, 1 - current_h, mode='full')
                        current_c5 = np.max(corr) * (2.0 / n_points)
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_h = current_h
                            best_n = n_points
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                break
    return (best_h, best_c5, best_n)
