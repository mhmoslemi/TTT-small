import numpy as np

from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    n_list = [47, 94, 188, 376]
    best_h = None
    best_c5 = float('inf')
    best_n = 47

        h = h * (n / 2) / total

        # Generate initial guess

            if best_h is None:

                    j = int(np.floor(x_i * (n_prev - 1) / 2.0))
                    current_h[i] = best_h[min(j, n_prev - 1)]
                initial_guess = current_h
        
        # Define objective function
        def objective(h_vec):
            h = h_vec.reshape(-1)
            dx = 2.0 / n
