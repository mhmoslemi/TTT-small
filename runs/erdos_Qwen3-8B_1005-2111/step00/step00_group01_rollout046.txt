import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values
    n_points_list = [47, 94, 188, 376]
    best_h = initial_h
    best_c5 = float('inf')
    start_time = time.time()

    def generate_random_start(n_points):
        h = np.random.uniform(0, 1, n_points)
        total = np.sum(h)
        scale = (n_points / 2) / total
        h *= scale
        h = np.clip(h, 0, 1)
        return h

    def generate_starting_points(n_points, initial_h=None):
        starts = []
        if initial_h is not None:
            if n_points % 2 == 0:
                new_h = np.repeat(initial_h, 2)
                starts.append(new_h)
        for _ in range(4):
            starts.append(generate_random_start(n_points))
        return starts

    def objective(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def optimize_grid(n_points, initial_h):
        starts = generate_starting_points(n_points, initial_h)
        best_h_grid = None
        best_c5_grid = float('inf')
        for start in starts:
            try:
                res = minimize(objective, start, method='SLSQP', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}])
                if res.success:
                    if res.fun < best_c5_grid:
                        best_c5_grid = res.fun
                        best_h_grid = res.x
            except:
                pass
            try:
                res_cobyla = minimize(objective, start, method='COBYLA', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}])
                if res_cobyla.success:
                    if res_cobyla.fun < best_c5_grid:
                        best_c5_grid = res_cobyla.fun
                        best_h_grid = res_cobyla.x
            except:
                pass

    current_initial = initial_h
