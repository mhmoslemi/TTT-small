import numpy as np
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    n_points = 45
    if 'initial_h_values' in globals():
        h_initial = initial_h_values
    else:
        h_initial = np.full(n_points, 0.5)
    dx = 2.0 / n_points
    bounds = [(0.0, 1.0) for _ in range(n_points)]

        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:

        scale = (n_points / 2) / s
        return h * scale
    
    def objective(h):
        h_repaired = repair(h)
        corr = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr) * dx
    
    result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, mutation=(0.5, 1), recombination=0.9, tol=1e-6, maxiter=1000, seed=seed)
    best_h = result.x
