def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    import numpy.random as npr
    from scipy.signal import correlate

    n_points = len(initial_h)

    # Compute initial C5
    corr = correlate(initial_h, 1 - initial_h, mode='full')
