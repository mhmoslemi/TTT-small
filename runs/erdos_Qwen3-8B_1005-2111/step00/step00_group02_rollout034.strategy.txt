def compute_smooth_objective(h, t=1000):
    """Compute smooth maximum correlation"""
    n_val = len(h)
    g = 1 - h
    # FFT-based correlation
    h_pad = np.fft.rfft(h)
    g_pad = np.fft.rfft(g)
    corr_fft = np.fft.irfft(h_pad * g_pad)
    # Smooth maximum
    max_val = np.max(corr_fft)
    # Handle negative correlation values
    if max_val < 0:
        # Shift to positive domain
        shift = -max_val + 1e-10
        corr_shifted = corr_fft + shift
        smooth_max_val = (1/t) * np.log(np.sum(np.exp(t * (corr_shifted + 1e-10))))
    else:
        smooth_max_val = (1/t) * np.log(np.sum(np.exp(t * (corr_fft + 1e-10))))
    return smooth_max_val * dx

def compute_smooth_gradient(h, t=1000):
    """Compute gradient of smooth objective using FFT"""
    n_val = len(h)
    g = 1 - h
    # FFT-based correlation
    h_pad = np.fft.rfft(h)
    g_pad = np.fft.rfft(g)
    corr_fft = np.fft.irfft(h_pad * g_pad)
    # Gradient computation
    grad = np.zeros(n_val)
    # Precompute exponential terms
    exp_corr = np.exp(t * (corr_fft + 1e-10))
    sum_exp = np.sum(exp_corr)
    # Matrix formulation for gradient
    # Compute gradient using the formula:
    # grad[k] = dx * (1/t) * (1/sum_exp) * sum_i [exp_corr[i] * g[i+k]] for valid i
    for k in range(n_val):
        # Valid indices for i: i from 0 to n_val-1-k
        i_max = n_val - 1 - k
        if i_max < 0:
            term = 0
        else:
            # Sum over valid i
            i_range = np.arange(0, i_max+1)
            term = np.sum(exp_corr[i_range] * g[i_range+k])
        grad[k] = dx * (1/t) * (1/sum_exp) * term
    return grad

def repair_solution(h, n, dx):
    """Ensure solution satisfies constraints"""
    # Clip values to [0,1]
    h_clip = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    total_clip = np.sum(h_clip) * dx
    if total_clip < 1 - 1e-5:
        scale = (1 / total_clip) * dx
        h_clip = np.minimum(1, h_clip * scale)
        total_clip = np.sum(h_clip) * dx
    elif total_clip > 1 + 1e-5:
        scale = (1 / total_clip) * dx
        h_clip = h_clip * scale
    # Final clip to ensure [0,1]
    h_clip = np.clip(h_clip, 0, 1)
    # Adjust sum if needed
    total_clip = np.sum(h_clip) * dx
    if abs(total_clip - 1) > 1e-5:
        scale = (1 / total_clip) * dx
        h_clip = h_clip * scale
    return h_clip
