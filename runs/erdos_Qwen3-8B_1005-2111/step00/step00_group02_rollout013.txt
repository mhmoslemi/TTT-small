import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    # Initialize with provided initial_h_values or generate a baseline
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_prev = len(initial_h)
    else:
        n = 45
        initial_h = np.full(n, 0.5)
        n_prev = n

    n_points_list = [45, 90, 180, 360]
    best_h = None
    best_c5 = float('inf')
    elapsed_time = 0

    for i, n_points in enumerate(n_points_list):
        # Determine initial guess
        if i == 0:
            initial_guess = initial_h.copy()
        else:
            if best_h is None:
                continue  # Skip if no previous solution
            prev_n = n_points_list[i-1]
            if n_points != 2 * prev_n:
                continue  # Skip if scaling not possible
            initial_guess = np.repeat(best_h, 2)

        # Optimization setup
        def objective(x):
            h = x
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            dx = 2.0 / n_points
            sum_penalty = 1e6 * (np.sum(h) - n_points/2)**2
            bounds_penalty = 1e6 * np.sum(np.maximum(h - 1, 0) + np.maximum(-h, 0))
            return max_corr * dx + sum_penalty + bounds_penalty

        start_time = time.time()
        result = minimize(objective, initial_guess, method='Nelder-Mead', tol=1e-6)
        elapsed = time.time() - start_time
        elapsed_time += elapsed
        if elapsed_time > budget_s:
            break

        h_opt = result.x
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        max_corr = np.max(corr)
        dx = 2.0 / n_points
        c5 = max_corr * dx

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
