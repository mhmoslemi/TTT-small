            current_h = result.x
            current_c5 = compute_c5(current_h, n)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        
        if elapsed_time >= budget_s:
            break
    
    return (best_h.tolist(), best_c5, len(best_h))
