            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        def obj_grad_up(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            max_idx = np.argmax(corr)
            k = max_idx - n_points + 1
            grad = np.zeros_like(h)
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    grad[i] -= h[i + k]
                if 0 <= i - k < n_points:
                    grad[i] += (1 - h[i - k])
            return grad * dx
        
        result_up = minimize(obj_up, h_upsampled, method='SLSQP', jac=obj_grad_up, bounds=bounds, constraints=cons, options={'maxiter': 100})
        elapsed_up = time.time() - start_time
        if elapsed_up < budget_s:
            best_h = result_up.x
            best_c5 = result_up.fun
    
    return (best_h, best_c5, n_points)
