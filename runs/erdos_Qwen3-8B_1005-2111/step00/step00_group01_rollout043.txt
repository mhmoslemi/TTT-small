                scale = (n / 2) / sum_clipped
                h = h * scale
            else:
                h = initial_h
        else:
            prev_n = n_list[n_list.index(n) - 1]
            prev_h = best_h if best_h is not None else initial_h
            if prev_h is None:
                h = np.random.rand(n)
                sum_h = np.sum(h)
                scale = (n / 2) / sum_h
                h = h * scale
                h = np.clip(h, 0, 1)
                sum_clipped = np.sum(h)
                scale = (n / 2) / sum_clipped
                h = h * scale
            else:
                h = interpolate(prev_h, prev_n, n)
                h = np.clip(h, 0, 1)
                sum_clipped = np.sum(h)
                scale = (n / 2) / sum_clipped
                h = h * scale
        
        def objective(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        def constraint(x):
            return np.sum(x) - n / 2
        
        bounds = [(0, 1) for _ in range(n)]
        
        result = minimize(objective, x0=h, method='SLSQP', constraints=[{'type': 'eq', 'fun': constraint}], bounds=bounds)
        
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        else:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
    
    return best_h, best_c5, n
