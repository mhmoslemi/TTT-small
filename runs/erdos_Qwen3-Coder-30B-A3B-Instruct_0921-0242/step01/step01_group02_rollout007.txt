import numpy as np
import time
from scipy.signal import correlate

import cvxpy as cp

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_constraints(h, n_points, target_sum):
    """Project h onto sum constraint and box constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    weights = np.ones_like(h)
    weights[np.where(h <= 0)] = 0
    weights[np.where(h >= 1)] = 0
    if np.sum(weights) > 0:
        h += diff * weights / np.sum(weights)
        h = np.clip(h, 0, 1)
    return h

def binary_to_continuous_step(h_binary, n_points):
    """Convert binary vector to continuous step function."""
    if n_points <= 0:
        return np.array([])
    
    # Find all 1-blocks
    i = 0
    blocks = []
    while i < n_points:
        if h_binary[i] == 1:
            start = i
            while i < n_points and h_binary[i] == 1:
                i += 1
            end = i
            blocks.append((start, end))
        else:
            i += 1
    
    # Create continuous representation
    h_cont = np.zeros(n_points)
    for start, end in blocks:
        h_cont[start:end] = 1
    
    return h_cont

def simulated_annealing(n_points, dx, target_sum, max_time):
    """Run simulated annealing to find a good binary start vector."""
    start_time = time.time()
    np.random.seed(42)
    h_current = np.random.rand(n_points)
    h_current = h_current * (target_sum / np.sum(h_current))
    # Round to binary
    h_current = (h_current > 0.5).astype(float)
    h_current = project_to_constraints(h_current, n_points, target_sum)
    
    best_h = h_current.copy()
    best_c5 = c5_of_h(h_current, dx)
    
    temp = 1.0
    cooling_rate = 0.995
    max_iter = 50000
    
    current_iter = 0
    while time.time() - start_time < max_time and current_iter < max_iter:
        # Randomly choose to swap one 0 with one 1
        ones = np.where(h_current > 0.5)[0]
        zeros = np.where(h_current < 0.5)[0]
        if len(ones) > 0 and len(zeros) > 0:
            idx_1 = np.random.choice(ones)
            idx_0 = np.random.choice(zeros)
            
            new_h = h_current.copy()
            new_h[idx_1] = 0
            new_h[idx_0] = 1
            
            new_c5 = c5_of_h(new_h, dx)
            delta = new_c5 - best_c5
            
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h_current = new_h
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
        
        temp *= cooling_rate
        current_iter += 1
    
    return best_h, best_c5

def solve_binary_mip(n_points, core_shifts, initial_h, target_sum, max_time, upper_bound):
    """
    Solve the MIP model for given shifts.
    """
    if max_time <= 0:
        return None, None
    
    # Setup the MIP model using cvxpy
    h = cp.Variable(n_points, boolean=True)
    y = cp.Variable((n_points, n_points), boolean=True)
    t = cp.Variable()
    
    # Cardinality constraint
    constraints = [
        cp.sum(h) == target_sum
    ]
    
    # Linearization constraints
    for i in range(n_points):
        for j in range(i+1, n_points):
            k = j - i
            if k in core_shifts:
                constraints += [
                    y[i, j] <= h[i],
                    y[i, j] <= h[j],
                    y[i, j] >= h[i] + h[j] - 1,
                    y[j, i] <= h[i],
                    y[j, i] <= h[j],
                    y[j, i] >= h[i] + h[j] - 1
                ]
    
    # Shift constraint
    for k in core_shifts:
        # For shift k, sum over y[i, i+k]
        if k == 0:
            continue
        summ = 0
        for i in range(n_points - k):
            summ += y[i, i + k]
        constraints.append(t + summ >= target_sum)
    
    # Bound constraint
    if upper_bound is not None:
        constraints.append(t <= upper_bound)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, max_seconds=max_time * 0.9)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            solution = h.value
            if solution is not None:
                return solution, prob.value
    except Exception:
        pass
    
    return None, None

    k_star = max_corr_idx - len(h_sol) + 1
