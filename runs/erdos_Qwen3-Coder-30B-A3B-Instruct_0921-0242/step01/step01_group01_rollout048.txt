def project_to_feasible(u, n):
    """Project vector u onto feasible set: 0 <= h <= 1, sum(h) = n/2"""
    # Clip to bounds
    h = np.clip(u, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - n/2) < 1e-10:
        return h
    
    # Binary search for lambda to match sum
    def sum_with_lambda(lam):
        h_test = np.clip(u + lam, 0.0, 1.0)
        return np.sum(h_test) - n/2
    
    a, b = -1.0, 1.0
    for _ in range(30):
        lam = (a + b) / 2
        val = sum_with_lambda(lam)
        if val > 0:
            b = lam
        else:
            a = lam
    
    h = np.clip(u + lam, 0.0, 1.0)
    return h

def compute_c5_fft(h, dx):
    """Compute C5 using FFT for efficient convolution."""
    n = len(h)
    # Zero-pad to 2*n-1 for full correlation
    H = rfft(h, n=2*n-1)
    G = rfft(1-h, n=2*n-1)
    conv = irfft(H * G, n=2*n-1)
    return dx * np.max(conv)

def symmetrize(h):
    """Enforce mirror symmetry: h[i] = h[n-1-i]"""
    n = len(h)
    mid = n // 2
    if n % 2 == 1:
        # Odd-length: center element unchanged
        h_sym = np.zeros(n)
        h_sym[:mid+1] = h[:mid+1]
        h_sym[mid+1:] = h_sym[mid-1::-1]
    else:
        # Even-length: pair elements
        h_sym = np.zeros(n)
        h_sym[:mid] = h[:mid]
        h_sym[mid:] = h_sym[mid-1::-1]
    return h_sym

            best_h = h_final.copy()

        h_start = project_to_feasible(np.random.uniform(size=d), n_coarse)
        h_start = np.hstack([h_start, h_start[::-1]])

        Z = np.random.randn(λ, d) * σ

            h_symmetric[:d] = h[:d]
            h_symmetric[d:] = h[:d][::-1]
            H_offspring.append(h_symmetric)
        
        # Evaluate offspring
        for h in H_offspring:

            conv = np.zeros(2*n_coarse-1)

            fft_g = rfft(1-h, n=2*n_coarse-1)
            
            # Compute with early stop
            spec = fft_h * fft_g
            conv_fft = irfft(spec, n=2*n_coarse-1)
            max_val = np.max(conv_fft)
            
            # Check if optimization should be stopped early
            if max_val >= best_c5:
                C5s.append(best_c5)  # assign worst possible to avoid early acceptance
                continue
                
            c5 = dx_coarse * max_val
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            C5s.append(c5)
            eval_count += 1
        
        # Sort offspring
        indices = np.argsort(C5s)
        if len(indices) < mu:
            mu = len(indices)
        indices = indices[:mu]
        
        # Update mean
        z_samples = Z[indices]
        h_samples = [H_offspring[i] for i in indices]
        
        # Compute mean-shift vector
        h_mean = np.mean(h_samples, axis=0)
        μ = μ + cm * (h_mean[:d] - μ)
        
        # Update covariance matrix 
        Z_mean = np.dot(weights, z_samples)
        C = (1 - c1) * C + c1 * np.outer(Z_mean, Z_mean)  # Rank-one update
        C = C + (1 - c1 - c1 * np.sum(np.diag(C)**2)) / (d - 1) * np.eye(d)  # Additional update
        
        # Update step size
        σ = σ * np.exp(σ_adapt * (np.sqrt(np.mean(np.linalg.norm(z_samples, axis=1)**2)) 
                                 - np.sqrt(d)) / np.sqrt(d) - 1)
        if σ < σ_min:
            σ = σ_min
        
        gen_no_improve += 1
        if gen_no_improve > max_gen_no_improve:
            # Restart
            if restarts < max_restarts:
                restarts += 1
                gen_no_improve = 0
                μ = best_h[:d].copy()
                σ /= 2
            else:
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):

    # Step 4: Local hill-climb polishing

    h_final = h_refined.copy()
    best_c5 = c5_refined
    
    p_moves = min(6, n_fine // 20)

        if time.time() - start_time_local > 200 - 0.5:
            break

        indices = np.random.choice(n_fine, size=p_moves, replace=False)
        h_trial = h_final.copy()

        for i in indices:
            delta = np.random.uniform(-0.004, 0.004)
            h_trial[i] = np.clip(h_trial[i] + delta, 0.0, 1.0)
        
        # Project
        h_trial = project_to_feasible(h_trial, n_fine)
        
        # Check if improvement
        c5_trial = dx_fine * np.max(np.correlate(h_trial, 1-h_trial, mode='full'))
        if c5_trial < best_c5:
            h_final = h_trial.copy()
            best_c5 = c5_trial
            
        # Re-symmetrize periodically
        if step % 50 == 0:
            h_final = symmetrize(h_final)
            
    # Final projection to ensure constraints exactly met

    return (h_final, best_c5, n_fine)
