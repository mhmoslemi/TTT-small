```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.interpolate import interp1d

def correlation_fft(h):
    """Compute all f_k(h) = sum h_i * (1-h_{i+k}) using FFT"""
    N = len(h)
    h_pad = np.pad(h, (0, N), 'constant')
    g_pad = np.pad(1 - h, (0, N), 'constant')
    c = ifft(fft(h_pad) * fft(g_pad)).real
    return c[:N]

def weighted_gradient_fft(h, p):
    """Compute weighted gradient of log-sum-exp surrogate"""
    N = len(h)
    p_pad = np.pad(p, (0, N), 'constant')
    h_pad = np.pad(h, (0, N), 'constant')
    g_pad = np.pad(1 - h, (0, N), 'constant')
    
    w1 = ifft(fft(p_pad) * fft(g_pad)).real[:N]
    w2 = ifft(fft(p_pad[::-1]) * fft(h_pad)).real[:N]
    return w1 - w2

def project_onto_Ω(v):
    """Project onto {h : 0 ≤ h_i ≤ 1, sum h_i = N/2}"""
    N = len(v)
    z = np.clip(v, 0, 1)
    r = N/2 - np.sum(z)
    if abs(r) < 1e-12:
        return z
    I = np.where((z > 0) & (z < 1))[0]
    count = len(I)
    while count > 0 and abs(r) > 1e-12:
        delta = r / count
        z[I] += delta
        r -= delta * count
        I = np.where((z > 0) & (z < 1))[0]
        count = len(I)
    return np.clip(z, 0, 1)

def binary_from_top(h, count):
    """Round h to a binary vector with exactly count ones"""
    idx = np.argsort(h)[::-1]
    mask = np.zeros_like(h)
    mask[idx[:count]] = 1.0
    return mask

def fractional_correction(h_bin):
    """Try to fix sum via small LP with at most 8 fractional entries"""
    N = len(h_bin)
    # Identify borderline values
    border = np.where(np.abs(h_bin - 0.5) < 0.2)[0]
    if len(border) <= 8:
        # Solve small LP
        x = cp.Variable(len(border))
        constraints = [
            0 <= x, x <= 1,
            cp.sum(x) == (N/2 - np.sum(h_bin[border==0]))
        ]
        objective = cp.Minimize(0)  # dummy, will be updated
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                for i, idx in enumerate(border):
                    if abs(x.value[i]) < 1e-10:
                        h_bin[idx] = 0.0
                    elif abs(x.value[i] - 1.0) < 1e-10:
                        h_bin[idx] = 1.0
                    else:
                        h_bin[idx] = x.value[i]
        except:
            pass
    return h_bin

def swap(h, i, j):
    """Swap h[i] and h[j]"""
    h_swap = h.copy()
    h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
    return h_swap

def c5_of_h(h):
    """Compute C5 using correlation"""
    dx = 2.0 / len(h)
    corr = correlation_fft(h) * dx
    return np.max(corr)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    
    N = 256
    h = np.zeros(N)
    h[:N//2] = 1.0
    np.random.shuffle(h)
    h = project_onto_Ω(h)

    τ = 0.05
    outer_cycles = 6
    inner_iters = 150
    for outer in range(outer_cycles):
        h_prev = h.copy()
        alpha = 1.0
        for t in range(inner_iters):
            y = h + ((t-1)/(t+2)) * (h - h_prev)
            f = correlation_fft(y)
            f_scaled = f / τ
            # Stable softmax
            f_scaled -= np.max(f_scaled)
            p = np.exp(f_scaled)
            p /= np.sum(p)
            grad = weighted_gradient_fft(y, p)
            
            # Line search
            while True:
                z = y - alpha * grad
                z_proj = project_onto_Ω(z)
                F_tau_y = τ * np.log(np.sum(np.exp(f_scaled)))
                F_tau_z = τ * np.log(np.sum(np.exp(correlation_fft(z_proj) / τ)))
                if F_tau_z <= F_tau_y - 0.5 * alpha * np.dot(grad, grad):
                    break
                alpha *= 0.5
            h_new = z_proj
            h_prev, h = h, h_new
            if time.time() - start > budget_s - 30:
                break
        τ *= 0.5
        if time.time() - start > budget_s - 30:
            break

    h_bin = binary_from_top(h, N//2)
    h_refined = fractional_correction(h_bin)

    best_h = h_refined
    best_c5 = c5_of_h(best_h)

    # Swap hill climbing
    while time.time() - start < budget_s - 80:
        # Here just take random swaps and see improvement
        i = np.random.randint(N)
        j = np.random.randint(N)
        if i == j:
            continue
        cand = swap(best_h, i, j)
        cand_c5 = c5_of_h(cand)
        if cand_c5 < best_c5 - 1e-7:
            best_h, best_c5 = cand, cand_c5

    # Maybe upsample
    if time.time() - start < budget_s - 5:
        h_up = np.interp(np.linspace(0, 2, 512), np.linspace(0, 2, 256), best_h)
        h_up = project_onto_Ω(h_up)
        # SLSQP refinement
        res = opt.minimize(
            fun=lambda x: c5_of_h(x),
            x0=h_up,
            method='SLSQP',
            bounds=[(0,1)] * 512,
            constraints={'type':'eq','fun':lambda x: np.sum(x)-256},
            options={'maxiter':100, 'ftol':1e-9}
        )
        if res.success:
            final_h = res.x
            final_c5 = c5_of_h(final_h)
            if final_c5 < best_c5:
                best_h, best_c5 = final_h, final_c5
        n_points = len(final_h)
    else:
        n_points = N

    return (best_h, best_c5, n_points)
```
