import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import math

def c5_fft(h):
    """Compute C5 using FFT for binary h. Returns max correlation."""
    g = 1 - h
    H = rfft(h)
    G = rfft(g)
    c = irfft(H * G, 2*len(h) - 1)
    return np.max(c)

def repair_weight(h, target_sum):
    """Repair to exact sum by flipping bits."""
    current_sum = np.sum(h)
    diff = int(target_sum - current_sum)
    if diff > 0:  # Need to add ones
        zero_indices = np.where(h == 0)[0]
        if len(zero_indices) >= diff:
            flip_indices = np.random.choice(zero_indices, diff, replace=False)
            h[flip_indices] = 1
    elif diff < 0:  # Need to remove ones
        one_indices = np.where(h == 1)[0]
        if len(one_indices) >= -diff:
            flip_indices = np.random.choice(one_indices, -diff, replace=False)
            h[flip_indices] = 0
    return h

def crossover(p1, p2):
    """Two-point crossover."""
    n = len(p1)
    c1 = np.random.randint(1, n-1)
    c2 = np.random.randint(c1+1, n)
    child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
    return repair_weight(child, n/2)

def mutate_swap(h):
    """Swap mutation."""
    one_indices = np.where(h == 1)[0]
    zero_indices = np.where(h == 0)[0]
    if len(one_indices) > 0 and len(zero_indices) > 0:
        i1 = np.random.choice(one_indices)
        i0 = np.random.choice(zero_indices)
        h[i1] = 0
        h[i0] = 1
    return h

def hill_climb(h, max_swaps=500):
    """Targeted hill-climbing based on worst shift."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    for _ in range(max_swaps):
        corr = correlate(best_h, 1 - best_h, mode='full')
        k_star = np.argmax(corr) - (n - 1)
        if corr[np.argmax(corr)] == 0:
            break
        best_swap = None
        best_delta = 0
        for _ in range(200):
            i1 = np.random.randint(0, n)
            i0 = np.random.randint(0, n)
            if best_h[i1] == 1 and best_h[i0] == 0:
                delta = 0
                if i1 + k_star >= 0 and i1 + k_star < n:
                    delta -= dx * best_h[i1] * best_h[i1 + k_star]
                    delta += dx * (1 - best_h[i1]) * best_h[i1 + k_star]
                if i0 + k_star >= 0 and i0 + k_star < n:
                    delta += dx * best_h[i0] * best_h[i0 + k_star]
                    delta -= dx * (1 - best_h[i0]) * best_h[i0 + k_star]
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i1, i0)
        if best_swap is None:
            break
        i1, i0 = best_swap
        best_h[i1] = 0
        best_h[i0] = 1
    return best_h

            best_c5 = c5_fine
            best_individual = h_up
            
    final_c5 = c5_fft(best_individual)
    return (best_individual, final_c5, len(best_individual))
