import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def proj_simplex_box(h, n):
    """Project h onto simplex {sum(h) = n/2, 0<=h<=1} using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    # If we're already at the right sum, just clip to [0,1]
    if abs(diff) < 1e-10:
        return h
    
    # Otherwise adjust by adding constant (water-filling)
    # Find lambda such that sum(max(0, h + lambda)) = n/2
    # This is done by binary search on lambda
    h_sorted = np.sort(h)[::-1]
    h_cumsum = np.cumsum(h_sorted)
    
    # We can fail if we have more than n/2 ones already
    # But we won't hit that since we just clipped
    
    # Find threshold index
    # For lambda, we have new values h' = max(0, h + lambda)
    # If lambda >= -min(h), all elements stay positive
    # If lambda <= -max(h), all elements become 0
    
    # Binary search for lambda to make sum = n/2
    low, high = -np.max(h), 1.0 - np.min(h)
    
    def check_sum(lam):
        h_new = np.clip(h + lam, 0, 1)
        return np.sum(h_new) - n/2
    
    eps = 1e-8
    if abs(check_sum(low)) < eps:
        return np.clip(h + low, 0, 1)
    if abs(check_sum(high)) < eps:
        return np.clip(h + high, 0, 1)

    for _ in range(50):
        mid = (low + high) / 2.0
        val = check_sum(mid)
        if abs(val) < eps:
            return np.clip(h + mid, 0, 1)
        elif val > 0:
            high = mid
        else:
            low = mid
    
    # If we fail, just use the last valid
    final_h = np.clip(h + (low + high)/2, 0, 1)
    return final_h

def c5(h):
    """Compute maximum overlap integral C5."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def evaluate_c5(h):
    """Compute C5 with explicit correlation."""
    dx = 2.0 / len(h)
    # Perform full convolution
    corr = correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def worst_shift(h):
    """Return index of shift k giving maximum integral."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1-h, mode="full")
    max_idx = np.argmax(corr)
    # The actual shift k is from corr indices:
    # len(h)-1 to 0, 0 to len(h)-1 
    # We want to map index in [0..2*len(h)-2] to k in [-len(h)+1..len(h)-1]
    k = max_idx - (len(h) - 1)
    return k, corr[max_idx]

def gradient_step(h, k, alpha):
    """Perform a gradient step for shift k."""
    n = len(h)
    dx = 2.0 / n
    # Compute gradient: dI_k/dh_i = dx * (1 - 2 * h_{i+k})
    grad = np.zeros_like(h)
    for i in range(n):
        # i = current index
        # k = shift, to be applied to i
        # h_{i+k} means i+k with modular boundary conditions
        idx = (i + k) % n
        grad[i] = -2 * dx * h[idx] + dx
    
    # Step
    h_new = h - alpha * grad
    # Project back to feasible set 
    return proj_simplex_box(h_new, n)

def solve_lp_relaxation(n):
    """Solve the LP relaxation of C5 optimization."""
    # We solve:
    # max t  
    # s.t.
    #   sum_i h_i = n/2
    #   0 <= h_i <= 1  
    #   sum_i h_i h_{i+k} >= t   for all k != 0
    
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Build product constraints using McCormick envelopes
    # For each i, k with k != 0, we want constraint y_{i,k} >= h_i h_{i+k} 
    # This is non-convex, but could be linearized using backtracking.
    # For now, work with a simplified model.
    constraints = [cp.sum(h) == n/2]
    constraints += [0 <= h[i] for i in range(n)]
    constraints += [h[i] <= 1 for i in range(n)]
    
    # Encourage most of h to be near 0 or 1 for binary-like structure.
    # Directly add some quadratic constraints, but these are non-convex.
    # Instead, start with strong primitives: 
    # Suppose a nice periodized solution exists with large period.
    # But directly solving makes problems indeterminate.
    # So let's just get a reasonable starting h.
    
    # Do not solve full LP; instead produce a good heuristic pattern
    # That's more efficient.
    
    # Fallback if we couldn't solve abstractly:
    # generate interesting pattern like blocked structure
    h_vec = np.zeros(n)
    # Put spikes (1s) at equidistant intervals. It works.
    interval = max(1, n // 4)
    for i in range(n):
        if i % interval == 0:
            h_vec[i] = 1
    
    # Normalize
    h_vec = h_vec * (n / 2) / np.sum(h_vec)
    h_vec = np.clip(h_vec, 0, 1)
    h_vec = proj_simplex_box(h_vec, n)
    
    return h_vec

def anneal(h, time_limit):
    """Run simulated annealing on h to optimize C5."""
    n = len(h)
    dx = 2.0 / n
    start_time = time.time()
    current_h = h.copy()
    current_c5 = c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    T = 0.1  # Initial temperature
    alpha = 0.95
    min_T = 1e-8
    
    step_count = 0
    while time.time() - start_time < time_limit and T > min_T:
        # Propose new solution
        idx1 = np.random.randint(0, n)
        idx2 = np.random.randint(0, n)
        delta = np.random.uniform(-0.02, 0.02)
        h_new = current_h.copy()
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Ensure within bounds before projecting
        h_new = np.clip(h_new, 0, 1)
        h_new = proj_simplex_box(h_new, n)
        
        new_c5 = c5(h_new)
        if new_c5 < current_c5:
            current_h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            # Accept with probability
            prob = np.exp(- (new_c5 - current_c5) / T)
            if np.random.rand() < prob:
                current_h = h_new
                current_c5 = new_c5
        T *= alpha
        step_count += 1
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 400  # Good for resolution; reasonable for fast optimization
    dx = 2.0 / n_points
    
    # Phase 1: LP Relaxation (ish) to get a good initial seed
    # Simple but effective starting point
    h = np.zeros(n_points)
    # Uniformly spaced 1s separated by about 1/4 period
    period = max(1, n_points // 4)
    for i in range(n_points):
        if i % period == 0:
            h[i] = 1
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    h = proj_simplex_box(h, n_points)

    best_h = h.copy()
    best_c5 = c5(h)
    
    # Phase 2: Improve with gradient steps on worst shift
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        for _ in range(1000):
            if time.time() - start_time > budget_s - 1.0:
                break
            k, corr_val = worst_shift(best_h)
            # Do gradient descent on most offending shift
            step_size = 0.001
            h_trial = gradient_step(best_h, k, step_size)
            trial_c5 = c5(h_trial)
            if trial_c5 < best_c5:
                best_h = h_trial
                best_c5 = trial_c5
            else:
                # Occasionally keep even if no improvement, to escape local minimum
                # Only do so with small chance
                if np.random.rand() < 0.05:
                    best_h = h_trial
                    best_c5 = trial_c5
    
    # Phase 3: Annealing for global exploration
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 15:
        h_anneal, c5_anneal = anneal(best_h, remaining_time - 5)
        if c5_anneal < best_c5:
            best_h = h_anneal
            best_c5 = c5_anneal

    # Final verification
    return (best_h, best_c5, n_points)
