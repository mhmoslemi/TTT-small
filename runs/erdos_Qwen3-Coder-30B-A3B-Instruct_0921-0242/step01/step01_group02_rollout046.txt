```python
import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *
import math

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_values = [200, 250, 300]
    max_iterations_sa = 50000
    max_swaps_local = 2000
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in n_values:
        if time.time() - start_time > budget_s - 5:
            break
            
        dx = 2.0 / n
        target_sum = n / 2.0
        
        # Heuristic: Simulated Annealing
        def simulated_annealing():
            h = np.random.randint(0, 2, n)
            h = h * (target_sum / np.sum(h))
            h = np.round(h).astype(int)
            while np.sum(h) != target_sum:
                if np.sum(h) > target_sum:
                    # Random 1 to 0 flip
                    idx = np.random.choice(np.where(h == 1)[0])
                    h[idx] = 0
                else:
                    # Random 0 to 1 flip
                    idx = np.random.choice(np.where(h == 0)[0])
                    h[idx] = 1
            
            best_h = h.copy()
            best_c5_val = c5_of_h(h, dx)
            
            temp = 1.0
            alpha = 0.995
            for _ in range(max_iterations_sa):
                if time.time() - start_time > budget_s - 5:
                    break
                # Swap two random positions
                i1 = np.random.randint(0, n)
                i0 = np.random.randint(0, n)
                
                if h[i1] == h[i0]:
                    continue
                
                h[i1], h[i0] = h[i0], h[i1]
                
                new_c5 = c5_of_h(h, dx)
                if new_c5 < best_c5_val:
                    best_c5_val = new_c5
                    best_h = h.copy()
                
                delta = new_c5 - best_c5_val
                if delta > 0 and np.random.rand() > np.exp(-delta / temp):
                    h[i1], h[i0] = h[i0], h[i1]
                
                temp *= alpha
                
            return best_h, best_c5_val
        
        h_start, c5_start = simulated_annealing()
        
        # MIP solver
        # Build the model with latest core shifts
        h_vars = Variable(n, boolean=True)
        t = Variable()
        
        # Core shifts
        K_core = [1, n//4, n//2, 3*n//4]
        # Update corr_k for each k in core
        corr_terms = {}
        y_vars = {}
        # Add auxiliary variables for the products
        aux_count = 0
        for i in range(n):
            for j in range(i+1, n):
                if abs(i - j) in K_core:
                    y_vars[(i,j)] = Variable(boolean=True)
                    aux_count += 1
        
        constraints = []
        constraints.append(sum(h_vars) == target_sum)
        
        for i in range(n):
            for j in range(i+1, n):
                k = abs(i - j)
                if k in K_core:
                    # McCormick constraints for y_ij = h_i * h_j
                    constraints.append(y_vars[(i,j)] <= h_vars[i])
                    constraints.append(y_vars[(i,j)] <= h_vars[j])
                    constraints.append(y_vars[(i,j)] >= h_vars[i] + h_vars[j] - 1)
                    corr_terms.setdefault(k, []).append(y_vars[(i,j)])
        
        # Objective: minimize t
        for k in K_core:
            term = sum(corr_terms[k]) 
            constraints.append(t + term >= target_sum)
        
        # Set initial upper bound
        initial_upper_bound = c5_start * n / target_sum
        
        if len(constraints) > 0:
            try:
                prob = Problem(Minimize(t), constraints + [t <= initial_upper_bound])
                
                # Gurobi is not available, try CBC or other solvers (if available)
                # Since we can't easily use different solvers in this runtime,
                # We'll solve it with default cvxpy which first tries to use 
                # ECOS or other compatible solvers that can tackle these MIPs
                # But ECOS doesn't support MIPs. Let's use a different strategy
                # For now we'll approximate the hard part via a subset of iteration
                # because we are in a solver limited environment
                
                # Simpler numerical trick to develop a better extended solution:
                # we re-iterate local search based on the suggested geometric cooling shape
                # after wrapping multiple passes
                
                # Split the remaining budget for solving
                remaining_fraction = 0.3 * (budget_s - (time.time() - start_time))
                if remaining_fraction < 10:
                    remaining_fraction = 10
                    
                # Local minima walking + evaluative
                # Evaluate the best solution
                if c5_start < best_c5:
                    best_c5 = c5_start
                    best_h = h_start.copy()
                    best_n = n
                    
                # Try to improve within the best solution using swaps
                best_local_h = h_start.copy()
                best_local_c5 = c5_start
                
                for _ in range(max_swaps_local):
                    if time.time() - start_time > budget_s - 5:
                        break
                        
                    i1 = np.random.randint(0, n)
                    i0 = np.random.randint(0, n)
                    
                    if best_local_h[i1] == best_local_h[i0]:
                        continue
                        
                    # Swap
                    best_local_h[i1], best_local_h[i0] = best_local_h[i0], best_local_h[i1]
                    
                    new_c5 = c5_of_h(best_local_h, dx)
                    if new_c5 < best_local_c5:
                        best_local_c5 = new_c5
                    else:
                        # Revert if bad move
                        best_local_h[i1], best_local_h[i0] = best_local_h[i0], best_local_h[i1]
                        
                if best_local_c5 < best_c5:
                    best_c5 = best_local_c5
                    best_h = best_local_h.copy()
                    best_n = n
                        
            except:
                pass
                
        # Do a final final selection among our runs with various configurations
        # Unroll if need to execute final bundle
        
    # Ensure best result is preserved
    assert best_h is not None
    return (best_h, best_c5, best_n)
```
