import numpy as np
import time
import cvxpy as cp
from scipy import sparse
from scipy.signal import correlate

def build_Q(k, n):
    """Build sparse matrix for term h_i * h_{i+k}"""
    if k == 0:
        return sparse.csr_matrix((n, n))
    data = np.ones(n - abs(k))
    rows = np.arange(n - abs(k))
    cols = rows + k if k >= 0 else rows - k
    Q = sparse.csr_matrix((data, (rows, cols)), shape=(n, n))
    return 0.5 * (Q + Q.T)

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def solve_qp(n, Kmax, warm_h=None):
    """Solve convex QP for optimal h"""
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2
    ]
    
    for k in range(-Kmax, Kmax + 1):
        Q = build_Q(k, n)
        constraints.append(cp.quad_form(h, Q) >= n/2 - t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    if warm_h is not None:
        h.value = warm_h
    
    prob.solve(solver=cp.OSQP, verbose=False, max_iter=5000)
    
    return h.value, t.value

def binary_refine(h_frac, n, Kmax, dx, time_limit):
    """Refine fractional solution to binary one"""
    h_bin = np.zeros(n)
    
    # Initialize with largest values
    idx = np.argsort(-h_frac)[:n//2]
    h_bin[idx] = 1
    
    best = c5_of_h(h_bin, dx)
    best_h = h_bin.copy()
    corr = correlate(h_bin, 1-h_bin, mode='full')
    
    while time.time() < time_limit:
        try:
            i1 = np.random.choice(np.where(h_bin == 1)[0])
            i0 = np.random.choice(np.where(h_bin == 0)[0])
            
            # Tentative swap
            h_bin[i1] = 0
            h_bin[i0] = 1
            
            new_corr = correlate(h_bin, 1-h_bin, mode='full')
            new_c5 = np.max(new_corr) * dx
            
            if new_c5 < best:
                best = new_c5
                best_h = h_bin.copy()
                corr = new_corr
            else:
                # Revert swap
                h_bin[i1] = 1
                h_bin[i0] = 0
        except IndexError:
            break
    
    return best_h, best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Sweep over different discretizations
    for n in [350, 400, 450]:
        if time.time() - start_time > budget_s - 5:
            break
            
        dx = 2.0 / n
        Kmax = min(n // 2, int(0.8 * n))
        warm_h = None
        
        # Solve convex QP
        h_opt, t_val = solve_qp(n, Kmax, warm_h)
        
        # Clamp to feasibility
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n / 2.0) / np.sum(h_opt)
        
        # Binary refinement
        refine_start = time.time()
        h_bin, c5_refined = binary_refine(h_opt, n, Kmax, dx, start_time + budget_s - 5)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_bin.copy()
            best_n = n
            
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final verification at higher resolution if needed
    if best_h is not None:
        # High resolution to compute more precise C5
        fine_n = 2000
        fine_dx = 2.0 / fine_n
        h_fine = np.interp(np.linspace(0, 2, fine_n), 
                          np.linspace(0, 2, best_n), best_h)
        best_c5 = c5_of_h(h_fine, fine_dx)
        
    return best_h, best_c5, best_n
