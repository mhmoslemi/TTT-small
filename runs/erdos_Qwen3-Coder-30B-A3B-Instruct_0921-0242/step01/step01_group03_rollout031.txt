import numpy as np
import scipy.optimize as opt
from scipy.fft import fft, ifft
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 256
    dx = 2.0 / N
    τ_start = 0.05
    τ_factor = 0.5
    outer_cycles = 6
    inner_iters = 150
    max_swap_iterations = 5000
    
    # Initialize h as binary
    h = np.zeros(N)
    h[:N//2] = 1.0
    np.random.shuffle(h)
    h = project_onto_Ω(h)
    
    τ = τ_start
    
    # Outer loop with decreasing τ
    for outer in range(outer_cycles):
        if time.time() - start_time > budget_s - 50:
            break
            
        α = 1.0
        h_prev = h.copy()
        
        # Inner Nesterov iterations
        for t in range(inner_iters):
            if time.time() - start_time > budget_s - 50:
                break
                
            # Nesterov extrapolation
            beta = (t - 1) / (t + 2) if t > 0 else 0
            y = h + beta * (h - h_prev)
            
            # Compute all correlations via FFT
            f = correlation_fft(y)
            
            # Softmax weights
            p = np.exp(f / τ - np.max(f / τ))
            p /= np.sum(p)
            
            # Compute weighted gradient
            grad = weighted_gradient_fft(y, p)
            
            # Backtracking line search
            while True:
                z = y - α * grad
                z_proj = project_onto_Ω(z)
                Fτ_y = τ * np.log(np.sum(np.exp(f / τ)))
                Fτ_z = τ * np.log(np.sum(np.exp(correlation_fft(z_proj) / τ)))
                if Fτ_z <= Fτ_y - 0.5 * α * np.dot(grad, grad):
                    break
                α *= 0.5
            
            h_new = z_proj
            h_prev, h = h, h_new
        
        τ *= τ_factor
    
    # Rounding to binary
    h_bin = binary_from_top(h, N // 2)
    
    # Optional fractional correction
    h_refined = fractional_correction(h_bin)
    
    # Swap hill-climbing
    best_h = h_refined
    best_c5 = c5_of_h(best_h)
    
    swap_count = 0
    while time.time() - start_time < budget_s - 20 and swap_count < max_swap_iterations:
        i, j = random_one_zero_pair(best_h)
        if i == -1:
            break
        cand = swap(best_h, i, j)
        cand_c5 = c5_of_h(cand)
        if cand_c5 < best_c5 - 1e-7:
            best_h, best_c5 = cand, cand_c5
        swap_count += 1
    
    # Final evaluation
    final_c5 = c5_of_h(best_h)
    
    return (best_h, final_c5, N)

def correlation_fft(h):
    """Compute all correlation values using FFT"""
    N = len(h)
    h_pad = np.pad(h, (0, N), 'constant')
    g_pad = np.pad(1 - h, (0, N), 'constant')
    c = ifft(fft(h_pad) * fft(g_pad)).real
    return c[:N]

def weighted_gradient_fft(h, p):
    """Compute weighted gradient from p and h (not using loops)"""
    N = len(h)
    # 1 - h rev
    g1 = np.pad(1 - h, (0, N), 'constant')
    # h rev
    g2 = np.pad(h, (0, N), 'constant')
    
    # convolve p with g1
    grad1 = ifft(fft(p) * fft(g1)).real[:N]
    # convolve p_rev with g2
    p_rev = p[::-1]
    grad2 = ifft(fft(p_rev) * fft(g2)).real[:N]
    
    return grad1 - grad2

def project_onto_Ω(v):
    """Project onto { h in R^N : 0 <= h_i <= 1, sum h_i = N/2 }"""
    z = np.clip(v, 0, 1)
    r = len(z) / 2 - np.sum(z)
    if abs(r) < 1e-12:
        return z
    
    I = np.where((z > 0) & (z < 1))[0]
    while I.size > 0 and abs(r) > 1e-12:
        z[I] += r / len(I)
        r = len(z) / 2 - np.sum(z)
        if abs(r) < 1e-12:
            break
        # After update, some values might have hit bounds
        I = np.where((z > 0) & (z < 1))[0]
    
    return z

def binary_from_top(h, count):
    """Convert h to binary mask with count ones"""
    sorted_indices = np.argsort(h)[::-1]
    h_bin = np.zeros(len(h))
    h_bin[sorted_indices[:count]] = 1.0
    return h_bin

def fractional_correction(h_bin):
    """Try to adjust a binary h to be closer to feasible."""
    # For fixed h_bin pattern, do tiny LP on 1-2 border points
    # This is very fast and handles the edge case.
    indices = np.where((h_bin == 0) | (h_bin == 1))[0]
    border_indices = []
    for i in indices:
        if i > 0 and h_bin[i-1] != h_bin[i]:
            border_indices.append(i)
        elif i < len(h_bin)-1 and h_bin[i+1] != h_bin[i]:
            border_indices.append(i)
    
    # Only pick at most a few at the borders to adjust (as extra points)
    border_indices = border_indices[:4]
    
    if len(border_indices) == 0:
        return h_bin.copy()
    
    # Now we solve a quick LP to adjust the sum
    x = cp.Variable(len(border_indices))
    h_extended = h_bin.copy()
    for i, idx in enumerate(border_indices):
        h_extended[idx] = x[i]
    
    obj = cp.Minimize(cp.sum_squares(x - 0.5))  # encourage near-1/2
    constr = [x >= 0, x <= 1,
              cp.sum(h_extended) == len(h_bin)/2]
    prob = cp.Problem(obj, constr)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            for i, idx in enumerate(border_indices):
                h_bin[idx] = x.value[i]
    except:
        pass  # fallback to original
    return h_bin

def swap(h, i, j):
    """Perform swap between 1 at i and 0 at j"""
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    return h_new

def random_one_zero_pair(h):
    """Return random pair (one_index, zero_index) or (-1,-1) if not possible"""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return -1, -1
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    return i, j

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)
