```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.interpolate import interp1d
from scipy.fft import fft, ifft
import cvxpy as cp

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 400  # Size of discretization
    r = 6
    dx = 2.0 / n_points
    tau0 = 0.05
    rho = 0.5
    n_temperatures = 6

    # FFT helper functions for circular convolution
    def convolution_fft(y):
        """y is a real vector of length n"""
        n = len(y)
        y_padded = np.zeros(2*n)
        y_padded[:n] = y
        fft_y = fft(y_padded)
        conv_result = ifft(fft_y * np.conj(fft_y)).real[:n]
        return conv_result
    
    def compute_correlations(Y):
        """Computes all correlations c_k(Y) in O(r N log N)"""
        n, r = Y.shape
        c = np.zeros((n,))  # Sum over all columns
        for d in range(r):
            y_d = Y[:, d]
            c_d = convolution_fft(y_d)
            c += c_d
        return c
    
    def compute_gradient(Y, c):
        """computes gradient of phi_tau = - tau * log(sum exp(-c_k / tau))"""
        n, r = Y.shape
        # Softmax weights p_k
        c_tau = c / tau0
        exp_neg_c = np.exp(-c_tau)
        Z = np.sum(exp_neg_c)
        p = exp_neg_c / Z
        # Construct gradient with FFTs
        G = np.zeros_like(Y)
        for d in range(r):
            y_d = Y[:, d]
            # convolve p with y_d, and with reflected y_d
            y_padded = np.zeros(2*n)
            y_padded[:n] = y_d
            p_padded = np.zeros(2*n)
            p_padded[:n] = p
            # The convolution of p and y from left
            p_y = ifft(fft(p_padded) * fft(y_padded)).real[:n]
            # The convolution of p and y from right (reflected)
            y_ref = np.flip(y_d)
            y_ref_padded = np.zeros(2*n)
            y_ref_padded[:n] = y_ref
            p_y_rev = ifft(fft(p_padded) * fft(y_ref_padded)).real[:n]
            G[:, d] = p_y + p_y_rev
        return G # Gradient scaled by -tau0

    def project_onto_constraints(Y):
        """Project onto the constraints:
        1. h_i = ||Y_i||^2
        2. sum h_i = n/2
        3. 0 <= h_i <= 1
        """
        n, r = Y.shape
        h = np.sum(Y**2, axis=1)
        # 1. Clip h_i to [0,1]
        h = np.clip(h, 0, 1)
        # 2. Adjust for sum = n/2
        sum_h = np.sum(h)
        if abs(sum_h) < 1e-10:
            alpha = 1.0
        else:
            alpha = np.sqrt(n / (2 * sum_h))
        Y_new = Y * alpha
        h_new = np.sum(Y_new**2, axis=1)
        h_new = np.clip(h_new, 0, 1)
        # Re-inflate if needed
        sum_h_new = np.sum(h_new)
        if abs(sum_h_new) > 1e-10:
            alpha = np.sqrt(n / (2 * sum_h_new))
            Y_new = Y_new * alpha
        return Y_new
    
    def deterministic_swap_polish(h_binary, n_iter=150):
        """Polish using deterministic swaps"""
        n = len(h_binary)
        # compute original correlation values
        c_orig = np.correlate(h_binary, 1 - h_binary, mode='full')
        max_idx = np.argmax(c_orig)
        k_max = max_idx - (n - 1)
        # find mismatches for k_max
        mismatches = []
        for i in range(n):
            if h_binary[i] == 1 and h_binary[i + k_max] == 0:
                mismatches.append(i)  # i is a 1 but i+k is 0
            elif h_binary[i] == 0 and h_binary[i - k_max] == 1:
                mismatches.append(i, i - k_max)  # i is 0 but i-k is 1
        # Then try exhaustive swaps.
        # For efficiency, try only swaps that involve one of the mismatched 
        # positions of the worst-shift
        best_h = h_binary.copy()
        best_c5 = c5_of_h(best_h)
        for i in range(n_iter):  # limit to some reasonable amount
            if time.time() - start_time > budget_s - 2:
                break
            improved = False
            for j in range(len(mismatches)):
                i_idx = mismatches[j]  # e.g., a 1 that should be a 0, or vice versa
                for k_idx in range(n):
                    if i_idx == k_idx:
                        continue
                    if i_idx < k_idx and (h_binary[i_idx] == 1 and h_binary[k_idx] == 0):
                        # check if swapping improves the worst correlation
                        old_max_correlation = c_orig[k_max + (n-1)]
                        test_h = h_binary.copy()
                        test_h[i_idx] = 0
                        test_h[k_idx] = 1
                        new_c = np.correlate(test_h, 1-test_h, mode='full')
                        new_max = np.max(new_c)
                        if new_max < old_max_correlation:
                            h_binary = test_h
                            c_orig = new_c
                            improved = True
                            break
                if improved:
                    break
        return h_binary

    # Initial Y matrix (random initialization)
    Y0 = np.random.randn(n_points, r)
    Y = project_onto_constraints(Y0)
    h0 = np.sum(Y**2, axis=1)
    
    c0 = compute_correlations(Y)
    phi0 = -tau0 * np.log(np.sum(np.exp(-c0 / tau0)))
    current_best_phi = phi0
    current_best_Y = Y.copy()

    # Optimization loop
    tau_list = [tau0 * (rho ** i) for i in range(n_temperatures)]
    for tau in tau_list:
        if time.time() - start_time > budget_s - 5:
            break
        for i in range(200):
            if time.time() - start_time > budget_s - 5:
                break
            # compute correlations and gradient
            c = compute_correlations(Y)
            grad = compute_gradient(Y, c)
            # line search for gradient ascent
            alpha = 1.0
            while alpha > 1e-12:  # small but more than zero
                Y_new = Y + alpha * grad
                Y_new = project_onto_constraints(Y_new)
                h_new = np.sum(Y_new**2, axis=1)
                c_new = compute_correlations(Y_new)
                phi_new = -tau * np.log(np.sum(np.exp(-c_new / tau)))
                if phi_new > current_best_phi:
                    Y = Y_new.copy()
                    current_best_phi = phi_new
                    break
                alpha *= 0.5
            # End while iterate
            

    h_final = np.sum(Y**2, axis=1)
    # Convert to binary mask
    h_sorted_idx = np.argsort(h_final)[::-1]
    h_binary = np.zeros(n_points)
    h_binary[h_sorted_idx[:n_points//2]] = 1.0
    if np.abs(np.sum(h_binary) - n_points/2) > 1e-9:
        h_binary = np.zeros(n_points)
        h_binary[h_sorted_idx[:n_points//2]] = 1.
    
    # check which bound is better
    c5_continuous = c5_of_h(h_final)
    c5_binary = c5_of_h(h_binary)
    if c5_binary < c5_continuous:
        final_h = h_binary
        final_c5 = c5_binary
    else:
        final_h = h_final
        final_c5 = c5_continuous

    # deterministic swap polishing
    h_binary = deterministic_swap_polish(h_binary, 100)
    final_c5_swap = c5_of_h(h_binary)
    if final_c5_swap < final_c5:
        final_h = h_binary
        final_c5 = final_c5_swap

    # refinement if still worse
    if final_c5 > 0.3807:
        # small LP to modify a few border values
        # Find indices near 0.5 (borderline)
        border_indices = np.where(np.abs(h_final - 0.5) < 0.2)[0]
        if len(border_indices) < 8:
            # refine the final_h using small LP
            n_border = len(border_indices)
            if n_border > 0:
                h_sub = h_final[border_indices]
                # construct small LP:
                # minimize t s.t. sum(h_i) = n/2 and for each k: n/2 - sum(h_i * h_{i+k}) <= t
                # only tight constraints
                new_h = cp.Variable(n_border)
                t = cp.Variable()
                obj = cp.Minimize(t)
                # sum constraints
                cons = [
                    cp.sum(new_h) == 0.5 * n_points - np.sum(h_final) + np.sum(h_sub),
                    t >= 0.0
                ]
                for k in range(1, n_points):
                    c = 0.0
                    for i in range(n_points):
                        j = (i + k) % n_points
                        if i in border_indices and j in border_indices:
                            c += new_h[border_indices.index(i)] * new_h[border_indices.index(j)]
                        elif i in border_indices:
                            c += new_h[border_indices.index(i)] * h_final[j]
                        elif j in border_indices:
                            c += h_final[i] * new_h[border_indices.index(j)]
                        else:
                            c += h_final[i] * h_final[j]
                    cons.append(n_points/2 - c <= t)
                prob = cp.Problem(obj, cons)
                try:
                    prob.solve(solver=cp.ECOS, verbose=False)
                    if prob.status == 'optimal':
                        # update the values
                        for i in range(n_border):
                            idx = border_indices[i]
                            h_final[idx] = max(0, min(1, new_h.value[i]))
                except:
                    pass
    
    # Re-project after small changes
    # set h_final again
    h_final = np.clip(h_final, 0, 1)
    sum_h = np.sum(h_final)
    if abs(sum_h) > 1e-10:
        # Normalize appropriately to sum = n/2
        h_final = h_final * (n_points / 2) / sum_h
    
    # Final upsampling through interpolation and sub-gradient descent
    if n_points != 800:  # if not already larger resolution
        x_upsampled = np.linspace(0, 2, 800)
        h_upsampled = np.interp(x_upsampled, np.linspace(0, 2, n_points), h_final)
        h_upsampled = h_upsampled * 200 / np.sum(h_upsampled)
        n_upsampled = 800
        # Now, do short sub-gradient steps
        # sqrt initial: but start from our best discrete h now
        for i in range(30):
            if time.time() - start_time > budget_s - 1:
                break
            # evaluate current maximum correlation
            c = np.correlate(h_upsampled, 1 - h_upsampled, mode='full')
            max_k_idx = np.argmax(c)
            k_best = max_k_idx - (n_upsampled - 1)
            
            # compute sub-gradient
            grad = np.zeros_like(h_upsampled)
            for i in range(n_upsampled):
                pos1 = (i + k_best) % n_upsampled
                pos2 = (i - k_best) % n_upsampled
                grad[i] = (1 - h_upsampled[pos1]) - h_upsampled[pos2]
            
            # step with small learning rate
            h_upsampled = h_upsampled - 0.1 * dx * grad
            h_upsampled = np.clip(h_upsampled, 0, 1)
            
            # renormalize
            sum_h = np.sum(h_upsampled)
            h_upsampled = h_upsampled * (n_upsampled / 2) / sum_h
        
        final_h = h_upsampled
        final_c5 = c5_of_h(final_h)
        
    return (final_h, final_c5, len(final_h))
```
