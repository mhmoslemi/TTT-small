import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp
from cvxpy import SCS
from scipy.linalg import eigh

def evaluate_c5(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def create_shift_matrices(n, max_k):
    """Create sparse shift matrices S_k."""
    S = {}
    for k in range(-max_k, max_k + 1):
        if k == 0:
            # Identity matrix
            S[k] = np.eye(n)
        else:
            # Diagonal shift
            S[k] = np.eye(n, k)
            if k < 0:
                S[k] = S[k][:, -k:]
            else:
                S[k] = S[k][:, :-k]
    return S

def solve_sdp(n, S, max_k):
    """Solve SDP relaxation."""
    m = (n + 1) // 2  # Reduced dimension due to symmetry
    X = cp.Variable((m, m), symmetric=True)
    t = cp.Variable()

    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) == 1.,  # Diagonal entries = 1
        cp.sum(X) == 0,  # Zero sum constraint (sum over all entries = 0)
    ]

    # Add constraints for each shift k
    for k in range(-max_k, max_k + 1):
        if k in S:
            constraints.append(t >= (n/4. - 0.25 * cp.trace(S[k][:, :m] @ X)))

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=SCS, eps=1e-6, max_iters=5000, verbose=False)
        return X.value, t.value
    except Exception:
        return None, None

def rounding(X_sol, n):
    """Randomized rounding using the SDP solution."""
    m = X_sol.shape[0]
    V, _ = eigh(X_sol)
    V = V[:, -m:]  # Keep only last m columns (for positive semidefinite matrix)
    samples = []
    
    for _ in range(200):
        g = np.random.randn(m)
        y = V.T @ g
        x = np.sign(y)
        
        # Adjust for zero sum
        s = int(np.sum(x))
        if s != 0:
            # Flip smallest entries
            flip_idx = np.argsort(np.abs(y))[:abs(s)//2]
            x[flip_idx] = -x[flip_idx]
        samples.append(x)
        
    return samples

def binary_refinement(h_start, max_swaps=5000):
    """Perform local refinement of binary solution."""
    h = h_start.copy()
    n = len(h)
    g = 1 - h
    
    corr = correlate(h, g, mode='full')
    cmax = np.max(corr)
    
    for _ in range(max_swaps):
        i0 = np.random.choice(np.where(h == 0)[0])
        i1 = np.random.choice(np.where(h == 1)[0])
        
        # Create new h by swapping positions i0 and i1
        h_new = h.copy()
        h_new[i0] = 1
        h_new[i1] = 0
        
        # Update correlation efficiently
        corr_new = corr.copy()
        # Only update relevant parts of correlation
        for k in range(-n+1, n):
            old_contribution_i0 = h[i0] * g[i0 - k] if 0 <= i0 - k < n else 0
            old_contribution_i1 = h[i1] * g[i1 - k] if 0 <= i1 - k < n else 0
            new_contribution_i0 = h_new[i0] * g[i0 - k] if 0 <= i0 - k < n else 0
            new_contribution_i1 = h_new[i1] * g[i1 - k] if 0 <= i1 - k < n else 0
            
            corr_new[k + n - 1] += (new_contribution_i0 - old_contribution_i0 +
                                   new_contribution_i1 - old_contribution_i1)
        
        cmax_new = np.max(corr_new)
        
        if cmax_new < cmax:
            h = h_new
            corr = corr_new
            cmax = cmax_new
            
    return h, cmax

def build_symmetric_x(X_full):
    """Construct the symmetric X matrix from the half-sized one."""
    n = 2 * X_full.shape[0] - 1
    X_sym = np.zeros((n, n))
    for i in range(X_full.shape[0]):
        for j in range(X_full.shape[0]):
            if i < X_full.shape[0] and j < X_full.shape[0]:
                X_sym[i, j] = X_full[i, j]
                X_sym[n - 1 - i, n - 1 - j] = X_full[i, j]
    return X_sym

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    

    grids = [(140, 70), (200, 100), (260, 130)]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    best_x_sol = None

    for (n_orig, m) in grids:
        if time.time() - start_time > budget_s - 30:
            break

        n = n_orig
        dx = 2. / n
        max_k = int(0.9 * n)
        
        # Create shift matrices
        S = create_shift_matrices(n, max_k)
        
        # Solve SDP
        S_reduced = {k: S[k][:m, :m] for k in S}
        X_sol, t_star = solve_sdp(n, S_reduced, max_k)
        
        if X_sol is None or t_star is None:
            continue
        
        # Rounding
        samples = rounding(X_sol, n)

        # Evaluate best sample and refine
        best_sample = None
        c5_min = float('inf')
        for x in samples:
            # Reconstruct full symmetric vector h
            h = np.zeros(n)
            h[:m] = (x + 1) / 2.
            h[m:] = (x[::-1] + 1) / 2.
            if np.abs(np.sum(h) - n/2) > 1e-6:
                # Adjust balancing manually
                diff = np.sum(h) - n/2
                n_flip = int(abs(diff))
                flip_indices = np.argsort(np.abs(h - 0.5))[:n_flip] if diff > 0 else np.argsort(np.abs(h - 0.5))[-n_flip:]
                for idx in flip_indices:
                    h[idx] = 1 - h[idx]
            c5 = evaluate_c5(h, dx)
            if c5 < c5_min:
                c5_min = c5
                best_sample = h.copy()

        if c5_min < best_c5:
            best_c5 = c5_min
            best_h = best_sample.copy()
            best_n = n
            
        # Local refinement
        if time.time() - start_time > budget_s - 30:
            break
            
        refined_h, refined_c5 = binary_refinement(best_sample, max_swaps=5000)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h.copy()

        if best_c5 <= 0.3807:
            break
            
    return (best_h, best_c5, best_n)
