```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 128
    dx = 2.0 / n_points
    
    # Stage 1: SDP relaxation
    if time.time() - start_time > budget_s - 30:
        return (best_h, best_c5, n_points) if best_h is not None else (np.ones(n_points) * 0.5, 1.0, n_points)
        
    # SDP model setup
    h = Variable(n_points)
    X = Variable((n_points, n_points), PSD=True)
    t = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        X.diag() == h,
        t <= 0
    ]
    
    for k in range(1, n_points):
        T_k = np.zeros((n_points, n_points))
        for i in range(n_points - k):
            T_k[i, i + k] = 1
        constraints.append(trace(T_k @ X) >= t)
    
    # Objective: maximize t
    objective = Maximize(t)
    prob = Problem(objective, constraints)
    
    # Solve SDP with SCS
    try:
        prob.solve(solver=SCS, eps=1e-4, max_iters=5000)
    except:
        # fallback if no solution found
        pass
    
    if prob.status != 'optimal':
        h_sdp = np.ones(n_points) * 0.5
    else:
        h_sdp = h.value
    
    # Clip for safety
    h_sdp = np.clip(h_sdp, 0, 1)
    h_sdp_sum = np.sum(h_sdp)
    if abs(h_sdp_sum - n_points/2) > 1e-8:
        # Waterfilling to adjust sum
        def adjust_sum(target_sum):
            # Linear interpolation to find lambda
            low, high = -1.0, 1.0
            for _ in range(20):
                mid = (low + high) / 2
                trial = np.clip(h_sdp + mid, 0, 1)
                if np.sum(trial) < target_sum:
                    low = mid
                else:
                    high = mid
            return np.clip(h_sdp + (low + high)/2, 0, 1)
        h_sdp = adjust_sum(n_points/2)
    
    c5_sdp = evaluate_c5(h_sdp)
    if c5_sdp < best_c5:
        best_h = h_sdp.copy()
        best_c5 = c5_sdp
        
    # Stage 2: Randomized rounding
    if time.time() - start_time > budget_s - 20:
        return (best_h, best_c5, n_points)
    
    R = 30
    best_binary = None
    best_c5_rounded = float('inf')
    
    def to_binary(help_vec, target_sum):
        # Sort indices by descending values
        sorted_idx = np.argsort(help_vec)[::-1]
        # Find threshold to get exactly target_sum ones
        num_ones = int(target_sum)
        # Set threshold to make sum equal to target_sum
        h = np.zeros_like(help_vec)
        h[sorted_idx[:num_ones]] = 1
        return h
    
    for _ in range(R):
        if time.time() - start_time > budget_s - 10:
            break
        try:
            # Get matrix factorization - first assume X = h_sdp @ h_sdp.T (rank one approximation)
            # We'll sample a Gaussian vector g, then project onto the row space of h_sdp
            # given it's already approximately rank-1, we use its components
            y = np.random.randn(n_points)
            h_candidate = h_sdp + 0.001 * np.random.randn(n_points)  # add noise
            h_binary = to_binary(h_candidate, n_points / 2)
            c5_trial = evaluate_c5(h_binary)
            if c5_trial < best_c5_rounded:
                best_c5_rounded = c5_trial
                best_binary = h_binary.copy()
        except Exception:
            continue
    
    if best_binary is not None and best_c5_rounded < best_c5:
        best_h = best_binary.copy()
        best_c5 = best_c5_rounded
    
    # Stage 3: Smooth-max gradient refinement
    if time.time() - start_time > budget_s - 10:
        return (best_h, best_c5, n_points)
    
    # Now do smooth max gradient refinement
    h_init = best_h.copy()
    h = h_init.copy()
    beta = 0.5
    max_beta = 50.0
    step_size = 0.01
    
    def smooth_max(overlap_vals, beta):
        return np.log(np.sum(np.exp(beta * overlap_vals))) / beta
        
    def compute_direct_overlap(h):
        # full convolution with 1-h
        overlap = np.correlate(h, 1 - h, mode='full')
        overlap_vals = []
        for k in range(1, len(h)):
            overlap_vals.append(overlap[k + len(h) - 1])
        return np.array(overlap_vals) * dx
    
    max_iter = 300
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        beta = min(max_beta, beta * 1.1)
        
        # Compute overlaps
        overlaps = compute_direct_overlap(h)
        # weights
        weights = np.exp(beta * overlaps)
        weights /= np.sum(weights)
        
        # Approximate gradient direction - we need to compute trace term per shifted pair
        grad = np.zeros_like(h)
        for k in range(1, len(h)):
            shift = np.zeros_like(h)
            for i in range(len(h) - k):
                shift[i] = (1 - h[i + k]) - h[i - k]
            grad += weights[k-1] * shift
        
        # Try gradient step:
        h_trial = h - step_size * grad
        # project
        h_trial = np.clip(h_trial, 0, 1)
        if abs(np.sum(h_trial) - n_points/2) > 1e-8:
            # waterfilling adjust
            sum_diff = n_points/2 - np.sum(h_trial)
            delta = sum_diff / len(h_trial)
            h_trial = np.clip(h_trial + delta, 0, 1)
            # if that still violates, adjust manually:
            h_trial = np.clip(h_trial + sum_diff / np.sum(h_trial), 0, 1)
            
        # evaluate
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        
        h = h_trial.copy()
        
    # Stage 4: Final local swaps
    if time.time() - start_time > budget_s - 2:
        return (best_h, best_c5, n_points)
    
    num_swaps = 1000
    for _ in range(num_swaps):
        if time.time() - start_time > budget_s - 1:
            break
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if h[i] == 0 or h[j] == 1:
            continue
        delta = min(h[i], 1 - h[j], 0.01)
        h_swap = h.copy()
        h_swap[i] -= delta
        h_swap[j] += delta
        # Ensure boundary conditions
        h_swap = np.clip(h_swap, 0, 1)
        sum_diff = n_points/2 - np.sum(h_swap)
        if abs(sum_diff) > 1e-6:
            # slight adjustment to make sum correct
            h_swap += sum_diff / len(h_swap)
            h_swap = np.clip(h_swap, 0, 1)
            
        c5_new = evaluate_c5(h_swap)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_swap.copy()
    
    return (best_h, best_c5, n_points)
```
