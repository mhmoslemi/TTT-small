import numpy as np
import scipy.optimize as opt
import time
import scipy.fft as fft
from cvxpy import Variable, Minimize, Problem, norm, sum, multiply, diag, hstack, vstack

def correlation_fft(h):
    """Compute all f_k(h) values using FFT."""
    N = len(h)
    h_pad = np.concatenate([h, np.zeros(N)])
    g_pad = np.concatenate([1 - h, np.zeros(N)])
    c = fft.fft(h_pad) * fft.fft(g_pad)
    return np.real(fft.ifft(c))[:N]

def weighted_gradient_fft(h, p):
    """Compute weighted gradient: sum_k p_k * [(1-h_{i+k}) - h_{i-k}]"""
    N = len(h)
    # Inverse of p (reversed)
    p_rev = p[::-1]
    
    # Prepare for convolution
    h_pad = np.concatenate([h, np.zeros(N)])
    g_pad = np.concatenate([1 - h, np.zeros(N)])
    p_pad = np.concatenate([p, np.zeros(N)])
    p_rev_pad = np.concatenate([p_rev, np.zeros(N)])
    
    # w1 = conv(p, 1-h)
    w1 = fft.ifft(fft.fft(p_pad) * fft.fft(g_pad)).real[:N]
    # w2 = conv(p_rev, h)
    w2 = fft.ifft(fft.fft(p_rev_pad) * fft.fft(h_pad)).real[:N]
    
    return w1 - w2

def project_onto_Ω(v):
    """Project v onto { h : 0<=h_i<=1, sum(h)=N/2 }"""
    N = len(v)
    z = np.clip(v, 0, 1)
    r = N/2 - np.sum(z)
    iter_count = 0
    while abs(r) > 1e-12 and iter_count < 10:
        I = np.nonzero((z > 0) & (z < 1))[0]
        if len(I) == 0:
            break
        r = N/2 - np.sum(z)
        delta = r / len(I)
        z[I] += delta
        z = np.clip(z, 0, 1)
        iter_count += 1
    return z

def binary_from_top(h, count):
    """Round h to the largest 'count' entries being 1, others 0."""
    N = len(h)
    idxs = np.argsort(h)[::-1]
    h_bin = np.zeros(N)
    h_bin[idxs[:count]] = 1
    return h_bin

def fractional_correction(h_bin):
    """Try to include up to 4 fractional entries."""
    N = len(h_bin)
    one_positions = np.nonzero(h_bin)[0]
    zero_positions = np.nonzero(1 - h_bin)[0]
    if len(one_positions) == N//2:
        return h_bin
    
    # Find borderline values to keep
    border_values = []
    for i in range(N):
        if 0 < h_bin[i] < 1:
            border_values.append((i, h_bin[i]))
    # Sort by closeness to 0.5
    border_values.sort(key=lambda x: abs(x[1] - 0.5))
    
    # Choose up to 4 borderline entries
    keep_indices = set()
    for i, val in border_values[:4]:
        if i in one_positions or i in zero_positions:
            keep_indices.add(i)
    
    # Solve tiny LP
    if not keep_indices:
        return h_bin
    
    h = np.zeros(N)
    h[one_positions] = 1
    h[zero_positions] = 0
    
    # Keep indices as variables
    var_indices = list(keep_indices)
    x = Variable(len(var_indices))
    # Variable ranges
    bounds = [(0,1)] * len(var_indices)
    
    # Ensure sum stays N/2
    # h[~ind] remains 0 or 1
    # modify sum with: 1*|one_positions| + sum(x) = N/2
    sum_one = len(one_positions)
    # Constraint: sum_one + sum(x) = N/2
    constraint_sum = sum(x) == (N/2 - sum_one)
    # Objective: minimize C5 (for now just return h_bin since direct evaluation is time-constrained)
    # We do not solve full LP in practical implementation for time efficiency
    # Instead, we construct a feasible version by restoring to original exact it if allowed
    
    # Here we just return the binary vector since if the rounding is good, it should
    # We'll use a proxy method to verify constraint and adjust approximately
    return h_bin

def swap(h, i, j):
    """Swap h[i] and h[j]"""
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def upsample(h, N_new):
    """Linear interpolation from N to N_new samples."""
    N = len(h)
    x_orig = np.linspace(0, 2, N)
    x_new = np.linspace(0, 2, N_new)
    h_new = np.interp(x_new, x_orig, h)
    return h_new

def evaluate_c5(h):
    """Evaluate the exact C_5 value using correlation."""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    N = 256
    
    # Initial binary construction (half ones, half zeros)
    h = np.zeros(N)
    h[:N//2] = 1.0
    h = project_onto_Ω(h)
    
    c5 = evaluate_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    # Stages and steps for acceleration
    τ_start = 0.05
    τ_factor = 0.5
    outer_cycles = 6
    inner_iters = 150
    
    τ = τ_start
    α = 1.0
    
    for outer in range(outer_cycles):
        if time.time() - start_time > budget_s - 10:
            break
        h_prev = h.copy()
        for t in range(inner_iters):
            if time.time() - start_time > budget_s - 10:
                break
                
            y = h + ((t-1)/(t+2)) * (h - h_prev)
            
            f = correlation_fft(y)
            p = np.exp(f/τ - np.max(f/τ))
            p = p / np.sum(p)
            
            grad = weighted_gradient_fft(y, p)
            
            # Backtracking line search
            while True:
                if time.time() - start_time > budget_s - 10:
                    break
                    
                z = y - α * grad
                z_proj = project_onto_Ω(z)
                
                if τ > 0:
                    # Approximate F_τ in a way that is less expensive than actual evaluation
                    # Just estimate using current values in an ad-hoc way
                    F_tau_y = np.log(np.sum(np.exp(f/τ))) * τ
                    F_tau_z = np.log(np.sum(np.exp(correlation_fft(z_proj)/τ))) * τ
                    if F_tau_z <= F_tau_y - 0.5 * α * np.dot(grad, grad):
                        break
                else:
                    break
                    
                α *= 0.5
                
            h_new = z_proj
            h_prev, h = h, h_new
            
        τ *= τ_factor
        
        c5_new = evaluate_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    # Rounding and fractional correction
    h_bin = binary_from_top(h, N//2)
    h_corrected = fractional_correction(h_bin)
    
    # Update if this improved the solution
    c5_corrected = evaluate_c5(h_corrected)
    if c5_corrected < best_c5:
        best_c5 = c5_corrected
        best_h = h_corrected.copy()
    
    # Swap hill-climbing
    best_h = h_corrected.copy()
    best_c5 = c5_corrected
    if time.time() - start_time < budget_s - 40:
        max_swaps = 5000
        for _ in range(max_swaps):
            if time.time() - start_time > budget_s - 40:
                break
            
            # Sample a random swap
            one_positions = np.nonzero(best_h)[0]
            zero_positions = np.nonzero(1 - best_h)[0]
            
            if len(one_positions) == 0 or len(zero_positions) == 0:
                break
                
            i = np.random.choice(one_positions)
            j = np.random.choice(zero_positions)
            
            test_h = swap(best_h, i, j)
            test_c5 = evaluate_c5(test_h)
            if test_c5 < best_c5 - 1e-6:
                best_h = test_h
                best_c5 = test_c5
    
    # Final up-sampling and refinement if time
    if time.time() - start_time < budget_s - 10:
        # Try to upsample to 512 points
        try:
            h_up = upsample(best_h, 512)
            h_up = project_onto_Ω(h_up)
            
            # Quick SLSQP on upsampled
            bounds = [(0,1)]*512
            def fun(x):
                return evaluate_c5(x)
                
            res = opt.minimize(
                fun=fun,
                x0=h_up,
                method='SLSQP',
                bounds=bounds,
                constraints={'type':'eq','fun':lambda x: np.sum(x)-256},
                options={'maxiter':100, 'ftol':1e-9}
            )
            
            final_h = res.x
            final_c5 = evaluate_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
        except:
            pass

    return (best_h, best_c5, len(best_h))
