import numpy as np
import time
import cvxpy as cp
from scipy import sparse as sp
from scipy.signal import correlate

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def build_Q(k, n):
    """Build sparse matrix for quadratic term h_i * h_{i+k}"""
    if k == 0:
        return sp.csr_matrix((n, n))
    data = np.ones(n - abs(k))
    rows = np.arange(n - abs(k))
    cols = rows + k if k >= 0 else rows - k
    Q = sp.csr_matrix((data, (rows, cols)), shape=(n, n))
    return 0.5 * (Q + Q.T)

def solve_qp(n, Kmax, warm_h=None):
    """Solve convex QP for optimal h"""
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    
    for k in range(-Kmax, Kmax + 1):
        Q = build_Q(k, n)
        constraints.append(cp.quad_form(h, Q) >= n/2 - t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    if warm_h is not None:
        h.value = warm_h
    
    prob.solve(solver=cp.OSQP, verbose=False, max_iter=5000, eps_abs=1e-9)
    
    return h.value, t.value

def binary_refine(h_frac, n, dx, time_limit):
    """Refine fractional solution to binary one"""
    # Initialize binary vector with top n/2 entries = 1
    idx = np.argsort(-h_frac)[:n//2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1
    
    best = c5_of_h(h_bin, dx)
    best_h = h_bin.copy()
    
    # Precompute correlation for binary vector
    corr = np.correlate(h_bin, 1 - h_bin, mode='full')
    
    while time.time() < time_limit:
        i1 = np.random.choice(np.where(h_bin == 1)[0])
        i0 = np.random.choice(np.where(h_bin == 0)[0])
        
        # Try swap
        h_bin[i1] = 0
        h_bin[i0] = 1
        
        new_corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        new_c5 = np.max(new_corr) * dx
        
        if new_c5 < best:
            best = new_c5
            best_h = h_bin.copy()
        else:
            # Revert
            h_bin[i1] = 1
            h_bin[i0] = 0
    
    return best_h, best

def evaluate_c5(h, n_points, dx):
    """Compute C5 with high precision"""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Grid sizes to try
    n_values = [300, 350, 400, 450]
    Kmax_values = [int(0.75 * n) for n in n_values]
    
    # Warm start from initial construction
    warm_h = None
    n_warm = None
    
    for i, (n, Kmax) in enumerate(zip(n_values, Kmax_values)):
        if time.time() - start_time > budget_s - 5.0:
            break
            
        try:
            # Solve convex QP
            h_sol, c5_sol = solve_qp(n, Kmax, warm_h)
            
            # Binary refinement
            h_refined, c5_refined = binary_refine(h_sol, n, 2.0/n, time.time() + 20.0)
            
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
                best_n = n
                
            # Save for warm start
            warm_h = h_sol
            n_warm = n
            
        except Exception as e:
            continue
    
    # Final high precision evaluation with coarse grid
    if best_h is not None:
        dx = 2.0 / best_n
        # Use refined to compute final evaluation
        final_c5 = evaluate_c5(best_h, best_n, dx)
        return (best_h, final_c5, best_n)
        
    # Fallback to previous method if needed
    raise RuntimeError("Failed to find valid solution within time budget")
