import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller grid for speed and memory efficiency
    n = 200
    dx = 2.0 / n
    d = n // 2 + 1  # number of independent variables due to symmetry
    
    # Initialize with symmetric h values
    def initialize_h():
        if 'initial_h_values' in globals():
            # Interpolate to current size
            x_orig = np.linspace(0, 2, len(initial_h_values))
            x_new = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_orig, initial_h_values)
            # Mirror for symmetry
            h = np.concatenate([h_interp[:d], h_interp[d-1::-1]])
        else:
            # Start with a simple symmetric function
            h = np.ones(n)
            # Apply reflection for symmetry: h[i] = h[n-1-i]
            h = np.array([h[i] if i < d else h[n-1-i] for i in range(n)])
        h = h * (n / 2) / np.sum(h)
        return h
    
    # Initialize best solution
    best_h = initialize_h()
    best_c5 = compute_c5(best_h)
    best_n = n

    # Double oracle cutting plane method
    S = {0}  # Set of active shifts
    lambda_penalty = 0.1
    epsilon = 1e-6
    max_iterations = 50
    iteration = 0

    # Build the LP model once for efficiency
    w = cp.Variable(d)
    M = cp.Variable()
    p = {}  # p[i,k] variables
    
    # Create constraint dictionaries
    constraints = []
    obj = cp.Minimize(M - lambda_penalty * cp.sum_squares(w))

    # Add box bounds on w
    constraints.extend([w >= 0, w <= 1])
    # Add sum constraint
    constraints.append(cp.sum(w) == n / 4 + 0.5)

    while iteration < max_iterations and time.time() - start_time < budget_s - 10:
        # Reset p variables
        p.clear()
        # Create new variables and constraints
        new_vars = {}
        new_cons = []
        
        # Add p variables for all active shifts
        all_p = []
        for k in S:
            for i in range(n):
                if 0 <= i + k < n:
                    if (i, k) not in new_vars:
                        # Create variable
                        p_var = cp.Variable()
                        p[(i, k)] = p_var
                        new_vars[(i, k)] = p_var
                        all_p.append(p_var)
                        # Add McCormick constraints
                        new_cons.extend([
                            p_var <= w[i] if i < d else p_var <= w[n-1-i],
                            p_var <= w[i+k] if i+k < d else p_var <= w[n-1-(i+k)],
                            p_var >= 0,
                            p_var >= w[i] + w[i+k] - 1 if i+k < d else p_var >= w[n-1-i] + w[n-1-(i+k)] - 1
                        ])
        # Add overlap constraints
        for k in S:
            overlap_term = n / 2 - cp.sum([p[(i, k)] for i in range(n) if 0 <= i + k < n])
            new_cons.append(overlap_term <= M)
            
        # Update problem
        constraints.extend(new_cons)
        problem = cp.Problem(obj, constraints)
        
        # Solve the LP
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status == cp.OPTIMAL:
                # Recover full h vector from solution
                w_sol = w.value
                h = np.array([w_sol[i] if i < d else w_sol[n-1-i] for i in range(n)])
                # Check if solution is good enough
                c5_exact = compute_c5(h)
                if c5_exact < best_c5:
                    best_c5 = c5_exact
                    best_h = h.copy()
                # Compute bound from LP
                bound = M.value
                if abs(c5_exact - bound) < epsilon:
                    break
                
                # Add the worst shift as a new cut
                k_max = find_worst_shift(h)
                S.add(k_max)
                # Increase penalty strength
                lambda_penalty = min(5.0, lambda_penalty * 1.5)
            else:
                break  # Could not solve optimally or problem is infeasible
        except:
            break
        iteration += 1

    # Polishing with projected subgradient descent
    h = best_h.copy()
    c5 = best_c5
    alpha = 0.02
    max_polish_it = 200
    polish_it = 0
    
    while polish_it < max_polish_it and time.time() - start_time < budget_s - 5:
        # Find the worst shift
        k_max = find_worst_shift(h, return_k=True)
        # Compute subgradient
        grad = np.zeros(n)
        for i in range(n):
            if i + k_max < n:
                grad[i] += 1 - h[i + k_max]
            if i - k_max >= 0:
                grad[i] -= h[i - k_max]
                
        # Update h with projected step
        h_new = h - alpha * grad
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Force integral constraint
        h_new = enforce_sum(h_new, n, dx)
        # Symmetrize
        h_new = (h_new + h_new[::-1]) / 2
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            alpha *= 0.9  # Reduce step size
        polish_it += 1

    # Final refinement: small stochastic hill climb
    h_refined = best_h.copy()
    c5_refined = best_c5
    max_hill_climb = 1000
    hill_climb_count = 0
    
    while hill_climb_count < max_hill_climb and time.time() - start_time < budget_s - 1:
        # Perturb some indices
        num_perturb = min(6, n // 20)
        indices = np.random.choice(n, size=num_perturb, replace=False)
        h_trial = h_refined.copy()
        feedback = 0.003
        for i in indices:
            delta = np.random.uniform(-feedback, feedback)
            h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
        # Re-project
        h_trial = enforce_sum(h_trial, n, dx)
        h_trial = (h_trial + h_trial[::-1]) / 2
        c5_trial = compute_c5(h_trial)
        
        if c5_trial < c5_refined:
            h_refined = h_trial
            c5_refined = c5_trial
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
        hill_climb_count += 1
    
    return (best_h, best_c5, best_n)

def compute_c5(h):
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    result = np.max(conv) * dx
    return result

def find_worst_shift(h, return_k=False):
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    idx = np.argmax(conv)
    k = idx - (len(h) - 1)
    if return_k:
        return k
    return k

def enforce_sum(h, n, dx):
    target_sum = n / 2
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Water filling approach
    diff = current_sum - target_sum
    # For small differences, adjust the average
    if abs(diff) > 1e-6:
        # Adjust uniformly for now, more sophisticated handling is possible
        adjustment = diff / n
        h -= adjustment
        # Clamp again to [0,1]
        h = np.clip(h, 0, 1)
        # Correct for exact sum enforcing by small correction to all values
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            # Re-normalize using difference to get remaining sum correction
            acc = 0
            for i in range(n):
                acc += h[i]
            if abs(acc - target_sum) > 1e-10:
                diff_factor = (target_sum - np.sum(h)) / np.sum(h)
                h = h * (1 + diff_factor)
                h = np.clip(h, 0, 1)
                # Final normalization just in case
                scale = target_sum / np.sum(h)
                h = h * scale
    return h
