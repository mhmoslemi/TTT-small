```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math

def evaluate_erdos_solution(h):
    # This replicates the exact same computation used by the evaluation framework
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    N = 140  # Grid size, keep it manageable for MILP solver

    # Precompute valid indices for each shift k to avoid recomputation
    index_sets = {}
    for k in range(N):
        index_sets[k] = list(range(N - k))

    # --- MILP Construction ---
    # Variables
    x = cp.Variable(N, boolean=True)
    t = cp.Variable()
    # Create wrapper for all w variables in one array (for efficient access)
    w_vars = {}
    for k in range(N):
        if k == 0:
            w_vars[k] = []
        else:
            w_vars[k] = [cp.Variable(1, boolean=True) for i in index_sets[k]]
    
    # Optimization specific to cvxpy handling
    # Flatten all w variables into a single list for easier constraints
    w_all = []
    w_index_map = {}
    i = 0
    for k in range(N):
        for idx in index_sets[k]:
            w_index_map[(idx, k)] = i
            w_all.append(w_vars[k][idx - (N - len(index_sets[k]))])
            i += 1

    constraints = []
    S = N // 2  # Sum constraint
    constraints.append(cp.sum(x) == S)
    constraints.append(t >= 0)
    # Enforce symmetry-breaking: x[0] = 1
    constraints.append(x[0] == 1)

    # Add the three McCormick inequalities for each w_i,k
    for k in range(1, N):
        for idx in index_sets[k]:
            # w_{i,k} <= x_i
            constraints.append(w_vars[k][idx - (N - len(index_sets[k]))] <= x[idx])
            # w_{i,k} <= x_{i+k}
            constraints.append(w_vars[k][idx - (N - len(index_sets[k]))] <= x[idx + k])
            # w_{i,k} >= x_i + x_{i+k} - 1
            constraints.append(w_vars[k][idx - (N - len(index_sets[k]))] >= x[idx] + x[idx + k] - 1)

    # Add the overlap bounds: sum_{i in valid range} w_{i,k} >= S - t
    for k in range(N):
        if k == 0:
            # This handles the empty sum (because index_sets[0] is just full list)
            w_sum = cp.Constant(0)
        else:
            w_sum = cp.sum(w_vars[k])
        constraints.append(w_sum >= S - t)

    # Binary search on t
    L = 0
    U = S
    best_t = U
    best_x = None

    while time.time() - start_time < budget_s - 2:
        if U - L <= 1e-4:
            break
        T = (L + U) / 2
        prob = cp.Problem(cp.Minimize(0),
                          [constraints[:-1], # exclude t constraint
                           t <= T])
        try:
            prob.solve(solver=cp.CBC, 
                       warm_start=True, 
                       **{'max_seconds': 30})
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE] and t.value <= T + 1e-5:
                best_t = T
                # Extract the solution x
                x_vals = np.array([var.value for var in x])
                # Clip to 0/1 if it's fractional (from solver inaccuracy)
                x_vals = np.round(x_vals).astype(int)
                best_x = x_vals
                U = T
            else:
                L = T
        except:
            L = T

    if best_x is None:
        # Fallback if MILP failed
        # Greedy solution: place 1s in first S positions
        x_vals = np.zeros(N)
        x_vals[:S] = 1
        best_x = x_vals

    # Convert binary solution to step function
    h = np.array(best_x, dtype=float)
    dx = 2.0 / N
    current_c5 = evaluate_erdos_solution(h)

    # --- Fractional polishing ---
    # To avoid complex temporary 2.0 tuning, perform a simpler local search

    # Get relevant shift k which gave maximum correlation
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    max_k = np.argmax(corr) - (N - 1)
    max_corr_val = corr[np.argmax(corr)]

    # Run simple simultaneous fractional moves to reduce overlap term
    best_c5 = current_c5
    best_h = h.copy()
    tol = 1e-6

    # Try swaps in vector, perform heuristic moves using efficient method
    # Try a small fixed number of small displacement steps to reduce worst overlap
    max_moves = 50  # Must reset early to avoid time issues

    s = 0
    while s < max_moves and time.time() - start_time < budget_s - 5:
        s += 1
        # Find contributing indices to the worst shift (k_max)
        idxs = list(range(N - abs(max_k)))
        if max_k < 0:
            # Inverted shift
            shifted = [i + max_k for i in idxs]
            idxs = [i for i in range(N) if i + max_k in idxs]
        elif max_k > 0:
            # Forward shift
            idxs = [i for i in range(N) if i + max_k < N]

        # Pick one specific bad pair (we can do local switch in greedy way)
        # Choose a pair (i, i+max_k) where h[i]=1 and h[i+max_k]=0
        selected_pairs = []
        for i in idxs:
            if h[i] > 0.5 and h[i + max_k] < 0.5:
                selected_pairs.append((i, i + max_k))
        if not selected_pairs:
            continue

        # Try a single delta move per pair, actually check several : pick one with best reduction
        best_delta_val = np.inf
        best_i, best_j = None, None
        best_delta = 0.0

        for (i,j) in selected_pairs:
            delta_arr = np.zeros(N)
            # How much would changing from (1, 0) to (1-d, d) change the overlap?
            # For that particular term: h[i](1 - h[i+k]) changes by -d*(1 - h[i+k]) + d = d*(h[i+k] - 1)
            # But very approximate using normal analytical update
            delta_arr[i] = -0.03
            delta_arr[j] = 0.03

            h_test = h + delta_arr
            h_test = np.clip(h_test, 0, 1)
            c5_new = evaluate_erdos_solution(h_test)
            if c5_new < best_c5 and c5_new < best_delta_val:
                best_delta_val = c5_new
                best_i, best_j = i, j
                best_delta = 0.03

        if best_i is not None:
            h[best_i] -= 0.03
            h[best_j] += 0.03
            h = np.clip(h, 0, 1)
            best_c5 = best_delta_val
            current_c5 = best_c5
        elif s == 1:
            # first step didn't work, try cheaper approach
            # basic random one
            i = np.random.randint(0, N)
            delta = 0.02 if h[i] > 0.5 else -0.02
            h[i] += delta
            h = np.clip(h, 0, 1)
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5

    # Final swap search (binary swaps)
    # Define an easy local optimizer to improve discrete solution
    swap_attempts = 10000  # Bounded in time at least
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    remaining_time = budget_s - (time.time() - start_time)
    
    # Now we have a good starting point from a MILP-based solution and polish it further
    # With local swapping approach

    # Threshold for good stopping heuristic
    count_improvements = 0
    count_attempts = 0
    while count_attempts < swap_attempts and time.time() - start_time < budget_s - 5:
        count_attempts += 1
        # Randomly pick two different positions
        i = np.random.randint(0, N)
        j = np.random.randint(0, N)
        if i == j:
            continue
        # Compute current correlations with the associated overlap(s)
        if abs(h[i] - h[j]) < tol:
            continue
        # Swap their values
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_c5 = evaluate_erdos_solution(new_h)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            count_improvements += 1
            if count_improvements > 1000:
                break

    h = best_h
    result_c5 = best_c5

    return h, result_c5, N
```
