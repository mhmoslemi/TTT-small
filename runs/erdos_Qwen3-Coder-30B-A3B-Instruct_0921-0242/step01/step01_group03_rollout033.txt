import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import scipy.optimize as opt

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def evaluate_erdos_solution(h_values):
    # Returns the C5 bound for provided h values that discretizes [0,2] with 300 samples
    # Not needed for our approach but included to preserve API compatibility
    return c5_of_h(h_values)

def run(seed=42, budget_s=600, **kwargs):
    """Optimize to get a low C5 value for Erdős minimum overlap problem using MILP"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameter settings
    N = 140  # Grid size for better tractability while still beating target
    dx = 2.0 / N
    S = N // 2  # Target sum
    
    # Step 1: Construct MILP model
    # Binary variables for the step function
    x = cp.Variable(N, boolean=True)
    # Scalar variable to bound maximum overlap
    t = cp.Variable()
    
    # Auxiliary variables for quadratic terms: w[i,k] = x[i] AND x[i+k]
    w_vars = {}
    w_list = []
    
    # Define constraints for each shift k
    constraints = []
    
    # Add the binary constraint on x
    constraints.append(cp.sum(x) == S)
    
    # Symmetry breaking to remove translation invariance
    constraints.append(x[0] == 1)
    
    # For each shift k, create the w variable matrix
    for k in range(N):
        w_vars[k] = []
        for i in range(N - k):
            w = cp.Variable(boolean=True)
            w_vars[k].append(w)
            w_list.append(w)
            # Each w is the AND of x[i] and x[i+k]
            constraints.append(w <= x[i])
            constraints.append(w <= x[i + k])
            constraints.append(w >= x[i] + x[i + k] - 1)
    
    # Constraint that each c_k <= t where c_k = S - sum(w_{i,k})
    # So: sum(w_{i,k}) >= S - t
    for k in range(N):
        if len(w_vars[k]) > 0:
            constraints.append(cp.sum(w_vars[k]) >= S - t)
    
    # Add trivial bound constraint
    constraints.append(t >= 0)
    
    # Step 2: Binary-search approach
    L = 0.0
    U = S
    best_t = float('inf')
    best_x = None
    search_done = False
    
    # Binary search within time budget
    iters = 0
    while U - L > 1e-4 and time.time() - start_time < budget_s - 10:
        if iters > 10:
            break
        T = (L + U) / 2
        
        # Set up feasibility problem
        prob = cp.Problem(
            cp.Minimize(0),
            constraints + [t <= T]
        )
        
        try:
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=30)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # This means feasible, check if better solution
                if t.value < best_t:
                    best_t = t.value
                    best_x = x.value.copy()
            else:
                L = T
        except Exception:
            L = T
        else:
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                U = T
            else:
                L = T
        
        iters += 1
    
    # If binary search failed to find feasible, fallback to greedy solution
    if best_x is None:
        h_fallback = np.zeros(N)
        h_fallback[:S] = 1.0
        return (h_fallback, c5_of_h(h_fallback), N)
    
    # Step 3: Exact integer solution found, convert to step function
    h = np.array(best_x, dtype=float)
    
    # Step 4: Local polishing with fractional moves
    if time.time() - start_time < budget_s - 5:
        
        # Find worst shift to refine
        c_vals = []
        for k in range(N):
            c = S - sum(w_vars[k]) if len(w_vars[k]) > 0 else S
            c_vals.append(c)
        worst_k = np.argmax(c_vals)
        worst_c = np.max(c_vals)
        
        # Try to improve with small fractional changes
        max_iter = 200
        iteration = 0
        while iteration < max_iter and time.time() - start_time < budget_s - 5:
            iteration += 1
            improved = False
            
            # Look for pairs where 1 -> 0 and 0 -> 1
            for i in range(N):
                # Index in the corr that contributes to shift-k
                if i + worst_k < N:
                    if h[i] == 1 and h[i + worst_k] == 1:
                        # Try reducing i -> 0 and i+k -> 1 for small delta
                        new_h = h.copy()
                        new_h[i] = 0.0
                        new_h[i + worst_k] = 1.0
                        c5_val = c5_of_h(new_h)
                        if c5_val < c5_of_h(h):
                            h = new_h
                            improved = True
                            break
            
            if not improved:
                break
    
    # Step 5: Swap hill climbing
    if time.time() - start_time < budget_s - 2:
        max_swaps = 30000
        swapped = 0
        while swapped < max_swaps and time.time() - start_time < budget_s - 2:
            swapped += 1
            # Pick random 0- and 1-position
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Swap
            h_swap = h.copy()
            h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
            
            # Recompute with FFT for speed
            if len(h_swap) < 1000:
                c5_val = c5_of_h(h_swap)
                if c5_val < c5_of_h(h):
                    h = h_swap
            else:
                # fall back to exact evaluation if too large
                # but be careful about overhead
                pass
    
    # Final normalization to maintain sum = 1 (which should already hold since sum(x)=S, dx=2/N)
    # But just double-check
    actual_sum = np.sum(h) * dx
    h = h * (1.0 / actual_sum)
    
    return (h, c5_of_h(h), len(h))
