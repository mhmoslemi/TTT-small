def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def c5_of_binary(b):
    """Compute C5 for a binary mask (1s and 0s)"""
    n = len(b)
    dx = 2.0 / n
    h = b.astype(float)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def project_to_feasible(h):
    """Ensure sum(h) = n/2 and h in [0,1]^n"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    # Simple projection: clip to bounds then adjust
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    # Distribute deficit uniformly
    adjustment = deficit / n
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def binary_to_continuous(b, indices_to_modify):
    """Convert binary mask b to continuous h using indices_to_modify to tweak"""
    h = b.astype(float)
    # Keep the same sum constraint
    n = len(b)
    dx = 2.0 / n
    
    # Change at specified indices to make sum = n/2
    old_sum = np.sum(h)
    diff = (n/2) - old_sum
    if abs(diff) < 1e-10:
        return h
    
    # Artificially set a few indices to fractional vals
    # Make indices_to_modify have at most 4 entries (arbitrary limit)
    if not indices_to_modify:
        # If we change nothing:
        return h
    # First, constrain indices to valid
    valid_indices = [i for i in indices_to_modify if 0 <= i < n]
    if not valid_indices:
        return h
    # Quantize diff across these indices so that total change = diff
    to_adjust = len(valid_indices)
    if to_adjust <= 0:
        return h
    delta = diff / to_adjust
    for i in valid_indices:
        h[i] += delta
    # Ensure constraint still holds exactly
    h = project_to_feasible(h)
    return h
