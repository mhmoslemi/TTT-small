import numpy as np
from scipy.optimize import minimize
import time
from scipy.interpolate import interp1d
from cvxpy import *
import cvxpy as cp
from scipy.linalg import eigh

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # ALL PARAMETERS
    n_coarse = 120  # Coarse resolution for SDP
    n_fine = 300    # Target resolution
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # --- Stage 1: Coarse SDP relaxation ---
    print("Stage 1: Coarse SDP relaxation...")
    if time.time() - start_time > budget_s - 10:
        fallback_to_uniform = True
    else:
        fallback_to_uniform = False
    
    try:
        # Create Toeplitz matrices R_k
        R = []
        for k in range(n_coarse):
            R_k = np.zeros((n_coarse, n_coarse))
            for i in range(n_coarse):
                if i+k < n_coarse:
                    R_k[i, i+k] = 1
                if i-k >= 0:
                    R_k[i, i-k] = 1
            R.append(R_k)
        
        # Variables
        t = cp.Variable()
        h = cp.Variable(n_coarse)
        X = cp.Variable((n_coarse, n_coarse), symmetric=True)
        
        # Build matrix M = [[1, h.T], [h, X]]
        M = cp.bmat([[1, h.T], [h, X]])
        
        # Constraints
        constraints = [
            cp.trace(R_k @ X) >= t for R_k in R
        ]
        constraints += [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_coarse/2,
            M >> 0  # PSD
        ]
        
        # Objective
        objective = cp.Maximize(t)
        
        prob = cp.Problem(objective, constraints)
        
        # Solve (cut short if needed)
        solve_start = time.time()
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, verbose=False)
        
        if time.time() - start_time > budget_s - 10:
            fallback_to_uniform = True
        
        if fallback_to_uniform:
            print("SDP fallback active")
            h_val_coarse = None
        else:
            print("SDP succeeded")
            h_val_coarse = h.value
            X_val = X.value
            t_val = t.value
        
    except Exception as e:
        print(f"SDP error: {e}")
        h_val_coarse = None
        fallback_to_uniform = True
        
    # Fallback to uniform if needed
    if fallback_to_uniform or h_val_coarse is None:
        h_val_coarse = np.full(n_coarse, 0.5)
        h_val_coarse = h_val_coarse * (n_coarse / 2) / np.sum(h_val_coarse)
        print("Using random start")
    
    # --- Stage 2: Extract and project (if SDP work)
    if not fallback_to_uniform:
        try:
            # Find leading eigenpair
            eigenvals, eigenvecs = eigh(X_val, eigvals=(n_coarse-1, n_coarse-1))
            lambda1 = eigenvals[0]
            v1 = eigenvecs[:, 0]
            
            # Realignment
            h_tilde = np.sqrt(lambda1) * v1
            
            # Box projection
            h_after_box = np.clip(h_tilde, 0, 1)
            
            # Sum projection (waterfilling style)
            def project_sum(h_init, target_sum):
                h = h_init.copy()
                excess = np.sum(h) - target_sum
                if abs(excess) < 1e-12:
                    return h
                while abs(excess) > 1e-12:
                    # indices with 0 < h < 1
                    indices = np.where((h > 0) & (h < 1))[0]
                    if len(indices) == 0:
                        # Shrink or expand all values
                        scale_factor = target_sum / np.sum(h)
                        h = h * scale_factor
                        break
                    delta = excess / len(indices)
                    # Distribute among active indices
                    h[indices] = np.clip(h[indices] - delta, 0, 1)
                    excess = np.sum(h) - target_sum
                return h
            
            h_proj = project_sum(h_after_box, n_coarse/2)
            
            # Mirror symmetry
            h_sym = (h_proj + h_proj[::-1]) / 2
            
            h_val_coarse = h_sym
            
        except Exception as e:
            print(f"Projection error: {e}")
            h_val_coarse = np.full(n_coarse, 0.5)
            h_val_coarse = h_val_coarse * (n_coarse / 2) / np.sum(h_val_coarse)
    
    # --- Stage 3: Coarse grid refinement ---
    print("Stage 3: Coarse grid refinement...")
    if time.time() - start_time > budget_s - 30:
        refine_coarse = False
    else:
        refine_coarse = True
    
    if refine_coarse:
        def smoothmax_objective(h, beta):
            conv_full = np.correlate(h, 1-h, mode='full')
            conv_values = conv_full[len(h)-1:]  # From shift 0 to n-1
            exp_vals = np.exp(beta * dx_coarse * conv_values)
            log_sum_exp = np.log(np.sum(exp_vals))
            objective = (1/beta) * log_sum_exp
            return objective

        def grad_smoothmax(h, beta):
            conv_full = np.correlate(h, 1-h, mode='full')
            conv_values = conv_full[len(h)-1:]
            exp_vals = np.exp(beta * dx_coarse * conv_values)
            w = exp_vals / np.sum(exp_vals)
            # Note: manual differentiation
            # For shift k, ∂c_k/∂h_i = (1-h_{i+k}) - h_{i-k}  or something similar
            # Gradient of soft-max surrogate
            grad = np.zeros_like(h)
            for i in range(len(h)):
                # ∂c_k/∂h_i = w_k * (1-h_{i+k} - h_{i-k}) 
                term1 = 0.0
                term2 = 0.0
                for k in range(len(h)):
                    if i+k < len(h):
                        term1 += w[k] * (1 - h[i+k])  # Safe but Tao-style for h[i+k]
                    if i-k >= 0:
                        term2 += w[k] * h[i-k]
                grad[i] = dx_coarse * (term1 - term2)
            return grad  # This is shaped as grad[i]

        # Optimizer loop
        h_coarse_refined = h_val_coarse.copy()
        beta_vals = [0.5, 1.0, 2.0, 5.0, 10., 20., 50.]
        c_prev = smoothmax_objective(h_coarse_refined, 50.0)
        for beta in beta_vals:
            if time.time() - start_time > budget_s - 30:
                break
            print(f"Refining with β={beta}")

            def f(h):
                return smoothmax_objective(h, beta)
            def grad(h):
                return grad_smoothmax(h, beta)
            
            # Projected L-BFGS-B
            bounds = [(0, 1)] * n_coarse
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_coarse / 2,
                    'jac': lambda h: np.ones_like(h)}

            try:
                for _ in range(80):  # Max iterations per β
                    if time.time() - start_time > budget_s - 30:
                        break
                    # Project gradient step
                    new_h = h_coarse_refined - 0.01 * grad(h_coarse_refined)
                    new_h = np.clip(new_h, 0, 1)
                    target_sum = n_coarse / 2
                    total = np.sum(new_h)
                    delta = (target_sum - total) / len(new_h)
                    # We don't do truncation because it might undo steps
                    new_h = new_h + delta
                    new_h = np.clip(new_h, 0, 1)
                    new_h = project_sum(new_h, target_sum)
                    h_coarse_refined = (new_h + new_h[::-1]) / 2  # Mirror
                    c = smoothmax_objective(h_coarse_refined, beta)
                    # Check convergence
                    if abs(c - c_prev) < 1e-6 and beta > 5.0:
                        break
                    c_prev = c
            except:
                pass  # Silent fallback if not working
    
    # --- Stage 4: Upsampling ---
    print("Stage 4: Upsampling...")
    if time.time() - start_time > budget_s - 20:
        upsample = False
    else:
        upsample = True
    
    if upsample:
        h_upsampled = np.interp(np.linspace(0, 2, n_fine), 
                                np.linspace(0, 2, n_coarse), 
                                h_coarse_refined)
        # Normalize
        h_upsampled = h_upsampled * (n_fine / 2) / np.sum(h_upsampled)
        # Mirror symmetry
        h_fine = (h_upsampled + h_upsampled[::-1]) / 2
        # Clip
        h_fine = np.clip(h_fine, 0, 1)
        # Project sum
        h_fine = project_sum(h_fine, n_fine / 2)
    else:
        h_fine = np.full(n_fine, 0.5)
        h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)

    # --- Stage 5: Fine grid refinement ---
    fine_grad = None
    print("Stage 5: Fine grid refinement...")
    def smoothmax_objective_fine(h, beta):
        conv_full = np.correlate(h, 1-h, mode='full')
        conv_values = conv_full[len(h)-1:]  # From shift 0 to n-1
        exp_vals = np.exp(beta * dx_fine * conv_values)
        log_sum_exp = np.log(np.sum(exp_vals))
        objective = (1/beta) * log_sum_exp
        return objective

    def grad_smoothmax_fine(h, beta):
        conv_full = np.correlate(h, 1-h, mode='full')
        conv_values = conv_full[len(h)-1:]
        exp_vals = np.exp(beta * dx_fine * conv_values)
        w = exp_vals / np.sum(exp_vals)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            term1 = 0.0
            term2 = 0.0
            for k in range(len(h)):
                if i+k < len(h):
                    term1 += w[k] * (1 - h[i+k])
                if i-k >= 0:
                    term2 += w[k] * h[i-k]
            grad[i] = dx_fine * (term1 - term2)
        return grad

    beta_vals = [0.5, 1.0, 2.0, 5.0, 10., 20., 50.]
    c_prev = smoothmax_objective_fine(h_fine, 50.0)
    for beta in beta_vals:
        if time.time() - start_time > budget_s - 15:
            break
        print(f"Fine refining with β={beta}")
        def f(h):
            return smoothmax_objective_fine(h, beta)
        def grad(h):
            return grad_smoothmax_fine(h, beta)
        try:
            for _ in range(50):
                if time.time() - start_time > budget_s - 15:
                    break
                new_h = h_fine - 0.005 * grad(h_fine)
                new_h = np.clip(new_h, 0, 1)
                target_sum = n_fine / 2
                total = np.sum(new_h)
                new_h = new_h + (target_sum - total) / len(new_h)
                new_h = np.clip(new_h, 0, 1)
                new_h = project_sum(new_h, target_sum)
                h_fine = (new_h + new_h[::-1]) / 2
                c = smoothmax_objective_fine(h_fine, beta)
                if abs(c - c_prev) < 1e-7 and beta > 3.0:
                    break
                c_prev = c
        except:
            pass  # No error here too

    # --- Stage 6: Stochastic polishing ---
    print("Stage 6: Stochastic polishing...")
    best_h = h_fine.copy()
    c5_best = np.max(np.correlate(h_fine, 1-h_fine, mode='full')) * dx_fine
    
    max_perturb = min(6, n_fine // 20)
    epsilon = 0.004
    iters = 0
    while True:
        if time.time() - start_time > budget_s - 10:
            break
        iters += 1
        indices = np.random.choice(n_fine, size=max_perturb, replace=False)
        trial_h = best_h.copy()
        for i in indices:
            trial_h[i] += np.random.uniform(-epsilon, epsilon)
        trial_h = np.clip(trial_h, 0, 1)
        trial_h = project_sum(trial_h, n_fine/2)
        trial_h = (trial_h + trial_h[::-1]) / 2
        
        c5_trial = np.max(np.correlate(trial_h, 1-trial_h, mode='full')) * dx_fine
        if c5_trial < c5_best:
            best_h = trial_h.copy()
            c5_best = c5_trial

    # --- Final report ---
    final_c5 = np.max(np.correlate(best_h, 1-best_h, mode='full')) * dx_fine
    return (best_h, final_c5, n_fine)

def project_sum(h_init, target_sum):
    h = h_init.copy()
    excess = np.sum(h) - target_sum
    if abs(excess) < 1e-12:
        return h
    while abs(excess) > 1e-12:
        indices = np.where((h > 0) & (h < 1))[0]
        if len(indices) == 0:
            scale_factor = target_sum / np.sum(h)
            h = h * scale_factor
            break
        delta = excess / len(indices)
        h[indices] = np.clip(h[indices] - delta, 0, 1)
        excess = np.sum(h) - target_sum
    return h
