def build_shift_matrices(N):
    """Build sparse CSR shift matrices A_k for k = 0,...,N-1"""
    A_list = []
    for k in range(N):
        # Create sparse matrix for shift k
        row = np.arange(N - k)
        col = np.arange(k, N)
        data = np.ones(N - k)
        A_k = sp.csr_matrix((data, (row, col)), shape=(N, N))
        A_list.append(A_k)
    return A_list

def project_to_feasible(h, dx):
    """Ensure sum(h) = n/2 and h in [0,1]^n"""
    n = len(h)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    # Simple bounded simplex projection
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    # Distribute deficit
    if deficit > 0:
        # Need to add more, distribute among non-1 entries
        adjustment = deficit / np.sum(h < 1)
        h = np.minimum(1.0, h + adjustment)
    else:
        # Need to subtract, distribute among non-0 entries
        adjustment = -deficit / np.sum(h > 0)
        h = np.maximum(0.0, h - adjustment)
        
    # Final adjustment to exactly match sum
    h = h * (n/2) / np.sum(h)
    return h

def eigen_round(X_sdp, h_sdp):
    """Principal eigenvector rounding"""
    # SVD of the matrix, get top eigenvector
    try:
        _, v = sp.linalg.eigs(X_sdp, k=1, which='LM', maxiter=500)
    except:
        # fallback to dense SVD
        U, s, Vt = np.linalg.svd(X_sdp.toarray())
        v = U[:, 0]
    
    # Form the vector (could be complex)
    v = np.real(v.flatten())  # Get real part
    # Start with sqrt(s) * v
    h_eig = np.abs(v) * np.sqrt(np.max(s))
    
    # Clip and normalize
    h_eig = np.clip(h_eig, 0, 1)
    h_eig = h_eig * (len(h_eig)/2) / np.sum(h_eig)
    return h_eig

def threshold_round(X_sdp):
    """Diagonal threshold rounding"""
    diag_vals = X_sdp.diagonal()
    # Sort indices by diagonal value descending
    sorted_indices = np.argsort(diag_vals)[::-1]
    # Set top N/2 as 1, rest 0
    h_thr = np.zeros(len(diag_vals))
    h_thr[sorted_indices[:len(sorted_indices)//2]] = 1.0
    return h_thr

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation
    corr = np.correlate(h, g, mode='full') * dx
    # Return maximum value
    return np.max(corr)

    best_h = h_current.copy()
    best_c5 = c5_current
    
    iter_count = 0
    while time.time() - start_time < budget_s - 2:
        # Pick two indices: 1 and 0
        ones = np.where(h_current == 1)[0]
        zeros = np.where(h_current == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        idx1 = np.random.choice(ones)
        idx2 = np.random.choice(zeros)
        
        # Try flipping
        h_new = h_current.copy()
        h_new[idx1] = 0
        h_new[idx2] = 1
        
        c5_new = c5_of_h(h_new)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
                
        iter_count += 1
        
        if iter_count % 50 == 0:
            # Periodically recompute
            c5_current = c5_of_h(h_current)
    
    return best_h, best_c5

def upsample_and_refine(h_coarse, n_points_high):
    """Upsample to higher resolution and refine"""
    start_time = time.time()
    h_up = np.zeros(n_points_high)
    dx_low = 2.0 / len(h_coarse)
    dx_high = 2.0 / n_points_high
    for i in range(n_points_high):
        x = i * dx_high
        i_low = int(x / dx_low)
        alpha = (x - i_low * dx_low) / dx_low
        j_low = min(i_low, len(h_coarse)-1)
        j_high = min(i_low + 1, len(h_coarse)-1)
        h_up[i] = (1 - alpha) * h_coarse[j_low] + alpha * h_coarse[j_high]
 
    # Renormalize
    h_up = h_up * (n_points_high / 2) / np.sum(h_up)
    
    # Refine using projected gradient descent with surrogate
    def logsumexp_surrogate(x, k_0):
        return np.log(np.sum(np.exp((x - x[k_0]) / 1e-4))) * 1e-4 + x[k_0]
    
    def proj_gradient_descent(h_start, max_iter=30):
        h = h_start.copy()
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            # Compute full marginal
            g = 1.0 - h
            corr = np.correlate(h, g, mode='full')
            k_max = np.argmax(corr)
            # Project gradient step
            try:
                # Simplified gradient direction, actual likely to be more nuanced
                grad = np.zeros_like(h)
                for k in range(len(h)):
                    if k == k_max:
                        k_shift = k
                        # Modify based on shift k
                        # This is approximated
                        for i in range(len(h)):
                            if i + k_shift < len(h):
                                grad[i] += (1 - h[i+k_shift])
                            if i - k_shift >= 0:
                                grad[i] += h[i-k_shift]
                # Projected gradient
                lr = 1e-2
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points_high / 2) / np.sum(h_new)
                h = h_new
            except:
                break
        return h
    
    # Run key refinement step
    h_opt = proj_gradient_descent(h_up, max_iter=30)
    return h_opt

        for i in range(N):

        constr.append(cp.sum(h) == N / 2)

        # Bounds on h
        constr.append(h >= 0)
        constr.append(h <= 1)

        # Constraint on t: tr(A_k * X) >= t
        for k in range(N):
            constr.append(cp.trace(A_list[k] @ X) >= t)

        # PSD constraint
        constr.append(X >> 0)
