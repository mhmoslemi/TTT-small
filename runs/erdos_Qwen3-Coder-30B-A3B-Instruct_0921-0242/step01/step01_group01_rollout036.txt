import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def c5_of_h(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution
    conv = np.correlate(h, g, mode='full')
    # The full convolution has 2*len(h) - 1 elements
    result = np.max(conv) * dx
    return result

def water_fill_projection(h, target_sum):
    """Project h to satisfy sum = target_sum using water-filling"""
    h = np.copy(h)
    total = np.sum(h)
    diff = total - target_sum
    
    if np.abs(diff) < 1e-12:
        return h
        
    # Identify indices that can be adjusted (0 < h_i < 1)
    candidates = np.where((h > 0) & (h < 1))[0]
    
    if len(candidates) == 0:
        # All values are at boundaries, just clip to target
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total > target_sum:
            # Reduce from top
            excess = total - target_sum
            while excess > 1e-12:
                max_indices = np.where(h == np.max(h))[0]
                for idx in max_indices:
                    if excess < 1e-12:
                        break
                    amount = min(excess, h[idx])
                    h[idx] -= amount
                    excess -= amount
        elif total < target_sum:
            # Increase from bottom
            deficit = target_sum - total
            while deficit > 1e-12:
                min_indices = np.where(h == np.min(h))[0]
                for idx in min_indices:
                    if deficit < 1e-12:
                        break
                    amount = min(deficit, 1 - h[idx])
                    h[idx] += amount
                    deficit -= amount
        return h
    
    # Apply water-filling
    step = diff / len(candidates)
    h[candidates] -= step
    
    # Ensure boundaries
    h = np.clip(h, 0, 1)
    
    # Re-project if necessary
    new_total = np.sum(h)
    new_diff = new_total - target_sum
    if np.abs(new_diff) > 1e-12:
        # Apply adjustment to already bounded elements
        adjustees = np.where((h > 0) & (h < 1))[0]
        if len(adjustees) > 0:
            new_step = new_diff / len(adjustees)
            h[adjustees] -= new_step
            h = np.clip(h, 0, 1)
    
    return h

def mirror_project(h):
    """Ensure h is symmetric w.r.t. center"""
    n = len(h)
    h_sym = (h + h[::-1]) / 2
    return h_sym

def smooth_max_gradient(h, beta, dx):
    """compute gradient of smooth max of correlations"""
    # h is length n
    n = len(h)
    g = 1.0 - h
    
    # Compute all correlations
    conv = np.correlate(h, g, mode='full')  # length 2*n-1
    # categories: k in [-n+1, n-1]
    # conv[i] corresponds to shift (i - (n-1)) = (i-n+1)
    
    # weights
    conv_beta = beta * dx * conv
    # avoid numerical overflow
    conv_beta = conv_beta - np.max(conv_beta)
    exp_conv = np.exp(conv_beta)
    weights = exp_conv / np.sum(exp_conv)
    # Compute w * (1-h) and w_rev * h  
    # we can use convolution for this!
    w_rev = weights[::-1]
    u = np.correlate(w_rev, g, mode='full')[:n]  # tanh precision issues may apply
    v = np.correlate(weights, h, mode='full')[n-1:]
    # these are now vectors of length n (note they're simpler)
    grad = dx * (u - v)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # parameters
    n_coarse = 120
    n_target = 300
    dx_coarse = 2.0 / n_coarse
    dx_target = 2.0 / n_target
    
    best_c5 = float('inf')
    best_h = None
    best_n = n_target
    
    # Step 1: Try to solve SDP relaxation
    h_best = None
    
    try:
        # Create a simple start vector if SDP fails
        h_start = np.zeros(n_coarse)
        h_start[:n_coarse//2] = 1.0
        h_start = h_start * (n_coarse / 2) / np.sum(h_start)

        # If SDP is too slow or problematic, abort and start from scratch
        if time.time() - start_time > budget_s - 100:
            h_start = np.ones(n_coarse) * 0.5
        
        # Precompute Toeplitz structure for autocorrelation constraints
        P = []
        # build for k = 0 to n_coarse - 1
        for k in range(n_coarse):
            R = np.zeros((n_coarse, n_coarse))
            for i in range(n_coarse):
                if i + k < n_coarse:
                    R[i, i+k] = 1
                if i - k >= 0:
                    R[i, i-k] = 1
            P.append(R)
        
        # SDP relaxation:
        # max t
        # s.t. trace(R_k * X) >= t  for k=0...n_coarse-1
        #      0 <= h_i <= 1
        #      sum(h) = n_coarse/2
        #      M = [[1., h.T],
        #           [h,  X]]  is PSD
        
        t = Variable()
        h = Variable(n_coarse)
        X = Variable((n_coarse, n_coarse), symmetric=True)
        M = [[1, h.T], [h, X]]
        
        problem = Problem(Maximize(t),
                          [trace(P[k] @ X) >= t for k in range(n_coarse)] + 
                          [0 <= h[i] for i in range(n_coarse)] + 
                          [h[i] <= 1 for i in range(n_coarse)] + 
                          [sum(h) == n_coarse / 2] + 
                          [M >> 0])  # is PSD

        # solve with limited time
        time_left = budget_s - (time.time() - start_time)
        if time_left < 100:
            raise Exception("No time for SDP")
        
        try:
            problem.solve(solver=SCS, max_iters=5000, eps=1e-5, verbose=False)
        except:
            raise Exception("SDP solve failed")
            
        # Check if we got iterations
        if problem.status in ['optimal', 'optimal_inaccurate']:
            h_val = h.value
            X_val = X.value
            t_val = t.value
        else:
            # fallback to seed
            h_val = h_start
            h_val = h_val * (n_coarse / 2) / np.sum(h_val)
            h_val = np.clip(h_val, 0, 1)
        
        # convert to vector
        h_coarse = np.array(h_val).flatten()
        
        # enforce constraints
        # Project to satisfy 0 <= h <= 1, sum=1/2
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = water_fill_projection(h_coarse, n_coarse / 2)
        h_coarse = mirror_project(h_coarse)
        
        # compute C5 coarse
        c5_coarse = c5_of_h(h_coarse)
        if c5_coarse < best_c5:
            best_c5 = c5_coarse
            best_h = h_coarse
            best_n = n_coarse

    except:
        print("SDP failed, fallback to local methods")
        # Use a good warm-start if available
        if 'initial_h_values' in globals():
            try:
                x_orig = np.linspace(0, 2, len(initial_h_values))
                x_new = np.linspace(0, 2, n_coarse)
                h_interp = interp1d(x_orig, initial_h_values, kind='linear', fill_value='extrapolate')
                h_start = h_interp(x_new)
                # Normalize to have integral = 1 (sum = n_coarse / 2)
                h_start = h_start * (n_coarse / 2) / np.sum(h_start)
            except:
                h_start = np.ones(n_coarse) * 0.5
        else:
            h_start = np.ones(n_coarse) * 0.5
            
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_coarse / 2) / np.sum(h_start)
        best_h = h_start
        
        # run local refinement with smooth max surrogate with varied beta
        h_current = h_start.copy()
        for i in range(100):
            if time.time() - start_time > budget_s - 20:
                break
            
            # fine tune with smooth max for different beta values
            for beta in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]:
                if time.time() - start_time > budget_s - 20:
                    break
                    
                grad = smooth_max_gradient(h_current, beta, dx_coarse)
                # L-BFGS-B step in numpy
                # projected gradient descent for simplicity of code
                h_trial = h_current - 0.001 * grad
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = water_fill_projection(h_trial, n_coarse / 2)
                h_trial = mirror_project(h_trial)
                
                c5_trial = c5_of_h(h_trial)
                if c5_trial < c5_coarse:
                    h_current = h_trial
                    c5_coarse = c5_trial
                    
        if c5_coarse < best_c5:
            best_c5 = c5_coarse
            best_h = h_current.copy()
            best_n = n_coarse

    # Step 2: Upsample to n_target
    if time.time() - start_time < budget_s - 100:
        h_upsampled = np.interp(np.linspace(0, 2, n_target),
                                np.linspace(0, 2, n_coarse),
                                best_h)
        # Clip and rescale
        h_upsampled = np.clip(h_upsampled, 0, 1)
        h_upsampled = h_upsampled * (n_target / 2) / np.sum(h_upsampled)
        h_upsampled = mirror_project(h_upsampled)
        
        # Smooth refinement on hoch grid
        h_refined = h_upsampled.copy()
        for _ in range(500):
            if time.time() - start_time > budget_s - 5:
                break
            beta = 0.5
            grad = smooth_max_gradient(h_refined, beta, dx_target)
            h_trial = h_refined - 0.0005 * grad
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = water_fill_projection(h_trial, n_target / 2)
            h_trial = mirror_project(h_trial)
            
            c5_trial = c5_of_h(h_trial)
            if c5_trial < best_c5:
                h_refined = h_trial
                best_c5 = c5_trial
        
        best_h = h_refined
        best_n = n_target

    # Step 3: Final stochastic polishing
    if time.time() - start_time < budget_s - 20:
        iterations = 0
        h_final = best_h.copy()
        c5_final = best_c5
        try:
            # Noise level
            epsilon = 0.004
            while time.time() - start_time < budget_s - 1:
                # Select indices to perturb
                p = min(6, n_target // 20)
                indices = np.random.choice(n_target, size=p, replace=False)
                h_trial = h_final.copy()
                for idx in indices:
                    delta = np.random.uniform(-epsilon, epsilon)
                    h_trial[idx] += delta
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = water_fill_projection(h_trial, n_target / 2)
                h_trial = mirror_project(h_trial)
                
                # Evaluate
                c5_trial = c5_of_h(h_trial)
                if c5_trial < c5_final:
                    h_final = h_trial
                    c5_final = c5_trial
                iterations += 1
        except:
            pass
        best_h = h_final
        best_c5 = c5_final
        best_n = n_target

    return (best_h, best_c5, best_n)
