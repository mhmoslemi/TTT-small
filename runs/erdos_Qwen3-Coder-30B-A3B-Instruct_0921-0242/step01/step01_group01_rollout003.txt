import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time
from scipy.interpolate import interp1d
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Target resolution
    n_points = 300
    
    # Phase 1: Binary MILP at coarse resolution
    n0 = 80
    dx0 = 2.0 / n0
    d = n0 // 2 + 1
    
    # Create variables
    x = Variable(d, boolean=True)
    M = Variable()
    
    # Constraints
    constraints = []
    
    # Mirror symmetry + fixed first element x[0]=1
    # x[i] = x[n0-1-i], x[0] = 1
    # We only vary the first d elements (0 to d-1)
    # But x[0]=1 is fixed, so our actual decision variables are just x[1:d]
    # For efficiency, we compute y[i,k] vars and their products directly
    
    # Reconstruct full x from first half including mirror symmetry
    def full_x(x_vals):
        """Reconstruct full binary vector from half with symmetry."""
        x_full = np.zeros(n0, dtype=bool)
        x_full[0] = True  # Fixed
        x_full[1:d] = x_vals[1:]  # First half
        x_full[d:] = x_vals[:0:-1]  # Mirror of second half excluding center
        return x_full
    
    # Reproduce the full binary solution 
    # Since it's tricky to encode the full logic in one go,
    # brute-force solutions (say, up to n0=100) are handy
    
    # Analytical construction based on the constraint:
    # We want set of indices with weight n0/2 = 40 that minimize max_k sum_i x_i * x_{i+k}
    
    # Simple construction: alternate 1s and 0s
    h_base = np.zeros(n0, dtype=bool)
    h_base[::2] = 1
    h_base = np.clip(h_base, 0, 1) # Cast to bool, then int
    
    # Anti-alternator 
    h_antialt = np.zeros(n0, dtype=bool)
    h_antialt[1::2] = 1
    h_antialt = np.clip(h_antialt, 0, 1)
    
    # A 'rough' but natural variant with 1s concentrated in centers
    h_centered = np.zeros(n0)
    mid = n0 // 2
    start = max(0, mid - 10)
    end = min(n0, mid + 10)
    h_centered[start:end] = 1
    h_centered = h_centered * (40 / np.sum(h_centered))
    
    # Another basic try: centered around middle indices
    h_centered2 = np.zeros(n0)
    # We place 40 ones starting from index n0/4
    h_centered2[n0//4:n0//4+40] = 1
    h_centered2 = h_centered2 * (40 / np.sum(h_centered2))
    
    # Let just go with one known good construction:
    # A more structured approach: divide into blocks
    h_structured = np.zeros(n0)
    blocksize = 5
    for i in range(0, n0, 2 * blocksize):
        h_structured[i:i+blocksize] = 1
    h_structured = np.clip(h_structured, 0, 1)
    h_structured = h_structured * (40 / np.sum(h_structured))  # Correct weight
    
    # Random construction with selected ones
    h_random = np.zeros(n0)
    indices = np.random.choice(n0, size=40, replace=False)
    h_random[indices] = 1
    h_random = h_random * (40 / np.sum(h_random))
    
    def compute_autocorrelations(h):
        """Compute auto-correlations at all shifts."""
        corr = np.correlate(h, h, mode='full')
        maxshift = len(h) - 1
        # Extract valid correlation values
        autocorr = []
        for k in range(-maxshift, maxshift + 1):
            idx = k + maxshift
            if 0 <= idx < len(corr):
                autocorr.append(corr[idx])
        return np.array(autocorr)
    
    # Compute overlaps for base candidates
    h_collection = [h_base, h_antialt, h_centered, h_centered2, h_random]
    valid_h = []
    for h in h_collection:
        # Normalize to exact integral
        h = h * (40 / np.sum(h))
        temp = np.zeros_like(h)
        temp[:] = h  
        valid_h.append(temp)
    
    # Pick best one based on min of overlap integrals ( = 1 - max of autocorrel)
    best_h_coarse = None
    best_overlap = float('inf')
    for h in valid_h:
        auto_corr = compute_autocorrelations(h)
        # Overlap integral = 1 - autocorrelation
        overlap = 1 - auto_corr
        max_overlap = np.max(overlap)
        if max_overlap < best_overlap:
            best_overlap = max_overlap
            best_h_coarse = h.copy()
    
    # Phase 2: Upscaling and integral constraint enforcement
    h_upscaled = np.repeat(best_h_coarse, 300 // n0, axis=0)  # 300/80 = 3.75
    h_upscaled = np.clip(h_upscaled, 0, 1)
    h_upscaled = h_upscaled * (n_points / 2) / np.sum(h_upscaled)
    # With exact upscaling (fallback if 3.75 not integral)
    if len(h_upscaled) > n_points:
        h_upscaled = h_upscaled[:n_points]
    elif len(h_upscaled) < n_points:
        # Pad with zeros at the end
        h_upscaled = np.pad(h_upscaled, (0, n_points - len(h_upscaled)), mode='constant', constant_values=0)
    
    # Robustly force integral constraint by converting to binary then restoring pos values
    h_binary = np.zeros_like(h_upscaled, dtype=int)
    h_binary[np.argsort(h_upscaled)[::-1][:n_points//2]] = 1
    
    # Convert to valid continuous h
    h_continuous = np.zeros_like(h_upscaled, dtype=float)
    h_vals = np.sort(h_upscaled)[::-1]
    top_indices = np.argsort(h_upscaled)[::-1][:n_points//2]
    value_weights = np.ones_like(top_indices) / (n_points//2)  # uniform width over top indices
    values = h_vals[:n_points//2] # just take sufficient values
    h_continuous[top_indices] = values
    h_continuous = h_continuous * (n_points / 2) / np.sum(h_continuous)
    
    # Apply symmetry now
    h_sym = (h_continuous + h_continuous[::-1]) / 2
    
    # Phase 3: Gradient-based refinement with soft max
    def soft_max_c5(h, beta):
        """Compute surrogate objective."""
        dx = 2.0 / len(h)
        h = np.asarray(h)
        g = 1.0 - h
        conj = np.correlate(h, g, mode='full')
        # Ignore the 0 index (shift of 0)
        shift_indices = np.arange(-(len(h)-1), len(h))
        conv = conj
        if beta is None:
            return np.max(conv) * dx
        # Compute soft max
        exp_terms = np.exp(beta * dx * conv)
        return np.log(np.sum(exp_terms)) / beta * dx
    
    def grad_soft_max_c5(h, beta):
        """Compute gradient of soft max objective."""
        dx = 2.0 / len(h)
        h = np.asarray(h)
        g = 1.0 - h
        # Get all the individual converolutions for shift k
        # Compute all shifts via 2×FFT
        conv = np.correlate(h, g, mode='full')
        # Compute autcorrelations:
        autocorr = np.correlate(h, h, mode='full')
        # Auto correlation does not include 1-h, just use it as shift for g
        shift_indices = np.arange(-(len(h)-1), len(h))
        
        if beta is None:
            idx_max = np.argmax(conv)
            k_max = idx_max - (len(h) - 1)
            # Gradient at that specific shift contribution
            grad = np.zeros_like(h)
            # For shift k_max:
            # Each entry contributes to: h_i * (1-h_{i+k_max})
            for i in range(len(h)):
                j = i + k_max  # note k_max may be negative
                if 0 <= j < len(h):
                    grad[i] += (1.0 - h[j])
                j = i - k_max  # we reverse notion below!
                if 0 <= j < len(h):
                    grad[i] -= h[j]
            return grad * dx
        else:
            exp_terms = np.exp(beta * dx * conv)
            weights = exp_terms / np.sum(exp_terms)
            # Keep gradients of each correlation
            # For one k shift: d conv_k / dh_i = (d / dh_i) (h_i (1 - h_{i+k}) )
            # This gives = (1 - h_{i+k}) if i is at index
            # Collect the shift vector affecting index i with weights
            
            # Very clean is to compute full Jacobian by fft-based computation
            # This part is tricky using numpy
        
            # Instead, let's compute the soft-gradient quite directly as
            # sub-instantiation to each correlation derivative
            grad = np.zeros_like(h)
            for k in range(len(conv)):
                # Pattern match shift k
                shift_val = shift_indices[k]
                shift_vector = np.zeros(len(h)) 
                if shift_val >= 0:
                    shift_vector[shift_val:] = h[:-shift_val]
                else:
                    shift_vector[:shift_val] = h[-shift_val:]
                # d_conv_k/dh_i = (1 - h_{i+k}) * (0 if not in range else 1) 
                # and - h_{i-k}  for i-k position.
                temp_grad = np.zeros_like(h)
                for i in range(len(h)):
                    # Backward diff contribution
                    j_fwd = i + shift_val  # Temporarily ignore bounds
                    if 0 <= j_fwd < len(h):
                        temp_grad[i] += (1.0 - h[j_fwd])
                    else:
                        pass  # ignore boundary contribution for now
                    # forward diff contribution
                    j_rev = i - shift_val  # or just i + (-shift_val))
                    if 0 <= j_rev < len(h) and j_rev != i:
                        temp_grad[i] -= h[j_rev]
                        
                        # Verify it will not double-count the self value
                        # For auto_correlation & crossCorrelation this needs care
                        
                # Accumulate error-weighted
                grad += weights[k] * temp_grad
                
            return grad * dx

    # Phase 3: Optimization with continuation schedule
    def project(h, integral_target):
        """Projection to unit integral constraint."""
        current_sum = np.sum(h)
        diff = current_sum - integral_target
        if abs(diff) < 1e-12:
            return h.copy()
        # Find all interior values (0<h<1)
        indices = np.where((h > 0) & (h < 1))[0]
        if len(indices) == 0:
            return h.copy()
        # Subtract evenly distribute excess
        delta_per = diff / len(indices)
        h_projected = h.copy()
        h_projected[indices] -= delta_per
        # Force into feasible range again
        h_projected = np.clip(h_projected, 0, 1)
        return h_projected
    
    # Warm up
    from scipy.optimize import minimize_scalar
    
    # Just minimize the soft version using soft min as proxy
    
    h_opt = h_sym.copy()
    target = n_points / 2
    
    # Soft max parameters
    beta_start = 0.5
    beta_max = 50.0
    beta_step = 2.0
    
    # Refinement loop
    beta_current = beta_start
    last_val = float('inf')
    epochs = 0
    
    violated_after_proj = False
    ties_count = 0
    
    convergence_threshold = 1e-6
    
    while time.time() - start_time < budget_s - 1:
        if violated_after_proj or beta_current >= beta_max:
            break
            
        def obj(h):
            return soft_max_c5(h, beta_current)
            
        def jac(h):
            return grad_soft_max_c5(h, beta_current)
            
        # Optimize one beta stage:
        # To use scipy optimize, make sure constraints are respected
        
        # Gradient descent step: use L-BFGS-B
        try:
            result = minimize(obj, h_opt, method='L-BFGS-B', jac=jac,
                              bounds=[(0.0, 1.0)] * len(h_opt),
                              options={'maxiter': 100, 'gtol': 1e-6})
            
            h_new = result.x
            
            # Ensure sum == target
            h_new = h_new * (target / np.sum(h_new))
            
            # Enforce symmetry
            h_new = (h_new + h_new[::-1]) / 2
            
            # After projection, how did the true value go?
            c5_actual = soft_max_c5(h_new, None)  # use hard version

            h_opt = h_new

            if c5_actual >= last_val - 1e-6 and epochs > 2:
                # Stay qualified yada
                ties_count += 1
                if ties_count > 2:
                    break
            else:
                ties_count = 0
            last_val = c5_actual
            epochs += 1
            
            beta_current *= beta_step
            
        except Exception as e:
            break
            
    # Final polishing: The high precision hill climb is better for the very last touch-up of final 
    # least constant.
    best_h_final = h_opt.copy()
    best_c5_final = soft_max_c5(h_opt, None)
    h_current = best_h_final.copy()
    c5_current = best_c5_final
    
    # Run at most 10k iterations
    # We can reuse the true integral check and do two batch iterations per outer loop
    MAX_TRIALS = 10000
    trial_count = 0
    
    while trial_count < MAX_TRIALS and time.time() - start_time < budget_s - 0.1:
        trial_count += 1
        # Pick small batch of indices to perturb
        npoints_perturb = min(6, n_points // 20)
        indices_to_change = np.random.choice(n_points, size=npoints_perturb, replace=False)
        h_trial = h_current.copy()
        for idx in indices_to_change:
            delta = np.random.uniform(-0.004, 0.004)
            h_trial[idx] += delta
            h_trial[idx] = min(1, max(0, h_trial[idx]))
        # Project to valid sum
        h_trial_sum = np.sum(h_trial)
        h_trial = h_trial * (target / h_trial_sum)
        # Re-enforce symmetry:
        h_trial = (h_trial + h_trial[::-1]) / 2
        c5_trial = soft_max_c5(h_trial, None)
        if c5_trial < c5_current:
            h_current = h_trial.copy()
            c5_current = c5_trial
            
    # The final answer    
    return (h_current, c5_current, n_points)
