import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy import sparse as sp

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def build_Q(k, n):
    """Build sparse symmetric PSD matrix for term h_i h_{i+k}."""
    data = np.ones(n - abs(k))
    rows = np.arange(n - abs(k))
    cols = rows + k if k >= 0 else rows - k
    Q = sp.csr_matrix((data, (rows, cols)), shape=(n, n))
    return 0.5 * (Q + Q.T)

def solve_qp(n, Kmax, warm_h=None):
    """Solve the convex QP for given n and Kmax."""
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2]
    for k in range(-Kmax, Kmax + 1):
        Q = build_Q(k, n)
        constraints.append(cp.quad_form(h, Q) >= n / 2 - t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    if warm_h is not None:
        h.value = warm_h
    prob.solve(solver=cp.OSQP, verbose=False, max_iter=5000, eps_abs=1e-9)
    return h.value, t.value

def binary_refine(h_frac, n, Kmax, dx, time_limit):
    """Refine fractional solution to binary with local search."""
    # Start with the top n/2 entries as 1s
    idx = np.argsort(-h_frac)[:n // 2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1
    best = c5_of_h(h_bin, dx)
    best_h = h_bin.copy()
    # Precompute correlation array for current h_bin
    corr = np.correlate(h_bin, 1 - h_bin, mode='full')
    max_k = Kmax
    # Local search with swaps
    while time.time() < time_limit:
        # Select a 1 and a 0 at random
        i1 = np.random.choice(np.where(h_bin == 1)[0])
        i0 = np.random.choice(np.where(h_bin == 0)[0])
        # Tentative swap
        h_bin[i1] = 0
        h_bin[i0] = 1
        # Update correlation for affected indices
        old_corr_max = np.max(corr)
        new_corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best:
            best = new_c5
            best_h = h_bin.copy()
            corr = new_corr
        else:
            # Revert
            h_bin[i1] = 1
            h_bin[i0] = 0
    return best_h, best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    # Parameters to sweep
    n_list = [350, 400]
    Kmax_list = [int(0.75 * n) for n in n_list]
    warm_h = None
    for idx, (n, Kmax) in enumerate(zip(n_list, Kmax_list)):
        if time.time() - start_time > budget_s - 2:
            break
        try:
            h_sol, c5_sol = solve_qp(n, Kmax, warm_h)
            h_sol = np.clip(h_sol, 0, 1)  # Ensure bounds
            # Post-process to ensure sum is correct (should already be)
            h_sol = h_sol * (n / 2) / np.sum(h_sol)
            dx = 2.0 / n
            # Binary refinement
            local_time_limit = start_time + budget_s - 50
            h_bin, c5_refined = binary_refine(h_sol, n, Kmax, dx, local_time_limit)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_bin.copy()
                best_n = n
            # Set warm start for next size
            warm_h = h_sol
        except Exception as e:
            continue
    # Final evaluation at higher resolution if needed
    if best_h is not None:
        dx = 2.0 / best_n
        # Want to make sure we are not off by much by finer evaluation
        h_test = best_h.copy()
        test_c5 = c5_of_h(h_test, dx)
        return (best_h, test_c5, best_n)
    # If we failed to improve, return a reasonable fallback
    n_fallback = 400
    h_fallback = np.zeros(n_fallback)
    h_fallback[:n_fallback//2] = 1
    dx_fallback = 2.0 / n_fallback
    h_fallback = h_fallback * (n_fallback / 2) / np.sum(h_fallback)
    fallback_c5 = c5_of_h(h_fallback, dx_fallback)
    return (h_fallback, fallback_c5, n_fallback)
