import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp
from scipy.optimize import minimize_scalar

def proj_simplex_box(h, n):
    """Project h onto the set {0 <= h_i <= 1, sum h_i = n/2}."""
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - n/2) < 1e-10:
        return h
    diff = n/2 - total
    # Add diff/n to each h_i to make the sum correct.
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def c5(h):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def worst_shift(h):
    """Return the shift index k that achieves max overlap."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # The actual shift is the center of the correlation
    # For full correlation it's 0 to 2*n-2, center is n-1
    return max_idx - (len(h) - 1)

def build_lp_relaxation(h, n):
    """Build and solve a linear relaxation of the problem."""
    # Variables
    h_var = cp.Variable(n)
    t = cp.Variable(1)
    z = cp.Variable((n, n), nonneg=True)
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n/2,
        t <= 0.5  # Upper bound on t
    ]
    
    # McCormick relaxation for z[i,j] >= h[i] + h[j] - 1
    for i in range(n):
        for j in range(n):
            k = (j - i) % n  # shift
            if k == 0:
                continue  # Skip the diagonal redirecting to sum of h^2 later
            # Add McCormick constraints
            constraints += [
                z[i,j] >= h_var[i] + h_var[j] - 1,
                z[i,j] <= h_var[i],
                z[i,j] <= h_var[j],
                z[i,j] >= 0
            ]
            # Connect to sum
            # We have sum_over_i z[i,i+k] = y_k for a given k. It's quadratic
            # Instead of reconstructing, we'll directly capture the outcome:
            # Each h_i h_{i+k} is bounded by McCormick.
    
    # Build sums for each shift
    sum_terms = []
    for shift in range(1, n):  # 1 to n-1
        # Compute sum over i of h[i] h[i+shift] using z variables
        total = 0
        for i in range(n):
            j = (i + shift) % n
            if i < j:
                total += z[i,j]
            else:
                total += z[j,i]
        sum_terms.append(total)
    
    # Minimize C5 = 1 - max sum of h_i h_{i+k} = 1 - max_k y_k
    # So maximize sum terms -> minimize min_k y_k = min_k sum_i h_i h_{i+k}
    for term in sum_terms:
        constraints.append(term >= t)
    
    # Problem
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        h_lp = h_var.value
    except:
        # Return initial guess if failed
        h_lp = h.copy()
    
    return h_lp

def gradient_step(h, k, alpha=0.01):
    """Step in the descent direction for shift k."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        # dI_k/dh_i = dx * (1 - 2 * h_{i+k})
        idx = (i + k) % n
        grad[i] = dx * (1 - 2 * h[idx])
        
    h_new = h - alpha * grad
    return proj_simplex_box(h_new, n)

def anneal(h, time_limit):
    """Perform simulated annealing."""
    np.random.seed(42)
    start_time = time.time()
    current_h = h.copy()
    current_c5 = c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    T = 0.1
    T_min = 1e-6
    alpha = 0.95
    iter_per_temp = 100
    
    while T > T_min and (time.time() - start_time) < time_limit:
        for _ in range(iter_per_temp):
            if (time.time() - start_time) > time_limit:
                break
            # Choose two indices and perturb
            i, j = np.random.choice(len(h), 2, replace=False)
            delta = np.random.uniform(-0.05, 0.05)
            # Ensure that the change remains within bounds
            h_trial = current_h.copy()
            h_trial[i] += delta
            h_trial[j] -= delta
            h_trial = proj_simplex_box(h_trial, len(h_trial))
            trial_c5 = c5(h_trial)
            
            if trial_c5 < current_c5 or np.random.rand() < np.exp(-(trial_c5 - current_c5) / T):
                current_h = h_trial
                current_c5 = trial_c5
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    
        T *= alpha
        
    return best_h, best_c5
