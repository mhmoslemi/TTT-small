def is_valid_binary_vector(x):
    return np.all(np.isin(x, [-1, 1]))

def create_shift_matrix(n, k):
    """Create a sparse shift matrix S_k"""
    # Create shift matrix as a sparse CSR matrix
    data = []
    row_ind = []
    col_ind = []
    
    for i in range(n):
        j = (i + k) % n
        if 0 <= j < n:
            data.append(1)
            row_ind.append(i)
            col_ind.append(j)
            
    S_k = np.zeros((n, n))
    S_k[row_ind, col_ind] = data
    return S_k

def solve_sdp(n, P, use_symmetry=True):
    """Solve the SDP relaxation for the Erdos problem"""
    if use_symmetry:
        m = (n + 1) // 2  # Reduced dimension
        X = SemiDefinite(m)
        # Reconstruct full matrix from reduced
        X_full = np.zeros((n, n))
        # Fill symmetric part
        for i in range(m):
            for j in range(m):
                if i == j:
                    X_full[i, j] = X[i, j]
                    X_full[n-1-i, n-1-j] = X[i, j]
                elif i < j:
                    X_full[i, j] = X[i, j]
                    X_full[n-1-j, n-1-i] = X[i, j]
                    X_full[j, i] = X[i, j]
                    X_full[n-1-i, n-1-j] = X[i, j]
    else:
        m = n
        X = SemiDefinite(m)
    
    # Constraint that all diagonal elements are 1
    constraints = [X[i, i] == 1 for i in range(m)]
    
    # Constraint that sum of entries is 0 (reduced to sum of first m entries)
    if use_symmetry:
        constraints.append(sum(X[i, j] for i in range(m) for j in range(m)) == 0)
    else:
        constraints.append(sum(X[i, j] for i in range(m) for j in range(m)) == 0)
        
    # Add shift constraints - only some shifts to keep problem tractable
    max_shift = min(int(n * 0.9), n//2)
    K = list(range(-max_shift, max_shift + 1))
    K = [k for k in K if k != 0]
    
    t = cp.Variable()
    for k in K:
        if use_symmetry:
            # For symmetric case, avoid redundant calculations
            pass
        else:
            S_k = create_shift_matrix(n, k)
            # Convert to CVXPY expression
            tr_Sk_X = cp.trace(S_k @ X)
            constraints.append(t >= n/4 - 0.25 * tr_Sk_X)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=SCS, eps=1e-6, max_iters=5000)
    except:
        # Fallback to a different method
        return None, None
        
    return prob.value, X

def gaussian_rounding(X_opt, n, use_symmetry=True):
    """Gaussian rounding procedure with zero-sum constraint"""
    if X_opt is None:
        return None
    
    if use_symmetry:
        m = (n + 1) // 2
        # Get a Cholesky-like decomposition of the matrix
        eigenvals, eigenvecs = np.linalg.eigh(X_opt.value)
        # Make sure eigenvals are non-negative
        eigenvals = np.maximum(eigenvals, 0)
        # Get eigen-decomposition
        V = eigenvecs @ np.diag(np.sqrt(eigenvals))
        # Get random vector
        g = np.random.randn(m)
        y = V.T @ g
        # Round to {-1,+1}
        x = np.sign(y)
        # Adjust to satisfy the zero-sum constraint
        s = np.sum(x)
        if s != 0:
            # Find the indices with the smallest absolute values to flip
            flip_indices = np.argsort(np.abs(y))[:int(abs(s)/2)]
            x[flip_indices] *= -1
    else:
        m = n
        # Get a Cholesky-like decomposition of the matrix
        eigenvals, eigenvecs = np.linalg.eigh(X_opt.value)
        # Make sure eigenvals are non-negative
        eigenvals = np.maximum(eigenvals, 0)
        # Get eigen-decomposition
        V = eigenvecs @ np.diag(np.sqrt(eigenvals))
        # Get random vector
        g = np.random.randn(m)
        y = V.T @ g
        # Round to {-1,+1}
        x = np.sign(y)
        # Adjust to satisfy the zero-sum constraint
        s = np.sum(x)
        if s != 0:
            # Find the indices with the smallest absolute values to flip
            flip_indices = np.argsort(np.abs(y))[:int(abs(s)/2)]
            x[flip_indices] *= -1
    
    # Convert to full vector if symmetric reduction used
    if use_symmetry:
        x_full = np.zeros(n)
        x_full[:m] = x
        x_full[m:] = x[::-1]
        x = x_full
        
    return x

def update_correlation(c, h, i_old, i_new, n, dx):
    """Update correlation efficiently when swapping h[i_old] and h[i_new]"""
    # h[i_old] changed from 1 to 0, h[i_new] changed from 0 to 1
    # When h changed from 1 to 0, subtract from correlations
    # When h changed from 0 to 1, add to correlations
    new_c = c.copy()
    
    for k in range(n):
        # Compute contribution to correlation shift k
        if (i_old - k) % n < n and (i_new - k) % n < n:
            new_c[k] -= h[i_old] * (1 - h[(i_old - k) % n]) - h[i_new] * (1 - h[(i_new - k) % n])
            # At i_old, we lose h[i_old] * (1 - h[i_old - k])
            # At i_new, we gain h[i_new] * (1 - h[i_new - k])
            # But we already have h[i_new] * (1 - h[i_new - k]) in new_c[k] 
            # and we are substracting the configuration that was in the previous state
            
    return new_c

def local_refinement(h, n_points, dx):
    """Local binary refinement via swaps"""
    c = np.correlate(h, 1-h, mode='full') * dx
    cmax = np.max(c)
    best_h = h.copy()
    best_cmax = cmax
    
    # Local search: perform 5000 swaps
    num_swaps = 5000
    for iteration in range(num_swaps):
        if iteration % 100 == 0 and iteration > 0:
            c = np.correlate(best_h, 1 - best_h, mode='full') * dx
            cmax = np.max(c)
            
        # Find indices with h[i] = 1 and h[j] = 0
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i1 = np.random.choice(len(ones))
        i0 = np.random.choice(len(zeros))
        
        # Create new h 
        h_new = best_h.copy()
        h_new[ones[i1]] = 0
        h_new[zeros[i0]] = 1
        
        # Update correlations efficiently
        c_new = c.copy()
        k_start = max(0, -ones[i1])
        k_end = min(n_points, n_points - zeros[i0])
        for k in range(k_start, k_end):
            # contribution to shift k
            contrib_i1 = h_new[ones[i1]] * (1 - h_new[(ones[i1] - k) % n_points])
            contrib_i0 = h_new[zeros[i0]] * (1 - h_new[(zeros[i0] - k) % n_points])
            contrib_old_i1 = best_h[ones[i1]] * (1 - best_h[(ones[i1] - k) % n_points])
            contrib_old_i0 = best_h[zeros[i0]] * (1 - best_h[(zeros[i0] - k) % n_points])
            c_new[k] += (contrib_i1 + contrib_i0 - contrib_old_i1 - contrib_old_i0)
            
        cmax_new = np.max(c_new)

        if cmax_new < cmax:
            best_h = h_new
            c = c_new
            cmax = cmax_new
        
    return best_h, cmax
