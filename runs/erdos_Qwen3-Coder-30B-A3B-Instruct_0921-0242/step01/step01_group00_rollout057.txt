import numpy as np
import time
from cvxpy import Problem, Minimize, Variable, sum, multiply, Convex
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def solve_spectral_lp(n):
    """Solve the linear program for the optimal power spectrum."""
    # Variables
    p = Variable(n)
    t = Variable()
    
    # Constraints
    constraints = [p >= 0]
    constraints += [p <= n]  # bound each power

    # Parseval constraint
    constraints += [sum(p) == n * n / 2]

    # Autocorrelation constraints
    # For all k=1 to n-1
    for k in range(1, n):
        # cos(2πkm/n) for m=0..n-1
        cos_term = np.cos(2 * np.pi * k * np.arange(n) / n)
        constraints += [sum(multiply(p, cos_term)) >= t]

    # Objective
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    prob.solve(solver='ECOS', verbose=False)
    
    if prob.status != 'optimal':
        # Fallback to simple bounds
        t_val = 0.0
        p_vals = np.ones(n) * n/2 / n
    else:
        t_val = t.value
        p_vals = p.value
        
    return p_vals, t_val

def gerchberg_saxton(p_spectrum, n, max_iter=200):
    """Gerchberg-Saxton phase retrieval."""
    # Random phases
    phases = np.random.uniform(0, 2*np.pi, n)
    # Complex spectrum with given power
    X = np.sqrt(p_spectrum) * np.exp(1j * phases)
    
    # Inverse transform
    x = irfft(X, n=n)
    
    for _ in range(max_iter):
        # Clip to [0,1]
        y = np.clip(x, 0, 1)
        # Water filling to enforce sum = n/2
        current_sum = np.sum(y)
        delta = n/2 - current_sum
        if abs(delta) > 1e-9:
            # Binary search for lambda
            lamb_min, lamb_max = -1.0, 1.0
            for _ in range(50):
                lambda_guess = (lamb_min + lamb_max) / 2
                y_try = np.clip(y + lambda_guess, 0, 1)
                sum_try = np.sum(y_try)
                if abs(sum_try - n/2) < 1e-9:
                    break
                elif sum_try < n/2:
                    lamb_min = lambda_guess
                else:
                    lamb_max = lambda_guess
            lambda_opt = (lamb_min + lamb_max)/2
            y = np.clip(y + lambda_opt, 0, 1)
        # Forward transform
        Y = rfft(y, n=n)
        # Spectral projection (magnitude to p_spectrum, keep phase)
        Y_proj = np.sqrt(p_spectrum) * np.exp(1j * np.angle(Y))
        # Inverse transform
        x = irfft(Y_proj, n=n)
    
    return x

def project_simplex_box(x, target_sum):
    """Project x onto the simplex with box constraints."""
    x = np.clip(x, 0, 1)
    s = np.sum(x)
    delta = target_sum - s
    if abs(delta) < 1e-9:
        return x
    # Add constant shift
    x = x + delta / len(x)
    x = np.clip(x, 0, 1)
    return x

def binary_sharpen(x, mu_vals=None):
    """Binary sharpening via projected gradient."""
    if mu_vals is None:
        mu_vals = [1e-4, 1e-3, 1e-2]
    x = x.copy()
    for mu in mu_vals:
        # Simple gradient step
        for _ in range(10):  # few steps
            grad = x * (1 - x) * mu
            x = x - 0.1 * grad
            x = project_simplex_box(x, len(x)/2)
    # Round near zeros and ones
    x = np.where(x < 0.01, 0, x)
    x = np.where(x > 0.99, 1, x)
    x = project_simplex_box(x, len(x)/2)
    return x

def gradient_worst_shift(x, dx):
    """Refine using projected gradient on worst shift."""
    max_corr = -1.0
    worst_k = -1
    # Do full correlation to find worst shift
    corr = correlate(x, 1 - x, mode="full")
    for k in range(1, len(x)):
        I_k = corr[k + len(x) - 1] * dx
        if I_k > max_corr:
            max_corr = I_k
            worst_k = k
    if worst_k == -1:
        return x
    # Gradient w.r.t. h_i for shift worst_k
    grad = np.zeros_like(x)
    for i in range(len(x)):
        # h_{i+k} term
        idx = (i + worst_k) % len(x)
        grad[i] -= dx * x[idx]
        # h_{i-k} term 
        idx = (i - worst_k) % len(x)
        grad[i] += dx * x[idx]
    # Perform gradient descent step
    step_size = 1.0
    max_iter = 100
    for i in range(max_iter):
        trial = x - step_size * grad
        trial = project_simplex_box(trial, len(x)/2)
        # Check value 
        trial_corr = correlate(trial, 1 - trial, mode="full")
        trial_max = max(trial_corr[1+len(x):]) * dx
        if trial_max < max_corr:
            return trial
        step_size /= 2
        if step_size < 1e-6:
            break
    return x

def swap_anneal(x, dx, time_remaining):
    """Simulated annealing swap refinement."""
    start_time = time.time()
    T = 1e-3
    while time.time() - start_time < time_remaining:
        # Pick two indices
        i = np.random.randint(len(x))
        j = np.random.randint(len(x))
        if x[i] <= 0 or x[j] >= 1:
            continue
        # Random transfer amount
        delta = np.random.uniform(0, min(x[i], 1 - x[j], 0.02))
        # Try swap
        x_new = x.copy()
        x_new[i] -= delta
        x_new[j] += delta
        # Project
        x_new = project_simplex_box(x_new, len(x)/2)
        # Check improvement
        c5_new = np.max(correlate(x_new, 1 - x_new, mode="full")) * dx
        c5_old = np.max(correlate(x, 1 - x, mode="full")) * dx
        if c5_new < c5_old or np.random.random() < np.exp(-(c5_new - c5_old) / T):
            x = x_new
        T *= 0.99
    return x

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 650
    dx = 2.0 / n
    
    # Stage 1: Spectral convex relaxation
    p_spectrum, t_opt = solve_spectral_lp(n)
    c5_relaxed = 2 * (1 - t_opt) / n
    if c5_relaxed < 0.3807:
        return (np.ones(n) * 0.5, c5_relaxed, n)
    
    time_after_symp = time.time()
    if time_after_symp - start_time > 0.9 * budget_s:
        h = np.ones(n) * 0.5
        return (h, 2 * (1 - t_opt) / n, n)
    
    # Stage 2: Phase retrieval with Gerchberg-Saxton
    h_gs = None
    for _ in range(10):
        h_gs_trial = gerchberg_saxton(p_spectrum, n)
        t_emp = np.min(correlate(h_gs_trial, 1 - h_gs_trial, mode="full")[1+n:])
        if t_emp >= 0.95 * t_opt:
            h_gs = h_gs_trial
            break
    if h_gs is None:
        h_gs = gerchberg_saxton(p_spectrum, n, max_iter=100)
        
    time_after_gs = time.time()
    if time_after_gs - start_time > 0.95 * budget_s:
        return (h_gs, evaluate_c5(h_gs), n)
    
    # Stage 3: Feasibility projection + binary sharpening
    h_gs = project_simplex_box(h_gs, n/2)
    h_rounded = binary_sharpen(h_gs)
    h_final = project_simplex_box(h_rounded, n/2)
    
    # Stage 4: Worst shift refinement
    time_remaining = budget_s - (time.time() - start_time)
    max_iter = 3000
    iter_count = 0
    while iter_count < max_iter and (time.time() - start_time) < 0.9 * budget_s:
        h_prev = h_final.copy()
        h_final = gradient_worst_shift(h_final, dx)
        if abs(evaluate_c5(h_final) - evaluate_c5(h_prev)) < 1e-8:
            break
        iter_count += 1
        
    # Stage 5: Swap refinement
    time_remaining = budget_s - (time.time() - start_time)
    if time_remaining > 5.0:
        h_final = swap_anneal(h_final, dx, time_remaining - 5.0)
    
    final_c5 = evaluate_c5(h_final)
    return (h_final, final_c5, n)
