def compute_block_autocorrelation(ell, h, n_eval=200):
    """Compute the autocorrelation for a block-based function."""
    K = len(ell)
    # Compute block endpoints
    cum_ell = np.concatenate([[0], np.cumsum(ell)])
    block_ends = cum_ell[1:]
    
    # Precompute block overlaps
    overlaps = np.zeros((K, K, n_eval))
    s_vals = np.linspace(0, 2, n_eval)
    
    for i in range(K):
        for j in range(K):
            for k, s in enumerate(s_vals):
                l1, r1 = cum_ell[i], cum_ell[i+1]
                l2, r2 = cum_ell[j] - s, cum_ell[j+1] - s
                overlap_len = max(0, min(r1, r2) - max(l1, l2))
                overlaps[i, j, k] = overlap_len
    
    # Compute full correlation for each shift
    corr_vals = np.zeros(n_eval)
    for k in range(n_eval):
        total = 0
        for i in range(K):
            for j in range(K):
                total += h[i] * h[j] * overlaps[i, j, k]
        corr_vals[k] = total
    
    # The objective is to minimize 1 - corr_vals. We use the max of (1 - c)
    return np.max(1 - corr_vals)

def compute_h_from_params(u, v, K):
    """Convert raw parameters to heights and lengths."""
    # Lengths via softmax
    ell = 2 * np.exp(u) / np.sum(np.exp(u))
    
    # Heights via sigmoid
    h_tilde = 1 / (1 + np.exp(-v))
    
    # Scale to satisfy integral constraint
    integral = np.sum(h_tilde * ell)
    alpha = 1.0 / integral
    h = np.minimum(1.0, alpha * h_tilde)
    
    # If any h equals 1, redistribute
    if np.any(h == 1):
        excess = np.sum(h) - 1
        mask = h < 1
        if np.sum(mask) > 0:
            h[mask] *= (1 - excess) / np.sum(h[mask])
    
    return ell, h

def block_objective(params, K=6, n_eval=200):
    """Objective for block-based optimization."""
    u = params[:K]
    v = params[K:]
    ell, h = compute_h_from_params(u, v, K)
    return compute_block_autocorrelation(ell, h, n_eval)

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    diff = n/2 - np.sum(h)
    
    if abs(diff) < 1e-10:
        return h
    
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def refine_solution(h, n):

    dx = 2.0 / n
    max_iter = 500
    best_c5 = float('inf')
    h_best = h.copy()
    
    for iteration in range(max_iter):
        c = correlate(h, 1 - h, mode='full') 
        worst_shift = np.argmax(c)
        c_worst = c[worst_shift]
        # Derivative of integral with respect to h_i for worst shift
        grad = np.zeros_like(h)
        for i in range(n):
            # index for left term
            left_idx = (i + worst_shift) % n
            # index for right term
            right_idx = (i - worst_shift) % n
            grad[i] = dx * (1 - h[left_idx] - h[right_idx])
        
        # Perform projected gradient step
        step_size = 1.0
        while step_size > 1e-10:
            h_new = h - step_size * grad
            h_new = project_to_feasible(h_new, n)
            c_new = correlate(h_new, 1 - h_new, mode='full')
            if np.max(c_new) < np.max(c):
                break
            step_size *= 0.5
            
        if step_size <= 1e-10:
            break
            
        h = h_new.copy()
        current_c5 = np.max(c_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            h_best = h.copy()
        else:
            break
            
    return h_best, best_c5
