```python
import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp
from cvxpy import CBC

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Try different discretizations
    for n in [200, 250, 300]:
        if time.time() - start_time > budget_s - 10:
            break
            
        dx = 2.0 / n
        target_sum = n / 2.0
        
        # Heuristic step: Simulated Annealing
        def simulated_annealing(max_iter):
            h = np.zeros(n)
            ones = np.random.choice(n, int(target_sum), replace=False)
            h[ones] = 1
            current_c5 = c5_of_h(h, dx)
            best_h = h.copy()
            best_c5_val = current_c5
            
            temp = 1.0
            min_temp = 1e-8
            cooling_rate = 0.9995
            
            for _ in range(max_iter):
                if time.time() - start_time > budget_s - 5:
                    break
                    
                # Pick two random indices
                i1 = np.random.randint(0, n)
                i0 = np.random.randint(0, n)
                
                # Make sure they're different and have opposite values
                if h[i1] == h[i0]:
                    continue
                    
                # Compute new configuration
                new_h = h.copy()
                new_h[i1] = 1 - new_h[i1]
                new_h[i0] = 1 - new_h[i0]
                
                new_c5 = c5_of_h(new_h, dx)
                delta = new_c5 - current_c5
                
                if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                    h = new_h
                    current_c5 = new_c5
                    if current_c5 < best_c5_val:
                        best_h = h.copy()
                        best_c5_val = current_c5
                
                temp *= cooling_rate
                if temp < min_temp:
                    temp = min_temp
                    
            return best_h, best_c5_val
            
        # Run SA with dynamic limit
        max_sa_iter = min(100000, max(1000, int((budget_s * 0.15) * 1000 / n)))
        h_heuristic, c5_heuristic = simulated_annealing(max_sa_iter)
        
        # MIP Solving
        if time.time() - start_time > budget_s - 10:
            break
            
        # Core shifts
        core_shifts = [1, n//4, n//2, 3*n//4]
        
        # MIP with progressive constraints (simplified model)
        # Variables
        h = cp.Variable(n, boolean=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h) == target_sum, t >= 0]
        
        # Core y variables for pairs
        # Just take a reasonable subset
        y_vars = []
        pairs = []
        
        # Use only specific pairs for a manageable problem
        base_pairs = []
        for k in core_shifts:
            for i in range(n - k):
                base_pairs.append((i, i + k))
                
        # Limit pairs to reduce problem size
        pairs = base_pairs[:min(len(base_pairs), 2000)]
        
        # For each pair, create y variable    
        y = cp.Variable(len(pairs), boolean=True)
        y_vars = y
        
        # Linearization
        for l, (i, j) in enumerate(pairs):
            constraints += [
                y[l] <= h[i],
                y[l] <= h[j],
                y[l] >= h[i] + h[j] - 1
            ]
        
        # Add constraints for core shifts
        for k in core_shifts:
            # Sum of relevant y_i,i+k's
            relevant_indices = [p for p, (i, j) in enumerate(pairs) if j - i == k]
            if relevant_indices:
                c_sum = cp.sum(y[relevant_indices])
                # t + c_sum >= target_sum => t >= target_sum - c_sum
                constraints.append(t >= target_sum - c_sum)
        
        # Add warm start constraint
        constraints.append(t <= c5_heuristic)
        
        # Objective
        objective = cp.Minimize(t)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 2:
            continue
            
        prob.solve(solver=CBC, maxSeconds=remaining_time*0.5)
        
        if prob.status == 'optimal':
            h_opt = np.round(h.value).astype(int)
        else:
            h_opt = h_heuristic.astype(int)
            
        # Fine check with full correlation
        best_c5_full = c5_of_h(h_opt, dx)
        
        # Local refinement via swaps
        if time.time() - start_time > budget_s - 3:
            continue
            
        current_h = h_opt.copy()
        current_c5 = best_c5_full
        
        max_local_iter = min(2000, int(10000 * (budget_s - (time.time() - start_time)) / 600))
        for i in range(max_local_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Pick two indices
            i1 = np.random.randint(0, n)
            i0 = np.random.randint(0, n)
            
            if i1 == i0:
                continue
                
            # Check if they have same value
            if current_h[i1] == current_h[i0]:
                continue
                
            # Try swap
            new_h = current_h.copy()
            new_h[i1] = 1 - new_h[i1]
            new_h[i0] = 1 - new_h[i0]
            
            new_c5 = c5_of_h(new_h, dx)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                
        c5_final = c5_of_h(current_h, dx)
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = current_h.copy()
            best_n = n
            
    return (best_h, best_c5, best_n)
```
