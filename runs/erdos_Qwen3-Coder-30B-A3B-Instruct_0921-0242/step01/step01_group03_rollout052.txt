def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def correlate_fft(Y):
    """Compute all cross-correlations using FFT"""
    N, r = Y.shape
    c_k = np.zeros((N, N))  # c_k[i] = sum of y_i * y_{i+k}
    for d in range(r):
        y_d = Y[:, d]
        # Circular convolution of y_d with its reverse
        fft_y_d = np.fft.fft(y_d, 2*N)
        c_d = np.fft.ifft(fft_y_d * np.conj(fft_y_d)).real[:N]
        c_k += c_d[:, None]
    return c_k

def lse_smooth(Y, t_vec):
    """Log-sum-exp smooth surrogate"""
    # t_vec[i] = sum of y_i * y_{i+k} over r columns
    # Note: this should be just c_k is the original k-space formulation
    # But we change sign convention for consistency with optimization
    # Original is min max, so our surrogate is max min
    # Flip sign of t_vec to avoid sign flips
    t_vec = -t_vec # this makes it as evoked in earlier parts
    tmp = np.exp(t_vec / 0.05)  # temperature tau = 0.05
    return -0.05 * np.log(np.sum(tmp)) # -tau * log(sum(exp(-t/tau)))

def min_correlations_of_h(h):
    """Return all c_k values"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return corr * dx

def project_to_feasible_sdpmode(Y):
    """ Project Y such that h = ||Y_i||^2, sum(h) = N/2, and h in [0,1]^n
    Y is N x r matrix
    """
    N, r = Y.shape
    h = np.sum(Y*Y, axis=1)

    #1. Clip norm
    norms = np.sqrt(np.sum(Y*Y, axis=1))
    safe_norms = np.clip(norms, 0, 1)
    mask = (norms > 1)
    Y[mask] = Y[mask] / norms[mask][:,None] * safe_norms[mask][:,None]

    #2. Enforce mass
    current_sum = np.sum(h)
    alpha = np.sqrt((N/2) / current_sum) if current_sum > 1e-10 else 1.0
    Y = Y * alpha

    #3. Re-clip (because scaling changes norms, new ones might exceed 1)
    norms = np.sqrt(np.sum(Y*Y, axis=1))
    safe_norms = np.clip(norms, 0, 1)
    mask = (norms > 1)
    if np.any(mask):
        Y[mask] = Y[mask] / norms[mask][:,None] * safe_norms[mask][:,None]

    return Y
