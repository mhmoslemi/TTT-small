import numpy as np
import scipy.optimize as opt

def correlation_fft(h):
    """Compute all f_k(h) values using FFT."""
    N = len(h)
    h_pad = np.pad(h, (0, N), 'constant')
    g_pad = np.pad(1 - h, (0, N), 'constant')
    c = fft.fft(h_pad) * fft.fft(g_pad)
    return np.real(fft.ifft(c))[:N]

def weighted_gradient_fft(h, p):
    """Compute ∇F_τ(h) using weighted convolution."""
    N = len(h)
    p_pad = np.pad(p, (0, N), 'constant')
    h_pad = np.pad(h, (0, N), 'constant')
    g_pad = np.pad(1 - h, (0, N), 'constant')
    
    w1 = fft.ifft(fft.fft(p_pad) * fft.fft(g_pad))
    w2 = fft.ifft(fft.fft(p_pad[::-1]) * fft.fft(h_pad))
    
    return np.real(w1[:N] - w2[:N])

def project_onto_Ω(v):
    """Euclidean projection onto {h : 0≤h≤1, sum(h)=N/2}."""
    z = np.clip(v, 0, 1)
    r = len(z) / 2 - np.sum(z)
    if abs(r) < 1e-12:
        return z
    
    I = np.where((z > 0) & (z < 1))[0]
    it = 0
    while abs(r) > 1e-12 and len(I) > 0:
        if it > 10:
            # Hard failure case. fallback to simple distribution.
            z[I] += r / len(I)
            z = np.clip(z, 0, 1)
            break
        r = len(z) / 2 - np.sum(z)
        if abs(r) < 1e-12:
            break
        dr = r / len(I)
        z[I] += dr
        z[I] = np.clip(z[I], 0, 1)
        I = np.where((z > 0) & (z < 1))[0]
        it += 1
    return z

def binary_from_top(h, count):
    """Convert continuous h into a binary vector with specified count of 1s."""
    idx = np.argsort(h)[::-1]
    bin_h = np.zeros_like(h)
    bin_h[idx[:count]] = 1.0
    return bin_h

def fractional_correction(h_bin):

    s = np.sum(h_bin)

    if abs(s - N/2) < 1e-10:
        return h_bin

    I1 = np.where(h_bin == 1)[0]
    # Find candidates for fractional correction (likely near 0 or 1)
    small = np.where(np.abs(h_bin - 0.5) < 0.3)[0]  # nearby 0.5
    if len(small) > 8:
        small = np.sort(small)[len(small)-8:]
    
    # Solve LP: keep bin_h[...] fixed, adjust h[small]
    if len(small) <= 0:
        return h_bin
    
    # Formulate LP
    x = cp.Variable(len(small))
    h_fix = h_bin.copy()
    h_fix[small] = 0.0
    # Build current fixed part: h_fix + x is our new h
    # sum((h_fix + x)[i]) = N/2, so sum(x[i]) = N/2 - sum(h_fix)
    sum_fixed = np.sum(h_fix)  # Known quantity
    target_sum = N/2
    sum_target = target_sum - sum_fixed
    
    # Objective is to minimize ∫ h(x)(1-h(x+k)) dx
    # We approximate this for full discrete representation: hard to compute properly
    # Instead, just relax the constraints slightly and return.
    # Still can solve a very tiny LP to handle very small correction.
    
    # Final: Just return original h_bin if no adjustment needed
    return h_bin

def swap(h, i, j):
    """Swap h[i] and h[j]."""
    h_swp = h.copy()
    h_swp[i], h_swp[j] = h_swp[j], h_swp[i]
    return h_swp

def c5_of_h(h):
    """Compute the worst overlap integral C5 for a given step function h."""
    corr = correlation_fft(h)
    return np.max(corr)
