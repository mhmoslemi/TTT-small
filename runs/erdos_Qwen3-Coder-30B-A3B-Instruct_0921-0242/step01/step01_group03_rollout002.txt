import scipy.optimize as opt
import time
import scipy.fft as fft

def run(seed=42, budget_s=600):

    start_time = time.time()
    
    # Parameters
    N = 256  # Base resolution (must be a power of 2)
    dx = 2.0 / N

    h = np.zeros(N)
    h[:N//2] = 1.0
    np.random.shuffle(h)

    # Smoothing parameters
    τ_start = 0.05
    τ_factor = 0.5
    outer_cycles = 6
    inner_iters = 150
    
    τ = τ_start
    
    for _ in range(outer_cycles):
        if time.time() - start_time > budget_s - 30:
            break
        h_prev = h.copy()
        α = 1.0  # Initial step size
        
        for t in range(inner_iters):
            if time.time() - start_time > budget_s - 30:
                break
                
            # Nesterov extrapolation
            β = (t - 1) / (t + 2)
            y = h + β * (h - h_prev)
            
            # Evaluate f_k for all k using FFT
            f = correlation_fft(y)
            # Compute weights (softmax)
            f_scaled = f / τ
            f_scaled -= np.max(f_scaled)
            p = np.exp(f_scaled)
            p /= p.sum()
            
            # Compute gradient
            grad = weighted_gradient_fft(y, p)
            
            # Backtracking line search
            while True:

                if Fτ(z_proj, τ) <= Fτ(y, τ) - 0.5 * α * np.dot(grad, grad):
                    break
                α *= 0.5

    h_bin = binary_from_top(h, N//2)

    h_refined = fractional_correction(h_bin)

    best_h = h_refined

                best_c5_save = final_c5

def project_onto_Ω(v):
    """Project onto the feasible set: 0 ≤ h ≤ 1, sum = N/2"""
    # Clip to [0,1]
    z = np.clip(v, 0, 1)
    r = len(v) / 2 - np.sum(z)
    
    # Water-filling correction
    iters = 0
    while abs(r) > 1e-12 and iters < 10:
        I = np.where((z > 0) & (z < 1))[0]
        if len(I) == 0:
            break
        dz = r / len(I)
        z[I] += dz
        z = np.clip(z, 0, 1)
        r = len(v) / 2 - np.sum(z)
        iters += 1
    return z

def Fτ(h, τ):
    """Smooth approximation to max_k f_k(h)"""
    f = correlation_fft(h)
    return τ * np.log(np.sum(np.exp(f / τ)))

def correlation_fft(h):
    """Use FFT for fast computation of f_k = Σ_i h_i * (1 - h_{i+k})"""
    h_pad = np.concatenate([h, np.zeros(len(h))])
    g_pad = np.concatenate([1 - h, np.zeros(len(h))])
    conv = fft.fft(h_pad) * fft.fft(g_pad)
    c = fft.ifft(conv).real
    return c[:len(h)]

def weighted_gradient_fft(h, p):
    """Compute weighted gradient ∇F_τ(h) = Σ_k p_k (1 - h_{i+k} - h_{i-k})"""
    h_pad = np.concatenate([h, np.zeros(len(h))])
    g_pad = 1 - h_pad
    p_pad = np.concatenate([p, np.zeros(len(h))])
    
    # w1 = conv(p, 1-h)
    w1 = fft.irfft(fft.rfft(p_pad) * fft.rfft(g_pad[:len(h)]))
    w1 = w1[:len(h)]
    
    # w2 = conv(p_rev, h)
    p_rev = p[::-1]
    p_rev_pad = np.concatenate([p_rev, np.zeros(len(h))])
    w2 = fft.irfft(fft.rfft(p_rev_pad) * fft.rfft(h_pad))
    w2 = w2[:len(h)]
    
    return w1 - w2

def binary_from_top(h, n_ones):
    """Convert h to binary vector with n_ones ones"""
    idxs = np.argsort(h)[::-1][:n_ones]
    h_bin = np.zeros_like(h)
    h_bin[idxs] = 1.0
    return h_bin

def fractional_correction(h_bin):

    N = len(h_bin)
    h_exact = h_bin.copy()

    error = N/2 - current_sum
    
    if abs(error) < 1e-10:
        return h_exact
    
    # Find fractional positions
    frac_idx = np.where((h_exact > 0) & (h_exact < 1))[0]
    
    # If not many fractional, marginally adjust
    if len(frac_idx) < 4:
        for i in frac_idx:
            h_exact[i] += error / len(frac_idx)

    h_exact = np.clip(h_exact, 0, 1)

def swap(h, i, j):
    """Swap h[i] and h[j] and return new vector"""
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def c5_of_h(h):
    """Compute C5 with the standard np.correlate method"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def upsample(h, N_new):
    """Linear interpolation upsampling by factor 2"""
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, N_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new
