import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 200  # grid size
    d = n // 2 + 1  # number of independent variables due to symmetry
    dx = 2.0 / n
    epsilon = 1e-6
    max_iterations = 50
    safety_margin = 5.0

    # Initialize variables
    w = Variable(d)
    h = np.zeros(n)
    h[:d] = w.value
    h[d:] = w.value[::-1]  # symmetry

    # Bounds, constraints
    constraints = [w >= 0, w <= 1, sum(w) == n/4 + 0.5]

    # LP master problem
    def solve_master(S, lambda_val):
        # Decision variables
        p = {}
        M = Variable()
        obj = Minimize(M - lambda_val * sum([w[i] * w[i] for i in range(d)]))

        # Constraints
        cons = constraints + [M >= n/2 - sum(p[(i, k)] for i in range(n) if i+k < n) for k in S]

        # McCormick envelopes
        for k in S:
            for i in range(n):
                if i + k < n:
                    p[(i, k)] = Variable()
                    cons += [
                        p[(i, k)] <= w[i],
                        p[(i, k)] <= w[i + k],
                        p[(i, k)] >= 0,
                        p[(i, k)] >= w[i] + w[i + k] - 1
                    ]

        # McCormick for k=0
        for i in range(n):
            p[(i, 0)] = Variable()
            cons += [
                p[(i, 0)] <= w[i],
                p[(i, 0)] <= w[i],
                p[(i, 0)] >= 0,
                p[(i, 0)] >= 2 * w[i] - 1
            ]

        prob = Problem(obj, cons)
        try:
            prob.solve(solver=ECOS, verbose=False)
            return prob.value, w.value, p
        except Exception:
            return float('inf'), None, None

    # Compute exact overlap
    def compute_overlap(h_vals):
        g = 1.0 - h_vals
        conv = np.correlate(h_vals, g, mode='full')
        return np.max(conv) * dx

    # Polishing with projected subgradient descent
    def polish_h(h_vals, max_iter=200):
        h = h_vals.copy()
        alpha = 0.02
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - safety_margin:
                break
            # Evaluate worst shift
            g = 1 - h  # compute using full convolution approach
            g_conv = np.correlate(h, g, mode='full')
            k_max = np.argmax(g_conv)
            # Get real overlap
            if k_max > n//2:
                k = k_max - n + 1
                g_vals = np.concatenate((np.zeros(k), h[:-k]))
            else:
                k = k_max
                g_vals = np.concatenate((h[k:], np.zeros(k)))
            # Compute gradient for this h wrt h[i]
            grad = np.zeros(n)
            # Derivative of h[i]*g_vals[i] w.r.t. h[i]
            for i in range(n):
                if i + k < n:
                    grad[i] += (1 - h[i + k])
                # From the other direction
                if i - k >= 0:
                    grad[i] += -h[i - k]
            # Projected step
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            scale = (n/2) / np.sum(h_new)
            h_new *= scale
            h = (h_new + h_new[::-1]) / 2  # enforce symmetry
            alpha *= 0.9
        return h

    # Initial shift set
    S = [0]
    lambda_val = 0.1
    best_c5 = float('inf')
    best_h = np.zeros(n)

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - safety_margin:
            break
        # Solve LP
        try:
            obj_val, w_vals, p_vars = solve_master(S, lambda_val)
            if w_vals is None:
                break
        except Exception:
            break
        # Reconstruct h vector
        h_full = np.zeros(n)
        h_full[:d] = w_vals
        h_full[d:] = w_vals[::-1]

        # Evaluate current overlap
        c5_eval = compute_overlap(h_full)
        if c5_eval < best_c5:
            best_c5 = c5_eval
            best_h = h_full.copy()
        
        # Check convergence
        if c5_eval <= obj_val + epsilon:
            break
        
        # Add new shift
        g = 1 - h_full
        conv = np.correlate(h_full, g, mode='full')
        k_max = np.argmax(conv)
        S.append(k_max)

        # Update lambda
        lambda_val = min(5.0, lambda_val * 1.5)

    # Apply polishing
    if time.time() - start_time < budget_s - 5.0:
        polished_h = polish_h(best_h, max_iter=200)
        polished_c5 = compute_overlap(polished_h)
        if polished_c5 < best_c5:
            best_c5 = polished_c5
            best_h = polished_h

    # Optional stochastic hill-climb
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10.0:
        p = min(6, n // 20)
        epsilon_hill = 0.003
        iter_max = 1000
        count = 0
        while count < iter_max and time.time() - start_time < budget_s - 5.0:
            indices = np.random.choice(n, size=p, replace=False)
            h_trial = best_h.copy()
            for i in indices:
                pert = np.random.uniform(-epsilon_hill, epsilon_hill)
                h_trial[i] = min(1.0, max(0.0, h_trial[i] + pert))
            # Ensure integral constraint
            sum_trial = np.sum(h_trial)
            scale_factor = (n / 2) / sum_trial
            h_trial *= scale_factor
            h_trial = (h_trial + h_trial[::-1]) / 2  # enforce symmetry
            c5_trial = compute_overlap(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
            count += 1

    return (best_h, best_c5, n)
