```python
import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Project h to the feasible set: h in [0,1] and sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    
    # Water-filling approach
    # Simple method: adjust by scaling + clipping
    try:
        if current_sum > 0:
            h = h * (n / 2) / current_sum
        h = np.clip(h, 0, 1)
        # Ensure exact sum
        diff = (n/2) - np.sum(h)
        if abs(diff) > 1e-10:
            h[0] += diff
            h = np.clip(h, 0, 1)
    except:
        h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Evaluate C5 for discrete function h."""
    dx = 2.0 / len(h)
    # Use FFT for fast full correlation
    h_fft = fft(h, 2*len(h))
    inv_fft = fft(1-h, 2*len(h))
    corr = ifft(h_fft * np.conj(inv_fft)).real
    # The full correlation has length 2*n - 1, centered at index n-1
    # Extract the valid overlap values (shifts from -n+1 to n-1)
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, k_star, n):
    """Compute gradient of C5 with respect to h for the k_star shift."""
    dx = 2.0 / n
    grad = np.zeros_like(h)
    # Compute derivative with respect to each h_i
    # d/dh_i of max_k int h(x)(1-h(x+k)) dx = dx * (1-h_{i+k*} - h_{i-k*})
    for i in range(n):
        # we know k* is negative, so i - k* > i, which means we look to the right
        # The actual indexing requires a mod operation or treating out of bounds
        # as 0 (zero-padding mode)
        left_idx = (i - k_star) % n  # for h_{i-k*}
        right_idx = (i + k_star) % n  # for h_{i+k*}
        
        grad[i] = dx * (1 - h[left_idx] - h[right_idx])
    return grad

def initialize_swarm(n, S):
    """Initialize S particles uniformly in feasible set."""
    swarm = []
    for _ in range(S):
        h = np.random.rand(n)
        h = project_to_feasible(h, n)
        swarm.append(h.copy())
    return swarm

def find_worst_shift(h):
    """Return the index of the worst shift that maximizes the overlap."""
    n = len(h)
    dx = 2.0 / n
    h_fft = fft(h, 2*n)
    inv_fft = fft(1-h, 2*n)
    corr = ifft(h_fft * np.conj(inv_fft)).real
    # Find the maximum except at k=0 (index n-1 in corr)
    corr = corr.copy()
    corr[n-1] = -np.inf  # ignore shift 0
    k_star = np.argmax(corr) - (n-1)  # Convert back to shift index
    return k_star

def memetic_refinement(h, n, max_steps=50):
    """Refine h using projected gradient descent on the worst shift."""
    h = h.copy()
    for _ in range(max_steps):
        k_star = find_worst_shift(h)
        # Compute gradient
        grad = compute_gradient(h, k_star, n)
        # Backtracking line search
        alpha = 1.0
        for _ in range(10):  # Max 10 halvings
            h_trial = h - alpha * grad
            h_trial = project_to_feasible(h_trial, n)
            if evaluate_c5(h_trial) < evaluate_c5(h) - 1e-12:
                h = h_trial
                break
            alpha /= 2.0
        if alpha < 1e-6:
            break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 300
    S = 30  # swarm size
    omega = 0.7
    c1 = c2 = 1.5
    n_points = n
    dx = 2.0 / n
    
    # Initialize swarm
    swarm = initialize_swarm(n, S)
    velocities = [np.zeros(n) for _ in range(S)]
    
    # Track best
    best_h = None
    best_c5 = float('inf')
    
    # PSO iterations (to allow memetic refines in between)
    max_pso_iter = 300
    last_improved_time = start_time
    
    # PSO main loop
    for iter_count in range(max_pso_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate particles and update best
        for i, h in enumerate(swarm):
            h = project_to_feasible(h, n)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            swarm[i] = h  # ensure feasible
            
        # Check for stagnation
        if time.time() - last_improved_time > 2.0:
            print(f"Stagnation detected at iter {iter_count}, injecting new particles.")
            # Inject new random particles
            for idx in np.random.choice(S, size=S//4, replace=False):
                h = np.random.rand(n)
                swarm[idx] = project_to_feasible(h, n)
                velocities[idx] = np.zeros(n)
            last_improved_time = time.time()
            
        # Update particles
        for i in range(S):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            pbest = swarm[i]
            gbest = best_h
            
            r1 = np.random.rand(n)
            r2 = np.random.rand(n)
            
            v = omega * v \
                + c1 * r1 * (pbest - h) \
                + c2 * r2 * (gbest - h)
                
            # Update position
            h_new = h + v
            h_new = project_to_feasible(h_new, n)
            swarm[i] = h_new
            velocities[i] = v
            
        # Inject memetic refinement every 5 iterations
        if iter_count % 5 == 0 and iter_count > 0:
            if best_h is not None:
                refined_best = memetic_refinement(best_h, n, 30)
                c5_refined = evaluate_c5(refined_best)
                if c5_refined < best_c5:
                    best_h = refined_best
                    best_c5 = c5_refined
                    
    # Final local hill-climb
    if best_h is not None:
        h_local = best_h.copy()
        current_c5 = evaluate_c5(h_local)
        
        for it in range(5000):
            if time.time() - start_time > budget_s - 2:
                break
            i, j = np.random.choice(n, size=2, replace=False)
            while h_local[i] <= 0:
                i = np.random.choice(n)
            while h_local[j] >= 1:
                j = np.random.choice(n)
            
            # Transfer small amount
            delta = np.random.uniform(0, min(h_local[i], 1 - h_local[j], 0.01))
            h_trial = h_local.copy()
            h_trial[i] -= delta
            h_trial[j] += delta
            h_trial = project_to_feasible(h_trial, n)
            trial_c5 = evaluate_c5(h_trial)
            
            if trial_c5 < current_c5:
                h_local = h_trial
                current_c5 = trial_c5
				
    # Final check before returning
    if best_h is None or current_c5 < best_c5:
        best_h = h_local
        best_c5 = current_c5

    return (best_h, best_c5, n_points)
```
