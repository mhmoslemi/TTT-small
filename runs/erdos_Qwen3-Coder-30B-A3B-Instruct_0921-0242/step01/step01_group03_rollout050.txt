def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def build_shift_matrices(n):
    """Build sparse shift matrices A_k for all k = 0, ..., n-1"""
    A = []
    for k in range(n):
        # Create shift matrix: (A_k)_{i,j} = 1 if j = i + k, else 0
        data = np.ones(n - k)
        row = np.arange(n - k)
        col = np.arange(k, n)
        A_k = csr_matrix((data, (row, col)), shape=(n, n))
        A.append(A_k)
    return A

def solve_sdp(n, A, max_time):
    """Solve the SDP relaxation of the problem"""
    start_time = time.time()
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.diag(X) == h,
        cp.sum(h) == n / 2,
        h >= 0,
        h <= 1,
        X >> 0,
    ]

    # Constraint for each shift
    for k in range(n):
        constraints.append(cp.trace(A[k] @ X) >= t)

    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    # Solve with SCS solver
    try:
        prob.solve(solver=cp.SCS, max_iters=15000, eps=1e-4, verbose=False)
    except Exception:
        return None, None

    end_time = time.time()
    if end_time - start_time > max_time * 0.8:
        return None, None

    return h.value, X.value

def eigen_rounding(X):
    """Principal eigenvalue rounding"""
   lam, v = np.linalg.eigh(X)
    idx = np.argsort(lam)[::-1]
    lam, v = lam[idx], v[:, idx]
    v1 = v[:, 0]
    h = np.sqrt(np.abs(lam[0])) * np.abs(v1)
    return np.clip(h, 0, 1)

def threshold_rounding(X):
    """Diagonal threshold rounding"""
    h = np.sqrt(np.diag(X))
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    h_bin[idx[:len(h)//2]] = 1.0
    return h_bin

def wrapped_shift(arr, k):
    """Wrap array shift for correlation computation"""
    n = len(arr)
    if k >= 0:
        return np.concatenate([arr[k:], arr[:k]])
    else:
        k = -k
        return np.concatenate([arr[-k:], arr[:-k]])

def compute_full_correlation(h):
    """Compute correlation for all shifts efficiently via FFT"""
    n = len(h)
    h_pad = np.pad(h, (0, n), mode='constant')
    g_pad = np.pad(1 - h, (0, n), mode='constant')
    corr = np.real(ifft(fft(h_pad) * np.conj(fft(g_pad))))
    return corr[n-1:]

def compute_correlation_at_shift(h, k):
    """Compute correlation at shift k"""
    n = len(h)
    h_shifted = wrapped_shift(h, k)
    return np.sum(h * (1 - h_shifted))

def compute_correlation_for_all_shifts(h):
    """Compute correlations for all shifts"""
    n = len(h)
    corrs = []
    for k in range(n):
        corr = compute_correlation_at_shift(h, k)
        corrs.append(corr)
    return np.array(corrs)

    start_time = time.time()

    # Use it more as a fast swap improvement, not embedding the worst-process
    iterations = 0

        iterations += 1
        # Randomly pick a 1 and a 0
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        
        if len(ones) < 1 or len(zeros) < 1:
            break

        p = np.random.choice(ones)
        q = np.random.choice(zeros)
        
        # Compute old correlation
        old_corr = compute_correlation_for_all_shifts(best_h)

        # Swap
        new_h = best_h.copy()
        new_h[p], new_h[q] = new_h[q], new_h[p]
        
        new_corr = compute_correlation_for_all_shifts(new_h)
        new_c5 = np.max(new_corr)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            
        # Occasionally recompute full spectrum to avoid drift
        if iterations % 1000 == 0:
            corr_all = compute_full_correlation(best_h)
            
    return best_h, best_c5

def bounded_simplex_projection(h, target_sum):
    """Project h to nearest point in simplex: sum = target and each h[i] in [0,1] """
    n = len(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    deficit = target_sum - current_sum
    if abs(deficit) < 1e-10:
        return h
    
    # Distribute deficit in a conservative way
    adjustment = deficit / n
    h += adjustment
    
    # Clamp all entries to [0,1]
    h = np.clip(h, 0, 1)
    
    # If we are still off, fix with more detailed projection
    # Union of projected points onto the simplex with some indices active
    # Quick and approximate estimation: re-projection using P^2 - simple O(n) routine
    
    # A simple iterative clipping can help:
    while abs(np.sum(h) - target_sum) > 1e-10:
        diff = np.sum(h) - target_sum
        if diff > 0:
            # Scale down
            mask = h > 0
            scaleby = min(1.0, -diff / np.sum(mask))
            h[mask] -= scaleby
        else:
            mask = h < 1
            scaleby = min(1.0, diff / np.sum(mask))
            h[mask] += scaleby
        
        h = np.clip(h, 0, 1)
    return h

def project_to_n_points(h, n_points):
    """Find nearest point to h that satisfies sum(h)*dx = n/2"""
    dx = 2.0 / len(h)

    # Correct sum to exactly n_points/2 (estimation quite stable)
    h_new = bounded_simplex_projection(h_new, n_points / 2)
    return h_new
