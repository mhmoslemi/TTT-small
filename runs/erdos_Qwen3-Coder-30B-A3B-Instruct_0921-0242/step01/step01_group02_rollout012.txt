import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.sparse import diags
from scipy.sparse.linalg import eigs
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Time budget allocation
    sdp_time = 300
    rounding_time = 150
    local_time = 130
    remaining_time = budget_s
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    grids = [(140, 70), (200, 100), (260, 130)]  # (n, m)
    
    # Helper functions
    def build_shift_matrix(n, k):
        """Build sparse shift matrix S_k for shift k."""
        diagonals = [1] * (n - abs(k))
        if k >= 0:
            offset = k
        else:
            offset = k
        return diags(diagonals, offset, shape=(n, n), format='csr')
    
    def solve_sdp(n, m, shift_indices):
        """Solve the SDP relaxation for size n, m."""
        X = cp.Variable((m, m), symmetric=True)
        t = cp.Variable(1)
        
        # Constraints
        constraints = [
            X >> 0,  # Positive semidefinite
            cp.diag(X) == 1,  # Diagonal ones
            cp.sum(X @ np.ones(m)) == 0,  # Zero sum condition
        ]
        
        # Add shift constraints
        for i, k in enumerate(shift_indices):
            S_k = build_shift_matrix(n, k)
            # Convert to CSR for efficiency
            if hasattr(S_k, 'toarray'):
                S_k_array = S_k.toarray()
            else:
                S_k_array = S_k
            constraints.append(t >= n/4.0 - 0.25 * cp.trace(S_k_array @ X))
        
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, eps=1e-6, max_iters=5000)
            return X.value, t.value
        except Exception:
            return None, None
    
    def round_sdp(X, m, n):
        """Round the SDP solution to binary vector."""
        # Use eigenvalue decomposition
        try:
            U, V = np.linalg.eigh(X)
            U = np.maximum(U, 0)
            U = np.sqrt(U)
            V = V * U
            Y = V.T
        
            # Get random Gaussian vector
            g = np.random.randn(m)
            y = Y @ g
            
            # Sign the values
            x = np.sign(y)
            
            # Enforce zero sum
            s = np.sum(x)
            if s != 0:
                indices = np.argsort(np.abs(y))[:int(abs(s)/2)]
                if s > 0:
                    x[indices] = -x[indices]
                else:
                    x[indices] = -x[indices]
            
            # Reconstruct full vector
            h_full = (x + 1) / 2
            if n % 2 == 1:
                h_full = np.concatenate([h_full, h_full[-2::-1]])
            else:
                h_full = np.concatenate([h_full, h_full[::-1]])
            
            # Normalize to correct sum
            h_full = h_full * (n / 2.0) / np.sum(h_full)
            
            return h_full
        except Exception:
            return None

    def compute_c5(h, dx):
        """Compute C5 for given h using full convolution."""
        g = 1 - h
        corr = correlate(h, g, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def compute_c5_exact(h, dx):
        """Compute exact C5 for given h."""
        g = 1 - h
        corr = correlate(h, g, mode='full')
        return np.max(corr) * dx

    def local_search(h, dx, max_steps=5000):
        """Perform local hill-climbing on binary vector."""
        corr = correlate(h, 1-h, mode='full')
        cmax = np.max(corr)
        n = len(h)
        
        for step in range(max_steps):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Pick two points to swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            
            # Make a copy and swap
            h_new = h.copy()
            h_new[i1] = 0
            h_new[i0] = 1
            
            # Update the correlation efficiently
            # Recompute correlation in the local region
            corr_new = corr.copy()
            
            # This approach is simplified, for more precise update,
            # we compute correlation in more localized regions
            dx = 2.0 / len(h)
            cmax_new = compute_c5_exact(h_new, dx)
            
            if cmax_new < cmax:
                h = h_new
                corr = correlate(h, 1-h, mode='full')
                cmax = np.max(corr)
                
        return h, cmax

    def project_to_constraints(h, n_points, target_sum):
        """Project h onto sum constraint."""
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-10:
            return h
        diff = target_sum - current_sum
        if abs(diff) < 1e-10:
            return h
        # Distribute proportionally
        weights = np.ones_like(h)
        weights[np.where(h <= 0)] = 0
        weights[np.where(h >= 1)] = 0
        if np.sum(weights) > 0:
            h += diff * weights / np.sum(weights)
            h = np.clip(h, 0, 1)
        return h

    for n, m in grids:
        if time.time() - start_time > remaining_time - 5:
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        grid_time = min(remaining_time, 200)
        
        # Full shift range for this size
        shift_range = int(0.9 * n)
        shift_indices = list(range(-shift_range, shift_range + 1))
        
        # Solve SDP
        X_opt, t_opt = solve_sdp(n, m, shift_indices)
        if X_opt is None or t_opt is None:
            continue
        
        # Round to get binary vector
        h_trial = round_sdp(X_opt, m, n)
        if h_trial is None:
            continue
        
        # Project to valid sum and box constraints
        target_sum = n / 2.0
        h_trial = project_to_constraints(h_trial, n, target_sum)
        h_trial = np.clip(h_trial, 0, 1)
        
        # Compute C5 for trial solution
        dx = 2.0 / n
        c5_trial = compute_c5(h_trial, dx)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            best_n = n
            
        # If we've found a good solution already, try local refinement to conserve time
        if best_c5 <= 0.3806:
            break
            
        # Local refinement
        try:
            h_refined, c5_refined = local_search(h_trial, dx, max_steps=2000 if grid_time > 50 else 1000)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined
                best_n = n
        except:
            pass
        
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final check to ensure best solution is within bounds
    if best_h is None:
        return (None, float('inf'), 0)
    
    # Ensure the result is valid (project)
    target_sum = best_n / 2.0 if best_n else 0
    if best_n:
        best_h = project_to_constraints(best_h, best_n, target_sum)
    else:
        best_n = 0
        
    final_c5 = compute_c5(best_h, 2.0 / best_n) if best_n > 0 else float('inf')
    
    return (best_h, final_c5, best_n)
