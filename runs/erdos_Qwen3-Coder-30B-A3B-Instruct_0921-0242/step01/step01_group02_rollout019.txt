import numpy as np
import time
from scipy.signal import fftconvolve
from scipy.optimize import minimize

def binarize_h(h, target_sum):
    """Convert a continuous h to binary with exact target sum."""
    h = np.clip(h, 0, 1)
    sorted_indices = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    h_bin[sorted_indices[:int(target_sum)]] = 1
    return h_bin

def compute_c5(h, dx):
    """Compute C5 using fast FFT convolution."""
    g = 1 - h
    corr = fftconvolve(h, g, mode='full')
    return np.max(corr) * dx

def update_convolution_delta(corr, h, i1, i0, old_h):
    """Fast incremental update of correlation for a swap."""
    n = len(h)
    
    # Compute shift indices
    shift1 = i1
    shift0 = i0
    
    # Create shifted versions of h and 1-h
    h_shift1 = np.roll(h, shift1)
    h_shift0 = np.roll(h, shift0)
    g_shift1 = np.roll(1 - h, shift1)
    g_shift0 = np.roll(1 - h, shift0)
    
    # Compute delta in cost
    delta = 0
    # Change for i1
    if old_h[i1] == 1:
        # Was 1, now 0
        delta -= np.sum(h_shift1)  # contribution from new_h[i1] = 0
        delta -= np.sum(g_shift1)  # contribution from 1 - new_h[i1] = 1
    else:
        # Was 0, now 1
        delta += np.sum(h_shift1)  # contribution from new_h[i1] = 1
        delta += np.sum(g_shift1)  # contribution from 1 - new_h[i1] = 0

    # Change for i0
    if old_h[i0] == 1:
        # Was 1, now 0
        delta -= np.sum(h_shift0)
        delta -= np.sum(g_shift0)
    else:
        # Was 0, now 1
        delta += np.sum(h_shift0)
        delta += np.sum(g_shift0)

    # Update corr by modifying regions
    new_corr = corr.copy()
    
    # Update corr[k] where k is the lag that changes
    # In full convolution mode, changes happen at appropriate lags
    # This is a simplified but still effective evaluation
    
    for k in range(-n + 1, n):
        new_corr[k + n - 1] += (h[i1] - old_h[i1]) * (1 - h[i1]) * np.sum(h[i1-k:i1-k+1]) + \
                               (h[i0] - old_h[i0]) * (1 - h[i0]) * np.sum(h[i0-k:i0-k+1])
    
    return new_corr

    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    n_list = [220, 260, 300]
    
    def evaluate_and_update(h, n, target_sum, dx):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
        c5 = compute_c5(h, dx)
        return h, c5
    
    # Evaluation routines
    def local_improvement(h, dx, n):
        """Use deterministic local search."""
        max_iters = 10000
        for _ in range(max_iters):
            improved = False
            for i in range(n):
                for j in range(i+1, n):
                    # Try swapping i and j
                    if h[i] == h[j]:
                        continue
                    new_h = h.copy()
                    new_h[i] = 1 - new_h[i]
                    new_h[j] = 1 - new_h[j]
                    
                    new_c5 = compute_c5(new_h, dx)
                    if new_c5 < compute_c5(h, dx):
                        h = new_h
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        return h, compute_c5(h, dx)
        
    def initialize_h(n, target_sum):
        h = np.random.rand(n)
        h = h * (target_sum / np.sum(h))
        return h
    
    # Main loop for different n
    for n in n_list:
        if time.time() - start_time > budget_s - 10:
            break
        dx = 2.0 / n
        target_sum = n / 2
        
        # Various starting designs
        starts = []
        
        # From initial_h_values
        try:
            x_orig = np.linspace(0, 2, len(initial_h_values))
            x_new = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_orig, initial_h_values)
            h_interp = h_interp * (target_sum / np.sum(h_interp))
            starts.append(h_interp)
        except:
            pass
        
        # Random
        h_random = np.random.rand(n)
        h_random = h_random * (target_sum / np.sum(h_random))
        starts.append(h_random)
        
        # Alternating
        h_alt = np.zeros(n)
        h_alt[::2] = 1
        h_alt = h_alt * (target_sum / np.sum(h_alt))
        starts.append(h_alt)
        
        # Two blocks
        h_blocks = np.zeros(n)
        h_blocks[:n//2] = 1
        np.random.shuffle(h_blocks)
        h_blocks = h_blocks * (target_sum / np.sum(h_blocks))
        starts.append(h_blocks)
        
        # Iterate over starting points
        for h_start in starts:
            h_local, c5_local = evaluate_and_update(h_start, n, target_sum, dx)
            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_local.copy()
                best_n = n
            # Differential local improvement
            h_local, c5_local = local_improvement(h_local, dx, n)

    if best_h is not None and time.time() - start_time < budget_s - 5:
