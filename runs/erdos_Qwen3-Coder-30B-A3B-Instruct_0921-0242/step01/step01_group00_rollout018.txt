import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def block_to_h(u, v, n_fine):
    """Convert block parameters to fine discretization h."""
    K = len(u)
    # Compute block lengths
    exp_u = np.exp(u)
    lengths = 2 * exp_u / np.sum(exp_u)
    # Compute block heights
    h_tilde = 1.0 / (1.0 + np.exp(-v))
    # Scale to satisfy integral constraint
    integral_tilde = np.sum(h_tilde * lengths)
    alpha = 1.0 / integral_tilde
    heights = np.minimum(1.0, alpha * h_tilde)
    # Adjust if any height hits 1
    if np.any(heights >= 1.0):
        # redistribute deficit
        integral_actual = np.sum(heights * lengths)
        residual = (1.0 - integral_actual) / np.sum(heights[heights < 1.0])
        # Adjust only the non-unit heights
        heights[heights < 1.0] += residual
        heights = np.minimum(1.0, heights)
        
    # Now build step function
    h_fine = np.zeros(n_fine)
    cumsum = np.concatenate([[0.0], np.cumsum(lengths)])
    dx_fine = 2.0 / n_fine
    for i in range(n_fine):
        x = i * dx_fine
        for k in range(K):
            if cumsum[k] <= x < cumsum[k+1]:
                h_fine[i] = heights[k]
                break
    return h_fine

def compute_autocorr(h, n_eval=200):
    """Compute full correlation."""
    dx = 2.0 / len(h)
    # Correlate h with (1-h) using full mode
    c = correlate(h, 1 - h, mode='full')
    # Pad to exact same length as correlation
    # compute correlation over shifts in [0, 2] at resolution dx
    shifts = np.arange(len(c)) * dx - 1
    mask = (shifts >= 0) & (shifts <= 2)
    values = c[mask] * dx
    if len(values) == 0:
        return 0
    return np.max(values)

def evaluate_block_params(u, v, n_fine=200):
    """Evaluate C5 for given block parameters."""
    h = block_to_h(u, v, n_fine)
    return compute_autocorr(h, n_fine)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Global hyperparams
    K = 6  # number of blocks
    popsize = 10 * K  # population size for DE
    
    # Bounds for DE: u and v are real-valued
    bounds = [(-5, 5)] * K + [(-5, 5)] * K
    
    def objective(x):
        u = x[:K]
        v = x[K:]
        return evaluate_block_params(u, v, 200)  # Evaluate with low resolution for speed
    
    # Time limit for global phase
    global_time_limit = 0.3 * budget_s
    best_c5 = float('inf')
    best_u = None
    best_v = None
    best_h = None

    # Run DE
    de_result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=1000,
        popsize=popsize,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False,
        atol=1e-6,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7
    )

    # Verify result
    u_opt = de_result.x[:K]
    v_opt = de_result.x[K:]
    best_c5 = de_result.fun
    best_u = u_opt
    best_v = v_opt

    # Second phase: Refine with fine-grid
    if best_u is not None:
        h_fine = block_to_h(best_u, best_v, 400)
        h_fine = np.clip(h_fine, 0, 1)
        dx = 2.0 / len(h_fine)
        
        # Projection function to ensure sum = 1
        def project_to_sum(h, target_sum=200.0):
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            diff = target_sum - current_sum
            h = h + diff / len(h)
            h = np.clip(h, 0, 1)
            return h
        
        # Ensure it satisfies sum(h) = 200
        h_fine = project_to_sum(h_fine, 200.0)
        
        # Local refinement
        max_iter = 5000
        for_iter = 0
        while for_iter < max_iter:
            if time.time() - start_time > budget_s - 0.5:
                break
            # Compute correlation
            c = correlate(h_fine, 1 - h_fine, mode='full')
            shifts = np.arange(len(c)) * dx - 1
            mask = (shifts >= 0) & (shifts <= 2)
            values = c[mask] * dx
            
            worst_k = np.argmax(values)
            max_val = np.max(values)

            # Compute gradient w.r.t worst shift
            grad = np.zeros_like(h_fine)
            k_idx = worst_k
            # For shift k_idx, dp/dh_i = (1 - h_{i+k_idx}) - h_{i-k_idx}
            for i in range(len(h_fine)):
                left_idx = (i - k_idx) % len(h_fine)
                right_idx = (i + k_idx) % len(h_fine)
                grad[i] = (1 - h_fine[left_idx]) - h_fine[right_idx]
            
            # Step size via backtracking line search
            alpha = 1.0
            alpha_max = 1.0
            while alpha > 1e-8:
                h_trial = h_fine - alpha * grad
                h_trial = project_to_sum(h_trial, 200.0)
                trial_c5 = compute_autocorr(h_trial, 800)
                if trial_c5 < max_val:
                    h_fine = h_trial
                    break
                alpha *= 0.5
                
            # Early stop if no improvement
            if alpha <= 1e-8:
                break
            for_iter += 1

        # Final refinement with hill-climbing
        max_hill_iter = 3000
        for_hill = 0
        while for_hill < max_hill_iter:
            if time.time() - start_time > budget_s - 0.5:
                break
            i1 = np.random.randint(0, len(h_fine))
            i2 = np.random.randint(0, len(h_fine))
            if h_fine[i1] > 0.0 and h_fine[i2] < 1.0:
                delta = min(h_fine[i1], 1.0 - h_fine[i2], 0.01)
                if delta > 0:
                    h_fine2 = h_fine.copy()
                    h_fine2[i1] -= delta
                    h_fine2[i2] += delta
                    h_fine2 = project_to_sum(h_fine2, 200.0)
                    c5_trial = compute_autocorr(h_fine2, 800)
                    c5_current = compute_autocorr(h_fine, 800)
                    if c5_trial < c5_current:
                        h_fine = h_fine2
            for_hill += 1

        # Final evaluation on finest resolution
        final_c5 = compute_autocorr(h_fine, 800)
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_fine
        else:
            best_h = h_fine
    
    # Return best solution
    if best_h is None:
        # Fallback
        best_h = np.ones(400) * 0.5
        best_c5 = compute_autocorr(best_h, 800)

    return (best_h, best_c5, len(best_h))
