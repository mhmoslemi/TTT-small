import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import math
from cvxpy import *

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_constraints(h, n_points, target_sum):
    """Project h onto sum constraint and box constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Distribute proportionally
    weights = np.ones_like(h)
    weights[np.where(h <= 0)] = 0
    weights[np.where(h >= 1)] = 0
    if np.sum(weights) > 0:
        h += diff * weights / np.sum(weights)
        h = np.clip(h, 0, 1)
    return h

                h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]

                        best_h = current_h.copy()

        h_initial = np.round(h_interp)
