import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.linalg import eigh
from scipy.sparse import diags

def build_shift_matrices(n, max_k):
    """Build sparse shift matrices S_k for k in [-max_k, max_k]."""
    S = []
    for k in range(-max_k, max_k + 1):
        off_diag = np.ones(n - abs(k))
        if k > 0:
            S_k = diags([off_diag], [k], shape=(n, n), format='csr')
        elif k < 0:
            S_k = diags([off_diag], [k], shape=(n, n), format='csr')
        else:
            S_k = diags([np.ones(n)], [0], shape=(n, n), format='csr')
        S.append(S_k)
    return S

def solve_sdp(n, S_k_list, symm=True):
    """Solve the SDP relaxation for the given shift matrices."""
    m = (n + 1) // 2 if symm else n
    X = cp.Variable((m, m), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == 1,  # diagonal ones
        cp.trace(X) == 0,  # zero sum approximation
        t >= n/4 - 0.25 * cp.trace(S_k_list[0] @ X) for S_k in S_k_list
    ]
    
    problem = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        problem.solve(solver=cp.SCS, eps=1e-6, max_iters=5000)
        return X.value, t.value
    except:
        return None, None

def gaussian_rounding(X_opt, m, n, symm=True):
    """Perform randomized Goemans-Williamson rounding with balancing."""
    U, V = eigh(X_opt)
    V = V[:, U > 1e-10]
    U = U[U > 1e-10]
    W = V * np.sqrt(U)
    
    best_x = None
    best_c5 = float('inf')
    
    for _ in range(200):
        g = np.random.randn(m)
        y = W.T @ g
        x = np.sign(y)
        
        # Ensure zero sum constraint
        s = np.sum(x)
        if s != 0:
            idx = np.argsort(np.abs(y))[:int(abs(s) / 2)]
            x[idx] = -x[idx]
        
        # Convert to h (0,1)
        h = (x + 1) / 2.
        
        if symm:
            h = np.concatenate([h, h[-2::-1]])
        else:
            h = h[:n]
        
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def local_search(h0, dx, n_points):
    """Perform hill-climbing local search."""
    h = h0.copy()
    c = np.correlate(h, 1 - h, mode='full')
    cmax = np.max(c)
    
    # Precompute shifts
    shifts = np.arange(-(n_points - 1), n_points)
    c = c[:2*n_points - 1]
    
    def update_corr(i0, i1):
        # Update correlation due to swapping h[i0], h[i1]
        old_c = np.copy(c)
        # find all k where k = i1 - i0 or k = i0 - i1 (assuming cyclic)
        for k in [i1 - i0, i0 - i1]:
            if 0 <= k < len(c):
                old_c[k] -= h[i0] * (1 - h[i1])
                old_c[k] += h[i1] * (1 - h[i0])
            if 0 <= -k < len(c):
                old_c[-k] -= h[i1] * (1 - h[i0])
                old_c[-k] += h[i0] * (1 - h[i1])
        return old_c
    
    for _ in range(5000):
        # Find a 0 and 1 bit
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        
        # Swap h[i0], h[i1] temporarily
        h[i0], h[i1] = h[i1], h[i0]
        # Recompute only affected correlations
        c_new = update_corr(i0, i1)
        cmax_new = np.max(c_new)
        
        if cmax_new < cmax:
            c = c_new
            cmax = cmax_new
        else:
            # Undo swap
            h[i0], h[i1] = h[i1], h[i0]
    
    return h, cmax

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Grid settings
    grids = [
        {"n": 140, "max_k": 126},
        {"n": 200, "max_k": 180},
        {"n": 260, "max_k": 234}
    ]
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for g in grids:
        if time.time() - start_time > budget_s - 30:
            break
            
        n = g["n"]
        max_k = g["max_k"]
        S_k_list = build_shift_matrices(n, max_k)
        
        # Solve SDP
        X_opt, t_opt = solve_sdp(n, S_k_list, symm=True)
        
        if X_opt is None or t_opt is None:
            continue
            
        # Rounding
        h_rounded, c5_rounded = gaussian_rounding(X_opt, (n+1)//2, n, symm=True)
        
        if c5_rounded >= best_c5:
            continue
            
        # Local refinement
        h_refined, c5_refined = local_search(h_rounded, 2.0/n, n)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
            best_n = n
            
        if best_c5 <= 0.3807:
            break
    
    if best_h is None:
        # Fallback to a simple construction
        n = 300
        dx = 2.0 / n
        h = np.zeros(n)
        h[:n//2] = 1
        c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
        return (h, c5, n)
    
    return (best_h, best_c5, best_n)
