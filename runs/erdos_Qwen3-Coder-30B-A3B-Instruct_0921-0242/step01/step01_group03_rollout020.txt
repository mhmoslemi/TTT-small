def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

        """Compute circular convolution (auto-correlation) of y with itself using FFT"""
        f = fft(y, fft_len)
        return (f * np.conj(f)).real[:N]
    
    def compute_correlations(Y):
        """Compute c_k(Y) for all k"""
        c = np.zeros(N)
        for d in range(r):
            c += convolve_fft(Y[:, d])
        return c
    
    def compute_lse_gradient(Y, c, tau):
        """Compute gradient of LSE smooth surrogate"""
        p = np.exp(-c / tau)
        p_sum = np.sum(p)
        p = p / p_sum if p_sum > 0 else np.ones_like(p) / N
        
        grad = np.zeros_like(Y)
        for d in range(r):
            y_d = Y[:, d]
            # convolve with weights (p) and reversed weights (p[::-1])
            p_fft = fft(p, fft_len)
            y_fft = fft(y_d, fft_len)
            # For left and right shifts
            left_shift = ifft(p_fft * y_fft).real[:N]
            right_shift = ifft(np.conj(p_fft * np.conj(y_fft))).real[:N]
            grad[:, d] = left_shift + right_shift
        
        return grad
    
    def project_onto_feasible(Y):
        """Project matrix Y onto feasible set"""
        h = np.sum(Y**2, axis=1)
        # Clip norms in range [0,1]
        norms = np.sqrt(h)
        safe_norms = np.clip(norms, 0, 1)

        Y[mask] = Y[mask] / norms[mask][:, np.newaxis]

        S = np.sum(h)
        if S > 0:
            alpha = np.sqrt((N / 2) / S)
            Y = Y * alpha
            # Re-clip norms after scaling
            norms_new = np.sqrt(np.sum(Y**2, axis=1))
            Y = Y * np.clip(1/norms_new, 0, 1)[:, np.newaxis]
            
        return Y
