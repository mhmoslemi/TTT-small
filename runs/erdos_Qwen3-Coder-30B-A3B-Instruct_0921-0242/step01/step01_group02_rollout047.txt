def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_full_correlation(h):
    """Evaluate full correlation to get all C(k) values."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    return corr

def project_to_constraints(h, n_points, target_sum):
    """Project h onto sum constraint and box constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Distribute proportionally
    weights = np.ones_like(h)
    weights[np.where(h <= 0)] = 0
    weights[np.where(h >= 1)] = 0
    if np.sum(weights) > 0:
        h += diff * weights / np.sum(weights)
        h = np.clip(h, 0, 1)
    return h

def simulated_annealing(n_points, dx, target_sum, max_iter=10000, start_temp=1.0, cooling_rate=0.995):
    h = np.random.randint(0, 2, n_points)
    current_sum = np.sum(h)
    while current_sum != target_sum:
        if current_sum < target_sum:
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                idx = np.random.choice(zeros)
                h[idx] = 1
        else:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                idx = np.random.choice(ones)
                h[idx] = 0
        current_sum = np.sum(h)
    
    current_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    temp = start_temp
    
    for i in range(max_iter):
        if temp < 1e-10:
            break
        # Try a random swap move
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            new_h = h.copy()
            new_h[i1] = 0
            new_h[i0] = 1
            
            new_c5 = c5_of_h(new_h, dx)
            delta = new_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
        temp *= cooling_rate
    
    return best_h, best_c5
