import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp
from scipy.signal import correlate
import math

def solve_spectral_lp(n):
    """Solve the spectral relaxation LP to maximize the minimum autocorrelation."""
    # Variables: p_m = |F(h)_m|^2, t (lower bound)
    # We use only n/2+1 unique values due to symmetry of real signal
    
    # Prepare the cosine matrix for autocorrelations
    # A_k = (1/n) * sum_m p_m * cos(2*pi*k*m/n)
    C = np.zeros((n-1, n))
    for k in range(1, n):
        for m in range(n):
            C[k-1, m] = np.cos(2 * np.pi * k * m / n)
    
    # LP setup
    # Objective: maximize t
    c = np.zeros(n + 1)
    c[-1] = 1  # maximize t
    
    # Equality constraint: sum p_m = n^2 / 2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = n * n / 2
    
    # Inequality constraints: (1/n) * sum_m p_m * cos(2*pi*k*m/n) >= t
    # => -sum_m p_m * cos(2*pi*k*m/n) + n * t <= 0
    A_ub = np.zeros((n - 1, n + 1))
    A_ub[:, :n] = -C
    A_ub[:, -1] = n  # t coefficient
    b_ub = np.zeros(n - 1)  # all >= 0
    
    # Bound constraints: 0 <= p_m <= n
    bounds = [(0, n) for _ in range(n)] + [(None, None)]  # t is unbounded above
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if result.success:
            t_opt = result.x[-1]
            p_opt = result.x[:-1]
            return p_opt, t_opt
        else:
            # fallback: try a simpler case
            p_flat = np.ones(n) * n/2
            t_flat = 0.0
            return p_flat, t_flat
    except:
        # fallback in case of solver issues
        p_flat = np.ones(n) * n/2
        t_flat = 0.0
        return p_flat, t_flat


def gerchberg_saxton(p, n, max_iter=200):
    """Apply Gerchberg-Saxton algorithm to recover a step function with the given power spectrum."""
    # Initial random phases
    phi = np.random.uniform(0, 2*np.pi, n)

    # Complex Fourier coefficients from the power spectrum
    Fh = np.sqrt(p) * np.exp(1j * phi)
    
    for _ in range(max_iter):
        # Inverse FFT to get time domain
        x = np.fft.irfft(Fh)
        
        # Phase-retrieval step
        # Clip values to box [0,1]
        y = np.clip(x, 0, 1)
        
        # Waterfilling to maintain sum = n/2
        s = np.sum(y)
        delta = n/2 - s
        if abs(delta) > 1e-10:
            # Linear search for the correct lambda to adjust
            # This is approximate: binary search for the right additive constant
            def adjust_lambda(lam):
                test = np.clip(y + lam, 0, 1)
                return np.sum(test) - n/2
            
            a, b = -10, 10
            for _ in range(50):
                mid = (a + b) / 2
                val = adjust_lambda(mid)
                if abs(val) < 1e-10:
                    break
                elif val > 0:
                    b = mid
                else:
                    a = mid
            lam = (a + b) / 2
            y = np.clip(y + lam, 0, 1)
            
        # Forward FFT
        Fy = np.fft.rfft(y)
        
        # Reconstruct with new phases, same magnitude
        if np.allclose(Fy, 0, atol=1e-15):
            # If all frequencies are zero, we use a simple binary vector for small changes
            Fh = np.fft.rfft(y)  # Keep it as identity
        else:
            magnitudes = np.abs(Fy)
            phases = np.angle(Fy)
            # Apply magnitudes to new phases
            Fh = magnitudes * np.exp(1j * phases)
            
    h = np.clip(np.fft.irfft(Fh), 0, 1)
    # Final waterfilling adjustment
    s = np.sum(h)
    delta = n/2 - s
    if abs(delta) > 1e-10:
        h = np.clip(h + delta/n, 0, 1)
    return h


def project_simplex_box(h, n):
    """Project h to the feasible region: sum(h) = n/2 and h in [0,1]"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    delta = n/2 - s
    if abs(delta) < 1e-10:
        return h
    
    # Add uniformly to all entries
    h = h + delta / n
    h = np.clip(h, 0, 1)
    return h


def binary_sharpen(h, n, mu=1e-2, max_iter=200):
    """Sharpen h toward a binary solution while maintaining feasibility"""
    def psi(h):
        # Penalty to push toward binary values
        return mu * np.sum(h * (1 - h))
    
    for _ in range(max_iter):
        # Compute gradient of sum over all terms h_i(1-h_i)
        grad = mu * (1 - 2*h)
        
        # Update in direction of -grad
        h_new = h - 0.01 * grad
        
        # Project
        h_new = np.clip(h_new, 0, 1)
        h_new = project_simplex_box(h_new, n)
        
        # Accept if improvement (or just move forward in theory)
        h = h_new
    
    # Final rounding
    h = np.where(h > 0.99, 1.0, h)
    h = np.where(h < 0.01, 0.0, h)
    h = project_simplex_box(h, n)
    return h


def evaluate_c5(h):
    """Evaluate the C5 bound for given h"""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr


def gradient_worst_shift(h, n, dx):
    """Perform gradient update on the worst shift"""
    # Get full correlation
    c = correlate(h, 1 - h, mode='full')
    # Find the worst shift (index of maximum in full correlation)
    max_k_idx = np.argmax(c)
    k_star = max_k_idx - n + 1  # shift key for the worst correlation
    
    # Compute Jacobian: d/dh_i of int h(x)(1 - h(x+k)) dx
    # = dx * (1 - h_{i+k*}) - dx * h_{i-k*}
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = dx * (1 - h[(i + k_star) % n]) - dx * h[(i - k_star) % n]
    
    # Gradient step
    alpha = 1.0
    for _ in range(10):  # bounded backtracking
        h_new = h - alpha * grad
        
        # Ensure bounds are preserved
        h_new = np.clip(h_new, 0, 1)
        
        # Waterfilling projection to maintain sum = n/2
        h_new = project_simplex_box(h_new, n)
        
        c_new = correlate(h_new, 1 - h_new, mode='full')
        if c_new[max_k_idx] < c[max_k_idx]:
            h = h_new
            break
        else:
            alpha *= 0.5
    return h


def swap_anneal(h, n, dx):
    """Simple swap heuristic with simulated annealing"""
    T = 1e-3
    for _ in range(min(10000, 100000)):  # many iterations but won't overload
        # Select two indices at random
        i, j = np.random.choice(n, size=2, replace=False)
        if h[i] <= 0 or h[j] >= 1:
            continue
        
        # Transfer a small amount
        delta = np.random.uniform(0, min(h[i], 1 - h[j], 0.02))
        
        h_old = h.copy()
        h[i] -= delta
        h[j] += delta
        
        # Project to feasible set
        h = project_simplex_box(h, n)
        
        # Evaluate
        c_old = evaluate_c5(h_old)
        c_new = evaluate_c5(h)
        
        if c_new < c_old:
            continue  # accept always
        else:
            # Accept with probability
            p_accept = math.exp((c_old - c_new) / T)
            if np.random.rand() > p_accept:
                h = h_old
        
        T *= 0.99995  # cooling schedule
        
    return h


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 650
    dx = 2.0 / n_points
    
    # Stage 1: Spectral convex relaxation
    if time.time() - start_time > budget_s - 10:
        h = np.ones(n_points) * 0.5
        c5 = evaluate_c5(h)
        return (h, c5, n_points)
        
    p_opt, t_opt = solve_spectral_lp(n_points)
    
    # Stage 2: Gerchberg-Saxton (phase retrieval) start
    h = gerchberg_saxton(p_opt, n_points)
    
    # Stage 3: Feasibility and binary sharpening
    h = project_simplex_box(h, n_points)
    h = binary_sharpen(h, n_points)
    
    c5 = evaluate_c5(h)
    
    # Stage 4: Targeted gradient update on worst shift
    for it in range(3000):  # allow up to 3000 iterations
        if time.time() - start_time > budget_s - 20:
            break
        h = gradient_worst_shift(h, n_points, dx)
        c5 = evaluate_c5(h)
    
    # Stage 5: Swap anneal
    h = swap_anneal(h, n_points, dx)
    c5 = evaluate_c5(h)
    
    return (h, c5, n_points)
