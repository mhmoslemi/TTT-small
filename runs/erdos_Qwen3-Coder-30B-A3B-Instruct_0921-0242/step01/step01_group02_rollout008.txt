def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_full_correlation(h, dx):
    """Evaluate all correlation values for bounds."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    return corr * dx

def binary_to_continuous_step_function(h_binary, n_points):
    """Convert binary vector to continuous step function."""
    # Find consecutive 1s
    steps = []
    i = 0
    while i < len(h_binary):
        if h_binary[i] == 1:
            start = i
            while i < len(h_binary) and h_binary[i] == 1:
                i += 1
            steps.append((start, i))
        else:
            i += 1
    return steps
