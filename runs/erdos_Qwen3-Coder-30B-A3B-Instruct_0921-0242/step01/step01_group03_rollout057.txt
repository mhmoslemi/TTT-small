def build_shift_matrices(N):
    """Build sparse CSR shift matrices A_k for k = 0, ..., N-1"""
    A_list = []
    for k in range(N):
        data = np.ones(N - k)
        row = np.arange(N - k)
        col = np.arange(k, N)
        A = sp.csr_matrix((data, (row, col)), shape=(N, N))
        A_list.append(A)
    return A_list

def bounded_simplex_projection(h, target_sum):
    """Project h onto the feasible set with sum = target_sum and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    deficit = target_sum - current_sum
    
    if abs(deficit) < 1e-10:
        return h
    
    # Distribute deficit proportionally
    if deficit > 0:
        # Want to increase values, so adjust interior points
        # Add deficit proportionally to points that can be increased
        to_increase = np.where(h < 1)[0]
        if len(to_increase) > 0:
            adjustment = deficit / len(to_increase)
            h[to_increase] += adjustment
            h = np.clip(h, 0, 1)
            # Correct any over-shoot
            h = bounded_simplex_projection(h, target_sum)
    else:
        # Deficit < 0: decrease values
        to_decrease = np.where(h > 0)[0]
        if len(to_decrease) > 0:
            adjustment = -deficit / len(to_decrease)
            h[to_decrease] -= adjustment
            h = np.clip(h, 0, 1)
            # Correct any over-shoot
            h = bounded_simplex_projection(h, target_sum)
            
    return h

def eigen_rounding(X, h0):
    """Rounding using leading eigenvector"""
    try:
        # Compute leading eigenpair
        w, v = sp.linalg.eigs(X, k=1, which='LM')
        lambda1 = np.real(w[0])
        v1 = np.real(v[:, 0])
        h_eig = np.sqrt(lambda1) * v1
        h_eig = np.abs(h_eig)
        # Project onto feasible set
        h_eig = bounded_simplex_projection(h_eig, len(h_eig) / 2)
        return h_eig
    except:
        return h0

def diagonal_threshold_rounding(X, h0):
    """Rounding via diagonal values"""
    diag = np.diag(X)
    idx_sorted = np.argsort(diag)[::-1]
    h_thresh = np.zeros_like(h0)
    h_thresh[idx_sorted[:len(h0)//2]] = 1.0
    return h_thresh

def compute_corr(h, A_list):
    """Compute correlation values efficiently"""
    dx = 2.0 / len(h)
    corr_vals = []
    for A in A_list:
        corr = np.trace(A @ (h.reshape(-1,1) @ h.reshape(1,-1)))
        corr_vals.append(corr)
    return np.array(corr_vals) * dx

def compute_c5_exact(h):
    """Compute exact C5 using full correlation"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def swap_update_correlation(h, A_list, corr_vals, p, q):
    """Incrementally update correlation values after swapping h[p] and h[q]"""
    new_corr = corr_vals.copy()
    # Correlation change due to swap
    for k in range(len(A_list)):
        # Old terms: h[p]h[p+k] + h[q]h[q+k] 
        # New terms: h[q]h[q+k] + h[p]h[p+k] if both indices present
        # Effect on k-th shift
        cor_old = 0
        cor_new = 0
        A = A_list[k]
        
        # That swaps only affect shifts where elements are actually present
        if p + k < len(h):
            cor_old += h[p] * h[p + k]
            cor_new += h[q] * h[p + k]
        if q + k < len(h):
            cor_old += h[q] * h[q + k]
            cor_new += h[p] * h[q + k]
        
        # Update new correlation for shift k
        new_corr[k] = corr_vals[k] - cor_old + cor_new
    
    return new_corr

def swap_search(h0, A_list, corr_vals, max_time_s):
    """Hill-climbing swap search to improve binary mask"""
    start_time = time.time()
    h = h0.copy()
    corr_vals = corr_vals.copy()
    best_c5 = compute_c5_exact(h)
    best_h = h.copy()
    
    iter_count = 0
    while time.time() - start_time < max_time_s - 5:
        p = np.random.choice(np.where(h == 1)[0])
        q = np.random.choice(np.where(h == 0)[0])
        
        # Compute new correlation values after swap
        new_corr = swap_update_correlation(h, A_list, corr_vals, p, q)
        # Estimate new C5 using min of correlations
        new_c5 = max(0.5 - np.min(new_corr) / 2.0, 0)  # Just approximation
        # Actually compute exact value for comparison
        h_test = h.copy()
        h_test[p], h_test[q] = h_test[q], h_test[p]
        new_c5_exact = compute_c5_exact(h_test)
        
        if new_c5_exact < best_c5:
            h = h_test
            best_c5 = new_c5_exact
            corr_vals = new_corr
            best_h = h.copy()
            if best_c5 < 0.3795:  # Early termination
                break
        iter_count += 1
    
    return best_h, best_c5

def solve_sdp(N, A_list):
    """Solve the SDP relaxation problem"""
    # Setup variables
    h = cp.Variable(N)
    X = cp.Variable((N, N), symmetric=True)
    t = cp.Variable()
    
    # Build constraints
    constraints = [
        cp.diag(X) == h,  # Consistency constraint
        cp.sum(h) == N/2,  # Sum constraint
        h >= 0, h <= 1,  # Box constraints
        X >> 0,  # PSD constraint
    ]
    
    # Add correlation constraints
    for A in A_list:
        constraints.append(cp.trace(A @ X) >= t)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.SCS, max_iters=15000, eps=1e-4)
        return h.value, X.value, t.value
    except:
        return None, None, None

def upsample_h(h_in, factor=2):
    """Upsample h by linear interpolation"""
    N_in = len(h_in)
    N_out = N_in * factor
    x_in = np.linspace(0, 2, N_in)
    x_out = np.linspace(0, 2, N_out)
    h_out = np.interp(x_out, x_in, h_in)
    
    # Renormalize
    h_out = h_out * (N_out / 2) / np.sum(h_out)
    return h_out
