def build_shift_matrices(N):
    """Build sparse CSR matrices A_k for each shift k = 0,...,N-1"""
    A_list = []
    for k in range(N):
        row = np.arange(N - k)
        col = np.arange(k, N)
        data = np.ones(N - k)
        A = csr_matrix((data, (row, col)), shape=(N, N))
        A_list.append(A)
    return A_list

def project_to_feasible(h):
    """Ensure sum(h) = N/2 and h in [0,1]^N"""
    N = len(h)
    dx = 2.0 / N
    sum_h = np.sum(h)
    deficit = N/2 - sum_h
    if abs(deficit) < 1e-10:
        return h

    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = N/2 - sum_h
    if abs(deficit) < 1e-10:
        return h

    # Distribute deficit uniformly
    adjustment = deficit / N
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def c5_of_h(h):
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def solve_sdp(N, A_k_list):
    """Solve the SDP relaxation for the Erdos problem"""
    # Variables
    h = cp.Variable(N)
    X = cp.Variable((N, N), symmetric=True)
    t = cp.Variable()

    constraints = [
        cp.diag(X) == h,  # Diagonal entry constraint
        cp.sum(h) == N/2,  # Total mass
        h >= 0, h <= 1,  # Bounds
        X >> 0,  # Positive semi-definite
    ]

    # Add constraints for all shifts
    for A_k in A_k_list:
        constraints.append(cp.trace(A_k @ X) >= t)

    # Minimize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    # Solve (try several solvers)
    try:
        prob.solve(solver=cp.SCS, max_iters=15000, eps=1e-4)
    except:
        # if SCS fails, try a coarser version
        # For fallback, try lbfgs if available or reduce complexity
        return None, None

    if prob.status != cp.OPTIMAL:
        return None, None

    return h.value, X.value

def eigen_rounding(X, N):
    """Principal eigenvalue rounding of the SDP solution"""
    try:
        # Compute leading eigenvalue and corresponding eigenvector
        eigvals, eigvecs = eigs(X, k=1, which='LM')
        v1 = eigvecs[:, 0].real
        lambda1 = np.real(eigvals[0])

        if lambda1 <= 0:
            return None

        h = np.sqrt(lambda1) * np.abs(v1)
        h = project_to_feasible(h)
        return h
    except:
        return None

def diagonal_rounding(X, N):
    """Threshold rounding based on diagonal entries"""
    diag = np.diag(X)
    idxs = np.argsort(diag)[::-1]
    h = np.zeros(N)
    h[idxs[:N//2]] = 1.0
    return h

def upsample(h_low, N_high):
    """Linearly upsample h_low to N_high points"""
    N_low = len(h_low)
    h_high = np.zeros(N_high)
    for i in range(N_high):
        idx0 = i // 2
        idx1 = (i + 1) // 2
        alpha = (i + 1) / 2 - idx0
        h_high[i] = (1 - alpha) * h_low[idx0] + alpha * h_low[idx1]
    
    # Renormalize to sum = N_high / 2
    h_high = h_high * (N_high / 2) / np.sum(h_high)
    return h_high
