import numpy as np
import time
from scipy.linalg import eigh
from scipy.signal import correlate
from cvxpy import*

def solve_sdp(n, K):
    # Create the shift matrices S_k
    S_k = []
    for k in K:
        S = np.zeros((n, n))
        for i in range(n):
            j = (i + k) % n
            if 0 <= j < n:
                S[i, j] = 1
        S_k.append(S)
    
    X = Semidef(n)
    t = Variable()
    
    # Constraints
    constraints = [X[i, i] == 1 for i in range(n)]
    constraints.append(sum(X[i, i] for i in range(n)) == 0)  # May not be correct, will adjust
    # Another way to encode sum(x_i) = 0:
    # Let e = np.ones(n), then e^T X e = 0 (this enforces the zero sum)
    constraints.append( sum(X[i, j] for i in range(n) for j in range(n)) == 0)
    # Actually, we want to truncate the diagonal to 1 and then construct a sum constraint that nets 0
    # Fixing xi = 2hi - 1, sum xi = 0 => sum (2hi - 1) = 0 => 2*Sum hi - n = 0 => sum hi = n/2
    # But this is the constraint we enforce to the original function.
    # Therefore, let's just use a vector constraint to do the enforcement of tr(e e^T X) = 0.
    ones = np.ones((n, 1))
    ones_vec = ones.flatten()
    # X is n x n, 11^T will be n x n, tr is a sum
    constraints.append(trace(ones @ ones.T @ X) == 0)
    
    for S in S_k:
        constraints.append(t >= (n/4) - (0.25 * trace(S @ X)))
    
    # Minimize t
    prob = Problem(Minimize(t), constraints)
    
    try:
        prob.solve(solver=SCS, eps=1e-6, max_iters=5000)
        return prob.value, X.value
    except:
        # Fallback method if SCS fails
        try:
            prob.solve(solver=ECOS, verbose=False)
            return prob.value, X.value
        except:
            # Silently fall back to spectral: use fixed point/rank 1 relaxation
            return None, None

def round_sdp(X, n):
    # Return a binary vector from the SDP matrix X
    # Using Goemans-Williamson rounding with Gaussian projection
    # X is approximately n x n matrix from SDP
    
    # Cholesky-like decomposition
    eigenvals, V = eigh(X)
    V = V[:, eigenvals > 1e-10]
    eigenvals = eigenvals[eigenvals > 1e-10]
    # use only positive eigenvalues and corresponding vectors
    sqrt_eigenvals = np.sqrt(eigenvals)
    
    # V * sqrt_eigenvals is the factorization needed for rounding:
    # We sample Gaussian vector g and generate y = V^T * g
    M = V.T * sqrt_eigenvals
    num_samples = 200
    best_h = None
    best_val = float('inf')
    
    for _ in range(num_samples):
        g = np.random.randn(n)
        y = M @ g  # project to the factor space
        
        x = np.sign(y)
        # enforce zero sum: change signs of the smallest absolute elements
        s = np.sum(x)
        if abs(s) > 0:
            m = int(abs(s)/2)
            idx_to_flip = np.argsort(np.abs(y))[:m]
            x[idx_to_flip] = -x[idx_to_flip]
        # Check if sum is correct (should be 0 +/- 1 in case of even number of elements)
        s = int(np.sum(x))
        # Now check feasibility  
        if s % 2 != 0: # Should not happen in exact rounding
            continue
            
        # Map back to h (0,1):  h = (x + 1)/2
        h = (x + 1) / 2

        # now compute cost
        dx = 2.0 / n
        corr = correlate(h, 1 - h, mode='full')
        cost = np.max(corr) * dx
        
        if cost < best_val:
            best_val = cost
            best_h = h.copy()
    
    return best_h, best_val

def refine_h(h, n):
    # Use in-place updates to improve over a local search
    dx = 2.0 / n
    # Compute initial correlation
    c = correlate(h, 1 - h, mode='full')
    cmax = np.max(c)
    
    max_swaps = 3000
    idle_count = 0
    limit_idle = 100
    
    for _ in range(max_swaps):
        i1 = np.random.choice(np.where(h == 1)[0])
        i0 = np.random.choice(np.where(h == 0)[0])
        
        # Simulate the effect of a swap on the correlation
        new_h = h.copy()
        new_h[i1] = 0
        new_h[i0] = 1
        
        # Update correlation
        # We do this efficiently by subtracting the old contributions and adding new ones
        delta = np.zeros_like(c)
        # Subtraction of old correlations
        # We need to find what positions in the convolution contain these
        # Time indices affected: i0 and i1 with k = 0 to n-1
        
        # We can now preprocess this more carefully:
        # Advantage: nothing special about forward-backward order, but we want
        # the change for what was i1 and i0 in their respective correlations
        # Compute the change in the correlation vector
        
        # This part is slow but fine; since we move one element, approximately 2*(2n-1) terms change
        # That's manageable.
        
        # Get the original current correlation array values (this might be outdated in loop but update once per swap)
        
        # More efficient way of doing the local update:
        # Since h (to) is set and we have the full correlation computed once, changes are simple updates:
        
        # Change for moving 1 from i1 to i0
        # Each new correlation includes new conv. terms at i1+k and (i0+k)
        # Those appear in correlation array at positions with index = i1+i2 - i2 = i1-k when i2 = k, 
        # But correlate returns index = i1+ k = i1 + (zero index of internal shift argument) = k + i1
        
        # Actually simpler way:
        old_coss = c.copy()
        # Calculate new c
        # via: h_new = h with i1 -> 0 and i0 -> 1
        # So we compute new convolution (quickly) by touching only relevant terms of 1-h (new) 
        # Let s1, s0 be locations considered for swap:
        # For correlation we want to compute sum over i of h[i] * h[i+k] for shifted by k
        
        # Now handle that by updating convolution values related to i1 or i0
        
        # Precompute other terms: seems tricky of we don't maintain the correlation array like in a contiguous addition algorithm.
        
        # Do this properly:
        # For every shift k, contributions of h[i] * (1 - h[i+k]) change for i = s1, s0  
        # Set old contribution to zero, new ones are to occur:
        # Negate contribution current:
        # In convolution, i corresponds to j = i 
        # C_{i,k} = h[i]*h[i+k]
        # We alter contribution of h[s1]*h[s1+k] and h[s0]*h[s0+k], however it's shifted
        # i1 being i -> never appears; 
        # i0 being i -> appears in convolution corr with other positions i_walk
        # i1 being i_walk -> appears in corr with i_start = i1
        # But there's no i because h[s1] = 0 after != 1 for h_new
        
        # Direct method computing correlation update efficiently
        # New: h_new[s1]=0, h_new[s0]=1
        # Old:
        # Correlation of original is sum over i of h[i]*(1-h[i+k]), i.e., i+k = j
        # So correlation at k = \sum_j h[j] (1-h[j-k])
        c_new = np.zeros_like(c)
        # Terms where
        #   h[j] * (1-h[j-k]) take new values for j = i1, i0:
        #   h[i1] *(1-h[i1-k]) is zero, and becomes h[i1] *(1-h[i1-k]) -> 0
        #   h[i0]* (1-h[i0-k]) is zero, and becomes 1*(1-h[i0-k]) = 1-h[i0-k]
        # Changed terms: old carries h[i]h[i-k], new carries (1-h[i-k]) where g is r'd used for h or not, not h.
        # OK, let's just compute c cleanly and reuse it for minor update:
        
        c_new = np.zeros(2*n - 1)
        for k in range(-n+1, n):
            v = 0
            for i in range(n):
                if 0 <= i + k < n:
                    if abs(k) != 0 or True:
                        # If we're updating pairs many zeros make it faster
                        # Easier to just recompute part:
                        # But for performance, with marking mismatch?
                        # Raw equivalen
                        v += h[i] * (1 - h[i+k])
            c_new[k + n - 1] = v
        
        # Or better yet: just take known diff:
        corr_old = c
        corr_new = c_new
        
        delta = corr_new - corr_old
        new_cmax = np.max(corr_new)
        
        # Accept new state
        if new_cmax < cmax:
            h[:] = new_h
            c[:] = corr_new
            cmax = new_cmax
            idle_count = 0
        else:
            idle_count += 1
            if idle_count > limit_idle:
                break
    return h, cmax

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None

    grid_sizes = [(140, 70), (200, 100), (260, 130)]
    remaining_budget = budget_s
    fixed_coarse_bound = 0.3807

    for n, m in grid_sizes:
        if remaining_budget < 30:
            break
            
        # Solve SDP relaxation
        # For symmetry induced matrix, we just reduce dimension
        # We now opt to solve compact version: symmetrized X matrix in  m x m format
        # We keep only first half of columns
        possible_ranges = range(-n+1, n)
        # Bias range to cover more extremes with less truncation because n grows
        # typical covering bests k ~ n//2
        K = [k for k in possible_ranges if abs(k) <= min(n//2, int(0.9*n))]

        t_opt, X_opt = solve_sdp(n, K)
        if t_opt is None or X_opt is None:
            continue
        
        # Apply rounding to get discrete solution
        h_before_refinement, c5_rounded = round_sdp(X_opt, n)
        if h_before_refinement is None:
            continue
            
        # Refine
        h_refined, c5_final = refine_h(h_before_refinement.copy(), n)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_refined.copy()
            best_n = n
            
        # Prune by achieving target
        if best_c5 <= fixed_coarse_bound:
            break    
        
        # Update timing
        elapsed = time.time() - start_time
        remaining_budget = budget_s - elapsed

    # Final check on the best result
    if best_h is not None:
        dx = 2.0 / best_n
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        return (best_h, final_c5, best_n)
    else:
        return (np.array([0.5] * 200), 0.5, 200)
