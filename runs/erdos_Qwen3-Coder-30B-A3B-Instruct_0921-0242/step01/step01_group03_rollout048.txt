import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.interpolate import interp1d

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Choose resolution
    N = 140  # Slightly smaller than previous to keep MILP lightweight
    while True:
        try:
            # Build the binary integer program (BIP)
            x = cp.Variable(N, boolean=True)
            t = cp.Variable()
            # Symmetry-breaking constraint
            constraints = [cp.sum(x) == N // 2, x[0] == 1, t >= 0]
            
            # Define auxiliary variables for AND terms
            # w[i, k] = x[i] AND x[i+k] 
            w_vars = {}
            w_cons = []
            for k in range(N):
                for i in range(N - k):
                    w = cp.Variable(boolean=True)
                    w_vars[(i,k)] = w
                    # McCormick inequalities: w <= x[i], w <= x[i+k], w >= x[i] + x[i+k] - 1
                    w_cons.append(w <= x[i])
                    w_cons.append(w <= x[i+k])
                    w_cons.append(w >= x[i] + x[i+k] - 1)
            constraints += w_cons

            # Overlap bounds for all shifts k
            for k in range(N):
                w_sum = cp.sum([w_vars[(i,k)] for i in range(N-k)])
                constraints.append(w_sum >= N//2 - t)
            
            # Binary search on t
            L = 0.0
            U = N // 2
            best_T = U
            best_x = None
            
            while U - L > 1e-4 and time.time() - start_time < budget_s - 10:
                T = (L + U) / 2
                # Temporarily change constraint to feasibility check
                old_constraints = constraints.copy()
                constraints.append(t <= T)
                prob = cp.Problem(cp.Minimize(0), constraints)
                result = prob.solve(solver=cp.CBC, warm_start=True, 
                                    solver_opts={'max_seconds': 30})
                if result != np.inf:
                    U = T
                    best_T = T
                    # Save the best feasible solution
                    if best_x is None or hasattr(x, 'value') and x.value is not None:
                        # This handles potential cvxpy v0.4.0+ style mgmt
                        try:
                            if hasattr(x, 'value') and x.value is not None:
                                best_x = np.array(x.value, dtype=int)
                            else: 
                                # fallback
                                best_x = np.floor(np.random.rand(N) > 0.5).astype(int)
                        except:
                            # fallback; in practice converges quickly
                            best_x = np.zeros(N, dtype=int)
                            best_x[:N//2] = 1
                            np.random.shuffle(best_x)
                else:
                    L = T
                # Clean constraints for next iteration
                constraints = old_constraints[:-1] # Remove the last t <= T constraint
            
            # If we got a result via binary search, use it
            if best_x is None and time.time() - start_time < budget_s - 20:
                # Try full optimization once cruder binary search didn't work robustly
                # Rebuild from scratch
                x = cp.Variable(N, boolean=True)
                t = cp.Variable()
                constraints = [cp.sum(x) == N // 2, x[0] == 1, t >= 0]
                
                # Rebuild constraints
                w_vars = {}
                w_cons = []
                for k in range(N):
                    for i in range(N - k):
                        w = cp.Variable(boolean=True)
                        w_vars[(i,k)] = w
                        w_cons.append(w <= x[i])
                        w_cons.append(w <= x[i+k])
                        w_cons.append(w >= x[i] + x[i+k] - 1)
                constraints += w_cons
                
                for k in range(N):
                    w_sum = cp.sum([w_vars[(i,k)] for i in range(N-k)])
                    constraints.append(w_sum >= N//2 - t)
                
                prob = cp.Problem(cp.Minimize(t), constraints)
                result = prob.solve(solver=cp.CBC, solver_opts={'max_seconds': 120})
                if result != np.inf:
                    try:
                        if hasattr(x, 'value') and x.value is not None:
                            best_x = np.array(x.value, dtype=int)
                        else:
                            best_x = np.ones(N, dtype=int)
                            best_x[N//2:] = 0
                    except:
                        best_x = np.ones(N, dtype=int)
                        best_x[N//2:] = 0
            elif best_x is None and time.time() - start_time < budget_s - 5:
                # fallback to greedy for binary, easy to do
                best_x = np.zeros(N, dtype=int)
                best_x[:N//2] = 1
                np.random.shuffle(best_x)
            
            # Convert back to step function h
            h = np.array(best_x, dtype=float)
            if N < 140:
                # interpolating to full N if needed
                h_large = np.interp(np.linspace(0, 2, 140), np.linspace(0, 2, N), h)
                h = h_large
            h_final = h.copy()
            c5_best = c5_of_h(h_final)
            
            # Phase 2: Local polishing
            # Fractional polishing (refine maybe a small amount)
            for _ in range(100):  # Limited iterations to avoid too much time
                if time.time() - start_time > budget_s - 10:
                    break
                # Find worst shift
                dx = 2.0 / len(h_final)
                g = 1.0 - h_final
                corr = np.correlate(h_final, g, mode='full') * dx
                max_corr_idx = np.argmax(corr)
                k = max_corr_idx - (len(h_final) - 1)  # k ranges from -(N-1) to N-1
                
                # Iterate pairs that could improve the worst case
                improve_positions = []
                for i in range(len(h_final)):
                    if i + k < len(h_final):
                        if h_final[i] > 0 and h_final[i+k] < 1:
                            # compute improvement
                            # currently h[i](1-h[i+k])
                            # change to (h[i]-delta)(1-h[i+k]+delta) = h[i]-delta - h[i]*h[i+k] + delta*h[i] + delta - delta**2
                            temp = h_final[i] - h_final[i+k] - h_final[i] * h_final[i+k] + h_final[i] - 2*0.01 + 0.01**2
                            improve_positions.append((i, 0.01))
                
                if not improve_positions:
                    break
                
                # choose a small perturbation that improves (if possible)
                for i, delta in improve_positions:
                    if time.time() - start_time > budget_s - 10:
                        break
                    # update delta to avoid numerical issues
                    h_new = h_final.copy()
                    if h_new[i] - delta >= 0 and h_new[i+k] + delta <= 1:
                        h_new[i] -= delta
                        h_new[i+k] += delta
                        
                        # Clamp to [0,1]
                        h_new = np.clip(h_new, 0, 1)
                        c5_new = c5_of_h(h_new)
                        if c5_new < c5_best:
                            h_final = h_new
                            c5_best = c5_new
                            break  # Accept and restart
            
            # Phase 3: Swap hill-climbing
            max_swaps = 30000
            swaps_done = 0
            last_improvement = 0
            while swaps_done < max_swaps and time.time() - start_time < budget_s - 10:
                if time.time() - start_time > budget_s - 10:
                    break
                # Pick two positions: one 1 and one 0
                ones = np.where(h_final > 0.9)[0]
                zeros = np.where(h_final < 0.1)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i1 = np.random.choice(ones)
                i0 = np.random.choice(zeros)
                
                # try swapping
                h_new = h_final.copy()
                h_new[i1] = 0
                h_new[i0] = 1
                # check constraint (should remain valid after swap)
                c5_new = c5_of_h(h_new)
                if c5_new < c5_best:
                    h_final = h_new
                    c5_best = c5_new
                    last_improvement = 0
                else:
                    # Occasionally stay in some non-improving yet still close swaps
                    last_improvement += 1
                    if last_improvement > 5000:
                        break  # stop after too many no-improvements
                swaps_done += 1
            
            # Optional:Upsample for very final improvement
            if N < 140:
                # interpolate to larger size if needed (but not in bin-search step)
                h_highres = np.interp(np.linspace(0, 2, 140), np.linspace(0, 2, N), h_final)
                h_highres = h_highres * (140 / 2) / np.sum(h_highres) 
                for iter in range(10):
                    h_new = h_highres.copy()
                    # Just make small non-zero differences to estimate improvement
                    if time.time() - start_time > budget_s - 5:
                        break
                    try:
                        c5_mod = c5_of_h(h_new)
                        if c5_mod < c5_best:
                            h_final = h_new
                            c5_best = c5_mod
                    except:
                        pass
                h_final = h_highres
            
            return (h_final, c5_best, len(h_final))
        except Exception:
            # reduce N if problems happen with cutoff
            N //= 2
            if N < 20:
                break
            continue

    # fallback if everything says no
    h_final = np.zeros(N)
    h_final[:N//2] = 1.
    return (h_final, c5_of_h(h_final), N)
