import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp
from scipy.optimize import minimize

def proj_simplex_box(h, n):
    """Project h onto {0 <= h_i <= 1, sum h_i = n/2} using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    
    # Use bisection to find lambda such that sum(clipped(h + lambda)) = n/2
    def sum_clipped(lam):
        h_lam = h + lam
        h_lam = np.clip(h_lam, 0, 1)
        return np.sum(h_lam)
        
    lo, hi = -1.0, 1.0
    while hi - lo > 1e-8:
        mid = (lo + hi) / 2
        s = sum_clipped(mid)
        if s > n/2:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2
    h = np.clip(h + lam, 0, 1)
    return h

def c5_of_h(h):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx using fast correlation."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def c5_of_h_vec(h):
    """Return full vector of overlap integrals for all shifts."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return corr * dx

def worst_shift(h):
    """Return shift index that causes maximum overlap (excluding k=0)."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    # Exclude first element (k=0) which is fixed
    corr = corr[1:]
    k_idx = np.argmax(corr)
    return k_idx + 1  # Return k in [1,2*n-1]

def compute_gradient_shift(h, k):
    """Return gradient of I_k(h) = int h(x)(1-h(x+k)) dx with respect to h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    
    # I_k = sum h_i * (1 - h_{i+k}) = sum h_i - sum h_i * h_{i+k}
    # Gradient w.r.t. h_i is: 1 - h_{i+k} (or dx * (1 - 2 * h_{i+k}) if we do not factor out dx)
    # But since the integral is discretized (you don't sum -> you average over all shifts)
    # The gradient of I_k is: dx * (1 - 2 * h_{i+k}) if we're at index i
    
    for i in range(n):
        j = (i + k) % n
        grad[i] = dx * (1 - 2 * h[j])
        
    return grad

def gradient_step(h, k, alpha):
    """One step of gradient ascent for the k-th overlap integral."""
    grad = compute_gradient_shift(h, k)
    h_new = h + alpha * grad
    return proj_simplex_box(h_new, len(h))

def c5_lp_relaxation(h_start, n):
    """Solve convex relaxation using CVXPY to get lower bound on max overlap."""
    try:
        h = cp.Variable(n)
        t = cp.Variable(1)
        
        # Build quadratic constraints via McCormick envelopes for h[i]*h[i+k]
        constraints = [
            cp.sum(h) == n/2,
            h >= 0,
            h <= 1,
        ]
        for k in range(1, n):
            # Introduce variables for the product h[i]*h[i+k]
            # Use McCormick envelopes for h[i] * h[i+k]
            yk = cp.Variable(1)
            # Constraint for pairwise McCormick: z >= h[i] + h[i+k] - 1, z <= h[i], z <= h[i+k]
            idx = np.arange(n)
            idx_k = (idx + k) % n
            # Approximate sum of products
            prod_sum = cp.sum(cp.multiply(h, h[idx_k]))
            # = minimize sum over i of h[i]*h[i+k] for k>=1
            constraints.append(yk >= 0)
            constraints.append(yk <= h[0])
            constraints.append(yk <= h[idx_k])
            constraints.append(yk >= h[0] + h[idx_k] - 1)
            # Instead, use direct link from sum to yk (not the correct way)
            # Better: Set sum_i h_i h_{i+k} >= t for all k.
            constraints.append(cp.sum(cp.multiply(h, h[idx_k])) >= t)
        
        prob = cp.Problem(cp.Maximize(t), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        return h.value if h.value is not None else h_start
    except:
        return h_start

def simple_hill_climb(h0, budget_s, start_time):
    """Optimize by repeatedly improving the worst shift."""
    h = h0.copy()
    best_c5 = c5_of_h(h)
    best_h = h.copy()
    
    time_left = budget_s - (time.time() - start_time)
    if time_left < 1:
        return best_h, best_c5
        
    # Hyperparameters
    max_steps = 1000
    alpha = 0.1
    patience = 50
    
    for step in range(max_steps):
        if time.time() - start_time > budget_s - 1:
            break
            
        k = worst_shift(h)
        h_new = gradient_step(h, k, alpha)
        c5_new = c5_of_h(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 50  
        else:
            patience -= 1
            if patience <= 0:
                break
                
        h = h_new.copy()
        
    return best_h, best_c5

def anneal_simplified(h0, budget_s, start_time):
    """Use a simple local search with simulated annealing-style moves."""
    h = h0.copy()
    current_c5 = c5_of_h(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    time_left = budget_s - (time.time() - start_time)
    if time_left < 1:
        return best_h, best_c5
        
    temp = 1.0
    cool_rate = 0.998
    max_iter = 10000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make a small random move
        idx1, idx2 = np.random.choice(len(h), size=2, replace=False)
        delta = np.random.uniform(-0.05, 0.05)
        
        h_trial = h.copy()
        h_trial[idx1] += delta
        h_trial[idx2] -= delta
        
        # Project back to feasible set
        h_trial = proj_simplex_box(h_trial, len(h))
        c5_trial = c5_of_h(h_trial)
        
        if c5_trial < current_c5 or np.random.rand() < np.exp(-(c5_trial - current_c5) / temp):
            h = h_trial
            current_c5 = c5_trial
            
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
                
        temp *= cool_rate
        
    return best_h, best_c5
