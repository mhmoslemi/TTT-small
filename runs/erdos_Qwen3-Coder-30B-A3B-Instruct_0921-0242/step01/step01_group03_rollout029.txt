import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def evaluate_binary_mask(b):
    """Evaluate the maximum overlap for a binary mask b"""
    n = len(b)
    dx = 2.0 / n
    h = b.astype(float)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def swap_positions(b, i, j):
    """Swap positions i and j in binary mask b"""
    b = b.copy()
    b[i], b[j] = b[j], b[i]
    return b

def simulate_annealing(n_points=200, seed=42, budget_s=350):
    """Run simulated annealing on binary configurations"""
    np.random.seed(seed)
    start_time = time.time()

    # Initial population
    best_mask = None
    best_c5 = np.inf

    # Generate diverse initial masks
    initial_masks = []

    # Random binary with 100 ones
    mask = np.zeros(n_points, dtype=int)
    mask[:n_points//2] = 1
    np.random.shuffle(mask)
    initial_masks.append(('random', mask))

    # Two blocks
    mask = np.zeros(n_points, dtype=int)
    mask[n_points//4:3*n_points//4] = 1
    initial_masks.append(('blocks', mask))

    # Three blocks
    mask = np.zeros(n_points, dtype=int)
    mask[n_points//5:2*n_points//5] = 1
    mask[2*n_points//5:3*n_points//5] = 1
    mask[3*n_points//5:4*n_points//5] = 1
    initial_masks.append(('three_blocks', mask))

    # Spiral
    mask = np.zeros(n_points, dtype=int)
    for i in range(n_points):
        mask[i] = 1 if (i % 3 == 0) else 0
    initial_masks.append(('spiral', mask))

    # Main search
    T = 0.02
    alpha = 0.995
    max_iter = 30000

    for method, init_mask in initial_masks:
        if time.time() - start_time > budget_s - 1:
            break

        current_mask = init_mask.copy()
        current_c5 = evaluate_binary_mask(current_mask)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_mask = current_mask.copy()

        # Within each initial, loop until accepted moves are reduced or time runs out
        last_accepted = 0
        iter_count = 0

        while iter_count < max_iter and time.time() - start_time < budget_s - 1:
            # Try to swap a 1 and a 0
            ones = np.where(current_mask == 1)[0]
            zeros = np.where(current_mask == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break

            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            new_mask = swap_positions(current_mask, i, j)
            new_c5 = evaluate_binary_mask(new_mask)

            delta = new_c5 - current_c5

            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_mask = new_mask
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_mask = current_mask.copy()
                last_accepted = iter_count

            iter_count += 1

            if iter_count - last_accepted > 1000:
                T *= alpha

    return best_mask, best_c5


def project_to_feasible(h):
    """Project h into the feasible set with sum(h) = n/2 and h in [0,1]^n"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h

    if abs(deficit) < 1e-10:
        return h

    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h

    if abs(deficit) < 1e-10:
        return h

    adjustment = deficit / n
    h += adjustment
    h = np.clip(h, 0, 1)
    return h


def solve_lp_refinement(b, max_iter=50):
    """Solve a linear program to refine the binary mask b into a continuous h"""
    n = len(b)
    dx = 2.0 / n

    # Variables
    h = cp.Variable(n)
    
    # Constraints
    constraints = []
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(cp.sum(h) == n/2)

    # Objective variable
    t = cp.Variable()

    # Max overlap constraint for each shift
    g = 1.0 - b.astype(float)
    corr_full = np.correlate(b.astype(float), g, mode='full') * dx
    max_shift = len(corr_full) // 2

    for k in range(len(corr_full)):
        shift = k - max_shift
        expr = 0
        for i in range(n):
            j = (i + shift) % n
            if b[i] == 1:
                expr += h[i] * (1 - b[j])
            else:
                expr += h[i] * (1 - b[j])  # This is actually just h[i] * 0 = 0, so no need to distinguish
        # Streamline: for each i, t >= expr. But expr is just a sum of h terms weighted by weights
        # Lets make a vectorized version of expr
        # Compute the linear expression for the k-th correlation:
        # For each entry i of h, we are computing h[i] * (1 - h[i+shift]) where indices wrap around
        # However, since we know the indices from b, we rewrite as:
        # Let's be efficient here by precomputing each taken from b for all shift values
        # Re-express the full expression as a sum of h terms and check weights.

        # Manually do correlation step:
        corr_k = 0
        for i in range(n):
            j = (i + shift) % n
            if b[i] == 1:  # Then contribution is h_i * (1 - b[j])
                corr_k += h[i] * (1 - b[j])
            else:  # Then contribution is h_i * (1 - b[j])  = h_i * (1 - 0) = h_i
                corr_k += h[i] * 1.0  # since b[i] = 0, 1 - 0 = 1
        constraints.append(corr_k <= t)

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        # Fallback: just return b converted to float
        return b.astype(float)

    if prob.status == cp.OPTIMAL:
        return np.array(h.value).flatten()
    else:
        # Fallback
        return b.astype(float)


def local_refinement(h_star, max_iter=2000):
    """Local search to improve h_star"""
    n = len(h_star)
    dx = 2.0 / n
    best_h = h_star.copy()
    best_c5 = c5_of_h(best_h)

    for i in range(max_iter):
        # Random perturbation
        idx = np.random.randint(0, n)
        delta = np.random.uniform(-0.01, 0.01)
        h_test = best_h.copy()
        h_test[idx] += delta

        # Project back
        h_test = project_to_feasible(h_test)
        c5_test = c5_of_h(h_test)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()

    return best_h, best_c5
