import numpy as np
from scipy.optimize import minimize
import time
from scipy.interpolate import interp1d
import math

def project_to_feasible(u, n):
    """Project vector u to feasible set: 0 <= h[i] <= 1, sum h[i] = n/2"""
    u = np.clip(u, 0.0, 1.0)
    s = np.sum(u)
    if abs(s - n/2) < 1e-12:
        return u
    
    # Bisection to find lambda
    def sum_with_lambda(lam):
        return np.sum(np.clip(u + lam, 0.0, 1.0)) - n/2
    
    # If sum is too small, we need to add to each
    # If sum is too large, we need to subtract from each
    a, b = -1.0, 1.0
    for _ in range(30):
        mid = (a + b) / 2.0
        val = sum_with_lambda(mid)
        if val > 0:
            b = mid
        else:
            a = mid
    lam = (a + b) / 2.0
    h = np.clip(u + lam, 0.0, 1.0)
    return h

def compute_c5_fast(h):
    """Fast evaluation with FFT-based convolution"""
    n = len(h)
    dx = 2.0 / n
    # Zero-pad to 2*n - 1 for full convolution
    H = np.fft.rfft(h, n=2*n-1)
    G = np.fft.rfft(1 - h, n=2*n-1)
    conv = np.fft.irfft(H * G)[:2*n-1]
    return dx * np.max(conv)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_coarse = 150
    n_fine = 300
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Initial guess: symmetric around middle of domain
    initial_h_coarse = np.zeros(n_coarse)
    # Full beta distribution in middle roughly half the domain
    # Place heuristically in domain [0, 2]
    mid_start = int(0.25 * n_coarse)
    mid_end = int(0.75 * n_coarse)
    initial_h_coarse[mid_start:mid_end] = 1.0
    initial_h_coarse = project_to_feasible(initial_h_coarse, n_coarse)
    
    # Enforce symmetric structure
    half = (n_coarse + 1) // 2
    for i in range(half):
        initial_h_coarse[n_coarse - 1 - i] = initial_h_coarse[i]
    
    # CMA-ES part: global optimisation on coarse grid
    best_c5_global = float('inf')
    best_h_global = None
    
    # Sample lambda offspring: ~30 samples
    lambda_offspring = 30
    population = []
    fitnesses = [] 
    
    num_gens = 0
    last_improvement_gen = 0
    
    # Scale down the search space to symmetry-reduced
    d = (n_coarse + 1) // 2  # Reduced dimension
    x_mean = np.ones(d) * 0.5  # Initial mean
    sigma = 0.2  # Stepsize
    cov_matrix = np.eye(d)
    
    # Avoid premature stopping
    max_generations = 200
    tolerance = 1e-4
    
    # Start time tracking for CMA-ES
    cma_es_time = 0
    
    while num_gens < max_generations and cma_es_time < budget_s - 150:
        start_cma = time.time()
        
        # Generate offspring
        offspring = []
        for _ in range(lambda_offspring):
            z = np.random.multivariate_normal(x_mean, sigma**2 * cov_matrix)
            # Apply projection and symmetrize
            h_produced = project_to_feasible(z, n_coarse)
            # Symmetrize the first part to get full vector
            half_size = (n_coarse + 1) // 2
            if n_coarse % 2 == 1:
                h_sym = np.concatenate((h_produced, h_produced[-2::-1]))
            else:
                h_sym = np.concatenate((h_produced, h_produced[::-1]))
            h_sym = project_to_feasible(h_sym, n_coarse)
            offspring.append(h_sym)
        
        # Compute fitness
        fitness_list = []
        for ind in offspring:
            # Early stop pruning
            H = np.fft.rfft(ind, n=2*n_coarse-1)
            G = np.fft.rfft(1 - ind, n=2*n_coarse-1)
            try:
                conv = np.fft.irfft(H * G)[:2*n_coarse-1]
                value = dx_coarse * np.max(conv)
            except:
                value = float('inf')
            if value < best_c5_global:
                fitness_list.append(value)
            else:
                fitness_list.append(float('inf'))
                
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_list)
        sorted_fits = [fitness_list[i] for i in sorted_indices]
        sorted_offspring = [offspring[i] for i in sorted_indices]
        
        # Select top 10% (as good heuristic in CMA-ES)
        selected = sorted_offspring[:lambda_offspring // 10]
        selected_fits = sorted_fits[:lambda_offspring // 10]
        
        # Update best
        if selected_fits[0] < best_c5_global:
            best_c5_global = selected_fits[0]
            best_h_global = selected[0].copy()
            
        # Update CMA-ES components
        if len(selected) > 0:
            # Update mean
            x_old = x_mean.copy()
            x_mean = np.mean([ offspring[i] for i in range(len(selected))], axis=0)
            # Covariance update (simplified)
            # For full CMA-ES, we'd use whole history
            # Here we just do basic update on recent bests
            x_diff = x_mean - x_old
            cov_matrix += 0.1 * np.outer(x_diff, x_diff)
            
            # Decrement sigma a bit
            sigma = max(1e-8, sigma * 0.99)
            
        num_gens += 1
        cma_es_time = time.time() - start_time
        
        # Check for adaptive restart after 150 gens if no progress
        if num_gens > 150 and (num_gens - last_improvement_gen) > 20:
            if sigma > 1e-5:
                x_mean = np.random.rand(d)
                cov_matrix = np.eye(d)
                sigma *= 0.5
                # constraint reset for prevention
                x_mean = np.ones(d) * 0.5
                last_improvement_gen = num_gens
        
        # Early stop if too much time or solution is already good
        if best_c5_global < 0.379 and cma_es_time > 300:
            break
    
    # Refinement phase with smooth max gradient descent on fine grid
    
    # Fine grid version from best coarse solution
    h_fine_in = np.zeros(n_fine)
    h_coarse = best_h_global
    
    # Interpolate to fine grid using linear interpolation
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    f = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
    h_fine_in = f(x_fine)
    
    # Project to exact constraints
    h_fine = project_to_feasible(h_fine_in, n_fine)
    
    # Apply symmetry
    half_fine = (n_fine + 1) // 2
    for i in range(half_fine):
        h_fine[n_fine - 1 - i] = h_fine[i]
    
    # Gradient-based polishing with suppression of high-frequency work
    beta = 0.5  # initial smoothness
    beta_max = 50.0
    tol = 1e-6
    max_iter_per_beta = 100
    
    # Ensuring value start is low to avoid overfitting during local polishing
    c5_current = np.inf
    best_c5 = best_c5_global  # from global phase
    best_h = h_fine.copy()
    
    while beta < beta_max and (time.time() - start_time) < budget_s - 50:
        # For this local search we'll use a simpler but effective method:
        # Implement weighted Cauchy-concentration or cutting skyline approach
        
        # Use in-batch coercive numerical optimization as plausible meat.
        # A simple projected gradient descent for "smooth max"
        # that makes C5 smooth and well behaved
        
        def grad_f(beta_val):
            """Gradient of smooth max approximation"""
            dx = 2.0 / n_fine
            
            # Compute current conv = h * (1-h)
            H = np.fft.rfft(h_fine, n=2*n_fine-1)
            G = np.fft.rfft(1 - h_fine, n=2*n_fine-1)
            conv = np.fft.irfft(H * G)[:2*n_fine-1]
            
            # Compute weights via normalization
            weights = np.exp(beta_val * dx * conv)
            total = np.sum(weights)
            if total == 0:
                weights = np.ones_like(weights)
                total = len(weights)
            weights /= total
            
            # Now compute the approximate gradient
            # Approx. as: dw/dh_i = dx * \sum_k (\partial conv_k / \partial h_i) * w_k
            
            # Instead of fully dense matrix, exploit Toeplitz structure:
            # \partial conv_k / \partial h_i =  (1 - h_{i+k}) - h_{i-k}
            # See computation:
            kmax = n_fine - 1
            grad_total = np.zeros(n_fine)
            
            # Vectorized loop over k
            # Broadcast right, left, compute difference
            for k in range(0, kmax):
                # k index goes from -kmax+1 to kmax
                current_k = k - kmax
                if current_k >= 0:
                    # left part
                    left = np.pad(h_fine, (0, current_k))[:n_fine]
                    right = np.pad(1 - h_fine, (current_k, 0))[:n_fine]
                else:
                    left = np.pad(h_fine, (0, -current_k))[:n_fine]
                    right = np.pad(1 - h_fine, (-current_k, 0))[:n_fine]
                
                diff = right - left
                weight = weights[k]
                # Accumulate gradients
                grad_total += weight * diff
            
            return dx * grad_total
                
        # Simple gradient update step
        max_iter = min(max_iter_per_beta, 80)
        learning_rate = 0.01 * beta**0.5  # ramp down slow improvement
                
        for _ in range(max_iter):
            if (time.time() - start_time) >= budget_s - 50:
                break
                
            g = grad_f(beta)
            # Update step
            h_new = h_fine - learning_rate * g
            h_new = np.clip(h_new, 0.0, 1.0)
            
            # Project to sum constraint
            h_new = project_to_feasible(h_new, n_fine)
            
            # Enforce symmetry again (no need for restart)
            for i in range(half_fine):
                h_new[n_fine - 1 - i] = h_new[i]
        
            c5_trial = compute_c5_fast(h_new)
            
            # Accept only improvement
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_new.copy()
                h_fine = h_new.copy()
                
            elif c5_trial > best_c5 * 1.01:
                # Might be local minimum; try small random perturbations
                eps = 0.002
                for i in range(5):
                    idx = np.random.randint(0, n_fine)
                    h_trial = h_fine.copy()
                    val = h_trial[idx] + np.random.uniform(-eps, eps)
                    h_trial[idx] = np.clip(val, 0.0, 1.0)
                    # Re-project to Feasible
                    h_trial = project_to_feasible(h_trial, n_fine)
                    # Where to mirror?
                    for j in range(half_fine):
                        h_trial[n_fine - 1 - j] = h_trial[j]
                    c5_trial = compute_c5_fast(h_trial)
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
                        h_fine = h_trial.copy()

            # Update learning rate or adjust beta
            learning_rate *= 0.999
            
        beta *= 2  # increase beta for sharper objective

    # Final polishing: small random swaps (similar to previous code)
    final_time = time.time()
    t_left = max(0, budget_s - (final_time - start_time))
    
    if t_left > 10:
        h_best = best_h.copy()
        c5_best = best_c5
        
        num_swaps = min(200, int(t_left * 50))  # adaptive based on time
        for i in range(num_swaps):
            if (time.time() - start_time) > budget_s - 0.5:
                break
            p = min(6, n_fine // 20)
            indices = np.random.choice(n_fine, p=p, replace=False)
            
            h_trial = h_best.copy()
            eps = 0.004
            for idx in indices:
                val = h_trial[idx] + np.random.uniform(-eps, eps)
                h_trial[idx] = np.clip(val, 0.0, 1.0)
            
            # Project sum again
            h_trial = project_to_feasible(h_trial, n_fine)
            
            # Enforce symmetry once more
            for j in range(half_fine):
                h_trial[n_fine - 1 - j] = h_trial[j]
                
            c5_trial = compute_c5_fast(h_trial)
            if c5_trial < c5_best:
                c5_best = c5_trial
                h_best = h_trial.copy()
        
        best_h = h_best
        best_c5 = c5_best
        
    return (best_h, best_c5, n_fine)
