import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate
import math

def c5_fft(h):
    """Compute C5 using FFT for speed."""
    g = 1 - h
    H = rfft(h)
    G = rfft(g)
    c = irfft(H * G, 2 * len(h) - 1)
    return np.max(c)

def repair_weight(h, target_sum):
    """Repair h to have exactly target_sum ones."""
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    diff = int(target_sum - s)
    if diff > 0:
        # Need to add diff ones
        zeros = np.where(h == 0)[0]
        idxs = np.random.choice(zeros, diff, replace=False)
        h[idxs] = 1
    elif diff < 0:
        # Need to remove -diff ones
        ones = np.where(h == 1)[0]
        idxs = np.random.choice(ones, -diff, replace=False)
        h[idxs] = 0
    return h

def crossover(p1, p2):
    """Two-point crossover."""
    n = len(p1)
    c1 = np.random.randint(1, n - 1)
    c2 = np.random.randint(c1 + 1, n)
    child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
    return child

def mutate_swap(h):
    """Swap mutation."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h
    i1 = np.random.choice(ones)
    i0 = np.random.choice(zeros)
    h[i1], h[i0] = 0, 1
    return h

def hill_climb(h):
    """Targeted local hill-climber focusing on the worst shift."""
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    max_iter = 500
    for _ in range(max_iter):
        c = correlate(h, 1 - h, mode="full")
        k_star = np.argmax(c) - n + 1
        # Compute changes for a subset of swaps
        best_delta = 0
        best_swap = None
        num_swaps = min(200, n * 2)
        for _ in range(num_swaps):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            delta = 0
            j1 = (i1 + k_star) % n
            j0 = (i0 + k_star) % n
            if h[i1] == 1 and h[j1] == 1:
                delta += dx
            elif h[i1] == 0 and h[j1] == 0:
                delta -= dx
            if h[i0] == 1 and h[j0] == 1:
                delta -= dx
            elif h[i0] == 0 and h[j0] == 0:
                delta += dx
            if delta < best_delta:
                best_delta = delta
                best_swap = (i1, i0)
        if best_swap is None:
            break
        i1, i0 = best_swap
        h[i1], h[i0] = 0, 1
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 280
    target_sum = n // 2
    dx = 2.0 / n
    pop_size = 200
    elite = 5
    p_mut = 0.2
    best_c5 = float("inf")
    best_h = None

    # Initial population
    population = []
    # Interpolated from initial_h_values
    h_orig = initial_h_values
    h_orig_len = len(h_orig)
    x_orig = np.linspace(0, 2, h_orig_len)
    x_new = np.linspace(0, 2, n)
    h_interp = np.interp(x_new, x_orig, h_orig)
    h_interp = h_interp * (target_sum / np.sum(h_interp))
    h_interp = np.round(h_interp).astype(int)
    h_interp = repair_weight(h_interp, target_sum)
    population.append(h_interp.copy())

    # Two-block deterministic
    for _ in range(10):
        h = np.zeros(n)
        h[:n//4] = 1
        h[n//2:n//2+n//4] = 1
        h = repair_weight(h, target_sum)
        # Shuffle slightly
        for _ in range(10):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            h[i1], h[i0] = 0, 1
        population.append(h.copy())

    # Alternating pattern
    for _ in range(10):
        h = np.zeros(n)
        h[::2] = 1
        h = repair_weight(h, target_sum)
        population.append(h.copy())

    # Random balanced
    while len(population) < pop_size:
        h = np.zeros(n)
        ones_indices = np.random.choice(n, target_sum, replace=False)
        h[ones_indices] = 1
        population.append(h.copy())

    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = c5_fft(h)
        fitness.append(-c5)

    # Main Evolutionary Loop
    t_start = time.time()
    gen = 0
    while time.time() - start_time < budget_s - 20 and gen < 5000:
        if time.time() - t_start > budget_s - 20:
            break
        # Selection
        mating_pool = []
        for _ in range(pop_size):
            i1, i2 = np.random.choice(pop_size, 2, replace=False)
            if fitness[i1] > fitness[i2]:
                mating_pool.append(population[i1].copy())
            else:
                mating_pool.append(population[i2].copy())

        # Crossover + Mutation
        new_pop = []
        for i in range(0, pop_size, 2):
            p1 = mating_pool[i]
            p2 = mating_pool[i + 1] if i + 1 < pop_size else mating_pool[0]
            child1 = crossover(p1, p2)
            child2 = crossover(p2, p1)
            if np.random.rand() < p_mut:
                child1 = mutate_swap(child1)
            if np.random.rand() < p_mut:
                child2 = mutate_swap(child2)
            child1 = repair_weight(child1, target_sum)
            child2 = repair_weight(child2, target_sum)
            new_pop.extend([child1, child2])

        # Elitism
        elite_indices = np.argsort(fitness)[:elite]
        elites = [population[i].copy() for i in elite_indices]
        new_pop = elites + new_pop[elite:]

        # Diversity control
        if gen % 20 == 0:
            fits = [c5_fft(h) for h in new_pop[:50]]
            avg_dist = np.mean([np.sum(h1 != h2) for i, h1 in enumerate(new_pop[:50]) for j, h2 in enumerate(new_pop[:50]) if i < j]) / n
            if avg_dist < 0.2:
                for i in range(len(new_pop) // 2):
                    h = np.zeros(n)
                    ones = np.random.choice(n, target_sum, replace=False)
                    h[ones] = 1
                    new_pop[i] = h.copy()

        # Keep population size
        population = new_pop[:pop_size]
        # Re-evaluate
        fitness = []
        for h in population:
            c5 = c5_fft(h)
            fitness.append(-c5)

        # Update best
        min_idx = np.argmin([-f for f in fitness])
        if -fitness[min_idx] < best_c5:
            best_c5 = -fitness[min_idx]
            best_h = population[min_idx].copy()

        # Local hill-climb every 10 generations
        if gen % 10 == 0:
            h = hill_climb(best_h)
            c5 = c5_fft(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        gen += 1

    if best_h is None:
        # Fallback to initial if needed
        h = initial_h_values
        n_orig = len(h)
        x_orig = np.linspace(0, 2, n_orig)
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_orig, h)
        h_interp = h_interp * (target_sum / np.sum(h_interp))
        h_interp = np.round(h_interp).astype(int)
        best_h = repair_weight(h_interp, target_sum)
        best_c5 = c5_fft(best_h)

    # Fine scale refinement
    n_fine = 500
    dx_fine = 2.0 / n_fine
    upscaled = np.zeros(n_fine)
    stride = int(np.ceil(n_fine / n))
    for i in range(n):
        start_idx = i * stride
        end_idx = min((i + 1) * stride, n_fine)
        if end_idx > start_idx:
            upscaled[start_idx:end_idx] = best_h[i]
    upscaled = np.round(upscaled)
    upscaled = repair_weight(upscaled, n_fine // 2)
    # Short GA on fine scale
    fine_pop = [upscaled.copy()]
    while len(fine_pop) < 50:
        h = np.zeros(n_fine)
        ones_indices = np.random.choice(n_fine, n_fine // 2, replace=False)
        h[ones_indices] = 1
        fine_pop.append(h.copy())
    for h in fine_pop:
        c5 = c5_fft(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    # Final hill-climb
    best_h = hill_climb(best_h)
    final_c5 = c5_fft(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Compute final c5 with high precision
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx

    return (best_h, final_c5, len(best_h))
