import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import ifft, fft
import math

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Use clip to reduce values if necessary, then add to lower ones
    # First add diff/n to each; then clip to [0,1]
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    # Check if we're still not exact.
    # If not, adjust with water filling.
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    if abs(diff) > 1e-8:
        # Binary search for the water filling: what should δ do to hit exactly n/2?
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        diff = n/2 - current_sum
        
        # Linear adjustment if diff small
        if abs(diff) < 1e-3:
            h += diff / n
            h = np.clip(h, 0, 1)
            return h
        
        # More precise binary search handling full amount.
        # But avoid overcomplicating: just scale.
        h = h * (n/2) / current_sum
        h = np.clip(h, 0, 1)
    
    return h

def compute_gradients(h, active_shift_idx):
    """Compute gradient at h for the active shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    
    # This computes derivative of max_k int h(x)(1-h(x+k)) dx 
    # with respect to h_i, which is:
    # dx * (1 - h_{i+active_shift}) - dx * h_{i-active_shift}
    
    for i in range(n):
        # Left term: -dx * h_{i+active_shift}
        left_idx = (i + active_shift_idx) % n
        grad[i] -= dx * h[left_idx]
        
        # Right term: dx * h_{i-active_shift}
        right_idx = (i - active_shift_idx) % n
        grad[i] += dx * h[right_idx]
    
    return grad

def safe_line_search(h, grad, dx, max_alpha=1.0):
    """Perform a safe line search with backtracking."""
    alpha = max_alpha
    max_trials = 10
    for _ in range(max_trials):
        h_trial = h - alpha * grad
        h_trial = project_to_feasible(h_trial, len(h))
        trial_c5 = evaluate_c5(h_trial)
        if trial_c5 < evaluate_c5(h):
            return h_trial, trial_c5
        else:
            alpha /= 2.0
    return h, evaluate_c5(h)

def memetic_improvement(h, max_iters=50):
    """Apply memetic refinement to improve the current solution."""
    n = len(h)
    dx = 2.0 / n
    h_current = h.copy()
    
    for _ in range(max_iters):
        # Compute correlations
        corr = correlate(h_current, 1 - h_current, mode='full')
        corr_real = corr.real
        # Exclude k=0 which corresponds to the count part of the sum
        corr_real[2*n - 1] = -1e10  # Make sure we never choose this
        active_shift_idx = np.argmax(corr_real)
        
        # Compute gradient for this shift
        grad = compute_gradients(h_current, active_shift_idx) 
        
        # Perform line search
        h_current, _ = safe_line_search(h_current, grad, dx)
        
    return h_current

def random_swap_local_improve(h, max_iters=5000):
    """Random swap hill climbing."""
    n = len(h)
    h_current = h.copy()
    current_c5 = evaluate_c5(h_current)
    
    for _ in range(max_iters):
        # Find indices with non-trivial values
        non_zero = np.where(h_current > 1e-6)[0]
        non_one = np.where(h_current < 1 - 1e-6)[0]
        
        if len(non_zero) == 0 or len(non_one) == 0:
            break
            
        i = np.random.choice(non_zero)
        j = np.random.choice(non_one)
        
        # Amount to transfer
        delta = np.random.uniform(0, min(h_current[i], 1 - h_current[j], 0.01))
        
        # Perform swap
        h_trial = h_current.copy()
        h_trial[i] -= delta
        h_trial[j] += delta
        
        # Project and evaluate
        h_trial = project_to_feasible(h_trial, n)
        trial_c5 = evaluate_c5(h_trial)
        
        if trial_c5 < current_c5:
            h_current = h_trial
            current_c5 = trial_c5
            
    return h_current

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    
    # Initialize particles
    num_particles = 30
    particles = []
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) < n_points:
            # interpolate
            x_orig = np.linspace(0, 2, len(orig_h))
            x_new = np.linspace(0, 2, n_points)
            h_interpolated = np.interp(x_new, x_orig, orig_h)
        else:
            h_interpolated = orig_h[:n_points]
        h_normalized = h_interpolated * (n_points / 2) / np.sum(h_interpolated)
        particles.append(h_normalized)
    else:
        particles.append(np.ones(n_points) * 0.5)
        
    # Add more random particles
    for _ in range(num_particles - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        
    # Initialize velocities and bests
    velocities = [np.zeros(n_points) for _ in range(num_particles)]
    pbest = particles[:]
    pbest_c5 = [evaluate_c5(h) for h in pbest]
    gbest_idx = np.argmin(pbest_c5)
    gbest_h = pbest[gbest_idx]
    gbest_c5 = pbest_c5[gbest_idx]
    
    omega = 0.7
    c1, c2 = 1.5, 1.5
    last_improvement = time.time()
    
    # PSO steps
    gen = 0
    while time.time() - start_time < 0.7 * budget_s:
        if time.time() - last_improvement > 2.0:
            # Inject new particles to restart diversity
            for i in range(num_particles // 4):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[i] = project_to_feasible(h, n_points)
                pbest[i] = particles[i]
                pbest_c5[i] = evaluate_c5(particles[i])
            gbest_idx = np.argmin(pbest_c5)
            gbest_h = pbest[gbest_idx]
            gbest_c5 = pbest_c5[gbest_idx]
            last_improvement = time.time()
            
        gen += 1
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] 
                           + c1 * r1 * (pbest[i] - particles[i])
                           + c2 * r2 * (gbest_h - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            current_c5 = evaluate_c5(particles[i])
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest_h = particles[i].copy()
                    gbest_c5 = current_c5
                    last_improvement = time.time()
        
        # Perform memetic refinement every 5 generations or after global phase
        if gen % 5 == 0 or gen == 1:
            gbest_h = memetic_improvement(gbest_h, max_iters=30)
            gbest_c5 = evaluate_c5(gbest_h)
            
    # Late-stage polishing
    while time.time() - start_time < 0.9 * budget_s:
        improved = False
        h_copy = gbest_h.copy()
        h_new = random_swap_local_improve(h_copy, max_iters=5000)
        new_c5 = evaluate_c5(h_new)
        if new_c5 < gbest_c5:
            gbest_h = h_new
            gbest_c5 = new_c5
            improved = True
            last_improvement = time.time()
        
        if not improved:
            break
    
    final_h = gbest_h
    final_c5 = gbest_c5
    final_n = n_points
    
    return (final_h, final_c5, final_n)
