import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize

def c5_of_h(h):
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    result = np.max(conv) * dx
    return result

def symmetrize_h(h):
    n = len(h)
    d = n // 2 + 1
    w = h[:d]
    # Mirror to reconstruct full h
    h_full = np.zeros(n)
    h_full[:d] = w
    h_full[d:] = w[-2::-1]
    return h_full

def build_symmetric_lp(n, S, lambda_pen):
    d = n // 2 + 1  # independent variables
    
    # Variables
    w = cp.Variable(d)
    M = cp.Variable()
    p_vars = {}
    
    # Create p variables for all k in S and valid i
    for k in S:
        for i in range(n):
            if 0 <= i + k < n:
                p_vars[(i, k)] = cp.Variable()
    
    # Constraints
    constraints = [
        0 <= w,
        w <= 1,
        cp.sum(w) == n / 4 + 0.5,  # Sum constraint
        M >= 0
    ]
    
    # McCormick envelopes
    for k in S:
        for i in range(n):
            if 0 <= i + k < n:
                p = p_vars[(i, k)]
                # p <= w[i]
                constraints.append(p <= w[i])
                # p <= w[i+k]
                constraints.append(p <= w[i+k])
                # p >= 0
                constraints.append(p >= 0)
                # p >= w[i] + w[i+k] - 1
                constraints.append(p >= w[i] + w[i+k] - 1)
    
    # Overlap constraints
    for k in S:
        expr = 0
        for i in range(n):
            if 0 <= i + k < n:
                expr += p_vars[(i, k)]
        constraints.append(0.5 * n - expr <= M)
    
    # Objective
    objective = cp.Minimize(M - lambda_pen * cp.sum([p_vars[(i, 0)] for i in range(n) if i < n]))
    
    prob = cp.Problem(objective, constraints)
    return prob, w, M, p_vars

def double_oracle_cutting_plane(n_points=200, budget_s=600):
    start_time = time.time()
    n = n_points
    d = n // 2 + 1
    
    # Initial shift
    S = {0}
    lambda_pen = 0.1
    
    best_h = None
    best_c5 = float('inf')
    
    # Safety margin for final polish
    safety_margin = 10
    
    max_iterations = 50
    
    for iteration in range(max_iterations):
        # Time
        if time.time() - start_time > budget_s - safety_margin:
            break
        
        # Build LP problem
        prob, w, M, p_vars = build_symmetric_lp(n, S, lambda_pen)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            break
        
        # Extract solution
        sol_w = w.value
        if sol_w is None:
            break
            
        # Reconstruct full h
        h_full = np.zeros(n)
        h_full[:d] = sol_w
        h_full[d:] = sol_w[-2::-1]
        h_full = np.clip(h_full, 0, 1)
        
        # Evaluate exact C5
        c5_exact = c5_of_h(h_full)
        
        if c5_exact < best_c5:
            best_c5 = c5_exact
            best_h = h_full.copy()
            
        current_M = M.value
        
        # Check if stopped by optimality
        if abs(c5_exact - current_M) < 1e-6:
            break
            
        # Add worst shift
        g = 1 - h_full
        conv = np.correlate(h_full, g, mode='full')
        max_shift = np.argmax(conv) - (n - 1)
        S.add(max_shift)
        
        # Increase lambda
        lambda_pen = min(5, lambda_pen * 1.5)
        
        if len(S) > 30:
            break
    
    return best_h, best_c5

def polish_solution(h, n_points=200, max_iter=200):
    h = h.copy()
    dx = 2.0 / n_points
    
    for _ in range(max_iter):
        g = 1 - h
        conv = np.correlate(h, g, mode='full')
        k_max = np.argmax(conv) - (n_points - 1)
        
        # Compute subgradient for shift k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max >= 0 and i + k_max < n_points:
                grad[i] += (1 - h[i + k_max])
            if i - k_max >= 0 and i - k_max < n_points:
                grad[i] -= h[i - k_max]
                
        # Take step
        alpha = 0.02 * (0.9 ** _)
        h_new = h - alpha * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        delta = (sum_h - n_points / 2) / n_points
        
        # Water filling
        h_new = h_new - delta
        h_new = np.clip(h_new, 0, 1)
        
        # Symmetrize
        h_new = symmetrize_h(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        h = h_new
        
        if _ % 10 == 0:
            c5_after = c5_of_h(h)
            if c5_after < 1e-6:
                break
                
    return h
