def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Adjust by distributing difference
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def block_to_step_functions(u, v, K, n_points):
    """Convert block parameters to a step function."""
    # Lengths
    u_exp = np.exp(u)
    lengths = 2 * u_exp / np.sum(u_exp)
    
    # Heights (with scaling to satisfy integral constraint)
    heights = 1 / (1 + np.exp(-v))
    area = np.sum(heights * lengths)
    alpha = 1 / area
    heights = np.minimum(1, alpha * heights)
    
    # Adjust for exact integral if needed
    if abs(np.sum(heights * lengths) - 1) > 1e-10:
        area = np.sum(heights * lengths)
        heights *= 1 / area
        heights = np.minimum(1, heights)
    
    # Reconstruct step function
    step_h = np.zeros(n_points)
    cumsum = 0
    for i in range(K):
        start = int(cumsum / 2 * n_points)
        end = int((cumsum + lengths[i]) / 2 * n_points)
        if end > start:
            step_h[start:end] = heights[i]
        cumsum += lengths[i]
    
    return step_h

    u = params[:K]
    v = params[K:]
    h_step = block_to_step_functions(u, v, K, n_points)
    c5 = evaluate_c5(h_step)
    return c5
