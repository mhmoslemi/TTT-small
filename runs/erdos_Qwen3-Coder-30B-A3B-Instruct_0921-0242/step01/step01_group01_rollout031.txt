import numpy as np
from scipy.optimize import differential_evolution
import time
import math

def c5_of_h(h):
    """Compute C5 for a given step function h using FFT for efficiency"""
    dx = 2.0 / len(h)
    # Use FFT for correlation
    H = np.fft.rfft(h, n=2*len(h)-1)
    G = np.fft.rfft(1-h, n=2*len(h)-1)
    conv = np.fft.irfft(H*G)[:2*len(h)-1]
    result = np.max(conv) * dx
    return result

def project_to_feasible(u, target_sum):
    """Project vector u onto feasible region with sum = target_sum"""
    u = np.clip(u, 0.0, 1.0)
    s = np.sum(u)
    if abs(s - target_sum) < 1e-10:
        return u
    
    # Use bisection to find lambda
    def sum_with_lambda(lam):
        return np.sum(np.clip(u + lam, 0.0, 1.0))
    
    # Bracket the root
    low, high = -1.0, 1.0
    while sum_with_lambda(low) > target_sum:
        low -= 1.0
    while sum_with_lambda(high) < target_sum:
        high += 1.0

    # Bisection
    for _ in range(30):
        mid = (low + high) / 2.0
        if sum_with_lambda(mid) > target_sum:
            high = mid
        else:
            low = mid
    
    return np.clip(u + (low + high)/2, 0.0, 1.0)

def symmetrize_and_project(h, target_sum):
    """Symmetrize h and project to feasible set"""
    n = len(h)
    h_sym = np.zeros(n)
    if n % 2 == 0:
        mid = n // 2
        h_sym[:mid] = h[:mid]
        h_sym[mid:] = h[mid-1::-1]
    else:
        mid = n // 2
        h_sym[:mid] = h[:mid]
        h_sym[mid] = h[mid]
        h_sym[mid+1:] = h[mid-1::-1]
    return project_to_feasible(h_sym, target_sum)

def cma_es_global_search(n_coarse, n_fine, budget_s):
    """Global search with CMA-ES in symmetric space"""
    import cma
    
    # Symmetric representation
    d = (n_coarse + 1) // 2
    target_sum = n_coarse / 2
    
    # Initialize CMA-ES
    es = cma.CMAEvolutionStrategy(d * [0.5], 0.2, {
        'bounds': [0.0, 1.0],
        'maxiter': 200,
        'verbose': -1
    })
    
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    generation = 0
    
    while not es.stop() and time.time() - start_time < budget_s-1:
        # Generate offspring
        X = es.ask()
        F = []
        for x in X:
            # Symmetrize and project
            h = symmetrize_and_project(x, target_sum)
            try:
                c5 = c5_of_h(h)
            except:
                c5 = float('inf')
            F.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        # Tell
        es.tell(X, F)
        generation += 1
    
    return best_h, best_c5

def smooth_max_gradient_descent(h_seed, n_fine, beta_max=50.0, max_iter=80):
    """Smooth max gradient descent with β continuation"""
    dx = 2.0 / n_fine
    h = h_seed.copy()
    
    def compute_convolution(h):
        H = np.fft.rfft(h, n=2*n_fine-1)
        G = np.fft.rfft(1-h, n=2*n_fine-1)
        conv = np.fft.irfft(H*G)[:2*n_fine-1]
        return conv
    
    def gradients_of_convolution(h, k_max):
        """Compute gradients of convolution in one pass"""
        right = np.pad(1-h, (k_max, 0))[:len(h)]
        left = np.pad(h, (0, k_max))[:len(h)]
        return right - left
    
    beta = 0.5
    lambda_reg = 10.0
    
    while beta <= beta_max:
        # Soft max function
        def soft_max_f(h):
            conv = compute_convolution(h)
            weights = np.exp(beta * dx * conv)
            weights = weights / np.sum(weights)
            return np.sum(weights * conv) * dx / beta
        
        # Gradient computation
        def compute_gradient(h):
            conv = compute_convolution(h)
            weights = np.exp(beta * dx * conv)
            weights = weights / np.sum(weights)
            
            # Create Toeplitz struct for shifts
            # For each i, we compute partial_derivative of the soft max function
            grad = np.zeros_like(h)
            # Right part (k >= 0) and left part (k < 0)
            temp_right = np.zeros_like(h)
            temp_left = np.zeros_like(h)
            
            # For k = 1 to len(h)-1 (equivalent to positive shifts):
            for k in range(1, len(h)):
                shifted = np.pad(1-h, (k,0))[:len(h)]
                temp_right += weights[k] * shifted
            
            for k in range(1, len(h)):
                shifted = np.pad(h, (0,k))[:len(h)]
                temp_left += weights[2*len(h)-2-k] * shifted
            
            grad = dx * (temp_right - temp_left)

            # Add penalty gradient
            penalty_grad = 2 * lambda_reg * (np.sum(h) - n_fine/2)
            grad += penalty_grad
            return grad

        # Gradient descent step (with symmetrization)
        for _ in range(max_iter):
            if time.time() > start_time + 550:
                break
            grad = compute_gradient(h)
            h = h - 0.001 * grad
            h = symmetrize_and_project(h, n_fine/2)
            h = project_to_feasible(h, n_fine/2)
            
        # Increase beta
        beta *= 2
        lambda_reg *= 2
    
    return h

def local_improvement(h, n_fine):
    """Local stochastic hill climbing"""
    dx = 2.0 / n_fine
    max_iter = 2000
    epsilon = 0.004
    best_c5 = c5_of_h(h)
    
    for _ in range(max_iter):
        if time.time() - start_time > 590:
            break
        # Random index subset
        p = min(6, n_fine // 20)
        indices = np.random.choice(n_fine, p, replace=False)
        h_trial = h.copy()
        for i in indices:
            delta = np.random.uniform(-epsilon, epsilon)
            h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
        h_trial = project_to_feasible(h_trial, n_fine/2)
        c5_trial = c5_of_h(h_trial)
        if c5_trial < best_c5:
            h = h_trial.copy()
            best_c5 = c5_trial
        
        # Symmetrize every 50 steps
        if _ % 50 == 0:
            h = symmetrize_and_project(h, n_fine/2)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global start_time
    start_time = time.time()
    
    np.random.seed(seed)
    n_coarse = 150
    n_fine = 300  
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Phase 1: Global search with CMA-ES to get macroscopic shape
    print("Starting global search with CMA-ES...")
    coarse_h, coarse_c5 = cma_es_global_search(n_coarse, n_fine, budget_s*0.6)
    if coarse_h is None:
        # Fallback to a simple pattern
        coarse_h = np.ones(n_coarse) * 0.5
        coarse_c5 = c5_of_h(coarse_h)
    
    # Phase 2: Refine with smooth dynamics
    print("Refining with smooth max optimization...")
    
    # Interpolate to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_coarse), coarse_h)
    h_fine = project_to_feasible(h_fine, n_fine/2)
    h_fine = symmetrize_and_project(h_fine, n_fine/2)
    
    # Smooth max optimization
    refined_h = smooth_max_gradient_descent(h_fine, n_fine, beta_max=50.0, max_iter=80)
    
    # Local stochastic hill climbing
    print("Performing local refinement...")
    final_h = local_improvement(refined_h, n_fine)
    
    # Final evaluation
    final_c5 = c5_of_h(final_h)
    
    return (final_h, final_c5, n_fine)
