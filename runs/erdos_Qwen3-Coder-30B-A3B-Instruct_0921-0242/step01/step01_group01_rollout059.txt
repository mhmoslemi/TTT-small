def project_sum(h, target_sum):
    """Project h to have target sum using water-filling"""
    h = np.copy(h)
    current_sum = np.sum(h)
    diff = current_sum - target_sum
    
    if abs(diff) < 1e-12:
        return h
    
    # Find indices that can be adjusted
    idxs = np.where((h > 0) & (h < 1))[0]
    
    if len(idxs) == 0:
        # All values are at boundaries - just clip and force sum
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    # Water-filling
    while abs(diff) > 1e-12:
        # Distribute evenly among adjustables
        delta = diff / len(idxs)
        if delta > 0:
            # Reduce by delta
            for i in idxs:
                if h[i] - delta >= 0:
                    h[i] -= delta
                else:
                    h[i] = 0
        else:
            # Increase by |delta|
            for i in idxs:
                if h[i] + (-delta) <= 1:
                    h[i] += (-delta)
                else:
                    h[i] = 1
        # Recompute
        current_sum = np.sum(h)
        diff = current_sum - target_sum
        # Update adjustable indices
        idxs = np.where((h > 0) & (h < 1))[0]
    
    return h

def compute_c5(h, dx):
    """Compute the C5 value for a given h"""
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    return np.max(conv) * dx

    # The shift k corresponds to index k + (n-1)

    conv_shifts = conv - (n - 1) # shift will be in [-n+1, n-1]

    # Shift all to avoid overflow
    conv_beta_shifted = conv_beta - np.max(conv_beta)
    exp_vals = np.exp(conv_beta_shifted)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    max_time = budget_s
    
    n_fine = 300
    dx_fine = 2.0 / n_fine
    
    # Step 1: Build coarse grid and SDP solve
    n_coarse = 120  # Smaller for speed
    dx_coarse = 2.0 / n_coarse
    
    # Define variables
    h = cp.Variable(n_coarse)
    X = cp.Variable((n_coarse, n_coarse), symmetric=True)
    t = cp.Variable()
    M = cp.bmat([[1, h.T], [h, X]])
    
    # Objective: max t
    objective = cp.Maximize(t)
    
    # Constraints
    constraints = []
    
    # Toeplitz matrices for autocorrelation (diagonals set)
    # For each k from -n+1 to n-1, build Rk
    # But we can compute explicitly via generate Toeplitz first
    # Create Rk such that trace(R_k X) = sum_i h_i * h_{i+k}
    for k in range(-n_coarse+1, n_coarse):
        if k >= 0:
            # on diagonals: Rk[i][i+k] = 1
            r = np.zeros((n_coarse, n_coarse))
            r[range(n_coarse-k), range(k, n_coarse)] = 1.0
            # Only define for off-diags (k>=0)
        else:
            # for negative k, Rk[i][i-k] = 1 => h_{i-k}
            r = np.zeros((n_coarse, n_coarse)) 
            r[range(-k, n_coarse), range(n_coarse+k)] = 1.0
        # Add constraint trace(R_k X) >= t
        constraints.append(cp.trace(r @ X) >= t)
    
    # Built-in conv constraints
    # 0 <= h <= 1
    constraints += [h >= 0, h <= 1]
    # sum h fixed
    constraints += [cp.sum(h) == n_coarse / 2]
    # M is PSD
    constraints += [M >> 0]  # PSD constraint
    
    # Formulate and solve
    prob = cp.Problem(objective, constraints)
    
    # Define timeout mechanism
    solution_time = 0
    intermediate_sol = None  
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, verbose=False)
        # Check time after solve:
        cur = time.time()
        solution_time = cur - start_time
    except:
        pass

    # Extract candidate h from solution
    # If solver succeeded, h.value is available
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        # SDP worked, extract h
        h_coarse = h.value
        X_coarse = X.value
        # Do eigenvalue decomposition to find best rank-1 approximation
        # SVD or eigval-decomp:
        # Actually,
        # If M = [1 h^T; h X], then det(M) = 1 * det(X) - h^T h = det(X) - ||h||^2
        # To extract a vector from the solution,
        # The SDP is a relaxation we should project it back to a approximation of 
        # solution that satisfies: h h^T = X and h has required norm constraints.
        # Take leading eigenvector of X or eigh(X)[1] ?:
        if X_coarse is not None:
            _, v = np.linalg.eigh(X_coarse)
            if len(v.shape) == 1:
                v = v.reshape(-1, 1)
            # Normalized one (eigvec with largest eigenval)
            v0 = v[:, -1]
            lambda0 = np.real(np.max(np.linalg.eigvals(X_coarse)))
            if lambda0 > 0:
                h0 = np.sqrt(lambda0) * v0
            else:
                h0 = v0.copy()  # fallback linear
            # Project onto feasible set
            # Violate bounds? too much sum? Projection needed:
            h0 = np.clip(h0, 0, 1)
            # Fix sum to exact:
            h0 = project_sum(h0, n_coarse / 2)
            # Enforce symmetry:
            h0 = (h0 + h0[::-1]) / 2.0
            h_coarse = h0
        else:
            h_coarse = np.random.uniform(0, 1, n_coarse)
            h_coarse = project_sum(h_coarse, n_coarse / 2)
            h_coarse = (h_coarse + h_coarse[::-1]) / 2.0
    else:
        # Fallback to simpler strategy
        h_coarse = np.random.uniform(0, 1, n_coarse)
        h_coarse = project_sum(h_coarse, n_coarse / 2)
        h_coarse = (h_coarse + h_coarse[::-1]) / 2.0
    
    # Do coarse refinement with smooth-max gradient
    # Store max improvement
    c5_current = compute_c5(h_coarse, dx_coarse)

    # Use smooth max technique to compute better gradient with non-linear proxy
    def smooth_max_objective(h_temp, beta=1.0):
        g = 1.0 - h_temp
        conv = np.correlate(h_temp, g, mode='full')
        conv_shifts = conv - (n_coarse - 1)
        conv_beta = beta * dx_coarse * conv_shifts
        # prevent overflow
        conv_beta_shift = conv_beta - np.max(conv_beta)
        return np.log(np.sum(np.exp(conv_beta_shift))) / beta
    
    def smooth_max_gradient(h_temp, beta=1.0):
        return compute_smooth_max_gradient(h_temp, dx_coarse, beta)
    
    # Continue optimization via increasing beta schedule
    beta_levels = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]
    
    h_refined = np.copy(h_coarse)
    c5_refined = c5_current
    
    for beta in beta_levels:
        if time.time() - start_time > max_time - 5:
            break
        # LBFGS-B as previous code  
        try:
            # Apply smoothed gradient with boundary conditions:
            # Per iteration, project steps, apply clip + sum + mirror
            last_c5 = c5_refined + 1e10  # dummy to start loop
            for _ in range(80):
                if time.time() - start_time > max_time - 5:
                    break
                
                grad = smooth_max_gradient(h_refined, beta)
                h_trial = h_refined - 0.0001 * grad  # stepsize of 0.0001 for now used.
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = project_sum(h_trial, n_coarse / 2)
                # Mirror symmetry  
                h_trial = (h_trial + h_trial[::-1]) / 2.0
                
                c5_trial = compute_c5(h_trial, dx_coarse)
                if c5_trial < c5_refined:
                    h_refined = h_trial
                    c5_refined = c5_trial
                else:
                    # No change after some stages
                    if abs(c5_trial - last_c5) < 1e-8:
                        break  # early exit
            
            # If convergence has stopped improving for last 3 trials, also skip
            if abs(c5_refined - last_c5) < 1e-8:
                break
                
        except:
            pass
    
    # Step 3: Upsampling to fine grid
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    
    if np.all(h_refined >= 0) and np.all(h_refined <= 1):
        f = interp1d(x_coarse, h_refined, kind='linear', fill_value='extrapolate')
        h_fine = f(x_fine)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = project_sum(h_fine, n_fine / 2)
        h_fine = (h_fine + h_fine[::-1]) / 2.0
    else:
        h_fine = np.ones(n_fine) * 0.5
        h_fine = project_sum(h_fine, n_fine / 2)
        h_fine = (h_fine + h_fine[::-1]) / 2.0
        
    # Step 4: Fine-grid refinement
    t_max = max_time - time.time()
    if t_max > 5:
        c5_current_fine = compute_c5(h_fine, dx_fine)
        c5_opt = c5_current_fine
        h_last = h_fine.copy()
        try:
            # Simpler optimization on fine grid
            beta_levels = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]
            for beta in beta_levels:
                if time.time() - start_time > max_time - 1:
                    break
                # L-BFGS step within reasonable limit
                for _ in range(50):
                    if time.time() - start_time > max_time - 1:
                        break
                    grad = smooth_max_gradient(h_last, beta)
                    h_trial = h_last - 0.0001 * grad
                    h_trial = np.clip(h_trial, 0, 1)
                    h_trial = project_sum(h_trial, n_fine / 2)
                    h_trial = (h_trial + h_trial[::-1]) / 2.0
                    c5_trial = compute_c5(h_trial, dx_fine)
                    # Accept improvement
                    if c5_trial < c5_opt:
                        h_last = h_trial
                        c5_opt = c5_trial
        except Exception as e:
            pass
        h_fine = h_last
        c5_fine = c5_opt
        
    # Final polishing
    t_max = max_time - time.time()
    if t_max > 20:
        # Simulate polishing with small stochastic changes in-region
        # Randomly select several index pairs and adjust
        h_best = h_fine.copy()
        c5_best = compute_c5(h_best, dx_fine)
        try:
            epsilon = 0.004
            steps_remaining = int(t_max / 0.05)
            for _ in range(steps_remaining):
                if time.time() - start_time > max_time - 0.1:
                    break
                # Small stochastic changes
                p = min(6, n_fine // 20)
                indices = np.random.choice(n_fine, size=p, replace=False)
                h_trial = h_best.copy()
                # Modify with small noise
                for i in indices:
                    h_trial[i] += np.random.uniform(-epsilon, epsilon)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = project_sum(h_trial, n_fine / 2)
                h_trial = (h_trial + h_trial[::-1]) / 2.0
                c5_trial = compute_c5(h_trial, dx_fine)
                if c5_trial < c5_best:
                    h_best = h_trial
                    c5_best = c5_trial
        except:
            pass
    else:
        h_best = h_fine
        c5_best = compute_c5(h_best, dx_fine)
        
    return (h_best, c5_best, 300)
