import numpy as np
import scipy.optimize as opt
import time
import scipy.fft as fft
from cvxpy import Variable, Minimize, Problem, sum_expr,.constraints, Binary

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 256  # Resolution
    dx = 2.0 / N
    τ_start = 0.05
    τ_factor = 0.5
    outer_cycles = 6
    inner_iters = 150
    
    # Initialize
    h = np.random.rand(N)
    h = h * (N/2) / np.sum(h)  # Ensure the sum constraint
    h = np.clip(h, 0, 1)  # Box constraints
    
    # Fast linear convolution using FFT
    def correlation_fft(h_vec):
        h_pad = np.pad(h_vec, (0, N), 'constant', constant_values=0)
        g_pad = np.pad(1 - h_vec, (0, N), 'constant', constant_values=0)
        c = fft.irfft(fft.rfft(h_pad) * fft.rfft(g_pad))
        return c[:N]
    
    # Weighted gradient via convolution
    def weighted_gradient_fft(h_vec, p):
        p_pad = np.pad(p, (0, N), 'constant', constant_values=0)
        h_pad = np.pad(h_vec, (0, N), 'constant', constant_values=0)
        w1 = fft.irfft(fft.rfft(p_pad) * fft.rfft(1 - h_vec))
        w2 = fft.irfft(fft.rfft(p_pad[::-1]) * fft.rfft(h_vec))
        return w1[:N] - w2[:N]
    
    # Projection onto feasible set
    def project_onto_Ω(v):
        # Clip to [0,1]
        z = np.clip(v, 0, 1)
        # Water-filling
        r = N/2 - np.sum(z)
        for _ in range(100):  # Max iterations
            if abs(r) < 1e-12:
                break
            I = np.where((z > 0) & (z < 1))[0]
            if len(I) == 0:
                break
            adjustment = r / len(I)
            z[I] += adjustment
            z = np.clip(z, 0, 1)
            r = N/2 - np.sum(z)
        return z
    
    # Smooth max via log-sum-exp
    def Fτ(h_vec, τ):
        f = correlation_fft(h_vec)
        return τ * np.log(np.sum(np.exp(f / τ)))
    
    # Get current best
    best_h = h.copy()
    best_c5 = np.max(correlation_fft(h))
    
    # Gradient-based optimization with smoothing
    τ = τ_start
    for _ in range(outer_cycles):
        if time.time() - start_time > budget_s - 100:
            break
        h_prev = h.copy()
        α = 1.0
        for t in range(inner_iters):
            if time.time() - start_time > budget_s - 100:
                break
            y = h + ((t - 1) / (t + 2)) * (h - h_prev)  # Nesterov extrapolation
            f_vals = correlation_fft(y)
            p = np.exp(f_vals / τ - np.max(f_vals / τ))
            p /= np.sum(p)
            grad = weighted_gradient_fft(y, p)
            
            # Backtracking line search
            while True:
                z = y - α * grad
                z_proj = project_onto_Ω(z)
                if Fτ(z_proj, τ) <= Fτ(y, τ) - 0.5 * α * np.dot(grad, grad):
                    break
                α *= 0.5
            
            h = z_proj
            h_prev = h.copy()
            
            # Check for improvement
            c5_new = np.max(correlation_fft(h))
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        
        τ *= τ_factor

    # Rounding to binary
    sorted_indices = np.argsort(best_h)[::-1]
    h_bin = np.zeros(N)
    h_bin[sorted_indices[:N//2]] = 1.0

    # Ensure mass constraint exactly
    h_bin = project_onto_Ω(h_bin)
    h_bin = np.clip(h_bin, 0, 1)  # Clip to [0,1]

    # Improve with local search (swap hill climbing)
    h_local = h_bin.copy()
    current_c5 = np.max(correlation_fft(h_local))

    for _ in range(1000):  # Heuristic number
        if time.time() - start_time > budget_s - 5:
            break
        # Try a few random swaps
        for _ in range(10):
            i = np.random.randint(0, N)
            j = np.random.randint(0, N)
            if h_local[i] != h_local[j]:
                h_test = h_local.copy()
                h_test[i], h_test[j] = h_test[j], h_test[i]
                test_c5 = np.max(correlation_fft(h_test))
                if test_c5 < current_c5 - 1e-8:
                    h_local = h_test
                    current_c5 = test_c5

    # Final check
    final_c5 = np.max(correlation_fft(h_local))
    final_h = h_local.copy()
    return (final_h, final_c5, N)
