def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def compute_gradient(h, k):
    """Compute gradient of C5 at h for shift k."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    for i in range(n):
        # Left term: -dx * h_{i + k}
        left_idx = (i + k) % n
        grad[i] -= dx * h[left_idx]
        
        # Right term: dx * h_{i - k}
        right_idx = (i - k) % n
        grad[i] += dx * h[right_idx]
        
    return grad

def line_search(h, grad, max_iter=10):
    """Backtracking line search to find optimal step size."""

        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h):
            return h_new, c5_new
        alpha /= 2.0
    return h, evaluate_c5(h)
