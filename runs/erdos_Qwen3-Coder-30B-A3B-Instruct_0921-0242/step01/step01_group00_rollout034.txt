import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_block_autocorrelation(lengths, heights, n_eval=200):
    """Compute the autocorrelation of the block function at discrete shifts."""
    K = len(lengths)
    # Compute block endpoints
    cumsum = np.concatenate([[0], np.cumsum(lengths)])
    endpoints = np.column_stack([cumsum[:-1], cumsum[1:]])
    
    # Vectorized computation of overlaps
    dx = 2.0 / n_eval
    shift_vals = np.linspace(0, 2, n_eval, endpoint=False)
    max_overlap = 0
    
    # For each shift
    for s in shift_vals:
        total = 0
        for i in range(K):
            for j in range(K):
                l1, r1 = endpoints[i]
                l2, r2 = endpoints[j]
                # Overlap of intervals [l1,r1] and [l2-s, r2-s]
                overlap_len = max(0, min(r1, r2-s) - max(l1, l2-s))
                total += heights[i] * heights[j] * overlap_len
        overlap = total * dx
        max_overlap = max(max_overlap, overlap)
    
    return max_overlap

def decode_block_params(u, v, K):
    """Decode raw parameters into lengths and heights."""
    # Lengths via softmax
    exp_u = np.exp(u)
    lengths = 2 * exp_u / np.sum(exp_u)
    
    # Heights via sigmoid
    heights = 1 / (1 + np.exp(-v))
    
    # Scale to satisfy integral constraint
    area_sum = np.sum(heights * lengths)
    if area_sum > 0:
        alpha = 1 / area_sum
        heights = np.minimum(1, alpha * heights)
    
    return lengths, heights

def block_objective(params, K, n_eval=200):
    """Objective function for block-based optimization."""
    u = params[:K]
    v = params[K:]
    lengths, heights = decode_block_params(u, v, K)
    return compute_block_autocorrelation(lengths, heights, n_eval)

def project_to_sum_n_half(h, n):
    """Ensure that sum(h) = n/2 with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
        
    h = h + diff / n
    
    # Re-clip
    h = np.clip(h, 0, 1)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Global parameters
    K = 6  # number of blocks
    n_fine = 400  # number of fine samples
    n_eval = 200  # number of shift evaluations for block objective
    
    # Set up bounds for CMA-ES
    bounds = [(-5, 5)] * (2 * K)
    
    # Stage 1: Block-based global optimization
    result = differential_evolution(
        block_objective,
        bounds,
        args=(K, n_eval),
        maxiter=500,
        popsize=10 * K,
        disp=False,
        seed=seed,
        workers=1
    )
    
    if time.time() - start_time > budget_s - 1:
        # fallback to a simple approach
        h_fine = np.ones(n_fine) * 0.5
        c5_bound = evaluate_c5(h_fine)
        return (h_fine, c5_bound, n_fine)
    
    # Decode best block parameters
    u_best = result.x[:K]
    v_best = result.x[K:]
    lengths, heights = decode_block_params(u_best, v_best, K)
    
    # Compute final block-based C5
    c5_block = compute_block_autocorrelation(lengths, heights, n_eval)
    
    # Stage 2: Discretization
    cumsum = np.concatenate([[0], np.cumsum(lengths)])
    endpoints = np.column_stack([cumsum[:-1], cumsum[1:]])
    
    # Create fine grid step function
    h_fine = np.zeros(n_fine)
    dx_fine = 2.0 / n_fine
    for i in range(n_fine):
        x = i * dx_fine
        for j in range(K):
            if endpoints[j, 0] <= x < endpoints[j, 1]:
                h_fine[i] = heights[j]
                break
    
    # Ensure integral constraint
    h_fine = project_to_sum_n_half(h_fine, n_fine)
    
    # Stage 3: Local refinement using projected gradient
    max_loops = 500
    for loop in range(max_loops):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find worst shift index
        c = correlate(h_fine, 1 - h_fine, mode='full')
        k_star = np.argmax(c)
        
        # Compute directional derivative of the integral over worst shift
        grad = np.zeros_like(h_fine)
        for i in range(n_fine):
            # Compute gradient at i for worst shift k_star
            # derivative of integral of h(x)(1-h(x+k_star)) w.r.t. h_i is:
            # -1 * (1 - h_{i+k_star}) + h_{i-k_star}
            # using modular indexing
            left_idx = (i - k_star) % n_fine
            right_idx = (i + k_star) % n_fine
            grad[i] = -(1 - h_fine[right_idx]) + h_fine[left_idx]
        
        # Perform projected gradient step
        alpha = 0.25
        h_trial = h_fine - alpha * grad
        h_trial = project_to_sum_n_half(h_trial, n_fine)
        
        # Check if this improves
        c_trial = correlate(h_trial, 1 - h_trial, mode='full')
        c5_new = np.max(c_trial) * (2.0 / n_fine)
            
        if c5_new < evaluate_c5(h_fine) - 1e-10:
            h_fine = h_trial
        else:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(h_fine)
    
    return h_fine, final_c5, n_fine
