import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    conv = correlate(h, g, mode='full')
    return np.max(conv) * dx

def decode_blocks_to_h(block_params, n_points):
    """Decode block parameters to a step function h"""
    M = len(block_params) // 3
    starts = block_params[0:M]
    lengths = block_params[M:2*M]
    heights = block_params[2*M:3*M]
    
    # Sort by start position
    indices = np.argsort(starts)
    starts = starts[indices]
    lengths = lengths[indices]
    heights = heights[indices]
    
    # Ensure non-overlapping
    for i in range(1, M):
        if starts[i] < starts[i-1] + lengths[i-1]:
            starts[i] = starts[i-1] + lengths[i-1]
    
    # Ensure starts are within [0, 2]
    starts = np.clip(starts, 0, 2)
    lengths = np.clip(lengths, 0, 2 - starts)
    
    # Create step function
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(M):
        if lengths[i] > 0:
            start_idx = int(starts[i] / dx)
            end_idx = int((starts[i] + lengths[i]) / dx)
            end_idx = min(end_idx, n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = heights[i]
    
    # Ensure sum = n/2
    total = np.sum(h)
    if total > 0:
        h = h * (n_points / 2) / total
    else:
        h = np.ones(n_points) * 0.5
    
    return h

def soft_max_conv(h, beta):
    """Evaluate soft max convolution with parameter beta"""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    conv = correlate(h, g, mode='full')
    return np.log(np.sum(np.exp(beta * dx * conv))) / beta

def gradient_soft_max_conv(h, beta):
    """Evaluate the gradient of soft max convolution"""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    conv = correlate(h, g, mode='full')
    
    # Soft weights
    weights = np.exp(beta * dx * conv)
    weights = weights / np.sum(weights)
    
    # Gradient computation
    grad = np.zeros(n)
    for i in range(n):
        local_grad = 0.0
        for k in range(2*n - 1):
            # shift index k corresponds to shift k - (n-1)
            shift = k - (n - 1)
            if 0 <= i + shift < n and 0 <= i - shift < n:
                # Progress does not account for all terms
                local_grad += dx * weights[k] * (1 - h[i + shift]) - dx * weights[k] * h[i - shift]
            elif 0 <= i + shift < n:
                local_grad += dx * weights[k] * (1 - h[i + shift])
            elif 0 <= i - shift < n:
                local_grad -= dx * weights[k] * h[i - shift]
        
        grad[i] = local_grad
    
    return grad

def project_sum(h, target_sum):
    """Project h to satisfy sum = target_sum"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-9:
        return h
    
    diff = current_sum - target_sum
    h = h - diff / len(h)
    h = np.clip(h, 0, 1)
    
    # Correction using water-filling to correct any over/under-flow.
    # This is simpler and robust
    new_sum = np.sum(h)
    if abs(new_sum - target_sum) < 1e-9:
        return h
        
    h = h * target_sum / new_sum
    h = np.clip(h, 0, 1)
    return h

def soft_max_objective_and_gradient(h, beta, target_sum, penalty_lambda):
    """Objective and gradient for soft max problem with penalty"""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    
    # Convolution
    conv = correlate(h, g, mode='full')
    
    # Soft max objective
    soft_max_val = np.log(np.sum(np.exp(beta * dx * conv))) / beta
    
    # Penalty for sum constraint
    penalty = penalty_lambda * (np.sum(h) - target_sum) ** 2
    objective = soft_max_val + penalty
    
    # Gradient of soft max part
    weights = np.exp(beta * dx * conv)
    weights = weights / np.sum(weights)
    
    grad_soft = np.zeros(n)
    for k in range(2*n - 1):
        shift = k - (n - 1)
        for i in range(n):
            # Gradient of term corresponding to correlation
            if 0 <= i + shift < n:
                grad_soft[i] += dx * weights[k] * (1 - h[i + shift])
            if 0 <= i - shift < n:
                grad_soft[i] -= dx * weights[k] * h[i - shift]
    
    # Gradient of penalty
    grad_penalty = 2 * penalty_lambda * (np.sum(h) - target_sum) * np.ones(n)
    
    return objective, grad_soft + grad_penalty

def refine_with_soft_max(h_start, max_beta=50.0, steps=5):
    """Refine h using soft-max surrogate with continuation"""
    h = h_start.copy()
    target_sum = len(h) / 2
    beta = 0.5
    lamb = 10.0
    dx = 2.0 / len(h)
    
    for i in range(steps):
        # Use L-BFGS-B with gradients
        prev_h = h.copy()
        try:
            res = minimize(
                lambda x: soft_max_objective_and_gradient(x, beta, target_sum, lamb),
                h,
                method='L-BFGS-B', 
                jac=True,
                bounds=[(0, 1)] * len(h),
                options={'maxiter': 150, 'ftol': 1e-6}
            )
            if res.success:
                h = res.x
            else:
                h = prev_h
        except:
            pass
        
        # Project sum constraint
        h = project_sum(h, target_sum)
        
        # Symmetrize
        h = (h + h[::-1]) / 2
        h = project_sum(h, target_sum)
        
        # Increase beta and lambda
        beta *= 2.0
        lamb *= 2.0
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 250
    dx = 2.0 / n_points
    max_time = budget_s - 1
    
    # Parametrize h as sum of M constant blocks
    M = 4
    n_params = 3 * M
    
    # Define bounds for the parametric representation
    bounds = []
    for i in range(M):
        bounds.append((0, 2))      # start
        bounds.append((0, 2))      # length
        bounds.append((0, 1))      # height
    
    best_c5 = float('inf')
    best_h = None
    
    # Global search via differential evolution
    def de_objective(block_params):
        h = decode_blocks_to_h(block_params, n_points)
        h = (h + h[::-1]) / 2    # enforce symmetry
        h = project_sum(h, n_points / 2)
        return evaluate_c5(h)
    
    try:
        de_res = differential_evolution(
            de_objective,
            bounds,
            seed=seed,
            maxiter=60,
            popsize=15,
            disp=False,
            atol=1e-6,
            tol=1e-6
        )
        if de_res.success:
            h_global = decode_blocks_to_h(de_res.x, n_points)
            h_global = (h_global + h_global[::-1]) / 2
            h_global = project_sum(h_global, n_points / 2)
            c5_global = evaluate_c5(h_global)
            
            if c5_global < best_c5:
                best_c5 = c5_global
                best_h = h_global.copy()
    except Exception:
        pass

    # Refinement of best solution
    if best_h is not None:
        refined_h = refine_with_soft_max(best_h.copy())
        refined_c5 = evaluate_c5(refined_h)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    # Hill climb to polish
    if best_h is not None and time.time() - start_time < max_time:
        h_current = best_h.copy()
        c5_current = evaluate_c5(h_current)
        
        max_hillclimb_time = max_time - (time.time() - start_time)
        i = 0
        while time.time() - start_time < max_hillclimb_time:
            h_trial = h_current.copy()
            indices = np.random.choice(n_points, size=min(5, n_points//20), replace=False)
            for idx in indices:
                pert = np.random.uniform(-0.005, 0.005)
                h_trial[idx] = np.clip(h_trial[idx] + pert, 0, 1)
            h_trial = project_sum(h_trial, n_points / 2)
            if time.time() - start_time > max_hillclimb_time - 0.1:
                break
            
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < c5_current:
                h_current = h_trial
                c5_current = c5_trial
            
            i += 1
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current
    
    # Final fallback if nothing found
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, n_points)
