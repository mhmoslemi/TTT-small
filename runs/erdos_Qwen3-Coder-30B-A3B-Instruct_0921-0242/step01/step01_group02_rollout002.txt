def binarize_h(h, target_sum):
    """Convert a continuous h to a binary one with target sum."""
    # Take the largest values to be 1, rest 0
    indices = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    h_bin[indices[:int(target_sum)]] = 1
    return h_bin

def evaluate_c5(h, dx):
    """Evaluate C5 using FFT for speed."""
    g = 1 - h
    corr = fftconvolve(h, g, mode='full')
    return np.max(corr) * dx

def update_convolution_delta(corr, h, i1, i0):
    """
    Efficiently update correlation for a single swap (i1, i0) from h to new_h.
    This modifies corr in-place to reflect the swap.
    """
    n = len(h)
    old_h = h.copy()
    new_h = h.copy()
    new_h[i1] = 1
    new_h[i0] = 0
    
    # For swapped positions, compute direct contribution
    # corr[shift] = sum_i h[i] * (1 - h[i + shift])
    # After swap (i1, i0)
    for k in range(n): # effectively ranges up to 2*(n-1)
        old_k = k - i1
        new_k = k - i1
        if 0 <= old_k < n:
            corr[k] -= old_h[i1] * (1 - old_h[old_k])
        if 0 <= new_k < n:
            corr[k] += new_h[i1] * (1 - new_h[new_k])
            
    for k in range(n):
        old_k = k - i0
        new_k = k - i0
        if 0 <= old_k < n:
            corr[k] -= old_h[i0] * (1 - old_h[old_k])
        if 0 <= new_k < n:
            corr[k] += new_h[i0] * (1 - new_h[new_k])
    
    return corr

def swap_positions(h, i1, i0):
    """
    Perform swap between position i1 and i0.
    """
    h[i1], h[i0] = h[i0], h[i1]
    return h
