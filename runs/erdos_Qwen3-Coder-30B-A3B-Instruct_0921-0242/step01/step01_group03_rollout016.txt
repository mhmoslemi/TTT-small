```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 140   # even grid size
    dx = 2.0 / N
    S = N // 2  # target sum
    
    # Precompute indices for potential pairs (i, i+k) for each k
    index_sets = {}  # key: k, value: list of valid i
    for k in range(N):
        index_sets[k] = [i for i in range(N - k)]
    
    # --- Model Setup ---
    # Variables
    x = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Auxiliary variables (one per pair (i,k))
    w = {}
    w_vars = []
    for k in range(N):
        w[k] = cp.Variable(len(index_sets[k]), boolean=True)
        w_vars.extend(w[k])
    
    # Constraints
    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(x) == S)
    
    # AND definition constraints for each w_{i,k}
    for k in range(N):
        indices = index_sets[k]
        for idx, i in enumerate(indices):
            # w_{i,k} <= x_i
            constraints.append(w[k][idx] <= x[i])
            # w_{i,k} <= x_{i+k}
            constraints.append(w[k][idx] <= x[i+k])
            # w_{i,k} >= x_i + x_{i+k} - 1
            constraints.append(w[k][idx] >= x[i] + x[i+k] - 1)
    
    # Bound on overlap terms
    for k in range(N):
        indices = index_sets[k]
        constraints.append(cp.sum(w[k]) >= S - t)
    
    # Symmetry breaking (to prevent translation symmetries)
    constraints.append(x[0] == 1)
    
    # --- Binary Search Loop ---
    L = 0.0
    U = float(S)
    best_t = float('inf')
    best_x = None
    best_sol_time = 0
    
    solver_time_limit = 30  # seconds per invoke
    while time.time() - start_time < budget_s - 10 and U - L > 1e-4:
        T = (L + U) / 2.0
        
        # Update the constraint to have t <= T
        # Remove the prior inequality constraint (can't use old reference)
        temp_constraints = constraints.copy()
        temp_constraints.append(t >= 0)
        temp_constraints.append(t <= T)
        
        p = cp.Problem(cp.Minimize(0), temp_constraints)
        
        try:
            start_solve = time.time()
            p.solve(solver=cp.CBC, verbose=False, 
                     time_limit=solver_time_limit,
                     solver_opts={'max_seconds': solver_time_limit})
            solve_time = time.time() - start_solve
            
            if p.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Restore the original constraints
                w_temp = {}
                for k in range(N):
                    w_temp[k] = cp.Variable(len(index_sets[k]), boolean=True)
                
                # assign new binary values
                x_val = x.value
                if x_val is not None and np.all(x_val.astype(int) == x_val):
                    # Check if this solution improves t
                    current_t = T
                    if current_t < best_t:
                        best_t = current_t
                        best_x = x_val.copy()
                        best_sol_time = time.time()
            else:
                L = T  # No solution found; set lower bound to T
        except Exception as e:
            L = T  # Error in solving; treat as infeasible
    
    # If still no solution, fallback to a greedy solution
    if best_x is None:
        x_val = np.zeros(N, dtype=int)
        x_val[:S] = 1
        best_x = x_val
        best_t = float('inf')
    
    # Convert to floating-point step function
    h = np.array(best_x, dtype=float)
    
    # Normalize for integral = 1 (already done via integer sum = S)
    # If needed, scale to dx*S = 1
    # Since we have N entries and dx = 2/N, scale factor is 1/N * N/2 = 1/2
    # wait, it's already such that sum(h) * dx = 1 because we force sum(x) = S and dx = 2/N
    # sum(h) * (2/N) = 1 leads to sum(h) = N/2 = S, which is the done condition.
    
    # --- Optional: Full objective MILP ---
    if time.time() - start_time < budget_s - 100 and best_t < float('inf'):
        # Rebuild the same problem with objective: minimize t
        # Replace constraints to track equality instead of inequalities
        # Ensure w and t are bound similarly to before
        constraints_final = [cp.sum(x) == S]
        constraints_final.append(x[0] == 1)  # Symmetry breaking
        constraints_final.append(t >= 0)
        
        # For exact definitions
        for k in range(N):
            indices = index_sets[k]
            for idx, i in enumerate(indices):
                constraints_final.append(w[k][idx] <= x[i])
                constraints_final.append(w[k][idx] <= x[i+k])
                constraints_final.append(w[k][idx] >= x[i] + x[i+k] - 1)
                
        for k in range(N):
            indices = index_sets[k]
            constraints_final.append(cp.sum(w[k]) >= S - t)
    
        # No need to recreate variable for w from scratch,
        # we reuse most of it from previous
        final_prob = cp.Problem(cp.Minimize(t), constraints_final)
        
        try:
            final_prob.solve(solver=cp.CBC, verbose=False, 
                            time_limit=120,
                            solver_opts={'max_seconds': 120})
            
            if final_prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                test_t = t.value
                if test_t is not None and test_t < best_t:
                    # Reextract the best x to update the bound
                    # Even if there is constraint misalignment recheck...
                    x_val = x.value
                    if x_val is not None:
                        if np.all(x_val == np.round(x_val)):
                            h = np.array(x_val, dtype=float)
                            best_t = test_t
        except Exception as e:
            pass  # ignore
        
    # --- Fractional polishing ---
    if best_x is not None and time.time() - start_time < budget_s - 50:
        # Recompute the most problematic shift for caching
        h_polish = h.copy()
        current_c5 = c5_of_h(h_polish)
        
        iter_count = 0
        max_iters = 15
        
        while iter_count < max_iters and time.time() - start_time < budget_s - 50:
            iter_count += 1
            
            # Compute exact constants for checking improvement
            g = 1 - h_polish
            corr = np.correlate(h_polish, g, mode='full')
            # Find worst shift
            max_k = np.argmax(corr)
            # k is some offset; index max_k corresponds to shift (max_k - (N-1))
            k_worst = max_k - (N - 1)
            
            # Explore all 1 -> 0 moves that contribute to this term.
            # Indices that contribute to k_worst: i with i+k_worst < N
            indices_worst = [i for i in range(N) if i + k_worst < N]
            
            best_improvement = 0
            best_i, best_j = None, None
            delta_schedule = [0.05, 0.02, 0.01]
            search_delta = 0.05
            for d in delta_schedule:
                if time.time() - start_time > budget_s - 20:
                    break
                search_delta = d
                # Find candidate pairs (i,j) where i currently 1 and j 0
                for i in indices_worst:
                    j = i + k_worst
                    if j < N and h_polish[i] > 0.9 and h_polish[j] < 0.1:
                        # Consider moving delta from i to j
                        h_temp = h_polish.copy()
                        h_temp[i] -= search_delta
                        h_temp[j] += search_delta
                        h_temp = np.clip(h_temp, 0, 1)
                        
                        # Compute new C5
                        new_c5 = c5_of_h(h_temp)
                        
                        # Check improvement
                        if new_c5 < current_c5 - 1e-10:
                            improvement = current_c5 - new_c5
                            if improvement > best_improvement:
                                best_improvement = improvement
                                best_i = i
                                best_j = j
        
            if best_improvement > 0:
                # Accept the move
                h_polish[best_i] -= search_delta
                h_polish[best_j] += search_delta
                current_c5 = current_c5 - best_improvement
            else:
                break
    
    # --- Swap hill-climbing ---
    if time.time() - start_time < budget_s - 100:
        # Given h = h_polish, compute overlaps and do swaps
        h_result = h_polish.copy()
        current_c5 = c5_of_h(h_result)
        
        attempts = 0
        max_attempts = 30000
        while attempts < max_attempts and time.time() - start_time < budget_s - 30:
            attempts += 1
            
            # Try a random swap (with float p)
            i = np.random.randint(N)
            j = np.random.randint(N)
            
            if i == j:
                continue
                
            # Test whether swapping positions i, j helps
            h_swap = h_result.copy()
            h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
            
            # regenerate overlap half-vector via FFT (what.c5_of_h uses)
            # An efficient way to get max of full cross-correlation
            # Currently memory intensive :(
            c5_new = c5_of_h(h_swap)
            if c5_new < current_c5 - 1e-12:
                h_result = h_swap
                current_c5 = c5_new
    
    # --- Final bound ---
    final_c5 = c5_of_h(h_result)
    
    return (h_result, final_c5, N)
```
