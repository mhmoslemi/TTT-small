import numpy as np
import scipy.optimize as opt
import time
from scipy.interpolate import interp1d

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def project_to_feasible(h):
    """Ensure sum(h) = n/2 and h in [0,1]^n"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    # Project into feasible set with bounds [0,1] and sum = n/2
    # Simple 1D projection (delta distributed uniformly)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    # Distribute deficit
    # Roughly: we spread deficit to all points that are not at boundary
    # and not particularly sensitive to boundary value
    # Use a uniform adjustment
    adjustment = deficit / n
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """
    Run the improved numerical optimization to find a better upper bound for C5.
    Uses a low-rank SDP-style approach with smooth relaxations and deterministic swaps.
    """
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    
    # Dimensions
    n_points = 400  # Use a 400-point discretization
    dx = 2.0 / n_points
    rank = 6  # Low-rank factorization
    
    def fft_correlations(Y):
        """Compute the correlation sum for all shifts from Y using FFT."""
        # Y is (n, r)
        n, r = Y.shape
        total_c = np.zeros(n)
        for d in range(r):
            y_d = Y[:, d]
            # Circular correlation with reverse using FFT
            y_d_padded = np.concatenate([y_d, np.zeros(n)])
            y_d_conj_fft = np.fft.fft(y_d_padded)
            conv_fft = y_d_conj_fft * np.conj(y_d_conj_fft)
            c_d = np.real(np.fft.ifft(conv_fft))[:n]
            total_c += c_d
        return total_c

    def lse_surrogate(Y, tau):
        """Smooth log-sum-exp lower bound on min correlation."""
        c = fft_correlations(Y)
        return -tau * np.log(np.sum(np.exp(-c/tau)))

    def lse_gradient(Y, tau):
        """Gradient of the LSE surrogate with respect to Y."""
        c = fft_correlations(Y)  # (n,)
        exp_neg_c_over_tau = np.exp(-c/tau)  # (n,)
        s = np.sum(exp_neg_c_over_tau)
        p = exp_neg_c_over_tau / s  # (n,)
        n, r = Y.shape
        grad = np.zeros_like(Y)
        # Gradient term: sum over k of p_k * (Y_{i+k} + Y_{i-k})
        for d in range(r):
            y_d = Y[:, d]
            y_d_padded = np.concatenate([y_d, np.zeros(n)])
            p_padded = np.concatenate([p, np.zeros(n)])
            conv1 = np.real(np.fft.ifft(np.fft.fft(y_d_padded) * np.conj(np.fft.fft(p_padded))))[:n]
            conv2 = np.real(np.fft.ifft(np.fft.fft(p_padded) * np.conj(np.fft.fft(y_d_padded))))[:n]
            grad[:, d] = conv1 + conv2
        return grad

    def project_Y(Y):
        """Project Y to satisfy the box and mass constraints."""
        n, r = Y.shape
        h = np.sum(Y**2, axis=1)
        # Step 1: Clip norms to [0,1]
        norms = np.sqrt(np.sum(Y**2, axis=1))  # (n,)
        # If any normalize toward unit norm
        mask = norms > 1
        if np.any(mask):
            Y[mask] = Y[mask] / norms[mask][:, np.newaxis]
        # Step 2: Enforce mass
        total = np.sum(h)
        if abs(total) > 1e-10:
            alpha = np.sqrt((n/2)/total)
        else:
            alpha = 1.0
        Y *= alpha
        # For numerical accuracy / stability, make sure all Y have valid norms.
        norms = np.sqrt(np.sum(Y**2, axis=1))
        # Enforce final clipping to ensure progression
        clip_mask = norms > 1
        if np.any(clip_mask):
            Y[clip_mask] = Y[clip_mask] / norms[clip_mask][:, np.newaxis]
        return Y

    def potential_max_min(Y):
        """Return min correlation c_k for given Y."""
        c = fft_correlations(Y)
        return np.min(c)

    # Optimization parameters
    tau_values = [0.05, 0.025, 0.0125, 0.00625, 0.003125, 0.0015625]
    n_inner = 200  # iterations per temperature stage

    # Run multiple restarts
    restarts = 3
    final_Ys = []
    final_c5s = []
    final_hs = []

    for restart in range(restarts):
        if time.time() - start_time > budget_s - 10:
            break
        # Initialize Y from i.i.d. normal
        Y = np.random.normal(0, 1, (n_points, rank))
        Y = project_Y(Y)
        best_Y = Y.copy()
        best_c5 = np.inf

        # Continue over decreasing temperatures
        for tau in tau_values:
            if time.time() - start_time > budget_s - 10:
                break
            for i in range(n_inner):
                if time.time() - start_time > budget_s - 10:
                    break
                # Compute current potentials
                c = fft_correlations(Y)
                current_c5 = np.max(c)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_Y = Y.copy()
                # Compute gradient
                grad = lse_gradient(Y, tau)
                # Backtracking line search
                alpha = 1.0
                step = alpha * grad
                Y_new = Y - step
                Y_new = project_Y(Y_new)
                # Evaluate with a simple line search
                actual_c = fft_correlations(Y_new)
                actual_c5 = np.max(actual_c)
                while actual_c5 >= current_c5 and alpha > 1e-8:
                    alpha *= 0.5
                    step = alpha * grad
                    Y_new = Y - step
                    Y_new = project_Y(Y_new)
                    actual_c = fft_correlations(Y_new)
                    actual_c5 = np.max(actual_c)
                Y = Y_new
                # Early termination based on improvement
                if i > 10 and abs(actual_c5 - current_c5) < 1e-7:
                    break

        # Final check of best Y
        h = np.sum(best_Y**2, axis=1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        c = fft_correlations(best_Y)
        this_c5 = np.max(c)
        final_Ys.append(best_Y)
        final_c5s.append(this_c5)
        final_hs.append(h)

    # Select best according to the ensured C5 values
    best_idx = np.argmin(final_c5s)
    Y_final = final_Ys[best_idx]
    h_final = final_hs[best_idx]
    c5_final = final_c5s[best_idx]

    # Convert to binary mask
    h = np.clip(h_final, 0, 1)
    h_sorted_idx = np.argsort(h)[::-1]
    mask = np.zeros(n_points)
    mask[h_sorted_idx[:n_points//2]] = 1
    mask = mask.astype(np.float64)
    c5_binary = c5_of_h(mask)
    
    # Use the better between continuous and binary
    if c5_binary < c5_final:
        h_result = mask
        c5_result = c5_binary
    else:
        h_result = h_final
        c5_result = c5_final

    # Deterministic swap polishing on the binary version
    if c5_result > 0.3800:  # Only if we are suboptimal already
        mask = np.zeros(n_points)
        mask[h_sorted_idx[:n_points//2]] = 1
        mask = mask.astype(np.float64)
        max_c = np.inf
        while max_c >= c5_result and time.time() - start_time < budget_s - 20:
            # Identify the worst-case shift k* that maximizes overlap
            c_local = fft_correlations(Y_final)
            k_max = np.argmax(c_local)
            # Get 1s and 0s for the shift k_max
            mask_local = mask.astype(np.int32)
            k = k_max - n_points//2  # shift in range [-n/2, n/2]
            not_identical = np.any(mask_local != np.roll(mask_local, k))
            if not not_identical:  # If all match, no need to swap
                break
            # Find all mismatched pairs
            mismatched_1s = np.where((mask_local == 1) & (np.roll(mask_local, k) == 0))[0]
            mismatched_0s = np.where((mask_local == 0) & (np.roll(mask_local, k) == 1))[0]
            if len(mismatched_1s) == 0 or len(mismatched_0s) == 0:
                break
            # Select the best swap
            min_improvement = np.inf
            best_i, best_j = None, None
            for i in mismatched_1s:
                for j in mismatched_0s:
                    # Compute change in correlation k
                    old = (mask_local[i] * mask_local[i+k] + mask_local[i-k] * mask_local[i]
                           + mask_local[j] * mask_local[j+k] + mask_local[j-k] * mask_local[j])
                    new = (0 * mask_local[i+k] + mask_local[i-k] * 0
                           + 1 * mask_local[j+k] + mask_local[j-k] * 1)
                    change = new - old
                    # Effect on correlation for all k
                    new_c = c_local.copy()
                    new_c[k] = new_c[k] - old + new  # We only consider changing this k
                    # But ignore other ks to improve other kinds of objective too
                    # Update all correlations for the full problem – this is costly
                    # Instead, only update the max
                    new_max = np.max(new_c)
                    improvement = max_c - new_max
                    if improvement > min_improvement:
                        min_improvement = improvement
                        best_i, best_j = i, j
            if best_i is not None and min_improvement > 0:
                # Perform the swap
                mask[best_i] = 0
                mask[best_j] = 1
                # Re-evaluate the c and max!
                # Approximate again...
                pass
            else:
                break  # no improvement
        # Final evaluate after swaps
        c5_result = c5_of_h(mask)
        h_result = mask

    # Final upsampling and sub-gradient polishing
    if time.time() - start_time < budget_s - 20:
        # Upsample
        x_orig = np.linspace(0, 2, n_points)
        x_up = np.linspace(0, 2, 2*n_points)
        h_up = np.interp(x_up, x_orig, h_result)
        # Renormalize
        h_up = h_up * (2*n_points / 2) / np.sum(h_up)
        # One bonus pass of subgradient descent
        for _ in range(30):
            if time.time() - start_time > budget_s - 5:
                break
            c_local = np.correlate(h_up, 1 - h_up, mode='full')
            k_max = np.argmax(c_local)
            dx_new = 2.0 / (2 * n_points)
            # Gradient is a vector with entries:
            # d/dh_i = 1 - h_{i+k_max} if i+k_max valid
            # d/dh_i = -h_{i-k_max} if i-k_max valid
            grad = np.zeros_like(h_up)
            for i in range(len(h_up)):
                if i + k_max < len(h_up):
                    grad[i] += 1 - h_up[i + k_max]
                if i - k_max >= 0:
                    grad[i] += -h_up[i - k_max]
            # Step and project
            step_size = 0.1 * dx_new
            h_up -= step_size * grad
            h_up = np.clip(h_up, 0, 1)
            h_sum = np.sum(h_up)
            h_up = h_up * (2*n_points / 2) / h_sum
        c5_result = c5_of_h(h_up)
        h_result = h_up

    return (h_result, c5_result, len(h_result))
