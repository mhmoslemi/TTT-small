import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import cvxpy as cp
from scipy.fft import fft, ifft
import math

def solve_spectral_lp(n):
    """Solve the spectral LP relaxation for maximum minimum autocorrelation."""
    # Variables: power spectrum p_m and lower bound t
    p = cp.Variable(n)
    t = cp.Variable(1)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    
    # Constraints
    constraints = [p >= 0, p <= n]
    
    # Sum constraint
    constraints.append(cp.sum(p) == n * n / 2)
    
    # Autocorrelation constraints for k = 1,...,n-1
    for k in range(1, n):
        # A_k = (1/n) * sum_m p_m * cos(2*pi*m*k/n)
        cos_term = np.cos(2 * np.pi * np.arange(n) * k / n)
        constraints.append(cp.sum(p * cos_term) >= n * t)
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if prob.status == cp.OPTIMAL:
        p_opt = p.value
        t_opt = t.value
        return p_opt, t_opt
    else:
        return None, None

def gerchberg_saxton(p_spectrum, n, max_iter=200):
    """Perform Gerchberg-Saxton phase retrieval to construct h with specified power spectrum."""
    # Random initial phases
    phi = np.random.uniform(0, 2 * np.pi, n)
    
    # Initial complex spectrum
    spectrum = np.sqrt(p_spectrum) * np.exp(1j * phi)
    
    for _ in range(max_iter):
        # Inverse FFT to get real-valued signal
        h = np.real(ifft(spectrum))
        
        # Amplitude projection
        y = np.clip(h, 0, 1)
        
        # Water-filling to enforce sum constraint
        current_sum = np.sum(y)
        diff = n / 2 - current_sum
        
        if abs(diff) > 1e-9:
            # Simple water-filling: add constant lambda
            y = y + diff / n
            y = np.clip(y, 0, 1)
        
        # Forward FFT
        y_spectrum = fft(y)
        
        # Spectral projection: keep magnitude from p_spectrum
        magnitudes = np.abs(y_spectrum)
        spectrum_new = np.sqrt(p_spectrum) * np.exp(1j * np.angle(y_spectrum))
        spectrum = spectrum_new
        # Check if this converged (optional, but debugging can help)
        
    return y  # returning the projected signal from GS

def project_simplex_box(h, n):
    """Project h to [0,1] box and enforce sum constraint using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-9:
        return h
    
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def binary_sharpen(h, n, mu_vals=[1e-4, 1e-3, 1e-2]):
    """Perform binary sharpening by projected gradient steps with penalty."""
    # Define penalty
    def penalty(h):
        return np.sum(h * (1 - h))
    
    for mu in mu_vals:
        # Simple gradient descent with penalty
        for _ in range(5):
            # Gradient of objective with penalty
            grad = -h + (1 - h)  # d/dh (h*(1-h)) = 1 - 2*h
            grad_penalty = mu * (1 - 2 * h)
            grad_total = grad + grad_penalty
            
            # Update with small stepsize
            stepsize = 1e-3
            h_new = h - stepsize * grad_total
            h_new = project_simplex_box(h_new, n)
            h = h_new
    
    # Final rounding
    h = np.clip(h, 0, 1)
    h_rounded = np.zeros_like(h)
    for i in range(n):
        if h[i] < 0.01:
            h_rounded[i] = 0.0
        elif h[i] > 0.99:
            h_rounded[i] = 1.0
        else:
            h_rounded[i] = h[i]
    
    h_rounded = project_simplex_box(h_rounded, n)
    return h_rounded

def compute_worst_shift_gradient(h, n, dx=2.0 / n):
    """Compute gradient of the worst shift integral with respect to h."""
    # WLOG we compute gradient wrt any vector indexing to compute the shift
    # As long as h*(1-h_k) integral is computed usin mode="full" correlation
    # Kept consistent with objective evaluated below.
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    
    # Indices in 'full' correlation array
    # The real indices are from -n+1 to n-1, offset by n-1 => 0 to 2n-2
    # The shift index is max_idx - (n-1) in [-n+1, n-1]
    shift_idx = max_idx - (n - 1)
    
    # Overload the definition of gradient to be just h's gradient wrt big shift
    grad = np.zeros_like(h)
    for i in range(n):
        # d/dh_i [int h_j(1-h_{j+k}) dx] = (1 - h_{i+k}) - h_{i-k}
        # The latter part, h_{i-k}, and former part, h_{i+k}, requires careful indexing
        # Consider a circular array
        j_plus = (i + shift_idx) % n
        j_minus = (i - shift_idx) % n
        grad[i] = (1 - h[j_plus]) - h[j_minus]
    
    return grad, shift_idx

def projected_gradient_worst_shift(h, n, dx=2.0 / n, max_iter=3000, max_time=30):
    """Projected gradient descent targeting the worst shift."""
    start_time = time.time()
    
    # Make a copy to avoid modifying input
    h = h.copy()
    
    c5_prev = evaluate_c5(h)
    
    for iter_no in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        grad, shift_idx = compute_worst_shift_gradient(h, n, dx)
        
        # Try step
        alpha = 1.0
        h_trial = h - alpha * grad
        
        # Project
        h_trial = project_simplex_box(h_trial, n)
        c5_trial = evaluate_c5(h_trial)
        
        # Backtracking
        if c5_trial < c5_prev:
            h = h_trial
            c5_prev = c5_trial
        else:
            # Try smaller step
            for _ in range(10):
                alpha /= 2
                h_trial = h - alpha * grad
                h_trial = project_simplex_box(h_trial, n)
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < c5_prev:
                    h = h_trial
                    c5_prev = c5_trial
                    break
            
    return h

def swap_anneal(h, n, dx=2.0 / n, max_time=40):
    """Stochastic swap refinement using simulated annealing."""
    start_time = time.time()
    h = h.copy()
    c5_prev = evaluate_c5(h)
    
    temp = 1e-3
    cooling_rate = 0.98
    min_temp = 1e-6
    
    iterations = 0
    unchanged = 0
    max_unchanged = 200
    
    while time.time() - start_time < max_time and unchanged < max_unchanged:
        # Pick two indices
        i, j = np.random.choice(n, size=2, replace=False)
        if h[i] <= 0 or h[j] >= 1:
            continue
            
        # Transfer amount
        delta = np.random.uniform(0, min(h[i], 1 - h[j], 0.02))
        
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Project
        h_new = project_simplex_box(h_new, n)
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        if c5_new < c5_prev:
            h = h_new
            c5_prev = c5_new
            unchanged = 0
        else:
            # Accept with probability exp(-(C5_new - C5_prev) / temp)
            prob = np.exp(-(c5_new - c5_prev) / temp)
            if np.random.rand() < prob:
                h = h_new
                c5_prev = c5_new
                unchanged = 0
            else:
                unchanged += 1
                
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
            
        iterations += 1
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 650
    dx = 2.0 / n_points
    
    # Stage 1: Spectral LP
    p_opt, t_opt = solve_spectral_lp(n_points)
    if p_opt is None or t_opt is None:
        # Fallback to some heuristic
        h = np.full(n_points, 0.5)
        return (h, evaluate_c5(h), n_points)
    
    # Stage 2: Gerchberg-Saxton Phase Retrieval
    h_gs = gerchberg_saxton(p_opt, n_points, max_iter=200)
    if h_gs is None:
        h = np.full(n_points, 0.5)
        return (h, evaluate_c5(h), n_points)
    
    # Stage 3: Feasibility Projection and Binary Sharpening
    h_feasible = project_simplex_box(h_gs, n_points)
    h_sharpened = binary_sharpen(h_feasible, n_points)
    
    # Stage 4: Local Worst-Shift Refinement
    h_refined = projected_gradient_worst_shift(h_sharpened, n_points, dx=dx, max_iter=3000, max_time=150)
    
    # Stage 5: Swap Annealing
    h_final = swap_anneal(h_refined, n_points, dx=dx, max_time=budget_s - (time.time() - start_time) - 5)
    
    # Final evaluation
    final_c5 = evaluate_c5(h_final)
    
    return (h_final, final_c5, n_points)
