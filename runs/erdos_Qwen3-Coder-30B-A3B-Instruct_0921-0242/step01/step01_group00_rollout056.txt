import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def solve_spectral_lp(n):
    """Solve the spectral relaxation LP for the maximum minimum autocorrelation."""
    # Variables
    p = cp.Variable(n)
    t = cp.Variable()
    
    # Constraints
    constraints = [p >= 0]
    # Upper bound on p
    constraints += [p <= n]
    
    # Parseval's identity constraint
    constraints += [cp.sum(p) == n * n / 2]  # n * sum(h_i^2) = n * (n/2) when sum(h) = n/2 and h_i in [0,1]
    
    # Autocorrelation constraints
    for k in range(1, n):
        # Compute cosine terms
        cos_terms = [np.cos(2 * np.pi * k * m / n) for m in range(n)]
        constraints += [cp.sum(p * cos_terms) >= t * n]
    
    # Objective
    objective = cp.Maximize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return p.value, t.value
    except:
        pass
        
    # If LP fails, fallback to using a bound
    p_fallback = np.full(n, n/2)
    p_fallback[0] = n/2
    t_fallback = 0.0
    return p_fallback, t_fallback

def gerchberg_saxton(p_star, n, max_iter=200):
    """Perform Gerchberg-Saxton phase retrieval."""
    # Initialize with random phases
    phi = np.random.uniform(0, 2*np.pi, n)
    x = np.random.rand(n)
    
    # Ensure correct sum
    x_sum = np.sum(x)
    delta = n/2 - x_sum
    if abs(delta) > 1e-10:
        x = x + delta/n
        x = np.clip(x, 0, 1)
    
    # HF projection
    def project_simplex_box(x_vec):
        x_proj = np.clip(x_vec, 0, 1)
        s = np.sum(x_proj)
        delta = n/2 - s
        if abs(delta) < 1e-10:
            return x_proj
        # Binary search for lambda
        a, b = -10, 10
        for _ in range(50):
            lam = (a + b) / 2
            x_test = np.clip(x_proj + lam, 0, 1)
            s_test = np.sum(x_test)
            if s_test < n/2:
                a = lam
            else:
                b = lam
        lam = (a + b) / 2
        x_final = np.clip(x_proj + lam, 0, 1)
        return x_final

    for _ in range(max_iter):
        # Inverse DFT
        X = np.fft.fft(x)  # complex
        # Replace magnitudes with target
        X_mag = np.sqrt(p_star)
        X_phase = X / np.abs(X)
        X = X_mag * X_phase
        # Back to time domain
        x = np.real(np.fft.ifft(X))
        # Project in time domain
        x = project_simplex_box(x)
    
    return x

def project_simplex_box(x):
    """Project x to have sum=n/2 and entries in [0,1]."""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    s = np.sum(x)
    delta = 0.5 * len(x) - s
    if abs(delta) < 1e-10:
        return x
    # Binary search for lambda
    a, b = -10, 10
    for _ in range(50):
        lam = (a + b) / 2
        x_test = np.clip(x + lam, 0, 1)
        s_test = np.sum(x_test)
        if s_test < 0.5 * len(x):
            a = lam
        else:
            b = lam
    lam = (a + b) / 2
    x_final = np.clip(x + lam, 0, 1)
    return x_final

def binary_sharpen(h, mu=1e-3):
    """Binary sharpening step."""
    def psi(h):
        return mu * np.sum(h * (1 - h))
    
    # Gradient of psi
    def grad_psi(h):
        return mu * (1 - 2 * h)
    
    # Simple projected gradient descent for psi
    for _ in range(100):
        grad = grad_psi(h)
        h_new = h - 0.01 * grad
        h_new = project_simplex_box(h_new)
        if psi(h_new) < psi(h):
            h = h_new
        else:
            break
    
    # Round near 0 or 1
    for i in range(len(h)):
        if h[i] < 0.01:
            h[i] = 0.0
        elif h[i] > 0.99:
            h[i] = 1.0
    h = project_simplex_box(h)
    return h

def compute_c5(h):
    # Compute dx
    n = len(h)
    dx = 2.0 / n
    # Use scipy to compute the full correlation
    c = np.correlate(h, 1 - h, mode="full")
    # Extract the max characteristic (excluding k=0)
    max_corr = np.max(c)
    # Scale by dx
    return max_corr * dx

def gradient_worst_shift(h):
    # Compute dx and identify the worst shift
    n = len(h)
    dx = 2.0 / n
    # Full correlation
    c = np.correlate(h, 1 - h, mode="full")
    # Location of max (excluding 0 shift)
    max_loc = np.argmax(c[1:]) + 1  # because we skip first 0
    # Retrieve shift index k from max_loc (in [0, 2n-2])
    if max_loc > n - 1:
        k = max_loc - (n - 1)
    else:
        k = - (n - 1) + max_loc
    
    # Gradient at each point
    grad = np.zeros_like(h)
    for i in range(n):
        # left term: -(1 - h[i+k])
        j = (i + k) % n
        grad[i] -= (1 - h[j])
        # right term: h[i-k]
        j = (i - k) % n
        grad[i] += h[j]
    grad *= dx
    return grad

def projected_gradient_descent(h, max_iter=3000, tol=1e-8):
    """Minimize C5 by gradient descent on the worst shift."""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    c0 = compute_c5(h)
    c_current = c0
    # Track last 5 values to check for stagnation
    recent_c = [c_current]*5
    
    for i in range(max_iter):
        if time.time() - start_time > 500:
            break
        grad = gradient_worst_shift(h)
        # Try different step sizes
        step_lr = 1.0
        for _ in range(10):
            h_try = h - step_lr * grad
            h_try = project_simplex_box(h_try)
            c_try = compute_c5(h_try)
            if c_try < c_current:
                h = h_try
                c_current = c_try
                break
            step_lr /= 2.0
        # Accept if improvement
        if c_current < c0:
            c0 = c_current
            recent_c = recent_c[1:] + [c_current]
            # Check if stagnated
            if abs(np.max(recent_c) - np.min(recent_c)) < tol:
                break
        old_c = c_current
        c_current = compute_c5(h)
        if abs(old_c - c_current) < tol:
            break
    
    return h, c_current

def swap_anneal(h, max_iter=10000):
    """Simulated annealing swap refinement."""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    c0 = compute_c5(h)
    T = 0.001
    decay = 0.9995
    c_current = c0
    
    for _ in range(max_iter):
        if time.time() - start_time > 50:
            break
        # Pick two indices
        i, j = np.random.randint(0, n, 2)
        if h[i] <= 0 or h[j] >= 1:
            continue
        # Try swapping small amount
        delta = np.random.uniform(0, min(h[i], 1 - h[j], 0.02))
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = project_simplex_box(h_new)
        c_new = compute_c5(h_new)
        if c_new < c_current:
            h = h_new
            c_current = c_new
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c_new - c_current)/T):
                h = h_new
                c_current = c_new
        T *= decay
    
    return h, c_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 650  # slightly larger than previous to get better resolution
    dx = 2.0 / n
    
    # Stage 1: Spectral Relaxation LP
    p_star, t_star = solve_spectral_lp(n)
    if t_star <= 0:
        # Fallback if LP is too tight or failed
        p_star = np.full(n, n/2)
        t_star = 0.0
    
    # Compute C5 bound
    c5_bound_relax = (2/n) * (1 - t_star)
    
    if c5_bound_relax < 0.3807:
        # Exact certificate, return relaxation result
        x = np.random.rand(n)
        x = project_simplex_box(x)  # ensure feasible
        return (x, c5_bound_relax, n)
    
    # Stage 2: Gerchberg-Saxton
    h = gerchberg_saxton(p_star, n, max_iter=200)  # Fast enough
    h = project_simplex_box(h)
    
    # Stage 3: Feasibility projection and binary sharpening
    h = project_simplex_box(h)
    h = binary_sharpen(h)
    
    # Stage 4: Local worst-shift refinement
    h, c5_local = projected_gradient_descent(h, max_iter=3000)
    
    # Stage 5: Swap annealing
    h, c5_final = swap_anneal(h, max_iter=10000)
    
    return (h, c5_final, n)
