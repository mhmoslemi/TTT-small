def binarize_h(h, target_sum):
    """Convert continuous h to binary vector with exact sum."""
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[target_sum - 1]
    h_binary = (h > threshold).astype(np.uint8)
    if np.sum(h_binary) > target_sum:
        # Reduce to exact sum
        indices = np.where(h_binary == 1)[0]
        for idx in indices[target_sum:]:
            h_binary[idx] = 0
    elif np.sum(h_binary) < target_sum:
        # Increase to exact sum
        indices = np.where(h_binary == 0)[0]
        for idx in indices[:target_sum - np.sum(h_binary)]:
            h_binary[idx] = 1
    return h_binary

def c5_of_h(h, dx):
    """Fast computation of C5 using FFT."""
    g = 1 - h
    corr = fftconvolve(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def update_convolution_incremental(corr, h, i1, i0, h_old):
    """Update correlation efficiently after a swap."""
    n = len(h)
    # Old contributions to corr
    corr_old_i1 = np.zeros(2*n - 1)
    corr_old_i0 = np.zeros(2*n - 1)
    corr_old_i1[i1:i1+n] = h_old
    corr_old_i0[i0:i0+n] = h_old
    corr_old_i0[i0:i0+n] = 1 - h_old

    # New contributions to corr
    corr_new_i1 = np.zeros(2*n - 1)
    corr_new_i0 = np.zeros(2*n - 1)
    corr_new_i1[i1:i1+n] = h
    corr_new_i0[i0:i0+n] = h
    corr_new_i0[i0:i0+n] = 1 - h

    # Compute differences
    diff = (corr_new_i1 - corr_old_i1) + (corr_new_i0 - corr_old_i0)

    # Update corr
    corr = corr + diff
    return corr

def compute_corr_vector(h):
    """Compute centered correlation vector."""
    g = 1 - h
    corr = fftconvolve(h, g, mode='full')
    return corr
