import numpy as np
import time
from scipy.signal import fftconvolve
from scipy.optimize import minimize
import math

def c5_of_h(h, dx):
    """Compute C5 for given h using FFT-based convolution."""
    g = 1 - h
    corr = fftconvolve(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_constraints(h, n_points, target_sum):
    """Project h onto sum constraint and box constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Distribute proportionally
    weights = np.ones_like(h)
    weights[np.where(h <= 0)] = 0
    weights[np.where(h >= 1)] = 0
    if np.sum(weights) > 0:
        h += diff * weights / np.sum(weights)
        h = np.clip(h, 0, 1)
    return h

        """Convert binary vector to continuous step function."""
        # Convert to continuous
        h = np.zeros(n_points)
        # Fill in constant intervals based on binary values
        ones = np.where(h_bin == 1)[0]
        if len(ones) > 0:
            # Group consecutive 1s into intervals
            intervals = []
            start = ones[0]
            prev = start
            for i in range(1, len(ones)):
                if ones[i] == prev + 1:
                    prev = ones[i]
                else:
                    intervals.append((start, prev))
                    start = ones[i]
                    prev = start
            intervals.append((start, prev))
            for s, e in intervals:
                h[s:e+1] = 1
        return h

        # 2. Random balanced binary
        random_h = np.zeros(n)
