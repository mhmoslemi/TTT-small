import numpy as np
import time
from scipy.signal import correlate
from scipy.linalg import eigh
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Compute C5 for given h using full convolution."""
    dx = 2.0 / len(h)
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    # This should be a function that returns a valid starting array
    # For now, reproduce the initial construction used in earlier version
    # You may need to adjust if a different one is known or referenced
    n = 300
    h = np.zeros(n)
    h[:n//2] = 1
    return h / np.sum(h) * (n / 2)

def create_shift_matrices(n, max_k):
    """Create sparse CSR shift matrices S_k for k in [-max_k, max_k]."""
    S_k_list = []
    k_list = []
    for k in range(-max_k, max_k + 1):
        data = []
        row = []
        col = []
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                data.append(1)
                row.append(i)
                col.append(j)
        S_k = np.array(data), (np.array(row), np.array(col)), (n, n)
        S_k_list.append(S_k)
        k_list.append(k)
    return S_k_list, k_list

def solve_sdp_grid(n, K=180):
    """
    Solve SDP relaxation for given grid size n.
    Returns the optimal t* from SDP.
    """
    m = (n + 1) // 2  # reduced size due to symmetry
    X = Semidef(m)
    
    # zero-sum constraint via trace(11^T X) = 0
    # This ensures that the sum of row sums is 0
    ones = np.ones(m)
    zero_sum = cp.sum(cp.multiply(ones, cp.sum(X, axis=1))) == 0
    
    # Create shift matrices
    S_k_list, k_list = create_shift_matrices(m, K)
    
    t = cp.Variable()
    
    # Constraints: t >= n/4 - 0.25 * tr(S_k * X) for each shift
    constraints = []
    constraints.append(X >> 0)  # Positive semidefinite
    for i in range(m):
        constraints.append(X[i, i] == 1)
    constraints.append(zero_sum)
    
    for S_k in S_k_list:
        S_k_sparse = S_k[0], (S_k[1][0], S_k[1][1]), S_k[2]
        # For sparse matrix conversion:
        # Reconstruct dense version for sum: tr(S_k * X)
        # We use a slower approach for clarity (only works for small m)
        if m <= 150:
            # Create a dense version
            S_dense = np.zeros((m,m))
            for i in range(len(S_k_sparse[1][0])):
                row, col = S_k_sparse[1][0][i], S_k_sparse[1][1][i]
                S_dense[row, col] = S_k_sparse[0][i]
            trace_term = cp.trace(S_dense @ X)
        else:
            # For larger dimensions, we do not compute dense S as it's memory-heavy
            # But given practical size we'll avoid solving very large SDPs in one go
            print("SDP too large to form dense matrix...")
            return None, None
            
        constraints.append(t >= n/4 - 0.25 * trace_term)
        
    # Minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        # Try with default solver first - SCS is good for sparse problems
        prob.solve(solver=SCS, eps=1e-6, max_iters=5000)
        return t.value, X.value
    except:
        # Fallback to ECOS for robustness if needed
        try:
            prob.solve(solver=ECOS, feastol=1e-6, reltol=1e-6)
            return t.value, X.value
        except:
            return None, None

def gaussian_rounding(X_hat, k):
    """Round the SDP solution to binary vector."""
    # Eigen-decompose: X = V D V^T
    _, V = eigh(X_hat)
    V = V[:, ::-1]  # Sort descending
    # Generate random Gaussian vector
    g = np.random.randn(V.shape[0])
    y = V.T @ g  # Should be Gaussian with covariance X_hat
    
    # Round to binary (sign of y_i)
    x = np.sign(y)
    
    # Adjust to have zero sum
    s = np.sum(x)
    if abs(s) > 1e-6:
        # Try to flip signs to get zero sum
        to_flip = int(abs(s)/2)
        if to_flip == 0:
            to_flip = 1
        idxs_abs = np.argsort(np.abs(y))
        if s > 0:
            # We have too many +1's, flip -1's to +1
            flip_ids = idxs_abs[:to_flip]
            x[flip_ids] = -x[flip_ids]
        else:
            # We have too many -1's, flip +1's to -1
            flip_ids = idxs_abs[:to_flip]
            x[flip_ids] = -x[flip_ids]
    
    # Reconstruct full sequence using symmetry
    x_full = np.concatenate([x, x[::-1]])
    x_full = x_full[:k]
    return x_full

def compute_correlation_sparse(h):
    """Efficiently compute max correlation including boundary handling."""
    # Easier to use full convolution
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr

def update_correlation_fast(H, c, i0, i1):
    """
    Update the correlation array after swapping H[i0] and H[i1].
    Assumes boolean (0,1) inputs.
    c = correlate(H, 1-H, mode='full')
    """
    n = len(H)
    # This will make changes to the full correlation array easily.
    # To optimize it, we compute the biases at shifts due to a swap
    old_c = c.copy()

    for k in range(2 * n - 1):
        # These changes occur to correlation due to h[i0] and h[i1]
        # When h[i0] was 1 and now 0,
        # and h[i1] was 0 and now 1.
        # Then:
        # c[k] += delta_k
        # where delta_k = (old_h[i0]*g[i0+k] + old_h[i1]*g[i1+k]) - (new_h[i0]*g[i0+k] + new_h[i1]*g[i1+k])
        old_h_i0 = H[i0]
        old_h_i1 = H[i1]
        new_h_i0 = 1 - old_h_i0
        new_h_i1 = 1 - old_h_i1
        if k < n:
            # case f(i0)=0, f(i1)=1
            g_i0 = 1 - H[i0]
            g_i1 = 1 - H[i1]
            c[k] += (g_i0 * (new_h_i0 + new_h_i1 - old_h_i0 - old_h_i1)) 
        else:
            # In general just do
            # Copy old and recompute these specific cases
            # Sacrifice for accuracy
            pass
            
    # Use full correlation for final checks in small cases
    g = 1 - H
    old_correlation = old_c
    new_correlation = correlate(H, g, mode='full')
    c[:] = new_correlation
    return c

def hill_climb_h(X_rounded, n):
    """Perform deterministic hill-climbing refinement."""
    # The actual problem defines H with values in [0,1], so only consider the binary values
    # For now approximating with true binary_h
    h = (X_rounded + 1) / 2
    h = h[:n]  # we're working with full sequence now
    h = np.clip(h, 0, 1)
    
    # Compute initial correlation
    g = 1 - h
    correlation = correlate(h, g, mode='full')
    max_corr = np.max(correlation)
    best_h = h.copy()
    best_c5 = max_corr * (2.0 / n)
    
    # Local search using swaps
    max_swaps = 3000
    swaps_done = 0
    
    while swaps_done < max_swaps:
        swaps_done += 1
        
        # Choose two indices to swap
        ones_idx = np.nonzero(h)[0]
        zeros_idx = np.nonzero(1 - h)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i0 = np.random.choice(ones_idx)
        i1 = np.random.choice(zeros_idx)
        
        # Make a copy for the swap
        new_h = h.copy()
        new_h[i0] = 1 - new_h[i0]
        new_h[i1] = 1 - new_h[i1]

        # Recompute correlation efficiently
        gc = 1 - new_h
        new_correlation = correlate(new_h, gc, mode='full')
        new_max_corr = np.max(new_correlation)
        new_c5 = new_max_corr * (2.0 / n)

        if new_c5 < best_c5:
            h = new_h
            best_h = new_h
            best_c5 = new_c5
            correlation = new_correlation
        else:
            pass  # Reject the move

    # Final validation
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 200

    # Grid sizes to try
    grids = [
        {"n": 140, "K": 100, "name": "coarse"},
        {"n": 200, "K": 180, "name": "medium"},
        {"n": 260, "K": 200, "name": "fine"}
    ]

    for i, grid in enumerate(grids):
        grid_n = grid["n"]
        grid_K = grid["K"]

        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 30:
            break

        # Solve SDP on this grid
        print(f"Solving SDP for grid {grid['name']} with n={grid_n}")
        t_opt, X_opt = solve_sdp_grid(grid_n, grid_K)

        if t_opt is None or X_opt is None:
            print(f"SDP failed for grid {grid['name']}")
            continue

        # Rounding step
        print(f"Rounding for {grid['name']}...")
        round_samples = 200
        best_rounded_c5 = float('inf')
        best_rounded_h = None

        for _ in range(round_samples):
            if time.time() - start_time > budget_s - 10:
                break
            try:
                x_rounded = gaussian_rounding(X_opt, grid_n)
                h_rounded = (x_rounded + 1) / 2
                c5_val = evaluate_erdos_solution(h_rounded)
                if c5_val < best_rounded_c5:
                    best_rounded_c5 = c5_val
                    best_rounded_h = h_rounded
            except Exception as e:
                continue

        if best_rounded_h is None:
            continue

        # Local refinement
        print(f"Refining {grid['name']}...")
        try:
            refined_h, refined_c5 = hill_climb_h(best_rounded_h, grid_n)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
                best_n = grid_n
        except Exception as e:
            continue

        # Early stopping if suffices target
        if best_c5 <= 0.3807:
            break

    if best_h is None:
        # Fallback to initial_h_values + refinement
        try:
            h_base = initial_h_values()
            h_base = h_base[:best_n]
            h_base = np.clip(h_base, 0, 1)
            c5 = evaluate_erdos_solution(h_base)
            best_c5 = c5
            best_h = h_base.copy()
        except:
            pass

    return (best_h, best_c5, best_n)
