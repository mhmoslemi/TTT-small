import numpy as np
import time
import cvxpy as cp
from scipy import sparse
from scipy.signal import correlate

def build_Q(k, n):
    # returns sparse symmetric PSD matrix for term h_i h_{i+k}
    if k == 0:
        return sparse.csr_matrix((n, n))
    data = np.ones(n - abs(k))
    rows = np.arange(n - abs(k))
    cols = rows + k if k >= 0 else rows - k
    Q = sparse.csr_matrix((data, (rows, cols)), shape=(n, n))
    return 0.5 * (Q + Q.T)

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def solve_qp(n, Kmax, warm_h=None):
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    for k in range(-Kmax, Kmax+1):
        Q = build_Q(k, n)
        constraints.append(cp.quad_form(h, Q) >= n/2 - t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    if warm_h is not None:
        h.value = warm_h
    prob.solve(solver=cp.OSQP, verbose=False, max_iter=5000, eps_abs=1e-9)
    return h.value, t.value

def binary_refine(h_frac, n, Kmax, dx, time_limit):
    # start with top n/2 entries =1
    idx = np.argsort(-h_frac)[:n//2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1
    best = c5_of_h(h_bin, dx)
    best_h = h_bin.copy()
    
    # precompute correlation array for current h_bin
    corr = correlate(h_bin, 1 - h_bin, mode='full')
    max_corr = np.max(corr)
    
    while time.time() < time_limit:
        i1 = np.random.choice(np.where(h_bin == 1)[0])
        i0 = np.random.choice(np.where(h_bin == 0)[0])
        # tentative swap
        h_bin[i1] = 0
        h_bin[i0] = 1
        
        # Update correlation efficiently
        # Only need to update terms involving i1 and i0
        new_corr = corr.copy()
        k_offset = len(h_bin) - 1
        for k in range(-Kmax, Kmax+1):
            new_corr[k + k_offset] += h_bin[i0] - h_bin[i1]
            if i1 + k >= 0 and i1 + k < n:
                new_corr[k + k_offset] -= h_bin[i1] - h_bin[i0]
            if i0 - k >= 0 and i0 - k < n:
                new_corr[k + k_offset] -= h_bin[i0] - h_bin[i1]
        
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best:
            best = new_c5
            best_h = h_bin.copy()
            corr = new_corr
            max_corr = np.max(corr)
        else:
            # revert
            h_bin[i1] = 1
            h_bin[i0] = 0
    return best_h, best
