import numpy as np
import scipy.optimize as opt
import time
from scipy.interpolate import interp1d
from scipy.fft import fft, ifft
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 400  # Recommended increasing size for better resolution
    dx = 2.0 / n_points

    # FFT helpers
    fft_len = 2 * n_points

    def compute_correlations(Y):
        c = np.zeros((n_points, n_points))
        for d in range(Y.shape[1]):
            y_d = Y[:, d]
            conv = ifft(fft(y_d, fft_len) * np.conj(fft(y_d, fft_len))).real[:n_points]
            c += conv.reshape(-1, 1)
        return c.sum(axis=1)

    def log_sum_exp_surrogate(Y, tau=0.05):
        c = compute_correlations(Y)
        p = np.exp(-c / tau)
        p = p / np.sum(p)
        return -tau * np.log(np.sum(p))

    def gradient_log_sum_exp(Y, tau=0.05):
        c = compute_correlations(Y)
        p = np.exp(-c / tau)
        p = p / np.sum(p)
        grad = np.zeros_like(Y)
        for d in range(Y.shape[1]):
            y_d = Y[:, d]
            # Convolution with p to get sum of weighting terms
            p_conv = ifft(fft(p, fft_len) * fft(y_d, fft_len)).real[:n_points]
            p_conv_rev = ifft(fft(p, fft_len) * fft(y_d[::-1], fft_len)).real[:n_points]
            grad[:, d] = p_conv + p_conv_rev
        # proper normalization and sign
        grad = grad * (-1.0 / tau)
        return grad

    def project_to_feasible(Y):
        h = np.sum(Y**2, axis=1)
        S = np.sum(h)
        alpha = np.sqrt((n_points / 2) / S)
        Y = Y * alpha  # Uniform scaling
        # Clip norms to 1 if needed
        for i in range(n_points):
            norm = np.linalg.norm(Y[i])
            if norm > 1:
                Y[i] = Y[i] / norm
        h = np.sum(Y**2, axis=1)
        return Y

    def restart_few():
        Y = np.random.randn(n_points, 6)
        Y = project_to_feasible(Y)
        return Y

    # Continue until timeout
    tau = 0.05  # Starting temperature
    rho = 0.5
    max_iters = 200  # per temperature stage
    best_Y = None
    best_val = -np.inf

    for stage in range(6):  # Zero-indexed stages
        if time.time() - start_time > budget_s - 5:
            break
        Y = restart_few() if stage == 0 else best_Y.copy()
        
        for iteration in range(max_iters):
            if time.time() - start_time > budget_s - 5:
                break
            # Compute surrogate
            val = log_sum_exp_surrogate(Y, tau)
            grad = gradient_log_sum_exp(Y, tau)
            # Gradient ascent step
            step_size = 0.01
            Y_new = Y + step_size * grad
            Y_new = project_to_feasible(Y_new)
            val_new = log_sum_exp_surrogate(Y_new, tau)
            if val_new >= val:
                Y = Y_new
            else:
                # Make small adjustment and try again
                step_size *= 0.5
                Y_new = Y + step_size * grad
                Y_new = project_to_feasible(Y_new)
                val_new = log_sum_exp_surrogate(Y_new, tau)
                if val_new > val:
                    Y = Y_new
            if val_new > best_val:
                best_val = val_new
                best_Y = Y.copy()
        
        # Reduce temperature
        tau *= rho

    # Reconstruct h from best_Y and round to binary
    h_continuous = np.sum(best_Y**2, axis=1)
    # Sort indices by h descending
    sort_idx = np.argsort(h_continuous)[::-1]
    binary_mask = np.zeros(n_points)
    binary_mask[sort_idx[:n_points//2]] = 1.0

    # Evaluate both binary and continuous estimates
    h_eval = h_continuous.copy()
    h_eval /= np.sum(h_eval) / (n_points / 2)
    
    c5_continuous = np.max(np.correlate(h_eval, 1-h_eval, mode='full') * dx)
    
    binary_eval = binary_mask.copy()
    c5_binary = np.max(np.correlate(binary_eval, 1-binary_eval, mode='full') * dx)
    
    if c5_binary < c5_continuous:
        final_h = binary_eval
        final_c5 = c5_binary
    else:
        final_h = h_eval
        final_c5 = c5_continuous
    
    # Up-sample to 800 points and slight polish with projected gradient
    n_upsample = 800
    dx_up = 2.0 / n_upsample
    x_up = np.linspace(0, 2, n_upsample)
    x_orig = np.linspace(0, 2, n_points)
    h_upsample = np.interp(x_up, x_orig, final_h)
    h_upsample = h_upsample * (n_upsample / 2) / np.sum(h_upsample)
    
    # Final polishing with subgradients on the full problem
    delta = 0.1 * dx_up
    for i in range(30):
        if time.time() - start_time > budget_s - 5:
            break
        # Max overlap shift
        corr = np.correlate(h_upsample, 1 - h_upsample, mode='full') * dx_up
        k_max = np.argmax(corr)
        shift = k_max - (n_upsample - 1)
        # Subgradient for k_max shift
        subgrad = np.zeros_like(h_upsample)
        # handle reflection for circular convolution
        for i in range(n_upsample):
            i_plus = (i + shift) % n_upsample
            i_minus = (i - shift) % n_upsample
            subgrad[i] = (1 - h_upsample[i_plus]) - h_upsample[i_minus]
        # Update
        h_new = h_upsample - delta * subgrad
        # Clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_upsample / 2) / np.sum(h_new)
        h_upsample = h_new

    final_c5 = np.max(np.correlate(h_upsample, 1 - h_upsample, mode='full') * dx_up)
    
    return (h_upsample, final_c5, n_upsample)
