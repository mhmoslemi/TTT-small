    def solve_binary_milp():

        # Total weight constraint for full pattern:
        # x[0] + 2 * sum_{i=1}^{d-1} x[i] - x[0] = n0/2  
        # => sum_{i=1}^{d-1} x[i] = n0/4
        weight_cons = sum(x[1:]) == n0/4

        # Binary auxiliary variables for products
        y = {}
        Z = {}  # Overlap counts at shift k

        # of shift k, certain position constraints. 
        # All shifts k = 0 to n0-1 (we generate as needed for valid pairs)
        # For a fixed shift k, pair (i, i+k) is valid if 0 <= i <= n0-1-k
        # Use symmetric construction, so (assuming k is positive)
        # i ranges from 0 to n0-1-k, so y[i,k] = x[i]*x[i+k]
        # Since we're using symmetry in the first half, only need to track combinations:
        # Groups of (i, i+k) for valid shift-k pairs and i

        # Use separation of index pairs into each k-shift (summing as Z_k = sum of y[i,k])

        # This requires careful modelling to ensure sparse constraints
        # Let's precompute valid pairs and make constraints for each shift.
        # Then A[k] = Z_k = sum_i y[i, k] and for each k we enforce Z_k >= n0/2 - M
        # Let's proceed carefully and avoid memory explosion
        # First define l=0,n0-1 indices
        # Now for each shift k (0 to n0-1), we have pairs (i, i+k) where i+k<n0
        # i ranges from 0 to n0-1-k inclusive (at most n0-1-k )
        # But with symmetry, we can have even more efficient representation

        # With:
        # x[i] = x[n0-1-i]   for all i < n0/2  
        # x[0]=1 fixed

        # x[i] = x[n0-1-i]
        # т.е. upper half: x[i] = x[n0-1-i] for i in range(40,80)
        
        # But it's simpler to make the discriminated indices explicitly
        # Let's express full process with system.
        # Symmetry constraint would make (at most) n0/2 variables active:
        # x[0], ..., x[n0//2], where for all i>0, x[i] = x[n0-i]

        # x[2] = x[78]
        # ...
        # x[39] = x[41]

        # x[41] = x[39]

        # So actual variable set: let's say y = x[1:41]
        # If y includes index 40, it will be used specially.
        # Better approach - think like block structure
        # [1] + [y1,..,y40] = 41 variables. 
        # Full x = [1, y1,..,y40, y39,..,y1]  if we index all from 0..79
        # But better exposed via indexing:
        
        # Let's encapsulate all by our reduced problem now:
        # FOR 0 <= i < d = n0//2+1 (41),
        # We declare x[i] with x[0] fixed = 1, x[i] = x[n0-i] for 1<=i <= n0//2
        
        # Define terms (we'll collect them carefully for y:
        # y[i, k] is x[i]*x[i+k] with i+k<n0, and comes into Z_k for shift k.
        
        # Now we build structure for yet another Z_k constraints:  
        # ∑_i y[i,k] >= n0 / 2 - M   (i.e. we minimize M)
        
        # To do that tractably:
        # We use constraint management.
        # Just iterate over shifts
        
        M = Variable()
        constraints = [x[0] == 1]  # x0 is constant  
        constraints.append(sum(x[1:]) == n0/4)
        
        # Initialize dictionary of y variables  
        y_cons = []
        # For each k=0 to n0-1, define pairs needed for that k
        for k in range(n0):
            # idx pairs for (i, i+k) with valid indices:
            # i = 0 to n0-1-k, valid finite sum.
            # pair: (i, i+k)
            # If i+k >= n0, invalid. If i+k < n0, valid.
            # Since full vector is from 0 to n0-1
            for i in range(n0-k): 
                i1 = i
                i2 = i+k
                
                # Ensure we are properly indexing in reduced variable space 
                # Consider x[0] = 1, x[1..n0//2] to be full range variables (we will map)
                # x[i] = x[n0-1-i]
                # Now store tuple (i1_check, i2_check) as relative to considered symmetry
                # Reference x[0] = 1, x[1] = x[79], etc.
                
                # Map to indices from reduced space:
                # Given a value i1, i2 (0 to n0-1)
                # We map to full x space: 
                # x[i] valid for all i (0 through n0-1 but derived from minimal representation)
                # Already done representation choice.
                # i refers to a position in x array
                # to record something like y[i,k], if we consider x[0] = 1 and others passed as x[1],x[2] ...
                # we define the addressing explicitly. 
                # So define :
                x_i1 = x[0] if i1 == 0 else x[n0 - i1] if i1 + (n0-i1) == n0 else x[i1]
                x_i2 = x[0] if i2 == 0 else x[n0 - i2] if i2 + (n0-i2) == n0 else x[i2]

                #   x[i1] = ( x[0]=1 ) or idx[1..40] (if i1=1..40)
                #   If x[i1] is in second half, it's symmetric to x[n0-1-i1]

                x_i1 = None; x_i2 = None

                    x_i1 = x[0]
                elif i1 <= 40:
                    x_i1 = x[i1]
                else:    # i1 >= 41
                    x_i1 = x[n0 - i1]
                    
                if i2 == 0:
                    x_i2 = x[0]
                elif i2 <= 40:
                    x_i2 = x[i2]
                else:    # i2 >= 41 (that's >40+1 = 41)
                    x_i2 = x[n0 - i2]

                # Let’s say indices for y instance correspond to: y[i1, k] is o.k.  
                # Then for index combinations:
                y_var_key = (i1, k)
                y_var = Variable(boolean=True)
                
                # Linearization first:
                # y_var <= x_i1
                # y_var <= x_i2  
                # y_var >= x_i1 + x_i2 - 1
                constraints.append(y_var <= x_i1)
                constraints.append(y_var <= x_i2)
                constraints.append(y_var >= x_i1 + x_i2 - 1)

                # Weighted using mapping properly according to
                # Let's imagine mapping back to integer for actual z直观
                # Linear track instead
                # General constraint: z_k = sum over i of y[i,k], gives forward sum of constraints
                # first chain - compute Z_k
                if k not in Z:  # create collection for shift k's   
                    Z[k] = 0
                Z[k] += y_var

        # Final constraints
        for k in range(n0):
            constraints.append(Z[k] >= n0/2 - M)

        obj = Minimize(M)
        prob = Problem(obj, constraints)
        
        try: 
            prob.solve(solver=CPLEX, verbose=False, time_limit=25, gap_abs=1e-8)
            if prob.status == OPTIMAL or prob.status == OPTIMAL_INACCURATE:
                soln = x.value  
                # Solve becomes x=(n0) vector
                soln_full = np.zeros(n0)
                soln_full[0] = 1.0
                soln_full[1:41] = x.value[1:]
                for j in range(1, 41): 
                    soln_full[n0-j] = x.value[j]
                # Or maybe just reconstruct properly:
                # x indexed from 0 to 40 (but full = 80)  
                for i in range(1, 41): 
                    soln_full[i] = x.value[i]
                for i in range(41, 80):
                    soln_full[i] = soln_full[n0 - i]
                return np.clip(soln_full, 0, 1)
            else:
                return None  
        except:
            return None

    # Recover best solution from symmetry-enforced MILP
    x_best = None
    try:
        x_best = solve_binary_milp()
    except:

    if x_best is None:
        # Fallback with simpler method.
        # This is the key flaw of the previous code: automatic scaling down to uniform
        # actually below.
        x_best = np.ones(n0) * 0.5
        x_best = x_best * (n0 / 2) / np.sum(x_best)
        

    # Upscale to target resolution (n = 300)
    # Rescale fine grid
    upscale_factor = n // n0
    h_kernel = np.repeat(x_best, upscale_factor)
    h_kernel = h_kernel[:n]  # trim to n points
    # adjust precisely to sum = n/2
    h_kernel = h_kernel * (n / 2) / np.sum(h_kernel)

    # Symmetrize the kernel – we want h[i] == h[n-1-i]
    h_sym = (h_kernel + h_kernel[::-1]) / 2.0

    # Continuous smooth-max ascent
    def smooth_max_objective(h, beta):
        # h must be valid in [0,1], sum = n/2
        dx = 2.0 / n
        # Compute all correlations using FFT for efficiency.
        conv_results = signal.correlate(h, 1-h, mode='full')
        # indexing: full convolution has n + n - 1 elements, from -n+1 to n-1
        # so the central part of 2n-1 contains onion shifts.
        # Center slice is from ending at index n-1 and starting at -n+1, which means indices (from n-1 to 2n-2)
        # still hard to explain...

        # centers it right with the convolution properties.
        # reflect from full array to get correct overlap values at all shifts.
        # note that 1-h is the same is at step size, let's verify in code
        # indices s.t. the full kernel is to 1-h segment aligned with h to compute其
        # Match edge behavioural effects to be precise.
        # just use computation directly first.
        # idx_vals = []
        # This all originates in time part, winding up in high frequency.

        # The basic correlation = convolution with 1-h (any zero-padding)
        # Let's go back to simple worry-free routine:
        # the indices are for negative k to positive k in full correlation.
        # the overlapping shift that degrees are computed, and masking gives no-zero region.
        # maximum over indices computed over it
        return 1.0 - np.max(conv_results) * dx

    def smooth_max_grad(h, beta):
        dx = 2.0 / n
        # Create padded h  
        padding = n - 1
        pad_h = np.pad(h, padding, mode='constant')
        neg_h = 1 - h
        # Marginalize over all the correlation functions from sliding with 1-h
        # e.g., compute 1st， 2nd，...truncated strength of h @ g
        # numerically evaluate <h, 1-h_k>
        # Instead: for each shift k in [-n+1, n-1]
        #  work fast over pairs.
        
        # Perform exponential softmax of overlapping values:
        # F_b(h) = (1/beta) log(sum_k exp(beta * O_k(h))) = 1 - (1/beta) log(sum_k exp(-beta * conv_k ))
        # Let's follow official formulas:
        convs = np.array([np.sum(pad_h[i : i + n] * (1 - h)) for i in range(0, 2 * n - 1)])
        
        # compute weights: exp(beta * conv) sum
        wts = np.exp(beta * convs) / np.sum(np.exp(beta * convs))  

        # derivative follows series logic in terms of grad contributions  
        grad = np.zeros(n)
        for k in range(0, 2 * n - 1):
            # Gradient shift = k -(n-1) which is valid only when shifted by that specific amount 
            # now compute gradient term-by-term of conv_k wrt h_i as given.
            # effectively a sliding window over pad_h  
            # This isn't fully correct.
            # On full; we consider final C_k * d/dh_i (conv_k)
            # Based on properties:
            # Diff of h_i with slide shift k is in terms of h(units at i and i+k)  positively
            # and h(i-k) from opposite, if it exists!
            
            # Derivative activity related to position is:
            # multiplying by ( (1 - h[i+k] ) * delta_{i=j} - h[i-k] * delta_{j=i-k} )  
            # But to accommodate for zero indexing:

            # Intuition check: derivative of <h, h_{i-k}>=∑_j h_j h_{j-i+k}, so:
            # for fixed j, values dependent on h_j and h_{j-i+k}  
            # written out:
            # d/dh_j [ h_j(1-h_{j+i-k}) ] = 1 - h_{j+i-k}
            # d/dh_j [ h_{j+i-k} (1-h_j) ] = -h_j
            # So contribution to gradient is:
            # if i_k + k = j: contribution is (1 - h_j)  
            # if j + k = i_k :    contribution is - h_i_k
            # Rearranging later.
            # Let’s fix the subdivision into overlapping subterms computation:
            # problem free – we emulate with python convolutions directly.

            # simpler Vec-diff use manual:
            # Let’s compute directional derivative for h

            # Recalculate: for >cheating method 
            # return correct gradient for each h.
            # Hand-coded:

            # idx = k + n - 1 
            # for specific idx, the sliding sum over valid i values all depend

            # kg = k - n + 1   # shift from 0-based canonical indexing
            # conv_k at index kg from (-n+1 to n-1), corresponds to i=kg + (n-1) of symmetry fallback
            grad_contrib = np.zeros(n)
            if k < n:
                # take subset  
                for i in range(k):
                    if i < n:
                        if i + k < n:
                            grad_contrib[i] += (1 - h[i + k])  # h_j* (1-h_{j+k})  
                        if i - k >= 0:
                            grad_contrib[i] -= h[i - k] 
            elif k >= n:
                idx_shift = k - n + 1  # summed index = [0, n-k) -> [length], but alternating 
                for i in range(n - idx_shift):
                    # The actual final length of overlap
                    j = i + idx_shift
                    if j < n:
                        if j >= 0:
                            grad_contrib[i] += (1 - h[j]) 
                        if j - k >= 0:
                            grad_contrib[i] -= h[j - k]                   
            grad += wts[k] * grad_contrib  

        return grad * dx

    # Optimise continuous version using modern gradient descent
    def optimize_continuous():
        h_current = h_sym.copy()
        # Different values of beta – increasing sharpness of softmax 
        beta = 0.5
        
        max_iter_per_beta = 100
        for iter in range(max_iter_per_beta):
            # Stop if time is done
            if time.time() - start_time > budget_s - 10:
                break
            
            # Optional tracking
            if iter % 10 == 0:
                exact_c5 = smooth_max_objective(h_current, beta)
            
            # Compute gradient  
            grad = smooth_max_grad(h_current, beta)
            
            # Apply step
            step_size = 0.1
            h_new = h_current - step_size * grad
            
            # Clamp to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Project sum to n/2
            old_sum = np.sum(h_new)
            delta = old_sum - n / 2
            h_new = h_new - delta / np.sum((0 < h_new) & (h_new < 1))  # projected waterfill
            # Here's a simple clean impl of waterfilling  
            # Normalize
            wrong = np.sum(h_new) - n / 2
            if abs(wrong) > 1e-12:
                changing_h = (0 < h_new) & (h_new < 1)
                h_new[changing_h] -= wrong / np.sum(changing_h)
            
            # Ensure symmetric 
            h_new = (h_new + h_new[::-1]) / 2
            
            h_current = h_new
            
            # Break early on convergence  
            if np.linalg.norm(grad) < 1e-6:
                break

        return h_current

    # Optimize optimization
    if time.time() - start_time < budget_s - 10:
        try:
            h_opt = optimize_continuous()
        except:
            h_opt = h_sym.copy()

    # Final polishing: small stochastic hill climb
    def polish_hill_climb(h_refined):
        h_best = h_refined.copy()
        max_trials = 10000
        epsilon = 0.004
        p = min(6, n // 20)  # number of indices to perturb (small enough)

        for trial in range(max_trials):
            if time.time() - start_time > budget_s - 1:
                break

            indices = np.random.choice(n, size=p, replace=False)
            h_trial = h_best.copy()
            for i in indices:
                h_trial[i] += np.random.uniform(-epsilon, epsilon)
                h_trial[i] = np.clip(h_trial[i], 0, 1)

            # correct integral
            old_sum = np.sum(h_trial)
            delta = old_sum - n / 2
            changing_h = (0 < h_trial) & (h_trial < 1)
            h_trial[changing_h] -= delta / np.sum(changing_h)
            h_trial = (h_trial + h_trial[::-1]) / 2

            # compute actual C5  
            dx = 2.0 / n
            convs = signal.correlate(h_trial, 1-h_trial, mode='full')
            c5_current = 1.0 - np.max(convs) * dx

            # Compute actual C5
            dx = 2.0 / n
            convs = signal.correlate(h_trial, 1 - h_trial, mode='full')
            c5_trial = 1.0 - np.max(convs) * dx

            if c5_trial < smooth_max_objective(h_best, 100):
                h_best = h_trial.copy()

        return h_best

    # Do global hill climb polish here.
    if time.time() - start_time < budget_s - 10:
        try:
            h_final = polish_hill_climb(h_opt)
        except:
            h_final = h_opt.copy()

    # Final evaluation of exact C5  
    dx = 2.0 / n
    convs = signal.correlate(h_final, 1 - h_final, mode='full')
    c5_result = 1.0 - np.max(convs) * dx

    return (h_final, c5_result, n)
