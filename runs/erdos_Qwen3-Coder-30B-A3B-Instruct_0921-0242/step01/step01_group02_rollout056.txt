def binarize_h(h, target_sum):
    """Convert continuous h to binary representation with exact target_sum."""
    h_copy = h.copy()
    # Sort indices by value in descending order
    sorted_indices = np.argsort(h_copy)[::-1]
    
    # Set first target_sum elements to 1
    h_binary = np.zeros_like(h_copy)
    h_binary[sorted_indices[:int(target_sum)]] = 1
    
    # Adjust if exactly target_sum is not achieved
    current_sum = np.sum(h_binary)
    if current_sum < target_sum:
        # Set some 0s to 1s
        for i in range(int(target_sum) - int(current_sum)):
            idx = sorted_indices[int(target_sum) + i]
            h_binary[idx] = 1
    elif current_sum > target_sum:
        # Set some 1s to 0s
        for i in range(int(current_sum) - int(target_sum)):
            idx = sorted_indices[int(target_sum) + i]
            h_binary[idx] = 0
    
    return h_binary

def c5_of_h_fft(h, dx):
    """Compute C5 for given h using FFT convolution."""
    h = h.astype(float)
    g = 1 - h
    corr = fftconvolve(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def update_convolution_fast(corr, h, i1, i0, old_h):
    """Fast update of correlation when swapping h[i1] and h[i0]."""
    n = len(h)
    delta = np.zeros(2 * n - 1)
    
    # Contributions to correlation from changing h[i1] and h[i0]
    if old_h[i1] == 1:
        # Old h[i1] was 1, now 0, contribution to -g_shifted
        delta += np.concatenate([g[i1:], np.zeros(i1)])
    else:
        # Old h[i1] was 0, now 1, contribution to h_shifted
        delta -= np.concatenate([h[i1:], np.zeros(i1)])
    
    if old_h[i0] == 1:
        # Old h[i0] was 1, now 0, contribution to -g_shifted
        delta += np.concatenate([g[i0:], np.zeros(i0)])
    else:
        # Old h[i0] was 0, now 1, contribution to h_shifted
        delta -= np.concatenate([h[i0:], np.zeros(i0)])
    
    # Return updated correlation
    return corr + delta

def get_indices_of_ones_zeros(h):
    """Return lists of indices where h=1 and h=0."""
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    return ones, zeros

def swap_bits(h, i1, i0, ones, zeros):
    """Swap elements at indices i1 and i0, and update lists."""
    old_val = h[i1]
    h[i1], h[i0] = h[i0], h[i1]
    if old_val == 1:
        ones.remove(i1)
        zeros.append(i1)
    else:
        zeros.remove(i1)
        ones.append(i1)
    if h[i0] == 1:
        ones.append(i0)
        zeros.remove(i0)
    else:
        zeros.append(i0)
        ones.remove(i0)
    return h, ones, zeros
