import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import scipy.optimize as opt

def evaluate_erdos_solution(h):
    """Compute C5 from h using the same method as the evaluation function."""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    N = 140
    S = N // 2
    dx = 2.0 / N
    
    # Very short preliminary version
    # 1. Define the MILP model with binary variables x and auxiliary w:
    x = cp.Variable(N, boolean=True)
    t = cp.Variable()
    # Symmetry-breaking constraint
    constraints = [cp.sum(x) == S, x[0] == 1]
    
    # Build index sets for each shift k
    shift_indices = []
    for k in range(N):
        indices = []
        for i in range(N - k):
            indices.append((i, i + k))
        shift_indices.append(indices)
    
    # List to hold w variables
    w_vars = []
    w_vars_by_k = []
    for k in range(N):
        w_list = []
        for i, j in shift_indices[k]:
            w_var = cp.Variable(boolean=True)
            w_vars.append(w_var)
            w_list.append(w_var)
        w_vars_by_k.append(w_list)
    
    # Add AND constraints
    for k in range(N):
        for i, j in shift_indices[k]:
            w_var = w_vars_by_k[k][shift_indices[k].index((i, j))]
            constraints.extend([
                w_var <= x[i],
                w_var <= x[j],
                w_var >= x[i] + x[j] - 1
            ])
    
    # Add overlap bounds
    for k in range(N):
        w_list = w_vars_by_k[k]
        constraints.append(cp.sum(w_list) >= S - t)
    
    constraints.append(t >= 0)
    
    # Binary search on t to find small feasible value
    L = 0
    U = S
    best_t = S
    best_x = None
    
    # Time for binary search
    search_time = 0
    max_search_time = 180
    
    while U - L > 1e-4 and search_time < max_search_time:
        T = (L + U) / 2
        # Solve feasibility problem with t <= T
        prob = cp.Problem(cp.Minimize(0), constraints + [t <= T])
        try:
            prob.solve(solver=cp.CBC, max_seconds=25, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Feasible, store this solution and update lower bound
                if t.value is not None and t.value < best_t:
                    best_t = t.value
                    best_x = x.value.copy()
                U = T
            else:
                L = T
        except Exception:
            L = T
        search_time = time.time() - start_time
    
    # If we got a good solution, use it
    if best_x is not None:
        # Convert to floating-point h
        h = np.array(best_x, dtype=float)
        # Try to improve with enrichment
        best_c5 = evaluate_erdos_solution(h)
        # Fractional polishing to improve solution, if time allows
        if time.time() - start_time < budget_s - 5:
            # Identify the worst shift (k that gives max c_k)
            g = 1.0 - h
            corr = np.correlate(h, g, mode='full')
            max_k = np.argmax(corr)
            worst_k = max_k - (N - 1)
            # Identify pairs that can be swapped to reduce overlap
            # Loop over candidate swaps (i,j) where h[i] = 1, h[j] = 0
            best_improve = 0.0
            best_swap = None
            num_pairs = 0
            num_moves = 0
            delta_schedule = [0.05, 0.02, 0.01]
            # Only test a subset of possible moves to stay within time budget
            for delta in delta_schedule:
                if num_moves > 100 or time.time() - start_time > budget_s - 5:
                    break
                for i in range(N):
                    if h[i] < 0.5:
                        continue
                    for j in range(i+1, min(i+100, N)):
                        if h[j] > 0.5:
                            continue
                        # Temporarily swap
                        h_temp = h.copy()
                        h_temp[i] -= delta
                        h_temp[j] += delta
                        # project to feasible set
                        h_temp = np.clip(h_temp, 0, 1)
                        h_temp = h_temp * (N / 2) / np.sum(h_temp)
                        # Check the new C5
                        c5_new = evaluate_erdos_solution(h_temp)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            h = h_temp.copy()
                        num_moves += 1
                        if num_moves > 300:
                            break
                    if num_moves > 300:
                        break
        # If no improvement, still use the base solution
    else:
        # fallback to random binary construction
        x = np.zeros(N, dtype=int)
        x[:S] = 1
        np.random.shuffle(x)
        h = x.astype(float)
        best_c5 = evaluate_erdos_solution(h)
        
    # Final binary swap refinement (check if improvement is possible)
    if time.time() - start_time < budget_s - 2:
        c5_before = evaluate_erdos_solution(h)
        attempts = 0
        max_attempts = 5000
        while attempts < max_attempts and time.time() - start_time < budget_s - 2:
            # Select a pair of indices (one 1, one 0), swap
            ones = np.where(h > 0.9)[0]
            zeros = np.where(h < 0.1)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                # Half move
                delta = 0.05
                h_temp = h.copy()
                h_temp[i] -= delta
                h_temp[j] += delta
                # Project to feasible set
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (N / 2) / np.sum(h_temp)
                c5_after = evaluate_erdos_solution(h_temp)
                if c5_after < c5_before:
                    h = h_temp
                    c5_before = c5_after
            attempts += 1
    
    # Ensure final sum constraint is satisfied
    h = h * (N/2) / np.sum(h)
    final_c5 = evaluate_erdos_solution(h)
    
    # Return h, C5 value, and number of points
    return (h, final_c5, N)
