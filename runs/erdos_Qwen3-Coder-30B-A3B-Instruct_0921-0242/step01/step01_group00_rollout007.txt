import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def proj_simplex_box(h, n_points):
    """Project h onto {0 ≤ h_i ≤ 1, sum h_i = n_points/2}."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n_points/2 - current_sum
    if abs(diff) < 1e-10:
        return h
    h = h + diff/n_points
    h = np.clip(h, 0, 1)
    return h

def c5(h):
    """Compute C5 constant."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def worst_shift(h):
    """Find the shift that maximizes the overlap integral."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.argmax(corr) - len(h) + 1

def solve_lp_relaxation(n_points, time_limit=20):
    """Solve a relaxed LP to get a good starting point."""
    try:
        # Variables
        h = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # McCormick envelopes for h_i * h_{i+k} (for k from 1 to n_points-1)
        # We introduce auxiliary variables z_{i,k} to model h_i * h_{i+k}
        # and add McCormick constraints
        
        # Since we'll be adding constraints, it's better to describe them directly
        # The full matrix setup would be O(n^2) and might be too much for this solver
        
        # Instead, let's approximate using a small subset of key lags
        # For sparse problem, control the number of lags we consider
        max_shifts = min(n_points // 2, 20)
        k_indices = list(range(1, max_shifts + 1))
        
        for k in k_indices:
            h_shifted = np.zeros(n_points)
            h_shifted[k:] = h[:-k]
            expr = cp.sum(cp.multiply(h, h_shifted))
            constraints.append(expr >= t)

        # Minimize t => Maximize C5
        prob = cp.Problem(cp.Maximize(t), constraints)
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=3000)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return h.value
    except Exception:
        pass
    # Fallback to some structured solution
    return np.ones(n_points) * 0.5

def gradient_step(h, k, alpha=0.01):
    """Perform one gradient step on shift k."""
    n = len(h)
    dx = 2.0 / n

    # Gradient of integral w.r.t h_i:
    # I_k = dx - dx * sum h_i h_{i+k}
    # dI_k/dh_i = -dx * h_{i+k}

    grad = np.zeros(n)
    for i in range(n):
        j = (i + k) % n
        grad[i] = -dx * h[j]

    # Move in the direction maximising I_k
    new_h = h - alpha * grad  # Gradient ascent
    return proj_simplex_box(new_h, n)

def anneal(h, time_limit):
    """Run simulated annealing heuristic for time_limit seconds."""
    start = time.time()
    current_h = h.copy()
    current_c5 = c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    temperature = 0.1
    
    while time.time() - start < time_limit:
        # Try to swap two values (randomly adjust)
        i, j = np.random.choice(len(h), 2, replace=False)
        d = np.random.uniform(-0.01, 0.01)
        h_try = current_h.copy()
        h_try[i] += d
        h_try[j] -= d
        h_try = proj_simplex_box(h_try, len(h_try))
        c5_try = c5(h_try)
        
        if c5_try < current_c5:
            current_h = h_try
            current_c5 = c5_try
            if c5_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c5_try
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c5_try - current_c5) / temperature):
                current_h = h_try
                current_c5 = c5_try
                
        # Cool down
        temperature *= 0.999
        if temperature < 1e-10:
            temperature = 1e-10
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 400
    dx = 2.0 / n_points
    
    # Phase 1: LP relaxation to get a seed
    h_seed = solve_lp_relaxation(n_points, time_limit=min(20, budget_s * 0.2))
    if h_seed is None:
        h_seed = np.ones(n_points) * 0.5
    h_seed = proj_simplex_box(h_seed, n_points)
    
    # Phase 2: Fine-tune with projective gradient on the worst shift
    h_refined = h_seed.copy()
    c5_best = c5(h_refined)
    remain_s = budget_s - (time.time() - start_time)
    if remain_s > 10:
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
            k = worst_shift(h_refined)
            h_refined = gradient_step(h_refined, k, alpha=0.01)
            c5_new = c5(h_refined)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_refined.copy()
    
    # Phase 3: Simulated annealing to explore space
    remain_s = budget_s - (time.time() - start_time)
    if remain_s > 30:
        h_annealed, c5_ann = anneal(h_refined, time_limit=min(30, remain_s))
        if c5_ann < c5_best:
            c5_best = c5_ann
            h_best = h_annealed.copy()
    
    # Phase 4: Final hill climbing
    remain_s = budget_s - (time.time() - start_time)
    if remain_s > 10:
        h_hill = h_best.copy()
        c5_hill = c5_best
        for _ in range(500):
            if time.time() - start_time > budget_s - 0.2:
                break
            # Modify a few entries
            indices = np.random.choice(n_points, size=min(5, n_points//20), replace=False)
            h_trial = h_hill.copy()
            for idx in indices:
                pert = np.random.uniform(-0.01, 0.01)
                h_trial[idx] += pert
            h_trial = proj_simplex_box(h_trial, n_points)
            c5_trial = c5(h_trial)
            if c5_trial < c5_hill:
                h_hill = h_trial
                c5_hill = c5_trial
        h_best = h_hill
        c5_best = c5_hill
    
    return (h_best, c5_best, n_points)
