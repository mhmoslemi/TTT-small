import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import math

def c5_fft(h):
    """Compute C5 using FFT for fast correlation."""
    g = 1 - h
    H = rfft(h)
    G = rfft(g)
    c = irfft(H * G, 2*len(h) - 1)
    return np.max(c)

def repair_weight(h, target_sum):
    """Repair vector to have exact sum target_sum."""
    current_sum = np.sum(h)
    diff = int(target_sum - current_sum)
    if diff == 0:
        return h
    
    if diff > 0:  # Need to add diff ones
        zero_indices = np.where(h == 0)[0]
        if len(zero_indices) < diff:
            diff = len(zero_indices)
        if diff > 0:
            indices = np.random.choice(zero_indices, diff, replace=False)
            h[indices] = 1
    else:  # Need to remove -diff ones
        one_indices = np.where(h == 1)[0]
        if len(one_indices) < -diff:
            diff = -len(one_indices)
        if diff < 0:
            indices = np.random.choice(one_indices, -diff, replace=False)
            h[indices] = 0
    return h

def crossover(p1, p2):
    """Two-point crossover."""
    n = len(p1)
    c1, c2 = np.random.randint(1, n, size=2)
    if c1 > c2:
        c1, c2 = c2, c1
    child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
    return repair_weight(child, n / 2)

def mutate_swap(h):
    """Swap one 1 with one 0."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h
    i1 = np.random.choice(ones)
    i0 = np.random.choice(zeros)
    h[i1], h[i0] = h[i0], h[i1]
    return h

def hill_climb(h0, max_swaps=500):
    """Local hill-climbing focused on reducing the worst shift."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    for _ in range(max_swaps):
        c = correlate(h, 1-h, mode='full')
        k_star = np.argmax(c) - n + 1
        best_delta = 0.0
        best_swap = None
        # Try 200 random swaps
        for _ in range(200):
            one_indices = np.where(h == 1)[0]
            zero_indices = np.where(h == 0)[0]
            if len(one_indices) == 0 or len(zero_indices) == 0:
                break
            i1 = np.random.choice(one_indices)
            i0 = np.random.choice(zero_indices)
            # Estimate change in c[k_star]
            old_val = h[i1] * h[i1 + k_star] + h[i0] * h[i0 + k_star]
            new_val = (1 - h[i1]) * h[i1 + k_star] + h[i0] * (1 - h[i0 + k_star])
            delta = (old_val - new_val) * dx
            if delta < best_delta:
                best_delta = delta
                best_swap = (i1, i0)
        if best_swap is None:
            break
        i1, i0 = best_swap
        h[i1], h[i0] = h[i0], h[i1]
    return h

    target_sum = n / 2

    best_h = None

    for _ in range(10):

    # 4. Random balanced
    for _ in range(pop_size - 21):
        indices = np.random.choice(n, int(target_sum), replace=False)

        h[indices] = 1

        P.append(h)

    # Evaluate initial population
    fitness = []

        c5 = c5_fft(h)
        fitness.append(-c5)

    if remaining_time < 20:
        return (best_h, best_c5, n) if best_h is not None else (np.zeros(n), float('inf'), n)
    
    # Main loop
    gen = 0
    while time.time() - start_time < budget_s - 20 and gen < 10000:

        # Selection - binary tournament
        new_P = []

            else:
                new_P.append(P[idx2].copy())
        
        # Crossover and Mutation

            idx_end = min(idx_start + scale, n_fine)
            h_upsampled[idx_start:idx_end] = best_h[i]

        # Steady state GA on fine grid
        pop_size_fine = 50

            h_fine[h] = 1

            P_fine.append(h_fine)
            
        for _ in range(100):
            new_P = []
            for _ in range(pop_size_fine - elite):
                idx1, idx2 = np.random.choice(pop_size_fine, 2, replace=False)

                else:
