def c5_of_h(h):
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def evaluate_binary_mask(b):
    """Evaluate C5 for a binary mask"""
    n = len(b)
    dx = 2.0 / n
    h = b.astype(float)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def compute_correlation_incremental(h, i1, i2, old_corr):
    """Incremental update to correlation when swapping h[i1] and h[i2]"""
    n = len(h)
    dx = 2.0 / n
    # Compute delta for the correlation
    delta = 0.0
    
    # For each position k, compute contribution to the correlation
    # when h changes by ±1 at i1 and i2
    for k in range(n):
        # Effect at position k of h[i1] change
        contrib1 = 0.0
        if i1 - k >= 0:
            contrib1 += h[i1 - k] * (1 - h[i1])  # contribution from i1
        else:
            # wrap around
            j = (i1 - k) % n
            contrib1 += h[j] * (1 - h[i1])
        if (i1 + k) % n < n:
            contrib1 -= h[i1] * (1 - h[(i1 + k) % n])  # removed effect
        else:
            # wrap around
            j = (i1 + k) % n
            contrib1 -= h[i1] * (1 - h[j])
            
        # Effect at position k of h[i2] change
        contrib2 = 0.0
        if i2 - k >= 0:
            contrib2 += h[i2 - k] * (1 - h[i2])  # contribution from i2
        else:
            # wrap around
            j = (i2 - k) % n
            contrib2 += h[j] * (1 - h[i2])
        if (i2 + k) % n < n:
            contrib2 -= h[i2] * (1 - h[(i2 + k) % n])  # removed effect
        else:
            # wrap around
            j = (i2 + k) % n
            contrib2 -= h[i2] * (1 - h[j])
            
        delta += (contrib2 - contrib1) * dx
        
    return delta

def project_to_feasible(h):
    """Ensure h has sum=1 and values in [0,1]"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    adjustment = deficit / n
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def lp_refinement(b_mask, h_ref, dx=0.01):
    """Refine binary mask b_mask into continuous h that satisfies new constraints"""
    # Convert binary mask to h mask where b_mask is 1, we want h_i >= 1, etc.
    n = len(b_mask)
    h = cp.Variable(n)
    # Maximal overlap constraint
    t = cp.Variable(1)
    # Constraint for each shift k 
    constraints = []
    # Add one big loop over shifts
    for k in range(2 * n - 1):
        shift_range = []
        cor_sum = 0.0
        for i in range(n):
            j = (i + k) % n
            if b_mask[i] == 1:
                cor_sum += h[i] * (1 - h[j])
            else:  # b_mask[i] == 0
                cor_sum += h[i] * (1 - 0)  # fixed to 0
        constraints.append(cor_sum <= t)
    # Feasibility objective
    constraints.append(cp.sum(h) == n / 2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    # Objective
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    if prob.status == "optimal":
        return h.value
    else:
        # Return a binary relaxation as fallback
        return h_ref
