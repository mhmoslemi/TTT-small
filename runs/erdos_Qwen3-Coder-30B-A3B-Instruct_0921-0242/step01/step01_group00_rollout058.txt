import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def gradient_worst_shift(h, worst_shift):
    """Compute gradient with respect to worst shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    
    # Since correlation uses circular shift, we take h_{i+k} and h_{i-k}
    # for k = worst_shift
    for i in range(n):
        left_idx = (i + worst_shift) % n
        right_idx = (i - worst_shift) % n
        grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
    
    return grad

def backtrack_line_search(h, grad, max_iter=10):
    """Backtracking line search for gradient update."""
    alpha = 1.0
    dx = 2.0 / len(h)
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        if c5_new < c5_old:
            return h_new
        alpha *= 0.5
    return h  # fallback

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Validate initial guess
    if 'initial_h_values' in globals():

    else:
        orig_h = np.ones(300) * 0.5
    
    # Use a moderate size
    n_points = min(400, len(orig_h))
    if len(orig_h) > n_points:
        h_start = orig_h[:n_points]
    else:
        x_orig = np.linspace(0, 2, len(orig_h))
        x_new = np.linspace(0, 2, n_points)
        h_start = np.interp(x_new, x_orig, orig_h)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = project_to_feasible(h_start, n_points)
    
    # Parameters for PSO
    S = 30  # swarm size
    max_iter_global = 1000
    omega = 0.7
    c1, c2 = 1.5, 1.5
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    for _ in range(S):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
        velocities.append(np.zeros(n_points))
    
    # Add the initial point
    swarm[0] = project_to_feasible(h_start, n_points)
    velocities[0] = np.zeros(n_points)
    
    # Track best
    global_best = None
    global_best_c5 = float('inf')
    global_best_idx = -1
    
    # Global PSO loop
    iter_count = 0
    last_improvement = time.time()
    
    while iter_count < max_iter_global and (time.time() - start_time < budget_s * 0.7):
        iter_count += 1
        if time.time() - start_time > budget_s * 0.7:
            break
            
        # Evaluate each particle
        for i in range(S):
            if time.time() - start_time > budget_s * 0.7:
                break
            h = swarm[i]
            c5 = evaluate_c5(h)
            
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best = h.copy()
                global_best_idx = i
                last_improvement = time.time()
        
        # Check for stagnation
        if time.time() - last_improvement > 2.0:
            # Inject some fresh particles
            for _ in range(S // 4):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                h = project_to_feasible(h, n_points)
                swarm[np.random.randint(S)] = h
                velocities[np.random.randint(S)] = np.zeros(n_points)
            last_improvement = time.time()
        
        # Update velocities and positions
        for i in range(S):
            if time.time() - start_time > budget_s * 0.7:
                break
            r1, r2 = np.random.rand(), np.random.rand()
            h = swarm[i]
            v = velocities[i]
            
            # Use current best and global best
            pbest = swarm[i]
            gbest = global_best
            
            v = omega * v + c1 * r1 * (pbest - h) + c2 * r2 * (gbest - h)
            h = h + v
            
            # Project to feasible
            h = project_to_feasible(h, n_points)
            swarm[i] = h
            velocities[i] = v
    
    # Memetic refinement on the global best
    if global_best is not None and time.time() - start_time < budget_s * 0.9:
        h_refined = global_best.copy()
        start_refine = time.time()
        for _ in range(50):
            if time.time() - start_refine > budget_s * 0.15 or time.time() - start_time > budget_s * 0.9:
                break
            # Compute worst shift
            corr = ifft(fft(h_refined, 2 * len(h_refined)) * np.conj(fft(1 - h_refined, 2 * len(h_refined))))
            # Ignore shift 0 (self-correlation)
            corr_vals = corr.real[1: len(h_refined)]  # exclude index 0
            worst_shift = np.argmin(corr_vals) + 1
            # Compute gradient
            grad = gradient_worst_shift(h_refined, worst_shift)
            # Update with backtracking
            h_refined = backtrack_line_search(h_refined, grad)
    
    # Final late-stage polishing
    if time.time() - start_time < budget_s * 0.995:
        h_final = h_refined.copy()
        c5_final = evaluate_c5(h_final)
        for _ in range(5000):
            if time.time() - start_time > budget_s * 0.995:
                break
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            if h_final[i] <= 0 or h_final[j] >= 1:
                continue
            delta = np.random.uniform(0, min(h_final[i], 1 - h_final[j], 0.01))
            h_test = h_final.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, n_points)
            c5_test = evaluate_c5(h_test)
            if c5_test < c5_final:
                c5_final = c5_test
                h_final = h_test
    
    # Return final result
    if h_final is not None:
        return (h_final, c5_final, n_points)
    else:
        return (np.ones(n_points) * 0.5, float('inf'), n_points)
