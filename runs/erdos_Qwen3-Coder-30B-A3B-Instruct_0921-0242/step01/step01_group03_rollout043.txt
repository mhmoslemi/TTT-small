import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import line_search
import cvxpy as cp
from scipy.spatial.distance import cdist

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def generate_fft_helpers(n):
    # For FFT-based convolution we need 2*N length arrays
    # We will do correlation as cross-correlation using FFT
    # In particular, for h and g, we compute conv(h, reverse(g))
    # This gives us correlation(direct, reverse)
    # But we need h[i] * g[i+k], so we reverse g and take the convolution
    # This means computing: h * reverse(g) where convolution is not circular
    # We'll pad with zeros to avoid wrap-around
    return 2 * n

def init_y(n, r, rng):
    # Random initialisation with normal
    Y = rng.normal(0, 1, (n, r))
    return Y

def project_to_feasible(Y):
    # Project onto the feasible region with constrs:
    # h_i = ||Y_i||^2, sum h_i = N/2, 0 <= h_i <= 1
    n = Y.shape[0]
    h = np.sum(Y**2, axis=1)
    
    # Clip and normalize (if needed)
    h = np.clip(h, 0, 1)
    S = np.sum(h)
    
    if abs(S - n/2) > 1e-10:
        alpha = np.sqrt((n/2) / S)
        Y *= alpha
    
    # Re-clip after possible scaling
    h = np.sum(Y**2, axis=1)
    h = np.clip(h, 0, 1)
    return Y

def compute_correlations(Y):
    # Y : (n, r) matrix
    n, r = Y.shape
    # Compute for each column, and sum them up
    correlations = np.zeros(n)
    for d in range(r):
        y_d = Y[:, d]
        # Real FFT method for circular cross-correlation
        y_fft = fft(y_d, 2*n)
        conv_fft = y_fft * np.conj(y_fft)
        c_d = ifft(conv_fft).real[:n]
        correlations += c_d
    
    return correlations

def compute_lse_gradient(Y, correlations, tau):
    # Compute p_k = exp(-c_k/tau) / sum_j exp(-c_j/tau)
    n = len(correlations)
    p = np.exp(-correlations / tau)
    p_sum = np.sum(p)
    p = p / p_sum if p_sum > 0 else p
    
    # Gradient is gradient wrt Y
    # At Y_i, we have: sum_k p_k (Y_{i+k} + Y_{i-k})
    # this is a convolution of p with each of the directions
    # We use FFT: this is circular convolution
    Y_grad = np.zeros_like(Y)
    for d in range(Y.shape[1]):
        y_d = Y[:, d]
        y_fft = fft(y_d, 2*n)
        p_fft = fft(p, 2*n)
        conv1_fft = p_fft * np.conj(y_fft)
        conv2_fft = np.conj(p_fft) * y_fft
        conv1 = ifft(conv1_fft).real[:n]
        conv2 = ifft(conv2_fft).real[:n]
        # Now compare with circular neighbor:
        # for i -> i+k, we compute shift.forward(p)[i] * y_d[i]
        # and shift.reverse(p)[i] * y_d[i]
        # each shift connects to itself, which is why we can compute it like above
        # But we want effect of SUM over k of p_k (Y_{i+k} + Y_{i-k})
        grad_d = conv1 + conv2
        Y_grad[:, d] = grad_d

    return Y_grad

def backtracking_line_search(Y, grad, func_val, c1=1e-4, rho=0.5):
    # Compute step size alpha that satisfies Armijo condition
    
    # Initialize with a large alpha to catch issues
    alpha = 1.0
    Y_new = Y - alpha * grad
    
    # Project Y_new
    Y_new = project_to_feasible(Y_new)
    
    # Function values to check
    func_new = compute_lse_func(Y_new)
    
    while func_new < func_val - c1 * alpha * np.sum(grad * grad):
        alpha *= rho
        Y_new = Y - alpha * grad
        Y_new = project_to_feasible(Y_new)
        func_new = compute_lse_func(Y_new)
        
    return alpha, Y_new

def compute_lse_func(Y, tau=0.05):
    # Compute the log-sum-exp function: -tau * log( sum(exp(-c_k/tau)) )
    correlations = compute_correlations(Y)
    if np.sum(np.exp(-correlations / tau)) > 0:
        return -tau * np.log(np.sum(np.exp(-correlations / tau)))
    else:
        return -tau * np.log(1e-15)  # Stable small value

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 400
    r = 6
    dx = 2.0 / n
    num_inner_iters = 200
    tau_values = [0.05, 0.025, 0.0125, 0.00625]  # Decreasing temperature
    
    best_c5 = np.inf
    best_h = None
    
    # helper for FFT-based computation
    fft_len = generate_fft_helpers(n)
    
    # Run multiple random restarts for robustness
    best_result = None
    num_restarts = 3  # reasonable number to balance exploration vs time
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        rng = np.random.default_rng(seed + restart)
        Y = init_y(n, r, rng)
        Y = project_to_feasible(Y)
        
        # Start temp annealing
        for tau in tau_values:
            if time.time() - start_time > budget_s - 1.0:
                break
                
            for _ in range(num_inner_iters):
                if time.time() - start_time > budget_s - 1.0:
                    break
                    
                # Compute current correlations
                correlations = compute_correlations(Y)
                
                # Compute function value
                func_val = compute_lse_func(Y, tau)
                
                # Compute gradient
                grad = compute_lse_gradient(Y, correlations, tau)
                
                # Backtracking line search
                alpha, Y_new = backtracking_line_search(Y, grad, func_val)
                
                # Update
                Y = Y_new
                
                # Re-project
                Y = project_to_feasible(Y)
                
        # After optimization, extract h
        h = np.sum(Y**2, axis=1)
        c5_val = c5_of_h(h)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            best_result = (best_h, best_c5, n)
            
    # Phase 2: rounding and swapping
    if time.time() - start_time > budget_s - 1.0:
        return best_result
    
    if best_h is not None:
        # Round to binary (n/2 ones) and evaluate exact C5
        h_rounded = np.zeros(n)
        sorted_indices = np.argsort(best_h)[::-1]
        h_rounded[sorted_indices[:n//2]] = 1.0
        c5_binary = c5_of_h(h_rounded)
        
        if c5_binary < best_c5:
            best_c5 = c5_binary
            best_h = h_rounded
            best_result = (best_h, best_c5, n)
    
    # Phase 3: deterministic swap polishing
    if time.time() - start_time > budget_s - 1.0:
        return best_result
        
    # Relaxation of Steps
    def swap_effect(h, i, j, k, c_k_vals):
        # Compute the effect of swapping h[i] with h[j]
        # i is from 1 to 0, j is from 0 to 1
        # But the h values (must be from mask, with 0/1 values) will be taken in the function below
        
        # Initialization
        new_c = c_k_vals.copy()  # So that we don't modify the input array
        # Compute changes from indices i and j affecting c_k for all k
        # The effect of a swap changes: c_k[i] += h[i, k] + h[i-k, i] 
        # But in our binary case, these are binary and we only need to track changes
        # In general, for a binary swap change:
        # c_k changes as: - (h[i + k] + h[i - k]) - (h[j + k] + h[j - k])
        # + (h'[i + k] + h'[i - k]) + (h'[j + k] + h'[j - k])  
        # Since h' = h[i] -> h[j], h[j] -> h[i], all others unchanged
        # and we only need to know if those are a 0 or 1
        
        # The indices that will be affected by the switches
        affected_ks = set([k for k in range(n) if abs(i - k) < n and abs(j - k) < n])
        
        # For each shift k, analyze how it's affected by the swap:
        delta = 0.0
        i_plus_k = (i + k) % n
        i_minus_k = (i - k) % n 
        j_plus_k = (j + k) % n
        j_minus_k = (j - k) % n 
        
        # undo contribution of the previous state for i and j
        delta += - (h[i_plus_k] + h[i_minus_k] + h[j_plus_k] + h[j_minus_k])
        # add contribution of new state
        delta += (h[j_plus_k] + h[j_minus_k] + h[i_plus_k] + h[i_minus_k])
        # Update c_k with the delta
        new_c[k] = c_k_vals[k] + delta
            
        return new_c
        
    def eval_c_k(h):
        dx = 2.0 / n
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full') * dx
        return corr
    
    # if we have h that is binary, and best_h is still valid:
    h_fin = best_h.copy()
    c_k_vals = eval_c_k(h_fin)
    
    # Find maximum shift k that attains the current c_k
    max_k = np.argmax(c_k_vals)
    
    # For deterministic swaps
    # Find positions where 1's don't match 0's for this k
    # This is key to greedy elimination of bad overlaps
    # e.g., any i such that h[i] == 1 and h[i+k] == 0
    # or h[i] == 0 but h[i+k] == 1
    
    swap_iter = 0
    max_swaps = 1000
    while swap_iter < max_swaps and time.time() - start_time < budget_s - 1.0:
        # Identify positions where c_k is maximum
        # Generally, the problem suggests we find:
        # mismatched indices for the max shift k: where h[i] is 1 but h[i+k] is 0
        # or vice versa
        max_k_idx = np.argmax(c_k_vals)
        mismatched_positions_i = [i for i in range(n) if (h_fin[i] == 1 and h_fin[(i + max_k_idx) % n] == 0) or 
                                  (h_fin[i] == 0 and h_fin[(i - max_k_idx) % n] == 1)]
        
        # For demonstration, we'll do just one swap
        # In layout: full policy, find the best move
        if len(mismatched_positions_i) >= 2:
            swap_i = mismatched_positions_i[0]
            swap_j = mismatched_positions_i[1]  # we're using slightest example
            
            # Make a temporary swap
            h_swap = h_fin.copy()
            h_swap[swap_i] = 1 - h_swap[swap_i]  
            h_swap[swap_j] = 1 - h_swap[swap_j]
            
            c_k_swap = eval_c_k(h_swap)
            new_c5 = np.max(c_k_swap)  # we want the new maximum overlap
            
            if new_c5 < best_c5:
                # Commit the swap
                best_c5 = new_c5
                h_fin = h_swap
                c_k_vals = c_k_swap
                swap_iter += 1
            else:
                # Skip it
                swap_iter += 1
                pass
        else:
            # So not enough mismatched elements
            break
    best_result = (h_fin, best_c5, n)
    
    # Phase 4: Upsample and fine-tune with sub-gradient
    if time.time() - start_time > budget_s - 1.0:
        return best_result
        
    # double resolution
    n_up = 2 * n
    dx_up = 2.0 / n_up
    h_up = np.zeros(n_up)
    for i in range(n):
        h_up[2*i] = best_h[i]
        h_up[2*i+1] = best_h[i]
    
    # Now renormalise via projection
    sum_h_up = np.sum(h_up)
    factor = (n_up/2) / sum_h_up
    h_up *= factor
    h_up = np.clip(h_up, 0, 1)

    # Final fine-tuning with projected sub-gradient:
    num_fine_steps = min(30, 100)  # limit number of steps
    for k in range(num_fine_steps):
        if time.time() - start_time > budget_s - 1.0:
            break
        # Find max overlap shift
        # To get current overlaps, we compute it exactly again
        dx = 2.0 / n_up
        g = 1.0 - h_up
        corr = np.correlate(h_up, g, mode='full') * dx
        k_max = np.argmax(corr)
        
        # Compute sub-gradient
        grad = np.zeros_like(h_up)
        for i in range(len(h_up)):
            # Forward term
            i_plus_k = (i + k_max) % len(h_up)
            grad[i] += -(1 - h_up[i_plus_k])
            grad[i] -= h_up[i_plus_k]
            
            # Backward term
            i_minus_k = (i - k_max) % len(h_up)
            grad[i] += 1 - h_up[i_minus_k]
            grad[i] += h_up[i_minus_k]
            
        # Step
        eta = 0.1 * dx_up
        h_up -= eta * grad
        
        # Project back
        h_up = np.clip(h_up, 0, 1)
        sum_h_up = np.sum(h_up)
        factor = (n_up/2) / sum_h_up
        h_up *= factor
        h_up = np.clip(h_up, 0, 1)
    
    # Final check
    final_c5 = c5_of_h(h_up)
    best_result = (h_up, final_c5, n_up)
    
    return best_result
