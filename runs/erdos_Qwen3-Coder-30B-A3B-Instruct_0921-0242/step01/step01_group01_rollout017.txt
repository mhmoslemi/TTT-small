import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time
from scipy.fft import rfft, irfft
import math

# Helper functions
def project_to_feasible(u, target_sum):
    """Project u onto the feasible set: 0 <= u <= 1, sum(u) = target_sum"""
    u = np.clip(u, 0.0, 1.0)
    s = np.sum(u)
    if abs(s - target_sum) < 1e-10:
        return u
    
    # Binary search for lambda
    def sum_with_lambda(lam):
        new_u = np.clip(u + lam, 0.0, 1.0)
        return np.sum(new_u) - target_sum
    
    low, high = -1.0, 1.0
    for _ in range(30):
        mid = (low + high) / 2.0
        val = sum_with_lambda(mid)
        if val > 0:
            high = mid
        else:
            low = mid
    
    final_u = np.clip(u + (low + high)/2.0, 0.0, 1.0)
    return final_u

def compute_C5_fft(h):
    """Fast computation using FFT for convolution and max"""
    n = len(h)
    # zero-padding to full length
    padded_len = 2*n - 1
    H = rfft(h, n=padded_len)
    G = rfft(1-h, n=padded_len)
    conv = irfft(H*G)[:padded_len]
    dx = 2.0 / n
    return dx * np.max(conv)

def symmetrize(h):
    """Enforce mirror symmetry h[i] = h[n-1-i]"""
    n = len(h)
    mid = n // 2
    # For even n, the middle two elements should be the same
    if n % 2 == 0:
        # Update from both directions
        h = (h + h[::-1]) / 2.0
        # Handle the central pair
        if mid > 0:
            h[mid-1] = h[mid] = (h[mid-1] + h[mid]) / 2.0
    else:
        # Odd case
        h = (h + h[::-1]) / 2.0
    return h

def cmaes_step(mean, sigma, d, pop_size, best_after_150):
    """one CMA-ES step with small population"""
    lambda_pop = pop_size
    z = np.random.randn(lambda_pop, d)
    offspring = mean + sigma * z
    weights = np.log(lambda_pop/2.0 + 1) - np.log(np.arange(1, lambda_pop+1))
    weights = weights / np.sum(weights)
    # Generate feasible solutions and compute fitness
    candidates = []
    for i in range(lambda_pop):
        # project back to feasible domain
        u = offspring[i]
        # enforce symmetry after projection
        h = np.zeros(2*d)
        h[:d] = u
        h[d:] = u[::-1]
        h = project_to_feasible(h, 2*d/2)
        h = symmetrize(h)
        candidates.append((h, compute_C5_fft(h)))
    
    candidates.sort(key=lambda x: x[1])
    best_candidate = candidates[0][0]
    best_fitness = candidates[0][1]
    
    # Update parameters
    best_h = best_candidate
    if best_fitness < best_after_150[0]:
        best_after_150[0] = best_fitness
        best_after_150[1] = best_h.copy()
    
    # for CMA-ES
    # weighted mean update
    h_weighted = np.zeros(d)
    for i in range(lambda_pop):
        h_weighted += weights[i] * offspring[i]
    
    mean = h_weighted
    
    # Adapt sigma
    C = np.eye(d)
    sigma = sigma * np.exp(0.3 * (np.std(offspring, axis=0) - np.std(z, axis=0)) / np.std(z, axis=0) ) # Simpler adjustment
    
    return mean, sigma, best_after_150

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_coarse = 150
    n_fine = 300
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Global coarse search
    d = (n_coarse + 1) // 2  # reduced dimension
    mu = np.ones(d) * 0.5
    sigma = 0.2
    
    # restart
    max_restarts = 3
    restart_count = 0
    
    pop_size = 4 + int(3 * np.log(d))
    # For increased stability
    if pop_size < 10:
        pop_size = 10
    print(f"Using population size {pop_size} for {d} dimensions")
    
    # Initial starting solution
    h_best_coarse = np.ones(n_coarse) * 0.5
    h_best_coarse = project_to_feasible(h_best_coarse, n_coarse/2)  # normalize
    h_best_coarse = symmetrize(h_best_coarse)
    best_c5 = compute_C5_fft(h_best_coarse)
    
    best_cost = [best_c5, h_best_coarse.copy()]
    
    iterations_since_improvement = 0
    last_improvement_cost = best_c5
    
    def cosine_schedule(t, t_max):
        return np.cos((t/t_max) * np.pi / 2)
    
    # Optimization loop
    while time.time() - start_time < budget_s - 5:
        t_remaining = budget_s - (time.time() - start_time)
        # roughly 500 generations for 350s, run ~150 generations then check restart
        max_gens = 500
        gens_since_restart = 0
        
        for gen in range(max_gens):
            if time.time() - start_time > budget_s - 2:
                break
                
            mu, sigma, best_cost = cmaes_step(mu, sigma, d, pop_size, best_cost)
            
            cost = best_cost[0]
            gens_since_restart += 1
            
            # Monitor improvements
            if abs(cost - last_improvement_cost) > 1e-4:
                last_improvement_cost = cost
                iterations_since_improvement = 0
            else:
                iterations_since_improvement += 1
            
            if iterations_since_improvement > 150:
                if restart_count < max_restarts:
                    # Restart
                    restart_count += 1
                    print(f"Restarting CMA-ES: {restart_count} times")
                    mu = best_cost[1][:d]
                    sigma *= 0.5
                    iterations_since_improvement = 0
                    gens_since_restart = 0
                    if sigma < 1e-8:
                        break
                else:
                    break
        
        # Check if we have time to do fine search
        if time.time() - start_time > budget_s - 10:
            break
        
        print(f"Coarse search finished at cost {best_cost[0]}")
        
        # Refine on fine grid using interpolation
        coarse_h = best_cost[1]
        coarse_x = np.linspace(0, 2, n_coarse)
        fine_x = np.linspace(0, 2, n_fine)
        fine_h_interp = interp1d(coarse_x, coarse_h, kind='linear', fill_value='extrapolate')
        h_fine_interp = fine_h_interp(fine_x)
        h_fine = project_to_feasible(h_fine_interp, n_fine/2)
        h_fine = symmetrize(h_fine)
        fine_c5 = compute_C5_fft(h_fine)
        print(f"Fine interpolation result: {fine_c5}")
        
        # Gradient-based refinement
        beta_max = 100
        beta = 0.5
        weight = 10
        max_beta_iter = 200
        lbfgs_iter = 0
        
        h_refined_fine = h_fine.copy()
        prev_cost = fine_c5
        
        while time.time() - start_time < budget_s - 10 and beta <= beta_max and lbfgs_iter < max_beta_iter:
            # Save current best as backup
            pb = prev_cost
            
            # Compute shifted convolutions efficiently
            
            def compute_conv_shift(h, k_shift):
                """Compute convolution term corresponding to shift k"""
                n = len(h)
                if k_shift > 0:
                    h_padded = np.pad(h, (k_shift, 0))[:n]
                    g = 1.0 - h_padded
                else:
                    h_padded = np.pad(h, (0, -k_shift))[-n:]
                    g = 1.0 - h_padded
                return np.sum(h * g) * dx_fine
            
            # compute all shifts once
            shifts = np.arange(-n_fine+1, n_fine)
            conv_vals = np.array([compute_conv_shift(h_refined_fine, k) for k in shifts])
            
            # Smoothing with soft max
            exp_conv_vals = np.exp(beta * conv_vals)
            exp_sum = np.sum(exp_conv_vals)
            weights = exp_conv_vals / exp_sum
            
            # Compute gradients:  This is executed efficiently using convolutions
            # compute ∇ compute_conv_k
            # ∂/∂h_i [ h_i * (1 - h_{i+k}) ] = 1 - h_{i+k} when i + k < n
            # ∂/∂h_i [ h_{i-k} * (1 - h_i) ] = -h_{i-k} when i - k >= 0
            
            # But for each coeff h_i we should compupute its influence on all these terms
            # Efficient way: we use Toeplitz structure
            
            # For shift k, gradient component (i) =  (1 - h_i+k) * [if i+k<n] - h_{i-k} * [if i-k>=0]
            # So we do this efficiently in batch for all k
            
            # We build shifted copies of h
            # one for positive shifts (1 - h_{i+k}), one for negative (h_{i-k})
            
            grad_list = []
            for k in shifts:
                grad_part = np.zeros(n_fine)
                # Shift indices 
                if k > 0:
                    # h_i+k term: 1-h_{i+k}  
                    h_shifted_forward = np.pad(1 - h_refined_fine, (k, 0))[:n_fine]
                    # 1 - h_{i+k} 
                    grad_part += h_shifted_forward
                else:
                    # h_{i-k} term: - h_{i-k} 
                    h_shifted_back = np.pad(h_refined_fine, (0, -k))[-n_fine:]
                    grad_part -= h_shifted_back
                
                grad_list.append(grad_part)
            
            grad_h = np.zeros(n_fine)
            for i, g in enumerate(grad_list):
                # gradient of weighted sum w_k * conv_k
                grad_h += weights[i] * g
                
            # Add normal penalty for constraint on sum
            extra_grad = 2 * weight * (np.sum(h_refined_fine) - n_fine / 2)
            grad_h += extra_grad
            
            # Simple gradient descent update
            step_size = 1e-3 / (beta + 1)
            h_refined_fine -= step_size * grad_h
            
            # project back
            h_refined_fine = project_to_feasible(h_refined_fine, n_fine/2)
            h_refined_fine = symmetrize(h_refined_fine)
            
            # Evaluate current cost
            new_c5 = compute_C5_fft(h_refined_fine)
            
            lbfgs_iter += 1
            if new_c5 < prev_cost:
                print(f"β={beta}, C5 = {new_c5}")
                prev_cost = new_c5
                
            if beta < beta_max:
                beta *= 2.0
                weight *= 2.0  # update penalty weight
                
            
            # Additional penalties or stops
            # break without further improvement
            if abs(new_c5 - pb) < 1e-5:
                pass # okay, converge in small steps 
            if lbfgs_iter > 150:
                break
                
        # Final polishing stage - hill climb
        h_final = h_refined_fine.copy()
        c5_final = compute_C5_fft(h_final)
        print(f"Before hill climb final C5={c5_final}")
        budget_remaining = budget_s - (time.time() - start_time)
        
        # حاجت 200s صعود
        if budget_remaining > 2:
            max_iters = int(1000 * (budget_remaining / 200.0))
            iter = 0
            for _ in range(max_iters):
                if time.time() - start_time > budget_s - 0.5:
                    break
                # Random swaps, small perturbations
                n_swaps = min(6, n_fine//20)
                idxs = np.random.choice(n_fine, size=n_swaps, replace=False)
                h_copy = h_final.copy()
                for i in idxs:
                    delta = np.random.uniform(-0.004, 0.004)
                    h_copy[i] += delta
                h_copy = project_to_feasible(h_copy, n_fine/2)
                h_copy = symmetrize(h_copy)
                c5_new = compute_C5_fft(h_copy)
                if c5_new < c5_final:
                    h_final = h_copy
                    c5_final = c5_new
                iter += 1
            print(f"Hill climb made {iter} steps")
            
        # Update best solution
        if c5_final < best_cost[0]:
            print(f"Better solution found: {c5_final}")
            best_cost[0] = c5_final
            best_cost[1] = h_final.copy()
        
        break # one refined and final bumping
    
    h_output = best_cost[1]
    c5_output = best_cost[0]
    
    return (h_output, c5_output, n_fine)
