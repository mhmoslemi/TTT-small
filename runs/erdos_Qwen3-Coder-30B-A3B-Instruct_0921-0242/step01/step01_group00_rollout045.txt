import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Adjust by distributing difference
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def sdp_relaxation(n_points):
    """Solve SDP relaxation for the Erdos problem"""
    dx = 2.0 / n_points
    
    # Variables
    h = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2,
        h >= 0,
        h <= 1,
        X >> 0,  # Positive semidefinite
    ]
    
    # Link X and h
    for i in range(n_points):
        constraints.append(X[i, i] == h[i])
    
    # Autocorrelation constraints
    for k in range(1, n_points):
        # Sum over valid indices (i, i+k) where both are within [0, n_points-1]
        expr = 0
        for i in range(n_points - k):
            expr += X[i, i + k]
        constraints.append(expr >= t)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sdp = h.value
            t_opt = t.value
            c5_sdp = dx * (1 - t_opt)
            return h_sdp, c5_sdp
    except Exception:
        pass
    return None, float('inf')

def randomized_rounding(h_sdp, n_points, R=30):
    """Do randomized rounding on SDP solution"""
    # Try to factor X = VV^T and then round
    try:
        U, s, Vt = np.linalg.svd(h_sdp.reshape(-1, 1) @ h_sdp.reshape(1, -1), full_matrices=False)
        V = Vt.T
        rank = np.sum(s > 1e-10)
        V = V[:, :rank]
        s = s[:rank]
        sqrt_s = np.sqrt(s)
        V_sqrt = V * sqrt_s.reshape(1, -1)
        best_c5 = float('inf')
        best_h = None
        for r in range(R):
            g = np.random.randn(rank)
            y = V_sqrt @ g
            sorted_y = np.sort(y)
            tau = sorted_y[n_points // 2]
            h_trial = (y > tau).astype(float)
            # Re-scale to sum = n_points/2 exactly
            h_trial = project_to_feasible(h_trial, n_points)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
        return best_h, best_c5
    except Exception:
        return None, float('inf')

def smooth_max_gradient_refinement(h_init, n_points, budget_s, start_time):
    # Create smooth max surrogate and refine
    dx = 2.0 / n_points
    # Initialize beta
    beta = 0.5
    h = h_init.copy()
    
    def smooth_max(h, beta):
        # compute I_k = dx * (1 - sum(h_i * h_{i+k}))
        # Approximate I_k for all k by doing correlation
        # Then logsumexp
        I = np.array([dx * (1 - np.sum(h[i:] * h[:len(h)-i])) for i in range(1, n_points)])
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * I)))
    
    def smooth_max_grad(h, beta):
        # Compute gradient of smooth max with respect to h
        I = np.array([dx * (1 - np.sum(h[i:] * h[:len(h)-i])) for i in range(1, n_points)])
        
        # Compute weights w_k = exp(beta*I_k) / sum(exp(beta*I_j))
        exp_beta_I = np.exp(beta * I)
        weights = exp_beta_I / np.sum(exp_beta_I)
        
        # Gradient is sum_k w_k * grad I_k
        grad = np.zeros(n_points)
        for k in range(1, n_points):
            if k < n_points:
                # Partial derivative of I_k w.r.t. h_i
                for i in range(n_points - k):
                    if i + k < n_points:
                        grad[i] += weights[k-1] * dx * (1 - h[i+k])
                        grad[i+k] += weights[k-1] * dx * (1 - h[i])
        # Subtract partial w.r.t. h_{i-k} for correct implementation
        # Actually we can reuse previous formula since correct correlation done in I_k
        
        # Add dummy gradient for initial term derivatives
        return grad
    
    # Initial evaluation
    current_max = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_max
    
    while (time.time() - start_time) < budget_s - 5:
        # Refine with increasing beta loop
        for _ in range(200):
            if (time.time() - start_time) >= budget_s - 5:
                break
            # Gradient step
            grad = smooth_max_grad(h, beta)
            # Step size
            alpha = 1.0
            for _ in range(10):
                h_trial = h - alpha * grad
                h_trial = project_to_feasible(h_trial, n_points)
                trial_max = evaluate_c5(h_trial)
                if trial_max < current_max:
                    h = h_trial
                    current_max = trial_max
                    if current_max < best_c5:
                        best_c5 = current_max
                        best_h = h.copy()
                    break
                else:
                    alpha /= 2.0
            beta *= 1.5
    return best_h, best_c5

def local_combinatorial_swap(h_init, n_points, budget_s, start_time):
    h = h_init.copy()
    dx = 2.0 / n_points
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    iterations = 0
    max_iter = int(1e5)
    while (time.time() - start_time) < budget_s - 5 and iterations < max_iter:
        iterations += 1
        # Pick two indices
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        if h[i] <= 0 or h[j] >= 1:
            continue
        # Make a small swap
        delta = min(h[i], 1 - h[j], 0.02)
        old_val = h[i] * (1 - h[j]) + h[j] * (1 - h[i])  # Approximate effect
        h[i] -= delta
        h[j] += delta
        # Re-project
        h = project_to_feasible(h, n_points)
        # Evaluate new
        c5_current = evaluate_c5(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 128
    best_h = None
    best_c5 = float('inf')
    
    # Stage 1: SDP relaxation
    h_sdp, c5_sdp = sdp_relaxation(n_points)
    if h_sdp is not None and c5_sdp < best_c5:
        best_c5 = c5_sdp
        best_h = h_sdp.copy()
    
    # Stage 2: randomized rounding
    if (time.time() - start_time) < budget_s - 30:
        h_rounded, c5_rounded = randomized_rounding(h_sdp, n_points, R=30)
        if h_rounded is not None and c5_rounded < best_c5:
            best_c5 = c5_rounded
            best_h = h_rounded.copy()
    
    # Stage 3: smooth max refinement
    if (time.time() - start_time) < budget_s - 20 and best_h is not None:
        h_refined, c5_refined = smooth_max_gradient_refinement(best_h, n_points, budget_s, start_time)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
            
    # Stage 4: local combinatorial swap
    if (time.time() - start_time) < budget_s - 10 and best_h is not None:
        h_final, c5_final = local_combinatorial_swap(best_h, n_points, budget_s, start_time)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Final evaluation of best result
    if best_h is None:
        # fallback
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_c5(best_h)
        
    return (best_h, best_c5, n_points)
