import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def worst_shift(h):
    """Return the shift index that gives maximal overlap."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # Return the shift that corresponds to this max index
    # For full correlation of length 2n-1, center is at n-1
    # Indices 0..n-2 are shifts -n+1..-1, n-1 is shift 0, n..2n-2 are 1..n-1
    n = len(h)
    shift = max_idx - (n - 1)
    return shift

def proj_simplex_box(h, n):
    """Project h to the feasible set: sum(h) = n/2, 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # adjust uniformly
    h = h + diff / n
    
    # re-clip
    h = np.clip(h, 0, 1)
    
    return h

def solve_lp_relaxation(n):
    """Solve LP relaxation to get a good initial h."""
    dx = 2.0 / n
    
    # Variables
    h = cp.Variable(n)
    
    # Auxiliary variables for the product terms
    y = cp.Variable(n)  # y[k] = sum_i h_i * h_{i+k}
    
    # Constraints
    constraints = [
        cp.sum(h) == n / 2,         # sum = n/2
        h >= 0, h <= 1,             # bounds
    ]
    
    # Build the objective
    # Maximize min_{k≠0} y[k] where y_k = sum_i h_i h_{i+k}
    # This is equivalent to maximize t subject to y_k >= t for all k≠0
    t = cp.Variable()
    for k in range(1, n):
        # For shift k, we compute sum_i h_i h_{i+k}
        # Here h_{i+k} uses circular indexing, so we use a trick
        # Let the shifted version be h.shifted = h[0] h[1] .. h[k-1] h[k] h[k+1] ...
        y_k = cp.sum(cp.multiply(h, cp.roll(h, -k)))
        constraints.append(y_k >= t)
    
    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
    except:
        pass
        
    # Fallback to simple uniform
    return np.ones(n) * 0.5

def gradient_step(h, k, alpha):
    """Perform one projected gradient step for shift k."""
    n = len(h)
    dx = 2.0 / n
    # Compute gradient for current shift k
    grad = np.zeros(n)
    for i in range(n):
        j = (i + k) % n
        # d/dh_i [ sum_j h_j h_{j+k} ] = h_{i+k} + h_{i-k}
        # But apply gradient of I_k = dx - dx*sum h_i h_{i+k}
        # So d/dh_i I_k = -dx h_{i+k}
        grad[i] = -dx * h[j]
        # Additional terms from h_{i-k}
        j2 = (i - k) % n
        grad[i] += dx * h[j2]

    # Compute new iterate
    h_new = h - alpha * grad
    # Project
    h_new = proj_simplex_box(h_new, n)
    return h_new

def anneal(h, time_limit):
    """Simulated annealing with exchange moves."""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    current_c5 = c5(h)
    
    T = 0.1  # Initial temperature
    T_min = 1e-7
    alpha = 0.99  # Cooling factor
    
    current_h = h.copy()
    
    while T > T_min and (time.time() - start_time) < time_limit:
        # Try a move
        i, j = np.random.choice(n, size=2, replace=False)
        delta = np.random.uniform(-0.05, 0.05)
        h_trial = current_h.copy()
        
        # Adjust first point
        h_trial[i] += delta
        # Adjust second point 
        h_trial[j] -= delta
        
        # Ensure bounds
        if h_trial[i] < 0: 
            h_trial[j] -= (h_trial[i] - 0)
            h_trial[i] = 0
        elif h_trial[i] > 1:
            h_trial[j] += (h_trial[i] - 1)
            h_trial[i] = 1
            
        if h_trial[j] < 0: 
            h_trial[i] -= (h_trial[j] - 0)
            h_trial[j] = 0
        elif h_trial[j] > 1:
            h_trial[i] += (h_trial[j] - 1)
            h_trial[j] = 1
            
        # Renormalize
        h_trial = proj_simplex_box(h_trial, n)
        trial_c5 = c5(h_trial)
        
        if trial_c5 < current_c5:
            current_h = h_trial
            current_c5 = trial_c5
        else:
            # Accept with prob
            prob = np.exp(-(trial_c5 - current_c5) / T)
            if np.random.rand() < prob:
                current_h = h_trial
                current_c5 = trial_c5
                
        T *= alpha
        
    return current_h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 400
    dx = 2.0 / n_points
    
    # Phase 1: LP relaxation to get a good structure
    print("Solving LP relaxation...")
    h_lp = solve_lp_relaxation(n_points)
    if h_lp is None:
        h_lp = np.ones(n_points) * 0.5
    h_lp = np.clip(h_lp, 0, 1)
    h_lp = proj_simplex_box(h_lp, n_points)
    if time.time() - start_time > budget_s - 5:
        return (h_lp, c5(h_lp), n_points)

    # Phase 2: Refine using projected gradient on worst shift
    print("Refining...")
    h_refined = h_lp.copy()
    current_c5 = c5(h_refined)
    for _ in range(50):
        if time.time() - start_time > budget_s - 2:
            break
        k = worst_shift(h_refined)
        h_refined = gradient_step(h_refined, k, 0.01)
        h_refined = proj_simplex_box(h_refined, n_points)
        new_c5 = c5(h_refined)
        if new_c5 < current_c5:
            current_c5 = new_c5
        else:
            # Avoid oscillation
            break
    
    # Phase 3: Annealing
    print("Annealing...")
    h_final, final_c5 = anneal(h_refined, (budget_s - (time.time() - start_time)) * 0.5)
    
    # Phase 4: Final coordinate descent
    print("Final polishing...")
    current_h = h_final.copy()
    current_c5 = final_c5
    for _ in range(300):
        if time.time() - start_time > budget_s - 0.2:
            break
        # Perturb a few entries
        modify_indices = np.random.choice(n_points, size=min(5, n_points//20), replace=False)
        h_trial = current_h.copy()
        for idx in modify_indices:
            pert = np.random.uniform(-0.01, 0.01)
            h_trial[idx] += pert
        h_trial = proj_simplex_box(h_trial, n_points)
        trial_c5 = c5(h_trial)
        if trial_c5 < current_c5:
            current_h = h_trial
            current_c5 = trial_c5
    
    print(f"Final C5: {current_c5}")
    return (current_h, current_c5, n_points)
