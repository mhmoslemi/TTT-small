import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from scipy.linalg import eigh
import time

def water_fill(h, target_sum):
    """Project h onto the simplex with sum = target_sum."""
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    diff = total - target_sum
    if abs(diff) < 1e-12:
        return h
    
    # Water filling algorithm
    while abs(diff) > 1e-12:
        # Find indices where h is not at bounds
        I = np.where((h > 0) & (h < 1))[0]
        if len(I) == 0:
            break
        # Distribute equally among active indices
        delta = diff / len(I)
        if delta > 0:
            # need to decrease
            h[I] -= delta
        else:
            # need to increase (delta < 0)
            h[I] -= delta
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        diff = total - target_sum
        
    return h

def nearest_rank1(X):
    """Extract the dominant rank-1 component from a PSD matrix X."""
    try:
        eigenvals, eigenvecs = eigh(X)
        idx = np.argmax(eigenvals)
        v = eigenvecs[:, idx]
        lam = eigenvals[idx]
        if lam > 0:
            return np.sqrt(lam) * v
        else:
            return v
    except:
        # Fallback to SVD if eigh fails
        U, s, Vt = np.linalg.svd(X)
        if s[0] > 0:
            return np.sqrt(s[0]) * U[:, 0]
        else:
            return U[:, 0]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: SDP relaxation with coarse grid
    n_coarse = 120
    dx_coarse = 2.0 / n_coarse
    n_target = 300
    dx_target = 2.0 / n_target
    
    # Build Toeplitz matrices R_k
    R_matrices = []
    for k in range(n_coarse):
        R_k = np.zeros((n_coarse, n_coarse))
        # Set off-diagonal k
        for i in range(n_coarse - k):
            R_k[i, i + k] = 1
            R_k[i + k, i] = 1
        R_matrices.append(R_k)
    
    # SDP model
    t = cp.Variable()
    h = cp.Variable(n_coarse)
    X = cp.Variable((n_coarse, n_coarse), symmetric=True)
    
    constraints = [
        cp.bmat([[1, h.T],
                 [h, X]]) >> 0,  # PSD constraint
        h >= 0,
        h <= 1,
        cp.sum(h) == n_coarse / 2,
    ]
    
    # Constraint on autocorrelations
    for k in range(n_coarse):
        constraints.append(cp.trace(R_matrices[k] @ X) >= t)
    
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve SDP (coarse grid)
    best_h_coarse = None
    best_c5_coarse = float('inf')
    
    # Settings for SDP solver
    settings = {
        'max_iters': 5000,
        'eps': 1e-5,
    }
    
    try:
        prob.solve(solver=cp.SCS, **settings, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_sol = h.value
            X_sol = X.value
            
            # Extract near rank-1 vector
            h_hat = nearest_rank1(X_sol)
            
            # Enforce constraints: clip and water fill
            h_projected = np.clip(h_hat, 0, 1)
            h_projected = water_fill(h_projected, n_coarse/2)
            
            # Mirror symmetry
            if np.allclose(h_projected, h_projected[::-1], rtol=1e-2):
                h_projected = (h_projected + h_projected[::-1]) / 2
            
            # Evaluate coarse C5
            c5_coarse = dx_coarse * np.max(np.correlate(h_projected, 1-h_projected, mode='full'))
            if c5_coarse < best_c5_coarse:
                best_c5_coarse = c5_coarse
                best_h_coarse = h_projected
        
    except:
        pass  # fallback
    
    # Fallback to simple start if needed
    if best_h_coarse is None:
        best_h_coarse = np.ones(n_coarse) * 0.5
        best_h_coarse = water_fill(best_h_coarse, n_coarse / 2)
        best_c5_coarse = dx_coarse * np.max(np.correlate(best_h_coarse, 1-best_h_coarse, mode='full'))

    # Stage 2: Smooth max gradient descent on coarse grid
    if time.time() - start_time > budget_s - 100:
        return (best_h_coarse, best_c5_coarse, n_coarse)

    def smooth_max_batch(h, beta, dx):
        # Compute full correlation array
        conv = np.correlate(h, 1-h, mode='full')
        # Apply soft max
        max_val = np.max(conv)  # For numerical stability
        exp_conv = np.exp(beta * dx * (conv - max_val))
        weights = exp_conv / np.sum(exp_conv)
        # Derivative of soft max
        w_rev = weights[::-1]
        u = np.correlate(weights, 1-h, mode='full')
        v = np.correlate(w_rev, h, mode='full')
        grad = dx * (u - v) 
        return grad

    h_refined = best_h_coarse.copy()
    beta = 0.5
    max_beta = 50.0
    patience = 0
    best_c5_refined = best_c5_coarse
    
    while beta <= max_beta:
        patience = 0
        prev_c5 = float('inf')
        while patience < 3 and beta <= max_beta and time.time() - start_time < budget_s - 100:
            grad = smooth_max_batch(h_refined, beta, dx_coarse)
            h_new = h_refined - 0.001 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = water_fill(h_new, n_coarse / 2)
            h_new = (h_new + h_new[::-1]) / 2
            c5_new = dx_coarse * np.max(np.correlate(h_new, 1-h_new, mode='full'))
            if c5_new < best_c5_refined:
                best_c5_refined = c5_new
                h_refined = h_new
            else:
                patience += 1

            # Continue to next beta if needed
            if c5_new > prev_c5 and beta < max_beta:
                beta *= 2
                prev_c5 = c5_new
                break
            else:
                prev_c5 = c5_new

        # Move to next beta
        if beta <= max_beta:
            beta *= 2
    
    # Stage 3: Upsample to target grid
    if time.time() - start_time > budget_s - 50:
        return (h_refined, best_c5_refined, n_coarse)
    
    # Linear interpolation
    x_coarse = np.linspace(0, 2, n_coarse)
    x_target = np.linspace(0, 2, n_target)
    h_upsampled = interp1d(x_coarse, h_refined, kind='linear')(x_target)
    h_upsampled = np.clip(h_upsampled, 0, 1)
    h_upsampled = water_fill(h_upsampled, n_target / 2)
    h_upsampled = (h_upsampled + h_upsampled[::-1]) / 2
    
    # Stage 4: Refinement on target grid
    if time.time() - start_time > budget_s - 20:
        return (h_upsampled, best_c5_refined, n_target)

    def smooth_max_full(h, beta, dx):
        conv = np.correlate(h, 1-h, mode='full')
        max_val = np.max(conv)
        exp_conv = np.exp(beta * dx * (conv - max_val))
        weights = exp_conv / np.sum(exp_conv)
        w_rev = weights[::-1]
        u = np.correlate(weights, 1-h, mode='full')
        v = np.correlate(w_rev, h, mode='full')
        grad = dx * (u - v) 
        return grad

    beta = 0.5
    max_beta = 50.0
    patience = 0
    h_final = h_upsampled.copy()
    prev_c5 = float('inf')
    
    while beta <= max_beta and time.time() - start_time < budget_s - 10:
        patience = 0
        while patience < 3 and beta <= max_beta and time.time() - start_time < budget_s - 10:
            grad = smooth_max_full(h_final, beta, dx_target)
            h_new = h_final - 0.0005 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = water_fill(h_new, n_target / 2)
            h_new = (h_new + h_new[::-1]) / 2
            c5_new = dx_target * np.max(np.correlate(h_new, 1-h_new, mode='full'))
            if c5_new < best_c5_refined:
                best_c5_refined = c5_new
                h_final = h_new
            else:
                patience += 1

            if c5_new > prev_c5 and beta < max_beta:
                beta *= 2
                prev_c5 = c5_new
                break
            else:
                prev_c5 = c5_new
    
    # Stage 5: Stochastic polishing
    # Run in short time
    safety_margin = 10
    calc_end_time = time.time() + safety_margin
    best_h_polished = h_final.copy()
    best_c5_polished = best_c5_refined

    epsilon = 0.004
    num_perturbations = min(6, n_target // 20)
    
    while time.time() < calc_end_time:
        # perturb
        indices = np.random.choice(n_target, size=num_perturbations, replace=False)
        h_trial = best_h_polished.copy()
        for idx in indices:
            delta = np.random.uniform(-epsilon, epsilon)
            h_trial[idx] += delta
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = water_fill(h_trial, n_target / 2)
        h_trial = (h_trial + h_trial[::-1]) / 2
        c5_trial = dx_target * np.max(np.correlate(h_trial, 1-h_trial, mode='full'))
        if c5_trial < best_c5_polished:
            best_c5_polished = c5_trial
            best_h_polished = h_trial
    
    return (best_h_polished, best_c5_polished, n_target)
