import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def project_to_feasible(h, target_sum):
    """Project h to feasible set with sum = target_sum and 0 <= h[i] <= 1"""
    # Clip to box
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Water filling: solve for lambda
    def sum_with_lambda(lam):
        return np.sum(np.clip(h + lam, 0.0, 1.0))
    
    # Binary search for lambda
    low, high = -1.0, 1.0
    for _ in range(30):
        mid = (low + high) / 2
        if sum_with_lambda(mid) > target_sum:
            high = mid
        else:
            low = mid
    lam = (low + high) / 2
    h_proj = np.clip(h + lam, 0.0, 1.0)
    return h_proj

def fast_convolution(h):
    """Compute full convolution h * (1-h) using FFT"""
    n = len(h)
    # Zero pad to 2*n-1
    H = np.fft.rfft(h, n=2*n-1)
    G = np.fft.rfft(1-h, n=2*n-1)
    conv = np.fft.irfft(H*G)[:2*n-1]
    return conv

def evaluate_c5(h, dx):
    """Compute C5 using fast convolution"""
    conv = fast_convolution(h)
    return dx * np.max(conv)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_coarse = 150
    n_fine = 300
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Global search parameters
    d = (n_coarse + 1) // 2  # number of free variables with symmetry
    mu0 = 0.5 * np.ones(d)
    sigma0 = 0.2
    lambda_ = 4 + int(3 * np.log(d))  # population size
    
    # Stopping conditions
    max_generations = 200
    improvement_threshold = 1e-4
    restart_threshold = 150
    
    # Fast Fourier transform
    def conv_fast(h, n=None):
        if n is None:
            n = len(h)
        H = np.fft.rfft(h, n=2*n-1)
        G = np.fft.rfft(1-h, n=2*n-1)
        conv = np.fft.irfft(H*G)[:2*n-1]
        return conv
    
    # Projection with symmetry
    def project_h(h_unsym, n):
        h_unsym = np.clip(h_unsym, 0.0, 1.0)
        h_proj = project_to_feasible(h_unsym, n / 2)
        # Apply symmetry
        if n % 2 == 0:
            h_proj[n//2:] = h_proj[n//2-1::-1]
        else:
            h_proj[n//2+1:] = h_proj[n//2-1::-1]
        return h_proj
    
    # Evolution Strategy
    best_h_coarse = None
    best_c5_coarse = np.inf
    current_h_coarse = mu0.copy()
    sigma = sigma0
    best_gen = 0
    n_restart = 0
    
    while time.time() - start_time < budget_s - 10:
        if n_restart >= 3:
            break
        
        # Sample offspring
        offspring = []
        for _ in range(lambda_):
            z = current_h_coarse + sigma * np.random.randn(d)
            h = np.ones(n_coarse)
            h[:d] = z
            h = project_h(h, n_coarse)
            offspring.append(h)
        
        # Evaluate offspring
        fitness = []
        for h_off in offspring:
            conv = conv_fast(h_off, n_coarse)
            max_val = np.max(conv) * dx_coarse
            fitness.append(max_val)
        
        # Sort
        indices = np.argsort(fitness)
        fitness = [fitness[i] for i in indices]
        offspring = [offspring[i] for i in indices]
        
        # Update best
        if fitness[0] < best_c5_coarse:
            best_c5_coarse = fitness[0]
            best_h_coarse = offspring[0].copy()
            best_gen = 0
        else:
            best_gen += 1
        
        # CMA-ES update
        # Simple rank-one update
        weights = np.log(lambda_ / 2 + 1) - np.log(np.arange(1, lambda_ + 1) + 1)
        weights = weights / np.sum(weights)
        weights = weights[weights > 0]
        weights = weights[:lambda_ // 2]
        weights = weights / np.sum(weights)
        
        # Update mean
        z_new = np.zeros(d)
        for i in range(len(weights)):
            z_new += weights[i] * (offspring[i][:d] - current_h_coarse)
        current_h_coarse += z_new
        
        # Update sigma
        sigma = sigma * np.exp(0.3 * (np.sum(z_new**2) / d - 1))
        
        # Check for restart
        if best_gen > restart_threshold and n_restart < 3:
            current_h_coarse = best_h_coarse[:d]
            sigma = sigma / 2
            n_restart += 1
            best_gen = 0
    
    # Interpolate coarse grid to fine grid
    if best_h_coarse is not None:
        x_coarse = np.linspace(0, 2, n_coarse)
        x_fine = np.linspace(0, 2, n_fine)
        h_fine_interp = interp1d(x_coarse, best_h_coarse, kind='linear', fill_value='extrapolate')
        h_fine = h_fine_interp(x_fine)
        h_fine = project_to_feasible(h_fine, n_fine / 2)
    else:
        # Fallback to uniform
        h_fine = np.ones(n_fine) * 0.5
        h_fine = project_to_feasible(h_fine, n_fine / 2)
    
    # Local refinement
    def batch_evaluate(h):
        h_sym = (h + h[::-1]) / 2  # enforce symmetry
        h_proj = project_to_feasible(h_sym, n_fine / 2)
        return evaluate_c5(h_proj, dx_fine)
    
    def compute_gradient(h, beta):
        h_sym = (h + h[::-1]) / 2
        h_proj = project_to_feasible(h_sym, n_fine / 2)
        n = len(h_proj)
        conv = conv_fast(h_proj, n)
        # Compute soft max probability
        soft_max = np.exp(beta * dx_fine * conv)
        soft_max = soft_max / np.sum(soft_max)
        # Gradients
        grad = np.zeros_like(h_proj)
        for k in range(2*n - 1):
            i = k - (n - 1)
            if i > 0:
                # Compute gradient of the k-th convolution
                grad += soft_max[k] * (1 - np.pad(h_proj, (i,0))[:n])
            else:
                # i <= 0
                grad += soft_max[k] * -np.pad(h_proj, (0,-i))[:n]
                # alt: use equivalent
        grad = dx_fine * grad
        return grad
    
    def update_h(h, steps, lr=1e-5, beta=0.5, penalty_lambda=10.0):
        for _ in range(steps):
            grad = compute_gradient(h, beta)
            h = h - lr * grad
            # Project to feasible set
            h = project_to_feasible(h, n_fine / 2)
            h = (h + h[::-1]) / 2 # maintain symmetry
            # Penalty term
            penalty = penalty_lambda * (np.sum(h) - n_fine / 2) ** 2
            if time.time() - start_time > budget_s - 5:
                break
        return h
    
    # Update with β continuation
    optimal_h = h_fine.copy()
    beta = 0.5
    penalty_lambda = 10.0
    for _ in range(20):  # Iterates to beta = 50
        optimal_h = update_h(optimal_h, 80, lr=1e-5, beta=beta, penalty_lambda=penalty_lambda)
        beta *= 2
        penalty_lambda *= 2
        if beta > 50:
            break
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final polishing by hill-climbing
    current_h = optimal_h.copy()
    current_c5 = batch_evaluate(current_h)
    for _ in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
        # Pick few indices to change
        p = min(6, n_fine // 20)
        indices = np.random.choice(n_fine, p, replace=False)
        new_h = current_h.copy()
        eps = 0.004
        
        # Perturb selected indices
        for i in indices:
            delta = np.random.uniform(-eps, eps)
            new_h[i] += delta
        
        # Project new_h
        new_h = project_to_feasible(new_h, n_fine / 2)
        new_h = (new_h + new_h[::-1]) / 2  # enforce symmetry
        
        new_c5 = batch_evaluate(new_h)
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            
    # Final refinement
    final_c5 = batch_evaluate(current_h)
    
    return (current_h, final_c5, n_fine)
