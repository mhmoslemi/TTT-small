```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.interpolate import interp1d

def correlation_fft(h):
    """Compute correlation f_k = sum h_i * (1 - h_{i+k}) for all k using FFT."""
    N = len(h)
    h_pad = np.concatenate([h, np.zeros(N)])
    g_pad = np.concatenate([1 - h, np.zeros(N)])
    c = ifft(fft(h_pad) * fft(g_pad)).real
    return c[:N]

def weighted_gradient_fft(h, p):
    """Compute weighted gradient of log-sum-exp smoothing."""
    N = len(h)
    p_pad = np.concatenate([p, np.zeros(N)])
    h_pad = np.concatenate([h, np.zeros(N)])
    g_pad = np.concatenate([1-h, np.zeros(N)])
    
    # w1 = conv(p, 1-h)
    w1 = ifft(fft(p_pad) * fft(g_pad)).real[:N]
    # w2 = conv(p_rev, h) where p_rev[k] = p[N-k]
    p_rev = np.concatenate([p[-1:0:-1], p[0:1]])
    w2 = ifft(fft(p_rev) * fft(h_pad)).real[:N]
    
    return w1 - w2

def project_onto_Ω(v):
    """Projection onto the set {h : 0 ≤ h_i ≤ 1, sum h_i = N/2}."""
    N = len(v)
    z = np.clip(v, 0, 1)
    r = N/2 - np.sum(z)
    if abs(r) < 1e-12:
        return z
    
    # Waterfilling
    I = np.where((z > 0) & (z < 1))[0]
    for _ in range(100):  # At most 100 iterations
        if len(I) == 0:
            break
        delta = r / len(I)
        z[I] += delta
        r -= delta * len(I)
        # Update I
        I = np.where((z > 0) & (z < 1))[0]
        if abs(r) < 1e-12:
            break
    else:
        # Should not happen
        z = np.clip(z, 0, 1)
    
    # Ensure sum = N/2 exactly
    z = z * (N / 2) / np.sum(z)
    return z

def binary_from_top(h, target_sum):
    """Create binary vector using thresholding"""
    N = len(h)
    indices = np.argsort(h)[::-1]  # descending order
    h_binary = np.zeros(N)
    h_binary[indices[:target_sum]] = 1
    return h_binary

def fractional_correction(h_binary):
    """Fix sum constraint with nearby fractional values."""
    N = len(h_binary)
    target = N/2
    current_sum = np.sum(h_binary)
    diff = target - current_sum
    
    if abs(diff) < 1e-10:
        return h_binary
    
    # Relax some zeros to fractions and some ones to fractions
    # Find entries around 0.5
    if abs(diff) <= 1:
        # For small changes, we pick two entries, one from 0-zone and one from 1-zone
        # Find entries near 0 and 1
        zero_indices = np.where(h_binary == 0)[0]
        one_indices = np.where(h_binary == 1)[0]
        
        if len(zero_indices) > 0 and len(one_indices) > 0:
            # Pick one zero and one one to adjust
            i_zero = zero_indices[np.random.randint(len(zero_indices))]
            i_one = one_indices[np.random.randint(len(one_indices))]
            
            # Adjust so score improves
            # We model this as linear programming since only a few variables
            # are modified, still within very low dimension
            v = np.copy(h_binary)
            # Try relaxing up to two variables
            # Make v[i_zero] = small fraction (say 0.5) and v[i_one] = 0.5
            # This changes the sum minimally
            if abs(diff) > 0.5:
                v[i_zero] = min(1.0, v[i_zero] + diff)
                v[i_one] = max(0.0, v[i_one] + (diff - (v[i_zero] - h_binary[i_zero])))
            else:
                v[i_zero] = 0.5
                v[i_one] = 0.5
                
            if abs(np.sum(v) - target) < 1e-10:
                return v
        return h_binary
    
    # For larger adjustments, just finalize round
    # Might not be needed in practice if rounding is good
    return h_binary

def swap(h, i, j):
    """Swap h[i], h[j]."""
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def c5_of_h(h):
    """Original function: compute the maximum overlap integral C5"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def evaluate_erdos_solution(h_values):
    """Wrapper for the original evaluator"""
    return c5_of_h(h_values)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    N = 256
    
    # Initial: random binary construction
    h = np.zeros(N)
    h[:N//2] = 1.
    np.random.shuffle(h)
    h = project_onto_Ω(h)
    
    # Solve smooth approximation with accelerated gradient
    τ = 0.05
    outer_cycles = 6
    inner_iters = 150

    for outer in range(outer_cycles):
        # Start with previous h if not first iteration
        h_old = h.copy()
        # Simple backtracking line search stepsize
        alpha = 1.0
        
        for t in range(inner_iters):
            if time.time() - start > budget_s - 30:
                break
                
            # Nesterov extrapolation
            beta = (t-1)/(t+2)
            y = h + beta*(h - h_old)
            
            # Evaluate f_k for all k via FFT
            f = correlation_fft(y)
            
            # Log-sum-exp to smooth max
            f_scaled = f / τ
            
            # Clamp for numerical stability
            max_f = np.max(f_scaled)
            f_stable = f_scaled - max_f
            exp_f = np.exp(f_stable)
            exp_sum = np.sum(exp_f)
            p = exp_f / exp_sum
            
            # Gradient via FFT
            grad = weighted_gradient_fft(y, p)
            
            # Backtracking line search to find alpha
            while True:
                z = y - alpha * grad
                z_proj = project_onto_Ω(z)
                F_current = τ * np.log(exp_sum) + max_f * τ
                F_new = τ * np.log(np.sum(np.exp((correlation_fft(z_proj) / τ - np.max(correlation_fft(z_proj) / τ))))) + np.max(correlation_fft(z_proj))
                if F_new <= F_current - 0.5 * alpha * np.dot(grad, grad) or alpha < 1e-10:
                    break
                alpha *= 0.5
            
            h_old = h.copy()
            h = z_proj
        
        if time.time() - start > budget_s - 30:
            break
        
        τ *= 0.5
    
    # Round to binary vector
    h_bin = binary_from_top(h, N//2)
    h_refined = fractional_correction(h_bin)
    
    # Local combinatorial polishing
    best_h = h_refined
    best_c5 = c5_of_h(best_h)
    
    # Weak annealing-style local searches with swaps
    n_swaps = 0
    max_swaps = 5000
    while time.time() - start < budget_s - 80 and n_swaps < max_swaps:
        i = np.random.randint(N)
        j = np.random.randint(N)
        if i == j:
            continue
        
        cand = swap(best_h, i, j)
        cand_c5 = c5_of_h(cand)
        
        if cand_c5 < best_c5 - 1e-9:
            best_h = cand
            best_c5 = cand_c5
            
        n_swaps += 1
    
    # Optional high-res refinement
    if time.time() - start < budget_s - 5:
        upsampled_h = np.interp(np.linspace(0, 2, 512), np.linspace(0, 2, 256), best_h)
        upsampled_h = project_onto_Ω(upsampled_h)
        
        # Run few iterations of SLSQP
        res = opt.minimize(
            fun=c5_of_h,
            x0=upsampled_h,
            method='SLSQP',
            bounds=[(0, 1)] * 512,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - 256},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        final_h = res.x
        final_c5 = c5_of_h(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    return (best_h, best_c5, len(best_h))
```
