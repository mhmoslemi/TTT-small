def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def compute_correlation_update(h, idx1, idx2):
    """Incrementally update correlation for swapping 1 and 0 at indices idx1 and idx2"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # For each shift, compute updated correlation
    old_corr = corr.copy()
    # We'll recompute the correlation for all relevant shifts
    h_new = h.copy()
    h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
    new_corr = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
    return np.max(new_corr)

def project_to_feasible(h):
    """Ensure sum(h) = n/2 and h in [0,1]^n"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    # Uniform adjustment to meet sum
    adjustment = deficit / n
    h += adjustment
    h = np.clip(h, 0, 1)
    return h
