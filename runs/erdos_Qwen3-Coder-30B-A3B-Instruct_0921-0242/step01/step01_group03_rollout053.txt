import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def c5_of_h(h):
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def compute_correlation_incremental(h, i1, i2):
    """Efficiently compute change in correlation when swapping h[i1] and h[i2]"""
    dx = 2.0 / len(h)
    # Get the difference in correlation function due to swapping
    # Create a temporary vector with swapped values for correlation
    h_temp = h.copy()
    h_temp[i1], h_temp[i2] = h_temp[i2], h_temp[i1]
    corr_new = np.correlate(h_temp, 1.0 - h_temp, mode='full') * dx
    corr_old = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr_new) - np.max(corr_old)

def project_to_feasible(h):
    """Project h to feasible set: clip to [0,1] and adjust to sum to n/2"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    adjustment = deficit / n
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def binary_to_float(b, n_points):
    """Convert binary mask b to a float vector with sum = n/2"""
    h = b.astype(float)
    omega = np.sum(h)
    h = h * (n_points / 2) / omega
    return h

def generate_seeds(n_points):
    """Generate diverse seed binary patterns"""
    seeds = []
    n_ones = n_points // 2
    
    # Random binary with exactly n_ones
    b = np.zeros(n_points, dtype=int)
    b[:n_ones] = 1
    np.random.shuffle(b)
    seeds.append(('random', b))
    
    # Two-blocks pattern: two intervals of length 0.5 each
    b = np.zeros(n_points, dtype=int)
    l = n_points // 4
    b[l:3*l] = 1
    seeds.append(('blocks2', b))
    
    # Three-blocks pattern
    b = np.zeros(n_points, dtype=int)
    b[n_points//4:2*n_points//4] = 1
    b[3*n_points//4:4*n_points//4] = 1
    seeds.append(('blocks3', b))
    
    # Alternating
    b = np.zeros(n_points, dtype=int)
    b[::2] = 1
    seeds.append(('alternating', b))
    
    return seeds

def simulated_annealing(n_points, budget_s):
    """Simulated annealing on binary mask"""
    np.random.seed(42)
    dx = 2.0 / n_points
    start = time.time()
    n_ones = n_points // 2
    
    # Generate seeds
    seeds = generate_seeds(200)  # Fixed 200 for speed
    current_best_val = np.inf
    current_best = None
    
    # Run SA on each seed
    for seed_name, b in seeds:
        if time.time() - start > budget_s - 1:
            break
            
        h = binary_to_float(b, n_points)
        current_val = c5_of_h(h)
        if current_val < current_best_val:
            current_best_val = current_val
            current_best = h.copy()
        
        # Run simulated annealing with small C5 units
        T = 0.02  # temperature
        cooling = 0.995
        max_iter = 30000
        accepted = 0
        for i in range(max_iter):
            if time.time() - start > budget_s - 1:
                break

            # swap indices
            i1, i2 = np.random.randint(0, n_points, size=2)
            if b[i1] == b[i2]:
                continue  # must swap 0 with 1 to maintain sum
            # Temporarily modify the binary mask
            b[i1], b[i2] = b[i2], b[i1]
            # evaluate change via correlation difference
            C5_change = compute_correlation_incremental(h, i1, i2)
            if C5_change < 0:
                current_val += C5_change
                if current_val < current_best_val:
                    current_best_val = current_val
                    current_best = h.copy()
                accepted += 1
            else:
                # Metropolis
                if np.random.rand() < np.exp(-C5_change / T):
                    current_val += C5_change
                    accepted += 1
                else:
                    b[i1], b[i2] = b[i2], b[i1]  # Revert
            
            if accepted % 100 == 0:
                T *= cooling
    
    return current_best_val, current_best

def lp_refinement(h_start, b):
    """Refine a binary mask b to continuous h with lower C5 using LP"""
    n_points = len(h_start)
    dx = 2.0 / n_points
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [h[i] >= 0 for i in range(n_points)]
    constraints += [h[i] <= 1 for i in range(n_points)]
    
    # Sum constraint
    constraints += [cp.sum(h) == n_points / 2] 
    
    # Max overlap constraint for each shift k
    corr = np.correlate(h_start, 1.0 - h_start, mode='full') * dx
    max_k = np.argmax(corr)
    max_shift = max_k - (n_points - 1)
    
    # Manually compute correlation indices for effectiveness of growth
    
    shifted = np.zeros(n_points)
    for k in range(2*n_points - 1):
        shift_k = k - (n_points - 1)  # from -n+1 to n-1 
        shifted.fill(0)
        for i in range(n_points):
            shifted[i] = h_start[(i + shift_k) % n_points]
        # Approximate bilinear term
        bilin = 0
        for i in range(n_points):
            if b[i] == 1:
                bilin += h[i] * (1 - shifted[i])
            else:
                # b[i] == 0, so h[i]*shifted[i] = 0 always
                bilin += 0.0 
        constraints.append(bilin <= t)
        
    # When too much complex structure, reduce to few shifts
    # In the full case use first few shifts
    for k in range(2*n_points - 1):
        shift_k = k - (n_points - 1)
        # Compute shifted version
        shifted = np.zeros(n_points)
        for i in range(n_points):
            shifted[i] = h_start[(i + shift_k) % n_points]
        # Compute overlap at shift
        # h[i]* (1-shifted[i]) 
        # But avoid quadratic terms because linearizing
        # Let's instead bound each term
        bilin = 0
        for i in range(n_points):
            if b[i] == 1:
                bilin += h[i] * (1-shifted[i])
            # else h_i * (1-shifted_i) = 0 because h_i=0, so bilin += 0  
        constraints.append(bilin <= t)
    
    # Actual constraint based on w: all individual overlaps <= t
    # Slightly hacky proportional estimate based on a few key shifts
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        # if successful, return refined h_vector
        refined = h.value
        return refined
    except:
        return h_start

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    n_points = 200  # Reduced from 300 for speed
        
    # Phase 1: Simulated Annealing
    print(f"Starting phase 1 (SA)")
    c5_best, h_best = simulated_annealing(n_points, budget_s * 0.55)
    print(f"Phase 1 ended with C5={c5_best}")
    
    # Phase 2: LP refinement
    print(f"Starting phase 2 (LP)")
    if time.time() - start_time > budget_s * 0.9:
        h_final = h_best.copy()
        return (h_final, c5_best, n_points)
        
    # Convert h_best to binary
    b_best = (h_best > 0.5).astype(int)
    # Smooth the refinement to reduce artifacts
    h_refined = lp_refinement(h_best, b_best)
    c5_refined = c5_of_h(h_refined)
    if c5_refined < c5_best:
        h_best = h_refined.copy()
        c5_best = c5_refined
    print(f"Phase 2 ended with C5={c5_best}")
    
    # Phase 3: further small random adjustments
    print(f"Starting phase 3 (random polishing)")
    for i in range(2000):
        if time.time() - start_time > budget_s * 0.95:
            break
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.02, 0.02)
        h_test = h_best.copy()
        h_test[idx] += delta
        h_test = project_to_feasible(h_test)
        c5_test = c5_of_h(h_test)
        if c5_test < c5_best:
            h_best = h_test.copy()
            c5_best = c5_test
    print(f"Phase 3 ended with C5={c5_best}")
        
    # Phase 4: Upsample and refine with SLSQP
    print(f"Starting phase 4 (upsample+SLSQP)")
    h_upsampled = np.interp(np.linspace(0, 2, 400), np.linspace(0, 2, n_points), h_best)
    h_upsampled = h_upsampled * (400 / 2) / np.sum(h_upsampled)
    
    # Optimization with SLSQP
    bounds = [(0,1)] * 400
    eq_constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - 200}
    
    # Decrease objective for convergence?
    def obj(h):
        return c5_of_h(h)
    
    # Updated SLSQP
    h_opt = minimize(
        fun=obj,
        x0=h_upsampled,
        method='SLSQP',
        bounds=bounds,
        constraints=eq_constraint,
        options={'ftol': 1e-12, 'maxiter': 200}
    ).x
    
    h_final = project_to_feasible(h_opt)
    c5_final = c5_of_h(h_final)
    
    print(f"Final result with C5={c5_final}")
    
    return (h_final, c5_final, 400)
