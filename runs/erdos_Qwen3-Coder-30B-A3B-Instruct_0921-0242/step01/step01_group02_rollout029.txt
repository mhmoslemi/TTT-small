import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import math

def c5_fft(h):
    """Compute C5 using FFT for efficiency."""
    g = 1 - h
    H = rfft(h)
    G = rfft(g)
    c = irfft(H * G, 2*len(h)-1)
    return np.max(c)

def repair_weight(h, target_sum):
    """Repair to exact sum constraint."""
    current_sum = np.sum(h)
    diff = int(target_sum - current_sum)
    if diff == 0:
        return h
    elif diff > 0:  # Need to increase sum
        zero_indices = np.where(h == 0)[0]
        if len(zero_indices) >= diff:
            indices = np.random.choice(zero_indices, size=diff, replace=False)
            h[indices] = 1
    else:  # Need to decrease sum
        one_indices = np.where(h == 1)[0]
        if len(one_indices) >= -diff:
            indices = np.random.choice(one_indices, size=-diff, replace=False)
            h[indices] = 0
    return h

def crossover(p1, p2):
    """Two-point crossover."""
    n = len(p1)
    c1 = np.random.randint(1, n-1)
    c2 = np.random.randint(c1+1, n)
    child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
    return child

def mutate_swap(h):
    """Swap mutation."""
    one_indices = np.where(h == 1)[0]
    zero_indices = np.where(h == 0)[0]
    if len(one_indices) > 0 and len(zero_indices) > 0:
        i1 = np.random.choice(one_indices)
        i0 = np.random.choice(zero_indices)
        h[i1], h[i0] = h[i0], h[i1]
    return h

def hill_climb(h, dx, max_swaps=500):
    """Targeted local hill-climber."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = c5_fft(h) * dx
    for _ in range(max_swaps):
        c = correlate(h, 1-h, mode='full')
        k_star = np.argmax(c) - len(h) + 1
        best_delta = 0
        best_swap = None
        # Sample a few swaps
        for _ in range(200):
            one_indices = np.where(h == 1)[0]
            zero_indices = np.where(h == 0)[0]
            if len(one_indices) == 0 or len(zero_indices) == 0:
                continue
            i1 = np.random.choice(one_indices)
            i0 = np.random.choice(zero_indices)
            delta = 0
            if k_star >= 0:
                if i1 + k_star < len(h) and i0 + k_star < len(h):
                    delta = -dx * (h[i1] * h[i1 + k_star] - (1 - h[i1]) * h[i1 + k_star] - 
                                  h[i0] * h[i0 + k_star] + (1 - h[i0]) * h[i0 + k_star])
            else:
                if i1 + k_star >= 0 and i0 + k_star >= 0:
                    delta = -dx * (h[i1] * h[i1 + k_star] - (1 - h[i1]) * h[i1 + k_star] - 
                                  h[i0] * h[i0 + k_star] + (1 - h[i0]) * h[i0 + k_star])
            if delta < best_delta:
                best_delta = delta
                best_swap = (i1, i0)
        if best_swap is not None:
            i1, i0 = best_swap
            h[i1], h[i0] = h[i0], h[i1]
            new_c5 = c5_fft(h) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            break
    return best_h, best_c5
