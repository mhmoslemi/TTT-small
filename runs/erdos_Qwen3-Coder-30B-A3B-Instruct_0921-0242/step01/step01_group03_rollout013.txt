def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def project_to_feasible(Y, N):
    """Project Y rows to feasible set with sum = N/2 and h_i in [0,1]"""
    # h_i = ||Y_i||^2 
    h = np.sum(Y**2, axis=1)
    
    # 1. clip norms to [0,1]
    norms = np.sqrt(np.sum(Y**2, axis=1))
    mask = norms > 1.0
    if np.any(mask):
        Y[mask] = Y[mask] / norms[mask][:, np.newaxis]
        
    # 2. enforce mass constraint
    S = np.sum(norms**2)
    if S > 0:
        alpha = np.sqrt((N/2)/S)
        Y *= alpha
    
    # 3. reclip if needed
    norms = np.sqrt(np.sum(Y**2, axis=1))
    mask = norms > 1.0
    if np.any(mask):
        Y[mask] = Y[mask] / norms[mask][:, np.newaxis]
    
    return Y

def compute_correlations(Y):
    """Fast computation of all c_k using FFT"""
    N, r = Y.shape
    c = np.zeros((N,))  # one row for each shift
    for d in range(r):
        y = Y[:, d]
        # Transform to frequency domain
        y_full = np.concatenate((y, np.zeros(N)))
        fft_y = fft(y_full)
        # circular autocorrelation: ifft(|fft(y)|^2)
        auto_corr_fft = fft_y * np.conj(fft_y)
        auto_corr = ifft(auto_corr_fft).real[:N]
        c += auto_corr  # add contributions from all dimensions
    return c

def gradient_of_phi(Y, c, tau):
    """Computes gradient of the LSE surrogate φ_τ(Y)"""
    # weights p_k
    exp_c = np.exp(-c/tau)
    Z = np.sum(exp_c)
    p = exp_c / Z
    
    N, r = Y.shape
    grad = np.zeros_like(Y)
    for d in range(r):
        y = Y[:, d]
        # convolve with p and reverse p
        p_full = np.concatenate((p, np.zeros(N)))
        y_full = np.concatenate((y, np.zeros(N)))
        y_rev_full = np.concatenate((y[::-1], np.zeros(N)))
        
        p_fft = fft(p_full)
        y_fft = fft(y_full)
        y_rev_fft = fft(y_rev_full)
        
        conv1 = ifft(p_fft * y_fft).real[:N]
        conv2 = ifft(p_fft * y_rev_fft).real[:N]
        # accumulate gradients
        grad[:, d] = conv1 + conv2
    return grad

def line_search(Y, grad, phi_old, c, tau, max_step=1.0):
    """Backtracking line search for scalar surrogate"""
    alpha = max_step
    # condition: phi(Y - alpha*grad) > phi_old 
    # for maximization, use sufficient decrease
    # we want: phi(Y - alpha*grad) >= phi_old + c1*alpha*(grad^T * (-grad))
    c1 = 1e-4
    
    for i in range(30):  # max iterations
        Y_new = Y - alpha * grad
        # project it
        Y_new = project_to_feasible(Y_new, len(c))
        c_new = compute_correlations(Y_new)
        phi_new = -tau * np.log(np.sum(np.exp(-c_new/tau)))
        
        if phi_new > phi_old:  # improvement
            return alpha, Y_new
        alpha *= 0.5
    return alpha, Y_new
