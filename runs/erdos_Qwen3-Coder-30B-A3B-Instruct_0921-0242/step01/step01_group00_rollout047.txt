import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def sample_step_function(block_lengths, block_heights, n_points):
    """Sample a step function with given block parameters."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    cum_len = 0
    for i, (length, height) in enumerate(zip(block_lengths, block_heights)):
        start_idx = int(cum_len / dx)
        end_idx = int((cum_len + length) / dx)
        if end_idx > n_points:
            end_idx = n_points
        h[start_idx:end_idx] = height
        cum_len += length
    return h

def parameter_to_h(u, v, n_points):
    """Convert raw parameters to feasible h vector."""
    # Normalize lengths to sum to 2
    exp_u = np.exp(u)
    lengths = 2 * exp_u / np.sum(exp_u)
    
    # Compute heights (clipped to [0,1])
    heights = 1.0 / (1.0 + np.exp(-v))
    
    # Scale heights to satisfy integral constraint
    area = np.sum(heights * lengths)
    if area > 0:
        alpha = 1.0 / area
        heights = np.minimum(1.0, alpha * heights)
        # If we hit upper bound, redistribute
        if np.any(heights >= 1.0):
            # Scale down others
            over = np.sum(heights - 1.0)  # negative
            under = np.sum(1.0 - heights[heights < 1.0])
            if under > 0:
                scale = over / under
                for i in range(len(heights)):
                    if heights[i] < 1.0:
                        heights[i] = max(0.0, heights[i] - scale * (1.0 - heights[i]))
    
    # Sample into fine grid
    h = sample_step_function(lengths, heights, n_points)
    return h

def objective_block_params(params, n_points=200, n_eval=200):
    """Objective function that evaluates the C5 bound for a given parameter set."""
    K = len(params) // 2
    u = params[:K]
    v = params[K:]
    h = parameter_to_h(u, v, n_points)
    
    # Evaluation on a coarse grid
    dx = 2.0 / n_eval
    n_shifts = 2 * n_eval  # full correlation length (after shifting)
    h_pad = np.concatenate([h, [0]*n_eval, h])
    corr = correlate(h_pad, 1 - h_pad, mode='full')
    corr = corr[n_eval:(n_eval + n_shifts)]
    corr = corr * dx
    # Find max over the cross-correlation (shifts)
    max_val = np.max(corr)
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 400  # target discretization
    n_eval = 800   # fine grid for evaluation
    K = 6          # number of blocks
    
    # Time budget splits
    global_search_time = 0.3 * budget_s
    fine_refinement_time = 0.3 * budget_s
    swap_time = 0.3 * budget_s
    
    # Initialize best solution
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Global optimization with differential evolution
    bounds = [(-5, 5)] * (2 * K)  # log length and logit height bounds
    remaining_time = global_search_time
    
    def objective(params):
        return objective_block_params(params, n_points, n_eval)
    
    res = differential_evolution(objective, bounds, seed=seed, 
                               maxiter=500, popsize=10*K, 
                               disp=False, 
                               callback=lambda x, convergence: None)
    
    # Extract best from DE
    u_best = res.x[:K]
    v_best = res.x[K:]
    h_best = parameter_to_h(u_best, v_best, n_points)
    c5_best = evaluate_c5(h_best)
    
    if c5_best < best_c5:
        best_c5 = c5_best
        best_h = h_best.copy()
    
    # Early stop if target reached
    if best_c5 <= 0.3807:
        return (best_h, best_c5, n_points)
    
    # Phase 2: Local refinement using projected gradient descent
    # Make a copy of best solution
    h_current = best_h.copy()
    c5_current = best_c5
    
    step_time = time.time() - start_time
    max_iter = 5000 // (0.1 + (time.time() - start_time) / (global_search_time + 0.1))
    for i_iter in range(max_iter):
        if time.time() - start_time > fine_refinement_time + step_time:
            break
        
        # Compute all correlations
        corr = correlate(h_current, 1 - h_current, mode='full') 
        # Get max index
        k_star = np.argmax(corr)
        dx = 2.0 / len(h_current)
        # compute gradient ∂I_{k*}/∂h_i = dx * (1-h_{i+k*} - h_{i-k*})
        g = np.zeros_like(h_current)
        # construct shifted versions using modularity
        shifts = np.arange(len(h_current))
        # perform vectorized gradient computation
        for i in range(len(h_current)):
            left_idx = (i - k_star) % len(h_current)
            right_idx = (i + k_star) % len(h_current)
            g[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Perform projected gradient step
        learning_rate = 0.1
        h_trial = h_current - learning_rate * g
        # Clip
        h_trial = np.clip(h_trial, 0, 1)
        # Project to sum=1
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_current:
            h_current = h_trial
            c5_current = c5_trial
        else:
            # Try with smaller step
            learning_rate /= 2
            h_trial = h_current - learning_rate * g
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < c5_current:
                h_current = h_trial
                c5_current = c5_trial
            
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
    
    # Phase 3: Conservative hill climbing with swaps
    h_final = h_current.copy()
    c5_final = c5_current
    
    step_time = time.time() - start_time
    max_swaps = 100000 // (0.1 + (time.time() - start_time) / (global_search_time + fine_refinement_time + 0.1))
    
    for i_swap in range(max_swaps):
        if time.time() - start_time > budget_s - 0.2:
            break
        # Pick two points, one with value > 0, one with value < 1
        idxs = np.where((h_final > 0) & (h_final < 1))[0]
        if len(idxs) < 2:
            break
        i, j = np.random.choice(idxs, 2, replace=False)
        delta = min(h_final[i], 1.0 - h_final[j], 0.01)
        if delta < 1e-6:
            continue
        
        # Transfer
        h_trial = h_final.copy()
        h_trial[i] -= delta
        h_trial[j] += delta
        # Project
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_final:
            c5_final = c5_trial
            h_final = h_trial.copy()
        
        # Update best
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            
    # Final check
    if best_c5 <= 0.3807:
        return (best_h, best_c5, n_points)
    
    # Ensure valid result if we haven't hit the target
    # Recompute final
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
