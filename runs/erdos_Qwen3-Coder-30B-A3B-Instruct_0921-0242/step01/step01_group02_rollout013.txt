import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import math

def c5_of_h_fft(h):
    """Compute C5 using FFT for fast convolution."""
    g = 1 - h
    n = len(h)
    n_full = 2 * n - 1
    H = rfft(h)
    G = rfft(g)
    c = irfft(H * G, n_full)
    max_corr = np.max(c)
    dx = 2.0 / n
    return max_corr * dx

def repair_weight(h, n_points, target_sum):
    """Repair h to satisfy sum constraint."""
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h.copy()
    diff = target_sum - s
    if diff > 0:  # Need to add ones
        zero_indices = np.where(h == 0)[0]
        if len(zero_indices) >= diff:
            to_flip = np.random.choice(zero_indices, int(diff), replace=False)
            h[to_flip] = 1
        else:
            # If we don't have enough zeros, clip
            h[zero_indices] = 1
            # Overcompensated, so clip some ones
            one_indices = np.where(h == 1)[0]
            to_flip = np.random.choice(one_indices, int(abs(diff)), replace=False)
            h[to_flip] = 0
    else:  # Need to subtract ones
        one_indices = np.where(h == 1)[0]
        if len(one_indices) >= -diff:
            to_flip = np.random.choice(one_indices, int(-diff), replace=False)
            h[to_flip] = 0
        else:
            # In case of error, clip
            h[one_indices] = 0
            # Add some zeros if necessary
            zero_indices = np.where(h == 0)[0]
            to_flip = np.random.choice(zero_indices, int(abs(diff)), replace=False)
            h[to_flip] = 1
    return h

def crossover(p1, p2):
    """Two-point crossover."""
    n = len(p1)
    c1 = np.random.randint(1, n-1)
    c2 = np.random.randint(c1+1, n)
    child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
    return repair_weight(child, n, n/2)

def mutate_swap(h):
    """Swap mutation."""
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > 0 and len(zeros) > 0:
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        h[i1], h[i0] = 0, 1
    return h

def mutate_block_swap(h):
    """Block swap mutation."""
    n = len(h)
    block_len = np.random.randint(1, min(5, n // 4))
    start_pos = np.random.randint(0, n - block_len)
    end_pos = start_pos + block_len
    
    block = h[start_pos:end_pos].copy()
    h[start_pos:end_pos] = 0  # Clear block
    new_start = np.random.randint(0, n - block_len)
    h[new_start:new_start + block_len] = block
    return repair_weight(h, n, n/2)

def hill_climb(h, max_swaps=500):
    """Local hill climbing around the maximum shift."""
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    best_c5 = c5_of_h_fft(h)
    swaps = 0
    
    while swaps < max_swaps:
        all_corr = correlate(h, 1 - h, mode='full')
        k_star = np.argmax(all_corr) - (n - 1)
        # Estimate contribution of swaps
        max_shift_corr = all_corr[np.argmax(all_corr)]
        new_c5 = best_c5
        best_swap = None
        
        for _ in range(50):  # sample 50 swaps
            if swaps >= max_swaps:
                break
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            delta = 0
            # Compute change to correlation at k*
            i1_k = (i1 + k_star) % n
            i0_k = (i0 + k_star) % n
            delta = 0
            # If i1_k exists as a 1, it contributes to 0
            # If i0_k exists as a 1, it contributes to 1
            if (h[i1_k] == 1):
                delta -= dx
            if (h[i0_k] == 0):
                delta += dx
            # Minimize C5, so minimize -corr
            if delta < 0:
                if best_swap is None or delta < best_swap[0]:
                    best_swap = (delta, i1, i0)
        
        if best_swap is None:
            break
        d, i1, i0 = best_swap
        h[i1], h[i0] = 0, 1
        swaps += 1
        new_c5 = c5_of_h_fft(h)
        if new_c5 < best_c5:
            best_c5 = new_c5

    return h, best_c5
