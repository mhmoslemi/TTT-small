def build_shift_matrices(N):
    """Build sparse CSR shift matrices A_k for k=0..N-1"""
    A = []
    for k in range(N):
        data = np.ones(N - k)
        row = np.arange(N - k)
        col = np.arange(k, N)
        A_k = csr_matrix((data, (row, col)), shape=(N, N))
        A.append(A_k)
    return A

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # correlate h with (1-h) using FFT
    corr = np.correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return dx * (len(h) / 2.0 - max_corr)

def bounded_simplex_projection(h, target_sum):
    """Project h onto the bounded simplex with sum=target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    deficit = target_sum - current_sum

    if abs(deficit) < 1e-10:
        return h

    # Distribute deficit uniformly across non-zero entries
    nz = np.where(h > 1e-10)[0]
    if len(nz) == 0:
        # All zero, set to be uniform and sum to target
        h.fill(target_sum / len(h))
        return h

    adjustment = deficit / len(nz)
    h[nz] += adjustment
    h = np.clip(h, 0, 1)
    
    # Fine tweak if needed
    current_sum = np.sum(h)
    deficit = target_sum - current_sum
    if abs(deficit) < 1e-10:
        return h
    
    # Use the complement approach: move from any
    # entry with h_i < 1 to one with h_i > 0
    while abs(deficit) > 1e-10:
        i = np.argmax(h) if deficit > 0 else np.argmin(h)
        delta = min(abs(deficit), 1 - h[i]) if deficit > 0 else min(abs(deficit), h[i])
        if delta < 1e-12:
            break
        h[i] += delta * (1 if deficit > 0 else -1)
        deficit -= delta * (1 if deficit > 0 else -1)
    
    return np.clip(h, 0, 1)

def eigen_rounding(X):
    """Principal eigenvalue rounding"""
    try:
        # Compute leading eigenpair
        evals, evecs = eigs(X, k=1, which='LM')
        v = evecs[:,0].real
        lambda_1 = evals[0].real
        h_eig = np.sqrt(lambda_1) * v
        h_eig = bounded_simplex_projection(h_eig, len(h_eig)/2)
        return h_eig
    except:
        return None

def threshold_rounding(X):
    """Diagonal thresholding rounding"""
    h_diag = np.sqrt(np.diag(X))
    idx_sort = np.argsort(h_diag)[::-1]
    h_thr = np.zeros_like(h_diag)
    h_thr[idx_sort[:len(h_diag)//2]] = 1.0
    return h_thr

def correlate_incremental(h0, h1, A):
    """Correlate h0 and h1 incrementally for small shifts"""
    corrs = np.array([np.sum(h0 * (A[k] @ h1)) for k in range(len(h0))])
    return corrs

def optimized_swap_search(h0, A, budget_s):
    """Perform deterministic swap-based local search on binary mask."""
    # Fast protocol
    N = len(h0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()
    
    # Use FFT for full correlation
    corr_full = np.correlate(best_h, 1-best_h, mode='full')
    corr_vals = corr_full
    
    iters = 0
    while time.time() - start_time < budget_s - 10.0:
        # Randomly select a "1" and a "0"
        idx1 = np.random.choice(np.where(best_h > 0.5)[0])
        idx0 = np.random.choice(np.where(best_h < 0.5)[0])
        
        # Construct swap candidate
        test_h = best_h.copy()
        test_h[idx1] = 0
        test_h[idx0] = 1
        
        test_c5 = evaluate_c5(test_h)
        if test_c5 < best_c5:
            best_h = test_h
            best_c5 = test_c5
            
            # Update full correlation
            corr_full = np.correlate(best_h, 1-best_h, mode='full')
            corr_vals = corr_full
        iters += 1
    return best_h, best_c5
