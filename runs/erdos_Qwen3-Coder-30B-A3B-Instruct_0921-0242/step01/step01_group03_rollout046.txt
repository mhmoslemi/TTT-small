```python
import numpy as np
import scipy.optimize as opt
import time
import scipy.fft as fft
import cvxpy as cp

def correlation_fft(h):
    """Compute all correlation values f_k(h) = sum h_i * (1 - h_{i+k}) using FFT"""
    N = len(h)
    h_pad = np.concatenate([h, np.zeros(N)])
    g_pad = np.concatenate([1 - h, np.zeros(N)])
    c = fft.fft(h_pad) * fft.fft(g_pad)
    return np.real(fft.ifft(c))[:N]

def weighted_gradient_fft(h, p):
    """Compute weighted gradient of smooth max with weights p"""
    N = len(h)
    p_pad = np.concatenate([p, np.zeros(N)])
    h_pad = np.concatenate([h, np.zeros(N)])
    g_pad = np.concatenate([1 - h, np.zeros(N)])
    
    w1 = fft.ifft(fft.fft(p_pad) * fft.fft(g_pad)).real[:N]
    w2 = fft.ifft(fft.fft(p_pad[::-1]) * fft.fft(h_pad)).real[:N]
    
    return w1 - w2

def project_onto_Ω(v):
    """Euclidean projection onto { h : 0 <= h_i <= 1, sum h_i = N/2 }"""
    N = len(v)
    z = np.clip(v, 0, 1)
    r = N/2 - np.sum(z)
    
    if abs(r) < 1e-12:
        return z
    
    # Water-filling algorithm
    for _ in range(100):  # At most 100 iterations
        I = np.where((z > 0) & (z < 1))[0]
        if len(I) == 0:
            break
        r = N/2 - np.sum(z)
        if abs(r) < 1e-12:
            break
        z[I] += r / len(I)
        z = np.clip(z, 0, 1)
    
    return z

def binary_from_top(h, count):
    """Convert h to binary with 'count' ones, at the top values"""
    indices = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    h_bin[indices[:count]] = 1.0
    return h_bin

def fractional_correction(h_bin):
    """Optionally adjust a binary vector to get exact sum = N/2 using a small LP"""
    N = len(h_bin)
    # Find indices that are near 0.5
    h_cont = h_bin.copy()
    # Keep near-binary entries as continuous variables (up to 4)
    diff = h_cont - np.round(h_cont)
    indices = np.argsort(np.abs(diff))[::-1]
    keep_indices = indices[:4]  # Keep at most 4 variables
    if len(keep_indices) < 2 or np.all(np.abs(diff[keep_indices]) < 0.25):
        return h_cont
    
    # Solve simple LP for these entries
    x = cp.Variable(len(keep_indices))
    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) + np.sum(h_cont[~np.isin(np.arange(N), keep_indices)]) == N/2
    ]
    objective = cp.Minimize(cp.sum_squares(x))  # Just to make it solvable
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        x_vals = x.value
        h_cont[keep_indices] = x_vals
        h_cont = project_onto_Ω(h_cont)
    except:
        pass
    return h_cont

def swap(h, i, j):
    """Exchange h[i] and h[j]"""
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def c5_of_h(h):
    """Compute C5 directly using correlation"""
    corr = correlation_fft(h)
    return np.max(corr)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    N = 256
    dx = 2.0 / N
    
    # Initialize with binary vector (half zeros, half ones)
    h = np.zeros(N)
    h[:N//2] = 1.0
    np.random.shuffle(h)
    h = project_onto_Ω(h)
    
    # Smoothing parameter and outer iteration parameters
    τ_start = 0.05
    τ_factor = 0.5
    outer_cycles = 6
    inner_iters = 150
    
    τ = τ_start
    for outer in range(outer_cycles):
        h_prev = h.copy()
        if time.time() - start_time > budget_s - 30:
            break
        # Inner acceleration loop: Nesterov
        for t in range(inner_iters):
            if time.time() - start_time > budget_s - 30:
                break
            # Nesterov extrapolation
            beta = (t - 1) / (t + 2) if t > 0 else 0
            y = h + beta * (h - h_prev)
            
            # Compute correlation values (all shifts in one pass using FFT)
            f = correlation_fft(y)
            
            # Compute exponential weights (softmax)
            p = np.exp(f / τ - np.max(f / τ))
            p /= np.sum(p)
            
            # Compute gradient using weighted convolution
            grad = weighted_gradient_fft(y, p)
            
            # Line search for step size
            α = 1.0
            while True:
                z = y - α * grad
                z_proj = project_onto_Ω(z)
                F_tau = τ * np.log(np.sum(np.exp(f / τ)))  # Smooth surrogate value
                if F_tau <= τ * np.log(np.sum(np.exp(f / τ))) - 0.5 * α * np.dot(grad, grad):
                    break
                α *= 0.5
            
            h_prev, h = h, z_proj
        
        τ *= τ_factor
    
    # Intermediate refinement using binary rounding and fractional corrections
    h_bin = binary_from_top(h, N // 2)
    h_refined = fractional_correction(h_bin)
    
    # Local swap refinement (hill climbing)
    current_h = h_refined.copy()
    current_c5 = c5_of_h(current_h)
    
    # Loop for swaps
    max_swaps = 5000
    for swap_it in range(max_swaps):
        if time.time() - start_time > budget_s - 80:
            break
        # Pick two random indices
        i, j = np.random.choice(N, 2, replace=False)
        if abs(current_h[i] - current_h[j]) < 1e-8:
            continue  # Skip if same value
        
        # Try swapping
        h_test = swap(current_h, i, j)
        test_c5 = c5_of_h(h_test)
        
        if test_c5 < current_c5 - 1e-7:
            current_h, current_c5 = h_test, test_c5
    
    best_h = current_h
    best_c5 = current_c5
    
    # Final upsample and refinement if time permits
    if time.time() - start_time < budget_s - 5:
        N_up = 512
        h_up = np.interp(np.linspace(0, 2, N_up), np.linspace(0, 2, N), best_h)
        h_up = project_onto_Ω(h_up)
        # SLSQP refinement on upsampling
        res = opt.minimize(
            fun=c5_of_h,
            x0=h_up,
            method='SLSQP',
            bounds=[(0, 1)] * N_up,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - N_up / 2},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if res.success:
            final_c5 = c5_of_h(res.x)
            if final_c5 < best_c5:
                best_h = res.x
                best_c5 = final_c5
    
    return (best_h, best_c5, len(best_h))
```
