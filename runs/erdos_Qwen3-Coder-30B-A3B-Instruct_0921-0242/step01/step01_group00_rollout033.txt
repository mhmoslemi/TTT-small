import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def compute_block_autocorrelation(ells, heights, n_eval=200):
    """Compute the full correlation of a block function for discrete shifts."""
    # Compute block endpoints
    cumsum = np.concatenate([[0], np.cumsum(ells)])
    lefts = cumsum[:-1]
    rights = cumsum[1:]
    
    # Grid of shifts
    delta = 2.0 / n_eval
    shifts = np.arange(n_eval) * delta
    
    # Precompute overlaps for all pairs
    K = len(ells)
    overlaps = np.zeros(n_eval)
    
    for i in range(K):
        l_a, r_a = lefts[i], rights[i]
        h_a = heights[i]
        for j in range(K):
            l_b, r_b = lefts[j], rights[j]
            h_b = heights[j]
            
            # overlap length for each shift
            for k, s in enumerate(shifts):
                overlap = max(0.0, min(r_a, r_b - s) - max(l_a, l_b - s))
                overlaps[k] += h_a * h_b * overlap
    
    # Compute C5 values
    c5_values = 1 - overlaps / (2.0 / n_eval)
    return np.max(c5_values)

def decode_block_params(u_raw, v_raw):
    """Convert raw parameters to block configuration."""
    K = len(u_raw)
    # Lengths via softmax
    ell = np.exp(u_raw)
    ell = ell / np.sum(ell) * 2.0
    # Heights via sigmoid
    h_tilde = 1.0 / (1.0 + np.exp(-v_raw))
    # Rescale to satisfy integral
    area_sum = np.sum(h_tilde * ell)
    alpha = 1.0 / area_sum
    h = np.minimum(1.0, alpha * h_tilde)
    # Handle capping
    if np.any(h == 1.0):
        excess = np.sum(h) - 1.0
        if excess > 1e-10:
            # Redistribute
            mask = h < 1.0
            if np.sum(mask) > 0:
                scale = excess / np.sum(h[mask])
                h[mask] = np.maximum(0.0, h[mask] - scale)
                # Rescale remaining
                total_unconstrained = np.sum(h[mask])
                if total_unconstrained > 0:
                    h[mask] *= (1.0 - excess) / total_unconstrained
                h = np.minimum(1.0, h)
    # Ensure to prevent any violation
    h = np.clip(h, 0.0, 1.0)
    return ell, h

def block_objective(x):
    """Objective function for block parameters."""
    u = x[:K]
    v = x[K:]
    try:
        ell, h = decode_block_params(u, v)
        return compute_block_autocorrelation(ell, h)
    except Exception:
        return 1e6

def project_h_to_feasible(h, n_points):
    """Project h to feasible region with sum = n_points / 2, and 0<=h<=1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n_points / 2 - current_sum
    if abs(diff) < 1e-10:
        return h
    h = h + diff / n_points
    h = np.clip(h, 0, 1)
    return h
