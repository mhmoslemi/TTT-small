def binarize_h(h, target_sum):
    """Convert continuous h to binary while maintaining sum constraint."""
    h_sorted = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    for i in range(int(target_sum)):
        h_bin[h_sorted[i]] = 1
    return h_bin.astype(np.uint8)

def compute_c5(h, dx):
    """Compute C5 using FFT convolution."""
    g = 1 - h
    corr = fftconvolve(h, g, mode='full')
    return np.max(corr) * dx

def swap_move(h, i1, i0):
    """Perform a swap move: exchange h[i1] and h[i0]."""
    new_h = h.copy()
    new_h[i1], new_h[i0] = new_h[i0], new_h[i1]
    return new_h

def evaluate_move(old_h, new_h, corr_old, i1, i0, n):

def local_improve(h, dx, max_iter=5000):
    """Apply local deterministic improvement."""
    old_c5 = compute_c5(h, dx)
    improved = True
    iter_count = 0
    
    while improved and iter_count < max_iter:
        improved = False
        better_move = None
        best_improvement = 0
        n = len(h)
        
        # Check all possible swaps
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i1 in ones[:200]:  # limit for speed
            for i0 in zeros[:200]:  
                new_h = swap_move(h, i1, i0)
                new_c5 = compute_c5(new_h, dx)
                improvement = old_c5 - new_c5
                
                if improvement > best_improvement:
                    best_improvement = improvement
                    better_move = (i1, i0)
        
        if better_move:
            i1, i0 = better_move
            h = swap_move(h, i1, i0)
            old_c5 -= best_improvement
            improved = True
            
        iter_count += 1

    return h, old_c5
