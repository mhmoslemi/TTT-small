import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import cvxpy as cp
from scipy.fft import rfft, irfft

def solve_spectral_lp(n):
    """Solve the spectral linear program to maximize the minimum autocorrelation."""
    # Variables for power spectrum
    p = cp.Variable(n)
    t = cp.Variable()

    # Constraints
    constraints = [p >= 0, p <= n]
    constraints.append(cp.sum(p) == n * n / 2.0)

    # Autocorrelation constraints
    for k in range(1, n):
        # cos(2*pi*k*m/n) for m = 0..n-1
        cos_term = np.cos(2 * np.pi * k * np.arange(n) / n)
        constraints.append(cp.sum(p * cos_term) >= t * n)

    # Objective
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)

    if prob.status == cp.OPTIMAL:
        return p.value, t.value
    else:
        # Fallback: return zeros and a conservative t
        return np.zeros(n), 0.0

def gerchberg_saxton(p_spectrum, n, max_iter=200):
    """Perform Gerchberg-Saxton phase retrieval."""
    # Random initial phases
    phi = np.random.uniform(0, 2 * np.pi, n)
    # Initialize complex FT
    X = np.sqrt(p_spectrum) * np.exp(1j * phi)
    
    for _ in range(max_iter):
        # Inverse DFT
        x = irfft(X, n=n)
        
        # Amplitude projection: clip to [0,1]
        y = np.clip(x, 0, 1)
        
        # Water-filling to maintain sum
        current_sum = np.sum(y)
        delta = n / 2.0 - current_sum
        
        if abs(delta) > 1e-10:
            # Simple bisection to find lambda
            low, high = -1.0, 1.0
            for _ in range(20):
                mid = (low + high) / 2
                test_y = np.clip(y + mid, 0, 1)
                test_sum = np.sum(test_y)
                if test_sum < n / 2:
                    high = mid
                else:
                    low = mid
            lambda_val = (low + high) / 2
            y = np.clip(y + lambda_val, 0, 1)
        
        # Forward DFT
        Y = rfft(y)
        
        # Spectral projection
        scale = np.abs(Y) + 1e-12  # avoid division by zero
        X = np.sqrt(p_spectrum) * (Y / scale)
    
    return y

def project_simplex_box(h, n):
    """Project h onto the sum = n/2 and box [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    delta = n / 2.0 - current_sum
    
    if abs(delta) < 1e-10:
        return h
    
    # Use bisection to find correct lambda
    low, high = -1.0, 1.0
    for _ in range(20):
        mid = (low + high) / 2
        test_h = np.clip(h + mid, 0, 1)
        test_sum = np.sum(test_h)
        if test_sum < n / 2:
            high = mid
        else:
            low = mid
    
    lambda_val = (low + high) / 2
    h = np.clip(h + lambda_val, 0, 1)
    return h

def binary_sharpen(h, n, mu_vals=[1e-4, 1e-3, 1e-2]):
    """Sharpen towards binary values."""
    for mu in mu_vals:
        # Define penalty
        def penalty(h):
            return mu * np.sum(h * (1 - h))
        
        # Simple gradient descent for a few steps
        for _ in range(10):
            # Compute penalty gradient
            grad = mu * (1 - 2 * h)
            h -= 0.01 * grad
            h = project_simplex_box(h, n)
    
    # Final rounding
    h = np.where(h < 0.01, 0.0, h)
    h = np.where(h > 0.99, 1.0, h)
    h = project_simplex_box(h, n)
    
    return h

def compute_worst_shift_gradient(h, n, dx):
    """Compute gradient of the worst shift correlation."""
    # Compute correlation
    corr = correlate(h, 1 - h, mode='full')
    # Find worst shift
    max_idx = np.argmax(corr)
    k_best = max_idx - (n - 1)
    
    # Gradient at worst shift
    grad = np.zeros(n)
    for i in range(n):
        # (1 - h[i + k_best])
        j = (i + k_best) % n
        grad[i] -= dx * (1 - h[j])
        
        # h[i - k_best]
        j = (i - k_best) % n
        grad[i] += dx * h[j]
        
    return grad, k_best

def swap_anneal(h, n, dx, max_time):
    """Stochastic swap refinement (simulated annealing like)."""
    current_c5 = evaluate_c5(h)
    T = 1e-3
    best_h = h.copy()
    best_c5 = current_c5
    
    start_ann = time.time()
    while time.time() - start_ann < max_time:
        # Random swap
        i, j = np.random.randint(0, n, 2)
        if h[i] < 1e-6 or h[j] > 1 - 1e-6:
            continue
            
        delta = np.random.uniform(0, min(h[i], 1 - h[j], 0.02))
        
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = project_simplex_box(h_new, n)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                h = h_new
                current_c5 = new_c5
        
        T *= 0.999
        
    return best_h, best_c5

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr
