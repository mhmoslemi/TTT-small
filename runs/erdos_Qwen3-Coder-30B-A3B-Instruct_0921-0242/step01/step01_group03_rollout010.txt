```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import scipy.optimize as opt

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    N = 140  # grid points (even number)
    dx = 2.0 / N
    S = N // 2  # target sum
    
    # Precompute indices for each k
    index_sets = {}
    for k in range(N):
        index_sets[k] = list(range(N - k))
    
    # Initialize best solution
    best_c5 = np.inf
    best_h = None
    
    # Binary search parameters
    L = 0.0
    U = S
    eps = 1e-4
    best_feasible_t = None
    best_feasible_x = None
    
    # Precompute how many auxiliary variables we'll need
    total_aux_vars = sum(N - k for k in range(N))  # N(N+1)/2
    
    print(f"Starting MILP binary search with N={N}, total auxiliary vars ~{total_aux_vars}")
    
    # Binary search loop via feasibility calls on MILP
    iter_count = 0
    while U - L > eps and time.time() - start_time < budget_s - 10:
        T = (L + U) / 2.0
        print(f"  Testing t = {T:.6f} (L={L:.6f}, U={U:.6f})...")
        
        # Construct MILP model
        x = cp.Variable(N, boolean=True)
        t = cp.Variable()
        
        # Auxiliary variables
        w = {}
        constraints = []

        # Mass constraint
        constraints.append(cp.sum(x) == S)
        # Symmetry breaking: x[0] = 1
        constraints.append(x[0] == 1)
        # Bounds for t
        constraints.append(t >= 0)
        
        w_count = 0
        for k in range(N):
            for i in index_sets[k]:
                w[(i, k)] = cp.Variable(boolean=True)
                w_count += 1
                # McCormick inequalities
                constraints.append(w[(i, k)] <= x[i])
                constraints.append(w[(i, k)] <= x[i + k])
                constraints.append(w[(i, k)] >= x[i] + x[i + k] - 1)

        # Overlap bound constraints
        for k in range(N):
            constraints.append(cp.sum([w[(i, k)] for i in index_sets[k]]) >= S - t)
        
        # Try to solve feasibility problem
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            result = prob.solve(solver=cp.CBC, 
                                verbose=False, 
                                max_seconds=30, 
                                warm_start=True)
            
            if result != np.inf:
                if best_feasible_t is None or T < best_feasible_t:
                    best_feasible_t = T
                    # Get solution
                    best_feasible_x = x.value
                    print(f"    Found feasible with t={T}, updated best")
            else:
                print(f"    Infeasible at t={T}")
        except Exception as e:
            print(f"    Exception solving feasibility t={T}: {e}")
        iter_count += 1
        if time.time() - start_time > budget_s - 10:
            break
        
        # Update binary search
        if best_feasible_t is not None and T < best_feasible_t:
            U = T
        else:
            L = T
    
    print(f"Binary search completed after {iter_count} iterations.")
    
    # At this point we should have the best feasible solution
    if best_feasible_x is not None:
        x_sol = best_feasible_x
    else:
        # Fall back to simple construction if no feasible found
        x_sol = np.zeros(N, dtype=int)
        x_sol[:S] = 1
        print("Falling back to greedy solution.")
    
    h_sol = np.array(x_sol, dtype=float)
    c5_bound = c5_of_h(h_sol)
    
    # Fractional polishing: small adjustments from pairs.
    if time.time() - start_time < budget_s - 10:
        # Get the current maximal shift
        g = 1.0 - h_sol
        corr = np.correlate(h_sol, g, mode='full')
        max_k = np.argmax(corr)
        k = max_k - (N - 1)
        print(f"Polishing: current max shift k={k}")
        
        # Refine using small swaps
        best_h = h_sol.copy()
        best_c5 = c5_bound
        
        num_iterations = 0
        delta_schedule = [0.05, 0.02, 0.01]
        for delta in delta_schedule:
            if time.time() - start_time > budget_s - 5:
                break
            num_iterations = 0
            changed = True
            while changed and num_iterations < 30 and time.time() - start_time < budget_s - 5:
                changed = False
                num_iterations += 1
                # Look for large contributions to overlap
                w_indices = index_sets[abs(k)] if k >= 0 else index_sets[abs(k)]
                # Find best pair swapping
                best_improvement = 0
                best_i, best_j = -1, -1
                for i in w_indices:
                    if h_sol[i] == 1.0:
                        for j in range(N):
                            if h_sol[j] == 0.0:
                                # Test moving delta from i to j
                                h_test = h_sol.copy()
                                h_test[i] -= delta
                                h_test[j] += delta
                                h_test = np.clip(h_test, 0, 1)
                                test_c5 = c5_of_h(h_test)
                                improvement = best_c5 - test_c5
                                if improvement > best_improvement and improvement > 1e-8:
                                    best_improvement = improvement
                                    best_i, best_j = i, j
                if best_i >= 0:
                    h_sol[best_i] -= delta
                    h_sol[best_j] += delta
                    h_sol = np.clip(h_sol, 0, 1)
                    best_c5 = c5_of_h(h_sol)
                    print(f"    Polished with move ({best_i}, {best_j}) delta={delta}, new C5={best_c5:.6f}")
                    changed = True
                    
    if best_c5 < 0.3807:
        print(f"Polishing achieved C5={best_c5:.6f} < 0.3807")
        
    # Final hill climbing on binary swap
    if time.time() - start_time < budget_s - 5:
        print("Performing swap hill climbing...")
        # Since we already have a good solution, try few swaps
        current_h = h_sol.copy()
        current_c5 = c5_of_h(current_h)
        
        # Use FFT to compute overlaps quickly
        def fast_overlap(h):
            dx = 2.0 / len(h)
            g = 1.0 - h
            # Use FFT for efficiency
            corr = np.correlate(h, g, mode='full') * dx
            return np.max(corr)
        
        # Swapping algorithm
        max_attempts = 30000
        i_iter = 0
        while i_iter < max_attempts and time.time() - start_time < budget_s - 5:
            # Pick two indices: one 1, one 0
            ones = np.where(current_h > 0.5)[0]
            zeros = np.where(current_h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Swap values
            h_test = current_h.copy()
            h_test[i], h_test[j] = h_test[j], h_test[i]
            
            test_c5 = fast_overlap(h_test)
            if test_c5 < current_c5:
                current_h = h_test
                current_c5 = test_c5
                if current_c5 < 0.3807:
                    print(f"Hill climbing improved to C5={current_c5:.6f}")
            i_iter += 1
        h_sol = current_h
        best_c5 = current_c5
            
    # Verify the bounds are correct
    if best_c5 > c5_of_h(h_sol):
        print("Warning final bound higher than computed")
        best_c5 = c5_of_h(h_sol)
    
    # Final renormalization for integral constraint
    integral = np.sum(h_sol) * dx
    h_sol = h_sol * (1.0 / integral)  # should be already correct

    return (h_sol, best_c5, N)
```
