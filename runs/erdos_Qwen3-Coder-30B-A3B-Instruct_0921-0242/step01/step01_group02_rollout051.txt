import numpy as np
import time
import cvxpy as cp
from scipy import sparse as sp
from scipy.signal import correlate

def build_Q(k, n):
    """Build sparse symmetric PSD matrix for term h_i h_{i+k}."""
    data = np.ones(n - abs(k))
    rows = np.arange(n - abs(k))
    cols = rows + k if k >= 0 else rows - k
    Q = sp.csr_matrix((data, (rows, cols)), shape=(n, n))
    return 0.5 * (Q + Q.T)

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def solve_qp(n, Kmax, warm_h=None):
    """Solve the convex QP formulation of the problem."""
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2]
    
    # Quadratic constraints for each shift in [-Kmax, Kmax]
    for k in range(-Kmax, Kmax + 1):
        Q = build_Q(k, n)
        constraints.append(cp.quad_form(h, Q) >= n / 2 - t)
    
    # Formulate and solve problem
    prob = cp.Problem(cp.Minimize(t), constraints)
    if warm_h is not None:
        h.value = warm_h
    
    prob.solve(solver=cp.OSQP, verbose=False, max_iter=5000, eps_abs=1e-9)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h.value, t.value
    else:
        return None, float('inf')

        else:
            # Revert swap
            h_bin[i1] = 1
            h_bin[i0] = 0
    
    return best_h, best
