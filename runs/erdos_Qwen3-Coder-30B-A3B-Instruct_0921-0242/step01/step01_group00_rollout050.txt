import numpy as np
import time
from scipy.optimize import linprog
from cvxpy import Variable, Minimize, Problem, sum, abs, pow, sqrt, multiply
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.signal import correlate

def solve_spectral_lp(n):
    # Spectral relaxation LP to maximize t such that all A_k >= t
    # Variables: p_m = |hat h_m|^2 for m = 0..n-1, and scalar t
    # LP formulation
    p = Variable(n)
    t = Variable(1)
    
    # Objective: maximize t
    objective = Minimize(-t)  # because linprog minimizes
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(sum(p) == n * n / 2)
    
    # Box constraints
    for i in range(n):
        constraints.append(p[i] >= 0)
        constraints.append(p[i] <= n)
    
    # Autocorrelation constraints
    # A_k = (1/n) * sum_m p_m * cos(2*pi*k*m/n)
    # We need A_k >= t for k=1..n-1
    
    # Compute cosine matrix
    cos_matrix = np.zeros((n-1, n))
    for k in range(1, n):
        for m in range(n):
            cos_matrix[k-1, m] = np.cos(2 * np.pi * k * m / n)
    
    # Add constraints: (1/n) * cos_matrix * p >= t
    for i in range(n-1):
        constraints.append(sum(cos_matrix[i,:] * p) >= n * t)
        
    # Solve the problem
    prob = Problem(objective, constraints)
    prob.solve(solver='ECOS', verbose=False)
    
    if prob.status in ['optimal', 'optimal_inaccurate']:
        return p.value, t.value
    else:
        # fallback to a simple approximation
        # Return p = n/2 for all m, and calculate corresponding t
        p_simple = np.full(n, n/2)
        cos_matrix = np.zeros((n-1, n))
        for k in range(1, n):
            for m in range(n):
                cos_matrix[k-1, m] = np.cos(2 * np.pi * k * m / n)
        
        # Evaluate at p=n/2
        A_vals = (1/n) * np.dot(cos_matrix, p_simple)
        t_val = np.min(A_vals)
        return p_simple, t_val

def gerchberg_saxton(p, n, max_iter=200):
    # Gerchberg-Saxton phase retrieval algorithm
    # Initialize with random phases
    phases = np.random.uniform(0, 2*np.pi, n)
    # complex spectrum with magnitude sqrt(p)
    X = np.sqrt(p) * np.exp(1j * phases)
    
    for _ in range(max_iter):
        # Inverse FFT
        x = ifft(X).real
        
        # Project to [-1, 1] and then to [0, 1] + waterfilling
        y = np.clip(x, 0, 1)
        
        # waterfilling to ensure sum = n/2
        current_sum = np.sum(y)
        if abs(current_sum - n/2) > 1e-9:
            # compute lambda such that sum(y + lambda) = n/2
            # y + lambda is clipped again to [0,1]
            # The problem is: sum(clamp(y + lambda, 0, 1)) = n/2
            # This requires bisection on lambda
            
            def waterfill_obj(lam):
                y_new = y + lam
                y_new = np.clip(y_new, 0, 1)
                return np.sum(y_new) - n/2
            
            # binary search
            low, high = -10, 10
            for _ in range(50):
                mid = (low + high) / 2 
                val = waterfill_obj(mid)
                if val > 0:
                    high = mid
                else:
                    low = mid
            
            lam = (low + high) / 2
            y = np.clip(y + lam, 0, 1)
        
        # Forward FFT
        Y = fft(y)
        
        # Update spectrum with target magnitude
        magnitudes = np.sqrt(p)
        phase = np.angle(Y)
        X = magnitudes * np.exp(1j * phase)
        
    return np.clip(y, 0, 1)

def project_simplex_box(x, target_sum):
    # Waterfilling projection onto box [0,1] with sum constraint
    x = np.array(x)
    x = np.clip(x, 0, 1)
    
    current_sum = np.sum(x)
    delta = target_sum - current_sum
    
    if abs(delta) < 1e-9:
        return x
        
    # sorting approach for water-fill
    n = len(x)
    # We are looking for a constant value to add to all entries to make sum = target_sum
    # But we have clipping to [0,1]
    # So, we should add lam to all entries, then clip to [0,1]
    # And find how much lam we should add to make the total sum = target_sum
    
    # Translate x to upper bounds (1 - x), and lower bounds to x
    if delta > 0:
        # Need to add
        # Do this using bisection on lambda
        lower_lam, upper_lam = 0, 1.0
        for _ in range(100):
            lam = (lower_lam + upper_lam) / 2
            x_proposed = x + lam
            x_proposed = np.clip(x_proposed, 0, 1)
            new_sum = np.sum(x_proposed)
            if new_sum > target_sum:
                upper_lam = lam
            else:
                lower_lam = lam
        lam_final = (lower_lam + upper_lam) / 2
        x = np.clip(x + lam_final, 0, 1)
    else:
        # Need to subtract
        lower_lam, upper_lam = -1.0, 0
        for _ in range(100):
            lam = (lower_lam + upper_lam) / 2
            x_proposed = x + lam
            x_proposed = np.clip(x_proposed, 0, 1)
            new_sum = np.sum(x_proposed)
            if new_sum < target_sum:
                lower_lam = lam
            else:
                upper_lam = lam
        lam_final = (lower_lam + upper_lam) / 2
        x = np.clip(x + lam_final, 0, 1)
        
    return x

def binary_sharpen(x, cnt_iter=2):
    # Simple gradient descent on penalty function
    # penalty \psi(h) = \mu \sum h_i(1-h_i)
    # We optimize S_\beta(h) - \mu \sum h_i(1-h_i)
    
    for _ in range(cnt_iter):
        # Loop over different mu
        for mu in [1e-4, 1e-3, 1e-2]:
            x = project_simplex_box(x, 0.5*n)
            # for now just hard threshold
            x = np.clip(x, 0.01, 0.99)
            x[x < 0.01] = 0
            x[x > 0.99] = 1
            x = project_simplex_box(x, 0.5*n)

    return x

def gradient_worst_shift(h, dx, max_iter=3000):
    # Optimizes the current worst shift
    n = len(h)
    # Compute full correlation
    corr = correlate(h, 1 - h, mode='full')
    # Find the index of the maximum overlap for the non-zero shifts
    max_idx = np.argmax(corr[1:]) + 1 # because we exclude the 0-th shift
    if max_idx == 0:
        # No improvement possible
        return h
    # And the actual shift
    shift = max_idx - (n-1)  # 0-indexed shift in correct range
  
    for i in range(max_iter):
        # Gradient of C5 at shift = shift
        # \partial I_{k}/\partial h_i = dx * [(1-h_{i+k}) - h_{i-k}]
        # We will use a finite diff
        grad = np.zeros(n)
        for j in range(n):
            # Compute the j-th entry of the gradient
            # Slope of sum (1-h_{j+shift}) * h_{j-shift}
            # for all corruptions
            grad[j] = dx * (
                (1 - h[(j + shift) % n]) - h[(j - shift) % n]
            )
        # Project the gradient
        h_trial = h - grad
        h_trial = project_simplex_box(h_trial, 0.5*n)
        current_corr = correlate(h, 1-h, mode='full')
        trial_corr = correlate(h_trial, 1-h_trial, mode='full')
        best_shift_before = np.argmax(current_corr[1:]) + 1 - (n-1)
        best_shift_after = np.argmax(trial_corr[1:]) + 1 - (n-1)
        # If we improved worst shift or maintain same shift but reduce objective
        if (abs(best_shift_after) == abs(shift) and trial_corr[best_shift_after+(n-1)] < corr[shift+(n-1)]):
            # Accept step
            h = h_trial
        # If worst shift got worse, stop early
        elif abs(best_shift_after) == abs(shift) and trial_corr[best_shift_after+(n-1)] > corr[shift+(n-1)]:
            break
    return h

def swap_anneal(h, dx, max_iter=1000):
    # Anneal-style swap exploration
    temp = 1e-3
    beta = 0.995
    n = len(h)
    original_c5 = np.max(correlate(h, 1-h, mode='full')) * dx
    for _ in range(max_iter):
        if temp < 1e-10:
            break
        # Pick two indices
        i, j = np.random.choice(n, 2, replace=False)
        # Check conditions
        if h[i] <= 0 or h[j] >= 1:
            continue
        # Choose delta
        delta = np.random.uniform(0, min(h[i], 1 - h[j], 0.02))
        # Try swap
        h[i] -= delta
        h[j] += delta
        # Project
        h = project_simplex_box(h, 0.5*n)
        # Check C5
        corr = correlate(h, 1-h, mode='full')
        c5_proposed = np.max(corr) * dx
        if c5_proposed < original_c5:
            original_c5 = c5_proposed
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c5_proposed - original_c5) / temp):
                original_c5 = c5_proposed
            else:
                # Undo swap
                h[i] += delta
                h[j] -= delta
                # Re-project
                h = project_simplex_box(h, 0.5*n)
        # Cooling schedule
        temp *= beta
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Total time per stage
    # 1. Spectral LP: ~4s
    # 2. GS: ~8s
    # 3. Projection + binary: ~3s
    # 4. Worst shift: ~180s
    # 5. Swap Anneal: ~50s
    
    n = 650  # Near to 300 from previous, but faster convergence
    dx = 2.0 / n
    
    # 1. Spectral convex relaxation
    print("Stage 1: Solving spectral LP...")
    spectrum, t_star = solve_spectral_lp(n)
    print("LP result: t_star =", t_star)
    if t_star <= 0:
        t_star = 0.001  # very conservative
    
    # 2. Phase retrieval
    print("Stage 2: Performing Gerchberg-Saxton...")
    h = gerchberg_saxton(spectrum, n)
    
    # 3. Feasibility projection and sharpening
    print("Stage 3: Projecting and sharpening...")
    h = project_simplex_box(h, n/2)
    h = binary_sharpen(h, 2)
    
    print("Stage 4: Targeted gradient refinement...")
    h = gradient_worst_shift(h, dx, 3000)
    
    print("Stage 5: Swap annealing...")
    h = swap_anneal(h, dx, 1000)
    
    # Final evaluation
    corr = correlate(h, 1-h, mode='full')
    c5 = np.max(corr) * dx
    
    return (h, c5, n)
