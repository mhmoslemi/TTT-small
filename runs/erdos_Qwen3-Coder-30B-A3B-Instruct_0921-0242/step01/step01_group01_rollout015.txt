import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def water_fill(h, target_sum):
    """Project h onto the sum constraint ∑h = target_sum"""
    h = np.copy(h)
    delta = np.sum(h) - target_sum
    if abs(delta) < 1e-12:
        return h
        
    while abs(delta) > 1e-12:
        # Find indices with 0 < h_i < 1
        mask = (h > 0) & (h < 1)
        indices = np.where(mask)[0]
        if len(indices) == 0:
            break
        avg_change = delta / len(indices)
        # Update and clip
        h[indices] += avg_change
        h = np.clip(h, 0, 1)
        delta = np.sum(h) - target_sum
        
    return h

def project_h(h, target_sum):
    """Apply both box and sum projections"""
    h = np.clip(h, 0, 1)
    h = water_fill(h, target_sum)
    return h

def smooth_max_gradient(h, beta, dx):
    """Compute gradient of smooth max surrogate"""
    n = len(h)
    # Full correlation
    g = 1 - h
    conv = np.correlate(h, g, mode='full')
    
    # Soft max weights
    conv_scaled = beta * dx * conv
    # Log-sum-exp trick for numerical stability
    conv_scaled -= np.max(conv_scaled)
    exp_conv = np.exp(conv_scaled)
    weights = exp_conv / np.sum(exp_conv)
    
    # Compute gradient via two convolutions
    u = np.correlate(weights, g, mode='full')[:n]
    v = np.correlate(weights[::-1], h, mode='full')[:n]
    grad = dx * (u - v)
    
    return grad

def refine_with_smooth_max(h, target_sum, beta_init=0.5, beta_max=50.0, max_iter=80, dx=2.0/120):
    """Smooth max refinement with continuation"""
    h = project_h(h, target_sum)
    beta = beta_init
    c5_prev = float('inf')
    
    for _ in range(20):  # At most 20 β-stages due to time budget
        if beta > beta_max:
            break
        for _ in range(max_iter):
            grad = smooth_max_gradient(h, beta, dx)
            h -= 0.1 * grad
            h = project_h(h, target_sum)
            h = (h + h[::-1]) / 2  # Mirror symmetry
            
        c5 = dx * np.max(np.correlate(h, 1-h, mode='full'))
        if abs(c5 - c5_prev) < 1e-6:
            break
        c5_prev = c5
        beta *= 2
        if beta > beta_max:
            break
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Coarse grid resolution
    n_coarse = 120
    target_sum_coarse = n_coarse / 2
    
    # SDP problem setup
    h = cp.Variable(n_coarse)
    t = cp.Variable()
    
    # Toeplitz matrices for correlation constraints
    R = []
    for k in range(n_coarse):
        R_k = np.zeros((n_coarse, n_coarse))
        for i in range(n_coarse):
            if i + k < n_coarse:
                R_k[i, i + k] = 1
            if i - k >= 0:
                R_k[i, i - k] = 1
        R.append(R_k)
    
    # Constraints for SDP
    constraints = [
        cp.sum(h) == target_sum_coarse,
        cp.min(h) >= 0,
        cp.max(h) <= 1,
        t <= cp.trace(R_k @ cp.reshape(h, (n_coarse, 1)) @ cp.reshape(h, (1, n_coarse))) 
        for R_k in R
    ]
    
    # Dummy constraint to select first one as -x <= 0
    # Re-formulate for simplicity:
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Solve SDP with fallback
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, warm_start=False, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_sol = h.value
        else:
            h_sol = np.random.uniform(0, 1, n_coarse)
            h_sol = h_sol * (n_coarse / 2) / np.sum(h_sol)
    except:
        h_sol = np.random.uniform(0, 1, n_coarse)
        h_sol = h_sol * (n_coarse / 2) / np.sum(h_sol)
    
    # Extracting near rank-1 solution from SDP (if it's a valid solution, we'll use it)
    # For this basic version, let's revert to a simple approach if SDP fails
    h_coarse = project_h(h_sol, target_sum_coarse)
    h_coarse = (h_coarse + h_coarse[::-1]) / 2  # Ensure symmetry
    
    # Refine using smooth max
    h_coarse = refine_with_smooth_max(h_coarse, target_sum_coarse, dx=2.0/n_coarse)
    c5_coarse = (2.0/n_coarse) * np.max(np.correlate(h_coarse, 1-h_coarse, mode='full'))
    
    # Upsample to final resolution
    n_final = 300
    dx_final = 2.0 / n_final
    x_coarse = np.linspace(0, 2, n_coarse)
    x_final = np.linspace(0, 2, n_final)
    
    # Interpolate
    interp_func = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
    h_final = interp_func(x_final)
    h_final = np.clip(h_final, 0, 1)
    h_final = water_fill(h_final, n_final / 2)
    h_final = (h_final + h_final[::-1]) / 2  # Enforce symmetry
    
    # Final refinement on full grid
    h_final = refine_with_smooth_max(h_final, n_final / 2, dx=dx_final)
    
    # Final polish
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        epsilon = 0.004
        best_h = h_final.copy()
        best_c5 = dx_final * np.max(np.correlate(h_final, 1-h_final, mode='full'))
        
        iterations = min(1000, int(remaining_time * 50))
        for _ in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
            num_perturb = min(6, n_final // 20)
            indices = np.random.choice(n_final, num_perturb, replace=False)
            h_trial = h_final.copy()
            delta = np.random.uniform(-epsilon, epsilon, num_perturb)
            h_trial[indices] += delta
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = water_fill(h_trial, n_final / 2)
            h_trial = (h_trial + h_trial[::-1]) / 2
            
            c5_trial = dx_final * np.max(np.correlate(h_trial, 1-h_trial, mode='full'))
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
                
        h_final = best_h
    
    final_c5 = dx_final * np.max(np.correlate(h_final, 1-h_final, mode='full'))
    
    return (h_final, final_c5, n_final)
