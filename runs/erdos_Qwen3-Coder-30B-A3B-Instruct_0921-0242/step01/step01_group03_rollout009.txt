```python
import numpy as np
import scipy.optimize as opt
import time
import scipy.fft as fft
from cvxpy import Variable, Minimize, Problem, affine_expr

def correlation_fft(h):
    """Compute correlation f_k = sum h_i (1 - h_{i+k}) for all k using FFT."""
    N = len(h)
    h_pad = np.pad(h, (0, N), mode='constant')
    g_pad = np.pad(1 - h, (0, N), mode='constant')
    c = fft.fft(h_pad) * fft.fft(g_pad)
    return np.real(fft.ifft(c))[:N]

def weighted_gradient_fft(h, p):
    """Compute gradient of log-sum-exp smoothing of correlations."""
    N = len(h)
    # Expand p to length 2N for zero-padding
    p_pad = np.pad(p, (0, N), mode='constant')
    h_pad = np.pad(h, (0, N), mode='constant')
    g_pad = np.pad(1-h, (0, N), mode='constant')
    # w1 = convolution(p, 1-h)
    w1 = np.real(fft.ifft(fft.fft(p_pad) * fft.fft(g_pad)))[:N]
    # w2 = convolution(p_rev, h)
    p_rev = p[::-1]
    p_rev_pad = np.pad(p_rev, (0, N), mode='constant')
    w2 = np.real(fft.ifft(fft.fft(p_rev_pad) * fft.fft(h_pad)))[:N]
    return w1 - w2

def project_onto_Ω(v):
    """Project onto { h : 0≤h≤1, sum(h)=N/2 } using water-filling."""
    N = len(v)
    z = np.clip(v, 0, 1)
    r = N/2 - np.sum(z)
    if abs(r) < 1e-12:
        return z
    # Apply water-filling
    I = np.where((z > 0) & (z < 1))[0]
    tat = 0
    while len(I) > 0 and abs(r) > 1e-12:
        # Distribute r among interior points
        r_add = r / len(I)
        z[I] += r_add
        # Update r
        r = N/2 - np.sum(z)
        # Remove saturated points
        I = np.where((z > 0) & (z < 1))[0]
        tat += 1
        if tat > 100:  # Safety break
            break
    return np.clip(z, 0, 1)

def binary_from_top(h, n_ones):
    """Round to a binary vector with exactly n_ones."""
    N = len(h)
    indices = np.argsort(h)[::-1]
    mask = np.zeros(N, dtype=bool)
    mask[indices[:n_ones]] = True
    return mask.astype(float)

def fractional_correction(binary_h):
    """Do tiny LP correction of a binary vector to make sum exact."""
    # This implementation simply returns the binary vector,
    # since binary sum is automatic and fractional correction
    # involves small LP, which is slow and not needed for this task.
    return binary_h

def swap(h, i, j):
    """Return a copy of h with h[i] and h[j] swapped."""
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    N = 256  # Base resolution
    h = np.random.choice([0, 1], size=N, p=[0.5, 0.5])
    h = project_onto_Ω(h * (N/2) / np.sum(h))  # Renormalize
    τ = 0.05
    for outer in range(6):
        if time.time() - start_time > budget_s - 100:
            break
        α = 1.0  # step size
        h_prev = h.copy()
        for t in range(150):
            if time.time() - start_time > budget_s - 100:
                break
            y = h + ((t-1)/(t+2)) * (h - h_prev)  # Nesterov extrapolation
            f = correlation_fft(y)
            p = np.exp(f/τ - np.max(f/τ))
            p /= p.sum()
            grad = weighted_gradient_fft(y, p)
            # Backtracking line search
            while True:
                z = y - α * grad
                z_proj = project_onto_Ω(z)
                if c5_of_h(z_proj) <= c5_of_h(y) - 0.5*α*np.dot(grad, grad):
                    break
                α *= 0.5
            h_new = z_proj
            h_prev, h = h, h_new
        τ *= 0.5
    # Rounding
    h_bin = binary_from_top(h, N//2)
    h_refined = fractional_correction(h_bin)
    # Swap hill-climbing
    best_h = h_refined
    best_c5 = c5_of_h(best_h)
    while time.time() - start_time < budget_s - 80:
        i = np.random.randint(N)
        j = np.random.randint(N)
        if i == j:
            continue
        cand = swap(best_h, i, j)
        cand_c5 = c5_of_h(cand)
        if cand_c5 < best_c5 - 1e-7:
            best_h, best_c5 = cand, cand_c5
    # Optional upsampling
    if time.time() - start_time < budget_s - 5:
        h_up = np.repeat(best_h, 2)
        h_up = project_onto_Ω(h_up)
        # Refine with SLSQP
        res = opt.minimize(
            fun=c5_of_h,
            x0=h_up,
            method='SLSQP',
            bounds=[(0, 1)] * (2*N),
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - N},
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        final_h = res.x
        final_c5 = c5_of_h(final_h)
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5
        n_points = len(final_h)
    else:
        n_points = N
    return (best_h, best_c5, n_points)
```
