import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def initial_binary_mask():
    # Start with roughly equal spacing of 1s
    n = 140  # Optimal for this algorithm
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    return h

def solve_milp_binary_search(n_points=140, budget_s=600):
    start_time = time.time()
    
    # Build index sets efficiently
    index_sets = {}
    for k in range(n_points):
        index_sets[k] = list(range(n_points - k))
    
    # Define variables
    x = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(x) == n_points // 2]
    constraints += [x[0] == 1]  # symmetry breaking

    # Add auxiliary variables and constraints for each shift k
    w_vars = {}
    for k in range(n_points):
        w_vars[k] = []
        for i in index_sets[k]:
            w = cp.Variable(boolean=True)
            w_vars[k].append(w)
            # McCormick constraints:
            constraints += [w <= x[i],
                            w <= x[i+k],
                            w >= x[i] + x[i+k] - 1]
    
    # Add bounds on sum of w terms
    for k in range(n_points):
        constraints += [cp.sum(w_vars[k]) >= (n_points//2) - t]
    
    # Add trivial bound
    constraints += [t >= 0]
    
    # Pre-allocate feasible solution
    best_x = None
    best_t = float('inf')
    
    # Binary search on t
    low_t = 0.0
    high_t = n_points // 2  # worst case
    epsilon = 1e-4
    
    while time.time() - start_time < budget_s - 2 and high_t - low_t > epsilon:
        mid_t = (low_t + high_t) / 2
        
        # Create feasibility problem for current t
        prob = cp.Problem(cp.Minimize(0), constraints + [t <= mid_t])
        
        try:
            result = prob.solve(solver=cp.CBC, verbose=False,
                               solver_opts={'max_seconds': 30})
            if result != np.inf:
                # Check if we got a solution with sum=x, respect that
                x_val = [int(round(val)) for val in x.value]
                if abs(np.sum(x_val) - n_points // 2) < 1e-3:
                    low_t = mid_t
                    if mid_t < best_t:
                        best_t = mid_t
                        best_x = x_val
                else:
                    high_t = mid_t
            else:
                high_t = mid_t
        except Exception:
            high_t = mid_t
    
    if best_x is None:
        # Fallback: use simple greedy solution
        try:
            x_val = initial_binary_mask()
            best_x = x_val.tolist()
            best_t = n_points // 2 + 1  # Should be overridden below
        except:
            # Just pick a simple binary pattern
            x_val = np.zeros(n_points, dtype=int)
            x_val[:n_points//2] = 1
            best_x = x_val.tolist()
            best_t = n_points // 2 + 1  # stable
            
    return best_x, best_t

def polish_solution(h, n_points, max_iters=100):
    # Apply coarse polishing based on modifying individual pairs of values
    original_h = h.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(best_h)
    
    # Identify which shift is currently the bottleneck
    dx = 2.0 / n_points
    g = 1.0 - best_h
    corr = np.correlate(best_h, g, mode='full') * dx
    max_idx = np.argmax(corr)
    k_best = max_idx - (n_points - 1)
    
    for it in range(max_iters):
        if time.time() > time.time() + 600 - 2:
            break
        improvement_found = False
        
        # For each support of worst shift
        start_idx = max(0, k_best) if k_best >= 0 else 0
        end_idx = n_points if k_best <= 0 else n_points + k_best
        max_shift_valid = min(end_idx, n_points - abs(k_best))
        
        # Look at pairs in the support of worst term
        for i in range(start_idx, max_shift_valid):
            if best_h[i] == 1 and i + k_best < n_points:
                j = i + k_best
                if best_h[j] == 0:
                    # Change a small amount: push value from i to j  
                    h_t = best_h.copy()
                    delta = 0.02
                    h_t[i] = max(0, h_t[i] - delta)
                    h_t[j] = min(1, h_t[j] + delta)
                    new_c5 = c5_of_h(h_t)
                    if new_c5 < best_c5:
                        best_h = h_t
                        best_c5 = new_c5
                        improvement_found = True
                        break
        if not improvement_found:
            break
            
    return best_h

def perform_swap_search(h, n_points, max_attempts=30000):
    """Perform hill Climbing swap search"""
    best_h = h.copy()
    best_c5 = c5_of_h(h)
    
    idx_1s = np.where(best_h > 0.5)[0]
    idx_0s = np.where(best_h < 0.5)[0]
    
    for attempt in range(max_attempts):
        if time.time() > time.time() + 600 - 2:
            break
        if len(idx_1s) == 0 or len(idx_0s) == 0:
            break
            
        i1 = np.random.choice(idx_1s)
        i0 = np.random.choice(idx_0s)
        
        # Make a swap
        h_test = best_h.copy()
        h_test[i1] = 0
        h_test[i0] = 1
        if abs(np.sum(h_test) - n_points / 2) > 1e-3:  # Keep feasible sum
            continue
            
        test_c5 = c5_of_h(h_test)
        if test_c5 < best_c5:
            best_h = h_test
            best_c5 = test_c5
            idx_1s = np.where(best_h > 0.5)[0]
            idx_0s = np.where(best_h < 0.5)[0]
    
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use N = 140 (smaller to make MILP tractable)
    n_points = 140
    dx = 2.0 / n_points
    
    # STEP 1: Solve MILP to get an initial binary mask
    try:
        x, t = solve_milp_binary_search(n_points, budget_s)
        h = np.array(x, dtype=float)
        # Re-scale h to ensure integral = 1 (probably already done)
        h *= (n_points / 2) / np.sum(h)
    except Exception:
        # Fallback: simple binary solution
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
        
    # STEP 2: Polish binary mask with fractional refinement
    h_refined = polish_solution(h, n_points)
    c5_bound = c5_of_h(h_refined)

    # STEP 3: Perform swap hill climbing
    h_final = perform_swap_search(h_refined, n_points)

    c5_final = c5_of_h(h_final)
    return (h_final, c5_final, n_points)
