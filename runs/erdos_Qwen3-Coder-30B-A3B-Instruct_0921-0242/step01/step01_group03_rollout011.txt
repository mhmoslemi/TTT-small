import numpy as np
import cvxpy as cp

def build_shift_matrices(N):
    """Build sparse shift matrices A_k for k = 0, ..., N-1."""
    A_list = []
    for k in range(N):
        rows = np.arange(N - k)
        cols = np.arange(k, N)
        data = np.ones(N - k)
        A = sp.csr_matrix((data, (rows, cols)), shape=(N, N))
        A_list.append(A)
    return A_list

def project_to_feasible(h):
    """Ensure sum(h) = N/2 and h in [0,1]^N."""
    N = len(h)
    dx = 2.0 / N
    sum_h = np.sum(h)
    deficit = N/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = N/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    # Distribute deficit uniformly
    adjustment = deficit / N
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def bounded_simplex_projection(x, target_sum, bounds):
    """Project x onto the simplex with target_sum and box constraints."""
    lower, upper = bounds
    # Simple projection algorithm for bounded simplex
    x = np.array(x)
    x = np.clip(x, lower, upper)
    current_sum = np.sum(x)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return x
    
    # For now, just rescale if diff is large enough to make it Sun)
    # simpler implementation
    if diff > 0:
        # distribute excess (or deficits)
        total_diff = np.sum(np.clip(x + diff / len(x) - upper, 0, 1) - (x + diff / len(x)))
        # For constraints on [0,1], we'll use a denoising-like approach:
        # Not full projection but quick modification
        adjust = diff / len(x)
        x += adjust
        x = np.clip(x, lower, upper)
    else:
        adjust = diff / len(x)
        x += adjust
        x = np.clip(x, lower, upper)
        
    # Normalization
    current_sum = np.sum(x)
    scale = target_sum / current_sum
    x *= scale
    return np.clip(x, lower, upper)

def solve_sdp(N, A_list):
    """Solve the SDP relaxation for this N."""
    # Variables
    h = cp.Variable(N)
    X = cp.Variable((N, N), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.diag(X) == h,  # Diagonal equals h
        cp.sum(h) == N / 2,   # Mass constraint
        cp.bounds(h, 0, 1),   # Bounds on h
        X >> 0,               # X positive semidefinite
        t <= cp.trace(A_k @ X) for A_k in A_list  # Lower bound on trace for each k
    ]
    
    # Objective
    objective = cp.Maximize(t)
    
    # Create and solve problem using SCS
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, max_iters=15000, eps=1e-4)
    
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        return h.value, X.value, t.value
    else:
        # Fallback to a coarser SDP with N//2
        N_fallback = N//2
        A_fallback = [A_list[i] for i in range(N_fallback)]
        h_f, X_f, t_f = solve_sdp(N_fallback, A_fallback)
        # Interpolate back to N
        h_f_interp = np.interp(np.linspace(0, N-1, N_fallback), np.linspace(0, N-1, N_fallback), h_f)
        return h_f_interp, X_f, t_f

def eigenvector_rounding(X_sol, h_sol, N):
    """Round the solution using principal eigenvector."""
    # Compute leading eigenpair
    u, v = spla.eigs(X_sol, k=1, which='LM')
    if np.iscomplex(u):
        u = u.real
    if np.iscomplex(v):
        v = v.real
    v = np.abs(v.flatten())
    h_eig = np.sqrt(u) * v
    h_eig = project_to_feasible(h_eig)
    return h_eig

def diagonal_threshold_rounding(X_sol, N):
    """Round the solution using diagonal entry thresholding."""
    diag = np.diag(X_sol)
    idxs = np.argsort(diag)[::-1]
    h_thr = np.zeros(N)
    h_thr[idxs[:N//2]] = 1.0
    return h_thr

def incremental_corr_update(h, corr, i, j, old_val=0):
    """Update correlation given a swap of index i and j."""
    # corr_full is of length 2*N - 1, centered at N-1
    # Add k rates from -N+1 to N-1
    # Incrementally update correlation for all changes
    
    # For positions i, j that changed from 0 to 1 or 1 to 0:
    # Changes in k-dependent terms for k such that i+k or j+k are within bounds
    # We compute this efficiently
    new_corr = corr.copy()
    n = len(h)
    delta = 1.0 if (h[i] == 0.0) else -1.0
    # Each shift contributes h[i]*h[i+k] and h[j]*h[j+k] 
    for k in range(-n+1, n):
        old_i = h[i] if (i+k >= 0 and i+k < n) else 0
        old_j = h[j] if (j+k >= 0 and j+k < n) else 0
        if h[i] > 0 and old_i > 0:  # i knows about i+k in old terms
            old_i = 1.0  # or use h[i] for numerical accuracy but will mess up grad
        new_i = h[i] if (i+k >= 0 and i+k < n) else 0
        new_j = h[j] if (j+k >= 0 and j+k < n) else 0

        # what we compute depends on k_in the range where it matters
        idx = k + n - 1  # 0-based index for correlation vector
        # linear update based on how h[i]/h[j] change in the cross terms for shift k
        if old_i > 0.5 and i+k >= 0 and i+k < n:
            old_corr = h[i] * old_i 
        else:
            old_corr = 0
        if new_i > 0.5 and i+k >= 0 and i+k < n:
            new_corr_val = h[i] * new_i
        else:
            new_corr_val = 0
        if old_j > 0.5 and j+k >= 0 and j+k < n:
            old_corr += h[j] * old_j
        if new_j > 0.5 and j+k >= 0 and j+k < n:
            new_corr_val += h[j] * new_j
            
        # No need for full recomputation here; just track side-effects
        # The update could be O(N), but it's best approximated with 2*
        # Since our resorting algorithm has full correlation precomputed:
        # For swapping position i with j:
        # Compute all shifts k for which either i or j is involved
        if i+k >= 0 and i+k < n:
            pass  # contributes to i+k
        if j+k >= 0 and j+k < n:
            pass  # contributes to j+k 
        # Apply effect of changing positions
        old_val = 0
        if old_i + old_j != 0:
            old_val = h[i]*old_i + h[j]*old_j
        new_val = 0

            new_val = h[i]*h[i+k] + h[j]*h[j+k] if i+k >=0 and i+k < n and j+k >=0 and j+k < n else 0
        # Assume that correlation values are getting new value due to a minimal change term
        # in practice, be non-confusing:
        return new_corr

def evaluate_corr_c5(h, N, dx):
    """Simple full C5 value from iterated correlation."""
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def perform_swap_search(h0, N, dx, max_time=200):
    """Perform deterministic swap search on a binary mask."""
    h = h0.copy()
    corr_full = np.correlate(h, 1-h, mode='full') * dx
    best_c5 = np.max(corr_full)
    
    # Initialize
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return h, best_c5
    
    # Some initial preprocessing
    corr_full = np.correlate(h, 1-h, mode='full') * dx
    best_c5 = np.max(corr_full)
    
    best_h = h.copy()
    time_start = time.time()
    
    iter_count = 0
    while time.time() - time_start < max_time:
        iter_count += 1
        if iter_count > 100000:
            break
            
        p = np.random.choice(ones)
        q = np.random.choice(zeros)
        
        # Trial swap
        h_trial = h.copy()
        h_trial[p] = 1.0
        h_trial[q] = 0.0
        
        corr_trial = np.correlate(h_trial, 1-h_trial, mode='full') * dx
        c5_trial = np.max(corr_trial)
        
        if c5_trial < best_c5:
            h[:] = h_trial[:]
            corr_full = corr_trial
            best_c5 = c5_trial
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            
            if time.time() - time_start > 0.95 * max_time:
                break
            
        if iter_count % 500 == 0:
            # Recompute exact correlation
            corr_full = np.correlate(h, 1-h, mode='full') * dx
            best_c5 = np.max(corr_full)
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            
    return h, best_c5

def upsample_and_refine(h, N_upsample):
    """Interpolate optimization on a finer grid."""
    # Build fine grid vector
    x_fine = np.linspace(0, 2, N_upsample)
    x_coarse = np.linspace(0, 2, len(h))
    
    h_interp = np.interp(x_fine, x_coarse, h)
    h_interp = h_interp * (N_upsample / 2) / np.sum(h_interp)
    
    return h_interp

def log_sum_exp_surrogate(h, k, temp=1e-4):
    """Log-sum-exp surrogate of max_k h(x)(1 - h(x+k))"""
    # Simple smoother to encourage smoothness
    return np.sum(h * (1.0 - h))

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    start_time = time.time()
    
    # Estimate grid size to fit in time (fewer samples for faster SDP)
    N = 150
    dx = 2.0 / N
    
    # Build shift matrices
    A_list = build_shift_matrices(N)
    
    # Solve SDP
    h_sdp, X_sdp, t_sdp = solve_sdp(N, A_list)
    
    # Rounding
    h_eig = eigenvector_rounding(X_sdp, h_sdp, N)
    h_thr = diagonal_threshold_rounding(X_sdp, N)
    
    # Normalize
    h_eig = project_to_feasible(h_eig)
    h_thr = project_to_feasible(h_thr)
    
    # Evaluate both
    c5_eig = evaluate_corr_c5(h_eig, N, dx)
    c5_thr = evaluate_corr_c5(h_thr, N, dx)
    
    best_h = h_eig if c5_eig < c5_thr else h_thr
    best_c5 = min(c5_eig, c5_thr)
    
    # Local swap improvement
    best_h, best_c5 = perform_swap_search(best_h, N, dx, max_time=200)
    
    # Upsample
    n_up = 2 * N
    h_up = upsample_and_refine(best_h, n_up)
    
    # Final refinement with few gradient steps on log-sum-exp objective
    dx_up = 2.0 / n_up
    # Placeholder for gradient descent, but since code is sparse we use simple `optimize`
    c5_up = evaluate_corr_c5(h_up, n_up, dx_up)
    
    return (h_up, c5_up, n_up)
