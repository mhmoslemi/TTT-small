def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Adjust by adding diff/n to each component
    h = h + diff / n
    
    # Re-clip
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_worst_shift(h):
    """Return the worst shift index for current h."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # max is at index n-1 (middle), so shift is at indices in [0, 2n-2]
    # we want to avoid the 0 shift (index n-1 which corresponds to shift 0) 
    # So we look at [0, n-2] and [n+1, 2n-2]
    max_idx = np.argmax(corr)
    # Pad left-to-right
    worst_shift = max_idx - (n-1)
    return worst_shift

def compute_gradient_for_shift(h, shift_idx):
    """Compute gradient for a specific shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    
    for i in range(n):
        # Handle boundary conditions (clip/zero)
        left_idx = (i + shift_idx) % n
        right_idx = (i - shift_idx) % n
        # gradient is dx * (h[left] - h[right])
        grad[i] = dx * (h[left_idx] - h[right_idx])
    return grad

def line_search(h, grad, max_iter=10):
    """Perform backtracking line search."""
    alpha = 1.0
    dx = 2.0 / len(h)
    h_current = h.copy()
    c5_current = evaluate_c5(h_current)

        h_trial = h_current - alpha * grad
        h_trial = project_to_feasible(h_trial, len(h))
        c5_trial = evaluate_c5(h_trial)
        
        if c5_trial < c5_current:
            return h_trial, c5_trial
        else:
            alpha *= 0.5
    
    return h_current, c5_current

def memetic_refinement(h, max_steps=50):
    """Perform memetic refinement using projections."""
    h_current = h.copy()
    c5_current = evaluate_c5(h_current)
    
    for _ in range(max_steps):
        worst_shift = compute_worst_shift(h_current)
        grad = compute_gradient_for_shift(h_current, worst_shift)
        h_new, c5_new = line_search(h_current, grad)
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
        else:
            break
    return h_current, c5_current
