import numpy as np
import time
import cvxpy as cp
from scipy import sparse as sp
from scipy.signal import correlate

def build_Q(k, n):
    """Build sparse matrix for quadratic term h_i * h_{i+k}"""
    if k == 0:
        return sp.csr_matrix((n, n), dtype=float)
    data = np.ones(n - abs(k))
    rows = np.arange(n - abs(k))
    cols = rows + k if k >= 0 else rows - k
    Q = sp.csr_matrix((data, (rows, cols)), shape=(n, n))
    return 0.5 * (Q + Q.T)

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def solve_qp(n, Kmax, warm_h=None):
    """Solve convex QP for optimal h"""
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2]
    
    for k in range(-Kmax, Kmax + 1):
        Q = build_Q(k, n)
        constraints.append(cp.quad_form(h, Q) >= n / 2 - t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    if warm_h is not None:
        h.value = warm_h
        
    prob.solve(solver=cp.OSQP, verbose=False, max_iter=5000, eps_abs=1e-9)
    
    return h.value, t.value

def binary_refine(h_frac, n, dx, time_limit):
    """Refine fractional solution to binary with swaps"""
    # Initialize binary from top n/2 entries
    idx = np.argsort(-h_frac)[:n // 2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1
    
    best = c5_of_h(h_bin, dx)
    best_h = h_bin.copy()
    
    # Precompute initial correlation
    corr = correlate(h_bin, 1 - h_bin, mode='full')
    max_corr = np.max(corr)
    
    current_c5 = max_corr * dx
    
    while time.time() < time_limit:
        # Pick two random indices
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        
        # Swap
        h_bin[i1] = 0
        h_bin[i0] = 1
        
        # Update correlations (efficient update)
        new_corr = correlate(h_bin, 1 - h_bin, mode='full')
        new_c5 = np.max(new_corr) * dx
        
        if new_c5 < best:
            best = new_c5
            best_h = h_bin.copy()
        else:
            # Revert
            h_bin[i1] = 1
            h_bin[i0] = 0
    
    return best_h, best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters to test
    n_list = [250, 300, 350, 400]
    kmax_ratio = 0.75
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    warm_h = None
    for n in n_list:
        Kmax = int(kmax_ratio * n)
        dx = 2.0 / n
        
        try:
            h_sol, t_val = solve_qp(n, Kmax, warm_h=warm_h)
            
            # Scale to match target sum (this is ensured by constraints in QP)
            h_sol = np.clip(h_sol, 0, 1)
            # Renormalize if needed
            h_sol = h_sol * (n / 2) / np.sum(h_sol)
            
            # Binary refinement
            refinement_time = start_time + budget_s * 0.3
            h_bin, c5_val = binary_refine(h_sol, n, dx, refinement_time)
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_bin
                best_n = n
                
            warm_h = h_sol  # For next iteration
            
        except Exception as e:
            continue
        
        if time.time() - start_time > budget_s - 1.0:
            break
    
    # Final refinement with higher resolution if time allows
    if best_h is not None and (time.time() - start_time) < budget_s - 0.5:
        # High-resolution check
        fine_n = 2000
        fine_dx = 2.0 / fine_n
        fine_h = np.interp(np.linspace(0, 2, fine_n), np.linspace(0, 2, best_n), best_h)
        c5_fine = c5_of_h(fine_h, fine_dx)
        if c5_fine < best_c5:
            best_c5 = c5_fine
    
    return (best_h, best_c5, best_n)
