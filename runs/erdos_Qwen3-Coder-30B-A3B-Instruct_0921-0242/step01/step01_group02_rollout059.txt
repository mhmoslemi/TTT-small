import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate
import math

def c5_fft(h):
    """Compute C5 using FFT for speed."""
    g = 1 - h
    H = rfft(h)
    G = rfft(g)
    c = irfft(H * G, 2*len(h) - 1)
    return np.max(c)

def evaluate_c5(h, dx):
    """Compute C5 using FFT and return max correlation."""
    max_corr = c5_fft(h)
    return max_corr * dx

def repair_weight(h, target_sum):
    """Repair h to have exact sum = target_sum."""
    current_sum = np.sum(h)
    diff = int(target_sum - current_sum)
    
    if diff > 0:
        # Need to flip diff zeros to ones
        zero_indices = np.where(h == 0)[0]
        if len(zero_indices) >= diff:
            flip_indices = np.random.choice(zero_indices, diff, replace=False)
            h[flip_indices] = 1
    elif diff < 0:
        # Need to flip |diff| ones to zeros
        one_indices = np.where(h == 1)[0]
        if len(one_indices) >= -diff:
            flip_indices = np.random.choice(one_indices, -diff, replace=False)
            h[flip_indices] = 0
    
    return h

def crossover(parent1, parent2):
    """Perform two-point crossover with repair."""
    n = len(parent1)
    c1 = np.random.randint(1, n-1)
    c2 = np.random.randint(c1+1, n)
    
    child = np.concatenate([parent1[:c1], parent2[c1:c2], parent1[c2:]])
    return repair_weight(child, n/2)

def mutate_swap(h):
    """Swap a 1 and a 0."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return h
    
    i1 = np.random.choice(ones)
    i0 = np.random.choice(zeros)
    
    h[i1] = 0
    h[i0] = 1
    return h

def hill_climb(h, max_moves=500):
    """Improve h by swaps that reduce the worst shift."""
    dx = 2.0 / len(h)
    max_corr = c5_fft(h)
    current_best = max_corr
    
    # Find the current worst shift
    g = 1 - h
    c = correlate(h, g, mode='full')
    worst_k = np.argmax(c) - len(h) + 1
    
    moves = 0
    improvement = True
    while moves < max_moves and improvement:
        improvement = False
        candidates = []
        for _ in range(200):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            
            # Predict change in contribution of worst shift
            ci1 = h[i1] * h[(i1 + worst_k) % len(h)]
            ci0 = h[i0] * h[(i0 + worst_k) % len(h)]
            
            delta = -(1 - 2 * h[i1]) * (1 - 2 * h[i0]) * dx

            # Simply calculate exact effect of move

            h_new[i1] = 0
            h_new[i0] = 1
            
            c_new = correlate(h_new, 1 - h_new, mode='full')
            delta_corr = c_new[np.argmax(c_new)] - max_corr
            
            if delta_corr < 0:
                candidates.append((delta_corr, i1, i0))
                
        if candidates:
            candidates.sort(key=lambda x: x[0])
            best_delta, best_i1, best_i0 = candidates[0]
            h[best_i1] = 0
            h[best_i0] = 1
            max_corr += best_delta
            improvement = True
            moves += 1
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 280
    dx = 2.0 / n
    target_sum = n / 2.0
    pop_size = 200
    elite = 5
    done = False
    best_c5 = float('inf')
    best_h = None
    
    # Population initialization
    pop = initial_population(n, pop_size, target_sum)
    for h in pop:
        h = repair_weight(h, target_sum)
    
    # Fitness evaluation
    fitness = []
    for h in pop:
        c5 = evaluate_c5(h, dx)
        fitness.append(-c5)  # Negate for maximization
    
    # GA main loop
    iters = 0
    while not done and time.time() - start_time < budget_s - 20:
        # Selection: tournament
        selected = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.randint(0, pop_size, 2)
            if fitness[idx1] > fitness[idx2]:
                selected.append(pop[idx1].copy())
            else:
                selected.append(pop[idx2].copy())
                
        # Crossover and mutation
        offspring = []
        for i in range(0, pop_size - elite, 2):
            child1 = crossover(selected[i], selected[i+1])
            child2 = crossover(selected[i+1], selected[i])
            
            # Mutation
            if np.random.rand() < 0.2:
                child1 = mutate_swap(child1)
            if np.random.rand() < 0.2:
                child2 = mutate_swap(child2)
                
            # Repair
            child1 = repair_weight(child1, target_sum)
            child2 = repair_weight(child2, target_sum)
            
            offspring.extend([child1, child2])
            
        # Elitism
        elite_indices = np.argsort(fitness)[-elite:]
        new_pop = [pop[i] for i in elite_indices]
        new_pop.extend(offspring[:pop_size - elite])
        pop = new_pop
        
        # Evaluate new population
        fitness = []
        for h in pop:
            c5 = evaluate_c5(h, dx)
            fitness.append(-c5)
            
        # Local search on best
        if iters % 10 == 0:
            best_idx = np.argmax(fitness)
            best_individual = pop[best_idx]
            improved = hill_climb(best_individual.copy())
            c5_improved = evaluate_c5(improved, dx)
            if c5_improved < best_c5:
                best_c5 = c5_improved
                best_h = improved.copy()
                
        iters += 1
        
        if best_c5 < 0.3807:
            done = True
            
    # If we haven't found a solution, use best from main loop
    if best_h is None:
        best_idx = np.argmax(fitness)
        best_h = pop[best_idx]
        best_c5 = evaluate_c5(best_h, dx)
        
    # Fine-scale refinement - upsample to 500 points
    if time.time() - start_time < budget_s - 10:
        n_fine = 500
        dx_fine = 2.0 / n_fine
        fine_h = np.repeat(best_h, np.ceil(n_fine/n).astype(int))
        fine_h = fine_h[:n_fine]
        fine_h = repair_weight(fine_h, n_fine / 2)
        
        # Run a small GA on fine grid
        fine_pop = []
        for _ in range(50):
            idxs = np.random.choice(n_fine, int(n_fine/2), replace=False)
            h = np.zeros(n_fine)
            h[idxs] = 1
            fine_pop.append(h)
            
        fine_fitness = []
        for h in fine_pop:
            c5 = evaluate_c5(h, dx_fine)
            fine_fitness.append(-c5)
            
        # Very brief improvement
        for _ in range(500):
            idx = np.random.randint(0, len(fine_pop))
            h = fine_pop[idx]
            h = mutate_swap(h)
            h = repair_weight(h, n_fine / 2)
            new_c5 = evaluate_c5(h, dx_fine)
            if -new_c5 < fine_fitness[idx]:
                fine_pop[idx] = h
                fine_fitness[idx] = -new_c5
        
        # Final hill climb
        best_fine_idx = np.argmax(fine_fitness)
        h_fine = fine_pop[best_fine_idx]
        hill_climb(h_fine, max_moves=2000)
        final_c5 = evaluate_c5(h_fine, dx_fine)
        
        if final_c5 < best_c5:
            best_h = h_fine
            best_c5 = final_c5
            
    # Final check of the solution
    c5_final = evaluate_c5(best_h, dx)
    return (best_h, c5_final, len(best_h))
