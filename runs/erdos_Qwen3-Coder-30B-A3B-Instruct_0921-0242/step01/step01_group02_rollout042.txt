import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import math

def c5_fft(h):
    """Compute C5 using FFT for speed."""
    g = 1 - h
    H = rfft(h)
    G = rfft(g)
    c = irfft(H * G, len(h) * 2 - 1)
    return np.max(c)

def repair_weight(h, target_sum):
    """Repair to exact weight (sum=target_sum) using random swaps."""
    s = np.sum(h)
    diff = int(s - target_sum)
    if diff > 0:
        # Need to zero some 1s
        ones_idx = np.where(h == 1)[0]
        flip_idx = np.random.choice(ones_idx, diff, replace=False)
        h[flip_idx] = 0
    elif diff < 0:
        # Need to one some 0s
        zeros_idx = np.where(h == 0)[0]
        flip_idx = np.random.choice(zeros_idx, -diff, replace=False)
        h[flip_idx] = 1
    return h

def crossover(p1, p2):
    """Two-point crossover."""
    n = len(p1)
    c1, c2 = sorted(np.random.choice(n, 2, replace=False))
    child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
    return child

def mutate_swap(h):
    """Swap mutation."""
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    if len(ones_idx) > 0 and len(zeros_idx) > 0:
        i1 = np.random.choice(ones_idx)
        i0 = np.random.choice(zeros_idx)
        h[i1], h[i0] = 0, 1
    return h

def hill_climb(h, max_iters=500):
    """Local hill-climbing on the worst shift."""
    n = len(h)
    best_h = h.copy()
    c = correlate(best_h, 1 - best_h, mode='full')
    k_star = np.argmax(c)
    c_val = c[k_star]
    for _ in range(max_iters):
        best_delta = 0
        best_i1, best_i0 = -1, -1
        num_tries = min(200, n * 2)
        for _ in range(num_tries):
            i1 = np.random.randint(0, n)
            i0 = np.random.randint(0, n)
            if best_h[i1] == 1 and best_h[i0] == 0:
                delta = 0
                i1k = (i1 + k_star) % n
                i0k = (i0 + k_star) % n
                if best_h[i1k] == 1:
                    delta -= 1
                if best_h[i0k] == 0:
                    delta += 1
                if delta < best_delta:
                    best_delta = delta
                    best_i1, best_i0 = i1, i0
        if best_delta < 0:
            best_h[best_i1], best_h[best_i0] = 0, 1
            c_val += best_delta
        else:
            break
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n = 280  # Better for memory and speed
    dx = 2.0 / n
    target_sum = n / 2.0
    
    pop_size = 200
    elite = 5
    p_mut = 0.2
    p_block_mut = 0.02
    
    def evaluate(h):
        return -c5_fft(h)
    
    def create_individual():
        # Random balanced binary string
        idx = np.random.permutation(n)[:int(target_sum)]
        h = np.zeros(n)
        h[idx] = 1
        return h
    
    # Initial population
    P = []
    # Interpolated from initial_h_values

    h_orig_len = len(h_orig)
    x_orig = np.linspace(0, 2, h_orig_len)
    x_new = np.linspace(0, 2, n)
    h_interp = np.interp(x_new, x_orig, h_orig)
    h_interp = np.round(h_interp)
    h_interp = repair_weight(h_interp, target_sum)
    P.append(h_interp.copy())

    # Deterministic two-blocks
    for _ in range(10):
        h = np.zeros(n)
        h[:n//4] = 1
        h[n//2:n//2+n//4] = 1
        np.random.shuffle(h)
        h = repair_weight(h, target_sum)
        P.append(h.copy())

    # Alternating pattern
    for _ in range(10):
        h = np.zeros(n)
        h[::2] = 1
        h = repair_weight(h, target_sum)
        P.append(h.copy())

    # Random balanced
    for _ in range(pop_size - 21):
        h = create_individual()
        P.append(h.copy())

    # Evaluate initial population
    fitness = [evaluate(ind) for ind in P]
    best_idx = np.argmax(fitness)
    best_h = P[best_idx].copy()
    best_c5 = -fitness[best_idx]

    # Main loop
    gen = 0
    while time.time() - start_time < budget_s - 20:
        gen += 1
        # Selection
        mating_pool = []
        for _ in range(pop_size):
            i1, i2 = np.random.randint(0, pop_size, 2)
            if fitness[i1] > fitness[i2]:
                mating_pool.append(P[i1].copy())
            else:
                mating_pool.append(P[i2].copy())
        
        # Crossover and mutation
        offspring = []
        for i in range(0, pop_size - elite, 2):
            p1, p2 = mating_pool[i], mating_pool[i + 1]
            child1 = crossover(p1, p2)
            child2 = crossover(p2, p1)
            if np.random.rand() < p_mut:
                mutate_swap(child1)
            if np.random.rand() < p_mut:
                mutate_swap(child2)
            if np.random.rand() < p_block_mut:
                # Block swap mutation
                blk_len = min(4, np.random.randint(1, n // 10 + 1))
                src_start = np.random.randint(0, n - blk_len)
                dst_start = np.random.randint(0, n - blk_len)
                if src_start + blk_len <= n and dst_start + blk_len <= n:
                    tmp = child1[src_start:src_start+blk_len].copy()
                    child1[src_start:src_start+blk_len] = child1[dst_start:dst_start+blk_len]
                    child1[dst_start:dst_start+blk_len] = tmp
                    child1 = repair_weight(child1, target_sum)
                else:
                    child1 = repair_weight(child1, target_sum)
            child1 = repair_weight(child1, target_sum)
            child2 = repair_weight(child2, target_sum)
            offspring.extend([child1, child2])
        offspring = offspring[:pop_size - elite]

        # New population
        new_P = [P[best_idx].copy()] + offspring  # Elitism

        P = new_P
        fitness = new_fitness

        # Diversity check
        if gen % 10 == 0:
            # Check if diversity is too low among worst individuals
            worst_fitness_idx = np.argsort(fitness)[-50:]
            worst_h = [P[i] for i in worst_fitness_idx]
            dists = []
            for i in range(len(worst_h)):
                for j in range(i+1, len(worst_h)):
                    dists.append(np.sum(worst_h[i] != worst_h[j]))
            mean_dist = np.mean(dists)

                for i in range(25):
                    ridx = len(worst_fitness_idx) // 2 + i
                    P[ridx] = create_individual()

        # Local hill-climb every 10 generations
        if gen % 10 == 0:
            best_h = hill_climb(best_h)
            best_c5 = c5_fft(best_h)
            if best_c5 < 0.3807:
                break
        if time.time() - start_time > budget_s - 5:
            break

    # Upsample and refine with fine-scale GA
    n_fine = 500
    fine_h = np.zeros(n_fine)
    for i in range(n):
        k = int(i * n_fine / n)
        fine_h[k] = best_h[i]
    fine_h = repair_weight(fine_h, n_fine / 2.0)
    
    # Finer GA run
    pop_size_fine = 50
    # Initial population from coarse
    fine_P = [fine_h.copy()]
    for _ in range(pop_size_fine - 1):
        h = create_individual()
        fine_P.append(h.copy())
    fine_fitness = [evaluate(h) for h in fine_P]

    gen_fine = 0
    while time.time() - start_time < budget_s - 3:
        gen_fine += 1
        # Selection
        mating_pool = []
        for _ in range(pop_size_fine):
            i1, i2 = np.random.randint(0, pop_size_fine, 2)
            if fine_fitness[i1] > fine_fitness[i2]:
                mating_pool.append(fine_P[i1].copy())
            else:
                mating_pool.append(fine_P[i2].copy())
        
        # Crossover and mutation
        offspring = []
        for i in range(0, pop_size_fine - elite, 2):
            p1, p2 = mating_pool[i], mating_pool[i + 1]
            child1 = crossover(p1, p2)
            child2 = crossover(p2, p1)
            if np.random.rand() < p_mut:
                mutate_swap(child1)
            if np.random.rand() < p_mut:
                mutate_swap(child2)
            child1 = repair_weight(child1, n_fine / 2.0)
            child2 = repair_weight(child2, n_fine / 2.0)
            offspring.extend([child1, child2])
        offspring = offspring[:pop_size_fine - elite]

        # New population
        new_P = [fine_P[best_idx].copy()] + offspring
        new_fitness = [fine_fitness[best_idx]] + [evaluate(ind) for ind in offspring]
        fine_P = new_P
        fine_fitness = new_fitness

        # Track best during fine GA
        best_idx = np.argmax(fine_fitness)
        best_c5 = -fine_fitness[best_idx]

    # Final hill climb on best from fine GA
    final_h = hill_climb(fine_P[best_idx], 2000)
    final_c5 = c5_fft(final_h)

    return (final_h, final_c5, len(final_h))
