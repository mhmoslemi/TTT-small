def compute_block_autocorrelation(h_blocks, lengths, n_eval=200):
    """Compute the full autocorrelation for a block-based function."""
    K = len(h_blocks)
    # Compute block endpoints
    cumsum = np.concatenate([[0], np.cumsum(lengths)])
    lefts = cumsum[:-1]
    rights = cumsum[1:]
    
    # Precompute all intervals
    intervals = [(lefts[i], rights[i]) for i in range(K)]
    
    # Precompute shift points
    shifts = np.linspace(0, 2, n_eval + 1)[:-1]  # from 0 to 2 in n_eval steps
    
    # For each shift, compute the correlation
    overlaps = []
    for s in shifts:
        total = 0.0
        for i in range(K):
            for j in range(K):
                l1, r1 = intervals[i]
                l2, r2 = intervals[j]
                # Compute intersection of [l1, r1] and [l2-s, r2-s]
                left = max(l1, l2 - s)
                right = min(r1, r2 - s)
                if right > left:
                    overlap_length = right - left
                    total += h_blocks[i] * h_blocks[j] * overlap_length
                else:
                    pass  # No overlap
        overlaps.append(total)
    
    return np.array(overlaps)

def block_to_h_vector(block_lengths, block_heights, n_fine=400):
    """Convert block structure to a fine grid step function."""
    # Ensure sum of lengths is 2
    lengths = np.array(block_lengths)
    lengths = lengths * 2.0 / np.sum(lengths)
    # Create cumulative intervals
    cumsum = np.concatenate([[0], np.cumsum(lengths)])
    lefts = cumsum[:-1]
    rights = cumsum[1:]
    # Create h vector
    h = np.zeros(n_fine)
    dx = 2.0 / n_fine
    for i in range(n_fine):
        x = i * dx
        for j in range(len(lengths)):
            if lefts[j] <= x < rights[j]:
                h[i] = block_heights[j]
                break
    return h

def compute_c5_from_blocks(block_lengths, block_heights, n_eval=200):
    """Compute C5 for a block structure."""
    overlaps = compute_block_autocorrelation(block_heights, block_lengths, n_eval)
    # Compute integral values for each shift k (k=0,1,...,n_eval-1)
    c5_values = 1.0 - overlaps  # c5_k = 1 - Phi(k)
    # Take maximum
    c5 = np.max(c5_values)
    return c5

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    if abs(diff) < 1e-10:
        return h
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h
