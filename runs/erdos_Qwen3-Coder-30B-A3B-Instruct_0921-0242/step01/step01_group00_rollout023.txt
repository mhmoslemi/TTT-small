import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2, with h in [0,1]. Uses water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum

    if abs(diff) < 1e-10:
        return h

    # Simple adjustment for small differences
    if abs(diff) < 0.001:
        h = h + diff / n
        h = np.clip(h, 0, 1)
        return h

    # Use water-filling type approach: find lambda that makes sum=n/2
    # Sort in descending order to compute lambda efficiently
    h_sorted = np.sort(h)[::-1]
    cumulative = np.cumsum(h_sorted)
    indices = np.arange(1, len(h_sorted) + 1)
    # Find the threshold lambda
    lambda_val = (n/2 - cumulative[-1]) / len(h_sorted)
    i_star = len(h_sorted)
    for i in reversed(range(len(h_sorted))):
        if (n/2 - cumulative[i]) / (i+1) < h_sorted[i]:
            i_star = i
            lambda_val = (n/2 - cumulative[i]) / (i+1)
    # Change the part of h that are below the threshold
    mask = h < lambda_val
    h[mask] = lambda_val
    # Now adjust the remaining to hit sum exactly
    if np.sum(h) > n/2:
        diff = np.sum(h) - n/2
        # Reduce all h[i] by diff / len(h)
        h = h - diff / len(h)
    else:
        diff = n/2 - np.sum(h)
        h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def worst_shift(h):
    """Return the shift index k that produces the maximal overlap I_k."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # Extract shift (full correlation is n-1 entries at shifts 0..n-1)
    # For full correlation returned by scipy, indices from n-1 to 2*n-2
    # shifted k = i-(n-1) where i=0 is max correlation index
    k = max_idx - (len(h) - 1)
    return k

def solve_lp_relaxation(n):
    """Solve convex relaxation of the optimization problem to get a good starting point."""
    # Define the variables
    h = cp.Variable(n)
    
    # Variables for products h[i] * h[i+k]
    # We enforce that y_k = sum_i h[i]*h[i+k] for all valid shifts k≠0
    # For scalability, we will not use z variables directly but use symbolic constraints
    # Instead, build feasibility constraints for variable y_k.
    # We'll build the 1D problem as a form of dual variable approach
    
    # Build constraints: sum h = n/2
    constraints = [cp.sum(h) == n/2, h >= 0, h <= 1]
    
    #_min h h_k such that h_k = h_i h_{i+k}
    # Introduce variable t for minimum auto-correlation and constraints
    # Use a scalar invariant t that is less than all y_k
    t = cp.Variable()
    
    # Instead of binaries and complex McCormick, let's take a simpler approach:
    # solve two-staged LP relaxation:
    # Problem: maximize t subject to:
    # - h_i h_{i+k} >= t  for all shifts k≠0 (elements of correlation matrix)
    # - sum(h) = n/2
    # - 0 <= h[i] <= 1

    # Since conjugate pairing can be computationally heavy, limit max shifts by computing 
    # only every few shifts. Let’s take every 2 to reduce problem size while still maintaining structure.
    step_size = 2
    
    # More Efficiently model ansatz:
    # Simplify constraint structure:
    # Instead of explicit bilinear terms:
    # let's assume approximating minimal correlation occurs over specific shifts only.
    # But ultimately, we are after a practical point for simulated annealing to refine.
    
    # Since CVXPY lacks standard support for disjunctive constraints like h_i h_{i+k},
    # and there's no direct QP solver that handles this well,
    # we will make minimal structure approximation for now and focus on good solver initialization.
    
    # Just return a locally structured binary or near-binary guess
    candidate = np.zeros(n)
    # Simple heuristic: dense blocks (like with alternating riffle pattern)
    block_size = max(1, n // 8)
    for i in range(0, n, block_size):
        end = min(i + block_size, n)
        candidate[i:end] = 1.0
    h_0 = project_to_feasible(candidate, n)
    return h_0

def gradient_descent_on_worst(h, attempts=50):
    """
    Run projected gradient descent on the worst shift for a fixed number of iterations.
    """
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    for _ in range(attempts):
        k = worst_shift(best_h)
        # Gradient for I_k(h) = dx - dx * sum h_i h_{i+k}
        grad = np.zeros_like(best_h)
        for i in range(n):
            kill = (i + k) % n
            kleft = (i - k) % n
            grad[i] = dx * (best_h[kill] - best_h[kleft])
        
        # Step size and update
        step_size = 0.01
        new_h = best_h - step_size * grad
        new_h = project_to_feasible(new_h, n)
        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        else:
            break
        
    return best_h, best_c5

def local_search(h, iterations=100):
    """
    Perform local hill-climb on h to improve C5.
    Modify few indices at a time.
    """
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    for _ in range(iterations):
        # Choose random set of indices to modify
        indices = np.random.choice(n, size=min(5, n//20), replace=False)
        delta = np.random.uniform(-0.01, 0.01, len(indices))
        new_h = best_h.copy()
        new_h[indices] += delta
        new_h = project_to_feasible(new_h, n)
        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    return best_h, best_c5
