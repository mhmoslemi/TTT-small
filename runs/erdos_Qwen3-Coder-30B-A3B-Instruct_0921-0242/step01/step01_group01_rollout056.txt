import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def project_to_feasible(u, n):
    """Project vector u to feasible region: 0 <= h <= 1 and sum(h) = n/2"""
    # Clip to box constraints
    h = np.clip(u, 0.0, 1.0)
    
    # Water-filling to adjust sum
    s = np.sum(h)
    if abs(s - n/2) < 1e-10:
        return h
    
    # Use bisection to find lambda
    def f(lam):
        new_h = np.clip(u + lam, 0.0, 1.0)
        return np.sum(new_h) - n/2
    
    a, b = -10.0, 10.0
    for _ in range(30):
        mid = (a + b) / 2.0
        val = f(mid)
        if val > 0:
            b = mid
        else:
            a = mid
    lam = (a + b) / 2.0
    h = np.clip(u + lam, 0.0, 1.0)
    return h

def compute_c5_fft(h):
    """Compute C5 efficiently using FFT"""
    n = len(h)
    # zero-pad to 2*n-1
    H = rfft(h, n=2*n-1)
    G = rfft(1-h, n=2*n-1)
    conv = irfft(H*G)[:2*n-1]
    # convolution normalizes to window size
    return np.max(conv)

def compute_c5_exact(h):
    """Compute C5 using direct computation"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution using numpy
    conv = np.correlate(h, g, mode='full')
    result = np.max(conv) * dx
    return result

def cmaes_global_search(n_coarse, budget_s):
    """Perform CMA-ES global search on coarse grid"""
    
    # Symmetric setup - only optimize first half
    d = (n_coarse + 1) // 2  # number of free variables
    n_half = (n_coarse + 1) // 2
    
    # Parameters
    lambda_pop = 4 + int(3 * np.log(d))
    mu = lambda_pop // 2
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / np.sum(weights)
    
    # Initial state
    mean = 0.5 * np.ones(d)
    sigma = 0.2
    C = np.eye(d)
    fitness_history = []
    
    t0 = time.time()
    restarts = 0
    max_restarts = 3
    generations = 0
    best_f = float('inf')
    best_x = None
    
    while time.time() - t0 < budget_s - 5:
        if generations > 0 and generations % 150 == 0:
            if len(fitness_history) and best_f <= min(fitness_history[-150:]):
                # Restart with current best
                if restarts < max_restarts:
                    restarts += 1
                    mean = best_x
                    sigma = sigma / 2.0
                else:
                    break  # stop after max restarts
                    
        # Sample population
        Z = np.random.randn(lambda_pop, d) @ np.linalg.cholesky(C)
        X = mean + sigma * Z  # (lambda_pop, d)
        
        # Evaluate
        fitness = []
        for i in range(lambda_pop):
            z = X[i]
            # Symmetrize
            h = np.zeros(n_coarse)
            h[:n_half] = z
            if n_coarse % 2 == 0:
                h[n_half:] = z[:0:-1]
            else:
                h[n_half:] = z[::-1]
            # Project to feasible set
            h = project_to_feasible(h, n_coarse)
            # Compute fitness
            f = compute_c5_fft(h)
            fitness.append(f)
            if f < best_f:
                best_f = f
                best_x = z.copy()
        
        # Sort by fitness
        idxs = np.argsort(fitness)
        fitness = [fitness[i] for i in idxs]
        X = X[idxs]
        
        # Update parameters
        z_w = np.sum(weights.reshape(-1,1) * Z[idxs[:mu]], axis=0)
        mean = mean + sigma * z_w
        
        # Update covariance matrix
        C = (1 - 0.1) * C + 0.1 * (Z[idxs[:mu]].T @ Z[idxs[:mu]]) / np.sum(weights)
        # Update sigma (this is a simplified version)
        # Use constant step size for simplicity
        if generations % 10 == 0:  # Adjust every 10 generations
            sigma *= 0.99
        if sigma < 1e-10:
            sigma = 1e-10
            
        fitness_history.extend(fitness[:5])  # Keep recent results
        generations += 1
    
    if best_x is not None:
        h_best = np.zeros(n_coarse)
        h_best[:n_half] = best_x
        if n_coarse % 2 == 0:
            h_best[n_half:] = best_x[:0:-1]
        else:
            h_best[n_half:] = best_x[::-1]
        h_best = project_to_feasible(h_best, n_coarse)
        return h_best, best_f
    
    return None, float('inf')

def smooth_max_grad_descent(h_seed, n_fine, beta_max=50, max_iter_per_beta=80):
    """Use smooth max to compute exact gradients and refine solution"""
    dx = 2.0 / n_fine
    h = h_seed.copy()
    
    # Add symmetry constraint
    h = (h + h[::-1]) / 2
    
    # Initialize lambda
    lambda_penalty = 10.0
    
    def compute_grad(h, beta):
        # Compute convolution
        g = 1 - h
        conv = np.correlate(h, g, mode='full')
        # for all indices
        n = len(h)
        kmax = n - 1
        
        # Soft weights
        exp_vals = np.exp(beta * dx * conv)
        w = exp_vals / np.sum(exp_vals)
        
        # Gradient computation - vectorized (Toeplitz structure)
        # Using the fast Toeplitz gradient approach
        # gradient per k is:
        # \partial conv_k / \partial h_i = (1-h_{i+k})[0<=i+k<n] - h_{i-k}[0<=i-k<n]
        # So we compute using padding trick:
        right = np.pad(1 - h, (kmax, 0))[:n]
        left = np.pad(h, (0, kmax))[:n]
        grad_per_k = right - left
        
        # Weighted sum (can be done as convolution)
        grad = dx * (grad_per_k.T @ w)
        return grad
    
    # Continue using smooth max with β increasing
    beta = 0.5
    max_iter_global = 0
    
    while beta <= beta_max:
        # We use a simple algorithm here - Iterate with L-BFGS style updates
        # but without linking to scipy's L-BFGS for portability.
        max_iter = min(max_iter_per_beta, max_iter_per_beta)
        for iter_i in range(max_iter):
            grad = compute_grad(h, beta)
            # Gradient step
            h_new = h - 0.01 * grad
            # Adjust step size to avoid overshooting and specific to problem
            h_new = h_new - lambda_penalty * (np.sum(h_new) - n_fine/2)
            h = project_to_feasible(h_new, n_fine)
            h = (h + h[::-1]) / 2  # enforce symmetry
            # Early stopping condition via gradient norm
            norm_grad = np.linalg.norm(grad)
            if norm_grad < 1e-6:
                break
                
            max_iter_global += 1
            if max_iter_global > 10000:
                break
                
        # Increase β
        beta = 2 * beta
        lambda_penalty *= 2
        if beta > beta_max:
            break
    
    return h

def stochastic_hill_climb(h_init, n_fine, budget_s):
    """Small random local search"""
    dx = 2.0 / n_fine
    h = h_init.copy()
    
    t0 = time.time()
    best_c5 = compute_c5_exact(h)
    
    # Random swap function
    n_swaps = min(6, n_fine//20)
    step = 0.004
    while time.time() - t0 < budget_s - 1:
        # Select random indices
        indices = np.random.choice(n_fine, size=n_swaps, replace=False)
        # Modify them
        h_trial = h.copy()
        for i in indices:
            delta = np.random.uniform(-step, step)
            h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
        # Project
        h_trial = project_to_feasible(h_trial, n_fine)
        h_trial = (h_trial + h_trial[::-1]) / 2  # enforce symmetry
        # Evaluate
        c5_trial = compute_c5_exact(h_trial)
        if c5_trial < best_c5:
            h = h_trial
            best_c5 = c5_trial
        # Symmetrize every 50 steps
        step_count = (step + 1) % 50
        if step_count == 0:
            h = (h + h[::-1]) / 2
    
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    # Coarse grid for global search
    n_coarse = 150
    n_fine = 300
    
    # Phase 1: CMA-ES global exploration (350s)
    budget_cmaes = 350.0
    h_coarse, c5_coarse = cmaes_global_search(n_coarse, budget_cmaes)
    
    # Return initial guess if CMA-ES failed
    if h_coarse is None:
        h_coarse = np.ones(n_coarse) * 0.5
        c5_coarse = compute_c5_fft(h_coarse)
    
    # Interpolate to fine grid
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    
    # Project to feasible coarse grid
    h_fine = project_to_feasible(h_fine, n_fine)
    
    # Phase 2: Gradient-based refinement (200s)
    h_refined = smooth_max_grad_descent(h_fine, n_fine)
    
    # Phase 3: Final hill-climb polishing (remaining time)
    remaining_time = budget_s - (time.time() - t0)
    h_final, c5_final = stochastic_hill_climb(h_refined, n_fine, remaining_time)
    
    return (h_final, c5_final, n_fine)
