import numpy as np
import scipy.optimize as opt
import time
import scipy.fft as fft

def correlation_fft(h):
    """Compute all f_k values using FFT"""
    N = len(h)
    h_pad = np.pad(h, (0, N), mode='constant')
    g_pad = np.pad(1 - h, (0, N), mode='constant')
    c = fft.fft(h_pad) * fft.fft(g_pad)
    return np.real(fft.ifft(c))[:N]

def weighted_gradient_fft(h, p):
    """Compute weighted gradient of F_τ using FFT"""
    N = len(h)
    p_pad = np.pad(p, (0, N), mode='constant')
    h_pad = np.pad(h, (0, N), mode='constant')
    g_pad = np.pad(1 - h, (0, N), mode='constant')
    
    w1 = fft.fft(p_pad) * fft.fft(g_pad)
    w2 = fft.fft(p_pad[::-1]) * fft.fft(h_pad)
    
    grad = np.real(fft.ifft(w1 - w2))[:N]
    return grad

def project_onto_Ω(v):
    """Project onto box + sum constraints via water filling"""
    z = np.clip(v, 0, 1)
    r = len(v) / 2 - np.sum(z)
    tol = 1e-12
    while abs(r) > tol:
        I = np.where((z > 0) & (z < 1))[0]
        if len(I) == 0:
            break
        delta = r / len(I)
        z[I] += delta
        z = np.clip(z, 0, 1)
        r = len(v) / 2 - np.sum(z)
    return z

def binary_from_top(h, count):
    """Convert to binary with 'count' ones, sorted by value"""
    indices = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    h_bin[indices[:count]] = 1
    return h_bin

def c5_of_h(h):
    """Compute max overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    N = 256
    h = np.random.rand(N)
    h = project_onto_Ω(h)
    
    # Nesterov accelerated gradient descent
    τ = 0.05
    for outer in range(6):
        if time.time() - start_time > budget_s - 30:
            break
        α = 1.0
        h_prev = h.copy()
        for t in range(150):
            if time.time() - start_time > budget_s - 30:
                break
            y = h + ((t-1)/(t+2)) * (h - h_prev)
            f = correlation_fft(y)
            p = np.exp(f/τ - np.max(f/τ))
            p /= p.sum()
            grad = weighted_gradient_fft(y, p)
            
            # Backtracking line search
            while True:
                z = y - α * grad
                z_proj = project_onto_Ω(z)
                F_tau_z = τ * np.log(np.sum(np.exp(f/τ)))  # This is approx
                F_tau_y = τ * np.log(np.sum(np.exp(f/τ)))  # This is approx 
                if F_tau_z <= F_tau_y - 0.5*α*np.dot(grad, grad):
                    break
                α *= 0.5
            h_new = z_proj
            h_prev, h = h, h_new
            
        τ *= 0.5
    
    # Rounding to binary
    h_bin = binary_from_top(h, N//2)
    h_refined = project_onto_Ω(h_bin)
    best_c5 = c5_of_h(h_refined)
    best_h = h_refined
    
    # Swap hill-climbing
    while time.time() - start_time < budget_s - 80:
        i = np.random.choice(N)
        j = np.random.choice(N)
        if h_refined[i] == h_refined[j]:
            continue
        cand = h_refined.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand = project_onto_Ω(cand)
        cand_c5 = c5_of_h(cand)
        if cand_c5 < best_c5 - 1e-7:
            best_h, best_c5 = cand, cand_c5
    
    return (best_h, best_c5, len(best_h))
