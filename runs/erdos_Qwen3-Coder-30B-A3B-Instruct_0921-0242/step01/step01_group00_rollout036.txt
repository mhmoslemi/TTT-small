```python
import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach: add a constant to all entries
    # Binary search for lambda
    tol = 1e-12
    low, high = -1.0, 1.0
    for _ in range(100):
        if high - low < tol:
            break
        lam = (low + high) / 2.0
        h_test = h + lam
        h_test = np.clip(h_test, 0, 1)
        sum_test = np.sum(h_test)
        if sum_test > n/2:
            high = lam
        else:
            low = lam
    lam = (low + high) / 2.0
    h = h + lam
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation of h with (1 - h) using FFT for efficiency
    h_fft = fft(h, 2*n - 1)
    inv_fft = fft(1 - h, 2*n - 1)
    corr = ifft(h_fft * np.conj(inv_fft))
    corr = np.real(corr)
    # The maximum is the one that matters, excluding index 0 (h * h shift 0)
    max_corr = np.max(corr[1:])  # Exclude self-correlation at lag 0
    return max_corr * dx

def update_velocity_and_position(particle, velocity, pbest, gbest, omega, c1, c2, n):
    """Update particle velocity and position."""
    r1, r2 = np.random.rand(2)
    velocity = omega * velocity + c1 * r1 * (pbest - particle) + c2 * r2 * (gbest - particle)
    new_particle = particle + velocity
    # Project to feasible set
    new_particle = project_to_feasible(new_particle, n)
    return new_particle, velocity

def compute_gradient(h, k_star, n):
    """Compute gradient for the worst shift."""
    dx = 2.0 / n
    grad = np.zeros_like(h)
    # Use zero-padding for out of bounds
    # For k* shift, the gradient for h_i is:
    # dx * [(1 - h_{i+k*}) - h_{i-k*}]
    h_shifted_left = np.roll(h, k_star)
    h_shifted_right = np.roll(h, -k_star)
    grad = dx * (1 - h_shifted_left - h_shifted_right)
    return grad

def backtracking_line_search(h, grad, n, max_iterations=10):
    """Perform backtracking line search along the gradient."""
    alpha = 1.0
    dx = 2.0 / n
    for _ in range(max_iterations):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n)
        c5_old = evaluate_c5(h)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_old:
            return h_new
        alpha /= 2.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 300
    n_particles = 30
    omega = 0.7
    c1 = c2 = 1.5
    max_pso_iter = 2000
    memetic_interval = 5
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    for _ in range(n_particles):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = project_to_feasible(h, n_points)
        swarm.append(h.copy())
        velocities.append(np.zeros_like(h))
        pbest.append(h.copy())
        pbest_fitness.append(evaluate_c5(h))
        
    # Global best
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_pso_iter:
        if time.time() - start_time > budget_s - 0.5:
            break
            
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Update particle
            new_h, new_v = update_velocity_and_position(h, v, pbest[i], gbest, omega, c1, c2, n_points)
            swarm[i] = new_h
            velocities[i] = new_v
            
            # Evaluate new particle
            c5_new = evaluate_c5(new_h)
            
            # Update personal best
            if c5_new < pbest_fitness[i]:
                pbest[i] = new_h.copy()
                pbest_fitness[i] = c5_new
                
            # Update global best
            if c5_new < gbest_fitness:
                gbest = new_h.copy()
                gbest_fitness = c5_new
                last_improvement = iter_count
            
        # Apply memetic refinement every few generations
        if iter_count % memetic_interval == 0 and iter_count > 0:
            # Take the best particle and refine it
            refine_h = gbest.copy()
            refine_c5 = gbest_fitness
            # Find worst shift in current best
            corr = np.correlate(refine_h, 1 - refine_h, mode="full")
            # Maximum is at 0, ignore it
            worst_shift_idx = np.argmax(corr[1:]) + 1
            # Refine with gradient descent
            for _ in range(30):
                grad = compute_gradient(refine_h, worst_shift_idx, n_points)
                new_h = backtracking_line_search(refine_h, grad, n_points)
                if new_h is not refine_h:
                    refine_h = new_h
                    refine_c5 = evaluate_c5(refine_h)
                else:
                    break
            # Update global best if improved
            if refine_c5 < gbest_fitness:
                gbest = refine_h.copy()
                gbest_fitness = refine_c5
                
        # Detect stall and inject new particles
        if iter_count - last_improvement > 20:
            # Inject new particles
            for _ in range(n_particles // 4):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                h = project_to_feasible(h, n_points)
                idx = np.random.choice(n_particles)
                swarm[idx] = h.copy()
                velocities[idx] = np.zeros_like(h)
                pbest[idx] = h.copy()
                pbest_fitness[idx] = evaluate_c5(h)
                # Update top if necessary
                if pbest_fitness[idx] < gbest_fitness:
                    gbest = pbest[idx].copy()
                    gbest_fitness = pbest_fitness[idx]
                    last_improvement = iter_count
                        
        iter_count += 1
    
    # Late-stage local polishing with random swaps
    h_final = gbest.copy()
    c5_final = gbest_fitness
    for _ in range(5000):
        if time.time() - start_time > budget_s - 0.2:
            break
        i, j = np.random.choice(n_points, 2, replace=False)
        if h_final[i] > 0 and h_final[j] < 1:
            # Transfer a small amount
            delta = np.random.uniform(0, min(h_final[i], 1 - h_final[j], 0.01))
            h_new = h_final.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5_final:
                h_final = h_new
                c5_final = c5_new
    
    return (h_final, c5_final, n_points)
```
