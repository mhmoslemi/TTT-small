import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from math import exp, log

def evaluate_c5_continuous(h):
    """Evaluates the C5 constant using full correlation.""" 
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def evaluate_c5_discrete(h, n_eval=200):
    """Fast discrete evaluation for optimization."""
    dx = 2.0 / len(h)
    # Use fewer eval points for speed
    shift_indices = np.arange(n_eval) * 2.0 / n_eval
    max_val = 0.0
    for s in shift_indices:
        # Shift by s, then integrate overlap
        s_idx = int(s / dx)
        if s_idx == 0:
            integral = np.sum(h * (1 - h))
        else:
            integral = np.sum(h[:-s_idx] * (1 - h[s_idx:]))
        val = integral * dx
        if val > max_val:
            max_val = val
    return max_val

def evaluate_block_params(u, v, K, n_eval=200):
    """Evaluate a block parameter vector."""
    u = np.array(u)
    v = np.array(v)
    # Reconstruct length parameters
    e_u = np.exp(u)
    lengths = 2 * e_u / np.sum(e_u)
    # Reconstruct heights
    h_raw = 1 / (1 + np.exp(-v))
    area = np.sum(h_raw * lengths)
    scale = 1.0 / area
    heights = np.minimum(1.0, scale * h_raw)
    # Refine if necessary
    while True:
        area = np.sum(heights * lengths)
        if abs(area - 1.0) < 1e-10:
            break
        scale = 1.0 / area
        heights = np.minimum(1.0, scale * h_raw)
        
    # Fast subgradient estimate
    # Trivial estimate: not optimal but acceptable
    h_eval = np.zeros(n_eval)
    step = 2.0 / n_eval
    cum_len = 0
    for i in range(K):
        start = cum_len
        end = cum_len + lengths[i]
        if start >= 2:
            break
        idx_start = int(start / step)
        idx_end = int(end / step)
        if idx_start >= n_eval:
            break
        if idx_end > n_eval:
            idx_end = n_eval
        h_eval[idx_start:idx_end] = heights[i]
        cum_len = end
        
    # Compute correlation
    dx = 2.0 / n_eval
    corr = correlate(h_eval, 1 - h_eval, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr
