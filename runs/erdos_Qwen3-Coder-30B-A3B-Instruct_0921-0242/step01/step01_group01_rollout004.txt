```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n0 = 80  # Coarse resolution for MILP
    n = 300  # Target resolution
    dx0 = 2.0 / n0
    dx = 2.0 / n
    
    # Compute the coarse binary solution using MILP
    def solve_binary_milp():
        # Compute number of free variables (with symmetry)
        d = n0 // 2 + 1  # First half + middle
        # Variables
        x = cp.Variable(d, boolean=True)
        y = cp.Variable((d, n0), boolean=True)
        M = cp.Variable()
        
        # Constraints
        constraints = []
        
        # Weight constraint: exactly n0/2 ones, enforced by symmetry and x[0]=1
        # Sum of first half + second half (excluding middle) + 1 (x[0]) = n0/2
        # So sum(first_half) + sum(first_half) - x[0] + x[0] = n0/2 => 2*sum(first_half) = n0/2 => sum = n0/4
        # Since x[0]=1, we have sum_{i=1}^{d-1} x[i] = n0/4 - 0.5
        constraints.append(sum(x[1:]) == n0/4 - 0.5)
        
        # Product constraints (McCormick)
        for i in range(d):
            for k in range(n0):
                if i + k < d:
                    # y[i,k] = x[i] * x[i+k]
                    constraints.extend([
                        y[i,k] <= x[i],
                        y[i,k] <= x[i+k],
                        y[i,k] >= x[i] + x[i+k] - 1
                    ])
        
        # Autocorrelation matching
        for k in range(n0):
            # Compute Z_k = sum_{i} y[i,k]
            Z_k = sum(y[i,k] for i in range(d) if i+k < d)
            # Overlap at shift k = (n0/2) - Z_k
            # Constraint: (n0/2) - Z_k <= M , equivalently Z_k >= n0/2 - M
            constraints.append(Z_k >= n0/2 - M)
        
        # Objective
        objective = cp.Minimize(M)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            # Use CBC as solver
            prob.solve(solver=CPLEX, verbose=False, time_limit=20)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                x_sol = [int(v.value.item()) for v in x]
                # Reconstruct full vector with symmetry
                full_x = np.array(x_sol + x_sol[-2::-1])  # Copy to mirror and exclude middle again
                return full_x
            else:
                print(f"MILP returned status: {prob.status}")
                return None
        except Exception as e:
            print(f"MILP solve error: {e}")
            return None
    
    # Prefer binary solution to fall back to
    binary_sol = None
    if time.time() - start_time < budget_s - 10:
        binary_sol = solve_binary_milp()
    
    # If binary solution found, upsample it to target resolution
    if binary_sol is not None and np.sum(binary_sol) == n0 // 2:
        # Stack with repeat and crop
        h0 = np.repeat(binary_sol, n // n0)  # Maybe more or less than exact
        if len(h0) > n:
            h0 = h0[:n]
        elif len(h0) < n:
            h0 = np.pad(h0, (0, n - len(h0)), 'constant', constant_values=0)
        
        h0 = np.clip(h0, 0, 1)
        # Scale to sum = n/2
        h0 = h0 * (n / 2) / np.sum(h0)
    else:
        # Fallback: uniform
        h0 = np.ones(n) * 0.5
    
    # Refine with smooth max descent
    def soft_max_c5(h, beta=1.0):
        """Computes soft maximum of correlation values"""
        dx = 2.0 / len(h)
        g = 1.0 - h
        # Compute all overlaps using convolution (full mode)
        corr = np.correlate(h, g, mode="full")  # Length 2*n - 1
        corr *= dx
        
        # Soft max
        exp_corr = np.exp(beta * corr)
        softmax = np.log(np.sum(exp_corr)) / beta
        return softmax
    
    def gradient_soft_c5(h, beta=1.0):
        """Soft max gradient"""
        dx = 2.0 / len(h)
        g = 1.0 - h
        # Compute overlaps
        corr = np.correlate(h, g, mode="full")
        corr *= dx
        
        # Compute weights and gradient
        exp_corr = np.exp(beta * corr)
        w = exp_corr / np.sum(exp_corr)
        
        # Gradient part
        grad = np.zeros_like(h)
        # Now go through formulations (weight times correlator derivative)
        for k in range(len(corr)):
            # Find the shift
            shift = k - len(h) + 1  # Shift at k -- from - (n-1) to n-1
            if shift >= 0:
                # positive shift
                # Term h[i] * (1 - h[i+shift]) contributes to correlation at k
                # derivative: 1 - h[i+shift] - h[i-shift] (only valid indices)
                left = np.pad(h, (0, abs(shift)), mode='constant')[shift:]
                right = np.pad(h, (abs(shift), 0), mode='constant')[:-shift]
                if shift == 0:
                    # We're reading h[i]*(1-h[i]) in same position
                    part_grad = 1 - h - h  # same problem
                else:
                    part_grad = 1 - left[abs(shift):] - right[:len(h)-abs(shift)]
                # Now we need padding to align with h
                part_grad = (part_grad * w[k] * dx)
                # Accumulate gradient
                if k < len(h):
                    grad[:len(h)] += part_grad
                else:
                    pad_left = 2 * len(h) - 2 - k
                    grad[:len(h)] += part_grad[pad_left:]
            else:
                # negative shift
                # shifting towards left, positive shift in direction
                left = np.pad(h, (0, abs(shift)), mode='constant')[abs(shift):]  
                right = np.pad(h, (abs(shift), 0), mode='constant')[:-abs(shift)]
                part_grad = 1 - left - right
                part_grad = (part_grad * w[k] * dx)
                # And we store at diff indices
                if k < len(h):
                    grad[k:] += part_grad[:-k]  # pad left when k<length
                else:
                    # Backup
                    grad[:len(h)] += part_grad[:len(h)]  # should not be needed in recurr logic
                pass  
        return grad
    
    # Water-filling projection onto integral constraint
    def project_sum(h_target, target_sum):
        h = h_target.copy()
        while True:
            current_sum = np.sum(h)
            diff = current_sum - target_sum
            if abs(diff) < 1e-12:
                break
            # Get indices that are strictly in (0,1)
            mask = (h > 0) & (h < 1)
            num_open = np.sum(mask)
            if num_open == 0:
                break
            step = diff / num_open
            h[mask] -= step
            # Clip
            h = np.clip(h, 0, 1)
        return h
    
    # Apply symmetry
    def enforce_symmetry(h):
        h_sym = (h + h[::-1]) / 2
        return h_sym
    
    # Optimization phases
    h_refined = h0.copy()
    h_refined = project_sum(h_refined, n / 2)
    h_refined = enforce_symmetry(h_refined)
    
    # Stage 1 to 3: refine with soft max
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]  # step size may vary for each beta
    
    for i, beta in enumerate(betas):
        # L-BFGS-B or similar solver works best
        # We loop manually one time solver-like, to use gradient descent
        
        def objective(h):
            h = np.clip(h, 0, 1)
            h = project_sum(h, n / 2)
            h = enforce_symmetry(h)
            return soft_max_c5(h, beta)
            
        def grad_func(h):
            h = np.clip(h, 0, 1)
            h = project_sum(h, n / 2)
            h = enforce_symmetry(h)
            return gradient_soft_c5(h, beta)
        
        max_iter = 80
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            grad = grad_func(h_refined)
            h_refined -= 0.1 * grad  # small step size
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = project_sum(h_refined, n / 2)
            h_refined = enforce_symmetry(h_refined)
            
            # Early stopping if improvement is small
            current_c5 = soft_max_c5(h_refined, beta)
            if iteration > 20 and iteration % 10 == 0:
                c5_values = [
                    soft_max_c5(h_refined, beta) for _ in range(3)
                ]
                if np.max(np.diff(c5_values)) < 1e-5:
                    break
        
        # Every beta, also compute true C5
        if time.time() - start_time > budget_s - 1:
            break
    
    h_final = h_refined.copy()
    h_final = project_sum(h_final, n / 2)
    h_final = enforce_symmetry(h_final)
    
    # Final polishing: small stochastic hill climb
    best_c5 = float('inf')
    best_h = h_final.copy()
    
    def eval_c5(h):
        dx = 2.0 / len(h)
        g = 1.0 - h
        conv = np.correlate(h, g, mode="full")
        return np.max(conv) * dx
    
    c5_final = eval_c5(best_h)
    
    # Polishing
    t_left = budget_s - (time.time() - start_time) - 1
    n_perturbed = min(6, n // 20)
    eps = 0.004
    max_trials = int(t_left * 1000)  # very generous trial estimate
    trials = 0
    while trials < max_trials and time.time() - start_time < budget_s - 1:
        # Make candidates
        perturb_indices = np.random.choice(n, size=n_perturbed, replace=False)
        h_trial = best_h.copy()
        for i in perturb_indices:
            delta = np.random.uniform(-eps, eps)
            h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
        
        h_trial = project_sum(h_trial, n / 2)
        h_trial = enforce_symmetry(h_trial)
        c5_trial = eval_c5(h_trial)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        
        trials += 1
    
    # Final check
    final_c5 = eval_c5(best_h)
    
    return (best_h, final_c5, n)
```
