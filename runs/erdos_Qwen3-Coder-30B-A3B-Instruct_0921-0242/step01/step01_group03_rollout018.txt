def build_shift_matrices(N):
    """Build sparse CSR shift matrices A_k for k=0..N-1."""
    A_list = []
    for k in range(N):
        data = np.ones(N - k)
        row = np.arange(N - k)
        col = np.arange(k, N)
        A = sp.csr_matrix((data, (row, col)), shape=(N, N))
        A_list.append(A)
    return A_list

def bounded_simplex_projection(h, target_sum):
    """Project to the feasible set with sum = target_sum and bounds [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    deficit = target_sum - current_sum
    if abs(deficit) < 1e-10:
        return h
    
    # Distribute deficit uniformly among interior points.
    # Sort indices by value, largest first.
    idx_sorted = np.argsort(h)[::-1]
    # Find how many we can raise to 1
    total_increase = 0
    for idx in idx_sorted:
        if h[idx] < 1:
            can_increase = 1 - h[idx]
            if total_increase + can_increase <= deficit:
                total_increase += can_increase
                h[idx] = 1.0
            else:
                h[idx] += deficit - total_increase
                total_increase = deficit
                break
        else:
            continue
    
    # If deficit remains, distributed to points below 1
    if abs(total_increase - deficit) > 1e-10:
        idx_sorted = np.argsort(h)
        for idx in idx_sorted:
            if h[idx] < 1 and abs(total_increase - deficit) > 1e-10:
                can_increase = 1 - h[idx]
                delta = min(can_increase, deficit - total_increase)
                h[idx] += delta
                total_increase += delta
            if abs(total_increase - deficit) <= 1e-10:
                break
    
    return h

def eval_c5_exact(h):
    """Compute the true C5 via full correlation."""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def eigen_rounding(X_sdp, h_sdp, n_points):
    """Principal eigenvector rounding."""
    _, V = np.linalg.eigh(X_sdp)
    v1 = V[:, -1]
    h_eig = np.abs(v1) * np.sqrt(np.max(np.linalg.eigvals(X_sdp)))
    h_eig = bounded_simplex_projection(h_eig, n_points / 2)
    return h_eig

def diagonal_rounding(X_sdp, n_points):
    """Diagonal threshold rounding."""
    h_diag = np.sqrt(np.diag(X_sdp))
    idx_sorted = np.argsort(h_diag)[::-1]
    h_thr = np.zeros_like(h_diag)
    h_thr[idx_sorted[:n_points // 2]] = 1.0
    return h_thr

def correlate_incremental_update(current_h, k_indices, corr_vals, old_i, old_j, new_i, new_j, A_list, dx):
    """Update correlation values for a swap."""
    # This updates contributions for all shifts k, which need to be looked up.
    # Given K indices that are new, and old pair (i,j)
    # only the contributions from shifts pointing to (i,j) and (new_i, new_j) change.
    corr_copy = corr_vals.copy()
    # Contributions change for these shifts
    for k_idx in k_indices:
        A_k = A_list[k_idx]
        # Old contributions to k_idx...
        # Contribution of old_i with old_j
        if old_i < len(current_h) and old_j < len(current_h) and A_k[old_i, old_j]:
            corr_copy[k_idx] -= current_h[old_i] * current_h[old_j] * dx
        # New contributions to k_idx...
        # Contribution of new_i with new_j
        if new_i < len(current_h) and new_j < len(current_h) and A_k[new_i, new_j]:
            corr_copy[k_idx] += current_h[new_i] * current_h[new_j] * dx
    return corr_copy

        h_trial = best_h.copy()
        h_trial[p], h_trial[q] = h_trial[q], h_trial[p]

        g_trial = 1.0 - h_trial

            accepted_swaps += 1

        iter_count += 1

    print(f"Local swap search done: {accepted_swaps} swaps accepted")
    return best_h

def solve_sdp(N, A_list):
    """Solve the SDP relaxation."""
    # Variables
    h = cp.Variable(N)
    X = cp.Variable((N, N), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [X >> 0]  # PSD
    for i in range(N):
        constraints.append(X[i, i] == h[i])
    constraints.append(cp.sum(h) == N / 2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    for k in range(N):
        constraints.append(cp.trace(A_list[k] @ X) >= t)

    # Objective
    objective = cp.Maximize(t)

    # Solve with SCS
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=15000, eps=1e-4)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return h.value, X.value, t.value
        else:
            print("SDP failed to solve successfully:", prob.status)
            return None, None, None
    except Exception as e:
        print("SDP solver error:", e)
        return None, None, None
