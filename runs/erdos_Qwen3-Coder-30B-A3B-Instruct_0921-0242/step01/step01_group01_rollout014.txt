import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set problem parameters
    n = 200  # discretization points
    dx = 2.0 / n
    d = n // 2 + 1  # number of independent variables due to symmetry

    # Initialize problem variables
    w = cp.Variable(d)  # independent h values
    p = {}  # product variables
    M = cp.Variable()  # upper bound on overlap

    # Binary penalty parameter
    lambda_param = 0.1

    # Constraints
    box_constraints = [0 <= w, w <= 1]
    sum_constraint = cp.sum(w) == n / 4 + 0.5  # same as sum(h) = n/2

    # Start with shift k=0 (this makes later relaxation easier)
    active_shifts = [0]

    # Build McCormick constraints (we'll add more as needed)
    # But first setup linear model formulation
    constraints = box_constraints + [sum_constraint] + [M >= 0]

    # Create sparse product variables - indexed by shift k and valid indices i
    # Helper to check bounds
    def is_valid_i(i, k):
        return 0 <= i < n and 0 <= i + k < n

    # Build the full problem structure
    best_c5 = float('inf')
    best_h = None

    # Define helper to make symmetric h from w
    def make_h(w_vec):
        h = np.zeros(n)
        for i in range(d):
            h[i] = w_vec[i]
            h[n - 1 - i] = w_vec[i]  # mirror symmetry
        return h

    # Phase 1: Cutting plane algorithm with linear relaxation
    # Initial master problem
    S = [0]
    k_max = 0

    # Setup linear relaxation form
    # We will keep extending the shifts until convergence
    print("Starting cutting-plane iteration")
    iteration = 0
    max_iterations = 50  # Safety
    epsilon = 1e-6  # tolerance to stop the iteration

    while (time.time() - start_time < budget_s - 5) and (iteration < max_iterations):
        iteration += 1

        # Create new LP model each time
        # This is repeated framework for solving updated subproblem
        w = cp.Variable(d)
        M = cp.Variable()
        p = {}

        # Create McCormick constraints for the current active shift set S
        constraints = [0 <= w, w <= 1, cp.sum(w) == n / 4 + 0.5, M >= 0]

        # Add term to model and constraints
        obj_terms = [M]

        # Only create partial constraints per shift
        p_vars = []

        for k in S:
            for i in range(n):
                if i + k < n and i + k >= 0:
                    # Create a separate variable p_{i,k}
                    p_var = cp.Variable()
                    p_vars.append(p_var)
                    p[(i, k)] = p_var
                    # Add McCormick constraints
                    # p_{i,k} <= h_i
                    constraints.append(p_var <= w[i])
                    # p_{i,k} <= h_{i+k}
                    constraints.append(p_var <= w[i + k])
                    # p_{i,k} >= 0
                    constraints.append(p_var >= 0)
                    # p_{i,k} >= h_i + h_{i+k} - 1
                    constraints.append(p_var >= w[i] + w[i + k] - 1)
                    # Also add to the overlap term
                    obj_terms.append(-p_var)

        # Add the overlap terms - previous std approach
        # We're building the objective and fields carefully
        obj = cp.Minimize(M - lambda_param * cp.sum([p[(i, 0)] for i in range(d)]))

        prob = cp.Problem(obj, constraints)

        # Solve the current subproblem
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status != cp.OPTIMAL:
                break
        except Exception:
            break

        w_sol = w.value
        M_val = M.value

        # Derive full h using mirror symmetry
        h = make_h(w_sol)

        # Evaluate exact overlap C5
        h_eval = h * dx
        conv = np.correlate(h_eval, 1 - h_eval, mode='full')
        c5_exact = np.max(conv)

        if c5_exact < best_c5:
            best_c5 = c5_exact
            best_h = h.copy()

        # Check for convergence (if we are close to M)
        # To terminate: check if the actual C5 matches the bound
        if abs(c5_exact - M_val) < epsilon:
            break

        # Find worst shift: compute C5_k(h) = int h(x) (1 - h(x+k)) dx for all shifts k
        # Let's evaluate over a smaller set for feasible computation.
        max_k = 0
        max_overlap = 0
        # Just check few k values to avoid overload in solver
        steps = 10
        for k in range(-n//3, n//3 + 1, max(1, n // steps)):
            if k == 0: continue  # already included
            # Instead of recomputing, we take vertical slices of convolution
            shift_conv = np.correlate(h_eval, 1 - h_eval, mode='full')[n-1+k:]
            overlap_k = np.max(shift_conv)
            if overlap_k > max_overlap:
                max_overlap = overlap_k
                max_k = k
        S.append(max_k)

        lambda_param = min(5.0, lambda_param * 1.5)

    if best_h is None:  # Fallback
        best_h = make_h(np.ones(d) * 0.5)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n / 2) / np.sum(best_h)
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Final polishing step with projected gradient
    h = best_h.copy()
    h = h * (n / 2) / np.sum(h)

    def step_size_alpha(t):
        return 0.02 * (0.9 ** t)

    def project_to_feasible(h_in):
        # Clip to [0,1]
        h_in = np.clip(h_in, 0, 1)
        total = np.sum(h_in)
        diff = total - n/2
        diff_per_elem = diff / n
        # Subtract uniformly from all elements
        h_in = h_in - diff_per_elem
        # Re-clip
        h_in = np.clip(h_in, 0, 1)
        # Re-symmetrize
        h_sym = (h_in + h_in[::-1]) / 2
        return h_sym

    max_iters = 200
    for t in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute k_max that gives max overlap
        h_eval = h * dx
        conv = np.correlate(h_eval, 1 - h_eval, mode='full')
        k_max = np.argmax(conv) - (n - 1)
        # Compute subgradient g_i = (1 - h_{i+k}) - h_{i-k} if both indices valid,
        # otherwise zero
        g = np.zeros(n)
        k = k_max
        for i in range(n):
            if i + k < n and i + k >= 0:
                g[i] += (1 - h[i + k])
            if i - k >= 0 and i - k < n:
                g[i] -= h[i - k]
        # Apply update
        alpha = step_size_alpha(t)
        h_trial = h - alpha * g
        h_trial = project_to_feasible(h_trial)
        c5_trial = np.max(np.correlate(h_trial, 1 - h_trial, mode='full')) * dx
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        h = h_trial

    # Optional stochastic hill climb
    if time.time() - start_time < budget_s - 3:
        p = min(6, n // 20)
        for _ in range(1000):
            if time.time() - start_time > budget_s - 0.5:
                break
            # Select p indices uniformly
            indices = np.random.choice(n, size=p, replace=False)
            # Perturb them
            h_trial = h.copy()
            for i in indices:
                delta = np.random.uniform(-0.003, 0.003)
                h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
            h_trial = h_trial * (n / 2) / np.sum(h_trial)
            # Project and re-symmetrize if necessary
            h_trial = project_to_feasible(h_trial)
            c5_trial = np.max(np.correlate(h_trial, 1 - h_trial, mode='full')) * dx
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
            # After 40 moves, increasingly re-symmetrize
            if _ % 40 == 0:
                best_h = (best_h + best_h[::-1]) / 2

    return (best_h, best_c5, n)
