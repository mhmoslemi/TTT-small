import numpy as np

def project_to_feasible(u, n):
    """Project vector u to feasible h with sum = n/2 and bounds [0,1]"""
    u = np.clip(u, 0.0, 1.0)
    s = np.sum(u)
    if abs(s - n/2) < 1e-10:
        return u
    
    # Use binary search for lambda
    def sum_with_lambda(lam):
        return np.sum(np.clip(u + lam, 0.0, 1.0))
    
    # Find appropriate lambda
    low, high = -1.0, 1.0
    for _ in range(30):
        mid = (low + high) / 2
        s_mid = sum_with_lambda(mid)
        if s_mid < n/2:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    
    h = np.clip(u + lam, 0.0, 1.0)
    return h

def evaluate_c5_fast(h, n):
    """Fast C5 evaluation using FFT with early stopping."""
    dx = 2.0 / n
    # Zero-pad to 2*n-1 for full correlation
    H = rfft(h, n=2*n-1)
    G = rfft(1-h, n=2*n-1)
    conv = irfft(H*G)[:2*n-1]
    
    # Early stopping
    max_val = np.max(conv)
    return max_val * dx

def symmetrize_h(h, n):
    """Symmetrize h with mirror symmetry h[i] = h[n-1-i]"""
    h_sym = h.copy()
    mid = n // 2
    for i in range(mid):
        h_sym[i] = h_sym[n-1-i] = (h_sym[i] + h_sym[n-1-i]) / 2.0
    return h_sym

def cma_es_stage(n_coarse, budget_s):
    """Global stage with CMA-ES on reduced symmetric space."""
    np.random.seed(42)  # For reproducibility
    
    # CMA-ES parameters
    d = (n_coarse + 1) // 2  # half the size due to symmetry
    popsize = 4 + int(3 * np.log(d))
    
    # Initial parameters
    mu = 0.5 * np.ones(d)
    sigma = 0.2
    lambda_ = popsize
    
    # Adaptive restart tracking
    best_so_far = float('inf')
    best_h = None
    iterations_no_improvement = 0
    restart_count = 0
    
    start_time = time.time()
    
    # CMA-ES main loop
    for gen in range(10000):  # large upper bound
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate offspring
        z = np.random.randn(lambda_, d) * sigma
        offspring = mu + z
        
        # Evaluate each candidate
        best_candidate_c5 = float('inf')
        best_candidate = None
        
        for i in range(lambda_):
            # Create full h from symmetric part
            z_i = offspring[i]
            h_i = np.zeros(n_coarse)
            h_i[:d] = z_i
            h_i = symmetrize_h(h_i, n_coarse)

            # Project to feasible set
            h_i = project_to_feasible(h_i, n_coarse)
            
            # Evaluate with fast FFT
            c5_i = evaluate_c5_fast(h_i, n_coarse)
            
            if c5_i < best_candidate_c5:
                best_candidate_c5 = c5_i
                best_candidate = h_i.copy()
        
        # Update with best from generation
        if best_candidate_c5 < best_so_far:
            best_so_far = best_candidate_c5
            best_h = best_candidate.copy()
            iterations_no_improvement = 0
        else:
            iterations_no_improvement += 1
            
        # Adaptation criteria
        if iterations_no_improvement >= 150:
            if restart_count < 3:
                mu = best_h[:d]  # restart from best
                sigma /= 2.0
                restart_count += 1
                iterations_no_improvement = 0
            else:
                break
                
    return best_h, best_so_far

def hill_climb(h, n, dx, budget_s):
    """Final stochastic hill climb with single moves"""
    start_time = time.time()
    
    for iter_count in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Select indices
        p = min(6, n//20)
        indices = np.random.choice(n, p, replace=False)
        
        # Perturb
        h_trial = h.copy()
        for i in indices:
            delta = np.random.uniform(-0.004, 0.004)
            h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
        
        # Project to feasible set
        h_trial = project_to_feasible(h_trial, n)
        
        # Evaluate
        c5_trial = evaluate_c5_fast(h_trial, n)
        c5_current = evaluate_c5_fast(h, n)
        
        if c5_trial < c5_current:
            h = h_trial
        
        # Symmetrize periodically
        if iter_count % 50 == 0:
            h = symmetrize_h(h, n)
    
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Stage 1: CMA-ES global search on coarse grid
    n_coarse = 150  # Coarse for speed
    n_fine = 300    # Final resolution
    
    h_coarse_best, c5_coarse = cma_es_stage(n_coarse, 350)  # ~350s budget
    
    # Stage 2: Interpolate and project to fine grid
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    
    # Linear interpolation from coarse to fine
    h_interpolated = interp1d(x_coarse, h_coarse_best, kind='linear', fill_value='extrapolate')(x_fine)
    
    # Project to feasible set
    h_fine_initial = project_to_feasible(h_interpolated, n_fine)
    h_fine_symmetric = symmetrize_h(h_fine_initial, n_fine)
    
    # Stage 3: Gradient-based refinement using smooth max
    h_refined = smooth_max_refinement(h_fine_symmetric, n_fine, beta_max=50.0, beta_start=0.5)
    
    # Stage 4: Final hill climbing
    dx = 2.0 / n_fine
    h_final = hill_climb(h_refined, n_fine, dx, budget_s - (time.time() - start_time) - 5)
    
    # Final evaluation of true C5
    c5_final = evaluate_c5_fast(h_final, n_fine)
    
    # Return with proper format
    return (h_final, c5_final, n_fine)
