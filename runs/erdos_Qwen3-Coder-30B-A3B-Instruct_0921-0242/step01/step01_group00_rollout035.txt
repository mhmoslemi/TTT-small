import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import cvxpy as cp
from scipy.fft import fft, ifft

def solve_spectral_lp(n):
    """Solve the spectral convex relaxation LP for the worst overlap problem."""
    # Variables: power spectrum p_m = |h_m|^2
    p = cp.Variable(n)
    
    # Target autocorrelation value t
    t = cp.Variable()
    
    # Constraints
    # p_m >= 0, p_m <= n
    constraints = [p >= 0, p <= n]
    
    # Sum constraint: sum p_m = n^2 / 2 (since sum h_i = n/2 and h_i <= 1)
    constraints.append(cp.sum(p) == n * n / 2)
    
    # Autocorrelation constraints for all non-zero shifts
    # A_k = (1/n) * sum_m p_m * cos(2*pi*m*k/n)
    for k in range(1, n):
        ac = cp.sum(p * np.cos(2 * np.pi * np.arange(n) * k / n)) / n
        constraints.append(ac >= t)
        
    # Objective: maximize t
    obj = cp.Maximize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return p.value, t.value
        else:
            # fallback to simpler version
            return np.full(n, n/2), 0.0
    except:
        return np.full(n, n/2), 0.0

def gerchberg_saxton(p_spectrum, n, max_iter=200):
    """Apply Gerchberg-Saxton phase retrieval algorithm."""
    # Initialize random phases
    phases = np.random.uniform(0, 2*np.pi, n)
    
    # Complex spectrum
    spectrum = np.sqrt(p_spectrum) * np.exp(1j * phases)
    
    # Inverse FFT to get candidate vector
    x = ifft(spectrum).real
    
    for _ in range(max_iter):
        # Apply clipping to [0,1] 
        y = np.clip(x, 0, 1)
        
        # Waterfilling to enforce sum constraint
        current_sum = np.sum(y)
        delta = n/2 - current_sum
        
        if abs(delta) > 1e-10:
            # Simple addition hack for waterfilling
            # Map to [0,1] and adjust
            new_y = y + delta/n
            new_y = np.clip(new_y, 0, 1)
            # Re-adjust if needed
            new_sum = np.sum(new_y)
            if abs(new_sum - n/2) > 1e-8:
                # Binary search
                scale = 1.0
                def adjust(alpha):
                    test_y = y + alpha * delta/n
                    test_y = np.clip(test_y, 0, 1)
                    return np.sum(test_y) - n/2
                lo, hi = -2, 2
                while hi - lo > 1e-6:
                    mid = (lo + hi) / 2
                    val = adjust(mid)
                    if val > 0:
                        hi = mid
                    else:
                        lo = mid
                final_alpha = (lo + hi) / 2
                final_y = y + final_alpha * delta/n
                final_y = np.clip(final_y, 0, 1)
                y = final_y
        
        # Forward FFT
        Y = fft(y)
        
        # Replace magnitude with target
        magnitudes = np.abs(Y)
        new_spectrum = (p_spectrum ** 0.5) * np.exp(1j * np.angle(Y))
        
        # Inverse FFT
        x = ifft(new_spectrum).real
    
    return x

def project_simplex_box(h, n):
    """Project h such that sum(h) = n/2 and all 0 <= h[i] <= 1."""
    # Clip first
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    delta = n/2 - current_sum
    
    if abs(delta) < 1e-10:
        return h
    
    # Add uniformly
    h = h + delta / n
    
    # Clip again
    h = np.clip(h, 0, 1)
    return h

def binary_sharpen(h, mu=1e-2, max_iter=30):
    """Sharpen h towards binary"""
    h = h.copy()
    for i in range(max_iter):
        # Soft gradient: smooth penalty
        grad = np.zeros_like(h)
        # Simple smooth approximation to h*(1-h)
        for j in range(len(h)):
            for k in range(len(h)):
                delta = k - j
                if delta < 0:
                    delta += len(h)
                if delta > 0:
                    grad[j] -= h[k]  # from j+delta
                    grad[k] += h[j]  # to k-delta
        # Soft gradient of penalty term
        penalty_grad = mu * (1 - 2*h)
        grad += penalty_grad
        
        # simple gradient step
        h_new = h - 0.01 * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum
        h = project_simplex_box(h_new, len(h))
        
        # Option: round final values
        h = np.clip(h, 0, 1)
        for j in range(len(h)):
            if h[j] < 0.01:
                h[j] = 0.0
            elif h[j] > 0.99:
                h[j] = 1.0

    return h

def evaluate_c5_batch(h):
    """Batch evaluation of all shifts’ overlaps."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def gradient_worst_shift(h, max_iter=1000):
    """Targeted gradient descent on the worst shift."""
    n = len(h)
    dx = 2.0 / n
    
    h = h.copy()
    current_c5 = evaluate_c5_batch(h)
    
    num_improvements = 0
    no_improvement_count = 0
    
    for i in range(max_iter):
        # Evaluate current correlations
        corr = correlate(h, 1-h, mode='full') 
        worst_shift_idx = np.argmax(corr)
        # Adjust to zero indexing
        actual_idx = worst_shift_idx - (n - 1)
        
        # Compute gradient dI/dh = dx * [ (1 - h_{i+shift}) - h_{i-shift} ]
        grad = np.zeros_like(h)
        for j in range(n):
            left_idx = (j - actual_idx) % n
            right_idx = (j + actual_idx) % n
            grad[j] = dx * ( (1 - h[left_idx]) - h[right_idx] )
            
        # Step size
        alpha = 1.0
        step = -alpha * grad
        
        # propose
        h_proposed = h + step
        
        # project back to feasible region
        h_proposed = np.clip(h_proposed, 0, 1)
        h_proposed = project_simplex_box(h_proposed, n)
        
        # evaluate
        new_c5 = evaluate_c5_batch(h_proposed)
        
        if new_c5 < current_c5 - 1e-10:
            h = h_proposed
            current_c5 = new_c5
            num_improvements += 1
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 10:
                break
    return h, current_c5

def swap_anneal(h, max_iter=3000, tmax=1e-3, tmin=1e-8):
    """Stochastic swap refinement."""
    n = len(h)
    dx = 2.0 / n
    current_c5 = evaluate_c5_batch(h)
    
    t = tmax
    rate = np.power(tmin / tmax, 1.0 / max_iter)
    
    for i in range(max_iter):
        if t < tmin:
            break
        # Pick two distinct indices
        i1, i2 = np.random.choice(n, size=2, replace=False)
        # Ensure h[i1] > 0 and h[i2] < 1
        if h[i1] <= 0 or h[i2] >= 1:
            continue
        # Transfer small amount
        delta = np.random.uniform(0, min(h[i1], 1-h[i2], 0.02))
        if delta == 0:
            continue
        # Update
        h[i1] -= delta
        h[i2] += delta
        # Project  
        h = project_simplex_box(h, n)
        # Evaluate
        new_c5 = evaluate_c5_batch(h)
        # Accept or reject
        if new_c5 < current_c5 - 1e-10:
            current_c5 = new_c5
        elif np.random.rand() < np.exp(-(new_c5 - current_c5) / t):
            current_c5 = new_c5
        else:
            # Revert
            h[i1] += delta
            h[i2] -= delta
            h = project_simplex_box(h, n)
        t *= rate
    return h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    """Final implementation using spectral relaxation + GS + refinement"""
    np.random.seed(seed)
    start_time = time.time()

    n_points = 650
    dx = 2.0 / n_points
    
    # Always start with initial_h_values if available
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) < n_points:
            x_orig = np.linspace(0, 2, len(orig_h))
            x_new = np.linspace(0, 2, n_points)
            h_interpolated = np.interp(x_new, x_orig, orig_h)
        else:
            h_interpolated = orig_h[:n_points]
        h_normalized = h_interpolated * (n_points / 2) / np.sum(h_interpolated)
    else:
        h_normalized = np.ones(n_points) * 0.5
    
    # 1. Solve the spectral LP
    if time.time() - start_time > budget_s - 2:
        # Not enough time for all stages
        return (h_normalized, evaluate_c5_batch(h_normalized), n_points)

    try:
        p_spectrum, t_opt = solve_spectral_lp(n_points)
        print(f"LP solved: t_opt = {t_opt}")
    except:
        p_spectrum, t_opt = np.full(n_points, n/2), 0.0
    
    # 2. Gerchberg-Saxton phase retrieval
    if time.time() - start_time > budget_s - 2:
        return (h_normalized, evaluate_c5_batch(h_normalized), n_points)

    h_trial = gerchberg_saxton(p_spectrum, n_points)
    
    # 3. Project to feasible bounds
    h_projected = project_simplex_box(h_trial, n_points)
    
    # 4. Binary sharpening
    h_sharpened = binary_sharpen(h_projected, mu=1e-3, max_iter=50)
    
    # 5. Targeted gradient descent on the worst shift
    if time.time() - start_time > budget_s - 2:
        return (h_sharpened, evaluate_c5_batch(h_sharpened), n_points)
    
    h_refined, refined_c5 = gradient_worst_shift(h_sharpened, max_iter=2000)
    
    # 6. Optional swap refinement
    if time.time() - start_time > budget_s - 1.5:
        return (h_refined, refined_c5, n_points)
    
    h_final, final_c5 = swap_anneal(h_refined)
    
    return (h_final, final_c5, n_points)
