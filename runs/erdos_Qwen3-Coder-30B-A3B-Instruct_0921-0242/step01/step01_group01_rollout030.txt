import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.ndimage import shift
import time

def c5_of_h(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution
    conv = np.correlate(h, g, mode='full')
    # The full convolution has 2*len(h) - 1 elements
    # The shifts are from -(len(h)-1) to (len(h)-1)
    # But we're only interested in the overlap integral
    # which is the maximum of the convolution values
    result = np.max(conv) * dx
    return result

def soft_max_conv(h, beta):
    """Compute soft max of convolution with parameter beta"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    exp_conv = np.exp(beta * dx * conv)
    sum_exp = np.sum(exp_conv)
    if sum_exp == 0:
        return 0.0
    return (1.0 / beta) * np.log(sum_exp)

def gradient_soft_max_conv(h, beta):
    """Compute gradient of soft max of convolution with parameter beta"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    exp_conv = np.exp(beta * dx * conv)
    sum_exp = np.sum(exp_conv)
    if sum_exp == 0:
        return np.zeros_like(h)
    
    weights = exp_conv / sum_exp
    
    # Compute gradient for each element of h
    grad = np.zeros_like(h)
    n = len(h)
    for i in range(n):
        # Derivative for each shift k
        # contribution from conv_k = sum_j h[j] * (1 - h[j+k])
        # d/dh_i conv_k = (1 - h[i+k]) * (if i+k < n) - h[i-k] * (if i-k >= 0)
        for k in range(-n + 1, n):
            shift_idx = k + (n - 1)
            if shift_idx < 0 or shift_idx >= 2 * n - 1:
                continue
            if k >= 0:
                if i + k < n:
                    grad[i] += weights[shift_idx] * (1 - h[i + k])
            else:
                if i - k >= 0:
                    grad[i] += weights[shift_idx] * (-h[i - k])
    grad *= dx * beta
    
    return grad

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum constraint by clipping and reweighting"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-9:
        return h
    
    # Use water filling
    h = np.clip(h, 0, 1) 
    delta = current_sum - target_sum
    if abs(delta) < 1e-10:
        return h
    
    # Simple iterative approach to adjust
    h = h - delta / len(h)
    h = np.clip(h, 0, 1)
    return h

def gradient_refinement(h0, target_sum, max_iter=150, beta_start=0.5, beta_final=50.0, lambda_start=10.0):
    """Refine h using smooth max and gradient descent"""
    h = h0.copy().astype(float)
    n = len(h)
    dx = 2.0 / n
    beta = beta_start
    lambda_ = lambda_start
    
    # L-BFGS-B approach
    def obj(h):
        c5 = soft_max_conv(h, beta)
        penalty = lambda_ * (np.sum(h) - target_sum) ** 2
        return c5 + penalty

    def grad(h):
        g = gradient_soft_max_conv(h, beta)
        g += 2 * lambda_ * (np.sum(h) - target_sum)
        return g
    
    # Loop over beta values
    beta_steps = 6  # 6 steps: 0.5, 1, 2, 4, 8, 16, 32, 50
    betas = np.logspace(np.log10(beta_start), np.log10(beta_final), beta_steps)
    
    for i, beta_i in enumerate(betas):
        if i > 0:
            # start from previous solution
            h = h.copy()
            
        lambda_ = lambda_start * (2 ** i)
        
        bounds = [(0, 1)] * n
        # Optimization inside the loop using L-BFGS-B for smooth approx 
        # But we are using scipy's minimize with default solver.
        options = {'maxiter': max_iter, 'gtol': 1e-6}
        
        try:
            result = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=bounds, options=options)
            if result.success:
                h = result.x
            else:
                h = result.x
        except:
            pass
            
        # Ensure h is within bounds and satisfies sum constraint
        h = np.clip(h, 0, 1)
        h = project_to_constraint(h, target_sum)

    return h

def decode_blocks_to_h(params, n):
    """Convert parameters to h vector"""
    # params is a list [start1, len1, height1, start2, len2, height2, start3, len3, height3, start4, len4, height4]  
    # it ensures 4 non-overlapping constant blocks  
    M = 4
    starts = params[0::3]
    lengths = params[1::3]
    heights = params[2::3]
    
    # Ensure valid positions and length
    for i in range(M):
        starts[i] = max(0, min(2 - lengths[i], starts[i]))
        lengths[i] = max(0, min(2, lengths[i]))
        heights[i] = max(0, min(1, heights[i]))
        
    # Sort by start positions and resolve overlaps
    blocks = [(starts[i], lengths[i], heights[i]) for i in range(M)]
    blocks.sort()
    final_blocks = []
    for start, length, height in blocks:
        if len(final_blocks) == 0:
            final_blocks.append((start, length, height))
        else:
            last_start, last_length, last_height = final_blocks[-1]
            if start < last_start + last_length:
                # overlapping sections: merge or keep separate piece
                new_start = max(last_start + last_length, start)
                new_length = length
                if new_start + new_length <= start + length:
                    final_blocks.append((new_start, new_length, height))
            else:
                final_blocks.append((start, length, height))
    
    # next, ensure non-overlapping condition by folding lower blocks into higher length blocks
    # Instead of full overlapping magic, we'll just keep valid sorted non-overlapping pieces
    # Approach: reprocess to make sure start positions and non-overflow  
    last_end = 0
    t_blocks = []
    total_len = 0
    for i in range(len(final_blocks)):
        s, l, h = final_blocks[i]
        s = max(s, last_end)
        if s >= 2:
            break
        l = min(l, 2 - s)
        if l <= 0:
            continue
        t_blocks.append((s, l, h))
        last_end = s + l
        total_len += l
        
    h = np.zeros(n)
    dx = 2.0 / n
    for s, l, h_val in t_blocks:
        i_s = int(np.round(s / dx))
        i_e = int(np.round((s+l) / dx))
        i_e = min(i_e, n)
        i_s = max(0, i_s)
        if i_s < i_e:
            h[i_s:i_e] = h_val
    
    # Normalize to sum = n/2
    actual_sum = np.sum(h)
    if actual_sum > 0:
        h = h * (n / 2) / actual_sum
    else:
        h.fill(0.5)
    
    # Ensure support is within [0,2] (implicit as boundaries are respected)
    return h

# Parameters for block-based optimization
def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 250  # steps across [0,2]
    dx = 2.0 / n
    target_sum = n / 2
    
    def objective(params):
        h = decode_blocks_to_h(params, n)
        c5 = c5_of_h(h)
        return c5
    
    # DE parameters
    # 4 blocks: start, length, height each (3 * 4 = 12 parameters)
    bounds = [(0, 2.0)] * 12  # Start in [0,2], length in [0,2], height in [0,1]
    
    # DE parameters
    de_popsize = 15
    de_n_gen = 60  # 60*15 = 900 evaluations
    
    try:
        result_de = differential_evolution(objective, bounds, seed=42, 
                                           popsize=de_popsize, maxiter=de_n_gen,
                                           workers=1, updating='deferred',
                                           disp=False)
        h_global = decode_blocks_to_h(result_de.x, n)
        c5_global = result_de.fun
    except Exception as e:
        # fallback to uniform solution
        h_global = np.ones(n) * 0.5
        c5_global = c5_of_h(h_global)
    
    # Refinement: smooth max gradient descent
    if time.time() - start_time < budget_s - 0.5:
        try:
            h_refined = gradient_refinement(h_global, target_sum, 
                                            max_iter=150, beta_start=0.5, 
                                            beta_final=50.0, lambda_start=10.0)
            c5_refined = c5_of_h(h_refined)
            if c5_refined < c5_global:
                h_global = h_refined
                c5_global = c5_refined
        except:
            pass
    
    # Symmetrize
    h_sym = (h_global + h_global[::-1]) / 2
    h_sym = project_to_constraint(h_sym, target_sum)
    c5_sym = c5_of_h(h_sym)
    if c5_sym < c5_global:
        h_global = h_sym
        c5_global = c5_sym
    
    # Final hill climbing
    h_best = h_global
    c5_best = c5_global
    
    while time.time() - start_time < budget_s - 0.5:
        h_trial = h_best.copy()
        r = min(5, n // 20)  # pick at most 5 indices
        indices = np.random.choice(n, size=r, replace=False)
        for i in indices:
            pert = np.random.uniform(-0.005, 0.005)
            h_trial[i] += pert
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = project_to_constraint(h_trial, target_sum)
        c5_trial = c5_of_h(h_trial)
        if c5_trial < c5_best:
            h_best = h_trial
            c5_best = c5_trial
    
    return (h_best, c5_best, n)
