import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import random

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2 and h in [0,1] using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum 
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    # Simple linear adjustment: add diff/n to each
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    
    # Full correlation using FFT for speed
    # Zero pad to 2*n-1 for linear convolution
    pad_n = 2 * n - 1
    h_padded = np.pad(h, (0, pad_n - n), 'constant')
    h_conj_padded = np.pad(1 - h, (0, pad_n - n), 'constant')
    
    corr_fft = ifft(fft(h_padded) * np.conj(fft(h_conj_padded)))
    corr = np.real(corr_fft[:pad_n])
    
    # max correlation value (excluding index 0)
    max_corr = np.max(corr[1:]) * dx
    return max_corr

def get_worst_shift(h):
    """Get the shift index that yields the minimum correlation."""
    n = len(h)
    dx = 2.0 / n
    
    pad_n = 2 * n - 1
    h_padded = np.pad(h, (0, pad_n - n), 'constant')
    h_conj_padded = np.pad(1 - h, (0, pad_n - n), 'constant')
    
    corr_fft = ifft(fft(h_padded) * np.conj(fft(h_conj_padded)))
    corr = np.real(corr_fft[:pad_n])
    
    # Find index of minimum correlation (excluding index 0)
    min_idx = np.argmin(corr[1:]) + 1
    return min_idx

def gradient_worst_shift(h, k_star):
    """Gradient of the overlap with respect to h for a specific shift k."""
    n = len(h)
    dx = 2.0 / n
    
    grad = np.zeros(n)
    for i in range(n):
        left_idx = (i - k_star) % n
        right_idx = (i + k_star) % n
        
        grad[i] = dx * (1 - h[left_idx] - h[right_idx])
    
    return grad

def backtracking_line_search(h, grad, max_iter=10):
    """Perform backtracking line search."""
    alpha = 1.0
    dx = 2.0 / len(h)
    
    h_new = h - alpha * grad
    h_new = project_to_feasible(h_new, len(h))
    c5_new = evaluate_c5(h_new)
    
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new)
        
        if c5_new < evaluate_c5(h):
            return h_new, c5_new, alpha
        alpha *= 0.5
    
    return h, evaluate_c5(h), alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use moderate size
    n_points = 300
    dx = 2.0 / n_points
    
    # Initialize swarm
    S = 30  # swarm size
    swarm = []
    for _ in range(S):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
    
    # Global best tracking
    global_best_h = None
    global_best_c5 = float('inf')
    
    # PSO parameters
    omega = 0.7
    c1 = c2 = 1.5
    
    velocities = [np.zeros(n_points) for _ in range(S)]
    personal_best_h = [h.copy() for h in swarm]
    personal_best_c5 = [evaluate_c5(h) for h in swarm]
    
    # Track best improvement time
    last_improvement_time = time.time()
    
    # PSO iterations
    generation = 0
    while time.time() - start_time < 0.7 * budget_s:
        if time.time() - start_time > budget_s - 0.5:
            break
        
        for i in range(S):
            if time.time() - start_time > budget_s - 0.5:
                break
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                            c1 * r1 * (personal_best_h[i] - swarm[i]) +
                            c2 * r2 * (global_best_h - swarm[i]) if global_best_h is not None else velocities[i])
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate
            c5 = evaluate_c5(swarm[i])
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best_h[i] = swarm[i].copy()
                personal_best_c5[i] = c5
                
                # Update global best
                if c5 < global_best_c5:
                    global_best_h = swarm[i].copy()
                    global_best_c5 = c5
                    last_improvement_time = time.time()
        
        # Memetic refinement every 5 generations
        if generation % 5 == 0 and generation > 0:
            if global_best_h is not None:
                k_star = get_worst_shift(global_best_h)
                grad = gradient_worst_shift(global_best_h, k_star)
                global_best_h, global_best_c5, _ = backtracking_line_search(global_best_h, grad)
                
                # Update personal bests for the global best
                for i in range(S):
                    if np.allclose(swarm[i], global_best_h):
                        personal_best_h[i] = global_best_h.copy()
                        personal_best_c5[i] = global_best_c5
    
        generation += 1
        
        # Early termination if no improvement for 3 seconds
        if time.time() - last_improvement_time > 3:
            # Add fresh particles
            for _ in range(S // 4):
                if time.time() - start_time > budget_s - 0.5:
                    break
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                h = project_to_feasible(h, n_points)
                swarm.append(h)
                personal_best_h.append(h.copy())
                personal_best_c5.append(evaluate_c5(h))
                velocities.append(np.zeros(n_points))
            
            # Clear extra particles
            while len(swarm) > S:
                swarm.pop()
                personal_best_h.pop()
                personal_best_c5.pop()
                velocities.pop()
    
    # Final polishing if we have time
    if time.time() - start_time < budget_s - 0.5 and global_best_h is not None:
        # Hill-climb swap
        h_final = global_best_h.copy()
        current_c5 = global_best_c5
        for i in range(5000):
            if time.time() - start_time > budget_s - 0.2:
                break
                
            # Choose move
            i1 = np.random.choice(n_points)
            i2 = np.random.choice(n_points)
            
            if h_final[i1] == 0 or h_final[i2] == 1:
                continue
                
            delta = np.random.uniform(0, min(h_final[i1], 1 - h_final[i2], 0.01))
            
            h_trial = h_final.copy()
            h_trial[i1] -= delta
            h_trial[i2] += delta
            
            # Project
            h_trial = project_to_feasible(h_trial, n_points)
            
            trial_c5 = evaluate_c5(h_trial)
            if trial_c5 < current_c5:
                h_final = h_trial
                current_c5 = trial_c5
    
    # Final check
    if global_best_h is None:
        h_final = np.ones(n_points) * 0.5
        h_final = project_to_feasible(h_final, n_points)
        current_c5 = evaluate_c5(h_final)
    else:
        h_final = global_best_h
        current_c5 = global_best_c5
    
    return (h_final, current_c5, n_points)
