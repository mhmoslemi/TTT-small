import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def solve_spectral_lp(n, verbose=False):
    """Solve the spectral relaxation LP to maximize minimum autocorrelation."""
    # Variables: p_m = |hat h_m|^2 for m = 0,...,n-1, and t
    # Constraints:
    # - sum p_m = n * n/2 = n^2/2
    # - p_m >= 0, p_m <= n
    # - sum p_m * cos(2*pi*m*k/n) >= t for k=1,...,n-1

    # Build the constraint matrix for cos(k*m/n) terms
    # Cosine matrix: C[k,m] = cos(2*pi*k*m/n)
    # We only use k=1,...,n-1 (excluding k=0) and m=0,...,n-1
    k_vals = np.arange(1, n)  # k=1,...,n-1
    m_vals = np.arange(n)     # m=0,...,n-1
    K, M = np.meshgrid(k_vals, m_vals, indexing='ij')
    cos_matrix = np.cos(2 * np.pi * K * M / n)
    
    # Add constraint for sum of p_m = n^2/2
    sum_constraint = np.ones(n)
    A_eq = np.array([sum_constraint])
    b_eq = np.array([n * n / 2])
    
    # Add constraints for each k: sum p_m * cos(2*pi*k*m/n) >= t
    A_ub = np.vstack([cos_matrix, -np.eye(n)])
    b_ub = np.hstack([np.zeros(k_vals.shape[0]), np.zeros(n) - np.inf])
    
    # Bounds on p_m
    bounds = [(0, n) for _ in range(n)] + [(None, None)]

    # Objective: maximize t
    c = np.zeros(n + 1)
    c[-1] = 1  # last component is t

    # Solve using linprog with method='highs'
    try:
        res = linprog(-c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            p_opt = res.x[:-1]
            t_opt = res.x[-1]
            return p_opt, t_opt
        else:
            # fallback to a simpler approach if fully infeasible
            p_opt = np.ones(n) * n / 2
            t_opt = 0.0
            if verbose:
                print("LP is infeasible, using conservative approach")
            return p_opt, t_opt
    except Exception as _:
        # fallback to simple initialization
        p_opt = np.ones(n) * n / 2
        t_opt = 0.0
        if verbose:
            print("LP failed, using simple initialization")
        return p_opt, t_opt

def gerchberg_saxton(p_spectrum, n, max_iter=200, seed=42):
    """Apply Gerchberg-Saxton phase retrieval algorithm."""
    np.random.seed(seed)
    
    # Initialize with random phases
    phi = np.random.uniform(0, 2 * np.pi, n)
    xhat = np.sqrt(p_spectrum) * np.exp(1j * phi)
    
    # Conjugate-symmetric extension for real signal
    if n % 2 == 0:
        # Even n: N/2+1 unique freqs with Hermitian symmetry
        xhat[n//2] = np.real(xhat[n//2])  # real
        xhat[n//2+1:] = np.conj(xhat[n//2-1:0:-1])  # Hermitian
    else:
        # Odd n: (n+1)/2 unique freqs with Hermitian symmetry
        xhat[n//2] = np.real(xhat[n//2])  # real
        xhat[n//2+1:] = np.conj(xhat[n//2-1::-1])  # Hermitian
    
    x = irfft(xhat, n=n)
    
    for iter in range(max_iter):
        # Amplitude projection: clip to box [0,1]
        y = np.clip(x, 0, 1)
        
        # Water-filling to enforce sum constraint
        current_sum = np.sum(y)
        target_sum = n / 2
        
        if abs(current_sum - target_sum) > 1e-9:
            # Binary search for lambda to get exact sum
            lambda_lo = -1.0
            lambda_hi = 1.0
            for _ in range(20):
                lambda_test = (lambda_lo + lambda_hi) / 2
                y_test = np.clip(y + lambda_test, 0, 1)
                test_sum = np.sum(y_test)
                if test_sum < target_sum:
                    lambda_hi = lambda_test
                else:
                    lambda_lo = lambda_test
                    
            lambda_final = (lambda_lo + lambda_hi) / 2
            y = np.clip(y + lambda_final, 0, 1)
            
        # Forward DFT
        yhat = rfft(y)
        
        # Spectral projection (maintain power spectrum)
        magnitude = np.abs(yhat)
        new_xhat = np.sqrt(p_spectrum) * (yhat / magnitude)  # Element-wise division
        
        # Mismatch on DC term:
        new_xhat[0] = np.real(new_xhat[0])
        
        # Hermitian symmetry for real signal
        if n % 2 == 0:
            new_xhat[n//2] = np.real(new_xhat[n//2])
            new_xhat[n//2+1:] = np.conj(new_xhat[n//2-1:0:-1])
        else:
            new_xhat[n//2] = np.real(new_xhat[n//2])
            new_xhat[n//2+1:] = np.conj(new_xhat[n//2-1::-1])
        
        x = irfft(new_xhat, n=n)
    
    return np.clip(x, 0, 1)

def project_simplex_box(h, n_points):
    """Project onto simplex with box constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) < 1e-9:
        return h.copy()
    
    # Use binary search to find optimal lambda
    delta = target_sum - current_sum
    
    # Just move all by delta / n_points if it keeps within bounds
    lambda_t = delta / n_points
    h_new = h + lambda_t
    h_new = np.clip(h_new, 0, 1)
    
    # Ensure exact target sum via 1-d bisection step if needed
    if abs(np.sum(h_new) - target_sum) > 1e-9:
        lambda_lo = -np.inf
        lambda_hi = np.inf
        
        for _ in range(20):
            lambda_test = (lambda_lo + lambda_hi) / 2
            h_test = h + lambda_test
            h_test = np.clip(h_test, 0, 1)
            test_sum = np.sum(h_test)
            
            if test_sum < target_sum:
                lambda_hi = lambda_test
            else:
                lambda_lo = lambda_test
                
        lambda_final = (lambda_lo + lambda_hi) / 2
        h_new = h + lambda_final
        h_new = np.clip(h_new, 0, 1)
    
    return h_new

def binary_sharpen(h, n_points, mu=1e-2, steps=50, verbose=False):
    """Attempt to push fractional values toward 0/1 by gradient steps."""
    h = h.copy()
    for mu_step in [1e-4, 1e-3, 1e-2]:
        for _ in range(steps):
            # Penalty function to push toward binary values
            # gradient of penalty = mu * (1 - 2*h)
            # Note: this is a simplified coarse update
            grad_penalty = mu_step * (1 - 2 * h)
            h = h - 0.01 * grad_penalty
            h = project_simplex_box(h, n_points)
    
    # Final rounding based on thresholds
    h = np.clip(h, 0, 1)
    h = np.where(h < 0.01, 0.0, h)
    h = np.where(h > 0.99, 1.0, h)
    h = project_simplex_box(h, n_points)
    
    return h

def gradient_worst_shift(h, n_points, dx, max_iter=3000, verbose=False):
    """Run targeted gradient descent on the worst shift."""
    h = h.copy()
    
    def evaluate_shift(idx, h):
        # Compute shift index corresponding to shift in circular correlation
        # Shift of k positions (k=0..n-1) appears at index k+n-1 of full correlation
        c = correlate(h, 1 - h, mode='full')
        # index of shift is (idx + n - 1)
        return dx * c[idx + n_points - 1]
    
    def compute_gradient(idx_shift, h):
        # Direct gradient of cor[h, 1-h] at shift idx_shift
        gr = np.zeros_like(h)
        # The derivative of correlation integral is:
        # d/dh_i [ sum_j h_j * (1 - h_{j+shift}) ] = (1 - h_{i+shift}) - h_{i-shift}
        shifted_pos = (np.arange(n_points) + idx_shift) % n_points
        shifted_neg = (np.arange(n_points) - idx_shift) % n_points
        gr = 1 - h[shifted_pos] - h[shifted_neg]
        return dx * gr
    
    best_c5 = np.inf
    best_h = h.copy()
    
    for iter in range(max_iter):
        # Compute actual full correlations
        c = correlate(h, 1 - h, mode='full')
        worst_k = np.argmax(c)
        worst_shift_idx = worst_k - n_points + 1  # convert from full index to shift index
        
        if iter == 0:
            if verbose:
                print(f"Initial worst shift: {worst_shift_idx}")
        
        # real shift index in [0, n)
        shift_idx = (worst_shift_idx + n_points) % n_points
        
        grad = compute_gradient(shift_idx, h)
        alpha = 1.0
        
        # Backtracking line search to stay in [0,1]
        for attempt in range(10):
            h_trial = h - alpha * grad
            h_trial = np.clip(h_trial, 0, 1)
            # re-project to simplex
            h_trial = project_simplex_box(h_trial, n_points)
            
            new_val = evaluate_shift(worst_k, h_trial)
            if new_val < best_c5:
                h = h_trial.copy()
                best_c5 = new_val
                break
            else:
                alpha *= 0.5
        else:
            if verbose:
                print(f"Line search failed at iter {iter}")
            break
        
        if iter > 0 and iter % 200 == 0:
            # Print periodic log
            if verbose:
                print(f"Gradient iter {iter}: worst shift = {worst_shift_idx}, value = {best_c5}")
    
    c_final = correlate(h, 1 - h, mode='full')
    final_c5 = np.max(c_final) * dx
    return h, final_c5

def swap_anneal(h, n_points, dx, max_time_s=60, verbose=False):
    """Stochastic swap phase using simulated annealing style."""
    t_start = time.time()
    
    def compute_c5(h):
        c = correlate(h, 1 - h, mode='full')
        return np.max(c) * dx
    
    h = h.copy()
    current_c5 = compute_c5(h)
    T = 1e-3  # initial temperature
    cooling = 0.95
    
    n_iter = 0
    while time.time() - t_start < max_time_s and n_iter < 5000:
        # Select indices i,j where h_i > 0 and h_j < 1
        idx_i = np.random.choice(n_points)
        idx_j = np.random.choice(n_points)
        while (h[idx_i] <= 0 or h[idx_j] >= 1) and n_iter < 200:
            idx_i = np.random.choice(n_points)
            idx_j = np.random.choice(n_points)
            n_iter += 1
            
        if h[idx_i] <= 0 or h[idx_j] >= 1:
            continue

        # Try moving small amount (up to 0.02 or feasible limits)
        delta = min(0.02, h[idx_i], 1 - h[idx_j])
        
        if delta <= 1e-10:
            continue
            
        h_new = h.copy()
        h_new[idx_i] -= delta
        h_new[idx_j] += delta
        h_new = project_simplex_box(h_new, n_points)

        new_c5 = compute_c5(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
        else:
            # Accept with Metropolis probability
            dE = new_c5 - current_c5
            if np.random.rand() < np.exp(-dE / T):
                h = h_new
                current_c5 = new_c5
                
        T *= cooling
        n_iter += 1
        
    return h, current_c5

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 650
    dx = 2.0 / n_points
    best_h = None
    best_c5 = np.inf
    
    # Stage 1: Solve spectral relaxation LP
    p_opt, t_opt = solve_spectral_lp(n_points, verbose=False)
    if time.time() - start_time > budget_s - 2:
        return (np.ones(n_points) * 0.5, 1.0, n_points)
    
    # Stage 2: Apply Gerchberg-Saxton
    h_gs = gerchberg_saxton(p_opt, n_points, max_iter=200, seed=seed)
    h_gs = project_simplex_box(h_gs, n_points)
    h_gs_c5 = evaluate_c5(h_gs)
    
    if h_gs_c5 < best_c5:
        best_c5 = h_gs_c5
        best_h = h_gs.copy()
    
    if time.time() - start_time > budget_s - 3:
        return (best_h, best_c5, n_points)
    
    # Stage 3: After GS, apply projection and binary sharpening
    h_binary = binary_sharpen(h_gs, n_points, mu=1e-2, steps=50)
    c5_binary = evaluate_c5(h_binary)
    
    if c5_binary < best_c5:
        best_c5 = c5_binary
        best_h = h_binary.copy()
    
    if time.time() - start_time > budget_s - 3:
        return (best_h, best_c5, n_points)
    
    # Stage 4: Local worst-shift refinement
    h_refined, c5_refined = gradient_worst_shift(best_h, n_points, dx,
                                               max_iter=3000, verbose=False)
    h_refined = project_simplex_box(h_refined, n_points)
    if c5_refined < best_c5:
        best_c5 = c5_refined
        best_h = h_refined.copy()
    
    if time.time() - start_time > budget_s - 2:
        return (best_h, best_c5, n_points)

    # Stage 5: Swap anneal
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 2:
        h_final, c5_final = swap_anneal(best_h, n_points, dx, 
                                       max_time_s=remaining_time - 1, verbose=False)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final evaluation
    return (best_h, best_c5, n_points)
