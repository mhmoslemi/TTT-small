import numpy as np

import scipy.sparse as sp
import cvxpy as cp
from scipy.signal import correlate

def build_Q(k, n):
    """Build sparse symmetric PSD matrix for term h_i h_{i+k}."""
    if k == 0:
        data = np.ones(n)
        rows = np.arange(n)
        cols = np.arange(n)
        Q = sp.csr_matrix((data, (rows, cols)), shape=(n,n))
    else:
        data = np.ones(n - abs(k))
        rows = np.arange(n - abs(k))
        cols = rows + k if k >= 0 else rows - k
        Q = sp.csr_matrix((data, (rows, cols)), shape=(n,n))
    return 0.5 * (Q + Q.T)

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def solve_qp(n, Kmax, warm_h=None):
    """Solve convex QP for optimal h."""
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    for k in range(-Kmax, Kmax+1):
        Q = build_Q(k, n)
        constraints.append(cp.quad_form(h, Q) >= n/2 - t)
    probs = cp.Problem(cp.Minimize(t), constraints)
    if warm_h is not None:
        h.value = warm_h
    probs.solve(solver=cp.OSQP, verbose=False, max_iter=5000, eps_abs=1e-9)
    return h.value, t.value

        else:
            h_bin[i1] = 1
            h_bin[i0] = 0
        if time.time() > time_limit - 1.0:
            break
    return best_h, best
