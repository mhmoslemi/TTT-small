```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math

def c5_of_h(h):
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 140  # Start here, can be adjusted if needed
    dx = 2.0 / n_points
    S = n_points // 2  # Target sum
    
    # Precompute indices for each shift k
    shift_indices = {}
    for k in range(n_points):
        shift_indices[k] = list(range(n_points - k))
    
    try:
        # Binary search on the bound t
        L = 0.0
        U = float(S)
        best_feasible_x = None
        best_feasible_t = U
        
        iteration = 0
        max_iterations = 8
        while L < U and iteration < max_iterations and (time.time() - start_time < budget_s - 5):
            T = (L + U) / 2.0
            t = cp.Variable()
            
            # Variables
            x = cp.Variable(n_points, boolean=True)
            w_vars = {}
            w_list = []
            constraints = []
            
            # Add mass constraint
            constraints.append(cp.sum(x) == S)
            # Add symmetry breaking
            constraints.append(x[0] == 1)
            
            # Add w variable and constraints for each k, i
            for k in range(n_points):
                if k not in shift_indices:
                    continue
                for i in shift_indices[k]:
                    w_name = f'w_{i}_{k}'
                    w = cp.Variable(boolean=True)
                    w_vars[(i,k)] = w
                    w_list.append(w)
                    # McCormick constraints
                    constraints.append(w <= x[i])
                    constraints.append(w <= x[i+k])
                    constraints.append(w >= x[i] + x[i+k] - 1)
            
            # Overlap constraint for this bound T
            for k in range(n_points):
                if k not in shift_indices:
                    continue
                indices = shift_indices[k]
                expr = cp.sum([w_vars[(i,k)] for i in indices])
                constraints.append(expr >= S - T)
            
            # Minimize only for feasibility testing
            prob = cp.Problem(cp.Minimize(0), constraints)
            
            # Solve with time limit
            prob.solve(solver=cp.CBC, warm_start=True, 
                       solver_opts={'max_seconds': 30})
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZE_NOT_COMPLETED]:
                # Check if solution is feasible
                if all(val.value is not None for val in x):
                    feasible = True
                    # Verify the computed t value by checking the overlap constraints separately
                    x_sol = [int(val.value) for val in x]
                    x_array = np.array(x_sol)
                    c5_val = c5_of_h(x_array)
                    if c5_val <= T * dx + 1e-10:  # Allow small tolerance
                        U = T
                        best_feasible_t = T
                        best_feasible_x = x_sol
                        # break early if already good enough
                        if T * dx < 0.3807:
                            break
                    else:
                        feasible = False
                else:
                    L = T
            else:
                L = T
                
            iteration += 1
            
        # If we found a feasible solution, use it as base
        if best_feasible_x is not None:
            h_final = np.array(best_feasible_x, dtype=float)
            # Optionally refine by full MILP solve with objective
            if (time.time() - start_time) < budget_s - 100:
                try:
                    # Reset model with objective
                    t_obj = cp.Variable()
                    x_obj = cp.Variable(n_points, boolean=True)
                    w_vars_obj = {}
                    w_list_obj = []
                    constraints_obj = []
                    
                    # Add mass and symmetry breaking
                    constraints_obj.append(cp.sum(x_obj) == S)
                    constraints_obj.append(x_obj[0] == 1)
                    
                    # Add auxiliary variables and McCormick constraints for each k, i
                    for k in range(n_points):
                        if k not in shift_indices:
                            continue
                        for i in shift_indices[k]:
                            w_name = f'w_{i}_{k}'
                            w = cp.Variable(boolean=True)
                            w_vars_obj[(i,k)] = w
                            w_list_obj.append(w)
                            # McCormick constraints  
                            constraints_obj.append(w <= x_obj[i])
                            constraints_obj.append(w <= x_obj[i+k])
                            constraints_obj.append(w >= x_obj[i] + x_obj[i+k] - 1)
                    
                    # Overlap bounds that force the objective upper bound t
                    for k in range(n_points):
                        if k not in shift_indices:
                            continue
                        indices = shift_indices[k]
                        expr = cp.sum([w_vars_obj[(i,k)] for i in indices])
                        constraints_obj.append(expr >= S - t_obj)
                    
                    # Minimize t as objective
                    prob_obj = cp.Problem(cp.Minimize(t_obj), constraints_obj)
                    prob_obj.solve(solver=cp.CBC, 
                                   solver_opts={'max_seconds': 120})
                    
                    if prob_obj.status in [cp.OPTIMAL, cp.OPTIMIZE_NOT_COMPLETED]:
                        if all(val.value is not None for val in x_obj):
                            x_sol_obj = [int(val.value) for val in x_obj]
                            h_final = np.array(x_sol_obj, dtype=float)
                            # Redo computation in case there's numeric difference
                            h_final_c5 = c5_of_h(h_final)
                            if h_final_c5 < best_c5:
                                best_c5 = h_final_c5
                                best_h = h_final
                except Exception as e:
                    pass  # Ignore for now
        
        # Final loop to polish the result if we found a feasible one
        if best_feasible_x is not None:
            h_local = np.array(best_feasible_x, dtype=float)
            if c5_of_h(h_local) < best_c5:
                best_c5 = c5_of_h(h_local)
                best_h = h_local
            
            # Simple local search to improve
            cycles = 0
            max_cycles = 15
            while (time.time() - start_time < budget_s - 50) and cycles < max_cycles:
                improved = False
                # Try random swaps
                for _ in range(10):
                    # Try poor indices first
                    idx0s = np.where(h_local == 0)[0]
                    idx1s = np.where(h_local == 1)[0]
                    if len(idx0s) > 0 and len(idx1s) > 0:
                        i0 = np.random.choice(idx0s)
                        i1 = np.random.choice(idx1s)
                        # Swap
                        h_test = h_local.copy()
                        h_test[i0] = 1
                        h_test[i1] = 0
                        c5_test = c5_of_h(h_test)
                        if c5_test < best_c5:
                            best_c5 = c5_test
                            best_h = h_test.copy()
                            h_local = h_test.copy()
                            improved = True
                if not improved:
                    break
                cycles += 1
        
    except Exception as e:
        pass  # fallback to GP construction below
    
    # If no good binary solution found, fallback to greedy
    if best_h is None:
        h_array = np.zeros(n_points)
        h_array[:n_points//2] = 1.0
        # Randomize
        np.random.shuffle(h_array)
        # Normalize to sum = 1 / dx = n_points / 2
        h_array = h_array * (n_points / 2) / np.sum(h_array)
        best_h = h_array
        best_c5 = c5_of_h(h_array)
    
    # Final refinement
    try:
        # Try local simple swaps
        h_test = best_h.copy()
        c5_bak = best_c5
        h_before = h_test.copy()
        max_swaps = 1000
        succeed_count = 0
        for i in range(max_swaps):
            if time.time() - start_time >= budget_s - 5:
                break
            idx0s = np.where(h_test == 0)[0]
            idx1s = np.where(h_test == 1)[0]
            if len(idx0s) > 0 and len(idx1s) > 0:
                i0 = np.random.choice(idx0s)
                i1 = np.random.choice(idx1s)
                h_new = h_test.copy()
                h_new[i0] = 1
                h_new[i1] = 0
                c5_new = c5_of_h(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_test = h_new.copy()
                    succeed_count += 1
                if succeed_count >= 5:
                    succeed_count = 0
            if i % 100 == 0:
                # Early stop if no improvement for a while
                if h_before is not None and (np.allclose(h_test, h_before, rtol=1e-6)):
                    break
                h_before = h_test.copy()
    except Exception as e:
        pass
    
    return (best_h, best_c5, n_points)
```
