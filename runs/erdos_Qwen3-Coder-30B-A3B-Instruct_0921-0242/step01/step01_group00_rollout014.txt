def c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def worst_shift(h):
    """Find shift k that achieves maximum overlap integral."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find shift index of maximum
    max_corr = np.max(corr) * dx
    max_idx = np.argmax(corr)
    # Convert back to shift k
    n = len(h)
    k = max_idx - (n - 1)
    return k

def proj_simplex_box(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Adjust by distributing difference
    # Affine projection: add diff/n to each component
    h = h + diff / n
    
    # Re-clip
    h = np.clip(h, 0, 1)
    return h

def gradient_step(h, k, alpha, n):
    """Perform a projected gradient descent step for the k-th overlap."""
    # Gradient of integral h(x)(1-h(x+k)) with respect to h_i
    # ∂/∂h_i = dx * (1 - 2 h_{i+k}) for the relevant indices
    grad = np.zeros_like(h)
    dx = 2.0 / n
    
    for i in range(n):
        # compute accumulate: for each i, its contribution to the score at shift k
        # derivation: I_k(h) = dx (sum h_i - sum h_i h_{i+k})
        # d I_k(h)/ d h_i = dx (1 - h_{i+k})
        # d I_k(h)/ d h_{i-k} = -dx h_i      (this is the constraint for shift k)
        idx_right = (i + k) % n
        idx_left = (i - k) % n
        if k > 0:
            grad[i] = dx * (1 - h[idx_right])
        else:
            grad[i] = 0
            
    # Update step: h -= alpha * grad
    h_new = h - alpha * grad
    # Project to feasible region
    h_new = proj_simplex_box(h_new, n)
    return h_new

def anneal(h_start, time_limit):
    """Simple simulated annealing to escape local minima."""
    np.random.seed(42)
    start_time = time.time()
    
    h_current = h_start.copy()
    c5_current = c5(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    temp = 0.5
    alpha = 0.005
    decay = 0.999

    while time.time() - start_time < time_limit:
        # Choose a random index to modify
        i = np.random.randint(len(h_current))
        # And random amount to change (± within bound) 
        delta = np.random.uniform(-0.1, 0.1)
        
        # Check feasibility 
        h_trial = h_current.copy()
        new_val = h_trial[i] + delta
        h_trial[i] = np.clip(new_val, 0, 1)
        
        # Must preserve sum exactly via waterfilling algorithm
        os = h_trial.sum()
        diff = len(h_trial)/2 - os
        
        # Redistribute
        for j in range(len(h_trial)):
            h_trial[j] += diff/len(h_trial)
        
        # Re-clip:
        h_trial = np.clip(h_trial, 0, 1)
        
        # New score
        c5_trial = c5(h_trial)
        
        # Accept or reject
        if c5_trial < c5_current:
            h_current = h_trial
            c5_current = c5_trial
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_trial.copy()
        else:
            # Accept with probability proportional to exp(-ΔE/T)
            if np.random.rand() < np.exp(-(c5_trial - c5_current)/temp):
                h_current = h_trial
                c5_current = c5_trial
                
        temp *= decay
                
    return best_h, best_c5
