import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from collections import defaultdict

def evaluate_erdos_solution(h):
    """
    Evaluate the C5 constant for a given step function h.
    Returns the maximum overlap integral.
    """
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    return np.max(conv) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 200  # Reduced for faster runs
    dx = 2.0 / n
    d = n // 2 + 1  # symmetry-reduced dimension
    
    # Create symmetric matrix indices
    def get_symmetric_indices(i, k):
        # i is the symmetric index (0 to d-1), k is shift
        if k >= 0:
            return i, i + k 
        else:  # k < 0
            return i + k, i
    
    # Symmetric indices: mapping from 0..d-1 to true indices and their mirrors
    sym_idx_map = []
    for i in range(d):
        sym_idx_map.append((i, i))  # For i and i+k with k=0
        for k in range(-n//2, n//2+1, 1):
            if k == 0: continue
            i1, i2 = get_symmetric_indices(i, k)
            if i1 >= 0 and i1 < n and i2 >= 0 and i2 < n:
                sym_idx_map.append((i, i1, i2, k))
    
    # Boolean variable to track which shifts were active
    active_k_values = dict()
    
    best_c5 = float('inf')
    best_h = None

    # Special handling for symmetry: the actual h will be built from w
    # such that h[i] = w[i] for i < d, and h[n-1-i] = w[i] for i < d
    # So h is symmetric: h[i] = h[n-1-i]
    
    def clear_constraints():
        for var in active_vars:
            active_vars.remove(var)
        active_k_values.clear()

    active_shifts = set([0])
    lambda_coeff = 0.1  # Binary penalty
    
    # Main cutting plane loop
    max_iters = 50
    epsilon = 1e-6
    last_improve_time = start_time

    try:
        # Decision variables
        w = cp.Variable(d)
        M = cp.Variable()
        p_vars = {}  # p_vars[(i,k)] = variable
        active_vars = []
        
        # Constraints
        constraints = []
        # Box constraints
        constraints += [0 <= w[i] for i in range(d)]
        constraints += [w[i] <= 1 for i in range(d)]
        # Sum constraint
        constraints += [cp.sum(w) == n/4 + 0.5]
        # McCormick envelopes for all possible shifts
        for k in range(-n//2, n//2+1):
            if abs(k) == 0:
                # for k = 0, use h_i * h_i = h_i^2
                for i in range(d):
                    # Since h_i = w[i] only for i < d
                    # Instead of implicit indexing, we'll create properly defined p_vars
                    pass
            else:
                # For |k| >= 1, \abs(i) < d and |i+k| < n
                for i in range(d):
                    if k > 0:
                        if i + k >= n: continue  # invalid range
                        i1, i2 = i, i+k  # These can be as it is even if i1 or i2 is greater than d
                        # Constraint for p[i,k] indexed at (i,i+k) in h
                        # Do not add overlapping constraints between different k shifts
                    else:  # k < 0
                        if i + k < 0: continue
                        i1, i2 = i+k, i
                        
                    # Create shared external variable p_value
                    p_key = (i, k)
                    p_temp = cp.Variable()
                    p_vars[p_key] = p_temp
                    active_vars.append(p_temp)
                    
                    # Add the 4 McCormick inequalities triplets:
                    # p <= h_i
                    constraints += [p_temp <= w[i]]
                    # p <= h_{i+k} 
                    constraints += [p_temp <= w[i+k if k < 0 else i+k]]
                    # p >= 0
                    constraints += [p_temp >= 0]
                    # p >= h_i + h_{i+k} - 1
                    constraints += [p_temp >= w[i] + w[i+k if k < 0 else i+k] - 1]
                    
                    # Ensure symmetry of p - consider binary structure
                    # p^k with k shifts do not fully fill all symmetries of far
                    # at best we preserve those needed at solving loop
                    # We add constraints per actual non-zero marginal k values
                
        # Outer set of active shifts refer to real indices
        # But let us construct set of valid shifts harder for LP
        # Fixing with reasonable computational limitation (30 rather than full n)
        
        # Implying non-redundant, case when indices of p_var are actually
        # used in valid overlaps. This transformation helps build small LP.
        def build_constraints_shifts(sset):
            local_vars = list()
            local_cons = []
            for k in sset:
                if k == 0:
                    continue
                # Add overlap constraints
                sum_p = 0.
                current_refs = []
                for i in range(d):
                    if k > 0:
                        if i + k >= n: continue
                        p_key = (i, k)
                        if p_key not in p_vars: continue
                        sum_p += p_vars[p_key]
                        current_refs.append(p_key)
                    else:
                        if i + k < 0: continue
                        p_key = (i, k)
                        if p_key not in p_vars: continue
                        sum_p += p_vars[p_key]
                        current_refs.append(p_key)
                        
                # Create proper lhs for constraint
                # \int h(1-h) = 1 - sum_p <= M
                local_cons += [1 - sum_p <= M]
                for rv in current_refs:
                    local_vars.append(p_vars[rv])
            return local_cons, local_vars
        
        # Main loop
        current_shifts = set([0])
        max_shifts = 30  # clamp it tránh overflow
        
        iteration = 0
        while iteration < max_iters and time.time() - start_time < budget_s - 5:
            iteration += 1
            # Try hard to predefine pool to speed up setup time
            
            # Only define from real use patterns of active set
            # recreate all constraints each step
            # Reconstruct with full variable list
            
            p_vars.clear()
            active_vars = []
            
            # Keep all the necessary constraints
            local_cons = []    
            # Box, line 131-136
            local_cons += [0 <= w[i] for i in range(d)]
            local_cons += [w[i] <= 1 for i in range(d)]
            local_cons += [cp.sum(w) == n/4 + 0.5]
            
            # Schedule differential assignments for symmetry processing
            # base = [0, 1, ..., d-1]
            
            for k in current_shifts:
                if k == 0:
                    # Handle k = 0 case specially for sum squares term
                    for i in range(d):
                        p_key = (i, k)
                        p_temp = cp.Variable()
                        p_vars[p_key] = p_temp
                        active_vars.append(p_temp)
                        # Add McCormick
                        local_cons += [p_temp <= w[i]]
                        local_cons += [p_temp >= 0]
                        local_cons += [p_temp >= 2*w[i] - 1]
                        local_cons += [p_temp <= w[i]]  # emulated quadratic term   
                else:
                    # General case: k != 0
                    for i in range(d):
                        if k > 0:
                            if i + k >= n:
                                continue
                            i1 = i
                            i2 = i+k
                        else:  # k < 0
                            if i + k < 0:
                                continue
                            i1 = i+k
                            i2 = i  
                        
                        # Add McCormick for this (i,k)
                        # Three constraints
                        p_key = (i, k)
                        p_temp = cp.Variable()
                        p_vars[p_key] = p_temp
                        active_vars.append(p_temp)
                        # p <= h_i, p <= h_{i+k}
                        local_cons += [p_temp <= w[i]]
                        local_cons += [p_temp <= w[i2]]  # or w[i+k]
                        # p >= 0
                        local_cons += [p_temp >= 0]
                        # p >= h_i + h_{i+k} - 1
                        local_cons += [p_temp >= w[i] + w[i2] - 1]
            
            # Now add overlap constraints 
            for k in current_shifts:
                if k == 0: continue
                sum_p = 0.
                for i in range(d):
                    p_key = (i, k)
                    if p_key not in p_vars:
                        continue
                    sum_p += p_vars[p_key]
                # Create overlap constraint
                local_cons += [1 - sum_p <= M]
            
            # Now assume all updates properly computed from above:
    
            # Objective function ends at full control:
            # Try to push toward binary value by penalizing sum square
            obj = M - lambda_coeff * sum(p_vars[(i, 0)] for i in range(d))
            prob = cp.Problem(cp.Minimize(obj), local_cons)
            
            if time.time() - last_improve_time > budget_s * 0.9:
                # Simulate cautious cancellation at late steps
                try:
                    prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
                except:
                    continue  # skip this try
            else:
                try:
                    prob.solve(solver=cp.ECOS, verbose=False)
                except:
                    print("Solver failed, next iteration")
                    continue
                    
            if prob.status != cp.OPTIMAL:
                print("LP infeasible or no optimal solution")
                break

            w_val = w.value
            M_val = M.value
            
            # Mirror and build full h
            h = np.zeros(n)            
            for i in range(d):
                h[i] = w_val[i]
                h[n-1-i] = w_val[i]
            
            # Correct normalization to match integral = 1
            # np.sum(h) * dx = 1 but we used sum(w) = n/4 + 0.5
            scale = (n/2) / np.sum(h)
            h = h * scale
            
            # Evaluate exact C5 (uses full computation)
            c5_val = evaluate_erdos_solution(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
                last_improve_time = time.time()

            # Check convergence
            if abs(c5_val - M_val) < epsilon:
                # We have reached the proven lower bound and found solution
                break
            
            # Determine shift which caused max overlap
            g = 1.0 - h
            conv = np.correlate(h, g, mode='full')
            k_max = np.argmax(conv) - (n - 1)
            assert k_max in range(-n//2, n//2+1)
            
            # Add to active set
            current_shifts.add(k_max)
            active_shifts.add(k_max)
            
            if len(current_shifts) > max_shifts:
                # Cap it
                sorted_k = sorted(current_shifts, reverse=True)
                # Remove elements that contribute less or to least defined data point
                keep = set()
                # Keep few most meaningful
                for k in sorted_k[:10]:
                    keep.add(k)
                current_shifts = keep
            
            # gradually increase binary pressure to 5.0 if we not tracking threes values
            lambda_coeff = min(5.0, lambda_coeff * 1.5)
            
            # Refresh constraint if nothing is parsed due to logic errors
            
        # End of cutting plane loop
        
        # Polish with subgradient descent
        
        # Final polishing on best found h
        if best_h is None:
            # fallback
            h0 = np.ones(n)
            h0 = h0 * (n / 2) / np.sum(h0)
            for _ in range(500):
                if time.time() - start_time > budget_s - 1:
                    break
                k_idx = -1
                g = 1.0 - best_h
                conv = np.correlate(best_h, g, mode='full')
                k_idx = np.argmax(conv) - (n - 1)  # get corresponding shift index
                if k_idx < 0:
                    # Generate a big vector for negative shift usage
                    # Formula: += (1 - h[i + k_idx]) * 1_{[i+k_idx >= 0]}
                    # and also -= h[i - k_idx] * 1_{[i-k_idx < n]}
                    # But to save memory and reindex we format it 'legbally' 
                    grad = np.zeros(n)
                    for i in range(n):
                        if i + k_idx >= 0:  # only valid reference
                            grad[i] += 1.0 - best_h[i+k_idx]
                        if i - k_idx < n:  # against that reference
                            grad[i] -= best_h[i-k_idx]
                    # Subgradient step with constrained update 
                    step_size = 0.05
                    new_h = best_h - step_size * grad
                    # Project to feasible set: [0,1]
                    new_h = np.clip(new_h, 0, 1)
                    # Ensure integral = 1  
                    new_h = new_h * (n / 2) / np.sum(new_h)
                    # Symmetrize
                    new_h = (new_h + new_h[::-1]) / 2
                    
                    candidate_c5 = evaluate_erdos_solution(new_h)
                    if candidate_c5 < best_c5:
                        best_c5 = candidate_c5
                        best_h = new_h.copy()
                else: 
                    # Handle positive k_idx
                    # Iterate each shift direction carefully then apply
                    grad = np.zeros(n)
                    for i in range(n):
                        if i + k_idx < n:
                            grad[i] += 1.0 - best_h[i+k_idx]
                        if i - k_idx >= 0:
                            grad[i] -= best_h[i-k_idx]
                    step_size = 0.05
                    new_h = best_h - step_size * grad
                    new_h = np.clip(new_h, 0, 1)
                    new_h = new_h * (n / 2) / np.sum(new_h)
                    new_h = (new_h + new_h[::-1]) / 2
                    
                    candidate_c5 = evaluate_erdos_solution(new_h)
                    if candidate_c5 < best_c5:
                        best_c5 = candidate_c5
                        best_h = new_h.copy()
                    
        # If no good found
        if best_h is None:
            # return uniform here
            best_h = np.ones(n) * (n / 2) / np.sum(np.ones(n))
            best_c5 = evaluate_erdos_solution(best_h)
            
    except Exception as e:
        # fallback to fixed pattern
        try:
            # Check imports to ensure functions available
            if 'initial_h_values' in globals():
                # interpolate initial h
                x_orig = np.linspace(0, 2, len(initial_h_values))
                x_new = np.linspace(0, 2, n)
                h_interp = interp1d(x_orig, initial_h_values, kind='linear', fill_value='extrapolate')
                h_init = h_interp(x_new)
                h_init = h_init * (n / 2) / np.sum(h_init)
                h_init = (h_init + h_init[::-1]) / 2
                best_c5 = evaluate_erdos_solution(h_init)
                best_h = h_init.copy()
            else:
                # Random or uniform start
                h_start = np.random.rand(n)
                h_start = h_start * (n / 2) / np.sum(h_start)
                h_start = (h_start + h_start[::-1]) / 2
                best_c5 = evaluate_erdos_solution(h_start)
                best_h = h_start.copy()
        except:
            best_h = np.ones(n) * 0.5
            best_c5 = evaluate_erdos_solution(best_h)
            
    final_h = best_h.copy()
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h, final_c5, n)
