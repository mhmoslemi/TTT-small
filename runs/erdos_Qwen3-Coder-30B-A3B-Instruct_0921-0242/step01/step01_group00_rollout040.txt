import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def evaluate_c5_discrete(h, n_eval=800):
    """Evaluates the C5 constant for a discrete step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def block_to_step_function(u, v, K, n_points):
    """Convert block parameters to a step function."""
    # Compute lengths using softmax
    u = np.array(u)
    lengths = 2 * np.exp(u) / np.sum(np.exp(u))
    
    # Compute heights
    heights = 1 / (1 + np.exp(-np.array(v)))
    
    # Scale heights to satisfy integral constraint
    area = np.sum(heights * lengths)
    if area > 0:
        scale = 1.0 / area
        heights = np.minimum(1.0, scale * heights)
    
    # Adjust if any height is exactly 1, redistribute
    if np.any(heights >= 1.0):
        # Simple redistribution
        total_height = np.sum(heights * lengths)
        deficit = 1.0 - total_height
        if deficit > 0:
            # Add deficit to lower heights
            for i in range(len(heights)):
                if heights[i] < 1.0:
                    add = min(deficit, 1.0 - heights[i])
                    heights[i] += add
                    deficit -= add
                    if deficit <= 0:
                        break

    # Create step function h
    h = np.zeros(n_points)
    cum_len = 0
    for i in range(K):
        start_idx = int(cum_len / 2 * n_points)
        end_idx = int((cum_len + lengths[i]) / 2 * n_points)
        if end_idx > n_points:
            end_idx = n_points
        h[start_idx:end_idx] = heights[i]
        cum_len += lengths[i]
    return h

def c5_block_objective(uv, K, n_points, n_eval=200):
    """Objective function for block parameters."""
    u = uv[:K]
    v = uv[K:]
    h = block_to_step_function(u, v, K, n_points)
    dx = 2.0 / n_points
    
    # Evaluate correlation with coarse grid
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr
