import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time
import cvxpy as cp
from cvxpy import Minimize, Problem

def upsample_binary_to_fine(binary_vec, n_fine):
    """Upsample a binary vector to fine resolution."""
    n_coarse = len(binary_vec)
    repeat_factor = n_fine // n_coarse
    if repeat_factor * n_coarse < n_fine:
        repeat_factor += 1
    h_fine = np.repeat(binary_vec, repeat_factor)
    h_fine = h_fine[:n_fine]
    return h_fine

def compute_c5(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution
    conv = np.correlate(h, g, mode='full')
    result = np.max(conv) * dx
    return result

def waterfill_projection(h, target_sum):
    """Project h to target_sum with box constraints and sum constraint."""
    h = h.copy()
    current_sum = np.sum(h)
    while abs(current_sum - target_sum) > 1e-12:
        diff = current_sum - target_sum
        # Find indices with 0 < h < 1
        interior_mask = (h > 0) & (h < 1)
        interior_count = np.sum(interior_mask)
        if interior_count == 0:
            break
        delta = diff / interior_count
        h[interior_mask] -= delta
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
    return h

def refine_smoothmax(h0, n_points, beta=50.0, max_iter=100):
    """Use smooth-max gradient descent to refine a discrete function."""
    dx = 2.0 / n_points
    h = h0.copy()
    h = waterfill_projection(h, n_points / 2)
    
    # Precompute the required padding
    def shift_h(h, k):
        if k > 0:
            return np.pad(h[:-k], (k, 0), mode='constant', constant_values=0)
        elif k < 0:
            return np.pad(h[-k:], (0, -k), mode='constant', constant_values=0)
        else:
            return h.copy()
    
    def compute_conv_and_grad(h, k):
        h_shifted = shift_h(h, k)
        conv = np.sum(h * (1 - h_shifted))
        # Gradient of h_i * (1 - h_{i+k}) w.r.t. h_i = (1 - h_{i+k}) - h_{i-k}
        # But we compute it for direct indexing correctness using np.pad
        g = np.zeros_like(h)
        if k > 0:
            g[:-k] = 1 - h_shifted[:-k]
            g[k:] += -h[k:]
        elif k < 0:
            g[-k:] = 1 - h_shifted[-k:]
            g[:-k] += -h[:-k]
        else:
            g = 1 - h_shifted
        return conv, g
    
    for i in range(max_iter):
        if beta < 1:
            bs = beta
        else:
            bs = beta
        # For smooth max target: sum with weights
        conv_values = []
        grad_components = []
        shifts = list(range(-(n_points-1), n_points))
        all_conv_vals = []
        
        for k in shifts:
            conv, g = compute_conv_and_grad(h, k)
            all_conv_vals.append(conv)
        all_conv_vals = np.array(all_conv_vals)
        # smoothing factor
        exp_vals = np.exp(bs * dx * all_conv_vals)
        Z = np.sum(exp_vals)
        weights = exp_vals / Z
        # gradient estimate
        grad = np.zeros_like(h)
        for k, w in zip(shifts, weights):
            conv, g = compute_conv_and_grad(h, k)
            grad += w * g
        grad = dx * grad
        h = h - 0.005 * grad
        # Projection
        h = np.clip(h, 0, 1)
        h = waterfill_projection(h, n_points / 2)
        
        # symmetry
        h = (h + h[::-1]) / 2
        
        # Check stopping condition by C5 value
        if i % 10 == 0:
            c5_refined = compute_c5(h)
            if c5_refined < 0.37999:  # early termination if near target
                break

    return h, compute_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Binary MILP formulation
    n_coarse = 80
    d = n_coarse // 2 + 1  # Free variables (with symmetry and fixed x[0] = 1)
    n_points = 300
    
    # Initialize binary variable with symmetry
    # x[i] = x[n_coarse-1-i] for i = 1..d-1, x[0] = 1 fixed
    x = cp.Variable(d, boolean=True)
    y = cp.Variable((d, n_coarse), boolean=True)
    M = cp.Variable()
    
    # Objective: minimize M
    objective = Minimize(M)
    
    # Constraints:
    # 1. Specified pattern: x[0] = 1, others are free
    # 2. sum{x[i] for i in [1..d-1]] + x[0] = n_coarse / 4 + 0.5 (with symmetry and fixed center)
    # The actual weight in mirror-type system is: 
    # sum_{i=0}^{d-1} x[i] + sum_{i=1}^{d-1} x[i] - 1.0 = n_coarse/2 (but x[0] = 1 counted explicitly)  
    constraints = [x[0] == 1]
    constraints.append(cp.sum(x) == n_coarse / 2)
    
    # 3. For y[i,k], product x[i] * x[i+k] with appropriate shifts
    for i in range(d):
        for k in range(n_coarse):
            if i + k >= d:
                break
            # y[i,k] represents x[i] * x[i+k] in the symmetry-reduced space
            # Asymmetric constraint due to 
            #         i      i+k
            #      ---|---|--|---|----- for symmetrically mirrored case
            #        j     j+k
            # Since we mirror about center: i + (n_coarse - 1 - (i+k)) = n_coarse - i - (k+1) 
            # Only consider valid pairs (i < d, i+k < d)
            if k == 0:
                # Generally y[i,0] <= x[i] and if x[i] == 1 then y[i,0] = x[i]
                constraints.append(y[i, k] <= x[i])  
                constraints.append(y[i, k] <= x[i])
                constraints.append(y[i, k] >= 2*x[i] - 1) 
            else:
                constraints.append(y[i, k] <= x[i])
                constraints.append(y[i, k] <= x[i+k])
                constraints.append(y[i, k] >= x[i] + x[i+k] - 1)
    
    # To simplify variance in the fill-reduce here:
    # Simplify expression to use only possible pairs within read bounds.
    # For each k = 0..n_coarse-1, we slow down a little and include how we'd have more elaborate
   
    # C_k overlap = dx * sum_j x_j x_{j+k} 
    # We break down overlaps over boundary conditions with d = 41 
    # For insight let’s generate each shift separately  
    max_k = n_coarse - 1
    Z_k = [cp.sum(y[i, k]) for k in range(max_k+1) for i in range(min(d, n_coarse - k))]
    
    Z = [[cp.sum([y[i, k] for i in range(d) if k+i < d]) for k in range(max_k+1) ]]
    
    # The z_k must bound overlaps
    # zk = sum_{i=0}^{n_coarse-k-1} x_i x_{i+k}
    # Symmetrized, this is for i = 0 to ⌊n_coarse/2⌋ 
    # then x_i x_{i+k} + x_{n_coarse-1-i} x_{n_coarse-1-i+k}
    for k in range(n_coarse):
        Z_sum = cp.sum([y[i, k] for i in range(min(d, n_coarse - k))])
        # We want: 
        # Z = sum_{i=0}^{n_coarse-k-1} x_i x_{i+k} 
        # which implies  
        # constraint (n_coarse/2 - Z) <= M
        constraints.append(Z_sum >= n_coarse/2 - M)
    
    # Build optimization model
    prob = Problem(objective, constraints)
    
    # Try solving several times
    best_x_sol = None
    try:
        prob.solve(solver=cp.CBC, verbose=False, maxSeconds=30)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Get x contribution
            x_sol = x.value
            # Restore full vector with symmetry and normalization
            h_binary_full = np.zeros(n_coarse)
            if x_sol is not None:
                half = x_sol.T
                # Skip zero for iteration to pad properly
                for i in range(len(half)):
                    h_binary_full[i] = half[i] 
                    h_binary_full[n_coarse - 1 - i] = half[i] 
                # adjust value for x[0] which is fixed at 1 but we have actual binary sol
                h_binary_full[0] = 1.0
                h_binary_full[n_coarse - 1] = 1.0
                # now remove extra copy  
                h_binary_full = h_binary_full[:len(h_binary_full)-1] 
                if len(h_binary_full) < n_coarse:
                     h_binary_full = np.pad(h_binary_full, (0, n_coarse-len(h_binary_full)), 'constant')
                # normalize for exact integration  
                s = np.sum(h_binary_full)
                h_binary_full = h_binary_full * (n_coarse/2)/s
                # make it cohesive
                if len(h_binary_full) != n_coarse:
                    print("unexpected size mismatch")
                # upsample to n_points fine resolution
                h_fine = upsample_binary_to_fine(h_binary_full, n_points)
                # add symmetry enforcement after upsampling
                h_fine = (h_fine + h_fine[::-1]) / 2
                # clip bounds
                h_fine = np.clip(h_fine, 0, 1)
                # normalize for target integral constraint
                h_fine = waterfill_projection(h_fine, n_points/2)
                c5_fine = compute_c5(h_fine)
                if c5_fine <= 0.38:
                    best_h = h_fine
                    best_c5 = c5_fine
                else:
                    # worsening path, continue to smooth refinement
                    best_h, best_c5 = refine_smoothmax(h_fine, n_points, beta=30, max_iter=80)
            else:
                # fallback 
                h_fine = np.ones(n_points) * 0.5
                best_h, best_c5 = refine_smoothmax(h_fine, n_points, beta=50, max_iter=100)
        else:
            # problem failed, proceed with cached solution or fallback
            pass

    except Exception as e:
        # Fallback to default if MILP fails
        pass

    # Fallback to more rigid loops to make it works reliably from default values
    # Final fallback to uniform initialization
    if 'best_h' not in locals() or best_c5 > 0.5:
        # simple assumption: fallback to linear interpolation or small block pattern
        h_linear = np.linspace(0, 1, n_points)
        h_linear = h_linear * (n_points / 2) / np.sum(h_linear)
        best_h, best_c5 = refine_smoothmax(h_linear, n_points, beta=30, max_iter=80)
        
    # Continuous stochastic polishing
    max_time_left = budget_s - (time.time() - start_time) - 5
    if max_time_left > 0.5:
        h_best = best_h.copy()
        c5_best = best_c5
        
        n_perturb = min(6, n_points // 20)
        epsilon = 0.004
        
        for _ in range(2000):
            if time.time() - start_time > budget_s - 0.5:
                break
                
            # Select positions to perturb
            indices = np.random.choice(n_points, n_perturb, replace=False)
            h_trial = h_best.copy()
            for idx in indices:
                h_trial[idx] += np.random.uniform(-epsilon, epsilon)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = waterfill_projection(h_trial, n_points / 2)
            h_trial = (h_trial + h_trial[::-1]) / 2
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best:
                h_best = h_trial.copy()
                c5_best = c5_trial
                
        best_h = h_best
        best_c5 = c5_best
    
    return (best_h, best_c5, n_points)
