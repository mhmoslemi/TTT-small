import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time
from scipy.signal import correlate

def waterfill(h, target_sum):
    """Project h to have exact sum = target_sum, maintaining bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    excess = current_sum - target_sum
    if abs(excess) < 1e-12:
        return h
    
    # Iteratively distribute excess
    while abs(excess) > 1e-12:
        indices = np.where((h > 0) & (h < 1))[0]
        if len(indices) == 0:
            break  # all bounded
        
        delta = excess / len(indices)
        if delta > 0:  # reduce excess
            for i in indices:
                h[i] = max(0, h[i] - delta)
        else:  # increase deficit
            for i in indices:
                h[i] = min(1, h[i] - delta)
            
        current_sum = np.sum(h)
        excess = current_sum - target_sum

    return h

def project_to_feasible(h, target_sum):
    """Project h to fit box constraints [0,1] and sum = target_sum"""
    return waterfill(h, target_sum)

def build_toeplitz_matrices(n):
    """Build Toeplitz matrices R_k for all k"""
    R = []
    for k in range(n):
        R_k = np.zeros((n, n))
        # Fill the k-th off-diagonal
        for i in range(n):
            if i + k < n:
                R_k[i, i + k] = 1
            if i - k >= 0:
                R_k[i, i - k] = 1
        R.append(R_k)
    return R

def sdp_coarse_refinement(h0, coarse_n=120):
    """Coarse SDP-based refinement"""
    dx0 = 2.0 / coarse_n
    target_sum = coarse_n / 2
    
    # Start with h0
    h = h0.copy()
    h = np.clip(h, 0, 1)
    h = project_to_feasible(h, target_sum)
    
    # Build Toeplitz matrices for the coarse grid
    R_mats = build_toeplitz_matrices(coarse_n)
    
    # SDP relaxation
    h_var = cp.Variable(coarse_n)
    X = cp.Variable((coarse_n, coarse_n), PSD=True)
    t = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target_sum,
    ]
    
    for k in range(coarse_n):
        constraints.append(cp.trace(R_mats[k] @ X) >= t)
    
    # Constraint that X = h h^T
    constraints.append(cp.bmat([[1, h_var.T], [h_var, X]]) >> 0)
    
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-5, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h_var.value
            if h_sol is not None:
                # Perform projection
                h_sol = project_to_feasible(h_sol, target_sum)
                
                # Mirror symmetry
                h_sol = (h_sol + h_sol[::-1]) / 2
                return h_sol
    except:
        pass
    
    return h

def smooth_max_gradient(h, beta, dx):
    """Compute smooth max gradient"""
    # Correlation matrix
    g = 1 - h
    # Full correlation 
    conv = correlate(h, g, mode='full')
    # Trim to the proper size (n-1) items for each shift, total = 2*n-1 items
    conv_length = 2 * len(h) - 1
    n = len(h)
    conv_raw = conv
    # Extract the actual autocorrelations by shifting
    conv_shifted = np.zeros(n)
    for k in range(n):
        # Shift k: h[i] * g[i+k], treat zeros outside:
        # Get the valid indices
        if k < n:
            # These are convolution terms h[i] * g[i+k] for i such that i+k < n
            # i from 0 to n-1-k
            conv_shifted[k] = sum(h[i] * g[i + k] for i in range(n - k))
        else:
            # These are convolution terms h[i] * g[i+k], where i+k >= n
            # i from 0 to n-1
            # i from (k-n+1) to n-1  
            conv_shifted[k] = sum(h[i] * (0 if i + k >= n else g[i + k]) for i in range(max(0, k - n + 1), n))

    conv_shifted = conv_raw[n-1:2*n-1] # The right half of full convolution, i.e., correlation shifts k = 0,...,n-1
    conv_shifted = conv_shifted[:n]  # Make sure it's size n  

    # Softmax weights
    beta_s = beta * dx * conv_shifted
    beta_s -= np.max(beta_s)  # Numerical stability (logsumexp trick)
    exp_beta_s = np.exp(beta_s)
    w = exp_beta_s / np.sum(exp_beta_s)

    # Weighted gradients of correlations
    u = correlate(w, 1 - h, mode='full')[n-1:2*n-1]  
    v = correlate(w[::-1], h, mode='full')[n-1:2*n-1]  

    grad = dx * (u - v)
    
    return grad

def refine_with_smooth_max(h, dx, beta_init=0.5, beta_max=50.0, max_iter=80):
    """Refine h with smooth max gradient descent"""
    h = h.copy()
    cur_beta = beta_init
    last_c5 = float('inf')
    
    while cur_beta <= beta_max:
        # Gradient descent with L-BFGS-B
        def obj(h):
            # Compute smooth max approximation
            g = 1 - h
            conv = correlate(h, g, mode='full')
            conv_shifted = conv[len(h)-1:2*len(h)-1]
            conv_vals = conv_shifted[:len(h)]
            beta_s = cur_beta * dx * conv_vals
            beta_s -= np.max(beta_s)
            log_sum_exp = np.log(np.sum(np.exp(beta_s))) 
            return -log_sum_exp / cur_beta
        
        def grad(h):
            return smooth_max_gradient(h, cur_beta, dx)
        
        # Project and invert to minimize -F_beta  
        try:
            # Use L-BFGS-B for gradient-based optimization
            result = minimize(obj, h, method='L-BFGS-B', 
                              jac=grad,
                              bounds=[(0, 1)] * len(h),
                              options={'ftol': 1e-6, 'maxiter': max_iter})
            h = result.x
            h = np.clip(h, 0, 1)
            h = project_to_feasible(h, len(h) / 2)
            h = (h + h[::-1]) / 2 # symmetrize
            
            # Compute exact C5
            c5 = dx * np.max(correlate(h, 1 - h, mode='full')[len(h)-1:2*len(h)-1])
            
            if c5 >= last_c5 * 0.9999: # stagnated
                break
            last_c5 = c5
            
            cur_beta *= 2
        except:
            break
            
    return h

def refine_fine_grid(h_coarse, n_fine=300):
    """Upsample and refine fine grid"""
    dx_fine = 2.0 / n_fine
    # Interpolate
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    
    # Clip and rescale
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = project_to_feasible(h_fine, n_fine / 2)
    h_fine = (h_fine + h_fine[::-1]) / 2  # ensure symmetry
    
    # Fine refinement
    h_fine = refine_with_smooth_max(h_fine, dx_fine, beta_init=0.5, beta_max=50.0, max_iter=50)
    
    return h_fine
