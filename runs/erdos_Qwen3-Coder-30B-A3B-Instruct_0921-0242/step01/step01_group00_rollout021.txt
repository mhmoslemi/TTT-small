import numpy as np
import time
from scipy.signal import correlate
from scipy.fftpack import fft, ifft

def project_to_feasible(h, n):
    """Project h onto the feasible set: h in [0,1] and sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum

    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    n_exceed = 0
    for i in range(len(sorted_h)):
        if sorted_h[i] + diff / (i + 1) > 1:
            break
        n_exceed = i + 1
    if n_exceed > 0:
        # Some elements will exceed 1, clip them and redistribute
        for i in range(n_exceed):
            h[np.argmax(h)] = 1.0  # Set one saturated element to 1
            diff = diff - (1 - np.max(h))  # Update diff
    # Distribute the remaining diff
    if diff > 1e-10:
        h += diff / n
        h = np.clip(h, 0, 1)
    else:
        h = h + diff / n
        h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def evaluate_c5_fft(h):
    """Use FFT for computing correlation - faster."""
    n = len(h)
    dx = 2.0 / n
    # Zero pad to 2*n
    h_padded = np.pad(h, (0, n), 'constant')
    h_conj_padded = np.pad(1 - h, (0, n), 'constant')
    corr = ifft(fft(h_padded) * np.conj(fft(h_conj_padded)))
    max_corr = np.max(corr.real[:n]) * dx
    return max_corr

def compute_worst_shift_gradient(h, k):
    """Compute gradient for the worst shift k."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    # For shift k, compute the derivative of the k'th term h[i]*h[i+k]
    # The partial w.r.t. h[i] for k'th shift is h[i+k] + h[i-k]
    # For the constraint min_k A_k, only consider the worst shift.
    # The gradient of the term A_k = sum(h[i]*h[i+k]) w.r.t. h[i]
    # is h[i+k] + h[i-k] (with mod n index)
    # But for the max of these min terms, we only consider one k.
    
    # Compute it using correct indices
    h_rot_k = np.roll(h, k)
    h_rot_neg_k = np.roll(h, -k)
    grad = dx*(h_rot_neg_k - h_rot_k)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    global start_time
    start_time = time.time()
    np.random.seed(seed)
    
    # Parameters
    n_points = 300
    dx = 2.0 / n_points
    S = 30  # Swarm size
    max_pso_iter = 300
    memetic_every = 5  # Do memetic refinement every 5 PSO generations
    time_limit = budget_s
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest_h = []
    pbest_c5 = []
    gbest_h = None
    gbest_c5 = float('inf')
    
    # Generate initial particles
    for _ in range(S):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = project_to_feasible(h, n_points)
        swarm.append(h)
        velocities.append(np.zeros_like(h))
        c5 = evaluate_c5_fft(h)
        pbest_h.append(h.copy())
        pbest_c5.append(c5)
        
        if c5 < gbest_c5:
            gbest_h = h.copy()
            gbest_c5 = c5
            
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_stagnation = 3  # seconds stagnation before reset
    stagnation_time = 0
    last_improvement = time.time()
    
    # Main PSO loop
    for iter in range(max_pso_iter):
        if time.time() - start_time > time_limit - 0.5:
            break
            
        for i in range(S):
            h = swarm[i]
            
            # Update velocity and position
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = c1 * r1 * (pbest_h[i] - h)
            social = c2 * r2 * (gbest_h - h)
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Update position
            h_new = h + velocities[i]
            
            # Project
            h_projected = project_to_feasible(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_c5_fft(h_projected)
            
            # Update personal best
            if c5_new < pbest_c5[i]:
                pbest_h[i] = h_projected.copy()
                pbest_c5[i] = c5_new
                
                # Update global best
                if c5_new < gbest_c5:
                    gbest_h = h_projected.copy()
                    gbest_c5 = c5_new
                    last_improvement = time.time()
                    stagnation_time = 0
                    
            swarm[i] = h_projected
            
        # Check stagnation
        if time.time() - last_improvement > max_stagnation:
            # Respawn part of swarm
            num_to_respawn = max(1, S // 4)
            for _ in range(num_to_respawn):
                idx = np.random.randint(S)
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                h = project_to_feasible(h, n_points)
                swarm[idx] = h
                velocities[idx] = np.zeros_like(h)
                pbest_h[idx] = h.copy()
                pbest_c5[idx] = evaluate_c5_fft(h)
                
                if pbest_c5[idx] < gbest_c5:
                    gbest_h = h.copy()
                    gbest_c5 = pbest_c5[idx]
                    last_improvement = time.time()
        
        # Memetic refinement every few iterations
        if iter % memetic_every == 0 and iter > 0:
            if time.time() - start_time > time_limit - 0.5:
                break
            for i in range(S):
                # Find the actual worst shift for current h
                h = swarm[i]
                corr = correlate(h, 1-h, mode='full')
                # Find the index of max correlation (excluding ghost entries)
                max_idx = np.argmax(corr)
                shift_k = max_idx - (n_points - 1)  # Normalize to shift in [-n+1, n-1]
                if shift_k == 0:
                    shift_k = np.argmax(corr[1:]) + 1  # Just pick a nontrivial shift
                # Compute the gradient for this shift (w.r.t. this max)
                grad = compute_worst_shift_gradient(h, shift_k)
                
                # Backtracking line search
                alpha = 1.0
                max_backtrack = 10
                old_c5 = evaluate_c5_fft(h)
                h_trial = h.copy()
                for _ in range(max_backtrack):
                    h_trial = h - alpha * grad
                    h_trial = project_to_feasible(h_trial, n_points)
                    trial_c5 = evaluate_c5_fft(h_trial)
                    
                    if trial_c5 < old_c5:
                        swarm[i] = h_trial
                        break
                    else:
                        alpha /= 2.0
                        if alpha < 1e-6:
                            break
                        
            # Update best after memetic updates
            for i in range(S):
                c5_new = evaluate_c5_fft(swarm[i])
                if c5_new < pbest_c5[i]:
                    pbest_h[i] = swarm[i].copy()
                    pbest_c5[i] = c5_new
                    
                    if c5_new < gbest_c5:
                        gbest_h = swarm[i].copy()
                        gbest_c5 = c5_new
                        
    # Late-stage polishing with random swap hill climb
    if time.time() - start_time < time_limit - 0.2:
        # Focus on global best
        h_best = gbest_h.copy()
        c5_best = gbest_c5
        
        for _ in range(5000):
            if time.time() - start_time > time_limit - 0.1:
                break
                
            # Random swap
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            
            # Only allow swapping between feasible halves
            if h_best[i] <= 1e-6 or h_best[j] >= 1 - 1e-6:
                continue

            delta = np.random.uniform(0.001, 0.01)
            if delta > h_best[i]:
                delta = h_best[i]
            if delta > (1 - h_best[j]):
                delta = 1 - h_best[j]
                
            h_new = h_best.copy()
            h_new[i] -= delta 
            h_new[j] += delta
            
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_c5_fft(h_new)
            
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
                
    # Final evaluation
    c5_final = evaluate_c5_fft(h_best)
    
    return (h_best, c5_final, n_points)
