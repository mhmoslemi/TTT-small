import numpy as np
from scipy.optimize import minimize
import time
from scipy.interpolate import interp1d
import cvxpy as cp
from cvxpy import SCS
from scipy.linalg import eigh
from scipy.signal import correlate

def compute_c5(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    result = np.max(conv) * dx
    return result

def project_sum(h, target_sum):
    """Project h onto the constraint that sum equals target_sum using water-filling"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-12:
        return h
    
    # If diff > 0, we need to add to elements that are < 1
    # If diff < 0, we need to subtract from elements that are > 0
    I = np.where((h > 0) & (h < 1))[0]
    
    while abs(diff) > 1e-12 and len(I) > 0:
        # Distribute diff among the set I
        delta = diff / len(I)
        for i in I:
            if diff > 0:
                h[i] = min(1.0, h[i] + delta)
            else:
                h[i] = max(0.0, h[i] + delta)
        # Recompute
        diff = target_sum - np.sum(h)
        I = np.where((h > 0) & (h < 1))[0]
    
    return h

def smooth_max_gradient(h, beta, dx):
    """Compute gradient of smooth max surrogate"""
    g = 1.0 - h
    convs = np.correlate(h, g, mode='full')
    # Add small epsilon to avoid numerical issues
    exp_convs = np.exp(beta * dx * convs)
    weights = exp_convs / np.sum(exp_convs)
    
    u = np.correlate(weights, g, mode='full')
    v = np.correlate(weights[::-1], h, mode='full')[::-1]
    
    grad = dx * (u - v)
    return grad

def project_box_sum(h):
    """Project h onto box and sum constraints"""
    h = np.clip(h, 0, 1)
    h = project_sum(h, len(h) / 2)
    return h

    start_time = time.time()
    
    # Budget for the different stages
    n_points = 300
    dx = 2.0 / n_points
    
    # First try the SDP approach
    n_coarse = 120
    dx_coarse = 2.0 / n_coarse
    target_sum_coarse = n_coarse / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # SDP solver parameters
    secs_for_sdp = 90
    sdp_start = time.time()
    
    # Coarse SDP solver attempt
    try:
        # Create Toeplitz matrices for autocorrelations
        R = []
        for k in range(n_coarse):
            R_k = np.zeros((n_coarse, n_coarse))
            for i in range(n_coarse):
                j = i + k
                if j < n_coarse:
                    R_k[i, j] = 1.0
                j = i - k
                if j >= 0:
                    R_k[i, j] = 1.0
            R.append(R_k)
        
        # CVXPY model
        h = cp.Variable(n_coarse)
        X = cp.Variable((n_coarse, n_coarse), symmetric=True)
        t = cp.Variable()
        
        # Constraints
        constraints = []
        
        # PSD constraint
        constraints.append(cp.bmat([[1, h.T], [h, X]]) >> 0)
        
        # Autocorrelation lower bounds
        for i in range(n_coarse):
            constraints.append(cp.trace(R[i] @ X) >= t)
        
        # Box constraints
        constraints.extend([h[i] >= 0 for i in range(n_coarse)])
        constraints.extend([h[i] <= 1 for i in range(n_coarse)])
        
        # Sum constraint
        constraints.append(cp.sum(h) == target_sum_coarse)
        
        # Objective
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Run with timeout
        prob.solve(solver=SCS, max_iters=5000, eps=1e-5)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = h.value
            X_sol = X.value
            
            # Extract solution by eigenvector method
            _, v = eigh(X_sol)
            v1 = v[:, -1]
            h_tilde = np.abs(v1) * np.sign(h_sol[0])
            h_tilde = np.abs(h_tilde)
            
            # Project onto constraints
            h_proj = project_box_sum(h_tilde)
            if not np.any(np.isnan(h_proj)):
                h_opt = h_proj.copy()
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
    except Exception as e:
        pass
    
    # If SDP doesn't help much, try refinement from initial start
    if best_h is None:
        if 'initial_h_values' in globals():
            initial = initial_h_values
            h_interp = interp1d(np.linspace(0, 2, len(initial)), initial, kind='linear', fill_value='extrapolate')
            h_initial = h_interp(np.linspace(0, 2, n_coarse))
            h_initial = h_initial * (n_coarse / 2) / np.sum(h_initial)
            h_initial = project_box_sum(h_initial)
            best_h = h_initial
            best_c5 = compute_c5(best_h)
        else:
            h_initial = np.ones(n_coarse) * 0.5
            best_h = h_initial
            best_c5 = compute_c5(best_h)
    
    # Refinement using smooth max
    h_refined = best_h.copy()
    decay_factor = 0.5
    min_beta = 0.5
    max_beta = 50.0
    β = min_beta
    iteration = 0
    local_best_c5 = best_c5
    
    refinement_start = time.time()
    
    while time.time() - refinement_start < 120 and β <= max_beta:
        limit = 80 if β < 10 else 20
        for _ in range(min(limit, 200)):
            if time.time() - start_time > budget_s - 5:
                break
            grad = smooth_max_gradient(h_refined, β, dx_coarse)
            h_new = h_refined - 0.01 * grad
            h_new = project_box_sum(h_new)
            h_new = (h_new + h_new[::-1]) / 2
            h_refined = h_new
            
        # Check if we made an improvement
        integrated_c5 = compute_c5(h_refined)
        if integrated_c5 < local_best_c5:
            local_best_c5 = integrated_c5
            
        β *= 2.0
        if β > max_beta:
            break
                
    # Upsample to target resolution
    if time.time() - start_time > budget_s - 10:
        pass
    else:
        h_fine_upsample = interp1d(np.linspace(0, 2, n_coarse), h_refined, kind='linear', fill_value='extrapolate')
        h_fine = h_fine_upsample(np.linspace(0, 2, n_points))
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        h_fine = project_box_sum(h_fine)
        h_fine = (h_fine + h_fine[::-1]) / 2
        c5_fine = compute_c5(h_fine)
        if c5_fine < local_best_c5:
            local_best_c5 = c5_fine
            best_h = h_fine.copy()

        h_fine_refined = h_fine.copy()
        β = min_beta
        while time.time() - start_time < budget_s - 10 and β <= max_beta:
            limit = 20
            for _ in range(limit):
                if time.time() - start_time > budget_s - 5:
                    break
                grad = smooth_max_gradient(h_fine_refined, β, dx)
                h_new = h_fine_refined - 0.01 * grad
                h_new = project_box_sum(h_new)
                h_new = (h_new + h_new[::-1]) / 2
                h_fine_refined = h_new
                
            c5_fine = compute_c5(h_fine_refined)
            if c5_fine < local_best_c5:
                local_best_c5 = c5_fine
                best_h = h_fine_refined.copy()
            
            β *= 2.0
            if β > max_beta:
                break
        
        # Final polishing
        eps = 0.004
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            idxs = np.random.choice(n_points, min(6, n_points//20), replace=False)
            h_trial = best_h.copy()
            for i in idxs:
                pert = np.random.uniform(-eps, eps)
                h_trial[i] = min(1.0, max(0.0, h_trial[i] + pert))
            h_trial = project_box_sum(h_trial)
            h_trial = (h_trial + h_trial[::-1]) / 2
            c5_trial = compute_c5(h_trial)
            if c5_trial < local_best_c5:
                local_best_c5 = c5_trial
                best_h = h_trial.copy()
    
    return (best_h, local_best_c5, n_points)
