import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *
import cvxpy as cp

def project_to_simplex_and_box(h, n):
    """Project h onto {0 <= h_i <= 1, sum h_i = n/2} using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Add constant to make sum = n/2
    if diff > 0:
        # Need to increase values
        h += diff / n
        # Re-clip
        h = np.clip(h, 0., 1.)
    else:
        # Need to decrease values
        h += diff / n
        # Re-clip
        h = np.clip(h, 0., 1.)
    
    return h

def c5_of_h(h):
    """Computes C5 using discrete correlation."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def worst_shift(h):
    """Returns shift index achieving max overlap."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # Adjust back to shift indices
    n = len(h)
    shift = max_idx - (n - 1)
    return shift

def compute_gradient(h, active_shift_idx):
    """Compute gradient for active shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    for i in range(n):
        # Derivative of int h(x)(1-h(x+k)) dx w.r.t. h_i:
        # dx * (1 - h_{i+shift}) - dx * h_{i-shift}
        left_idx = (i + active_shift_idx) % n
        right_idx = (i - active_shift_idx) % n
        grad[i] = dx * (1 - h[left_idx]) - dx * h[right_idx]
    return grad

def projected_gradient_step(h, shift, alpha=0.01):
    """Perform a projected gradient step."""
    grad = compute_gradient(h, shift)
    h_new = h - alpha * grad
    h_new = project_to_simplex_and_box(h_new, len(h))
    return h_new

def solve_lp_relaxation(n):
    """Solve LP relaxation to get a good starting point."""
    # Bilinear terms z_{i,k} = h_i * h_{i+k}
    h = cp.Variable(n)
    y = cp.Variable(n-1)  # For each shift k=1 to n-1
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    
    # McCormick envelopes for z_{i,k} = h_i * h_{i+k}
    # z >= h_i + h_{i+k} - 1
    # z <= h_i
    # z <= h_{i+k}
    # z >= 0
    
    # We will directly use values y_k = sum_i h_i h_{i+k}
    # Build constraints: y[k-1] >= sum_i h_i * h_{i+k}
    
    for k_idx in range(n - 1):
        k = k_idx + 1
        expr = 0
        for i in range(n):
            j = (i + k) % n
            expr += h[i] * h[j]
        constraints.append(y[k_idx] >= expr)  # Approximation
        
    # Objective: minimize max of y_k, equivalent to minimize t s.t. y_k >= t
    t = cp.Variable(1)
    constraints.append(y >= t)
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=CPLEX, verbose=False)
    
    if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        return h.value
    else:
        return np.ones(n) * 0.5  # fallback

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 400
    dx = 2.0 / n_points
    
    # LP relaxation for good starting point
    h_lp = solve_lp_relaxation(n_points)
    h_lp = project_to_simplex_and_box(h_lp, n_points)
    
    h_best = h_lp.copy()
    best_c5 = c5_of_h(h_best)
    
    # Refinement using projected gradient descent on worst shift
    while time.time() - start_time < budget_s - 20:
        shift = worst_shift(h_best)
        h_new = projected_gradient_step(h_best, shift, alpha=0.02)
        c5_new = c5_of_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h_new.copy()
        else:
            break
    
    # Simulated annealing
    current_h = h_best.copy()
    current_c5 = best_c5
    
    temp = 0.1
    cooling_rate = 0.999
    max_iter = 1000
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        # Randomly shuffle indices
        indices = np.random.permutation(n_points)
        # Pick two indices
        i, j = indices[0], indices[1]
        # Try to change small amount delta
        delta = np.random.uniform(-0.02, 0.02)
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        # Clamp
        new_h = np.clip(new_h, 0, 1)
        # Renormalize to sum = n/2
        new_h = project_to_simplex_and_box(new_h, n_points)
        new_c5 = c5_of_h(new_h)
        
        # Accept
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
        elif np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
            current_h = new_h
            current_c5 = new_c5
            
        temp *= cooling_rate
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            h_best = current_h.copy()
    
    # Final refinement
    while time.time() - start_time < budget_s - 2:
        shift = worst_shift(h_best)
        h_new = projected_gradient_step(h_best, shift, alpha=0.01)
        c5_new = c5_of_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h_new.copy()
        else:
            break
    
    return (h_best, best_c5, n_points)
