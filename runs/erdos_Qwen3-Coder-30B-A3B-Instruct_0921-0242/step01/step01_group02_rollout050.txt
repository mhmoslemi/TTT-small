import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import math

def c5_fft(h):
    """Compute C5 using FFT for maximum efficiency."""
    g = 1 - h
    H = rfft(h)
    G = rfft(g)
    c = irfft(H * G, 2*len(h) - 1)
    return np.max(c)

def repair_weight(h, target_sum):
    """Repair a binary vector to have exactly target_sum ones."""
    s = np.sum(h)
    if s == target_sum:
        return h
    diff = int(s - target_sum)
    if diff > 0:
        # Need to flip some ones to zeros
        idx_ones = np.where(h == 1)[0]
        flip = np.random.choice(idx_ones, diff, replace=False)
        h[flip] = 0
    elif diff < 0:
        # Need to flip some zeros to ones
        idx_zeros = np.where(h == 0)[0]
        flip = np.random.choice(idx_zeros, -diff, replace=False)
        h[flip] = 1
    return h

def crossover(p1, p2):
    """Two-point crossover."""
    n = len(p1)
    c1, c2 = sorted(np.random.choice(n, 2, replace=False))
    child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
    return child

def mutate_swap(h):
    """Swap mutation: pick one 1 and one 0, swap them."""
    idx_ones = np.where(h == 1)[0]
    idx_zeros = np.where(h == 0)[0]
    if len(idx_ones) == 0 or len(idx_zeros) == 0:
        return h
    i1 = np.random.choice(idx_ones)
    i0 = np.random.choice(idx_zeros)
    h[i1], h[i0] = h[i0], h[i1]
    return h

def hill_climb(h):
    """Hill-climb optimized for the worst shift."""
    n = len(h)
    max_iter = 500
    for _ in range(max_iter):
        g = 1 - h
        corr = correlate(h, g, mode='full')
        k_star = np.argmax(corr) - n + 1
        best_delta = 0
        best_swap = None
        tried = 0
        while tried < 200 and best_delta >= 0:
            i1 = np.random.randint(0, n)
            if h[i1] == 0:
                continue
            i0 = np.random.randint(0, n)
            if h[i0] == 1:
                continue
            delta = 0
            if i1 + k_star >= 0 and i1 + k_star < n:
                delta -= h[i1 + k_star]
            if i0 + k_star >= 0 and i0 + k_star < n:
                delta += g[i0 + k_star]
            # We compute the change in the correlation value (which is just dx * delta)
            delta = delta * 0.00714  # dx = 2/n = 0.00714
            tried += 1
            if delta < best_delta:
                best_delta = delta
                best_swap = (i1, i0)
        if best_swap is None:
            break
        i1, i0 = best_swap
        h[i1], h[i0] = h[i0], h[i1]
    return h

            best_c5 = c5_val

                best_h = local_h.copy()

            break

    # Multi-scale refinement phase
    if time.time() - start_time < budget_s - 10:
        # Upsample to finer grid
        n_fine = 500
        dx_fine = 2.0 / n_fine
        h_upsampled = np.zeros(n_fine)
        scale = n_fine // n
        for i in range(n):
            start_idx = i * scale
            end_idx = min((i + 1) * scale, n_fine)
            h_upsampled[start_idx:end_idx] = best_h[i]

        h_upsampled = repair_weight(h_upsampled, target_sum)

        c5_fine = c5_fft(h_upsampled)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_upsampled
            
    # Final hill climb on the best solution
    final_h = best_h.copy()
    final_h = hill_climb(final_h)
    final_c5 = c5_fft(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h

    # Final evaluation with high precision
    g = 1 - best_h
    corr = correlate(best_h, g, mode='full')
    best_c5 = np.max(corr) * dx    
    
    return (best_h, best_c5, len(best_h))
