import numpy as np

from scipy.linalg import eigh
from scipy.sparse import csr_matrix

def build_shift_matrices(N):
    """Build sparse shift matrices A_k for k=0,...,N-1"""
    A_k_list = []
    for k in range(N):
        # Create sparse matrix for shift k
        rows = np.arange(N - k)
        cols = np.arange(k, N)
        data = np.ones(N - k)
        A_k = csr_matrix((data, (rows, cols)), shape=(N, N))
        A_k_list.append(A_k)
    return A_k_list

def project_to_feasible(h, N):
    """Ensure sum(h) = N/2, 0 <= h[i] <= 1 for all i"""
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = N/2 - sum_h
    if abs(deficit) < 1e-10:
        return h

    # Distribute deficit uniformly
    adjustment = deficit / N
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def eval_corr_exact(h):
    """Evaluate C5 with exact correlation calculation."""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def eigen_round(X, N):
    """Principal eigenvector rounding for SDP solution"""
    # Get leading eigenpair
    try:
        eigvals, eigvecs = eigh(X.toarray(), eigvals=(N-1, N-1))
        v = eigvecs[:, -1]
        lam = eigvals[-1]
        if lam < 0:
            lam = 0
        h_eig = np.sqrt(lam) * np.abs(v)
        h_eig = project_to_feasible(h_eig, N)
    except:
        h_eig = np.ones(N) * 0.5  # fallback
    return h_eig

def diag_round(X, N):
    """Diagonal thresholding rounding"""
    diag = X.diagonal()
    # Sort indices by diagonal entries descending
    idxs = np.argsort(diag)[::-1]
    h_diag = np.zeros(N)
    h_diag[idxs[:N//2]] = 1.0
    return h_diag

def compute_corr_incrementally(h, corr_full, p, q, A_k_list):
    """Update correlation values after a swap of p and q"""
    corr_new = corr_full.copy()
    # All shifts k such that p+k or q+k are valid indices:
    for k in range(len(A_k_list)):
        # For each shift k, if p+k < N and q+k < N:
        if p+k < len(h) and q+k < len(h):
            # Update contribution from pairs (p, p+k) and (q, q+k)
            corr_new[k] -= h[p] * h[p+k] + h[q] * h[q+k]
            corr_new[k] += h[q] * h[p+k] + h[p] * h[q+k]
        # When q+k < len(h) and p+k >= len(h), the elements in first list of shift are not updated correctly. 
        # But with the actual finite-length shifts, q+k is valid, only if p+k is also valid.
        # we don't make this distinction in order to prevent error cascade.
        # Better we compute all actual shifts(subset) for update, as below:
        # Use a more precise scan of updates needed:
    return corr_new

def apply_swap(h, corr_full, p, q, A_k_list):
    """Apply a single swap (p, q) to h, return new h and updated correlations"""
    h_new = h.copy()
    h_new[p], h_new[q] = h_new[q], h_new[p]
    corr_new = compute_corr_incrementally(h_new, corr_full, p, q, A_k_list)
    return h_new, corr_new

def stochastic_swap_improve(h, corr_full, A_k_list, max_iter=10000):
    """Improve binary h with stochastic swap hill climber"""
    # Ensure h is binary 0/1 (rounding to nearest)
    h = np.round(h).astype(int)
    
    best_h = h.copy()
    best_c = eval_corr_exact(best_h)
    corr = corr_full.copy()
    
    # Precompute full correlation so we're not constantly re-computing
    # We only do a full re-eval at fixed intervals to check global improvement
    n_swaps = 0
    full_eval_count = 0
    for _ in range(max_iter):
        if n_swaps % 500 == 0:
            # Full recompute
            corr = np.array([np.sum(h * (A_k @ h)) for A_k in A_k_list])
            # Compute C5 from the smallest autocorrelation
            c = 2.0/len(h) * (len(h)/2 - np.min(corr))
            if c < best_c:
                best_c = c
                best_h = h.copy()
        
        # Random swap
        p = np.random.randint(len(h))
        q = np.random.randint(len(h))
        while h[p] == h[q]:
            p = np.random.randint(len(h))
            q = np.random.randint(len(h))
        
        # Apply swap
        h_new, corr_new = apply_swap(h, corr, p, q, A_k_list)
        corr_new = np.array([np.sum(h_new * (A_k @ h_new)) for A_k in A_k_list])
        c_new = 2.0/len(h) * (len(h)/2 - np.min(corr_new))
        
        # Accept if better
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            corr = corr_new
            full_eval_count = 0
        else:
            full_eval_count += 1
            if full_eval_count > 20000:  # every 20k steps, just enforce a full recomputation
                full_eval_count = 0
                corr = np.array([np.sum(h * (A_k @ h)) for A_k in A_k_list])
                # Recomputed C5
                c = 2.0/len(h) * (len(h)/2 - np.min(corr))
                if c < best_c:
                    best_c = c
                    best_h = h.copy()

        n_swaps += 1
        if best_c < 0.3795:  # early exit early
            break
    return best_h, best_c

def solve_sdp(N, A_k_list, max_iter=15000):
    """Solve SDP to maximize minimum correlation"""
    # Variables
    h = cp.Variable(N)
    X = cp.Variable((N, N), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == h,  # diagonal entries
        cp.sum(h) == N/2,  # mass constraint
        0 <= h, h <= 1  # bounds
    ]

    # Tr(A_k X) >= t for all k
    for A_k in A_k_list:
        constraints.append(cp.trace(A_k @ X) >= t)

    # Objective
    objective = cp.Maximize(t)

    # Problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=max_iter, eps=1e-4)
        return h.value, X.value
    except Exception as e:
        return None, None

def refine_with_logsumexp(h, A_k_list, max_iter=30, dx=2.0/len(h)):

    h_smooth = h.copy()
    
    # A helper to compute the smooth approximation of max
    def logsumexp_max(arr, temp=1e-4):
        # avoid numerical overflow
        m = np.max(arr)
        exps = np.exp((arr - m) / temp)
        return m + temp * np.log(np.sum(exps)) 

    def infinite_sharpness(x):
        # This is an approximation to the true max operator
        # We won't use this in full form due to gradient complexity
        # But a simplified smooth estimator can help in fine-tuning
        return logsumexp_max
        
    # We'll just do gradient descent on a log-sum-exp smooth version of that max
    # to get coarser structure refined smoothly
    
    # Simple coordinate descent over each h_i (still approximate)
    # Compute default max_k by full matrix
    try:
        corr_values = []
        for A_k in A_k_list:
            corr_values.append(np.sum(h_smooth * (A_k @ h_smooth)))
        max_corr = max(corr_values)
        c_current = dx * (N/2 - max_corr)
    except:
        return h_smooth
    
    for _ in range(max_iter):
        # We don't trust gradient of max, but here we dampen values
        # clamping to [0,1], but we might move it a bit toward interior
        clipped = np.clip(h_smooth, 0.01, 0.99)
        
        # try to rebalance around median 
        exp_val = np.exp((0.5 - clipped) * 20)
        h_smooth = clipped + (0.5 - clipped) * 0.2
        
        # Ensure constraint:
        h_smooth = project_to_feasible(h_smooth, N)
        
        # Re-eval after iteration
        corr_values = []
        for A_k in A_k_list:
            corr_values.append(np.sum(h_smooth * (A_k @ h_smooth)))
        max_corr = max(corr_values)
        c_current = dx * (N/2 - max_corr)
        
        if c_current < 0.379:
            break
            
    return h_smooth
