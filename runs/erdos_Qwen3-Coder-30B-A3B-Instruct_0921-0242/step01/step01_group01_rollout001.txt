import numpy as np

def project_to_feasible(u, n):
    """Project vector u to feasible set: 0 <= h <= 1, sum(h) = n/2"""
    u = np.clip(u, 0.0, 1.0)
    s = np.sum(u)
    if np.isclose(s, n/2):
        return u
    
    # Use bisection to find lambda such that sum(clip(u + lambda, 0, 1)) = n/2
    def sum_with_lambda(lam):
        return np.sum(np.clip(u + lam, 0.0, 1.0))
    
    # Binary search for correct lambda
    low, high = -1.0, 1.0
    for _ in range(30):
        mid = (low + high) / 2
        val = sum_with_lambda(mid)
        if val < n/2:
            low = mid
        else:
            high = mid
    
    lam = (low + high) / 2
    h = np.clip(u + lam, 0.0, 1.0)
    return h

def evaluate_c5_fast(h):
    """Fast evaluation using FFT for convolution"""
    n = len(h)
    H = rfft(h, n=2*n-1)
    G = rfft(1-h, n=2*n-1)
    conv = irfft(H*G)[:2*n-1]
    dx = 2.0 / n
    return dx * np.max(conv)

def symmetrize_and_project(h, n):
    """Symmetrize h and make it feasible"""
    # If odd, leave central element as is
    d = (n + 1) // 2
    if n % 2 == 0:
        h_sym = np.zeros(n)
        h_sym[:d] = h[:d]
        h_sym[d:] = h[:d-1][::-1]
    else:
        h_sym = np.zeros(n)
        h_sym[:d] = h[:d]
        h_sym[d:] = h[:d-1][::-1]
    return project_to_feasible(h_sym, n)

def cma_es_global_search(n_coarse, n_fine, budget_s):
    """CMA-ES global search on coarse grid"""
    np.random.seed(42)
    start_time = time.time()
    
    d = (n_coarse + 1) // 2  # dimension of free variables
    mu = 0.5 * np.ones(d)
    sigma = 0.2
    lambda_ = 4 + int(3 * np.log(d))
    max_iter = 200
    
    best_c5 = float('inf')
    best_h = None
    
    # Initialize
    pop = []
    fitnesses = []

    # Generate initial population
    for _ in range(lambda_):
        z = np.random.normal(0, 1, d)
        h = mu + sigma * z
        h = symmetrize_and_project(h, n_coarse)
        c5 = evaluate_c5_fast(h)
        pop.append(h)
        fitnesses.append(c5)
        
    # Sort population
    idxs = np.argsort(fitnesses)
    pop = [pop[i] for i in idxs]
    fitnesses = [fitnesses[i] for i in idxs]
    
    # CMA-ES main loop
    no_improvement_count = 0
    restart_count = 0
    restart_threshold = 150
    
    for gen in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get surviving parents
        parents = pop[:lambda_ // 2]
        parent_fitnesses = fitnesses[:lambda_ // 2]
        
        # Update mean and covariance
        # Simple update for now
        # This could be made more robust with full CMA-ES
        mu = np.mean(parents, axis=0)
        
        # Sample offspring
        offspring = []
        for _ in range(lambda_):
            z = np.random.normal(0, 1, d)
            h = mu + sigma * z
            h = symmetrize_and_project(h, n_coarse)
            offspring.append(h)
            
        # Evaluate offspring
        fitness_offspring = []
        for h in offspring:
            # early stop if we detect a too large value
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                # Use a nice early stop trick to estimate when it's likely still good
                pass
            fitness_offspring.append(c5)
            
        # Combine and select
        pop = parents + offspring
        fitnesses = parent_fitnesses + fitness_offspring
        idxs = np.argsort(fitnesses)
        pop = [pop[i] for i in idxs]
        fitnesses = [fitnesses[i] for i in idxs]
        
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = pop[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive restart
        if no_improvement_count > restart_threshold:
            if restart_count < 3:
                mu = best_h[:d].copy()
                sigma *= 0.5
                restart_count += 1
                no_improvement_count = 0
            else:
                break
    
    return best_h, best_c5
