import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import cvxpy as cp
from cvxpy import SCS

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorrelations(h):
    """Compute all autocorrelation terms I_k = dx * (1 - sum(h_i * h_{i+k}))."""
    dx = 2.0 / len(h)
    n = len(h)
    autocorr = []
    for k in range(1, n):
        # Compute sum of h_i * h_{i+k} for valid indices
        sum_h_prod = 0.0
        for i in range(n - k):
            sum_h_prod += h[i] * h[i + k]
        I_k = dx * (1 - sum_h_prod)
        autocorr.append(I_k)
    return np.array(autocorr)

def solve_sdp(n_points, tolerance=1e-4):
    """Solve the SDP relaxation to find a fractional h that maximizes minimum autocorrelation."""
    dx = 2.0 / n_points
    # Variables
    h = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.trace(X) == n_points / 2,  # Sum constraint
        h >= 0, h <= 1,               # Box constraints
        X >> 0,                       # Positive semidefinite
    ]

    # Add X_{ii} = h_i
    for i in range(n_points):
        constraints.append(X[i, i] == h[i])

    # Add autocorrelation constraints for all shifts k = 1, ..., n-1
    T_k = np.zeros((n_points, n_points))
    for k in range(1, n_points):
        # Create the shift matrix T_k for shift k
        for i in range(n_points - k):
            T_k[i, i + k] = 1
        # Trace(T_k * X) = sum_{i=0}^{n-k-1} X[i, i+k] = sum of h_i * h_{i+k}
        constraints.append(cp.trace(T_k @ X) >= t)
        # Reset T_k for next k
        T_k.fill(0)

    # Objective
    objective = cp.Maximize(t)

    # Solve
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=SCS, verbose=False, eps=tolerance, max_iters=5000)

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        # Extract h_sdp
        h_sdp = h.value
        # Adjust sum if needed
        current_sum = np.sum(h_sdp)
        diff = n_points / 2 - current_sum

        if abs(diff) > 1e-8:
            # Water-filling: add same value to each component
            h_sdp = h_sdp + diff / n_points
            # Clip
            h_sdp = np.clip(h_sdp, 0, 1)
            # Ensure we got there by checking
            assert abs(np.sum(h_sdp) - n_points / 2) < 1e-5
        return h_sdp
    else:
        # return a feasible solution if SDP fails
        return np.ones(n_points) * 0.5

def random_round_sdp(h_sdp, n_points, n_rounds=30):
    """Randomized rounding of fractional SDP solution."""
    # Slightly perturb to avoid exact ties
    h_sdp = h_sdp + np.random.normal(0, 1e-8, n_points)
    
    # One method: threshold at 0.5 and adjust
    h_binary = (h_sdp >= 0.5).astype(float)
    current_sum = np.sum(h_binary)
    diff = n_points / 2 - current_sum
    if abs(diff) > 1e-8:
        # Simple adjustment: set some 0s to 1s or vice versa
        if diff > 0:
            # We need more 1s
            indices_0 = np.where(h_binary == 0)[0]
            if len(indices_0) >= diff:
                indices_to_flip = np.random.choice(indices_0, size=int(diff), replace=False)
                h_binary[indices_to_flip] = 1.0
        else:
            # We need more 0s
            indices_1 = np.where(h_binary == 1)[0]
            if len(indices_1) >= -diff:
                indices_to_flip = np.random.choice(indices_1, size=int(-diff), replace=False)
                h_binary[indices_to_flip] = 0.0
    
    return h_binary

def project_sum(h, n_points, target_sum):
    """Ensure sum(h) = target_sum with 0 <= h[i] <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-8:
        return h
    
    # Simple uniform adjustment
    h = h + diff / n_points
    h = np.clip(h, 0, 1)
    return h

    # It's a weighted average of ∇I_k: w_k * ∇I_k

def projected_gradient_step(h, step_size, beta=0.5, max_iter=50):
    """Perform projected gradient descent for smooth max objective."""
    n_points = len(h)
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    for _ in range(max_iter):
        grad = smooth_max_gradient(h, beta)
        
        # Try the step with backtracking
        h_new = h - step_size * grad
        h_new = project_sum(h_new, n_points, target_sum)
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate using true C5 for termination check
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        
        if c5_new < c5_old:
            h = h_new
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                break
        # Increase β
        beta = beta*1.1
    return h, beta
