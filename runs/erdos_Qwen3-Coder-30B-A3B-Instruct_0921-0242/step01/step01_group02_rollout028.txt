import cvxpy as cp
from cvxpy import SCS

def build_shift_matrices(n, max_k):
    """Build sparse shift matrices S_k."""
    S_k_list = []
    for k in range(-max_k, max_k + 1):
        # Create shift matrix
        offsets = np.arange(n)
        cols = (offsets + k) % n
        rows = offsets
        data = np.ones(n)
        S_k = diags(data, k, shape=(n, n), format='csr')
        S_k_list.append(S_k)
    return S_k_list

def solve_sdp(n, shift_matrices, symmetry_reduced=True):
    """Solve the SDP relaxation for binary sequence optimization."""
    # Determine dimension
    m = n // 2 + 1 if symmetry_reduced else n
    
    # Convex variable
    X = cp.Variable((m, m), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [X >> 0,  # Positive semidefinite
                   cp.diag(X) == 1,  # Diagonal ones
                   cp.sum(X @ np.ones(m)) == 0]  # Zero sum constraint
    
    # Add correlation constraints
    for S_k in shift_matrices:
        constraints.append(t >= n / 4 - 0.25 * cp.trace(S_k @ X))
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    prob.solve(solver=SCS, eps=1e-6, max_iters=5000)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        print(f"Warning: SDP solver status: {prob.status}")
        return None, None
    
    return X.value, t.value

def round_binary(X, n, symmetry_reduced=True):
    """Round PSD solution to binary vector."""
    m = X.shape[0]
    if symmetry_reduced:
        # Use Cholesky-like factorization (eigen-decomposition)
        U, V = np.linalg.eigh(X)
        U = np.maximum(U, 0.0)  # Ensure non-negative
        V = V[:, U > 1e-4]
        U = np.sqrt(U[U > 1e-4])
        V = V @ np.diag(U)
        
        # Generate samples
        best_h = None
        best_c5 = np.inf
        for _ in range(200):  # Sample pool
            g = np.random.randn(m)
            y = V.T @ g
            x = np.sign(y)
            # Fix zero-sum
            s = np.sum(x)
            if s != 0:
                if s > 0:  # Flip |s|/2 positive entries to negative
                    flip_indices = np.argsort(-np.abs(y))[:s//2]
                    x[flip_indices] = -x[flip_indices]
                else:  # Flip |s|/2 negative entries to positive
                    flip_indices = np.argsort(-np.abs(y))[:(-s)//2]
                    x[flip_indices] = -x[flip_indices]
            # Symmetrize
            h = (x + 1) / 2
            full_h = np.concatenate((h, h[::-1]))
            h_full = full_h[:n]
            # Check sum constraint
            if abs(np.sum(h_full) - n/2) < 1e-10:
                # Evaluate
                dx = 2.0 / n
                c5 = max(correlate(h_full, 1 - h_full, mode='full')) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_full
        return best_h
    else:
        # Non-symmetric case - simpler rounding
        U, V = np.linalg.eigh(X)
        U = np.maximum(U, 0.0)
        V = V[:, U > 1e-6]
        U = np.sqrt(U[U > 1e-6])
        V = V @ np.diag(U)
        g = np.random.randn(m)
        y = V.T @ g
        x = np.sign(y)
        s = np.sum(x)
        if s != 0:
            if s > 0:
                flip_indices = np.argsort(-np.abs(y))[:s//2]
                x[flip_indices] = -x[flip_indices]
            else:
                flip_indices = np.argsort(-np.abs(y))[:(-s)//2]
                x[flip_indices] = -x[flip_indices]
        h = (x + 1) / 2
        return h
