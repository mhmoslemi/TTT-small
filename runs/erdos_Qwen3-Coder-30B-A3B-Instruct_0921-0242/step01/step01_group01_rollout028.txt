import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def c5_of_h(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution
    conv = np.correlate(h, g, mode='full')
    # The full convolution has 2*len(h) - 1 elements
    result = np.max(conv) * dx
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 200  # Discretization points
    n_half = n // 2
    dx = 2.0 / n
    
    # Symmetric version of h with only n_half + 1 independent variables
    # Total variables: n_half + 1 (w_i values) (first half), symmetrically mirrored
    # We'll assume w_i = h_i for i in [0, n_half], and h_{n-1-i} = w_i for i in [0, n_half]
    # So independent variables: w_0, w_1, ..., w_{n_half} where w_{n_half} = h_{n_half} (if n odd)
    # let's enforce w_i = h_i for i = 0..n_half (this means h = w for the first half)
    # and h_{n-1-i} = h_i for i = 0..n_half
    
    # In the final full h, we'll have:
    # h[i] = w[i] for i = 0 .. n_half
    # h[n-1-i] = w[i] for i = 0 .. n_half 
    # If n is even, then h[n_half] is unpaired, use h[n_half] = w[n_half]
    # If n is odd, h[n_half] = w[n_half], h[n_half+1] = w[n_half-1], etc.
    # The full array builds as w[:n_half] + [w[n_half]] + w[n_half-1::-1] 
    # But keep symmetric copies for initialization.
    
    # Anyway, the primary model variables are the first n_half + 1 values:
    # w_i = h_i for i = 0, ..., n_half (using symmetry)
    # Build full h from those later.

    # Define the problem as min M s.t. for each shift k:
    # sum_i h_i * h_{i+k} <= n/2 - M
    # and also bounds on h_i
    # When working with symmetric h, the usual shifts are k = -n+1, ..., n-1
    # but due to symmetry we can limit k to just 0, ..., n/2  
    # because shift k = -j matches shift j in the symmetric case.
    # We'll keep track of all relevant shifts ((0,n-1) as in the usual case, 
    # but due to symmetry, we can index shifts such that |k| <= n/2

    
    # ICC says that a feasible shift k must satisfy that 
    # sum_{i: i+k < n} h_i * h_{i+k} <= n/2 - M  
    # In our symmetric case, the full correlation sum depends on a 
    # fewer subsetting because there's duplication.
    
    # We define a set of valid active shifts from the set [0, n//2]. 
    # The corresponding symmetric shifts only if k != 0.
    # Maybe better to consider expand the full set of shifts for completeness
    
    def build_full_h_from_w(w):
        # build full symmetric h array from d first w_i (d = n_half + 1)
        # full length n
        h = np.zeros(n)
        h[:n_half+1] = w
        # symmetry
        for i in range(n_half):
            h[n-1-i] = w[i]
        return h

    # Here's the main algorithm:
    # 1. Define LP variables
    # 2. Solve it
    # 3. Check convergence
    # 4. Add violated constraint
    # 5. Repeat loop until convergence or timeout
    
    # Definition of variables:
    # w[i] for i in 0 to n_half inclusive
    # p[i,k] for valid (i,k) pairs where k is in shifts set S  
    # M - scalar for overbound
    
    # Initialization:
    # Start with shift k = 0, so just our sum constraint.
    # S = set of active shifts in LP
    S = set([0])
    # Also keep a history of best
    best_c5 = float('inf')
    best_h = None

    # Binary penalty strength: starts small, grows
    lambda_penalty = 0.1

    max_iterations = 50
    epsilon = 1e-6  # tolerance
    safety_margin = 10.0  # time safety margin (seconds)

    # Build the base model

    # We'll have a standard cvxpy LP approach but single-shot invocation 
    # as we *include* the cuts one by one until convergence
    
    # File real LP parametrization version of this problem:
    
    # Let's recognize the grasp: 
    # Objective: minimize M.
    # Variables: w[0:n_half+1], p[i,k], M.
    # The full convolution provides exact overlaps over all k. In theory, 
    # if we know the actual position at which the max overlaps occur, 
    # we'd know which constraint needs adding 
    # (i.e., for shifts of large |overlap|).
    
    # For now, just build the simplest deterministic version; 
    # assumption: after checking, we can use a standard simple cut-loop
    
    # For better determinism:
    iterations = 0
    while iterations < max_iterations and (time.time() - start_time) < budget_s - safety_margin:
        iterations += 1
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
            
        # In this version, we solve one LP model, but update the set of shifts S dynamically.
        
        # We want to maintain a small list of bad shifts in S, and a model with only 
        # those in hand with their McCormick inequalities.
        
        # Optimization.
        # Try to build a small MILP or LP model on current S
        
        # So let's_precisely name the variables now:
        # There are n_half + 1 core values w_i where i = 0..n_half
        w = cp.Variable(n_half + 1, name="w")
        M = cp.Variable(name="M")
        
        # Keep 2D dict of f_{i,k} = h_i h_{i+k}
        # If k=0, then it's h_i * h_i = h_i^2
        # There's symmetry involved - only i such that i, i+k < n are active
        # we'll make sure we never index out of bounds.
        
        # List all possible pairs (i,k) for current S
        # Don't yet build all p variables, but associate them later.
        
        # Rederive all problems from a single set-up approach
        
        # First set up an array of 2D indices (pairwise combinations)
        constraints = []
        
        # Sum constraint: Ensure sum w = n/4 + 0.5  -> sum(w) = n/4 + 0.5
        constraints.append(cp.sum(w) == (n / 4) + 0.5)
        
        # Box constraints on w
        constraints.append(w >= 0)
        constraints.append(w <= 1)
        
        # P_{i,k} variables
        # For computational simplicity define all p variables now
        
        log_p = {}  # p_{i,k} for current S (dict of 2D arrays)
        for k in S:
            p = cp.Variable((n, 1), name="p_k"+str(k))
            for i in range(n):
                if 0 <= i+k < n:
                    log_p[(i,k)] = p[i]
                else:
                    log_p[(i,k)] = 0  # invalid
            # The P_{i,k} variables  
        # Above: do not define P variable if i or i+k out of bounds (those will not affect anything)
        
        # MC inequalities matrices
        for k in S:
            for i in range(n):
                if not (0 <= i+k < n):
                    continue
                i1, i2 = i, i+k
                # Directly stored in ? No, we need to define
                # But more precise here:
                log_p[(i,k)] = cp.Variable(name=f'p_{i}_{k}')
                
                # Add McCormick inequalities:
                # p >= 0
                constraints.append(log_p[(i,k)] >= 0)
                # p <= w[i],  p<=w[i+k] 
                constraints.append(log_p[(i,k)] <= w[i])
                constraints.append(log_p[(i,k)] <= w[i+k])
                # p >= w[i] + w[i+k] - 1
                constraints.append(log_p[(i,k)] >= w[i] + w[i+k] - 1)
        
        # So now add the overlap constraints:
        # \sum_i p[i,k] <= n/2 - M  for all k in S
        for k in S:
            # Build actual overlap sum (defined by shifted version)
            total_p = 0
            for i in range(n):
                if 0 <= i+k < n:
                    total_p += log_p[(i,k)]
            # bound: \sum p_i,k <= n/2 - M
            constraints.append(total_p <= (n/2) - M)
        
        # Objective: minimize M - lambda * \sum p_{i,0}
        objective_terms = []
        for i in range(n):
            if 0 <= i+0 < n:
                objective_terms.append(log_p[(i,0)])
        # sum of p_i,0 (where i+0 = i)  
        obj_sum_p0 = cp.sum(objective_terms) 
        objective = cp.Minimize(M - lambda_penalty * obj_sum_p0)
        
        # Build and solve the LP
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            # For any robust methods we could handle infeasibility
        except Exception:
            # fallback if problem construction fails
            break  # fallback or break if error
    
        # Upon convergence:
        # Get current w values
        w_val = w.value
        M_val = M.value
        # To recover h, just replicate and mirror
        h = build_full_h_from_w(w_val)
        
        # Compute C5 precisely
        c5_estimate = c5_of_h(h)
        
        if c5_estimate < best_c5:
            best_c5 = c5_estimate
            best_h = h.copy()
            
        # Check convergence (if true overlap is within epsilon of M)
        if abs(c5_estimate - M_val) <= epsilon:
            break
        
        # Find the most violated constraint (i.e., the shift that gives the largest overlap)
        # For computational speed, let's compute the full correlation (a one-time operation with O(n^2))
        # Simple: compute full overlap score on current h and check best one
        
        g = 1.0 - h
        full_corr = np.correlate(h, g, mode='full')
        # shift indices: from -n+1 to n-1
        # but we only care about the positive shifts (mirror symmetric)  
        k_max = np.argmax(full_corr) - (n-1)

        if abs(k_max) > n//2:
            # Try trimming to in-bound
            cur_max = 0
            k_best = 0
            for k in range(-n//2, n//2+1):
                if (k == 0): continue  # skip 0
                corr_val = np.sum(h * np.roll(g, k))
                if corr_val > cur_max:
                    cur_max = corr_val
                    k_best = k
            k_max = (k_best)
                
        # Possibly additional debate over which = best. How to ensure k is in [-n+1, n-1]
        S.add(k_max)
        
        # Optionally increase penalty lambda for binary push
        lambda_penalty = min(5.0, lambda_penalty * 1.5)
    
    # Polishing finishes using projected subgradient descent:
    if time.time() - start_time > budget_s - 20:
        return (best_h, best_c5, n)
        
    # Polishing phase (longer) with projected step
    # Re-use best_h, make a few steps
    polishing_h = best_h.copy()
    polishing_c5 = best_c5
    
    # Function to compute a subgradient vector w.r.t. k_max, for the overlap function defined by h
    def subgradient_k(h, k):
        # h is a vector of length n
        # indices h[i], h[i+k] are paired in overlap integral
        # subgradient is \nabla C5 at h.
        dx = 2.0 / len(h)
        # Following insight: the derivative of overlap w.r.t h_i is known separately at each i
        # If overlap = \int h(x) (1 - h(x+k)) dx 
        # then
        # d/dh_i \int h(x) (1 - h(x+k)) dx = (1 - h[i+k])  * [i+k < n] - h[i-k] * [i-k >= 0]
        # (excluding omitted dx here for simplicity)
        g = np.zeros_like(h)
        for i in range(len(h)):
            if i+k < len(h):
                g[i] += 1 - h[i+k]
            if i-k >= 0:
                g[i] -= h[i-k] 
        # The way we use gradients in our case is by scalarizing them using distance to k_max:
        return g

    # Main loop:
    max_steps = 200
    s = 0.02  # initial step size
    
    while s > 1e-10 and max_steps > 0 and time.time() - start_time < budget_s - 20:
        max_steps -= 1
        # Find the current max overlap k
        g = 1.0 - polishing_h
        full_corr = np.correlate(polishing_h, g, mode='full')
        k_max = np.argmax(full_corr) - (n-1)
        
        # Compute stochastic step with subgradient
        grad = subgradient_k(polishing_h, k_max) 
        h_new = polishing_h - s * grad
        
        # Project back to feasible set (components in [0,1])
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum = n/2 (waterfill technique)  
        # sum of w is fixed, maintain that sum
        current_sum = np.sum(h_new) 
        diff = current_sum - (n/2)
        h_new = h_new - diff / len(h_new)  # subtract average diff from all
        
        # Ensure still clipped
        h_new = np.clip(h_new, 0, 1)
        
        # Apply symmetry (h_(n-1-i) := (h_(n-1-i) + h_i) / 2 for i < n/2)
        for i in range(n_half):
            h_new[n-1-i] = (h_new[n-1-i] + h_new[i]) / 2
            h_new[i] = h_new[n-1-i]
        
        # Ensure sum is still right:
        current_sum = np.sum(h_new)
        diff = current_sum - (n/2)
        h_new = h_new - diff / len(h_new) 
        
        # Evaluate new overlap
        polishing_c5_new = c5_of_h(h_new)
        
        if polishing_c5_new < polishing_c5:
            polishing_h = h_new.copy()
            polishing_c5 = polishing_c5_new
            s = 0.9 * s  # decay rate
        else:
            # Make a smaller uniform step and try again
            s /= 2

    if polishing_c5 < best_c5:
        best_c5 = polishing_c5
        best_h = polishing_h.copy()

    # Final hill climb add-on to perhaps squeeze a bit more improvement
    if time.time() - start_time < budget_s - 5:
        # random and meaningful perturbations
        for iter in range(200):
            if time.time() - start_time > budget_s - 10:
                break
            # Random bias: we'll perturb up to 6 entries
            p = min(6, n//20)
            indices = np.random.choice(n, p, replace=False)
            h_trial = best_h.copy()
            for i in indices:
                # Add small perturbation
                delta = np.random.uniform(-0.003, 0.003)
                h_trial[i] = min(1.0, max(0.0, h_trial[i] + delta))
            # Re-normalize (clip sum to n/2)
            h_trial = h_trial * (n / 2) / np.sum(h_trial)
            # Apply symmetry
            for i in range(n_half):
                h_trial[n-1-i] = (h_trial[n-1-i] + h_trial[i]) / 2
                h_trial[i] = h_trial[n-1-i]
            h_trial = h_trial * (n / 2) / np.sum(h_trial)
            c5_trial = c5_of_h(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
    
    return (best_h, best_c5, n)
