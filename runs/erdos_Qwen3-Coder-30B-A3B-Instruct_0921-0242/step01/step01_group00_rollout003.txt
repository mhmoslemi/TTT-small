```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import cvxpy as cp
from cvxpy import SCS

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_shift_matrix(n, k):
    """ Computes a sparse shift matrix T_k where T_k[i, i+k] = 1 """
    T = np.zeros((n, n))
    for i in range(n - k):
        T[i, i + k] = 1
    return T

def project_sum(h, n_points):
    """ Projects h to sum to n/2, clipping to [0,1] """
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n_points / 2 - current_sum
    if abs(diff) < 1e-10:
        return h
    # Water-filling approach
    h = h + diff / n_points
    h = np.clip(h, 0, 1)
    return h

# Multiple optimization stages
def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: SDP relaxation
    n_points = 128
    dx = 2.0 / n_points
    print(f"Starting SDP stage with n_points = {n_points}")
    
    # Define SDP variables
    h = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [cp.trace(X) == n_points / 2,
                   h >= 0, h <= 1,
                   X.diag() == h,
                   X >> 0,  # Positive semidefinite
    ]

    # Add mutual autocorrelation constraints
    for k in range(1, n_points):
        T_k = compute_shift_matrix(n_points, k)
        # trace(T_k @ X) = sum over i of X[i, i+k], which should be >= t
        constraints.append(cp.trace(T_k @ X) >= t)

    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=SCS, eps=1e-4, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            print(f"SDP solved with status {prob.status}")
            h_sdp = h.value
            t_opt = t.value
            
            # Ensure sum constraint is satisfied
            h_sdp = project_sum(h_sdp, n_points)
            C5_sdp = dx * (1 - t_opt)
            print(f"Best SDP C5 = {C5_sdp:.6f}")
        else:
            print(f"SDP failed with status {prob.status}")
            # fallback
            h_sdp = np.ones(n_points) * 0.5
            C5_sdp = evaluate_c5(h_sdp)
    except Exception as e:
        print("SDP failed, using fallback:", str(e))
        h_sdp = np.ones(n_points) * 0.5
        C5_sdp = evaluate_c5(h_sdp)

    # Stage 2: Randomized Rounding
    print("Starting randomized rounding stage...")
    best_C5 = C5_sdp
    best_h = h_sdp.copy()
    
    # Try deterministic rounding
    h_rounded = np.zeros(n_points)
    for i in range(n_points):
        # Simple deterministic binary rounding with threshold 0.5
        h_rounded[i] = 1.0 if h_sdp[i] >= 0.5 else 0.0
    h_rounded = project_sum(h_rounded, n_points)
    C5_rounded = evaluate_c5(h_rounded)
    
    if C5_rounded < best_C5:
        best_C5 = C5_rounded
        best_h = h_rounded.copy()
        
    # Try several random roundings from SDP
    R = 30
    for r in range(R):
        if time.time() - start_time > budget_s - 10:
            break
        # Generate random Gaussian vector and form linear combination
        V = np.linalg.eigh(X.value)[1]
        g = np.random.randn(V.shape[1])
        y = V @ g
        # Sort y
        sorted_y = np.sort(y)
        # Find threshold so exactly n/2 are > threshold
        i_thresh = n_points // 2
        if i_thresh < len(sorted_y):
            tau = sorted_y[i_thresh]
        else:
            tau = sorted_y[-1] + 1.0  # Should never happen

        h_random = (y > tau).astype(float)
        h_random = project_sum(h_random, n_points)
        C5_random = evaluate_c5(h_random)
        
        if C5_random < best_C5:
            best_C5 = C5_random
            best_h = h_random.copy()
            
    print(f"After rounding, best C5 = {best_C5:.6f}")

    # Stage 3: Smooth Max Gradient Descent Refinement
    print("Starting smooth max gradient refinement...")
    h_refined = best_h.copy()
    beta_0 = 0.5
    max_beta = 50.0
    beta_factor = 1.5
    inner_iter = 50

    def smooth_max(h, beta):
        """ Get the smooth max over all k */
        dx = 2.0 / len(h)
        corr = correlate(h, 1 - h, mode='full')
        corr = corr[len(corr)//2:] * dx  # Take right half
        assert len(corr) == len(h) - 1
        exp_corr = np.exp(beta * corr)
        log_sum = np.log(np.sum(exp_corr))
        return log_sum / beta

    def smooth_max_der(h, beta):
        """ Compute gradient of smooth max, with respect to h """
        dx = 2.0 / len(h)
        corr = correlate(h, 1 - h, mode='full')
        corr = corr[len(corr)//2:] * dx  # Take right half
        exp_corr = np.exp(beta * corr)
        weights = exp_corr / np.sum(exp_corr)
        
        # gradient of each I_k = dx*(1 - sum of h_i h_{i+k})
        grad = np.zeros_like(h)
        for k in range(1, len(h)):
            if k <= len(h) - 1:
                shifted = np.roll(h, k)
                laplace = h * shifted
                # we have integrated: for i, density is: dx*(1 - h_i * h_{i+k})  
                # so for partial w.r.t. h_i: dx*(-h_{i+k} + h_{i-k})
                g = dx * (shifted + np.roll(h, -k) - 2*laplace)
                # Shift pad to return len(h) gradient
                if k > 0:
                    g = np.concatenate([g[k:], np.zeros(k)])
                else:
                    g = g
            else:
                g = np.zeros_like(h)
            grad += weights[k-1] * g
        
        return grad

    beta = beta_0
    refines = 0
    while time.time() - start_time < budget_s - 20 and beta < max_beta:
        # Optimization step
        for _ in range(inner_iter):
            if time.time() - start_time > budget_s - 20:
                break
            grad = smooth_max_der(h_refined, beta)
            # Armijo-type line-search
            step_size = 1.0
            h_trial = h_refined - step_size * grad
            h_trial = project_sum(h_trial, n_points)
            s_max = smooth_max(h_refined, beta)
            s_trial = smooth_max(h_trial, beta)
            
            # Backtrack until improvement
            while s_trial > s_max and step_size > 1e-8:
                step_size *= 0.5
                h_trial = h_refined - step_size * grad
                h_trial = project_sum(h_trial, n_points)
                s_trial = smooth_max(h_trial, beta)

            h_refined = h_trial.copy()
        
        # Update beta  
        beta *= beta_factor
        refines += 1
        
        # Update best
        C5_refined = evaluate_c5(h_refined)
        if C5_refined < best_C5:
            best_C5 = C5_refined
            best_h = h_refined.copy()
    
    print(f"After refinement, best C5 = {best_C5:.6f}")

    # Stage 4: Final Discrete Mutation
    # Exploring small discrete changes
    print("Entering final local swap stage...")
    h_final = best_h.copy()
    C5_final = best_C5
    C5_best = C5_final
    h_best = h_final.copy()
    max_swaps = 2000  # arbitrary bound
    swaps = 0
    while swaps < max_swaps and time.time() - start_time < budget_s - 5:
        if time.time() - start_time > budget_s - 5:
            break
            
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if h_final[i] == 0 or h_final[j] == 1:
            continue
            
        delta = np.random.uniform(0.01, 0.05)
        delta = min(delta, h_final[i], 1. - h_final[j])
        if delta < 1e-8:
            continue
            
        h_trial = h_final.copy()
        h_trial[i] -= delta
        h_trial[j] += delta
        h_trial = project_sum(h_trial, n_points)
        
        C5_trial = evaluate_c5(h_trial)
        if C5_trial < C5_final:
            h_final = h_trial.copy()
            C5_final = C5_trial
            
            if C5_final < C5_best:
                C5_best = C5_final
                h_best = h_final.copy()
                
        swaps += 1

    print(f"Final C5 = {C5_best:.6f}")
    return (h_best, C5_best, n_points)
```
