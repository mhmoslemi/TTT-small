import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import math

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water filling approach
    h = h + diff / n 
    h = np.clip(h, 0, 1)
    return h

def compute_worst_shift_grad(h, n, shift_idx):
    """Compute gradient w.r.t. h of the worst shift term."""
    dx = 2.0 / n
    grad = np.zeros_like(h)
    
    # Compute derivative of A_{shift_idx}(h) = sum h_i * h_{i+shift_idx}
    # grad_i = h_{i+shift_idx} + h_{i-shift_idx}
    for i in range(n):
        left_idx = (i - shift_idx) % n
        right_idx = (i + shift_idx) % n
        grad[i] = dx * ((1 - h[right_idx]) - h[left_idx])
    
    return grad

def line_search(h, grad, max_iter=10):
    """Backtracking line search to update h and return new C5"""
    alpha = 1.0
    dx = 2.0 / len(h)
    for _ in range(max_iter):
        h_trial = h - alpha * grad
        h_trial = project_to_feasible(h_trial, len(h))
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < evaluate_c5(h):
            return h_trial, c5_trial
        alpha /= 2.0
    return h, evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 300
    dx = 2.0 / n_points

    # Initialize swarm
    S = 30  # swarm size
    particles = []
    velocities = []
    
    # Initial particles 
    for _ in range(S):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros_like(h))
    
    # Keep track of bests
    pbests = [p.copy() for p in particles]
    pbest_c5s = [evaluate_c5(p) for p in particles]
    
    gbest_idx = np.argmin(pbest_c5s)
    gbest = pbests[gbest_idx].copy()
    gbest_c5 = pbest_c5s[gbest_idx]
    
    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Global PSO phase
    last_improvement_time = time.time()
    gen = 0
    
    while time.time() - start_time < 0.7 * budget_s:
        if time.time() - last_improvement_time > 2.0:
            # Inject new particles
            for _ in range(S // 4):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                h = project_to_feasible(h, n_points)
                idx = np.random.randint(S)
                particles[idx] = h.copy()
                velocities[idx] = np.zeros_like(h)
                pbests[idx] = h.copy()
                pbest_c5s[idx] = evaluate_c5(h)
                if evaluate_c5(h) < gbest_c5:
                    gbest = h.copy()
                    gbest_c5 = evaluate_c5(h)
            last_improvement_time = time.time()
        
        for i in range(S):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbests[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Update personal best
            c5_new = evaluate_c5(particles[i])
            if c5_new < pbest_c5s[i]:
                pbest_c5s[i] = c5_new
                pbests[i] = particles[i].copy()
                if c5_new < gbest_c5:
                    gbest_c5 = c5_new
                    gbest = particles[i].copy()
                    last_improvement_time = time.time()
        
        gen += 1
        if gen % 5 == 0:
            # Apply memetic refinement
            memetic_h = gbest.copy()
            for _ in range(30):
                if time.time() - start_time > 0.9 * budget_s:
                    break
                # Find worst shift
                corr = correlate(memetic_h, 1 - memetic_h, mode='full')
                worst_shift = np.argmax(corr)
                worst_shift = worst_shift - (2*n_points - 1) // 2  # convert to shift index
                grad = compute_worst_shift_grad(memetic_h, n_points, abs(worst_shift))
                memetic_h, new_c5 = line_search(memetic_h, grad)
                if new_c5 < gbest_c5:
                    gbest_c5 = new_c5
                    gbest = memetic_h.copy()
                    
        if time.time() - start_time > 0.7 * budget_s - 1.0:
            break

    # Final polishing hill climb
    final_h = gbest.copy()
    final_c5 = gbest_c5
    
    for i in range(5000):
        if time.time() - start_time > budget_s - 5.0:
            break
        i_idx = np.random.choice(n_points)
        j_idx = np.random.choice(n_points)
        if final_h[i_idx] <= 0.0 or final_h[j_idx] >= 1.0:
            continue
        delta = np.random.uniform(0.0, min(final_h[i_idx], 1.0 - final_h[j_idx], 0.01))
        h_trial = final_h.copy()
        h_trial[i_idx] -= delta
        h_trial[j_idx] += delta
        h_trial = project_to_feasible(h_trial, n_points)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < final_c5:
            final_h = h_trial.copy()
            final_c5 = c5_trial

    return (final_h, final_c5, n_points)
