import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp
from cvxpy import SCS

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: SDP relaxation to get a good initial solution
    n_points = 128
    dx = 2.0 / n_points
    
    # Solve SDP relaxation
    h_sdp = np.ones(n_points) * 0.5  # default
    sdp_time = 0
    try:
        # Variables
        h = cp.Variable(n_points)
        X = cp.Variable((n_points, n_points), symmetric=True)
        t = cp.Variable()

        # Constraints
        constraints = [
            cp.trace(X) == n_points / 2.0,
            h >= 0,
            h <= 1,
            X >> 0,
            cp.diag(X) == h
        ]

        # Add autocorrelation constraints
        for k in range(1, n_points):
            # Shift matrix T_k for shift k
            T_k = np.zeros((n_points, n_points))
            for i in range(n_points - k):
                T_k[i, i + k] = 1

            # Add constraint: trace(T_k @ X) >= t
            constraints.append(cp.trace(T_k @ X) >= t)

        # Objective: maximize t
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=SCS, eps=1e-4, max_iters=5000)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sdp = h.value
            sdp_time = time.time() - start_time
        else:
            print("SDP solver failed:", prob.status)
    except Exception as e:
        print("SDP solver error:", e)
    
    # Post-process SDP solution to enforce exact sum
    current_sum = np.sum(h_sdp)
    diff = n_points / 2 - current_sum
    if abs(diff) > 1e-8:
        h_sdp = h_sdp + diff / n_points
        h_sdp = np.clip(h_sdp, 0, 1)
    h_sdp = np.clip(h_sdp, 0, 1)

    # Evaluate SDP result
    c5_sdp = evaluate_c5(h_sdp)
    best_h = h_sdp.copy()
    best_c5 = c5_sdp

    # Stage 2: Binary Rounding
    if (time.time() - start_time) < budget_s - 10:
        try:
            # Factorize X = V V^T (from the SDP)
            # We use the fact that X ≈ h_sdp h_sdp^T, so we take h_sdp as the factor
            # Now create a vector to transform it
            R = 30
            best_h_bin = h_sdp.copy()
            best_c5_bin = c5_sdp
            
            for r in range(R):
                if time.time() - start_time > budget_s - 10:
                    break
                # Sample g from standard normal
                g = np.random.randn(n_points)
                # v = V * g \approx one scaled sample of the random vector
                v = h_sdp * g
                
                # Sort v to find suitable threshold
                sorted_v = np.sort(v)
                # Choose threshold so that sum of 1s is exactly n/2
                idx = np.searchsorted(sorted_v, np.partition(sorted_v, n_points // 2)[n_points // 2])
                threshold = sorted_v[idx]
                h_bin = (v > threshold).astype(float)
                # Ensure exact sum constraint
                current_sum = np.sum(h_bin)
                diff_sum = n_points / 2 - current_sum
                if abs(diff_sum) > 1e-10:
                    # Adjust
                    indices = np.argsort(v)
                    for i in range(int(np.abs(diff_sum))):
                        if diff_sum < 0:  # Need more ones
                            h_bin[indices[i]] = 1.0
                        else:  # Need more zeros
                            h_bin[indices[-(i+1)]] = 0.0
            
                # Evaluate
                c5_trial = evaluate_c5(h_bin)
                if c5_trial < best_c5_bin:
                    best_c5_bin, best_h_bin = c5_trial, h_bin.copy()
                
            if best_c5_bin < best_c5:
                best_c5, best_h = best_c5_bin, best_h_bin
        except Exception as e:
            print("Rounding error:", e)
    
    # Stage 3: Smooth max gradient refinement using continuation
    if (time.time() - start_time) < budget_s - 10:
        # Parameters
        max_iter_inner = 50
        beta_max = 50
        beta_min = 0.5
        beta_factor = 1.5
        step_size = 1.0
        
        h_refined = best_h.copy()
        current_c5 = evaluate_c5(h_refined)
        
        beta = beta_min
        while beta <= beta_max:
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute the smooth max gradient
            def smooth_max_grad(h, beta):
                dx = 2.0 / len(h)
                overlap = []  # storing the individual overlaps I_k(h)
                grads = []   # storing gradients ∇I_k
                
                corr = correlate(h, 1 - h, mode='full')
                # Scale by dx
                scaled_corr = corr * dx
                
                # Only relevant for positive shifts, and excluding the self term  
                for shift in range(1, len(h)):
                    # For each shift k, compute I_k(h)
                    # We need to handle the correlation term carefully
                    # corr is full length: len(h) + len(h) - 1 = 2*n - 1
                    # We pick out the appropriate correlation term
                    corr_val = scaled_corr[len(h) - 1 - shift]
                    overlap.append(1 - corr_val)
                
                # Renormalize weights to sum to 1
                exp_overlap = np.exp(beta * np.array(overlap))
                weights = exp_overlap / np.sum(exp_overlap)
                
                # Approximate gradient via finite differences (crude, but works safely)
                grad = np.zeros_like(h)
                eps = 1e-6
                for i in range(len(h)):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    I_plus = (1 - (corr_plus * dx)[len(h) - 1 - 1])  # use 1 for simplicity
                    # Very rough linear estimate for gradient
                    grad[i] = (I_plus - (1 - scaled_corr[len(h) - 1 - 1])) / eps

                # Smooth approximation should be weighted average with I_k using weights
                # since we don't do exact gradient logic here, use a proxy:
                final_grad = np.zeros_like(h)
                for i in range(len(h)):
                    gs = 0.0
                    for k in range(1, len(h)):
                        # Try a proxy for smaller derivative
                        idx = len(h) - 1 - k
                        if i >= k:
                            # derivative of h_i h_{i-k}
                            gs -= dx * h[i - k]
                        if i + k < len(h):
                            # derivative of h_{i+k} h_i
                            gs += dx * h[i + k]
                    final_grad[i] = gs
                
                # Weighted average of gradients
                weighted_grad = np.average([final_grad], axis=0, weights=weights)
                return weighted_grad
        
            for _ in range(max_iter_inner):
                if time.time() - start_time > budget_s - 10:
                    break
                    
                grad = smooth_max_grad(h_refined, beta)
                
                # Backtracking line search
                alpha = step_size
                h_trial = h_refined - alpha * grad
                h_trial = np.clip(h_trial, 0, 1)
                # Ensure sum constraint
                current_sum = np.sum(h_trial)
                diff = n_points / 2 - current_sum
                h_trial = h_trial + diff / n_points
                h_trial = np.clip(h_trial, 0, 1)
                c5_trial = evaluate_c5(h_trial)
                
                # Reduce step
                attempts = 0
                while c5_trial > current_c5 and attempts < 10:
                    alpha *= 0.5
                    attempts += 1
                    h_trial = h_refined - alpha * grad
                    h_trial = np.clip(h_trial, 0, 1)
                    current_sum = np.sum(h_trial)
                    diff = n_points / 2 - current_sum
                    h_trial = h_trial + diff / n_points
                    h_trial = np.clip(h_trial, 0, 1)
                    c5_trial = evaluate_c5(h_trial)
                    
                if c5_trial < current_c5:
                    h_refined = h_trial.copy()
                    current_c5 = c5_trial
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h_refined.copy()
                else:
                    break
                    
            beta *= beta_factor
            
    # Stage 4: Final local search with discrete swaps
    if (time.time() - start_time) < budget_s - 5:
        max_swaps = 5000  # roughly 5000 swaps if time is not constrained
        t_0 = time.time()
        
        # Helper function to evaluate the overlap I_k for some shift k
        # Here we'll compute it with correlations
        def compute_I_k(h, k):
            dx = 2.0 / len(h)
            shifted = np.roll(h, k)
            corr = np.sum(h * (1 - shifted))
            return dx * (1 - corr)  # I_k
            
        # Helper to compute max of I_k for all k
        def get_max_I_k(h):
            all_I = [compute_I_k(h, k) for k in range(1, len(h))]
            return max(all_I)
            
        for _ in range(max_swaps):
            if time.time() - t_0 > (budget_s - 5 - (time.time() - start_time)):
                break
            i, j = np.random.randint(0, len(best_h), size=2)
            if i == j:
                continue
                
            # If h_i > 0 and h_j < 1
            if not (best_h[i] > 0 and best_h[j] < 1):
                continue
                
            delta = min(best_h[i], 1 - best_h[j], 0.02)
            
            # Try to swap delta from i to j
            # Update the h vector
            h_new = best_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            
            # Make sure the sum is preserved (this is exact)  
            # But preserve sum to n/2 exactly
            current_sum = np.sum(h_new)
            diff_sum = n_points / 2 - current_sum
            h_new += diff_sum / n_points
            h_new = np.clip(h_new, 0, 1)
            
            c5_new = get_max_I_k(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    return (best_h, best_c5, n_points)
