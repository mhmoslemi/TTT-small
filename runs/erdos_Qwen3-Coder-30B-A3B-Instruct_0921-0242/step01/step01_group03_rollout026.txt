def build_shift_matrices(N):
    """Build sparse CSR shift matrices A_k"""
    A_list = []
    for k in range(N):
        data = np.ones(N - k)
        row = np.arange(N - k)
        col = np.arange(k, N)
        A = sp.csr_matrix((data, (row, col)), shape=(N, N))
        A_list.append(A)
    return A_list

def c5_of_h_exact(h):
    """Compute exact C5 with full correlation."""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def bounded_simplex_project(h, target_sum):
    """Project h to the simplex with bounds [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    deficit = target_sum - current_sum
    if abs(deficit) < 1e-10:
        return h
    # Simple strategy: distribute deficit uniformly
    adjustment = deficit / len(h)
    h += adjustment
    h = np.clip(h, 0, 1)
    # Fine tune to get exact sum
    current_sum = np.sum(h)
    h += (target_sum - current_sum) / len(h)
    h = np.clip(h, 0, 1)
    return h

def principal_eigenvector_rounding(X):
    """Perform principal eigenvector rounding."""
    eigenvals, eigenvecs = np.linalg.eigh(X)
    idx = np.argmax(eigenvals)
    v = eigenvecs[:, idx]
    lambda_val = eigenvals[idx]
    h_eig = np.sqrt(lambda_val) * np.abs(v)
    h_eig = bounded_simplex_project(h_eig, len(h_eig) / 2)
    return h_eig

def diagonal_threshold_rounding(X):
    """Perform diagonal-threshold rounding."""
    diag = np.diag(X)
    sort_idx = np.argsort(-diag)  # descending order
    h_thr = np.zeros_like(diag)
    h_thr[sort_idx[:len(diag)//2]] = 1.0
    # Already filled sum constraint
    return h_thr

    while time.time() - start_time < max_time:
        if accepted_swaps > 0 and accepted_swaps % 500 == 0:
            # Recompute full once in a while to correct any numerical drift
            g = 1.0 - h
            corr_full = np.correlate(h, g, mode='full') * dx
            current_max = np.max(corr_full)
            
        # Pick two indices: one where it's 1, another where it's 0
        one_indices = np.where(h > 0.5)[0]
        zero_indices = np.where(h < 0.5)[0]
        if len(one_indices) == 0 or len(zero_indices) == 0:
            break
        p = np.random.choice(one_indices)
        q = np.random.choice(zero_indices)
        
        # Simulate the swap by temporarily updating
        h_new = h.copy()
        h_new[p] = 0.0
        h_new[q] = 1.0
        
        # Recompute corr for the new h
        g_new = 1.0 - h_new
        corr_new = np.correlate(h_new, g_new, mode='full') * dx
        new_max = np.max(corr_new)
        
        if new_max < current_max:
            h = h_new
            corr_full = corr_new
            current_max = new_max
            accepted_swaps += 1
        else:
            # Occasionally just accept a random swap for diversity
            if np.random.rand() < 0.01:
                h = h_new
                corr_full = corr_new
                current_max = new_max
                accepted_swaps += 1
                
    return h

def solve_sdp(A_list, N):
    """Solve the SDP relaxation with CVXPY."""
    # Build optimization variables
    h = cp.Variable(N)
    X = cp.Variable((N, N), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Diagonal consistency: X_{ii} = h_i
    for i in range(N):
        constraints.append(X[i, i] == h[i])
    
    # Sum constraint
    constraints.append(cp.sum(h) == N / 2)
    
    # Bounds on h
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Minimum correlation constraints
    for k in range(N):
        constraints.append(cp.trace(A_list[k] @ X) >= t)
    
    # PSD constraint
    constraints.append(X >> 0)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=SCS, max_iters=15000, eps=1e-4)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value, X.value, t.value
        else:
            print(f"Infeasible or unbounded. Status: {prob.status}")
            return None, None, None
    except Exception as e:
        print(f"SDP solver failed: {e}")
        return None, None, None

def upsample_vec(h, n_up):
    """Linearly upsample h to n_up points."""
    h_up = np.zeros(n_up)
    for i in range(n_up):
        i_lo = i // 2
        i_hi = (i + 1) // 2
        alpha = i / 2.0 - i_lo
        if i_hi == len(h):
            h_up[i] = h[i_lo]
        else:
            h_up[i] = (1 - alpha) * h[i_lo] + alpha * h[i_hi]
    # Scale to sum = n_up / 2
    h_up = h_up * (n_up / 2) / np.sum(h_up)
    return h_up
