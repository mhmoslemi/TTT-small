import numpy as np
import time
import scipy.optimize as opt
import scipy.fft as fft
from cvxpy import Variable, Minimize, Problem, Constraint, sum as cvx_sum, norm

def correlation_fft(h):
    """Compute all f_k(h) values using FFT, returning vector of size N"""
    N = len(h)
    h_pad = np.pad(h, (0, N), mode='constant')
    g_pad = np.pad(1 - h, (0, N), mode='constant')
    c = fft.fft(h_pad) * fft.fft(g_pad)
    return np.real(fft.ifft(c))[:N] * 2.0 / N

def weighted_gradient_fft(h, p):
    """Compute weighted gradient ∇F_τ(h) using FFT, where p is probability distribution over shifts"""
    N = len(h)
    p_pad = np.pad(p, (0, N), mode='constant')
    h_pad = np.pad(h, (0, N), mode='constant')
    g_pad = np.pad(1 - h, (0, N), mode='constant')
    
    w1 = np.real(fft.ifft(fft.fft(p_pad) * fft.fft(g_pad)))[:N]
    w2 = np.real(fft.ifft(fft.fft(p_pad[::-1]) * fft.fft(h_pad)))[:N]
    return w1 - w2

def project_onto_Ω(v):
    """Project vector v onto the set { h : 0 ≤ h_i ≤ 1, Σ h_i = N/2 }"""
    N = len(v)
    z = np.clip(v, 0, 1)
    r = N/2 - np.sum(z)
    
    I = np.where((z > 0) & (z < 1))[0]
    while abs(r) > 1e-12 and len(I) > 0:
        delta = r / len(I)
        z[I] += delta
        # Clip and recompute residual
        z = np.clip(z, 0, 1)
        r = N/2 - np.sum(z)
        I = np.where((z > 0) & (z < 1))[0]
    
    return z

def binary_from_top(h, count_ones):
    """Create binary vector selecting top `count_ones` values"""
    N = len(h)
    indices = np.argsort(h)[::-1]
    mask = np.zeros(N, dtype=int)
    mask[indices[:count_ones]] = 1
    return mask

def fractional_correction(h_bin):
    """Use LP to fix small errors in binary solution."""
    N = len(h_bin)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    # Find indices that are at decision boundary (close to 0.5)
    if len(ones) + len(zeros) >= N:
        # Simple fix: ensure sum is right
        adjusted = h_bin.astype(float)
        diff = N/2 - np.sum(adjusted)
        if abs(diff) > 1e-6:
            adjust_indices = zeros[:int(np.ceil(abs(diff)))]
            if diff < 0:
                adjusted[adjust_indices] = 1.0
            else:
                # diff > 0, adjust some 1s to 0s
                adjust_indices = ones[:int(np.ceil(abs(diff)))]
                adjusted[adjust_indices] = 0.0
        return adjusted
    return h_bin

def swap(h, i, j):
    """Swap values at position i and j in h."""
    h_copy = h.copy()
    h_copy[i], h_copy[j] = h_copy[j], h_copy[i]
    return h_copy

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h."""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    N = 256  # power-of-two grid size
    h = np.random.rand(N)
    h = project_onto_Ω(h)
    
    τ_start = 0.05
    τ_factor = 0.5
    outer_cycles = 6
    inner_iters = 150
    τ = τ_start
    
    for outer in range(outer_cycles):
        α = 1.0
        h_prev = h.copy()
        for t in range(inner_iters):
            if time.time() - start_time > budget_s - 30:
                break
            # Nesterov extrapolation
            y = h + ((t-1)/(t+2)) * (h - h_prev)
            f = correlation_fft(y)
            p_unnorm = np.exp(f/τ - np.max(f/τ))  # numerical stability
            p = p_unnorm / np.sum(p_unnorm)
            grad = weighted_gradient_fft(y, p)
            
            # Backtracking line search
            while True:
                z = y - α * grad
                z_proj = project_onto_Ω(z)
                # Approximate F_τ
                Fτ_y = τ * np.log(np.sum(np.exp(f/τ)))  # Approximate F_τ(y)
                Fτ_z = τ * np.log(np.sum(np.exp(correlation_fft(z_proj)/τ)))
                if Fτ_z <= Fτ_y - 0.5*α*np.dot(grad, grad):
                    break
                α *= 0.5
            h_prev, h = h, z_proj
        if time.time() - start_time > budget_s - 30:
            break
        τ *= τ_factor
    
    # Rounding and refinement
    h_bin = binary_from_top(h, N//2)
    h_refined = fractional_correction(h_bin)
    
    # Local swap hill-climbing
    best_h = h_refined
    best_c5 = c5_of_h(best_h)
    swap_counter = 0
    while time.time() - start_time < budget_s - 80:
        i = np.random.randint(0, N)
        j = np.random.randint(0, N)
        if (h_refined[i] != h_refined[j]):
            cand = swap(h_refined, i, j)
            cand_c5 = c5_of_h(cand)
            if cand_c5 < best_c5 - 1e-10:
                best_h, best_c5 = cand, cand_c5
                if time.time() - start_time > budget_s - 5:
                    break
        swap_counter += 1
        if swap_counter > 5000:
            break
    
    # Optional upsample and finalize
    if time.time() - start_time < budget_s - 5:
        h_up = np.interp(np.linspace(0, 2, 512), np.linspace(0, 2, 256), best_h)
        h_up = project_onto_Ω(h_up)
        res = opt.minimize(
            fun=c5_of_h,
            x0=h_up,
            method='SLSQP',
            bounds=[(0,1)]*512,
            constraints={'type':'eq','fun':lambda x: np.sum(x)-256},
            options={'maxiter':100,'ftol':1e-9}
        )
        final_h = res.x
        final_c5 = c5_of_h(final_h)
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5
        n_points = len(final_h)
    else:
        n_points = len(best_h)

    return (best_h, best_c5, n_points)
