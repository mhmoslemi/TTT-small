def c5_of_h(h):
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

    start_time = time.time()

    n_points = 400  # Grid size
    rank = 6        # Rank of the factorization

    max_iter_per_temp = 200
    temp_schedule = [0.05, 0.025, 0.0125, 0.00625, 0.003125, 0.0015625]
    temp_schedule = [t for t in temp_schedule if t >= 0.0015625]

    def compute_correlations(Y):
        c_k = np.zeros(n_points)
        for i in range(rank):
            y = Y[:, i]
            y_padded = np.pad(y, (0, n_points), mode='constant')
            conv = fftconvolve(y_padded, y_padded[::-1], mode='full')
            c_k += conv[:n_points]
        return c_k
    
    def log_sum_exp_bound(Y, tau):
        c_k = compute_correlations(Y)
        exp_vals = np.exp(-c_k / tau)
        return -tau * np.log(np.sum(exp_vals))
    
    def gradient_log_sum_exp(Y, tau):
        c_k = compute_correlations(Y)
        exp_vals = np.exp(-c_k / tau)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        
        # Gradients for each row
        grad = np.zeros_like(Y)
        for i in range(rank):
            y = Y[:, i]
            # Compute convolutions with weights
            weight_padded = np.pad(weights, (0, n_points), mode='constant')
            y_padded = np.pad(y, (0, n_points), mode='constant')
            conv1 = fftconvolve(y_padded[::-1], weight_padded, mode='full')[:n_points]
            conv2 = fftconvolve(weight_padded[::-1], y_padded, mode='full')[:n_points]
            grad[:, i] = conv1 + conv2
        return grad / tau
    
    def project_onto_feasible(Y):
        h = np.sum(Y**2, axis=1)
        # Clip norms to [0, 1]
        norms = np.sqrt(np.sum(Y**2, axis=1))
        # Normalize rows that are not zero
        for i in range(n_points):
            if norms[i] > 1e-8:
                Y[i] = Y[i] / norms[i] * min(1.0, norms[i])
        # Enforce mass constraint
        current_sum = np.sum(norms**2)
        scale = np.sqrt((n_points / 2) / current_sum)
        Y *= scale
        # Final clipping
        norms = np.linalg.norm(Y, axis=1)
        for i in range(n_points):
            if norms[i] > 1:
                Y[i] = Y[i] / norms[i]
        return Y
    
    def deterministic_swap_improvement(h_binary, target_c5):
        # Compute current correlations
        c_k = compute_correlations(np.sqrt(h_binary.reshape(-1,1)))
        k_max = np.argmax(c_k)  # shift causing max overlap
        current_c5 = n_points / 2 - np.max(c_k)
        if current_c5 <= target_c5:
            return h_binary, current_c5
        # Identify positions that contribute to worst shift
        positions_1 = np.where(h_binary == 1)[0]
        positions_0 = np.where(h_binary == 0)[0]
        worst_mismatch = positions_1[np.where(h_binary[(positions_1 + k_max) % n_points] == 0)[0]]
        # Construct pairs and evaluate impact
        best_idx_p = best_idx_q = -1
        max_improvement = 0
        for p in worst_mismatch:
            for q in positions_0:
                if p == q:
                    continue
                # Update correlations affected by swap
                delta = 0
                for k in range(n_points):
                    old_val = int(h_binary[p]) * int(h_binary[(p+k)%n_points]) + int(h_binary[(p-k)%n_points]) * int(h_binary[p])
                    new_val = int(h_binary[q]) * int(h_binary[(q+k)%n_points]) + int(h_binary[(q-k)%n_points]) * int(h_binary[q])
                    delta += (old_val - new_val)
                if delta > max_improvement:
                    max_improvement = delta
                    best_idx_p, best_idx_q = p, q
        if max_improvement > 0:
            h_binary[best_idx_p] = 0
            h_binary[best_idx_q] = 1
            return deterministic_swap_improvement(h_binary, target_c5)
        return h_binary, current_c5

                break
            # Random initialization
            Y = np.random.randn(n_points, rank) * 0.1
            Y = project_onto_feasible(Y)

            max_val_last = np.max(c_k_last)
            for temp in temp_schedule:
                for iter_ in range(max_iter_per_temp):
                    if time.time() - start_time > budget_s - 10:
                        break
                    # Evaluate surrogate
                    phi = log_sum_exp_bound(Y, temp)
                    # Gradient
                    grad = gradient_log_sum_exp(Y, temp)
                    # Backtracking line search
                    alpha = 1.0
                    for backtrack_iter in range(10):
                        Y_new = Y - alpha * grad
                        Y_new = project_onto_feasible(Y_new)
                        phi_new = log_sum_exp_bound(Y_new, temp)
                        if phi_new >= phi - 1e-8 and backtrack_iter > 0:
                            break
                        alpha *= 0.5

                if time.time() - start_time > budget_s - 10:
                    break
            h_cont = np.sum(Y**2, axis=1)
            h_binary = np.zeros(n_points)
            sort_idx = np.argsort(h_cont)[::-1]
            h_binary[sort_idx[:n_points//2]] = 1.0
            h_binary = h_binary.astype(bool)
            
            # Swap polishing
            h_binary_polished, c5_val = deterministic_swap_improvement(h_binary.copy(), best_c5_val)
            h_binary = h_binary_polished
            
            if c5_val < best_c5_val:
                best_c5_val = c5_val
                best_result = (h_binary.astype(float), best_c5_val, n_points)
            
            if best_c5_val <= 0.379:
                break
        return best_result
    
    # Main optimization
    try:
        result = smooth_optimization()
        if result:
            best_h, best_c5, n_points = result
            # Compute the final C5 using the original method
            c5_improved = c5_of_h(best_h)
            if c5_improved < best_c5:
                best_c5 = c5_improved

            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = c5_of_h(best_h)
    except Exception as e:
        # Fall back
        best_h = np.array(initial_h_values)
        x_init = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points)
        best_h = np.interp(x_new, x_init, best_h)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = c5_of_h(best_h)
    
    # Upsample and refine with directional derivatives
    n_points_upsample = 800
    dx_upsample = 2.0 / n_points_upsample
    x_upsample = np.linspace(0, 2, n_points_upsample)
    t_upsample = np.linspace(0, 2, n_points_upsample)
    h_upsample = interp1d(np.linspace(0, 2, n_points), best_h, kind='linear')(x_upsample)
    h_upsample = h_upsample * (n_points_upsample / 2) / np.sum(h_upsample)
    
    # Project it back
    h_upsample = np.clip(h_upsample, 0, 1)
    sum_h = np.sum(h_upsample)
    adjust = (n_points_upsample/2) / sum_h
    h_upsample *= adjust
    h_upsample = np.clip(h_upsample, 0, 1)
    
    # Some refining steps using subgradients
    for _ in range(30):
        if time.time() - start_time > budget_s - 2:
