import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_full_correlation(h):
    """Evaluate full correlation for integer vector."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    return corr

def project_to_constraints(h, n_points, target_sum):
    """Project h onto sum constraint and box constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Distribute proportionally
    weights = np.ones_like(h)
    weights[np.where(h <= 0)] = 0
    weights[np.where(h >= 1)] = 0
    if np.sum(weights) > 0:
        h += diff * weights / np.sum(weights)
        h = np.clip(h, 0, 1)
    return h

def fast_anneal_binary(h_init, max_iter, temp_start, temp_decay, dx, target_sum, n_points):
    """Fast simulated annealing for binary vectors."""
    h_current = h_init.copy()
    h_best = h_current.copy()
    c5_best = c5_of_h(h_current, dx)
    
    temp = temp_start
    for i in range(max_iter):
        # Pick random 1 and 0 positions
        ones = np.where(h_current == 1)[0]
        zeros = np.where(h_current == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
            
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        
        # Swap
        h_new = h_current.copy()
        h_new[i1] = 0
        h_new[i0] = 1
        
        # Check constraints
        h_new = project_to_constraints(h_new, n_points, target_sum)
        
        c5_new = c5_of_h(h_new, dx)
        delta = c5_new - c5_best
        
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h_current = h_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
        
        temp *= temp_decay
        if temp < 1e-10:
            temp = 1e-10
            
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):

        h0 = np.random.rand(n_points)

        # Ensure sum is correct
        current_sum = np.sum(h0)
        if current_sum != target_sum:
            # Adjust to sum exactly to target_sum
            # We'll adjust by swapping
            missing = int(target_sum - current_sum)
            if missing > 0:
                # Need to make more 1's
                zero_indices = np.where(h0 == 0)[0]
                if len(zero_indices) >= missing:
                    for i in range(missing):
                        h0[zero_indices[i]] = 1
                else:
                    # Fill in the rest
                    h0[:missing] = 1
            elif missing < 0:
                # Too many 1's
                one_indices = np.where(h0 == 1)[0]
                if len(one_indices) >= -missing:
                    for i in range(-missing):
                        h0[one_indices[i]] = 0
                else:
                    h0[n_points+missing:] = 0
                    
        # Run heuristic optimization
        max_anneal_iter = min(50000, int(remaining_time * 2000))
        if max_anneal_iter > 0:
            h_heuristic, c5_heuristic = fast_anneal_binary(
                h0, max_anneal_iter, 0.5, 0.9999, dx, target_sum, n_points
            )
        else:
            h_heuristic, c5_heuristic = h0, c5_of_h(h0, dx)
            
        # Now solve MIP using CVXPY with CBC (if available)

        try:

            pass
            
        if c5_heuristic < best_c5:
            best_c5 = c5_heuristic
            best_h = h_heuristic.copy()
            best_n = n_points
            
        if best_c5 < 0.379:
            # Early termination if target met
            break
            
    # Postprocessing with local search
    if best_h is not None and budget_s - (time.time() - start_time) > 0.5:
        h_temp = best_h.copy()
        max_iter = min(10000, int((budget_s - (time.time() - start_time)) * 100))
        iter_counter = 0
        
        while iter_counter < max_iter:
            iter_counter += 1
            # Swap two points
            i1 = np.random.randint(n_points)
            i2 = np.random.randint(n_points)
            if h_temp[i1] == h_temp[i2]: continue
            
            val1, val2 = h_temp[i1], h_temp[i2]
            h_temp[i1], h_temp[i2] = val2, val1
            
            # Project
            h_temp = project_to_constraints(h_temp, n_points, target_sum)
            c5_new = c5_of_h(h_temp, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_temp.copy()
            else:
                h_temp[i1], h_temp[i2] = val1, val2
                
    return (best_h, best_c5, best_n)
