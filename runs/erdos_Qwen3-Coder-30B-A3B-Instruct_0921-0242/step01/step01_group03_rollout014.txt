import numpy as np

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def evaluate_binary_mask(b):
    """Evaluate C5 for a binary mask (assumed to have exactly 100 ones)"""
    dx = 2.0 / len(b)
    g = 1.0 - b.astype(float)
    corr = np.correlate(b.astype(float), g, mode='full') * dx
    return np.max(corr)

def neighbor_swap(b, i, j):
    """Swap positions i and j in the binary mask b"""
    b_new = b.copy()
    b_new[i], b_new[j] = b_new[j], b_new[i]
    return b_new

def binary_mask_to_h(b, h_vals):
    """Convert binary mask to h values"""
    return b.astype(float) * h_vals

def project_to_feasible(h):
    """Project onto the set {h: 0<=h<=1, sum(h)=n_points/2}"""
    n = len(h)
    dx = 2.0 / n
    total = np.sum(h)
    deficit = n/2 - total
    if abs(deficit) < 1e-10:
        return h
    adjustment = deficit / n
    h = np.clip(h + adjustment, 0, 1)
    # Correct the sum
    total = np.sum(h)
    deficit = n/2 - total
    if abs(deficit) > 1e-10:
        # Distribute error
        adjustment = deficit / n
        h = np.clip(h + adjustment, 0, 1)
    return h

def solve_refinement_lp(b_strong, h_best):
    """Iteratively refine mask by solving LP with variable intensities"""
    n = len(b_strong)
    dx = 2.0 / n
    
    # Set up LP
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2, t >= 0]
    
    # Add constraints for each shift k
    for k in range(1 - n, n):
        if k < 0:
            # h_i * (1 - h_{i+k}) <= t for all i
            # Since we have b_strong, set which values are fixed to 0 or 1
            expr = 0
            for i in range(n):
                # For shift k, neighbor is at i+k
                j = (i + k) % n
                if b_strong[i] == 1:
                    expr += h[i] * (1 - h[j])
                else:  # b_strong[i] == 0
                    expr += h[i] * (1 - 0)
            constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return np.array(h.value)
    except:
        pass
    
    return h_best

def perturb_and_refine(h_init, max_evals=2000):
    """Small random perturbations followed by projection"""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = c5_of_h(h)
    
    for _ in range(max_evals):
        h_new = h.copy()
        # Make small change
        idx = np.random.randint(0, len(h))
        delta = np.random.uniform(-0.05, 0.05)
        h_new[idx] += delta
        # Project it back
        h_new = project_to_feasible(h_new)
        c5_new = c5_of_h(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best_h, best_c5
