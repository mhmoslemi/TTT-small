def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    return np.max(corr)

def project_to_feasible(h):
    """Ensure sum(h) = n/2 and h in [0,1]^n"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    # Project into feasible set with bounds [0,1] and sum = n/2
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    # Distribute deficit
    adjustment = deficit / n
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def binary_optimize(n_points, max_time_s=350):
    """
    Simulated Annealing for binary mask optimization
    """
    def make_binary_mask_v1(n):
        h = np.zeros(n)
        h[:n//2] = 1.0
        np.random.shuffle(h)
        return h

    def make_binary_mask_v2(n):
        h = np.zeros(n)
        # 5 intervals of alternating 1s and 0s
        for i in range(5):
            start = i * (n // 5)
            end = min((i + 1) * (n // 5), n)
            if i % 2 == 0:
                h[start:end] = 1.0
        return h

    def objective(mask):
        h = mask.astype(float)
        return c5_of_h(h)

    def swap_pos(mask, a, b):
        new_mask = mask.copy()
        new_mask[a], new_mask[b] = new_mask[b], new_mask[a]
        return new_mask
    
    def diff_corr(mask, a, b):
        h = mask.astype(float)
        new_h = h.copy()
        new_h[a], new_h[b] = new_h[b], new_h[a]
        return c5_of_h(new_h) - c5_of_h(h)

    masks = [make_binary_mask_v1(n_points), make_binary_mask_v2(n_points)]
    start_time = time.time()
    
    best_mask = masks[0]
    best_value = objective(best_mask)
    T = 0.02
    counter = 0
    accept_count = 0
    swap_count = 0
    
    for _ in range(1000000):
        if time.time() - start_time > max_time_s:
            break

        # propose a random swap
        mask = best_mask.copy()
        while True:
            a = np.random.randint(n_points)
            if mask[a] == 1:
                break
        while True:
            b = np.random.randint(n_points)
            if mask[b] == 0:
                break

        new_mask = swap_pos(mask, a, b)
        
        # compute loss difference
        change = diff_corr(mask, a, b)
        
        if change < 0 or np.random.rand() < np.exp(-change / T):
            best_mask = new_mask
            best_value = objective(best_mask)
        
        T *= 0.995
        counter += 1
        if counter % 100 == 0:
            swap_count += 100

    return best_mask, best_value

def linear_program_refine(binary_mask, n_points):
    """Use linear programming to refine the binary mask"""
    # Convert binary to integer
    b = binary_mask.astype(int)

    # Variables for h_i
    h = cp.Variable(n_points)
    t = cp.Variable(1)

    # Objective
    objective = cp.Minimize(t)

    constraints = [0 <= h[i] <= 1 for i in range(n_points)]
    constraints += [cp.sum(h) == n_points / 2]
    
    # For each shift k, we must have sum_{i} h[i] * (1 - h[i+k]) <= t
    # This uses a rounding approach to linearize: 
    # for i where b[i]=1, replace h[i]*h[i+k] by h[i] 
    # otherwise replace it by 0.
    dx = 2.0 / n_points
    
    corr = np.correlate(1-b, b, mode='full')
    shift_indices = range(-(n_points-1), n_points)
    
    for k in shift_indices:
        # For all i in range(n):
        # if i+k >= 0 and i+k < n:
        # then get correlation term: h[i] * (1 - b[i+k]) if b[i]=1
        # else if b[i] = 0, then h[i] * 1 = h[i] (or otherwise 0)
        expr = 0.0
        for i in range(n_points):
            if k >= 0:
                if i-k >= 0:
                    if b[i] == 1:
                        expr += h[i] * (1 - b[i-k])
                    else:
                        expr += h[i] * 1.0
                else:
                    expr += h[i]    
            else:
                # k < 0
                if i-k < n_points:
                    if b[i] == 1:
                        expr += h[i] * (1 - b[i-k])
                    else:
                        expr += h[i] * 1.0
                else:
                    expr += h[i]
                    
        constraints += [expr <= t]

    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return h.value
    except:
        pass
    return binary_mask

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 200
    dx = 2.0 / n_points

    # Run binary optimization phase
    binary_best_mask, binary_c5 = binary_optimize(n_points, max_time_s=300)
    best_c5 = binary_c5
    best_h = binary_best_mask.astype(float)
    
    # Linear program refinement
    h_refined = linear_program_refine(binary_best_mask, n_points)
    if h_refined is not None:
        c5_val = c5_of_h(h_refined)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_refined
        
    # Local perturbation
    for i in range(1000):
        if time.time() - start_time > budget_s - 100:
            break
        idxs = np.random.choice(n_points, min(3, n_points//10), replace=False)
        h_test = best_h.copy()
        for idx in idxs:
            delta = np.random.uniform(-0.01, 0.01)
            h_test[idx] += delta
        h_test = project_to_feasible(h_test)
        c5_test = c5_of_h(h_test)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()

    # Final upsampling and optimization
    if time.time() - start_time < budget_s - 100:
        # Interpolate to higher resolution
        x_coarse = np.linspace(0, 2, n_points)
        x_fine = np.linspace(0, 2, 400)
        h_upsampled = np.interp(x_fine, x_coarse, best_h)
        h_upsampled = h_upsampled * (400.0 / 2.0) / np.sum(h_upsampled)

        # Optimize with SLSQP using the upsampled version
        bounds = [(0, 1)] * 400
        eq_constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - 200.0, 'jac': lambda x: np.ones_like(x)}

        res = minimize(
            fun=c5_of_h,
            x0=h_upsampled,
            method='SLSQP',
            bounds=bounds,
            constraints=eq_constraint,
            options={'ftol': 1e-10, 'maxiter': 200, 'disp': False}
        )

        if res.success:
            h_final = res.x
        else:
            h_final = h_upsampled
        final_c5 = c5_of_h(h_final)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final

    return (best_h, best_c5, len(best_h))
