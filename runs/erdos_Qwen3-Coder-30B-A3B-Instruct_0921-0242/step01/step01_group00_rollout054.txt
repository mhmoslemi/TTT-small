import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from math import exp, sqrt

def project_to_feasible(h, n_points):
    """Project h to feasible region with sum = n_points/2 and bounds [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n_points / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Add diff/n to each element
    h += diff / n_points
    
    # Re-clip
    h = np.clip(h, 0, 1)
    
    return h

def block_to_h(block_lengths, block_heights, n_points):
    """Convert block parameters to step function h."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Set up cumulative block boundaries
    cumsum = np.concatenate([[0], np.cumsum(block_lengths)])
    assert abs(cumsum[-1] - 2.0) < 1e-10
    
    for i in range(n_points):
        x = i * dx
        # Find which block x falls into
        block_idx = np.searchsorted(cumsum, x, side='right') - 1
        if block_idx < len(block_heights):
            h[i] = block_heights[block_idx]
    
    return h

def compute_autocorrelation(h, n_eval=200):
    """Compute maximum overlap for given h."""
    dx = 2.0 / len(h)
    # Create correlation function for all shifts
    corr_full = correlate(h, 1 - h, mode='full')
    
    # We are interested in the overlap for shifts from 0 to 2 in steps
    # of dx, so check relevant midpoints of correlation
    k_max = len(corr_full) - 1
    max_k = k_max // 2
    # Max overlap is for shifts in [0,2], corresponding to full correlation indices from range
    # Let's directly compute for discrete shift points.
    ds = 2.0 / n_eval
    shifts = np.arange(0, 2 + ds/2, ds)
    
    max_overlap = 0.0
    
    for shift in shifts:
        # Map shift to appropriate k in correlation
        shift_idx = int(shift / dx)
        if shift_idx >= len(corr_full):
            shift_idx = len(corr_full) - 1
        elif shift_idx < 0:
            shift_idx = 0
        corr_val = corr_full[shift_idx]
        overlap = dx * (1 - corr_val)
        max_overlap = max(max_overlap, overlap)
    
    return max_overlap

def c5_objective_block_params(params, K, n_points=200, n_eval=200):
    """Objective function that evaluates C5 for block representation."""
    assert len(params) == 2 * K
    u = params[:K]
    v = params[K:]
    
    # Compute block lengths from u using softmax
    exp_u = np.exp(u)
    block_lengths = 2 * exp_u / np.sum(exp_u)
    
    # Compute block heights from v using sigmoid
    block_heights = 1.0 / (1.0 + np.exp(-v))
    
    # Ensure sum of h*len = 1 by scaling
    area_sum = np.sum(block_heights * block_lengths)
    if area_sum > 0:
        alpha = 1.0 / area_sum
        block_heights = np.minimum(1.0, alpha * block_heights)
        # Rebalance if necessary
        total_area = np.sum(block_heights * block_lengths)
        if abs(total_area - 1.0) > 1e-10:
            # Distribute deficit to other heights via reweighting
            excess_scaled = (1.0 - total_area) / np.sum(block_heights)
            block_heights = np.clip(block_heights + excess_scaled, 0, 1)
    
    # Create h function via block interpolation
    h = block_to_h(block_lengths, block_heights, n_points)
    
    # Compute overlap
    try:
        c5 = compute_autocorrelation(h, n_eval)
    except:
        c5 = 1.0  # worst case fallback
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define block parameters: K=6 blocks
    K = 6
    n_points = 400  # fine grid for final h
    n_eval = 200    # number of shifts to sample for overlap

    # Sampled from the blocked construction with random shifts
    def initialize_block_params():
        u = np.random.normal(0, 1, K)
        v = np.random.normal(0, 1, K)
        return np.concatenate([u, v])

    # Use initial guess
    initial_block_params = initialize_block_params()
    
    # Global search:
    # With 2*K params, population of size 10*K = 60
    bounds = [(-5, 5)] * (2 * K)  # Reasonable bounds on log lengths and logit heights

    # Function to wrap
    obj_wrapper = lambda x: c5_objective_block_params(x, K, n_points, n_eval)

    best_c5 = float('inf')
    best_indices = None
    
    #----------------------------------------------------------------------#
    # PHASE 1: GLOBAL OPTIMIZATION (DE)                                   #
    #----------------------------------------------------------------------#
    # We try DE only first (it's simpler)
    try:
        de_res = differential_evolution(obj_wrapper, bounds, seed=seed,
                                        maxiter=int(1e6), popsize=60,
                                        mutation=(0.5, 1.5), recombination=0.7,
                                        disp=False, polish=True,
                                        strategy='best1bin')
        
        if not de_res.success:
            pass  # Fall back below
        else:
            final_params = de_res.x
            # Evaluate this one to get true value
            test_c5 = c5_objective_block_params(final_params, K, n_points, n_eval)
            if test_c5 < best_c5:
                best_c5 = test_c5
                best_indices = final_params
        
        # Check for early termination
        if time.time() - start_time > 0.3 * budget_s:
            pass  # Continue to refine
            
    except:
        pass
    
    # Manually do one iteration just in case
    if best_indices is None:
        best_indices = initial_block_params.copy()
    
    #----------------------------------------------------------------------#
    # PHASE 2: DISCRETISATION AND PROJECTED GRADIENT                      #
    #----------------------------------------------------------------------#
    # Decode best block parameters to fine grid h
    u = best_indices[:K]
    v = best_indices[K:]
    exp_u = np.exp(u)
    block_lengths = 2 * exp_u / np.sum(exp_u)
    block_heights = 1.0 / (1.0 + np.exp(-v))
    
    # Re-scale for integral constraint
    area_sum = np.sum(block_heights * block_lengths)
    if area_sum > 0:
        alpha = 1.0 / area_sum
        block_heights = np.minimum(1.0, alpha * block_heights)
        # And balance it
        total_area = np.sum(block_heights * block_lengths)
        if abs(total_area - 1.0) > 1e-10:
            excess_scaled = (1.0 - total_area) / np.sum(block_heights)
            block_heights = np.clip(block_heights + excess_scaled, 0, 1)
    
    h = block_to_h(block_lengths, block_heights, n_points)
    h = project_to_feasible(h, n_points)
    
    #----------------------------------------------------------------------#
    # PHASE 3: PROJECTED GRADIENT DESCENT TO WORST SHIFT                  #
    #----------------------------------------------------------------------#
    current_c5 = compute_autocorrelation(h, n_eval)
    dx = 2.0 / n_points
    grad_step = 0.01

    max_iter = 5000
    for iter in range(max_iter):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Compute grad for worst shift
        corr = correlate(h, 1 - h, mode='full')
        max_k = len(corr) // 2
        # Find worst shift
        corr_shifts = np.arange(-max_k, max_k + 1)
        worst_idx = np.argmax(corr)
        worst_shift = corr_shifts[worst_idx]
        # Compute gradient (the gradient of the integral w.r.t. h_i)
        g = np.zeros(n_points)
        # Slow method for now (can be optimized further)
        for i in range(n_points):
            # Find the amount that A_{i+k} contributes to integral in shift k
            # Derivative of \int h(x)(1-h(x+k)) dx = dx (1 - h_{i+k} - h_{i-k})
            g[i] = dx * ((1 - h[(i + worst_shift) % n_points]) - (h[(i - worst_shift) % n_points]))
        
        # Gradient descent step
        h_new = h - grad_step * g
        h_new = project_to_feasible(h_new, n_points)
        
        new_c5 = compute_autocorrelation(h_new, n_eval)
        
        if new_c5 < current_c5 - 1e-10:
            h = h_new
            current_c5 = new_c5
        else:
            grad_step *= 0.9  # Reduce step size if not helping
            
    #----------------------------------------------------------------------#
    # PHASE 4: MEMETIC IMPROVEMENT + FINAL EVALUATION                  #
    #----------------------------------------------------------------------#
    # Simple stochastic hill climb with swaps
    for iter in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        i, j = np.random.randint(0, n_points, size=2)
        if h[i] > 0 and h[j] < 1:
            delta = min(h[i], 1.0 - h[j], 0.01)
            h[i] -= delta
            h[j] += delta
            h = project_to_feasible(h, n_points)
            new_c5 = compute_autocorrelation(h, n_eval)
            if new_c5 >= current_c5 - 1e-10:
                # undo
                h[i] += delta
                h[j] -= delta
                h = project_to_feasible(h, n_points)
            else:
                current_c5 = new_c5

    # Final check at highest resolution for true value
    final_c5 = compute_autocorrelation(h, 800)
    
    try:
        if final_c5 < 0.3807:
            return (h, final_c5, n_points)
    except:
        pass

    return (h, final_c5, n_points)
