import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import math

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Grid sizes to try
    n_values = [200, 250, 300]
    for n in n_values:
        if time.time() - start_time > budget_s - 5:
            break
        dx = 2.0 / n
        target_sum = n / 2.0
        
        # Step 1: Simulated Annealing for warm start
        h_heuristic = None
        best_h_heuristic = None
        best_c5_heuristic = float('inf')
        temp = 1.0
        cool_rate = 0.9995
        max_iter = int(1e6)
        if time.time() - start_time > budget_s - 5:
            break

        # Initialize a random binary vector with exactly target_sum ones
        h_start = np.zeros(n)
        ones_indices = np.random.permutation(n)[:int(target_sum)]
        h_start[ones_indices] = 1
        h_current = h_start.copy()
        current_c5 = c5_of_h(h_current, dx)
        
        if current_c5 < best_c5_heuristic:
            best_c5_heuristic = current_c5
            best_h_heuristic = h_current.copy()

        iter_count = 0
        while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
            i1 = np.random.randint(0, n)
            i0 = np.random.randint(0, n)
            if h_current[i1] == 0 or h_current[i0] == 1:
                if h_current[i1] == 0:
                    i1, i0 = i0, i1
                else:
                    pass
            if h_current[i1] == 1 and h_current[i0] == 0:
                h_trial = h_current.copy()
                h_trial[i1] = 0
                h_trial[i0] = 1
                trial_c5 = c5_of_h(h_trial, dx)
                delta = trial_c5 - current_c5
                if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                    h_current = h_trial
                    current_c5 = trial_c5
                    if current_c5 < best_c5_heuristic:
                        best_c5_heuristic = current_c5
                        best_h_heuristic = h_current.copy()
            temp *= cool_rate
            iter_count += 1

        h_heuristic = best_h_heuristic
        initial_c5 = best_c5_heuristic

        # Step 2: Integer Programming with core shift constraints
        # Just doing it for 300 points, with core shifts only
        if n != 300:
            continue
        # Core shifts
        core_shifts = [1, n//4, n//2, 3*n//4]
    
        # Formulate and solve BIP
        try:  # This is a Problem with possibly memory limits
            # Variables:
            h = Variable(n, boolean=True)
            y = Variable(int(n*(n-1)/2), boolean=True)  # Products
            t = Variable()
            
            # Sum constraint
            constraints = [sum(h) == target_sum]
            
            # Linearization: y[i,j] = h[i]*h[j] iff i < j
            y_idx = 0
            for i in range(n):
                for j in range(i+1, n):
                    k = j-i
                    if k in core_shifts:
                        # Constraint:
                        # y[i,j] <= h[i]
                        # y[i,j] <= h[j]
                        # y[i,j] >= h[i] + h[j] - 1
                        # This is a binary variable
                        constraints.append(y[y_idx] <= h[i])
                        constraints.append(y[y_idx] <= h[j])
                        constraints.append(y[y_idx] >= h[i] + h[j] - 1)
                    else:
                        pass  # Skip for non-core shifts currently
                    y_idx += 1
            
            # Core shift constraints:
            y_idx = 0
            shift_idx = 0
            for i in range(n):
                for j in range(i+1, n):
                    k = j-i
                    if k in core_shifts:
                        # Sum of y terms for shift k
                        sum_y_k = y[y_idx]  # Already the y_i,j variables
                        constraints.append(t + sum_y_k >= target_sum)
                        y_idx += 1
            constraints.append(t >= 0)
            
            # Objective
            objective = Minimize(t)
            prob = Problem(objective, constraints)
            
            # Setting solver cutoff to half remaining time
            time_remaining = budget_s - (time.time() - start_time) - 1
            if time_remaining <= 0.1:
                break
            # Using CBC solver
            prob.solve(solver=CPLEX, maxSeconds=time_remaining, verbose=False)
            if prob.status == 'optimal':
                h_opt = np.array([float(h[i].value) for i in range(n)])
                t_opt = t.value
                if t_opt < best_c5:
                    best_c5 = t_opt
                    best_h = h_opt.copy()
                    best_n = n
        except Exception as e:
            pass

        # If MIP failed, fallback to heuristic
        if best_h is None or best_c5 > initial_c5:
            best_h = h_heuristic
            best_c5 = initial_c5
            best_n = n
        else:
            pass

    # Local refinement: try swaps around the best solution
    if best_h is not None and (time.time() - start_time) < budget_s - 1:
        h_refined = best_h.copy()
        current_c5 = c5_of_h(h_refined, dx)
        step_limit = 5000
        for _ in range(step_limit):
            if (time.time() - start_time) > budget_s - 1:
                break
            i1 = np.random.randint(0, n)
            i0 = np.random.randint(0, n)
            if h_refined[i1] == 0 or h_refined[i0] == 1:
                if h_refined[i1] == 0:
                    i1, i0 = i0, i1
                else:
                    pass
            if h_refined[i1] == 1 and h_refined[i0] == 0:
                h_trial = h_refined.copy()
                h_trial[i1] = 0
                h_trial[i0] = 1
                trial_c5 = c5_of_h(h_trial, dx)
                if trial_c5 < current_c5:
                    h_refined = h_trial
                    current_c5 = trial_c5

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined

    # Return best
    return (best_h, best_c5, best_n)
