import numpy as np
import time
from scipy.signal import fftconvolve
import math

def binarize_h(h, target_sum):
    """Convert continuous h to binary with exact sum constraint."""
    h = np.clip(h, 0, 1)
    # Sort indices by decreasing value
    idx_sorted = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    # Set top target_sum entries to 1
    for i in range(int(target_sum)):
        h_bin[idx_sorted[i]] = 1
    return h_bin

def evaluate_overlap(h, dx):
    """Fast evaluation of C5 using FFT."""
    # Use fftconvolve for fast convolution
    corr = fftconvolve(h, 1 - h, mode='full')
    return np.max(corr) * dx

def update_convolution_fast(h, corr, i1, i0, n):
    """Incrementally update convolution after a swap."""
    # Precompute shifted versions once for efficiency
    h1 = h[i1]
    h0 = h[i0]
    # New values after swap
    new_h1 = 1 - h1
    new_h0 = 1 - h0
    
    # Update corr vector: subtract old contributions and add new ones
    # Each element in corr is affected by shift from index
    # The change in corr due to swapping h[i1] and h[i0]
    delta_corr = np.zeros(2 * n - 1)
    
    # The contribution of h[i1] to convolution
    if i1 < n:
        # 1 - h shifted to position i1
        if i1 == 0:
            delta_corr[:n] -= 1 - h
        else:
            delta_corr[i1:i1+n] -= 1 - h
    # The contribution of h[i0] to convolution
    if i0 < n:
        # 1 - h shifted to position i0
        if i0 == 0:
            delta_corr[:n] -= 1 - h
        else:
            delta_corr[i0:i0+n] -= 1 - h

    # Add new contributions
    # 1 - h0 shifted to position i1
    if i1 < n:
        if i1 == 0:
            delta_corr[:n] += 1 - h
        else:
            delta_corr[i1:i1+n] += 1 - h
    # 1 - h1 shifted to position i0
    if i0 < n:
        if i0 == 0:
            delta_corr[:n] += 1 - h
        else:
            delta_corr[i0:i0+n] += 1 - h
            
    # Update the sums
    new_corr = corr + delta_corr
    return new_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    np.random.seed(42)
    
    n_list = [220, 260, 300]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    max_swap_attempts = 5 * 10**4  # Safe limit
        
    target_sum = None
    initial_h = None
    
    # Handle the initial_h_values provided
    assert 'initial_h_values' in globals(), "initial_h_values not available"
    # We'll approximate it as:
    # interpolate to 300 points:
    h_orig = initial_h_values
    n_points_orig = len(h_orig)
    x_orig = np.linspace(0, 2, n_points_orig)
    x_new = np.linspace(0, 2, 300)
    h_interp = np.interp(x_new, x_orig, h_orig)
    initial_h = h_interp.copy()
    
    for n in n_list:
        if time.time() - start_time > budget_s - 5:
            break
            
        dx = 2.0 / n
        target_sum = n / 2.0
        
        # Create multiple initial states
        initial_states = []
        
        # Interpolated from initial_h_values
        h_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, 300), initial_h)
        h_bin = binarize_h(h_interp, target_sum)
        initial_states.append(h_bin)
        
        # Random binary vector
        h_rand = np.random.rand(n)
        h_bin_rand = binarize_h(h_rand, target_sum)
        initial_states.append(h_bin_rand)
        
        # Alternating pattern (e.g., 1,0,1,0,...)
        h_alt = np.ones(n)
        h_alt[1::2] = 0
        initial_states.append(h_alt)
        
        # Two-block pattern (with shuffle)
        h_block = np.zeros(n)
        h_block[:n//2] = 1
        np.random.shuffle(h_block)
        initial_states.append(h_block)
        
        temp_rates = [0.9995] * 4  # P chains at different temps
        T_start = 0.05
        temp_schedule = [T_start * (temp_rates[0] ** i) for i in range(10000)]
        
        num_chains = 4
        # Use 4 independent chains
        num_sweeps_per_chain = 1000
        swap_interval = 50
        
        num_sweeps_per_chain = 500  # not too much, aim for budget
        best_c5_each_n = float('inf')
        best_h_each_n = None
        
        # Run each chain
        for init_idx, h_start in enumerate(initial_states[:2]):  # up to 2
            if time.time() - start_time > budget_s - 5:
                break
                
            # Initialize a chain
            h_current = h_start.copy().astype(np.uint8)
            current_c5 = evaluate_overlap(h_current, dx)
            current_corr = fftconvolve(h_current, 1 - h_current, mode='full')
            best_this_chain = current_c5
            h_best_chain = h_current.copy()
            T = T_start
            
            swap_counter = 0
            
            # Monitor for stagnation
            last_improvement = 0  # number of sweeps since last improvement
            
            for sweep in range(num_sweeps_per_chain):
                if time.time() - start_time > budget_s - 5:
                    break
                    
                # Perform swaps
                if swap_counter >= max_swap_attempts:
                    break
                    
                # Perform about n/2 swaps per sweep for simplicity
                num_swap_per_sweep = n//2
                for _ in range(num_swap_per_sweep):
                    # Pick two indices such that one is 1 and the other is 0
                    ones = np.where(h_current == 1)[0]
                    zeros = np.where(h_current == 0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i1 = np.random.choice(ones)
                    i0 = np.random.choice(zeros)
                    
                    # Compute proposed new overlap value using efficient update
                    old_corr = current_corr.copy()
                    try:
                        current_corr = update_convolution_fast(h_current, current_corr, i1, i0, n)
                    except Exception:
                        # Fall back to exact computation
                        current_corr = fftconvolve(h_current, 1 - h_current, mode='full')
                    
                    new_c5 = np.max(current_corr) * dx
                    
                    # Acceptance rule
                    delta = new_c5 - current_c5
                    
                    if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                        # Accept
                        h_current[i1] = 0
                        h_current[i0] = 1
                        current_c5 = new_c5
                        if current_c5 < best_this_chain:
                            best_this_chain = current_c5
                            h_best_chain = h_current.copy()
                        last_improvement = 0
                    else:
                        # Update correlation back to old value
                        current_corr = old_corr
                    swap_counter += 1
                    
                    if swap_counter >= max_swap_attempts:
                        break
                        
                # Update temperature
                if sweep % 10 == 0:
                    T = T * 0.9995
                    
                if sweep % swap_interval == 0:
                    # Reject any temperature that might be extremely low
                    if T < 1e-5:
                        break
                        
                last_improvement += 1
                if last_improvement > 150:
                    # Restart with new random seed
                    h_current = np.random.rand(n)
                    h_current = binarize_h(h_current, target_sum)
                    current_c5 = evaluate_overlap(h_current, dx)
                    current_corr = fftconvolve(h_current, 1 - h_current, mode='full')
                    last_improvement = 0
                    
            if best_this_chain < best_c5_each_n:
                best_c5_each_n = best_this_chain
                best_h_each_n = h_best_chain.copy()
                
        if best_c5_each_n < best_c5:
            best_c5 = best_c5_each_n
            best_h = best_h_each_n
            best_n = n
            
        # Early stopping if we've already beaten target
        if best_c5 <= 0.3807:
            break
                
    # Deterministic improvement
    if best_h is not None and time.time() - start_time < budget_s - 1:
        h_improved = best_h.copy()
        current_c5 = evaluate_overlap(h_improved, dx)
        improvements = True
        iter_count = 0
        while improvements and time.time() - start_time < budget_s - 1 and iter_count < 1000:
            iter_count += 1
            improvements = False
            # Enumerate all possible swaps
            ones = np.where(h_improved == 1)[0]
            zeros = np.where(h_improved == 0)[0]
            for i1 in ones:
                for i0 in zeros:
                    h_temp = h_improved.copy()
                    h_temp[i1] = 0
                    h_temp[i0] = 1
                    
                    temp_c5 = evaluate_overlap(h_temp, dx)
                    if temp_c5 < current_c5:
                        current_c5 = temp_c5
                        h_improved = h_temp
                        improvements = True
                        break
            if not improvements:
                break
                
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_improved.copy()
            
    # Return result
    return (best_h, best_c5, best_n)
