import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def project_to_feasible(u, n):
    """Project vector u onto the feasible set: 0 <= h[i] <= 1, sum h = n/2"""
    u = np.clip(u, 0.0, 1.0)
    s = np.sum(u)
    if abs(s - n/2) < 1e-12:
        return u
    
    # Use bisection to find appropriate lambda
    def sum_with_lambda(lam):
        return np.sum(np.clip(u + lam, 0.0, 1.0)) - n/2
    
    # Find bounds for lambda
    lo, hi = -1.0, 1.0
    while sum_with_lambda(lo) > 0:
        lo -= 1.0
    while sum_with_lambda(hi) < 0:
        hi += 1.0
    
    # Bisection
    for _ in range(30):
        mid = (lo + hi) / 2
        val = sum_with_lambda(mid)
        if abs(val) < 1e-10:
            break
        if val > 0:
            hi = mid
        else:
            lo = mid
    
    u = np.clip(u + (lo + hi)/2, 0.0, 1.0)
    return u

def c5_of_h(h):
    """Compute C5 for a given step function h using FFT"""
    dx = 2.0 / len(h)
    # FFT-based correlation
    H = np.fft.rfft(h, n=2*len(h)-1)
    G = np.fft.rfft(1-h, n=2*len(h)-1)
    conv = np.fft.irfft(H*G)[:2*len(h)-1]
    return dx * np.max(conv)

def c5_of_h_pruned(h, best_so_far):
    """Compute C5 with early-stop pruning"""
    dx = 2.0 / len(h)
    # FFT-based correlation
    H = np.fft.rfft(h, n=2*len(h)-1)
    G = np.fft.rfft(1-h, n=2*len(h)-1)
    conv = np.fft.irfft(H*G)[:2*len(h)-1]
    max_val = np.max(conv)
    if max_val > best_so_far / dx:
        return float('inf')  # Early stop
    return dx * max_val

def symmetrize(h):
    """Make h symmetric: h[i] = h[n-1-i] after projected"""
    n = len(h)
    if n % 2 == 0:
        mid = n // 2
        h_sym = np.zeros(n)
        for i in range(mid):
            mean_val = (h[i] + h[n-1-i]) / 2
            h_sym[i] = mean_val
            h_sym[n-1-i] = mean_val
    else:
        mid = n // 2
        h_sym = np.zeros(n)
        for i in range(mid):
            mean_val = (h[i] + h[n-1-i]) / 2
            h_sym[i] = mean_val
            h_sym[n-1-i] = mean_val
        h_sym[mid] = h[mid]
    return h_sym

def smooth_max_grad(h, beta, dx):
    """Gradient of smooth-max approximation to C5 with β-continuation"""
    n = len(h)
    # Convolve with 1-h and h for gradient computation
    H = np.fft.rfft(h, n=2*n-1)
    G1 = np.fft.rfft(1-h, n=2*n-1)
    G2 = np.fft.rfft(h, n=2*n-1)
    
    conv1 = np.fft.irfft(H*G1)[:2*n-1]
    conv2 = np.fft.irfft(H*G2)[:2*n-1]
    
    # Weight with exp(beta * dx * conv_k)  
    exp_conv = np.exp(beta * dx * conv1)
    w = exp_conv / np.sum(exp_conv)
    
    # Use convolution trick: 
    # grad_i = dx * sum_k w_k * (1 - h_{i+k}) - h_{i-k}
    # This gives two shifts with proper extrapolation
    
    w_pad = np.pad(w, (0, n-1), constant_values=0)  # for sum k >= 0
    h_pad = np.pad(h, (0, n-1), constant_values=0)
    h_pad2 = np.pad(h, (n-1, 0), constant_values=0)
    one_h_pad = np.pad(1-h, (0, n-1), constant_values=0)
    
    grad1 = np.convolve(w_pad[:n], one_h_pad[:n], mode='full')[:n]  # for +k terms 
    grad2 = np.convolve(w_pad[n-1:n], h_pad2[:n], mode='full')[:n]  # for -k terms 
    
    grad = dx * (grad1 - grad2)
    return grad

def local_refine(h, dx, beta_init=0.5, beta_max=50.0, max_iter=80):
    """Local refinement via smooth max with β continuation"""
    h = h.copy()
    beta = beta_init
    n = len(h)
    lambda_reg = 10.0
    
    for _ in range(10000):  # Max iterations, but stop early
        grad = smooth_max_grad(h, beta, dx)
        # Gradient step with line search (simple fixed step)
        step_size = 1e-3
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n)
        # Symmetrize
        h_new = symmetrize(h_new)
        # Project again to enforce sum = n/2 after symmetrization
        h_new = project_to_feasible(h_new, n)
        
        # Add equality constraint penalty and project
        h = h_new
        
        if beta < beta_max:
            beta *= 2
            lambda_reg *= 2
        else:
            break
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Coarse CMA-ES with n=150
    n_coarse = 150
    dx_coarse = 2.0 / n_coarse
    
    # Symmetric region: first n//2 entries
    n_sym = (n_coarse + 1) // 2
    μ = 0.5 * np.ones(n_sym)
    σ = 0.2
    
    # CMA-ES parameters
    λ = int(4 + 3 * np.log(n_sym))  # Population size
    μ_eff = λ / (2 * np.log(λ))
    c_1 = 2 / (n_sym + 2)
    c_μ = 2 / (n_sym + 2)
    
    # Initialize
    p_c = np.zeros(n_sym)
    p_s = np.zeros(n_sym)
    C = np.eye(n_sym)
    σ_min = 1e-8
    best_c5_coarse = float('inf')
    best_h_coarse = None
    no_improvement_count = 0
    
    # Conditionally restart
    restart_threshold = 1e-4
    max_restarts = 3
    num_restarts = 0
    
    # Precompute convolutions for early stopping
    best_coarse_value = float('inf')
    
    while time.time() - start_time < budget_s - 50:
        # Generate offspring
        pop = []
        pop_c5 = []
        for _ in range(λ):
            z = np.random.randn(n_sym)
            x = μ + σ * z
            
            # Symmetrize x to get full h
            h_full = np.zeros(n_coarse)
            h_full[:n_sym] = x
            if n_coarse % 2 == 0:
                h_full[n_sym:] = x[:0:-1]
            else:
                h_full[n_sym:] = x[:n_sym-1:-1]
            
            h_full = project_to_feasible(h_full, n_coarse)
            c5_val = c5_of_h_pruned(h_full, best_coarse_value)
            if c5_val < float('inf'):
                pop.append(h_full)
                pop_c5.append(c5_val)
            else:
                pop_c5.append(float('inf'))
        
        if not pop:
            continue
            
        # Sort by objective
        sort_idx = np.argsort(pop_c5)
        pop = [pop[i] for i in sort_idx]
        pop_c5 = [pop_c5[i] for i in sort_idx]
        
        # Keep best ones
        pop = pop[:int(λ/2)]
        pop_c5 = pop_c5[:int(λ/2)]
        
        # Update best
        if pop_c5 and pop_c5[0] < best_c5_coarse:
            best_c5_coarse = pop_c5[0]
            best_h_coarse = pop[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > 200:
            if num_restarts < max_restarts:
                μ = best_h_coarse[:n_sym]
                σ = σ / 2
                no_improvement_count = 0
                num_restarts += 1
            else:
                break
        
        # Update parameters
        y_k = np.zeros(n_sym)
        for i in range(len(pop)):
            diff = pop[i][:n_sym] - μ
            y_k += diff / σ
        
        p_s = (1-c_1) * p_s + np.sqrt(c_1*(2-c_1)*μ_eff) * y_k / np.linalg.norm(y_k)
        
        # Update C matrix
        rank_one = np.outer(p_s, p_s)
        rank_mu = np.zeros((n_sym, n_sym))
        for i in range(len(pop)):
            diff = (pop[i][:n_sym] - μ) / σ
            rank_mu += 1.0/len(pop) * np.outer(diff, diff)
        
        C = (1 - c_1 - c_μ) * C + c_1 * rank_one + c_μ * rank_mu
        _, eigenvectors = np.linalg.eigh(C)
        
        # Update the solution vector μ
        μ = μ + σ * (np.sqrt(c_1*(2-c_1)*μ_eff) * y_k / np.linalg.norm(y_k))
        
        # Update σ
        σ = σ * np.exp(1/4 * (np.sqrt(np.sum(p_s**2)) / np.sqrt(n_sym) - 1)/2)
        
        if σ < σ_min:
            σ = σ_min
    
    if best_h_coarse is None:
        best_h_coarse = np.ones(n_coarse) * 0.5
    
    # Phase 2: Interpolation to fine grid
    h_interpolated = np.zeros(300)
    x_orig = np.linspace(0, 2, n_coarse)
    x_new = np.linspace(0, 2, 300)
    u_interp = interp1d(x_orig, best_h_coarse, kind='linear', fill_value='extrapolate')
    h_interpolated = u_interp(x_new)
    h_interpolated = project_to_feasible(h_interpolated, 300)
    h_interpolated = symmetrize(h_interpolated)
    h_interpolated = project_to_feasible(h_interpolated, 300)
    
    # Phase 3: β continuation local refinement
    refined = local_refine(h_interpolated, 2.0/300)
    
    # Final polishing with hill climb 
    h_best = refined.copy()
    c5_best = c5_of_h(h_best)
    
    # Final hill climb to fine tune
    remaining_budget = budget_s - (time.time() - start_time)
    if remaining_budget > 1.0:
        p = min(6, 300 // 20)
        for _ in range(5000):
            if time.time() - start_time > budget_s - 0.5:
                break
            # Select indices at random 
            indices = np.random.choice(300, size=p, replace=False)
            
            # Generate trial
            h_trial = h_best.copy()
            for i in indices:
                δ = np.random.uniform(-0.004, 0.004)
                h_trial[i] = np.clip(h_trial[i] + δ, 0, 1)
            h_trial = project_to_feasible(h_trial, 300)
            h_trial = symmetrize(h_trial)
            h_trial = project_to_feasible(h_trial, 300)
            
            c5_trial = c5_of_h(h_trial)
            if c5_trial < c5_best:
                h_best = h_trial.copy()
                c5_best = c5_trial
    
    return (h_best, c5_best, 300)
