from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.special import softmax

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Adjust by distributing difference
    h = h + diff / n
    
    # Re-clip
    h = np.clip(h, 0, 1)
    
    return h

def block_to_h(u, v, K, n_points):
    """Convert block parameters to step function h."""
    # Compute lengths from softmax of u
    l = 2 * softmax(u)
    
    # Compute provisional heights
    h_provisional = 1.0 / (1.0 + np.exp(-v))
    
    # Compute scaling factor
    area = np.sum(h_provisional * l)
    scale = 1.0 / area
    
    # Apply scaling with clamping
    h = np.minimum(1.0, scale * h_provisional)
    
    # Ensure integral is exactly correct
    h = project_to_feasible(h, n_points)
    return h

    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr
