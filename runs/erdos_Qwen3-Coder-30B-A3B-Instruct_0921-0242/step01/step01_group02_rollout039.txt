import numpy as np
import time
from scipy.signal import fftconvolve
import math

def binary_to_step_function(h_binary, n_points):
    """Convert binary vector to piecewise constant step function."""
    dx = 2.0 / n_points
    h_fine = np.zeros(n_points)
    i = 0
    while i < n_points:
        if h_binary[i] == 1:
            j = i
            while j < n_points and h_binary[j] == 1:
                j += 1
            h_fine[i:j] = 1
            i = j
        else:
            j = i
            while j < n_points and h_binary[j] == 0:
                j += 1
            i = j
    return h_fine

def compute_c5_from_binary(h_binary):
    """Fast C5 computation via FFT convolution."""
    n = len(h_binary)
    dx = 2.0 / n
    # For numerical stability, we use float in the computation
    h = h_binary.astype(float)
    g = 1.0 - h
    # Full convolution
    corr = fftconvolve(h, g, mode='full')
    # Equivalent to max_k integrate h(x)(1-h(x+k)) = max_k (h*k)(k)
    max_corr = np.max(corr)
    return max_corr * dx

def compute_c5_incremental_update(h_binary, corr_old, i1, i0, n):
    """Update the convolution vector after a swap at positions i1->1 and i0->0."""
    # Convert to floats
    h = h_binary.astype(float)
    g = 1.0 - h
    
    # Old contributions to correlation
    old_contributions = np.zeros(2*n - 1)
    old_contributions[i1:i1+n] += (1 - h[i1]) * g
    old_contributions[i0:i0+n] += (1 - h[i0]) * g
    
    # New contributions to correlation  
    new_contributions = np.zeros(2*n - 1)
    new_contributions[i1:i1+n] += (1 - h[i1]) * (1 - g)
    new_contributions[i0:i0+n] += (1 - h[i0]) * (1 - g)
    
    # Difference
    delta = new_contributions - old_contributions
    
    # Compute new correlation
    corr_new = corr_old - delta
    
    return corr_new

def sample_initial_binary_vectors(n):
    """Generate different types of initial balanced binary vectors."""
    h_list = []
    
    # 1. Initial state with high resolution
    h_orig = initial_h_values
    h_orig_len = len(h_orig)
    x_orig = np.linspace(0, 2, h_orig_len)
    x_new = np.linspace(0, 2, n)
    h_interp = np.interp(x_new, x_orig, h_orig)
    h_interp = (h_interp > np.percentile(h_interp, 50 - 50 * n / h_orig_len)).astype(int)
    h_list.append(h_interp)
    
    # 2. Random balanced binary vector
    h_random = np.zeros(n, dtype=np.uint8)
    h_random[:n//2] = 1
    np.random.shuffle(h_random)
    h_list.append(h_random)
    
    # 3. Alternating pattern
    h_alternating = np.zeros(n, dtype=np.uint8)
    h_alternating[::2] = 1
    h_list.append(h_alternating)
    
    # 4. Two block pattern with slight shuffling
    h_two_block = np.zeros(n, dtype=np.uint8)
    h_two_block[n//4:3*n//4] = 1
    # Slightly shuffle the boundary
    for i in range(n//4, n//4 + 5):
        if i < n and np.random.rand() > 0.5:
            h_two_block[i] = 1 - h_two_block[i]
    for i in range(3*n//4 - 5, 3*n//4):
        if i >= 0 and np.random.rand() > 0.5:
            h_two_block[i] = 1 - h_two_block[i]
    h_list.append(h_two_block)
    
    return h_list

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h_binary = None
    best_n = None
    
    # Grid sizes to try, all even
    n_list = [220, 260, 300]
    
    # Parameters for simulated annealing
    T0 = 0.05
    alpha = 0.9995
    T_min = 1e-5
    sweeps_per_temp = 500
    num_temps = 4
    swap_interval = 50
    
    for n in n_list:
        if time.time() - start_time > budget_s - 5:
            break
        dx = 2.0 / n
        target_sum = n // 2
        
        # Get initial binary vectors
        initial_h_list = sample_initial_binary_vectors(n)
        
        # Track best for this n
        n_best_c5 = float('inf')
        n_best_h_binary = None
        
        # Run multiple chains for parallel tempering
        swap_positions = list(range(n))
        np.random.shuffle(swap_positions)
        
        # Parallel tempering: multiple chains
        num_chains = num_temps
        chains = []
        temps = []
        for i in range(num_chains):
            temp = T0 * (alpha ** (i * 5))
            temps.append(temp)
            chains.append(None)  # Will be initialized
            
        # Initialize chains with best feasible starting configurations
        worst_h = None
        worst_c5 = -1
        for h_binary in initial_h_list:
            sum_h = np.sum(h_binary)
            if abs(sum_h - target_sum) < 1e-10:
                c5 = compute_c5_from_binary(h_binary)
                if c5 > worst_c5:
                    worst_h = h_binary.copy()
                    worst_c5 = c5
            
        if worst_h is not None:
            # Initialize all chains at same starting configuration
            for i in range(num_chains):
                chains[i] = worst_h.copy()
                if i == 0:
                    chain_temp = 0.05
                else:
                    chain_temp = temps[i]
                chains[i] = worst_h.copy()
                # Store indices of 1s and 0s for quick updates
                ones_idxs = np.where(chains[i] == 1)[0]
                zeros_idxs = np.where(chains[i] == 0)[0]
                chan = {
                    'h': chains[i],
                    'ones': ones_idxs,
                    'zeros': zeros_idxs,
                    'temp': chain_temp,
                    'c5': c5,
                    'last_improve': 0
                }
        else:
            # Fallback to random balanced vector
            h_initial = np.zeros(n, dtype=np.uint8)
            h_initial[:n//2] = 1
            np.random.shuffle(h_initial)
            c5_start = compute_c5_from_binary(h_initial)
            ones_idxs = np.where(h_initial == 1)[0]
            zeros_idxs = np.where(h_initial == 0)[0] 
            for i in range(num_chains):
                if i == 0:
                    chain_temp = 0.05
                else:
                    chain_temp = temps[i]
                chan = {
                    'h': h_initial.copy(),
                    'ones': ones_idxs,
                    'zeros': zeros_idxs,
                    'temp': chain_temp,
                    'c5': c5_start,
                    'last_improve': 0
                }
                chains[i] = h_initial.copy()
                
        # Actual SA process on chains
        ms_counter = 0
        max_swaps_per_chain = 2 * n * 20000  # bits * max_iterations
        
        # For deterministic refinement later
        refinement_h = None
        refinement_c5 = float('inf')
        
        while True:
            if time.time() - start_time > budget_s - 5:
                break
            if ms_counter > max_swaps_per_chain * num_chains:
                break
                
            # Run one iteration across all chains
            # Each chain attempts sweep_per_temp swaps
            for c_idx in range(num_chains):
                if time.time() - start_time > budget_s - 5:
                    break
                if ms_counter > max_swaps_per_chain * num_chains:
                    break
                    
                c_temp = temps[c_idx]
                cur_h = chains[c_idx] 
                
                # Compute current C5
                corr = fftconvolve(cur_h.astype(float), 1.0-cur_h.astype(float), mode='full')
                cur_c5 = np.max(corr) * dx
                
                # Try sweep moves for this chain
                for i in range(sweeps_per_temp):
                    if time.time() - start_time > budget_s - 5:
                        break
                    if ms_counter >= max_swaps_per_chain * num_chains:
                        break
                        
                    ms_counter += 1
                    
                    # Find indices with 1s and 0s
                    ones_list = np.where(cur_h == 1)[0]
                    zeros_list = np.where(cur_h == 0)[0]
                    if len(ones_list) == 0 or len(zeros_list) == 0:
                        break
                        
                    # Pick random indices to swap
                    idx1 = np.random.choice(ones_list)
                    idx0 = np.random.choice(zeros_list)
                    
                    # Make temporary swap
                    new_h = cur_h.copy()
                    new_h[idx1] = 0
                    new_h[idx0] = 1
                    
                    # Compute new C5
                    new_corr = fftconvolve(new_h.astype(float), 1.0-new_h.astype(float), mode='full')
                    new_c5 = np.max(new_corr) * dx
                    
                    # Accept or reject
                    delta = new_c5 - cur_c5
                    if delta <= 0 or np.random.rand() < np.exp(-delta / c_temp):
                        # Accept
                        cur_h[:] = new_h[:]
                        cur_c5 = new_c5
                        # Update listing
                        ones_list = np.where(cur_h == 1)[0] 
                        zeros_list = np.where(cur_h == 0)[0]
                        # Update chains array directly
                        chains[c_idx] = cur_h
                        
                        # Track improvement //
                        if new_c5 < n_best_c5:
                            n_best_h_binary = cur_h.copy()
                            n_best_c5 = new_c5
                            
                        if new_c5 < refinement_c5:
                            refinement_h = cur_h.copy()
                            refinement_c5 = new_c5
                        
                    else:
                        # No change
                        pass
                        
            # Periodically exchange states between temperature levels
            if ms_counter % (swap_interval * num_chains) == 0:
                # Metropolis exchange of adjacent temperatures
                for i in range(num_chains - 1):
                    if time.time() - start_time > budget_s - 5:
                        break
                    t1 = temps[i]
                    t2 = temps[i+1]
                    h1 = chains[i]
                    h2 = chains[i+1]
                    if h1 is not None and h2 is not None:
                        corr1 = fftconvolve(h1.astype(float), 1.0-h1.astype(float), mode='full')
                        c1 = np.max(corr1) * dx
                        
                        corr2 = fftconvolve(h2.astype(float), 1.0-h2.astype(float), mode='full')
                        c2 = np.max(corr2) * dx
                        
                        delta = (c1 - c2) * (1/t1 - 1/t2)
                        if delta <= 0 or np.random.rand() < np.exp(-delta):
                            # Exchange
                            chains[i] = h2
                            chains[i+1] = h1
                            
            # Temperature cooling
            if c_temp > T_min:
                for i in range(num_chains):
                    temps[i] = temps[i] * alpha
                    
            # Check progress
            if n_best_c5 < 0.3807:
                break
                
        # Keep this best result overall
        if n_best_c5 < best_c5:
            best_c5 = n_best_c5
            best_h_binary = n_best_h_binary
            best_n = n
            
        if best_c5 < 0.3807:
            break
    
    # Now do deterministic local search to improve further
    if best_h_binary is not None and time.time() - start_time < budget_s - 2:
        # Local improvement phase - hill climb on good solution
        local_h = best_h_binary.copy()
        corr = fftconvolve(local_h.astype(float), 1.0-local_h.astype(float), mode='full')
        local_c5 = np.max(corr) * dx
        
        improved = True
        tries = 0
        while improved and tries < 5000:
            improved = False
            tries += 1
            # Enumerate all swaps that might improve
            ones_list = np.where(local_h == 1)[0]
            zeros_list = np.where(local_h == 0)[0]
            
            for i1 in ones_list:
                for i0 in zeros_list:
                    if i1 == i0:
                        continue
                    # Proposal
                    candidate = local_h.copy() 
                    candidate[i1] = 0
                    candidate[i0] = 1
                    
                    # Check if better
                    new_corr = fftconvolve(candidate.astype(float), 1.0-candidate.astype(float), mode='full')
                    c_candidate = np.max(new_corr) * dx
                    
                    if c_candidate < local_c5:
                        local_h[:] = candidate[:]
                        local_c5 = c_candidate
                        improved = True
                        break
                if improved:
                    break
                    
    # Final check
    if best_h_binary is not None:
        h_fine = binary_to_step_function(best_h_binary, 2000)
        # Integrate in return format with fine grid
        dx_new = 2.0 / 2000
    else:
        h_fine = None
        dx_new = 2.0 / 300
        
    return (best_h_binary, best_c5, best_n)
