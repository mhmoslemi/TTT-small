import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from math import exp, log
import heapq

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_block_autocorrelation(block_lengths, block_heights, n_eval=200):
    """
    Compute the maximum overlap for given block parameters.
    """
    K = len(block_lengths)
    # Compute block endpoints
    block_ends = np.cumsum(block_lengths)
    block_starts = np.concatenate([[0], block_ends[:-1]])
    
    # Create the shifted version
    dx = 2.0 / n_eval
    # Shifts (in steps of dx)
    shifts = np.arange(n_eval) * dx
    # Vectorized overlap computation (slightly adjusted to avoid overhead with small K)
    max_value = 0.0
    for s in shifts:
        total = 0.0
        for i in range(K):
            for j in range(K):
                left_i, right_i = block_starts[i], block_ends[i]
                left_j, right_j = block_starts[j] - s, block_ends[j] - s
                overlap = max(0.0, min(right_i, right_j) - max(left_i, left_j))
                total += block_heights[i] * block_heights[j] * overlap
        # Compute integral of h(x) * (1 - h(x+s))
        value = 1.0 - total / 2.0  # extra 2 from integral normalizations
        if value > max_value:
            max_value = value
    return max_value

def sample_from_blocks(block_lengths, block_heights, n_points):
    """Sample block structure to fine-grid vector."""
    h = np.zeros(n_points)
    total_length = sum(block_lengths)
    # Check unitization
    if abs(total_length - 2.0) > 1e-10:
        block_lengths = np.array(block_lengths) * 2.0 / total_length
    
    block_ends = np.cumsum(block_lengths)
    block_starts = np.concatenate([[0], block_ends[:-1]])
    
    dx = 2.0 / n_points
    
    for i in range(n_points):
        x = i * dx
        # Find which block contains x
        for j in range(len(block_starts)):
            if block_starts[j] <= x < block_ends[j]:
                h[i] = block_heights[j]
                break
    return h

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2 with h in [0,1] using bisection."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    desired_sum = n / 2.0
    
    if abs(current_sum - desired_sum) < 1e-10:
        return h
    
    # Binary search for lambda (delta to add to all entries)
    low, high = -1.0, 1.0
    for _ in range(50):  # Bisection iterations
        mid = (low + high) / 2.0
        trial = np.clip(h + mid, 0, 1)
        trial_sum = np.sum(trial)
        if trial_sum > desired_sum:
            high = mid
        else:
            low = mid
            
    lambda_val = (low + high) / 2.0
    h = np.clip(h + lambda_val, 0, 1)
    return h

def block_objective(params, K=6, n_eval=200):
    """Evaluate objective function from 2*K block parameters."""
    u = params[:K]
    v = params[K:]
    
    # Convert u to lengths
    u_exp = np.exp(u)
    lengths = 2.0 * u_exp / np.sum(u_exp)
    
    # Convert v to heights
    heights = 1.0 / (1.0 + np.exp(-v))
    
    # Scale to integral = 1
    area_sum = np.sum(heights * lengths)
    alpha = 1.0 / area_sum
    heights = np.minimum(1.0, alpha * heights)
    
    # redistribution to accommodate for final scaling
    total_h = np.sum(heights * lengths)
    if abs(total_h - 1.0) > 1e-10:
        alpha = 1.0 / total_h 
        heights = np.minimum(1.0, alpha * heights)
    
    return compute_block_autocorrelation(lengths, heights, n_eval)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    K = 6  # number of blocks
    n_fine = 400  # fine grid resolution
    
    # Global optimization
    bounds = [(-10, 10)] * (2 * K)  # u,v parameters
    result = differential_evolution(
        block_objective,
        bounds,
        maxiter=500,
        popsize=10*K,
        seed=seed,
        disp=False,
        atol=1e-8,
        rtol=1e-8
    )
    if time.time() - start_time > budget_s - 0.5:
        pass  # We have no time for fine sampling
    u_opt = result.x[:K]
    v_opt = result.x[K:]
    
    u_opt_exp = np.exp(u_opt)
    lengths = 2.0 * u_opt_exp / np.sum(u_opt_exp)
    heights = 1.0 / (1.0 + np.exp(-v_opt))
    area_sum = np.sum(heights * lengths)
    alpha = 1.0 / area_sum
    heights = np.minimum(1.0, alpha * heights)
    
    # Generate fine sample h vector
    h_initial = sample_from_blocks(lengths, heights, n_fine)
    h_initial = project_to_feasible(h_initial, n_fine)
    
    if evaluate_c5(h_initial) < 0.3807:
        return (h_initial, evaluate_c5(h_initial), n_fine)
    
    # Local optimization:
    last_improvement = start_time
    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Projected gradient on worst shift  
    n_steps = 0
    while time.time() - start_time < budget_s - 1.0 and time.time() - last_improvement < 5.0:
        # Get full correlation
        corr = correlate(best_h, 1 - best_h, mode='full')
        # Find worst shift k*
        k_star = np.argmax(corr)
        dx = 2.0 / n_fine
        # Compute gradient
        grad = np.zeros_like(best_h)
        for i in range(len(best_h)):
            left_idx = (i - k_star) % len(best_h)
            right_idx = (i + k_star) % len(best_h)
            grad[i] = 1.0 - best_h[left_idx] - best_h[right_idx]
        # Refined learning rate search
        step = 1.0
        improved = False
        for _ in range(10):
            trial = best_h - step * grad
            trial = project_to_feasible(trial, n_fine)
            trial_c5 = evaluate_c5(trial)
            if trial_c5 < best_c5:
                improved = True
                best_h = trial
                best_c5 = trial_c5
                last_improvement = time.time()
                break
            step *= 0.5
        if not improved:
            break  # No more progress
        n_steps += 1
        
        if n_steps >= 1000:  # Limit to prevent infinite loop
            break
    
    # Final hill climbing
    if time.time() - start_time < budget_s - 1.0:
        for _ in range(3000):
            if time.time() - start_time > budget_s - 0.5:
                break
            i, j = np.random.randint(0, n_fine, 2)
            if best_h[i] > 0 and best_h[j] < 1:
                delta = min(best_h[i], 1 - best_h[j], 0.01)
                if delta > 1e-8:
                    best_h[i] -= delta
                    best_h[j] += delta
                    best_h = project_to_feasible(best_h, n_fine)
                    new_c5 = evaluate_c5(best_h)
                    if new_c5 >= best_c5:
                        best_h[i] += delta
                        best_h[j] -= delta
                        best_h = project_to_feasible(best_h, n_fine)
                    else:
                        best_c5 = new_c5
                        last_improvement = time.time()
    
    # Final accuracy check
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n_fine)
