import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import math

def c5_fft(h):
    """Fast C5 computation using FFT."""
    g = 1 - h
    n = len(h)
    n_full = 2 * n - 1
    H = rfft(h)
    G = rfft(g)
    c = irfft(H * G, n_full)
    return np.max(c)

def repair_weight(h, target_sum):
    """Repair to exact weight."""
    current_sum = np.sum(h)
    diff = int(target_sum - current_sum)
    if diff > 0:  # Need to change some zeros to ones
        zero_indices = np.where(h == 0)[0]
        if len(zero_indices) >= diff:
            change_indices = np.random.choice(zero_indices, diff, replace=False)
            h[change_indices] = 1
    elif diff < 0:  # Need to change some ones to zeros
        one_indices = np.where(h == 1)[0]
        if len(one_indices) >= -diff:
            change_indices = np.random.choice(one_indices, -diff, replace=False)
            h[change_indices] = 0
    return h

def crossover(p1, p2):
    """Two-point crossover."""
    n = len(p1)
    c1 = np.random.randint(1, n-1)
    c2 = np.random.randint(c1+1, n)
    child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
    return repair_weight(child, n/2)

def mutate_swap(h):
    """Swap mutation."""
    one_indices = np.where(h == 1)[0]
    zero_indices = np.where(h == 0)[0]
    if len(one_indices) > 0 and len(zero_indices) > 0:
        i1 = np.random.choice(one_indices)
        i0 = np.random.choice(zero_indices)
        h[i1], h[i0] = h[i0], h[i1]
    return h

def hill_climb(h):
    """Local hill-climbing on the worst shift."""
    n = len(h)
    best_h = h.copy()
    best_c5 = c5_fft(best_h)
    
    # Relax to regular correlation for efficient gradient estimation
    g = 1 - best_h
    c = correlate(best_h, g, mode='full')
    k_star = np.argmax(c) - n + 1
    
    for _ in range(500):
        # Try a set of random swaps
        swaps_tried = 0
        best_delta = 0
        best_swap = None
        while swaps_tried < 200:
            # Pick two random indices to swap
            one_indices = np.where(best_h == 1)[0]
            zero_indices = np.where(best_h == 0)[0]
            if len(one_indices) == 0 or len(zero_indices) == 0:
                break
            i1 = np.random.choice(one_indices)
            i0 = np.random.choice(zero_indices)
            
            # Compute delta change in correlation for shift k_star
            # h[i1] and h[i0] are swapped
            old_corr = (best_h[i1] * best_h[i1 + k_star] if i1 + k_star < n else 0) + \
                       (best_h[i0] * best_h[i0 + k_star] if i0 + k_star < n else 0)
            new_corr = ((1 - best_h[i1]) * best_h[i1 + k_star] if i1 + k_star < n else 0) + \
                       (best_h[i0] * (1 - best_h[i0 + k_star]) if i0 + k_star < n else 0)
            delta = new_corr - old_corr  # Negative means better!
            
            if delta < best_delta:
                best_delta = delta
                best_swap = (i1, i0)
            swaps_tried += 1
        if best_swap is None:
            break
        i1, i0 = best_swap
        best_h[i1], best_h[i0] = best_h[i0], best_h[i1]
        # Direct update of the correlation for k_star
        # Only update local values, make real check later
        new_c5 = c5_fft(best_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # Revert
            best_h[i1], best_h[i0] = best_h[i0], best_h[i1]
    return best_h
