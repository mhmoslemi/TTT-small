```python
import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def evaluate_c5(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    result = np.max(conv) * dx
    return result

def construct_symmetric_h(w):
    """Construct full symmetric h vector from independent w values"""
    n = 2 * (len(w) - 1)
    h = np.zeros(n)
    h[:len(w)] = w
    h[len(w):] = w[-2:0:-1]
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Grid size
    n = 200
    dx = 2.0 / n
    d = n // 2 + 1  # number of independent variables
    
    # Initial variable setup
    w = cp.Variable(d)
    M = cp.Variable()
    lambda_param = cp.Parameter(nonneg=True)
    lambda_param.value = 0.1
    
    # Set up constraints and objective
    constraints = []
    # Box constraints
    constraints += [0 <= w, w <= 1]
    # Sum constraint
    constraints += [cp.sum(w) == n/4 + 0.5]
    
    # Linear relaxation of bilinear terms
    p_vars = {}
    S = {0}  # active shifts
    
    # Add initial constraints
    # For k=0
    k = 0
    for i in range(n):
        if i in p_vars:
            continue
        p_vars[(i, k)] = cp.Variable()
        # p[i,0] <= w[i]  
        constraints += [p_vars[(i, k)] <= w[i]]
        # p[i,0] <= w[i]  
        constraints += [p_vars[(i, k)] <= w[i]]
        # p[i,0] >= 0  
        constraints += [p_vars[(i, k)] >= 0]
        # p[i,0] >= w[i] + w[i] - 1 = 2*w[i] - 1  
        constraints += [p_vars[(i, k)] >= 2*w[i] - 1]  # Actually this is for k=0
        
    # Initial constraints for k=0 (laplace estimation limited)
    # We adaptively build shift set during loop
    
    # Objective 
    objectives = [M - lambda_param * cp.sum([p_vars[(i, 0)] for i in range(n)])]

    best_c5 = float('inf')
    best_h = None

    # Double-oracle cutting plane loop
    S = {0}
    lambda_param.value = 0.1
    max_shifts = min(30, n)
    loop_iters = 0
    ε = 1e-6
    safety_margin = 5.0
    max_iters = 50

    while time.time() - start_time < budget_s - safety_margin and loop_iters < max_iters and len(S) <= max_shifts:
        # Build lp problem for current shifts
        # Variables: w (d), p (n*len(S)), M (1) 
        p_vars = {}  # maps (i,k) to variable
        rel_constraints = constraints[:]

        # Add bilinear constraints for active shifts
        # We assume the solution is already symmetrized
        for k in S:
            if k == 0:
                # p[i,0] = w[i]^2 
                for i in range(0, n):
                    p_vars[(i, k)] = cp.Variable()
                    rel_constraints += [p_vars[(i, k)] == w[i] * w[i]]
            else:
                # For k>0, consider i+k < n, i-k >=0 
                # We'll look at overlap shifts k
                # Only subdomain is considered
                for i in range(max(0, k), min(n, n-k)):
                    p_vars[(i, k)] = cp.Variable()
                    # p[i,k] <= w[i]
                    rel_constraints += [p_vars[(i, k)] <= w[i]]
                    # p[i,k] <= w[i+k]
                    rel_constraints += [p_vars[(i, k)] <= w[i+k]]
                    # p[i,k] >= 0
                    rel_constraints += [p_vars[(i, k)] >= 0]
                    # p[i,k] >= w[i] + w[i+k] - 1 
                    rel_constraints += [p_vars[(i, k)] >= w[i] + w[i+k] - 1]

        # Overlap constraints  
        overlap_constraints = []
        for k in S:
            # Convolution must be at most M
            # Overlap = 1 - sum(p[i, k])
            overlap_expr = 1.0 - cp.sum([p_vars[(i, k)] for i in range(len(w)) if (i, k) in p_vars])
            overlap_constraints += [overlap_expr <= M]
        rel_constraints += overlap_constraints

        # Objective
        obj = cp.Minimize(M - lambda_param * cp.sum([p_vars[(i, 0)] for i in range(len(w)) if (i, 0) in p_vars]))

        # Solve the LP
        problem = cp.Problem(obj, rel_constraints)
        solver = cp.ECOS
        try:
            problem.solve(solver=solver)
        except:
            break  # break if infeasible or problem with solver 

        if problem.status not in ["optimal", "optimal_inaccurate"]:
            break  # Skip bad iterations
        M_val = M.value if M.value is not None else float('inf')
        
        # Extract w solution and reproduce full h vector
        if w.value is None:
            break
        w_sol = np.array(w.value)
        h = construct_symmetric_h(w_sol)
        # Re-evaluate true C5
        c5_exact = evaluate_c5(h)
        
        if c5_exact < best_c5:
            best_c5 = c5_exact
            best_h = h.copy()

        # Check convergence to M_val (within ε)
        # Solve exactly for maximum overlap 
        if time.time() - start_time > budget_s - 0.5:
            break
        if abs(c5_exact - M_val) < ε:
            break

        # Add the most violated shift as a new cut
        # Compute C5 by current h at all shifts k
        g = 1.0 - h
        conv = np.correlate(h, g, mode='full')
        max_idx = np.argmax(conv) 
        k_best = max_idx - (len(h) - 1) 
        # k_best is our new constraint to add
        S.add(k_best)
        loop_iters += 1

        # Gradually increase binary penalty  
        lambda_param.value = min(5.0, lambda_param.value * 1.5)

    # Subtract 10% from total time for final steps if needed
    finalize_time = time.time() - start_time
    if finalize_time < budget_s - 3.0:
        # Local polish via projected subgradient
        polishing_time = min(200, budget_s - finalize_time - 3.0)
        best_h = polish_with_projected_subgradient(best_h, polishing_time)

    # Final optional hill-climb on remaining time
    if time.time() < start_time + budget_s - 2.0:
        remaining_budget = budget_s - (time.time() - start_time)
        if remaining_budget > 1.0:
            best_h, best_c5 = stochastic_hill_climb(best_h, n, best_c5, remaining_budget)

    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, n)

def polish_with_projected_subgradient(h, timeout_s):
    """Polish h using projected subgradient descent"""
    n = len(h)
    start = time.time()
    step_size = 0.02
    epsilon = 1e-6
    max_iter = 200
    h_old = h.copy()
    c5_prev = evaluate_c5(h_old)
    for i in range(max_iter):
        if time.time() - start > timeout_s - 0.1:
            break
        # Compute gradient at current solution
        g = np.zeros(n)  
        # Use maximum overlap shift
        g1 = 1.0 - h
        conv = np.correlate(h, g1, mode='full')
        idx_max = np.argmax(conv)
        k_max = idx_max - (n - 1)
        # Build g[i] = (1-h[i+k_max]) - h[i-k_max] 
        for i in range(n):
            contrib = 0.0
            if i + k_max < n: 
                contrib += 1.0 - h[i + k_max]
            if i - k_max >= 0:
                contrib -= h[i - k_max] 
            g[i] = contrib
        # Update
        h_new = h_old - step_size * g
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Project to sum = 1 (approximate, n=200)
        h_new = h_new * (n / 2) / np.sum(h_new)
        # Symmetrize
        h_new = (h_new + h_new[::-1]) / 2
        h_old = h_new
        c5 = evaluate_c5(h_old) 
        if c5 < c5_prev - epsilon:
            c5_prev = c5
            step_size *= 0.9  # decay
        else:
            step_size *= 0.95
    return h_old

def stochastic_hill_climb(h, n_points, c5_current, remaining_time_s):
    """Do stochastic hill-climb on remaining time"""
    start = time.time()
    c5_best = c5_current
    h_best = h.copy()
    p = min(6, n_points // 20) 
    epsilon = 0.003
    attempts = 0
    successes = 0
    max_iter = 10000
    for i in range(max_iter):
        if time.time() - start > remaining_time_s - 0.5:
            break
        # Select random indices to perturb
        indices = np.random.choice(n_points, p)
        h_trial = h_best.copy()
        for idx in indices:
            delta = np.random.uniform(-epsilon, epsilon)
            h_trial[idx] += delta
        # Clip and re-project  
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial) 
        if n_points % 2 == 0:
            h_trial = (h_trial + h_trial[::-1]) / 2
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_best:
            c5_best = c5_trial 
            h_best = h_trial
            successes += 1
            if successes % 40 == 0:
                # Re-symmetrize
                h_best = (h_best + h_best[::-1]) / 2
        attempts += 1
    return h_best, c5_best
```
