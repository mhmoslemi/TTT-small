import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import cvxpy as cp
from cvxpy import ECOS
import math

def solve_spectral_lp(n_points):
    """Solve the spectral relaxation LP."""
    # Variables: p_m = |h_m|^2
    p = cp.Variable(n_points)
    t = cp.Variable()
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    
    # Constraints
    constraints = []
    
    # Sum constraint: sum p_m = n^2/2 (due to sum h_i = n/2 and h_i <= 1)
    constraints.append(cp.sum(p) == n_points * n_points / 2)
    
    # Bound constraints on p_m
    constraints.append(p >= 0)
    constraints.append(p <= n_points)
    
    # Autocorrelation constraints
    # For k = 1 to n-1
    for k in range(1, n_points):
        # cos(2pi k m / n) for m = 0 to n-1
        cos_term = np.cos(2 * np.pi * k * np.arange(n_points) / n_points)
        # Sum over m: p_m * cos(2pi*k*m/n) >= t
        constraints.append(cp.sum(p * cos_term) >= t)
    
    # Form and solve LP
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        return p.value, t.value
    except Exception:
        # Fallback: use a simple bound
        return None, None

def gerchberg_saxton(p_spectrum, n_points, max_iter=200):
    """Run the Gerchberg-Saxton phase retrieval algorithm."""
    # Initialize with random phases
    phi = np.random.rand(n_points) * 2 * np.pi
    
    # Set up complex spectrum with given power
    H = np.sqrt(p_spectrum) * np.exp(1j * phi)
    
    # Inverse FFT
    h = np.fft.irfft(H)
    h = np.real(h)
    
    # Precompute conjugate for easier inverse DFT later
    H_conj = np.conj(H)
    H_conj[0] = H_conj[0].real
    if n_points % 2 == 0:
        H_conj[n_points//2] = H_conj[n_points//2].real
    
    for _ in range(max_iter):
        # Amplitude projection (box constraints)
        y = np.clip(h, 0, 1)
        # Water-filling to maintain sum = n/2
        total = np.sum(y)
        delta = n_points / 2 - total
        # Add a constant to each entry
        if np.abs(delta) > 1e-10:
            # Use binary search to find proper lambda
            # But for simplicity, just add it directly and clip then adjust
            # However this inevitably leads to suboptimal results
            # So let's do it correctly: we know the sum is off by delta
            # We want to scale this in projection
            if delta > 0: # need to increase sum
                # For stable solution, we average the additions + clip and re-add
                # But we do it simpler: just add the delta/n to all elements
                y = y + delta / n_points
                y = np.clip(y, 0, 1)
                # Compensate for any clippings by adjustment to sum
                total = np.sum(y)
                diff = n_points / 2 - total
                # Now we can increase some elements if positive differences
                if diff > 1e-10:
                    indices = np.argsort(y)
                    for i in range(n_points):
                        j = indices[-(i+1)]
                        if y[j] < 1.0:
                            add = min(diff, 1.0 - y[j])
                            y[j] += add
                            diff -= add
                            if diff <= 1e-10:
                                break
            else: # need to decrease sum
                # Similar idea: decrease uniformly (or subject to clip)
                y = y + delta / n_points
                y = np.clip(y, 0, 1)
                # Then try to clip on large ones to raise sum
                total = np.sum(y)
                diff = n_points / 2 - total
                if diff < -1e-10: # need to reduce sum
                    indices = np.argsort(y)
                    for i in range(n_points):
                        j = indices[i]
                        if y[j] > 0.0:
                            subtract = min(-diff, y[j])
                            y[j] -= subtract
                            diff += subtract
                            if diff >= -1e-10:
                                break
                
        # Polyfill: a simpler but more robust approach to sum
        y = np.clip(y, 0, 1)
        total = np.sum(y)
        diff = n_points / 2 - total
        if abs(diff) > 1e-10:
            if diff > 0: # we need to push up
                # Try pushing uniformly
                y += diff / n_points
            else:
                y += diff / n_points
            # Clip again with safe adjustments
            y = np.clip(y, 0, 1)
            
        # Forward DFT
        Y = np.fft.rfft(y)
        
        # Spectral projection
        # Replace magnitudes but preserve phases
        mags = np.abs(Y)
        # Avoid division by zero
        mask = mags != 0
        # Note: H is redefined in each phase as it represents what should be projected
        H[0] = np.sqrt(p_spectrum[0]) * np.exp(1j * phi[0])
        if n_points % 2 == 0:
            H[n_points // 2] = np.sqrt(p_spectrum[n_points // 2]) * 1j * 0 # since we req. real valued
        if len(H) >= len(Y):
            H[1:-1] = np.sqrt(p_spectrum[1:-1]) * Y[1:-1] / mags[1:-1]
            
        # But we should truncate appropriately
        for m in range(n_points):
            if mask[m]:
                H[m] = np.sqrt(p_spectrum[m]) * Y[m] / mags[m]
            else:
                # Keep original phase or set to zero
                pass
        
        # Inverse FFT
        h = np.fft.irfft(H)
        h = np.real(h)
        
    return h

def project_simplex_box(v, sum_value):
    """Project vector onto the simplex defined by sum(v) = sum_value and 0 <= v_i <= 1."""
    # Clip to [0,1]
    v = np.clip(v, 0, 1)
    
    # Water-filling to maintain exact sum
    total = np.sum(v)
    if np.abs(total - sum_value) < 1e-10:
        return v
    
    # Use iterative approach
    v_old = v.copy()
    diff = sum_value - total
    if diff > 0: # need to increase
        # Add delta uniformly to elements in increasing order of value
        indices = np.argsort(v)
        # Need it increasing order with idxs
        for idx in indices:
            old_val = v[idx]
            if v[idx] < 1.0:
                add = min(diff, 1.0 - v[idx])
                v[idx] += add
                diff -= add
                if diff <= 1e-10:
                    break
    else: # diff < 0, decrease
        diff = -diff
        indices = np.argsort(v)[::-1]
        for idx in indices:
            old_val = v[idx]
            if v[idx] > 0.0:
                sub = min(diff, v[idx])
                v[idx] -= sub
                diff -= sub
                if diff <= 1e-10:
                    break
                    
    return v

def binary_sharpen(h, mu=1e-3, num_iter=50):
    """Apply soft binary sharpening on the step function."""
    for epoch in range(num_iter):
        # Approximate gradient assuming smooth surrogate
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        diff = 1.0 * len(h) / 2 - total
        # Gradient: h += 1/n * (diff)
        h += diff / len(h)
        h = project_simplex_box(h, len(h) / 2)
        
        # Apply penalty
        penalty = mu * np.sum(h * (1 - h))
        # Gradient descent with small learning rate to mitigate the penalty
        h = np.clip(h, 0, 1) 
        
        # After few iterations sharpen
        if epoch % 10 == 0 and epoch > 0:
            h = np.round(h)
            h = project_simplex_box(h, len(h) / 2)
            
    return h

def gradient_worst_shift(h, dx, max_iter=3000):
    """Optimize the worst shift with explicit gradient descent."""
    num_points = len(h)
    c = correlate(h, 1 - h, mode="full")
    # Get the maximum k index (shift) which corresponds to [-n+1, n-1]
    # So k = argmax(c) - (n-1) should give the actual shift we expect
    idx_max_shift = np.argmax(c)
    max_idx = idx_max_shift - (num_points - 1)
    
    # Keep old value of C5 until proven otherwise.
    old_max = c[idx_max_shift]
    old_c5 = dx * old_max
    
    for _ in range(max_iter):
        # Update based on gradient of worst shift
        grad = np.zeros_like(h)
        # Gradient of I_{max_shift} = dx * sum(h_i*(1-h_{i+shift}) + h_{i-shift})
        # = dx * [sum(1-h_{i+shift}) - sum(h_{i-shift})] 
        # This becomes log term [n - sum_{i}(h_{i+shift}) - sum_{i}(h_{i-shift})]
        # But expansion is:
        # For each i, gradient of I = dx*(1-h_{i+shift}) - dx*(h_{i-shift})
        
        # This means:
        k = max_idx
        for i in range(num_points):
            if i + k >= 0 and i + k < num_points:
                grad[i] -= dx * h[i + k]
            if i - k >= 0 and i - k < num_points:
                grad[i] += dx * h[i - k]
            # Potential elliptic fix
            if not (i + k >= 0 and i + k < num_points):
                # treat as zero
                grad[i] -= 0
            if not (i - k >= 0 and i - k < num_points):  
                # treat as zero
                grad[i] += 0
                
        # Simple gradient step
        step_size = 0.001
        h_new = h - step_size * grad
        h_new = project_simplex_box(h_new, num_points / 2)
        
        # Check if it improves
        c_new = correlate(h_new, 1 - h_new, mode="full")
        idx_max_new = np.argmax(c_new)
        old_max_new = c_new[idx_max_new]
        new_c5 = dx * old_max_new
        
        if new_c5 < old_c5:
            h = h_new
            old_c5 = new_c5
            idx_max_shift = idx_max_new
            max_idx = idx_max_shift - (num_points - 1)
            old_max = old_max_new
        else:
            step_size *= 0.5
            if step_size < 1e-6:
                break
                
    return h

def swap_anneal(h, dx, time_remaining):
    """Perform simulated-annealing-like swap approach."""
    start_time = time.time()
    T = 1e-3  # initial temperature
    cooling_rate = 0.95  # geometric cooling
    max_iter = 1000
    iter_count = 0
    
    # Set up the worst shift
    c = correlate(h, 1 - h, mode="full")
    idx_max_shift = np.argmax(c)
    current_c5 = dx * c[idx_max_shift]
    
    while iter_count < max_iter and time.time() - start_time < time_remaining:
        # pick two random indices with h[i] > 0 and h[j] < 1
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        if h[i] > 1e-6 and h[j] < 1.0 - 1e-6:
            # Transfer amount
            max_transfer = min(h[i], 1.0 - h[j], 0.02)
            if max_transfer <= 0:
                continue
            delta = np.random.uniform(0, max_transfer)
            
            # Try this change
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            
            # Project
            h_new = project_simplex_box(h_new, len(h) / 2)
            
            # see if new c5 is better
            c_new = correlate(h_new, 1 - h_new, mode="full")
            current_max_idx = np.argmax(c_new)
            new_c5 = dx * c_new[current_max_idx]
            
            # acceptance probability
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
            else:
                # Metropolis
                dE = new_c5 - current_c5
                if np.random.rand() < np.exp(-dE / T):
                    h = h_new
                    current_c5 = new_c5
                    
            iter_count += 1
            
        T *= cooling_rate
        
    return h

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 650
    dx = 2.0 / n_points
    
    print(f"Starting optimization with n_points={n_points}")
    
    # 1. Spectral convex relaxation
    print("Solving spectral LP...")
    p_opt, t_opt = solve_spectral_lp(n_points)
    if p_opt is None or t_opt is None:
        # fallback to bin-based h from previous
        h = np.ones(n_points) * 0.5
        return (h, evaluate_c5(h), n_points)
        
    print(f"Solved LP: t_opt={t_opt:.6f}")
    
    # 2. Phase retrieval with Gerchberg-Saxton
    print("Running Gerchberg-Saxton phase retrieval...")
    
    h_best = None
    best_c5 = float('inf')
    
    for trial in range(10):  # max 10 restarts
        if time.time() - start_time > budget_s - 5:
            break
            
        h_gs = gerchberg_saxton(p_opt, n_points, max_iter=200)
        
        if h_gs is not None:
            h_gs = project_simplex_box(h_gs, n_points / 2)
            c5_gs = evaluate_c5(h_gs)
            
            if c5_gs < best_c5:
                best_c5 = c5_gs
                h_best = h_gs.copy()
                if best_c5 <= 0.3807:
                    break  # early stop
                    
    print(f"Best GS result before refinement: {best_c5:.6f}")
    
    # 3. Binary sharpening
    if time.time() - start_time < budget_s - 5:
        print("Applying binary sharpening...")
        h_best = binary_sharpen(h_best, num_iter=100)
        c5_sharpen = evaluate_c5(h_best)
        if c5_sharpen < best_c5:
            best_c5 = c5_sharpen
        print(f"After binary sharpening: {best_c5:.6f}")
    
    # 4. Worst-shift gradient refinement
    if time.time() - start_time < budget_s - 5:
        print("Running worst-shift gradient refinement...")
        h_refined = gradient_worst_shift(h_best, dx, max_iter=3000)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            h_best = h_refined
            best_c5 = c5_refined
        print(f"After gradient refinement: {best_c5:.6f}")
    
    # 5. Swap annealing
    if time.time() - start_time < budget_s - 5:
        print("Running swap annealing...")
        remaining_time = budget_s - (time.time() - start_time)
        h_final = swap_anneal(h_best, dx, remaining_time)
        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            h_best = h_final
            best_c5 = c5_final
        print(f"After swap annealing: {best_c5:.6f}")
    
    return (h_best, best_c5, n_points)
