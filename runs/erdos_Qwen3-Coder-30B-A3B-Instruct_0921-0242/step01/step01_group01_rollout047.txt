import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.ndimage import shift
import time

def evaluate_c5(h):
    """Evaluate C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution
    conv = np.correlate(h, g, mode='full')
    # The full convolution has 2*len(h) - 1 elements
    # The shifts are from -(len(h)-1) to (len(h)-1)
    # But we're only interested in the overlap integral
    # which is the maximum of the convolution values
    result = np.max(conv) * dx
    return result

def decode_blocks_to_h(block_params, n_points):
    """Decode M blocks to a full h vector"""
    M = len(block_params) // 3
    starts = block_params[0::3]
    lengths = block_params[1::3]
    heights = block_params[2::3]
    
    # Sort by start position
    indices = np.argsort(starts)
    starts = starts[indices]
    lengths = lengths[indices]
    heights = heights[indices]
    
    # Ensure non-overlap
    for i in range(1, M):
        if starts[i] < starts[i-1] + lengths[i-1]:
            starts[i] = starts[i-1] + lengths[i-1]
    
    # Create h vector
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(M):
        start_idx = int(starts[i] / dx)
        end_idx = int((starts[i] + lengths[i]) / dx)
        if end_idx > start_idx:
            h[start_idx:end_idx] = heights[i]
    
    # Normalize to satisfy integral constraint
    h_sum = np.sum(h)
    if h_sum > 1e-10:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * 0.5
        
    return h

def soft_max_conv(h, beta=1.0):
    """Compute soft-max approximation of the convolution max"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    return np.log(np.sum(np.exp(beta * dx * conv))) / beta

def gradient_soft_max_conv(h, beta=1.0):
    """Compute gradient of soft-max convolution with respect to h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    weights = np.exp(beta * dx * conv)
    weights = weights / np.sum(weights)
    
    # Compute gradient for each position in the full convolution
    grad = np.zeros_like(h)
    
    # For each shift s in the full convolution
    n = len(h)
    full_len = 2 * n - 1
    
    for s in range(full_len):
        k = s - (n - 1)  # shift parameter
        
        # Compute weight for this k
        weight_s = weights[s]
        
        # Now we'll compute contributions to h[i] gradient
        for i in range(n):
            # Pattern: for h[i] its contribution to conv[s] when 
            # k = s - (n - 1) means i = s - k' or k' = i - s + (n-1)
            # So conv[s] = ... h[i] * (1 - h[i+k]) ... s.t. h[i+k] term
            # corresponds to shift k from h[i] 
            # In convolution logic: conv[s] has h[j] * (1 - h[j+k_perm])
            
            # When k > 0, i+k corresponds to position in array
            # when k < 0, i-k corresponds to position in array

            # The contribution is for each i:
            # 1. Term from (1 - h[i+k]) if j = i
            # 2. Term from - h[i-k] if j = i+k
            # But as a derivative of sum h[j] * (1 - h[j+k]),  
            # d/dh_i (h_i * (1 - h[i+k])) = (1 - h[i+k]) - h[i-k]  
            
            # Limited by valid indices
            term1 = 0.0
            if 0 <= i + k < n:
                term1 = -(1 - h[i + k])  # sign flipped
            term2 = 0.0
            if 0 <= i - k < n and k != 0:
                term2 = h[i - k]
                
            grad[i] += weight_s * (term1 - term2) * dx * beta
    
    return grad

def project_sum(h, target_sum):
    """Project h onto sum = target_sum using water filling"""
    current_sum = np.sum(h)
    diff = current_sum - target_sum
    n = len(h)
    
    if abs(diff) < 1e-9:
        return h
    
    if n == 0:
        return h
    
    # Distribute the difference evenly
    avg_diff = diff / n
    h = np.clip(h - avg_diff, 0, 1)
    
    # Fine adjustment
    current_sum = np.sum(h)
    diff = current_sum - target_sum

    if abs(diff) > 1e-9:
        diff_per_unit = diff / (np.sum(h > 0) if np.sum(h > 0) > 0 else 1)
        # Only adjust non-zero values
        h = np.clip(h - diff_per_unit, 0, 1)
        # If we run over bounds we project again
        h = np.clip(h - (np.sum(h) - target_sum) / n, 0, 1)
    
    return h

def refine_h_with_soft_max(h0, max_iter=150, beta_max=50.0, beta_ini=0.5):
    """Refine h using soft-max gradient descent with continuation"""
    h = h0.copy()
    dx = 2.0 / len(h)
    target_sum = len(h) / 2
    
    # Initialize penalty coefficient
    lambda_penalty = 10.0
    beta = beta_ini
    
    # L-BFGS-B optimization parameters
    bounds = [(0, 1)] * len(h)
    
    for iteration in range(20):  # 20 stages with beta continuation
        if beta > beta_max:
            break
            
        # Define objective and gradient for this beta
        def obj(h):
            c5 = soft_max_conv(h, beta)
            penalty = lambda_penalty * (np.sum(h) - target_sum)**2
            return c5 + penalty
            
        def grad(h):
            grad_c5 = gradient_soft_max_conv(h, beta)
            grad_penalty = 2 * lambda_penalty * (np.sum(h) - target_sum) * np.ones_like(h)
            return grad_c5 + grad_penalty
            
        # Optimization with bounds
        try:
            result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, jac=grad, options={'maxiter': max_iter//5})
            h = result.x
        except Exception:
            pass
            
        # Project and clip
        h = project_sum(h, target_sum)
        h = np.clip(h, 0, 1)
        # Update beta and lambda
        beta *= 2.0
        lambda_penalty *= 2.0
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 250
    dx = 2.0 / n_points
    
    # Parametric block representation: 3 parameters per block: start, length, height
    # Let's use 4 blocks
    n_blocks = 4
    n_params = 3 * n_blocks
    
    # Define bounds for parameters: 
    # start_j ∈ [0, 2 - len_j], len_j ∈ (0, 2], height_j ∈ [0, 1]
    # For simplicity, we'll use fixed range:
    bounds = []
    for _ in range(n_blocks):
        bounds.extend([(0, 2), (0.001, 2), (0, 1)])
    
    # Best so far
    best_c5 = float('inf')
    best_h = None
    
    # Global search with Differential Evolution
    def de_objective(params):
        h = decode_blocks_to_h(params, n_points)
        h = (h + h[::-1]) / 2  # enforce symmetry  
        c5 = evaluate_c5(h)
        return c5
    
    # Run DE
    try:
        result = differential_evolution(de_objective, bounds, maxiter=60, popsize=15, seed=seed, disp=False)
        if result.success:
            h = decode_blocks_to_h(result.x, n_points)
            h = (h + h[::-1]) / 2  # enforce symmetry  
            h = project_sum(h, n_points / 2)
            c5_opt = evaluate_c5(h)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h.copy()
    except Exception as e:
        pass
    
    # Local refinement with soft max gradients
    if best_h is not None and time.time() - start_time < budget_s - 1:
        h_refined = refine_h_with_soft_max(best_h, max_iter=100)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Stochastic hill-climbing
    if best_h is not None and time.time() - start_time < budget_s - 0.5:
        h_best = best_h.copy()
        c5_best = best_c5
        n_perturb = min(5, n_points // 20)
        
        for _ in range(500):  # Inner loop limit
            if time.time() - start_time > budget_s - 0.5:
                break
                
            # Pick random indices
            indices = np.random.choice(n_points, size=n_perturb, replace=False) 
            
            # Perturb with small uniform noise
            h_trial = h_best.copy()
            for idx in indices:
                delta = np.random.uniform(-0.005, 0.005)
                h_trial[idx] = min(1.0, max(0.0, h_trial[idx] + delta))
            
            # Re-project to satisfy integral constraint
            h_trial = project_sum(h_trial, n_points / 2)
            
            # Evaluate
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < c5_best:
                h_best = h_trial
                c5_best = c5_trial
                
        best_h = h_best
        best_c5 = c5_best
    
    if best_h is None:
        # Fallback to uniform
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_c5(best_h)
        
    return (best_h, best_c5, n_points)
