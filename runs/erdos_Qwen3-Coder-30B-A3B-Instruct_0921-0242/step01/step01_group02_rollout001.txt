import time
from scipy.signal import correlate
import cvxpy as cp

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx


def evaluate_full_correlation(h):
    """Return full correlation array."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    return corr


def project_to_constraints(h, n_points, target_sum):
    """Project h onto sum constraint and box constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    weights = np.ones_like(h)
    weights[np.where(h <= 0)] = 0
    weights[np.where(h >= 1)] = 0
    if np.sum(weights) > 0:
        h += diff * weights / np.sum(weights)
        h = np.clip(h, 0, 1)
    return h


def simulated_annealing_binary(h_initial, dx, max_iter=50000, initial_temp=1e-2, cool_rate=0.9999, target_sum=150):
    """Simulated Annealing to find good binary h."""
    h = h_initial.copy()
    current_c5 = c5_of_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    temp = initial_temp
    for i in range(max_iter):
        if i % 10000 == 0:
            temp *= cool_rate
            
        # Select two random indices to swap
        idx1 = np.random.randint(0, len(h))
        idx2 = np.random.randint(0, len(h))
        if idx1 == idx2:
            continue
            
        # Check if indices are compatible (i.e., swapping maintains the constraint)
        if h[idx1] + h[idx2] in [0, 1, 2]:
            h[idx1], h[idx2] = h[idx2], h[idx1]
            new_c5 = c5_of_h(h, dx)
            
            # Acceptance probability
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                # Revert if not accepted
                h[idx1], h[idx2] = h[idx2], h[idx1]
    
    return best_h, best_c5
