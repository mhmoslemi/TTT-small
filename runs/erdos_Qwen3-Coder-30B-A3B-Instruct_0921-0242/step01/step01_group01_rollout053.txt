```python
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time
from scipy.linalg import eigh
import warnings

warnings.filterwarnings('ignore')

def waterfill(h, target_sum):
    """Waterfill projection to enforce sum constraint"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = current_sum - target_sum
    if abs(diff) < 1e-12:
        return h

    while abs(diff) > 1e-12:
        # Find indices where 0 < h < 1
        idxs = np.where((h > 0) & (h < 1))[0]
        if len(idxs) == 0:
            break
    
        # Distribute the deviation
        avg_change = diff / len(idxs)
        h[idxs] -= avg_change
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        diff = current_sum - target_sum
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_coarse = 120  # Coarse grid resolution
    n_fine = 300    # Target resolution
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    target_sum_coarse = n_coarse / 2
    target_sum_fine = n_fine / 2
    
    h_best = None
    c5_best = float('inf')
    
    # Step 1-3: SDP relaxation on coarse grid
    try:
        # Build Toeplitz matrices R_k for k=0 to n_coarse-1        
        R_matrices = []
        for k in range(n_coarse):
            R = np.zeros((n_coarse, n_coarse))
            # Add ones on the k-th off-diagonal
            if k == 0:
                np.fill_diagonal(R, 1.0)
            else:
                R[np.arange(n_coarse - k), np.arange(k, n_coarse)] = 1.0
                R[np.arange(k, n_coarse), np.arange(n_coarse - k)] = 1.0
            R_matrices.append(R)
        
        # SDP Variables
        t = cp.Variable()
        h = cp.Variable(n_coarse)
        X = cp.Variable((n_coarse, n_coarse), symmetric=True)
        
        # Constraints
        constraints = [X >> 0]  # PSD
        constraints.append(cp.trace(X) == 1.0)  # trace(X) = 1
        # matrix M must be positive semidefinite, with the structure:
        # [[1 , h.T],
        #  [h , X]]
        M = cp.bmat([[1, h.T], [h, X]])
        constraints.append(M >> 0)
        
        # Set pinv constraint manually to prevent rank issues:
        # Trace of X = 1 (zero-mean)
        # Identical to rank 1 constraint via Schur complement
        for k in range(n_coarse):
            constraints.append(cp.trace(R_matrices[k] @ X) >= t)
        
        # Box constraints on h and sum constraint
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        constraints.append(cp.sum(h) == target_sum_coarse)
        
        # Objective: maximize t
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solver settings
        solver_options = {
            "max_iters": 5000,
            "eps": 1e-5
        }
        
        # Solve in time-bound manner, extract best solution seen
        prob.solve(solver=solver=solver_options, verbose=False)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_val = h.value
            X_val = X.value
            t_val = t.value
        else:
            # Fallback to a simpler approach
            h_val = np.ones(n_coarse) * 0.5
            X_val = None
            t_val = -1
    except:
        # Fallback if SDP fails
        h_val = np.ones(n_coarse) * 0.5
        X_val = None
        t_val = -1

    # Step 4: If SDP gave a solution, get approximate vector via eigenvalue decomposition
    if X_val is not None and np.isfinite(X_val).all():
        try:
            eigenvals, eigenvecs = eigh(X_val)
            # Use top eigenpair
            idx_max = np.argmax(eigenvals)
            v1 = eigenvecs[:, idx_max]
            lambda1 = eigenvals[idx_max]
            if lambda1 > 0:  # Makes sense to proceed
                h_hat = np.sqrt(lambda1) * v1
            else:
                h_hat = np.ones(n_coarse) * (target_sum_coarse / n_coarse)
        except:
            h_hat = np.ones(n_coarse) * (target_sum_coarse / n_coarse)
    else:
        h_hat = np.ones(n_coarse) * (target_sum_coarse / n_coarse)
    
    # Perform box and sum projection
    h_hat = waterfill(h_hat, target_sum_coarse)
    if np.any(h_hat > 0):
        h_hat = np.clip(h_hat, 0, 1)
        h_hat = h_hat * (target_sum_coarse / np.sum(h_hat))
    else:
        h_hat = np.ones(n_coarse) * (target_sum_coarse / n_coarse)
    
    # Enforce mirror symmetry
    h_hat = (h_hat + h_hat[::-1]) / 2
    
    # Step 5: Exact evaluation on coarse grid
    def compute_c5_coarse(h):
        g = 1.0 - h
        conv = np.correlate(h, g, mode='full')
        return dx_coarse * np.max(conv)
    
    c5_coarse = compute_c5_coarse(h_hat)
    
    # Use as starting point if promising
    if c5_coarse < c5_best:
        h_best = h_hat.copy()
        c5_best = c5_coarse
    
    # Step 6: Smooth max refinement algorithm
    def smooth_max_objective(h, beta, dx=dx_coarse):
        h = np.clip(h, 0, 1)
        # Full correlation
        g = 1.0 - h
        conv = np.correlate(h, g, mode='full')
        # Shift to allow index access
        conv = conv[n_coarse-1:]
        # Softmax weights w_k = exp(beta*dx*conv_k) / sum(exp(...))
        conv_beta = beta * dx * conv
        conv_beta_max = np.max(conv_beta)
        # Log-sum-exp trick
        log_sum_exp = conv_beta_max + np.log(np.sum(np.exp(conv_beta - conv_beta_max)))
        return log_sum_exp / beta
    
    def smooth_max_gradient(h, beta, dx=dx_coarse):
        h = np.clip(h, 0, 1)
        g = 1.0 - h
        conv = np.correlate(h, g, mode='full')
        conv = conv[n_coarse-1:]
        conv_beta = beta * dx * conv
        conv_beta_max = np.max(conv_beta)
        # Log-sum-exp trick for numerical stability
        log_weights = conv_beta - conv_beta_max
        weights = np.exp(log_weights)
        total_weight = np.sum(weights)
        weights /= total_weight
        
        # Let w = weights * dx
        # Gradient = w * gradient_val
        u = np.correlate(weights, g, mode='full')  # u[i] = sum_k w_k * (1 - h[i+k])
        v = np.correlate(weights[::-1], h, mode='full')  # v[i] = sum_k w_k * h[i-k]
        # Trim to match original x size
        u = u[n_coarse-1:2*n_coarse-1]
        v = v[n_coarse-1:2*n_coarse-1]
        grad = dx * (u - v)
        return grad
    
    # Gradient descent with increasing beta
    h_refined = h_hat.copy()
    beta = 0.5
    beta_max = 50
    patience = 0
    prev_c5 = float('inf')
    
    while beta <= beta_max and time.time() - start_time < budget_s - 10:
        # L-BFGS-B optimization
        def fun(h):
            return smooth_max_objective(h, beta, dx_coarse)
        def jac(h):
            return smooth_max_gradient(h, beta, dx_coarse)
        
        # Need to ensure feasible set: h in [0,1] with sum fixed
        bounds = [(0, 1)] * n_coarse
        constrain_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum_coarse}
        
        try:
            res = minimize(fun, h_refined, jac=jac, 
                           method='L-BFGS-B', bounds=bounds,
                           constraints=constrain_sum,
                           options={'maxiter': 80, 'ftol': 1e-4, 'gtol': 1e-6},
                           callback=lambda h: None)  # meaningless callback
            h_refined = res.x
        except:
            pass
            
        # Projection steps
        h_refined = waterfill(h_refined, target_sum_coarse)
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (target_sum_coarse / np.sum(h_refined))
        # Apply mirror symmetry
        h_refined = (h_refined + h_refined[::-1]) / 2
        
        # Check improvement
        c5_pos = compute_c5_coarse(h_refined)
        if c5_pos < prev_c5 - 1e-7:
            prev_c5 = c5_pos
            patience = 0
        else:
            patience += 1
        if patience > 2:
            break
        beta *= 2
    
    # Step 7: Upsample to fine grid resolution
    h_fine = interp1d(np.arange(n_coarse) * (2.0 / n_coarse),
                     h_refined,
                     kind='linear',
                     fill_value='extrapolate')(np.arange(n_fine) * (2.0 / n_fine))
    
    # Apply clipping and rescaling
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (target_sum_fine / np.sum(h_fine))
    
    # Enforce mirror symmetry at fine grid too
    h_fine = (h_fine + h_fine[::-1]) / 2
    
    # Step 8: Refine on fine grid
    def smooth_max_objective_fine(h, beta, dx=dx_fine):
        h = np.clip(h, 0, 1)
        g = 1.0 - h
        conv = np.correlate(h, g, mode='full')
        conv = conv[n_fine-1:]
        conv_beta = beta * dx * conv
        conv_beta_max = np.max(conv_beta)
        # Log-sum-exp trick
        log_sum_exp = conv_beta_max + np.log(np.sum(np.exp(conv_beta - conv_beta_max)))
        return log_sum_exp / beta
    
    def smooth_max_gradient_fine(h, beta, dx=dx_fine):
        h = np.clip(h, 0, 1)
        g = 1.0 - h
        conv = np.correlate(h, g, mode='full')
        conv = conv[n_fine-1:]
        conv_beta = beta * dx * conv
        conv_beta_max = np.max(conv_beta)
        # Log-sum-exp trick for numerical stability
        log_weights = conv_beta - conv_beta_max
        weights = np.exp(log_weights)
        total_weight = np.sum(weights)
        weights /= total_weight
        
        u = np.correlate(weights, g, mode='full')
        v = np.correlate(weights[::-1], h, mode='full')
        u = u[n_fine-1:2*n_fine-1]
        v = v[n_fine-1:2*n_fine-1]
        grad = dx * (u - v)
        return grad
    
    # L-BFGS-B optimization on fine grid
    beta = 0.5
    beta_max = 50
    patience = 0
    prev_c5 = float('inf')
    h_opt_fine = h_fine.copy()
    
    while beta <= beta_max and time.time() - start_time < budget_s - 20:
        def fun(h):
            return smooth_max_objective_fine(h, beta, dx_fine)
        def jac(h):
            return smooth_max_gradient_fine(h, beta, dx_fine)
        
        bounds = [(0, 1)] * n_fine
        constrain_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum_fine}
        
        try:
            res = minimize(fun, h_opt_fine, jac=jac, 
                           method='L-BFGS-B', bounds=bounds,
                           constraints=constrain_sum,
                           options={'maxiter': 50, 'ftol': 1e-4, 'gtol': 1e-6},
                           callback=lambda h: None)
            h_opt_fine = res.x
        except:
            pass
            
        # Projection steps
        h_opt_fine = waterfill(h_opt_fine, target_sum_fine)
        h_opt_fine = np.clip(h_opt_fine, 0, 1)
        h_opt_fine = h_opt_fine * (target_sum_fine / np.sum(h_opt_fine))
        h_opt_fine = (h_opt_fine + h_opt_fine[::-1]) / 2
        
        # Check improvement
        c5_current = dx_fine * np.max(np.correlate(h_opt_fine, 1-h_opt_fine, mode='full'))
        if c5_current < prev_c5 - 1e-7:
            prev_c5 = c5_current
            patience = 0
            if c5_current < c5_best:
                c5_best = c5_current
                h_best = h_opt_fine.copy()
        else:
            patience += 1
        if patience > 2:
            break
        beta *= 2

    # Step 9: Stochastic polishing
    if h_best is None:
        h_best = h_opt_fine.copy()
        c5_best = dx_fine * np.max(np.correlate(h_best, 1-h_best, mode='full'))
        
    # Apply polishing for last improvements
    epsilon = 0.004
    h_polish = h_best.copy()
    c5_polish = c5_best
    max_polish_steps = int((budget_s - time.time()) * 500)
    
    for step in range(max_polish_steps):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Pick few points to update
        num_perturb = min(6, n_fine // 20)
        update_indices = np.random.choice(n_fine, size=num_perturb, replace=False)
        
        h_trial = h_polish.copy()
        for i in update_indices:
            perturbation = np.random.uniform(-epsilon, epsilon)
            h_trial[i] += perturbation
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (target_sum_fine / np.sum(h_trial))
        h_trial = (h_trial + h_trial[::-1]) / 2
        
        c5_trial = dx_fine * np.max(np.correlate(h_trial, 1-h_trial, mode='full'))
        
        if c5_trial < c5_polish:
            h_polish = h_trial
            c5_polish = c5_trial
    
    # Final return
    return (h_polish, c5_polish, n_fine)
```
