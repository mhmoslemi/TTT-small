import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def decode_blocks_to_h(positions, lengths, heights, n_points):
    """Convert block parameters to step function h"""
    # Ensure non-overlapping and proper ordering
    blocks = sorted(zip(positions, lengths, heights))
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    total_area = 0.0
    for start, length, height in blocks:
        if length <= 0:
            continue
        start_idx = int(start / dx)
        end_idx = int((start + length) / dx)
        end_idx = min(end_idx, n_points)
        if end_idx > start_idx:
            h[start_idx:end_idx] = height
            total_area += height * length
    
    # Normalize to sum = n_points / 2
    if total_area > 0:
        scale = (n_points / 2) / total_area
        h *= scale
    else:
        h.fill(0.5)
    
    return h

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = correlate(h, g, mode='full')
    c5 = np.max(conv) * dx
    return c5

def soft_max_conv(h, beta=1.0):
    """Soft maximum of convolution values"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = correlate(h, g, mode='full')
    return np.log(np.sum(np.exp(beta * dx * conv))) / beta

def soft_max_conv_grad(h, beta=1.0):
    """Gradient of soft max convolution"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    
    # Compute convolution values for all shifts
    conv = correlate(h, g, mode='full')
    
    # Weights for each shift
    exp_conv = np.exp(beta * dx * conv)
    weights = exp_conv / np.sum(exp_conv)
    
    # Compute gradient using proper shifts
    grad = np.zeros_like(h)
    
    # For each shift k in convolution (index s = k + len(h) - 1)
    # We want d/dh_i of conv_s = sum_j h_j * (1 - h_{j+k})
    # So d/dh_i h_j * (1 - h_{j+k}) = (1 - h_{i+k}) * (j=i) 
    #                                 - h_{i-k} * (j=i+k)
    # But we are compute the total gradient for fixed k
    # Where (i+k) and (i-k) are valid indices
    
    # Compute shifted arrays for computing derivatives
    for k in range(-len(h) + 1, len(h)):
        # Build the derivative contribution for shift k (which corresponds to index k + len(h) - 1)  
        if k >= 0:
            # h[i] contributes to shift k by (1 - h[i+k]) term
            # This corresponds to summing over j = i-k, so for h_j(1-h_{j+k}) term
            # Derivative w.r.t. h_i is (i = j+k) * (1-h_{j+k}) - (j = i-k) * h_{i-k}
            # Since j = i-k, this term is 0 unless i = j+k, i.e., i = i-k+k
            # which is always true in loop. This means we go back to the h[i+k] above.
            # For k >= 0: shift right (i+k), we represent as right shift and left shift
            # For shift k from i, contribution to h[i]: (1-h_{i+k})
            # For shift -k: i-k is valid, it contributes to h[i]: - h_{i-k}
            # Therefore we build the shift k capability
            s = k + len(h) - 1  # Convert shift k to index in full convolution array
            # This should be handled in a vectorized way
            
        # Method: displacement using np.roll
        # h[i] contributes to shift k = s - (n-1) via (1 - h[i+k]) and -h[i-k] terms
        shifted_right = np.roll(h, k * (-1))
        shifted_left = np.roll(h, k * 1)
        g_right = 1.0 - shifted_right
        g_left = shifted_left
        
        # The contribution of term (1-h[i+k]) * (1) which is a derivative of h[i] term
        # And of term - h[i-k] * (1) (which is a derivative of h[i] term if some trick)
        # To compute ∂_i [ \sum_j h_j * (1 - h_{j+k}) ] = (1 - h_{i+k}) if only that term contributes
        # This is:
        # For ∂ φ_s / ∂ h_i with φ_s = h_i * (1 - h_{i+k}) 
        # Taking derivative w.r.t. h_i, it's the contribution from this φ_s term:
        # 1 - h_{i+k} if i+k is valid, else 0
        # And from term h_{i-k} * (1 - h_i): file additional if valid.
        # Neat approach: create extended shifted array.
        
        # Direct more precise vectorization approach:
        # We inflow mainly from: (1 - h_{i+k}) 
        # and from contribution of h_{i-k} * (1 - h_i), which we rewrite as -h_{i-k}
        # So we generalize `- h_{i-k}` for structuring
        
        # More robust: we use np.pad
        # Consider padded arrays to overlap computation with np.pad j? nope  
        # Let's create the right vector we want:
        # A contribution to final answer ∂_i = ∑ s weights[s] * (1 - h_{i+k[s]})
        # for conv_s = ∑_{j} h_j (1 - h_{j+k}) we compute d/dh_i
        # Realization of this is the row/column:
        # After padding h with zeros at both ends, left shift and right shift
        
        # Pad h for overflow handling - same as scipy correlate with mode='full' -> no padding needed?
        # Actually, h concatenate with g (1-h) creates full signal
        # conv is of size 2*N - 1 (h -> N, 1-h -> N)  
        # Index for shift k = s - N + 1 
        # For simplicity, we'll distribute in correct locations
        # For j such that h[j] * (1-h[j+k]) contributes:
        #   if k>0: value contributes when j = i => h[i] *(1 - h[i+k]) studied
        #   if k<0: value contributes when j = i+k => h[j+k] *(1 - h[j]) = h[i]* (1 - h[i-k])  
        
        # By the entire mapping of conv_s = \sum_j h_j * (1 - h_{j+s}) 
        # -> d/dh_i = sum_s  weights[s] * (1 - h_{i+s}) * (i = j[s])  
        #               if i+j = s, then j=s-i 
        #               So (i=j) * (1 - h_{s})
    
    # let's place that into a more clean and working different form:
    grad = np.zeros_like(h)
    full_conv_len = 2 * len(h) - 1
    for k in range(-len(h) + 1, len(h)):
        s = k + len(h) - 1
        # contribution to h[i] from term in conv[s]
        if k >= 0:
            g_contrib = 1.0 - h[-k:]  # right indices: h[0:k], but reverse
            g_contrib = np.pad(g_contrib, (0, k), mode='constant', constant_values=0)
            grad += weights[s] * g_contrib
        else:
            # mapping to h[i] in term k = i - j
            # k = i - j, so j = i - k
            # and the h_j * (1 - h_{j+k}) term  
            # where j+k = i
            # h_{i-k} * (1 - h_i) contributes to i in 1-h_i (i.e. + weights[s] * - h[i-k])
            # s = k + N - 1 => k = s - (N-1)
            h_left = h[:len(h)+k]  # from index 0 to len(h)+k-1 (if k<0)
            h_left = np.pad(h_left, (-k, 0), mode='constant', constant_values=0)  # pad left
            h_left = -h_left * weights[s]
            grad += h_left[:len(h)]
            
    return dx * grad

def project_onto_sum(h, target_sum):
    """Project h to have sum == target_sum"""
    current_sum = np.sum(h)
    diff = current_sum - target_sum
    if abs(diff) < 1e-9:
        return h
    if np.all(h <= 0):
        h.fill(target_sum / len(h))
    else:
        delta = diff / len(h)
        h = h - delta
        h = np.clip(h, 0, 1)

    # Re-clip to maintain sum (= target_sum). Ensure sum within tolerance
    # Iterate until satisfied by exact sum if needed w/ simple Newton-like method
    # Or simple tweaks
    current_sum = np.sum(h)
    while abs(current_sum - target_sum) > 1e-9:
        # Waterfilling approach: subtract adjustment from all and clip
        adjust = (current_sum - target_sum) / len(h)
        h = h - adjust
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)

    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2  # That's sum(h) = 1 = integral of h

    # Parameters for global search: M=4 blocks
    M = 4
    bounds = [(0, 2)] * M + [(0, 2)] * M + [(0, 1)] * M  # start, length, height
    # Define global optimizer
    def obj_blocks(x):
        if time.time() - start_time > budget_s - 0.5:
            raise RuntimeError("Time limit approached")
        positions = x[:M]
        lengths = x[M:2*M]
        heights = x[2*M:]
        h = decode_blocks_to_h(positions, lengths, heights, n_points)
        # Apply symmetry for cheap normalization: not required in case of internal handling
        h_sym = (h + h[::-1]) / 2
        h_sym = project_onto_sum(h_sym, target_sum)
        try:
            return evaluate_c5(h_sym)
        except:
            return float('inf')
    
    # Differential Evolution: only (M*3) dimensions, exploit global structure
    try:
        de_result = differential_evolution(obj_blocks, bounds, seed=seed, maxiter=60,
                                          popsize=15, disp=False, polish=False)
        h_global = decode_blocks_to_h(
            de_result.x[:M], 
            de_result.x[M:2*M], 
            de_result.x[2*M:], 
            n_points
        )
        h_global = (h_global + h_global[::-1]) / 2  # symmetry
        h_global = project_onto_sum(h_global, target_sum)
        c5_global = evaluate_c5(h_global)
        best_c5 = c5_global
        best_h = h_global.copy()
    except:
        # Fallback
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_c5(best_h)
        c5_global = best_c5
        
    # Gradient descent refinement on refined solution:
    # Try increasing β smoothly
    h_refined = best_h.copy()
    c5_refined = best_c5
    beta_vals = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]  # soft max scaling
    lambda_vals = [10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0]
    
    for beta_idx, (beta, lambda_coeff) in enumerate(zip(beta_vals, lambda_vals)):
        if time.time() - start_time > budget_s - 1:
            break
            
        def obj_func(h):
            if time.time() - start_time > budget_s - 0.5:
                raise RuntimeError("Time limit approached")
            h = project_onto_sum(h, target_sum)
            return soft_max_conv(h, beta) + lambda_coeff * (np.sum(h) - target_sum)**2

        def grad_func(h):
            if time.time() - start_time > budget_s - 0.5:
                raise RuntimeError("Time limit approached")
            h = project_onto_sum(h, target_sum)
            grad = soft_max_conv_grad(h, beta)
            # Penalty gradient on sum
            grad += 2 * lambda_coeff * (np.sum(h) - target_sum) * np.ones_like(h)
            return grad

        # Use L-BFGS-B 
        try:
            res = minimize(
                obj_func, 
                h_refined, 
                method='L-BFGS-B',
                jac=grad_func,
                options={'disp': False, 'maxiter': min(150, int(2000/beta))},
                bounds=[(0,1)] * n_points
            )
            h_refined = res.x
            h_refined = project_onto_sum(h_refined, target_sum)
            # Check exact C5 after modification
            c5_refined = evaluate_c5(h_refined)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
        except:
            pass

    # Final local search with hill climbing  
    num_hill = 0
    while time.time() - start_time < budget_s - 0.5:
        num_hill += 1
        h_trial = best_h.copy()
        # Perturb
        r = min(5, n_points // 20)
        indices = np.random.choice(n_points, r, replace=False)
        for i in indices:
            noise = np.random.uniform(-0.005, 0.005)
            h_trial[i] += noise
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = project_onto_sum(h_trial, target_sum)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial

    return (best_h, best_c5, n_points)
