import numpy as np

def proj_sum(h, target_sum):
    """Project h to have the given target sum using binary search for lambda"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        lamb = (a + b) / 2
        h_new = np.clip(h + lamb, 0.0, 1.0)
        s_new = np.sum(h_new)
        if s_new < target_sum:
            b = lamb
        else:
            a = lamb
    lamb = (a + b) / 2
    h_new = np.clip(h + lamb, 0.0, 1.0)
    return h_new

def c5_of_h(h):
    """Compute C5 for a given step function h using FFT"""
    dx = 2.0 / len(h)
    n = len(h)
    # Zero-pad for full convolution
    H = np.fft.rfft(h, n=2*n-1)
    G = np.fft.rfft(1-h, n=2*n-1)
    conv = np.fft.irfft(H*G)[:2*n-1]
    result = dx * np.max(conv)
    return result

def compute_convolutions(h):
    """Compute all convolutions efficiently using FFT approach"""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, n=2*n-1)
    G = np.fft.rfft(1-h, n=2*n-1)
    conv = np.fft.irfft(H*G)[:2*n-1]
    return conv * dx

def smooth_max_gradient(h, beta=0.5):
    """Compute gradient of smooth max approximation"""
    n = len(h)
    dx = 2.0 / n
    conv = compute_convolutions(h)
    # Avoid overflow in exponential
    conv_scaled = beta * conv
    max_val = np.max(conv_scaled)
    exp_vals = np.exp(conv_scaled - max_val)
    Z = np.sum(exp_vals)
    weights = exp_vals / Z
    
    # Compute weighted gradient using a trick:
    # see: (1-h_{i+k}) - h_{i-k}
    right_term = np.pad(1-h, (0, n-1))[:n]
    left_term = np.pad(h, (n-1, 0))[n-1:]
    grad_per_k = right_term - left_term
    
    # For each k, the gradient term contribution
    # weight_k * [ (1-h_{i+k}) - h_{i-k} ] 
    # Take padded versions for offsetting correctly
    g = np.zeros(n)
    for k in range(2*n-1):
        if k < n:
            g += weights[k] * grad_per_k[k]
        else:
            # Reflect index for negative shifts
            idx = 2*n-2-k
            g -= weights[k] * grad_per_k[idx]
    
    return g

def local_refinement(h_start, beta=0.5):
    """Refinement using smooth max gradient descent"""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    t_start = time.time()
    lambda_pen = 10.0
    
    # Apply symmetry
    h = (h + h[::-1]) / 2.0
    
    for _ in range(200):  # Max iter
        if time.time() - t_start > 150:
            break
        grad = smooth_max_gradient(h, beta)
        
        # Apply constraint (project on sum)
        h -= 0.01 * grad
        h = proj_sum(h, n/2)
        h = (h + h[::-1]) / 2.0
        
        if np.linalg.norm(grad) < 1e-6:
            break
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_coarse = 150  # coarse grid for everything but final ref
    n_fine = 300    # final grid size
    dx_coarse = 2.0 / n_coarse

    # Start search from finest grid size with initial_h_values

    # Interpolate initial guess to coarse grid size
    if len(initial) != n_coarse:
        x_orig = np.linspace(0, 2, len(initial))
        x_new = np.linspace(0, 2, n_coarse)
        h_interp = np.interp(x_new, x_orig, initial)
        h_start = proj_sum(h_interp, n_coarse/2)
    else:
        h_start = proj_sum(initial, n_coarse/2)
    
    # Symmetrize for initial
    h_sym = (h_start + h_start[::-1]) / 2
    h_sym = proj_sum(h_sym, n_coarse/2)
    
    # Evaluate coarse
    best_c5_coarse = c5_of_h(h_sym)
    best_h_coarse = h_sym.copy()
    
    # CMA-ES Stage (on coarse grid)
    # Use simplified CMA-ES-inspired uniform search
    best_c5 = best_c5_coarse
    best_h = best_h_coarse.copy()
    
    # update to 200 iterations
    for run_iter in range(200):
        if time.time() - start_time > budget_s - 100:
            break
            
        # Take a random perturbation of the current solution
        trial_h = best_h.copy()
        num_perturb = int(n_coarse * 0.1)
        indices = np.random.choice(n_coarse, size=num_perturb, replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            trial_h[i] += delta
        
        # Project to feasible set
        trial_h = np.clip(trial_h, 0.0, 1.0)
        trial_h = proj_sum(trial_h, n_coarse/2)
        trial_h = (trial_h + trial_h[::-1]) / 2
        
        c5_trial = c5_of_h(trial_h)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = trial_h.copy()
    
    # Local refinement with beta continuation
    h_fine = best_h.copy()
    n = n_coarse
    for _ in range(150):
        if time.time() - start_time > budget_s - 100:
            break
        
        # Interpolate
        h_interp = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n), best_h)

        h_fine = h_interp.copy()
        
        # Symmetrize
        h_fine = (h_fine + h_fine[::-1]) / 2
        
        # beta continuation refinement
        beta = 0.5
        h_refined = h_fine.copy()

            if time.time() - start_time > budget_s - 50:
                break
            h_refined = local_refinement(h_refined, beta)
            beta *= 2.0
            if np.linalg.norm(h_refined - h_fine) < 1e-4:
                break
            h_fine = h_refined.copy()
        
        # Evaluate C5 on fine grid
        c5_fine = c5_of_h(h_refined)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_refined.copy()
        else:
            h_refined = h_fine.copy()

    # For approximate 50s budget
    while time.time() - start_time < budget_s - 5:
        h_trial = best_h.copy()
        num_swap = min(6, n_fine // 20)
        indices = np.random.choice(n_fine, size=num_swap, replace=False)
        epsilon = 0.004
        for i in indices:
            delta = np.random.uniform(-epsilon, epsilon)
            h_trial[i] += delta
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = proj_sum(h_trial, n_fine/2)
        h_trial = (h_trial + h_trial[::-1]) / 2
        
        c5_trial = c5_of_h(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
    
    # Final check
    result_c5 = c5_of_h(best_h)
    return (best_h, result_c5, n_fine)
