import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.interpolate import interp1d

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 140  # Discretization points
    dx = 2.0 / N
    S = N // 2  # Target sum
    
    # Early exit if not enough time
    if time.time() - start_time > budget_s - 10:
        # Fallback to a simple construction
        h = np.zeros(N)
        h[:S] = 1.0
        h = h * (N / 2) / np.sum(h)
        return (h, c5_of_h(h), N)
    
    # Precompute indices for each shift
    shift_indices = {}
    for k in range(N):
        indices = [i for i in range(N - k)]
        shift_indices[k] = indices
    
    # Build the model
    x = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(x) == S, t >= 0]
    
    # Auxiliary variables and constraints for each shift
    w_vars = {}
    for k in range(N):
        indices = shift_indices[k]
        if indices:
            w_vars[k] = cp.Variable(len(indices), boolean=True)
            # McCormick inequalities
            for idx, i in enumerate(indices):
                constraints.append(w_vars[k][idx] <= x[i])
                constraints.append(w_vars[k][idx] <= x[i+k])
                constraints.append(w_vars[k][idx] >= x[i] + x[i+k] - 1)
            # Overlap bound: sum of w_vars >= S - t
            constraints.append(cp.sum(w_vars[k]) >= S - t)
    
    # Fix first variable to break symmetry
    constraints.append(x[0] == 1)
    
    # Binary search for tight bound
    L, U = 0.0, float(S)
    feasible_sol = None
    best_t = float('inf')
    
    # Binary search loop
    while U - L > 1e-4 and time.time() - start_time < budget_s - 10:
        T = (L + U) / 2
        # Fix the constraint for this T
        temp_constraints = constraints[:-1] + [t <= T]  # Reuse other constraints
        
        # Create and solve problem
        prob = cp.Problem(cp.Minimize(0), temp_constraints)
        
        try:
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=30)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Check if we got a better solution
                if t.value < best_t:
                    best_t = t.value
                    feasible_sol = x.value.copy()
                U = T
            else:
                L = T
        except:
            L = T
    # Finalization
    if feasible_sol is None:
        # Fallback construction
        h = np.zeros(N)
        h[:S] = 1.0
        h = h * (N / 2) / np.sum(h)
        return (h, c5_of_h(h), N)
    
    # Convert binary solution to step function h
    h = np.array(feasible_sol, dtype=float)
    
    # Apply fractional polishing
    max_C5 = c5_of_h(h)
    for _ in range(20):  # Limited iterations
        if time.time() - start_time > budget_s - 5:
            break
        # Find current worst shift
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (N - 1)
        # Find indices contributing to this shift
        indices = shift_indices[k] if k >= 0 else shift_indices[-k]
        if not indices:
            continue
        # Try to improve by swapping two elements
        best_improvement = 0.0
        best_i, best_j = None, None
        delta_step = 0.02
        for i in indices:
            for j in range(N):
                if (h[i] > 0.001 and h[j] < 0.999 and i != j):
                    # Attempt a small swap
                    test_h = h.copy()
                    test_h[i] = min(1.0, test_h[i] + delta_step)
                    test_h[j] = max(0.0, test_h[j] - delta_step)
                    new_C5 = c5_of_h(test_h)
                    improvement = max_C5 - new_C5
                    if improvement > best_improvement and improvement > 1e-8:
                        best_improvement = improvement
                        best_i, best_j = i, j
        if best_i is not None:
            h[best_i] += delta_step
            h[best_j] -= delta_step
            max_C5 = c5_of_h(h)
        else:
            break
    
    # Final fix-up to ensure mass constraint
    sum_h = np.sum(h)
    if abs(sum_h - S*dx) > 1e-10:
        # Ensure sum == S*dx
        scale_factor = S * dx / sum_h
        h *= scale_factor
    
    # Clamp and project
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h - S*dx) > 1e-10:
        # Adjust uniformly to maintain sum
        adjustment = (S*dx - sum_h) / N
        h += adjustment
        h = np.clip(h, 0, 1)
    
    return (h, c5_of_h(h), N)
