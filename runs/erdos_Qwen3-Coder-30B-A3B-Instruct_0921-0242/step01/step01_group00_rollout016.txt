import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_sdp_solution(n):
    """Solve SDP relaxation to get optimal fractional h"""
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.diag(X) == h,  # Diagonal constraint
        X >> 0,           # PSD constraint
        cp.sum(h) == n / 2,  # Sum constraint
        0 <= h, h <= 1    # Box constraints
    ]
    
    # Add autocorrelation constraints for k = 1, ..., n-1
    for k in range(1, n):
        # Create shift matrix T_k 
        T_k = np.zeros((n, n))
        for i in range(n - k):
            T_k[i, i + k] = 1
        constraints.append(cp.trace(T_k @ X) >= t)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-4, max_iters=5000)
    
    if prob.status != cp.OPTIMAL:
        # fallback to approximate or return regular solution
        return None
        
    h_sdp = h.value
    if h_sdp is not None:
        # ensure full feasibility and sum constraint
        h_sdp = np.clip(h_sdp, 0, 1)
        current_sum = np.sum(h_sdp)
        diff = n/2 - current_sum
        
        if abs(diff) > 1e-8:
            # Waterfilling approach to correct sum precisely
            # Ensure we can actually add diff/n without hitting bounds
            if diff/n < 1 - np.max(h_sdp):
                h_sdp += diff / n
                h_sdp = np.clip(h_sdp, 0, 1)
        return h_sdp
    return None

def round_sdp_vector(h_sdp, n):
    """Randomized rounding of SDP solution to binary candidate"""
    # Factorize X = V @ V^T using eigen-decomposition
    # Here we assume h_sdp has been used to construct approximate X with correct diag
    # and good rank-1 structure.
    V = np.random.randn(n, n)
    # Use the fact that h_sdp represents diagonal of X for that approximation
    
    best_h = h_sdp.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Try a few samples
    num_rounding = 20
    for r in range(num_rounding):
        g = np.random.randn(n)
        y = h_sdp + 0.1*g  # small Gaussian perturbation to make better after sorting
        
        # Create binary vector by thresholding to get exact sum n/2
        sorted_y = np.sort(y)[::-1]
        # Find the threshold that makes exactly n/2 entries >= threshold
        k = int(n/2)
        if k == 0:
            threshold = -1e10
        else:
            threshold = sorted_y[k-1]
        if k < n and sorted_y[k] == threshold:
            # There are ties at that threshold; prefer a slightly higher threshold
            # to avoid having more than n/2 ones
            threshold = np.min(y[y >= threshold]) - 1e-12
        
        b = (y > threshold).astype(float)
        
        # Ensure sum remains exactly n/2 using waterfilling if necessary
        current_sum = np.sum(b)
        diff = n/2 - current_sum
        if abs(diff) > 1e-8:
            b = np.clip(b + diff/n, 0, 1)
            b = np.clip(b, 0, 1)
        # Double check
        actual_sum = np.sum(b)
        if abs(actual_sum - n/2) > 1e-8:
            tol = 1e-8
            # Check if they're close
            if abs(actual_sum - n/2) > tol:
                # Manually construct it exactly with O(n) approach
                indices = np.argsort(b)[::-1]
                adjust_count = int(actual_sum - n/2)
                if adjust_count > 0:
                    for i in range(adjust_count):
                        if b[indices[i]] > 0:
                            b[indices[i]] = 0
                elif adjust_count < 0:
                    for i in range(-adjust_count):
                        if b[indices[n-i-1]] < 1:
                            b[indices[n-i-1]] = 1
                # Re-normalize if necessary
                
        c5 = evaluate_c5(b)
        if c5 < best_c5:
            best_c5 = c5
            best_h = b.copy()
    return best_h, best_c5

def compute_smooth_max_gradient(h, beta=0.5):
    """Compute gradient of smooth maximum approximation"""
    dx = 2.0 / len(h)
    # Compute all I_k values
    I_k = []
    n = len(h)
    for k in range(1, n):
        # Compute sum over valid i's
        corr = 0
        for i in range(n - k):
            corr += h[i] * h[i + k]
        # I_k = dx * (1 - corr)
        I_k.append(dx * (1 - corr))
    
    # Compute log-sum-exp of I_k * beta
    I_array = np.array(I_k)
    exp_beta_I = np.exp(beta * I_array)
    Z = np.sum(exp_beta_I)
    W = exp_beta_I / Z  # weights
    
    # Now compute gradient
    grad = np.zeros(n)
    for k in range(1, n):
        # k corresponds to column of T_k matrix (for primal variable indices i)
        # ∂I_k/∂h_i: 
        #   for i: i is in the sum, contribution is -dx * h_{i+k}  
        #   for i+k: contribute -dx * h_{i+k} 
        # but it is easier to loop through i
        for i in range(n - k):
            # If we're computing d/dh_i
            # ∂I_k/∂h_i = -dx * h_{i+k} and ∂I_k/∂h_{i+k} = -dx * h_i
            if i < len(grad):
                grad[i] -= dx * W[k - 1] * h[i + k]
            if i + k < len(h):
                grad[i + k] -= dx * W[k - 1] * h[i]
    return grad

def project_to_feasible(h, n, eps=1e-10):
    """Ensure sum = n/2 and h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) <= eps:
        return h
    # Add correction uniformly
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def smooth_max_refinement(h_start, max_time_s):
    """Apply smooth-max gradient descent to refine h_start"""
    start_time = time.time()
    h = h_start.copy()
    dx = 2.0 / len(h)
    
    # Initialize with a rough value for beta
    beta = 0.5  # starting beta for smooth max
    max_beta = 50.0
    beta_growth = 1.5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iter in range(1000):
        if time.time() - start_time > max_time_s:
            break
        # compute smooth max gradient with current beta
        grad = compute_smooth_max_gradient(h, beta)
        
        # Projected gradient descent with backtracking line search
        alpha = 1.0
        max_backtrack = 10
        for _ in range(max_backtrack):
            h_trial = h - alpha * grad
            # Project to feasible set
            h_trial = project_to_feasible(h_trial, len(h))
            c5_trial = evaluate_c5(h_trial)
            if c5_trial <= best_c5:
                # Accept
                h = h_trial
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h.copy()
                break
            else:
                alpha *= 0.5
        if iter % 20 == 0:
            beta = min(beta * beta_growth, max_beta)
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameter choices
    n = 128
    max_time = budget_s - 10  # save at least 10s
    
    best_c5 = float('inf')
    best_h = None
    
    # Stage 1: SDP relaxation
    print("Starting SDP stage...")
    h_sdp = compute_sdp_solution(n)
    if h_sdp is None:
        print("SDP failed, falling back to uniform vector")
        h_sdp = np.ones(n) * 0.5  # fallback
        
    c5_sdp = evaluate_c5(h_sdp)
    print(f"SDP finished with C5 = {c5_sdp}")
    
    # Stage 2: Randomized rounding
    print("Starting rounding stage...")
    h_rounded, c5_rounded = round_sdp_vector(h_sdp, n)
    if c5_rounded < best_c5:
        best_c5 = c5_rounded
        best_h = h_rounded.copy()
    print(f"Rounding finished with C5 = {c5_rounded}")
    
    # Stage 3: Smooth max refinement 
    print("Starting refinement stage...")
    refine_time = max_time - (time.time() - start_time)
    if refine_time > 1:
        h_refined, c5_refined = smooth_max_refinement(h_rounded, refine_time)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    print(f"Refinement finished with C5 = {best_c5}")
    
    # Stage 4: Local discrete swaps
    print("Starting local swaps...")
    if time.time() - start_time < max_time - 1:
        # Start with best_h found so far
        h = best_h.copy()
        dx = 2.0 / n
        max_iter = 500
        for i in range(max_iter):
            if time.time() - start_time > max_time - 1:
                break
            idx1 = np.random.randint(0, n)
            idx2 = np.random.randint(0, n)
            if idx1 == idx2:
                continue
            h1, h2 = h[idx1], h[idx2]
            # Only perform if both can be swapped within bounds
            if 0 <= h1 <= 1. and 0 <= h2 <= 1.:
                delta = min(h1, 1. - h2, 0.02)
                if delta == 0:
                    continue
                # Try to swap
                h_trial = h.copy()
                h_trial[idx1] -= delta
                h_trial[idx2] += delta
                # Ensure feasibility after swap
                h_trial = project_to_feasible(h_trial, n)
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
    
    return (best_h, best_c5, n)
