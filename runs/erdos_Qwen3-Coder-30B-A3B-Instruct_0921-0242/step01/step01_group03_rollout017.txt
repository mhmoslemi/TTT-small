```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    N = 140  # Use step size that allows for quick MILP solve
    dx = 2.0 / N
    
    # For fast correlation computation (FFT-based)
    def compute_correlation(h):
        # compute overlaps for all shifts k
        g = 1 - h
        corr_full = np.correlate(h, g, mode='full')  # length 2*N - 1
        # Take max of all valid k values
        max_corr = np.max(corr_full)
        return max_corr * dx

    # Build the binary MILP model
    def build_milp_model(N, t_bound=None):
        x = cp.Variable(N, boolean=True)
        
        # list of w variables, grouped by shift
        w_vars = []
        w_vars_dict = {}
        total_w_vars = 0
        w_indices_start = []

        # Precompute for each shift k, list of pairs (i, i+k) that are valid
        # i ranges from 0 to N-1-k (inclusive)
        shifts_w_pairs = {} # k -> [(i, i+k), ...] with i+k<N
        for k in range(N):
            if k == 0:
                # just indices, no pairs needed
                shifts_w_pairs[k] = [(i, i+k) for i in range(N)]
            else:
                shifts_w_pairs[k] = [(i, i+k) for i in range(N - k)]

            w_vars_dict[k] = [] # list of cp.Variable(k) 
            for i, j in shifts_w_pairs[k]:
                w_vars_dict[k].append(cp.Variable(boolean=True))
                total_w_vars += 1

        # Flatten w list to pass to constraints
        w_list = [w for k in range(N) for w in w_vars_dict[k]]
        # Create constraint dictionary from index pairs
        w_indices_per_shift = {}
        for k in range(N):
            w_indices_per_shift[k] = [idx for idx in range(len(w_list)) if idx in range(len(w_list)) and any((i, j) == (idx % (N-k), (idx % (N-k)) + k) for i,j in shifts_w_pairs[k])]

        # Constraints
        constraints = []

        # Mass constraint: sum x_i = N/2
        constraints.append(cp.sum(x) == N/2)

        # AND constraints: for each pair (i,j) with j = i+k
        for k in range(N):
            pairs = shifts_w_pairs[k]
            w_list_k = w_vars_dict[k]
            w_idx = 0
            for i, j in pairs:
                w = w_list_k[w_idx]
                # w <= x[i]
                constraints.append(w <= x[i])
                # w <= x[j]
                constraints.append(w <= x[j])
                # w >= x[i] + x[j] - 1
                constraints.append(w >= x[i] + x[j] - 1)
                w_idx += 1

        # Overlap bound: sum( w_{i,k} ) >= N/2 - t  (for each shift k)
        t = cp.Variable()
        if t_bound is not None:
            t = t_bound  # fixed value, for feasibility
  
        for k in range(N):
            pairs = shifts_w_pairs[k]
            w_list_k = w_vars_dict[k]
            sum_w = cp.sum(w_list_k)
            # Add constraint sum(w_{i,k}) >= N/2 - t
            if k == 0:
                constraints.append(sum_w <= N/2 - t)
            else:
                constraints.append(sum_w <= N/2 - t)

        # Optionally add symmetry breaking
        constraints.append(x[0] == 1)
        
        # Objective (or placeholder if solving feasibility)
        if t_bound is None:
            objective = cp.Minimize(t)
        else:
            objective = cp.Minimize(0)  # feasible problem
            
        return cp.Problem(objective, constraints), x, w_list, t

    # Binary search to find solution
    L = 0
    U = N/2
    best_feasible_t = N/2
    best_feasible_x = None
    best_feasible_c5 = N/2 * dx  # worst possible

    # Main loop
    while (U - L > 1e-4) and (time.time() - start_time < budget_s - 2):
        T = (L + U) / 2.0
        prob, x, w_list, t = build_milp_model(N, T)
        try:
            # Solve with a timeout
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=30)  # 30 seconds timeout
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # get x values and convert to array
                x_sol = np.array([x[i].value for i in range(N)])
                # validation
                if np.isclose(np.sum(x_sol), N/2, atol=1e-3):
                    # compute c5
                    h = x_sol.astype(float) 
                    c5_val = compute_correlation(h)
                    if c5_val < best_feasible_c5:
                        best_feasible_c5 = c5_val
                        best_feasible_x = x_sol.copy()
                    U = T
                else:
                    L = T
            else:
                L = T # reject this T
        except:
            L = T # reject this T
            
        # early termination if good enough
        if best_feasible_c5 <= 0.379:
            break

    if best_feasible_x is not None:
        h_opt = best_feasible_x
        c5_opt = best_feasible_c5
    else:
        # fallback: simple construction
        h_opt = np.zeros(N)
        h_opt[:N//2] = 1.0
        c5_opt = compute_correlation(h_opt)

    # Polish with fractional moves
    def improve_with_fractional_moves(h, num_passes=20, delta_schedule=[0.05, 0.02, 0.01]):
        for iteration in range(num_passes):
            if time.time() - start_time > budget_s - 0.5:
                break
                
            worst_k = np.argmax(np.correlate(h, 1-h, mode='full'))
            # full_width = len(h)*2 - 1
            if worst_k < len(h):
                k = worst_k
            else:
                k = worst_k - len(h) + 1
                
            # Find all contributing pairs
            i_vals = range(min(N, N - k))  # valid i such that i+k < N
            best_delta = 0
            best_i = -1
            best_j = -1
            # Try small changes to improve
            for delta in delta_schedule:
                for i in i_vals:
                    j = i + k
                    if j < N and h[i] > delta and h[j] < 1 - delta:
                        new_h = h.copy()
                        new_h[i] -= delta
                        new_h[j] += delta
                        # compute overlap integral with new h
                        new_c5 = compute_correlation(new_h)
                        if new_c5 < c5_opt:
                            c5_opt = new_c5
                            h = new_h.copy()
                            best_delta = delta
                            best_i = i 
                            best_j = j

    # Post-process polishing
    improve_with_fractional_moves(h_opt, num_passes=20)

    # Final swap hill climbing
    def swap_hill_climbing(h, max_attempts=30000):
        nonlocal c5_opt
        for attempt in range(max_attempts):
            if time.time() - start_time > budget_s - 0.5:
                break
            # Select two random indices
            i = np.random.randint(0, N)
            j = np.random.randint(0, N)
            if i == j or (h[i] == 0 and h[j] == 1):
                continue  # avoid invalid swaps
            # Make the swap and check
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            c5_new = compute_correlation(h_new)
            if c5_new < c5_opt:
                c5_opt = c5_new
                h = h_new.copy()
        return h

    # Final swap hill climbing
    h_opt = swap_hill_climbing(h_opt, max_attempts=30000)

    return (h_opt, c5_opt, N)
```
