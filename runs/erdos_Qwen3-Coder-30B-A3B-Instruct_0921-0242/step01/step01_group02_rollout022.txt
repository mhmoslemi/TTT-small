import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import math

def c5_fft(h):
    """Compute C5 using FFT for faster correlation."""
    g = 1 - h
    H = rfft(h)
    G = rfft(g)
    corr = irfft(H * G, len(h) * 2 - 1)
    return np.max(corr)

def repair_weight(h, target_sum):
    """Repair to exact weight."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = int(target_sum - current_sum)
    if diff > 0:  # need more 1s
        zeros = np.where(h == 0)[0]
        if len(zeros) >= diff:
            idxs = np.random.choice(zeros, diff, replace=False)
            h[idxs] = 1
    elif diff < 0:  # need more 0s
        ones = np.where(h == 1)[0]
        if len(ones) >= -diff:
            idxs = np.random.choice(ones, -diff, replace=False)
            h[idxs] = 0
    return h

def crossover(p1, p2):
    """Two-point crossover."""
    n = len(p1)
    c1, c2 = np.random.randint(1, n-1, 2)
    if c1 > c2:
        c1, c2 = c2, c1
    child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
    return repair_weight(child, n/2)

def mutate_swap(h):
    """Swap mutation."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > 0 and len(zeros) > 0:
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        h[i1], h[i0] = h[i0], h[i1]
    return h

def hill_climb(h, dx, max_swaps=500):
    """Targeted hill climbing."""
    h = h.copy()
    for _ in range(max_swaps):
        c = correlate(h, 1 - h, mode="full")
        k_star = np.argmax(c) - len(h) + 1
        best_delta = 0
        best_i0, best_i1 = 0, 0
        candidate_swaps = []
        for _ in range(200):
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break
            i0 = np.random.choice(zeros)
            i1 = np.random.choice(ones)
            delta = 0
            if 0 <= i1 + k_star < len(h):
                delta -= dx * h[i1 + k_star]
            if 0 <= i0 + k_star < len(h):
                delta += dx * h[i0 + k_star]
            if delta < best_delta:
                best_delta = delta
                best_i0, best_i1 = i0, i1
        if best_delta >= 0:
            break
        h[best_i0], h[best_i1] = h[best_i1], h[best_i0]
    return h

    start_time = time.time()
    n = 280
    dx = 2.0 / n
    target_sum = n / 2.0
    pop_size = 200
    elite = 5
    p_mut = 0.2
    
    # Initial population
    population = []
    
    def add_individual(h):

        population.append(h)
    
    # Interpolated initial_h_values

    h_orig_len = len(h_orig)
    x_orig = np.linspace(0, 2, h_orig_len)
    x_new = np.linspace(0, 2, n)
    h_interp = np.interp(x_new, x_orig, h_orig)
    add_individual(h_interp)
    
    # Two block deterministic
    for _ in range(10):
        h = np.zeros(n)
        h[:n//4] = 1
        h[n//2:n//2+n//4] = 1
        np.random.shuffle(h)
        add_individual(h)
    
    # Alternating pattern
    for _ in range(10):
        h = np.zeros(n)
        h[::2] = 1
        add_individual(h)
    
    # Random balanced
    for _ in range(179):
        h = np.zeros(n)
        ones_idx = np.random.choice(n, int(n/2), replace=False)
        h[ones_idx] = 1
        add_individual(h)
    
    # Evaluate initial population
    fitness = [-c5_fft(h) for h in population]
    
    # Evolve
    best_h = None
    best_c5 = float('inf')
    
    gen = 0
    while time.time() - start_time < budget_s - 20:
        # Selection (tournament)
        new_pop = []
        for _ in range(pop_size):
            i1, i2 = np.random.randint(0, pop_size, 2)
            if fitness[i1] > fitness[i2]:
                new_pop.append(population[i1].copy())
            else:
                new_pop.append(population[i2].copy())
        
        # Crossover and mutation
        offspring = []
        for i in range(0, pop_size - elite, 2):
            parent1 = new_pop[i]
            parent2 = new_pop[i+1]
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            
            if np.random.rand() < p_mut:
                child1 = mutate_swap(child1)
            if np.random.rand() < p_mut:
                child2 = mutate_swap(child2)
            
            offspring.append(repair_weight(child1, target_sum))
            offspring.append(repair_weight(child2, target_sum))
        
        # Keep elite
        elite_indices = np.argsort(fitness)[:elite]
        for i in elite_indices:
            offspring.append(population[i].copy())
        
        # Update population
        population = offspring
        fitness = [-c5_fft(h) for h in population]
        
        # Update best
        current_best_idx = np.argmax(fitness)
        current_c5 = -fitness[current_best_idx]

            best_c5 = current_c5
            best_h = population[current_best_idx].copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Local hillclimb every 10 generations
        if gen % 10 == 0:
            best_h = hill_climb(best_h, dx)
            best_c5 = c5_fft(best_h)
            
        gen += 1
    
    if best_h is None:
        # fallback if not found
        idx = np.argmax(fitness)
        best_h = population[idx].copy()
        best_c5 = c5_fft(best_h)
    
    # Up-sample and refine
    if time.time() - start_time < budget_s - 5:
        n_fine = 500
        dx_fine = 2.0 / n_fine
        h_fine = np.zeros(n_fine)
        scale = n_fine // n
        for i in range(n):
            h_fine[i * scale:(i + 1) * scale] = best_h[i]
        h_fine = h_fine[:n_fine]
        h_fine = repair_weight(h_fine, n_fine/2)
        h_fine = hill_climb(h_fine, dx_fine, max_swaps=2000)
        best_c5 = c5_fft(h_fine)
        best_h = h_fine
    
    # Final verification with high precision
    if time.time() - start_time < budget_s - 1:
        c5_precise = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
        best_c5 = c5_precise
    
    return (best_h, best_c5, len(best_h))
