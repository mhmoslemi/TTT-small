```python
import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
import math

def evaluate_erdos_solution(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Use N=140 as suggested in strategy
    N = 140
    dx = 2.0 / N
    S = N // 2  # Sum should be N/2
    max_iterations = 20000  # max attempts for swap search
    
    # Precompute index sets for k-shifts where i+k < N
    shift_indices = {}
    for k in range(N):
        shift_indices[k] = [i for i in range(N - k)]
    
    # Step 1: Binary search feasibility MILP solver
    L = 0.0
    U = S
    t_best = float('inf')
    best_x = None
    
    # Set a reasonable upper bound for MILP time limit
    solver_time_limit = 30
    
    while U - L > 1e-4 and (time.time() - start_time) < budget_s - 10:
        T = (L + U) / 2.0
        # Build model
        x = cp.Variable(N, boolean=True)
        w = {}
        constraints = [cp.sum(x) == S, x[0] == 1]  # Symmetry breaking x[0] = 1
        
        # Add AND definition constraints for all w_{i,k} 
        for k in range(N):
            indices = shift_indices[k]
            for i in indices:
                w_i_k = cp.Variable(boolean=True)
                w[(i, k)] = w_i_k
                # McCormick constraints
                constraints.append(w_i_k <= x[i])
                constraints.append(w_i_k <= x[i+k])
                constraints.append(w_i_k >= x[i] + x[i+k] - 1) 
        
        # Add overlap bound constraints
        for k in range(N):
            indices = shift_indices[k]
            sum_w = cp.sum([w[(i,k)] for i in indices])
            # The constraint is sum_w >= S - t where t <= T
            constraints.append(sum_w >= S - T)
        
        # Try to solve the feasibility problem with this T
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.CBC, warm_start=True, 
                       **{'max_seconds': solver_time_limit})
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Check if this is feasible and better
                x_val = np.array([int(v.value) for v in x])
                # Compute actual C5 for this x
                h = x_val.astype(float)
                c5_val = evaluate_erdos_solution(h)
                # We actually want to find the minimal T s.t. feasibility holds
                # This is a minimization of T, so we update if we found feasible
                if c5_val * dx < t_best:
                    t_best = c5_val * dx
                    best_x = x_val.copy()
                U = T
            else:
                L = T
        except:
            L = T
            
    # Check if we found anything feasible
    if best_x is None:
        # Fall back to a simple solution
        h = np.zeros(N)
        h[:S] = 1.0
        best_x = h.astype(int)
        t_best = evaluate_erdos_solution(h) * dx
    
    # Convert best binary solution to continuous h
    h = best_x.astype(float)
    
    # Step 2: Fractional polishing
    if (time.time() - start_time) < budget_s - 50:
        best_c5 = t_best  # This holds the best bound from first method
        num_iters = 0
        max_iters = 50
        schedule = [0.05, 0.02, 0.01, 0.005]
        for delta in schedule:
            improved = False
            for _ in range(max_iters):
                if (time.time() - start_time) > budget_s - 50:
                    break
                # Find worst shift
                corr = np.correlate(h, 1-h, mode='full')
                k_worst = np.argmax(corr)
                indices = []
                for i in range(N):
                    if i + k_worst < N and h[i] == 1 and h[i + k_worst] == 0:
                        indices.append(i)
                if not indices:
                    break
                # Find best incremental move
                best_move = None
                best_delta_c5 = 0
                for i in indices:
                    i_plus_k = i + k_worst
                    # Get the full correlation array at offset 
                    # (this is more complex; we just use a slight tolerance heuristic )
                    h_test = h.copy()
                    h_test[i] -= delta
                    h_test[i_plus_k] += delta
                    # Keep within bounds
                    if h_test[i] < 0 or h_test[i_plus_k] > 1:
                        continue
                    h_test = np.clip(h_test, 0, 1)
                    c5_test = evaluate_erdos_solution(h_test)
                    delta_c5 = (c5_test - best_c5)
                    if delta_c5 < best_delta_c5:
                        best_delta_c5 = delta_c5
                        best_move = (i, i_plus_k, delta)
                if best_move:
                    i, i_plus_k, delta = best_move
                    h[i] -= delta
                    h[i_plus_k] += delta
                    h = np.clip(h, 0, 1)
                    best_c5 += best_delta_c5
                    improved = True
                num_iters += 1
                if not improved:
                    break
            if not improved:
                break
    
    # Step 3: Swap hill-climbing
    if (time.time() - start_time) < budget_s - 50:
        min_c5 = evaluate_erdos_solution(h)
        attempts = 0
        num_swap_attempts = 30000
        while attempts < num_swap_attempts and (time.time() - start_time) < budget_s - 10:
            attempts += 1
            # Find current maximal overlap for swap evaluation
            corr = np.correlate(h, 1-h, mode='full')
            max_idx = np.argmax(corr)
            best_idx = max_idx - (N-1)  # shift index k
            
            # Try a swap between two random indices
            #     Choose 1-index i and 0-index j such that i+j < N
            i = np.random.choice(np.where(h == 1)[0])
            # Find 0-positions which could benefit from this
            valid_indices = np.where(h == 0)[0]
            if len(valid_indices) == 0:
                continue
            j = np.random.choice(valid_indices)
            
            # Check if swapping i and j improves C5
            h_test = h.copy()
            h_test[i] = 0
            h_test[j] = 1
            c5_test = evaluate_erdos_solution(h_test)
            if c5_test < min_c5:
                h = h_test
                min_c5 = c5_test
        # Get final updated value
        final_c5 = min_c5
    
    final_c5 = evaluate_erdos_solution(h)
    
    # Optionally upsample to a finer grid
    # But since we're already using 140 points, and target is about 0.3807, 
    # better to stick with current grid and refine more carefully
    return (h, final_c5, N)
```
