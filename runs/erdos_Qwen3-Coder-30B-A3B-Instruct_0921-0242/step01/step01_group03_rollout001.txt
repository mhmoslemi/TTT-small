import numpy as np
import scipy.optimize as opt
import time
from scipy.interpolate import interp1d
import cvxpy as cp

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def evaluate_binary_mask(b):
    """Evaluate C5 for a binary mask (faster evaluation)"""
    n = len(b)
    dx = 2.0 / n
    h = b.astype(float)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def project_to_feasible(h):
    """Ensure sum(h) = n/2 and h in [0,1]^n"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    
    if abs(deficit) < 1e-10:
        return h
    
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    adjustment = deficit / n
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def binary_local_search(seed_mask, iterations=1000):
    """Use simulated annealing to improve binary mask"""
    n = len(seed_mask)
    mask = seed_mask.copy()
    dx = 2.0 / n
    current_c5 = evaluate_binary_mask(mask)
    best_c5 = current_c5
    best_mask = mask.copy()
    
    # Temperature schedule
    temp = 0.02
    cooling_rate = 0.995
    accepted = 0
    start_time = time.time()
    
    for i in range(iterations):
        if time.time() - start_time > 580:
            break
        # Pick two random positions: one 1, one 0
        ones = np.where(mask == 1)[0]
        zeros = np.where(mask == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        idx1 = np.random.choice(ones)
        idx2 = np.random.choice(zeros)
        
        # Swap and compute new C5
        new_mask = mask.copy()
        new_mask[idx1] = 0
        new_mask[idx2] = 1
        
        new_c5 = evaluate_binary_mask(new_mask)
        delta = new_c5 - current_c5
        
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            mask = new_mask
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_mask = mask.copy()
            accepted += 1
                    
        if accepted % 100 == 0:
            temp *= cooling_rate
            
    return best_mask, best_c5

def build_lp_refinement(h, binary_mask):
    """Use binary pattern to build LP for refinement"""
    n = len(h)
    dx = 2.0 / n
    h_vars = cp.Variable(n)
    t = cp.Variable()
    
    # Constraints
    constraints = [0 <= h_vars[i] <= 1 for i in range(n)]
    constraints.append(cp.sum(h_vars) == n/2)
    
    # Linear inequality constraints for each shift
    for k in range(2*n - 1):
        # Use the binary mask to determine linearization point
        expr = 0
        for i in range(n):
            j = (i + k - n + 1) % n
            if binary_mask[i] == 1:
                expr += h_vars[i] * (1 - binary_mask[j])
            else:
                expr += h_vars[i] * (1 - 0)  # = h_vars[i]
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == 'optimal':
            refined_h = h_vars.value
            return refined_h
    except:
        pass
    return h
