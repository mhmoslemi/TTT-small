def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def get_shift_matrix(n, k):
        """Create sparse shift matrix S_k for circular shift by k."""
        S = np.zeros((n, n))
        for i in range(n):
            j = (i + k) % n
            if 0 <= j < n:
                S[i, j] = 1
        return S

    def solve_sdp(n, K_list):
        """Solve the SDP relaxation for given n and shift indices K_list."""
        m = (n + 1) // 2   # Reduced dimension due to symmetry
        X = Semidef(m)
        t = Variable()
        
        # Build shift matrices that are redundant for symmetry
        # Effective for first m elements
        constraints = [X[i, i] == 1 for i in range(m)]
        constraints.append(sum(X[i, i] for i in range(m)) == 0)  # Zero-sum constraint
        constraints.append(t >= n / 4 - 0.25 * sum(X[i, j] for i in range(m) for j in range(m)))

        for k in K_list:
            S = np.zeros((m, m))
            for i in range(m):
                j = (i + k) % n
                if j < m:
                    S[i, j] = 1
            constraints.append(t >= n / 4 - 0.25 * trace(S @ X))

        objective = Minimize(t)
        problem = Problem(objective, constraints)
        try:
            problem.solve(solver=SCS, eps=1e-6, max_iters=5000)
            return problem.value, X.value
        except:
            return None, None

    def round_sdp(X_sol, n):
        """Round the SDP solution using Gaussian projection."""
        # Compute a Cholesky-like factorization of X_sol
        if X_sol is None:
            return None

        v, U = eigh(X_sol)
        v = np.maximum(v, 0)  # ensure non-negative
        sqrt_v = np.sqrt(v)
        V = U @ np.diag(sqrt_v)

        m = X_sol.shape[0]
        best_h = None
        best_c5 = float('inf')
        
        # Sample a few vectors
        for _ in range(200):
            # Sample Gaussian vector
            g = np.random.randn(m)
            y = V.T @ g
            x = np.sign(y)
            
            # Correct the sum to zero
            s = np.sum(x)
            if s != 0:
                # Determine how many flips we need
                num_to_flip = abs(s) // 2
                if num_to_flip > 0:
                    # Create a list of indices sorted by absolute y values
                    idxs = np.argsort(np.abs(y))
                    # Flip the ones with smallest |y|
                    flip_i = idxs[:num_to_flip]
                    x[flip_i] = -x[flip_i]
            
            # Construct the full h vector (mirror)
            h = np.zeros(n)
            h[:m] = (x + 1) / 2
            h[m:] = h[:m][::-1]  # Symmetric
            
            c5_val = c5_of_h(h, 2.0 / n)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
        
        return best_h

        h = h_orig.copy()

        cmax = np.max(c)
        n = len(h)
        iteration = 0
        max_iterations = 5000
        
        while iteration < max_iterations:
            if time.time() - start_time > budget_s - 10:
                break
                
            # Pick two random indices to swap
            i1 = np.random.choice(np.where(h == 1)[0])
            i0 = np.random.choice(np.where(h == 0)[0])

                    delta += h[i0] * (1 - h[(k+i1)%n]) - h[i1] * (1 - h[(k+i0)%n])
                else:
                    # Non-trivial case
                    pass
                # Simplified computation
                    
            # Alternatively, recompute contributions for the relevant parts of correlation:
            # Reconstruct correlation changes efficiently
            # Indices in rows/columns affected by swapping i0 and i1
            
            # Optimized: just update the portions
            # For shift k: update c[i1+k] and c[i0+k] etc
            new_h = h.copy()
            new_h[i1] = 0
            new_h[i0] = 1
            new_c = correlate(new_h, 1 - new_h, mode='full')
            new_cmax = np.max(new_c)
            
            if new_cmax < cmax:
                # Accept the improvement
                h = new_h
                c = new_c
                cmax = new_cmax
                iteration += 1
            else:
                iteration += 1
                
        return h, cmax

    # Grid sizes: coarse, medium, fine
    grid_configs = [
        {"n": 140, "m": 70},
        {"n": 200, "m": 100},
        {"n": 260, "m": 130}
    ]
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    for grid in grid_configs:
        if time.time() - start_time > budget_s - 30:
            break
            
        n = grid["n"]
        m = grid["m"]
        K_list = list(range(-int(0.9 * n), int(0.9 * n) + 1))
        
        print(f"Solving SDP for n={n}")
        
        t_min, X_sol = solve_sdp(n, K_list)
        
        if t_min is None:
            continue
            
        print(f"SDP min: {t_min}")
        
        # Round
        h_rounded = round_sdp(X_sol, n)
        if h_rounded is None:
            continue

        dx = 2.0 / n
        c5_rounded = c5_of_h(h_rounded, dx)
        print(f"Rounded C5: {c5_rounded}")
        
        if c5_rounded < best_c5:
            best_h = h_rounded.copy()
            best_c5 = c5_rounded
            best_n = n
            
        # Local refinement
        print(f"Refining with local hill climbing...")
        refined_h, refined_c5 = local_refinement(h_rounded, dx)

        if refined_c5 < best_c5:

        best_h = h0
        best_n = len(h0)
    
    # Final verification
    dx = 2.0 / best_n
    # Ensure it's within acceptable bound
