import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import math

def c5_fft(h):
    """Compute C5 using FFT for speed."""
    g = 1 - h
    H = rfft(h)
    G = rfft(g)
    c = irfft(H * G, 2 * len(h) - 1)
    return np.max(c)

def repair_weight(h, target_sum):
    """Repair h to have exact sum."""
    current_sum = np.sum(h)
    diff = int(target_sum - current_sum)
    if diff == 0:
        return h
    if diff > 0:
        # Need to add diff ones. Flip zeros to ones.
        zeros = np.where(h == 0)[0]
        flip_indices = np.random.choice(zeros, size=diff, replace=False)
        h[flip_indices] = 1
    else:
        # Need to remove -diff ones. Flip ones to zeros.
        ones = np.where(h == 1)[0]
        flip_indices = np.random.choice(ones, size=-diff, replace=False)
        h[flip_indices] = 0
    return h

def crossover(p1, p2):
    """Two-point crossover."""
    n = len(p1)
    c1 = np.random.randint(0, n)
    c2 = np.random.randint(c1, n)
    child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
    return repair_weight(child, n / 2)

def mutate_swap(h):
    """Mutation: swap a one and a zero."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h
    i1 = np.random.choice(ones)
    i0 = np.random.choice(zeros)
    h[i1], h[i0] = h[i0], h[i1]
    return h

def hill_climb(h, max_swaps=500):
    """Local hill climb focusing on worst shift."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for _ in range(max_swaps):
        corr_full = correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr_full) - n + 1
        max_corr = corr_full[k_star + n - 1]
        best_delta = 0
        best_i1 = -1
        best_i0 = -1
        # Check a set of random swaps
        for _ in range(200):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            delta = 0
            # Direct computation of change in correlation at k_star
            # Only terms involving i1 and i0 change
            if (i1 + k_star) >= 0 and (i1 + k_star) < n:
                delta += -dx * h[i1 + k_star]
            if (i0 + k_star) >= 0 and (i0 + k_star) < n:
                delta += dx * h[i0 + k_star]
            if (i1 + k_star) >= 0 and (i1 + k_star) < n:
                delta += dx * (1 - h[i1 + k_star])
            if (i0 + k_star) >= 0 and (i0 + k_star) < n:
                delta += -dx * (1 - h[i0 + k_star])
            if delta < best_delta:
                best_delta = delta
                best_i1 = i1
                best_i0 = i0
        if best_delta < 0:
            h[best_i1], h[best_i0] = h[best_i0], h[best_i1]
        else:
            break
    return h

        h[n//2:n//2 + n//4] = 1

        h = repair_weight(h, target_sum)
        P.append(h)

    for _ in range(10):
        h = np.tile([1, 0], n // 2)[:n]

        P.append(h)
    
    # 4. Random balanced
    for _ in range(pop_size - 21):
        pos = np.random.choice(n, int(target_sum), replace=False)
        h = np.zeros(n)
        h[pos] = 1
        P.append(h)
    
    # Evaluate initial population
    fitness = []
    for i, h in enumerate(P):
        c5 = c5_fft(h)
        fitness.append(-c5)
    fitness = np.array(fitness)
    
    best_h = P[np.argmax(fitness)].copy()
    best_c5 = -np.max(fitness)
    
    # Evolution loop
    generation = 0
    while time.time() - start_time < budget_s - 20:
        generation += 1
        # Selection: binary tournament
        mating_pool = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            if fitness[idx1] > fitness[idx2]:
                mating_pool.append(P[idx1].copy())
            else:
                mating_pool.append(P[idx2].copy())
        
        # Crossover and mutation
        offspring = []
        for i in range(0, pop_size - elite, 2):
            p1, p2 = mating_pool[i], mating_pool[i+1]
            child1 = crossover(p1, p2)
            child2 = crossover(p2, p1)
            
            if np.random.rand() < p_mut:
                child1 = mutate_swap(child1)
            if np.random.rand() < p_mut:
                child2 = mutate_swap(child2)
            
            child1 = repair_weight(child1, target_sum)
            child2 = repair_weight(child2, target_sum)
            
            offspring.append(child1)
            offspring.append(child2)
        
        # Elitism
        elite_indices = np.argsort(fitness)[-elite:]
        new_P = [P[i].copy() for i in elite_indices]
        new_P.extend(offspring)
        
        # Evaluate new population
        new_fitness = []
        for i, h in enumerate(new_P):
            c5 = c5_fft(h)
            new_fitness.append(-c5)
        new_fitness = np.array(new_fitness)

        if generation % 10 == 0:

            dists = []
            for i in range(len(recent)):

                    dists.append(np.sum(recent[i] != recent[j]) / n)

                # Inject random individuals
                for i in range(len(recent) // 2):
                    pos = np.random.choice(n, int(target_sum), replace=False)
                    h = np.zeros(n)
                    h[pos] = 1
                    new_P[-(i+1)] = h

        P = new_P

            best_h = P[current_best_idx].copy()
        
        # Early stopping
        if best_c5 <= 0.3807:

    # Local hill climb on best individual
    local_h = best_h.copy()

    upsampled = repair_weight(upsampled, n_fine / 2)
    
    # Refinement GA on fine grid
    pop_size_fine = 50
    P_fine = [upsampled.copy()]
    for _ in range(pop_size_fine - 1):
        pos = np.random.choice(n_fine, int(n_fine / 2), replace=False)
        h = np.zeros(n_fine)
        h[pos] = 1
        P_fine.append(h)
    
    # Evaluate fine population
    fitness_fine = []
    for i, h in enumerate(P_fine):
        c5 = c5_fft(h)
        fitness_fine.append(-c5)
    fitness_fine = np.array(fitness_fine)
    
    # Fine GA loop
    for gen in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        mating_pool_fine = []
        for _ in range(pop_size_fine):
            idx1, idx2 = np.random.choice(pop_size_fine, 2, replace=False)
            if fitness_fine[idx1] > fitness_fine[idx2]:
                mating_pool_fine.append(P_fine[idx1].copy())
            else:
                mating_pool_fine.append(P_fine[idx2].copy())
        
        offspring_fine = []
        for i in range(0, pop_size_fine - elite, 2):
            p1, p2 = mating_pool_fine[i], mating_pool_fine[i+1]
            child1 = crossover(p1, p2)
            child2 = crossover(p2, p1)
            if np.random.rand() < p_mut:
                child1 = mutate_swap(child1)
            if np.random.rand() < p_mut:
                child2 = mutate_swap(child2)
            child1 = repair_weight(child1, n_fine / 2)
            child2 = repair_weight(child2, n_fine / 2)
            offspring_fine.append(child1)
            offspring_fine.append(child2)
        
        elite_indices = np.argsort(fitness_fine)[-elite:]
        new_P_fine = [P_fine[i].copy() for i in elite_indices]
        new_P_fine.extend(offspring_fine)
        
        # Evaluate
        new_fitness_fine = []
        for i, h in enumerate(new_P_fine):
            c5 = c5_fft(h)

        new_fitness_fine = np.array(new_fitness_fine)

        fitness_fine = new_fitness_fine

        if -fitness_fine[best_idx_fine] < best_c5:

    final_h = hill_climb(best_h, max_swaps=2000)
    
    # Use final high-res value of C5 (i.e., use correlate)
    corr_final = correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (final_h, final_c5, len(final_h))
