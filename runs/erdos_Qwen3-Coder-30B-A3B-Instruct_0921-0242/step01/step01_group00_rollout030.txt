import numpy as np

def proj_simplex_box(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2} using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    # Adjust to target sum
    diff = n/2 - current_sum
    # Water-filling approach: add diff/n to all entries (subject to bounds)
    h = h + diff / n
    # Re-clip to [0,1]
    h = np.clip(h, 0, 1)
    # Re-adjust to exact sum
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    if abs(diff) > 1e-10:
        # Adjust by adding compensation to lower entries
        # More precisely, we'll do one step of iterative projection
        h = h + diff / n
        h = np.clip(h, 0, 1)
    return h

def c5(h):
    """Computes C5 = max_k int h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def worst_shift(h):
    """Returns the shift index that maximizes overlap."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    return max_idx - (len(h) - 1)

def gradient_step(h, k, alpha):
    """Perform a gradient step w.r.t. shift k."""
    n = len(h)
    dx = 2.0 / n
    h_new = h.copy()
    # Compute gradient: d/dh_i of I_k(h) = dx * sum h_i (1 - h_{i+k})
    # = dx * (1 - 2 * h_{i+k} )
    grad = np.zeros(n)
    for i in range(n):
        left_idx = (i + k) % n
        grad[i] = dx * (1 - 2 * h[left_idx])
        
    h_new = h_new - alpha * grad
    
    # Project back
    h_new = proj_simplex_box(h_new, n)
    return h_new

def solve_lp_relaxation(n):
    """Solve the LP relaxation of the problem to get a structured h."""

    h = cp.Variable(n)
    # Objective: maximize t (minimize -t)
    # With constraints: sum h_i = n/2, 0 <= h_i <= 1
    # And for each shift k != 0: sum h_i h_{i+k} >= t

    # We'll approximate with a small subset of constraints for speed
    # Only include shifts k = 1, 2, ..., n//4 and n//4*3, ... n//2
    # Skip redundant / symmetric.
    # For n=400, we'll use up to 200 shifts (more than enough).

    constraints = []
    constraints.append(cp.sum(h) == n/2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Now add the bilinear constraint (via McCormick relaxation) 
    # For each k, we want sum_i h_i h_{i+k} >= t.
    # Use variables z_{i,k} where z_i,k approximates h_i * h_{i+k}
    # Enforce McCormick envelopes:
    # z >= h_i + h_{i+k} - 1
    # z >= 0
    # z <= h_i
    # z <= h_{i+k}

    t = cp.Variable(1)
    
    # Use subset of interesting shifts
    shift_subset = list(range(1, n//2)) # Up to half the sample size (fast)

    # Generate McCormick envelopes for each shift.
    z_vars = []
    z_matrix = {}
    for k in shift_subset:
        z_k = cp.Variable((n, 1), name=f'z_{k}')
        z_vars.append(z_k)
        z_matrix[k] = z_k
        
        # Implement McCormick:
        # z_i >= 0
        # z_i >= h_i + h_{i+k} - 1
        # z_i <= h_i
        # z_i <= h_{i+k}
        for i in range(n):
            left_idx = (i + k) % n
            constraints.append(z_k[i] >= 0)
            constraints.append(z_k[i] >= h[i] + h[left_idx] - 1)
            constraints.append(z_k[i] <= h[i])
            constraints.append(z_k[i] <= h[left_idx]) 

    # Add the constraint: sum of z_{i,k} >= t for all k (in subset)
    for k in shift_subset:
        constraints.append(cp.sum(z_matrix[k]) >= t)
        
    # Solve
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
        return None
    except:
        return None
