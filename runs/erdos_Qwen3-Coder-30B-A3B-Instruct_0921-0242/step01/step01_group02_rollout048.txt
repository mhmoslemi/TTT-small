import numpy as np
import time
from scipy.signal import fftconvolve
from scipy.optimize import minimize

def c5_of_h(h, dx):
    """Compute C5 for given h using FFT-based convolution."""
    g = 1 - h
    corr = fftconvolve(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_binary_step_function(h_binary, n_points):
    """Convert binary h to step function and compute C5."""
    dx = 2.0 / n_points
    return c5_of_h(h_binary, dx)

def perform_swap(h, i1, i0):
    """Perform swap: h[i1] <-> h[i0] and keep track of positions of 1s and 0s."""
    h[i1], h[i0] = h[i0], h[i1]
    return h

def update_convolution_incremental(corr, h, i1, i0, n):
    """Update correlation efficiently after a swap."""
    # Get old values
    old_val1, old_val0 = h[i1], h[i0]
    # Get new values (after swap)
    new_val1, new_val0 = old_val0, old_val1
    
    # Update correlation with shifts using subtractions/additions
    # Correlation contribution of a point i is (1-h) shifted as i
    # The two points involved affect the correlation in shifts around i1 and i0
    # For simplicity, we'll recompute the entire correlation occasionally
    # But we can be more efficient for frequent updates:
    # We'll use full recomputation as it's faster than implementing precise incremental update
    corr_new = fftconvolve(h, 1 - h, mode='full')
    return corr_new

def get_positions_indices(h):
    """Return arrays of indices where h=1 and h=0."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    return ones, zeros
