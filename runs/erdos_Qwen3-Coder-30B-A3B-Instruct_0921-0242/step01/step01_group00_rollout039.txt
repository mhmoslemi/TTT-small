import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp
from cvxpy import SCS

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def waterfilling_projection(h, target_sum, lb=0.0, ub=1.0):
    """Project h to sum to target_sum with box constraints [lb, ub]."""
    h = np.clip(h, lb, ub)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Scale up or down
    h = h + diff / len(h)
    h = np.clip(h, lb, ub)
    return h

def round_sdp_solution(X, n):
    """Round an SDP solution to binary vector"""
    # Extract the dominant factorization
    try:
        eigvals, eigvecs = np.linalg.eigh(X)
        # Keep only positive eigenvalues
        mask = eigvals > 1e-8
        V = eigvecs[:, mask] * np.sqrt(eigvals[mask])
        if V.shape[1] == 0:
            raise ValueError("No positive eigenvectors found")
    except:
        # Fallback to an SVD-like approach
        V = np.random.randn(n, n // 8)
        V = np.linalg.qr(V)[0]
    
    min_c5 = float('inf')
    best_binary = None
    n_rounds = 30
    for _ in range(n_rounds):
        g = np.random.randn(V.shape[1])
        y = V @ g
        sorted_y = np.sort(y)
        # Find threshold
        target = n // 2
        # Binary search for threshold to get exactly n/2 ones
        lo, hi = sorted_y[0], sorted_y[-1]
        try:
            if abs(sorted_y[0] - sorted_y[-1]) < 1e-10:
                threshold = sorted_y[0]
            else:
                # Simple binary search with max iterations
                for _ in range(50):
                    mid = (lo + hi) / 2
                    count = np.sum(y > mid)
                    if count < target:
                        hi = mid
                    elif count > target:
                        lo = mid
                    else:
                        threshold = mid
                        break
                else:
                    threshold = (lo + hi) / 2
        except:
            # Fallback: simple median
            threshold = np.median(y)
            
        b = (y > threshold).astype(float)
        # Re-scale to sum = n/2 exactly
        b = waterfilling_projection(b, n / 2, 0, 1)
        c5 = evaluate_c5(b)
        if c5 < min_c5:
            min_c5 = c5
            best_binary = b
    return best_binary, min_c5

def smooth_max_gradient(h, beta):
    """Compute gradient of smooth max of overlaps with parameter beta"""
    n = len(h)
    dx = 2.0 / n
    # compute all I_k for k=1,...,n-1
    I_vals = []
    for k in range(1, n):
        corr_k = np.sum(h[:-k] * h[k:]) if k < n else 0
        I_k = dx * (1 - corr_k)
        I_vals.append(I_k)
    
    I_vals = np.array(I_vals)
    exp_vals = np.exp(beta * I_vals)
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute gradient of smooth max
    grad = np.zeros(n)
    for k in range(1, n):
        corr_grad = np.zeros(n)
        if k < n:
            corr_grad[:-k] += h[k:]
            corr_grad[k:] += h[:-k]
        # Gradient of I_k is dx*(1-h_{i+k} - h_{i-k})
        # for i+shift and i-shift, set boundary to zero (as in corr_grad)
        
        # Compute more efficiently
        g_k = np.zeros(n)
        g_k[:-k] += 1 - h[k:]  # i-k doesn't matter for BC, so assume 0
        # But we are computing for delta I_k over h_i
        g_k[k:] -= h[:-k]   # contribution from h_{i-k}
        g_k = dx * g_k
        
        grad += weights[k-1] * g_k

    # Project to feasible set
    return grad

def proj(h, target_sum):
    """Project h onto the feasible set. Keep in [0,1] and sum = target_sum"""
    h = np.clip(h, 0.0, 1.0)
    h = waterfilling_projection(h, target_sum, 0, 1)
    return h

def project_gradient_step(h, grad, beta, step_size=0.01, max_backtrack=10):
    """Do gradient descent step with backtracking line search"""
    h0 = h.copy()
    h_new = h - step_size * grad
    h_new = proj(h_new, len(h) / 2)
    # backtracking line search to stay in feasible region and decrease objective
    # However, as we are computing the max directly use a simple one
    current_c5 = evaluate_c5(h)
    new_c5 = evaluate_c5(h_new)
    
    if new_c5 < current_c5:
        return h_new
    else:
        # do a backtracking line search
        for _ in range(max_backtrack):
            step_size *= 0.5
            h_new = h - step_size * grad
            h_new = proj(h_new, len(h) / 2)
            new_c5 = evaluate_c5(h_new)
            if new_c5 < current_c5:
                return h_new
        return h
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: SDP relaxation, run for max ~45 seconds
    n = 128
    dx = 2.0 / n
    t_max = 0
    h_best = None
    c5_best = float('inf')
    
    if time.time() - start_time > budget_s - 40:
        # fallback
        h_final = np.ones(n) * 0.5
        h_final = waterfilling_projection(h_final, n/2)
        return (h_final, evaluate_c5(h_final), n)

    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.trace(X) == n / 2,  # Trace = sum of diagonal = n/2
        h >= 0,
        h <= 1,
        X >> 0,  # X is positive semidefinite
    ]
    # Link X and h
    constraints += [X[i, i] == h[i] for i in range(n)]
    
    # Add each shift constraint
    for k in range(1, n):
        T_k = np.zeros((n, n))
        for i in range(n - k):
            T_k[i, i + k] = 1
        constraints.append(cp.trace(T_k @ X) >= t)
    
    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=SCS, eps=1e-4, max_iters=5000)
        h_sdp = h.value
        t_max = prob.value
        if h_sdp is None:
            h_sdp = np.ones(n) * 0.5
        h_sdp = waterfilling_projection(h_sdp, n/2)
        c5_sdp = dx * (1 - t_max)
        if c5_sdp < c5_best:
            c5_best = c5_sdp
            h_best = h_sdp.copy()
    except Exception as e:
        h_sdp = np.ones(n) * 0.5
        h_sdp = waterfilling_projection(h_sdp, n/2)
        c5_sdp = evaluate_c5(h_sdp)
        if c5_sdp < c5_best:
            c5_best = c5_sdp
            h_best = h_sdp.copy()
    
    # Stage 2: Rounding
    if time.time() - start_time > budget_s - 30:
        # fallback
        return (h_best, c5_best, n)

    # Try to compute the rounded solution
    try:
        # Try to compute coarse version of X from h_sdp
        if h_sdp.ndim == 1:
            X_sdp = np.outer(h_sdp, h_sdp)
        else:
            X_sdp = np.outer(h_sdp.flatten(), h_sdp.flatten())
            
        h_rounded, c5_rounded = round_sdp_solution(X_sdp, n)
        if c5_rounded < c5_best:
            c5_best = c5_rounded
            h_best = h_rounded.copy()
    except Exception as e:
        pass

    # Stage 3: Smooth max gradient descent
    if time.time() - start_time > budget_s - 30:
        return (h_best, c5_best, n)

    h_refined = h_best.copy()
    beta = 0.5
    max_beta = 50
    inner_steps = 50
    for i in range(50):  # max iterations
        if time.time() - start_time > budget_s - 10:
            break
        grad = smooth_max_gradient(h_refined, beta)
        h_refined = project_gradient_step(h_refined, grad, beta)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < c5_best:
            c5_best = c5_refined
            h_best = h_refined.copy()
        if beta < max_beta:
            beta = 1.2 * beta

    # Stage 4: Discrete local search
    if time.time() - start_time > budget_s - 10:
        return (h_best, c5_best, n)

    def update_I(h, k):
        corr = np.sum(h[:-k] * h[k:]) if k < n else 0
        return dx * (1 - corr)
    
    def compute_all_I(h):
        return [update_I(h, k) for k in range(1, n)]
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
        i, j = np.random.choice(n, size=2, replace=False)
        if h_best[i] <= 0 or h_best[j] >= 1:
            continue
        # Swap a small amount
        delta = min(h_best[i], 1 - h_best[j], 0.02)
        if delta <= 1e-6:
            continue
        h_try = h_best.copy()
        h_try[i] -= delta
        h_try[j] += delta
        h_try = waterfilling_projection(h_try, n/2)
        c5_try = evaluate_c5(h_try)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_try.copy()

    return (h_best, c5_best, n)
