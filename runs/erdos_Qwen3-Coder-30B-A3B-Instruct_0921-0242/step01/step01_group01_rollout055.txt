import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def c5_of_h(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution
    conv = np.correlate(h, g, mode='full')
    # The full convolution has 2*len(h) - 1 elements
    result = np.max(conv) * dx
    return result

def decode_blocks_to_h(block_params, n_points):
    """Decode block parameters to a full h vector"""
    # block_params: [start1, len1, height1, start2, len2, height2, ...]
    M = len(block_params) // 3
    starts = block_params[0::3]
    lengths = block_params[1::3]
    heights = block_params[2::3]

    # Sort blocks by start position (non-overlapping)
    idxs = np.argsort(starts)
    starts = starts[idxs]
    lengths = lengths[idxs]
    heights = heights[idxs]

    # Ensure no overlaps
    for i in range(1, M):
        if starts[i] < starts[i-1] + lengths[i-1]:
            starts[i] = starts[i-1] + lengths[i-1]

    h = np.zeros(n_points)
    dx = 2.0 / n_points
    total_area = 0

    # Fill blocks
    for i in range(M):
        start = starts[i]
        length = lengths[i]
        height = heights[i]
        # Determine indices
        start_idx = int(np.floor(start / dx))
        end_idx = int(np.ceil((start + length) / dx))
        end_idx = min(end_idx, n_points)
        if end_idx > start_idx:
            h[start_idx:end_idx] = height
            total_area += height * (end_idx - start_idx) * dx

    # Normalize to sum = n/2
    if np.isclose(total_area, 0.0):
        h.fill(0.5)
    else:
        h = h * (n_points / 2) / total_area

    return h

def soft_max_conv(h, beta, dx):
    """Compute soft max of the convolution, for a given beta"""
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    # Convert to log-space for numerical stability
    conv_scaled = beta * dx * conv
    # Subtract max for numerical stability
    max_conv = np.max(conv_scaled)
    if max_conv > 1000:
        logsumexp = max_conv + np.log(np.sum(np.exp(conv_scaled - max_conv)))
    else:
        logsumexp = max_conv + np.log(np.sum(np.exp(conv_scaled - max_conv)))
    return logsumexp / beta

def gradient_soft_max_conv(h, beta, dx):
    """Compute gradient of soft max of convolution"""
    n = len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')

    # Weights for soft max
    conv_scaled = beta * dx * conv
    max_conv = np.max(conv_scaled)
    if max_conv > 1000:
        logsumexp = max_conv + np.log(np.sum(np.exp(conv_scaled - max_conv)))
    else:
        logsumexp = max_conv + np.log(np.sum(np.exp(conv_scaled - max_conv)))
    weights = np.exp(conv_scaled - logsumexp)

    # Create gradient vector
    grad = np.zeros_like(h)
    for k in range(2*n - 1):
        # k corresponds to shift k - (n-1)
        shift = k - (n - 1)
        if shift >= 0:
            # shift right: h[i+k] in the original
            # But in conv we have gathers h[i] * (1 - h[i+k])
            # grad wrt h[i] includes (1 - h[i+shift])
            if k < n:
                # left padding
                h_pad = np.pad(h, (0, shift), 'constant')
                g_pad = np.pad(g, (0, shift), 'constant')
                d_i = (1 - h_pad[shift:-shift] if shift > 0 else g_pad[:n]) * weights[k] * dx
                grad[:-shift] += d_i
            else:
                # Valid part once we have enough elements
                d_i = (1 - h[shift:]) * weights[k] * dx
                grad[:n-shift] += d_i
        else:
            # shift left: h[i+shift] in the original
            if k >= shift:
                # The valid entry of (1-h_{i-shift}) in the original
                h_pad = np.pad(h, (-shift, 0), 'constant')
                d_i = (-h_pad[0:n+shift]) * weights[k] * dx
                grad[-shift:] += d_i
            else:
                # Valid part on the right (k case between -n+1 and n-1)
                # We have h[i-shift] in convolution, so we get +h[i-shift] term
                d_i = -h[shift:] * weights[k] * dx
                grad[:-shift] += d_i
    return grad

def project_sum(h, target):
    """Project h onto the constraint sum = target by water filling"""
    total = np.sum(h)
    diff = target - total
    if abs(diff) < 1e-9:
        return h
    
    # keep h in [0,1]
    h = h.copy()
    if diff > 0:
        # Adding mass
        # Fill sorted from smallest to largest
        sorted_indices = np.argsort(h)
        for idx in sorted_indices:
            if diff <= 0:
                break
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
    else:
        # Subtracting mass
        # Remove from sorted largest to smallest
        sorted_indices = np.argsort(h)[::-1]
        for idx in sorted_indices:
            if diff >= 0:
                break
            subtract = min(-diff, h[idx])
            h[idx] -= subtract
            diff += subtract
    
    return np.clip(h, 0, 1)

def smooth_max_minimize(h_init, sum_constraint, dx, max_iter=150, beta_init=0.5):
    """Refine h using smooth max gradient descent"""
    beta = beta_init
    lam = 10.0
    h = h_init.copy()
    
    # Precompute h and g for correlation
    n = len(h)
    
    def func(h):
        # Soft max objective
        soft_max = soft_max_conv(h, beta, dx)
        penalty = lam * (np.sum(h) - sum_constraint)**2
        return soft_max + penalty
    
    def grad_func(h):
        g_soft = gradient_soft_max_conv(h, beta, dx)
        g_penalty = 2 * lam * (np.sum(h) - sum_constraint)
        return g_soft + g_penalty

    # L-BFGS-B
    result = minimize(func, h, method='L-BFGS-B', jac=grad_func, bounds=[(0, 1)] * n,
                      options={'maxiter': max_iter, 'gtol': 1e-6})
    
    h_final = result.x
    h_final = np.clip(h_final, 0, 1)
    h_final = project_sum(h_final, sum_constraint)
    
    return h_final

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Parametric block representation: M=4 blocks
    # 3*M = 12 parameters: start_i, len_i, height_i (i=1..4)
    # Bounds for each block: start ∈ [0,2], len ∈ [0,2], height ∈ [0,1]
    bounds = []
    for i in range(4):
        bounds.append((0, 2))   # start_i
        bounds.append((0, 2))   # len_i
        bounds.append((0, 1))   # height_i
    
    # Step 1: Differential evolution global search
    def de_objective(block_params):
        h = decode_blocks_to_h(block_params, n_points)
        return c5_of_h(h)
    
    # Run DE
    timeout = budget_s * 0.25
    de_result = differential_evolution(de_objective, bounds, maxiter=60, popsize=15, 
                                       seed=seed, disp=False, maxtime=timeout)
    
    if de_result.success:
        h_global = decode_blocks_to_h(de_result.x, n_points)
        h_global = (h_global + h_global[::-1]) / 2  # symmetry
        h_global = project_sum(h_global, target_sum)
        c5_global = c5_of_h(h_global)
        if c5_global < best_c5:
            best_c5 = c5_global
            best_h = h_global.copy()
    
    # Step 2: Smooth max refinement
    refine_timeout = timeout * 2  # Saved from first step
    if best_h is not None:
        # Several stages of increasing beta
        h_cur = best_h.copy()
        stages = [(0.5, 50), (1.0, 50), (2.0, 50), (5.0, 50), (10.0, 50), (50.0, 50)]
        for beta, max_iter in stages:
            if time.time() - start_time > budget_s - 0.5:
                break
            h_cur = smooth_max_minimize(h_cur, target_sum, dx, max_iter=max_iter, beta_init=beta)
            # Symmetry
            h_cur = (h_cur + h_cur[::-1]) / 2
            h_cur = project_sum(h_cur, target_sum)
            c5_cur = c5_of_h(h_cur)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
    
    # Possibly re-run the above with better initial h...
    if time.time() - start_time > budget_s - 0.5:
        pass
    
    # Step 3: Local hill climb
    if best_h is not None and time.time() - start_time < budget_s - 0.5:
        h_best = best_h.copy()
        c5_best = best_c5
        n_it = 0
        while time.time() - start_time < budget_s - 0.5:
            n_it += 1
            num_perturb = min(5, n_points // 20)
            indices = np.random.choice(n_points, size=num_perturb, replace=False)
            h_trial = h_best.copy()
            for i in indices:
                delta = np.random.uniform(-0.005, 0.005)
                h_trial[i] = min(1.0, max(0.0, h_trial[i] + delta))
            
            h_trial = project_sum(h_trial, target_sum)
            c5_trial = c5_of_h(h_trial)
            if c5_trial < c5_best:
                h_best = h_trial
                c5_best = c5_trial
        best_h = h_best
        best_c5 = c5_best
    
    # Fallback
    if best_h is None:
        h = np.ones(n_points) * 0.5
        h = project_sum(h, target_sum)
        best_h = h
        best_c5 = c5_of_h(h)
    
    return (best_h, best_c5, n_points)
