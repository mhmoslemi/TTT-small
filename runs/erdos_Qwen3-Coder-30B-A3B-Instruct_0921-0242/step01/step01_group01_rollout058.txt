import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from math import floor, ceil

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Discretization parameters
    n = 200  # Use lower resolution for faster computation
    dx = 2.0 / n
    d = n // 2 + 1  # Number of independent variables due to symmetry
    
    # Initialize best solution
    best_c5 = float('inf')
    best_h = None
    
    # Helper to compute overlap exactly
    def compute_c5_exact(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    # Initialize h from initial_h_values if available, else use zero
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
        # Interpolate to n points
        x_original = np.linspace(0, 2, len(h_initial))
        x_new = np.linspace(0, 2, n)
        f_interp = interp1d(x_original, h_initial, kind='linear', fill_value='extrapolate')
        h = f_interp(x_new)
    else:
        h = np.ones(n) * 0.5  # Fallback to uniform

    # Enforce integral constraint
    h = h * (n / 2) / np.sum(h)
    # Apply symmetry: h[i] = h[n-1-i]
    h_sym = np.zeros(n)
    for i in range(n):
        h_sym[i] = h[i] if i <= n//2 else h[n-1-i]
    h = h_sym

    # Double-oracle cutting plane method with linear relaxation
    # Variables:
    # - w_i: independent variables (0..d-1)
    # - p_{i,k}: products for shifts k
    # - M: upper bound on overlap
    
    def solve_lp_current(S, lambda_reg=0.1):
        # Build the LP
        # w: d variables (symmetric)
        w = cp.Variable(d)
        # M: scalar bound
        M = cp.Variable()
        # p: each k in S, each valid i
        p = {}
        # All shifted pairs, dicts of sparse arrays indexed by k
        # To avoid problems during refactor later, each tuple (i,k) maps to a variable
        
        # build constraints (linear system)
        constraints = []
        
        # Box constraints
        constraints += [0 <= w[i] for i in range(d)]
        constraints += [w[i] <= 1 for i in range(d)]
        
        # Sum constraint: sum(w_i) = n/4 + 0.5 = 50.5
        constraints += [cp.sum(w) == 50.5]
        
        # Define p_{i,k} alias dictionary
        for k in S:
            p[k] = {}
            for i in range(n):
                # Only valid p_{i,k} if both indices are in range
                i_k = i + k
                if 0 <= i_k < n:
                    p[k][i] = cp.Variable()
                    # McCormick envelopes
                    # p_{i,k} <= w[i] (when w is indexed by i in d-array)
                    # p_{i,k} <= w[i+k]
                    # p_{i,k} >= 0
                    # p_{i,k} >= w[i] + w[i+k] - 1
                    
                    # We're using meta indexing; after mirroring,
                    # w indexed [0,d) maps to full [0,n)
                    w_idx_i = i if i <= n//2 else n - 1 - i
                    w_idx_i_k = i_k if i_k <= n//2 else n - 1 - i_k
                    
                    constraints += [p[k][i] <= w[w_idx_i]]
                    constraints += [p[k][i] <= w[w_idx_i_k]]
                    constraints += [p[k][i] >= 0]
                    constraints += [p[k][i] >= w[w_idx_i] + w[w_idx_i_k] - 1]
        
        # Back into exact overlap frames, issue overlap constraint for each k in S
        for k in S:
            total = cp.sum([p[k][i] for i in range(n) if i + k < n])
            constraints += [0.5 - total <= M]  # Since overlap is 0.5 - sum(p_{i,k})
        
        # Objective: minimize M - lambda * sum(p_{i,0}) for k=0
        obj_expr = M - lambda_reg * cp.sum([p[0][i] for i in range(n) if i < n])
        prob = cp.Problem(cp.Minimize(obj_expr), constraints)
        return prob, w, p, M

    # Iteration loop
    S = [0]  # Start with base shift
    lambda_reg = 0.1
    max_iter = 50
    epsilon = 1e-6
    iter_count = 0
    local_start = time.time()
    
    # Iteration of cutting plane
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 10:
            break
            
        try:
            prob, w, p, M = solve_lp_current(S, lambda_reg)
            prob.solve(solver=cp.GLOP, verbose=False)  # Solves faster for small problem
            
            # Extract solution
            h_sol_raw = w.value.copy()
            # Reconstruct full h vector with symmetry
            h_sol = np.zeros(n)
            for i in range(d):
                # Simple mapping from index
                h_sol[i] = h_sol_raw[i]
                h_sol[n - 1 - i] = h_sol_raw[i]
            
            # Project back to valid h values
            h_sol = np.clip(h_sol, 0, 1)
            h_sol = h_sol * (n / 2) / np.sum(h_sol)  # Renormalize
            
            # Evaluate C5 of exact solution
            c5_exact = compute_c5_exact(h_sol)
            
            if c5_exact < best_c5:
                best_c5 = c5_exact
                best_h = h_sol.copy()
                
            # Termination: if bound is tight
            if np.abs(c5_exact - M.value) < epsilon:
                break
                
            # Find worst shift to add
            conv = np.correlate(h_sol, 1 - h_sol, mode='full')
            k_max = np.argmax(conv)
            # Since we're dealing with full conv range = 2*n - 1
            k_actual = k_max - (n - 1)
            S.append(k_actual)
            
            # Increase lambda gradually
            lambda_reg = min(5.0, lambda_reg * 1.5)
            
            iter_count += 1
        except:
            break

    # Sometimes refactor before dropping into polishing
        
    if best_h is None:
        # fallback
        best_h = np.ones(n) * 0.5
        best_h = best_h * (n / 2) / np.sum(best_h)
        best_c5 = compute_c5_exact(best_h)

    # Polish the best solution with projected subgradient descent
    # Subgradient computation
    
    def project_h(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Redistribute to make sum = n/2
        total = np.sum(h)
        deficit = n/2 - total
        if abs(deficit) > 1e-9:
            # Add deficit uniformly if positive, subtract if negative
            h += deficit / n
            
        return h

    def step_subgradient(h, k):  # k is the shift that gives max overlap
        # Compute subgradient w.r.t. h (but for full miss/hessian -> not feasible)
        # For actual C5_subgradient, define once
        # The issue is with index issues
        # Instead of analytically computing ∂[∫ h(1-h^{(k)})]/∂h_i,
        # Use numerics or appealing representation
        # Gradient:
        # It's the sum over i of 1-h_{i+k} (where h_{i+k} existed) - h_{i-k} (when valid)
        # But we're fixing the shift k, so form d_i = 1 - h_{i+k} - h_{i-k} 
        n = len(h)
        # Subgradient for i-th component of overlap integral:
        # g_i = (1 - h_{i+k}) * 1_{i+k < n} - h_{i-k} * 1_{i-k >= 0} 
        g = np.zeros(n)
        # Assume zero-padding outside bounds
        # Do not check for valid indices here as this is K*K convolution like
        # but and treat it as linear rule over an V₁ vector (row-wise)
        if k > 0:
            g[k:] = -h[:n-k]
            g[:-k] += (1 - h[k:])
        elif k < 0:
            g[:k] = -h[-k:]
            g[-k:] += (1 - h[:k])
        else:
            g = 1 - 2 * h  # k=0 case: d/dh_i [h_i(1-h_i)] = 1 - 2*h_i
        return g

    # Polish
    h_p = best_h.copy()
    c5_p = compute_c5_exact(h_p)
    
    step_sizes = [0.02 * (0.9 ** t) for t in range(200)]
    max_polish_iter = min(200, 200)
    
    for t in range(max_polish_iter):
        if time.time() - start_time > budget_s - 5:
            break
        # Identify the k which maximizes overlap
        conv = np.correlate(h_p, 1 - h_p, mode='full')
        k_max = np.argmax(conv)
        actual_k = k_max - (n - 1)
        g = step_subgradient(h_p, actual_k)
        h_new = h_p - step_sizes[t] * g
        # Project back
        h_new = project_h(h_new)
        # Force integral constraint
        h_new = h_new * (n / 2) / np.sum(h_new)
        # Force symmetry
        h_new = (h_new + h_new[::-1]) / 2
        c5_new = compute_c5_exact(h_new)
        if c5_new < c5_p:
            h_p = h_new
            c5_p = c5_new
        else:
            # Revert if no improvement
            pass
    
    # Final check
    final_c5 = compute_c5_exact(h_p)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_p.copy()

    return (best_h, best_c5, n)
