```python
import numpy as np
import scipy.fft as fft
import scipy.optimize as opt
import time
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, Constraint

def correlation_fft(h):
    """Compute all f_k(h) = sum_i h_i * (1 - h_{i+k}) using FFT."""
    N = len(h)
    h_pad = np.concatenate([h, np.zeros(N)])
    g_pad = np.concatenate([1 - h, np.zeros(N)])
    c = fft.fft(h_pad) * fft.fft(g_pad)
    return np.real(fft.ifft(c))[:N]

def weighted_gradient_fft(h, p):
    """Compute weighted sum of shift differences: sum_k p_k * [(1-h_{i+k}) - h_{i-k}] """
    N = len(h)
    # p is length N, process as probability distribution
    p_pad = np.concatenate([p, np.zeros(N)])
    h_pad = np.concatenate([h, np.zeros(N)])
    
    w1 = fft.ifft(fft.fft(p_pad) * fft.fft(1 - h_pad)).real[:N]
    w2 = fft.ifft(fft.fft(p_pad[::-1]) * fft.fft(h_pad)).real[:N]
    return w1 - w2

def project_onto_Ω(v):
    """Projection onto {h : 0 ≤ h_i ≤ 1, sum h_i = N/2}"""
    N = len(v)
    z = np.clip(v, 0, 1)
    r = N/2 - np.sum(z)
    if abs(r) < 1e-10:
        return z
    I = np.where((z > 0) & (z < 1))[0]
    if len(I) == 0:
        # all values are either 0 or 1
        return z
    while abs(r) > 1e-10:
        incr = r / len(I)
        z[I] += incr
        # might have pushed some entries beyond bounds
        z = np.clip(z, 0, 1)
        r = N/2 - np.sum(z)
        # update interior indices
        I = np.where((z > 0) & (z < 1))[0]
        if len(I) == 0:
            break
    return z

def binary_from_top(h, target_sum):
    """Get binary mask with exact target_sum ones"""
    N = len(h)
    indices = np.argsort(h)[::-1]  # descending order
    h_bin = np.zeros(N)
    h_bin[indices[:target_sum]] = 1
    return h_bin

def fractional_correction(h_bin):
    """If too many values are near 0.5, fix with linear program"""
    # count values near 0.5
    h = h_bin.copy()
    diff = np.abs(h - 0.5)
    borderline_idx = np.where(diff < 0.3)[0]
    if len(borderline_idx) <= 4:
        return h
    # Try to do nearest binary fix with a little fractional freedom
    indices = borderline_idx[np.argsort(diff[borderline_idx])[:4]]
    # Create simple problem: allow these four values to vary while fixing sum
    # Define variables for these positions, other are in {0,1}
    x = Variable(4)
    # sum constraint
    fixed_sum = np.sum(h_bin) - np.sum(h_bin[indices])
    constraints = [0 <= x, x <= 1, cp_sum(x) + fixed_sum == len(h_bin)/2]
    # minimize some proxy of smoothness or C5, just use 0 for now
    prob = Problem(Minimize(0), constraints)
    try:
        prob.solve(solver='ECOS', verbose=False)
        if prob.status == 'optimal':
            for i, idx in enumerate(indices):
                h[idx] = x.value[i]
    except:
        pass
    return h

def swap(h, i, j):
    """Return copy of h with positions i,j swapped"""
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def c5_of_h(h):
    """Compute C5 for a step function"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()

    N = 256
    h = np.random.rand(N)
    h = project_onto_Ω(h)

    τ_start = 0.05
    τ_factor = 0.5
    outer_cycles = 6
    inner_iters = 150
    τ = τ_start
    
    for outer in range(outer_cycles):
        α = 1.0
        h_prev = h.copy()
        for t in range(inner_iters):
            if time.time() - start > budget_s - 30:
                break
            y = h + ((t-1)/(t+2)) * (h - h_prev)
            f = correlation_fft(y)
            p = np.exp(f/τ - np.max(f/τ))
            p /= p.sum()
            grad = weighted_gradient_fft(y, p)
            while True:
                z = y - α * grad
                z_proj = project_onto_Ω(z)
                Fτ_val = τ * np.log(np.sum(np.exp(f/τ))) if τ > 0 else np.max(f)
                Fτ_z = τ * np.log(np.sum(np.exp(correlation_fft(z_proj)/τ))) if τ > 0 else np.max(correlation_fft(z_proj))
                if Fτ_z <= Fτ_val - 0.5*α*np.dot(grad, grad):
                    break
                α *= 0.5
            h_prev, h = h, z_proj
        if time.time() - start > budget_s - 30:
            break
        τ *= τ_factor

    h_bin = binary_from_top(h, N//2)
    h_refined = fractional_correction(h_bin)

    best_h = h_refined
    best_c5 = c5_of_h(best_h)
    
    while time.time() - start < budget_s - 80:
        i, j = np.random.choice(N, 2, replace=False)
        if i == j:
            continue
        cand = swap(best_h, i, j)
        cand_c5 = c5_of_h(cand)
        if cand_c5 < best_c5 - 1e-7:
            best_h, best_c5 = cand, cand_c5

    # optional upsample
    if time.time() - start < budget_s - 5:
        h_up = np.interp(np.linspace(0, 2, 512), np.linspace(0, 2, 256), best_h)
        h_up = project_onto_Ω(h_up)
        res = opt.minimize(
                fun=lambda x: c5_of_h(x),
                x0=h_up,
                method='SLSQP',
                bounds=[(0,1)]*512,
                constraints={'type':'eq','fun':lambda x: np.sum(x)-256},
                options={'maxiter':100,'ftol':1e-9})
        final_h = res.x
        final_c5 = c5_of_h(final_h)
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5
        n_points = len(final_h)
    else:
        n_points = N

    return (best_h, best_c5, n_points)
```
