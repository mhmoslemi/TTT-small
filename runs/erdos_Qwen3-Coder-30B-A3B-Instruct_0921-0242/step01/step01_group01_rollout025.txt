import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time
import math

def c5_of_h(h, dx):
    """Compute C5 for a given step function h using FFT for efficiency."""
    n = len(h)
    # Zero-pad to 2*n-1 for full correlation
    H = np.fft.rfft(h, n=2*n-1)
    G = np.fft.rfft(1-h, n=2*n-1)
    conv = np.fft.irfft(H*G)[:2*n-1]
    return dx * np.max(conv)

def project_to_feasible(u, n):
    """Project vector u to feasible region with sum = n/2 and 0 <= u <= 1."""
    u = np.clip(u, 0, 1)
    s = np.sum(u)
    if abs(s - n/2) < 1e-10:
        return u
    
    # Binary search for lambda
    def sum_at_lambda(lam):
        return np.sum(np.clip(u + lam, 0, 1))
    
    low, high = -1.0, 1.0
    for _ in range(30):
        mid = (low + high) / 2
        if sum_at_lambda(mid) > n/2:
            high = mid
        else:
            low = mid
    
    lam = (low + high) / 2
    result = np.clip(u + lam, 0, 1)
    return result

def symmetrize(h):
    """Make h symmetric by averaging with its reverse."""
    n = len(h)
    h_sym = (h + h[::-1]) / 2
    return h_sym

def cma_es_global_search(n_coarse, n_fine, time_budget):
    """Global search with CMA-ES."""
    start_time = time.time()
    # Parameters
    d = (n_coarse + 1) // 2  # half-size for symmetry
    mu = 0.5 * np.ones(d)
    sigma = 0.2
    lmbda = 4 + int(3 * np.log(d))  # population size
    
    # Initialize
    best_c5 = float('inf')
    best_h_coarse = None
    
    # CMA-ES parameters
    C = np.eye(d)
    mu_eff = lmbda / 4
    c_m = 1
    c_1 = 2 / (d + 2)
    c_mu = 2 / (d + 2)
    chi = np.sqrt(d) * (1 - 1/(4*d) + 1/(21*d*d))
    
    # Restart tracking
    restart_cnt = 0
    last_best = float('inf')
    gen = 0
    
    while time.time() - start_time < time_budget - 10:
        gen += 1
        # Sample offspring
        z = np.random.randn(lmbda, d)
        x = mu + sigma * z  # (lmbda, d)
        
        # Evaluate and update
        fitnesses = []
        candidates = []
        
        for i in range(lmbda):
            # Copy symmetrically
            h_trial = np.zeros(n_coarse)
            h_trial[:d] = x[i]
            h_trial[d:] = x[i][::-1]
            
            # Project to feasible set
            h_trial = project_to_feasible(h_trial, n_coarse)
            h_trial = symmetrize(h_trial)
            candidates.append(h_trial)
            
            # Evaluate
            c5 = c5_of_h(h_trial, 2.0 / n_coarse)
            fitnesses.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h_coarse = h_trial.copy()
        
        # Update
        idx_sorted = np.argsort(fitnesses)
        fitnesses = np.array(fitnesses)[idx_sorted]
        x_sorted = x[idx_sorted]
        
        # Update mean
        h_mu = np.sum(x_sorted[:lmbda//2], axis=0) / (lmbda//2)
        mu = (1 - c_m) * mu + c_m * h_mu
        
        # Update covariance(matrix) and step-size
        # Update C and sigma
        z_norm = np.zeros_like(z)
        for i in range(lmbda):
            z_norm[i] = z[i] - np.dot(C, h_mu - mu)
        C = (1 - c_1) * C + c_1 * np.outer(z_norm[0], z_norm[0])  # Simple update
        sigma = sigma * np.exp((c_m * np.linalg.norm(h_mu - mu)) / chi)
        
        # Reseting
        if gen > 150 and abs(fitnesses[0] - last_best) < 1e-4:
            if restart_cnt < 3:
                mu = best_h_coarse[:d]  # Use best as new mean
                sigma = sigma / 2
                restart_cnt += 1
                last_best = float('inf')
        else:
            last_best = fitnesses[0] 
        
        if gen > 200:
            break
    
    return best_h_coarse, best_c5

def smooth_max_gradient_descent(h_init, n_points, time_budget):
    """Optimize using smooth max gradient descent."""
    start_time = time.time()
    beta = 0.5  # smoothness
    beta_max = 50
    lambda_pen = 10  # penalty parameter
    max_iter = 80
    dx = 2.0 / n_points
    h_current = h_init.copy()
    current_c5 = c5_of_h(h_current, dx)
    
    # Make symmetric
    h_current = symmetrize(h_current)
    
    while beta <= beta_max and time.time() - start_time < time_budget - 20:
        # Compute full convolution gradient
        def compute_grad(h):
            # Use FFT for circular correlation
            n = len(h)
            H = np.fft.rfft(h, 2*n - 1)
            G = np.fft.rfft(1 - h, 2*n - 1)
            conv = np.fft.irfft(H * G)[:2*n - 1]
            w = np.exp(beta * dx * conv) / np.sum(np.exp(beta * dx * conv))
            
            # Use Toeplitz structure
            h_padded = np.pad(h, (0, n - 1), 'constant')
            g_padded = np.pad(1 - h, (0, n - 1), 'constant')
            
            # Convolve for the two parts: 1-h with w, h with -w
            grad_part1 = np.convolve(w, g_padded)[:n]
            grad_part2 = -np.convolve(w, h_padded)[:n]
            return dx * (grad_part1 + grad_part2)
        
        # Gradient descent step
        for i in range(max_iter):
            if time.time() - start_time > time_budget - 5:
                break
                
            g = compute_grad(h_current)
            h_new = h_current - 0.01 * g
            h_new = project_to_feasible(h_new, n_points)
            h_new = symmetrize(h_new)
            
            # Penality update
            penalty = lambda_pen * (np.sum(h_new) - n_points / 2)**2
            old_penalty = lambda_pen * (np.sum(h_current) - n_points / 2)**2
            
            if penalty < old_penalty:
                h_current = h_new
                current_c5 = c5_of_h(h_current, dx)
            else:
                lambda_pen *= 2  # Increase penalty
                
        beta *= 2
        if beta <= beta_max:
            # Interpolate to higher resolution
            h_interp = interp1d(np.linspace(0, 2, len(h_current)), h_current, kind='linear', fill_value='extrapolate')
            x_new = np.linspace(0, 2, n_points)
            h_current = h_interp(x_new)
            h_current = project_to_feasible(h_current, n_points)
            current_c5 = c5_of_h(h_current, dx)
    
    return h_current, current_c5

def final_hill_climb(h_init, n_points, time_budget):
    """Final stochastic hill climbing."""
    start_time = time.time()
    h_current = h_init.copy()
    dx = 2.0 / n_points
    current_c5 = c5_of_h(h_current, dx)
    min_c5 = current_c5
    h_best = h_current.copy()
    
    eps = 0.004
    max_iter = 200
    
    for i in range(max_iter):
        if time.time() - start_time > time_budget - 5:
            break
        p = min(6, n_points // 20)
        indices = np.random.choice(n_points, size=p, replace=False)
        h_trial = h_current.copy()
        
        for idx in indices:
            delta = np.random.uniform(-eps, eps)
            h_trial[idx] = np.clip(h_trial[idx] + delta, 0, 1)
            
        h_trial = project_to_feasible(h_trial, n_points)
        h_trial = symmetrize(h_trial)
        c5_trial = c5_of_h(h_trial, dx)
        
        if c5_trial < min_c5:
            min_c5 = c5_trial
            h_best = h_trial.copy()
            h_current = h_trial.copy()
            
        if i % 50 == 0:
            h_current = symmetrize(h_best)
    
    return h_best, min_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_coarse = 150
    n_fine = 300
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # 1. Global search with CMA-ES
    h_coarse, c5_coarse = cma_es_global_search(n_coarse, n_fine, budget_s * 0.5)
    
    # 2. Interpolate to fine resolution and project
    if h_coarse is not None:
        h_interp = interp1d(np.linspace(0, 2, n_coarse), h_coarse, kind='linear', fill_value='extrapolate')
        x_new = np.linspace(0, 2, n_fine)
        h_fine = h_interp(x_new)
        h_fine = project_to_feasible(h_fine, n_fine)
        h_fine = symmetrize(h_fine)
        c5_fine = c5_of_h(h_fine, dx_fine)
    else:
        h_fine = np.ones(n_fine) * 0.5
        c5_fine = c5_of_h(h_fine, dx_fine)
    
    # 3. Smooth max optimization
    h_refined, c5_refined = smooth_max_gradient_descent(h_fine, n_fine, budget_s * 0.4)
    
    # 4. Final hill climbing
    h_final, c5_final = final_hill_climb(h_refined, n_fine, budget_s * 0.05)
    
    return (h_final, c5_final, n_fine)
