import numpy as np

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def evaluate_binary_mask(b):
    """Evaluate a binary mask and return its C5 value"""
    h = b.astype(float)
    return c5_of_h(h)

def generate_seeds(n_points):
    """Generate diverse initial binary masks"""
    seeds = []
    
    # 1. Random binary with exactly n/2 ones
    for _ in range(5):
        b = np.zeros(n_points, dtype=int)
        ones = np.random.choice(n_points, n_points//2, replace=False)
        b[ones] = 1
        seeds.append(b.copy())
    
    # 2. Two blocks: 0.25, 0.75 intervals
    b = np.zeros(n_points, dtype=int)
    idx1, idx2 = int(n_points * 0.25), int(n_points * 0.75)
    b[idx1:idx2] = 1
    seeds.append(b.copy())
    
    # 3. Three blocks: 0.2, 0.2, 0.1 spaced
    b = np.zeros(n_points, dtype=int)
    idx1, idx2 = int(n_points * 0.1), int(n_points * 0.3)
    idx3, idx4 = int(n_points * 0.4), int(n_points * 0.6)
    b[idx1:idx2] = 1
    b[idx3:idx4] = 1
    seeds.append(b.copy())
    
    # 4. Sporadic ones
    b = np.zeros(n_points, dtype=int)
    for i in range(0, n_points, 5):
        if i < n_points:
            b[i] = 1
    seeds.append(b.copy())
    
    return seeds

def improve_binary_mask(b, max_iter=30000, T_start=0.02, T_decay=0.995):
    """Improve binary mask using simulated annealing"""
    n_points = len(b)
    dx = 2.0 / n_points
    
    best_b = b.copy()
    best_c5 = evaluate_binary_mask(best_b)
    current_b = b.copy()
    current_c5 = best_c5
    
    T = T_start
    accepted = 0
    
    start_time = time.time()
    timeout = 350  # seconds for SA
    
    # Precompute all correlations for efficient updates
    def compute_correlation(h):
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        return corr
    
    # Do a full correlation evaluation if needed
    corr_current = compute_correlation(current_b)
    corr_best = corr_current.copy()
    
    # Pair indices where one is 0 and one is 1
    ones = np.where(current_b == 1)[0]
    zeros = np.where(current_b == 0)[0]
    
    # Indices to track which correlation values are affected
    likely_swaps = list(zip(np.random.permutation(ones)[:10], np.random.permutation(zeros)[:10]))
    
    for i in range(max_iter):
        if time.time() - start_time > timeout:
            break
            
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Pick a random swap index
        a = np.random.choice(ones)  # a is currently 1
        b = np.random.choice(zeros)  # b is currently 0
        
        # Evaluate new correlation by modifying correlation values
        new_c5 = None
        corr_new = corr_current.copy()
        
        # For each shift k, correlate with (1-h), so we need to update only where h changes
        # In a correlation, when position a goes 1->0 and b goes 0->1
        # at shift k, the delta is:
        # d_k = (h_a * (1-h_{a+k}) - h_a * (1-h_{a+k}) when a was 1) - (h_b * (1-h_{b+k}) when b was 0) 
        #   = h_b * (1-h_{b+k}) - h_a * (1-h_{a+k})
        #  = (1-h_{b+k}) - (1-h_{a+k}) = h_{a+k} - h_{b+k}
        # But calculating delta is complex so just redo correlation
        temp_b = current_b.copy()
        temp_b[a] = 0
        temp_b[b] = 1
        new_c5 = evaluate_binary_mask(temp_b)
        
        if new_c5 < current_c5:
            # Accept improvement
            current_b = temp_b.copy()
            current_c5 = new_c5
            corr_current = compute_correlation(current_b)
            
            if current_c5 < best_c5:
                best_b = current_b.copy()
                best_c5 = current_c5
                corr_best = corr_current.copy()
            
            ones = np.where(current_b == 1)[0]
            zeros = np.where(current_b == 0)[0]
            accepted += 1
        else:
            # Accept with probability
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                current_b = temp_b.copy()
                current_c5 = new_c5
                corr_current = compute_correlation(current_b)
                ones = np.where(current_b == 1)[0]
                zeros = np.where(current_b == 0)[0]
                accepted += 1

        # Cooling
        if accepted % 100 == 0:
            T *= T_decay

    return best_b, best_c5

def project_to_feasible(h):
    """Project h into box [0,1]^n and makes sum=1"""
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h - 1.0) < 1e-10:
        return h
    # Adjust uniformly
    h = h * (1.0 / sum_h)
    return np.clip(h, 0, 1)
