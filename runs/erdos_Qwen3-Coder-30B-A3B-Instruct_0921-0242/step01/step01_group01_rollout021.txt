```python
import numpy as np
from scipy.optimize import minimize
import time
from scipy.interpolate import interp1d
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_coarse = 120
    n_fine = 300
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Convolution-based C5 computation
    def compute_C5(h):
        dx = 2.0 / len(h)
        g = 1.0 - h
        conv = np.correlate(h, g, mode='full')
        return np.max(conv) * dx

    # Water-filling projection for sum constraint
    def project_sum(h, target_sum):
        h = h.copy()
        excess = np.sum(h) - target_sum
        if abs(excess) < 1e-12:
            return h
        # Find indices that can be modified
        I = np.where((h > 0) & (h < 1))[0]
        while abs(excess) > 1e-12 and len(I) > 0:
            adjust = excess / len(I)
            for i in I:
                if adjust > 0:  # Need to increase
                    delta = min(1 - h[i], adjust)
                    h[i] += delta
                else:  # Need to decrease
                    delta = min(h[i], -adjust)
                    h[i] -= delta
            excess = np.sum(h) - target_sum
            I = np.where((h > 0) & (h < 1))[0]
        return h
    
    # Mirror symmetry
    def enforce_symmetry(h):
        return (h + h[::-1]) / 2
    
    # SDP-based coarse solution extraction
    def extract_from_sdp(X, h_vec):
        # Eigenvalue decomposition
        evals, evecs = np.linalg.eigh(X)
        # Most dominant
        idx = np.argmax(evals)
        lambda1 = evals[idx]
        v1 = evecs[:, idx]
        # Smoothed vector
        h_tilde = np.sqrt(lambda1) * np.abs(v1)
        # Apply projections
        h_proj = np.clip(h_tilde, 0, 1)
        h_proj = project_sum(h_proj, n_coarse / 2)
        if np.allclose(h_proj, h_proj[::-1], atol=1e-4):
            pass  # already symmetric, skip
        h_final = enforce_symmetry(h_proj)
        return h_final

    # Smooth max surrogate gradient
    def compute_smooth_grad(h, beta, dx):
        # Full correlation
        g = 1 - h
        corr = np.correlate(h, g, mode='full')  # length 2*n - 1
        # Ensure all values are finite to avoid overflow
        corr = np.clip(corr, -100, 100)
        # Log-sum-exp trick for stable computation
        max_corr = np.max(corr)
        corr_shifted = corr - max_corr
        exp_corr = np.exp(beta * dx * corr_shifted)
        Z = np.sum(exp_corr)
        weights = exp_corr / Z  # Softmax weights
        # Gradients: weighted sum of the derivative terms
        # The derivative term for shift k: 
        # d/dh_i of [h_i * (1 - h_{i+k}) + h_{i-k} * (1 - h_i) + ...]
        # = (1 - h_{i+k}) * delta_{i,i+k} - h_{i-k} * delta_{i,i-k}
        # Instead, calc with convolutions since that's how it's computed in CC
        # just make weights be full length not shifted, shape 2*N - 1
        # and use direct convolution again but g is 1-h
        weights_rev = weights[::-1]
        u = np.convolve(weights, 1 - h, mode='same')
        v = np.convolve(weights_rev, h, mode='same')
        # Contribution to gradient at i = u[i] - v[i]
        gradient = dx * (u - v)
        return gradient[:len(h)]
    
    # Gradient-based refinement with smooth max surrogate
    def refine_with_smoothmax(h0, max_iter=50, beta0=0.5, beta_max=50):
        h = h0.copy()
        beta = beta0
        for i in range(100):  # Just a failsafe, if it's converging quickly, it'll exit before
            if time.time() - start_time > budget_s - 1:
                break
            grad = compute_smooth_grad(h, beta, dx_coarse)
            # L-BFGS step (very basic gradient descent for now)
            step_size = 0.01
            h_new = h - step_size * grad
            # Clip to feasible box
            h_new = np.clip(h_new, 0, 1)
            # Project sum constraint (water filling)
            h_new = project_sum(h_new, n_coarse / 2)
            # Enforce symmetry
            h_new = enforce_symmetry(h_new)
            if np.linalg.norm(h_new - h) < 1e-6:
                break
            h = h_new
            beta = min(beta_max, beta * 2.0)
        return h

    # Initial solution
    best_h = None
    best_C5 = float('inf')
    n_best = n_fine

    # Try SDP relaxation
    try:
        # Build Toeplitz matrices for autocorrelations at all lags
        # Create the constraint matrices R_k
        R_matrices = []
        for k in range(n_coarse):
            R = np.zeros((n_coarse, n_coarse))
            for i in range(n_coarse):
                j = i + k
                if j < n_coarse:
                    R[i, j] = 1
                j = i - k
                if j >= 0:
                    R[i, j] = 1
            R_matrices.append(R)
        
        h_var = Variable(n_coarse)
        X = Variable((n_coarse, n_coarse), symmetric=True)
        t = Variable()
        
        constraints = [X >> 0,  # PSD
                       h_var >= 0,
                       h_var <= 1,
                       sum(h_var) == n_coarse / 2,
                       ]
        for k, R in enumerate(R_matrices):
            constraints.append(trace(R @ X) >= t)
        
        objective = Maximize(t)
        prob = Problem(objective, constraints)
        
        # Early time check for more safety
        if time.time() - start_time > budget_s - 5:
            raise Exception("Time is up")
        
        # Solve with SCS solver
        try:
            prob.solve(solver=SCS, max_iters=5000, eps=1e-5)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_sol = h_var.value
                X_sol = X.value
                if h_sol is not None and X_sol is not None:
                    h_init = extract_from_sdp(X_sol, h_sol)
                else:
                    raise Exception("No solution")
            else:
                raise Exception("Couldn't solve SDP")
        except:
            raise Exception("SDP solver failed or timed out")
            
        # Now refine the solution using smooth max
        h_refined = refine_with_smoothmax(h_init, max_iter=50)
        if time.time() - start_time > budget_s - 1:
            pass  # Just return whatever best we have
        C5_test = compute_C5(h_refined)
        if C5_test < best_C5:
            best_C5 = C5_test
            best_h = h_refined.copy()
        
    except Exception as e:
        # Fallback to simpler method
        pass
    
    if best_h is None:
        # Fallback to uniform start
        h = np.ones(n_coarse) * 0.5
        h = project_sum(h, n_coarse / 2)
        h = enforce_symmetry(h)
        best_h = h
        best_C5 = compute_C5(h)

    # Upsample to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_coarse), best_h)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = project_sum(h_fine, n_fine / 2)
    h_fine = enforce_symmetry(h_fine)
    
    best_h = h_fine
    best_C5 = compute_C5(h_fine)

    # Fine-grid optimization with smooth max surrogate
    if time.time() - start_time < budget_s - 10:
        final_h = refine_with_smoothmax(h_fine, max_iter=50)
        if time.time() - start_time < budget_s - 5:
            c5_final = compute_C5(final_h)
            if c5_final < best_C5:
                best_C5 = c5_final
                best_h = final_h.copy()

    # Final polishing
    if time.time() - start_time < budget_s - 1:
        epsilon = 0.004
        current_h = best_h.copy()
        current_c5 = compute_C5(current_h)
        
        while time.time() - start_time < budget_s - 1:
            try:
                try_indices = np.random.choice(n_fine, size=min(6, n_fine//20), replace=False)
                h_trial = current_h.copy()
                for i in try_indices:
                    pert = np.random.uniform(-epsilon, epsilon)
                    h_trial[i] = np.clip(h_trial[i] + pert, 0, 1)
                h_trial = project_sum(h_trial, n_fine / 2)
                h_trial = enforce_symmetry(h_trial)
                c5_trial = compute_C5(h_trial)
                if c5_trial < current_c5:
                    current_c5 = c5_trial
                    current_h = h_trial.copy()
                if c5_trial < best_C5:
                    best_C5 = c5_trial
                    best_h = current_h.copy()
            except:
                pass

    return (best_h, best_C5, n_best)
```
