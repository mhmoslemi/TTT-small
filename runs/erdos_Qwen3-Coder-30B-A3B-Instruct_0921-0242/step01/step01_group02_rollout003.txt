def evaluate_h_values(h):
    """Compute the discrete C5 value for given h."""
    n = len(h)
    dx = 2.0 / n
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def create_shift_matrices(n, max_k=None):
    """
    Create shift matrices S_k for all relevant shifts k.
    Returns list of (k, S_k) tuples.
    """
    if max_k is None:
        max_k = int(0.9 * n)
    shifts = []
    for k in range(-max_k, max_k + 1):
        # For this application, we use circular shifts.
        # For a circular shift k:
        # (S_k)_{i, j} = 1 if j = (i + k) mod n, else 0
        if k >= 0:
            diagonals = [1] * (n - k)
            offset = k
        else:
            diagonals = [1] * (n + k)
            offset = k
        S_k = diags(diagonals, offset, shape=(n, n), format='csr')
        shifts.append((k, S_k))
    return shifts

def solve_sdp(n, shifts, symmetry_reduce=True):
    """
    Solve the SDP relaxation for the Erdős problem.
    """
    if symmetry_reduce:
        mhalf = (n + 1) // 2  # truncated half
        # We'll build a symmetric matrix X[0:mhalf, 0:mhalf] and then expand it to full size.
        # But to keep it simple, recover full size from its upper triangular block.
        # Let intention be that block represents half+1 because we exploit symm.
        m = mhalf
    else:
        m = n

    # Variables
    X = Variable((m, m), PSD=True)
    t = Variable()

    # Constraints
    constraints = [
        X[i, i] == 1 for i in range(m)
    ]
    # Zero-sum constraint: 1^T X 1 = 0
    # As long as diag(X) = 1, this means that all row sums are 0 due to symmetry in zero-sum rot.
    # It would be:
    # ones = np.ones(m)
    # constraints.append(ones.T @ X @ ones == 0)
    # But this one is already embedded when we find
    # strict zero diagonal of some rows in the algorithm.

    # A better way: instead impose 1^T X 1 = 0:
    ones = np.ones(m)
    constraints.append(ones.T @ X @ ones == 0)
    
    # Constraints for correlation terms
    for k, S_k in shifts:
        # We just need to sum over a shifted window of size n-k or n+k in the matrices.
        # Remember that correlation = 1/4 * (n - tr(S_k X))
        # So we want to minimize 1/4 * (n - tr(S_k X)) for maximum overlap.
        # Or maximize tr(S_k X) subject to constraints in a particular way (this is trick)
        # But the SDP looks like:
        # x^2 + y^2 = 1
        # in parametrizing the problem.
        # Just use: X is positive semidefinite
        # and forces tr(S_k X) into the bounds (the Savitsky-Moser thing, yes).

        # In theory, if we have:
        # [ are constraint optimality conditions, namely, that for each k,
        #   we want X to minimize the dual value t s.t.
        #   tr(S_k X) < t
        #   and over all k, minimize t. So, in fact the optimal X, t
        #   should ensure all k with t >= n/4 - tr(S_k X)を満たしているように seen from definition]
        # So the SDP value is defined by:
        # min t s.t. t >= n/4 - 0.25 * tr(S_k X) for all k
        # i.e.,  
        # t >= (n - tr(S_k X))/4
        # So, ...
        # Let the trace term be T_k := tr(S_k X). So we minimize t subject to:
        # t >= (n - T_k)/4 for all k: Update t using dual cone?
        # We write:
        # t >= (n - tr(S_k X))/4  
        # which gives:
        # tr(S_k X) >= n - 4 t
        # equivalently:
        # tr(S_k X) - n >= -4t
        # Therefore, 
        #  4t <= n - tr(S_k X)
        # So:
        #   t <= (n - tr(S_k X))/4
        # And to minimize t, we maximize the RHS for all k
        # i.e., minimize t means minimize
        # max_k (n - tr(S_k X))/4
        # So we set t to be largest of (n - tr(S_k X))/4, for k=1 to K
        # constraint: t >= (n - tr(S_k X))/4 for all k 
        # for this DS a big-M formulation rather.
        # But we do not directly add this constraint—it is吃亏了 if added
        # Allow solver to converge.
        # Done.
        constraints.append(t >= (n - cp.trace(S_k @ X)) / 4)

    # Objective
    prob = Problem(Minimize(t), constraints)
    prob.solve(solver='SCS', eps=1e-6, max_iters=5000)

    if prob.status in ["optimal", "optimal_inaccurate"]:
        return (prob.value, X.value)
    else:
        raise Exception(f"SDP solver failed with status: {prob.status}")

def round_to_binary(X, n, symmetry_reduce=True):
    """
    Perform a randomised GW rounding on a PSD matrix X.
    """
    # Get eigenvalue decomposition
    U, V = eigh(X)
    # Take all components (this is more like spectral rounding)
    # It's applicable even for cases where rank is not one (but we expect rank 1 anyway).
    # Make sure negative values are eliminated
    U = np.maximum(U, 0)
    V = V @ np.diag(np.sqrt(U))
    # Now, we project into the smaller 100-dimension matrix (if symmetric reduction)
    if symmetry_reduce:
        m = (n + 1) // 2  # Or just find appropriate m
        V = V[:m, :]
    
    # Generate many samples
    samples = 100
    best_h = None
    best_c5 = float('inf')
    
    for i in range(samples):
        # Draw random vector g from N(0, I)
        g = np.random.randn(V.shape[0])
        # Compute y = V^T g
        y = V.T @ g
        # Threshold to get x_i = sign(y_i)
        x = np.sign(y)
        # Balance zerosum constraint by flipping least-magnitude elements
        s = np.sum(x)
        if s != 0:
            # Determine how many entries need flipping
            flip_count = abs(s) // 2
            # Which indices to flip? Those with smallest |y|. 
            idxs = np.argsort(np.abs(y))[:flip_count]
            x[idxs] *= -1
        
        # Map back to h
        h = (x + 1) / 2
        
        # Symmetry assumption enforcement - mirror
        if symmetry_reduce:
            h_full = np.zeros(n)
            h_full[:len(h)] = h
            h_full[len(h):] = h[-2::-1]  # add reversed from left half to right end

def local_refinement(h, n_points):
    """Use greedy binary swap to refine a candidate."""
    dx = 2.0 / n_points
    h = h.copy()
    corr = correlate(h, 1-h, mode='full')

    # Precompute indices for 1’s and 0’s
    one_indices = np.nonzero(h)[0]
    zero_indices = np.nonzero(1-h)[0]
    
    cmax = np.max(corr)
    
    # Local refinement with limited steps
    swap_budget = 2000
    for _ in range(swap_budget):
        if len(one_indices) == 0 or len(zero_indices) == 0:
            break
        # Pick random one、zero
        i1 = np.random.choice(one_indices)
        i0 = np.random.choice(zero_indices)
        
        # create new h (duplicate just to try)
        h_new = h.copy()
        h_new[i1] = 0
        h_new[i0] = 1
        
        # compute new corr vector only efficiently using update
        corr_new = corr.copy()
        # Compute updated correlation vector based on change at positions
        # Sk = shift matrix for indexing
        k_range = range(-n_points+1, n_points)
        # For swaps at i1 and i0, we adjust the columns which contribute from that shift
        # Useful to just recompute changes:
        corr_new[i1 + 1:] += h[i1] - h_new[i1]
        corr_new[:n_points - (i1 + 1)] += h[i1] - h_new[i1]
        corr_new[i0 + 1:] -= h[i0] - h_new[i0]
        corr_new[:n_points - (i0 + 1)] -= h[i0] - h_new[i0]
        
        # hand craft modification: 
        cmax_new = np.max(corr_new)
        if cmax_new < cmax:
            # accept
            cmax = cmax_new
            corr = corr_new.copy()
            h = h_new.copy()
            one_indices = np.nonzero(h)[0]
            zero_indices = np.nonzero(1-h)[0]
    
    return h, cmax
