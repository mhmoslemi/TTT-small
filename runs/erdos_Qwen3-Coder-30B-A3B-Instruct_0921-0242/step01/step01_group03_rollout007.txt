def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

    h = np.clip(h, 0, 1)
    # Renormalize
    h = h * (N / 2) / np.sum(h)
    return h

def project_Y(Y, N, r):
    """Project Y into the feasible set with h = ||Y_i||^2 and normalization"""
    # Evaluate h
    h = np.sum(Y**2, axis=1)
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # Renormalize to sum to N/2
    h_total = np.sum(h)
    scale = (N / 2) / h_total
    h = h * scale
    # Reclip
    h = np.clip(h, 0, 1)
    # Recompute Y norms to match h
    norms = np.sqrt(h)
    # Only update Y if norms are non-zero
    for i in range(N):
        if norms[i] > 0:
            Y[i] = Y[i] / np.linalg.norm(Y[i]) * norms[i]
        else:
            Y[i] = np.zeros(r)
    return Y

def initialize_Y(N, r):
    """Initialize Y with gaussian and then project"""
    Y = np.random.randn(N, r)
    Y = project_Y(Y, N, r)
