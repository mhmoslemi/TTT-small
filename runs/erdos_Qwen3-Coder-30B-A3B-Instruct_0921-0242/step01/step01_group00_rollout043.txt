def proj_simplex_box(h, n):
    """Project h onto {0 <= h_i <= 1, sum(h) = n/2} using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Add diff/n to each element (this is a simple linear projection)
    # But check bounds after
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def c5(h):
    """Evaluate overlap integral C5 for step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def worst_shift(h):
    """Return shift index that maximizes overlap."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    return np.argmax(corr) - (len(h) - 1)

def gradient_step(h, k, alpha):
    """Perform one gradient step along worst shift k."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Gradient of I_k(h) = dx * (1 - sum h_i h_{i+k})
    # dI_k/dh_i = -dx * h_{i+k}
    for i in range(n):
        j = (i + k) % n
        grad[i] = -dx * h[j]
    
    # Step
    h_new = h + alpha * grad
    return proj_simplex_box(h_new, n)

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    elapsed = lambda: time.time() - start_time

    h = None
    if elapsed() < budget_s - 30:

            h = solve_lp_relaxation(n_points)
        except:
            pass
            
    # Phase 2: Gradient-based refinement on worst shift (at most 10s)
    if h is None or elapsed() >= budget_s - 15:
        # Fallback to previous method
        h = np.ones(n_points) * 0.5
    if elapsed() < budget_s - 10:
        try:
            # Refinement: targeted gradient descent on worst shift
            max_iter = 100
            for iter in range(max_iter):
                if elapsed() > budget_s - 1: break
                k = worst_shift(h)
                h = gradient_step(h, k, alpha=0.1)
        except Exception as e:
            pass
            
    # Phase 3: Annealing (at most 30s)
    if elapsed() < budget_s - 10:
        try:
            best_h = h.copy()
            best_c5 = c5(h)
                
            # Simulated Annealing
            T = 1.0
            T_min = 1e-6
            alpha = 0.995
            max_iter = 1000
            iter = 0
            while T > T_min and iter < max_iter:
                if elapsed() > budget_s - 0.5: break
                    
                # Pick two indices to modify
                i, j = np.random.choice(n_points, 2, replace=False)
                delta = np.random.uniform(-0.02, 0.02)
                
                # Save old values
                old_h_i, old_h_j = h[i], h[j]
                
                # Update
                h[i] += delta
                h[j] -= delta
                
                # Project
                h = proj_simplex_box(h, n_points)
                
                # Compute new value
                new_c5 = c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                else:
                    # Accept with probability
                    if np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
                        pass  # Keep
                    else:
                        # Undo
                        h[i], h[j] = old_h_i, old_h_j
                        
                T *= alpha
                iter += 1
                
            h = best_h
        except Exception as e:
            pass
            
    # Final local hill climbing
    if elapsed() < budget_s - 0.5:
        try:
            best_h = h.copy()
            best_c5 = c5(h)
            
            for _ in range(1000):
                if elapsed() > budget_s - 0.1: break
                    
                h_trial = h.copy()
                # Modify 5 random indices at once
                indices = np.random.choice(n_points, size=5, replace=False)
                for i in indices:
                    h_trial[i] += np.random.uniform(-0.01, 0.01)

                    best_c5 = trial_c5
                    best_h = h_trial.copy()
                    h = h_trial.copy()
        except Exception as e:
            pass
            
    return (best_h, best_c5, n_points)
