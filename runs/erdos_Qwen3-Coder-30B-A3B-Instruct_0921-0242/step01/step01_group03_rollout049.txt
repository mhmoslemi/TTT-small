import numpy as np
import cvxpy as cp
import time
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import diags
import warnings
warnings.filterwarnings('ignore')

def build_shift_matrices(N):
    """Build sparse CSR shift matrices A_k"""
    A_list = []
    for k in range(N):
        # A_k[i,j] = 1 if j = i+k and i+k < N, else 0
        row = np.arange(N - k)
        col = np.arange(k, N)
        data = np.ones(N - k)
        A = csr_matrix((data, (row, col)), shape=(N, N))
        A_list.append(A)
    return A_list

def bounded_simplex_projection(h, target_sum):
    """Project h to the feasible set with sum = target_sum and h in [0,1]^N"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    deficit = target_sum - current_sum
    
    if abs(deficit) < 1e-10:
        return h
    
    # Distribute deficit
    adjustment = deficit / len(h)
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def eigen_rounding(X):
    """Principal eigenvector rounding"""
    try:
        # Compute largest eigenvalue and eigenvector
        lam, v = np.linalg.eigh(X)
        i_max = np.argmax(lam)
        v1 = v[:,i_max]
        h_eig = np.sqrt(lam[i_max]) * np.abs(v1)
        h_eig = bounded_simplex_projection(h_eig, len(h_eig)/2)
        return h_eig
    except:
        return None

def diagonal_rounding(X):
    """Diagonal threshold rounding"""
    h_diag = np.sqrt(np.diag(X))  # estimates of h_i
    idx_sorted = np.argsort(h_diag)[::-1]
    h_thr = np.zeros_like(h_diag)
    h_thr[idx_sorted[:len(h_diag)//2]] = 1.0
    return h_thr

def evaluate_c5(h):
    """Compute exact C5 using full correlation"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def compute_full_correlation(h):
    """Compute all correlations using FFT"""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), mode='constant')
    g_padded = np.pad(1-h, (0, n-1), mode='constant')
    fft_h = fft(h_padded)
    fft_g = fft(g_padded)
    correlation = ifft(fft_h * np.conj(fft_g)).real
    return correlation[n-1:]

def update_correlation_incremental(corr, h, p, q, A_list):
    """Update correlation after swapping h[p] and h[q]"""
    # corr contains the correlation values for shifts k=0,1,...,n-1
    # indices p and q are swapped, so we can update efficiently
    
    # For each k, the changes are in terms of indices (p+k) and (q+k)
    # and (p-k) and (q-k) if k < p and k < q respectively
    n = len(h)
    for k in range(n):
        # Old contributions to correlation k:
        # If p+k < n: h[p] * h[p+k]  If q+k < n: h[q] * h[q+k]
        # If p-k >= 0: h[p] * h[p-k]  If q-k >= 0: h[q] * h[q-k]
        old_val = 0
        new_val = 0
        
        # contribution of pair (p, p+k) if p+k is valid
        if p+k < n:
            old_val += h[p] * h[p+k]
            new_val += h[q] * h[q+k]
        # contribution of pair (q, q+k) if q+k is valid  
        if q+k < n:
            old_val += h[q] * h[q+k]
            new_val += h[p] * h[p+k]
        # contribution of pair (p-k, p) if p-k >= 0
        if p-k >= 0:
            old_val += h[p] * h[p-k]
            new_val += h[q] * h[q-k]
        # contribution of pair (q-k, q) if q-k >= 0
        if q-k >= 0:
            old_val += h[q] * h[q-k]
            new_val += h[p] * h[p-k]
        
        # Changes in correlation for shift k:
        delta = new_val - old_val
        corr[k] -= delta
    return corr

def swap_improvement(h, A_list, corr):
    """Deterministic swap hill-climbing"""
    n = len(h)
    best_h = h.copy()
    best_corr = corr.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Determine current worst shift direction
    worst_k = np.argmax(best_corr)
    
    iterations = 0
    while time.time() < time.time() + 180:  # 3 min limit
        iterations += 1
        # Pick a random 1 and 0
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = np.random.choice(ones)
        q = np.random.choice(zeros)
        
        # Compute new correlation vector after swap
        corr_new = best_corr.copy()
        corr_new = update_correlation_incremental(corr_new, best_h, p, q, A_list)
        new_h = best_h.copy()
        new_h[p], new_h[q] = new_h[q], new_h[p]  # Swap
        new_c5 = np.max(corr_new)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_corr = corr_new
            best_c5 = new_c5
        else:
            # Occasionally full recomputation to avoid drift
            if iterations % 500 == 0:
                corr_full = compute_full_correlation(best_h)
                current_c5 = np.max(corr_full)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_corr = corr_full
    
    return best_h, best_c5

def solve_sdp(N):
    """Solve SDP relaxation of the problem"""
    # Build the shift matrices
    A_list = build_shift_matrices(N)
    # Formulate the SDP problem
    h = cp.Variable(N)
    X = cp.Variable((N, N), symmetric=True)
    t = cp.Variable()
    
    constraints = [cp.diag(X) == h]
    constraints += [cp.sum(h) == N/2]
    constraints += [h >= 0]
    constraints += [h <= 1]
    constraints += [X >> 0]
    for k in range(N):
        constraints += [cp.trace(A_list[k] @ X) >= t]
    
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=15000, eps=1e-4)
        return h.value, X.value
    except:
        return None, None

def refine_solution(h, X):
    """Try both rounding methods and select the better one"""
    h_eig = eigen_rounding(X)
    h_diag = diagonal_rounding(X)
    if h_eig is not None and h_diag is not None:
        c5_eig = evaluate_c5(h_eig)
        c5_diag = evaluate_c5(h_diag)
        if c5_eig < c5_diag:
            return h_eig
        else:
            return h_diag
    elif h_eig is not None:
        return h_eig
    elif h_diag is not None:
        return h_diag
    else:
        return None

def upsample_solution(h_coarse, n_final):
    """Linear interpolation and renormalize"""
    h_up = np.zeros(n_final)
    dx_coarse = 2.0 / len(h_coarse)
    dx_fine = 2.0 / n_final
    for i in range(n_final):
        x = i * dx_fine
        k = x / dx_coarse
        lower = int(np.floor(k))
        upper = int(np.ceil(k))
        alpha = k - lower
        if lower >= len(h_coarse):
            h_up[i] = 0.0
        elif upper >= len(h_coarse):
            h_up[i] = h_coarse[lower]
        else:
            h_up[i] = alpha * h_coarse[upper] + (1 - alpha) * h_coarse[lower]
    # Renormalize
    h_up = h_up * (n_final / 2) / np.sum(h_up)
    return h_up

def apply_fine_tuning(h_fine, n_iters=30):
    """Projected gradient descent using log-sum-exp surrogate"""
    n = len(h_fine)
    dx = 2.0 / n
    # Heuristic: good temperature for log-sum-exp
    tau = 1e-4
    
    def objective_log_sum_exp(h):
        g = 1 - h
        corr_full = np.correlate(h, g, mode='full') * dx
        # log-sum-exp surrogate for max
        return -np.log(np.sum(np.exp(-corr_full / tau))) * tau
    
    def gradient_log_sum_exp(h):
        g = 1 - h
        corr_full = np.correlate(h, g, mode='full') * dx
        # Normalized softmax weights
        weights = np.exp(-corr_full / tau)
        weights /= np.sum(weights)
        # Compute gradient of each term
        grad_h = np.zeros(n)
        for k in range(n):
            # Derivative w.r.t h[i] of term h[i] * h[i+k] for each k 
            # We compute only non-trivial part for simplicity of implementation
            # Approximate using derivative of log-sum-exp 
            i = np.arange(n)
            i_plus_k = (i + k) % n
            # d/dh_i of h_i * h_{i+k} equals h_{i+k}
            grad_h += weights[k] * (1.0 - h[i_plus_k])  # weighted by softmax
        return -grad_h
    
    # Gradient descent steps with projection
    for _ in range(n_iters):
        grad = gradient_log_sum_exp(h_fine)
        h_new = h_fine - 0.001 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n/2) / np.sum(h_new)  # Renormalize
        h_fine = h_new
    
    return h_fine

def run(seed=42, budget_s=600, **kwargs):
    """Main function for computing optimal h with minimal C5"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose initial N
    N = 150
    if time.time() - start_time > budget_s - 100:
        N = 100
    
    # Step 1: Solve SDP
    h_sdp, X_sdp = solve_sdp(N)
    if h_sdp is None or X_sdp is None:
        # Fallback
        h_fallback = np.ones(N) * 0.5
        h_fallback = bounded_simplex_projection(h_fallback, N/2)
        return (h_fallback, evaluate_c5(h_fallback), N)
    
    # Step 2: Rounding
    h_rounded = refine_solution(h_sdp, X_sdp)
    if h_rounded is None:
        h_fallback = np.ones(N) * 0.5
        h_fallback = bounded_simplex_projection(h_fallback, N/2)
        return (h_fallback, evaluate_c5(h_fallback), N)
        
    h_bin = h_rounded  # binary mask-like
    c5_bin = evaluate_c5(h_bin)
    
    # Step 3: Swap improvement
    A_list = build_shift_matrices(N)
    corr_full = compute_full_correlation(h_bin)
    h_improved, c5_improved = swap_improvement(h_bin, A_list, corr_full)
    if c5_improved < c5_bin:
        h_final = h_improved
        c5_final = c5_improved
    else:
        h_final = h_bin
        c5_final = c5_bin

    # Step 4: Upsample
    N_up = 2 * N
    h_up = upsample_solution(h_final, N_up)
    c5_up = evaluate_c5(h_up)
    
    # Step 5: Fine-tune
    h_fine = apply_fine_tuning(h_up)
    c5_fine = evaluate_c5(h_fine)
    
    # Final solution:
    h_result = h_fine
    c5_result = c5_fine
    n_points = N_up
    
    # Rare: if falling back to coarser grid
    if N == 100:
        # This does not typically happen, but spell out just in case
        pass
    
    # Return best found
    return (h_result, c5_result, n_points)
