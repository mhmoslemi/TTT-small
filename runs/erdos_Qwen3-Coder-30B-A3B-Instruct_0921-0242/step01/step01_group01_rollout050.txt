```python
import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def c5_of_h(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution
    conv = np.correlate(h, g, mode='full')
    # The full convolution has 2*len(h) - 1 elements
    # The shifts are from -(len(h)-1) to (len(h)-1)
    result = np.max(conv) * dx
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Model parameters
    n = 200  # grid points (even, so n/2+1 independent vars)
    dx = 2.0 / n
    d = n // 2 + 1  # number of independent variables (due to symmetry)
    n_points = n
    max_iterations = 50
    eps = 1e-6

    # Start with an initial construction from globals if available
    if 'initial_h_values' in globals():
        initial = initial_h_values
        # interpolate to n points
        x_orig = np.linspace(0, 2, len(initial))
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_orig, initial)
        # Force sum to equal n/2
        h_interp = h_interp * (n / 2) / np.sum(h_interp)
    else:
        # Initialize a simple symmetric h
        h_interp = np.ones(n) * 0.5  # uniform case
        h_interp = h_interp * (n / 2) / np.sum(h_interp)

    # Create the full h vector (symmetric)
    h_full = h_interp.copy()
    if n % 2 == 0:
        h_full[n//2:] = h_full[n//2-1::-1]
    else:
        h_full[n//2:] = h_full[n//2::-1]

    # Compute the initial C5
    best_c5 = c5_of_h(h_full)
    best_h = h_full.copy()

    # Symmetrized variables:
    # w is the independent half of h (index 0 to d-1)
    # Symmetry makes indices w_i and w_{n-1-i} equal in the full vector
    w = Variable(d)
    M = Variable()
    lambda_reg = 0.1  # binary penalty parameter

    # Define constraints
    # Sum constraint
    sum_cons = cp.sum(w) == n/4 + 0.5

    # Box constraints
    box_cons = [w >= 0, w <= 1]

    # Graph of the set of feasible (h,h^{(k)}) points
    # Asymptotic independence (no constraints in the rest)
    constraints = [sum_cons] + box_cons

    # Always include shift 0
    S = [0]

    # To build a list of key terms without actual objects earlier
    shift_k = 0
    p_vars = {}
    p_vars[shift_k] = []
    for i in range(n):
        p_vars[shift_k].append(Variable())

    # Here we construct the problem structure lazily, reusing solver if possible

    # Double-oracle loop:
    iteration = 0
    while iteration < max_iterations and (time.time() - start_time < budget_s - 5):
        iteration += 1
        if time.time() - start_time > budget_s - 5:
            break

        # Build LP structure
        new_p_vars = {}   # to store new p_{i,k} vars for active k in S
        all_p = []
        all_constraints = constraints[:]  # Copy base constraints
        all_p.append(M)  # M is part of the objective

        # Create variables for all p_{i,k}'s in the current active set S
        # As in loop we'll add p_{i,k} for all k in S
        all_shift_data = []  # [k_start, k_end] for reversibility

        for k in S:
            p_vars[k] = []
            start_idx = max(0, -k)
            end_idx = min(n, n - k)
            for i in range(start_idx, end_idx):
                p_vars[k].append(Variable())
                all_p.append(p_vars[k][-1])
            all_shift_data.append((k, start_idx, end_idx))

        # Add McCormick envelope constraints
        p_list = []
        for k in S:
            p_vars_list = p_vars[k]
            start_idx = max(0, -k)
            end_idx = min(n, n - k)
            for i, pval in enumerate(p_vars_list):
                idx = i + start_idx  # in the full list context
                # All four McCormick bounds
                if idx + k >= 0 and idx + k < n:
                    all_constraints.append(pval <= w[idx])
                    all_constraints.append(pval <= w[idx + k])
                    all_constraints.append(pval >= 0)
                    all_constraints.append(pval >= w[idx] + w[idx + k] - 1)
                    p_list.append(pval)
                else:
                    # Don't add infeasible (OutOfBounds)
                    pass  # we're inside the domain

        # Objective: Minimize M - lambda_reg * sum of p_{i,0}
        if 0 in S:
            sum_p_0 = sum(p_vars[0])
        else:
            sum_p_0 = 0

        objective = Minimize(M - lambda_reg * sum_p_0)

        # Add all overlap constraints:
        for k in S:
            p_vars_list = p_vars[k]
            start_idx = max(0, -k)
            end_idx = min(n, n - k)
            sum_p_k = sum(p_vars_list)
            # Overlap = (n/2) - sum_p_k <= M
            all_constraints.append((n / 2) - sum_p_k <= M)
        # Solve LP using ECOS
        prob = Problem(objective, all_constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
        except:
            # If it fails to solve, use a default good state
            break

        # Extract solution
        w_sol = w.value  # np array of length d
        if w_sol is None:
            break

        # Reconstruct h (full vector) from w
        h_sol = np.zeros(n)
        for i in range(d):
            h_sol[i] = w_sol[i]
            h_sol[n-1-i] = w_sol[i]

        # Compute C5 of this solution
        c5_sol = c5_of_h(h_sol)

        # Update best solution found
        if c5_sol < best_c5:
            best_c5 = c5_sol
            best_h = h_sol.copy()

        # Termination check
        if np.abs(c5_sol - M.value) < eps:
            break

        # Add new constraint: the actual maximum overlap shift
        g = 1.0 - h_sol
        conv = np.correlate(h_sol, g, mode='full')
        idx_max = np.argmax(conv)
        k_max = idx_max - (n - 1)
        if k_max not in S:
            S.append(k_max)

        # Increase lambda_reg
        lambda_reg = min(5.0, lambda_reg * 1.5)

    # Polish the solution using projected subgradient method
    h_current = best_h.copy()
    c5_current = c5_current
    stepsize = 0.02
    max_polish_iters = 200
    polish_iter = 0
    while polish_iter < max_polish_iters and (time.time() - start_time < budget_s - 5):
        polish_iter += 1
        if time.time() - start_time > budget_s - 5:
            break

        # Compute deviation from symmetry at current point
        # Symmetrize for protection if needed
        h_reflect = h_current[::-1]
        h_symmetric = (h_current + h_reflect) / 2.0
        h_current = h_symmetric.copy()
        g = 1.0 - h_current
        conv = np.correlate(h_current, g, mode='full')
        idx_max = np.argmax(conv)
        k_max = idx_max - (n - 1)

        # Define subgradient g_i = (1-h_{i+k_max}) - h_{i-k_max}
        # with correct boundary clipping
        grad = np.zeros(n)
        for i in range(n):
            if i + k_max >= 0 and i + k_max < n:
                grad[i] += (1 - h_current[i + k_max])
            if i - k_max >= 0 and i - k_max < n:
                grad[i] -= h_current[i - k_max]

        # Project gradient step: h ← P(h - alpha*grad)
        h_propose = h_current - stepsize * grad

        # 1. Clip to [0,1]
        h_propose = np.clip(h_propose, 0, 1)

        # 2. Enforce sum to n/2 using waterfilling approach
        current_sum = np.sum(h_propose)
        diff = (n / 2) - current_sum
        scalar = diff / n
        h_propose = h_propose + scalar
        h_propose = np.clip(h_propose, 0, 1)  # Clip again after scalar adjustment

        # Symmetrize again
        h_reflect = h_propose[::-1]
        h_propose = (h_propose + h_reflect) / 2.0

        # Compute C5
        c5_propose = c5_of_h(h_propose)
        if c5_propose < c5_current:
            h_current = h_propose.copy()
            c5_current = c5_propose
            stepsize *= 0.9   # Decay step size
            if abs(c5_current - c5_propose) < 1e-6:
                break
        else:
            stepsize *= 0.9   # Decay step size if no improvement

    best_c5 = c5_current
    best_h = h_current

    # Final optional stochastic hill-climb
    # Within remaining budget, run a few random moves
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1.0:
        num_moves = min(2000, int(remaining_time / 0.015))
        for _ in range(num_moves):
            if time.time() - start_time > budget_s - 0.5:
                break
            indices = np.random.choice(n, size=min(10, n//20), replace=False)
            h_trial = best_h.copy()
            for i in indices:
                delta = np.random.uniform(-0.003, 0.003)
                h_trial[i] += delta
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n / 2) / np.sum(h_trial)

            # Symmetrize
            h_reflect = h_trial[::-1]
            h_trial = (h_trial + h_reflect) / 2.0

            # Compute exact overlap
            c5_trial = c5_of_h(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()

    return best_h, best_c5, n_points
```
