import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time
import math

def project_to_feasible(u, n):
    """Project u to feasible set: 0 <= u <= 1, sum(u) = n/2"""
    u = np.clip(u, 0.0, 1.0)
    s = np.sum(u)
    if abs(s - n/2) < 1e-12:
        return u
    
    # Find lambda via bisection
    def sum_at_lambda(lam):
        return np.sum(np.clip(u + lam, 0.0, 1.0)) - n/2

    lam_low, lam_high = -10.0, 10.0
    for _ in range(30):
        lam_mid = (lam_low + lam_high) / 2
        if sum_at_lambda(lam_mid) > 0:
            lam_high = lam_mid
        else:
            lam_low = lam_mid
    lam = (lam_low + lam_high) / 2
    
    u = np.clip(u + lam, 0.0, 1.0)
    return u

def evaluate_c5(h, dx):
    """Fast FFT-based evaluation of C5"""
    n = len(h)
    # Zero-padded FFT
    H = np.fft.rfft(h, n=2*n-1)
    G = np.fft.rfft(1-h, n=2*n-1)
    conv = np.fft.irfft(H*G)[:2*n-1]
    # Early stopping optimization
    max_val = np.max(conv)
    return max_val * dx

def evaluate_c5_naive(h):
    """Naive evaluation - for verification"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    return np.max(conv) * dx

def cma_es_global(n_points_c, budget_s):
    """Global search using CMA-ES with symmetry"""
    np.random.seed(42)
    start_time = time.time()
    
    # Use symmetric representation
    d = (n_points_c + 1) // 2  # First d variables
    n_points = n_points_c
    
    # Initial parameters
    mu = 0.5 * np.ones(d)
    sigma = 0.2
    lambda_pop = 4 + int(3 * np.log(d))
    
    # CMA-ES parameters
    mu_eff = lambda_pop / 2
    c1 = 2.0 / (d + 2.0)**2
    cmu = min(1 - c1, 2 * (mu_eff - 2 + 1/mu_eff) / (d**2 + 5))
    sigma_decay = 0.95
    
    # For convergence detection
    best_val = float('inf')
    best_h = None
    no_improvement = 0
    restart_count = 0
    
    pop_z = np.zeros((lambda_pop, d))
    pop_h = np.zeros((lambda_pop, n_points))
    fitness = np.zeros(lambda_pop)
    weights = np.log(lambda_pop/2 + 1) - np.log(np.arange(1, lambda_pop + 1))
    weights = weights / np.sum(weights)
    
    # Optimization loop
    generation = 0
    while time.time() - start_time < budget_s - 1:
        if no_improvement > 200 or restart_count > 3:
            break
            
        # Sample offspring
        for i in range(lambda_pop):
            pop_z[i] = np.random.normal(0, 1, d)
            h_trial = mu + sigma * pop_z[i]
            # Apply symmetry: h[i] = h[n-1-i]
            h_sym = np.zeros(n_points)
            h_trial_full = project_to_feasible(h_trial, n_points)
            h_sym[:d] = h_trial_full[:d]
            if n_points % 2 == 0:
                h_sym[d:] = h_trial_full[:d-1][::-1]
            else:
                h_sym[d:] = h_trial_full[:d][::-1]
            pop_h[i] = h_sym
            # Fast evaluation
            fitness[i] = evaluate_c5(h_sym, 2.0/n_points)
            
        # Sort by fitness
        idx = np.argsort(fitness)
        best_f = fitness[idx[0]]
        if best_f < best_val:
            best_val = best_f
            best_h = pop_h[idx[0]].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Update
        z_avg = np.sum(weights[:, None] * pop_z[idx[:mu_eff]], axis=0)
        mu = mu + sigma * z_avg
        
        # Adapt sigma
        if no_improvement < 50:
            sigma *= sigma_decay ** (1 if no_improvement > 10 else 1.0)
        else:
            mu = best_h[:d] if best_h is not None else 0.5 * np.ones(d)
            # Restart with smaller sigma
            sigma = 0.15
            restart_count += 1
            no_improvement = 0
            
        generation += 1
    
    return best_h, best_val

def smooth_max_grad_descent(h_init, beta_start=0.5, beta_max=50.0, max_iters=80):
    """Local refinement using smooth-max gradient descent"""
    h = h_init.copy()
    dx = 2.0 / len(h)
    
    # Symmetrize as part of update
    def symmetrize(h_in):
        return (h_in + h_in[::-1]) / 2
        
    # Define loss function and gradient
    def compute_conv(h_in):
        n = len(h_in)
        H = np.fft.rfft(h_in, n=2*n-1)
        G = np.fft.rfft(1-h_in, n=2*n-1)
        return np.fft.irfft(H*G)[:2*n-1]
    
    beta = beta_start
    lam = 10.0  # penalty weight for constraint
    convergence_tol = 1e-6
    
    for _ in range(20):  # Number of beta stages
        if beta > beta_max:
            break
            
        # We only keep using the previous h for start (careful for gradient computation)
        for _ in range(max_iters):
            # Compute convolution
            conv_vals = compute_conv(h)
            # Compute weight (smooth max)
            weights = np.exp(beta * dx * conv_vals)
            weights = weights / np.sum(weights)
            
            # Compute gradient efficiently using convolution
            #∇conv_k = (1-h_{i+k}) - h_{i-k}
            # We consider correlation: 
            # For k=0 to n-1: h_{i} * (1 - h_{i+k}) 
            # For k=-n+1 to -1: h_{i} * (1 - h_{i+k})
            # The pure Jacobian part
            # Instead, we extract all affine gradients efficiently by reusing convolutions
            # Compute -∇(conv_k) which is h_{i-k} [when k < 0] - (1-h_{i+k}) [when k > 0]
            # For later, conv of (h_i, -1 + h_{i+k})
            # But a simpler way: Consider for all conv_k
            # Gradient wrt h_i is 1-h_{i+k} when i+k < n and -h_{i-k} when i-k >= 0
            # Build the contribution
            # Weights are convolved with these shifted terms efficiently
            # This can be 2 convolutions: 
            # First weight * (1 - h) and weight * h  
            # Then compute shifted by k convolutions to get full Jacobian
            # But better, work with the reformulated equivalent:
            
            # Split into shifts:
            # We compute weighted sum of ∇conv_k, where ∇conv_k = (1-h_{i+k}) - h_{i-k}
            # So for k >= 0: term is - h_{i-k} (when i-k >=0)  
            #         k < 0:  term is 1 - h_{i+k} (when i+k < n)
            
            # Get:
            # Right shifted by k >= 0 (⊝): pad zeros, crop, this handles h_{i-k} 
            # Left shifted by k < 0 (⊖): pad zeros, crop, this handles 1-h_{i+k}
            
            lambda_kernel = np.zeros_like(h)
            lambda_kernel[1:] = weights[1:]  # k > 0 part: -h_{i-k} -> 1 shifts 1
            right_shifted = -np.pad(h, (0, 1))[:-1]
            right_term = np.convolve(lambda_kernel, right_shifted, mode='valid')
            
            # flip for k<0: 
            lambda_kernel2 = weights[2:][::-1] # y[1:] -> i+k < n, so k = 1..n-1 -> index 2..2n-3 etc.
            lambda_kernel2 = np.pad(lambda_kernel2, (0,1), mode='constant', constant_values=0)
            left_term = np.convolve(lambda_kernel2, h[::-1], mode='valid')[::-1]
            
            grad = dx * (right_term + left_term)
            
            # Constraint gradient ((sum h) - n/2 ) 
            grad_constr = lam * np.ones_like(h)
            delta = np.sum(h) - len(h) / 2
            grad = grad + grad_constr * delta
            
            # Update
            step_size = 0.1
            h_new = h - step_size * grad
            
            # Project
            h_new = project_to_feasible(h_new, len(h))
            
            # Enforce symmetry
            h_new = symmetrize(h_new)
            
            # Check convergence
            norm_grad = np.linalg.norm(grad)
            if norm_grad < convergence_tol:
                break                
            
            h = h_new
            
        beta *= 2
        lam *= 2
    
    # Final projected h
    return h

def stochastic_hill_climb(h_init, n_iters=20000):
    """Final polishing by stochastic hill climbing"""
    h = h_init.copy()
    dx = 2.0 / len(h)
    n = len(h)
    current_c5 = evaluate_c5(h, dx)
    start_time = time.time()
    
    # Interleaving symmetry check
    for i in range(n_iters):
        elapsed = time.time() - start_time
        if elapsed > 190:
            break
        if i % 50 == 0:
            h = (h + h[::-1]) / 2
            
        # Select indices
        p = min(6, n//20)
        indices = np.random.choice(n, p, replace=False)
        h_new = h.copy()
        
        for idx in indices:
            delta = np.random.uniform(-0.004, 0.004)
            h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
        
        # Project to feasible
        h_new = project_to_feasible(h_new, n)
        h_new = (h_new + h_new[::-1]) / 2  # enforce symmetry
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < current_c5:
            h = h_new
            current_c5 = c5_new
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_coarse = 150
    n_fine = 300
    
    # Phase 1: Global search with CMA-ES
    print("Starting CMA-ES global optimization", flush=True)
    h_global, c5_global = cma_es_global(n_coarse, 350)
    if h_global is None:
        h_global = np.ones(n_coarse) * 0.5
        c5_global = evaluate_c5(h_global, 2.0/n_coarse)
    
    # Phase 2: Interpolate for fine grid
    print("Interpolating to fine grid", flush=True)
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine_interpolated = interp1d(x_coarse, h_global, kind='linear', fill_value='extrapolate')(x_fine)
    
    # Project to feasible on fine grid
    h_fine = project_to_feasible(h_fine_interpolated, n_fine)
    h_fine = (h_fine + h_fine[::-1]) / 2  # symmetry
    
    # Phase 3: Local refinement
    print("Starting smooth-max gradient descent", flush=True)
    h_local = smooth_max_grad_descent(h_fine, beta_start=0.5, beta_max=50.0, max_iters=80)
    
    # Phase 4: Final hill climbing
    print("Starting stochastic hill climbing", flush=True)
    h_final = stochastic_hill_climb(h_local, 20000)
    
    # Final evaluation
    dx_fine = 2.0 / n_fine
    final_c5 = evaluate_c5(h_final, dx_fine)
    
    # Return best solution found
    return (h_final, final_c5, n_fine)
