import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time
import cvxpy as cp
from cvxpy import SCS
import math

def water_fill(h, target_sum):
    """Project h onto the sum constraint using water-filling."""
    h = np.array(h, dtype=float)
    current_sum = np.sum(h)
    delta = current_sum - target_sum
    
    if abs(delta) < 1e-12:
        return h
    
    # Identify indices not at bounds
    active_indices = np.where((h > 0) & (h < 1))[0]
    while abs(delta) > 1e-12 and len(active_indices) > 0:
        num_active = len(active_indices)
        # Distribute delta evenly among active indices
        step = delta / num_active
        if step > 0:
            # Adding will increase the sum
            for idx in active_indices:
                if h[idx] + step >= 1.0:
                    h[idx] = 1.0
                else:
                    h[idx] += step
        else:
            # Removing will decrease the sum
            for idx in active_indices:
                if h[idx] + step <= 0.0:
                    h[idx] = 0.0
                else:
                    h[idx] += step
        current_sum = np.sum(h)
        delta = current_sum - target_sum
        active_indices = np.where((h > 0) & (h < 1))[0]
    
    return h

def compute_correlation(h, k, n_points):
    """Compute single correlation value for shift k."""
    dx = 2.0 / n_points
    if k == 0:
        return np.sum(h * (1 - h)) * dx
    elif k > 0:
        # h[i] * (1 - h[i+k])
        corr = 0.0
        for i in range(n_points - k):
            corr += h[i] * (1 - h[i + k])
        return corr * dx
    else:
        # h[i+k] * (1 - h[i])
        corr = 0.0
        for i in range(-k, n_points):
            corr += h[i + k] * (1 - h[i])
        return corr * dx

def c5_of_h(h):
    """Compute C5 for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Full convolution with 1-h
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    # Compute the maximum value (corresponds to max overlap)
    max_corr = np.max(conv)
    return max_corr * dx

def smooth_max_gradient(h, b, n_points):
    """Compute gradient of smooth max function with temperature b."""
    dx = 2.0 / n_points
    # Full convolution with 1-h
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full') # length = 2*n-1
    # Apply temperature
    conv_b = conv * dx * b
    # Subtract max to prevent overflow
    conv_b = conv_b - np.max(conv_b)
    exps = np.exp(conv_b)
    weights = exps / np.sum(exps)
    # Compute gradient using convolutions
    # Let's compute weights * (1-h) and reverse-weight * h
    u = np.correlate(weights, 1 - h, mode='full')
    v = np.correlate(weights[::-1], h, mode='full')
    # Final gradient component is dx * (u - v)
    grad = dx * (u - v)
    # We want the gradient w.r.t. the first n_points values
    return grad[:n_points]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_coarse = 120
    n_fine = 300
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # 1. Solve SDP relaxation to get coarse grid vector
    print("Solving SDP relaxation...")
    time_check = time.time()
    if time_check - start_time > budget_s - 90:
        # Fallback if not enough time
        final_h = np.ones(n_fine) * 0.5
        final_h = final_h * (n_fine / 2) / np.sum(final_h)
        return (final_h, c5_of_h(final_h), n_fine)
    
    # Prepare Toeplitz matrices for SDP
    R_k_matrices = []
    for k in range(n_coarse):
        R = np.zeros((n_coarse, n_coarse))
        for i in range(n_coarse):
            if i + k < n_coarse:
                R[i, i + k] = 1.0
            if i - k >= 0:
                R[i, i - k] = 1.0
        R_k_matrices.append(R)
    
    # Formulate SDP in cvxpy
    t = cp.Variable()
    h_vec = cp.Variable(n_coarse)
    X = cp.Variable((n_coarse, n_coarse), symmetric=True)
    M = cp.bmat([[1, h_vec.T], [h_vec, X]])

    constraints = [cp.trace(R_k_matrices[k] @ X) >= t for k in range(n_coarse)]
    constraints += [0 <= h_vec[i] for i in range(n_coarse)]
    constraints += [h_vec[i] <= 1 for i in range(n_coarse)]
    constraints += [cp.sum(h_vec) == n_coarse / 2]
    constraints += [M >> 0]

    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve the SDP with timeout
    try:
        prob.solve(solver=SCS, max_iters=5000, eps=1e-5, verbose=False)
        if prob.status != cp.OPTIMAL:
            print(f"SDP status: {prob.status}. Using fallback.")
            h_coarse = np.ones(n_coarse) * 0.5
            h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
        else:
            h_coarse = h_vec.value
            print(f"SDP successful: t = {t.value:.6f}")
    except:
        print("SDP failed, using fallback...")
        h_coarse = np.ones(n_coarse) * 0.5
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    print(f"SDP time used: {time.time() - time_check:.2f}s")
    time_check = time.time()
    
    # 2. Extract a near-rank-1 solution
    if h_coarse is not None:
        # Start from h_coarse
        pass
    else:
        # Fallback
        h_coarse = np.ones(n_coarse) * 0.5
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # 3. Refine with smooth gradient descent (coarse grid)
    print("Coarse refinement...")
    h_coarse = np.clip(h_coarse, 0, 1)
    h_coarse = water_fill(h_coarse, n_coarse / 2)
    
    t_start = time.time()
    b_vals = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]
    patience_counter = 0
    patience_thresh = 3
    prev_c5 = float('inf')
    
    for b in b_vals:
        if time.time() - start_time > budget_s - 100:
            break
        
        def obj_func(h):
            return np.log(np.sum(np.exp(b * dx_coarse * np.correlate(h, 1 - h, mode='full')))) / b
        
        def grad_func(h):
            return smooth_max_gradient(h, b, n_coarse)
        
        for i in range(80):
            if time.time() - start_time > budget_s - 60:
                break
            # Standard gradient step
            grad = grad_func(h_coarse)
            h_new = h_coarse - 0.1 * grad  # Simple step size
            h_new = np.clip(h_new, 0, 1)
            h_new = water_fill(h_new, n_coarse / 2)
            h_new = (h_new + h_new[::-1]) / 2  # Mirror symmetry
            
            # Check improvement
            if time.time() - start_time > budget_s - 60:
                break
            h_coarse = h_new
            c5_new = c5_of_h(h_coarse)
            if c5_new < prev_c5:
                prev_c5 = c5_new
                patience_counter = 0
            else:
                patience_counter += 1
            if patience_counter >= patience_thresh:
                break
    
    print(f"Coarse refinement time used: {time.time() - time_check:.2f}s")
    time_check = time.time()

    # 4. Upsample to fine grid
    print("Upsampling...")
    if len(h_coarse) != n_coarse:
        h_coarse = np.ones(n_coarse) * 0.5
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    
    # Interpolate to fine grid
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    
    # Apply constraints
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = water_fill(h_fine, n_fine / 2)
    
    # Apply symmetry
    h_fine = (h_fine + h_fine[::-1]) / 2
    h_fine = water_fill(h_fine, n_fine / 2)
    
    # Final optimization on fine grid
    print("Fine-grid refinement...")
    b_vals = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]
    prev_c5 = float('inf')
    patience_counter = 0
    patience_thresh = 3
    
    for b in b_vals:
        if time.time() - start_time > budget_s - 30:
            break
        def obj_func(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.log(np.sum(np.exp(b * dx_fine * corr))) / b
        
        def grad_func(h):
            return smooth_max_gradient(h, b, n_fine)
        
        for i in range(50):
            if time.time() - start_time > budget_s - 10:
                break
            
            # Standard gradient step
            grad = grad_func(h_fine)
            h_new = h_fine - 0.1 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = water_fill(h_new, n_fine / 2)
            h_new = (h_new + h_new[::-1]) / 2
            h_new = water_fill(h_new, n_fine / 2)
            
            h_fine = h_new
            c5_new = c5_of_h(h_fine)
            if c5_new < prev_c5:
                prev_c5 = c5_new
                patience_counter = 0
            else:
                patience_counter += 1
            if patience_counter >= patience_thresh:
                break
    
    print(f"Fine-grid refinement time used: {time.time() - time_check:.2f}s")
    
    # Final polishing
    print("Final polishing...")
    original_h = h_fine.copy()
    original_c5 = c5_of_h(h_fine)
    best_h = h_fine.copy()
    best_c5 = original_c5
    
    eps = 0.004
    time_left = budget_s - (time.time() - start_time)
    time_per_iter = 0.01
    
    n_iter_start = int((time_left - 50) / time_per_iter)
    n_iter_start = max(1, n_iter_start)
    
    for _ in range(n_iter_start):
        if time.time() - start_time > budget_s - 1:
            break
        # Choose indices
        p = min(6, n_fine // 20)
        indices = np.random.choice(n_fine, size=p, replace=False)
        h_trial = h_fine.copy()
        # Perturb
        for idx in indices:
            delta = np.random.uniform(-eps, eps)
            h_trial[idx] += delta
        # Clip
        h_trial = np.clip(h_trial, 0, 1)
        # Re-scale
        h_trial = water_fill(h_trial, n_fine / 2)
        h_trial = (h_trial + h_trial[::-1]) / 2
        h_trial = water_fill(h_trial, n_fine / 2)
        
        trial_c5 = c5_of_h(h_trial)
        if trial_c5 < best_c5:
            best_h = h_trial.copy()
            best_c5 = trial_c5
        h_fine = h_trial
    
    print(f"Polishing time: {time.time() - time.time():.2f}s")
    return (best_h, best_c5, n_fine)
