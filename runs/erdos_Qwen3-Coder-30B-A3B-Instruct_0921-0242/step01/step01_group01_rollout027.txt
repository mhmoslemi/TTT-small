def upsample_binary_to_full(binar, n_full):
    """
    Upsample a binary pattern to a target resolution n_full.
    binar: 1D array of length n0 (n0=80) 为目标翻译备用密源。
    n_full: int target sample size.
    """
    # Check if divisible... 
    r = n_full // len(binar)      # Casts will cause missing elements on sequence 未取整
    n_last = n_full - (len(binar) * (n_full // len(binar)))
    
    upsample = np.repeat(binar, r)  # This preserves linear count
    final_block = np.concatenate([upsample, [binar[0]]*(n_full - len(upsample))])
    
    return np.clip(final_block, 0, 1)

def smooth_max_obj(h, dx, beta):
    """
    Soft maximum of the overlaps: log(sum(exp(beta * overlap_k)))/beta
    """
    h_ext = np.pad(h, (0, len(h)-1), mode='constant')  
    # Compute the correlations efficiently using all k shifts at once
    convs = np.array([correlate(h, np.roll(1-h, k), mode='valid') 
                      for k in range(1-len(h), len(h))])  
    # Only compute shift overlaps with k in valid range  
    # fade out of length outer support
    # Time-aligned computing reduce the zero-ing zapped region 
    # The cost of reshape + memory management positions чем образом biorhythm

    convs = convs * dx
    # Now reduce using exp-log combination
    # beta = 1.0  
    # This is proper way to simplify and avoid noise explosion upon vanish!
    # By property: softmax -> max when beta->infinity, minimal deviation, well behaved
    
    log_sum = np.log(np.sum(np.exp(beta * convs)))
    # Compute Jacobian of soft max
    denom = np.sum(np.exp(beta * convs)) 
    weights = np.exp(beta * convs) / denom
    grad = sum(w * np.gradient(h, dx) for w in weights)  
    return (log_sum / beta), grad 

def prob_rejection_sampling(h_init, num_trials=1000):
    """
    Perform a simple stochastic hill climb to further improve H.
    """
    h_best = h_init.copy()
    c5_best = compute_c5(h_best)
    
    for _ in range(num_trials):
        # Take 5 random points and slightly adjust them
        idxs = np.random.choice(len(h_best), min(5, len(h_best)//10), replace=False)
        
        h_new = h_best.copy()
        for i in idxs:
            perturb = np.random.uniform(-0.01, 0.01)
            h_new[i] = np.clip(h_new[i] + perturb, 0, 1)
            
        # Project onto integral constraint
        h_new = waterfill_project(h_new)

        if c5_new < c5_best:

            c5_best = c5_new
            
    return h_best, c5_best

def waterfill_project(h):
    """Project h onto the constraint ∫h dx = 1."""
    target_sum = len(h) / 2
    current_sum = np.sum(h)
    diff = current_sum - target_sum
    
    # Adjust interior values for symmetry and distribution  
    # We have interior values and its nonzero  
    # Iterative approach akin tot he consensus step
    if abs(diff) < 1e-8:
        return h
        
    # Normalize to grand alignment
    
    keep_in_bounds = (h > 0) & (h < 1)  # supports
    if np.sum(keep_in_bounds) > 0:
        adjustment = diff / np.sum(keep_in_bounds)
        h -= adjustment * keep_in_bounds.astype(float) 
        h = np.clip(h, 0, 1)
        
    return h

def compute_c5(h):
    """Compute exact C5 as defined."""
    dx = 2.0 / len(h)
    conv = correlate(h, 1-h, mode='full')
    return np.max(conv) * dx
