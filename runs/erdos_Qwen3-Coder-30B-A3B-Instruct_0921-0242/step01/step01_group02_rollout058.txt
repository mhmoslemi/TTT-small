def evaluate_overlap(h):
    """Evaluate the overlap integral C5 for given h."""
    dx = 2.0 / len(h)
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def max_corr_of_h(h):
    """Compute max correlation term."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    return np.max(corr)

def shift_matrix(n, k):
    """Create n x n shift matrix S_k for a cyclic shift by k."""
    S = np.zeros((n,n))
    for i in range(n):
        j = (i + k) % n
        if 0 <= j < n:
            S[i, j] = 1
    return S

def project_to_constraints(h, target_sum):
    """Project h onto sum constraint [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    weights = np.ones_like(h)
    weights[np.where(h <= 0)] = 0
    weights[np.where(h >= 1)] = 0
    if np.sum(weights) > 0:
        h += diff * weights / np.sum(weights)
        h = np.clip(h, 0, 1)
    return h

def round_SDP_solution(X, m, n):
    """Round the SDP solution using Gaussian sampling method."""
    # Perform eigenvalue decomposition to construct V such that VV^T = X.
    try:
        # Use eigenvalue decomposition
        evals, evecs = eigh(X)
        # Use only positive eigenvalues, and instead of numerical zero,
        # treat eigenvalues near zero as zero.
        evals = np.maximum(evals, 0)
        # drop small negative values
        small = np.where(evals < 1e-8)[0]
        if len(small) > 0:
            evals[small] = 0
        V = evecs @ np.diag(np.sqrt(evals))
    except np.linalg.LinAlgError:
        # Fallback to SVD if eigenvectors fail
        U, s, Vt = np.linalg.svd(X)
        s = np.maximum(s, 0)
        V = U @ np.diag(np.sqrt(s))
    
    best_h = None
    best_c5 = float('inf')
    
    # Try multiple rounds
    for _ in range(200):
        g = np.random.randn(m)
        y = V.T @ g
        # Convert to binary
        x = np.sign(y)
        # Balance sign vector to have zero sum
        s = np.sum(x)
        if abs(s) > 1e-8:
            # Ties are broken by index if any, but we simply make enough flips
            # Find indices with smallest |y_i|
            abs_y = np.abs(y)
            idxs = np.argsort(abs_y)
            # Flip sign of |s|/2 least absolute y values to make sum zero
            to_flip = int(abs(s) / 2)
            if to_flip > 0:
                for i in range(to_flip):
                    idx = idxs[i]
                    x[idx] = -x[idx]
        
        # Create full length vector with symmetric property
        h_full = np.zeros(n)
        # only the first half
        h_full[:m] = (x + 1) / 2
        h_full[m:] = h_full[:m][::-1]
        
        # Ensure sum is exactly n/2
        h_full *= (n / 2) / np.sum(h_full)
        # Project to box constraints
        h_full = project_to_constraints(h_full, n / 2)
        
        c5 = evaluate_overlap(h_full)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_full.copy()
    return best_h, best_c5

def solve_grid(n, m, time_budget, start_time, total_time):
    """Solve SDP on a single grid of size n."""
    if time.time() - start_time > time_budget - 10:
        return None, None, None
    remaining_time = time_budget - total_time

    # Build shift matrices S_k for k in range(-floor(0.9 * n), floor(0.9 * n) + 1)
    k_range = int(0.9 * n) + 1
    S_list = {}
    for k in range(-k_range, k_range + 1):
        S_list[k] = shift_matrix(n, k)

    # SDP
    X = Semidef(m)
    t = Variable()

    # Constraints
    constraints = [X[i, i] == 1 for i in range(m)]
    # Zero sum condition (sum x_i = 0)
    # tr(1*1^T * X) = sum_i sum_j 1_ij * X_ij = sum_i X_ii = m
    # No, the zero sum is more subtle: the sum of entries in `1 * x = 0`
    # Remind that x = [x_0, ..., x_{n-1}] (length n)
    # With X = x x^T, we have:
    # sum_i x_i = 0  means 1^T x = 0 => 1^T X 1 = 0  
    # We pose: tr(1 * 1^T * X) = 0
    # But we only know M = V * V^T = X for m even.  
    # The submatrix built from X_{i,j} with i,j < m (after symmetry reduction).
    # The real sum constraint x_i = 2 * h_i - 1, sum h_i = n/2 => sum x_i = 0.
    one = np.ones(m)
    constraints.append(1.0 * trace(one @ one.T * X) == 0)
    
    # Add constraints for each S_k: t >= n/4 - 0.25 * tr(S_k * X)
    # If X was full-size, we would have all n constraints. But just if using m elements
    # and using symmetry*, the make sure to handle all s_k such that tr(S_k * X) non-zero on the (m, m) block.
    # Instead, construct a full set of shift matrices and filter those contributing.
    for k, S in S_list.items():
        # Project to the reduced matrix
        try:
            S_reduced = S[:m, :m]
            constraints.append(t >= n/4 - 0.25 * trace(S_reduced @ X))
        except Exception:
            pass  # Skip bad shift
        
    # Objective
    objective = Minimize(t)

    # Try to solve SDP
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=SCS, eps=1e-6, max_iters=5000)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            if not np.isnan(t.value) and not np.isinf(t.value):
                t_star = t.value
                X_opt = X.value
                h_best, c5_best = round_SDP_solution(X_opt, m, n)
                return h_best, c5_best, n
    except Exception as e:
        pass
    return None, None, None

def local_search(h, n_points):
    """Perform local swapping to refine a solution."""
    h_ref = h.copy()
    c = correlate(h_ref, 1 - h_ref, mode='full')
    cmax = np.max(c)
    max_swaps = 3000
    for _ in range(max_swaps):
        # Pick a 1 and a 0 to swap
        ones = np.where(h_ref > 0.5)[0]
        zeros = np.where(h_ref < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        # Swap them
        h_new = h_ref.copy()
        h_new[i1] = 0
        h_new[i0] = 1
        # Recompute the correlation efficiently
        # Get indices affected
        affected = []
        if i1 > 0: affected.append((i1-1, 1))
        if i1 < n_points - 1: affected.append((i1+1, 1))
        if i0 > 0: affected.append((i0-1, 1))
        if i0 < n_points - 1: affected.append((i0+1, 1))
        # For simplicity avoid fast update; much harder to compute efficiently
        # So, we'll just recompute the total
        c_new = correlate(h_new, 1 - h_new, mode='full')
        cmax_new = np.max(c_new)
        if cmax_new < cmax:
            h_ref = h_new
            cmax = cmax_new
            c = c_new
    return h_ref, cmax
