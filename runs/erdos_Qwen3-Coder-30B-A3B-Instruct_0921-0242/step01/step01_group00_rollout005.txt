import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import cvxpy as cp
from cvxpy import ECOS
from scipy.fft import fft, ifft

def solve_spectral_lp(n):
    """Solve the spectral LP relaxation to maximize the minimum autocorrelation bound."""
    # Variables: power spectrum p_m = |h_m|^2
    p = cp.Variable(n)
    t = cp.Variable()

    # Constraints
    constraints = []
    # Autocorrelation constraints: sum p_m * cos(2pi m k / n) >= t for all k != 0
    for k in range(1, n):
        # We can compute the DFT of the cosine terms using the fact that 
        # sum_m p_m cos(2pi m k / n) = 0.5 * (sum_m p_m exp(2pi i m k / n) + sum_m p_m exp(-2pi i m k / n))
        # = 0.5 * (F{p}_k + F{p}_{-k})
        # Here we just compute it directly using cosines
        cos_term = np.array([np.cos(2 * np.pi * m * k / n) for m in range(n)])
        constraints.append(cp.sum(cp.multiply(p, cos_term)) >= t)

    # Sum constraint: sum p_m = n * n/2 = n^2/2
    constraints.append(cp.sum(p) == n * n / 2)
    
    # Bounds: 0 <= p_m <= n
    constraints.extend([p >= 0, p <= n])

    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return p.value, t.value
        else:
            # Fallback: treat as infeasible
            return None, None
    except Exception:
        return None, None

def gerchberg_saxton(p_spectrum, n, max_iter=200):
    """
    Perform Gerchberg-Saxton phase retrieval to construct a real-valued vector
    from the given power spectrum.
    """
    # Generate a random starting phase  
    phi = np.random.uniform(0, 2*np.pi, n)
    
    # Start with complex spectrum
    Fx = np.sqrt(p_spectrum) * np.exp(1j * phi)
    
    # Iteration loop
    for _ in range(max_iter):
        # Inverse FFT to time domain
        x = np.real(ifft(Fx))
        
        # Amplitude projection (clip to [0,1]) 
        y = np.clip(x, 0, 1)
        
        # Water filling to enforce exact sum = n/2
        current_sum = np.sum(y)
        diff = n/2 - current_sum
        # Add uniform constant to adjust sum
        if abs(diff) > 1e-10:
            # Simple iterative search for lambda
            # We can compute this more carefully but for now just use a small change
            # A better implementation would use bisection, but approximation is fine
            y += diff / n
            y = np.clip(y, 0, 1)

        # Forward FFT
        Fy = fft(y)
        
        # Spectral projection: keep magnitudes of p_spectrum, keep phases from Fy
        # Handle zero-magnitude case carefully
        Fx = np.sqrt(p_spectrum) * (Fy / (np.abs(Fy) + 1e-15))
    
    return np.real(ifft(Fx))

def project_simplex_box(h, n):
    """Project h onto the set satisfying: sum(h) = n/2, 0 <= h[i] <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum

    if abs(diff) < 1e-10:
        return h
    
    # Add diff/n to each entry
    h += diff / n
    h = np.clip(h, 0, 1)
    
    return h

def binary_sharpening(h, n, mu_vals=[1e-4, 1e-3, 1e-2]):
    """
    Perform binary sharpening with penalty.
    """
    # Use a simple gradient descent approach
    for mu in mu_vals:
        # Define penalty: psi(h) = mu * sum(h_i * (1 - h_i))
        def penalty_grad(h):
            # d/dh_i (mu * h_i * (1 - h_i)) = mu * (1 - 2*h_i)
            return mu * (1 - 2 * h)
        
        # Perform gradient descent steps
        for _ in range(20):  # Fixed number of iterations
            grad = penalty_grad(h)
            h_new = h - 0.01 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = project_simplex_box(h_new, n)
            
            # If h_new is better or close enough, update
            if np.sum(h_new) <= n/2 + 1e-10:
                h = h_new
            else:
                break
    
    # Final rounding to binary
    h_rounded = np.zeros_like(h)
    for i in range(n):
        if h[i] >= 0.99:
            h_rounded[i] = 1.0
        elif h[i] <= 0.01:
            h_rounded[i] = 0.0
        else:
            h_rounded[i] = h[i]
    
    # Adjust sum to exactly n/2
    h_rounded = project_simplex_box(h_rounded, n)
    return h_rounded

def gradient_worst_shift(h, n, dx):
    """Conduct a gradient descent for the worst shift."""
    # Compute full correlations
    corr = correlate(h, 1-h, mode='full')
    
    # Find worst shift (highest correlation)
    max_idx = np.argmax(corr)
    k_star = max_idx - (n - 1)  # Convert full index to shift index
    
    # Compute gradient of the worst shift
    grad = np.zeros(n)
    # For each i:
    # dI_k*/dh_i = dx * [ (1 - h_{i+k*}) - h_{i-k*} ]
    for i in range(n):
        i_plus = (i + k_star) % n
        i_minus = (i - k_star) % n
        grad[i] = dx * ((1 - h[i_plus]) - h[i_minus])
    
    # Gradient descent step
    alpha = 1.0
    max_steps = 50
    for _ in range(max_steps):
        h_new = h - alpha * grad
        # Project h_new to ensure [0,1] bounds and sum = n/2
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        diff = n/2 - current_sum
        h_new += diff / n
        h_new = np.clip(h_new, 0, 1)
        h_new = project_simplex_box(h_new, n)
        
        # Evaluate new value
        new_corr = correlate(h_new, 1-h_new, mode='full')
        new_max = np.max(new_corr) * dx
        
        if new_max < np.max(corr) * dx or alpha < 1e-8:
            if new_max < np.max(corr) * dx:
                h = h_new
            break
        else:
            alpha /= 2.0

    return h

def swap_anneal(h, n, dx, max_time=None):
    """Perform simulated annealing-like swaps to improve solution."""
    T = 0.001
    alpha = 0.99
    max_iter = 10000
    start_time = time.time()
    
    h_best = h.copy()
    c5_cur = np.max(correlate(h, 1-h, mode='full')) * dx
    c5_best = c5_cur
    
    for i in range(max_iter):
        # Break early if time is running out
        if max_time and time.time() - start_time > max_time - 0.5:
            break
            
        # Choose two indices with non-zero values
        indices = np.where((h > 0) & (h < 1))[0]
        if len(indices) < 2:
            break
            
        i1, i2 = np.random.choice(indices, 2, replace=False)
        delta = np.random.uniform(0, min(h[i1], 1 - h[i2], 0.02))
        
        h_new = h.copy()
        h_new[i1] -= delta
        h_new[i2] += delta
        
        # Project back to feasible
        h_new = project_simplex_box(h_new, n)
        
        # Evaluate
        corr = correlate(h_new, 1-h_new, mode='full')
        c5_new = np.max(corr) * dx
        
        # Accept or reject
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
        else:
            # Accept with probability based on temperature
            if np.random.uniform() < np.exp(-(c5_new - c5_cur) / T):
                h = h_new
                c5_cur = c5_new
        # Cooling
        T *= alpha
        
    return h_best, c5_best

def evaluate_c5(h):
    """Compute C5 value for the step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 650  # Target granularity
    dx = 2.0 / n_points
    
    # Stage 1: Spectral LP relaxation
    p_opt, t_opt = solve_spectral_lp(n_points)
    if p_opt is None or t_opt is None:
        # Fallback to simpler approach
        h_best = np.ones(n_points) * 0.5
        c5_bounds = evaluate_c5(h_best)
        return (h_best, c5_bounds, n_points)
    
    # Stage 2: Gerchberg-Saxton phase retrieval
    h_est = gerchberg_saxton(p_opt, n_points)
    h_gs = project_simplex_box(h_est, n_points)
    
    # Stage 3: Binary sharpening and projection
    h_after_sharpen = binary_sharpening(h_gs, n_points)
    h_final = h_after_sharpen
    
    # Stage 4: Worst shift refinement
    h_after_grad = gradient_worst_shift(h_final, n_points, dx)
    h_grad = h_after_grad
    
    # Stage 5: Swap annealing refinement
    c5_prev = evaluate_c5(h_grad)
    total_time = time.time() - start_time
    remaining_time = budget_s - total_time
    
    if remaining_time > 2:  # Leave room for final steps
        h_swap, c5_swap = swap_anneal(h_grad, n_points, dx, max_time=remaining_time)
        if c5_swap < c5_prev:
            h_final = h_swap
            c5_prev = c5_swap
    
    # Final evaluation of best construction
    c5_result = evaluate_c5(h_final)
    
    return (h_final, c5_result, n_points)
