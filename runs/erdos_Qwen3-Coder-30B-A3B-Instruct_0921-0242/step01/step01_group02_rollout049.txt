import numpy as np
import time
from scipy.signal import correlate
from scipy.linalg import eigh
from cvxpy import *
import cvxpy as cp
from cvxpy.reductions.solvers import choose_solver

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Grid sizes for multi-scale approach
    grids = [
        (140, 70),   # coarse
        (200, 100),  # medium
        (260, 130)   # fine
    ]
    
    # Target bound
    target_bound = 0.3807
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def compute_c5(h, dx):
        """Compute C5 using full correlation."""
        g = 1 - h
        corr = correlate(h, g, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    def project_to_constraints(h, n_points, target_sum):
        """Project h onto sum constraint and box constraints."""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-10:
            return h
        diff = target_sum - current_sum
        if abs(diff) < 1e-10:
            return h
        weights = np.ones_like(h)
        weights[np.where(h <= 0)] = 0
        weights[np.where(h >= 1)] = 0
        if np.sum(weights) > 0:
            h += diff * weights / np.sum(weights)
            h = np.clip(h, 0, 1)
        return h

    def get_shift_matrix(n, k):
        """Construct the shift matrix S_k."""
        S = np.zeros((n, n))
        if k >= 0:
            S[np.arange(n-k), np.arange(k, n)] = 1
        else:
            S[np.arange(-k, n), np.arange(n+k)] = 1
        return S

    def solve_sdp(n, m, k_vals, dx):
        """Solve the SDP relaxation."""
        # All the shift matrices S_k
        S_k_list = [get_shift_matrix(m, k) for k in k_vals]
        
        # Define CVXPY problem
        X = Semidef(m)
        t = Variable()
        ones_m = np.ones(m)
        
        # Constraints
        constraints = [
            X >> 0,                    # PSD
            [X[i, i] == 1 for i in range(m)],  # diagonal ones
            trace(ones_m @ ones_m.T @ X) == 0,  # zero sum relaxation
        ]
        
        # Add the constraints on t
        for S_k in S_k_list:
            constraints.append(t >= n/4 - 0.25 * trace(S_k @ X))
        
        # Objective
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        
        # Solve with warm start
        try:
            prob.solve(solver=SCS, eps=1e-6, max_it=5000, verbose=False)
            if prob.status == 'optimal':
                return X.value, t.value, True
        except Exception:
            pass
        return None, None, False

    def round_sdp(X_opt, m):
        """Rounding using Gaussian sampling."""
        if X_opt is None:
            return None
            
        # Get the eigenvector decomposition
        evals, evecs = eigh(X_opt)
        V = evecs @ np.diag(np.sqrt(np.maximum(evals, 0)))
        
        best_x = None
        best_c5 = float('inf')
        
        for _ in range(200):
            g = np.random.randn(m)
            y = V.T @ g
            x = np.sign(y)
            
            # Ensure zero sum by flipping smallest values
            s = np.sum(x)
            if abs(s) > 1e-6:
                idxs = np.argsort(np.abs(y))
                if s > 0:
                    # Flip from +1 to -1
                    flip_indices = idxs[:int(s//2)]
                    x[flip_indices] = -x[flip_indices]
                else:
                    # Flip from -1 to +1
                    flip_indices = idxs[:int(-s//2)]
                    x[flip_indices] = -x[flip_indices]
            
            # Convert to h values
            h_full = (x + 1) / 2
            h_full = h_full.T
            # Mirror to get full vector
            h_full = np.concatenate((h_full, h_full[::-1]))
            
            # Ensure the right sum
            h_full = project_to_constraints(h_full, 2*m, m)
            c5 = compute_c5(h_full, 2.0/(2*m))
            
            if c5 < best_c5:
                best_c5 = c5
                best_x = h_full
        
        return best_x, best_c5

    def local_search(h_in, dx, budget_time):
        """Improve the solution via swaps."""
        h = h_in.copy()
        corr = correlate(h, 1 - h, mode='full')
        cmax = np.max(corr)
        
        # Get the indices with 1s and 0s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            return h, cmax
        
        remaining_time = budget_time - (time.time() - start_time)
        if remaining_time < 5:
            return h, cmax
        
        num_swaps = min(2000, int(remaining_time * 10))
        
        for _ in range(num_swaps):
            if time.time() - start_time > budget_time - 0.5:
                break
                
            idx1 = np.random.choice(ones)
            idx0 = np.random.choice(zeros)
            
            h_new = h.copy()
            h_new[idx1] = 0
            h_new[idx0] = 1
            
            # Recompute convolution efficiently
            corr_new = corr.copy()
            # Subtract old values of old bits
            for k in range(len(h)):
                idx_k = (idx1 - k) % len(h)
                if 0 <= idx_k < len(h):
                    corr_new[k] -= h[idx1] * (1 - h[idx_k])
                idx_k = (idx0 - k) % len(h)
                if 0 <= idx_k < len(h):
                    corr_new[k] -= h[idx0] * (1 - h[idx_k])
                    
            # Add new values of new bits
            for k in range(len(h)):
                idx_k = (idx1 - k) % len(h)
                if 0 <= idx_k < len(h):
                    corr_new[k] += h_new[idx1] * (1 - h_new[idx_k])
                idx_k = (idx0 - k) % len(h)
                if 0 <= idx_k < len(h):
                    corr_new[k] += h_new[idx0] * (1 - h_new[idx_k])
            
            cmax_new = np.max(corr_new)
            
            if cmax_new < cmax:
                h = h_new
                corr = corr_new
                cmax = cmax_new
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        
        return h, cmax

    # Loop through the grid sizes
    for n, m in grids:
        if time.time() - start_time > budget_s - 2:
            break
            
        # Set up shifts to consider
        k_max = int(0.9 * n)
        k_vals = list(range(-k_max, k_max + 1))
        if len(k_vals) > 200:
            k_vals = list(range(int(-0.6 * n), int(0.6 * n) + 1))
        
        dx = 2.0 / n
        
        # Solve SDP
        try:
            X_opt, t_opt, success = solve_sdp(n, m, k_vals, dx)
            if not success:
                continue
        except:
            continue
            
        # Get the rounded solution
        try:
            h_round, c5_round = round_sdp(X_opt, m)
            if h_round is not None:
                # Run local refinement
                h_refined, c5_final = local_search(h_round, dx, start_time + budget_s)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_refined
                    best_n = n
                if c5_final <= target_bound:
                    return (h_refined, c5_final, n)
        except Exception:
            pass

    # Fallback if no good solution found yet
    if best_h is None:
        # Use the initial construction
        h_orig = initial_h_values
        n_orig = len(h_orig)
        dx_orig = 2.0 / n_orig
        h_interp = np.interp(np.linspace(0, 2, 300), np.linspace(0, 2, n_orig), h_orig)
        h_interp = h_interp * (150 / np.sum(h_interp))  # scale to sum 150
        best_c5 = compute_c5(h_interp, dx_orig)
        best_h = h_interp
        best_n = 300

    return (best_h, best_c5, best_n)
