import numpy as np
import time
from scipy.signal import correlate
from numpy.fft import fft, ifft
import math

def get_worst_shift(h):
    """Find the shift index that gives the minimum correlation."""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation using FFT (zero-padded)
    hfft = fft(np.concatenate([h, [0]*(n-1)]))
    hconj = np.conj(hfft)
    onefft = fft(np.concatenate([1-h, [0]*(n-1)]))
    corr = ifft(hfft * onefft)
    corr = np.real(corr)
    # Remove the zero shift (index 0) to find minimum
    corr = corr[1:n]  # Skip index 0 (the zero shift)
    k = np.argmax(corr)
    return k + 1  # Convert to 1-based indexing

def project_to_feasible(h, n):
    """Project h onto the feasible set: sum(h) = n/2 and h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling method
    # Binary search for the correct λ
    if diff > 0:  # Need to increase h
        # Find minimum value to raise (must remain ≤ 1)
        h_sorted = np.sort(h)
        h_sorted = h_sorted[::-1]  # descending
        cumsum = np.cumsum(h_sorted)
        n_exceeds = 0
        for i, val in enumerate(h_sorted):
            new_sum = cumsum[i] + (n - i - 1) * (val + diff / (n - i - 1))
            if new_sum <= n/2:
                n_exceeds = i + 1
            else:
                break
        if n_exceeds == 0:
            # All can be increased evenly
            h += diff / n
        elif n_exceeds == n:
            # All can be increased
            h += diff / n
        else:
            # Some will hit 1, others increase
            lambda_val = diff / (n - n_exceeds)
            for i in range(n_exceeds):
                h[i] += lambda_val
            # For remaining ones, we just spread the remainder
            extra = diff - n_exceeds * lambda_val
            h[n_exceeds:] += extra / (n - n_exceeds)
    else:
        # Need to reduce h
        diff = -diff
        h_sorted = np.sort(h)
        h_sorted = h_sorted  # ascending
        cumsum = np.cumsum(h_sorted)
        n_under = 0
        for i, val in enumerate(h_sorted):
            new_sum = cumsum[i] + (n - i - 1) * (val - diff / (n - i - 1))
            if new_sum >= n/2:
                n_under = i + 1
            else:
                break
        if n_under == 0:
            # All can be decreased evenly
            h -= diff / n
        elif n_under == n:
            # All must be decreased
            h -= diff / n
        else:
            # Some will go to 0, others decrease
            lambda_val = diff / (n - n_under)
            for i in range(n_under):
                h[i] -= lambda_val
            extra = diff - n_under * lambda_val
            h[n_under:] -= extra / (n - n_under)
    
    h = np.clip(h, 0, 1)
    # Final correction
    h = h * (n / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    return h

def modify_gradient(h, k):
    """Compute gradient at h with respect to shift k."""
    n = len(h)
    grad = np.zeros_like(h)
    dx = 2.0 / n
    
    # Gradient of correlation term:
    # d/dh_i [h_i * h_{i+k}] = h_{i+k} + h_{i-k} (with cycles)
    # But: C5 = dx(max(A_k) = dx * (1 - min(A_k))
    # We only pick the worst k (using np.argmax) in compute_C5
    
    # Also, for min(A_k) over all k != 0, we compute gradient of one term:
    # For A_k = sum_i h_i * h_{i+k}
    # Gradient at h_i = h_{i+k} + h_{i-k}
    # So: 
    for i in range(n):
        left_ind = (i - k) % n
        right_ind = (i + k) % n
        grad[i] = dx * (h[left_ind] + h[right_ind])
    
    return grad

def compute_C5(h):
    """Evaluate the C5 value of the given h."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation between h and (1-h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 300
    dx = 2.0 / n_points
    
    # Initialize swarm
    S = 30  # swarm size
    particles = []
    velocities = []
    
    # Initial particles
    for _ in range(S):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros_like(h))
    
    # Current best
    best_global = None
    best_C5 = float('inf')
    
    omega = 0.7
    c1 = c2 = 1.5
    
    last_improvement = time.time()
    
    gen = 0
    while time.time() - start_time < budget_s - 2.0:
        gen += 1
        if gen % 5 == 0:  # memetic refinement at every 5 iterations
            for i in range(S):
                h = particles[i]
                k = get_worst_shift(h)
                delta = modify_gradient(h, k)
                
                # backtracking line search
                step_size = 1.0
                h_t = h.copy()
                
                max_iters = 10
                for _ in range(max_iters):
                    h_trial = h_t - step_size * delta
                    h_trial = project_to_feasible(h_trial, n_points)
                    C5_trial = compute_C5(h_trial)
                    if C5_trial < compute_C5(h):
                        h = h_trial
                        particles[i] = h.copy()
                        break
                    step_size /= 2.0
                if C5_trial < compute_C5(h):
                    particles[i] = h.copy()
        
        # Update particles
        for i in range(S):
            h = particles[i]
            v = velocities[i]
            
            # Velocity update
            r1, r2 = np.random.rand(2)
            p_best = particles[i]
            g_best = best_global
            
            if g_best is not None:
                v = omega * v + c1 * r1 * (p_best - h) + c2 * r2 * (g_best - h)
                velocities[i] = v
            
            # Position update
            h_new = h + v
            
            # Apply constraints
            h_new = project_to_feasible(h_new, n_points)
            particles[i] = h_new
            
            # Update best
            C5 = compute_C5(h_new)
            if C5 < best_C5:
                best_C5 = C5
                best_global = h_new.copy()
                last_improvement = time.time()
                
        # Termination condition for global phase
        if time.time() - last_improvement > 2.0:
            # Inject fresh particles
            inject_count = max(1, S // 4)
            for _ in range(inject_count):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                h = project_to_feasible(h, n_points)
                # Replace the oldest particle
                particles.append(h.copy())
                velocities.append(np.zeros_like(h))
                particles.pop(0)
                velocities.pop(0)
    
    # Late-stage local refinement: random swaps
    if best_global is not None:
        best_h = best_global.copy()
        c5_best = compute_C5(best_h)
        n = len(best_h)
        
        # Random swap hill-climbing
        for _ in range(5000):
            if time.time() - start_time > budget_s - 2.0:
                break
                
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            
            # Ensure i has a value > 0 and j < 1
            if best_h[i] <= 0 or best_h[j] >= 1:
                continue
            
            delta = np.random.uniform(0, min(best_h[i], 1 - best_h[j], 0.01))
            h_trial = best_h.copy()
            h_trial[i] -= delta
            h_trial[j] += delta
            
            # project
            h_trial = project_to_feasible(h_trial, n)
            c5_trial = compute_C5(h_trial)
            
            if c5_trial < c5_best:
                best_h = h_trial
                c5_best = c5_trial
    
    # Final evaluation
    if best_global is None:
        # Fallback to flat profile
        h_fallback = np.ones(n_points) * 0.5
        c5_best = compute_C5(h_fallback)
        best_h = h_fallback
    else:
        best_h = best_global
    
    return (best_h, c5_best, n_points)
