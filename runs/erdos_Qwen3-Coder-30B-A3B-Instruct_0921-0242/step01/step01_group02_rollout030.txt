import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import rfft, irfft
import math

def c5_fft(h):
    """Compute C5 using FFT for speed."""
    g = 1 - h
    H = rfft(h)
    G = rfft(g)
    c = irfft(H * G, 2 * len(h) - 1)
    return np.max(c)

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def repair_weight(h, target_sum):
    """Repair h to have exactly the target sum."""
    current_sum = np.sum(h)
    diff = int(target_sum - current_sum)
    if diff > 0:
        # Need to turn zeros into ones
        zero_indices = np.where(h == 0)[0]
        if len(zero_indices) >= diff:
            indices_to_flip = np.random.choice(zero_indices, diff, replace=False)
            h[indices_to_flip] = 1
    elif diff < 0:
        # Need to turn ones into zeros
        one_indices = np.where(h == 1)[0]
        if len(one_indices) >= -diff:
            indices_to_flip = np.random.choice(one_indices, -diff, replace=False)
            h[indices_to_flip] = 0
    return h

def crossover(p1, p2):
    """Two-point crossover."""
    n = len(p1)
    c1 = np.random.randint(1, n-1)
    c2 = np.random.randint(c1, n)
    child = np.concatenate([p1[:c1], p2[c1:c2], p1[c2:]])
    return repair_weight(child, n/2)

def mutate_swap(h):
    """Swap mutation."""
    one_indices = np.where(h == 1)[0]
    zero_indices = np.where(h == 0)[0]
    if len(one_indices) and len(zero_indices):
        i1 = np.random.choice(one_indices)
        i0 = np.random.choice(zero_indices)
        h[i1], h[i0] = h[i0], h[i1]
    return h

def mutate_block_swap(h):
    """Block swap mutation."""
    n = len(h)
    block_len = min(4, n//10)
    if block_len < 1:
        return h
    i1 = np.random.randint(0, n - block_len)
    i2 = np.random.randint(0, n - block_len)
    if i1 != i2:
        block = h[i1:i1+block_len].copy()
        h[i1:i1+block_len] = h[i2:i2+block_len]
        h[i2:i2+block_len] = block
    return repair_weight(h, n/2)

def hill_climb(h, max_iterations=500):
    """Hill climb to improve solution by swapping positions to reduce worst shift."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    current_h = h.copy()
    c = correlate(current_h, 1 - current_h, mode='full')
    best_c = np.max(c)
    
    for _ in range(max_iterations):
        # Find worst shift
        k_star = np.argmax(c) - n + 1
        if k_star == 0:
            break
            
        # Evaluate change in convolution due to swap
        best_delta = 0
        best_i1 = None
        best_i0 = None
        
        # For efficiency, sample a limited number of swaps
        for _ in range(200):
            i1 = np.random.randint(n)
            i0 = np.random.randint(n)
            if current_h[i1] == 1 and current_h[i0] == 0:
                delta = 0
                # Only compute the term that changes due to swap at these indices
                # For shift k*, contributions are at i1+k*, i0+k*
                j1 = i1 + k_star
                j0 = i0 + k_star
                if 0 <= j1 < n:
                    delta += dx * (current_h[j1] - (1 - current_h[j1]))
                if 0 <= j0 < n:
                    delta += dx * (current_h[j0] - (1 - current_h[j0]))
                delta *= -1  # because we're minimizing
            
            if delta < best_delta:
                best_delta = delta
                best_i1 = i1
                best_i0 = i0
                
        if best_i1 is not None:
            current_h[best_i1], current_h[best_i0] = current_h[best_i0], current_h[best_i1]
            c = correlate(current_h, 1 - current_h, mode='full')
            new_c = np.max(c)
            if new_c < best_c:
                best_c = new_c
                best_h = current_h.copy()
        else:
            break
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 280
    dx = 2.0 / n
    target_sum = n / 2

    # Initialize population
    pop_size = 200
    elite = 5
    diversity_threshold = 0.2
    min_diversity_generations = 10
    
    population = generate_initial_population(n, pop_size, target_sum)
    
    if not len(population):
        # Fallback: create a random balanced population
        population = []
        for i in range(pop_size):
            h = np.zeros(n)
            ones = np.random.choice(n, int(target_sum), replace=False)
            h[ones] = 1
            population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = c5_fft(h) * dx
        fitness.append(-c5)  # negative because lower C5 is better
        
    # Main evolutionary loop
    generation = 0
    time_remaining = budget_s - (time.time() - start_time) - 20  # Keep some buffer
    
    while time_remaining > 0 and generation < 10000:
        # Selection by binary tournament
        mating_pool = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.choice(pop_size, 2, replace=False)
            if fitness[idx1] > fitness[idx2]:
                mating_pool.append(population[idx1])
            else:
                mating_pool.append(population[idx2])
                
        # Generate offspring
        offspring = []
        for _ in range(pop_size - elite):
            p1, p2 = np.random.choice(mating_pool, 2, replace=False)
            child = crossover(p1, p2)
            
            # Apply mutation with probability 0.2
            if np.random.rand() < 0.2:
                child = mutate_swap(child)
                
            # Block swap mutation with small probability
            if np.random.rand() < 0.02:
                child = mutate_block_swap(child)
            
            # Repair to exact weight
            child = repair_weight(child, target_sum)
            offspring.append(child)
            
        # Form new population with elitism
        elite_indices = np.argsort(fitness)[-elite:]
        new_population = [population[i] for i in elite_indices] + offspring
        
        # Compute fitness for new population
        fitness = []
        for h in new_population:
            c5 = c5_fft(h) * dx
            fitness.append(-c5)
            
        # Update population
        population = new_population
        
        # Diversity check
        if generation % min_diversity_generations == 0:
            # Compute pairwise Hamming distances for the worst 50
            worst_indices = np.argsort(fitness)[:50]
            if len(worst_indices) > 10:
                dists = []
                for i in range(len(worst_indices)):
                    for j in range(i+1, len(worst_indices)):
                        idx1 = worst_indices[i]
                        idx2 = worst_indices[j]
                        dist = np.sum(population[idx1] != population[idx2])
                        dists.append(dist / n)
                avg_dist = np.mean(dists)
                if avg_dist < diversity_threshold:
                    for _ in range(len(worst_indices) // 2):
                        idx = np.random.choice(worst_indices)
                        h = np.zeros(n)
                        ones = np.random.choice(n, int(target_sum), replace=False)
                        h[ones] = 1
                        population[idx] = h
                        
        # Periodic hill climb
        if generation % 10 == 0:
            best_idx = np.argmax(fitness)
            h_best = population[best_idx]
            h_improved = hill_climb(h_best)
            # Update if better
            c5_improved = c5_fft(h_improved) * dx
            c5_current = c5_fft(h_best) * dx
            if c5_improved < c5_current:
                population[best_idx] = h_improved
                fitness[best_idx] = -c5_improved
                
        generation += 1
        time_remaining = budget_s - (time.time() - start_time) - 20
        
        # Early stop if target reached
        if -min(fitness) <= 0.3807:
            break

    # Final selection
    final_fitness = []
    for h in population:
        c5 = c5_fft(h) * dx
        final_fitness.append(c5)
    best_idx = np.argmin(final_fitness)
    best_h = population[best_idx]
    best_c5 = final_fitness[best_idx]
    
    # Run final hill climb and optimize on fine grid if time remains
    time_left = budget_s - (time.time() - start_time)
    if time_left > 30 and best_c5 > 0.3807:
        fine_h = hill_climb(best_h, max_iterations=1000)
        fine_c5 = c5_fft(fine_h) * dx
        if fine_c5 < best_c5:
            best_h = fine_h
            best_c5 = fine_c5
            
        # If still low time, try upsampling
        n_fine = 500
        dx_fine = 2.0 / n_fine
        if time_left > 20:
            # Upsample
            coarse_h = best_h
            scale = n_fine // n
            fine_h = np.repeat(coarse_h, scale)
            if len(fine_h) > n_fine:
                fine_h = fine_h[:n_fine]
            else:
                fine_h = np.pad(fine_h, (0, n_fine - len(fine_h)), 'constant')
                fine_h = fine_h[:n_fine]
            fine_h = repair_weight(fine_h, n_fine / 2)
            
            # Fine GA
            pop_fine = [fine_h]
            remaining_time = budget_s - (time.time() - start_time)
            max_generations = int(remaining_time / 0.5) if remaining_time > 5 else 5
            
            for gen in range(max_generations):
                if time.time() - start_time > budget_s - 5:
                    break
                # Select
                indices = np.random.choice(len(pop_fine), 50, replace=True)
                selected = [pop_fine[i] for i in indices]
                # Crossover and mutation
                offspring = []
                for _ in range(45):
                    p1, p2 = np.random.choice(selected, 2)
                    child = crossover(p1, p2)
                    if np.random.rand() < 0.2:
                        child = mutate_swap(child)
                    if np.random.rand() < 0.02:
                        child = mutate_block_swap(child)
                    child = repair_weight(child, n_fine / 2)
                    offspring.append(child)
                pop_fine += offspring
                # Evaluate all and keep best
                fitness_fine = []
                for h in pop_fine:
                    c5 = c5_fft(h) * dx_fine
                    fitness_fine.append(c5)
                best_fine_index = np.argmin(fitness_fine)
                pop_fine = [pop_fine[best_fine_index]]  # Keep only the best
                
            # Final hill climb on fine grid
            if time.time() - start_time < budget_s - 5:
                final_fine = hill_climb(pop_fine[0], max_iterations=2000)
                final_c5 = c5_fft(final_fine) * dx_fine
                if final_c5 < best_c5:
                    best_h = final_fine
                    best_c5 = final_c5

    return (best_h, best_c5, len(best_h))
