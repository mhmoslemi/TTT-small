import numpy as np

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    return np.max(corr)

def evaluate_binary_mask(b):
    """Evaluate C5 for a binary mask"""
    n = len(b)
    dx = 2.0 / n
    h = b.astype(float)
    return c5_of_h(h)

def get_correlation_incremental(b, idx1, idx2):
    """Compute the change in correlation when swapping b[idx1] and b[idx2]"""
    n = len(b)
    dx = 2.0 / n
    # Original correlation values (could be precomputed)
    h = b.astype(float)
    g = 1.0 - h
    
    # Compute the full correlation for current h
    corr = np.correlate(h, g, mode='full')
    # corr has length 2*n - 1, centered at index n - 1
    orig_max = np.max(corr)
    
    # Construct new binary mask:
    b_new = b.copy()
    b_new[idx1], b_new[idx2] = b_new[idx2], b_new[idx1]
    
    # Re-evaluate correlation to get the actual delta
    h_new = b_new.astype(float)
    g_new = 1.0 - h_new
    corr_new = np.correlate(h_new, g_new, mode='full')
    new_max = np.max(corr_new)
    
    return (orig_max - new_max), b_new

def simulated_annealing_binary(b, max_iter=30000):
    """Run simulated annealing on binary mask"""
    n = len(b)
    current_mask = b.copy()
    current_c5 = evaluate_binary_mask(current_mask)
    
    best_mask = current_mask.copy()
    best_c5 = current_c5
    
    temp = 0.02
    
    for i in range(max_iter):
        if i % 100 == 0:
            temp *= 0.995
        if temp < 1e-10:
            break
            
        # Find indices where current_mask is 0 or 1
        zero_indices = np.where(current_mask == 0)[0]
        one_indices = np.where(current_mask == 1)[0]
        
        if len(zero_indices) == 0 or len(one_indices) == 0:
            break
            
        idx1 = np.random.choice(one_indices)
        idx2 = np.random.choice(zero_indices)
        
        delta, new_mask = get_correlation_incremental(current_mask, idx1, idx2)
        
        if delta > 0 or np.random.rand() < np.exp(-delta / temp):
            current_mask = new_mask
            current_c5 = evaluate_binary_mask(current_mask)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_mask = current_mask.copy()
                
    return best_mask, best_c5

def refine_binary_mask_with_lp(b):
    """Refine binary mask h to a continuous h via linear programming"""
    n = len(b)
    dx = 2.0 / n
    
    # Variables for the continuous h
    h = cp.Variable(n)
    
    # Objective: minimize max overlap t
    t = cp.Variable()
    constraints = [0 <= h, h <= 1, cp.sum(h) == n / 2]
    
    # Compute overlap constraint for each shift k
    # We use: sum_i h_i * (1 - h_{i+k})
    for k in range(1-n, n):
        # Using numpy convention for indexing
        if k >= 0:
            left = h[k:]
            right = h[:n-k]
        else:
            left = h[:n+k]
            right = h[-k:]
        
        # Linearize: use b[i] to decide if we use h[i] or 0 for h_{i+k}
        if k >= 0:
            # h_{i+k} -> h[i] if b[i] = 1, otherwise 0
            expr = cp.sum(cp.multiply(h, 1 - b) + cp.multiply(h[k:], b[:n-k]))
        else:
            expr = cp.sum(cp.multiply(h, 1 - b) + cp.multiply(h[-k:], b[:n+k]))
        constraints.append(expr <= t)
        
    # Solve using ECOS
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if prob.status == cp.OPTIMAL:
        return h.value
    else:
        # Fallback to original binary
        return b.astype(float)

def project_to_feasible(h):
    """Ensure sum(h) = n/2 and h in [0,1]^n"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    # Project into feasible set with bounds [0,1] and sum = n/2
    # Simple 1D projection (delta distributed uniformly)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    # Distribute deficit
    adjustment = deficit / n
    h += adjustment
    h = np.clip(h, 0, 1)
    return h
