def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_full_correlation(h):
    """Returns the full correlation array for the given h."""
    g = 1 - h

def simulated_annealing_binary(n_points, target_sum, max_iter, temp_init=1.0, cooling_rate=0.995):
    """Run simulated annealing to find a good binary vector."""
    h = np.zeros(n_points, dtype=int)
    positions_ones = np.random.choice(n_points, size=int(target_sum), replace=False)
    h[positions_ones] = 1
    
    best_h = h.copy()
    best_c5 = c5_of_h(h, 2.0 / n_points)
    temp = temp_init
    
    for _ in range(max_iter):
        # Choose two random positions
        i1 = np.random.randint(0, n_points)
        i0 = np.random.randint(0, n_points)
        
        # Make sure i1 is 1 and i0 is 0
        if h[i1] == 1 and h[i0] == 0:
            # Swap them
            h[i1] = 0
            h[i0] = 1
            
            c5 = c5_of_h(h, 2.0 / n_points)
            delta = c5 - best_c5
            
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                best_c5 = c5
                best_h = h.copy()
            else:
                # Revert the swap
                h[i1] = 1
                h[i0] = 0
                
        temp *= cooling_rate
        if temp < 1e-10:
            break
            
    return best_h, best_c5
