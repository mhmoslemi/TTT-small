import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.sparse import diags
from scipy.linalg import eigh

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Grid sizes and corresponding dimensions
    grids = [
        (140, 70),  # coarse
        (200, 100), # medium
        (260, 130)  # fine
    ]
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    def create_shift_matrix(n, k):
        """Create shift matrix for shift k."""
        if k == 0:
            return diags([1] * n, 0, shape=(n, n), format='csr')
        elif k > 0:
            # Shift right: x[i] -> x[i+k]
            rows = np.arange(n - k)
            cols = np.arange(k, n)
            data = np.ones(len(rows))
            return diags(data, k, shape=(n, n), format='csr')
        else:
            # Shift left: x[i] -> x[i+k] = x[i-k]
            rows = np.arange(-k, n)
            cols = np.arange(n + k)
            data = np.ones(len(rows))
            return diags(data, k, shape=(n, n), format='csr')
    
    def solve_sdp(n, m, shift_indices):
        """Solve SDP relaxation to find lower bound on max correlation."""
        # Initialize shift matrices
        S = [create_shift_matrix(n, k) for k in shift_indices]
        
        # Variables
        X = cp.Variable((m, m), symmetric=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            X >> 0,  # X is positive semidefinite
            cp.diag(X) == 1,  # diagonal entries are 1
            cp.trace(X) == 0,  # trace constraint (equivalent to sum of x = 0)
            t >= n/4 - 0.25 * cp.trace(S[k] @ X) for k in range(len(shift_indices))
        ]
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.SCS, eps=1e-6, max_iters=5000, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return t.value, X.value
        except:
            pass
        return None, None
    
    def round_vector(X, m, n):
        """Round PSD matrix using Goemans-Williamson."""
        # Eigenvalue decomposition
        eigenvals, eigenvecs = eigh(X)
        # Keep only positive eigenvalues
        idx = eigenvals > 1e-8
        eigenvals = eigenvals[idx]
        eigenvecs = eigenvecs[:, idx]
        # Cholesky-like decomposition
        V = eigenvecs @ np.diag(np.sqrt(eigenvals))
        
        # Generate samples
        for sample in range(200):  # Sample pool
            g = np.random.randn(m)
            y = V.T @ g
            x = np.sign(y).astype(int)
            
            # Fix zero-sum constraint
            s = np.sum(x)
            if s != 0:
                if s > 0:
                    # Flip s/2 smallest entries from +1 to -1
                    flip_indices = np.argsort(y)[:s//2]
                    x[flip_indices] = -1
                else:
                    # Flip |s|/2 smallest entries from -1 to +1
                    flip_indices = np.argsort(-y)[:(-s)//2]
                    x[flip_indices] = 1
            
            # Construct full vector by symmetry
            h = np.zeros(n)
            h[:m] = (x + 1) / 2
            h[m:] = h[:m][::-1]
            
            # Normalize to sum n/2
            current_sum = np.sum(h)
            h *= (n/2) / current_sum
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = project_to_constraints(h, n, n/2)
            
            # Evaluate C5
            dx = 2 / n
            c5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n
                
        return best_c5, best_h, best_n

        """Project h onto sum constraint and box constraints."""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-10:
            return h
        diff = target_sum - current_sum
        if abs(diff) < 1e-10:
            return h
        weights = np.ones_like(h)
        weights[np.where(h <= 0)] = 0
        weights[np.where(h >= 1)] = 0
        if np.sum(weights) > 0:
            h += diff * weights / np.sum(weights)
            h = np.clip(h, 0, 1)
        return h

        dx = 2.0 / n

                break

            hnew = h.copy()
            hnew[i1] = 0
            hnew[i0] = 1

            # Remove contribution from old bits
            for k in range(len(corr)):
                if (i1 + k) < n: cnew[k] -= h[i1] * (1 - h[i1+k])
                if (i0 + k) < n: cnew[k] -= h[i0] * (1 - h[i0+k])
                if k > 0:
                    if (k + i1) < n: cnew[k] -= (1 - h[i1]) * h[k+i1]
                    if (k + i0) < n: cnew[k] -= (1 - h[i0]) * h[k+i0]
            # Add contribution from new bits
            for k in range(len(cnew)):
                if (i1 + k) < n: cnew[k] += hnew[i1] * (1 - hnew[i1+k])
                if (i0 + k) < n: cnew[k] += hnew[i0] * (1 - hnew[i0+k])
                if k > 0:
                    if (k + i1) < n: cnew[k] += (1 - hnew[i1]) * hnew[k+i1]
                    if (k + i0) < n: cnew[k] += (1 - hnew[i0]) * hnew[k+i0]
            
            new_max = np.max(cnew)
            if new_max < cmax:
                cmax = new_max
                h = hnew
                corr = cnew
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        
        return h, cmax

            break

        max_k = int(0.9 * n)
        shift_indices = list(range(-max_k, max_k + 1))
        
        # Solve SDP

            continue

            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined
                best_n = n_rounded

            break
