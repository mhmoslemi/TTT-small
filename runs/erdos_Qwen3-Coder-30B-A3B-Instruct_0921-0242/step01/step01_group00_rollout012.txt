import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling approach
    h_sorted = np.sort(h)[::-1]
    h_cumsum = np.cumsum(h_sorted)
    
    # Find the threshold
    threshold = 0.0
    for i in range(len(h_sorted)):
        if h_sorted[i] > threshold:
            if (i + 1) * threshold + h_cumsum[i] >= n / 2:
                threshold = (n / 2 - h_cumsum[i]) / (i + 1)
                break
            threshold = h_sorted[i]
    else:
        threshold = (n / 2 - h_cumsum[-1]) / len(h_sorted)
    
    # Adjust all elements
    h = np.maximum(np.minimum(h, 1.0), threshold)
    h = h + (n / 2 - np.sum(h)) / n
    
    # Final clip
    h = np.clip(h, 0.0, 1.0)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation of h with (1 - h) via FFT
    h_fft = fft(np.pad(h, (0, n-1), mode='constant'))
    h_conj_fft = fft(np.pad(1 - h, (0, n-1), mode='constant'), overwrite_x=True)
    corr = ifft(h_fft * h_conj_fft.conj()).real
    # The valid (non-zero) part is from index n-1 to 2n-2
    corr = corr[n-1:] 
    # Scale and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_worst_shift_index(h):
    """Computes the shift index that gives the minimum correlation."""
    n = len(h)
    dx = 2.0 / n
    h_fft = fft(np.pad(h, (0, n-1), mode='constant'))
    h_conj_fft = fft(np.pad(1 - h, (0, n-1), mode='constant'), overwrite_x=True)
    corr = ifft(h_fft * h_conj_fft.conj()).real
    corr = corr[n-1:]  # non-zero part
    max_corr = np.max(corr)
    worst_shift = np.argmax(corr)  # 0-based index, but we want the shift
    if worst_shift >= n:
        worst_shift -= n
    return worst_shift

def compute_gradient_for_shift(h, shift_idx):
    """Compute gradient for a specific shift idx."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    for i in range(n):
        left_idx = (i + shift_idx) % n
        right_idx = (i - shift_idx) % n
        grad[i] = dx * (1 - h[left_idx] - h[right_idx])
    return grad

def backtracking_line_search(h, grad, max_iter=10):
    """Perform backtracking line search."""
    alpha = 1.0
    dx = 2.0 / len(h)
    current_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        h_trial = h - alpha * grad
        h_trial = project_to_feasible(h_trial, len(h))
        trial_c5 = evaluate_c5(h_trial)
        if trial_c5 < current_c5:
            return h_trial, trial_c5
        alpha /= 2.0
    
    return h, current_c5
