def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def build_fft_helpers(n):

def project_to_feasible(Y, n):
    """Project matrix Y to satisfy: sum(h) = n/2, h in [0,1]^n"""
    # Compute h = ||Y_i||^2
    h = np.sum(Y**2, axis=1)
    
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    
    # Normalize to sum = n/2
    total = np.sum(h)
    if abs(total) > 1e-10:
        alpha = (n / 2.0) / total
        h *= alpha

    # Re-clip after normalization
    h = np.clip(h, 0, 1)
    
    # Update Y accordingly (transpose matrix to sanity check)
    for i in range(n):
        if h[i] > 0:
            Y[i] = Y[i] * np.sqrt(h[i]) / np.sqrt(np.sum(Y[i]**2))
        else:
            Y[i] = np.zeros(Y.shape[1])
    
    return Y

def compute_correlations(Y, fft_len):
    """Compute correlations c_k using FFT over columns of Y"""
    r = Y.shape[1]  # rank
    n = Y.shape[0]
    c = np.zeros(n)
    for d in range(r):
        y_d = Y[:, d]
        # Compute zero-padded FFT: length 2*n
        fft_y = fft(y_d, fft_len)
        # Compute convolution using FFT
        conv_fft = fft_y * np.conj(fft_y)
        conv_result = ifft(conv_fft).real[:n]
        c += conv_result
    return c

def gradient_phi_tau(Y, c, tau, fft_len):
    """Compute gradient of log-sum-exp surrogate"""
    # Compute weights
    exp_neg_c_tau = np.exp(-c / tau)
    sum_exp_neg_c_tau = np.sum(exp_neg_c_tau)
    p = exp_neg_c_tau / sum_exp_neg_c_tau
    
    # Initialize gradients per row
    grad = np.zeros_like(Y)
    
    r = Y.shape[1]
    n = Y.shape[0]
    
    for d in range(r):
        y_d = Y[:, d]
        # convolve p with y_d and reverse y_d  
        # First, we need to efficiently compute p * y_{i+k} and p * y_{i-k}
        # Using circular convolution (everything zero-padded to 2*n)
        fft_y = fft(y_d, fft_len)
        fft_p = fft(p, fft_len)
        # compute the circulant convolution for both y and (-Y)
        conv1 = ifft(fft_p * np.conj(fft_y)).real[:n]
        conv2 = ifft(fft_p * fft_y).real[:n]
        
        grad[:, d] = conv1 - conv2
    
    return grad

def line_search_phi_tau(Y, grad, step_size, tau, c, fft_len, n):
    """Backtracking line search on phi_tau"""
    # This is a simple gradient ascent step which will work with small stepsize
    c0 = c.copy()
    # Try some values
    # Since this is an smooth upper bound, we assume it behaves well
    # But we just do iteration until a reasonable increase
    for _ in range(10):  # 10 steps max at most
        Y_trial = Y - step_size * grad
        Y_trial = project_to_feasible(Y_trial, n)
        h_trial = np.sum(Y_trial**2, axis=1)
        c_trial = compute_correlations(Y_trial, fft_len)
        phi_trial = -tau * np.log(np.sum(np.exp(-c_trial/tau)))
        phi_current = -tau * np.log(np.sum(np.exp(-c0/tau)))
        if phi_trial > phi_current or step_size < 1e-10:
            break
        step_size *= 0.5
    return Y_trial, step_size
