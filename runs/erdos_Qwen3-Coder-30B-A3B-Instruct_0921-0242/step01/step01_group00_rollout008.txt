import numpy as np
import time
from cvxpy import*
import cvxpy as cp
from scipy.signal import correlate

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_sum(h, target_sum):
    """Ensure that sum(h) = target_sum, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Adjust by distributing difference
    h = h + diff / len(h)
    
    # Re-clip
    h = np.clip(h, 0, 1)
    
    return h

def solve_sdp(n, dx):
    """Solve the SDP relaxation for the approximate optimal h"""
    h = Variable(n)
    X = Variable((n, n), symmetric=True)
    t = Variable()

    constraints = []
    constraints += [trace(X) == sum(h)]  # tr(X) = sum(h) = n/2
    constraints += [X >> 0]  # X is positive semidefinite
    constraints += [X[i, i] == h[i] for i in range(n)]  # diag(X) = h
    
    # autocorrelation constraints
    for k in range(1, n):
        T_k = np.zeros((n, n))
        for i in range(n - k):
            T_k[i, i + k] = 1
        constraints += [trace(T_k @ X) >= t]  # trace(T_k * X) >= t
    
    # box constraints
    constraints += [h >= 0]
    constraints += [h <= 1]

    # objective
    objective = Maximize(t)

    prob = Problem(objective, constraints)
    prob.solve(solver=SCS, eps=1e-4, max_iters=3000)

    if prob.status != 'optimal':
        return None, float('inf')

    h_sdp = np.array(h.value).flatten()
    h_sdp = project_to_sum(h_sdp, n/2)
    t_sdp = t.value
    c5_sdp = dx * (1 - t_sdp)
    return h_sdp, c5_sdp

def randomized_rounding(h_sdp, n, n_rounds=30):
    """Try to find a good binary rounding of h_sdp"""
    # compute cholesky factor
    V = np.linalg.cholesky(np.outer(h_sdp, h_sdp))  # This won't work because h_sdp is not exactly rank-1
    eigvals, eigvecs = np.linalg.eigh(np.outer(h_sdp, h_sdp))
    # Keep only positive eigenvalues
    idx = eigvals > 1e-8
    V = eigvecs[:, idx] * np.sqrt(eigvals[idx])
    
    best_h = None
    best_c5 = float('inf')
    
    for r in range(n_rounds):
        g = np.random.randn(V.shape[1])
        y = V @ g
        # sort y and find threshold
        sorted_y = np.sort(y)
        # We want exactly n/2 elements to be > threshold
        threshold = sorted_y[n//2] 
        b = (y > threshold).astype(float)
        
        # ensure the sum is exactly n/2
        b = project_to_sum(b, n/2)
        c5 = evaluate_c5(b)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = b.copy()

    return best_h, best_c5

def smooth_max_objective(h, k_range, dx, beta):
    """Smooth log-sum-exp approximation to the max"""
    total = 0.0
    for k in k_range:
        # compute I_k(h)
        I_k = 0.0
        for i in range(len(h) - k):
            I_k += h[i] * h[i + k]
        I_k = dx * (1 - I_k)
        total += np.exp(beta * I_k)
    return (1.0 / beta) * np.log(total)

def gradient_smooth_max(h, k_range, dx, beta):
    """Gradient of the smooth max"""
    grad = np.zeros_like(h)
    
    I = np.zeros(len(k_range))
    exp_terms = np.zeros(len(k_range))
    for i, k in enumerate(k_range):
        # compute I_k(h)
        I_k = 0.0
        for j in range(len(h) - k):
            I_k += h[j] * h[j + k]
        I_k = dx * (1 - I_k)
        I[i] = I_k
        exp_terms[i] = np.exp(beta * I_k)
    
    # compute weights
    total_exp = np.sum(exp_terms)
    weights = exp_terms / total_exp if total_exp > 0 else np.ones_like(exp_terms) / len(exp_terms)
    
    # compute gradient
    for i in range(len(h)):
        grad[i] = 0.0
        for j, (k, weight) in enumerate(zip(k_range, weights)):
            #  derivative of I_k w.r.t. h_i
            #  ∂I_k/∂h_i = dx *(1 - h_{i+k}) - h_{i-k}
            left_term = 0.0
            right_term = 0.0
            if i + k < len(h):
                left_term = -dx * h[i+k]
            if i - k >= 0:
                right_term = dx * h[i-k]
            grad[i] += weight * (left_term + right_term)
    
    return grad

def project_gradient(h, grad, alpha, target_sum):
    """Take a gradient step and project back to feasible set."""
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def refine_with_smooth_max(h, c5, n, dx, max_time_s):
    """Refine h using smooth-max optimization"""
    start_t = time.time()
    k_range = list(range(1, n))
    beta = 0.5
    max_beta = 50.0
    step_size = 1.0
    
    best_h = h.copy()
    best_c5 = c5
    
    # Inner iterations
    while beta < max_beta and (time.time() - start_t) < max_time_s:
        inner_iter = 0
        while inner_iter < 50 and (time.time() - start_t) < max_time_s:
            grad = gradient_smooth_max(best_h, k_range, dx, beta)
            # Backtracking to find good alpha
            trial_h = best_h
            trial_alpha = step_size
            trial_c5 = float('inf')
            for _ in range(10):
                trial_h = project_gradient(best_h, grad, trial_alpha, n/2)
                trial_c5 = evaluate_c5(trial_h)
                if trial_c5 < best_c5 or trial_alpha < 1e-6:
                    break
                trial_alpha *= 0.5
                
            if trial_c5 < best_c5:
                best_h = trial_h
                best_c5 = trial_c5
                
            inner_iter += 1
            
        if beta >= 5.0:
            step_size = 0.1
        beta *= 1.5
        if beta > max_beta:
            beta = max_beta

    return best_h, best_c5

def local_search(h, c5, n, dx, max_time_s):
    """Perform discrete local search to refine"""
    start_t = time.time()
    best_h = h.copy()
    best_c5 = c5
    
    while time.time() - start_t < max_time_s:
        i, j = np.random.randint(0, n, 2)
        if i == j: continue
        if best_h[i] <= 0 or best_h[j] >= 1: continue
        
        delta = min(best_h[i], 1 - best_h[j], 0.02)
        if delta <= 0: continue
        
        h_trial = best_h.copy()
        h_trial[i] -= delta
        h_trial[j] += delta
        h_trial = project_to_sum(h_trial, n/2)
        c5_trial = evaluate_c5(h_trial)
        
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 128
    dx = 2.0 / n
    target_sum = n/2

    best_c5 = float('inf')
    best_h = None

    # Stage 1: SDP Relaxation
    if (time.time() - start_time) < budget_s - 30:
        h_sdp, c5_sdp = solve_sdp(n, dx)
        if h_sdp is not None and c5_sdp < best_c5:
            best_c5 = c5_sdp
            best_h = h_sdp.copy()
        print(f"SDP result: C5 = {c5_sdp}")

    # Stage 2: Rounding
    if (time.time() - start_time) < budget_s - 20:
        if best_h is not None:
            h_rounded, c5_rounded = randomized_rounding(best_h, n)
            if h_rounded is not None and c5_rounded < best_c5:
                best_c5 = c5_rounded
                best_h = h_rounded.copy()
            print(f"Rounding result: C5 = {c5_rounded}")

    # Stage 3: Smooth max refinement
    if (time.time() - start_time) < budget_s - 10:
        if best_h is not None:
            remaining_time = budget_s - (time.time() - start_time) - 10
            refined_h, refined_c5 = refine_with_smooth_max(best_h, best_c5, n, dx, remaining_time)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
            print(f"Smooth max refinement: C5 = {refined_c5}")

    # Stage 4: Final discrete local search
    if (time.time() - start_time) < budget_s - 5:
        if best_h is not None:
            remaining_time = budget_s - (time.time() - start_time) - 5
            final_h, final_c5 = local_search(best_h, best_c5, n, dx, remaining_time)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
            print(f"Local search: C5 = {final_c5}")

    if best_h is None:
        # fallback: random sample
        best_h = np.random.rand(n)
        best_h = project_to_sum(best_h, target_sum)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, n)
