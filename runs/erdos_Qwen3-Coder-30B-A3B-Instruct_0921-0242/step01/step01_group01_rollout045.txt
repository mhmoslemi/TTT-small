import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configurable parameters
    n = 200  # grid points (half the full size due to symmetry)
    dx = 2.0 / n
    d = n // 2 + 1  # number of independent variables due to symmetry
    max_shifts = 30  # max number of constraints to add
    epsilon = 1e-6
    safety_margin = 5.0
    max_iterations = 50
    
    # Initialize best solution
    best_c5 = float('inf')
    best_h = None
    
    # Symmetric binary penalty parameter
    lambda_param = 0.1
    lambda_growth = 1.5
    
    # Get initial solution (if available)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0[:n]  # truncate or pad as necessary
        h0 = h0 * (n / 2) / np.sum(h0)
        # Symmetrize the initial guess
        h0 = (h0 + h0[::-1]) / 2.0
    else:
        # Create an initial symmetric step function
        h0 = np.zeros(n)
        h0[:n//2] = 1.0
        h0 = h0 * (n / 2) / np.sum(h0)
        h0 = (h0 + h0[::-1]) / 2.0

    # Build LP model
    # We will use symmetric approach: variables w_i = h_i for i = 0..d-1 
    # then h_i = w_{n-1-i} for i >= d 
    w = cp.Variable(d)
    M = cp.Variable()
    p_vars = {}
    shift_set = set()
    
    # Create a dict to store the mapping from indices to variables
    # p_vars[(i, k)] is a list of the variable(s) corresponding to the bilinear product.
    # Here we make an assumption: all entries of h from 0 to n-1 are used in the shifts
    # stored in shift_set
    
    # Helper to add constraints for a fixed shift k
    def add_shift(k):
        # Determine valid indices where this shift applies
        valid_indices = []
        for i in range(n):
            if 0 <= i + k < n:
                valid_indices.append(i)
        
        # For each such index, create a variable p_{i,k}
        if k not in p_vars:
            p_vars[k] = []
        for i in valid_indices:
            if len(p_vars[k]) <= i:
                p_vars[k].extend([None for _ in range(i - len(p_vars[k]) + 1)])
            # Get or create the variable for p_{i,k}
            if p_vars[k][i] is None:
                p_vars[k][i] = cp.Variable()
                
        return valid_indices
    
    # Handle initial shift 0
    shift_set.add(0)
    valid_indices_0 = add_shift(0)
    
    # Constraint list
    constraints = []
    
    # Box constraints
    constraints.extend([w[i] >= 0 for i in range(d)])
    constraints.extend([w[i] <= 1 for i in range(d)])
    
    # Sum constraint
    sum_constraint = cp.sum(w) == n / 4 + 0.5
    constraints.append(sum_constraint)
    
    # McCormick constraints for k=0
    for i in valid_indices_0:
        p_idx = p_vars[0][i]
        # Apply McCormick: p >= 0, p >= w[i] + w[n-1-i] - 1
        # p <= w[i], p <= w[n-1-i]
        constraints.append(p_idx >= 0)
        constraints.append(p_idx >= w[i] + w[n-1-i] - 1)
        constraints.append(p_idx <= w[i])
        constraints.append(p_idx <= w[n-1-i])
        # Ignoring p = h_i * h_{n-1-i}, note that this is not technically safe
        # Better: just ensure w[i]==w[n-1-i] (the symmetric part) but that's not the case
        # Actually the challenge arises in infinite-dimensional extension
        # We continue usingfull here.
    
    # Make the LP problem
    objective = cp.Minimize(M - lambda_param * cp.sum([p_vars[0][i] for i in valid_indices_0]))
    
    prob = cp.Problem(objective, constraints)
    
    iteration = 0
    while time.time() < start_time + budget_s - safety_margin and iteration < max_iterations:
        if len(shift_set) > max_shifts:
            break
            
        # Create the most current problem version
        # Rebuild the model for active constraints ((dynamic addition)
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status != cp.OPTIMAL:
                break
            
            # Extract solution
            w_val = w.value
            if w_val is None:
                break  # revert to error
            
            h_current = np.zeros(n)
            for i in range(d):
                h_current[i] = w_val[i]
                h_current[n-1-i] = w_val[i]
                
            # Or make it correctly symmetric
            h_symmetric = np.zeros(n)
            for i in range(d):
                h_symmetric[i] = w_val[i]
                h_symmetric[n-1-i] = w_val[i]
            # Ensure two endpoints are handled
            h_current = h_symmetric
            
            # Calculate overlap using full convolution
            g = 1.0 - h_current
            conv_full = np.correlate(h_current, g, mode='full')
            c5_current = np.max(conv_full) * dx
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
            
            # Check convergence
            if abs(c5_current - M.value) < epsilon:
                break
            
            # Find the worst shift
            k_max = np.argmax(conv_full)
            k = k_max - (n - 1)  # maps back to actual k shift (positive or negative)
            
            # Add this as new constraint
            shift_set.add(k)
            valid_indices = add_shift(k)
            
            # McCormick constraints for shift k
            for i in valid_indices:
                p_idx = p_vars[k][i]
                # The index of h_{i+k} due to how we're indexing
                j = i + k
                if j < 0 or j >= n:
                    continue
                # Constraint variable mapping: need w[i] and w[j] where j = i + k
                if j >= d and j < n:
                    j_sym = n - 1 - j
                    if j_sym < 0 or j_sym >= d:
                        continue
                    j_used = j_sym
                else:
                    j_used = j if j < n else n - 1 - (j - n)
                
                if j_used >= d:
                    # dout-of-bound lexically... but may be ignorable
                    continue
                
                # Using near-symmetric handling: mine all are indexed effectively 
                # Work from index {i, j} to create p_{i,k}
                # Ignore numerical or edge imprecision
                constraints.append(p_idx >= 0)
                constraints.append(p_idx >= w[i] + w[j_used] - 1)
                constraints.append(p_idx <= w[i])
                constraints.append(p_idx <= w[j_used])
            
            # Add the overlap constraint for the new k
            sum_p = cp.sum([p_vars[k][i] for i in valid_indices])
            overlap_expr = (n/2) - sum_p
            constraints.append(overlap_expr <= M)

            # Increase lambda to encourage binary solutions
            lambda_param = min(5.0, lambda_param * lambda_growth)

            iteration += 1

        except Exception as e:
            # Debug
            break

    if best_h is None:
        # Fallback to simple symmetric
        h_temp = np.ones(n) * 0.5
        h_temp[100-n//2:100+n//2] = 1.0
        h_temp = h_temp * (n / 2) / np.sum(h_temp)
        best_h = (h_temp + h_temp[::-1]) / 2.0
        best_c5 = np.max(np.correlate(best_h, 1-best_h, mode='full')) * dx

    # Polyhedral polishing phase - projected subgradient descent
    h_polish = best_h.copy()
    c5_final = best_c5
    alpha = 0.02
    max_polish_iter = 200
    
    for polish_iter in range(max_polish_iter):
        if time.time() >= start_time + budget_s - 0.5:
            break
            
        # Find maximum overlap shift and compute its subgradient
        g = 1.0 - h_polish
        conv = np.correlate(h_polish, g, mode='full')
        max_shift_idx = np.argmax(conv)
        k_max = max_shift_idx - (n - 1)
        
        # Compute subgradient
        gradient = np.zeros(n)
        for i in range(n):
            # Subgradient calculation symbolic:
            # grad_i = (1 - h_{i+k_max}) if i+k_max < n, else 0
            #        - h_{i-k_max} if i-k_max >= 0, else 0
            
            if i + k_max < n:
                gradient[i] += (1 - h_polish[i + k_max])
            if i - k_max >= 0:
                gradient[i] -= h_polish[i - k_max]
                
        # Update h
        h_new = h_polish - alpha * gradient
        
        # Project back to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum = n/2
        total = np.sum(h_new)
        diff = (n / 2) - total
        if abs(diff) > 1e-9:
            # Water filling step: distribute the difference in areas evenly
            valid_idx = np.where((h_new > 0.0001) & (h_new < 0.9999))[0]
            if len(valid_idx) == 0:
                pass
            else:
                increment = diff / len(valid_idx)
                # Adjust
                for idx in valid_idx:
                    if h_new[idx] + increment > 1:
                        h_new[idx] = 1
                    elif h_new[idx] + increment < 0:
                        h_new[idx] = 0
                    else:
                        h_new[idx] += increment
                # Normalize again if necessary
                norm_total = np.sum(h_new)
                if abs(norm_total - n/2) > 1e-9:
                    h_new = h_new * (n/2) / norm_total
                    
        # Symmetrize
        h_polish = (h_new + h_new[::-1]) / 2.0
        
        # Compute new overlap
        c5_new = np.max(np.correlate(h_polish, 1-h_polish, mode='full')) * dx
        if c5_new < c5_final:
            c5_final = c5_new
            best_h = h_polish.copy()
            
        # Adapt step size
        alpha *= 0.9
        if abs(c5_final - best_c5) < 1e-6:
            break
    
    return (best_h, c5_final, n)
