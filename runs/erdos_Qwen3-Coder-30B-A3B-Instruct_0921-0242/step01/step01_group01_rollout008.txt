def water_fill_projections(h, target_sum):
    """Project h to the feasible set with sum = target_sum."""
    h = np.copy(h)
    current_sum = np.sum(h)
    diff = current_sum - target_sum
    
    if abs(diff) < 1e-12:
        return h
    
    # Apply water filling
    if diff > 0:
        # Need to reduce sum: distribute excess among interior points
        indices = np.where((h > 0) & (h < 1))[0]
        while abs(diff) > 1e-12 and len(indices) > 0:
            avg_diff = diff / len(indices)
            for i in indices:
                if h[i] - avg_diff >= 0:
                    h[i] -= avg_diff
                else:
                    h[i] = 0
                    diff += h[i] + avg_diff
            indices = np.where((h > 0) & (h < 1))[0]
    else:
        # Need to increase sum: distribute among interior points
        indices = np.where((h > 0) & (h < 1))[0]
        while abs(diff) > 1e-12 and len(indices) > 0:
            avg_diff = diff / len(indices)
            for i in indices:
                if h[i] + avg_diff <= 1:
                    h[i] += avg_diff
                else:
                    h[i] = 1
                    diff -= (h[i] - avg_diff)
            indices = np.where((h > 0) & (h < 1))[0]
    
    return h

def evaluate_c5(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    result = np.max(conv) * dx
    return result

def smooth_max_gradient(h, beta):
    """Compute gradient of smooth max function at h with temperature beta."""
    dx = 2.0 / len(h)
    g = 1.0 - h
    
    # Full correlation
    conv = np.correlate(h, g, mode='full')
    
    # Softmax weights
    max_val = np.max(beta * dx * conv)
    exp_vals = np.exp(beta * dx * conv - max_val)
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient calculation using convolutions (circular shift)
    u = np.correlate(weights, 1 - h, mode='full')
    v = np.correlate(weights[::-1], h, mode='full')
    
    # The gradient at i-th element of h is: dx * (u[i] - v[i]) 
    grad = dx * (u - v)
    return grad[:len(h)]  # Trim to original length
