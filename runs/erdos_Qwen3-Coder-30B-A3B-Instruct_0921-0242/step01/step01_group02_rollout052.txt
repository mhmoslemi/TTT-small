```python
import numpy as np
import time
import cvxpy as cp
from scipy import sparse as sp
from scipy.signal import correlate

def build_Q(k, n):
    """Build sparse matrix for quadratic term h_i * h_{i+k}"""
    if k == 0:
        data = np.ones(n)
        rows = np.arange(n)
        cols = np.arange(n)
        Q = sp.csr_matrix((data, (rows, cols)), shape=(n,n))
    else:
        data = np.ones(n - abs(k))
        rows = np.arange(n - abs(k))
        cols = rows + k if k > 0 else rows - k
        Q = sp.csr_matrix((data, (rows, cols)), shape=(n,n))
    return 0.5 * (Q + Q.T)

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def solve_qp(n, Kmax, warm_h=None, solver=cp.OSQP):
    """Solve convex QP formulation"""
    h = cp.Variable(n)
    t = cp.Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    
    # Build quadratic constraints for each shift k
    for k in range(-Kmax, Kmax+1):
        Q = build_Q(k, n)
        # Note: Convert to dense for CVXPY compatibility
        constraints.append(cp.quad_form(h, Q) >= n/2 - t)
    
    # Objective: minimize t
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    if warm_h is not None:
        h.value = warm_h
        
    prob.solve(solver=solver, verbose=False, max_iter=5000, eps_abs=1e-9)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h.value, t.value
    else:
        return None, None

def binary_refine(h_frac, n, dx, time_limit):
    """Refine fractional solution to binary with local search"""
    # Start with the top n/2 entries set to 1
    idx = np.argsort(-h_frac)[:n//2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1
    
    # Precompute correlation for current binary vector
    corr = correlate(h_bin, 1-h_bin, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h_bin.copy()
    
    while time.time() < time_limit:
        # Pick two random indices: one 1, one 0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        
        # Try swap
        h_bin[i1] = 0
        h_bin[i0] = 1
        
        # Update correlation efficiently
        new_corr = np.correlate(h_bin, 1-h_bin, mode='full')
        new_c5 = np.max(new_corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_bin.copy()
        else:
            # Revert swap
            h_bin[i1] = 1
            h_bin[i0] = 0
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different grid sizes
    n_values = [250, 300, 350, 400]
    Kmax_values = [int(0.75*n) for n in n_values]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    warm_h = None
    for i, (n, Kmax) in enumerate(zip(n_values, Kmax_values)):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
            
        print(f"Solving QP for n={n}, Kmax={Kmax}")
        h_solution, t_value = solve_qp(n, Kmax, warm_h)
        if h_solution is None:
            continue
            
        # Clip to [0, 1] and renormalize
        h_solution = np.clip(h_solution, 0, 1)
        h_solution = h_solution * (n/2) / np.sum(h_solution)
        warm_h = h_solution
        
        # Local refinement with binary rounding and swaps
        dx = 2.0 / n
        refine_time = time.time() + min(100, remaining/2)
        h_binary, c5_refined = binary_refine(h_solution, n, dx, refine_time)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_binary.copy()
            best_n = n
            
        print(f"n={n}: C5 = {c5_refined:.6f}")
        
        if time.time() - start_time > remaining - 5:
            break
    
    # Final high-precision evaluation
    if best_h is not None:
        dx = 2.0 / best_n
        final_corr = correlate(best_h, 1-best_h, mode='full')
        final_c5 = np.max(final_corr) * dx
        best_c5 = final_c5
    
    return (best_h, best_c5, best_n)
```
