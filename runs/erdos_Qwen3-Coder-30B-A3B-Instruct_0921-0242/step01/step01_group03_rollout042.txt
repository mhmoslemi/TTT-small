def build_shift_matrices(N):
    """Build sparse CSR shift matrices A_k for k = 0, ..., N-1"""
    As = []
    for k in range(N):
        # A_k[i,j] = 1 if j = i + k and i + k < N, else 0
        rows = np.arange(N - k)
        cols = np.arange(k, N)
        data = np.ones(N - k)
        A = sp.csr_matrix((data, (rows, cols)), shape=(N, N))
        As.append(A)
    return As

def project_to_feasible(h):
    """Ensure sum(h) = n/2 and h in [0,1]^n"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    # Distribute deficit
    adjustment = deficit / n
    h += adjustment
    h = np.clip(h, 0, 1)
    return h

def max_correlation_from_fft(mask, A_k_list):
    """Compute maximum correlation using FFT-based method for the mask"""
    N = len(mask)
    # Horner's method is a bit tricky here.
    # Use standard FFT approach as used in the original evaluate_erdos_solution
    # Originally used numpy.correlate, but let's stick to what is well known in code
    # Here we use g = 1-mask, compute full correlation of h with g
    g = 1.0 - mask
    corr = np.correlate(mask, g, mode='full')
    max_corr = np.max(corr)
    # corr has length 2*N-1 centered at index N-1
    return max_corr

def evaluate_c5(h):
    """Evaluate C5 for the function h on [0,2]"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation with mode='full' to capture all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # Take max over all shifts
    return np.max(corr)

def principal_eigenvector_rounding(X):
    """Wrapper for principal eigenvector rounding"""
    # Get the leading eigenpair of X
    # Use scipy to handle the large matrix efficiently
    _, v = sp.linalg.eigs(X, k=1, which='LM')
    v = np.real(v[:, 0])
    # Form the candidate vector (should be real)
    h_eig = np.abs(v) * np.sign(v[0])  # Make it positive, aligned with first entry
    h_eig = np.sqrt(np.abs(h_eig))  # Use square root to promote near-binary structure
    # Clip to [0, 1]
    h_eig = np.clip(h_eig, 0, 1)
    h_eig = project_to_feasible(h_eig)
    return h_eig

def diagonal_threshold_rounding(X):
    """Diagonal threshold rounding"""
    # X_{ii} approximates h_i^2
    diag_entries = np.diag(X)
    # Sort indices by diagonal entries descending
    idxs_sorted = np.argsort(diag_entries)[::-1]
    # Set top N/2 entries to 1, rest to 0
    mask = np.zeros(len(diag_entries))
    mask[idxs_sorted[:len(diag_entries)//2]] = 1.0
    return mask

def solve_sdp(N, A_list):
    """Solve the SDP relaxation for maximum minimum correlation"""
    # Define variables
    h = cp.Variable(N)
    X = cp.Variable((N, N), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = []
    # X_{ii} = h_i
    for i in range(N):
        # Add to constraints for diagonal
        constraints.append(X[i, i] == h[i])
    
    # Mass constraint
    constraints.append(cp.sum(h) == N / 2)
    
    # Bounds on h
    constraints.extend([h[i] >= 0 for i in range(N)])
    constraints.extend([h[i] <= 1 for i in range(N)])
    
    # PSD constraint
    constraints.append(X >> 0)
    
    # Correlation constraints
    for k in range(N):
        # tr(A_k X) >= t
        constraints.append(cp.trace(A_list[k] @ X) >= t)
    
    # Objective
    objective = cp.Maximize(t)
    
    # Formulate and solve the SDP
    prob = cp.Problem(objective, constraints)
  
    try:
        prob.solve(solver=cp.SCS, max_iters=15000, eps=1e-4)
    except Exception:
        return None, None
    
    if prob.status != cp.OPTIMAL:
        return None, None
    
    h_sol = h.value
    X_sol = X.value
    t_sol = t.value
    
    return h_sol, X_sol, t_sol
