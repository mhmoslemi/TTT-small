import numpy as np

def proj_simplex_box(h, n):
    """Project h onto {0 <= h_i <= 1, sum h_i = n/2}."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - n/2) < 1e-10:
        return h
    # Use water-filling
    diff = n/2 - s
    # Adjust using constant shift
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def worst_shift(h):
    """Get the shift index k that maximizes the overlap integral."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    # Mask out k=0
    corr[len(h)-1] = -np.inf  # exclude zero shift
    max_k = np.argmax(corr)
    # Convert index back to shift k
    k = max_k - (len(h) - 1)
    return k

def gradient_step(h, k, alpha=0.01):
    """Do one step of projected gradient descent on shift k."""
    n = len(h)
    dx = 2.0 / n
    
    # Gradient w.r.t. h of integral h(x) (1 - h(x+k))
    grad = np.zeros(n)
    for i in range(n):
        # Derivative is -dx * h[i] if i is shifted by k
        if i == (k + 0) % n:
            grad[i] -= dx
        # plus dx * 1 if it is a source handle?
        # No wait, that's not the sign
        # d/dh_i (h_i (1-h_{i+k})) = (1 - h_{i+k}) - h_i = 1 - h_{i+k} - h_i
        # But we want d/dh_i sum h_i (1-h_{i+k}) = (1 - h_{i+k})
        # YES, except make sure indexing is correct on the "shifted"

        # Actually:  d/dh_i (sum_h_i (1 - h_{i+k})) = 1 - h_{i+k} if sum index i
        # That’s correct
        grad[i] = -dx * (1 - h[(i - k) % n])
        # And for the shift term to loop we need to .
        # diff = -dx * (1 - h[(i-k)%n]) - dx * h[i]
        # wait no. gradient of sum x_i y_{i+k} 

    new_h = h - alpha * grad
    new_h = proj_simplex_box(new_h, n)
    return new_h

def anneal(h, time_limit):
    """Run Simulated Annealing."""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    current_h = h.copy()
    current_c5 = c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    temp = 1.0
    cooling_rate = 0.995
    
    while time.time() - start_time < time_limit:
        if temp < 1e-6:
            break
            
        # Propose change
        idx1, idx2 = np.random.choice(n, size=2, replace=False)
        delta = np.random.uniform(-0.1, 0.1)
        new_h = current_h.copy()
        new_h[idx1] += delta
        new_h[idx2] -= delta
        
        # Ensure feasibility
        new_h = proj_simplex_box(new_h, n)
        
        # Accept or reject
        new_c5 = c5(new_h)
        
        if new_c5 < current_c5:
            current_h = new_h.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        else:
            # Accept with probability
            p = np.exp(-(new_c5 - current_c5) / temp)
            if np.random.rand() < p:
                current_h = new_h.copy()
                current_c5 = new_c5

        temp *= cooling_rate
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Reduce problem size to make fast
    n_points = 400
    dx = 2.0 / n_points
    
    # Phase 1: Relaxation to get structure
    h = solve_lp_relaxation(n_points)
    if np.mean(h) != 0.5 * n_points / n_points:  # if not already set correctly
        h = np.ones(n_points) * 0.5
    h = proj_simplex_box(h, n_points)
    
    # Phase 2: Local steps on worst shift
    current_c5 = c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    if (time.time() - start_time) < budget_s - 30:
        for _ in range(200):
            if time.time() - start_time > budget_s - 1: break
            k = worst_shift(h)
            h = gradient_step(h, k, alpha=0.001)
            h = proj_simplex_box(h, n_points)
            current_c5 = c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    # Phase 3: Annealing
    if (time.time() - start_time) < budget_s - 30:
        h, c5_trial = anneal(best_h, budget_s - (time.time() - start_time) - 10)
        if c5_trial < best_c5:
            best_h = h
            best_c5 = c5_trial
    
    # Phase 4: Final fine-tuning
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        best_c5 = c5(best_h)
        
    # Eval final
    final_c5 = c5(best_h)
    return (best_h, final_c5, n_points)
