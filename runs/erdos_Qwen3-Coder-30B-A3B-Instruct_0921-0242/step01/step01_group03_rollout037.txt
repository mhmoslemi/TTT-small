import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def evaluate_binary_mask(mask):
    """Evaluate the binary mask for C5 and return normalized h"""
    dx = 2.0 / len(mask)
    h = mask.astype(float)
    h = h * (len(mask) / 2) / np.sum(h)  # normalize to sum = n/2
    return c5_of_h(h)

def perturb_binary_mask(mask, num_swaps=2):
    """Perturb a binary mask by swapping 0 and 1"""
    new_mask = mask.copy()
    ones = np.where(new_mask == 1)[0]
    zeros = np.where(new_mask == 0)[0]
    
    for _ in range(num_swaps):
        if len(ones) > 0 and len(zeros) > 0:
            idx1 = np.random.choice(ones)
            idx2 = np.random.choice(zeros)
            new_mask[idx1] = 0
            new_mask[idx2] = 1
            ones = np.where(new_mask == 1)[0]
            zeros = np.where(new_mask == 0)[0]
    return new_mask

def project_to_feasible(h):
    """Ensure sum(h) = n/2 and h in [0,1]^n"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    # Uniformly adjust to meet exact sum constraint
    if deficit > 0:
        h += deficit / n
    else:
        h += deficit / n
    h = np.clip(h, 0, 1)
    return h
