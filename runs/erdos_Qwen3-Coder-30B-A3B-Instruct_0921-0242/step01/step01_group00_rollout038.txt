import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def worst_shift(h):
    """Find the shift k that maximizes the overlap integral."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # Adjust index to get the shift k in [-n+1, n-1]
    k = max_idx - (len(h) - 1)
    return k

def proj_simplex_box(h, n):
    """Project h onto the set {0 <= h <= 1, sum(h) = n/2}."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    # Simple adjustment to meet the sum constraint
    diff = n/2 - current_sum
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def gradient_step(h, k, alpha=0.01):
    """Perform one gradient step to improve the k-th shift."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros_like(h)
    # Compute gradient w.r.t. h_i for I_k = dx*(1 - sum h_i h_{i+k})
    # d/dh_i I_k = dx * (-h_{i+k})
    # d/dh_{i-k} I_k = dx * (-h_i)
    # but for all i compute ∂I_k/∂h_i

    # This is corrected gradient: sum over i of h_{i+k} dx = 1, so
    # to reduce I_k, push h_i to make h_i h_{i+k} smaller:
    # For each i: dI_k/dh_i = dx*(1 - 2 h_{i+k})
    # But we need to match the actual changing variable h_i: 
    # ∂I_k/∂h_i = dx * (-h_{i-k} - h_{i+k}) if k≠0

    # Actual implementation by replacing formula: 
    # ∂I_k/∂h_i = dx * ( - h_{i-k} - h_{i+k} ) 
    # Wrap-around using % to simulate circular convolution
    # Avoid including h_i itself

    # For current k indexing: roll h to left by k positions
    h_k = np.roll(h, k)
    # Note: h_k[i] = h[(i+k)%n], so we want squared difference for min
    # But better way:
    # ∂I_k/∂h_i = dx * ( - h_{i-k} - h_{i+k} );
    for i in range(n):
        left_idx = (i - k) % n
        right_idx = (i + k) % n
        g[i] = -dx * (h[left_idx] + h[right_idx])
        
    h_new = h - alpha * g
    h_new = proj_simplex_box(h_new, n)
    return h_new

def anneal(h_start, time_limit):
    """Perform simulated annealing on h."""
    start_time = time.time()
    h = h_start.copy()
    current_c5 = evaluate_c5(h)
    T = 0.1
    cooling_rate = 0.9999
    iter_count = 0
    h_best = h.copy()
    c5_best = current_c5
    
    while time.time() - start_time < time_limit - 0.1 and T > 1e-5:
        # Try small move
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        delta = np.random.uniform(-0.05, 0.05)
        # Cap the change
        delta = max(-0.1, min(0.1, delta))
        h_new = h.copy()
        old_val_i = h_new[i]
        old_val_j = h_new[j]
        h_new[i] += delta
        h_new[j] -= delta
        # Clipping and re-normalizing 
        h_new = np.clip(h_new, 0, 1)
        h_new = proj_simplex_box(h_new, len(h_new))
        c5_new = evaluate_c5(h_new)
        
        if c5_new < current_c5:
            h = h_new
            current_c5 = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        elif np.random.rand() < np.exp(-(c5_new - current_c5) / (T + 1e-10)):
            h = h_new
            current_c5 = c5_new
            
        if iter_count % 100 == 0:
            T *= cooling_rate
        iter_count += 1
    return h_best, c5_best
