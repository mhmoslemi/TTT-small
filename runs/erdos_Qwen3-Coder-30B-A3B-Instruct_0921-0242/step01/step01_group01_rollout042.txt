def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 300
    dx = 2 / n_points

    # Step 1: We'll directly use coarse MILP (n0=80)
    n0 = 80

        x_l = solve_binary_milp(n0)
        print(f"MILP at coarse level (n0={n0}) found some result.")
    except Exception as e:

        x_l = np.random.choice([0, 1], size=n0, p=[0.5, 0.5]) 
    
    # Step 2: Upscale coarse solution to fine resolution
    # Repeat each coarse cell with n//n0 elements (more than integer multiples are okay)
    if n_points % n0 != 0:
        up_ratio = n_points // n0 + 1
        # Not perfect crop—truncation pushed to right
        x_fine = np.repeat(x_l, up_ratio)
        x_fine = x_fine[:n_points]
        x_fine = x_fine * (n_points / 2) / np.sum(x_fine)
    else:
        x_fine = np.tile(x_l, n_points // n0)
        x_fine = x_fine * (n_points / 2) / np.sum(x_fine)

    # Now x_fine should have integral = n_points/2 and assume it's valid, 
    # applying smoothing to allow this to be valid until projection

    # Step 3: Trajectory smoothing
    # We'll debias with minimize_c5 with proximal oscillator and water_fill projection
    
    def c5_func(h):
        conv = np.correlate(h, 1-h, mode='full') * dx
        return np.max(conv)
        
    def project_sum(h):
        delta = np.sum(h) - n_points / 2
        if abs(delta) < 1e-12:
            return h
        # Water-filling
        count = np.count_nonzero((h > 0) & (h < 1))
        if count == 0:
            return h
        delta_per = delta / count
        new_h = h.copy()
        # Reduce interior entries by delta_per
        for i in range(n_points):
            if 0 < h[i] < 1:
                new_h[i] -= delta_per
        return np.clip(new_h, 0, 1)

    best_h = x_fine.copy()
    best_c5 = c5_func(best_h)
    print(f"Initial C5 = {best_c5}")

    # Now smooth-induce refinement via differentiable approximation with 
    # soft-max gradient-based method
    nref = 100
    beta = 0.5
    
    # Create approximate negative log sum (i.e., approximate C5)
    def relax_c5(h, beta=1.0):
        # Convert to ensures 1d layout with correct convolution handling
        convs = np.correlate(h, 1-h, mode='full')
        scaled_convs = dx * convs
        exp_convs = np.exp(beta * scaled_convs)
        numerator = np.sum(exp_convs)
        log_sum = np.log(numerator/len(convs))
        return log_sum / beta
    
    def grad_of_c5(h, beta):
        # Build normalized masses (no longer convolving because using full pattern aware)
        convs = np.correlate(h, 1-h, mode='full')
        scaled_convs = dx * convs
        
        # Weights w_k for soft summation = exp(beta * scaled_convs) / sum_x(exp(...))  
        weights = np.exp(beta*scaled_convs) / np.sum(np.exp(beta*scaled_convs))
        
        # Compute derivatives of each conv_k wrt h_i
        # v_k(i) = (1 - h[i+k]) if exists else 0 
        #        - h[i-k] if exists else 0  
        grad = np.zeros_like(h)
        # For each convolution index k,
        # with legend for extrusions: 
        # part_abs = np.abs(k)  # offset for along axis
        
        # Optimized loop inside real dual gaming continuous step
        k_range = range(-n_points+1, n_points) 
        for i in range(n_points):
            contrib = 0.0
            for k in k_range:
                # Compute fix triple direction effect
                if 0 <= i+k < n_points:
                    contrib += weights[k+n_points-1] * (1 - h[i+k])
                if 0 <= i-k < n_points and k != 0:
                    contrib -= weights[k+n_points-1] * h[i-k] 
            grad[i] = contrib
            
        return grad
    
    # Run L-BFGS-B
    max_iter_stages = 50
    for stage in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        # minimal L_bfgs_b powder settings mirror run
        h = project_sum(best_h.copy())
        # Take iter-action few wide:
        rectx = minimize(relax_c5, h.copy(), args=(beta),
                         method='L-BFGS-B', jac=grad_of_c5,
                         options={'maxiter': max_iter_stages, 'disp': False})
        
        h = project_sum(rectx.x)
        best_c5 = c5_func(h)

    p = min(6, n_points // 20) 
    eps = 0.004
    for _ in range(tries):
        if time.time() - start_time > budget_s - 10:
            break
        # Choose small random subset
        indices = np.random.choice(n_points, p, replace=False)
        h_trial = best_h.copy()
        disp = np.random.uniform(-eps, eps, p)
        for i, delta in zip(indices, disp):
            h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
        
        h_trial = project_sum(h_trial)
        c5_trial = c5_func(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
            
    return (best_h, best_c5, n_points)
