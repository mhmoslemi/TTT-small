```python
import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Problem parameters
    n = 200  # Grid size (even for symmetry)
    dx = 2.0 / n
    d = n // 2 + 1  # Number of independent variables (due to symmetry)
    half_sum = n / 4 + 0.5  # Target sum for independent variables
    epsilon = 1e-6  # Tolerance for termination
    safety_margin = 5  # Safety margin in seconds
    max_iterations = 50
    max_shifts = 30
    max_lp_solves = 30

    # Generate symmetric indices
    indices = np.arange(d)
    reversed_indices = np.arange(d-1, -1, -1)
    # Mirror symmetric indices
    sym_index_map = np.concatenate([indices, reversed_indices[1:]])
    
    # Objective: min M - lambda * sum of p_{i,0}
    # We model h_{i+k} using shifted variables properly
    # But taking symmetry, we can build the constraints efficiently

    # Helper to build convex relaxation with symmetry
    def build_master_lp(S, lambda_reg=0.1):
        # Create variables
        w = cp.Variable(d)  # independent h values
        M = cp.Variable(1)  # upper bound

        # Constraints
        constraints = []

        # Box constraints on w
        constraints += [w >= 0, w <= 1]

        # Sum constraint
        constraints += [cp.sum(w) == half_sum]

        # Add variables p_{i,k} for each k in S and valid i
        p_vars = {}
        p_constraints = []

        # For each shift, compute which indices are valid
        p_flat = []  # variables in order
        p_flat_constraints = []  # constraints for each p multipliers

        # Compute list of (i,k) pairs for all active shifts
        p_indices = []
        p_shifts = []
        p_start_idx = 0
        p_shape = []

        for k in S:
            if k == 0:
                # diagonal terms p[i,0] = w[i]^2 (but use McCormick)
                valid_i = np.arange(n)  # not just d indices for zero shift
            else:
                # explore only valid: i, i+k should be inside [0,n)
                valid_i = np.arange(n - abs(k))  # ensure i+k < n
                if k > 0:
                    valid_i = valid_i[:-k]  # for i, i+k to be both in [0,n)

            # For each i in valid_i, we have a constraint with h[i] and h[i+k]
            # The second term should be r_{i,k} where index in full h is (i+k) but this also depends on
            # if k is positive or negative and how we use mirroring
            
            # Suddenly, let's simplify: for a symmetric vector we have full symbolic indices with wrapping but in 
            # our simplified framework as long as we are consistent with symmetry
            # i goes up to d-1 for understanding, in full array mapping
            # But indexing always uses the same logic → resymmetrize after solving
            
            k_index = k
            # In real indexing:
            Irange = valid_i  # valid starting indices in actual array
            # Ensure that i + k < n
            if k > 0:
                Irange = valid_i[: -k]
            else:
                Irange = valid_i[ -k :]

            for i in Irange:
                # Define p_{i,k} if it makes sense
                p_indices.append((i, k))
                p_shifts.append(k)
                
        # Now create vec of all p vars
        p_count = len(p_indices)
        p_vars = cp.Variable(p_count)
        p_vars_exprs = []

        # Constraint expressions for McCormick envelope per p[i,k]
        # p <= h_i, p <= h_{i+k}, p >= 0, p >= h_i + h_{i+k} -1
        p_cons = []
        for idx, (i, k) in enumerate(p_indices):
            p = p_vars[idx]
            p_expr = p
            p_vars_exprs.append(p_expr)

            # mirrored indices in full array
            if k < 0:
                j = i + k
                # j is left of i
                assert j >= 0, f"Invalid j={j} for i={i}, k={k}"
                # get actual indices for h_i and h_j in symmetric mirrored array
                # main full indices
                h_index_i = i
                h_index_j = j
                h_val_i = w[i]  # for independent
                h_val_j = w[j] if j < d else w[n - 1 - j]
            else:
                # k > 0
                j = i + k
                assert j < n, f"Invalid j={j} for i={i}, k={k}"
                # inverted indexing: h_index_j = j matches, h_index_i = i
                h_index_i = i
                h_index_j = j
                h_val_i = w[i]
                h_val_j = w[j] if j < d else w[n - 1 - j]

            p_cons.append(p <= h_val_i)
            p_cons.append(p <= h_val_j)
            p_cons.append(p >= 0)
            p_cons.append(p >= h_val_i + h_val_j - 1)

        # Add all constraints
        constraints.extend(p_cons)

        # Overlap constraints for each shift k
        for k in S:
            # Compute sum of p_{i,k} terms corresponding to this k
            sum_p = sum(p_vars_exprs[i] for i, (i_val, shift_val) in enumerate(p_indices) if shift_val == k)
            # Consider integral (1 - integral) — the overlap
            # Current shift overlap = (n/2) - sum_p, target <= M
            if k == 0:
                # For k=0, is sum p[i,0] = sum h[i]^2
                # So overlap is (n/2) - sum_p = (n/2) - sum(w[i]^2)
                sum_p = sum(p_vars_exprs[i] for i, (i_val, shift_val) in enumerate(p_indices) if shift_val == 0)
                constraints.append((n/2) - sum_p <= M)
            else:
                # Original constraint
                constraints.append((n/2) - sum_p <= M)

        # Objective: minimize M - lambda * sum_{i,k} p_{i,0}
        # Sum p_{i,0}: assumes k=0 bits
        p_0_sum = sum(p_vars_exprs[i] for i, (i_val, shift_val) in enumerate(p_indices) if shift_val == 0)

        objective = cp.Minimize(M - lambda_reg * p_0_sum)

        # Construct problem
        prob = cp.Problem(objective, constraints)

        return prob, w, M, p_vars, p_indices, p_vars_exprs
    
    # Optimization loop
    iteration_count = 0
    S = [0]  # set of shifts to consider
    lambda_reg = 0.1
    
    best_c5 = float('inf')
    best_h_full = None

    while iteration_count < max_iterations and len(S) <= max_shifts and (time.time() - start_time) < budget_s - safety_margin:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Solve master LP
        try:
            prob, w_var, M_var, p_vars, p_indices, p_exprs = build_master_lp(S, lambda_reg)
            
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status not in ["optimal", "optimal_inaccurate"]:
                break
                
            # Extract solution w
            w_sol = w_var.value
            if w_sol is None:
                break
            # Build full h
            h_sym = np.zeros(n)
            h_sym[:d] = w_sol
            h_sym[d:] = w_sol[1::-1]  # mirror, exclude center as it's symmetric

            # Check if C5 has improved
            conv_full = np.correlate(h_sym, 1 - h_sym, mode='full')
            c5_exact = np.max(conv_full) * dx
            
            if c5_exact < best_c5:
                best_c5 = c5_exact
                best_h_full = h_sym.copy()

            # Check convergence
            m_val = float(M_var.value)
            if abs(c5_exact - m_val) < epsilon:
                break

            # Add most violated shift k_max
            # compute overlaps of h_sym with 1-h_sym for all valid shifts
            kmax = np.argmax(conv_full)
            k_shift = kmax - (n - 1)  # mapping back
            if k_shift not in S:
                S.append(k_shift)

            if len(S) > 20:
                lambda_reg = min(5.0, lambda_reg * 1.2)
            iteration_count += 1
            
        except:
            break

    # Polishing step: Gradient descent on the worst shift away from LP solution
    if best_h_full is not None:
        h_current = best_h_full.copy()
        max_iter = 200
        alpha = 0.02
        tol = 1e-6

        for i in range(max_iter):
            if (time.time() - start_time) >= budget_s - 1:
                break
            # find worst shift k
            conv_full = np.correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(conv_full) - (n - 1)
            
            # Compute subgradient of the overlap associated with k_max (full convolution contains all shifts)
            g = np.zeros(n)
            sign = 1 if k_max > 0 else -1  # direction
            
            # constructive subgradient computation
            if k_max > 0:
                # positive shift: h_current[i] * (1 - h_current[i + k_max])
                # find gradient for each i:
                # d/dh_i [ h_i * (1 - h_{i+k}) ] = (1 - h_{i+k}) if i+k < n
                # d/dh_i [ h_{i+k} * (1 - h_i) ] = -h_{i+k} if i-k >= 0
                g[:n-k_max] = (1 - h_current[k_max:])  # gradient term from first part
                g[k_max:] -= h_current[:-k_max]  # from second part; note: flowed right k_max positions
            else:
                # negative shift
                g[-k_max:] = (1 - h_current[:n+k_max])
                g[:-k_max] -= h_current[-k_max:]

            # Check norm
            g_norm = np.linalg.norm(g)
            if g_norm <= tol:
                break
            # step
            h_new = h_current - alpha * g / g_norm
            # Project onto feasible set: [0,1] box + constraint sum = 1
            # Clip and renormalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n / 2) / np.sum(h_new)
            # Mirror again if needed for symmetry
            h_current = (h_new + h_new[::-1]) / 2

            # compute exact overlap again to check improvement
            conv_full = np.correlate(h_current, 1 - h_current, mode='full')
            new_c5 = np.max(conv_full) * dx
            if new_c5 >= best_c5:
                break
            best_c5 = new_c5
            if i % 10 == 0:  # for low logging
                pass
                
        best_h_full = h_current

    # Final refinement with randomized hill climbing
    if best_h_full is not None and (time.time() - start_time) < budget_s - 10:
        h_current = best_h_full.copy()
        #_Adjust number of samples if necessary to meet base resolution condition (n=200)
        refinement_factor = int(round(300 / 200.0))
        refined_n = 300  # since problem says: prefer <1k, use 300
        dx_refine = 2.0 / refined_n
        h_refined = np.interp(np.linspace(0, 2, refined_n), 
                              np.linspace(0, 2, min(200, refined_n)), h_current)
        h_refined = h_refined * (refined_n / 2) / np.sum(h_refined)

        # Local stochastic perturbation
        num_perturb = min(6, refined_n // 20)
        for _ in range(100 if refined_n >= 200 else 20):
            if (time.time() - start_time) >= budget_s - 1:
                break
            indices = np.random.choice(refined_n, num_perturb, replace=False)
            h_trial = h_refined.copy()
            epsilon = 0.003
            for i in indices:
                delta = np.random.uniform(-epsilon, epsilon)
                h_trial[i] += delta
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (refined_n / 2) / np.sum(h_trial)
            conv = np.correlate(h_trial, 1 - h_trial, mode='full') * dx_refine
            c5_trial = np.max(conv)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                h_refined = h_trial

        best_h = h_refined
        best_n = refined_n
    else:
        best_h = best_h_full if best_h_full is not None else np.ones(n)*0.5
        best_n = n

    return (best_h, best_c5, best_n)
```
