import numpy as np
import time
from scipy.optimize import linprog
import cvxpy as cp
from scipy.signal import correlate
from scipy.fft import fft, ifft
import math

def solve_spectral_lp(n):
    """Solve the spectral convex relaxation LP for the minimum overlap problem."""
    # Set up the LP
    # Variables: p_m = |hat h_m|^2 for m = 0..n-1, and t (the lower bound to maximize)
    # Objective: maximize t
    # Constraints:
    # 1. sum(p_m) = n * n/2  (Parseval's theorem with sum h_i = n/2 and h_i <= 1)
    # 2. for k = 1..n-1: sum_m p_m * cos(2*pi*m*k/n) >= t
    # 3. 0 <= p_m <= n
    
    # Build the constraint matrix
    A = []
    b = []
    
    # Constraint: sum(p_m) = n * n/2 = n^2/2
    a_sum = np.ones(n)
    A.append(a_sum)
    b.append(n * n / 2)
    
    # Constraints: sum_m p_m * cos(2*pi*m*k/n) >= t for k=1..n-1
    for k in range(1, n):
        row = np.array([np.cos(2 * np.pi * m * k / n) for m in range(n)])
        A.append(row)
        b.append(0.0)  # will be adjusted below when t is added
    
    # Add t terms to the left-hand side of the constraints
    A = np.vstack(A)
    # add [-1] at the end of each row except the first to account for t
    A_aug = np.hstack([A[:, :n], -np.ones((A.shape[0], 1))])
    
    # Objective: maximize t, so minimize -t
    c = np.zeros(n + 1)
    c[-1] = -1
    
    # Bounds
    bounds = [(0, n)] * n + [(None, None)]  # p_m in [0, n], t unbounded
    
    # Solve as a linear program using linprog
    try:
        res = linprog(c, A_ub=-A_aug, b_ub=-np.array(b), bounds=bounds, method='highs')
        if res.success:
            p_opt = res.x[:-1]  # ignore the last element (t)
            t_opt = res.x[-1]
        else:
            # fallback to simple approach
            p_opt = np.ones(n)  # uniform power spectrum (very conservatively)
            t_opt = 0.0
    except Exception:
        # fallback if numerical issues or solver fails
        p_opt = np.ones(n)
        t_opt = 0.0
        
    return p_opt, t_opt

def gerchberg_saxton(p_spectrum, n, max_iter=200):
    """Apply Gerchberg-Saxton phase retrieval to get a feasible vector"""
    # Start with random phases
    phase = np.random.uniform(0, 2 * np.pi, n)
    # Create complex Fourier coefficients
    X = np.sqrt(p_spectrum) * np.exp(1j * phase)
    
    # Reconstruct in time-domain and project
    for it in range(max_iter):
        # Inverse transform
        x = np.real(ifft(X))
        
        # Amplitude projection: clip to [0,1] and enforce sum constraint
        y = np.clip(x, 0, 1)
        # Waterfilling correction
        current_sum = np.sum(y)
        diff = n/2 - current_sum
        
        if abs(diff) > 1e-10:
            # Scale all entries by some amount and re-clip
            # Simple solution: add constant to minimize error
            if diff > 0:
                # Adding positive constant to increase sum
                offset = diff / n
                y = np.clip(y + offset, 0, 1)
            else:
                # Subtracting negative constant to decrease sum
                offset = -diff / n
                y = np.clip(y - offset, 0, 1)
            
            # Re-adjust using bisection search to get exact sum
            # Just a simple bisection
            minval = 0
            maxval = 1
            while maxval - minval > 1e-9:
                mid = (minval + maxval) / 2
                temp = y + mid
                temp = np.clip(temp, 0, 1)
                temp_sum = np.sum(temp)
                if temp_sum > n/2:
                    maxval = mid
                else:
                    minval = mid
                y = temp.copy()
        
        # Forward transform
        Y = fft(y)
        
        # Re-project onto the target spectrum
        abs_Y = np.abs(Y)
        X = (p_spectrum / (abs_Y + 1e-12))**0.5 * np.exp(1j * np.angle(Y))
        
    return np.real(ifft(X))

def project_simplex_box(h, n):
    """Project h onto the set of vectors satisfying sum(h) = n/2, and 0<=h[i]<=1.
    Uses waterfilling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Waterfilling approach: add diff/n to each entry
    # but that may go above 1. 
    # So let's use binary search on the correct amount to add
    # By adjusting a constant offset
    
    if diff > 0:
        # Need to add: exact add_val to reach sum = n/2
        # Let's binary search how much to add
        low = 0
        high = 1 - np.min(h)
        while high - low > 1e-12:
            mid = (low + high) / 2
            h_trial = np.clip(h + mid, 0, 1)
            sum_trial = np.sum(h_trial)
            if sum_trial < n/2:
                low = mid
            else:
                high = mid
        h = np.clip(h + low, 0, 1)
    else:
        # Need to subtract
        diff = -diff
        low = 0
        high = np.max(h)
        while high - low > 1e-12:
            mid = (low + high) / 2
            h_trial = np.clip(h - mid, 0, 1)
            sum_trial = np.sum(h_trial)
            if sum_trial > n/2:
                low = mid
            else:
                high = mid
        h = np.clip(h - low, 0, 1)
    
    # ensure it's exactly sum(n/2)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    h += diff / n
    h = np.clip(h, 0, 1)
    
    return h

def binary_sharpen(h, mu=1e-2, max_iter=10):
    """Sharpen fractional values to 0 or 1 with a penalty."""
    h = h.copy()
    for _ in range(max_iter):
        # gradient
        # approximate h as real-valued
        grad_h = np.zeros_like(h)
        for i in range(len(h)):
            # gradient of sum_{j} h_i * h_{i+j} w.r.t h_i
            # (NOTE: there are more terms when inclusively considering all possible translations)
            # We estimate for the full range of k 
            # but just compute a single gradient with the worst shift
            # For polynomial gradient descent it's more efficient
            grad = 0.0
            # small loop: if cpu makes sense to invest
            # This is a bit different than standard projector along gradient
            # Just a heuristic that often improves concrete c5
            grad += h[(i+1) % len(h)]  # from j = i+1 in sum
            grad += h[i-1]  # from j = i-1
            grad_h[i] = grad
            
        # Normally for x_i = h_i (on feasible simplex)
        # The quadratic penalty is: μ * h_i * (1 - h_i) = μ * h_i - μ * h_i^2
        # Gradient is μ - 2 * μ * h_i = μ * (1 - 2 * h_i)
        # Add the gradients and compute a projected/proximal update
        
        # but since this is binary sharpening using gradient descent:
        # Apply Newton-like update based on the common cubic approximation:
        # h_i = h_i - step * (gradient + mu(1 - 2 * h_i))
        step = 0.1
        new_h = h - step * (grad_h + mu * (1 - 2 * h))
        new_h = np.clip(new_h, 0, 1)
        
        # Clip back to [0,1] and renormalize
        new_h = project_simplex_box(new_h, len(h))
        
        # sharpen to binary states  
        h = new_h.copy()
        
        # Final clipping then reproject
        h = np.clip(h, 0, 1)
        h = project_simplex_box(h, len(h))
        
    # round so that entries close to 0 or 1 are integers
    for i in range(len(h)):
        if h[i] < 0.01:
            h[i] = 0.0
        elif h[i] > 0.99:
            h[i] = 1.0
    h = project_simplex_box(h, len(h))
    return h

def gradient_worst_shift_step(h, dx, worst_k):
    """Take one projected gradient step towards reducing the overlap at the worst shift k."""
    n = len(h)
    # Compute gradient for overlap at k
    grad = np.zeros(n)
    for i in range(n):
        # Gradient of integral over i of h_i (1 - h_{i+k})
        # to get d/dh_i = (1 - h_{i+k}) - h_{i-k}
        grad[i] = dx * ((1 - h[(i+worst_k) % n]) - h[(i-worst_k) % n])

    alpha = 1.0
    max_alpha = 1.0
    # Explore acceptable alpha
    for _ in range(20):
        h_new = h - alpha * grad
        # Projected update
        h_new = project_simplex_box(h_new, n)
        
        # Evaluate new c5
        corr = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr) * dx
        
        if c5_new <= 0.3806:
            return h_new.copy(), c5_new
            
        # reduce alpha and retry
        alpha = alpha / 2.0
        if alpha < 1e-6:
            break

    # If no improvement at acceptable step sizes
    return h.copy(), evaluate_c5(h)

def swap_anneal(h, dx, time_remaining):
    """Run a stochastic swap phase for a short time."""
    n = len(h)
    start = time.time()
    T = 1e-3  # initial temperature
    decay = 0.99  # geometric decay
    
    current_c5 = evaluate_c5(h)
    
    # We'll perform at least a few iterations
    min_iter = 50
    count = 0
    
    while time.time() - start < time_remaining and count < min_iter:
        i, j = np.random.choice(n, 2, replace=False)
        # transfer amount
        delta = np.random.uniform(0, min(h[i], 1 - h[j], 0.02))
        # make sure to stay within bounds
        old_i = h[i]
        old_j = h[j]
        
        h[i] = h[i] - delta
        h[j] = h[j] + delta
        h = project_simplex_box(h, n)
        
        new_c5 = evaluate_c5(h)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
        else:
            # Metropolis criterion
            p_accept = np.exp(-(new_c5 - current_c5) / T)
            if np.random.rand() < p_accept:
                current_c5 = new_c5
            else:
                # revert change
                h[i] = old_i
                h[j] = old_j
                
        count += 1
        T *= decay
        
    return h, current_c5

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 650
    dx = 2.0 / n_points
    time_left = budget_s - (time.time() - start_time)
    
    # Stage 1: Spectral relaxation
    p_opt, t_opt = solve_spectral_lp(n_points)
    c5_relax = 2 * (1 - t_opt) / n_points
    if c5_relax >= 0.3807:
        # The bound is too loose, drift to a better solution
        pass
        
    # Stage 2: Phase retrieval via Gerchberg-Saxton
    h_gs = gerchberg_saxton(p_opt, n_points)

    # Stage 3: Feasible projection and binary sharpening
    h_feas = project_simplex_box(h_gs, n_points)
    h_binary = binary_sharpen(h_feas, 1e-3, 5)
    
    # Stage 4: Local worst-shift refinement
    current_h = h_binary.copy()
    c5_val = evaluate_c5(current_h)
    ref_time = budget_s * 0.3  # approximately 30% of the time is used for this part
    early_stop = False
        
    # repeatedly find the worst shift then reduce it
    for i in range(10000):  # limited for safety 
        if time.time() - start_time > budget_s - 0.1:
            break
            
        corr_full = correlate(current_h, 1 - current_h, mode='full')
        # find the index of max value (i.e., worst shift)
        worst_k_idx = np.argmax(corr_full)
        worst_k = worst_k_idx - (n_points - 1)
        
        # take step to reduce overlap at that shift
        new_h, new_c5 = gradient_worst_shift_step(current_h, dx, worst_k)
        if new_c5 < c5_val:
            current_h = new_h
            c5_val = new_c5
        else:
            current_h = new_h
            c5_val = new_c5
            
    # Final time allocation: swap subtle improvements
    time_left = budget_s - (time.time() - start_time)
    if time_left > 20:
        current_h, c5_val = swap_anneal(current_h, dx, time_left)

    # Final project to ensure constraint is met exactly 
    final_h = project_simplex_box(current_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
