```python
import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.ndimage import shift
import time

def c5_of_h(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution
    conv = np.correlate(h, g, mode='full')
    # The full convolution has 2*len(h) - 1 elements
    result = np.max(conv) * dx
    return result

def soft_max_conv(h, beta=1.0):
    """Compute soft max of convolution with parameter beta"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    exp_conv = np.exp(beta * dx * conv)
    return np.log(np.sum(exp_conv)) / beta

def grad_soft_max_conv(h, beta=1.0):
    """Compute gradient of soft max of convolution"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    exp_conv = np.exp(beta * dx * conv)
    weights = exp_conv / np.sum(exp_conv)
    
    # Gradient computation
    grad = np.zeros_like(h)
    n = len(h)
    for i in range(n):
        # For each shift k, contribution to grad[i]
        for k in range(-n+1, n):
            if k >= 0:
                j = i + k
                if 0 <= j < n:
                    grad[i] += weights[k + n - 1] * (1 - h[j])
            else:
                j = i + k
                if 0 <= j < n:
                    grad[i] += weights[-k + n - 1] * (-h[j])
    return dx * grad * beta

def project_sum(h, target_sum):
    """Project h onto sum = target_sum while keeping in [0,1]"""
    current_sum = np.sum(h)
    diff = current_sum - target_sum
    if abs(diff) < 1e-9:
        return h
    delta = diff / len(h)
    h_new = h - delta
    h_new = np.clip(h_new, 0, 1)
    # Adjust via water-filling if necessary
    actual_sum = np.sum(h_new)
    diff2 = target_sum - actual_sum
    if abs(diff2) > 1e-9:
        # Add evenly to elements that aren't already at limit
        indices = np.where((h_new > 0) & (h_new < 1))[0]
        if len(indices) > 0:
            # Add diff2 evenly to these
            add_per = diff2 / len(indices)
            for i in indices:
                h_new[i] += add_per
                h_new[i] = np.clip(h_new[i], 0, 1)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def decode_blocks(params):
        # 12 parameters: 4 blocks with (start, length, height) each
        M = 4
        starts = params[0:M]
        lengths = params[M:2*M]
        heights = params[2*M:3*M]
        
        # Normalize lengths to be in [0,2]
        # Ensure starts are valid in [0,2-lengths]
        lengths = np.clip(lengths, 0, 2)
        starts = np.clip(starts, 0, 2 - lengths)
        
        # For ordering 
        combined = list(zip(starts, lengths, heights))
        combined.sort(key=lambda x: x[0])
        starts, lengths, heights = zip(*combined)
        
        # Reconstruct h
        h = np.zeros(n_points)
        # Convert start and lengths to int indices
        for i in range(M):
            length = lengths[i]
            if length <= 0:
                continue
            start = starts[i]
            start_idx = int(start / dx)
            end_idx = min(start_idx + int(length / dx) + 1, n_points)
            h[start_idx:end_idx] = heights[i]
        
        # Normalize to have sum = target_sum (equivalent to integral = 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        else:
            h = np.ones(n_points) * 0.5
            h = h * (target_sum / np.sum(h))
        return h
    
    def objective(params):
        h = decode_blocks(params)
        if time.time() - start_time > budget_s - 1:
            return float('inf')
        return c5_of_h(h)
    
    # DE parameters
    bounds = [(0, 2)] * 4 + [(0, 2)] * 4 + [(0, 1)] * 4  # start, length, height for 4 blocks
    de_options = {'popsize': 15, 'maxiter': 60, 'tol': 1e-5, 'recombination': 0.7, 'mutation': (0.5, 1)}
    de_result = differential_evolution(objective, bounds, strategy='best1bin', seed=seed, **de_options)
    h_global = decode_blocks(de_result.x)
    h_global = (h_global + h_global[::-1]) / 2  # Symmetrize
    h_global = project_sum(h_global, target_sum)
    c5_global = c5_of_h(h_global)
    best_h = h_global.copy()
    best_c5 = c5_global

    # Gradient refinement with smooth max surrogate
    def soft_max_obj(h):
        return soft_max_conv(h, beta=0.5)
    
    def soft_max_obj_der(h):
        return grad_soft_max_conv(h, beta=0.5)
    
    def penalty_obj(h):
        return soft_max_obj(h) + 10.0 * (np.sum(h) - target_sum)**2

    def penalty_grad(h):
        g = soft_max_obj_der(h)
        g += 20.0 * (np.sum(h) - target_sum)
        return g

    # Refinement using L-BFGS-B
    # Multi-scale refinement
    beta_current = 0.5
    lambda_penalty = 10.0
    for _ in range(5):  # multiple stages of increasing beta
        beta_current *= 2.0
        lambda_penalty *= 2.0
        
        def smooth_obj(h):
            dx = 2.0 / len(h)
            # Gradient computation using soft max
            conv = np.correlate(h, 1.0 - h, mode='full')
            exp_conv = np.exp(beta_current * dx * conv)
            return np.log(np.sum(exp_conv)) / beta_current
        
        def smooth_obj_grad(h):
            dx = 2.0 / len(h)
            g = 1.0 - h
            conv = np.correlate(h, g, mode='full')
            exp_conv = np.exp(beta_current * dx * conv)
            weights = exp_conv / np.sum(exp_conv)
            
            grad = np.zeros_like(h)
            n = len(h)
            for i in range(n):
                for k in range(-n+1, n):
                    if k >= 0:
                        j = i + k
                        if 0 <= j < n:
                            grad[i] += weights[k + n - 1] * (1 - h[j])
                    else:
                        j = i + k
                        if 0 <= j < n:
                            grad[i] += weights[-k + n - 1] * (-h[j])
            return dx * grad * beta_current
            
        lbfgs_options = {'maxiter': 150, 'gtol': 1e-6, 'ftol': 1e-9, 'maxcor': 10}
        
        # L-BFGS from a warm start
        proj_h = project_sum(h_global, target_sum)
        try:
            result = minimize(smooth_obj, proj_h, method='L-BFGS-B', jac=smooth_obj_grad, 
                              bounds=[(0,1)]*n_points, options=lbfgs_options)
            h_best = result.x
            h_best = project_sum(h_best, target_sum)
            h_global = h_best
            c5_refined = c5_of_h(h_global)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_global.copy()
        except:
            pass # continue with current best if failed
    
    # Local stochastic hill-climbing
    h_refined = best_h.copy()
    c5_refined = best_c5
    idx_hill = 0
    while time.time() - start_time < budget_s - 0.5:
        r = min(5, n_points // 20)
        perturb_indices = np.random.choice(n_points, size=r, replace=False)
        h_trial = h_refined.copy()
        for i in perturb_indices:
            pert = np.random.uniform(-0.005, 0.005)
            h_trial[i] = min(1.0, max(0.0, h_trial[i] + pert))
        
        h_trial = project_sum(h_trial, target_sum)
        c5_trial = c5_of_h(h_trial)
        if c5_trial < c5_refined:
            h_refined = h_trial
            c5_refined = c5_trial
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
        
        idx_hill += 1

    return (best_h, best_c5, n_points)
```
