def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def evaluate_erdos_solution(h_values, n_points):
    """Given a solution, return the C5 constant."""
    h = np.array(h_values)
    return c5_of_h(h)

#-------- Helper functions for optimization phases --------

def project_to_feasible(Y):
    """Projects the rank-r factor Y onto the feasible set of h = ||Y_i||^2 
       with sum(h) = n/2 and 0 <= h_i <= 1"""
    n = Y.shape[0]
    # Compute h = norm^2
    h = np.sum(Y**2, axis=1)
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # Enforce mass constraint
    S = np.sum(h)
    if S > 0:
        alpha = np.sqrt((n/2) / S)
        Y = Y * alpha
        h = np.sum(Y**2, axis=1)
        h = np.clip(h, 0, 1)
    return Y, h

def compute_correlations_fft(Y):
    """Compute all c_k = sum h_i h_{i+k} for all k via FFT"""
    n, r = Y.shape
    c_k = np.zeros(n)
    for d in range(r):
        y_d = Y[:, d]
        # Circular convolution y_d * reverse(y_d)  
        # with zero-padding to 2n for correct circular convolution
        fft_len = 2 * n
        y_fft = np.fft.fft(y_d, fft_len)
        c_fft = np.fft.fft(y_d, fft_len) * np.conj(y_fft)
        c_d = np.fft.ifft(c_fft).real[:n]
        c_k += c_d
    return c_k

def lse_smooth(Y, c_k, tau):
    """Smooth lower bound log-sum-exp on min_k c_k"""
    exp_neg = np.exp(-c_k / tau)
    Z = np.sum(exp_neg)
    phi = -tau * np.log(Z)
    return phi

def gradient_lse(Y, c_k, tau):
    """Gradient of log-sum-exp smooth approx to min_k c_k"""
    exp_neg = np.exp(-c_k / tau)
    Z = np.sum(exp_neg)
    p = exp_neg / Z
    # Derivative is a weighted sum of 
    # Y_{i+k} + Y_{i-k} for each k controling p_k
    # We'll precompute these contributions
    n, r = Y.shape
    grad = np.zeros_like(Y)
    # Precompute the weights
    # For each column d,
    #   sum over k of p_k * (Y[d,i+k] + Y[d,i-k])
    for d in range(r):
        y_d = Y[:, d]
        # We compute: k convols with `y_d` and `-y_d`
        # Term for Y_{i+k} with weights p_k: convolve p with y_d
        p_pad = np.concatenate([p, np.zeros(n)])
        y_pad = np.concatenate([y_d, np.zeros(n)])
        conv_plus = np.fft.irfft(np.fft.rfft(p_pad) * np.fft.rfft(y_pad))[:n]
        # And the minus term: convolve p with reverse(y_d) 
        y_rev = y_d[::-1]
        y_rev_pad = np.concatenate([y_rev, np.zeros(n)])
        conv_minus = np.fft.irfft(np.fft.rfft(p_pad) * np.fft.rfft(y_rev_pad))[:n]
        # Now add weights
        grad[:, d] = conv_plus + conv_minus
    
    return grad

            best_c5 = current_c5

        best_c5 = c5_of_h(best_h)

    # Now try to polish the binary solution using deterministic swaps
    if best_h is not None and time.time() - start_time < max_time - 5:
        h = best_h.copy()
        if np.all(np.abs(h - np.round(h)) < 1e-10):
            # Already binary
            h_binary = h
        else:
            # Round to binary
            idxs = np.argsort(h)[::-1]

        max_iter_swaps = 300   # Fixed upper bound

        upsampled_h = upsampled_h * (n_points / 2) / (np.sum(upsampled_h) * 2.0 / (2 * n_points))

            # Sub-gradient
            step_size = 0.01 * dx
            subgrad = np.zeros_like(best_h)
            for i in range(len(best_h)):
                # Define sub-gradient associated with shift k
                i_plus_k = (i + max_k) % len(best_h)
                i_minus_k = (i - max_k) % len(best_h)
                subgrad[i] = (1 - best_h[i_plus_k]) - best_h[i_minus_k]
            # Apply gradient step and project
            best_h = best_h - step_size * subgrad
            # Project back
            best_h = np.clip(best_h, 0, 1)
            # Renormalize
            total = np.sum(best_h)
            best_h = best_h * (n_points / 2) / total
            best_c5 = c5_of_h(best_h)

    return (best_h, best_c5, len(best_h))
