import numpy as np
import time
from scipy.signal import fftconvolve
from scipy.optimize import minimize
import math

def evaluate_h(h, dx):
    """Compute C5 for given h using FFT-based convolution."""
    g = 1 - h
    corr = fftconvolve(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def update_convolution(corr, h, i1, i0, old_h):
    """Update convolution with a swap of indices i1 and i0."""
    n = len(h)
    # Get the current values
    v1_new = h[i1]
    v0_new = h[i0] 
    v1_old = old_h[i1]
    v0_old = old_h[i0]
    
    # Compute changes to the convolution
    delta = np.zeros(2*n - 1)
    
    # Contribution from i1 swaps
    if v1_new != v1_old:
        shift1 = i1
        # Need to compute (1-h) shifted by i1
        if v1_new == 1:
            # New: 1, Old: 0
            delta[shift1:shift1 + n] += (1 - h)  # Add (1-h) shifted by i1
        else:
            # New: 0, Old: 1
            delta[shift1:shift1 + n] -= (1 - h)  # Subtract (1-h) shifted by i1
            
    # Contribution from i0 swaps
    if v0_new != v0_old:
        shift0 = i0
        if v0_new == 1:
            # New: 1, Old: 0
            delta[shift0:shift0 + n] += (1 - h)  # Add (1-h) shifted by i0
        else:
            # New: 0, Old: 1
            delta[shift0:shift0 + n] -= (1 - h)  # Subtract (1-h) shifted by i0
            
    # So we have: new_corr = old_corr + delta
    new_corr = corr + delta
    return new_corr

def binary_to_step_function(binary_h, n_points):
    """Convert binary vector to continuous step function."""
    # Create dense representation with the piece-wise constant intervals
    step_func = np.zeros(n_points)
    dx = 2.0 / n_points
    # Find contiguous runs of 1s
    i = 0
    while i < len(binary_h):
        if binary_h[i] == 1:
            j = i
            while j < len(binary_h) and binary_h[j] == 1:
                j += 1
            # Set interval [i*dx, j*dx) to 1  
            start_idx = int(np.floor(i * 2.0 / len(binary_h) * n_points))
            end_idx = int(np.ceil(j * 2.0 / len(binary_h) * n_points))
            step_func[start_idx:end_idx] = 1
            i = j
        else:
            i += 1
    return step_func

def get_random_binary_vector(n):
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.uint8)
    h[:n//2] = 1
    np.random.shuffle(h)
    return h

def get_alternating_pattern(n):
    """Create alternating pattern."""
    h = np.zeros(n, dtype=np.uint8)
    h[::2] = 1
    return h

def get_two_block_pattern(n):
    """Create two block pattern with random shuffle."""
    h = np.zeros(n, dtype=np.uint8)
    h[:n//2] = 1
    np.random.shuffle(h)
    return h

def get_initial_h_from_interpolated(n):

    n_orig = len(h_orig)
    # Interpolate onto n points  
    x_orig = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n)
    h_interp = np.interp(x_new, x_orig, h_orig)
    # Bin the continuous values and scale sum
    h_binary = (h_interp >= 0.5).astype(np.uint8)
    # Fix sum to exactly n/2
    current_sum = np.sum(h_binary)
    diff = n//2 - current_sum
    if diff > 0:
        # Need to add more 1s
        positions = np.where(h_binary == 0)[0]
        indices = np.random.choice(positions, diff, replace=False)
        h_binary[indices] = 1
    elif diff < 0:
        # Need to remove some 1s  
        positions = np.where(h_binary == 1)[0]
        indices = np.random.choice(positions, -diff, replace=False)
        h_binary[indices] = 0
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Grid sizes
    n_list = [220, 260, 300]
    dx_list = [2.0/n for n in n_list]
    
    # Parameters
    max_iter = int(1e6)
    temp_schedule = lambda T, alpha=0.9995: T * alpha
    temp_start = 0.05
    temp_low = 1e-5
    swap_interval = 50
    num_chains = 4
    max_swaps_no_improve = 50000
    
    for n_idx, n in enumerate(n_list):
        dx = dx_list[n_idx] 
        target_sum = n // 2
        
        def is_valid(h):
            return np.sum(h) == target_sum
            
        # Helper to compute correlation using good precision
        def compute_convolution(h, dx):
            g = 1 - h
            corr = fftconvolve(h, g, mode='full')
            return corr
        
        # Generate initial configurations
        initial_configs = []
        initial_configs.append(get_initial_h_from_interpolated(n))
        initial_configs.append(get_random_binary_vector(n))
        initial_configs.append(get_alternating_pattern(n))
        initial_configs.append(get_two_block_pattern(n))
        
        # Simulated Annealing with Parallel Tempering
        num_swaps = 0
        best_c5_for_n = float('inf')
        best_h_for_n = None
        
        # Start several chains
        current_states = []
        temps = [temp_start * (0.7 ** i) for i in range(num_chains)]
        
        # Initialize all chains
        for i in range(num_chains):
            cfg = initial_configs[i % len(initial_configs)]
            current_states.append(cfg.copy())
            
        # Track best for this n
        t_total = 0
        
        while num_swaps < max_iter and time.time() - start_time < budget_s - 5:
            proceed = True
            t_total += 1
            
            # Select a chain
            chain_idx = np.random.randint(0, num_chains)
            h = current_states[chain_idx]
            old_corr = compute_convolution(h, dx)
            old_c5 = np.max(old_corr) * dx
            
            # Attempt a swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                continue
            
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            
            # Make swap
            h_new = h.copy()
            h_new[i1] = 0
            h_new[i0] = 1
            
            new_corr = update_convolution(old_corr, h_new, i1, i0, h)
            new_c5 = np.max(new_corr) * dx
            
            # Accept with Metropolis probability
            T = temps[chain_idx]
            if new_c5 <= old_c5 or np.random.rand() < np.exp(-(new_c5 - old_c5) / T):
                current_states[chain_idx] = h_new
                num_swaps += 1
                
                # Update best for this n
                if new_c5 < best_c5_for_n:
                    best_c5_for_n = new_c5
                    best_h_for_n = h_new.copy()
                    
                # Early stopping
                if new_c5 < 0.3807:  
                    break
                    
            # Temperature update - after each "sweep"
            if t_total % (n // 2) == 0:
                # Update temperature
                temps[chain_idx] = temp_schedule(temps[chain_idx])
                if temps[chain_idx] < temp_low:
                    # Restart with new config
                    current_states[chain_idx] = initial_configs[np.random.randint(0, len(initial_configs))]
                    temps[chain_idx] = temp_start
                    
            # Exchange neighboring temperatures
            if t_total % swap_interval == 0:
                for i in range(num_chains - 1):
                    if np.random.rand() < 0.1:
                        # Metropolis acceptance for exchange
                        T1, T2 = temps[i], temps[i+1]
                        h1, h2 = current_states[i], current_states[i+1]
                        
                        c1 = np.max(compute_convolution(h1, dx))
                        c2 = np.max(compute_convolution(h2, dx))  
                        
                        delta = (c1/T1 - c2/T2) - (c2/T2 - c1/T1)
                        if delta < 0 or np.random.rand() < np.exp(-delta):
                            temps[i], temps[i+1] = temps[i+1], temps[i]
                            current_states[i], current_states[i+1] = current_states[i+1], current_states[i]
        
        # Finish this n size if it's not already beaten
        if best_c5_for_n < best_c5:
            best_c5 = best_c5_for_n
            best_h = best_h_for_n
            best_n = n
            if best_c5 < 0.3807:
                break
                
    # Local deterministic improvement around best
    if best_h is not None:
        # Try to improve deterministicly
        h_current = best_h.copy()
        improved = True
        iterations = 0
        max_iter_local = 10000
        while improved and iterations < max_iter_local and time.time() - start_time < budget_s - 2:
            improved = False
            corr_current = compute_convolution(h_current, dx)
            current_max = np.max(corr_current)
            
            # Try all possible 1-0 positions 
            swaps = []
            ones = np.where(h_current == 1)[0]
            zeros = np.where(h_current == 0)[0]
            
            for i1 in ones:
                for i0 in zeros:
                    # Check if a swap improves the max correlation
                    test_h = h_current.copy()
                    test_h[i1] = 0
                    test_h[i0] = 1
                    test_corr = update_convolution(corr_current, test_h, i1, i0, h_current)
                    test_max = np.max(test_corr)
                    if test_max < current_max:
                        swaps.append((i1, i0, test_max))
            
            if swaps:
                best_swap = min(swaps, key=lambda x: x[2])
                h_current[best_swap[0]] = 0
                h_current[best_swap[1]] = 1
                improved = True
            iterations += 1
            
        # Re-evaluate final state
        corr_final = compute_convolution(h_current, dx)
        final_c5 = np.max(corr_final) * dx  
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_current
            
    return (best_h, best_c5, best_n)
