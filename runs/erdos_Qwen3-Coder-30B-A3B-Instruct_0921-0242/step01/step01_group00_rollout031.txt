import numpy as np
import time
from cvxpy import *
import cvxpy as cp
from scipy.signal import correlate
from scipy.linalg import eigh

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum

    if abs(diff) < 1e-10:
        return h

    # Water-filling adjustment
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def solve_sdp(n, dx):
    h = Variable(n)
    t = Variable()
    X = Variable((n, n), symmetric=True)
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n / 2,
        X >> 0,  # positive semidefinite
        X[range(n), range(n)] == h  # diagonal constraint
    ]
    
    # Add correlation constraints
    for k in range(1, n):
        # Compute the k-shifted correlation sum
        # For shift k, we sum h_i * h_{i+k} for i = 0..n-k-1
        # This is equivalent to summing the diagonals of X with offset k
        
        # Actually, let's use a more direct formulation via matrix trace
        # T_k is a matrix with 1s on (i, i+k) for valid i
        # trace(T_k * X) gives the sum of correlations            
        I = np.zeros((n, n))
        for i in range(n - k):
            I[i, i + k] = 1
        
        constraints.append(trace(I @ X) >= t)
    
    # Objective
    objective = Maximize(t)
    
    prob = Problem(objective, constraints)
    prob.solve(solver=SCS, eps=1e-4, max_it=5000)
    
    if prob.status == 'optimal':
        h_sdp = h.value
        return h_sdp
    else:
        # Fallback to initial construction if SDP fails
        return initial_h_values[:n]

def randomized_rounding(h_sdp, n, R=30):
    # Assuming rank-1 structure via SVD trick
    X = np.outer(h_sdp, h_sdp)
    # Add small perturbation to ensure full rank
    # In real SDP we'd have X = V V^T properly
    V, s, _ = np.linalg.svd(X)
    V = V[:, :len(h_sdp)]  # keep relevant columns

    best_h = None
    best_c5 = float('inf')
    
    for _ in range(R):
        g = np.random.randn(len(h_sdp))
        y = V @ g
        # Sort y
        srt = np.sort(y)
        # Select middle elements to get exactly n/2 ones
        idx = np.argsort(-y)  # indices that sort in descending order
        num_ones = int(n / 2)
        indices = idx[:num_ones]
        b = np.zeros(n)
        b[indices] = 1
        b = project_to_feasible(b, n)
        c5 = evaluate_c5(b)
        if c5 < best_c5:
            best_c5 = c5
            best_h = b
            
    return best_h, best_c5

def smooth_max_grad_refinement(h_start, n, dx, budget_s, start_time):
    # Define smooth max with β
    def smooth_max(h, beta):
        # Get I_k values for all k = 1..n-1
        I_k = [0.0] * (n - 1)
        for k in range(1, n):
            s = 0.0
            for i in range(n - k):
                s += h[i] * h[i + k]
            I_k[k-1] = dx * (1.0 - s)
        
        # Ensure all I_k are negative so the log-sum-exp is valid
        # But keep them consistent with original
        # We compute the smooth max directly
        exp_terms = [np.exp(beta * I_kk) for I_kk in I_k]
        numerator = sum(exp_terms)
        if numerator == 0:
            expected = 0
        else:
            expected = (1.0 / beta) * np.log(numerator)
        return expected
    
    def compute_grad(h, beta):
        I_k = [0.0] * (n - 1)
        for k in range(1, n):
            s = 0.0
            for i in range(n - k):
                s += h[i] * h[i + k]
            I_k[k-1] = dx * (1.0 - s)
        
        exp_terms = [np.exp(beta * I_kk) for I_kk in I_k]
        Z = sum(exp_terms)
        weights = [exp_term / Z for exp_term in exp_terms]

        grad = np.zeros(n)
        for k in range(1, n):
            weights_k = weights[k-1]
            for i in range(n - k):
                # Derivative of I_k w.r.t. h_i is:
                # dx * ( (1 - h_{i+k}) - h_{i-k} )
                # But h_{i-k} is zero for negative indices
                grad[i] += dx * weights_k * (1.0 - h[i + k])
                grad[i + k] += dx * weights_k * (- h[i])
        return grad
    
    h = h_start.copy()
    beta = 0.5
    max_iter = 2000
    beta_max = 50.0
    beta_factor = 1.5
    step_size = 0.01
    inner_iter = 50
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 10.0:
            break
            
        grad = compute_grad(h, beta)
        h_new = h - step_size * grad
        
        # Project to feasibility
        h_new = project_to_feasible(h_new, n)
        
        # Accept if better
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Increase beta occasionally
        if _ % 100 == 0 and beta < beta_max:
            beta *= beta_factor
            # Recompute grad with new beta
            grad = compute_grad(h, beta)
    
    return best_h, best_c5

def discrete_local_search(h, n, dx, budget_s, start_time):
    # Try small swaps
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    max_swaps = 2000
    for _ in range(max_swaps):
        if time.time() - start_time > budget_s - 5.0:
            break
            
        i, j = np.random.choice(n, 2, replace=False)
        if h[i] > 0 and h[j] < 1:
            delta = min(h[i], 1 - h[j], 0.02)
            # Compute approximate change dC5
            # Hook up to overlapping neighbors
            old_c5 = evaluate_c5(best_h)
            h_trial = best_h.copy()
            
            # Update erosion effect (shifted versions)
            h_trial[i] -= delta
            h_trial[j] += delta
            h_trial = project_to_feasible(h_trial, n)
            new_c5 = evaluate_c5(h_trial)
            
            if new_c5 < best_c5:
                best_h = h_trial.copy()
                best_c5 = new_c5
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 128
    dx = 2.0 / n
    best_c5 = float('inf')
    best_h = None
    
    # Stage 1: SDP relaxation
    if time.time() - start_time < budget_s - 40:
        h_sdp = solve_sdp(n, dx)
        h_sdp = project_to_feasible(h_sdp, n)
        c5_sdp = evaluate_c5(h_sdp)
        if c5_sdp < best_c5:
            best_c5 = c5_sdp
            best_h = h_sdp.copy()
    
    # Stage 2: Rounding
    if time.time() - start_time < budget_s - 30:
        h_rounded, c5_rounded = randomized_rounding(h_sdp, n)
        if c5_rounded < best_c5:
            best_c5 = c5_rounded
            best_h = h_rounded.copy()
    
    # Stage 3: Smooth-max refinement
    if time.time() - start_time < budget_s - 10:
        h_refined, c5_refined = smooth_max_grad_refinement(best_h, n, dx, budget_s, start_time)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Stage 4: Final discrete local search
    if time.time() - start_time < budget_s - 5:
        h_final, c5_final = discrete_local_search(best_h, n, dx, budget_s, start_time)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Final:
    if best_h is None:
        best_h = np.ones(n) * 0.5
        best_c5 = evaluate_c5(best_h)
        
    return (best_h, best_c5, n)
