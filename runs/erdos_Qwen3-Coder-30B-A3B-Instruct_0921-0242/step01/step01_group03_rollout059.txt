```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.interpolate import interp1d
from scipy.linalg import norm
from scipy.fft import ifft, fft
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    N = 400  # grid size
    r = 6    # rank for low-rank factorization
    dx = 2.0 / N
    tau0 = 0.05  # initial temperature for LSE surrogate
    rho = 0.5    # temperature reduction factor
    max_iter_per_tau = 200  # iterations per temperature stage
    
    # Precompute FFT helpers (for circular convolution)
    fft_len = 2 * N
    
    def compute_correlations(Y):
        # Y is N x r
        c = np.zeros(N)
        for d in range(r):
            y_d = Y[:, d]
            y_d_conj = np.conj(fft(y_d, fft_len))
            corr_d = np.real(ifft(y_d_conj * y_d_conj)).real[:N]
            c += corr_d
        return c
    
    def lse_surrogate(c, tau):
        # log-sum-exp smooth lower bound
        return -tau * np.log(np.sum(np.exp(-c / tau)))
    
    def gradient_lse(c, Y, tau):
        # Gradient of LSE surrogate, using FFT for efficiency
        p = np.exp(-c / tau)
        p = p / np.sum(p)
        
        grad = np.zeros_like(Y)
        for d in range(r):
            y_d = Y[:, d]
            y_d_conj = np.conj(fft(y_d, fft_len))
            p_fft = fft(p, fft_len)
            conv1 = np.real(ifft(p_fft * y_d_conj)).real[:N]
            conv2 = np.real(ifft(y_d_conj * p_fft)).real[:N]
            # Combine the two convolutions (symmetric sum)
            grad[:, d] = conv1 + conv2
        return grad
    
    def project_to_feasible(Y):
        # Project Y to feasible set
        h = np.sum(Y**2, axis=1)
        # Clip norms to [0, 1] for diagonal entries
        norms = np.sqrt(np.maximum(h, 0))
        # Clip norms that are > 1
        for i in range(N):
            if norms[i] > 1.0:
                Y[i, :] /= norms[i]
        h = np.sum(Y**2, axis=1)
        # Enforce mass constraint
        total = np.sum(h)
        if total > 0:
            alpha = np.sqrt((N / 2) / total)
            Y *= alpha
        # Re-clip norms after scaling
        h = np.sum(Y**2, axis=1)
        norms = np.sqrt(np.maximum(h, 0))
        for i in range(N):
            if norms[i] > 1.0:
                Y[i, :] /= norms[i]
        return Y
    
    # Optimization
    best_c5 = np.inf
    best_h_continuous = None
    best_h_binary = None
    
    # Try multiple random restarts
    num_restarts = 4
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 5.0:
            break
        Y = np.random.randn(N, r)
        Y = project_to_feasible(Y)
        
        current_tau = tau0
        for stage in range(5):
            if time.time() - start_time > budget_s - 5.0:
                break
            for iteration in range(max_iter_per_tau):
                if time.time() - start_time > budget_s - 5.0:
                    break
                c = compute_correlations(Y)
                phi = lse_surrogate(c, current_tau)
                grad = gradient_lse(c, Y, current_tau)
                # Line search
                step_size = 1.0
                Y_test = Y - step_size * grad
                Y_test = project_to_feasible(Y_test)
                c_test = compute_correlations(Y_test)
                phi_test = lse_surrogate(c_test, current_tau)
                # Backtracking line search
                while phi_test < phi:
                    step_size *= 0.5
                    Y_test = Y - step_size * grad
                    Y_test = project_to_feasible(Y_test)
                    c_test = compute_correlations(Y_test)
                    phi_test = lse_surrogate(c_test, current_tau)
                    if step_size < 1e-10:
                        break
                Y = Y_test
            current_tau *= rho
    
    # Extract the final h
    h_continuous = np.sum(Y**2, axis=1)
    h_continuous = np.clip(h_continuous, 0, 1)
    # Normalize sum to N/2
    h_continuous = h_continuous * (N / 2) / np.sum(h_continuous)
    
    # Convert to binary mask
    h_sorted = np.argsort(h_continuous)[::-1]
    h_binary = np.zeros(N)
    h_binary[h_sorted[:N//2]] = 1.0
    
    def c5_of_h(h):
        # Compute the maximum overlap integral C5 for a given step function h
        dx = 2.0 / len(h)
        g = 1.0 - h
        # Full correlation: this computes all shifts
        corr = np.correlate(h, g, mode='full') * dx
        # The correlation is of length 2*n - 1, centered at index n-1
        # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
        # We want the maximum over all k, which is max of correlation values
        return np.max(corr)
    
    # Evaluate C5 for both continuous and binary
    c5_continuous = c5_of_h(h_continuous)
    c5_binary = c5_of_h(h_binary)
    
    if c5_continuous < c5_binary:
        best_h = h_continuous
        best_c5 = c5_continuous
    else:
        best_h = h_binary
        best_c5 = c5_binary
    
    # Post-processing: improve binary mask with swaps
    if best_c5 > 0.379:
        def update_correlations(h):
            c = np.zeros(N)
            for i in range(N):
                for k in range(N):
                    c[k] += h[i] * h[(i + k) % N]
            return c
            
        def compute_correlation_diff(h, i, j, k):
            # Compute change in correlation when swapping h[i] and h[j]
            delta = 0
            for shift in range(N):
                i_val = h[i]
                j_val = h[j]
                i_k = h[(i + shift) % N]
                j_k = h[(j + shift) % N]
                delta += i_val * j_k + j_val * i_k
                delta -= i_k * j_k + i_val * i_k
            return delta
            
        # deterministic swap polishing
        max_iter_swaps = 1000
        iter_count = 0
        max_iter_total = 10000
        
        while iter_count < max_iter_swaps and time.time() - start_time < budget_s - 10.0:
            if iter_count > max_iter_total or time.time() - start_time > budget_s - 10.0:
                break
            # Find the maximum overlap shift
            c = update_correlations(best_h)
            max_k = np.argmax(c)
            max_k_corr = c[max_k]
            
            # Find positions that are 1 but have 0 at shiftmax, and 0 at shiftmax but 1 currently
            bad_i = []
            bad_j = []
            for i in range(len(best_h)):
                if best_h[i] == 1 and best_h[(i + max_k) % N] == 0:
                    bad_i.append(i)
                elif best_h[i] == 0 and best_h[(i + max_k) % N] == 1:
                    bad_j.append(i)
            
            if len(bad_i) == 0 or len(bad_j) == 0:
                break
                
            # Try swapping pairs
            best_change = 0
            best_i, best_j = None, None
            for i in bad_i:
                for j in bad_j:
                    if (i == j): continue
                    change = compute_correlation_diff(best_h, i, j, max_k)
                    if change < best_change:
                        best_change = change
                        best_i, best_j = i, j
            
            if best_i is not None and best_change < -0.0001:
                # swap
                best_h[best_i] = 0.0
                best_h[best_j] = 1.0
                iter_count += 1
            else:
                break

        # recompute C5 for improved version
        c5_improved = c5_of_h(best_h)
        if c5_improved < best_c5:
            best_c5 = c5_improved
    
    # Final upsampling and finer optimization
    if time.time() - start_time < budget_s - 5.0:
        # interpolate to N' = 800
        n_points_new = 800
        h_upsampled = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, N), best_h)
        # re-normalize to sum N/2
        h_upsampled = h_upsampled * (n_points_new / 2) / np.sum(h_upsampled)
        dx_new = 2.0 / n_points_new
        
        def evaluate_h(h):
            dx = 2.0 / len(h)
            g = 1.0 - h
            corr = np.correlate(h, g, mode='full') * dx
            return np.max(corr)
            
        # Sub-gradient descent on max overlap integral
        max_iter = 30
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 2.0:
                break
            c = evaluate_h(h_upsampled)
            # find the max shift
            dx = 2.0 / len(h_upsampled)
            g = 1.0 - h_upsampled
            corr = np.correlate(h_upsampled, g, mode='full')
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (len(h_upsampled) - 1)
            # Compute sub-gradient for current max_k
            grad = np.zeros_like(h_upsampled)
            for i in range(len(h_upsampled)):
                k = max_k
                grad[i] = (1 - h_upsampled[(i + k) % len(h_upsampled)]) - h_upsampled[(i - k) % len(h_upsampled)]
            
            # take small step
            eta = 0.1 * dx_new
            h_new = h_upsampled - eta * grad
            # project to [0, 1]
            h_new = np.clip(h_new, 0, 1)
            # renormalize
            h_new = h_new * (n_points_new / 2) / np.sum(h_new)
            h_upsampled = h_new
            # update final bound
            best_c5 = evaluate_h(h_upsampled)
        
        # final result
        best_h = h_upsampled
        best_c5 = evaluate_h(h_upsampled)
        
    return (best_h, best_c5, len(best_h))
```
