import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 200  # grid size
    dx = 2.0 / n
    d = n // 2 + 1  # independent variables (due to symmetry)
    time_remaining = lambda: budget_s - (time.time() - start_time)
    
    # Initial guess: interpolate from the provided initial_h_values or start with a simple structure
    h_initial = None
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
        if len(initial_h) == n:
            h_initial = initial_h
        else:
            x_orig = np.linspace(0, 2, len(initial_h))
            x_new = np.linspace(0, 2, n)
            h_interp = interp1d(x_orig, initial_h, kind='linear', fill_value='extrapolate')
            h_initial = h_interp(x_new)
    if h_initial is None:
        # Start with a symmetric pattern
        h_initial = np.zeros(n)
        # Create symmetric pattern with two bumps
        mid = n // 2
        width = n // 4
        h_initial[mid - width//2:mid + width//2] = 1.0
        h_initial = h_initial * (n / 2) / np.sum(h_initial)
    
    # enforce symmetry
    h_initial = (h_initial + h_initial[::-1]) / 2
    h_initial = np.clip(h_initial, 0, 1)

    # Estimate # of shifts to add before convergence
    max_shifts = 20
    epsilon = 1e-6

    # Warm start with a coarse solution
    h_best = h_initial.copy()
    
    # Define symbolic progression to reduce complexity
    h_symmetric = h_initial[:d]
    h_symmetric = np.clip(h_symmetric, 0, 1)
    h_symmetric = h_symmetric * (n / 2) / np.sum(h_symmetric)
    h_full = np.concatenate([h_symmetric, h_symmetric[-2::-1]])
    
    # Symmetric indices
    indices = np.arange(d)
    # Generate independent variables for symmetric pattern
    w = cp.Variable(d)
    M = cp.Variable()
    # Initialize set of active shifts
    active_shifts = [0]
    
    # Binary penalty coefficient
    lambda_reg = 0.1
    best_c5 = float('inf')
    best_h = None
    
    # Model preparation
    # Store all p variables by shifts to be indexed cleanly
    p_vars = {}

    # Build linearized model using for loop to deal with symmetry
    while time_remaining() > 2 and len(active_shifts) <= max_shifts:
        # Prepare room for this batch of constraints and variables
        p_vars = {}
        
        # Build equality constraint for sum (dealing with symmetry correctly)
        constraint_sum = cp.sum(w) == (n / 4) + 0.5
        
        # Define left-hand side overlap expressions and linear constraints per shift k
        constraints = [w >= 0, w <= 1, constraint_sum]
        overlap_exprs = {}
        
        # Process all active shifts k
        for k in active_shifts:
            # Create p variables for this shift
            # For symmetry, only need to consider k from 0 to d-1, but also -k up to n-1 vice versa
            # Strike redundancy: use k in ]0, n//2] to avoid duplicates.
            if k > 0:
                p_vars[k] = cp.Variable((n - k,), pos=True)
            else:
                p_vars[k] = cp.Variable((n,), pos=True)
            
            # Define linear constraints
            for i in range(n - abs(k)):
                if k > 0:
                    # h_i * h_{i+k}
                    p_idx = i
                    h_i = w[i] if i < d else w[n - 1 - i]
                    h_j = w[i + k] if i + k < d else w[n - 1 - (i + k)]
                    # McCormick envelopes
                    constraints.append(p_vars[k][p_idx] <= h_i)
                    constraints.append(p_vars[k][p_idx] <= h_j)
                    constraints.append(p_vars[k][p_idx] >= 0)
                    constraints.append(p_vars[k][p_idx] >= h_i + h_j - 1)
                else: # k == 0 case for p_i,0 = h_i^2
                    p_idx = i
                    h_i = w[i] if i < d else w[n - 1 - i]
                    constraints.append(p_vars[k][p_idx] <= h_i)
                    constraints.append(p_vars[k][p_idx] <= h_i)
                    constraints.append(p_vars[k][p_idx] >= 0)
                    constraints.append(p_vars[k][p_idx] >= 2 * h_i - 1)
            
            # Overlap
            # h_i * h_{i+k} for valid ranges
            total_p = cp.sum(p_vars[k])
            constraint_overlap = (n / 2) - total_p <= M
            constraints.append(constraint_overlap)
            overlap_exprs[k] = (n / 2 - total_p)
        
        # Objective
        objective = cp.Minimize(M - lambda_reg * cp.sum(p_vars[0]))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=200)
        except:
            # Fallback or break
            break

        # Extract solution
        h_opt = w.value.copy()
        h_opt = np.clip(h_opt, 0, 1)
        h_full = np.concatenate([h_opt, h_opt[-2::-1]])
        
        # Evaluate true C5 value
        corr_values = np.correlate(h_full, 1 - h_full, mode='full')
        true_c5 = np.max(corr_values) * dx

        if true_c5 < best_c5:
            best_c5 = true_c5
            best_h = h_full.copy()
        
        # Check convergence
        if time.time() - start_time > budget_s - 0.5:
            break

        # Determine worst shift used in the model
        # Find all k in current list of active shifts that dominate one another
        # Compute overlaps for all valid shifts
        shifted_correlations = {}
        for k in range(n):
            if k <= n - 1:
                offset = k - (n - 1)  # the zero padding offset
                corr = np.correlate(h_full, np.roll(1 - h_full, k), mode='valid')
                curr_max = max(corr) * dx
                shifted_correlations[k] = curr_max
        
        k_max = max(shifted_correlations, key=shifted_correlations.get)
        rel_shifts = list(shifted_correlations.keys())
        rel_values = list(shifted_correlations.values())
        k_max = rel_shifts[np.argmax(rel_values)]
        if abs(shifted_correlations[k_max] - true_c5) < epsilon and k_max in active_shifts:
            break  # converged!

        # Add new shift to the model (just add the most violated shift)
        active_shifts.append(k_max)
        # Possibly update lambda_reg
        lambda_reg = min(5.0, lambda_reg * 1.5)
    
    # Now polish the best known solution with simpler subgradient methods
    if best_h is None:
        best_h = h_full
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx

    # Subgradient optimization
    def project_nonnegative(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        deficit = (n / 2) - total
        # Waterfilling
        if deficit != 0:
            x += deficit / n
            x = np.clip(x, 0, 1)
        return x

    h_polished = best_h.copy()
    steps = 0
    max_steps = 200
    alpha = 0.02
    last_improvement = float('inf')
    
    for step in range(max_steps):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate best k
        corr_vals = np.correlate(h_polished, 1 - h_polished, mode='full')
        k_best = np.argmax(corr_vals)
        corr_max = corr_vals[k_best]
        
        # compute grad
        grad = np.zeros_like(h_polished)
        if k_best < n:
            roll_amount = k_best
            h_shift = np.roll(h_polished, -roll_amount)
            h_shift = h_shift[:n]
            grad[:n] = 1 - h_shift
            grad += np.roll(h_polished, roll_amount)  # allow the head part to crop
            grad[n:] = grad[:-n-1:-1]  # mirrored portion
        else:
            roll_amount = n - k_best
            h_shift = np.roll(h_polished, roll_amount)
            h_shift = h_shift[n:]
            grad[n:] = 1 - h_shift
            grad += np.roll(h_polished, -roll_amount)  # avoid indexing error
            grad[:n] = grad[:-n-1:-1]  # mirrored portion

        # take projected step
        h_trial = h_polished - alpha * grad
        h_trial = project_nonnegative(h_trial)
        h_trial = h_trial * (n / 2) / np.sum(h_trial)  # re-normalize
        
        corr trial = np.correlate(h_trial, 1 - h_trial, mode='full')
        new_c5 = np.max(corr_trial) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_trial.copy()
            
        # adaptive learning rate
        if new_c5 > last_improvement:
            alpha *= 0.9
        else:
            last_improvement = new_c5
            
        h_polished = h_trial
        
        steps += 1
    
    # Return the final result
    return (best_h, best_c5, n)
