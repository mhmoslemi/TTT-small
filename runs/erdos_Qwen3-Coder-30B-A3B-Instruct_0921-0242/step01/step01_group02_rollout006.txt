import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy import sparse as sp

def evaluate_c5(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def build_Q(k, n):
    """Build sparse symmetric PSD matrix for term h_i h_{i+k}."""
    if k == 0:
        return sp.diags(np.ones(n), format='csr')
    data = np.ones(n - abs(k))
    rows = np.arange(n - abs(k))
    cols = rows + k if k >= 0 else rows - k
    Q = sp.csr_matrix((data, (rows, cols)), shape=(n,n))
    return 0.5 * (Q + Q.T)

def solve_qp(n, Kmax, warm_h=None):
    """Solve convex QP for optimal h with quadratic constraints."""
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    
    for k in range(-Kmax, Kmax+1):
        Q = build_Q(k, n)
        constraints.append(cp.quad_form(h, Q) >= n/2 - t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    if warm_h is not None:
        h.value = warm_h
    
    prob.solve(solver=cp.OSQP, verbose=False, max_iter=5000)
    return h.value, t.value

def binary_refine(h_frac, n, Kmax, dx, time_limit):
    """Refine fractional solution to binary with local search."""
    # Initialize binary vector by thresholding
    idx = np.argsort(-h_frac)[:n//2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1
    
    # Compute initial correlation
    corr = correlate(h_bin, 1 - h_bin, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h_bin.copy()
    
    while time.time() < time_limit:
        i1 = np.random.choice(np.where(h_bin == 1)[0])
        i0 = np.random.choice(np.where(h_bin == 0)[0])
        
        # Tentative swap
        h_bin[i1] = 0
        h_bin[i0] = 1
        
        # Update correlation only for involved shifts
        new_corr = correlate(h_bin, 1 - h_bin, mode='full')
        new_c5 = np.max(new_corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_bin.copy()
        else:
            # Revert
            h_bin[i1] = 1
            h_bin[i0] = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_values = [250, 300, 350, 400]
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    warm_h = None
    
    for n in n_values:
        if time.time() - start_time > budget_s - 50:
            break
        
        dx = 2.0 / n
        Kmax = int(0.75 * n)
        
        h_opt, c5_val = solve_qp(n, Kmax, warm_h)
        
        # Binary refinement
        refine_time = start_time + budget_s - 50
        h_refined, c5_refined = binary_refine(h_opt, n, Kmax, dx, refine_time)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
            best_n = n
        
        # Use as warm start for next
        warm_h = h_opt
        
        if time.time() - start_time > budget_s - 50:
            break
            
    return (best_h, best_c5, best_n)
