```python
import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def waterfilling(h, target_sum, eps=1e-8):
    """Adjust h to target_sum by adding a constant to all entries."""
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < eps:
        return h
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def build_shift_matrix(n, k):
    """Build sparse matrix T_k that represents shift by k."""
    T = np.zeros((n, n))
    for i in range(n - k):
        T[i, i + k] = 1
    return T

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 128
    dx = 2.0 / n_points
    target_sum = n_points / 2  # since dx * sum(h) = 1
    
    best_c5 = float('inf')
    best_h = None
    
    # Stage 1: SDP Relaxation
    if time.time() - start_time > budget_s - 30:
        pass
    else:
        try:
            # Variables
            h = cp.Variable(n_points)
            X = cp.Variable((n_points, n_points), symmetric=True)
            t = cp.Variable()

            # Constraints
            constraints = [
                cp.sum(h) == target_sum,
                h >= 0,
                h <= 1,
                cp.diag(X) == h,
                X >> 0,  # X is positive semidefinite
            ]

            # Autocorrelation constraints via trace
            for k in range(1, n_points):
                T_k = build_shift_matrix(n_points, k)
                # Trace(T_k * X) = sum_{i=0}^{n-k-1} X_{i,i+k}
                constraints.append(cp.trace(T_k @ X) >= t)

            # Objective
            objective = cp.Maximize(t)
            prob = cp.Problem(objective, constraints)
            
            prob.solve(solver=cp.SCS, verbose=False, eps=1e-4, max_iters=5000)
            
            h_sdp = h.value
            if h_sdp is not None:
                h_sdp = waterfilling(h_sdp, target_sum)
                c5_sdp = evaluate_c5(h_sdp)
                if c5_sdp < best_c5:
                    best_c5 = c5_sdp
                    best_h = h_sdp.copy()
        except Exception as e:
            pass
    
    # Stage 2: Randomized Rounding
    if best_h is not None and time.time() - start_time < budget_s - 20:
        try:
            if hasattr(cp, 'eig') and len(best_h) > 0:
                # For rounding, if SDP solution is not near-binary, try some randomizations
                # we take the diagonal of the SDP solution's rank-1 approximation
                h_rounded = best_h.copy()
                best_candidate_c5 = best_c5
                
                # Try a few refinements
                for _ in range(30):
                    # Generate random rounding
                    V = np.random.randn(n_points, n_points)  # dummy matrix to simulate rank
                    y = V @ np.random.randn(n_points)  # sample random vector to sort
                    sorted_y = np.sort(y)[::-1]
                    # Find median of rotations (adjust for exact median)
                    threshold = sorted_y[n_points // 2]
                    h_trial = (y > threshold).astype(float)
                    h_trial = waterfilling(h_trial, target_sum)
                    c5_trial = evaluate_c5(h_trial)
                    if c5_trial < best_candidate_c5:
                        best_candidate_c5 = c5_trial
                        h_rounded = h_trial.copy()
                
                if best_candidate_c5 < best_c5:
                    best_c5 = best_candidate_c5
                    best_h = h_rounded.copy()
        except Exception as e:
            pass
    
    # Stage 3: Smooth Max Gradient Descent with Continuation
    if best_h is not None and time.time() - start_time < budget_s - 15:
        try:
            h_refine = best_h.copy()
            c5_refine = evaluate_c5(h_refine)
            
            def smooth_max(h, beta=0.5):
                """Compute a smooth surrogate for max_k I_k."""
                dx = 2.0 / len(h)
                I = np.zeros(len(h))
                corr = correlate(h, 1 - h, mode='full')
                for k in range(len(corr)):
                    I[k] = dx * (1 - corr[k] / dx)
                exp_I = np.exp(beta * I)
                return np.log(np.sum(exp_I)) / beta
            
            def grad_smooth_max(h, beta=0.5):
                """Compute the gradient of the smooth max."""
                dx = 2.0 / len(h)
                n = len(h)
                # Full correlation of h with (1-h) in full mode
                corr_h = correlate(h, 1 - h, mode='full')
                # Compute I_k values directly as smooth shifting
                I = np.zeros(len(corr_h))
                for k in range(len(corr_h)):
                    I[k] = dx * (1 - corr_h[k] / dx)
                # Consider weights for gradient contribution
                exp_I = np.exp(beta * I)  # Exponent for each shift
                W = exp_I / np.sum(exp_I)
                grad = 0  # shape (n,)
                for k in range(len(I)):
                    partial_I = np.zeros(n)
                    for i in range(n):
                        if i + k < n:
                            partial_I[i] += dx * (1 - h[i + k])
                        if i - k >= 0:
                            partial_I[i] -= dx * h[i - k]
                    grad += W[k] * partial_I
                return grad
            
            beta = 0.5
            num_inner_steps = 50
            
            while beta < 50 and time.time() - start_time < budget_s - 10:
                for _ in range(num_inner_steps):
                    if time.time() - start_time > budget_s - 10:
                        break
                    grad = grad_smooth_max(h_refine, beta)
                    # Backtracking line search
                    alpha = 1.0
                    for _ in range(10):
                        h_trial = h_refine - alpha * grad
                        h_trial = np.clip(h_trial, 0, 1)
                        h_trial = waterfilling(h_trial, target_sum)
                        c5_trial = evaluate_c5(h_trial)
                        if c5_trial < c5_refine or alpha < 1e-8:
                            break
                        alpha *= 0.5
                    h_refine = h_trial.copy()
                    c5_refine = c5_trial
                beta *= 1.5
                if c5_refine < best_c5:
                    best_c5 = c5_refine
                    best_h = h_refine.copy()
                    
        except Exception as e:
            pass
    
    # Stage 4: Final discrete switching
    if best_h is not None and time.time() - start_time < budget_s - 5:
        try:
            h_final = best_h.copy()
            c5_final = evaluate_c5(h_final)
            
            for _ in range(500):  # do many swaps
                if time.time() - start_time > budget_s - 2:
                    break
                i, j = np.random.choice(n_points, 2, replace=False)
                if h_final[i] <= 0 or h_final[j] >= 1:
                    continue
                # Perturbation (move some mass)
                delta = min(h_final[i], 1 - h_final[j], 0.02)
                h_trial = h_final.copy()
                h_trial[i] -= delta
                h_trial[j] += delta
                # Ensure constraints
                h_trial = waterfilling(h_trial, target_sum)
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < c5_final:
                    h_final = h_trial.copy()
                    c5_final = c5_trial
                    if c5_final < best_c5:
                        best_c5 = c5_final
        except Exception as e:
            pass
    
    if best_h is None:
        # Fallback
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_c5(best_h)
        
    return (best_h, best_c5, n_points)
```
