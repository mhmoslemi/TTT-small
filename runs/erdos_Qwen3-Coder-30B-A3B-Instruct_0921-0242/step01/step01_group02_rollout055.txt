def c5_of_h(h, dx):
    """Compute C5 for given h using fast FFT convolution."""
    g = 1 - h
    corr = fftconvolve(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def compute_corr_update(corr, h, i1, i0, old_h_i1, old_h_i0):
    """Compute updated correlation vector after swapping h[i1] and h[i0]."""
    n = len(h)
    # Calculate new values after swap
    new_h_i1 = h[i1]  # Will become 0 if old_h_i1 was 1
    new_h_i0 = h[i0]  # Will become 1 if old_h_i0 was 0
    
    # Update the correlation vector incrementally
    # Subtract contribution of old positions
    if old_h_i1 == 1:
        corr -= np.concatenate([np.zeros(i1), h[:n-i1]]) * (1 - old_h_i1)
    else:
        corr -= np.concatenate([np.zeros(i1), (1-h)[:n-i1]]) * old_h_i1
        
    if old_h_i0 == 0:
        corr -= np.concatenate([np.zeros(i0), h[:n-i0]]) * (1 - old_h_i0)
    else:
        corr -= np.concatenate([np.zeros(i0), (1-h)[:n-i0]]) * old_h_i0

    # Add contribution of new positions
    if new_h_i1 == 1:
        corr += np.concatenate([np.zeros(i1), h[:n-i1]]) * (1 - new_h_i1)
    else:
        corr += np.concatenate([np.zeros(i1), (1-h)[:n-i1]]) * new_h_i1
        
    if new_h_i0 == 0:
        corr += np.concatenate([np.zeros(i0), h[:n-i0]]) * (1 - new_h_i0)
    else:
        corr += np.concatenate([np.zeros(i0), (1-h)[:n-i0]]) * new_h_i0

    return corr

def binary_to_dense(h_binary, n_fine=2000):
    """Convert a binary h vector to a dense piecewise constant function."""
    h_fine = np.zeros(n_fine)
    dx_fine = 2.0 / n_fine
    n = len(h_binary)
    dx_coarse = 2.0 / n
    
    # Determine how many fine points each coarse-point covers
    for i in range(n):
        start_idx = int(i * n_fine / n)
        end_idx = int((i+1) * n_fine / n)
        h_fine[start_idx:end_idx] = h_binary[i]
        
    return h_fine

def get_best_shift_from_corr(corr):
    """Extract the maximum shift index."""
    return np.argmax(corr) - len(corr) // 2

def evaluate_with_fine_grid(h_binary, n_fine=2000):
    """Evaluate the piecewise-constant function on fine grid."""
    h_fine = binary_to_dense(h_binary, n_fine)
    dx_fine = 2.0 / n_fine
    return c5_of_h(h_fine, dx_fine)

def binary_neighbor_moves(h, ones_list, zeros_list):
    """Generate all possible single swap moves (i1, i0) such that h[i1]=1 and h[i0]=0."""
    moves = []
    if len(ones_list) == 0 or len(zeros_list) == 0:
        return moves
    # Sample pairs of indices
    max_moves = min(1000, len(ones_list) * len(zeros_list))
    for _ in range(max_moves):
        i1 = np.random.choice(ones_list)
        i0 = np.random.choice(zeros_list)
        moves.append((i1, i0))
    return moves
