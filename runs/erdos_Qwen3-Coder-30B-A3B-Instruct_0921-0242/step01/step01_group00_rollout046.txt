import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.linalg import eigh

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def waterfilling_projection(h, target_sum):
    """Projects h onto the feasible set with exact sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Add diff/n to each component
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def solve_sdp(n_points):
    """Solve the SDP relaxation for the maximum t such that all correlations >= t."""
    dx = 2.0 / n_points
    h = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable(1)

    constraints = [
        cp.trace(X) == np.sum(h),
        h >= 0,
        h <= 1,
        X >> 0,
        X[cp.arange(n_points), cp.arange(n_points)] == h,
    ]
    
    # Add autocorrelation constraints
    for k in range(1, n_points):
        # Sum over i from 0 to n-k-1
        expr = 0
        for i in range(n_points - k):
            expr += X[i, i + k]
        constraints.append(expr >= t)
    
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sdp = np.array(h.value).flatten()
            h_sdp = waterfilling_projection(h_sdp, n_points / 2)
            return h_sdp
    except Exception:
        pass
    return None

def randomized_rounding(h_sdp, n_points, num_rounds=30):
    """Use randomized rounding to convert fractional solution to binary."""
    try:
        # Compute a partial SVD with positive eigenvalues
        X = np.outer(h_sdp, h_sdp)
        eigenvals, eigenvecs = eigh(X)
        # Keep only positive eigenvalues
        mask = eigenvals > 1e-8
        eigenvals = eigenvals[mask]
        eigenvecs = eigenvecs[:, mask]
        
        best_h = None
        best_c5 = float('inf')
        
        for r in range(num_rounds):
            g = np.random.randn(eigenvecs.shape[1])
            y = eigenvecs @ g
            y_sorted = np.sort(y)
            # Find threshold so that exactly n/2 entries are > threshold
            target_count = n_points // 2
            idx = target_count - 1
            if idx < 0:
                idx = 0
            elif idx >= len(y_sorted):
                idx = len(y_sorted) - 1
            threshold = y_sorted[idx]
            
            h_binary = (y > threshold).astype(float)
            h_binary = waterfilling_projection(h_binary, n_points / 2)
            c5 = evaluate_c5(h_binary)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_binary.copy()
        return best_h, best_c5
    except Exception:
        return None, float('inf')

def smooth_max_gradient_descent(h_start, n_points, budget_s):
    """Use smooth max surrogate to optimize the function."""
    dx = 2.0 / n_points
    beta = 0.5  # start with small beta for smoothness
    max_beta = 50.0
    steps_per_beta = 50
    max_steps = 1000

    h_current = h_start.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current)

    start_time = time.time()
    while beta <= max_beta and (time.time() - start_time) < budget_s:
        # Compute I_k values
        corr = correlate(h_current, 1 - h_current, mode='full')
        corr = corr * dx
        I_values = 1 - corr  # I_k = dx*(1 - sum(h_i * h_{i+k}))
        
        # Compute smooth max
        exp_I = np.exp(beta * I_values)
        total_exp = np.sum(exp_I)
        weights = exp_I / total_exp
        
        # Compute gradient
        gradient = np.zeros_like(h_current)
        for k in range(1, n_points):
            # compute dI_k/dh_i for k shift
            # I_k = dx * (1 - sum(h_i * h_{i+k}))
            # dI_k/dh_i = -dx * h_{i+k} if i+k < n, else 0
            # dI_k/dh_i = dx * h_{i-k} if i-k >= 0, else 0
            dI_k = np.zeros(n_points)
            for i in range(n_points):
                if i + k < n_points:
                    dI_k[i] -= dx * h_current[i + k]
                if i - k >= 0:
                    dI_k[i] += dx * h_current[i - k]
            
            gradient += weights[k] * dI_k
        
        # Projected gradient step
        step_size = 1.0
        for _ in range(10):  # backtrack steps
            # try h' = h_current - step_size * gradient
            h_trial = h_current - step_size * gradient
            h_trial = waterfilling_projection(h_trial, n_points / 2)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < evaluate_c5(h_current):
                h_current = h_trial
                break
            step_size /= 2
        else:
            h_current = waterfilling_projection(h_current, n_points / 2)
        
        # Update best if needed
        current_c5 = evaluate_c5(h_current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
        
        # Update beta for continuation
        beta *= 1.5
        if beta > max_beta:
            beta = max_beta
    
    return best_h, best_c5

def discrete_local_search(h_start, n_points, budget_s):
    """Do a cheap integer-local search to further improve."""
    h_current = h_start.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current)
    start_time = time.time()
    
    while (time.time() - start_time) < budget_s:
        # Try a few random swaps
        for _ in range(100):
            if (time.time() - start_time) >= budget_s:
                break
                
            i, j = np.random.choice(n_points, size=2, replace=False)
            delta = min(h_current[i], 1 - h_current[j], 0.02)
            
            h_trial = h_current.copy()
            h_trial[i] -= delta
            h_trial[j] += delta
            h_trial = waterfilling_projection(h_trial, n_points / 2)
            
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h_current = h_trial.copy()
            elif c5_trial < evaluate_c5(h_current):
                h_current = h_trial.copy()
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 128
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Stage 1: solve SDP
    h_sdp = solve_sdp(n_points)
    if h_sdp is None:
        h_sdp = np.ones(n_points) * 0.5
    c5_sdp = evaluate_c5(h_sdp)
    
    best_h = h_sdp.copy()
    best_c5 = c5_sdp
    
    # Stage 2: Randomized Rounding
    if (time.time() - start_time) < budget_s - 30:
        h_rounded, c5_rounded = randomized_rounding(h_sdp, n_points, num_rounds=30)
        if h_rounded is not None and c5_rounded < best_c5:
            best_h = h_rounded
            best_c5 = c5_rounded
    
    # Stage 3: Smooth Max Gradient Descent
    if (time.time() - start_time) < budget_s - 20:
        h_refined, c5_refined = smooth_max_gradient_descent(best_h, n_points, budget_s - 20 - (time.time() - start_time))
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Stage 4: Discrete Local Search
    if (time.time() - start_time) < budget_s - 10:
        h_final, c5_final = discrete_local_search(best_h, n_points, budget_s - 10 - (time.time() - start_time))
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (best_h, best_c5, n_points)
