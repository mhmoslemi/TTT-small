import numpy as np

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n_values = [200, 250, 300]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in n_values:
        if time.time() - start_time > budget_s - 5:
            break
            
        dx = 2.0 / n
        target_sum = n / 2.0
        
        # Run simulated annealing to get heuristic starting point
        h0, c5_initial = heuristic_sa(n, target_sum, dx, budget_s * 0.15, start_time)
        if c5_initial < best_c5:
            best_c5 = c5_initial
            best_h = h0.copy()
            best_n = n
        
        # Run MIP solver with core shift constraints
        h_mip, c5_mip = solve_mip_core(n, target_sum, dx, c5_initial, budget_s * 0.3, start_time)
        if h_mip is not None and c5_mip < best_c5:
            best_c5 = c5_mip
            best_h = h_mip.copy()
            best_n = n
            
        # Local refinement with binary swaps
        h_refined, c5_refined = local_binary_refinement(best_h, dx, n, budget_s * 0.05, start_time)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Final evaluation
    if best_h is not None:
        final_c5 = c5_of_h(best_h, dx)
        return (best_h, final_c5, best_n)
    else:
        raise RuntimeError("No valid solution found within the allocated time budget.")

def heuristic_sa(n, target_sum, dx, max_time, start_time):
    """Perform simulated annealing to find good starting binary vector."""
    best_h = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
    while np.sum(best_h) != target_sum:
        idx = np.random.randint(0, n)
        if best_h[idx] == 0:
            # Swap to 1
            ones = np.where(best_h == 1)[0]
            if len(ones) == 0:
                continue
            flip_idx = np.random.choice(ones)
            best_h[flip_idx] = 0
            best_h[idx] = 1
        else:
            # Swap to 0
            zeros = np.where(best_h == 0)[0]
            if len(zeros) == 0:
                continue
            flip_idx = np.random.choice(zeros)
            best_h[flip_idx] = 1
            best_h[idx] = 0

    best_c5 = c5_of_h(best_h, dx)
    
    temp = 1.0
    cooling_rate = 0.999
    max_iter = int(max_time / 0.0001) if max_time > 0 else 100000
    current_h = best_h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        idx1 = np.random.randint(0, n)
        idx2 = np.random.randint(0, n)
        while idx1 == idx2 or current_h[idx1] == current_h[idx2]:
            idx1 = np.random.randint(0, n)
            idx2 = np.random.randint(0, n)
        
        # Swap
        new_h = current_h.copy()
        new_h[idx1], new_h[idx2] = new_h[idx2], new_h[idx1]
        
        # Ensure exact sum constraint
        ones = np.where(new_h == 1)[0]
        zeros = np.where(new_h == 0)[0]
        if len(ones) > target_sum:
            remove_idx = np.random.choice(ones, len(ones) - int(target_sum), replace=False)
            new_h[remove_idx] = 0
        elif len(ones) < target_sum:
            add_idx = np.random.choice(zeros, int(target_sum) - len(ones), replace=False)
            new_h[add_idx] = 1
            
        new_c5 = c5_of_h(new_h, dx)
        
        delta = new_c5 - best_c5
        if delta < 0 or np.exp(-delta / temp) > np.random.rand():
            current_h = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        
        temp *= cooling_rate
        
        # Reduce time budget if necessary
        if time.time() - start_time > max_time - 0.1:
            break
    
    return best_h, best_c5

def local_binary_refinement(h_start, dx, n, max_time, start_time):
    """Perform fast hill climb with binary swaps."""
    current_h = h_start.copy()
    best_c5 = c5_of_h(current_h, dx)
    num_swaps = 0
    max_swaps = int(max_time / 0.00001) if max_time > 0 else 5000
    
    while num_swaps < max_swaps:
        if time.time() - start_time > max_time:
            break
            
        # Select two random positions to swap
        idx1 = np.random.randint(0, n)
        idx2 = np.random.randint(0, n)
        if idx1 == idx2:
            continue
        if current_h[idx1] == current_h[idx2]:
            continue
            
        # Swap them
        new_h = current_h.copy()
        new_h[idx1], new_h[idx2] = new_h[idx2], new_h[idx1]
        
        # Reconcile sum to target
        ones = np.where(new_h == 1)[0]
        zeros = np.where(new_h == 0)[0]
        if len(ones) > n / 2:
            remove = len(ones) - int(n / 2)
            idx_remove = np.random.choice(ones, remove, replace=False)
            new_h[idx_remove] = 0
        elif len(ones) < n / 2:
            add = int(n / 2) - len(ones)
            idx_add = np.random.choice(zeros, add, replace=False)
            new_h[idx_add] = 1
            
        new_c5 = c5_of_h(new_h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            current_h = new_h.copy()
            
        num_swaps += 1
        
    return current_h, best_c5
