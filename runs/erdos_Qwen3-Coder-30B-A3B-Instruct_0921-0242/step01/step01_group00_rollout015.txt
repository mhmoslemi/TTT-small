import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp
from scipy.linalg import eigh

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def solve_sdp(n):
    """Solve the SDP relaxation of the problem."""
    dx = 2.0 / n
    
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.trace(X) == 1,  # sum(h) = n/2 is same as sum(h*dx) = 1 with dx=2/n
        h >= 0,
        h <= 1,
        X[i, i] == h[i] for i in range(n),
        X >> 0,  # Positive semidefinite
        t <= 0.5  # Safe bound
    ]
    
    # Autocorrelation constraints for all shifts k = 1,...,n-1
    for k in range(1, n):
        # Build the shift matrix T_k
        Tk = np.zeros((n, n))
        for i in range(n - k):
            Tk[i, i + k] = 1
        # Add constraint
        constraints.append(cp.trace(Tk @ X) >= t)
    
    # Objective
    objective = cp.Maximize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, max_iters=5000, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_sdp = h.value
            h_sdp = project_to_feasible(h_sdp, n)
            t_val = t.value
            return h_sdp, t_val
    except:
        pass
    return None, None

def randomized_rounding(h_sdp, n, R=30):
    """Perform randomized rounding to get binary candidates."""
    # Get an approximate factorization of the SDP solution
    try:
        eigenvals, eigenvecs = eigh(h_sdp @ h_sdp.T)
        # Build V such that V @ V.T ≈ h_sdp * h_sdp^T, but keep all components
        V = eigenvecs @ np.diag(np.sqrt(np.maximum(eigenvals, 0)))
        
        best_h_rounded = None
        best_c5 = float('inf')
        
        # Sample multiple candidates
        for _ in range(R):
            g = np.random.randn(V.shape[1])
            y = V @ g
            # Sort and find threshold such that sum = n/2
            sorted_y = np.sort(y)[::-1]
            threshold = sorted_y[n // 2]
            candidate = (y > threshold).astype(float)
            
            # Do a waterfilling projection to ensure sum = n/2
            candidate = project_to_feasible(candidate, n)
            
            c5 = evaluate_c5(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h_rounded = candidate.copy()
                
        return best_h_rounded, best_c5
    except:
        # Fallback to simple threshold rounding
        threshold = 0.5
        binary = (h_sdp >= threshold).astype(float)
        binary = project_to_feasible(binary, n)
        c5 = evaluate_c5(binary)
        return binary, c5

def smooth_max_gradient_refinement(h0, n, budget_s):
    """Refine solution using smooth max gradients."""
    dx = 2.0 / n
    # Prepare shift matrices for autocorrelations
    shift_matrices = []
    for k in range(1, n):
        Tk = np.zeros((n, n))
        for i in range(n - k):
            Tk[i, i + k] = 1
        shift_matrices.append(Tk)
    
    # Objective and its smooth approximation
    def smooth_max(h, beta):
        I_values = [dx * (1 - np.sum(h * np.roll(h, k))) for k in range(1, n)]
        I_values = np.array(I_values)
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * I_values)))
    
    def smooth_max_grad(h, beta):
        I_values = [dx * (1 - np.sum(h * np.roll(h, k))) for k in range(1, n)]
        I_values = np.array(I_values)
        weights = np.exp(beta * I_values)
        weights = weights / np.sum(weights)
        
        grad = np.zeros_like(h)
        for k in range(1, n):
            # Derivative of I_k(h)
            # h_{i-k} + h_{i+k} = 0, for indices < 0 or >= n, we assume 0
            # So,
            # for i: G[i] += (1 - h[i+k]) - h[i-k]
            kernel = np.zeros_like(h)
            if k < n:
                kernel[k:] = h[:-k]
            if k < n:
                kernel[:-k] += h[k:]
            # More precise
            backward_term = np.zeros_like(h)
            backward_term[k:] = h[:-k]
            forward_term = np.zeros_like(h)
            forward_term[:-k] = h[k:]
            grad += weights[k-1] * ((1 - h) - backward_term - (forward_term))
        return grad
    
    # Partition into chunks for better handling
    h = h0.copy()
    t_start = time.time()
    beta = 0.5
    max_iterations = 1000
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for iteration in range(max_iterations):
        if time.time() - t_start > budget_s:
            break
        # Gradient and update
        grad = smooth_max_grad(h, beta)
        
        # Step size selection
        alpha = 1.0
        for _ in range(10):
            h_trial = h - alpha * grad
            h_trial = project_to_feasible(h_trial, n)
            trial_c5 = evaluate_c5(h_trial)
            
            if trial_c5 < current_c5:
                h = h_trial
                current_c5 = trial_c5
                if trial_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = trial_c5
                break
            else:
                alpha /= 2.0
                
        # Increase beta with continuation
        beta = min(50.0, beta * 1.5)
        
    return best_h, best_c5

def final_local_search(h0, n, budget_s):
    """Local discrete search to tighten the value."""
    h = h0.copy()
    dx = 2.0 / n
    current_c5 = evaluate_c5(h)
    t_start = time.time()
    
    while time.time() - t_start < budget_s:
        # Pick two random indices
        i, j = np.random.choice(n, size=2, replace=False)
        if h[i] > 0 and h[j] < 1:
            # Trial: increase h[i], decrease h[j]
            delta = min(h[i], 1 - h[j], 0.02)
            
            h_trial = h.copy()
            h_trial[i] -= delta
            h_trial[j] += delta
            
            h_trial = project_to_feasible(h_trial, n)
            trial_c5 = evaluate_c5(h_trial)
            if trial_c5 < current_c5:
                h = h_trial
                current_c5 = trial_c5
                
    return h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Use a lower number of points to save time and avoid memory issues?
    n = 128
    dx = 2.0 / n
    
    # Stage 1: SDP relaxation
    if time.time() - start_time < budget_s - 40:
        h_sdp, t_sdp = solve_sdp(n)
        if h_sdp is not None:
            # Estimate C5 from SDP
            C5_sdp = dx * (1 - t_sdp)
            # Remember that C5 = dx*(1 - max_autocorr) so this is an upper bound
            if C5_sdp < best_c5:
                best_c5 = C5_sdp
                best_h = h_sdp.copy()
                best_n = n

    # Stage 2: Randomized Rounding
    if time.time() - start_time < budget_s - 30:
        if best_h is not None:
            rounded_h, rounded_c5 = randomized_rounding(best_h, n)
            if rounded_c5 < best_c5:
                best_c5 = rounded_c5
                best_h = rounded_h.copy()
                best_n = n

    # Stage 3: Smooth Max Gradient Refinement
    if time.time() - start_time < budget_s - 10:
        if best_h is not None:
            refine_h, refine_c5 = smooth_max_gradient_refinement(
                best_h, n, min(budget_s - (time.time() - start_time) - 10, 50))
            if refine_c5 < best_c5:
                best_c5 = refine_c5
                best_h = refine_h.copy()
                best_n = n

    # Stage 4: Final Local Search
    if time.time() - start_time < budget_s:
        if best_h is not None:
            final_h, final_c5 = final_local_search(best_h, n, budget_s - (time.time() - start_time))
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
                best_n = n

    # Final check
    if best_h is None:
        best_h = np.ones(n) * 0.5
        best_c5 = evaluate_c5(best_h)
        best_n = n

    return (best_h, best_c5, best_n)
