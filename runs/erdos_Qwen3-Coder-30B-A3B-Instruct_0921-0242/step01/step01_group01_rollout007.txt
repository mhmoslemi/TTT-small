import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp
from scipy.interpolate import interp1d
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Time tracking
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)
    
    # Parameters
    n_fine = 300
    dx_fine = 2.0 / n_fine
    
    if time_left() < 10:
        # Fallback to simple approach if not enough time
        h = np.ones(n_fine) * 0.5
        h = h * (n_fine / 2) / np.sum(h)
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx_fine
        return (h, c5, n_fine)
    
    # Step 1: SDP relaxation coarse grid
    n_coarse = 120  # Make it simpler, faster
    dx_coarse = 2.0 / n_coarse
    h_coarse = None
    
    # Añadir un fallback en caso de fallo en la SDP
    try:
        # SDP model

        # Define variables
        t = cp.Variable()
        h = cp.Variable(n_coarse)
        X = cp.Variable((n_coarse, n_coarse), symmetric=True)

        constraints = [cp.bmat([[1, h.T], [h, X]]) >> 0]  # Block matrix PSD
        constraints += [x >= 0 for x in h]  # h >= 0
        constraints += [x <= 1 for x in h]  # h <= 1
        constraints += [cp.sum(h) == n_coarse / 2]  # Sum constraint

        # Autocorrelation constraints
        for k in range(n_coarse):
            R = np.zeros((n_coarse, n_coarse))
            R[np.arange(n_coarse), np.arange(k, n_coarse)] = 1
            R[np.arange(k, n_coarse), np.arange(n_coarse)] = 1
            constraints += [cp.trace(R @ X) >= t]

        prob = cp.Problem(cp.Maximize(t), constraints)

        # Solve SDP with SCS solver
        prob.solve(solver=SCS, max_iters=1000, eps=1e-5, verbose=False)
        
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_opt = h.value
            if h_opt is not None:
                h_coarse = h_opt.copy()
            else:
                h_coarse = np.ones(n_coarse) * 0.5
        else:
            h_coarse = np.ones(n_coarse) * 0.5

    except Exception as ex:
        h_coarse = np.ones(n_coarse) * 0.5
   
    if time_left() < 100:
        # Not enough time for refinement, return result
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
        c5_coarse = np.max(np.correlate(h_coarse, 1 - h_coarse, mode='full')) * dx_coarse
        h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_coarse), h_coarse)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
        c5_fine = np.max(np.correlate(h_fine, 1 - h_fine, mode='full')) * dx_fine
        return (h_fine, c5_fine, n_fine)
    
    # Step 2: Refinement via gradient descent on smooth approximation
    def smooth_max(x, beta):
        return (1/beta) * np.log(np.sum(np.exp(beta * x)))
    
    def gradient_of_smooth_max(h, k):
        # Use analytical gradient with given shift k
        h_pad = np.pad(h, (0, len(h)), mode='constant')
        # Derivative of h_i (1 - h_{i+k})
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + k < len(h):
                grad[i] += (1 - h[i + k])
            if i - k >= 0:
                grad[i] -= h[i - k]
        return grad
    
    def apply_soft_max(h, beta, k, dx):
        # Compute the full convolution
        conv = np.correlate(h, 1 - h, mode='full')
        # Apply softmax
        max_con = np.max(conv)
        exp_vals = np.exp(beta * dx * (conv - max_con))
        weights = exp_vals / np.sum(exp_vals)
        # Sum of weighted gradients
        grad_sum = np.zeros_like(h)
        for i in range(len(h)):
            grad_i = gradient_of_smooth_max(h, i)
            grad_sum += weights[i] * grad_i
        return grad_sum
    
    def water_filling(h, target_sum):
        # Water filling projection
        h = np.copy(h)
        current_sum = np.sum(h)
        if np.isclose(current_sum, target_sum):
            return h
        
        excess = current_sum - target_sum
        if np.abs(excess) < 1e-12:
            return h

        # identify indices that can be modified (0 < h[i] < 1)
        valid = np.where((h > 0) & (h < 1))[0]
        if len(valid) == 0:
            # All are at boundaries; clamp
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            return h

        while np.abs(excess) > 1e-12:
            if excess > 0:
                # Reduce evenly
                delta = excess / len(valid)
                h[valid] -= delta
                # Update valid indices
                new_valid = np.where((h > 0) & (h < 1))[0]
                if len(new_valid) == 0:
                    # All below zero or above one
                    h = np.clip(h, 0, 1)
                    h = h * (target_sum / np.sum(h))
                    return h
                valid = new_valid
            else:
                # Increase evenly
                delta = -excess / len(valid)
                h[valid] += delta
                # Update valid indices
                new_valid = np.where((h > 0) & (h < 1))[0]
                if len(new_valid) == 0:
                    # All on boundaries
                    h = np.clip(h, 0, 1)
                    h = h * (target_sum / np.sum(h))
                    return h
                valid = new_valid
            current_sum = np.sum(h)
            excess = current_sum - target_sum
            
        h = np.clip(h, 0, 1)
        return h
    
    # Refinement on coarse grid
    h_current = np.copy(h_coarse)
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_coarse / 2) / np.sum(h_current)
    
    # Mirror symmetry
    h_current = (h_current + h_current[::-1]) / 2.0
    
    prev_c5 = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx_coarse
    for beta in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]:
        if time_left() < 100:
            break
            
        def fun_to_minimize(h):
            # Soft max surrogate
            conv = np.correlate(h, 1 - h, mode='full')
            return -smooth_max(conv, beta)
        
        # Compute gradient
        def grad_to_minimize(h):
            conv = np.correlate(h, 1 - h, mode='full')
            # It could be a long vector here since we're dealing with mode='full'
            # We compute the derivative of smooth_max at the shift that maximized conv
            # So we take only the points that contribute the max value in the convolution
            
            # Moment of truth: eval directly
            return -np.fromfunction(lambda i: gradient_of_smooth_max(h, i)[0] * (np.exp(beta * dx_coarse * conv[0]) / np.sum(np.exp(beta * dx_coarse * conv))), shape=(len(h),), dtype=int)
            
        
        res = minimize(fun_to_minimize, h_current, method='L-BFGS-B', bounds=[(0,1)]*len(h_current), options={'maxiter': 50, 'gtol': 1e-6})
        if res.success:
            h_new = res.x
            h_new = water_filling(h_new, n_coarse / 2)
            h_new = (h_new + h_new[::-1]) / 2.0
            c5_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx_coarse
            if c5_new < prev_c5:
                h_current = h_new.copy()
                prev_c5 = c5_new
            else:
                break  # No improvement
        else:
            break  # Optimization failed
    
    # Step 3: Upsampling and final refinement
    if time_left() < 100:
        h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_coarse), h_current)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
        h_fine = (h_fine + h_fine[::-1]) / 2.0
        c5_fine = np.max(np.correlate(h_fine, 1 - h_fine, mode='full')) * dx_fine
        return (h_fine, c5_fine, n_fine)
    
    # Fine grid refinement: Double the soft max precision
    h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_coarse), h_current)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    h_fine = (h_fine + h_fine[::-1]) / 2.0

    prev_c5 = np.max(np.correlate(h_fine, 1 - h_fine, mode='full')) * dx_fine
    
    for beta in [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0]:
        if time_left() < 30:
            break
            
        def fun_to_minimize(h):
            # Soft max surrogate
            conv = np.correlate(h, 1 - h, mode='full')
            return -smooth_max(conv, beta)
            
        def grad_to_minimize(h):
            # Numerical gradient by finite differences
            eps = 1e-5
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_minus = h.copy()
                h_minus[i] -= eps
                v_plus = -smooth_max(np.correlate(h_plus, 1 - h_plus, mode='full'), beta)
                v_minus = -smooth_max(np.correlate(h_minus, 1 - h_minus, mode='full'), beta)
                grad[i] = (v_plus - v_minus) / (2 * eps)
            return grad
                
        res = minimize(fun_to_minimize, h_fine, method='L-BFGS-B', bounds=[(0,1)]*len(h_fine), options={'maxiter': 30, 'gtol': 1e-6})
        if res.success:
            h_new = res.x
            h_new = water_filling(h_new, n_fine / 2)
            h_new = (h_new + h_new[::-1]) / 2.0
            c5_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx_fine
            if c5_new < prev_c5:
                h_fine = h_new.copy()
                prev_c5 = c5_new
            else:
                break  # No improvement
        else:
            break
            
    # Step 4: Stochastic polishing
    if time_left() < 30:
        return (h_fine, prev_c5, n_fine)
    
    best_h = h_fine.copy()
    best_c5 = prev_c5
    
    # Small random hill climbing
    num_moves = 500
    step_size = 0.004
    for _ in range(num_moves):
        if time_left() <= 1:
            break
        # Choose random indices
        indices = np.random.choice(n_fine, size=min(6, n_fine//20), replace=False)
        h_trial = best_h.copy()
        for i in indices:
            perturbation = np.random.uniform(-step_size, step_size)
            h_trial[i] += perturbation

        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_fine / 2) / np.sum(h_trial)
        h_trial = (h_trial + h_trial[::-1]) / 2.0
        
        c5_trial = np.max(np.correlate(h_trial, 1 - h_trial, mode='full')) * dx_fine
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial

    return (best_h, best_c5, n_fine)
