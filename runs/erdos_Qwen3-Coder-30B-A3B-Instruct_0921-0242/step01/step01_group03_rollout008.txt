import numpy as np
import scipy.optimize as opt
import time
from scipy.interpolate import interp1d
import cvxpy as cp

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Metaheuristic 1: Simulated Annealing on binary mask
    def binary_mask_to_h(b):
        h = b.astype(float)
        # Renormalize sum to exactly target_sum
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-10:
            scale = target_sum / h_sum
            h *= scale
        return h

    def compute_corr_mask(b):
        h = b.astype(float)
        g = 1.0 - h
        return np.correlate(h, g, mode='full')

    def evaluate_mask(b):
        corr = compute_corr_mask(b)
        return np.max(corr) * dx

    def propose_swap(b, idx1, idx2):
        b_new = b.copy()
        b_new[idx1] = 1 - b_new[idx1]
        b_new[idx2] = 1 - b_new[idx2]
        return b_new

    def update_corr_after_swap(b_old, corr_old, idx1, idx2):
        # update full correlation
        h = b_old.astype(float)
        g = 1.0 - h
        corr_old = np.correlate(h, g, mode='full')
        return corr_old

    def binary_sa(n_points, target_sum, budget_s, seed):
        np.random.seed(seed)
        best_c5 = np.inf
        best_binary = None
        current_binary = None

        # Generate multiple starting points
        seed_masks = []
        # Random
        b = np.zeros(n_points, dtype=int)
        b[:int(target_sum)] = 1
        np.random.shuffle(b)
        seed_masks.append(b.copy())

        # Two blocks
        b = np.zeros(n_points, dtype=int)
        b[int(n_points*0.25):int(n_points*0.75)] = 1
        seed_masks.append(b.copy())

        # Three blocks
        b = np.zeros(n_points, dtype=int)
        b[int(n_points*0.1):int(n_points*0.3)] = 1
        b[int(n_points*0.4):int(n_points*0.6)] = 1
        b[int(n_points*0.7):int(n_points*0.9)] = 1
        seed_masks.append(b.copy())

        # Get a better starting point from initialization if available
        if 'initial_h_values' in globals():
            init_h = np.array(initial_h_values)
            x_init = np.linspace(0, 2, len(init_h))
            x_new = np.linspace(0, 2, n_points)
            init_h_interp = np.interp(x_new, x_init, init_h)
            # Convert to binary
            b_from_init = (init_h_interp > 0.5).astype(int)
            # Adjust for sum
            sum_current = np.sum(b_from_init)
            diff = int(target_sum) - int(sum_current)
            if diff != 0:
                indices_zero = np.where(b_from_init == 0)[0]
                indices_one = np.where(b_from_init == 1)[0]
                if diff > 0:
                    # Add ones
                    indices_to_flip = indices_zero[:diff]
                    b_from_init[indices_to_flip] = 1
                else:
                    # Remove ones
                    indices_to_flip = indices_one[:abs(diff)]
                    b_from_init[indices_to_flip] = 0
            seed_masks.append(b_from_init)

        best_c5 = np.inf
        best_b = None

        threshold = 0.02  # Initial temperature-like scalar
        max_iters = 30000
        accepted = 0
        for i in range(max_iters):
            if time.time() - start_time > budget_s - 0.5:
                break
            # Sample starting mask
            current_b = seed_masks[i % len(seed_masks)]
            # Propose a random swap
            indices_1 = np.where(current_b == 1)[0]
            indices_0 = np.where(current_b == 0)[0]
            if len(indices_1) == 0 or len(indices_0) == 0:
                continue
            idx1 = np.random.choice(indices_1)
            idx2 = np.random.choice(indices_0)
            new_b = propose_swap(current_b, idx1, idx2)
            
            c0 = evaluate_mask(current_b) 
            c1 = evaluate_mask(new_b)

            if c1 < c0:
                current_b = new_b
                new_c5 = c1
            else:
                r = np.random.rand()
                if r < np.exp(-(c1 - c0) / threshold):
                    current_b = new_b
                    new_c5 = c1
                else:
                    new_c5 = c0

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = current_b.copy()

            # Update temperature
            accepted += 1
            if accepted % 100 == 0:
                threshold *= 0.995

        return best_c5, best_b

    # Run simulated annealing to obtain best binary mask
    best_mask_c5, best_binary = binary_sa(n_points, target_sum, budget_s, seed)

    # Convert binary to continuous h and clamp to [0,1]
    h_init = binary_mask_to_h(best_binary)
    best_c5 = best_mask_c5

    # Phase 2: Refine with LP
    if time.time() - start_time < budget_s - 10:
        # Build LP: introduce h variables, include max constraint
        # Variable vector: h[0], ..., h[n_points-1]  
        # Sum constraint: sum(h) = target_sum
        # For each shift k, sum_{i} h[i] * (1 - h[i+k]) <= t
        # Accept fragment indices from the binary mask

        b = best_binary
        t = cp.Variable()
        h_vars = cp.Variable(n_points)
        
        # Constraint: sum(x) = target_sum 
        sum_constraint = cp.Equality(cp.sum(h_vars), target_sum)
        
        # Maximal overlap constraint for each shift k
        constraints = [sum_constraint, t >= 0]
        for k in range(1 - n_points, n_points):
            if k >= 0:
                # h[i] * (1 - h[i+k])
                # Let g[i] = (1 - h[i+k])  
                expr = cp.sum(cp.multiply(h_vars, cp.roll(1 - h_vars, k)))
            else:
                # h[i] * (1 - h[i+k])
                # h[i] * (1 - h[i+k]) where k < 0 means we shift forward in indices
                expr = cp.sum(cp.multiply(h_vars, 1 - cp.roll(h_vars, -k)))
            constraints.append(expr <= t)

        # We want to minimize the maximum overlap t
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_refined = h_vars.value
                refined_c5 = c5_of_h(h_refined)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    h_init = h_refined
        except:
            pass

    # Phase 3: Small perturbations and projection
    if time.time() - start_time < budget_s - 10:
        h_best = h_init.copy()
        best_c5_refined = best_c5
        for _ in range(2000):
            if time.time() - start_time > budget_s - 2:
                break
            idxs = np.random.choice(n_points, size=np.random.randint(1, 10), replace=False)
            h_test = h_best.copy()
            for i in idxs:
                delta = np.random.uniform(-0.05, 0.05)
                h_test[i] += delta
            # Project to feasible set
            h_test = np.clip(h_test, 0, 1)
            h_sum = np.sum(h_test)
            h_test *= target_sum / h_sum
            c5_test = c5_of_h(h_test)
            if c5_test < best_c5_refined:
                best_c5_refined = c5_test
                h_best = h_test.copy()
        best_c5 = best_c5_refined
        h_init = h_best

    # Phase 4: Upsampling and SLSQP refinement
    h_upsampled = h_init
    final_n_points = 400
    dx_up = 2.0 / final_n_points
    h_upsampled = interp1d(np.linspace(0, 2, len(h_init)), h_init, kind='linear', fill_value='extrapolate')(np.linspace(0, 2, final_n_points))
    h_upsampled = h_upsampled * (final_n_points / 2) / np.sum(h_upsampled)
    
    if time.time() - start_time < budget_s - 5:
        # Final SLSQP minimization 
        bounds = [(0, 1)] * final_n_points
        def constraint_sum(h):
            return np.sum(h) - final_n_points / 2

        eq_constraint = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)}
        try:
            res = opt.minimize(
                fun=c5_of_h,
                x0=h_upsampled,
                method='SLSQP',
                bounds=bounds,
                constraints=eq_constraint,
                options={'ftol': 1e-10, 'maxiter': 200, 'disp': False},
                jac=None  # Try with no gradient
            )
            if res.success:
                best_c5_final = c5_of_h(res.x)
                h_upsampled = res.x
                if best_c5_final < best_c5:
                    best_c5 = best_c5_final
        except:
            pass
    
    return (h_upsampled, best_c5, final_n_points)
