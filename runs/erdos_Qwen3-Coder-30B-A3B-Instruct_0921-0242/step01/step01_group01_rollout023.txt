import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.ndimage import shift
import time

def decode_blocks_to_h(positions, lengths, heights, n_points):
    """Convert block parameters to a step function h"""
    # Each block contributes a constant value over a range
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    total_len = 0
    for i in range(len(positions)):
        start = int(positions[i] / dx)
        end = int((positions[i] + lengths[i]) / dx)
        end = min(end, n_points)
        start = max(0, start)
        if end > start:
            h[start:end] = heights[i]
            total_len += lengths[i]
    # Normalize to have integral = 1
    if np.sum(h) > 0:
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    return h

def c5_of_h(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution
    conv = np.correlate(h, g, mode='full')
    result = np.max(conv) * dx
    return result

def soft_max_conv(h, beta=1.0):
    """Compute soft-max approximation to C5"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    # Use log-sum-exp trick for numerical stability
    max_val = np.max(conv)
    exp_vals = np.exp(beta * dx * (conv - max_val))
    log_sum_exp = max_val + np.log(np.sum(exp_vals)) / beta
    return log_sum_exp

def gradient_soft_max_conv(h, beta=1.0):
    """Compute gradient of soft-max approximation"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = np.correlate(h, g, mode='full')
    
    # Compute weights
    max_val = np.max(conv)
    exp_vals = np.exp(beta * dx * (conv - max_val))
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute each shift's gradient
    grad = np.zeros_like(h)
    n = len(h)
    for i in range(2 * n - 1):
        k = i - (n - 1)
        # build shifted indices, taking into account padding
        if k >= 0:
            # h[j] * (1 - h[j+k]) 
            # if k >= 0: conv[i] sums over h[j] * (1 - h[j+k]) 
            # d/dh_j [ h[j] * (1 - h[j+k]) ] = (1 - h[j+k]) - (j+k)*h[j+k] 
            shifted_g_right = np.concatenate([g[k:], np.zeros(k)])
            grad += weights[i] * shifted_g_right
            # also -h[j]*h[j-k] but we compute that via index shuffle
            shifted_h_left = np.concatenate([np.zeros(-k), h[:-k]])
            grad -= weights[i] * shifted_h_left
        else:
            # k < 0 so h[j] * (1 - h[j-k])
            # shift g right by negative k
            shifted_g_left = np.concatenate([np.zeros(-k), g[:-k]])
            grad += weights[i] * shifted_g_left
            # also -h[j]*h[j-k] but shifted differently
            shifted_h_right = np.concatenate([h[-k:], np.zeros(-k)])
            grad -= weights[i] * shifted_h_right
            
    # Normalize by dx
    grad *= dx
    return grad

def project_sum(h, target_sum):
    """Project h onto the hyperplane sum(h) = target_sum"""
    current_sum = np.sum(h)
    diff = current_sum - target_sum
    if abs(diff) < 1e-9:
        return h
    # Subtract diff/n from all entries
    h = h - diff / len(h)
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    return h

def proj_sum_l2(h, target_sum):
    """L2 projection to sum=h, with clipping"""
    diff = np.sum(h) - target_sum
    # Handle the case that we need to make positive
    if abs(diff) < 1e-9:
        return h
    # Subtract average
    h = h - diff / len(h)
    # Clip to feasible bounds
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 250
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_params = 4 * 3  # 4 blocks, (pos,len,height) each

    def evaluate_candidate(x):
        # x contains 12 parameters (positions, lengths, heights)
        positions = x[:4]
        lengths = x[4:8]
        heights = x[8:12]
        
        # Apply constraints: positions in [0,2], lengths in (0,2], heights in [0,1]
        positions = np.clip(positions, 0, 2)
        lengths = np.clip(lengths, 1e-6, 2)
        heights = np.clip(heights, 0, 1)
        
        # Ensure positions are increasing
        positions = np.sort(positions)
        
        # Enforce non-overlapping constraint manually
        total_len = 0
        for i in range(len(positions)):
            pos = positions[i]
            leng = lengths[i]
            end = pos + leng
            total_len += leng
            # Bound overlaps
            if i != 0:
                pos_prev = positions[i-1]
                leng_prev = lengths[i-1]
                if pos < pos_prev + leng_prev:
                    # need to shift this one back
                    pos = pos_prev + leng_prev + 1e-8
                    positions[i] = pos
                    # now adjust to ensure we don't go over 2
                    if pos + leng > 2:
                        leng = 2 - pos
                        lengths[i] = leng
                        if leng < 0:
                            lengths[i] = 0.1
                            positions[i] = 2 - 0.1  # Push back
            
        # Decode parameters to full vector
        try:
            h = decode_blocks_to_h(positions, lengths, heights, n_points)
        except:
            # fallback to something valid
            h = np.ones(n_points) * 0.5
        # Expect sums are correct due to forced normalize
        return c5_of_h(h)

    # Global search space
    bounds = [(0,2), (0,2), (0,2), (0,2),  # positions
              (1e-6,2), (1e-6,2), (1e-6,2), (1e-6,2),  # lengths
              (0,1), (0,1), (0,1), (0,1)]  # heights

    print("Starting global DE...")
    # Differential Evolution search on blocks
    de_result = differential_evolution(evaluate_candidate, bounds,
                                       seed=seed, maxiter=60, popsize=15,
                                       disp=False, tol=1e-6)
    
    best_c5 = de_result.fun
    best_h = decode_blocks_to_h(de_result.x[:4], de_result.x[4:8], de_result.x[8:12], n_points)
    best_h = best_h * (n_points / 2) / np.sum(best_h)  # renormalize in any case
    
    # Refinement using smooth max
    print("Starting gradient refinement...")
    # Be cautious with larger time budget
    time_remaining = budget_s - (time.time() - start_time)
    max_iterations = min(500, int(time_remaining / 0.5))  # reasonable time per run
    
    # Bundle multiple passes of L-BFGS-B with smooth max
    beta = 0.5
    lambda_penalty = 10.0
    max_beta = 50.0
    max_iter_per_stage = 150
    idx = 0
    current_h = best_h.copy()
    current_c5 = c5_of_h(current_h)
    
    while beta < max_beta and time.time() - start_time < budget_s - 0.5:
        # Compute penalty term for sum
        def obj_and_grad(h):
            # Compute penalty
            sum_diff = np.sum(h) - target_sum
            penalty = lambda_penalty * sum_diff**2
            f = soft_max_conv(h, beta) + penalty
            grad = gradient_soft_max_conv(h, beta) + 2 * lambda_penalty * sum_diff * np.ones_like(h)
            return f, grad

        try:
            # Run L-BFGS-B 
            res = minimize(obj_and_grad, current_h, method='L-BFGS-B', jac=True,
                           bounds=[(0,1)]*len(current_h), options={'maxiter': max_iter_per_stage, 'ftol': 1e-6})
            current_h = res.x
            current_h = np.clip(current_h, 0, 1)
            current_h = proj_sum_l2(current_h, target_sum)
            current_c5 = c5_of_h(current_h)
        except:
            # fallback to previous step
            pass

        # Apply continuous refinement through continued β
        beta *= 2
        lambda_penalty *= 2
        if beta > max_beta:
            beta = max_beta
            
        # Check beta
        print(f"β = {beta:.3f}, C5 = {current_c5:.5f}")

        # Adjust budgeting
        if time.time() - start_time > budget_s - 0.5:
            break
    
    # Final symmetry preserving step
    current_h = (current_h + current_h[::-1]) / 2
    current_h = proj_sum_l2(current_h, target_sum)
    current_c5 = c5_of_h(current_h)
    print(f"Final refined C5 = {current_c5:.5f}")
    
    # Polishing with hill climb
    print("Starting hill climb...")
    h_final = current_h.copy()
    c5_final = current_c5
    n_hill = 0
    
    # Allow up to 5000 or until timeout
    for i in range(5000):
        if time.time() - start_time > budget_s - 0.1:
            break
        # Perturb a few indices
        r = min(5, n_points // 20)
        indices_perturb = np.random.choice(n_points, size=r, replace=False)
        h_trial = h_final.copy()
        for idx in indices_perturb:
            pert = np.random.uniform(-0.005, 0.005)
            h_trial[idx] = min(1.0, max(0.0, h_trial[idx] + pert))
        
        h_trial = proj_sum_l2(h_trial, target_sum)
        c5_trial = c5_of_h(h_trial)
        if c5_trial < c5_final:
            c5_final = c5_trial
            h_final = h_trial
        n_hill = i

    print(f"Polishing finished ({n_hill} steps). Final C5 = {c5_final:.5f}")
    return (h_final, c5_final, n_points)
