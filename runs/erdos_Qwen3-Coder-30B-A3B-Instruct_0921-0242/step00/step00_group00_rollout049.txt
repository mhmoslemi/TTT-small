import numpy as np
import time
from scipy.linalg import block_diag
from scipy.fft import fft, ifft

# Global constants
M = 8  # Number of frequency components per side (a_m, b_m)
n_points_1 = 256
n_points_2 = 512
n_freqs = 2 * M  # Total number of Fourier coefficients (2*M for a and b terms)

# Precompute the Fourier basis matrix Phi for n_points_1
def build_fourier_basis(n):
    dx = 2.0 / n
    i_vals = np.arange(n)
    Phi = np.zeros((n, n_freqs))
    for m in range(1, M + 1):
        Phi[:, m - 1] = np.cos(2 * np.pi * m * i_vals / n)  # a_m term
        Phi[:, M + m - 1] = np.sin(2 * np.pi * m * i_vals / n)  # b_m term
    return Phi, dx

Phi_1, dx_1 = build_fourier_basis(n_points_1)
Phi_2, dx_2 = build_fourier_basis(n_points_2)

# Precompute the pinv of Phi^T * Phi for fast projection
# Phi^T * Phi is a 2M x 2M matrix
PhiTPhi_1 = Phi_1.T @ Phi_1
pinv_1 = np.linalg.pinv(PhiTPhi_1)

PhiTPhi_2 = Phi_2.T @ Phi_2
pinv_2 = np.linalg.pinv(PhiTPhi_2)

# Optimize a single batch of restarts
def optimize_batch(cma_seed, n_points, Phi, dx, time_limit, best_c=None):
    np.random.seed(cma_seed)
    c_best = None
    c5_best = float('inf')
    
    # CMA-ES parameters
    lambda_ = 4 + int(np.floor(3 * np.log(n_freqs)))
    mu = lambda_ // 2

    # Initialize population
    pop = np.random.randn(lambda_, n_freqs) * 0.2  # Mean 0, std 0.2
    
    # Iterate with time limit
    start_time = time.time()
    while time.time() - start_time < time_limit:
        # Project all individuals to feasible
        pop_projects = []
        for ind in pop:
            h = Phi @ ind + 0.5  # Add offset to mean
            h_clipped = np.clip(h, 0, 1)
            err = h_clipped - h
            delta = pinv_1 @ err
            ind_corr = ind + delta
            pop_projects.append(ind_corr)
        
        pop_projects = np.array(pop_projects)
        
        # Evaluate objective for all individuals
        c5_vals = []
        for ind in pop_projects:
            h = Phi @ ind + 0.5
            corr = np.correlate(h, 1 - h, mode='full') * dx
            c5 = np.max(corr)
            c5_vals.append(c5)
        
        # Sort by objective and select best
        idxs = np.argsort(c5_vals)
        pop = pop_projects[idxs[:mu]]
        c5_vals = np.array(c5_vals)[idxs[:mu]]
        
        # Update best
        if np.min(c5_vals) < c5_best:
            c5_best = np.min(c5_vals)
            c_best = pop[np.argmin(c5_vals)].copy()  # Make a copy
            
        # CMA-ES update step
        if len(pop) >= 2:
            # Compute mean
            mean = np.mean(pop, axis=0)
            # Compute covariance matrix
            centered = pop - mean
            
            # Keep the mean feasible after update
            mean_h = Phi @ mean + 0.5
            mean_h_clipped = np.clip(mean_h, 0, 1)
            mean_err = mean_h_clipped - mean_h
            mean_corr = pinv_1 @ mean_err
            mean = mean + mean_corr
            
            # Update populations with mutations
            # Just sample from normal
            pop = np.random.randn(lambda_, n_freqs) * 0.2
            pop += mean
        
        # Occasionally re-inject diversity
        if np.random.rand() < 0.2:
            pop[np.random.randint(0, lambda_)] = np.random.randn(n_freqs) * 0.2
            
    return c_best, c5_best

# Project feasible h to interval [0, 1] (interior projection for bound constraints)
def project_h_to_box(h):
    return np.clip(h, 0, 1)

# Refined gradient descent local search
def refine_with_gradient(c, Phi, dx, n_points):
    h = Phi @ c + 0.5
    
    # The maximum will be at some lag k* which we can find with convex search
    # But for speed, we'll optimize the function by finding actual k*
    corr = np.correlate(h, 1 - h, mode='full') * dx
    k_max = np.argmax(corr)
    max_val = corr[k_max]
    
    # Now find the gradient using analytically-derived set of indices
    # We will directly apply the functional derivative to get approximate gradient
    # ∂C5/∂h_i = (1 - h_{i+k*}) - h_{i-k*}
    # This needs to be approximated carefully using the exact correlation structure
    
    # Simple finite difference for demonstration of method
    # Fill in a rough gradient approximation - better than nothing
    
    # The fastest way for exact gradient in full correlation is from the example from literature:
    # A naive stamping of ∂C5/∂h
    # ∂C5/∂h_i = 1 - h_{i+1} (for 1-period) if we consider padding blank
    # Actually, for circulant correlation we know:
    # Components of the derivative w.r.t h_i
    # C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    # So ∂C5 / ∂h_i = -U_i + V_i, where
    # U_i = h_{i+k_star}, V_i = h_{i-k_star} (with periodic indexing)
    max_k = np.argmax(corr)
    
    # Build in place the "gradient" matrix for r.h.s of the Hessian update:
    # Use condition from correlation: it's the convolution, so symmetrized propagator:
    h_grad = np.zeros_like(h)
    # Renormalized from'full'-mode get the same support
    # Think we can extract k by full corrupt and taking ones with ...
    # Instead, build simple gradient in periodic fashion:
    k = max_k - (n_points-1)
    for i in range(n_points):
        idx1 = (i + k) % n_points
        idx2 = (i - k) % n_points
        h_grad[i] = (1 - h[idx1]) - h[idx2]
        
    # Back-propagate
    grad_c = Phi.T @ h_grad
    
    # Perform gradient descent step
    steps = 200
    alpha0 = 1e-3
    for t in range(1, steps + 1):
        alpha = alpha0 / (1 + t)
        c -= alpha * grad_c
        
        # Project result
        h = Phi @ c + 0.5
        h = project_h_to_box(h)
        err = h - (h - 0.5)
        delta = pinv_1 @ err
        c += delta
    
    return c

def run(seed=42, budget_s=600, **kwargs):
    # Global time limit
    start_time = time.time()
    
    # Init
    n_points = n_points_1
    
    # Perform first batch on coarser grid
    time_split_1 = 0.4 * budget_s
    best_c = None
    best_c5 = float('inf')
    
    # Many restarts for good coverage
    num_restarts = 15
    for i in range(num_restarts):
        if time.time() - start_time > budget_s - 0.5:
            break
        cma_seed = seed + i
        c_opt, c5_opt = optimize_batch(cma_seed, 
                                       n_points, 
                                       Phi_1, 
                                       dx_1, 
                                       time_split_1 / num_restarts, 
                                       best_c)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_c = c_opt.copy()
    
    # Early stop if we've hit our target
    if best_c5 <= 0.3807:
        pass  # Continue to refine...
    else:
        # Double resolution and continue refining
        n_points = n_points_2
        Phi = Phi_2
        dx = dx_2
        pinv = pinv_2
        
        # Interpolate coeffs
        c_before = best_c
        
        # The way to preserve same structured solution is to keep coefficients as is
        # and recompute on finer grid. That is not easily possible due to different basis sizes.
        # Better to reinitialize optimally
        best_c = np.copy(c_before)
        
        # Do second round of search with higher resolution
        time_split_2 = (1.0 - 0.4) * budget_s
        num_restarts = 10
        for i in range(num_restarts):
            if time.time() - start_time > budget_s - 0.5:
                break
            cma_seed = seed + num_restarts + i
            c_opt, c5_opt = optimize_batch(cma_seed,
                                           n_points, 
                                           Phi_2, 
                                           dx_2,
                                           time_split_2 / num_restarts,
                                           best_c)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_c = c_opt.copy()
    
    # Final refinement with gradient descent
    refined_c = refine_with_gradient(best_c, Phi_1, dx_1, n_points_1)
    h_vals = Phi_1 @ refined_c + 0.5
    if n_points == 512:
        # Interpolate h from coarse version
        h_coarse = Phi_1 @ best_c + 0.5
        # Reconstruct at finer grid
        h_vals = Phi_2 @ (best_c[:2*M]) + 0.5
    
    # Ensure normalization
    integral = np.sum(h_vals) * (2.0 / len(h_vals))
    correction = (1.0 - integral) / len(h_vals)
    h_vals = np.clip(h_vals + correction, 0, 1)

    # Final C5
    corr = np.correlate(h_vals, 1 - h_vals, mode='full') * (2.0 / len(h_vals))
    final_c5 = np.max(corr)
    
    return (h_vals.tolist(), float(final_c5), len(h_vals))
