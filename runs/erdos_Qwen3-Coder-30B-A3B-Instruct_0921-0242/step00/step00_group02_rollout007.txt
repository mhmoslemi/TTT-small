import time
import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from math import floor, ceil

# Global variables for time tracking
t0 = None
budget_s = 600

def evaluate_time_left():
    global t0, budget_s
    return time.time() - t0 < budget_s - 5

def clamp(x, low, high):
    return max(low, min(high, x))

def compute_overlap_intervals(a, l):
    """Compute max overlap O(k) for intervals a with length l."""
    m = len(a)
    n = 2 * m * m
    breakpoints = []
    for i in range(m):
        for j in range(m):
            # Breakpoints: (a_i - a_j) ± l
            k1 = (a[i] - a[j]) - l
            k2 = (a[i] - a[j]) + l
            # Clamp to [0, 2]
            breakpoints.append(clamp(k1, 0, 2))
            breakpoints.append(clamp(k2, 0, 2))
    # Make unique and sorted
    breakpoints = sorted(set(breakpoints))
    max_o = 0
    for k in breakpoints + [0, 2]:
        total = 0
        for i in range(m):
            for j in range(m):
                diff = abs((a[i] - a[j]) - k)
                contrib = max(0, l - diff)
                total += contrib
        max_o = max(max_o, total)
    return max_o

def project_to_ordered_simplex(a, l):
    """Project vector a to interval endpoints of length l satisfying ordering."""
    m = len(a)
    gaps = np.diff(a)
    total_gap = 2 - m * l
    if total_gap < 0:
        # Already too long, shrink the intervals
        a = np.array(a) - (np.sum(a) - (2 - m * l)) / m
        gaps = [l] * (m - 1)
        gaps = np.array(gaps)
    else:
        # Add total gap to the first gap and redistribute
        gaps = np.array([l] * (m - 1))
        # We'll recompute a directly now
    # Correct the ordering
    a = np.zeros(m)
    a[0] = 0
    cum_sum = 0
    for i in range(1, m):
        cum_sum += gaps[i-1]
        a[i] = cum_sum
        if i < m - 1:
            a[i] = max(a[i], a[i-1] + l)
    return a

def direct_optimization(m, l):
    """Phase A - deterministic Lipschitz branch-and-bound."""
    # Initialize
    box_low = np.zeros(m)
    box_high = np.ones(m) * 2
    best_val = float('inf')
    best_a = None
    # Sample points for estimation
    n_samples = 2000
    for _ in range(n_samples):
        if not evaluate_time_left():
            break
        a = np.random.uniform(0, 2-l, m)
        # Arrange so that they are ordered
        a = np.sort(a)
        # But we need gaps > 0 and final interval at most 2
        for i in range(m-1):
            if i == 0:
                # a[0] can be anything in [0, 2 - l - (m-1)*l] 
                # but minimum is 0
                a[i+1] = max(a[i] + l, a[i+1])
            else:
                # Must satisfy a[i+1] >= a[i] + l
                a[i+1] = max(a[i] + l, a[i+1])
        a[-1] = min(a[-1], 2 - l)
        # Check feasibility
        if a[-1] < a[0] + l:
            continue
        # Compute overlap
        o = compute_overlap_intervals(a, l)
        if o < best_val:
            best_val = o
            best_a = a.copy()
    return best_a, best_val

def discretize_and_evaluate(a, l, n):
    """Convert interval vector to bin vector, adjust for integral constraint."""
    # Ensure m = len(a)
    m = len(a)
    dx = 2.0 / n
    h = np.zeros(n)
    # Add intervals to h
    for i in range(m):
        start = max(0, int(np.floor(a[i] / dx)))
        end = min(n, int(np.ceil((a[i] + l) / dx)))
        for j in range(start, end):
            if (j * dx >= a[i] and j * dx < a[i] + l) or \
               ((j+1)*dx > a[i] and (j+1)*dx <= a[i] + l) or \
               (j*dx <= a[i] and (j+1)*dx >= a[i] + l):
                  h[j] = 1
    # Ensure integral is 1 (i.e. sum = n / 2)
    actual_sum = np.sum(h)
    target_sum = n / 2
    if actual_sum == target_sum:
        # All fine
        pass
    elif actual_sum < target_sum:
        # Add 1s to both sides
        # find closest lower and upper bounds
        diffs_up = []
        diffs_down = []
        for i in range(n):
            if h[i] == 0:
                left = np.abs(i * dx - a[0])
                right = np.abs(i * dx + dx - (a[-1] + l))
                diffs_up.append((left, i))
                diffs_down.append((right, i))
        # Sort by diff (closest first)
        diffs_up.sort()
        diffs_down.sort()
        count = 0
        for _, idx in diffs_up:
            if count < target_sum - actual_sum:
                h[idx] = 1
                count += 1
            else:
                break
        for _, idx in diffs_down:
            if count < target_sum - actual_sum:
                h[idx] = 1
                count += 1
            else:
                break
    else:
        # Remove some 1s
        # Find closest to edges
        diffs_up = []
        diffs_down = []
        for i in range(n):
            if h[i] == 1:
                left = np.abs(i * dx - a[0])
                right = np.abs(i * dx + dx - (a[-1] + l))
                diffs_up.append((left, i))
                diffs_down.append((right, i))
        diffs_up.sort()
        diffs_down.sort()
        count = 0
        for _, idx in diffs_up:
            if count < actual_sum - target_sum:
                h[idx] = 0
                count += 1
            else:
                break
        for _, idx in diffs_down:
            if count < actual_sum - target_sum:
                h[idx] = 0
                count += 1
            else:
                break
    # Recompute C5 = max_k int h(x) (1 - h(x + k))
    corr = correlate(h, 1-h, mode='full')
    c5 = 1 - np.max(corr) * dx
    return h, c5

def local_swap_improvement(h):
    # Find critical k
    corr = correlate(h, 1-h, mode='full')
    k_star = np.argmax(corr)
    # k_star is in the middle of correlation array, so shift it to be index in [-1,1]
    # Correlation index from 0:2n-2, k_star = n-1 mapping to shift k=0.
    k_star_shift = k_star - (len(h) - 1)
    # This is the shift we want to penalize least
    start_pos = max(0, k_star_shift)  # If k_star_shift is negative, it's already a good point
    # Enumerate swaps
    best_improvement = 0
    best_i, best_j = None, None
    n = len(h)
    # Evaluate some candidate swaps, limit to reasonable number (say 500)
    for _ in range(500):
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        # Compute exact change in correlation at k_star
        coef = (1 - 2 * h[i]) * (1 - 2 * h[j])
        old_i = 0
        if i + k_star_shift >= 0 and i + k_star_shift < n:
            old_i = h[i]
        old_j = 0
        if j + k_star_shift >= 0 and j + k_star_shift < n:
            old_j = h[i]
        change = coef * (1 - int(old_i) - int(old_j))  
        if change > best_improvement:
            best_improvement = change
            best_i, best_j = i, j
    # Perform best swap
    if best_i is not None:
        h[best_i], h[best_j] = h[best_j], h[best_i]
    return h

# Initialize global timer
def run(seed=42, budget_s=600):
    global t0, budget_s
    t0 = time.time()
    budget_s = budget_s
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = None
    m_values = [4, 5, 6, 7, 8]
    for m in m_values:
        if not evaluate_time_left():
            break
        l = 1.0 / m
        # Phase A: direct optimization
        # Start with a uniform initial guess
        a = np.linspace(0, 2 - l, m)
        # Direct optimization on m-dimensional space
        # Uniform sampling to see performance 
        sample_points = 100
        best_val = float('inf')
        best_a = None
        for k in range(sample_points):
            if not evaluate_time_left():
                break
            # Create a random initialization with spacing l
            a_try = np.random.rand(m) * (2 - l)
            a_try = np.sort(a_try)
            a_try[0] = 0
            for i in range(1, m):
                a_try[i] = max(a_try[i-1] + l, a_try[i])
            a_try[-1] = min(a_try[-1], 2 - l)
            o = compute_overlap_intervals(a_try, l)
            if o < best_val:
                best_val = o
                best_a = a_try.copy()
        a = best_a
        # Phase B: differential evolution
        bounds = [(0, 2-l)] * m
        res = differential_evolution(
            lambda x: compute_overlap_intervals(x, l),
            bounds,
            seed=seed,
            maxiter=500,
            popsize=10 * m,
            mutation=(0.5, 1.0),
            recombination=0.7,
            callback=lambda x, convergence: None,
            disp=False
        )
        a_opt = res.x
        a_opt = np.sort(a_opt)
        a_opt[0] = 0
        for i in range(1, m):
            a_opt[i] = max(a_opt[i-1] + l, a_opt[i])
        o_opt = compute_overlap_intervals(a_opt, l)
        best_a = a_opt
        # Discretize with different n values
        n_values = [36, 42, 48, 60]
        for n in n_values:
            if not evaluate_time_left():
                break
            h, c5 = discretize_and_evaluate(best_a, l, n)
            h = local_swap_improvement(h)
            corr = correlate(h, 1-h, mode='full')
            c5 = 1 - np.max(corr) * (2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_n = n
        # Do partial swap polishing so it's off anyway
        if best_h is not None:
            h_p = best_h.copy()
            for _ in range(100):
                h_p = local_swap_improvement(h_p)
                corr = correlate(h_p, 1-h_p, mode='full')
                c5_temp = 1 - np.max(corr) * (2.0 / len(h_p))
                if c5_temp < best_c5:
                    best_c5 = c5_temp
                    best_h = h_p.copy()
                    best_n = len(h_p)
    return (best_h.tolist(), best_c5, best_n)
