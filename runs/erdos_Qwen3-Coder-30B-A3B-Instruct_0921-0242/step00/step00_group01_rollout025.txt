import numpy as np

def project_box_affine(h, target_sum):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Project onto affine constraint sum = target_sum
    mean_diff = (np.sum(h) - target_sum) / len(h)
    h -= mean_diff
    # Clip again after projection
    h = np.clip(h, 0, 1)
    return h

def autocorr(h):
    # Returns circular autocorrelation: sum h[i] h[i+k] over all i
    # Using FFT for efficiency
    H = fft(h)
    autocorr_vals = np.real(ifft(H * np.conj(H)))
    # Return only first half to match standard definition
    return autocorr_vals[:len(h)]

def compute_c5(h, dx):
    # Use full convolution to compute overlaps
    conv = np.correlate(h, 1-h, mode='full')
    # Scale by dx; the duplicates are already handled in mode='full'
    # For a circular pattern, the useful part is of size 2*n - 1
    # But we want only the "max_k" over valid lag values. 
    # Specific interpretation: For lag k=0,...,n-1, exactly what overlap is
    # what we compute for maximum lag size. But use mode='full' for maximum
    # overlap range, then grab relevant central portions (n terms centered about 0 lag).
    n = len(h)
    # shifts from 0 to n-1, and back, mapping to full result of size 2*n-1
    # we're computing k=0,1,...,n-1 as shifts
    # Shift indices: 0,...,n-1 are valid in this, and the src is not wrapped.
    result = []
    for k in range(n):
        # convolution index = n-1 + k (from full) 
        # indexing conv[0..2n-2]  
        # But we want result of mode 'full' in valid range k=0..n-1  
        # Actually confirm using numpy.correlate, works directly.
        # conv is size[2*n-1]. We want k=0 in result[0] to k=n-1 in result[n-1]:
        # if mode == full, np.correlate(h, 1-h, mode="full") makes result of length 2*n - 1
        # conv[n-1] corresponds to k=0  
        # conv[m] corresponds to shift (m - (n-1)) 
        # so, shift k corresponds to index k + (n-1)
        # For k = 0..n-1, we get indices n-1 to 2*n-2
        # But for shift (k) or index (k+n-1), we should pass index (k + (n-1)) in conv
        # The index to access for shift k is k + (n - 1)
        val = dx * conv[k + (n - 1)]
        result.append(val)
    # Now find C5 = max result
    return max(result)

def compute_c5_fast(h, dx):
    # Analytic method for computation of C5
    # The long array convolution mode != 'valid' for the mutual overlaps
    # Reduce convolution to the max of the simply computed overlap for lags k=0,...,n-1
    # But 

    # Directly using manual loop or a faster method
    n = len(h)
    max_val = 0.0
    for k in range(n):
        # Compute h(i) * (1 - h(i + k)), wrapping around
        tmp = 0.0
        for i in range(n):
            j = (i + k) % n  # This is circular
            tmp += h[i] * (1 - h[j])
        val = dx * tmp
        if val > max_val:
            max_val = val
    return max_val


def optimize_h(h0, dx, max_iter, lr_schedule, seed=None):
    h = np.array(h0)
    n = len(h)
    target_sum = n / 2.0
    best_c5 = compute_c5(h, dx) 
    best_h = h.copy()
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > 580:  # time buffer
            break
        # Autocorrelation: imagine we want max h_i * (1 - h_{i+j}) for small j. 
        # We want to maximize min( h_i * (1 - h_{i+j}) for fixed j ) over j
        # That's hard, so we will do gradient ascent on the worst-case overlap. 

        # Compute all overlaps O_k = dx * ( sum h_i - sum h_i h_{i+k} )
        # That is, a matrix where rows correspond to lag k:
        # O_k = dx * ( n/2  - sum_j h_j h_{j+k} )  
        # O_k = dx * ( S - (h * h)_{k} )  where(*) is circular conv
        # Use FFT-based autocorrelation, 
        aut = np.real(ifft(fft(h) * np.conj(fft(h))))
        # Index stores the lag that is worst, i.e., lowest value of overlap 
        lag = np.argmin(aut) # k such that (dx*(S - aut[k])) achieves max O_k, i.e. min aut[k]
        # The gradient of O_k w.r.t. h at a fixed k is:
        # - dx * ( h_{k} + h_{-k} ) ( evenly distributed signs, circular)
        # Derivative of O_k in terms of h = dx * ( 0 - (d/dh) (h * h)_k ) 
        # (d/dh) (h * h)_k = h_{k} + h_{-k }, in circular notation  
        # So, gradient w.r.t h for k* = argmin aut,
        # d/dh_k (O_k*) = -dx * (h_{k*} + h_{-k*})  
        grad = -dx * (np.roll(h, lag) + np.roll(h, -lag))  # That gives appropriate indexing
        h_new = h + grad * lr_schedule(it)
        h = project_box_affine(h_new, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
