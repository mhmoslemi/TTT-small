import numpy as np
import time
from scipy import signal

def project_simplex_box(v, total, lo=0., hi=1.):
    # Project v onto the simplex {x: sum(x) = total, lo <= x <= hi}
    v = np.array(v)
    n = len(v)
    # Sort in descending order
    s = np.sort(v)[::-1]
    # Find threshold
    cumsum = np.cumsum(s)
    rho = np.where(s > (total - cumsum) / np.arange(1, n + 1))[0]
    if len(rho) == 0:
        theta = (cumsum[-1] - total) / n
        projected = np.maximum(lo, np.minimum(hi, v - theta))
    else:
        rho = rho[-1] + 1
        theta = (cumsum[rho - 1] - total) / rho
        projected = np.maximum(lo, np.minimum(hi, v - theta))
    return projected

def compute_c5(h, dx):
    # Use FFT for fast correlation computation
    corr = np.real(np.fft.ifft(np.fft.fft(h) * np.fft.fft(1 - h).conj()))
    return np.max(corr) * dx

def worst_lag(h):
    # Compute full correlation with 1-h
    n = len(h)
    corr = np.real(np.fft.ifft(np.fft.fft(h) * np.fft.fft(1 - h).conj()))
    k_max = np.argmax(corr)
    return k_max, corr[k_max]

def gradient_worst(h, k):
    # Gradient of overlap at lag k
    n = len(h)
    grad = np.zeros(n)
    # For periodic boundary conditions
    for i in range(n):
        grad[i] = h[(i + k) % n] + h[(i - k) % n]
    return grad
