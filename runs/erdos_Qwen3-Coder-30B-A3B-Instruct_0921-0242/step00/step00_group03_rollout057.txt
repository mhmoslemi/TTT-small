import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate C5 using the correlation approach."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_c5_binary(h):
    """Evaluate C5 for binary h assuming N=100."""
    # This matches the logic of the problem
    n = len(h)
    # Normalize to area 1
    dx = 2.0 / n
    # Direct computation of non-zero shifts correlation (automatically done by numpy)
    corr = correlate(h, 1 - h, mode='full')
    # Focus on non-zero shifts (that's what we're interested in)
    # The maximum among non-zero shifts
    non_zero_corr = corr[n-1:]  # picks all non-zero
    max_corr = np.max(non_zero_corr)
    return max_corr * dx

def compute_overlaps(h, n):
    """Compute all overlaps for a binary h."""
    A = np.zeros(n - 1)
    for s in range(1, n):
        A[s - 1] = sum(h[i] * h[i + s] for i in range(n - s))
    return A

def improve_binary_solution(h_bin):
    """Run hill-climbing to improve the binary solution."""
    n = len(h_bin)
    A = compute_overlaps(h_bin, n)
    t = int(np.min(A))
    best_t = t
    best_h = h_bin.copy()
    # Track execution time
    start_time = time.time()
    time_limit = 100  # seconds

    while time.time() - start_time < time_limit:
        improved = False
        # Sample possible swaps
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Test some random swaps
        for _ in range(20):  # tried 20 swaps per iteration
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            # Simulate swap
            h_new = h_bin.copy()
            h_new[i1] = 0
            h_new[i0] = 1
            # Compute overlap changes
            A_new = compute_overlaps(h_new, n)
            new_t = int(np.min(A_new))
            if new_t > best_t:
                best_t = new_t
                best_h = h_new.copy()
                improved = True
                break  # Once we find one good improvement, restart

        if not improved:
            break

    # Final check
    final_A = compute_overlaps(best_h, n)
    final_t = int(np.min(final_A))
    final_c5 = 1.0 - 2*final_t/n

    return best_h, final_c5

def solve_milp(n_points, bin_seed=None):
    """Solve the MILP to maximumimize the minimal exposure."""
    np.random.seed(bin_seed)
    if n_points % 2 != 0:
        raise ValueError("n must be even")
    half = n_points // 2

    # Variables
    h = cp.Variable(n_points, boolean=True)
    y = cp.Variable((n_points - 1, n_points - 1), boolean=True)
    t = cp.Variable(integer=True)

    # Constraints
    constraints = [
        cp.sum(h) == half,
    ]

    # Linearization constraints for y[i,s] = h[i] * h[i+s]
    for s in range(1, n_points):
        for i in range(n_points - s):
            constraints.append(y[i, s - 1] <= h[i])
            constraints.append(y[i, s - 1] <= h[i + s])
            constraints.append(y[i, s - 1] >= h[i] + h[i + s] - 1)

    # Overlap lower bound for each shift s
    for s in range(1, n_points):
        constraints.append(t <= cp.sum([y[i, s - 1] for i in range(n_points - s)]))

    # Objective
    objective = cp.Maximize(t)

    # Problem
    problem = cp.Problem(objective, constraints)

    # Solve
    max_seconds = 300
    try:
        problem.solve(solver=cp.CBC, verbose=False, max_seconds=max_seconds)
        sol_h = np.round(h.value).astype(int)
    except Exception as e:
        print("MILP failed:", e)
        sol_h = np.random.randint(0, 2, n_points)
        while np.sum(sol_h) != half:
            # adjust to make it sum to half
            diff = np.sum(sol_h) - half
            if diff > 0:
                # Remove a 1
                idx = np.where(sol_h == 1)[0][0]
                sol_h[idx] = 0
            else:
                # Add a 1
                idx = np.where(sol_h == 0)[0][0]
                sol_h[idx] = 1

    return sol_h

def run(seed=42, budget_s=600, **kwargs):

    best_h = None

    # Try first with n=100
    n = 100

        h_bin = solve_milp(n, seed)
        h_bin_100 = h_bin.copy()
        # Now improve locally
        h_after_improvement, c5_after_improvement = improve_binary_solution(h_bin_100)
        if c5_after_improvement < best_c5:
            best_h = h_after_improvement
            best_c5 = c5_after_improvement

    # If still room, try n=120 if it gives a better result
    if time.time() - start_time < budget_s - 20 and best_c5 > 0.37:
        n_large = 120
        h_bin_large = solve_milp(n_large, seed + 1)  # avoid exact seeds
        h_bin_large = h_bin_large.copy()
        h_improved_large, c5_large = improve_binary_solution(h_bin_large)
        if c5_large < best_c5:
            best_h = h_improved_large
            best_c5 = c5_large

    # Optional polishing (if we have enough time and not too tight already)
    time_left = budget_s - (time.time() - start_time)
    if time_left > 30 and best_h is not None and best_c5 > 0.375:
        # Convert to continuous
        h_cont = best_h.astype(float)
        h_cont += np.random.uniform(-0.01, 0.01, len(h_cont))
        h_cont = np.clip(h_cont, 0, 1)
        # Optimization parameters
        dx = 2.0 / len(h_cont)
        # Use trust-constr or SLSQP for local refinement
        bounds = [(0, 1) for _ in range(len(h_cont))]
        # Add one linear constraint: sum(h) * dx = 1
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1}
        ]

        # Objective
        def obj(h):
            corr = correlate(h, 1-h, mode='full')
            c5 = np.max(corr) * dx
            return c5

        # Local refine
        # Rerun to get proper result if results are substantial
        try:
            # We keep it simple here by repeating the evaluation directly
            # to make it fast.
            c5_final = evaluate_c5(h_cont)
            if c5_final < best_c5:
                best_h = h_cont
                best_c5 = c5_final
        except:

    # Upsample if we have enough space left and time and it's still favorable
    if best_h is not None and len(best_h) <= 400:
        # Use simple replication: replicate each sample 3 times
        r = 3
        h_upsampled = np.repeat(best_h, r)

            best_c5 = new_c5
            best_h = h_upsampled

    # Ensure the final h satisfies the integral constraint
    if best_h is not None:

        assert abs(np.sum(best_h) * dx - 1) < 1e-10, "Integral constraint violated"
