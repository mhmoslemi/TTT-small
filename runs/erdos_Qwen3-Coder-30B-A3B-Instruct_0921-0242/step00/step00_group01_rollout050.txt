import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    n = len(h)
    # Compute the convolution h * (1 - h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Take only the relevant part (overlapping shifts)
    corr = conv[len(h)-1:]
    # Total integral is dx * (sum(h) - max_k corr_k)
    # Since sum(h) * dx = 1 (so sum(h) = n/2), and all values x in [0,1], 
    # we have 
    return dx * (n/2 - np.max(corr))

def block_sweep_3block(n_base=400, delta=0.02):
    best_h = None
    best_c5 = float('inf')
    h_grid = np.linspace(0, 2, n_base)
    # sweep over a,b in [0,2], fixed spacing
    linspace_a = np.arange(0, 2-delta, delta)
    linspace_b = np.arange(delta, 2, delta)
    
    for a in linspace_a:
        for b in linspace_b:
            if b <= a:
                continue
            # p = (1-a)/(b-a)  
            p = (1-a)/(b-a)
            if p < 0 or p > 1:
                continue
            h = np.zeros(n_base)
            idx_a = np.searchsorted(h_grid, a, side='left')
            idx_b = np.searchsorted(h_grid, b, side='left') 
            h[:idx_a] = 1.0
            h[idx_a:idx_b] = p
            h[idx_b:] = 0.0
            # adjust for exact constraint
            total = np.sum(h) * (2/n_base)
            if total > 0:
                h = h * (n_base / (2 * total))
                h = np.clip(h, 0, 1)
            c5 = evaluate_c5(h, 2/n_base)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def block_sweep_4block(n_base=400, delta=0.04):
    best_h = None
    best_c5 = float('inf')
    h_grid = np.linspace(0, 2, n_base)

    linspace_a = np.arange(0, 2-delta, delta)
    linspace_b = np.arange(delta, 2-delta, delta)
    linspace_c = np.arange(delta, 2, delta)
    
    for a in linspace_a:
        for b in linspace_b:
            if b <= a:
                continue
            for c in linspace_c:
                if c <= b:
                    continue
                # Determine two unknowns p, q from 3 equations
                # Integral constraint: a*1 + (b-a)*p + (c-b)*q + (2-c)*0 = 1
                # Two unknowns, assume equal mass for middle blocks:
                # (b-a)*p = (c-b)*q => p = q * (c-b)/(b-a)
                # Integral -> (b-a)*p = 1 - a - (c-b)*q
                # Sub into above:
                # q * (c-b)/(b-a) = (1-a - (c-b)*q)/(b-a)
                # => q * (c-b) = (1-a - (c-b)*q)
                # => q * (c-b + (c-b)) = (1-a)
                # => q * 2*(c-b) = 1-a => q = (1-a)/(2*(c-b))
                # => p = q * (c-b)/(b-a) = (1-a)/[2*(c-b)] * (c-b)/(b-a) = (1-a)/(2*(b-a))
                q = (1-a) / (2 * (c - b))
                p = (1-a) / (2 * (b - a))
                if p < 0 or p > 1 or q < 0 or q > 1:
                    continue
                h = np.zeros(n_base) 
                idx_a = np.searchsorted(h_grid, a, side='left')
                idx_b = np.searchsorted(h_grid, b, side='left')
                idx_c = np.searchsorted(h_grid, c, side='left')
                h[:idx_a] = 1.0
                h[idx_a:idx_b] = p
                h[idx_b:idx_c] = q
                h[idx_c:] = 0.0
                # Normalize to sum  = n/2
                total = np.sum(h) * (2/n_base)
                if total > 0:
                    h = h * (n_base / (2*total))
                    h = np.clip(h, 0, 1)
                c5 = evaluate_c5(h, 2/n_base)
                if c5 < best_c5:
                    best_c5 = c5  
                    best_h = h.copy()
    return best_h, best_c5

# Stochastic parameterization: mixture of indicator functions
def gaussian_bump_iter(h, centers, amplitudes, lengths, n):
    # Compute indicator functions
    grid = np.linspace(0, 2, n)
    h_total = np.zeros(n)
    for c, a, l in zip(centers, amplitudes, lengths):
        lo = c - l/2
        hi = c + l/2
        mask = (grid >= lo) & (grid <= hi)
        h_total += a * mask
    h_total = np.clip(h_total, 0, 1)
    mass = np.sum(h_total) * (2.0/n)
    if mass > 0:
        h_total *= (n/2) / mass
        h_total = np.clip(h_total, 0, 1)
    return h_total

# DE fitness function on low dim
def de_fitness(params, n=200):
    # parameters: center1, amp1, center2, amp2, center3, amp3, lengths  
    centers = params[0:3]
    amplitudes = params[3:6]
    lengths = [0.1] * 3  
    h = gaussian_bump_iter(np.zeros(n), centers, amplitudes, lengths, n)
    return -evaluate_c5(h, 2.0/n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    log = []
    best_c5 = float('inf')
    best_h = None

    # Stage 1: Block sweep (3-block) - fast
    if time.time() - start_time < budget_s - 100:
        h1, c5_1 = block_sweep_3block()
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1
        log.append(("Block sweep 3block", c5_1))

    if time.time() - start_time < budget_s - 100:
        h2, c5_2 = block_sweep_4block()
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
        log.append(("Block sweep 4block", c5_2))

    # Stage 2: DE search on mix of gaussian bumps
    if time.time() - start_time < budget_s - 150:
        # Define bounds for the 6-D parameterization 
        bounds = [(0,2), (0,1), (0,2), (0,1), (0,2), (0,1)]  # centers, amplitudes
        # Add length and a multistart to prevent local optima  
        def de_func(params):
            return de_fitness(params, n=200)
        optimization_result = differential_evolution(
            de_func, 
            bounds, 
            seed=seed,
            maxiter=500, 
            popsize=30,
            callback=lambda x, convergence: None,
            disp=False
        )
        h_de = gaussian_bump_iter(np.zeros(200), 
                                  [optimization_result.x[0], optimization_result.x[2], optimization_result.x[4]], 
                                  [optimization_result.x[1], optimization_result.x[3], optimization_result.x[5]], 
                                  [0.1]*3, 200)
        c5_de = evaluate_c5(h_de, 2.0/200)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de
        log.append(("DE", c5_de))

    # Stage 3: Full-scale gradient ascent with multi-resolution
    if time.time() - start_time > budget_s - 100:
        return (best_h.tolist(), best_c5, len(best_h))

    # Start gradient ascent from best obtained so far
    n_curr = 150 
    ratio = 2
    h = np.interp(np.linspace(0, 2, n_curr), np.linspace(0,2, len(best_h)), best_h)
    h = np.clip(h, 0, 1)
    h = h * (n_curr / (2 * np.sum(h) * 2/n_curr)) # normalize to sum=n/2
    h = np.clip(h, 0, 1)

    best_h_fin = h.copy()
    best_c5_fin = evaluate_c5(h, 2/n_curr)

    step_size = 0.5
    min_step = 1e-6
    decay = 0.98
    iter_since_improve = 0
    max_iter_no_improve = 200

    def project(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h) * (2/n_curr)
        if total > 0:
            scale_factor = (n_curr / 2) / total  
            h = h * scale_factor
        h = np.clip(h, 0, 1)
        return h

    def circular_shift(h, k):
        return np.roll(h, k)

    max_iter = 5000 
    curr_iter = 0
    
    while (curr_iter < max_iter) and (time.time() - start_time < budget_s - 50):
        if iter_since_improve > max_iter_no_improve:
            break
            
        # FFT-based autocorrelation
        H = fft(h)
        A = ifft(np.conj(H) * H).real
        k_star = np.argmin(A)
        grad = 2 * circular_shift(h, k_star)
        h_new = h + step_size * grad
        h_new = project(h_new)
        c5_new = evaluate_c5(h_new, 2/n_curr)
        
        if c5_new < best_c5_fin:
            best_h_fin = h_new.copy()
            best_c5_fin = c5_new
            iter_since_improve = 0
        else:
            iter_since_improve += 1
            
        h = h_new
        step_size *= decay
        if step_size < min_step:
            step_size = min_step

        # Up/downscale resolution  
        if curr_iter == 1500:
            n_new = 300
            if n_new <= 1000:
                h = np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)
                n_curr = n_new
                step_size = 0.1
                ratio = 2
                
        curr_iter += 1

    final_h = project(best_h_fin)
    final_c5 = evaluate_c5(final_h, 2/n_curr)
    return (final_h.tolist(), final_c5, len(final_h))
