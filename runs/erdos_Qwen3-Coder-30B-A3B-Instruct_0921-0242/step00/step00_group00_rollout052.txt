import numpy as np
import time
from scipy.fft import ifft, fft
from scipy.optimize import minimize
from math import floor

def project_simplex_box(v, total, lo=0., hi=1.):
    """
    Project v onto the box-constrained simplex: sum(v') = total, lo <= v' <= hi.
    """
    v = np.asarray(v)
    n = len(v)
    # Clip to [lo, hi]
    v_clipped = np.clip(v, lo, hi)
    s = np.sum(v_clipped)
    if abs(s - total) < 1e-12:
        return v_clipped
    elif s < total:
        # Add to all at upper bound to reach total
        delta = (total - s) / n
        v_result = v_clipped + delta
        v_result = np.clip(v_result, lo, hi)
    else:
        # Need to subtract from all elements
        # Sort v_clipped descending
        sorted_idx = np.argsort(v_clipped)[::-1]
        sorted_v = v_clipped[sorted_idx]
        # Find threshold such that sum of clipped values is total
        # Iterate to find the "break point"
        cumsum = np.cumsum(sorted_v[::-1])
        # Find largest i such that (cumsum[i] + i * sorted_v[i]) <= total
        # This is a bit tricky, we'll use a simpler method
        # The change from projecting to simplex is small.
        # Use Newton-like method by sorting
        v_sorted =np.sort(v_clipped)[::-1]
        excess = s - total
        i = 0
        # Greedy thresholding
        while i < len(v_sorted) and excess > 0:
            # Value we can decrease for remaining elements
            if i == 0:
                threshold = v_sorted[i]
            else:
                threshold = (cumsum[i-1] - (total - (n-i)*v_sorted[i]) - (n-i)*v_sorted[i])/(n-i)
                threshold = v_sorted[i]
            v_sorted[i] = max(lo, min(hi, max(threshold, v_sorted[i])))
            new_sum = np.sum(v_sorted)
            if new_sum <= total:
                excess = total - new_sum
                i += 1
                continue
            else:
                v_sorted[i] = v_sorted[i] - excess / (n-i)
                excess = 0
            break
        # Re-sort
        v_result = np.zeros_like(v)
        v_result[sorted_idx] = v_sorted[::-1]
    # Ensure sum is exactly total
    s_r = np.sum(v_result)
    if abs(s_r - total) > 1e-12:
        # Do not change sum too much
        v_result = v_result * (total / s_r)
    return v_result

def compute_c5(h, dx):
    """Evaluate the constant C5 using FFT for full correlation."""
    h = np.asarray(h)
    # Use cross-correlation with 1-h
    # Correlation of h with (1-h)
    # np.correlate(h, 1-h, mode="full") can be done via FFT:
    corr = ifft(fft(h) * fft(1-h).conj()).real
    # ext is n-1
    # corr size = 2*n -1
    c5 = np.max(corr) * dx
    return c5

def worst_lag(h):
    """Return the lag index and the value with maximum correlation."""
    h = np.asarray(h)
    corr = ifft(fft(h) * fft(1-h).conj()).real
    kmax = np.argmax(corr)
    v = corr[kmax]
    return kmax, v

def gradient_worst(h, k):
    """Return gradient vector of overlap at lag k."""
    # partial of sum_i h_i h_{i+k} wrt h_i is h_{i+k} + h_{i-k}
    # indices modulo n
    n = len(h)
    g = np.zeros_like(h)
    for i in range(n):
        j1 = (i + k) % n
        j2 = (i - k) % n
        g[i] = h[j1] + h[j2]
    return g

def optimize(n, time_limit, seed=None):
    if seed is not None:
        np.random.seed(seed)
    dx = 2.0 / n
    total = n / 2.0  # required sum
    
    # Initialize h
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values)
    else:
        # Start with uniform + small noise
        h = np.ones(n) * 0.5
        h += np.random.normal(0, 0.05, n)
        h = np.clip(h, 0, 1)
        h = project_simplex_box(h, total)
        h = np.clip(h, 0, 1)
        h = project_simplex_box(h, total)
    
    # Half precision workaround: ensure sum constraint exactly
    h = project_simplex_box(h, total)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    t_start = time.time()
    
    alpha = 0.01
    iter_count = 0
    while time.time() - t_start < time_limit:
        iter_count += 1
        # Find worst lag and its value
        k_star, overlap_star = worst_lag(h)
        # Gradient of overlap at that lag
        grad = gradient_worst(h, k_star)
        # Gradient step (need dx in numerator but it appears in both parts so cancels)
        h_new = h - alpha * grad
        # Project
        h_new = project_simplex_box(h_new, total)
        # Evaluate
        c5_new = compute_c5(h_new, dx)
        # Accept step if improvement
        if c5_new <= best_c5:
            h = h_new
            best_c5 = c5_new
        else:
            # Reduce step if no improvement
            alpha *= 0.95
        # Early stopping
        if abs(alpha) < 1e-8:
            alpha = 1e-6
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    t_start = time.time()
    
    n_values = [150, 250, 350]  # Small enough to be fast and avoiding any issues
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n in n_values:
        if time.time() - t_start > budget_s * 0.95:
            break
        time_left = budget_s - (time.time() - t_start)
        if time_left < 1:  
            break
        h, c5 = optimize(n, time_left/len(n_values), seed+23*n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
            
        # Alternative: multiple restarts (especially useful if time allows)
        # but here we may limit to single run per n to have more variety across n sizes
    
    if best_h is None:
        # Fallback
        h = np.array(initial_h_values)
        if len(h) <= 200:  # reasonable size for fast eval
            h = h.copy()
            best_c5 = compute_c5(h, 2.0/len(h))
            best_n = len(h)
            best_h = h
        else:
            best_h = np.array(initial_h_values[:200])
            best_c5 = compute_c5(best_h, 2.0/len(best_h))
            best_n = 200

    # Ensure the array holds the right number of samples
    hh = list(best_h)
    return hh, best_c5, best_n
