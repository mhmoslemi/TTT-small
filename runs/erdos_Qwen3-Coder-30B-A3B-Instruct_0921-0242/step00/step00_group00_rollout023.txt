def project_capped_simplex(v, total):
    """Project vector v onto capped simplex: 0 <= v_i <= 1, sum(v) = total."""
    v = np.asarray(v)
    n = len(v)
    # Binary search for lambda such that sum(p_i(lambda)) = total
    v_sorted = np.sort(v)[::-1]
    sum_prefix = np.cumsum(v_sorted)
    # Find the index where the sum crosses total
    idx = np.searchsorted(sum_prefix, total, side='right')
    if idx == 0:
        # All elements should be set to total/n, but since they are capped at 1,
        # we check if total/n > 1.
        lambda_star = total / n - (total / n - 1) * (total / n > 1)
        lambda_star = max(0, lambda_star)
    else:
        # The condition is sum( max(v_sorted[i] - lambda_star, 0) ) = total
        # This simplifies to: sum(v_sorted[:idx]) - idx * lambda_star = total
        # Therefore, lambda_star = (sum(v_sorted[:idx]) - total) / idx
        lambda_star = (sum_prefix[idx - 1] - total) / idx
        # Ensure lambda_star is within [0, v_sorted[0]]
        # Actually, the projection ensures that all entries with lambda_star > real_value are 0.
        # But we use the condition that p_i(lambda_star) = max(v_i - lambda_star, 0)
        lambda_star = max(0, min(lambda_star, v_sorted[0]))
    proj = np.maximum(0, np.minimum(1, v - lambda_star))
    return proj

def auto_corr(h):
    """Compute zero-padded auto-correlation w.r.t. lag, 
       returning array [O_0, O_1, ..., O_{n-1}]"""    
    h = np.asarray(h)
    n = len(h)
    # Do a zero-padded convolution of h with itself:
    # we want h * h where indices are aligned properly
    # But since we want correlation with lag k, we do correct indexing
    # For h[i] * h[i+k], which is valid only if i+k < n, i.e. k < n-i
    # For 0 <= k < n, we care about:
    # O_k = sum_{i=0}^{n-1-k} h[i] * h[i+k]
    # For this, do FFT of zero-padded h to length 2*n
    h_fft = np.fft.fft(h, 2 * n)
    # Convolution of h with itself:
    # Note that conv(h, h) would give the standard definition of correlation
    # But we need correlation with lag, not multiplication
    h_auto = np.fft.ifft(h_fft * np.conj(h_fft)).real[:n]
    return h_auto

    t0 = time.time()  

    for beta in beta_schedule:
        inner_iter = 0
        last_improvement = np.inf
        for _ in range(10000):
            # Check timeout
            if time.time() - t0 > time_limit:
                break                
            # Compute auto correlation  
            O = auto_corr(h)
            # compute new F_beta
            F_beta = 1 - (1/beta) * np.log(np.sum(np.exp(beta * O)))
            
            # Compute weights w
            w = np.exp(beta * O)
            w = w / np.sum(w)
            
            # gradient
            g = grad_F(beta, h, O, w)
            
            # stock gradient descent step
            alpha = 1e-3
            while True:
                h_try = h - alpha * g
                h_try = project_capped_simplex(h_try, total=n/2)
                # compute corresponding F_beta on h_try
                O_try = auto_corr(h_try)
                F_try = 1 - (1/beta) * np.log(np.sum(np.exp(beta * O_try)))
                if F_try <= F_beta - 1e-4 * alpha * np.linalg.norm(g) **2:
                    h = h_try
                    break
                alpha *= 0.5
                if alpha < 1e-8:
                    break
            
            # compute current c5 (overlast/actual prob)
            corr = np.convolve(h, 1-h, mode='full') * (2 / n)
            c5 = np.max(corr) 
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = time.time() - t0
                
            inner_iter += 1
            if last_improvement > 20 and time.time() - t0 > time_limit - 1:
                break
                
    # Fine-tune using exact objective (non-smooth one)
    corr_best = np.convolve(best_h, 1-best_h, mode='full') * (2 / n)
    best_c5_exact = np.max(corr_best)
    
    if best_c5_exact < best_c5:
        # Better value obtained from final computation
        best_c5 = best_c5_exact
        
    return best_h, best_c5, n

    best_n = None

        beta_schedule = np.logspace(np.log10(0.5), np.log10(50), 12)

                h, c5, n_points = optimize_one_start(n, beta_schedule, budget_s - (time.time() - t0) - 1, seed + restart)
            except:
                continue

                current_best = c5
                best_h_solution = h.copy()
                best_n = n_points
