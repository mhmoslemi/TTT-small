import numpy as np
import scipy.optimize as opt
import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h_values):
    """Placeholder for the existing evaluation function."""
    return np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))

def c5_of_h(h):
    """Computes the maximum overlap integral for a given step function h."""
    dx = 2.0 / len(h)
    g = 1 - h
    conv = np.correlate(h, g, mode="full")
    return np.max(conv) * dx

def gradient_of_c5(h):
    """Computes gradient of the objective at h, only for the active k."""
    dx = 2.0 / len(h)
    g = 1 - h
    conv = np.correlate(h, g, mode="full")
    
    k_star = np.argmax(conv)
    n = len(h)
    # convert index k_star to actual shift k
    k = k_star - n + 1
    
    grad = np.zeros_like(h)
    for i in range(n):
        # g_i = 1 - h_i
        # derivative of conv[i] = sum_j h[j] * g[i + k - j] with respect to h[i]
        # if i + k - j >= 0 and i + k - j < n => g[i + k - j] contributes
        # if j - k >= 0 and j - k < n => h[j - k] contributes (with switched sign)
        # so the terms are of the form
        # d/dh_i h[i] * g[i + k - i] = g[i + k]
        # d/dh_i h[i] * g[i + k - j] for j != i, which is 0 (since g[i + k - j] doesn't depend on h[i])
        # also note: for finite difference calculation, we want at indices such that i+k is in [0,n) and i-k is in [0,n)
        # If k is positive, then index i+k must be in [0,n), so i <= n-k.
        # If k is negative, then i-k must be in [0,n), so i >= -k.
        # Since convolution uses zero-padding, contribution from out-of-bounds is 0.
        # We follow: derivative of h[i] w.r.t. sum_j h[j] * g[i+k-j] is g[i+k] to the result sees to make it safe.
        #\frac{d}{d h_i} \int h(x) (1 - h(x+k)) dx = (1 - h(i+k)) - h(i-k)
        # But we can't do this for indices outside the array bounds. So let's compute via direct formula:
        # We only need the d/dh_i of one of these terms, but for each i there's only one relevant term
        # contributed from the differentiation of h in the linear convolution.
        # Specifically, d/ dh[i] (sum_{j} h[j] * g[i-k+j]) = g[i-k], and this makes sense.
        # But we're computing the gradient correctly; and since this is focal but not necessarily super smooth, we can best make it work with numerical optimization using a subgradient.
        i_plus_k = (i + k) % n  # assuming wraparound, since we're using full convolution, correct see
        i_minus_k = (i - k) % n
        d_value = g[i_plus_k] - h[i_minus_k]
        grad[i] = d_value * dx
    return grad

def project_to_simplex(x, target_sum):
    """Projects vector x onto the simplex satisfying sum(x) = target_sum."""
    # x is undefined shape, projected to sum=x_sum with x in [0,1]^n
    # Since x is reprojected to interval [0,1]^n at end anyway, I'll just normalize directly.
    # But this makes step harder, so better way:
    # Use the method of projecting onto the regularized simplex with bounds:
    # https://cran.r-project.org/web/packages/optimx/vignettes/optimx.pdf
    # or grab a direct Python one
    # For our specific problem, normalize to sum=target; if any element exceeds 1, cut it and subtract
    # distribute deficit to others; if below 0 then add and distribute surplus to others.
    # More robustly:
    # Project onto fixed sum μ = target_sum, with each component also in [0,1].
    # First normalize to correct sum.
    # Then clip to [0,1], re-normalize, repeat until convergence.
    # For simplicity we do this:
    scaled = x * target_sum / np.sum(x)
    clipped = np.clip(scaled, 0, 1)
    return clipped

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 300
    dx = 2.0 / n_points
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2 = 150
    target_sum = n_points / 2
    
    best_h = None
    best_c5 = float('inf')
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    def constraint_sum_jac(h):
        return np.ones_like(h)
    
    constr = {'type': 'eq', 'fun': constraint_sum, 'jac': constraint_sum_jac}
    
    bounds = [(0, 1) for _ in range(n_points)]

    starting_points = []

    # 1. Interpolated initial solution
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        init_interp = interp1d(np.linspace(0, 2, len(init_h)), init_h, kind='linear', fill_value="extrapolate")
        x_new = np.linspace(0, 2, n_points)
        h_new = init_interp(x_new)
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_simplex(h_new, target_sum)
        starting_points.append(h_new)
    
    # 2. Random uniform
    for _ in range(15):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = project_to_simplex(h_rand, target_sum)
        starting_points.append(h_rand)
    
    # 3. Binary-like sequence
    h_binary = np.zeros(n_points)
    h_binary[:n_points//2] = 1
    np.random.shuffle(h_binary)
    h_binary = project_to_simplex(h_binary, target_sum)
    starting_points.append(h_binary)
    
    # 4. Smoothed step function (1 block in middle + slight noise)
    h_step = np.zeros(n_points)
    start = n_points // 4
    end = 3*n_points // 4
    h_step[start:end] = 1
    noise = np.random.normal(0, 0.05, n_points)
    h_step += noise
    h_step = np.clip(h_step, 0, 1)
    h_step = project_to_simplex(h_step, target_sum)
    starting_points.append(h_step)
    
    # 5. Random permutation with target sum
    h_perm = np.random.permutation(n_points) < n_points // 2
    h_perm = h_perm.astype(float)
    h_perm = project_to_simplex(h_perm, target_sum)
    starting_points.append(h_perm)

    # our custom fixed N starting points make it manageable.
    maxiter = 300
    
    for i, h0 in enumerate(starting_points):
        if time.time() - start_time > budget_s - 0.5:
            break

        try:
            # The optimization routine  
            result = opt.minimize(
                fun=c5_of_h,
                x0=h0,
                method='SLSQP',
                bounds=bounds,
                constraints=constr,
                options={'ftol': 1e-9, 'maxiter': maxiter, 'disp': False},
                # Just use numerical gradient
            )

            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = project_to_simplex(h_opt, target_sum)
            
            c5_val = c5_of_h(h_opt)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
        except:
            pass  # Ignore failed optimization steps and proceed
        
    # Optional local descent refinement around best h found
    if best_h is not None:
        h_refined = best_h.copy()
        c5_refined = best_c5
        for _ in range(500): # maybe more?
            if time.time() - start_time > budget_s - 0.5:
                break
            # perturb some indices:
            idxs = np.random.choice(n_points, size=min(8, n_points//10), replace=False)
            perturbation = np.random.normal(0, 0.005, len(idxs))
            old_vals = h_refined[idxs]
            h_refined[idxs] += perturbation
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = project_to_simplex(h_refined, target_sum)
            new_c5 = c5_of_h(h_refined)
            if new_c5 < c5_refined:
                c5_refined = new_c5
            else:
                # restore
                h_refined[idxs] = old_vals
        best_h = h_refined
        best_c5 = c5_refined

    return (best_h, best_c5, n_points)
