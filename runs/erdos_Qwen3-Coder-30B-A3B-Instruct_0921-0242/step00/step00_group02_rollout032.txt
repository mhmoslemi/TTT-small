import numpy as np
import time
import random
from scipy.signal import correlate

def correlation_vector(b):
    # Compute correlation vector R_k = sum b_i * b_{i+k} for k=1..n-1
    R = correlate(b, b, mode='full')
    return R[len(b)-1:]

def update_correlation(R, i, j, b):
    # Update R after swapping positions i (1) and j (0)
    n = len(b)
    new_R = R.copy()
    for k in range(1, n):
        if i >= k and b[i-k] == 1:
            new_R[k-1] -= 1
        if i+k < n and b[i+k] == 1:
            new_R[k-1] -= 1
        if j >= k and b[j-k] == 1:
            new_R[k-1] += 1
        if j+k < n and b[j+k] == 1:
            new_R[k-1] += 1
    return new_R

def local_hill_climb(b, R, max_swaps=None):
    # Hill climb from current b with given R
    n = len(b)
    best_b = b.copy()
    best_R = R.copy()
    best_t = np.min(R)
    improved = True
    iterations = 0
    while improved:
        improved = False
        kstar = np.argmin(best_R)
        tstar = best_R[kstar]
        candidates = []
        for i in range(n):
            if best_b[i] == 1:
                for j in range(n):
                    if best_b[j] == 0:
                        candidates.append((i, j))
        if max_swaps is not None:
            random.shuffle(candidates)
            candidates = candidates[:max_swaps]
        for i, j in candidates:
            new_R = update_correlation(best_R, i, j, best_b)
            new_t = np.min(new_R)
            if new_t > best_t or (new_t == best_t and new_R[kstar] > best_R[kstar]):
                best_R = new_R
                best_t = new_t
                best_b[i], best_b[j] = best_b[j], best_b[i]
                improved = True
        iterations += 1
        if iterations > 100:
            break
    return best_b, best_R

def fitness(b):
    R = correlation_vector(b)
    return np.min(R)

def crossover(p1, p2):
    # Union-based crossover
    s1 = set(np.where(p1 == 1)[0])
    s2 = set(np.where(p2 == 1)[0])
    union = list(s1.union(s2))
    n = len(p1)
    if len(union) > n//2:
        # Remove extras
        diff = len(union) - n//2
        for _ in range(diff):
            union.remove(random.choice(union))
    elif len(union) < n//2:
        # Add extras
        complement = set(range(n)) - set(union)
        diff = n//2 - len(union)
        add = random.sample(complement, diff)
        union += add
    child = np.zeros(n, dtype=np.uint8)
    for i in union:
        child[i] = 1
    return child

def mutate(b, num_swaps=1):
    # Apply random swaps
    n = len(b)
    b = b.copy()
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for _ in range(num_swaps):
        if len(ones) > 0 and len(zeros) > 0:
            i = random.choice(ones)
            j = random.choice(zeros)
            b[i], b[j] = b[j], b[i]
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
    return b

def sample_candidate(n):
    # Sample random binary vector of exact weight n//2
    b = np.zeros(n, dtype=np.uint8)
    selected = random.sample(range(n), n//2)
    for i in selected:
        b[i] = 1
    return b

        for i in range(len(initial_h_values)):
            for _ in range(int(scale) + (1 if random.random() < scale - int(scale) else 0)):
                if len(h_interp) < n:
                    h_interp.append(initial_h_values[i])
        h_interp = h_interp[:n]
    # Convert to binary
    sorted_indices = np.argsort(h_interp)[::-1]
    b = np.zeros(n, dtype=np.uint8)
    for i in range(n//2):
        b[sorted_indices[i]] = 1
    # Population initialization
    population = [b]
    for i in range(pop_size - 1):
        bnew = sample_candidate(n)
        population.append(bnew)
    
    best_t = -1
    best_b = None
    for gen in range(generations):
        if time.time() - t0 > budget - 2:
            break
        # Evaluate
        pop_fitness = []
        for ind in population:
            R = correlation_vector(ind)
            t = np.min(R)
            pop_fitness.append(t)
        # Track global best
        max_idx = np.argmax(pop_fitness)
        if pop_fitness[max_idx] > best_t:
            best_t = pop_fitness[max_idx]
            best_b = population[max_idx].copy()
        # Selection Tournament
        new_pop = []
        for _ in range(pop_size):
            tournament = [random.choice(population) for _ in range(3)]
            tournament_fitness = [np.min(correlation_vector(t)) for t in tournament]
            winner = tournament[np.argmax(tournament_fitness)]
            new_pop.append(winner)
        # Crossover and Mutation
        next_pop = []
        for i in range(0, pop_size, 2):
            p1 = new_pop[i]
            p2 = new_pop[(i+1)%pop_size]
            if random.random() < 0.8:
                c = crossover(p1, p2)
            else:
                c = p1.copy()
            c = mutate(c, random.randint(1, 2))
            # Local improvement
            R = correlation_vector(c)
            c, R = local_hill_climb(c, R, max_swaps=200)
            next_pop.append(c)
        population = next_pop
    return best_b, best_t

def upsample_and_fix(b, target_n):
    # upsample
    scale = target_n / len(b)
    upsampled = []
    for i in range(len(b)):
        for _ in range(int(scale) + (1 if random.random() < scale - int(scale) else 0)):
            if len(upsampled) < target_n:
                upsampled.append(b[i])
    upsampled = upsampled[:target_n]
    # convert to array
    b_up = np.array(upsampled, dtype=np.uint8)
    current_weight = np.sum(b_up)
    # adjust weight
    if current_weight > target_n // 2:
        ones = np.where(b_up == 1)[0]
        to_change = int(current_weight - target_n // 2)
        for i in range(to_change):
            b_up[ones[i]] = 0
    elif current_weight < target_n // 2:
        zeros = np.where(b_up == 0)[0]
        to_change = int(target_n // 2 - current_weight)
        for i in range(to_change):
            b_up[zeros[i]] = 1
    return b_up

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    t0 = time.time()
    # Sequence lengths to use in hierarchy
    lengths = [30, 36, 42, 48, 60]
    results = []
    for i, n in enumerate(lengths):
        if time.time() - t0 > budget_s - 5:
            break
        if i == 0:
            # Stage 1: short baseline
            pop_size = 50
            generations = 20
            # Use small extra time for small problems
            budget = min(100, budget_s//6)
        elif i < len(lengths) - 1:
            pop_size = max(30, 2 * n)
            generations = 100
            budget = min(120, 2.0 * n)
        else:
            # Final stage
            pop_size = max(30, 2 * n)
            generations = 200
            budget = min(120, 2.0 * n)
        if time.time() - t0 > budget_s - budget - 5:
            budget = budget_s - (time.time() - t0) - 5
        best_b, t = optimize_stage(n, pop_size, generations, budget)
        results.append((n, best_b, t))
    
    # Find final best
    best_idx = np.argmax([t for _, _, t in results])
    n_best, best_b, best_t = results[best_idx]
    
    # Final polishing with exhaustive search on critical shift
    R = correlation_vector(best_b)
    kstar = np.argmin(R)
    best_R = R.copy()
    improve = True
    limit = 1000
    while improve and limit > 0:
        improve = False
        limit -= 1
        candidates = []
        for i in range(n_best):
            if best_b[i] == 1:
                for j in range(n_best):
                    if best_b[j] == 0:
                        candidates.append((i, j))
        max_delta = -np.inf
        best_swap = None
        for i, j in candidates:
            new_R = update_correlation(best_R, i, j, best_b)
            new_t = np.min(new_R)
            delta = new_R[kstar] - best_R[kstar]
            if delta > max_delta or (delta == max_delta and new_t > np.min(best_R)):
                max_delta = delta
                best_swap = (i, j)
        if best_swap:
            i, j = best_swap
            best_b[i], best_b[j] = best_b[j], best_b[i]
            best_R = update_correlation(best_R, i, j, best_b)
            improve = True

    # Final computation of C5
    dx = 2.0 / n_best
    R_final = correlate(best_b, 1 - best_b, mode='full')
    c5_bound = 1 - dx * np.min(R_final[len(best_b)-1:])  # The critical term
    
    # Convert to float array for output
    h_values = best_b.astype(float).tolist()

    return (h_values, c5_bound, n_best)
