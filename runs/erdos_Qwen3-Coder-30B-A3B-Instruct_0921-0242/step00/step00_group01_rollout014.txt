```python
import time
import numpy as np
from cvxpy import *
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    # Binary optimization is only used for coarse resolution
    n0 = 120  # coarse discretization
    dx = 2.0 / n0
    S = n0 / 2  # desired sum
    
    # Define the solver and problem structure
    # Build the MIP model using cvxpy
    b = cp.Variable(n0, boolean=True)
    
    # Product variables for binary products b[i] * b[i+k] 
    # We'll generate these constraints on the fly
    
    # Objective: minimize w (the maximum overlap = S - min_k A_k)
    w = cp.Variable()
    
    # All constraints will be temporarily stored
    constraints = []
    
    # Setup overlap (autocorrelation) constraints
    # For each k, we define: w >= S - sum b[i] * b[(i+k)%n]
    # To model the product b[i] * b[j], we create auxiliary variable z[i,k]
    
    # Obs: since we have spatial symmetry, we only consider k = 0 to n0//2
    # and exploit that O_k = O_{n0-k}
    
    # Exact model approach via constraints on binary products
    
    # Precompute the constraints as a generator
    def z_constraints():
        for i in range(n0):
            for k in range(n0):  # could optimize by only ranging k <= n0//2
                idx1 = i
                idx2 = (i + k) % n0
                # We will use a sparse constraint form
                yield b[idx1] + b[idx2] - 1 <= 1  # prod = 1 only if both are 1
                yield b[idx1] >= z[i,k]
                yield b[idx2] >= z[i,k]
    
    # Create all the product variables z[i,k]
    z = {}
    all_z_vars = []
    for i in range(n0):
        for k in range(n0):
            z[i,k] = cp.Variable(boolean=True)
            all_z_vars.append(z[i,k])
    
    # Setup constraints
    # 1. Product identity constraints
    for i in range(n0):
        for k in range(n0):
            constraints.extend([
                z[i,k] <= b[i],
                z[i,k] <= b[(i+k) % n0],
                z[i,k] >= b[i] + b[(i+k) % n0] - 1
            ])
    
    # 2. Overlap value constraints (we compute A_k = sum_i z[i,k] = sum_i b[i]*b[i+k])
    for k in range(n0):
        # sum of the products over i
        sum_z = sum(z[i,k] for i in range(n0))  # This is A_k 
        constraints.append(w >= S - sum_z)
    
    # 3. Mass constraint
    constraints.append(cp.sum(b) == S)
    
    # 4. Symmetry breaking constraint:
    # Fix first element to 1 (weaker than fixing b[0] = 1)
    constraints.append(b[0] == 1)
    
    # 5. Construct the problem
    obj = cp.Minimize(w)
    prob = cp.Problem(obj, constraints)
    
    # Actually, our majority of time will go in solving the MIP, so we...
    # Place the solver within a limited runtime and gracefully handle timeout
    remaining = budget_s - (time.time() - start)
    if remaining < 20:
        # Fall back to a greedy method if time is tight
        h = greedy_balanced_erdos(n0)
    else:
        try:
            prob.solve(solver=cp.GLPK_MI, time_limit=remaining)
            # Check if solution is available
            if b.value is not None:
                h = np.array(b.value, dtype=float)
                w_val = w.value
                assert abs(np.sum(h) - S) < 1e-9
                C5 = w_val * dx
            else:
                # Solve didn't finish; fallback
                h = greedy_balanced_erdos(n0)
        except:
            # GLPK failed, fallback to heuristic too
            h = greedy_balanced_erdos(n0)
    
    # Do a small postprocessing refinement using local search swaps:
    # improve min_k sum_i h[i]*h[i+k] to less than previous value
    best_h, best_C5 = local_swap_improve(h, n0, dx)
    
    # Upscaling step: inflate to multiples of original size
    # Want n > 1000 but as close as possible to 1000 and n0 | n
    N = 360
    # Do duplication
    h_final = [best_h[i // (N // n0)] for i in range(N)]
    C5_final = best_C5
    # Ensure integral constraint to within float error
    integral_err = abs(np.sum(h_final) - 1.0)
    assert integral_err < 1e-9, f"Integral constraint violated: {integral_err}"
    
    # Re-evalutate best_C5 from h_final using standard method
    C5_final = compute_C5_via_correlation(h_final, N)
    
    return (h_final, C5_final, N)

def greedy_balanced_erdos(n):
    """Greedy method to produce a binary approximation for the Erdos problem"""
    # Start from initial_h_values, or an empty set
    # Initialize:
    b = [0] * n
    count = 0
    S = n // 2
    while count < S:
        best_take = None
        min_gain = np.inf
        for i in range(n):
            if b[i] == 1: continue
            
            # Compute benefit of taking this index
            # This requires evaluating impacts of taking index `i`
            # For simplicity, we use a basic strategy:
            # compute impact on current overlaps if we add `i`
            temp_b = b[:]
            temp_b[i] = 1
            # Dummy select an index that improves worst case overlap
            # We don't compute full updates.
            # Instead, first get current overlaps, then simulate `i` being added
            
            # This greedy strategy is heuristic with poor time complexity,
            # but we run on a small problem (n=120), so it's okay.
            # Better approach: model beneficial vs. disruptive: 
            # Interest is in the change in   O_k = S - A_k
            
            # Specific case: let k = 0 (no shift), O_0 = sum(h[i]^2) = count
            # Approximate the buffer as a greedy selection of one 'i' 
            # that has least negative effect into one of the correlations
            
            # For now just run with a basic strategy that selects proper number
            # without always doing all the work
            
            # For this purpose, assume a prioritized selection
            # of index to include based on overall gains
            
            # Compute overlap excess from adding i 
            # But that's O(n^2)ic to compute all of them  
            # Use a quick scoring matrix of approx >V
            # Let's just do greedy selecting based on a revertible area score
            
            # Avoiding exhaustive overlaps calculation:
            # Simple strategy: weighted sum with current selection status, 
            # relative to how many 1s already placed
            # Better than nothing
            
            # Let's just prioritize placing near center for simple symmetric to get going 
            # Li Wei-like
            if np.abs(i - n//2) < np.abs(min_gain) or min_gain == np.inf:
                min_gain = i - n // 2
                best_take = i
        if best_take is not None:
            b[best_take] = 1
            count += 1
            
    return b


def local_swap_improve(h, n0, dx):
    """Perform a fast deterministic local search improvement"""
    h = h.copy()
    S = n0 / 2
    MAX_SWAPS = 200
    best_h = h[:]
    best_C5 = compute_C5_via_correlation(best_h, n0)
    
    for _ in range(MAX_SWAPS):
        smallest_correlation = np.inf
        worst_k = -1
        for k in range(n0):
            # Compute A_k = sum h[i] * h[i+k] for lag k
            s = 0
            for i in range(n0):
                s += h[i] * h[(i+k)%n0]
            if s < smallest_correlation:
                smallest_correlation = s
                worst_k = k
                
        if worst_k == -1:
            break
        
        # Find pair (i, j) such that i has value 1, j has 0 
        # and swapping them improves the worst overlap
        found = False
        for i in range(n0):
            if h[i] == 0:
                continue
            for j in range(n0):
                if h[j] == 1:
                    continue
                # Simulate swap using changes to A_k
                old_val = 0
                new_val = 0
                
                # compute delta_k = change in A_k due to swapping i&j?
                for k in range(n0):
                    expected_val = h[i]*h[(i+k)%n0] + h[j]*h[(j+k)%n0]
                    swapped = h[i]*h[(j+k)%n0] + h[j]*h[(i+k)%n0]
                    
                    if expected_val > swapped:
                        old_val += expected_val - swapped
                    else:
                        new_val += swapped - expected_val
                        
                if new_val < old_val and new_val < 0:
                    # Swap indices i and j
                    h[i], h[j] = h[j], h[i]
                    found = True
                    break
            if found:
                break
                
        if not found:
            break
        new_C5 = compute_C5_via_correlation(h, n0)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h[:]
            
    return best_h, best_C5


def compute_C5_via_correlation(h_vec, n):
    """Compute C5 using standard discretization and correlate"""
    h = np.array(h_vec, dtype=float)
    # compute the maximum overlap = maximum value of integral 
    # for all valid shifts k
    dx = 2.0 / n
    max_overlap = 0
    for i in range(n):
        # Quick integration of h(x) * (1 - h(x+i))
        h_sh = np.roll(h, i)
        corr = h * (1 - h_sh)
        overlap = np.sum(corr) * dx
        if overlap > max_overlap:
            max_overlap = overlap
            
    return max_overlap
```
