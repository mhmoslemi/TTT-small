import numpy as np
import time
from scipy import fftpack
from scipy.optimize import minimize
import math


def project_to_F(h, n):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Compute current deficit/excess
    deficit = n/2 - h.sum()
    # Indices that are not at a bound
    free = np.where((h>0) & (h<1))[0]
    while np.abs(deficit) > 1e-12 and free.size>0:
        # Distribute equally over free indices
        delta = deficit / free.size
        # Apply delta, respecting bounds, and recompute deficit
        h[free] += delta
        # Clamp again and update free set
        h = np.clip(h, 0.0, 1.0)
        deficit = n/2 - h.sum()
        free = np.where((h>0) & (h<1))[0]
    return h


def fast_corr(h, g, dx, L):
    # Zero-pad both h and g to length L
    h_pad = np.pad(h, (0, L - len(h)), 'constant')
    g_pad = np.pad(g, (0, L - len(g)), 'constant')
    # Compute FFTs
    H = fftpack.fft(h_pad)
    G = fftpack.fft(g_pad)
    # Full correlation
    conv = fftpack.ifft(H * np.conj(G)).real * dx
    return conv


def evaluate_fitness(h, n, dx):
    g = 1 - h
    # Use next power of 2 for FFT
    L = 2**int(np.ceil(np.log2(2*n)))
    conv = fast_corr(h, g, dx, L)
    # Only take the valid correlations (leading ones, which are for full overlap)
    # Index k = 0 corresponds to the middle of the full correlation array
    # Since we pad to length 2*n, the indices have bias = n-1
    # The valid range is from -(n-1) to (n-1)
    # So, the true overlap indices lie between n-1 and 3*n-3
    # We want max c_k for k from -n+1 to n-1 (all valid non-zero shifts)
    # However, the rightmost n-1 points of correlation array have k=0,...,n-2,
    # so we take from indices n-1 to 3*n-3, which covers elements corresponding
    # to shifts 0 to 2*n-3.
    # But we want k from -(n-1) to (n-1), hence indices n-1 to n-1+2*(n-1) = 3*n-3
    # however we actually want the largest C5, which corresponds to maximum of
    # conv[n-1:3*n-3]  but we take only few elements near the middle (the overlapping ones)
    # Actually, we just want max over all k excluding k=0, 
    # and for our use case this is easily done: 
    # shift k from 0 to 2*(n-1) gives index k+n-1 in convolution array
    # so we want to exclude k=0, i.e. index n-1
    # Max over all indices except n-1
    c5_val = np.max(conv[:n-1])  # k = -(n-1) to -1
    c5_val = max(c5_val, np.max(conv[n+1:]))  # k = 1 to n-1
    return c5_val


def local_improve(h, n, dx, max_steps=30):
    # Using the full convolution array
    # Use next power of 2 for FFT
    L = 2**int(np.ceil(np.log2(2*n)))
    g = 1 - h
    conv = fast_corr(h, g, dx, L)
    # get the shift which gives the max value (i.e., worst overlap)
    max_val = -np.inf
    k_star = 0
    for k in range(-n+1, n):
        idx = k + n - 1  # fix index in conv array
        if idx < 0 or idx >= len(conv): continue
        if k == 0: continue
        if conv[idx] > max_val:
            max_val = conv[idx]
            k_star = k
    # Now we know what the worst k is.
    # Now trying pairwise swaps to reduce max conv[k] for k=k_star.
    # Trying to improve the shift quality, one pair at a time
    for iteration in range(max_steps):
        # Find candidate indices (have some room for exchange)
        ind1 = np.where(h > 0)[0]
        ind0 = np.where(h < 1)[0]
        if len(ind1) == 0 or len(ind0) == 0: break
        i1 = np.random.choice(ind1)
        i0 = np.random.choice(ind0)
        # Proposed change
        epsilon = min(h[i1], 1-h[i0], 0.05)
        if epsilon <= 0: continue
        
        # Compute new array
        h_new = h.copy()
        h_new[i1] -= epsilon
        h_new[i0] += epsilon
        # project to feasible set:
        h_new = project_to_F(h_new, n)
        # Re-evaluate the target shift efficiently
        # We need to compute the change in correlation at shift k_star only
        # the change is delta = -epsilon * dx * (g[i1+k_star] - g[i0+k_star])
        delta = -epsilon * dx * (g[i1 + k_star] - g[i0 + k_star])  # assuming index wraps
        # corresponds to above logic, but carefully adjust for zero padding.
        # For simplicity, let's recompute relevant parts exactly.
        # However given that only small changes occur and we are mostly concerned with
        # k_star, computation can be made significantly sparse.
        # It suffices to compute conv' = conv - delta for that k_star
        # For array this:
        # This step isn't easy to compute by hand because long correlation is mostly zero
        # So we just bluff by computing the new conv. unless it's too slow:
        # Let's recompute full correlation. For small arrays this is fine.
        g_new = 1 - h_new
        conv_new = fast_corr(h_new, g_new, dx, L)
        
        # find index for k_star in new conv
        k_idx = k_star + n - 1
        if 0 <= k_idx < len(conv_new) and k_idx != n - 1:
            new_val_k = conv_new[k_idx]
        else:
            new_val_k = -np.inf  # invalid 
        # Only accept if it leads to reduction in that entry
        if new_val_k < max_val:
            h = h_new
            max_val = new_val_k
        else:
            # let's also recompute with small decrease in tolerance:
            if new_val_k < max_val * 1.001:
                h = h_new
    return h


def create_initial_population(n, pop_size, initial_h):
    pop = []
    # Seed 1: interpolate from provided initial_h_values (length ~70)
    if len(initial_h) <= n:
        h_seed_orig = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h)), initial_h)
    else:
        # Downsample to n points if needed
        h_seed_orig = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h)), initial_h)
    h_seed_orig = project_to_F(h_seed_orig, n)
    c5_seed = evaluate_fitness(h_seed_orig, n, 2.0/n)
    pop.append((h_seed_orig.copy(), c5_seed))
    
    # Seed 2: Random continuous
    for _ in range(pop_size-1):
        h = np.random.rand(n)
        h = project_to_F(h, n)
        c5 = evaluate_fitness(h, n, 2.0/n)
        pop.append((h.copy(), c5))
        
    return pop


def crossover(p1, p2, n):
    a = np.random.rand()
    child = a * p1 + (1-a) * p2
    child = project_to_F(child, n)
    return child


def mutate(h, n, sigma):
    m = max(1, int(0.02 * n))
    indices = np.random.choice(n, m, replace=False)
    noise = np.random.normal(0, sigma, m)
    h_mut = h.copy()
    h_mut[indices] += noise
    h_mut = project_to_F(h_mut, n)
    return h_mut


def adaptive_mutate(h, n, sigma, sigma_min=0.02):
    if sigma < sigma_min:
        # restart mutation strength  
        sigma = 0.1
        # half population reinit
        h = np.random.rand(n)
        h = project_to_F(h, n)
    else:
        h = mutate(h, n, sigma)
    return h, sigma
