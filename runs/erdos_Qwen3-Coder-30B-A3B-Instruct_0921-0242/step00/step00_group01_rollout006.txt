import time
import numpy as np
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    # Initial setup
    n0 = 120  # Base grid size
    dx = 2.0 / n0
    S = n0 / 2  # Target integral value
    
    # Try to use SCIP via cvxpy if available, otherwise fallback
    solver = None
    for s in [cp.SCIP, cp.CBC, cp.GLPK]:
        try:
            test_prob = cp.Problem(cp.Minimize(0), [])
            test_prob.solve(solver=s, verbose=False)
            solver = s
            break
        except:
            continue
    
    # If no solver is available, use a greedy fallback
    if solver is None:
        return run_greedy_fallback(n0, dx, S, start, budget_s)
    
    # Build MILP model
    try:
        # Create binary decision variables
        b = cp.Variable(n0, boolean=True)
        
        # Product variables z[i,k] = b[i] * b[(i+k)%n0]
        # We only create z for k = 0 to n0//2 (due to symmetry)
        z_vars = {}
        for k in range(n0):
            for i in range(n0):
                z_vars[(i, k)] = cp.Variable(boolean=True)
        
        # Constraints
        constraints = []
        
        # Symmetry breaking
        constraints.append(b[0] == 1)
        
        # Product linearization constraints
        for k in range(n0):
            for i in range(n0):
                j = (i + k) % n0
                z = z_vars[(i, k)]
                # z <= b[i]
                constraints.append(z <= b[i])
                # z <= b[j]
                constraints.append(z <= b[j])
                # z >= b[i] + b[j] - 1
                constraints.append(z >= b[i] + b[j] - 1)
        
        # Overlap constraints
        w = cp.Variable()
        for k in range(n0):
            sum_z = cp.sum([z_vars[(i, k)] for i in range(n0)])
            constraints.append(w >= S - sum_z)
        
        # Mass constraint
        constraints.append(cp.sum(b) == S)
        
        # Objective
        objective = cp.Minimize(w)
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        
        remaining_time = budget_s - (time.time() - start)
        if remaining_time <= 0:
            return run_greedy_fallback(n0, dx, S, start, budget_s)
            
        # Solve with timeout
        prob.solve(solver=solver, verbose=False, max_seconds=remaining_time)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INaccurate]:
            b_val = b.value
        else:
            return run_greedy_fallback(n0, dx, S, start, budget_s)
            
    except Exception as e:
        return run_greedy_fallback(n0, dx, S, start, budget_s)
    
    # Refine with pairwise swaps if needed
    best_h = b_val.round().astype(int)
    w_opt = prob.value
    c5_opt = w_opt * dx
    
    # Apply pairwise swap refinement
    iterations = 0
    max_iter = 200
    while iterations < max_iter:
        improved = False
        
        # Compute autocorrelations
        A = np.zeros(n0)
        for k in range(n0):
            A[k] = np.sum(best_h * np.roll(best_h, k))
        
        k_star = np.argmin(A)
        worst_autocorr = A[k_star]
        
        # Try all pairs for improvement
        for i in range(n0):
            if best_h[i] == 0:
                continue
            
            for j in range(i+1, n0):
                if best_h[j] == 1:
                    continue
                    
                # Update autocorrelation at k_star
                change = 0
                
                # Original would be 1 if both are 1 at lag k_star
                # Check i with i+k_star, j with j+k_star
                if (best_h[i] == 1 and (i + k_star) % n0 == j) or \
                   (best_h[j] == 1 and (j + k_star) % n0 == i):
                    # This influences the count
                    continue
                
                # Compute change if we swap h[i] and h[j]
                orig_i = best_h[i]
                orig_j = best_h[j]
                
                # Test number of matches at k_star when we swap
                new_h = best_h.copy()
                new_h[i] = orig_j
                new_h[j] = orig_i
                
                # Compute new autocorrelations at k_star
                new_corr = np.sum(new_h * np.roll(new_h, k_star))
                if new_corr > worst_autocorr:
                    # Perform the swap
                    best_h[i] = orig_j
                    best_h[j] = orig_i
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
        iterations += 1
    
    # Compute final C5
    final_h = best_h
    A_final = np.zeros(n0)
    for k in range(n0):
        A_final[k] = np.sum(final_h * np.roll(final_h, k))
    w_final = np.min(A_final)
    c5_final = w_final * dx
    
    # Upscale to target size (prefer 360 for speed/accuracy trade-off)
    N_target = 360  # Even number, <= 1000
    upscale_factor = N_target // n0
    h_upscaled = np.tile(final_h, upscale_factor)
    
    # Final verification
    integral_check = np.sum(h_upscaled) * (2.0 / N_target)
    if abs(integral_check - 1.0) > 1e-9:
        print("Warning: Integral constraint violated in upscaled version.")
    
    # Return as requested
    return (h_upscaled.tolist(), float(c5_final), N_target)

def run_greedy_fallback(n0, dx, S, start, budget_s):
    # Simple greedy strategy: try to construct a greedy solution
    # This is a fallback that works faster than full MILP
    import time
    
    # Start with binary vector (0 and 1s)
    h = np.zeros(n0)
    current = set()
    
    while len(current) < S:
        best_i = -1
        min_increase = float('inf')
        for i in range(n0):
            if i in current:
                continue
            # Simulate adding i and evaluate impact
            current.add(i)
            min_overall = float('inf')
            counts = []  # For tracking overlap impacts at each k
            for k in range(n0):
                count = 0
                for j in range(n0):
                    if j in current and (j + k) % n0 in current:
                        count += 1
                counts.append(count)
                min_overall = min(min_overall, count)
            current.remove(i)
            if min_overall < min_increase:
                min_increase = min_overall
                best_i = i
        if best_i != -1:
            current.add(best_i)
    
    final_h = np.zeros(n0)
    for i in current:
        final_h[i] = 1
    
    # Apply refinement if time allows
    if time.time() - start < budget_s - 5:
        # Refinement
        max_refine = 200
        for _ in range(max_refine):
            A = np.zeros(n0)
            for k in range(n0):
                A[k] = np.sum(final_h * np.roll(final_h, k))
            k_star = np.argmin(A)
            best_swap = None
            best_improvement = 0
            for p in range(n0):
                if final_h[p] == 1:
                    for q in range(n0):
                        if final_h[q] == 0:
                            # Test swap of p and q
                            new_h = final_h.copy()
                            new_h[p] = 0
                            new_h[q] = 1
                            new_A = np.zeros(n0)
                            for k in range(n0):
                                new_A[k] = np.sum(new_h * np.roll(new_h, k))
                            new_worst = np.min(new_A)
                            its_worst = np.min(A)
                            improvement = its_worst - new_worst
                            if improvement > best_improvement:
                                best_improvement = improvement
                                best_swap = (p, q)
            if best_swap:
                p, q = best_swap
                final_h[p] = 0
                final_h[q] = 1
            else:
                break
            
    # Compute final C5
    A_final = np.zeros(n0)
    for k in range(n0):
        A_final[k] = np.sum(final_h * np.roll(final_h, k))
    w_final = np.min(A_final)
    c5_final = w_final * dx
    
    # Scale to target size
    N_target = 360
    upscale_factor = N_target // n0
    h_upscaled = np.tile(final_h, upscale_factor)
    
    return (h_upscaled.tolist(), float(c5_final), N_target)
