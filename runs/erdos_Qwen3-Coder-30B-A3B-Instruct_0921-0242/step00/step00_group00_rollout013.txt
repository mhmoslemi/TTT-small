import numpy as np
import time
from scipy.linalg import qr
from scipy.fft import fft, ifft

# The original constant from the brainwave approach
initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 256
    M = 8  # Number of frequency modes (excluding constant term)
    n_freqs = 2 * M  # Total number of Fourier coefficients
    dx = 2.0 / n_points
    
    # Precompute reconstruction matrix
    # Φ is n x 2M matrix of cosine/sine basis
    i_vals = np.arange(n_points)
    freqs = np.arange(1, M + 1)
    cos_basis = np.cos(2 * np.pi * np.outer(i_vals, freqs) / n_points)
    sin_basis = np.sin(2 * np.pi * np.outer(i_vals, freqs) / n_points)
    Phi = np.hstack([cos_basis, sin_basis])  # n x n_freqs
    Phi_T_Phi = Phi.T @ Phi  # n_freqs x n_freqs
    
    # Compute pseudo-inverse: (Phi^T Phi)^{-1} Phi^T
    # This should be used for projection (but not during optimization!)
    # Avoid computing twice in loop.
    Q, R = qr(Phi_T_Phi)
    Phi_T_Phi_inv = np.linalg.solve(R.T, Q.T)
    
    def project_coefficients(c):
        # c is vector of length 2*M
        # Φ c + 0.5 gives h values
        h = Phi @ c + 0.5
        # Clip to [0, 1]
        h_clipped = np.clip(h, 0, 1)
        # Compute correction d so that Φ(d) + h_clipped = h_clipped exactly
        correction = h_clipped - h
        delta_c = Phi_T_Phi_inv @ (Phi.T @ correction)
        c_projected = c + delta_c
        return c_projected
    
    def evaluate_c5(c):
        h = Phi @ c + 0.5
        # Compute convolution
        conv = np.convolve(h, 1 - h, mode="full") * dx
        return conv.max()
    
    # CMA-ES algorithm
    def cma_es(n_points, M, Phi, Phi_T_Phi_inv, budget_s, seed):
        np.random.seed(seed)
        time_budget = budget_s * 0.4 if n_points == 256 else budget_s * 0.6
        start_t = time.time()
        
        lambda_ = 4 + int(np.floor(3 * np.log(2 * M)))
        mu = lambda_ // 2
        
        # Parameters
        sigma = 0.2
        c_sigma = 0.3
        c_c = 0.1
        c_1 = 0.1
        c_mu = 0.1
        
        # Initialize
        mean = np.zeros(2 * M)
        C = np.eye(2 * M)
        chi = np.sqrt(2 * M) * (1 - 1/(4 * M) + 1/(21 * M**2))
        
        best_c = None
        best_value = np.inf
        
        # For multi-restart
        restarts = 10 if n_points == 256 else 10
        time_per_restart = time_budget / restarts
        
        for r in range(restarts):
            if time.time() - start_time > budget_s - 0.5:
                break
                
            restart_seed = seed + r
            np.random.seed(restart_seed)
            
            gen_start = time.time()
            # Initialize population
            pop = []
            pop_fitness = []
            for _ in range(lambda_):
                c = np.random.randn(2 * M) * sigma
                pop.append(c)
                
            while time.time() - gen_start < time_per_restart and time.time() - start_time < budget_s - 0.5:
                # Project all individuals in batch
                projected_pop = [project_coefficients(c) for c in pop]
                
                # Evaluate
                fitness = [evaluate_c5(c) for c in projected_pop]
                # Keep best
                best_idx = np.argmin(fitness)
                if fitness[best_idx] < best_value:
                    best_value = fitness[best_idx]
                    best_c = projected_pop[best_idx].copy()
                    
                # Resample
                indices = np.argsort(fitness)[:mu]
                parents = [projected_pop[i] for i in indices]
                
                # Update mean
                mu_vec = np.mean(parents, axis=0)
                
                # Update C
                diff = np.array(parents) - mean
                C = (1 - c_1 - c_mu) * C + c_1 * np.outer(mean - mu_vec, mean - mu_vec) + c_mu * np.dot(diff.T, diff) / mu
                
                # Update mean
                mean = (1 - c_c) * mean + c_c * mu_vec
                
                # Mutate
                pop = []
                for _ in range(lambda_):
                    # Sample new individual
                    z = np.random.randn(2 * M)
                    c = mean + sigma * np.dot(C, z)
                    pop.append(c)
                    
                # Occasionally inject new individuals
                if np.random.rand() < 0.2:
                    for _ in range(lambda_ // 5):
                        c = np.random.randn(2 * M) * sigma
                        pop.append(c)
            
            # Check for early exit
            if time.time() - start_time > budget_s - 0.5:
                break
                
        return best_c, best_value
    
    # Run optimization on 256 grid
    best_c_256, best_value_256 = cma_es(n_points, M, Phi, Phi_T_Phi_inv, budget_s, seed)
    
    # If time permits and we haven't hit target yet, do finer grid
    if budget_s - (time.time() - start_time) > 10.0 and best_value_256 > 0.3807:
        # Double n points to 512
        n_points = 512
        dx = 2.0 / n_points
        i_vals = np.arange(n_points)
        cos_basis = np.cos(2 * np.pi * np.outer(i_vals, freqs) / n_points)
        sin_basis = np.sin(2 * np.pi * np.outer(i_vals, freqs) / n_points)
        Phi = np.hstack([cos_basis, sin_basis])
        Phi_T_Phi = Phi.T @ Phi
        Q, R = qr(Phi_T_Phi)
        Phi_T_Phi_inv = np.linalg.solve(R.T, Q.T)
        best_c_512, best_value_512 = cma_es(n_points, M, Phi, Phi_T_Phi_inv, budget_s, seed)
        
        # Use the best of the two
        if best_value_512 < best_value_256:
            best_c = best_c_512
            best_value = best_value_512
        else:
            best_c = best_c_256
            best_value = best_value_256
    else:
        best_c = best_c_256
        best_value = best_value_256
    
    # Refinement step: projected gradient descent
    def refine_c5(c, max_iter=200):
        best_c = c.copy()
        best_val = evaluate_c5(best_c)
        step_size = 0.1
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 0.5:
                break
                
            h = Phi @ best_c + 0.5
            # Find the position k that gives the max value
            # implement convolution and find the index
            conv = np.convolve(h, 1 - h, mode="full")
            max_k = np.argmax(conv) - (len(h) - 1)  # shift it back to -n+1 to n-1
            
            # Assemble gradient: ∂C5/∂h_i = (1 - h_{i+k}) - h_{i-k}
            n = len(h)
            grad = np.zeros(n)
            sign = 1
            for i in range(n):
                j_left = (i - max_k) % n
                j_right = (i + max_k) % n
                grad[i] = (1 - h[j_right]) - h[j_left]
            
            # Map to coefficient space
            grad_coeff = Phi.T @ grad
            
            # Take step
            new_c = best_c - step_size * grad_coeff
            
            # Project
            new_c = project_coefficients(new_c)
            
            new_val = evaluate_c5(new_c)
            if new_val < best_val:
                best_c = new_c
                best_val = new_val
                
        return best_c, best_val
    
    # Run refinement
    c_refined, best_value = refine_c5(best_c)

    # Final reconstruction
    h = Phi @ c_refined + 0.5
    # Enforce integral constraint
    integral = np.sum(h) * dx
    correction = (1 - integral) / n_points
    h = np.clip(h + correction, 0, 1)
    
    # Final evaluation of C5
    conv = np.convolve(h, 1 - h, mode="full") * dx
    c5_bound = conv.max()
    
    # Return final h as list, final bound, n points
    return (h.tolist(), float(c5_bound), n_points)
