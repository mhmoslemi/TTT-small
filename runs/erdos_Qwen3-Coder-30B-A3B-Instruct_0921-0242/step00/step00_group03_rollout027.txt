import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations with mode='full'
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Extract the valid k values (excluding k=0)
    valid_corr = corr[n-1:]  # This corresponds to k=1,2,...,n-1
    # The overlap integral is 1 - min of the correlations
    return 1.0 - np.min(valid_corr)

def solve_milp(n, max_seconds):
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True, low=0)
    y = cp.Variable((n-1, n-1), boolean=True)
    
    # Constraints
    constraints = [
        cp.sum(h) == n // 2,  # weight constraint
    ]
    
    # Enforce y[i,s] = h[i] * h[i+s] for valid i,s
    for s in range(1, n):
        for i in range(n-s):
            # y[i,s] <= h[i]
            constraints.append(y[i,s-1] <= h[i])
            # y[i,s] <= h[i+s]
            constraints.append(y[i,s-1] <= h[i+s])
            # y[i,s] >= h[i] + h[i+s] - 1
            constraints.append(y[i,s-1] >= h[i] + h[i+s] - 1)
    
    # Enforce t <= A_s = sum_{i=0}^{n-1-s} y[i,s-1]
    for s in range(1, n):
        # j = s-1, sum over i from 0 to n-1-s
        sum_ys = cp.sum(y[:n-s, s-1])
        constraints.append(t <= sum_ys)
    
    # Objective
    objective = cp.Maximize(t)
    
    # Problem
    problem = cp.Problem(objective, constraints)
    
    # Solve
    problem.solve(solver=cp.CBC, max_seconds=max_seconds)
    
    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = np.round(h.value).astype(int)
        t_sol = t.value
        return h_sol, t_sol
    else:
        # Get the best feasible solution if not optimal
        h_sol = np.round(h.value).astype(int)
        return h_sol, -1

def compute_autocorrelations(h):
    """Compute the min overlap for all non-zero shifts"""
    n = len(h)
    min_overlap = float('inf')
    for k in range(1, n):
        s = sum(h[i] * h[i+k] for i in range(n-k))
        min_overlap = min(min_overlap, s)
    return min_overlap

def hill_climbing(h, max_time, verbose=False):
    n = len(h)
    best_h = h.copy()
    best_t = compute_autocorrelations(best_h)
    
    start_time = time.time()
    iterations = 0
    
    while True:
        if time.time() - start_time > max_time:
            break
            
        # Try to improve by swapping
        # Pick two indices: one with 1 (index1) and one with 0 (index0)
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        
        # Temporarily swap
        test_h = best_h.copy()
        test_h[i1] = 0
        test_h[i0] = 1
        
        # Compute changes in overlaps
        old_t = best_t
        new_t = compute_autocorrelations(test_h)
        
        if new_t > old_t:
            # Accept the swap
            best_h = test_h
            best_t = new_t
            iterations += 1
        elif iterations > 0 and iterations % 100 == 0:
            # Minor restart occasionally
            pass
            
    if verbose:
        print(f"Hill-climbing: {iterations} swaps attempted, final min overlap: {best_t}")
    
    return best_h, best_t

def polish_continuous(h_binary):
    """Try to slightly improve score using continuous optimization."""
    h = h_binary.astype(float)
    n = len(h)
    dx = 2.0 / n
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1-h_vec, mode='full') * dx
        valid_corr = corr[n-1:]
        return 1.0 - np.min(valid_corr)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    def constraint_bounds(h_vec):
        return np.min(h_vec), 1.0 - np.max(h_vec)
    
    def fun(h_vec):
        return objective(h_vec)
    
    def jac(h_vec):
        # Approximate gradient
        delta = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_vec_plus = h_vec.copy()
            h_vec_plus[i] += delta
            h_vec_minus = h_vec.copy()
            h_vec_minus[i] -= delta
            grad[i] = (fun(h_vec_plus) - fun(h_vec_minus)) / (2*delta)
        return grad
    
    # SLSQP optimizer
    bounds = [(0.0, 1.0) for _ in range(len(h))]
    x0 = h + np.random.normal(0, 0.01, n)
    x0 = np.clip(x0, 0, 1)
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(fun, x0, jac=jac, bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 100})
        if result.success and result.fun < objective(h):
            return result.x
    except:
        pass
    return h

def upsample(h, factor):
    """Replicate each entry 'factor' times"""
    upsampled = np.repeat(h, factor)
    return upsampled

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    remaining = budget_s
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    n_list = []
    if remaining > 300:
        n_list.append(100)
        remaining = remaining - 300  # reserve for later work
    
    if remaining > 150:
        n_list.append(120)
        remaining = remaining - 150
    
    for n in n_list:
        if time.time() - start_time > budget_s - 30:
            break
            
        solver_time = min(remaining // 2, 300 if n==100 else 150)
        
        print(f"[INFO] Solving MILP for n={n}")
        h_bin, t_opt = solve_milp(n, solver_time)
        
        if h_bin is not None and t_opt >= 0:
            print(f"[MILP] n={n}, t={t_opt}, estimated C5 = {1.0-2*t_opt/n}")
            
            # Improve with hill climbing
            h_improved, t_improved = hill_climbing(h_bin, 30, verbose=False)  # Limited time
            
            # Compute final C5
            final_c5 = 1.0 - 2*t_improved/n
            
            if final_c5 < best_c5:
                best_h = h_improved
                best_c5 = final_c5
                best_n = n
                
    # Optional polishing with continuous optimizer
    if best_h is not None and time.time() - start_time < budget_s - 30:
        final_h = polish_continuous(best_h)
        final_c5 = evaluate_erdos_solution(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    # Ensure best_h still evaluates
    c5_eval = evaluate_erdos_solution(best_h)
    
    # Optionally upsample to make smoother plot (not required by judge, but helpful)
    upsampled = upsample(best_h, 4)
    final_c5_eval = evaluate_erdos_solution(upsampled)
    
    # Return the best version
    return (upsampled, final_c5_eval, len(upsampled))
