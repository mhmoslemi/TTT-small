import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def project_simplex(h, target_sum):
    """Project h onto the unit simplex with sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    if abs(delta) < 1e-12:
        return h

    free = np.where((h > 0) & (h < 1))[0]
    if len(free) == 0:
        return h

    sorted_indices = np.argsort(h[free])[::-1]
    free = free[sorted_indices]

    while abs(delta) > 1e-12 and len(free) > 0:
        num_free = len(free)
        shift = delta / num_free
        for i in range(num_free):
            new_val = h[free[i]] + shift
            if new_val <= 0:
                h[free[i]] = 0
                delta -= -h[free[i]]
                free = np.delete(free, i)
                break
            elif new_val >= 1:
                h[free[i]] = 1
                delta -= (1 - h[free[i]])
                free = np.delete(free, i)
                break
        else:
            h[free] += shift
            delta = 0

    return h


def compute_autocorrelations(h):
    """Compute linear autocorrelations r_k(h) for k = 1,...,n-1."""
    n = len(h)
    padded_h = np.pad(h, (0, n), mode='constant')
    H = fft(padded_h)
    r = np.real(ifft(H * np.conj(H)))
    return r[1:n]


def soft_min_gradient(h, r, alpha):
    """Compute weighted gradient of soft minimum with given weights."""
    n = len(h)
    # Compute weights
    exp_r = np.exp(-alpha * r)
    w = exp_r / np.sum(exp_r)
    # Gradient computation using FFT-based convolution
    # ∇S_α = w * h + reverse(w) * h
    w_rev = w[::-1]
    # Padding for linear convolution
    padded_w = np.pad(w, (0, n), mode='constant')
    padded_w_rev = np.pad(w_rev, (0, n), mode='constant')
    padded_h = np.pad(h, (0, n), mode='constant')
    g1 = np.real(ifft(fft(padded_w) * fft(padded_h)))
    g2 = np.real(ifft(fft(padded_w_rev) * fft(padded_h)))
    grad = g1[:n] + g2[:n]
    return grad

def backtracking_line_search(h, grad, alpha, eta=0.1, max_iter=10):

def round_to_step_function(h, dx):
    """Round h to a step function with same integral."""
    n = len(h)
    # Sort by value
    sorted_indices = np.argsort(h)[::-1]
    # Select first n//2 entries to be 1
    new_h = np.zeros(n)
    # Set learned values to 1
    for i in range(n // 2):
        new_h[sorted_indices[i]] = 1
    # Make sure sum constraint is maintained
    scale_factor = (n // 2) / np.sum(new_h)
    new_h = new_h * scale_factor
    return new_h
