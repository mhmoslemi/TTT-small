import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fftpack import fft, ifft

def evaluate_C5(h_values, dx):
    """Evaluate the C5 bound for given h values."""
    n = len(h_values)
    # Compute the circular autocorrelations using FFT
    H = fft(h_values)
    autocorr = ifft(np.abs(H)**2).real
    # The C5 is 1 - min_autocorr because of the integral setup
    # But our original problem defines C5 = max_k int h(x)(1 - h(x+k))dx
    # Which is dx * (n/2 - min_k A_k) where A_k = sum h_i * h_{i+k}
    min_autocorr = np.min(autocorr)
    return dx * (n/2 - min_autocorr)

def block_family_search(n_points=400, delta=0.02):
    """Search through 3-block functions."""
    best_C5 = float('inf')
    best_h = None
    dx = 2.0 / n_points
    grid = np.linspace(0, 2, n_points)
    
    for a in np.arange(0, 2-delta, delta):
        for b in np.arange(a+delta, 2, delta):
            p = (1 - a) / (b - a)
            if 0 <= p <= 1:
                h = np.zeros(n_points)
                h[grid < a] = 1.0
                h[(grid >= a) & (grid < b)] = p
                h[grid >= b] = 0.0
                C5 = evaluate_C5(h, dx)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h.copy()
    return best_h, best_C5

def four_block_search(n_points=400, delta=0.04):
    """Search through 4-block functions."""
    best_C5 = float('inf')
    best_h = None
    dx = 2.0 / n_points
    grid = np.linspace(0, 2, n_points)
    
    for a in np.arange(0, 2-delta, delta):
        for b in np.arange(a+delta, 2-delta, delta):
            for c in np.arange(b+delta, 2, delta):
                # Balance constraint on middle blocks
                p = (1 - a) / (b - a)  # p = (1 - a) / (b - a), only valid if a < b
                q = (1 - a - p*(b-a)) / (2 - c)  # q = (1 - a - p*(b-a)) / (2 - c)
                if 0 <= p <= 1 and 0 <= q <= 1:
                    h = np.zeros(n_points)
                    h[grid < a] = 1.0
                    h[(grid >= a) & (grid < b)] = p
                    h[(grid >= b) & (grid < c)] = q
                    h[grid >= c] = 0.0
                    C5 = evaluate_C5(h, dx)
                    if C5 < best_C5:
                        best_C5 = C5
                        best_h = h.copy()
    return best_h, best_C5

def gaussian_mixture(h0, n_points, seed=42):
    """Generate a Mixture of Gaussians-like h from current h0."""
    np.random.seed(seed)
    n_bumps = 3  # Number of bumps
    centers = np.random.uniform(0, 2, n_bumps)
    scales = np.random.uniform(0.02, 0.1, n_bumps)
    amplitudes = np.random.uniform(0.1, 1.0, n_bumps)
    
    h = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    
    for i in range(n_bumps):
        # Gaussian-shaped bump
        bump = amplitudes[i] * np.exp(-0.5 * ((x - centers[i]) / scales[i])**2)
        h += bump
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to have integral = 1 (same as original h0)
    total_mass = np.trapz(h, x)
    if total_mass > 0:
        h = h * (1.0 / total_mass)
    return h

def gradient_ascent(h0, n_points, budget_s=300):
    """Perform projected gradient ascent."""
    dx = 2.0 / n_points
    h = h0.copy()
    h = h / np.sum(h) * (n_points / 2)
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Start gradient descent
    start_t = time.time()
    eta = 0.5
    decay = 0.98
    max_iter = 50000
    for i in range(max_iter):
        if time.time() - start_t > budget_s - 1:
            break
        # Compute autocorrelation via FFT
        H = fft(h)
        autocorr = ifft(np.abs(H)**2).real
        kstar = np.argmin(autocorr)
        # Gradient: 2 * roll(h, kstar)
        g = 2 * np.roll(h, kstar)
        h_new = h + eta * g
        # Projection
        h_new = np.clip(h_new, 0, 1)
        # Re-scale to match sum constraint
        mean_offset = (np.sum(h_new) - n_points / 2) / n_points
        h_new = h_new - mean_offset
        h_new = np.clip(h_new, 0, 1)
        # Second pass correction if needed
        mean_offset = (np.sum(h_new) - n_points / 2) / n_points
        h_new = h_new - mean_offset
        h_new = np.clip(h_new, 0, 1)
        h = h_new
        C5 = evaluate_C5(h, dx)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
        eta *= decay
    
    return best_h, best_C5
