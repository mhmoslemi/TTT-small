import numpy as np

def decode(x, M):
    """Decode flat parameter vector to segment lengths u and heights a."""
    u = x[:M]
    a0 = x[M]
    v = x[M+1:]
    a = np.empty(M+1)
    a[0] = a0
    a[1:] = np.maximum(0, a0 - np.cumsum(v))
    a[M] = max(0, a[M-1] - v[M-2]) if M > 1 else a0
    return u, a

def overlap_piecewise(s, a):
    """Compute the exact max overlap integral for piecewise constant h."""
    M = len(a) - 1
    # Construct list of breakpoints k where F(k) can change slope
    K = set()
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0 <= k1 <= 2: K.add(k1)
            if 0 <= k2 <= 2: K.add(k2)
    # Add interval ends
    K.add(0)
    K.add(2)
    K = sorted(K)
    
    if len(K) < 2:
        # Only one interval: fixed value
        F = 0
        for j in range(M+1):
            for l in range(M+1):
                if s[j+1] <= s[l] or s[j] >= s[l+1]:
                    continue
                # Intersection length
                L = max(0, min(s[j+1], s[l+1] - 0) - max(s[j], s[l] - 0))
                F += a[j] * (1 - a[l]) * L
        return F
    
    F_vals = []
    for k in K:
        F = 0.0
        for j in range(M+1):
            for l in range(M+1):
                if s[j+1] <= s[l] or s[j] >= s[l+1] - k:
                    continue
                # Intersection length
                L = max(0, min(s[j+1], s[l+1] - k) - max(s[j], s[l] - k))
                F += a[j] * (1 - a[l]) * L
        F_vals.append(F)
    
    return max(F_vals)

def project_capped_simplex(v, total):
    """Project vector v onto the capped simplex {sum = total, 0 <= v[i] <= 1}"""
    v_clipped = np.clip(v, 0, 1)
    total_clipped = np.sum(v_clipped)
    if abs(total_clipped - total) < 1e-12:
        return v_clipped
    # Use binary search to find proper lambda
    diff = total_clipped - total
    if diff < 0:
        # Must increase values
        # Since they're already clipped, the only way to increase is to break the upper bounds.
        # Into view: solve np.sum(v + x) = total with constraints v + x <= 1,
        # subject to x >= 0 and v + x <= 1 => x <= 1 - v, x >= 0
        # So we can increase by min(1-v, max(0, required_increase))
        # But since v = 1 (boundary), we're trying too hard.
        # Simplest approach: add uniform adjustment
        adjustment = (total - total_clipped) / len(v)
        v = np.clip(v + adjustment, 0, 1)
        # Now renormalize to total
        v = v * total / np.sum(v)
        return v
    else:
        # Need to decrease values
        # Binary search on lambda to reconstruct v such that sum = total
        def sum_diff(lambda_factor):
            v_test = np.clip(v - lambda_factor, 0, 1)
            return np.sum(v_test) - total

        left, right = 0, np.max(v)
        while abs(right - left) > 1e-12:
            mid = (left + right) / 2
            if sum_diff(mid) >= 0:
                left = mid
            else:
                right = mid
        lambda_val = left
        v_proj = np.clip(v - lambda_val, 0, 1)
        # Final normalization
        v_proj = v_proj * total / np.sum(v_proj)
        return v_proj

def evaluate(x, M, n_points, integral_target=1.0):
    """Evaluate objective with penalties for constraint violations."""
    x = np.array(x)
    try:
        u, a = decode(x, M)
        # Check constraints on segments and heights:
        if np.any(u < 0):
            return 1e12
        s = np.concatenate([[0], np.cumsum(u)])
        if s[-1] != 2:
            return 1e12
        if a[0] < 0 or a[0] > 1:
            return 1e12
        if np.any(a < 0):
            return 1e12
        s_integrals = np.diff(s)
        integral_check = np.sum(a * s_integrals)
        if abs(integral_check - integral_target) > 1e-8:
            penalty = 1e6 * (integral_check - integral_target)**2
        else:
            penalty = 0
        # Compute overlap C5
        C5 = overlap_piecewise(s, a)
        return C5 + penalty
    except Exception:
        return 1e12

def make_h_from_params(x, M, n_points):
    u, a = decode(x, M)
    s = np.concatenate([[0], np.cumsum(u)])
    dx = 2.0 / n_points
    h = np.empty(n_points)
    for i in range(n_points):
        x_pos = (i + 0.5) * dx
        # Find j such that x_pos in [s[j], s[j+1])
        j = 0
        while j < len(s) - 1 and x_pos >= s[j+1]:
            j += 1
        h[i] = a[j]
    # Adjust by capped simplex projection
    h = project_capped_simplex(h, n_points / 2)
    return h
