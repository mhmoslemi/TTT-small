import time
import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize_scalar

def project_capped_simplex(v, total=0.5):
    """Project vector v onto capped simplex: 0 <= x_i <= 1, sum x_i = total"""
    n = len(v)
    # Sort v in descending order with indices
    sorted_indices = np.argsort(-v)
    sorted_v = v[sorted_indices]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_v)
    
    # Find the projection parameter
    tmp = (cumsum - total) / np.arange(1, n+1)
    gamma = tmp[np.argmax(sorted_v >= tmp)]
    
    # Compute the projection
    projected = np.maximum(0, np.minimum(1, v - gamma))
    
    # Adjust if sum is not exactly total
    diff = total - np.sum(projected)
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            idx = np.where(projected < 1)[0]
            to_add = min(diff, np.sum(1 - projected[idx]))
            if to_add > 0:
                for i in idx:
                    if diff <= 0:
                        break
                    can_add = min(1 - projected[i], diff)
                    projected[i] += can_add
                    diff -= can_add
        else:
            idx = np.where(projected > 0)[0]
            to_sub = min(-diff, np.sum(projected[idx]))
            if to_sub > 0:
                for i in idx:
                    if diff >= 0:
                        break
                    can_sub = min(projected[i], -diff)
                    projected[i] -= can_sub
                    diff += can_sub
    
    return projected

def auto_corr(h):
    """Compute zero-padded linear auto-correlation O_k(h) efficiently via FFT."""
    n = len(h)
    h_pad = np.pad(h, (0, n), mode='constant')
    # Compute auto-correlation via FFT (convention: correlate(x, y) = fft(x)*conj(fft(y)), which is the convolution of x and reverse(y))
    # We want h * reverse(h), so compute fft(h) * conj(fft(reverse(h)))
    # Actually, let's just use fftconvolve with zero padding
    ac = fftconvolve(h, h[::-1], mode='full')
    ac = ac[n-1:2*n-1]
    return ac

def grad_F(beta, h, O, w):
    """Compute gradient of F_beta(h) = 1 - (1/beta)*log(sum_k exp(beta * O_k))"""
    n = len(h)
    g = np.zeros_like(h)
    # Compute w * conv(h) + w * conv_rev(h) where conv is zero padded at both ends
    # Use fftconvolve for fast linear convolution
    # w * conv(h) (compress convolution)
    conv_forward = fftconvolve(w, h, mode='full')
    conv_forward = conv_forward[n-1 : 2*n-1]  # keep only the relevant part
    
    # w * conv_rev(h) where conv_rev is like conv but reversed
    conv_reverse = fftconvolve(w, h[::-1], mode='full')
    conv_reverse = conv_reverse[n-1 : 2*n-1][::-1]  # reverse the result
    
    g = conv_forward + conv_reverse
    return -g  # negative gradient since we're minimizing

def optimize_one_start(n, beta_schedule, time_limit):
    """Run continuation with projected gradient descent from one start."""
    start_time = time.time()
    np.random.seed(int(10000*time.time()) % 1000000)
    
    # Initial point: interpolate from initial_h_values or random
    # We're going to use random initialization for now
    u = np.random.exponential(1, n)
    h = u / np.sum(u) * (n/2)
    h = np.clip(h, 0, 1)
    h = project_capped_simplex(h, n/2)

    # Initialize best solution
    best_h = h.copy()
    best_C5 = 1e10
    prev_Fbeta = 1e10
    
    # β schedule
    beta_list = [np.exp(np.log(beta_schedule[0]) + t*(np.log(beta_schedule[1]) - np.log(beta_schedule[0])) / (len(beta_schedule)-1)) for t in range(len(beta_schedule))]
    
    for beta_idx, beta in enumerate(beta_list):
        if time.time() - start_time > time_limit:
            break
            
        # Inner loop with projected gradient descent
        max_iter_inner = 5000  # reasonable limit
        iter_count = 0
        last_improvement = 0
        beta_time_start = time.time()
        
        while iter_count < max_iter_inner and time.time() - start_time < time_limit - 0.3:
            # compute all O_k and weights
            O = auto_corr(h)
            w = np.exp(beta * O)
            w /= np.sum(w)
            
            # compute gradient
            g = grad_F(beta, h, O, w)
            
            # project gradient + backtracking line search
            alpha = 0.1
            lr = 1e-6
            min_norm = 0.0001
            recent_improvements = [] # track previous steps to detect stagnation
            
            for _ in range(30):  # max backtrack steps
                h_new = h - alpha * g
                h_new = project_capped_simplex(h_new, n/2)
                O_new = auto_corr(h_new)
                curr_F = 1. - (1./beta)*np.log(np.sum(np.exp(beta * O_new)))
                
                # Armijo condition
                gradient_norm_sq = np.sum(g**2)
                if curr_F <= (1 - (1./beta)*np.log(np.sum(np.exp(beta * O))) - lr * alpha * gradient_norm_sq) or alpha < 1e-10:
                    # accept
                    h = h_new
                    break
                alpha *= 0.5
                
            # check for convergence by checking improvement in Fbeta
            if beta_idx == 0:
                curr_F = 1 - (1./beta)*np.log(np.sum(np.exp(beta * O)))
            else:
                curr_F = 1 - (1./beta)*np.log(np.sum(np.exp(beta * O)))
            delta = (prev_Fbeta - curr_F) if prev_Fbeta != 1e10 else 1e10
            prev_Fbeta = curr_F
            
            # Track best ever solution
            corr = fftconvolve(h, 1-h[::-1], mode='full')
            corr = corr[len(h)-1:2*len(h)-1]
            C5 = np.max(corr) * 2.0 / n  # dx = 2.0/n
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
            
            iter_count += 1
            last_improvement += 1
            
            if last_improvement > 100:
                break
    
    # Exact refinement using projected gradient, no smoothing, following worst
    # direction
    max_iter_final = 200
    last_C5 = best_C5
    final_time_start = time.time()
    for it in range(max_iter_final):
        if time.time() - final_time_start > 0.3 or time.time() - start_time > time_limit - 0.3:
            break
        corr = fftconvolve(best_h, 1-best_h[::-1], mode='full')
        corr = corr[len(best_h)-1:2*len(best_h)-1]
        C5 = np.max(corr) * 2.0 / n  # dx = 2.0/n
        max_k = np.argmax(corr)
        # compute O_{max_k} gradient
        # O_{max_k}(h) = sum_i h[i] * h[i + max_k] where i ranges in valid region
        # Derivative vector is [h[i+max_k], h[i-max_k]] (but watch boundaries)
        # For δO / δh_i, we're looking at: 
        #  1. the h[i + max_k] term (if i + max_k < len(h))
        #  2. the h[i - max_k] term (if i >= max_k)
        grad_corr = np.zeros_like(best_h)
        if max_k < len(best_h):
            grad_corr[max_k:] += best_h[:-max_k]
        if max_k > 0:
            grad_corr[:-max_k] += best_h[max_k:]
        # Gradient based on C5 = 1 - min_k O_k ≈ 1 - O(max_k), so we want to minimize O(max_k)
        # The gradient of O_max_k w.r.t. h is: \sum_{i in support} (h[i+k] on positions i and h[i-k] on positions i)
        # Include a fixed stepsize α = l / sqrt(it+200) where l = 0.1
        alpha = 0.1 / np.sqrt(it+200)
        h_new = best_h - alpha * grad_corr
        h_new = project_capped_simplex(h_new, n/2)
        # Recalculate C5
        corr = fftconvolve(h_new, 1-h_new[::-1], mode='full')
        corr = corr[len(h_new)-1:2*len(h_new)-1]
        C5_new = np.max(corr) * 2.0 / n
        if C5_new < last_C5:
            last_C5 = C5_new
            best_h = h_new.copy()
    
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_t = time.time()
    n_values = [256, 384]  # Try multiple resolutions
    
    best_C5 = 1e10
    best_h = None
    best_n = 256
    
    restarts = 8  # number of random seeds
    
    for n in n_values:
        if time.time() - start_t > budget_s - 1.0:
            break
        # Beta schedule for continuation (small β early, large β later)
        beta_schedule = [0.5, 40]
        # adjust schedule length for compute
        n_schedule = 12
        
        for r in range(restarts):
            if time.time() - start_t > budget_s - 1.0:
                break
                
            h, C5 = optimize_one_start(n, beta_schedule, budget_s - (time.time() - start_t) - 1.0)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h
                best_n = n
                
    # Final polished evaluation
    if best_h is not None:
        # re-evaluate exact C5
        corr = fftconvolve(best_h, 1-best_h[::-1], mode='full')
        corr = corr[len(best_h)-1:2*len(best_h)-1]
        final_C5 = np.max(corr) * 2.0 / best_n
        return (best_h.tolist(), final_C5, best_n)
    else:
        # Default return if nothing works quickly
        return ([0.5]*100, 0.5, 100)
