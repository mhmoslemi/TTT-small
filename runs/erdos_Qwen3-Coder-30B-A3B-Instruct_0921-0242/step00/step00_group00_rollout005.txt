import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.linalg import cho_factor, cho_solve

# Global constants
MAX_TIME_S = 600.0
TARGET_C5 = 0.3807
N_INITIAL = 256
N_REFINE = 512
M = 8  # Number of frequencies (cos and sin each)
N_VARS = 2 * M  # Total variables: a_1..a_M, b_1..b_M

# Pre-compute the Fourier basis matrix for n=256
def build_fourier_basis(n, m):
    """Builds the n x 2m Fourier basis matrix for cosine/sine terms."""
    i = np.arange(n)
    mat = np.zeros((n, 2 * m))
    for j in range(m):
        mat[:, 2 * j] = np.cos(2 * np.pi * (j + 1) * i / n)
        mat[:, 2 * j + 1] = np.sin(2 * np.pi * (j + 1) * i / n)
    return mat

# Compute inv(Phi^T Phi) * Phi^T projection matrix once
def build_projection_matrix(Phi):
    """Compute the projection matrix P = inv(Phi^T Phi) * Phi^T"""
    PhiTPhi = Phi.T @ Phi
    try:
        # Factorize for efficiency
        L = cho_factor(PhiTPhi)
        return lambda v: cho_solve(L, Phi.T @ v)
    except:
        return lambda v: np.linalg.solve(PhiTPhi, Phi.T @ v)

# Reconstruct h from coefficients c (assumes a0 = 0.5)
def reconstruct_h(c, Phi):
    return Phi @ c + 0.5

# Check if all values in h are in [0,1]
def is_feasible(h):
    return np.all((h >= 0) & (h <= 1))

# Project a coefficient vector c onto the feasible set
def project_c(c, Phi, proj_func):
    h = reconstruct_h(c, Phi)
    # Clip to [0, 1] and then project back
    h_clamped = np.clip(h, 0, 1)
    # Solve for correction: Phi * delta_c = (h_clamped - h)
    delta_c = proj_func(h_clamped - h)
    return c + delta_c

# Evaluate the max overlap integral C5
def compute_C5(h, dx):
    # compute the full correlation
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c = None
    best_C5 = np.inf
    
    Phi_initial = build_fourier_basis(N_INITIAL, M)
    Phi_refine = build_fourier_basis(N_REFINE, M)
    proj_func_initial = build_projection_matrix(Phi_initial)
    proj_func_refine = build_projection_matrix(Phi_refine)
    
    # CMA-ES parameters
    lambda_pop = 4 + int(3 * np.log(N_VARS))  # 30
    mu = lambda_pop // 2
    sigma = 0.2
    alpha = 0.1  # gradient step size decay
    
    def eval_c5(c, n_grid, Phi, proj_func):
        try:
            h = reconstruct_h(c, Phi)
            h = np.clip(h, 0, 1)
            h = project_h_to_unit_integral(h, n_grid)
            dx = 2.0 / n_grid
            return compute_C5(h, dx)
        except Exception:
            return np.inf
    
    def project_h_to_unit_integral(h, n_grid):
        integral = np.sum(h) * 2.0 / n_grid
        if abs(integral - 1.0) > 1e-12:
            diff = (1.0 - integral) / n_grid
            h += diff
            # Clamp to [0, 1] again
            h = np.clip(h, 0, 1)
        return h
    
    # Multi restart loop with both resolutions
    n_grid = N_INITIAL
    Phi = Phi_initial
    proj_func = proj_func_initial
    
    restart_count = 0
    while time.time() - start_time < (budget_s - 0.5) and n_grid <= N_REFINE:
        for restart in range(15):
            if time.time() - start_time > (budget_s - 0.5):
                break
                
            # Warm up from known construction if available
            c_start = np.random.normal(0, sigma, N_VARS)
            
            # Evolve population
            population = []
            fitness = []
            
            # Initialize population
            for _ in range(lambda_pop):
                individual = np.random.normal(0, sigma, N_VARS)
                individual = project_c(individual, Phi, proj_func)
                fitness_individual = eval_c5(individual, n_grid, Phi, proj_func)
                population.append(individual)
                fitness.append(fitness_individual)
            
            # Sort by fitness (ascending)
            indices = np.argsort(fitness)
            population = [population[i] for i in indices]
            fitness = [fitness[i] for i in indices]
            
            # Main CMA-ES loop
            gen = 0
            while time.time() - start_time < (budget_s - 0.5) and gen < 200:
                # Selection
                selected = population[:mu]
                fitness_selected = fitness[:mu]
                
                # Compute mean
                mean = np.mean(selected, axis=0)
                mean = project_c(mean, Phi, proj_func)
                
                # Update distribution
                sigma = max(1e-7, sigma * 0.999)
                
                # Generate offspring
                offspring = []
                for _ in range(lambda_pop):
                    # Sample from distribution
                    z = np.random.normal(0, sigma, N_VARS)
                    c = mean + z
                    c = project_c(c, Phi, proj_func)
                    offspring.append(c)
                
                # Evaluate offspring
                fitness_offspring = []
                for off in offspring:
                    f = eval_c5(off, n_grid, Phi, proj_func)
                    fitness_offspring.append(f)
                
                # Combine and sort
                population = selected + offspring
                fitness = fitness_selected + fitness_offspring
                indices = np.argsort(fitness)
                population = [population[i] for i in indices]
                fitness = [fitness[i] for i in indices]
                
                # Update best
                if fitness[0] < best_C5:
                    best_C5 = fitness[0]
                    best_c = population[0].copy()
                
                gen += 1
                
                # Check timeout early
                if time.time() - start_time > (budget_s - 0.5):
                    break
            
            restart_count += 1
            
            # Early stopping if we're close enough
            if best_C5 <= TARGET_C5:
                break
        
        # Move to higher resolution if time allows
        if n_grid == N_INITIAL and best_C5 <= 0.380:
            n_grid = N_REFINE
            Phi = Phi_refine
            proj_func = proj_func_refine
            # Interpolate best_c to this resolution
            # Just use the same coefficients (will be interpolated naturally)
            # Remains best_c from previous resolution
            
        elif n_grid == N_REFINE:
            break
            
    # Final refinement with projected gradient descent
    if best_c is None or time.time() - start_time > (budget_s - 0.5):
        # Use a simple fallback or return current best
        pass
    else:
        # Perform randomized gradient descent refinement
        c_refined = best_c.copy()
        h_refined = reconstruct_h(c_refined, Phi)
        
        # Ensure feasible
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = project_h_to_unit_integral(h_refined, n_grid)
        
        dx = 2.0 / n_grid
        corr = np.correlate(h_refined, 1-h_refined, mode='full')
        max_k = np.argmax(corr)
        C5_eval = corr[max_k] * dx
        
        # Simple gradient descent refinement step
        num_steps = 200
        step_size = 0.2
        for step in range(num_steps):
            if time.time() - start_time > budget_s - 0.5:
                break
                
            # Compute approximate gradient (at max_k indexed overlap)
            h = h_refined
            result_h = np.zeros_like(h)
            k_best = max_k - (len(h) - 1)  # Convert to shift index
            
            shift_back = np.roll(h, k_best)
            shift_front = np.roll(h, -k_best)
            
            # Approx gradient is approximated via derivative of correlation
            # Let's try a coarse one using finite difference-like approach on the 'best_k'
            # The energy difference can be viewed via finite difference of valid h.
            # This corresponds to: d(C5)/d(h_i) = (1 - h_{i+k}) - h_{i-k} 
            
            # where indices are periodic
            gradient_h = np.zeros_like(h)
            for i in range(len(h)):
                j_forward = (i + k_best) % len(h)
                j_backward = (i - k_best) % len(h)
                gradient_h[i] = (1 - h[j_forward]) - h[j_backward]
            
            # Project back via linearization of basis:
            # Reconstruct with small adjustment in coefficient space
            # gradient_c = Phi.T @ gradient_h
            gradient_c = Phi.T @ gradient_h
            
            c_refined -= step_size / (1.0 + step) * gradient_c
            c_refined = project_c(c_refined, Phi, proj_func)
            
            h_refined = reconstruct_h(c_refined, Phi)
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = project_h_to_unit_integral(h_refined, n_grid)
            
            corr = np.correlate(h_refined, 1-h_refined, mode='full')
            C5_eval = np.max(corr) * dx
            if C5_eval <= best_C5:
                best_C5 = C5_eval
                best_c = c_refined.copy()
    
    # Final verification
    h_final = reconstruct_h(best_c, Phi)
    h_final = np.clip(h_final, 0, 1)
    h_final = project_h_to_unit_integral(h_final, n_grid)
    dx_final = 2.0 / n_grid
    final_C5 = compute_C5(h_final, dx_final)
    
    return (h_final.tolist(), float(final_C5), n_grid)
