import numpy as np
import time
import random
from scipy.signal import correlate

# Global constants
N_POINTS_LIST = [30, 36, 42, 48, 60]
MAX_ITER_PER_STAGE = 150
TIME_MARGIN = 5

def correlation_vector(b):
    """Compute correlation vector R_k for binary sequence b."""
    # Full correlation of b with itself
    corr = correlate(b, b, mode='full')
    # Extract the relevant part (excluding shift 0)
    R = corr[len(b)-1:]
    return R

def update_correlation(R, i, j, b):
    """Update correlation vector R after swapping positions i and j."""
    # Precompute shifts that could be affected by the swap
    n = len(b)
    # For each k, update R_k as needed
    for k in range(1, n):
        # Check if i-k contributes to R_k
        if i - k >= 0 and b[i - k] == 1:
            R[k - 1] -= 1
        # Check if i+k contributes to R_k
        if i + k < n and b[i + k] == 1:
            R[k - 1] -= 1
        # Check if j-k contributes to R_k
        if j - k >= 0 and b[j - k] == 1:
            R[k - 1] += 1
        # Check if j+k contributes to R_k
        if j + k < n and b[j + k] == 1:
            R[k - 1] += 1

def hill_climb_binary(b, R, max_iter=1000):
    """Perform hill climbing on a binary sequence."""
    n = len(b)
    t = np.min(R)
    count = 0
    while count < max_iter:  # prevent infinite loop
        k_star = np.argmin(R) + 1  # 1-based index
        t = R[k_star - 1]
        best_d = -np.inf
        best_i, best_j = -1, -1
        # Iterate through all admissible swaps
        for i in range(n):
            if b[i] != 1:
                continue
            for j in range(n):
                if b[j] != 0:
                    continue
                # Compute delta for k_star
                d = 0
                if i - k_star >= 0 and b[i - k_star] == 1:
                    d -= 1
                if i + k_star < n and b[i + k_star] == 1:
                    d -= 1
                if j - k_star >= 0 and b[j - k_star] == 1:
                    d += 1
                if j + k_star < n and b[j + k_star] == 1:
                    d += 1
                # higher t for k_star is better
                if d > best_d:
                    best_d = d
                    best_i, best_j = i, j
        if best_d <= 0:
            break
        # Apply the swap
        b[best_i] = 0
        b[best_j] = 1
        update_correlation(R, best_i, best_j, b)
        count += 1
    return b, R

def evaluate_solution(b):
    """Evaluate solution b on the full range."""
    n = len(b)
    dx = 2.0 / n
    R = correlation_vector(b)
    t = np.min(R)
    c5 = 1 - dx * t
    return c5

def upsample_sequence(b, target_n):
    """Upsample a sequence to target size."""
    source_n = len(b)
    ratio = target_n // source_n
    if ratio > 0:
        expanded = np.repeat(b, ratio)
    else:
        expanded = b.copy()
    if len(expanded) < target_n:
        expanded = np.pad(expanded, (0, target_n - len(expanded)), 'constant', constant_values=(0, 0))
    elif len(expanded) > target_n:
        expanded = expanded[:target_n]
    # correct weight
    current_weight = np.sum(expanded)
    target_weight = target_n // 2
    diff = int(target_weight - current_weight)
    if diff > 0:
        # add diff ones
        zero_indices = np.where(expanded == 0)[0]
        for i in range(diff):
            expanded[zero_indices[i]] = 1
    elif diff < 0:
        # remove diff ones
        one_indices = np.where(expanded == 1)[0]
        for i in range(-diff):
            expanded[one_indices[i]] = 0
    return expanded

def crossover(p1, p2):
    """Single-point crossover of two binary sequences."""
    n = len(p1)
    point = random.randint(1, n - 1)
    child = p1[:point].copy()
    child = np.append(child, p2[point:])
    # Fix weight
    current_weight = np.sum(child)
    target_weight = n // 2
    diff = target_weight - current_weight
    if diff > 0:
        zero_indices = np.where(child == 0)[0]
        for i in range(diff):
            if i < len(zero_indices):
                child[zero_indices[i]] = 1
    elif diff < 0:
        one_indices = np.where(child == 1)[0]
        for i in range(-diff):
            if i < len(one_indices):
                child[one_indices[i]] = 0
    return child

def mutate(b, num_swaps=1):
    """Mutate by swapping 0's and 1's."""
    n = len(b)
    for _ in range(num_swaps):
        i = np.random.choice(np.where(b == 1)[0])
        j = np.random.choice(np.where(b == 0)[0])
        b[i], b[j] = b[j], b[i]
    return b

def initial_population(n, num_pop=50, elite=None):
    """Generate initial population."""
    population = []
    if elite is not None:
        population.append(elite.copy())
    # Generate random individuals
    for _ in range(num_pop - len(population)):
        indices = np.random.choice(n, size=n//2, replace=False)
        b = np.zeros(n, dtype=np.uint8)
        b[indices] = 1
        population.append(b)
    return population

def run(seed=42, budget_s=600):
    """Run the optimization procedure."""
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # tracked best
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    # Multi-scale optimization
    for n_idx, n_points in enumerate(N_POINTS_LIST):
        if time.time() - start_time + TIME_MARGIN > budget_s:
            break

        remaining_time = budget_s - (time.time() - start_time)

        # Stage specific time allocation
        gen_time_estimate = 0.01 * n_points  # Time estimate per generation
        if remaining_time < gen_time_estimate * 20:
            break  # Not enough time to proceed

        allocated_time = min(remaining_time - TIME_MARGIN, 120)  # Cap 120s per stage

        # Initialize population
        if n_idx == 0:  # Initial population from seed
            # Use initial_h_values or generate randomly
            if 'initial_h_values' in globals():
                h_init = np.array(initial_h_values)
                if len(h_init) != n_points:
                    h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)
                h_init = h_init > 0.5  # Map to binary
            else:
                # Generate random binary
                h_init = np.zeros(n_points, dtype=np.uint8)
                indices = np.random.choice(n_points, size=n_points//2, replace=False)
                h_init[indices] = 1
        else:
            # Use upsampling from previous best
            h_init = upsample_sequence(best_h, n_points)
            
        pop_size = max(30, 2 * n_points)
        population = initial_population(n_points, pop_size, h_init)

        # Evaluate all individuals
        fitness = []
        for b in population:
            R = correlation_vector(b)
            t = np.min(R)
            fitness.append(t)

        # Main evolutionary loop
        max_gen = int(allocated_time / (gen_time_estimate + 1e-6))
        gen_count = 0

        for g in range(max_gen):
            if time.time() - start_time + TIME_MARGIN > budget_s:
                break
            gen_count += 1

            # Tournament selection
            selected = []
            for _ in range(pop_size):
                tournament = random.sample(list(zip(population, fitness)), 3)
                best_in_tournament = max(tournament, key=lambda x: x[1])
                selected.append(best_in_tournament[0])

            # Create children
            children = []
            for i in range(0, pop_size, 2):
                p1, p2 = selected[i], selected[i+1] if i+1 < pop_size else selected[0]
                # Crossover
                if random.random() < 0.8:
                    child = crossover(p1, p2)
                else:
                    child = p1.copy()
                # Mutation
                child = mutate(child, num_swaps=1 + (n_points // 40))
                # Local improvement
                R = correlation_vector(child)
                child, R = hill_climb_binary(child, R)
                assert np.sum(child) == n_points // 2
                children.append(child)

            # Update population
            population = children
            fitness = [np.min(correlation_vector(b)) for b in population]

            # Update global best
            best_idx = np.argmax(fitness)  # higher fitness is better
            if fitness[best_idx] > np.min([f for f in fitness if f < float('inf')]):  # Safeguard against strange values
                pass
            if fitness[best_idx] > np.min(fitness):
                continue  # avoid suboptimal
            best_t = fitness[best_idx]
            best_b = population[best_idx]
            R_best = correlation_vector(best_b)
            current_c5 = 1 - (2.0 / n_points) * best_t
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = best_b.copy()
                best_n = n_points

        # Final polishing for this stage's best
        if time.time() - start_time + TIME_MARGIN > budget_s:
            break
        # Convert to float for compatibility with eval
        b_float = best_h.astype(float)
        c5_eval = evaluate_solution(best_h)

        if c5_eval < best_c5:
            # Reevaluate with improved accuracy
            best_c5 = c5_eval
            best_h = best_h.copy()
            best_n = n_points

    # Final polishing
    if best_h is None:
        # Fallback
        n_points = 60
        h_init = np.zeros(n_points, dtype=np.uint8)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h_init[indices] = 1
        R = correlation_vector(h_init)
        b_final, R = hill_climb_binary(h_init, R)
        best_h = b_final
        best_c5 = evaluate_solution(b_final)
        best_n = n_points

    # Convert best_h to float array for output
    h_values = best_h.astype(float)

    return (h_values, best_c5, best_n)
