import numpy as np
import time
from scipy.signal import correlate

def correlation_vector(b):
    """Compute correlation vector R_k for binary sequence b."""
    R = correlate(b, b, mode='full')
    return R[len(b)-1:]

def update_correlation(R, i, j, b):
    """Update correlation vector R after swapping b[i] and b[j]."""
    # For each shift k, we must update R_k according to changes
    n = len(b)
    for k in range(1, n):
        # Changing b[i] from 1 to 0
        if i - k >= 0 and b[i - k] == 1:
            R[k - 1] -= 1
        if i + k < n and b[i + k] == 1:
            R[k - 1] -= 1
        # Changing b[j] from 0 to 1
        if j - k >= 0 and b[j - k] == 1:
            R[k - 1] += 1
        if j + k < n and b[j + k] == 1:
            R[k - 1] += 1

def hill_climb(b, R, max_swaps=500, escape_prob=0.05):
    """
    Perform local hill climb on binary sequence b with correlation R.
    """
    n = len(b)
    best_b = b.copy()
    best_R = R.copy()
    best_t = np.min(best_R)
    improved = True
    n_swaps = 0
    
    while improved and n_swaps < max_swaps:
        improved = False
        min_shift = np.argmin(best_R)
        min_correlation = best_R[min_shift]
        
        # Try to improve correlation at critical shift
        best_delta = 0
        best_i, best_j = None, None
        best_changes = []

        # Enumerate all swaps that are likely to help improve critical shift
        positions_1 = np.where(best_b == 1)[0]
        positions_0 = np.where(best_b == 0)[0]
        candidates = list(np.ndindex((len(positions_1), len(positions_0))))
        for _ in range(min(len(candidates), 200)):
            idx_i, idx_j = np.random.choice(len(positions_1)), np.random.choice(len(positions_0))
            i, j = positions_1[idx_i], positions_0[idx_j]
            if i == j: continue
            
            # Simulate update of R_k for all shifts during swap
            temp_R = best_R.copy()
            temp_b = best_b.copy()
            temp_b[i], temp_b[j] = 0, 1
            
            # Update R efficiently for the relevant shifts
            prev_min_k = np.argmin(best_R)
            for k in range(1, n): 
                if i - k >= 0 and best_b[i - k] == 1:
                    temp_R[k - 1] -= 1
                if i + k < n and best_b[i + k] == 1:
                    temp_R[k - 1] -= 1
                if j - k >= 0 and best_b[j - k] == 1:
                    temp_R[k - 1] += 1
                if j + k < n and best_b[j + k] == 1:
                    temp_R[k - 1] += 1
            
            new_t = np.min(temp_R)
            new_delta = new_t - min_correlation
            if new_delta > best_delta or (new_delta == best_delta and np.random.rand() < escape_prob):
                best_delta = new_delta
                best_i = i
                best_j = j
                best_changes = temp_R
            # Restore temp arrays before next iteration

        if best_delta > 0:
            # Apply the best swap
            best_b[best_i], best_b[best_j] = 0, 1
            update_correlation(best_R, best_i, best_j, best_b)
            best_t = np.min(best_R)
            improved = True
            n_swaps += 1

    return best_b, best_R, best_t

def crossover(p1, p2):
    """One-point binary crossover with correction to preserve weight."""
    n = len(p1)
    mid = np.random.randint(1, n)
    child = (p1[:mid] + p2[mid:]).astype(np.uint8)
    
    # Ensure child has exactly n//2 ones
    ones_count = np.sum(child)
    if ones_count > n // 2:
        zeros = np.where(child == 0)[0]
        for _ in range(ones_count - n // 2):
            idx = np.random.choice(zeros)
            child[idx] = 1
            zeros = np.delete(zeros, np.where(zeros == idx))
    elif ones_count < n // 2:
        ones = np.where(child == 1)[0]
        for _ in range(n // 2 - ones_count):
            idx = np.random.choice(ones)
            child[idx] = 0
            ones = np.delete(ones, np.where(ones == idx))
    
    return child

def mutate(individual, num_swaps=1):
    """Perform num_swaps random bit flips."""
    n = len(individual)
    zero_positions = np.where(individual == 0)[0]
    one_positions = np.where(individual == 1)[0]
    
    for _ in range(num_swaps):
        if len(zero_positions) > 0 and len(one_positions) > 0:
            idx_zero = np.random.choice(zero_positions)
            idx_one = np.random.choice(one_positions)
            individual[idx_zero] = 1
            individual[idx_one] = 0
            zero_positions = np.where(individual == 0)[0]
            one_positions = np.where(individual == 1)[0]
            
    return individual

def binary_to_float(b):
    """Convert binary sequence of length n to a floating point array of length n."""
    return b.astype(float)
