def project_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum and bounds [0,1]."""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    
    # Sort indices by h values in descending order
    sorted_indices = np.argsort(h)[::-1]
    free_set = set(sorted_indices)
    
    # Perform the projection
    while abs(diff) > 1e-12 and len(free_set) > 0:
        # Compute the size of the free set and the value of the smallest entry in it
        free_list = list(free_set)
        free_values = h[free_list]
        sorted_free_indices = np.argsort(free_values)[::-1]
        
        # Check which entries can still be increased
        delta = diff / len(free_list)
        for idx in sorted_free_indices:
            i = free_list[idx]
            new_val = h[i] + delta
            if new_val >= 1:
                # This component will hit its upper bound
                diff -= (1 - h[i])
                h[i] = 1
                free_set.remove(i)
            elif new_val <= 0:
                # This component will hit its lower bound
                diff -= (0 - h[i])
                h[i] = 0
                free_set.remove(i)
            else:
                h[i] = new_val
                diff = 0
    
    return h

def soft_min_alpha(h, alpha=1.0):
    """Computes the soft minimum of the correlations with temperature parameter alpha."""
    n = len(h)
    padded = np.pad(h, (0, n), mode='constant')
    H = fft(padded)
    r_vec = np.real(ifft(H * np.conj(H)))[:n]
    r = r_vec[1:]  # k = 1,...,n-1
    
    # Soft minimum via log-sum-exp trick
    exp_neg_alpha_r = np.exp(-alpha * r)
    log_sum = np.log(np.sum(exp_neg_alpha_r))
    return - log_sum / alpha

def compute_correlations(h):
    """Compute the vector of correlations r_k = sum h[i] * h[i+k] for k=1..n-1."""
    n = len(h)
    padded = np.pad(h, (0, n), mode='constant')
    H = fft(padded)
    r_vec = np.real(ifft(H * np.conj(H)))[:n]
    return r_vec[1:]

def gradient_soft_min_alpha(h, alpha=1.0):
    """Computes the gradient of the soft minimum w.r.t. h."""
    n = len(h)
    r = compute_correlations(h)  # r_k for k=1..n-1
    
    # Weights (probabilities)
    exp_neg_alpha_r = np.exp(-alpha * r)
    normalizer = np.sum(exp_neg_alpha_r)
    w = exp_neg_alpha_r / normalizer

    # Weights for the convolution
    # convolve h with w and reverse(w)
    # Loop through each element and compute the weighted sum of shifted values
    grad = np.zeros(n)

    # Compute gradient in terms of convolutions
    w_rev = w[::-1]
    # Pad w_rev and h
    padded_h = np.pad(h, (0, n), mode='constant')
    padded_w = np.pad(w_rev, (0, n), mode='constant')

    # Inverse FFT of w_rev * h. Then constrain to leading n entries
    H = fft(padded_h)
    W = fft(padded_w)
    conv1 = ifft(W * H)

    # Repeat for w * h
    padded_w = np.pad(w, (0, n), mode='constant')
    W = fft(padded_w)
    conv2 = ifft(W * H)

    sum_conv = np.real(conv1 + conv2)
    gradient = sum_conv[:n]  # Take first n

    return gradient

def nesterov_update(h_prev, h_cur, beta=0.9):
    """Perform the Nesterov update: y = h + β(h - h_prev)."""
    return h_cur + beta * (h_cur - h_prev)

def line_search(f, grad_f, x, eta_max=1.0):
    """Perform backtracking line search: find step size that satisfies Armijo condition."""
    eta = eta_max
    c1 = 1e-4
    if isinstance(f(x), float):
        f_x = f(x)
    else:
        f_x = f(x)[0]  # Should be the function value as scalar

    g_x = grad_f(x)
    while True:
        x_new = x - eta * g_x
        if isinstance(f(x_new), float):
            f_new = f(x_new)
        else:
            f_new = f(x_new)[0]
        
        if f_new <= f_x - c1 * eta * np.dot(g_x, g_x):
            break
        eta *= 0.5
        if eta < 1e-6:
            break
    return eta

def homotopy_ascent(h0, n_iter=1000, alpha_init=0.5, alpha_max=80, factor=2, beta=0.9):
    """Run projected gradient ascent with increasing alpha (homotopy)."""
    h_prev = h0.copy()
    h_cur = h0.copy()
    
    alpha = alpha_init
    
    best_h = h_cur.copy()
    best_obj = soft_min_alpha(h_cur, alpha)  # maximize this
    
    # Track the last few objective values for early stopping
    obj_history = []
    
    for iter_num in range(n_iter):
        # Get current gradient
        g = gradient_soft_min_alpha(h_cur, alpha)
        
        # Nesterov update
        y = nesterov_update(h_prev, h_cur, beta=beta)
        
        # Line search to find step size
        eta = line_search(lambda x: soft_min_alpha(x, alpha), 
                          lambda x: gradient_soft_min_alpha(x, alpha), y)
        
        # Update
        h_new = y - eta * g
        h_new = project_simplex(h_new, len(h_new)/2)
        
        h_prev = h_cur
        h_cur = h_new 
        
        # Track best solution
        obj_val = soft_min_alpha(h_cur, alpha)
        if obj_val > best_obj:
            best_obj = obj_val
            best_h = h_cur.copy()
            
        # Early stopping if improvement is small
        obj_history.append(obj_val)
        if len(obj_history) > 20:
            obj_history.pop(0)
            if abs(obj_history[-1] - obj_history[0]) < 1e-9:
                break

        # Update alpha if not too high
        if alpha < alpha_max:
            alpha = min(alpha * factor, alpha_max)

    return best_h, best_obj, alpha

def refine_binary(h_frac, n):
    """Refine fractional solution into binary form with local swaps."""
    # Get correlation vector
    r = compute_correlations(h_frac)
    c5_candidates = 1 - (r * 2.0 / n)
    
    # Greedy rounding to binary
    # Set top n/2 to 1
    idxs = np.argsort(h_frac)[::-1]
    h_bin = np.zeros(n)
    for i in range(n // 2):
        h_bin[idxs[i]] = 1
    h_bin = project_simplex(h_bin, n / 2)
    
    best_h = h_bin.copy()
    # Compute current correlations
    r = compute_correlations(h_bin)
    best_c5 = 1 - np.min(r) * 2.0 / n  # reverse the normalizing
    
    # Local swap search to improve
    # Enumerate the worst k
    k_star = np.argmin(r) + 1
    
    # Modify r to not include k_star directly
    r_modified = r.copy()
    r_modified[k_star - 1] = np.inf
    
    # Find the critical shifts and their ranges
    critical_shift = np.argmin(r) + 1
    # Let's just do 500 random swaps in nearby range
    for _ in range(200):
        # Pick two indices
        one_indices = np.argwhere(h_bin > 0.5).flatten()
        zero_indices = np.argwhere(h_bin < 0.5).flatten()
        if len(one_indices) == 0 or len(zero_indices) == 0:
            break
        i = np.random.choice(one_indices)
        j = np.random.choice(zero_indices)
        
        # See if swapping improves C5
        # For shift k*, compute Δ in r_{k*}
        # We need to compute the change accurately
        # r_{k*}[i] += h_bin[i] * h_bin[i + k*]
        # r_{k*}[j] += h_bin[j] * h_bin[j + k*]
        # but be careful with the bounds
        
        old_i = r[k_star - 1]  # The old r_{k*}
        old_j = r[k_star - 1]  
        
        # Estimate value of old_i and old_j (they're the contributions of i and j to k*)
        # r_k = sum_{i=0}^{n-k} h[i]*h[i+k]
        # So contribution of h[i] to r_k is h[i+k]  if k = k*
        # So if we swap h[i] with h[j], then contributions change:
        # h[i] now contributes h[i+k*] (if exists) instead of h[i+k*] (before)
        # h[j] now contributes h[j+k*] instead of h[j+k*] (before)
        # Actually, better:
        # When i goes from 1 to 0, it adds -h[i+k*] to r_{k*} if valid
        # When j goes from 0 to 1, it adds h[j+k*] to r_{k*}
        # Change = (h[j+k*] - h[i+k*]) for k* = k_star
        # But we must ensure valid indices
        if j + k_star < n:
            diff1 = h_bin[j+k_star]
        else:
            diff1 = 0
        if i + k_star < n:
            diff2 = h_bin[i+k_star]
        else:
            diff2 = 0
        delta_r_k = diff1 - diff2  # implied contribution to r_k*

        # We actually want to increase r_k* (lower objective)
        # So we're increasing the worst correlation, which might not improve
        # Let's instead compute the change in C5 directly
        # C5 = max_k integral terms = max_k (sum h[i](1-h[i+k]))
        # Which we optimize by minimizing the minimum of r_k 
        # If we swap h[i] and h[j]
        
        r_old = r.copy()
        r_old[k_star - 1] += delta_r_k  # Update r_{k*}
        
        # New C5
        new_c5 = 1 - np.min(r_old) * 2.0 / n
        if new_c5 < best_c5:
            # Actually swap
            h_bin[i] = 0
            h_bin[j] = 1
            h_bin = project_simplex(h_bin, n / 2)
            best_c5 = new_c5
            r = compute_correlations(h_bin)
            
    return h_bin, best_c5

        # Deterministic initial point

            if c5_val < best_solution:

        # Early stopping if target is achieved
