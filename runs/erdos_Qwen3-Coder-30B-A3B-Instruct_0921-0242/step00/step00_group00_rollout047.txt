import numpy as np
import cvxpy as cp
from scipy.sparse import diags
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose grid sizes: coarse, medium, fine
    grid_sizes = [120, 200, 300]
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Set timing limits
    sdp_budget = 0.6 * budget_s
    polish_budget = 0.3 * budget_s
    safety_margin = 0.05 * budget_s
    
    def project_to_simplex(v, target_sum):
        """Projection of v onto the capped simplex with sum = target_sum"""
        v = np.clip(v, 0, 1)
        v_sorted = np.sort(v)[::-1]
        tmp_sum = 0
        for i, val in enumerate(v_sorted):
            tmp_sum += val
            if tmp_sum <= target_sum:
                continue
            else:
                threshold = (tmp_sum - target_sum) / (i + 1)
                break
        else:
            threshold = 0
        return np.clip(v - threshold, 0, 1)
    
    def compute_c5(h, dx):
        """Compute C5 using convolution."""
        h = np.array(h)
        conv = np.correlate(h, 1 - h, mode='full')
        max_k = len(conv) // 2
        conv = conv[max_k:] * dx
        return np.max(conv)
    
    def construct_shift_matrices(n):
        """Construct sparse shift matrices as a list of scipy matrices."""
        shift_matrices = []
        for k in range(n):
            if k == 0:
                shift_matrices.append(diags([1.0]*n, 0, shape=(n, n)))
            else:
                # For k > 0, shift to the right
                diag = [1.0] * (n - k)
                shift_matrices.append(diags(diag, k, shape=(n, n)))
        return shift_matrices
    
    def solve_sdp(n, shift_matrices, max_iters=4000, eps=5e-4):
        """Solve SDP relaxation and return primal vector."""
        dx = 2.0 / n
        L = cp.Variable()
        X = cp.Variable((n, n), symmetric=True)
        h = cp.Variable(n)
        
        # Objective: maximize L
        objective = cp.Maximize(L)
        
        # Constraints
        constraints = [
            X >> 0,  # PSD
            h >= 0,
            h <= 1,
            cp.sum(h) == n / 2,
            cp.diag(X) == h,
        ]
        
        # Add bounds on X entries
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append(X[i, j] <= h[i])
                constraints.append(X[i, j] <= h[j])
                constraints.append(X[i, j] >= 0)
        
        # Add correlation constraints
        for k in range(n):
            if k == 0:
                constraints.append(cp.trace(shift_matrices[k] @ X) >= L)
            else:
                # For non-zero shifts, there are overlaps to compute
                constraints.append(cp.trace(shift_matrices[k] @ X) >= L)
        
        # Formulate and solve SDP
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=max_iters, eps=eps)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                X_sol = X.value
                return h_sol, X_sol
            else:
                print("SDP status:", prob.status)
                return None, None
        except Exception as e:
            print(f"SDP solve error: {e}")
            return None, None
    
    def gradient_polish(h_start, shift_matrices, dx, max_iter=500):
        """Local gradient descent polishing with projected updates."""
        h = np.array(h_start.copy())
        for iter in range(max_iter):
            # Find current maximum overlap index
            conv = np.correlate(h, 1 - h, mode='full')
            max_k = len(conv) // 2
            conv = conv[max_k:] * dx
            k_star = np.argmax(conv)
            
            # Compute gradient
            g = np.zeros_like(h)
            n = len(h)
            if k_star == 0:
                for i in range(n):
                    g[i] = 1 - h[i]
            else:
                # Build gradient exact as per maximization objective
                # Overlap for shift k_star is sum h_i * h_{i+k_star}
                # Gradient d/dh_i = (1 - h_{i+k_star}) if valid, 0 otherwise
                for i in range(n):
                    j = i + k_star
                    if j < n:
                        g[i] = 1 - h[j]
            
            # Update (with diminishing stepsize)
            alpha = 1.0 / np.sqrt(iter + 1)
            h = h - alpha * g
            
            # Project back to feasible set
            h = project_to_simplex(h, n/2)
        return h
    
    # Multi-resolution procedure
    current_h = None
    for n in grid_sizes:
        if time.time() - start_time > budget_s - safety_margin:
            break
            
        shift_matrices = construct_shift_matrices(n)
        
        # Warm-start from previous solution if available
        if current_h is not None:
            h_try = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(current_h)), current_h)
            # Round to improve numerical properties
            h_try = np.clip(h_try, 0, 1)
        else:
            # Random initial guess
            h_try = np.random.rand(n)
            h_try = project_to_simplex(h_try, n/2)
        
        # Solve SDP on this grid
        h_sol, X_sol = solve_sdp(n, shift_matrices)
        if h_sol is None:
            continue
            
        # Rounding via eigen-decomposition
        try:
            _, V = np.linalg.eigh(X_sol)
            v = V[:, -1]  # Take largest eigenvalue
            hat_h = np.abs(v) * np.sqrt(np.maximum(0, np.linalg.eigvals(X_sol)[-1]))
            hat_h = np.clip(hat_h, 0, 1)
            # Project to feasible simplex
            h_feas = project_to_simplex(hat_h, n / 2)
        except:
            h_feas = h_try
        
        # Local polishing
        dx = 2.0 / n
        h_polished = gradient_polish(h_feas, shift_matrices, dx, max_iter=min(500, int(100 + polish_budget/100)))
        c5_val = compute_c5(h_polished, dx)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_polished.copy()
            best_n = n
        
        current_h = h_polished.copy()
