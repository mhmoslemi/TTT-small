import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def decode(x, M):
    """
    Decodes a flattened parameter vector x into segment lengths u and heights a.
    x = (u0, ..., u_{M-1}, a0, v0, ..., v_{M-2})
    """
    u = np.array(x[:M])
    a0 = x[M]
    v = np.array(x[M+1:])
    # Compute heights
    a = np.zeros(M+1)
    a[0] = a0
    for j in range(1, M):
        a[j] = a[j-1] - v[j-1]
        if a[j] < 0:
            a[j] = 0
    a[M] = max(0, a[M-1] - v[M-2] if M > 1 else a0)
    # End of segment
    s = np.concatenate([[0], np.cumsum(u)])
    s = np.append(s, 2.0)
    return s, a

def overlap_piecewise(s, a):
    """Computes exact maximum overlap for piecewise-constant h."""
    M = len(a) - 1
    # Compute all breakpoints from all pairs of segment endpoints
    breakpoints = set()
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0 <= k1 <= 2:
                breakpoints.add(k1)
            if 0 <= k2 <= 2:
                breakpoints.add(k2)
    K = sorted(breakpoints)
    K = [0] + K + [2]
    max_val = 0
    for k in K:
        val = 0
        for j in range(M+1):
            for l in range(M+1):
                x1 = max(s[j], s[l] - k)
                x2 = min(s[j+1], s[l+1] - k)
                if x2 > x1:
                    length = x2 - x1
                    val += a[j] * (1 - a[l]) * length
        max_val = max(max_val, val)
    return max_val

def project_capped_simplex(v, total):
    """Projects vector v onto capped simplex: sum = total, 0 <= v_i <= 1."""
    eps = 1e-12
    # Clip to [0,1]
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - total) < eps:
        return v
    # Simple projection: project onto sum = total
    delta = (s - total) / len(v)
    v -= delta
    # Ensure all are in [0,1]
    v = np.clip(v, 0, 1)
    return v

def fitness_func(x, M, n_points, target_integral=1.0, penalty_factor=1e6):
    s, a = decode(x, M)
    # Reconstruct a to be valid (constraints may have been violated)
    a = np.array([max(0, min(1, a[i])) for i in range(len(a))])
    u = s[1:] - s[:-1]
    
    # Check integral
    integral_error = np.sum(u * a[:-1]) - target_integral
    if abs(integral_error) > 1e-8:
        penalty = penalty_factor * integral_error**2
    else:
        penalty = 0
    
    # Check bounds
    bounds_penalty = 0
    if np.any(a < 0) or np.any(a > 1):
        bounds_penalty += 1e6
    if np.any(u < 0) or np.any(u > 2):
        bounds_penalty += 1e6
        
    # Skip if invalid
    if bounds_penalty > 0:
        return penalty + bounds_penalty
        
    # Evaluate overlap
    max_overlap = overlap_piecewise(s, a)
    return max_overlap + penalty + bounds_penalty

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    M = 4
    n_points = 128
    dx = 2.0 / n_points
    
    # Phase A: DE with coarse model M=4
    print(f"Starting Phase A: DE with M={M}, n={n_points}")
    
    def de_fitness(x):
        return fitness_func(x, M, n_points)
    
    bounds = [(0, 2)] * M + [(0, 1)] + [(0, 1)] * (M - 1)  # bounds for initial guess
    res = differential_evolution(de_fitness, bounds, seed=seed, maxiter=200, popsize=15*len(bounds), strategy='best1bin', recombination=0.9)
    
    best_x_A = res.x
    best_value_A = res.fun
    print(f"Phase A finished, best value: {best_value_A}")
    
    # Phase B: DE with finer model M=6
    print("Starting Phase B: DE with M=6")
    
    M = 6
    n_points = 256
    dx = 2.0 / n_points
    
    # Recover candidate from A
    s_A, a_A = decode(best_x_A, 4)
    
    # Generate start population near A's solution
    popsize = 15 * (M + M - 1 + 1)  # M lengths, 1 height, M-1 drops
    bounds = [(0, 2)] * M + [(0, 1)] + [(0, 1)] * (M - 1)
    initial_pop = [res.x]  # Use previous solution to start
    for _ in range(popsize - 1):
        # Small variation around current best
        x = np.array(res.x)
        # Modify parameters randomly
        for i in range(len(x)):
            if np.random.rand() < 0.3:  # Randomly modify 30% of parameters
                delta = np.random.normal(0, 0.1)
                x[i] += delta
                x[i] = np.clip(x[i], bounds[i][0], bounds[i][1])
        initial_pop.append(x)
    
    def de_fitness_new(x):
        return fitness_func(x, M, n_points)
    
    res2 = differential_evolution(
        de_fitness_new, bounds,
        seed=seed * 2, maxiter=200,
        popsize=popsize, strategy='best1bin', recombination=0.9,
        initial=initial_pop
    )
    
    best_x_B = res2.x
    best_value_B = res2.fun
    print(f"Phase B finished, best value: {best_value_B}")
    
    # Phase C: Local refinement with SLSQP and subgradient
    print("Starting Phase C: Local refinement")
    
    def smooth_objective(x):
        s, a = decode(x, M)
        a = np.array([max(0, min(1, a[i])) for i in range(len(a))])
        u = s[1:] - s[:-1]
        integral_error = np.sum(u * a[:-1]) - 1.0
        penalty = 1e6 * integral_error**2
        
        # Use log-sum-exp to approximate max
        beta = 50
        K_set = set()
        for j in range(M+1):
            for l in range(M+1):
                k1 = s[l] - s[j+1]
                k2 = s[l+1] - s[j]
                if 0 <= k1 <= 2:
                    K_set.add(k1)
                if 0 <= k2 <= 2:
                    K_set.add(k2)
        K = sorted(K_set)
        K = [0] + K + [2]
        log_vals = []
        for k in K:
            val = 0
            for j in range(M+1):
                for l in range(M+1):
                    x1 = max(s[j], s[l] - k)
                    x2 = min(s[j+1], s[l+1] - k)
                    if x2 > x1:
                        length = x2 - x1
                        val += a[j] * (1 - a[l]) * length
            log_vals.append(val)
        smooth_max = (1/beta) * np.log(np.sum(np.exp(beta * np.array(log_vals))))
        return smooth_max + penalty
    
    # Optimize with smooth objective
    res_smooth = minimize(smooth_objective, res2.x, method='SLSQP', bounds=bounds, 
                          constraints={'type': 'eq', 'fun': lambda x: np.sum(decode(x, M)[0][1:] * decode(x, M)[1][:-1]) - 1.0},
                          options={'maxiter': 200, 'ftol': 1e-8})
    
    best_x_C = res_smooth.x
    print(f"Phase C SLSQP finished, final value: {res_smooth.fun}")
    
    # Now step 1: build piecewise function
    s, a = decode(best_x_C, M)
    a = np.array([max(0, min(1, a[i])) for i in range(len(a))])
    
    # discrete grid
    h_values = np.zeros(n_points)
    for i in range(n_points):
        x = (i + 0.5) * dx
        # find the segment containing x
        for j in range(len(s) - 1):
            if s[j] <= x < s[j+1]:
                h_values[i] = a[j]
                break
        else:
            # if x == 2, assign last segment
            h_values[i] = a[-1]
    
    # Project to exact integral
    total_sum = np.sum(h_values) * dx
    if abs(total_sum - 1.0) > 1e-8:
        h_values = project_capped_simplex(h_values, 1.0)
        
    # Compute final evaluation
    corr = np.convolve(h_values, 1 - h_values, mode='full') * dx
    c5_bound = np.max(corr)
    
    return (h_values.tolist(), float(c5_bound), n_points)
