import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def c5_of_h(h):
    """Compute C5 for a given step function h."""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution
    conv = correlate(h, g, mode='full')
    # The maximum is over all shifts from -n+1 to n-1
    # Shifts in real space: k = (idx - (n-1)) * dx
    # We want the maximum over all valid shifts
    max_val = np.max(conv)
    return max_val * dx

def gradient_of_c5(h):
    """Compute the gradient w.r.t. h for the active shift."""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = correlate(h, g, mode='full')
    idx_max = np.argmax(conv)
    n = len(h)
    k_opt = idx_max - (n - 1)  # corresponding shift in real space
    
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # The gradient for h[i] is dx * (1 - h[i+k_opt]) - dx * h[i-k_opt]
        # We treat out-of-range indices as 0 (zero-padding)
        idx1 = i + k_opt
        idx2 = i - k_opt
        term1 = 0 if idx1 < 0 or idx1 >= n else g[idx1]
        term2 = 0 if idx2 < 0 or idx2 >= n else h[idx2]
        grad[i] = dx * (term1 - term2)
    return grad * dx

def project_to_constraints(h, n_points):
    """Adjust h to satisfy the integral constraint exactly."""
    sum_h = np.sum(h)
    target = n_points / 2.0
    diff = target - sum_h
    
    if abs(diff) < 1e-12:
        return h
    
    # Distribute the difference uniformly among internal points that aren't at bounds
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    diff = target - sum_h
    
    # Distribute the error uniformly
    if abs(diff) > 0:
        # Uniform distribution of error across the vector
        delta = diff / n_points
        h += delta
        
    # Re-clip to maintain bounds in case of overshoot
    h = np.clip(h, 0.0, 1.0)
    
    # Ensure the sum is accurate (exact)
    sum_h = np.sum(h)
    diff = target - sum_h
    if abs(diff) > 1e-10:
        # Apply correction to one index to get the exact sum
        idx = np.argmax(h) if diff > 0 else np.argmin(h)
        h[idx] += diff
    
    return h

            break

            return c5_of_h(h)
            
        def jac(h):
            try:
                return gradient_of_c5(h)
            except Exception:
                # Fallback to finite difference if gradient fails
                return None

            return np.sum(h) - n_points / 2

    return (best_h, best_c5, n_points)
