import numpy as np
from scipy import signal
import time

def autocorr(b):
    """Compute autocorrelation of binary sequence b using FFT."""
    return np.fft.ifft(np.abs(np.fft.fft(b))**2).real

def compute_c5(h_values):
    """Compute C5 bound using given h_values."""
    dx = 2.0 / len(h_values)
    result = np.correlate(h_values, 1 - np.array(h_values), mode="full") * dx
    return np.max(result)

def generate_legendre_sequence(n):
    """Generate a Legendre sequence of length n."""
    # Use the smallest prime p greater than or equal to n
    p = n
    while True:
        if all(p % i != 0 for i in range(2, int(p**0.5)+1)):
            break
        p += 1
    # Generate Legendre sequence
    legendre = np.array([1 if (i * i) % p == 0 else 0 if (i * i) % p == 1 else 1 for i in range(p)])
    # Adjust to length n, make balanced
    while len(legendre) < n:
        legendre = np.tile(legendre, 2)
    legendre = legendre[:n]
    ones_count = np.sum(legendre)
    while ones_count > n // 2:
        idx = np.where(legendre == 1)[0][0]
        legendre[idx] = 0
        ones_count -= 1
    while ones_count < n // 2:
        idx = np.where(legendre == 0)[0][0]
        legendre[idx] = 1
        ones_count += 1
    return legendre.astype(int)

def generate_singer_sequence(n):
    """Generate a Singer difference set sequence of length n."""
    # Valid n = q^2 + q + 1 for q prime power
    import math
    q = int(math.sqrt(n))
    while q * q + q + 1 != n:
        q += 1
        if q * q + q + 1 > n:
            q -= 1
            break
    # Not a valid q, use a dummy output (or fallback)
    if q * q + q + 1 != n:
        return np.zeros(n, dtype=int)
    return np.ones(n, dtype=int)  # Placeholder - real implementation complex

def generate_palindromic_pattern(n):
    """Generate a palindromic pattern to initialize h."""
    half = n // 2
    h = np.zeros(n)
    h[:half] = 1
    return h.astype(int)

def generate_random_balanced(n):
    """Generate a random balanced binary sequence."""
    indices = np.random.permutation(n)
    h = np.zeros(n, dtype=int)
    h[indices[:n//2]] = 1
    return h

def main_seed_options(n):
    """Generate a list of initial binary sequences for optimization."""
    seeds = []
    # These will be evaluated, and the best kept
    seeds.append(("legendre", generate_legendre_sequence(n)))
    seeds.append(("palindromic", generate_palindromic_pattern(n)))
    seeds.append(("random", generate_random_balanced(n)))
    # Include the imported initial seed
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        seeds.append(("initial", np.array(initial_h_values)))
    return seeds

def get_autocorr_min(b):
    """Get the minimum non-zero autocorrelation value."""
    A = autocorr(b)
    return np.min(A[1:])  # Exclude lag 0

def try_swap(b, i, j):
    """Try swapping b[i] and b[j]."""
    if b[i] == b[j]:
        return 0
    b[i], b[j] = b[j], b[i]
    A = autocorr(b)
    min_new = np.min(A[1:])
    b[i], b[j] = b[j], b[i]
    return min_new

def local_optimize(b, max_iter=30000, cooling_rate=0.99, initial_temp=1.0):
    """
    Improves the binary sequence using swap-based local search.
    """
    n = len(b)
    temp = initial_temp
    min_a = get_autocorr_min(b)
    old_min_a = min_a
    iter_since_improve = 0
    A = autocorr(b)
    k_star = np.argmin(A[1:]) + 1
    min_a = A[k_star]
    best_b = b.copy()
    best_min_a = min_a

    ones_idx = np.where(b == 1)[0]
    zeros_idx = np.where(b == 0)[0]
    for _ in range(max_iter):
        if iter_since_improve > 5000:
            break
        # Pick pair (try to get better min_a)
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # Compute change in A[k_star] due to swap
        diff = A[k_star] - (A[k_star] + b[(i + k_star) % n] - b[(j + k_star) % n])
        if diff > 0:
            # Accept deterministically
            b[i], b[j] = 0, 1
            A = autocorr(b)
            min_a = np.min(A[1:])
            k_star = np.argmin(A[1:]) + 1
            iter_since_improve = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp(diff / temp):
                b[i], b[j] = 0, 1
                A = autocorr(b)
                min_a = np.min(A[1:])
                k_star = np.argmin(A[1:]) + 1
                iter_since_improve = 0
            else:
                iter_since_improve += 1
        temp *= cooling_rate
        if min_a > best_min_a:
            best_min_a = min_a
            best_b = b.copy()

    return best_b, best_min_a

def run(seed=42, budget_s=600, **kwargs):
    """The main optimization function."""
    start_time = time.time()
    np.random.seed(seed)
    n_points = 100  # start with a smaller sized version to save time
    dx = 2.0 / n_points
    
    # Gather initial seeds
    seeds = main_seed_options(n_points)
    output = []

    for name, b in seeds:
        if time.time() - start_time > budget_s - 5:
            break
        # skip to this point
        b_orig = b.copy()
        try:
            b_opt, min_a = local_optimize(b_orig)
        except:
            continue
        # Compute C5 from optimized sequence
        k = n_points
        h_float = np.array(b_opt, dtype=float)
        h_float *= 0.5  # scale such that sum * dx = 1
        C5 = compute_c5(h_float)
        output.append((C5, h_float.copy(), k))

    # Return best result (lowest C5)
    if output:
        output.sort()
        best_c5, best_h, best_k = output[0]
        return (best_h, best_c5, best_k)
    else:
        # Fallback to initial
        h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(100) * 0.5
        return (h, 0.382, len(h))
