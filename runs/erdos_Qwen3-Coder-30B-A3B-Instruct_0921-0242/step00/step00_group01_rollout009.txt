import numpy as np
import time
from scipy import fftpack
from scipy.optimize import minimize

def project_box_affine(h, target_sum):
    """Project h onto box [0,1] and affine constraint sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    h = h - (h.sum() - target_sum) / len(h)
    h = np.clip(h, 0, 1)
    return h

def autocorr(h):
    """Circular autocorrelation using FFT"""
    H = fftpack.fft(h)
    autocorr_values = fftpack.ifft(H * np.conj(H)).real
    return autocorr_values

def compute_c5(h, dx):
    """Compute C5 using full convolution"""
    h_full = np.concatenate([h, np.zeros_like(h)])
    h_rev = np.flip(h)
    conv = np.convolve(h_full, h_rev, mode='full')
    conv = conv[len(h)-1:2*len(h)-1]
    # conv[i] = sum h[j] * h[j+i] (circular)
    overlap = dx * (len(h) * 0.5 - conv)
    return np.max(overlap)

def initialize_candidates(n_points, initial_h_values, seed=42):
    np.random.seed(seed)
    candidates = []
    dx = 2.0 / n_points
    
    # Interpolate initial solution
    h_init = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, len(initial_h_values)), 
                       initial_h_values)
    candidates.append(project_box_affine(h_init, n_points / 2))
    
    # Random binary patterns with smoothing
    h_rand = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h_rand[indices] = 1.0
    h_smooth = np.convolve(h_rand, np.ones(5)/5, mode='same')
    candidates.append(project_box_affine(h_smooth, n_points / 2))
    
    # Alternating block
    h_block = np.zeros(n_points)
    mid = n_points // 2
    h_block[:mid] = 1.0
    candidates.append(project_box_affine(h_block, n_points / 2))
    
    return candidates

def single_gradient_step(h, dx, step_size):
    h = h.copy()
    aut = autocorr(h)
    lag = np.argmin(aut)
    #grad = -(np.roll(h, lag) + np.roll(h, -lag))
    grad = np.zeros_like(h)
    grad = grad - np.roll(h, lag) - np.roll(h, -lag)
    
    h_new = h + step_size * grad
    h = project_box_affine(h_new, len(h) / 2)
    return h

def optimize_one_candidate(h_start, dx, iter_max=1000, step_schedule=None):
    h = h_start.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    if step_schedule is None:
        step_schedule = [0.1 * (0.95 ** i) for i in range(iter_max)]
    
    for i, step_size in enumerate(step_schedule):
        if i >= iter_max:
            break
        h = single_gradient_step(h, dx, step_size)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def block_search(n_points, dx, best_so_far):
    # Try two-blocks: [0,a] = 1, [a,b] = c, [b,2] = 0
    best_c5 = best_so_far
    best_h = None
    step = 0.02
    for a in np.arange(dx, 2-dx, step):
        for b in np.arange(a+dx, 2-dx, step):
            L1 = a
            L2 = b - a
            L3 = 2 - b
            if L2 <= 0:
                continue
            c = (1 - L1) / L2
            if c < 0 or c > 1:
                continue
            
            h_candidate = np.zeros(n_points)
            indices1 = np.arange(0, int(np.round(a/dx)), dtype=int)
            indices2 = np.arange(indices1[-1]+1, int(np.round(b/dx)), dtype=int)
            if len(indices1) > 0:
                h_candidate[indices1] = 1.0
            if len(indices2) > 0:
                h_candidate[indices2] = c
            h_candidate = project_box_affine(h_candidate, n_points / 2)
            c5 = compute_c5(h_candidate, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = 1e10
    best_h = None
    n_points = 200
    dx = 2.0 / n_points
    
    candidates = initialize_candidates(n_points, initial_h_values, seed)
    candidates = [project_box_affine(h, n_points / 2) for h in candidates]
    
    print("Starting local search")
    
    for i, h_start in enumerate(candidates):
        if time.time() - start_time > budget_s - 5:
            break
        h_opt, c5_opt = optimize_one_candidate(h_start, dx, iter_max=2000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    print(f"Best after local search: {best_c5}")
    
    # Block search
    if time.time() - start_time < budget_s - 10:
        h_block, c5_block = block_search(n_points, dx, best_c5)
        if h_block is not None and c5_block < best_c5:
            best_c5 = c5_block
            best_h = h_block.copy()
        print(f"Best after block search: {best_c5}")
    
    if best_h is None:
        best_h = initial_h_values
        best_h = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, len(initial_h_values)), 
                           initial_h_values)
        best_h = project_box_affine(best_h, n_points / 2)
        best_c5 = compute_c5(best_h, dx)
    
    # Final refinement with targeting
    if time.time() - start_time < budget_s - 10:
        print("Trying refinement in resolution")
        # Downsample and reoptimize with higher resolution
        n_downsample = 100
        dx_down = 2.0 / n_downsample
        best_h_down = np.zeros(n_downsample)
        for i in range(n_downsample):
            start_idx = int(i * 2)
            end_idx = int((i+1) * 2)
            if end_idx > n_points:
                end_idx = n_points
            avg_val = np.mean(best_h[start_idx:end_idx])
            best_h_down[i] = avg_val
        best_h_down = project_box_affine(best_h_down, n_downsample / 2)
        # Optimize again
        h_opt_ref, c5_ref = optimize_one_candidate(best_h_down, dx_down, iter_max=500)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_opt_ref.copy()
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
