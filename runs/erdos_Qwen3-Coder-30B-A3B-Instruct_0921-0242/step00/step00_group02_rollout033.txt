import numpy as np
import time
import random
from scipy.signal import correlate
from copy import deepcopy

# Global constants
MAX_N = 60
STAGES = [30, 36, 42, 48, 60]
P_MIN = 30
P_MAX = 2 * MAX_N
TOURNAMENT_SIZE = 3
CROSSOVER_PROB = 0.8
MUTATION_STRENGTH = 2
DIVERSIFICATION_FREQ = 20
ESCAPE_PROB = 0.05
FINAL_POLISHING_SWAPS = 200

def correlation_vector(b):
    # Compute the auto-correlation vector for binary sequence b
    # Returns R=[R_1, ..., R_{n-1}] where R_k = sum b_i * b_{i+k}
    b = np.array(b, dtype=np.uint8)
    full_corr = correlate(b, b, mode='full')
    # Remove the middle element (index n-1 if b.length = n)
    return full_corr[len(b):]

def update_correlation_delta(R, b, i, j):
    # i is set to 0, j is set to 1
    # Bilinear update rule, O(n) operation
    old_min_k = np.argmin(R)
    old_min_val = R[old_min_k]
    n = len(b)
    for k in range(1, n):
        if i - k >= 0 and b[i - k] == 1:
            R[k - 1] -= 1
        if i + k < n and b[i + k] == 1:
            R[k - 1] -= 1
        if j - k >= 0 and b[j - k] == 1:
            R[k - 1] += 1
        if j + k < n and b[j + k] == 1:
            R[k - 1] += 1

    new_min_k = np.argmin(R)
    new_min_val = R[new_min_k]
    improvement = new_min_val - old_min_val
    return improvement, new_min_k

def local_hill_climb(b, R):
    n = len(b)
    # Convert to list for easier manipulation
    b = list(b)
    R = list(R)
    best_hill_climb = deepcopy(b)
    best_R = deepcopy(R)
    
    # Keep looping until no better swap exists
    steps_no_better = 0
    while steps_no_better < 50:
        critical_k = np.argmin(best_R)
        critical_value = best_R[critical_k]
        # Find all 1-0 pairs (i,j) where i is a 1 and j is a 0
        ones = [i for i in range(n) if best_hill_climb[i] == 1]
        zeros = [j for j in range(n) if best_hill_climb[j] == 0]
        candidates = [(i,j) for i in ones for j in zeros]
        
        best_improvement = -np.inf
        best_pair = None
        # Sample or enumerate candidates
        if len(candidates) > 1000:
            random.shuffle(candidates)
            candidates = candidates[:1000]
        for i, j in candidates:
            new_R = deepcopy(best_R)
            old_R_critical = new_R[critical_k]
            # Update new_R for critical_k only, efficiently
            for k in range(1, n):
                if i - k >= 0 and best_hill_climb[i - k] == 1:
                    new_R[k - 1] -= 1
                if i + k < n and best_hill_climb[i + k] == 1:
                    new_R[k - 1] -= 1
                if j - k >= 0 and best_hill_climb[j - k] == 1:
                    new_R[k - 1] += 1
                if j + k < n and best_hill_climb[j + k] == 1:
                    new_R[k - 1] += 1
            new_critical_value = new_R[critical_k]
            improvement = new_critical_value - old_R_critical
            if improvement > best_improvement:
                best_improvement = improvement
                best_pair = (i, j)
    
        if best_pair is None or best_improvement <= 0:
            steps_no_better += 1
        else:
            i, j = best_pair
            # Apply swap
            best_hill_climb[i], best_hill_climb[j] = 0, 1
            # Update R efficiently
            for k in range(1, n):
                if i - k >= 0 and best_hill_climb[i - k] == 1:
                    best_R[k - 1] -= 1
                if i + k < n and best_hill_climb[i + k] == 1:
                    best_R[k - 1] -= 1
                if j - k >= 0 and best_hill_climb[j - k] == 1:
                    best_R[k - 1] += 1
                if j + k < n and best_hill_climb[j + k] == 1:
                    best_R[k - 1] += 1
            steps_no_better = 0

    return best_hill_climb, np.array(best_R)

def generate_random_binary_sequence(n, weight):
    # Generate a binary sequence with exactly `weight` ones
    indices = sorted(random.sample(list(range(n)), weight))
    b = [0] * n
    for i in indices:
        b[i] = 1
    return b

def upscale_sequence(b, n):
    # Upscale binary vector b to new length n
    factor = (n + len(b) - 1) // len(b)
    upsampled = np.repeat(b, factor)
    if len(upsampled) > n:
        upsampled = upsampled[:n]
    else:
        # Duplicate until it reaches at least n
        while len(upsampled) < n:
            upsampled = np.concatenate([upsampled, upsampled])
        upsampled = upsampled[:n]

    # Ensure correct weight by flipping excess 1s or adding missing 0s
    ones_count = np.sum(upsampled)
    diff = ones_count - n // 2
    if diff > 0:
        # Flip diff excess 1s to 0s
        ones_indices = [i for i in range(n) if upsampled[i] == 1]
        for i in range(diff):
            upsampled[ones_indices[i]] = 0
    elif diff < 0:
        # Flip -diff 0s to 1s
        zeros_indices = [i for i in range(n) if upsampled[i] == 0]
        for i in range(-diff):
            upsampled[zeros_indices[i]] = 1
    return list(upsampled)

def compute_c5(h):
    # h: array of length n in [0,1] with sum(h)*dx = 1
    n = len(h)
    dx = 2.0 / n
    h = np.array(h)
    # Standard approach (can be optimized later if necessary)
    conv = correlate(h, 1 - h, mode='full')
    return max(conv) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    random.seed(seed)
    t0 = time.time()
    
    best_sequence = None
    best_C5 = float('inf')
    
    # Iterate through different lengths
    for stage_idx, n in enumerate(STAGES):
        if time.time() - t0 > budget_s - 5:
            break
        remaining_time = budget_s - (time.time() - t0)
        if remaining_time <= 5:
            break
        
        # Time allocation proportional to n
        max_time_stage = min(120, remaining_time / 2.0)  # Caps stage time
        
        # Stage 1 (fast early stages)
        if stage_idx == 0:
            # Fast stage, small population, lots of early diversification
            pop_size = max(P_MIN, min(P_MAX, 20*2))  # Very small for speed
            generations = 200  # some of these work best for 30
        elif stage_idx == 1:
            # mid
            pop_size = max(P_MIN, min(P_MAX, n*2))
            generations = 150
        else:
            # later stages use more time
            pop_size = max(P_MIN, min(P_MAX, n*2))
            generations = min(200, 3*max(1, n//3))
            
        # Stop if not enough time left
        if remaining_time < 5 + 10:
            break
            
        # Start of stage population logic:
        population = []
        
        # Initialize with the current best known solution, if available
        # Otherwise generate random binary sequences
        if stage_idx == 0:
            if 'initial_h_values' in globals():
                # Convert intial_h_values to binary sequence
               ivals = np.array(initial_h_values)
                threshold = np.partition(ivals, len(ivals) - len(ivals) // 2)[-len(ivals) // 2]
                b_initial = (ivals >= threshold).astype(int)
                
                if np.sum(b_initial) != n // 2:
                    candidates = []
                    weights = []
                    for i in range(len(b_initial)):
                        if b_initial[i] == 1:
                            candidates.append(i)
                            weights.append(ivals[i])
                    candidates = [c for _, c in sorted(zip(weights, candidates), reverse=True)]
                    # convert enough 1s to 0s to match target
                    for i in range(len(candidates) - n//2):
                        b_initial[candidates[i]] = 0
                    # add 0s to reach n//2 total
                    zeros = [i for i in range(n) if b_initial[i] == 0]
                    for i in range(n//2 - np.sum(b_initial)):
                        b_initial[zeros[i]] = 1

                population.append(b_initial)
            else:
                # Use a backup
                population.append(generate_random_binary_sequence(n, n // 2))
        else:
            # Use previous best + upsampled population starter
            if best_sequence is None:
                # Fallback
                population.append(generate_random_binary_sequence(n, n // 2))
            else:
                upscaled = upscale_sequence(best_sequence, n)
                population.append(upscaled)
            
        # Fill the rest with random
        for i in range(pop_size - len(population)):
            pop_element = generate_random_binary_sequence(n, n // 2)
            population.append(pop_element)
            
        # Eval initial population
        for i, b in enumerate(population):
            R = correlation_vector(b)
            population[i] = (b, R)
        
        # Track best individual so far
        best_individual_idx = min(range(len(population)), 
                                  key=lambda i: np.min(population[i][1]))
        global_best = deepcopy(population[best_individual_idx])
        global_worst = np.min(population[best_individual_idx][1])
        
        # Adaptive generation limiter
        gen_count = 0
        max_gens = min(generations, int(remaining_time / max(0.001, 0.01*n)) + 1)
        
        for gen in range(max_gens):
            if time.time() - t0 > budget_s - 5:
                break
            gen_count += 1
            
            # Elite individuals
            elites = [i for i in range(len(population)) if np.min(population[i][1]) >= global_worst]
            if len(elites) == 0:
                elites = [best_individual_idx]
            elif len(elites) > 1:
                # choose elite with highest min_correlation
                elites = [min(elites, key=lambda i: np.min(population[i][1]))]
            elite = elites[0]
            
            offspring = []
            for _ in range(pop_size):
                # Selection
                selected_indices = [random.choice(list(range(len(population)))) for _ in range(TOURNAMENT_SIZE)]
                winner_idx = min(selected_indices, key=lambda i: np.min(population[i][1]))
                parent1 = population[winner_idx][0]
                
                # Crossover with probability
                if random.random() < CROSSOVER_PROB:
                    parent2_idx = random.choice(list(range(len(population))))
                    parent2 = population[parent2_idx][0]
                    # Perform a union-and-trim crossover
                    s1 = set([i for i in range(n) if parent1[i] == 1])
                    s2 = set([i for i in range(n) if parent2[i] == 1])
                    union_set = s1.union(s2)
                    if len(union_set) > n // 2:
                        # Remove excess bits
                        elements = list(union_set)
                        while len(elements) != n // 2:
                            to_remove = random.choice(elements)
                            elements.remove(to_remove)
                    elif len(union_set) < n // 2:
                        # Add missing bits
                        elements = list(union_set)
                        all_positions = set(range(n))
                        complement = all_positions - union_set
                        to_add = random.sample(list(complement), n//2 - len(elements))
                        elements.extend(to_add)
                    else:
                        elements = list(union_set)
                    # Now convert back to vector
                    child = [0] * n
                    for el in elements:
                        child[el] = 1
                else:
                    child = parent1
            
                # Mutate (swap two 1s and 0s)
                for _ in range(np.random.randint(1, MUTATION_STRENGTH + 1)):
                    ones = [i for i in range(n) if child[i] == 1]
                    zeros = [j for j in range(n) if child[j] == 0]
                    if ones and zeros:
                        i = random.choice(ones)
                        j = random.choice(zeros)
                        child[i], child[j] = 0, 1

                offspring.append(child)
            
            # Local search on new offspring
            for i, b in enumerate(offspring):
                # Use deterministic approach for robustness
                R = correlation_vector(b)
                b_improved, R_improved = local_hill_climb(b, R)
                offspring[i] = (b_improved, R_improved)
            
            # Replace worst individuals with offspring
            worst_indices = sorted(range(len(population)), 
                                   key=lambda i: np.min(population[i][1]))[-pop_size // 2:]
            for i, idx in enumerate(worst_indices):
                population[idx] = offspring[i]
            
            # Track global best
            for i in range(len(population)):
                min_r = np.min(population[i][1])
                if min_r > global_worst:
                    global_worst = min_r
                    global_best = deepcopy(population[i])
            
            # Periodically add a new random individual (diversify)
            if gen % DIVERSIFICATION_FREQ == 0 and gen > 0:
                r = generate_random_binary_sequence(n, n//2)
                r_R = correlation_vector(r)
                population[0] = (deepcopy(r), r_R)
                # Update global best if needed
                min_r = np.min(r_R)
                if min_r > global_worst:
                    global_worst = min_r
                    global_best = (r, r_R)
    
        # Final polishing of global best
        best_b = global_best[0]
        best_R = global_best[1]
        while True:
            old_min = np.min(best_R)
            critical_k = np.argmin(best_R)
            ones = [i for i in range(n) if best_b[i] == 1]
            zeros = [j for j in range(n) if best_b[j] == 0]
            candidates = [(i,j) for i in ones for j in zeros]
            
            # Exhaustive?
            best_improvement = -np.inf
            best_pair = None
            for i, j in candidates[:FINAL_POLISHING_SWAPS]:  # limit to keep short time
                new_R = list(best_R)
                # Update it for all k using the update_rule for each k
                for k in range(1, n):
                    if i - k >= 0 and best_b[i - k] == 1:
                        new_R[k - 1] -= 1
                    if i + k < n and best_b[i + k] == 1:
                        new_R[k - 1] -= 1
                    if j - k >= 0 and best_b[j - k] == 1:
                        new_R[k - 1] += 1
                    if j + k < n and best_b[j + k] == 1:
                        new_R[k - 1] += 1
                # Check improvement of critical shift
                new_min = min(new_R)
                improvement = new_min - old_min
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_pair = (i, j)
            
            if best_improvement <= 0:
                break
            i, j = best_pair
            best_b[i], best_b[j] = 0, 1
            best_R = [c for c in best_R] 
            for k in range(1, n):
                if i - k >= 0 and best_b[i - k] == 1:
                    best_R[k - 1] -= 1
                if i + k < n and best_b[i + k] == 1:
                    best_R[k - 1] -= 1
                if j - k >= 0 and best_b[j - k] == 1:
                    best_R[k - 1] += 1
                if j + k < n and best_b[j + k] == 1:
                    best_R[k - 1] += 1
        
        # Compute actual C5 value
        dx = 2.0 / n
        conv = correlate(np.array(best_b, dtype=float), 1 - np.array(best_b, dtype=float), mode='full')
        result_c5 = max(conv) * dx
        
        if result_c5 < best_C5:
            best_C5 = result_c5
            best_sequence = np.array(best_b, dtype=float)

    # Final evaluation using continuous convolution (not discrete)
    final_h = best_sequence
    final_c5 = compute_c5(final_h)
    final_n = len(final_h)
    
    return (final_h, final_c5, final_n)
