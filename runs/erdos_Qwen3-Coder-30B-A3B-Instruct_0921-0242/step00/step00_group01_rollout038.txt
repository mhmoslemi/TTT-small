import numpy as np
from scipy import fftpack

import time

def autocorr(b):
    """Compute cyclic autocorrelation of a binary sequence."""
    n = len(b)
    # Use FFT for efficiency
    B = np.fft.fft(b)
    A = np.fft.ifft(B * np.conj(B)).real
    return A

def evaluate_c5(h, dx):
    """Evaluate C5 using numpy correlate as specified."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def evaluate_c5_exact(h, dx):
    """Evaluate the exact C5 bound."""
    conv = np.correlate(h, 1 - h, mode="full")
    return np.max(conv) * dx

def evaluate_erdos_solution(h_values, dx=2.0 / len(h_values)):
    """Evaluate the solution using the given function."""
    return np.max(np.correlate(h_values, 1 - h_values, mode="full")) * dx

def create_legendre_sequence(n):
    """Generate a Legendre sequence of length n."""
    # Find a prime p >= n
    p = n
    while not is_prime(p):
        p += 1
    # Create Legendre sequence
    s = np.zeros(p, dtype=int)
    for i in range(1, p):
        if pow(i, (p-1)//2, p) == 1:
            s[i] = 1
    # Truncate to n
    b = s[:n]
    return b

def is_prime(n):
    """Check if a number is prime."""
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(math.sqrt(n)) + 1, 2):
        if n % i == 0:
            return False
    return True

def create_singer_difference_set(n):
    """Create a Singer difference set if possible."""
    # Work with q = 13 => q^2 + q + 1 = 183
    q = 13
    if q*q + q + 1 <= n:
        n_orig = q*q + q + 1
        b = np.zeros(n, dtype=int)
        ones = np.arange(n_orig)
        # Create difference set in 0..n_orig-1 by shifting 
        # e.g., the set {0,1,2,...,((n_orig-1)//2)} ∪ {n_orig // 2 + 1,...,n_orig-1}
        # but actually simplify: spread ones at equiangular positions 
        b[:n_orig] = 1  # Set first n_orig positions to 1
        # Ensure balance (n//2 ones) by only taking enough ones
        excess = np.sum(b) - n // 2
        if excess > 0:
            # Shrink by dropping from the end
            ones_indices = np.nonzero(b)[0]
            for i in range(excess):
                b[ones_indices[len(ones_indices) - 1 - i]] = 0
        return b
    else:
        return None

def create_palindromic_block(n):
    """Create a simple palindromic block structure."""
    half = n // 2
    b = np.zeros(n, dtype=int)
    # Place 1s in an asymmetric block with a center section
    left_block = n // 4
    right_block = n // 4
    b[left_block:left_block + (n - 2 * left_block)] = 1
    # Ensure balance
    ones_count = np.sum(b)
    if ones_count > n // 2:
        indices = np.where(b == 1)[0]
        extra = ones_count - n // 2
        for i in range(extra):
            b[indices[-1 - i]] = 0
    elif ones_count < n // 2:
        indices = np.where(b == 0)[0]
        missing = n // 2 - ones_count
        for i in range(missing):
            b[indices[i]] = 1
    return b

def create_random_balanced(n):
    """Create a random balanced binary vector."""
    b = np.zeros(n, dtype=int)
    ones = np.random.permutation(n)[:n // 2]
    b[ones] = 1
    # Force balance if needed
    if np.sum(b) != n // 2:
        idx = np.where(b == 1)[0]
        missing = n // 2 - len(idx)
        if missing > 0:
            zeros = np.where(b == 0)[0][:missing]
            b[zeros] = 1
        elif missing < 0:
            extra = np.random.choice(idx, -missing, replace=False)
            b[extra] = 0
    return b

def get_seeds(n):
    """Generate a library of seed vectors for optimization."""
    seeds = []
    # Create Legendre, Singer, Palindromic, and Random sequences
    # Legendre sequence
    b1 = create_legendre_sequence(n)
    if b1 is not None:
        seeds.append(b1.copy())
    
    # Singer difference set 
    b2 = create_singer_difference_set(n)
    if b2 is not None:
        seeds.append(b2.copy())
    
    # Palindromic blocks
    b3 = create_palindromic_block(n)
    seeds.append(b3.copy())
    
    # Random sequences
    for i in range(5):
        b = create_random_balanced(n)
        seeds.append(b.copy())
    
    return seeds

def update_acorr(A_b, old_i, new_i, b, k, mod_n):
    """Update the cyclic autocorrelation vector after a swap at index k."""
    # Rearranging of A due to swap 
    # For any k, A_k is updated due to terms involving b_i, b_{i+k}
    # and b_j, b_{j+k}
    update = 0.0
    old_k = (old_i + k) % mod_n
    new_k = (new_i + k) % mod_n
    # b_i contributes to A_k and b_j to A_k
    # When b_i changes from 1 to 0 and b_j from 0 to 1, 
    # b_i+k and b_j+k flip their contribution to A_k
    update = b[old_k] - b[new_k]
    return update

def extract_min_autocorr(b):
    """Extract minimal autocorrelation of non-zero lags."""
    if len(b) == 0:
        return 0
    ac = autocorr(b)
    return np.min(ac[1:])

def swap_improve(b, dx, max_time=30, swap_attempts=10000, initial_temp=1.0, cooling=0.99):
    """Use simulated annealing to improve binary sequence."""
    # We will work with the binary vector b in place
    n = len(b)
    A = autocorr(b)
    m = np.min(A[1:])  # worst autocorrelation at non-zero lags
    best_b = b.copy()
    best_m = m
    T = initial_temp
    t0 = time.time()
    
    # Precompute shifts
    B_shift = np.array([np.roll(b, k) for k in range(n)])
    
    # While time remains and we haven't plateaued
    no_improvement_count = 0
    iter_count = 0
    while time.time() - t0 < max_time and no_improvement_count < 5000:
        iter_count += 1
        # Find worst lag
        k_star = np.argmin(A[1:]) + 1
        # Random swap
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break

        try:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
        except:
            # If fails (very unlikely), just skip
            continue

        # Evaluate change at k_star
        delta = 0
        # b_i used to contribute to A_{k_star} if b_i = 1
        # but b_i = 0 now, so contrib = 0
        # similarly for b_j  
        # the formula for the change in A_k is: -b_{i+k} + b_{j+k} 
        # but note Double-counting the effect at k_star
        # contribution change:  no longer b_i * b_{i+k_star} 
        # - b_j * b_{j+k_star} + 0 * b_{j+k_star} + 0 * b_{i+k_star}
        b_i_k_star = b[(i + k_star) % n]
        b_j_k_star = b[(j + k_star) % n]
        delta = -b_i_k_star + b_j_k_star
        
        if delta > 0:
            # Accept strictly improving swap
            b[i] = 0
            b[j] = 1
            # Update for all lags
            A = autocorr(b)
            new_m = np.min(A[1:])
            if new_m > m:
                m = new_m
                best_b = b.copy()
                best_m = m
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            # Accept with probability exp(delta/T)
            r = np.random.rand()
            if r < np.exp(delta / T):
                b[i] = 0
                b[j] = 1
                A = autocorr(b)
                new_m = np.min(A[1:])
                if new_m > m:
                    m = new_m
                    best_b = b.copy()
                    best_m = m
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        
        T *= cooling

    return best_b, best_m

def duplicate_and_refine(b, target_length):
    """Duplicate each bit and interpolate locally."""
    n = len(b)
    if n >= target_length:
        return b[:target_length]
    new_b = np.repeat(b, 2)
    # Add fractional elements if needed
    if new_b.shape[0] < target_length:
        new_b = np.pad(new_b, (0, target_length - new_b.shape[0]), mode='constant')
    return new_b

def project_to_feasible(h, dx):
    """Project h to satisfy the constraints."""
    h = np.array(h, dtype=float)
    # Ensure sum * dx = 1
    total = np.sum(h) * dx
    if abs(total - 1.0) > 1e-9:
        h = h * (1.0 / total)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    return h

def int_to_float_h(b, n_points, dx):
    """Convert a binary vector to a uniform float one."""
    # Normalize value to satisfy sum h*dx = 1, so sum h = 1/dx = n_points/2
    h = b.astype(float) * (n_points / (2 * np.sum(b)))
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600):
    """Main optimization loop."""
    np.random.seed(seed)
    t0 = time.time()
    
    # Standard parameters
    n_seed = 200
    desired_n = 500  # Uniform grid size for best results
    
    # Stage 1: Generate seeds (0 - 60s)
    seeds = get_seeds(n_seed)
    print(f"Generated {len(seeds)} seeds")
    
    # Compute initial C5 for all seeds
    best_C5 = float('inf')
    best_h = None
    best_n = 0
    
    # Keep track of the best non-binary reported for use in final steps  
    best_h_score = float('inf')
    best_h_final = None
    best_h_length = 0
    
    iterations = 0
    for b in seeds:
        iterations += 1
        if time.time() - t0 > budget_s - 10:
            break
        # Convert to float vector
        h = int_to_float_h(b, n_seed, 2.0 / n_seed)
        c5 = evaluate_c5_exact(h, 2.0 / n_seed)
        if c5 < best_h_score:
            best_h_score = c5
            best_h_final = h.copy()
            best_h_length = len(h)
            
        # Try to improve b with local swaps
        t_start = time.time()
        improved_b, improved_m = swap_improve(b.copy(), 2.0 / n_seed, 
                                              max_time=min(30, budget_s - (time.time() - t0) - 5))
        if time.time() - t_start > 30:
            print("Swap iteration took too long.")
        # Convert back to continuous h
        h_impr = int_to_float_h(improved_b, n_seed, 2.0 / n_seed)
        c5_impr = evaluate_c5_exact(h_impr, 2.0 / n_seed)
        if c5_impr < best_C5:
            best_C5 = c5_impr
            best_h = h_impr.copy()
            best_n = n_seed
    
    # Stage 2: Scale up if found a good solution (360 - 540s)
    if best_C5 < float('inf') and time.time() - t0 < 180:
        # Duplicate to nearly desired length
        h_exact = best_h.copy()
        h_up = duplicate_and_refine(h_exact, desired_n)
        h_final = project_to_feasible(h_up, 2.0 / desired_n)
        c5_final = evaluate_c5_exact(h_final, 2.0 / desired_n)
        if c5_final < best_C5:
            best_C5 = c5_final
            best_h = h_final
            best_n = len(h_final)
    
    # Final evaluation (540 - 590s) 
    if time.time() - t0 > 590:
        pass  # Finalize at edge
    else:
        c5_final = evaluate_c5_exact(best_h, 2.0 / best_n)
        # Final ensure it's within tolerance
        final_rounded_sum = np.sum(best_h) * (2.0 / best_n)
        if abs(final_rounded_sum - 1.0) > 1e-9:
            print(f"Sum deviation = {abs(final_rounded_sum - 1.0)}")
    
    # Reuse already computed result
    return (best_h, best_C5, best_n)
