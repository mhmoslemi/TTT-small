import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def decode(x, M):
    """Transform flattened parameters to segment lengths u and heights a."""
    u = x[:M]
    a0 = x[M]
    v = x[M+1:]
    
    # Ensure monotonicity by setting a_j = max(0, a_{j-1} - v_{j-1})
    a = np.empty(M + 1)
    a[0] = a0
    for j in range(1, M):
        a[j] = max(0, a[j-1] - v[j-1])
    a[M] = max(0, a[M-1] - v[M-1])
    
    # Compute breakpoints
    s = np.concatenate(([0], np.cumsum(u)))
    return u, a, s

def overlap_piecewise(s, a):
    """Compute exact C5 for piecewise constant function."""
    M = len(a) - 1
    # All possible breakpoints for overlap function
    K = set()
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0 <= k1 <= 2:
                K.add(k1)
            if 0 <= k2 <= 2:
                K.add(k2)
    K = sorted(list(K))
    K = [0] + K + [2]
    
    # Evaluate F(k) for each breakpoint
    F_values = []
    for k in K:
        F = 0.0
        for j in range(M+1):
            for l in range(M+1):
                # Length of intersection of [s[j], s[j+1]) and [s[l]-k, s[l+1]-k)
                left = max(s[j], s[l] - k)
                right = min(s[j+1], s[l+1] - k)
                if right > left:
                    length = right - left
                    F += a[j] * (1 - a[l]) * length
                else:
                    pass  # Length 0
        F_values.append(F)
    
    return max(F_values)

def project_capped_simplex(v, total):
    """Project vector onto capped simplex sum = total, 0 <= v_i <= 1."""
    v = np.array(v)
    n = len(v)
    if np.sum(v) == total:
        return v
    # Use binary search for the Lagrange multiplier
    low, high = -100, 100
    epsilon = 1e-10
    for _ in range(50):
        mu = (low + high) / 2
        x = np.clip(v - mu, 0, 1)
        s = np.sum(x)
        if abs(s - total) < epsilon:
            break
        if s > total:
            low = mu
        else:
            high = mu
    x = np.clip(v - mu, 0, 1)
    return x

def evaluate_penalty(x, M, target_integral=1.0, penalty_weight=1e6):
    u, a, s = decode(x, M)
    integral_residual = np.sum(a * u) - target_integral
    penalty = penalty_weight * integral_residual ** 2
    
    # Bounds check
    for val in a:
        if val < 0 or val > 1:
            return 1e12 + penalty
    for val in u:
        if val < 0:
            return 1e12 + penalty
    
    C5 = overlap_piecewise(s, a)
    return C5 + penalty

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase A: DE with M=4 on n=128
    M_A = 4
    n_A = 128
    dim_A = M_A + 1 + (M_A - 1)
    bounds_A = [(0, 2)] * dim_A
    bounds_A[0] = (0, 1)  # a0 is in [0,1]
    
    t_remaining = budget_s - (time.time() - start_time)
    if t_remaining < 0.5:
        best_x = None
    else:
        de_res_A = differential_evolution(
            lambda x: evaluate_penalty(x, M_A),
            bounds_A,
            maxiter=50,
            popsize=15 * dim_A,
            seed=seed,
            polish=False,
            timeout=t_remaining * 0.3
        )
        best_x = de_res_A.x
    
    # Phase B: DE with M=6 on n=256
    M_B = 6
    n_B = 256
    dim_B = M_B + 1 + (M_B - 1)
    bounds_B = [(0, 2)] * dim_B
    bounds_B[0] = (0, 1)  # a0 is in [0,1]
    
    t_remaining = budget_s - (time.time() - start_time)
    if t_remaining < 0.5:
        final_x = best_x
    else:
        init_pop = []
        if best_x is not None:
            init_pop = [best_x] * 5  # Inject best solution as samples
        de_res_B = differential_evolution(
            lambda x: evaluate_penalty(x, M_B),
            bounds_B,
            maxiter=100,
            popsize=15 * dim_B,
            seed=seed,
            polish=False,
            x0=init_pop,
            timeout=t_remaining * 0.4
        )
        final_x = de_res_B.x
    
    # Local refinement
    def smooth_objective(x):
        u, a, s = decode(x, M_B)
        integral_residual = np.sum(a * u) - 1.0
        penalty = 1e6 * integral_residual ** 2
        C5 = overlap_piecewise(s, a)
        return C5 + penalty
    
    def smooth_objective_slsqp(x):
        return smooth_objective(x)
    
    def smooth_constraint(x):
        u, a, s = decode(x, M_B)
        return np.sum(a * u) - 1.0
    
    # Refinement
    t_remaining = budget_s - (time.time() - start_time)
    if t_remaining < 1.0:
        result_x = final_x
    else:
        bounds_refine = [(0, 2)] * dim_B
        bounds_refine[0] = (0, 1)
        con = {'type': 'eq', 'fun': smooth_constraint}
        x0 = final_x
    
        try:
            ref_result = minimize(
                smooth_objective_slsqp,
                x0,
                method='SLSQP',
                bounds=bounds_refine,
                constraints=con,
                options={'maxiter': 300, 'ftol': 1e-8},
                tol=1e-8
            )
            result_x = ref_result.x
        except:
            result_x = final_x
    
    # Final discretization
    u, a, s = decode(result_x, M_B)
    n_points = 256
    dx = 2.0 / n_points
    h_values = []
    
    x_coords = (np.arange(n_points) + 0.5) * dx
    for x in x_coords:
        for j in range(len(s)-1):
            if s[j] <= x < s[j+1]:
                h_values.append(a[j])
                break
        else:
            h_values.append(a[-1])
    
    h_array = np.array(h_values)
    h_array = project_capped_simplex(h_array, n_points / 2)
    
    corr = np.convolve(h_array, 1-h_array, mode='full') * dx
    c5_bound = np.max(corr)
    
    return (h_array.tolist(), float(c5_bound), n_points)
