def evaluate(h):
    """Compute C₅ = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The max is over the full correlation, but the first and last segments aren't valid.
    # The actual shifts are from -len(h)+1 to len(h)-1
    # We only care about k = 1..len(h)-1 (positive shifts)
    # Similarly for negative shifts: -len(h)+1..-1
    # But in the negative shift case, k = -len(h)+1 gives us the largest offset.
    # However, we can avoid indexing issues by looking at all correlations.
    # The first len(h)-1 elements are from shift k = -len(h)+1 .. -1
    # The middle len(h) elements are from shift k = 0 to len(h)-1
    # The last len(h)-1 elements are from shift k = 1 to len(h)-1

    # corr is of length 2*len(h) - 1
    # Take the central part that represents valid shifts 0..len(h)-1, and also negative shifts
    # But centers at 0 (shift k=0) are treated specially, etc.
    # The shifted version should be only those valid overlaps
    # Actually let's directly compute the result to see "correct" overlap values
    # We compute the positive overlaps explicitly for k in [1, 2*len(h) - 2]
    
    # Better approach:
    full_corr = corr * dx  # scale by dx
    # extract all the valid shift values from 1 to 2*len(h)-2 inclusive (which corresponds to k from 1 to len(h)-1, 1 to len(h)-1 respectively)  
    # This is a bit tricky
    # k goes from -len(h)+1 to len(h)-1, and we want the overlap values
    # Let us compute it properly:
    max_val = np.max(full_corr[len(h)-1:2*len(h)-2])  # valid shifts in [1, len(h)-1]
    # another interpretation - it's the full correlation,
    # and it's symmetric about the center. Zero shift gives sum(h^2)
    # Positive shift k ranges from 1 to len(h)-1:
    # correspond to indices from len(h) to 2*len(h)-2
    # Negative shift k from -len(h)+1 to -1 # correspond to from 0 to len(h)-2  
    # But in our case we only take absolutes values, so viz we have from last to first
    # We just take upper left to lower right, and find maximum between valid indices
    # Let's cleanly extract maximum of overlap for k < len(h) and k > 0 two ways:
    # See the shape for k = 0 (center), k = -len(h)+1 to -1, k = 1 to len(h)-1
    # Valid positive and negative shifts valid around the central peak k=0
    # Valid shift k = 1..len(h)-1: corresponding to indices len(h):2*len(h)-2
    # Valid shift k = -len(h)+1..-1: corresponding to indices 0:len(h)-1
    # So extremes of valid shifts: up to 2*len(h)-2
    # Actually, let's just directly compute valid positive/negative overlaps:
    # negative shifts: k = -len+1 to -1
    # positive shifts: k = 1 to len-1 (len shifts)
    # Center shift k = 0 (no shift): also valid but we ignore it since it's (1-h)^2
    valid_corr = full_corr[0:len(h)-1]  # negative shifts -len+1 to -1 as shifts 0 to len-2  (indices 0 to len-2)
    valid_corr = np.append(valid_corr, full_corr[len(h):])  # append positive shifts 1 to len-1 (indices len to 2*len-2)
    return np.max(valid_corr)

def project_feasible(x, target):
    """Project x onto [0,1] and scale to sum = target (water filling)"""
    # Ensure x is array
    x = np.array(x)
    # Clip to 0,1
    x = np.clip(x, 0, 1)
    # Initial sum
    s = np.sum(x)
    # If sum already matches target
    if abs(s - target) < 1e-9:
        return x

    # Water filling algorithm:
    # Find how much to equalize to achieve desired sum
    # Do 10 iteration max to be safe
    for _ in range(10):
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Compute how much to raise:
        # Among entries <= 1, find count of ones (those must increase),
        # and among remaining, compute how much bits we can distribute evenly to reach target
        # Get indices that are below 1
        non_oners = x < 1
        count = np.sum(non_oners)  # How many to raise        #if count == 0:
        if count == 0:
            break
        change_per = diff / count
        # Possibly we would go over 1, so clip
        x = np.minimum(x + change_per, 1)
        s = np.sum(x)

    return x

def active_shifts(h, eps=1e-9):
    """Find shifts that achieve current max"""
    dx = 2.0 / len(h)
    corr_all = np.correlate(h, 1-h, mode="full") * dx
    max_val = np.max(corr_all)
    # The valid range of k that are really shifted values:
    # We're looking at correlation indices from 0 to 2*n-2
    # We ignore k = 0 shift and consider k = 1 to n-1 as positive shifts
    # and k = -n+1 to -1 as negative shifts, so total of 2n-2 shifts. but we can't form |k|=n so up to n-1.
    # Work from the central point
    # 0 1 2 3  ... 2n-2  
    # Correlation indices 0 to 2n-2, corresponding to shifts -n+1 to n-1 
    # We only look at valid 'positive shifts' from 1 to n-1
    # In full correlation: index 0=-n+1, ..., n-1=0, ..., 2n-2=n-1  
    # Thus shift k=-n+1 to -1 = indices 0 to n-2, 
    # and k=1 to n-1 = indices n to 2n-2.  (zero shift is index n-1)
    # So within these ranges we decide where shift is "active" if it's within eps of max:
    indices_positive = list(range(n, 2 * n - 2))
    indices_negative = list(range(0, n - 1))
    # Find for positive or negative all shifts that are within threshold of the max.
    active = []
    
    for i in indices_positive:
        if abs(corr_all[i] - max_val) < eps:
            k = i - (n - 1)  # Convert index to shift = i - (n-1)  = 1 to n-1
            active.append(k)
    for i in indices_negative:
        if abs(corr_all[i] - max_val) < eps:
            k = i - (n - 1)  # Convert index to shift = i - (n-1)  = -n+1 to -1
            active.append(k)
    return active

def gradient(h, ks):

    g = np.zeros_like(h)

        # A sum of integrals, each integral is integral of h(x)(1-h(x+k)) 
        # The gradient for h[i] would be:
        # for positive shift k: (1-h[i+k])  (sum over shifts where i+k<T)
        # for negative shift k: (1-h[i+k]) (sum over shifts where i+k>=0)

        # An element h[i], when contributing its integral for

        # if k>=0 we have h[t](1-h[t+k])
        # For k>0:
        #   t = i: it is h[i](1-h[i+k]), so gradient component related to h[i] is (1-h[i+k])

        # (sum for t fixed of h[t]*(1-h[t+k]) ) wrt h[t] = (1-h[t+k])  
        # But that doesn't match the expected behavior when taking derivative.
        
        # Since we do x -> h(x) * (1-h(x+k)), let's see it algebraically.
        # For fixed t, take derivative of x*(1-x_{t+k}) wrt h[t]  
        # If t != t+k -> derivative 0 of h_{t+k} = 0  
        # If t = t+k -> derivative   = (1-h(t))  
        # This interpretation is wrong though.
        # Let's think again: 
        # Components of correlation sum (k shift): h[t] * h[t+k]
        # So for given k, consider the overlap sum:
        #  Sum_i h_i * h_{i+k} 
        # Taking derivative with respect to h_j:
        #  sum_i d/dh_j [ h_i * h_{i+k} ] 
        # = sum_i [h_{i+k} * delta_{ij} + h_i * delta_{i+k,j} ]  
        #   = h_{j+k} + h_{j-k}  
        # So then this is the full gradient goes:
        # For each shift k:

        # But for the actual integral itself that we are supervising:

        # Handling differently, reconstruction backwards...
        # We compute overlaps through correlations or direct matrix multiplication
        # But it’s beter: 
        # Analogously:
        # Let y = [h_0, ..., h_{n-1}] 
        # Then we want to compute over k all values of:
        # sum_i h[i] * (1-h[i+k])  
        # = sum_i h[i] - sum_i h[i] * h[i+k]  
        # = sum h - sum_i h[i] * h[i+k]  
        # The first is a scalar
        # Therefore, gradient of h[i] is then:

        # This is direct result.

        # So gradient of h_i wrt this is:
        # 1 - sum (where j=i+k) h_j  = 1 - h[i+k]

        # But for the full output,

def line_search(h, g, t_current, budget_s):
    """Try step sizes and find best improvement."""
    dx = 2.0 / len(h)
    alphas = [0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1]
    best_h = h.copy()
    best_val = evaluate(h)
    for a in alphas:
        if a == 0:
            continue  # skip alpha=0, since this is identity step
        h_new = h - a * g
        h_projected = project_feasible(h_new, 1.0)
        val = evaluate(h_projected)
        if val < best_val:
            best_val = val
            best_h = h_projected.copy()
    return best_h, best_val

def local_optimize(h0, max_iter, t_end):
    """Gradient descent with line search."""
    h = h0.copy()
    best_h = h0.copy()
    best_val = evaluate(h0)
    dx = 2.0 / len(h)

    for it in range(max_iter):
        t = time.time()
        if t > t_end:
            break
        # Determine active shifts
        ks = active_shifts(h)
        if not ks:
            ks = [1]  # fallback
        # Compute subgradient
        g = gradient(h, ks)
        # Do line search with projected gradient
        h_new, val = line_search(h, g, best_val, t_end)
        h = h_new
        if val < best_val:
            best_val = val
            best_h = h.copy()
        if np.abs(val - best_val) < 1e-10:
            break
    return best_h, best_val

def anneal(h0, t_end):
    """Simple simulated Annealing."""
    # Try if optimal found by local search is not touching global
    dx = 2.0 / len(h0)
    h = h0.copy()
    current_val = evaluate(h)
    best_h = h.copy()
    best_val = current_val

    T = 0.01
    decay = 0.995
    steps = 5000

    for step in range(steps):
        t = time.time()
        if t > t_end or T < 1e-8:
            break
        # randomized proposal
        # Pick two points to exchange a small value (push value from i to j)
        i, j = np.random.randint(0, len(h)), np.random.randint(0, len(h))
        # Ensure they are different and both within valid range  
        if i == j:
            continue
        # Amount to transfer
        delta = np.random.rand() * 0.01  # range of transfer
        # Make sure j is not max or i is not min
        # If we try to move too much:
        if h[i] < delta:
            delta = h[i]
        if h[j] + delta > 1:
            delta = 1 - h[j]
        if delta <= 0:
            continue

        # Apply changes
        h[i] -= delta
        h[j] += delta

        # Project
        h = project_feasible(h, 1.0)
        new_val = evaluate(h)

        # Accept with probability
        if new_val < current_val or np.random.rand() < np.exp(-(new_val - current_val) / T):
            current_val = new_val
        else:
            # revert if reject
            h[i] += delta
            h[j] -= delta

        if current_val < best_val:
            best_val = current_val
            best_h = h.copy()

        T *= decay

    return best_h, best_val

def random_feasible(n):
    """Generate a random feasible vector."""
    x = np.random.rand(n)
    x = x / np.sum(x) * 1.0  # scale to sum = 1
    return project_feasible(x, 1.0)

                # sum_i h[i]*h[i+k] is max/min in sense of select one shift 
                # to be max and use it
                constraints.append(an[i,k-1] <= h[i])
                constraints.append(an[i,k-1] <= h[i+k])
                constraints.append(an[i,k-1] >= h[i] + h[i+k] - 1)
                constraints.append(an[i,k-1] >= 0)
            
        obj = cp.Minimize(cp.sum(an, axis=(0, 1)) / (2.0/n))   # weight toward unit

        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CBC, verbose=False, time_limit=(time_limit/1000))  # 1000 ms

        if prob.status == cp.OPTIMAL:
            h0 = np.round(h.value).astype(int)
            # get any valid overlap for CLIP  
            # Binary overlaps with C_5 over shifts:  
            # report current best due to data amount
            # It is okay if integer is feassible, return such.  
            return h0 / (2.0/n)  # scale again to unit vector...
        else:
            # return null 
            return np.array([])
    except Exception as e:
        # fallback or skip for now
        return np.array([])

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_val = float('inf')
    best_n = None

    # Candidate sizes
    candidates = [30, 36, 40, 45, 50, 60]
    
    # actual initial value from ``initial_h_values``  
    # As usually "you are free to initialize" 
    initial_h = np.array(initial_h_values)
    initial_n = len(initial_h)
    print("Starting with initial h of size", initial_n)
    
    for n in candidates:  
        if time.time() - t0 > budget_s - 5:
            break

        if n == initial_n:
            h0_raw = initial_h  # raw input prep.
        else:
            # Linearly interpolate to target length
            t_orig = np.linspace(0, 2, initial_n)
            t_new = np.linspace(0, 2, n)
            h0_raw = np.interp(t_new, t_orig, initial_h)

        # Clip and project: consider raw 
        h0 = project_feasible(h0_raw, 1.0)
        print("Interpolated h size", n, " initial eval", evaluate(h0))

        # Binary MILP for small n
        if n <= 36:
            try:
                # Compute time left
                time_left = budget_s - (time.time() - t0)
                if time_left > 10:
                    # Call MILP binary solver
                    h_binary = milp_binary(n, time_left)
                    if len(h_binary) > 0:
                        val = evaluate(h_binary)
                        if val < best_val - 1e-6:
                            best_val = val
                            best_h = h_binary
                            best_n = n
                print(f"Binary search for n={n} returned")
            except:
                pass  # skip

        # Local optimization with gradients from starting points
        starts = [h0]
        # Generate random sensible points
        for r in range(3):
            if time.time() - t0 > budget_s - 5:
                break
            starts.append(random_feasible(n))
        
        for i, start_h in enumerate(starts):
            if time.time() - t0 > budget_s - 5:
                break
            
            max_iter = 2000
            t_end = t0 + budget_s - 5
            
            h_opt, val_opt = local_optimize(start_h, max_iter, t_end)
            if val_opt < best_val - 1e-6:
                best_val = val_opt
                best_h = h_opt.copy()
                best_n = n
            # Pull down with Annealing from best local optimizer
            h_ann, val_ann = anneal(h_opt, t_end)
            if val_ann < best_val - 1e-6:
                best_val = val_ann
                best_h = h_ann.copy()
                best_n = n

    # Final check to upsample if needed from coarse result
    if best_n is None:
        # Fall back on the initial one just in case
        h0 = project_feasible(initial_h, 1.0)
        best_h = h0
        best_val = evaluate(h0)
        best_n = initial_n

    # Make sure final answer has correct structure:
    final_h = np.array(best_h)
    c5_bound = best_val
    n_points = int(best_n)
    return (final_h, c5_bound, n_points)
