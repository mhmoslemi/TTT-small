import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
import time
from math import sqrt

def evaluate_erdos_solution(h, dx=2.0 / len(h)):
    """Returns the C5 value given h."""
    # make sure h is a numpy array
    h = np.array(h)
    # compute the correlation vector
    corr = np.correlate(h, 1 - h, mode='full')
    # get the max (excluding the trivial zero correlation)
    max_corr = np.max(corr[len(h):])
    return max_corr * dx

def initial_h_values(n=45):
    """Return an initial feasible h vector of length n."""
    h = np.zeros(n)
    # Start with a simple rectangular function with mass 1
    # For n=45, dx = 2/45,
    # So the mass should be 1, so we can use 1/2 * (2/45) * 90 = 1
    h[:22] = 1.0  # First 22 elements set to 1
    h[22:] = 0.0  # Remainder set to 0
    return h

def solve_sdp(n, eps=1e-4):
    """Solve the SDP relaxation for size n."""
    # Create variable for h
    h = cp.Variable(n)
    # Create variable for X = h * h^T
    X = cp.Variable((n, n), symmetric=True)

    # Objective: maximize s
    s = cp.Variable()

    # Constraints
    constraints = []
    # Box constraints
    constraints += [0 <= h[i] for i in range(n)]
    constraints += [h[i] <= 1 for i in range(n)]
    # Mass constraint
    constraints += [cp.sum(h) == n / 2.0]
    # PSD constraint
    constraints += [X >> 0]
    # Diagonal values of X = h
    for i in range(n):
        constraints += [X[i, i] == h[i]]

    # We have exactly n-1 shifts to check
    # Add the constraint that all Phi_k(X) >= s
    for k in range(1, n):
        # Phi_k(X) = sum over i of X[i, i+k]
        # Create expression for that sum
        sum_term = 0
        for i in range(n - k):
            sum_term += X[i, i + k]
        constraints += [sum_term >= s]

    # Solve
    prob = cp.Problem(cp.Maximize(s), constraints)
    try:
        prob.solve(solver=cp.SCS, eps=eps, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            X_opt = X.value
            return h_opt, X_opt
    except Exception:
        pass
    return None, None

def project_psd(X):
    """Project matrix X onto the nearest PSD matrix."""
    evals, evecs = eigh(X)
    evals = np.maximum(evals, 0.0)
    X_proj = evecs @ np.diag(evals) @ evecs.T
    return X_proj

def random_round(X, n, num_rounds=300):
    """Perform random hyperplane rounding."""
    # Compute the Cholesky decomposition of X
    # Try to factor as X = V V^T
    try:
        V = np.linalg.cholesky(X)
        if V.ndim == 1:
            V = V.reshape(-1, 1)
    except np.linalg.LinAlgError:
        X_pos = project_psd(X)
        try:
            V = np.linalg.cholesky(X_pos)
            if V.ndim == 1:
                V = V.reshape(-1, 1)
        except np.linalg.LinAlgError:
            # Fallback: use SVD for decomposition
            U, s, Vt = np.linalg.svd(X)
            s = np.maximum(s, 0)
            V = U @ np.sqrt(np.diag(s))
    # Ensure V is shape (n, r) with r to be determined
    k = V.shape[1] if V.ndim > 1 else 1
    n_rounds = max(1, num_rounds)
    best_h = None
    best_c5 = float('inf')
    for _ in range(n_rounds):
        # Draw a random vector g of size k
        g = np.random.randn(k)
        # Compute z = V g
        z = V @ g
        # Thresholding: set top n/2 elements to 1 and rest to 0
        z_sorted = np.sort(z)[::-1]
        threshold = z_sorted[n//2]
        h_rounded = (z >= threshold).astype(float)
        # If mass is not correct, adjust
        sum_h = np.sum(h_rounded)
        if abs(sum_h - n/2) > 1e-8:
            # Simple correction, just flip one or two elements
            # in oscillation around the target
            diff = sum_h - n/2
            if diff > 0:
                idxs = np.argsort(z)[:int(diff)]
                h_rounded[idxs] = 0
            else:
                idxs = np.argsort(z)[::-1][:int(-diff)]
                h_rounded[idxs] = 1
        # Compute c5
        c5 = evaluate_erdos_solution(h_rounded)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_rounded.copy()
    return best_h, best_c5

def local_improvement(h, X, n, max_swaps=500):
    """Improve h via local swaps."""
    dx = 2.0 / n
    # Compute initial correlation vector
    corr = np.correlate(h, 1 - h, mode='full')
    shifts = np.arange(len(corr) // 2 + 1, len(corr))
    corr_shifts = corr[shifts]
    min_corr_idx = np.argmin(corr_shifts)
    k_star = shifts[min_corr_idx]
    
    # labels = np.arange(n)
    # The set of indices pairs (i, j) such that h[i] = 1 and h[j] = 0
    one_idx = np.where(h == 1)[0]
    zero_idx = np.where(h == 0)[0]
    # Make lists of all such pairs
    candidate_pairs = []
    for i in one_idx:
        for j in zero_idx:
            # There's no real need to be exhaustive; let's do a sample
            candidate_pairs.append((i, j))
            if len(candidate_pairs) >= 2000:
                break
        if len(candidate_pairs) >= 2000:
            break

    if len(candidate_pairs) == 0:
        return h

    # In each step, find the pair that maximizes the decrease in C5
    # We compute change in correlation with shift k_star
    best_swap = None
    best_delta = 0
    current_corr = corr_shifts[min_corr_idx]

    # Instead of exhaustive scan, sample only a subset
    if len(candidate_pairs) > 500:
        sampled_pairs = np.random.choice(len(candidate_pairs), 500, replace=False)
        candidate_pairs = [candidate_pairs[i] for i in sampled_pairs]

    for i, j in candidate_pairs:
        delta = 0
        # We measure increase in correlation of k_star shift (i.e. decrease in C5)
        # Change depends on how many terms in R_{k_star}(h) involve either i or j
        # The contributing terms are:
        # For i in h: terms h_{i}h_{i+k_star} and h_{i-k_star}h_{i}
        # For j in 1-h: terms, h_{j}h_{j+k_star} and h_{j-k_star}h_{j}
        for k in range(n - k_star):
            # This is what the contribution is before swap:
            contrib_before = h[k] * h[k + k_star]
            # and after:
            contrib_after = h[k] * h[k + k_star]
            if k_star == 0:
                continue
            # Part due to i being flipped from 1 to 0:
            if i == k:
                contrib_after -= h[k + k_star]
            if i == k + k_star:
                contrib_after -= h[k]
            # Part due to j being flipped from 0 to 1:
            if j == k:
                contrib_after += h[k + k_star]
            if j == k + k_star:
                contrib_after += h[k]
            delta += (contrib_after - contrib_before)

        if delta > best_delta:
            best_delta = delta
            best_swap = (i, j)

    if best_swap is not None:
        # Apply the swap i -> 0, j -> 1
        h[best_swap[0]] = 0
        h[best_swap[1]] = 1
        # update corr
        corr = np.correlate(h, 1 - h, mode='full')
        shifts = np.arange(len(corr) // 2 + 1, len(corr))
        corr_shifts = corr[shifts]
        min_corr_idx = np.argmin(corr_shifts)
        k_star = shifts[min_corr_idx]

    return h

def run(seed=42, budget_s=600, verbose=False):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    # Use a multi-scale approach
    # We'll try different n
    sizes = [30, 34, 38, 42]
    times_per_stage = [80, 120, 150, 350]  # total time so far: [80, 200, 350, 700]
    time_offset = 0
    for si, (n, t) in enumerate(zip(sizes, times_per_stage)):
        if time.time() - start_time + 10 > budget_s:
            break

        if verbose:
            print(f"Stage {si+1}: n={n}")
        stage_start = time.time()
        stage_end_time = start_time + time_offset + t

        # Solve SDP
        if time.time() - start_time > stage_end_time - 10:
            break
        h_sdp, X_sdp = solve_sdp(n, eps=1e-4)
        if h_sdp is None:
            continue

        if time.time() - start_time > stage_end_time - 8:
            break
        # Do rounding
        h_rounded, c5_rounded = random_round(X_sdp, n, 300)
        if h_rounded is None:
            continue

        if time.time() - start_time > stage_end_time - 3:
            break

        # Local improvement
        h_improved = h_rounded.copy()  # avoid modifying original
        h_improved = local_improvement(h_improved, X_sdp, n, max_swaps=500)
        c5_improved = evaluate_erdos_solution(h_improved)

        if c5_improved < best_c5:
            best_c5 = c5_improved
            best_h = h_improved
            best_n = n
            if verbose:
                print(f"New best C5 = {c5_improved:.5f} at n={n}")

        time_offset += t

        if best_c5 <= 0.3807:
            break

    if best_h is None:
        # Fallback
        h_init = initial_h_values(45)
        best_c5 = evaluate_erdos_solution(h_init)
        best_h = h_init
        best_n = 45

    return (best_h, best_c5, best_n)
