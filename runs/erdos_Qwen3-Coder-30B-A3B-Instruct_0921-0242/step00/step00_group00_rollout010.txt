import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import bisect
import time


def project_capped_simplex(v, total):
    """
    Project vector v onto the capped simplex: 0 <= x_i <= 1, sum x_i = total
    Uses bisection on the Lagrange multiplier.
    """
    v = np.asarray(v)
    n = len(v)
    if n == 0:
        return np.array([])
    
    def proj_sum(lam):
        return np.sum(np.clip(v - lam, 0, 1)) - total
    
    # Find appropriate lambda
    low, high = np.min(v), np.max(v) + 1.0
    try:
        lam = bisect(proj_sum, low, high)
    except:
        # fallback if bisect fails
        lam = np.mean(v)
        try:
            lam = bisect(proj_sum, low, high)
        except:
            pass
    
    return np.clip(v - lam, 0, 1)


def auto_corr(h):
    """
    Compute zero-padded linear auto-correlation of h.
    Returns array of length len(h)
    """
    n = len(h)
    # Convert to numpy and pad with zeros
    h_pad = np.pad(h, (0, n), mode='constant')
    # Use FFT-based convolution (negative correlation)
    corr = fftconvolve(h_pad, h[::-1], mode='valid')[n-1:]
    # Return only non-negative lags to match problem definition
    return corr[:n]


def grad_F(beta, h, O, w):
    """
    Compute gradient of smooth max surrogate F_beta(h) = 1 - (1/beta) * log(sum(exp(beta * O_k)))
    """
    # w_k = exp(beta * O_k) / sum(exp(beta * O_k))
    # grad F_beta = -sum(w_k * grad O_k)
    # O_k = sum(h_i * h_{i+k}) for i < n-k
    # grad O_k / dh_i = h_{i+k} for i+k < n (forward term)
    #                  + h_{i-k} for i-k >= 0 (backward term)
    n = len(h)
    # w is already computed
    # Compute gradient as weighted average of gradients of o_k:
    # g_i = sum_k w_k (h_{i+k} + h_{i-k}) 
    #  = w * (h_{:n} + h_{k:} for k = 0,...,n-1)
    # More precisely:
    # x = conv(w, h) + conv_rev(w, h) 
    # but must handle boundaries correctly and k ranges from 0..n-1
    
    # Let's do the full gradient computation manually:
    # g_i = sum_{k=0}^{n-1} w_k h_{i+k} if i+k < n
    #     + sum_{k=0}^{n-1} w_k h_{i-k} if i-k >= 0
    g = np.zeros_like(h)
    for k in range(n):
        w_k = w[k]
        if k == 0:
            g += w_k * h
        else:
            # both forward and backward terms contribute to g_i
            # g[i] += w_k * h[i+k]
            # g[i+k] += w_k * h[i]
            # Both are Toeplitz like structures; compute as rank 1 updates
            g += w_k * np.pad(h[:-k], (0, k), mode='constant')
            g += w_k * np.pad(h[k:], (k, 0), mode='constant')
    return g


def optimize_one_start(n, beta_schedule, time_limit):
    time_start = time.time()
    
    # Initial point from the best known construction or random
    initial = np.array(initial_h_values)  # Should be already defined
    if len(initial) != n:
        # Interpolate
        x_orig = np.linspace(0, 2, len(initial))
        x_new = np.linspace(0, 2, n)
        initial = np.interp(x_new, x_orig, initial)
    
    h = project_capped_simplex(initial, total=n/2)
    
    best_C5 = np.inf
    best_h = h.copy()
    
    # Inner loop running tolerances
    stop_inner_iter = False
    for t, beta in enumerate(beta_schedule):
        if time.time() - time_start > time_limit:
            break
            
        # Inner loop with projected gradient descent
        h_old = h.copy()
        # Compute all O_k
        O = auto_corr(h)
        # Weighting vectors
        exp_beta_O = np.exp(beta * O)
        w = exp_beta_O / np.sum(exp_beta_O)
        
        # gradient of F_beta
        g = grad_F(beta, h, O, w)
        
        # Backtracking line search
        alpha = 0.1
        min_iter = 20
        iter_since_improvement = 0
        while True:
            h_new = h - alpha * g
            h_new = project_capped_simplex(h_new, total=n/2)
            # New F_beta
            O_new = auto_corr(h_new)
            F_new = 1.0 - (1.0 / beta) * np.log(np.sum(np.exp(beta * O_new)))
            
            # Eval actual C5
            corr = np.convolve(h_new, 1 - h_new, mode="full")
            C5_new = np.max(corr * (2.0 / n))
            
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_new.copy()
            # Armijo condition
            if F_new <= (1 - (1/beta) * np.log(np.sum(np.exp(beta * O))) - 1e-4 * alpha * np.dot(g, g)):
                h = h_new
                break
            else:
                alpha /= 2.0
                iter_since_improvement += 1
                if iter_since_improvement > 20:
                    break
                    
        # Test early stopping on improvements
        # If improvement is less than 1e-7, we can stop inner loop
        delta = np.linalg.norm(h - h_old)
        if t > 0 and delta < 1e-7:
            pass  # Continue with next beta
        if t > 0 and iter_since_improvement > 20:
            break
    
    # Final refinement: Exact C5 maximization under fixed beta (largest beta)
    # Use greedy search where we move in the direction of the worst lag
    O_final = auto_corr(best_h)
    best_lag = np.argmin(O_final)
    
    for iter in range(1000): # quick polish
        if time.time() - time_start > time_limit:
            break
        h_old = best_h.copy()
        
        # direction of the worst lag
        # gradient of - O_k* = - sum h_i * h_{i+k*}
        # grad = [0,...,0, h_{k*},..., h_{n-1+k*}]  + [h_{k*}, ..., h_{n-1}], 0, ...
        # so reconstruct complete vector using forward backward
        lag = best_lag
        
        b_grad = np.zeros_like(best_h)
        b_grad[lag:] += best_h[:-lag] if lag > 0 else best_h
        b_grad[:-lag] += best_h[lag:] if lag > 0 else best_h
        
        # Maximize direction of actual -O_kvec
        step = 0.001 / np.sqrt(iter + 1.0)
        h_candidate = best_h - step * b_grad
        h_candidate = project_capped_simplex(h_candidate, total=n/2)
        
        corr = np.convolve(h_candidate, 1 - h_candidate, mode="full")
        C5_new = np.max(corr * (2.0 / n))
        
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h_candidate.copy()
        else:
            break  # No improvement
    
    # Final evaluation
    final_corr = np.convolve(best_h, 1 - best_h, mode="full")
    final_C5 = np.max(final_corr * (2.0 / n))
    
    return best_h, final_C5


def run(seed=42, budget_s=600):

    best_C5 = np.inf

    best_n = None
    
    # Time management
    time_start = time.time()
    
    # Hyperparameters for the optimization
    n_choices = [256, 384]
    beta_min = 0.5
    beta_max = 40.0

    # Multi-start using random seeds
    max_restarts = 8
    for restart in range(max_restarts):
        if time.time() - time_start > 0.8 * budget_s:
            break
            
        current_seed = seed + restart
        np.random.seed(current_seed)
        
        for n in n_choices:
            if time.time() - time_start > 0.7 * budget_s:
                break
                
            time_for_this_run = 0.7 * budget_s / (max_restarts * len(n_choices))  # adjust for parallelism
            try:
                h_opt, C5_opt = optimize_one_start(n, beta_schedule, time_for_this_run)
                if C5_opt < best_C5:
                    best_C5 = C5_opt
                    best_h = h_opt.copy()
                    best_n = n
            except Exception as e:
                continue
                
        # Early exit if we beat target
        if best_C5 <= 0.379:
            break
            
    # Final polishing of best solution
    time_remaining = budget_s - (time.time() - time_start)
    if best_h is not None and time_remaining > 0.2:
        try:
            # \(O(n \log n)\) number of manual gradient steps with optimal step sizes
            # now do additional polishing to maybe ensure correctness
            h_final = best_h.copy()
            O_final = auto_corr(h_final)
            for i in range(500):
                if time.time() - time_start > 0.8 * budget_s:
                    break
                best_lag = np.argmin(O_final)
                # Maximize along the gradient of -O_k*
                grad = np.zeros_like(h_final)
                lag = best_lag
                if lag != 0:
                    grad[lag:] += h_final[:-lag]
                    grad[:-lag] += h_final[lag:]
                else:
                    grad += h_final
            
                step = 0.0001 / (i+1.0)
                h_new = h_final - step * grad
                h_new = project_capped_simplex(h_new, total=best_n/2)

                    h_final = h_new.copy()
                else:
                    break
            best_h = h_final
        except Exception:
            pass
    
    if best_h is not None:
        return (best_h.tolist(), best_C5, best_n)
    else:
