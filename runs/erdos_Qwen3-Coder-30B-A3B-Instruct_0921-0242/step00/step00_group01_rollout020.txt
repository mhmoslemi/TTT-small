import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
import math

def evaluate_C5(h, dx):
    # Compute autocorrelations efficiently using FFT
    H = fft(h)
    autocorr = ifft(np.abs(H)**2).real
    # The first element of autocorr is the maximum correlation (when k=0)
    # We want the minimum of correlations for k=1,2,...,n-1
    # But the problem is asking for max_k of \int h(x)(1 - h(x+k)) dx
    # Which equals sum(h) - sum(h*(shifted h)) = n*dx - A_k
    # So minimizing C5 = max_k [n*dx - A_k] is the same as maximizing min_k A_k
    # But our task is to minimize C5, i.e. minimize max_k [ n*dx - A_k ]
    # which means maximizing min_k A_k
    # The argsort finds the indices that would sort the array
    # We want the minimum A_k, so we return the minimum of autocorr[1:]
    return np.max(autocorr[1:])

def evaluate_C5_full(h, dx):
    # Evaluate all overlaps
    n = len(h)
    max_overlap = 0
    for k in range(1, n):
        # Correlation h * (1 - h[k:])
        corr = np.sum(h * (1 - np.roll(h, k)))
        max_overlap = max(max_overlap, corr)
    return max_overlap

def block_method_3(n_points=400):
    dx = 2.0 / n_points
    best_C5 = float('inf')
    best_h = None
    
    # Sweep a and b (breakpoints)
    step = 0.02
    a_vals = np.arange(0, 2, step)
    b_vals = np.arange(step, 2, step)
    
    for a in a_vals:
        if a >= 2:
            break
        for b in b_vals:
            if b <= a or b >= 2:
                continue
            p = (1 - a) / (b - a)
            if 0 <= p <= 1:
                h = np.zeros(n_points)
                start_a = int(a / dx)
                end_a = int(b / dx)
                h[:start_a] = 1
                h[start_a:end_a] = p
                # Normalize to sum = n/2
                h_sum = np.sum(h)
                if h_sum > 0:
                    h *= (n_points / 2) / h_sum
                
                # Evaluate C5
                C5 = evaluate_C5_full(h, dx)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h.copy()
    
    return best_h, best_C5

def block_method_4(n_points=400):
    dx = 2.0 / n_points
    best_C5 = float('inf')
    best_h = None
    
    # Sweep a, b, c (breakpoints)
    step = 0.04
    a_vals = np.arange(0, 2, step)
    b_vals = np.arange(step, 2, step)
    c_vals = np.arange(step * 2, 2, step)
    
    for a in a_vals:
        for b in b_vals:
            if b <= a or b >= 2:
                continue
            for c in c_vals:
                if c <= b or c >= 2:
                    continue
                # Solve for p, q from two constraints:
                # 1. a*1 + (b-a)*p + (c-b)*q = 1
                # 2. (b-a)*p = (c-b)*q (equal mass in middle blocks)
                A = np.array([[1, b-a, 0], [0, b-a, c-b]])
                b_vec = np.array([1, 0])
                try:
                    sol = np.linalg.solve(A, b_vec)
                    p, q = sol[0], sol[1]
                    if 0 <= p <= 1 and 0 <= q <= 1:
                        h = np.zeros(n_points)
                        start_a = int(a / dx)
                        start_b = int(b / dx)
                        start_c = int(c / dx)
                        h[:start_a] = 1
                        h[start_a:start_b] = p
                        h[start_b:start_c] = q
                        # Normalize to sum = n/2
                        h_sum = np.sum(h)
                        if h_sum > 0:
                            h *= (n_points / 2) / h_sum
                        
                        # Evaluate C5
                        C5 = evaluate_C5_full(h, dx)
                        if C5 < best_C5:
                            best_C5 = C5
                            best_h = h.copy()
                except np.linalg.LinAlgError:
                    continue
    
    return best_h, best_C5

def gaussian_mixture(h_params, n_points=200):
    # h_params: list of [center, amplitude, width] - we have 3 mixture components
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for c, a, w in h_params:
        # Convert center to index
        center_idx = int(max(0, min(n_points - 1, c / dx)))
        # Half-width in indices
        hw = int(max(1, w / dx))
        start_idx = max(0, center_idx - hw)
        end_idx = min(n_points, center_idx + hw)
        h[start_idx:end_idx] += a
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Force sum = n/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h *= (n_points / 2) / h_sum
    return h

def de_optimize(n_points=200, budget_s=120):
    def objective(params):
        h = gaussian_mixture(params, n_points)
        dx = 2.0 / n_points
        C5 = evaluate_C5_full(h, dx)
        return C5
    
    bounds = []
    for i in range(3):
        bounds.append((0.0, 2.0))        # x center in [0,2]
        bounds.append((0.0, 1.0))        # amplitude in [0,1]
        bounds.append((0.01, 0.5))       # width in [0.01, 0.5]
        
    start_time = time.time()
    result = differential_evolution(objective, bounds, seed=42, maxiter=200, 
                                    popsize=30, mutation=(0.5, 1.0), recombination=0.7)
    elapsed = time.time() - start_time
    
    # evaluate final solution
    h = gaussian_mixture(result.x, n_points)
    dx = 2.0 / n_points
    C5 = evaluate_C5_full(h, dx)
    
    # interpolate to fine grid for final result
    h_fine = gaussian_mixture(result.x, 400)
    dx_fine = 2.0 / 400
    C5_fine = evaluate_C5_full(h_fine, dx_fine)
    
    return h_fine, C5_fine

def project_h(h, n):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Enforce sum = n/2 by subtracting mean offset
    offset = (h.sum() - n / 2) / n
    h -= offset
    # Clip again and reproject one more time if needed
    h = np.clip(h, 0, 1)
    offset2 = (h.sum() - n / 2) / n
    h -= offset2
    h = np.clip(h, 0, 1)
    return h

def gradient_ascent_optimize(initial_h, n_points, budget_s=300):
    # Initialize
    h = initial_h.copy()
    dx = 2.0 / n_points
    start_time = time.time()
    best_h = h.copy()
    best_C5 = evaluate_C5_full(h, dx) 
    
    # Early stopping tracking
    last_C5 = best_C5
    patience = 0
    patience_limit = 200
    
    # Multi-resolution
    coarse_grid = [150, 300, 500] if n_points < 300 else [300, 500]
    grid_choices = []
    for grid_size in coarse_grid:
        if grid_size <= n_points:
            grid_choices.append(grid_size)
    if not grid_choices:
        grid_choices = [n_points]
    
    grid_idx = 0
    iterations = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        current_n = grid_choices[grid_idx] if grid_idx < len(grid_choices) else n_points
        
        if current_n != len(h):
            # Interpolate
            h = np.interp(np.linspace(0, 2, current_n), np.linspace(0, 2, len(h)), h)
            h = project_h(h, current_n)
            dx = 2.0 / current_n
            
        # Compute autocorrelations via FFT
        H = fft(h)
        A = ifft(np.abs(H)**2).real
        # A[0] corresponds to k=0 (always max), others are for k=1,...,n-1
        # We're interested in minimizing C5 = max_k (1 - A_k), which is equivalent to
        # maximizing min_k A_k for k from 1 to n-1
        k_star = np.argmin(A[1:]) + 1  # Account for index shift: 0 to n-1, but k=1 to n-1

        # Gradient is 2 * roll(h, k_star)
        g = 2 * np.roll(h, k_star)
        
        # Step size decreases geometrically
        eta = max(1e-6, 0.5 * (0.98 ** iterations))
        
        # Update
        h_new = h + eta * g
        
        # Project back
        h_new = project_h(h_new, len(h))
        
        h = h_new
        
        # Evaluate
        C5 = evaluate_C5_full(h, dx)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        iterations += 1
        if patience > patience_limit:
            if grid_idx < len(coarse_grid) - 1:
                grid_idx += 1
                patience = 0
            else:
                break
        
        # Update step size
        eta = max(1e-6, 0.5 * (0.98 ** iterations))
        
    return best_h, best_C5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Stage 1: Block Enumeration for 3- and 4-blocks
    # 3-blocks
    print("Stage 1a: 3-block")
    h1, C5_1 = block_method_3(400)
    print(f"3-block C5: {C5_1}")
    
    # 4-block
    print("Stage 1b: 4-block")
    h2, C5_2 = block_method_4(400)
    print(f"4-block C5: {C5_2}")
    
    best_block_h = h1 if C5_1 < C5_2 else h2
    best_block_C5 = min(C5_1, C5_2)
    
    # Stage 2: Stochastic optimization (DE)
    print("Stage 2: Differential Evolution")
    h3, C5_3 = de_optimize(n_points=200, budget_s=120)
    print(f"DE C5: {C5_3}")
    
    best_stochastic_h = h3
    best_stochastic_C5 = C5_3
    
    # Choose best from block and DE to start gradient ascent
    if best_block_C5 < best_stochastic_C5:
        start_h = best_block_h
        start_C5 = best_block_C5
    else:
        start_h = best_stochastic_h
        start_C5 = best_stochastic_C5

    print("Stage 3: Gradient Ascent")
    # Start with a coarse resolution for speed, then improve
    final_h, final_C5 = gradient_ascent_optimize(start_h, 400, budget_s=300)
    print(f"Gradient ascent C5: {final_C5}")

    # Final evaluation
    final_h_eval = final_h.tolist()
    # Re-evaluate with full precision if low res
    dx_final = 2.0 / len(final_h_eval)
    final_C5_eval = evaluate_C5_full(final_h_eval, dx_final)
    return (final_h_eval, final_C5_eval, len(final_h_eval))
