import numpy as np
import time
from scipy import signal

def project_simplex_box(v, total, lo=0., hi=1.):
    # Project v onto the simplex {x : sum(x) = total, lo <= x_i <= hi}
    v = np.array(v, dtype=float)
    v_clipped = np.clip(v, lo, hi)
    v_sum = np.sum(v_clipped)
    if abs(v_sum - total) < 1e-12:
        return v_clipped
    # Find threshold theta such that sum(min(1, max(0, v_i - theta))) = total
    # Sort values
    s = np.sort(v_clipped)[::-1]
    cumsum = np.cumsum(s)
    n = len(s)
    # Find largest k such that s_k - (total - cumsum[k-1]) / k >= 0
    theta = (total - cumsum[0]) / 1
    for k in range(1, n):
        theta_k = (total - cumsum[k - 1]) / k
        if theta_k >= s[k]:
            theta = theta_k
            break
    # Apply threshold and ensure sum = total
    projection = np.clip(v - theta, lo, hi)
    # Adjust for numerical error
    adjustment = (total - np.sum(projection)) / len(projection)
    projection += adjustment
    return projection

def compute_c5(h, dx):
    # Compute the maximum overlap: max_k ∫ h(x)(1 - h(x+k)) dx
    # Using FFT
    h1 = np.array(h, dtype=float)
    h2 = 1 - h1
    # Compute full cross-correlation
    corr = signal.correlate(h1, h2, mode='full')
    # This gives full correlation with lags from -(n-1) to n-1
    # We only care about positive lags (k = 0 to n-1)
    corr = corr[len(h1)-1:]
    # We want to compute max_k dx * sum h_i (1 - h_{i+k})
    # Which is max_k dx * (sum h_i - sum h_i h_{i+k})
    # i.e., max_k dx * (1 - sum h_i h_{i+k}) since sum h_i = 1
    # So we want max_k (1 - sum h_i h_{i+k}) = max_k (1 - corr_k/dx)
    # Thus max_k (1 - corr_k/dx) = 1 - min_k corr_k/dx = 1 - min_k corr_k * dx
    # But we're concerned with max_k of the correlation (positive overlap)
    corr_max = np.max(corr)
    c5 = corr_max * dx
    return c5

def worst_lag(h):
    # Return (k, overlap) where overlap is the maximum value
    # Use FFT
    h1 = np.array(h, dtype=float)
    h2 = 1 - h1
    corr = signal.correlate(h1, h2, mode='full')
    corr = corr[len(h1)-1:]
    k_max = np.argmax(corr)
    overlap = corr[k_max]
    return k_max, overlap

def gradient_worst(h, k):
    # Compute gradient of overlap_k = sum h_i h_{i+k} wrt h_i
    # ∂O_k/∂h_i = h_{i+k} + h_{i-k}
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = h[(i + k) % n] + h[(i - k) % n]
    return grad

def optimize(n, time_limit):
    # Build initial solution from initial_h_values if length matches
    h_base = initial_h_values
    if len(h_base) == n:
        h = np.array(h_base)
    else:
        h = np.random.dirichlet(np.ones(n)) * (n / 2)
    
    dx = 2.0 / n
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    start_time = time.time()
    np.random.seed(42)
    step_size = 0.001  # adaptive step size
    max_iter = 10000
    for iter in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Find worst lag and its overlap
        k_max, overlap = worst_lag(h)
        # Compute gradient of worst overlap
        grad = gradient_worst(h, k_max)
        # Take gradient step to decrease overlap
        h_new = h - step_size * grad
        # Project step back
        h_new = project_simplex_box(h_new, n / 2.0, 0., 1.)
        # Evaluate new value
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        # Adaptive step size
        if iter > 0 and iter % 100 == 0:
            step_size = max(0.000001, step_size * 0.95)
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    elapsed = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    # Start with small samples, then increase
    n_vals = [150, 250, 350] # Less than 1000
    
    for i, n in enumerate(n_vals):
        if time.time() - elapsed > budget_s - 2:
            break
        if i < len(n_vals) - 1:
            time_for_n = budget_s / len(n_vals) * 0.8
        else:
            time_for_n = budget_s - (time.time() - elapsed) - 1
        if time_for_n < 1:
            break
        h, c5 = optimize(n, time_for_n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n
