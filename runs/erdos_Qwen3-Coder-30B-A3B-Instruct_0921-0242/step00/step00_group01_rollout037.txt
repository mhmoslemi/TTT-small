import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
import math

def evaluate_overlap(h):
    """Evaluate C5 for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute circular autocorrelation
    H = fft(h)
    autocorr = ifft(np.abs(H)**2).real
    # The overlap integral becomes: dx * (n/2 - min_autocorr)
    min_autocorr = np.min(autocorr)
    C5 = dx * (n/2 - min_autocorr)
    return C5

def block_family_search(n_base=400):
    """Enumerate 3-block and 4-block patterns and return the best."""
    best_C5 = float('inf')
    best_h = None

    # 3-block: [1, p, 0]
    for a in np.arange(0.01, 2.0, 0.02):
        for b in np.arange(a + 0.01, 2.0, 0.02):
            p = (1.0 - a) / (b - a)
            if 0 <= p <= 1:
                h = np.zeros(n_base)
                h[:int(a*n_base/2)] = 1.0
                h[int(a*n_base/2):int(b*n_base/2)] = p
                h[int(b*n_base/2):] = 0.0
                # Normalize to sum = n_base/2
                h = h * (n_base/2) / np.sum(h)
                C5 = evaluate_overlap(h)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h.copy()
    
    # 4-block: [1, p, q, 0], make p*q = 1-a-b, and p = q for balance
    for a in np.arange(0.02, 2.0, 0.04):
        for b in np.arange(a + 0.02, 2.0, 0.04):
            for c in np.arange(b + 0.02, 2.0, 0.04):
                if (c - b) <= 0 or (1 - a) <= 0:
                    continue
                q = (1 - a) / (c - b)
                p = q
                if 0 <= p <= 1:
                    h = np.zeros(n_base)
                    h[:int(a*n_base/2)] = 1.0
                    h[int(a*n_base/2):int(b*n_base/2)] = p
                    h[int(b*n_base/2):int(c*n_base/2)] = q
                    h[int(c*n_base/2):] = 0.0
                    h = h * (n_base/2) / np.sum(h)
                    C5 = evaluate_overlap(h)
                    if C5 < best_C5:
                        best_C5 = C5
                        best_h = h.copy()
    
    return best_h, best_C5

def stochastic_optimization(n_points=200, budget_s=120):
    """Use stochastic population-based search with simple Gaussians."""
    np.random.seed(42)
    
    def mix_gaussians(params):
        centers = params[:3]
        amplitudes = params[3:6]
        h = np.zeros(n_points)
        for c, a in zip(centers, amplitudes):
            left = max(0, int((c - 0.1) * n_points / 2))
            right = min(n_points, int((c + 0.1) * n_points / 2))
            h[left:right] += a
        # Clip is [0,1]
        h = np.clip(h, 0, 1)
        # Enforce sum = n_points/2
        scale = (n_points / 2) / np.sum(h)
        h *= scale
        h = np.clip(h, 0, 1)
        return h
    
    # Objective is -C5 for minimization
    def objective(params):
        h = mix_gaussians(params)
        val = evaluate_overlap(h)
        return -val  # Negative because we minimize
    
    bounds = [(0.01, 1.99), (0.01, 1.99), (0.01, 1.99),
              (0.01, 1.0), (0.01, 1.0), (0.01, 1.0)]
    
    de = differential_evolution(objective, bounds, maxiter=50,
                                popsize=30, seed=42, disp=False)
    
    # Rerun on high-res grid
    h_final = mix_gaussians(de.x)
    # Interpolate to higher resolution
    h_high = np.interp(np.linspace(0, 2, 400), 
                       np.linspace(0, 2, n_points), h_final)
    # Normalize
    h_high = h_high * (400 / 2) / np.sum(h_high)
    C5 = evaluate_overlap(h_high)
    
    return h_high, C5

def project(h, n):
    """Project h onto feasible set."""
    h = np.clip(h, 0, 1)
    # Enforce sum = n/2
    offset = (np.sum(h) - n/2) / n
    h -= offset
    h = np.clip(h, 0, 1)
    offset = (np.sum(h) - n/2) / n
    h -= offset
    h = np.clip(h, 0, 1)
    return h

def gradient_ascent_step(h, kstar, stepsize, n):
    """Perform one step of projected gradient ascent."""
    g = np.roll(h, kstar)
    h_new = h + stepsize * g
    return project(h_new, n)

def projected_gradient_ascent(initial_h, budget_s):
    """Run gradient ascent, upscaling resolution as needed."""
    start_time = time.time()
    best_h = initial_h.copy()
    best_C5 = evaluate_overlap(best_h)
    
    # Multi-resolution schedule
    resolutions = [150, 300, 500]
    step_sizes = [0.5, 0.25, 0.1]
    tolerances = [1e-6, 1e-6, 1e-6]
    
    last_improvement = 0
    iter_count = 0
    
    for res, stepsize, tol in zip(resolutions, step_sizes, tolerances):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Interpolate to current resolution
        n = res
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h)), initial_h)
        
        # Make sure sum = n/2
        h = h * (n/2) / np.sum(h)
        
        iter_start = iter_count
        for step in range(2000):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute autocorrelation
            H = fft(h)
            autocorr = ifft(np.abs(H)**2).real
            kstar = np.argmin(autocorr)
            C5_prev = evaluate_overlap(h)
            
            # Take gradient step
            h = gradient_ascent_step(h, kstar, stepsize, n)
            
            # Evaluate
            C5 = evaluate_overlap(h)
            
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
                
            iter_count += 1
            
            # Early stopping
            if iter_count - iter_start > 200 and abs(C5_prev - C5) < tol:
                break
                
        initial_h = h.copy()  # For next iteration
    
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    np.random.seed(seed)
    
    best_C5 = float("inf")
    best_h = None
    
    # Stage 1: Block families
    if time.time() - start_time < budget_s * 0.2:
        h_block, C5_block = block_family_search()
        if C5_block < best_C5:
            best_C5 = C5_block
            best_h = h_block.copy()
    
    # Stage 2: Stochastic optimization
    if time.time() - start_time < budget_s * 0.4:
        h_de, C5_de = stochastic_optimization()
        if C5_de < best_C5:
            best_C5 = C5_de
            best_h = h_de.copy()
    
    # Stage 3: Gradient ascent with multi-resolution
    if time.time() - start_time < budget_s - 10:
        h_grad, C5_grad = projected_gradient_ascent(best_h, budget_s)
        if C5_grad < best_C5:
            best_C5 = C5_grad
            best_h = h_grad.copy()
    
    # Final evaluation
    n_points = len(best_h)
    return (best_h.tolist(), best_C5, n_points)
