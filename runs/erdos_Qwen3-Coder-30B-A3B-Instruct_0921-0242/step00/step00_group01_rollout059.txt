def evaluate_erdos_solution(h_values):
    """Returns the C5 constant for the given h_values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute all correlations
    corr_full = correlate(h_values, 1 - np.array(h_values), mode="full")
    # Only consider valid overlaps
    mid = len(corr_full) // 2
    corr_valid = corr_full[mid:]
    # Return the maximum overlap
    return np.max(corr_valid) * dx
