import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize_scalar
import time
import math

def project_capped_simplex(v, total=None):
    """Project vector v onto the capped simplex: 0 <= vi <= 1, sum vi = total."""
    if total is None:
        total = np.sum(v)
    v = np.array(v)
    min_v, max_v = 0.0, 1.0

    def proj_sum(lam):
        return np.sum(np.clip(v - lam, min_v, max_v)) - total

    if proj_sum(0.0) < 0:
        # Sum is already less than total, clip to 1 and renormalize
        proj = np.minimum(v, 1.0)
        proj *= total / np.sum(proj)
        return proj
    elif proj_sum(1.0) > 0:
        # Sum is already more than total, clip to 0 and renormalize
        proj = np.maximum(v, 0.0)
        proj *= total / np.sum(proj)
        return proj

    # Bisection search for lambda
    a, b = 0.0, 1.0
    for _ in range(100):
        lam = (a + b) / 2
        s = proj_sum(lam)
        if s > 0:
            a = lam
        else:
            b = lam
    lam = (a + b) / 2

    proj = np.clip(v - lam, min_v, max_v)
    return proj

def auto_corr(h):
    """Returns auto-correlation array O_k of length n = len(h)"""
    # zero-padded FFT convolution of h with itself
    h_pad = np.concatenate((h, np.zeros_like(h)))
    O = fftconvolve(h_pad, h, mode='valid')
    return O[:len(h)]

def grad_F(beta, h, O, w):
    """Compute ∇F_β(h) = - Σ w_k ∇ O_k(h) using FFT-based convolutions."""
    h_pad = np.concatenate((h, np.zeros_like(h)))
    # forward autograd
    g1 = fftconvolve(w, h_pad, mode='valid')
    g1 = g1[:len(h)]
    
    # backward autograd
    g2 = fftconvolve(w, h_pad[::-1], mode='valid')
    g2 = g2[:len(h)]
    g2 = g2[::-1]  # revert
    
    # combine
    g = g1 + g2
    return -g

def optimize_one_start(n, beta_schedule, time_limit):
    """Optimize a single start from random initialization."""
    np.random.seed()
    start_t = time.time()
    
    # Initial h
    h = np.random.exponential(1.0, n)
    h = h / np.sum(h) * (n / 2)
    h = np.clip(h, 0, 1)
    h = project_capped_simplex(h, total=n/2)
    
    best_c5 = 1e9
    best_h = h.copy()
    
    # Differentiable surrogate
    for t, beta in enumerate(beta_schedule):
        if time.time() - start_t > time_limit - 0.3:
            break
            
        # Inner loop
        O = auto_corr(h)
        w = np.exp(beta * O)
        w /= np.sum(w)
        g = grad_F(beta, h, O, w)
        
        # Line search for step size
        alpha = 0.01
        max_iter = 20
        for i in range(max_iter):
            h_new = h - alpha * g
            h_new = project_capped_simplex(h_new, total=n/2)
            O_new = auto_corr(h_new)
            F_new = 1 - (1 / beta) * np.log(np.sum(np.exp(beta * O_new)))
            F_old = 1 - (1 / beta) * np.log(np.sum(np.exp(beta * O)))
            
            if F_new <= F_old - 1e-4 * alpha * np.dot(g, g):
                break
            else:
                alpha /= 2
        else:
            alpha = max(alpha, 1e-8)
            
        h = h_new
        O = auto_corr(h)
        c5 = 1 - np.min(O)
        if c5 < best_c5:
            best_c5 = min(c5, best_c5)
            best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main entry point for optimization."""
    np.random.seed(seed)
    start_t = time.time()
    best_c5 = 1e9
    best_h = None
    best_n = None
    
    # Multi-restart + multi-resolution strategy
    n_list = [256, 384]
    n_restart_seeds = 8
    time_for_restart = budget_s * 0.2
    time_for_schedule = budget_s * 0.7
    
    beta_schedule = [0.5 * (40/0.5)**(t/11) for t in range(12)]
    
    for n in n_list:
        if time.time() - start_t > budget_s - 0.3:
            break
            
        for i in range(n_restart_seeds):
            if time.time() - start_t > budget_s - 0.3:
                break
                
            h, c5 = optimize_one_start(
                n,
                beta_schedule,
                time_limit=time_for_restart
            )
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_n = n
            if time.time() - start_t > time_for_schedule:
                break
                
    if best_h is None:
        print('No solution found in time budget.')
        return np.array([]), 1.0, 0
        
    # Final polishing
    if time.time() - start_t < budget_s - 0.3:
        # Exact C5 minimization (non-smooth, better to use worst lag gradient)
        def compute_c5(h):
            h = np.array(h)
            dx = 2.0 / len(h)
            corr = np.convolve(h, 1-h, mode='full') * dx
            return np.max(corr)
        
        # Optimize by gradient descent on worst lag
        h = best_h.copy()
        max_iter = 1000
        for i in range(max_iter):
            if time.time() - start_t > budget_s - 0.3:
                break
                
            # Compute correlations
            h = np.array(h)
            dx = 2.0 / len(h)
            corr = np.convolve(h, 1-h, mode='full') * dx
            k_max = np.argmax(corr)
            
            # -∇O_k_max (approximate)
            if k_max < len(h):
                g = np.zeros_like(h)
                for i in range(len(h)):
                    if i + k_max < len(h):
                        g[i] += 1 - h[i + k_max]
                    if i >= k_max:
                        g[i] += h[i - k_max]
            else:
                # this case is rare; compute gradient using direct method
                # not approximating, we're done after few tens iterations
                break
                
            # Step size
            alpha = 1e-6 / np.sqrt(i + 1)
            h_new = h - alpha * g
            h_new = project_capped_simplex(h_new, total=len(h)/2)
            h = h_new
            
        # Final evaluation
        best_c5 = compute_c5(h)
        
    # Return final result
    return h, best_c5, len(h)
