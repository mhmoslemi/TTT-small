def evaluate_overlap_intervals(intervals, k_values):
    """Evaluate the overlap function O(k) for a given list of intervals."""
    m = len(intervals)
    total_overlap = np.zeros_like(k_values)
    
    for i in range(m):
        a_i, b_i = intervals[i]
        for j in range(m):
            a_j, b_j = intervals[j]
            # Compute overlap contribution for pair (i,j)
            # o_ij(k) = max(0, l - |(a_i - a_j) - k|)
            diff = a_i - a_j
            l = b_i - a_i
            for k_idx, k in enumerate(k_values):
                overlap_val = max(0, l - abs(diff - k))
                total_overlap[k_idx] += overlap_val
    
    return total_overlap

def compute_breakpoints(intervals):
    """Compute all breakpoints for the overlap function."""
    m = len(intervals)
    breakpoints = set()
    
    for i in range(m):
        a_i, b_i = intervals[i]
        for j in range(m):
            a_j, b_j = intervals[j]
            diff = a_i - a_j
            l = b_i - a_i
            breakpoints.add(diff - l)
            breakpoints.add(diff + l)
    
    return sorted(list(breakpoints))

def max_overlap_for_intervals(intervals):
    """Return the maximum overlap O(k) for given intervals."""
    breakpoints = compute_breakpoints(intervals)
    # Keep only those in [0,2]
    valid_breakpoints = [k for k in breakpoints if 0 <= k <= 2]
    # Also include 0 and 2
    valid_breakpoints.extend([0, 2])
    valid_breakpoints = sorted(set(valid_breakpoints))
    
    # Evaluate O(k) at all valid breakpoints
    k_vals = np.array(valid_breakpoints)
    overlaps = evaluate_overlap_intervals(intervals, k_vals)
    
    return np.max(overlaps)

def interval_to_discrete(intervals, n):
    """Convert interval representation to discrete step function."""
    dx = 2.0 / n
    h = np.zeros(n)
    
    # Convert interval tuples to actual points
    for a, b in intervals:
        start_idx = int(np.floor(a / dx))
        end_idx = int(np.ceil(b / dx))
        for i in range(start_idx, min(end_idx, n)):
            if (i * dx >= a) and (i * dx < b) or ((i+1)*dx > a) and ((i+1)*dx <= b):
                h[i] = 1.0
                
    # Adjust so that sum(h) = n/2
    total = np.sum(h)
    target = n / 2.0
    if abs(total - target) < 1e-6:
        return h
    
    diff = int(total - target)
    if diff > 0:  # Too many 1's, change some to 0
        # Find the closest points to interval boundaries to change to 0
        boundary_points = set()
        for a, b in intervals:
            boundary_points.add(int(np.floor(a/dx)))
            boundary_points.add(int(np.ceil(b/dx)))
        boundary_points = sorted(list(boundary_points))
        for p in boundary_points:
            if diff <= 0:
                break
            if p < n and h[p] == 1:
                h[p] = 0.0
                diff -= 1
        # If we still (or now) have too many 1s, flip ones that are far from boundaries
        p_indices = np.where(h == 1)[0]
        for p in reversed(p_indices):
            if diff <= 0:
                break
            h[p] = 0.0
            diff -= 1
    elif diff < 0:  # Too few 1's, change some to 1
        # Find points not covered by current intervals but closest to boundaries
        boundary_points = set()
        for a, b in intervals:
            boundary_points.add(int(np.floor(a/dx)))
            boundary_points.add(int(np.ceil(b/dx)))
        boundary_points = sorted(list(boundary_points))
        for p in boundary_points:
            if diff >= 0:
                break
            if p >= 0 and p < n and h[p] == 0:
                h[p] = 1.0
                diff += 1
                
        # Try all indices not yet set if necessary
        p_indices = np.where(h == 0)[0]
        for p in p_indices:
            if diff >= 0:
                break
            h[p] = 1.0
            diff += 1
    
    # Final check
    total = np.sum(h)
    return h

def validate_intervals(intervals):
    """Validate if a list of intervals satisfies all constraints."""
    # Must be non-overlapping, ordered, and total length 1
    if len(intervals) == 0:
        return False
    a_prev = 0.0
    total_length = 0.0
    for a, b in intervals:
        if a < 0 or a > 2 or b < 0 or b > 2:
            return False
        if a >= b:
            return False
        if a < a_prev:
            return False
        length = b - a
        total_length += length
        a_prev = b
    if abs(total_length - 1.0) > 1e-6:
        return False
    return True

def project_onto_valid_order(intervals):
    """Project intervals onto valid ordering and gaps."""
    # Assume intervals are assumed to be in a valid order
    # We just make sure gaps are non-negative
    sorted_intervals = sorted(intervals)
    new_intervals = []
    
    a_prev = 0.0
    for a, b in sorted_intervals:
        # Make sure no overlap
        a = max(a, a_prev)
        if a >= b:
            continue  # Skip overlapping interval
        new_intervals.append((a, b))
        a_prev = b
    
    if sum(b - a for a, b in new_intervals) != 1.0:
        # May need to rescale to exactly length 1
        total_length = sum(b - a for a, b in new_intervals)
        factor = 1.0 / total_length
        new_intervals = [(a, a + (b - a) * factor) for a, b in new_intervals]
    
    return new_intervals

def evaluate_h(h):
    """Evaluate overlap integral C5 for discrete function h."""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    overlap = np.correlate(h, h_rev, mode='full')
    max_overlap = np.max(overlap) * dx
    return max_overlap

def evaluate_objective(intervals):
    """Evaluate objective function for intervals."""
    global t0, budget_s
    if time.time() - t0 > budget_s - 5:
        raise RuntimeError("Timeout")
    
    try:
        # Project to a valid ordering
        valid_intervals = project_onto_valid_order(intervals)
        if len(valid_intervals) == 0 or not validate_intervals(valid_intervals):
            # Penalize invalid projections by pushing them to max
            return 1.0
        
        # Evaluate object
        max_overlap = max_overlap_for_intervals(valid_intervals)
        
        # Convert to discrete and evaluate C5
        n_vals = [30, 36, 42, 48, 60]
        best_h = None
        best_C5 = float('inf')
        best_n = None
        
        for n in n_vals:
            # Discretize to a resolution
            h = interval_to_discrete(valid_intervals, n)
            C5 = evaluate_h(h)  # This should be the complement
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
                best_n = n
                
        global best_h as be_h, best_C5 as be_C5 
        be_h = best_h
        be_C5 = best_C5
        best_n = best_n
        
        # return C5 directly as "the same" as max overlap
        return max_overlap   # max O(k) = Minimize this over all intervals
        
    except Exception as e:
        # If any transformation fails, ignore (try again later)
        return 1.0

def binary_improve(h): 
    """Perform binary swaps to decrease the maximum value of overlap."""
    global t0, budget_s
    if time.time() - t0 > budget_s - 5:
        return h, float('inf')
        
    corr = np.correlate(h, h[::-1], mode="full")
    k_star = np.argmin(corr)
    corr_k_star = corr[k_star]
    max_corr = np.max(corr)  

    # We want to minimize max_overlap = 1 - corr_k_star
    # That means we maximize corr_k_star to decrease C5 value
    
    best_update = 0
    best_i, best_j = -1, -1
    max_iter = 500
    restarts = 3
    last_corr = corr.copy()
    
    best_h = h.copy()
    best_C5 = max_corr
    
    for rep in range(restarts):
        current_h = h.copy()
        corr = np.correlate(current_h, current_h[::-1], mode="full")
        corr_k_star = corr[k_star]
        
        for it in range(max_iter):
            if time.time() - t0 > budget_s - 5:
                break
                
            i = np.random.randint(0, len(h))
            j = np.random.randint(0, len(h))
            
            # Only swap 1 and 0
            if current_h[i] == current_h[j]:
                continue
                
            # Swap
            current_h[i], current_h[j] = current_h[j], current_h[i]  
            
            # Compute new overlap contribution
            # For only ~k_star, effectively, we should check 
            corr_new = np.correlate(current_h, current_h[::-1], mode="full")            
            new_k_star = np.argmin(corr_new)
            new_corr_k_star = corr_new[new_k_star] if new_k_star < len(corr_new) else 0
            
            # Best to improve max_corr of the full function:
            delta_corr = new_corr_k_star - corr_k_star
            if delta_corr < best_update:
                best_update = delta_corr
                best_i, best_j = i, j
                best_h = current_h.copy()
                corr = corr_new.copy()
                corr_k_star = new_corr_k_star
            else:
                # Revert
                current_h[i], current_h[j] = current_h[j], current_h[i]
    
    # Return best improvement
    updated_h = best_h
    updated_corr = np.correlate(updated_h, updated_h[::-1], mode="full")
    updated_k_star = np.argmin(updated_corr)
    updated_C5 = updated_corr[updated_k_star]
    return updated_h, updated_C5

# DE-style single-stage evaluation  
def phase_b_de(intervals, m):
    """Run DE on interval parameters."""
    global t0, budget_s, best_h, best_C5
    
    def fun_to_optimize(x):
        # x is array of a_i values
        # Assuming we work in reverse indices, a_i = x[i]
        intervals_trial = [(x[i], x[i] + 1./m) for i in range(m)]
        try:
            return evaluate_objective(intervals_trial)
        except:
            return 1.0

    # Set range bounds for [0, 2]
    bounds = [(0.0, 2.0 - 1./m) for _ in range(m)]
    
    res = differential_evolution(fun_to_optimize, bounds, seed=42, maxiter=100, 
                                mutation=(0.5, 1.0), recombination=0.8, 
                                disp=False, polish=False)  
    
    intervals_opt = [(res.x[i], res.x[i] + 1./m) for i in range(m)]
    try:
        C5_val = evaluate_objective(intervals_opt)
        return intervals_opt, C5_val
    except:
        return intervals, 1.0
