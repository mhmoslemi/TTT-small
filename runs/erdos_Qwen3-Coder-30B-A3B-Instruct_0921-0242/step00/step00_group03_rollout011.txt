import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar
from scipy.sparse import diags

def run(seed=42, budget_s=600, **kwargs):

    n_points = 240
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if time.time() - start_time > budget_s - 30:

    print(f"Solving SDP for n={n_points}...")
    h = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()

    # Constraints for SDP
    constraints = [
        cp.diag(X) == h,  # This ensures X_{ii} = h_i since h_i in [0,1], X_{ii} = h_i^2 = h_i
        X >> 0,  # X is PSD
        cp.sum(h) == target_sum,
        0 <= h,
        h <= 1,
        t >= 0
    ]

    # Add correlation constraints for all shifts
    for shift in range(-n_points + 1, n_points):
        # [0, 1, ..., n-1], we must shift indices properly
        P_shift = np.zeros((n_points, n_points))
        if shift >= 0:
            P_shift[shift:, :n_points-shift] = np.eye(n_points - shift)
        else:
            P_shift[:n_points+shift, -shift:] = np.eye(n_points + shift)

        # However, for convolution-like constraint, compute trace correctly
        # Stricter implementation follows
        corr = np.zeros(n_points)
        corr[shift:] = np.ones(n_points - shift) if shift < 0 else np.ones(n_points - shift)
        corr = corr[:n_points]  # truncate if necessary

        # OR Better: use shift operator matrix
        # build a permutation matrix for the shift
        shift_idx = np.arange(n_points)
        shift_idx = (shift_idx - shift) % n_points
        Ps = np.zeros((n_points, n_points))
        Ps[np.arange(n_points), shift_idx] = 1

        # Add constraint t ≥ const - dx * Tr(Ps * X)  
        const_term = dx * target_sum
        constr = t >= const_term - dx * cp.trace(Ps @ X)
        constraints.append(constr)

    # Set up problem
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=20000, 
                   eps_abs=1e-6, eps_rel=1e-6)
    except Exception as e:
        print(f"SDP Solver failed with error: {e}")
        return initial_h_values, evaluate_erdos_solution(initial_h_values), len(initial_h_values)
    
    if prob.status != cp.OPTIMAL:
        print(f"SDP Status: {prob.status}")
        # Fallback
        test_h = np.clip(initial_h_values, 0, 1)
        test_h = test_h * (target_sum / np.sum(test_h))
        return test_h, evaluate_erdos_solution(test_h), len(test_h)

    h_opt = h.value
    print(f"SDP Lower bound: {t.value:.6f}")
    relax_bound = t.value
    
    # Rounding steps
    best_h = None
    best_c5 = float('inf')
    best_n = n_points
    rounds = 30
    print(f"Performing {rounds} rounds of randomized rounding...")
    
    for _ in range(rounds):
        # Eigen-decomposition of solution
        X_s = X.value
        try:
            _, U = np.linalg.eigh(X_s)
        except np.linalg.LinAlgError:
            print("Cannot perform Eigh, using pseudo-inverse approach.")
            U = np.eye(n_points)  # fallback, terrible but safe

        g = np.random.randn(n_points)
        y = U @ (np.diag(np.sqrt(np.maximum(0, np.linalg.eigvals(X_s)))) @ g)
        
        # Clip y to [0,1]
        y_clipped = np.clip(y, 0, 1)
        # Scale to sum to target
        current_sum = np.sum(y_clipped)
        deficit = target_sum - current_sum

        if abs(deficit) < 1e-8:
            h_rounded = y_clipped
        else:
            # Distribute deficit
            mask = (y_clipped > 0) & (y_clipped < 1)
            weights = y_clipped[mask]
            weights /= np.sum(weights)
            to_add = np.zeros_like(y_clipped)
            to_add[mask] = weights * deficit
            h_rounded = y_clipped + to_add
            
        # Normalize again to ensure sum constraint exactly
        h_rounded = np.clip(h_rounded, 0, 1)
        scale_factor = target_sum / np.sum(h_rounded)
        h_rounded *= scale_factor
        
        # Final validation
        h_rounded = np.maximum(0, np.minimum(1, h_rounded))
        
        val = evaluate_erdos_solution(h_rounded)
        if val < best_c5:
            best_c5 = val
            best_h = h_rounded
            best_n = n_points

        if time.time() - start_time > max_time - 30:
            break

    print(f"Best after rounding: {best_c5:.6f}")

    # Try refinement with coordinate search
    if best_h is not None:
        print("Starting coordinate descent refinement...")
        refined_h = best_h.copy()

        def compute_corr(h_in):
            return np.correlate(h_in, 1-h_in, mode='full') * dx

        def objective(h_in):
            corr = compute_corr(h_in)
            return np.max(corr)

        def update_coordinate(i, h_in, step_size=1e-3):
            h_new = h_in.copy()
            h_new[i] += step_size
            avg = np.sum(h_in) / n_points
            h_new += (avg - np.sum(h_new) / n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            return h_new
        
        def compute_max(shift_indices, h_in):
            corr_vals = np.correlate(h_in, 1-h_in, mode='full') * dx
            return np.max(corr_vals[shift_indices])
        
        max_iter = 2000
        iters = 0

        while iters < max_iter and time.time() - start_time < max_time - 30:
            improved = False
            # Loop over all indices
            for i in range(n_points):
                if not improved:
                    break
                step = 1e-3
                old_val = objective(refined_h)
                candidates = []
                for direction in [-1, 1]:
                    test_h = refined_h.copy()
                    test_h[i] += step * direction
                    
                    if test_h[i] < 0 or test_h[i] > 1:
                        continue

                    # Adjust remaining to maintain sum constraint
                    delta_sum = test_h[i] - refined_h[i]
                    avg = np.sum(refined_h) / n_points
                    adjusted_sum = np.sum(refined_h) + delta_sum
                    correction = target_sum - adjusted_sum

                    # Recompute
                    test_h[i] = max(0, min(1, test_h[i]))
                    test_h = test_h + correction / n_points
                    test_h = np.clip(test_h, 0, 1)
                    test_h = test_h * (target_sum / np.sum(test_h))
                    new_val = objective(test_h)
                    if new_val < old_val - 1e-8:
                        candidates.append((test_h.copy(), new_val, direction))

                if candidates:
                    h_new, new_val, _ = min(candidates, key=lambda x: x[1])
                    if new_val < old_val - 1e-8:
                        refined_h = h_new
                        improved = True
                        iters += 1
        
        final_c5 = objective(refined_h)
        print(f"Refined to: {final_c5:.6f}")
    else:
        final_c5 = best_c5
        refined_h = best_h

    if refined_h is not None:
        # Final projection to ensure constraint
        mean_val = np.sum(refined_h) / n_points
        refined_h = refined_h * (target_sum / np.sum(refined_h))
        refined_h = np.clip(refined_h, 0, 1)
        final_c5 = evaluate_erdos_solution(refined_h)
    else:
        # Fall back
        refined_h = best_h if best_h is not None else initial_h_values.copy()
        final_c5 = evaluate_erdos_solution(refined_h)

    return refined_h, final_c5, n_points
