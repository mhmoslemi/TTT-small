def project_simplex(h, target_sum):
    """Project h onto the simplex with given target sum using bounded simplex projection."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    
    if abs(delta) < 1e-12:
        return h.copy()
    
    # Create sorted indices
    sorted_indices = np.argsort(h)[::-1]
    
    # Keep track of free variables
    free_set = set(range(len(h)))
    
    while abs(delta) > 1e-12 and len(free_set) > 0:
        # Compute the amount we can change
        # We want to increase the values in free set by delta/len(free_set)
        delta_change = delta / len(free_set)
        
        # Apply change to all free variables
        new_h = h.copy()
        for i in list(free_set):
            new_h[i] += delta_change
            if new_h[i] > 1:
                new_h[i] = 1.0
                free_set.remove(i)
            elif new_h[i] < 0:
                new_h[i] = 0.0
                free_set.remove(i)
        
        # Recompute deficit
        current_sum = np.sum(new_h)
        delta = target_sum - current_sum
        h = new_h
        
    return h

def compute_autocorrelations(h):
    """Compute linear autocorrelations and their indices."""
    n = len(h)
    # Pad with zeros
    h_padded = np.pad(h, (0, n), mode='constant')
    H = fftpack.fft(h_padded)
    autocorr = fftpack.ifft(H * np.conj(H)).real[:n]
    # Return correlation values from 1 to n-1
    return autocorr[1:]

def compute_S_alpha(h, alpha):
    """Compute soft minimum of r_k values."""
    r = compute_autocorrelations(h)
    # Avoid computing log of near-zero exps:
    r = r[r > 0]
    if len(r) == 0:
        return -1e6
    # Prevent numerical overflow  
    r_alpha = -alpha * r - np.max(-alpha * r)
    exp_r = np.exp(r_alpha)
    sum_exp = np.sum(exp_r)
    if sum_exp <= 0:  # unlikely but safe
        return -1e6
    s = -(1.0 / alpha) * np.log(sum_exp)
    return s

def nesterov_update(h, g, h_prev, beta):
    """Perform Nesterov-style update."""
    y = h + beta * (h - h_prev)
    return y

def line_search_step(f, df, x, d, max_iter=10):
    """Simple backtracking line search."""
    alpha = 1.0
    c1 = 1e-4
    rho = 0.5
    
    fx = f(x)
    gt = df(x)
    
    while True:
        x_new = x + alpha * d
        f_new = f(x_new)
        slope = np.dot(gt, d)
        
        if f_new <= fx + c1 * alpha * slope:
            return alpha, x_new
        if alpha < 1e-10:
            return alpha, x
        alpha *= rho

def projected_gradient_ascent(h0, alpha, max_iter=200, tol=1e-9):
    """Projected gradient ascent with Nesterov optimization."""
    h = h0.copy()
    h_prev = h0.copy()
    
    best_h = h.copy()
    best_S = compute_S_alpha(h, alpha)
    
    beta = 0.9  # Nesterov momentum
    tol_count = 0
    
    for i in range(max_iter):
        g = grad_s_alpha(h, alpha)
        
        # Nesterov update
        y = nesterov_update(h, g, h_prev, beta)
        
        # Line search
        step_size, h_new = line_search_step(
            lambda x: compute_S_alpha(x, alpha),
            lambda x: grad_s_alpha(x, alpha),
            y, g, max_iter=5
        )
        
        h_old = h.copy()
        h = h_new.copy()
        
        # Project onto feasible set
        target_sum = len(h) * 0.5  # because integral = 1 and dx = 2/n
        h = project_simplex(h, target_sum)
        
        # Check improvement
        current_S = compute_S_alpha(h, alpha)
        if current_S > best_S:
            best_S = current_S
            best_h = h.copy()
            
        # Convergence check
        if abs(current_S - best_S) < tol:
            tol_count += 1
        else:
            tol_count = 0
        
        if tol_count > 20:
            break
            
        h_prev = h_old
        
    return best_h, best_S

def round_to_binary(h, target_sum):
    """Round continuous h to binary step function."""
    # Determine positions of maximum entries to set to 1
    n = len(h)
    indices = np.argsort(h)[::-1]
    # Set exactly `target_sum` elements to 1
    binary = np.zeros(n)
    binary[indices[:int(target_sum)]] = 1.0
    return binary
