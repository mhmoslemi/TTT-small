def project_simplex(h, target_sum):
    """Project h onto the simplex with sum equal to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    if abs(delta) < 1e-12:
        return h
    
    # Sort indices by h values
    idx_sorted = np.argsort(h)[::-1]
    
    # Process from largest to smallest
    running_sum = 0
    for i in range(len(h)):
        idx = idx_sorted[i]
        running_sum += h[idx]
        if i+1 < len(h):
            gap = h[idx_sorted[i+1]]
        else:
            gap = 0
        if current_sum - running_sum + (i+1) * (h[idx] - gap) >= target_sum:
            break
    else:
        i += 1
    
    threshold = (current_sum - target_sum) / (i + 1e-12)
    h = np.clip(h - threshold, 0, 1)
    
    # Rescale if needed
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-12:
        h = h * target_sum / current_sum
    
    return h

def soft_min_gradient(h, r, alpha):
    """Compute gradient of soft minimum with temperature alpha."""
    r = np.array(r)
    # Use log-sum-exp trick to avoid numerical overflow
    max_r = np.max(-alpha * r)
    if max_r > 100:
        # Avoid numerical overflow
        r_shifted = -alpha * (r - max_r)
        exp_r = np.exp(r_shifted)
    else:
        exp_r = np.exp(-alpha * r)
    
    weights = exp_r / np.sum(exp_r)
    
    # Gradient is weighted sum of shifted copies of h
    # w * h + reverse(w) * h where * is convolution
    w = weights
    # Pad with zeros for linear convolution
    n = len(h)
    h_ext = np.concatenate([h, np.zeros(n)])
    w_ext = np.concatenate([w, np.zeros(n)])
    
    # This effectively computes w * h using FFT for efficient linear convolution with zero-padding
    fft_h = fftpack.fft(h_ext)
    fft_w = fftpack.fft(w_ext)
    grad = np.real(fftpack.ifft(fft_h * np.conj(fft_w)))[:n]
    
    # Do the same for reverse(w) * h (which is reverse(w) conv h)  
    w_rev = np.flip(w)
    fft_w_rev = fftpack.fft(w_rev)
    grad_rev = np.real(fftpack.ifft(fft_h * np.conj(fft_w_rev)))[:n]
    
    gradient = grad + grad_rev
    return gradient

def compute_autocorrelations(h):
    """Compute linear autocorrelations r_k for all k."""
    n = len(h)
    # Pad with zeros to get proper linear convolution (zero-padding)
    h_pad = np.concatenate([h, np.zeros(n)])
    fft_h = fftpack.fft(h_pad)
    cross = fft_h * np.conj(fft_h)
    r = np.real(fftpack.ifft(cross))[:n]
    return r[1:n]  # k = 1, ..., n-1

    return (best_h, final_c5, best_n)

def optimize_single_h(h, n, dx, target_sum, budget_rem):
    """Carry out projected gradient with soft minimum surrogate."""
    # Optimization parameters  
    alpha_start = 0.5
    alpha_max = 80
    alpha_factor = 2
    max_iter_per_alpha = 500
    early_stop_tol = 1e-9
    nesterov_beta = 0.9
    
    h_best = h.copy() 
    c5_best = evaluate_f(h_best)
    alpha = alpha_start
    
    # Prepare FFT buffers
    h_fft = np.zeros(2*n, dtype=np.complex128)
    h_ext = np.zeros(2*n, dtype=np.float64)
    
    # Start timing
    start = time.time()
    
    while alpha <= alpha_max and (time.time() - start) < budget_rem:
        # Run gradient ascent
        prev_h = np.zeros_like(h)
        prev_c5 = float('inf')
        
        for iter in range(max_iter_per_alpha):
            if (time.time() - start) > budget_rem:
                break
                
            # Compute r_k = ∫ h(x)h(x+k) dx  
            r_values = compute_autocorrelations(h)  # Returns r_1..r_{n-1}
            # Which is from r_k for indices 1 to n-1 used down below.  
            if not r_values.size:
                break
                
            # Compute gradient based on augmented surrogate 
            grad = soft_min_gradient(h, r_values, alpha)
            
            # Projected gradient descent (with Nesterov step)
            y = h + nesterov_beta * (h - prev_h)  # Nesterov extrapolation  
            g = grad
            # Project y
            y = project_simplex(y, target_sum)
            
            # Backtrack line search
            step_size = 1.0
            while step_size > 1e-10:
                h_new = project_simplex(y - step_size * g, target_sum)
                c5_new = evaluate_f(h_new)
                # Armijo condition check
                if c5_new <= c5_best + 1e-6:  # tolerance for early
                    h = h_new.copy()
                    break
                step_size *= 0.5  # Reduce step size
            
            prev_h = h.copy()
            
            # Update best
            c5_current = evaluate_f(h)
            if c5_current < c5_best:
                c5_best = c5_current  
                h_best = h.copy()
                
            # Check for convergence  
            if abs(prev_c5 - c5_current) < early_stop_tol or iter > 100:
                break
                
        # Increase alpha 
        alpha *= alpha_factor
        
    # Return final best
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    h_values, c5_bound, n_points = main_erdos_optimization(seed, budget_s)
    return (h_values, c5_bound, n_points)
