import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.special import logsumexp

def decode(x, M):
    """Recover segment lengths u and heights a from the transformed variables x."""
    u = x[:M]
    a0 = x[M]
    v = x[M+1:]
    a = np.empty(M+1)
    a[0] = a0
    for j in range(1, M):
        a[j] = max(0, a[j-1] - v[j-1])
    a[M] = max(0, a[M-1] - v[M-1])
    return u, a

def compute_breakpoints(u, a):
    """Compute breakpoints s from segment lengths u and heights a."""
    s = np.empty(len(u)+1)
    s[0] = 0
    for j in range(len(u)):
        s[j+1] = s[j] + u[j]
    return s

def overlap_piecewise(s, a):
    """Compute exact C5 for a piecewise constant function h defined by breakpoints s and heights a."""
    M = len(a) - 1
    # All possible breakpoints where F(k) changes slope:
    K = []
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0 <= k1 <= 2:
                K.append(k1)
            if 0 <= k2 <= 2:
                K.append(k2)
    K = sorted(set(K + [0, 2]))
    
    # Compute for each k in K:
    C5_vals = []
    for k in K:
        F_k = 0.0
        for j in range(M+1):
            for l in range(M+1):
                s_j, s_jp1 = s[j], s[j+1]
                s_l, s_lp1 = s[l], s[l+1]
                # Intersection length goes to 0 if s_jp1 <= s_l - k or s_j >= s_lp1 - k
                # Otherwise it is min(s_jp1, s_lp1 - k) - max(s_j, s_l - k)
                left = max(s_j, s_l - k)
                right = min(s_jp1, s_lp1 - k)
                if right > left:
                    L_jl = right - left
                    F_k += a[j] * (1 - a[l]) * L_jl
        C5_vals.append(F_k)
    return max(C5_vals)

def project_capped_simplex(v, total):
    """Project v onto the capped simplex such that sum(v) = total."""
    v = np.asarray(v)
    s = np.sum(v)
    if abs(s - total) < 1e-10: 
        return v
    # Use binary search for the Lagrange multiplier
    if s < total:
        return v + (total - s) / len(v)
    else:
        # Project to simplex first, then to capped simplex
        # Simplex: x_i >= 0 and sum x_i = total
        def proj_simplex(x):
            # sort and find threshold
            sorted_x = np.sort(x)[::-1]
            cumsums = np.cumsum(sorted_x)
            n_sorted = len(cumsums)
            theta = 0
            found = False
            for i in range(n_sorted):
                if sorted_x[i] - (cumsums[i] - total) / (i + 1) > 0:
                    theta = (cumsums[i] - total) / (i + 1)
                    found = True
                    break
            if not found:
                # All will be zero
                return np.zeros_like(x)
            # Apply threshold
            y = np.maximum(x - theta, 0)
            return y
        # Project to simplex, then to cap
        proj = proj_simplex(v)
        need = total - np.sum(proj)
        if need > 0:
            # We need to increase some values, all equal
            to_add = need / np.count_nonzero(proj)
            proj += to_add * (proj > 0)
        else:
            # If we have excess (due to numerical errors)
            return proj
        # Ensure exact sum: adjust all equal values
        # But just do it correctly with sigmoid and add any remaining error evenly
        return proj

def evaluate_fitness(x, M, n_points, penalize_integral=True):
    """Evaluate fitness of a candidate solution."""
    try:
        u, a = decode(x, M)
        s = compute_breakpoints(u, a)
        # Check bounds
        if np.any(u < 0) or np.any(u > 2) or np.any(a < 0) or np.any(a > 1):
            return 1e6  # penalty
        integral_residual = np.sum(a*u) - 1.0
        if abs(integral_residual) > 1e-8 and penalize_integral:
            penalty = 1e6 * (integral_residual**2)
        else:
            penalty = 0.0
        C5 = overlap_piecewise(s, a)
        return C5 + penalty
    except:
        return 1e6

def solve_coarse(budget_s):
    M = 4
    n_points = 128
    dx = 2.0 / n_points
    bounds = [(0,2) for _ in range(M)] + [(0,1)] + [(0,1) for _ in range(M-1)]
    def func(x):
        return evaluate_fitness(x, M, n_points)
    
    start = time.time()
    res = differential_evolution(func, bounds, seed=42, maxiter=200, popsize=15*(M+1))
    elapsed = time.time() - start
    if elapsed >= 0.8 * budget_s:
        # Sub-optimal but we have a result
        pass
    return res.x, M, n_points

def solve_fine(budget_s, init_x, M=6):
    n_points = 256
    dx = 2.0 / n_points
    bounds = [(0,2) for _ in range(M)] + [(0,1)] + [(0,1) for _ in range(M-1)]
    def func(x):
        return evaluate_fitness(x, M, n_points)
    
    start = time.time()
    # Initialize population around the best from coarse
    init_pop = [init_x]
    for _ in range(14 * (M+1)):
        # Sample neighbors in box
        sample = []
        for i in range(len(bounds)):
            lb, ub = bounds[i]
            s = np.random.uniform(lb, ub)
            sample.append(s)
        init_pop.append(sample)
        
    res = differential_evolution(func, bounds, seed=42, maxiter=200, popsize=15*(M+1), 
                                init=init_pop, callback=None)
    elapsed = time.time() - start
    if elapsed >= 0.8 * budget_s:
        # Sub-optimal but we have a result  
        pass
    return res.x, M, n_points

def refine_solution(x_opt, M, n_points):
    """Use local refinement: logsumexp smoother then SLSQP then subgradients."""
    dx = 2.0 / n_points
    u, a = decode(x_opt, M)
    s = compute_breakpoints(u, a)
    integral_residual = np.sum(a*u) - 1.0
    if abs(integral_residual) > 1e-8:
        # Possibly penalize more
        pass

    def smooth_fitness(x, beta=50):
        """Smooth approximation to max using log-sum-exp."""
        u_, a_ = decode(x, M)
        s_ = compute_breakpoints(u_, a_)
        # Compute overlaps at all breakpoints:
        K = []
        for j in range(M+1):
            for l in range(M+1):
                k1 = s_[l] - s_[j+1]
                k2 = s_[l+1] - s_[j]
                if 0 <= k1 <= 2:
                    K.append(k1)
                if 0 <= k2 <= 2:
                    K.append(k2)
        K = sorted(set(K + [0, 2]))
        F_vals = []
        for k in K:
            F_k = 0.0
            for j in range(M+1):
                for l in range(M+1):
                    s_j, s_jp1 = s_[j], s_[j+1]
                    s_l, s_lp1 = s_[l], s_[l+1]
                    left = max(s_j, s_l - k)
                    right = min(s_jp1, s_lp1 - k)
                    if right > left:
                        L_jl = right - left
                        F_k += a_[j] * (1 - a_[l]) * L_jl
            F_vals.append(F_k)
        return -logsumexp(F_vals, b=beta) / beta

    # SLSQP iteration:
    bounds = [(0,2) for _ in range(M)] + [(0,1)] + [(0,1) for _ in range(M-1)]
    x0 = x_opt.copy()
    constr = {'type': 'eq', 'fun': lambda x: np.sum(decode(x, M)[1] * decode(x, M)[0]) - 1.0}

    # Use final solution from previous stage as initial point
    # Local call to minimize via SLSQP
    try:
        res = minimize(smooth_fitness, x0, method='SLSQP', bounds=bounds, constraints=[constr], 
                       options={'maxiter': 100})
        if res.success:
            x_opt = res.x
    except:
        pass

    # Subgradient descent to refine current estimate of corner
    return x_opt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Coarse solution M = 4
    x_coarse, M_coarse, n_points_coarse = solve_coarse(budget_s * 0.3)
    best_x = x_coarse
    best_M = M_coarse
    best_n = n_points_coarse

    elapsed = time.time() - start_time
    if elapsed >= 0.9 * budget_s:
        # Fallback to make sure we return something
        pass
    
    # Phase 2: Fine solution, but only if we have time
    if elapsed < 0.8 * budget_s:
        x_fine, M_fine, n_points_fine = solve_fine(budget_s, x_coarse, M=6)
        best_x = x_fine
        best_M = M_fine
        best_n = n_points_fine
        elapsed = time.time() - start_time

    # Phase 3: Local refinement
    if elapsed < 0.95 * budget_s:
        try:
            x_refined = refine_solution(best_x, best_M, best_n)
            best_x = x_refined
        except:
            pass

    # Final vector
    u, a = decode(best_x, best_M)
    s = compute_breakpoints(u, a)
    
    # Make discrete h_values
    h_values = []
    dx = 2.0 / best_n
    for i in range(best_n):
        x = (i + 0.5) * dx
        # Find relevant segment
        j = 0
        while j < len(s) - 1 and s[j+1] <= x:
            j += 1
        h_values.append(a[j])

    # Exact integral adjustment
    corrected_h = project_capped_simplex(np.array(h_values), total=best_n/2)
    
    # Evaluate final C5 using full convolution
    corr = np.correlate(corrected_h, 1 - corrected_h, mode='full') * dx
    c5_bound = np.max(corr)
    
    return (corrected_h.tolist(), float(c5_bound), best_n)
