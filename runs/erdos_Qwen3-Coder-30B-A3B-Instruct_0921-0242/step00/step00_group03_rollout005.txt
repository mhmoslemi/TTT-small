import numpy as np
import scipy.fft as fft
import time
from scipy.optimize import minimize
from math import floor

# Predefined initial_h_values from the environment
# Initial solution for reference
initial_h_values = None

def project_to_F(h, n):
    """ Projects vector h onto the feasible set {h : 0 <= h_i <= 1, sum(h) = n/2} """
    h = np.clip(h, 0.0, 1.0)
    deficit = n/2 - h.sum()
    free = np.where((h>0) & (h<1))[0]
    while np.abs(deficit) > 1e-12 and free.size>0:
        delta = deficit / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        deficit = n/2 - h.sum()
        free = np.where((h>0) & (h<1))[0]
    return h

def convolve_fast(h, g, dx):
    """ Fast correlation using FFT """
    L = 2 * len(h)
    h_pad = np.pad(h, (0, L - len(h)), mode='constant')
    g_pad = np.pad(g, (0, L - len(g)), mode='constant')
    H = fft.fft(h_pad)
    G = fft.fft(g_pad)
    conv = fft.ifft(H * np.conj(G)).real
    return conv[:2*len(h)-1] * dx

def compute_C5(h, dx):
    """ Computes C5 = max_k int h(x)(1 - h(x + k)) dx using fast FFT """
    g = 1 - h
    conv = convolve_fast(h, g, dx)
    return conv.max()

def c5_gradient(h, k_star, dx):
    """ Computes gradient of C5 with respect to h for active shift k* """
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        i_k = i + k_star
        i_neg_k = i - k_star
        grad[i] = dx * ((1 - h[i_k] if 0 <= i_k < n else 0) - (h[i_neg_k] if 0 <= i_neg_k < n else 0))
    return grad

def local_improve(h, dx, max_steps=30):
    """ Memetic local improvement via coordinate swaps """
    g = 1 - h
    conv = convolve_fast(h, g, dx)
    k_star = np.argmax(conv)
    c_orig = conv[k_star]
    c = conv.copy()
    
    for _ in range(max_steps):
        # Choose two indices
        # i1 must have h[i1] > 0, i0 must have h[i0] < 1
        i1 = np.random.choice(np.where(h > 0)[0])
        i0 = np.random.choice(np.where(h < 1)[0])
        
        # Compute epsilon
        epsilon = min(h[i1], 1 - h[i0], 0.05)
        if epsilon < 1e-6:
            continue
            
        # Propose new h
        h_new = h.copy()
        h_new[i1] -= epsilon
        h_new[i0] += epsilon
        
        # Compute updated correlation incrementally
        new_conv = c.copy()
        for k in range(2*len(h)-1):
            # Indices affected by changes at i1 and i0
            i1_shift = k - i1
            i0_shift = k - i0
            # If indices are valid
            if 0 <= i1_shift < len(h):
                new_conv[k] -= epsilon * dx * (g[i1_shift] if 0 <= i1_shift < len(h) else 0)
            if 0 <= i0_shift < len(h):
                new_conv[k] += epsilon * dx * (g[i0_shift] if 0 <= i0_shift < len(h) else 0)
        
        new_c_star = new_conv.max()
        if new_c_star < c_orig - 1e-10:
            h = h_new.copy()
            c = new_conv.copy()
            c_orig = new_c_star
        else:
            # Early stopping if no substantial improvement
            if new_c_star >= c_orig - 1e-8:
                break
    return h

def initialize_population(n, pop_size, initial_seed):
    """ Creates initial population diverse from the seed """
    population = []
    
    # Seed 1: Interpolated from initial_h_values
    if initial_seed is not None:
        orig_len = len(initial_seed)
        x_orig = np.linspace(0, 2, orig_len)
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_orig, initial_seed)
        h_interp = project_to_F(h_interp, n)
        population.append((h_interp, compute_C5(h_interp, 2.0/n)))

    # Seed 2: Random continuous
    h_rand = np.random.rand(n)
    h_rand = project_to_F(h_rand, n)
    population.append((h_rand, compute_C5(h_rand, 2.0/n)))

    # Seed 3: Binary block
    h_block = np.zeros(n)
    h_block[:n//2] = 1.0
    np.random.shuffle(h_block)
    h_block = project_to_F(h_block, n)
    population.append((h_block, compute_C5(h_block, 2.0/n)))

    # Seed 4: Gaussian
    h_gauss = np.full(n, 0.5) + np.random.normal(0, 0.1, n)
    h_gauss = project_to_F(h_gauss, n)
    population.append((h_gauss, compute_C5(h_gauss, 2.0/n)))

    # Remaining random individuals
    for _ in range(pop_size - 4):
        h = np.random.rand(n)
        h = project_to_F(h, n)
        population.append((h, compute_C5(h, 2.0/n)))
    
    return population

def blend_crossover(p1, p2, alpha):
    """ Blend crossover """
    child = alpha * p1 + (1 - alpha) * p2
    return child

def mutate(h, n, sigma):
    """ Gaussian noise mutation """
    m = max(1, int(0.02 * n))
    indices = np.random.choice(n, size=m, replace=False)
    noise = np.random.normal(0, sigma, m)
    h_mutated = h.copy()
    h_mutated[indices] += noise
    return h_mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    
    n_points = 240
    dx = 2.0 / n_points
    pop_size = 30
    elite_size = 2
    
    # Initiates timing
    start = time.time()
    
    # Initial population
    population = initialize_population(n_points, pop_size, initial_h_values)
    
    # Main evolutionary loop
    sigma = 0.2
    gen = 0
    max_generations = 2000
    
    h_best = population[0][0]
    c5_best = population[0][1]
    previous_c5_best = c5_best + 1.0
    
    while time.time() - start < budget_s - 20 and gen < max_generations:
        gen += 1
        
        # Select parents using tournament
        parents = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(len(population), 3, replace=False)
            tournament = [population[i] for i in tournament_indices]
            best_in_tournament = min(tournament, key=lambda x: x[1])
            parents.append(best_in_tournament[0])
            
        # Generate offsprings
        offsprings = []
        for i in range(pop_size):
            p1 = parents[i]
            p2 = parents[(i + 1) % pop_size]
            alpha = np.random.rand()
            child = blend_crossover(p1, p2, alpha)
            child = project_to_F(child, n_points)
            child = mutate(child, n_points, sigma)
            child = project_to_F(child, n_points)
            
            # Memetic local improvement
            child = local_improve(child, dx)
            
            # Compute fitness
            c5 = compute_C5(child, dx)
            
            offsprings.append((child, c5))
            
        # Merge population with offspring and elite
        all_individuals = population[0:elite_size] + offsprings
        all_individuals.sort(key=lambda x: x[1])
        population = all_individuals[:pop_size]
        
        # Update global best
        best_c5 = population[0][1]
        if best_c5 < c5_best:
            h_best = population[0][0].copy()
            c5_best = best_c5
        
        # Adaptive mutation
        sigma *= 0.995
        if sigma < 0.02:
            sigma = 0.1
            # Randomize half
            half_pop = pop_size // 2
            for i in range(half_pop):
                h_rnd = np.random.rand(n_points)
                population[i] = (project_to_F(h_rnd, n_points), compute_C5(h_rnd, dx))

        # Yield progress if necessary
        if time.time() - start > 5:
            pass  # progress update is optional

    # Final SLSQP refinement
    def obj(h):
        c5 = compute_C5(h, dx)
        return c5
    
    def con_eq(h):
        return np.sum(h) - n_points / 2

    def jac(h):
        # Use finite difference gradient from current C5 evaluation if needed
        g = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = project_to_F(h_plus, n_points)
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            h_minus = project_to_F(h_minus, n_points)
            g[i] = (obj(h_plus) - obj(h_minus)) / (2e-6)
        return g
    
    h_opt = h_best.copy()
    h_opt = project_to_F(h_opt, n_points)
    
    try:
        bounds = [(0.0, 1.0)] * n_points
        constraints = {'type': 'eq', 'fun': con_eq}
        # Small random perturbation to avoid stagnation
        h_init = h_opt + np.random.uniform(-1e-4, 1e-4, n_points)
        h_init = project_to_F(h_init, n_points)
        
        # SLSQP isn't very fast, but we try to squeeze a bit more
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=constraints, jac=None, options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_opt = project_to_F(res.x, n_points)
            c5_opt = res.fun
        else:
            c5_opt = compute_C5(h_opt, dx)
    except:
        c5_opt = compute_C5(h_opt, dx)
    
    # Re-compute final C5
    final_c5 = compute_C5(h_opt, dx)
    c5_bound = final_c5
    
    return (h_opt.tolist(), c5_bound, n_points)
