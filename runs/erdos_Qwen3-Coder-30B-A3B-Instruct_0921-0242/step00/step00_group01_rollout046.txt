import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    corr = correlate(h, 1-h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def initial_h_values():
    """Provide an initial guess."""
    # Use the current known good construction
    n = 47
    dx = 2.0 / n
    h = np.zeros(n)
    # Fill with a simple pattern
    for i in range(n):
        h[i] = 1.0 if (i % 3 == 0) else 0.0
    # Normalize to integral = 1
    h = h * (1.0 / np.sum(h) * 2.0)
    return h.tolist()

def run(seed=42, budget_s=600, **kwargs):
    """Run the optimization process."""
    start_time = time.time()
    # For speed, start with a modest size for exact optimization
    n0 = 120  # Even number
    dx = 2.0 / n0
    # Set the target size for final output (must be <= 1000)
    target_n = 360  # 3x duplication, should be even
    
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    remaining_budget = budget_s

    # Initial solution
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values())
    else:
        h_initial = np.zeros(n0)
        h_initial[:n0//2] = 1.0
        h_initial = h_initial / np.sum(h_initial) * 2.0

    # Build the MIP for n0=120
    x = cp.Variable(n0, boolean=True)
    w = cp.Variable()
    S = n0 / 2

    # Add equality constraint for sum
    constraints = [cp.sum(x) == S]

    # For building z[i, k] = x[i] * x[(i+k) % n0]
    # We'll use a variable for each unique pair and model the constraints properly
    # Here we compute all z
    
    max_k = n0 // 2  # Because of symmetry
    z_vars = {}
    z_indices = []

    # Variables for z[i, k], where z[i,k] = x[i] * x[(i+k) % n0]
    z_vars_list = []
    # We introduce all necessary z vars
    for k in range(n0):
        for i in range(n0):
            z_vars[(i, k)] = cp.Variable(boolean=True)
            z_indices.append((i, k))

    # Linearization constraints for z[i, k] = x[i] * x[(i+k) % n0]
    linearization_constraints = []
    for k in range(n0):
        for i in range(n0):
            j = (i + k) % n0
            z = z_vars[(i, k)]
            linearization_constraints.append(z <= x[i])
            linearization_constraints.append(z <= x[j])
            linearization_constraints.append(z >= x[i] + x[j] - 1)

    # Overlap constraints: for each k
    for k in range(n0):
        # s_k = sum_{i} z[i,k] = sum_{i} x[i] * x[(i+k)%n0]
        # O_k = S - s_k = S - sum_{i} x[i] * x[(i+k)%n0]
        # We want w >= O_k = S - sum_{i} x[i] * x[(i+k)%n0]
        # equivalently, sum_{i} x[i] * x[(i+k)%n0] <= S - w
        # sum_{i} z[i,k] <= S - w
        expr = cp.sum([z_vars[(i, k)] for i in range(n0)])
        linearization_constraints.append(expr <= S - w)

    # Final constraints
    constraints = constraints + linearization_constraints

    # Symmetry breaking
    constraints.append(x[0] == 1)
    # Lexicographic order for first half
    for i in range(n0//2):
        if i+1 < n0//2:
            constraints.append(x[i] >= x[i+1])

    objective = cp.Minimize(w)
    prob = cp.Problem(objective, constraints)
    
    # Solve like this with a timeout (using solver internal param)
    try:
        # Set the time limit for the solver
        elapsed = time.time() - start_time
        remaining = max(1, budget_s - elapsed - 5)  # Leave some buffer
        prob.solve(solver=cp.CBC, max_seconds=remaining)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Get solution
            x_opt = x.value > 0.5
            h_opt = np.array(x_opt, dtype=float)
            c5_opt = prob.value * dx
            print(f"MILP Solution: c5 = {c5_opt:.6f}")
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        else:
            print("MILP failed or timeout. Falling back to greedy method.")
            # Fallback
    except Exception as e:
        print(f"MILP solver error: {e}. Falling back to greedy.")
        # fall back to greedy

    # Greedy fallback if needed
    if best_h is None:
        # GreedyBalanced what we can do quickly
        n = 120
        dx = 2.0 / n
        # Initialize empty set S
        S = set()
        while len(S) < n // 2:
            best_index = -1
            min_delta = float('inf')
            # For each index not in S
            for i in range(n):
                if i in S:
                    continue
                # Temporarily add it
                temp_S = S | {i}
                # Calculate max overlap with this addition
                temp_h = np.zeros(n)
                for j in temp_S:
                    temp_h[j] = 1.0
                # Compute overlaps
                corr = correlate(temp_h, 1-temp_h, mode="full")
                max_o = np.max(corr) * dx
                # The effect is to reduce the worst overlap if it's beneficial
                # In greedy we're minimizing expansion of worst overlap
                # So we try to minimize the effect of adding i
                worst_before = float('inf')
                temp_h_crop = temp_h.copy()
                for j in range(n):
                    temp_h_crop[j] = 1.0 if j in S else 0.0
                corr_before = correlate(temp_h_crop, 1-temp_h_crop, mode="full")
                worst_before = np.max(corr_before) * dx
                if max_o < worst_before:
                    if max_o < min_delta:
                        min_delta = max_o
                        best_index = i
            # Add the chosen index
            if best_index != -1:
                S.add(best_index)
            else:
                # If no improvement possible, greedily add least constraint-violating
                best_index = next((i for i in range(n) if i not in S), None)
                if best_index is not None:
                    S.add(best_index)
            if time.time() - start_time > budget_s - 2:
                break
        h_greedy = np.zeros(n)
        for i in S:
            h_greedy[i] = 1.0
        h_greedy = h_greedy / np.sum(h_greedy) * 2.0
        c5_greedy = evaluate_erdos_solution(h_greedy)
        print(f"Greedy Solution: c5 = {c5_greedy:.6f}")
        if c5_greedy < best_c5:
            best_c5 = c5_greedy
            best_h = h_greedy

    # Now perform a fast local improvement on the best solution found (even if greedy)
    h_improved = best_h.copy()
    if h_improved is not None and len(h_improved) >= 120:
        # Run a bounded swap refinement
        for _ in range(200):
            if time.time() - start_time > budget_s - 2:
                break
            # Compute all autocorrelations
            corr = correlate(h_improved, h_improved, mode="full")
            # Find the minimum one
            min_k = np.argmin(corr)
            # Find the line corresponding to min_k
            # In the table of correlations, the value is at index `min_k`
            # The minimal (index) overlap is in the middle of full correlation
            # corr[-half : half] width 2*half + 1, with corr[0] being k=0
            # shift = half = n/2
            shift = len(h_improved) // 2
            # min_corr = corr[min_k] = sum_{i} h[i+k] * h[i]
            # We want to find a pair that improves this overlap
            found = False
            for p in range(len(h_improved)):
                if h_improved[p] == 0:
                    continue
                for q in range(len(h_improved)):
                    if h_improved[q] == 1:
                        continue
                    # Swap p and q
                    pcorr = 0
                    qcorr = 0
                    # Recalculate changes in correlations that involve p or q
                    for i in range(len(h_improved)):
                        i_p = i
                        i_q = (i + (min_k - shift)) % len(h_improved)
                        pcorr += h_improved[i_p] * h_improved[i_q]
                        qcorr += h_improved[i_p] * h_improved[i_q]
                    # Swap p and q and check
                    temp = h_improved[p]
                    h_improved[p] = h_improved[q]
                    h_improved[q] = temp
                    pcorr_new = 0
                    qcorr_new = 0
                    for i in range(len(h_improved)):
                        i_p = i
                        i_q = (i + (min_k - shift)) % len(h_improved)
                        pcorr_new += h_improved[i_p] * h_improved[i_q]
                        qcorr_new += h_improved[i_p] * h_improved[i_q]
                    diff = (pcorr_new + qcorr_new) - (pcorr + qcorr)
                    if diff > 0:
                        found = True
                        break
                    # Undo swap
                    temp = h_improved[p]
                    h_improved[p] = h_improved[q]
                    h_improved[q] = temp
                if found:
                    break
            if not found:
                break

    # Duplicate to final size
    m = target_n // n0
    if m * n0 != target_n:
        # Make fit by going to larger n0 or adjust size
        pass
    h_final = np.tile(h_improved, m)
    final_c5 = evaluate_erdos_solution(h_final)
    
    return (h_final.tolist(), final_c5, target_n)
