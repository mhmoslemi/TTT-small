import numpy as np
import time
from scipy import fft
from scipy.optimize import minimize

def project_simplex_box(v, total, lo=0., hi=1.):
    """Project vector v onto the simplex with sum = total, box constraints [lo, hi]."""
    v = np.asarray(v)
    n = len(v)
    # Sort values in descending order
    sorted_v = np.sort(v)[::-1]
    # Find threshold
    cumsum = np.cumsum(sorted_v)
    threshold = (cumsum - total) / np.arange(1, n + 1)
    # The largest k such that threshold_k <= sorted_v[k]
    k = np.argmax(sorted_v > threshold)
    if sorted_v[k] <= threshold[k]:
        k += 1
    # Adjust the threshold (in case we went too far)
    if k == 0:
        theta = (cumsum[-1] - total) / n
    else:
        theta = threshold[k - 1]
    # Compute projection
    projected = np.clip(v - theta, lo, hi)
    # Scale to exact total if needed
    actual_sum = np.sum(projected)
    if np.abs(actual_sum - total) > 1e-9:
        # Linear correction to maintain sum
        correction = (total - actual_sum) / n
        projected += correction
    return projected

def compute_c5(h, dx):
    """Compute the C5 bound using FFT-based correlation."""
    h = np.asarray(h)
    h_conj = 1 - h
    # Full cross-correlation: use FFT
    corr = fft.ifft(fft.fft(h) * fft.fft(h_conj).conj()).real
    # Take only the meaningful part (indices 0 to n-1)
    corr = corr[:len(h)]
    return np.max(corr) * dx

def worst_lag(h):
    """Returns the index k of worst lag and its correlation value."""
    h = np.asarray(h)
    h_conj = 1 - h
    corr = fft.ifft(fft.fft(h) * fft.fft(h_conj).conj()).real
    corr = corr[:len(h)]
    k_max = np.argmax(corr)
    return k_max, corr[k_max]

def gradient_worst(h, k):
    """Compute gradient of overlap at lag k with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        # In cyclic indices: i+k and i-k
        j1 = (i + k) % n
        j2 = (i - k) % n
        grad[i] = h[j1] + h[j2]
    return grad

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None

    n_values = [150, 250, 350]

        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
