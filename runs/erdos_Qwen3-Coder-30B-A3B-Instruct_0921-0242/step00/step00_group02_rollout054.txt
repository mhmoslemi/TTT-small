import numpy as np
import time
import cvxpy as cp
from scipy import optimize as opt
from scipy.signal import correlate

def evaluate(h):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # The overlap integral for shift k is sum(h[i] * (1 - h[i+k]))
    # which is equivalent to: sum(h[i]) - sum(h[i]*h[i+k])
    # We compute cross-correlation of h with (1-h) using full mode
    # The result is a vector of length 2*n-1
    cross_corr = correlate(h, 1-h, mode='full')
    max_corr = np.max(cross_corr) * dx
    return max_corr

def project_feasible(x, target):
    """Project a real vector x to the feasible set {0≤h≤1, ∑h = target}"""
    h = np.array(x, dtype=float)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Water-filling to adjust sum to target
    tol = 1e-12
    for _ in range(10):
        current_sum = np.sum(h)
        diff = target - current_sum
        if abs(diff) < tol:
            break
        # Determine how much we can increase uniformly
        free = np.sum(1.0 - h)
        if free <= 0:
            break
        incr = min(diff, free)
        incr_per = incr / np.sum(h < 1)
        if incr_per <= 0:
            break
        h = np.clip(h + incr_per, 0, 1)
    return h

def active_shifts(h, eps=1e-9):
    """Return all shift indices k for which the overlap O(k) is within eps of current C5"""
    dx = 2.0 / len(h)
    cross_corr = correlate(h, 1-h, mode='full')
    max_corr = np.max(cross_corr)
    # For each k, the overlap is cross_corr[n-1+k]
    ks = []
    n = len(h)
    for k in range(2*n - 1):
        if abs(cross_corr[k] - max_corr) < eps:
            ks.append(k - (n-1))
    return ks

def gradient(h, ks):
    """Compute sub-gradient g[i] for active shifts"""
    n = len(h)
    g = np.zeros(n)
    
    # For each active shift k:
    for k in ks:
        # Compute contribution to gradient for shift k
        if k >= 0:
            # h[i] * (1 - h[i+k]) for i = 0..n-k-1, else 0
            for i in range(n - k):
                g[i] += 1 - h[i+k]
            # h[i+k] * (1 - h[i]) for i = 0..k-1, else 0
            for i in range(k):
                g[i+k] += -h[i]
        else:
            # k < 0, i.e., i+k >= 0, so h[i+k] * (1 - h[i]), h[i]
            for i in range(n + k):
                g[i] += -h[i-k]  # Position i+k, so we are at index i...
                g[i-k] += 1 - h[i]  # Position i, so we are at index i-k...
        # compute for i = 0...n-1 the derivative wrt h_i:
        # There are vectors from h to (1-h) at all shift positions k
        # It's a long peculiar convolution...
    # Simpler computation:
    # For each k in ks, let's compute the local effect.
    # Now we'll correctly build the sub-gradient for all k
    g = np.zeros(n)
    for k in ks:
        # contribution is 1 for h[i] iff i+k is valid and within range
        # If k is positive, then i from 0 to n-k-1, and we add (1-h[i+k])
        # If k negative, then i from -k to n-1, and we subtract h[i+k]
        dk = np.zeros(n)
        if k >= 0:
            for i in range(n - k):
                dk[i] += 1 - h[i+k]
            for i in range(k):
                dk[i+k] -= h[i]
        else:
            for i in range(-k, n):
                dk[i] -= h[i+k]
            for i in range(n + k):
                dk[i+k] += 1 - h[i]
        g += dk
    # Normalize
    g /= len(ks) if ks else 1
    return g

def line_search(h, g, t_current, budget):
    """Try a small list of step sizes."""
    dx = 2.0 / len(h)
    # Compute current value
    f0 = evaluate(h)
    best_h = h.copy()
    best_val = f0
    step_sizes = [0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1]
    for alpha in step_sizes:
        if alpha == 0:
            continue
        h_new = h - alpha * g
        h_new = project_feasible(h_new, len(h) / 2.0 * dx)
        f_new = evaluate(h_new)
        if f_new < best_val:
            best_val = f_new
            best_h = h_new
    return best_h, best_val

def local_optimize(h0, max_iter, t_end):
    """Projected gradient descent with line search."""
    h = h0.copy()
    current_val = evaluate(h)
    best_h = h.copy()
    best_val = current_val
    dx = 2.0 / len(h)
    t0 = time.time()
    for iteration in range(max_iter):
        if time.time() > t_end - 2:
            break
        try:
            ks = active_shifts(h)
            g = gradient(h, ks)
            h, val = line_search(h, g, current_val, t_end)
            current_val = val
            if val < best_val:
                best_h = h.copy()
                best_val = val
        except:
            break
    return best_h, best_val

def anneal(h0, t_end):
    """Simple simulated annealing"""
    h = h0.copy()
    current_val = evaluate(h)
    best_h = h.copy()
    best_val = current_val
    dx = 2.0 / len(h)
    # initial temp
    temp = 0.01
    max_steps = 5000
    # cooling schedule: 0.995 after each step
    for step in range(max_steps):
        if time.time() > t_end - 1:
            break
        i, j = np.random.randint(0, len(h), 2)
        delta = np.random.random() * 0.01
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_feasible(h_new, len(h) / 2.0 * dx)
        new_val = evaluate(h_new)
        if new_val < current_val:
            h = h_new
            current_val = new_val
            if new_val < best_val:
                best_h = h_new
                best_val = new_val
        else:
            # Metropolis rule
            if np.random.random() < np.exp(-(new_val - current_val) / temp):
                h = h_new
                current_val = new_val
        # decrease temperature
        temp *= 0.995
    return best_h, best_val

def random_feasible(n):
    """Sample from uniform distribution over feasible set."""
    x = np.random.uniform(0, 1, n)
    desired_sum = n / 2.0
    x = x * desired_sum / np.sum(x)  # scale to match sum
    return project_feasible(x, desired_sum)

def milp_binary(n, time_limit):
    """Solve binary MILP for h[i] ∈ {0,1} with exactly n/2 ones."""
    try:
        # Try CBC if available
        solver = cp.CBC
    except:
        try:
            solver = cp.GLPK
        except:
            return None, np.inf

    # Create variables
    h = cp.Variable(n, boolean=True)
    # Constraint sum(h) == n/2
    constraints = [cp.sum(h) == n / 2.0]
    # Objective: maximize overlap (minimize 1 - overlap)
    # We introduce auxiliary variables for each h[i] * h[i+k]
    # Linearization:
    # a[i,k] <= h[i]
    # a[i,k] <= h[i+k]
    # a[i,k] >= h[i] + h[i+k] - 1
    # All a[i,k] >= 0
    vars_dict = {}
    coeffs = []

    for k in range(1, n):
        k_vars = []
        for i in range(n):
            j = i + k
            if j >= n:
                continue  # Skip out-of-bounds
            # Introduce binary variable a[i,k], representing h[i]*h[j]
            a = cp.Variable(boolean=True)
            vars_dict[(i,k)] = a
            # Constraints
            constraints.append(a <= h[i])
            constraints.append(a <= h[j])
            constraints.append(a >= h[i] + h[j] - 1)
            constraints.append(a >= 0)
            k_vars.append(a)
        if k_vars:
            coeffs.append(cp.sum(k_vars))

    # The objective is to minimize max_k \sum a[i,k]
    # But we want min max_k (1-dx*sum a[i,k]) = max_k dx*sum a[i,k] ?
    # No, we want minimum overlaps.
    # Let's say for some large M, bound each overlap from below by (1 - M*delta).
    # But due to model shrinkage, any concrete distribution of h allows computing all the overlaps exactly.

    # Let's assume each a[i,k] <= h[i]*h[j], let that be via big-M, and minimize bounds on
    # the worst-case... which should be structured with linear inequalities.
    
    # Simpler: find maximum overlap for a function
    # What we want is overlap = sum_h[i]*(1-h[i+k]) for all k, length [0..n-1]
    # For constraint sum(h) = n/2, input coeffs = [1 1 ... 1], and we just minimize max over index.
    
    # But here, we want any optimal integer solution (to beat any bound). Just return the solution.
    tolerance = 1e-6
    # Instead, we solve a problem that penalizes sum of mismatches for non-binary etc... using the right integer-linear constraints.
    
    # To get exact value for overlap of binary pattern with itself shifted, we do binary overlap
    # So the LP relaxes the binary constraints, but to make it harder to create a gap,  
    # we approach a bit differently with minimax.
    
    # We model max_k sum h[i]*h[i+k] = minimize t such that t >= sum h[i]*h[i+k] for all k in (1..n-1)
    # But that's still more complex due to replacement of products with binary vars.
    
    # Just solve it by trying to build small solutions:
    # Much simpler: let's simulate with a slightly different formulation of the overlap itself:
    
    # This is a classic case of a big-M or product linearization trick.
    # Instead solve actual little maximum of sum element-wise overlapping terms with 1-h.

    # Simplest yet correct thing: enumerate and optimize order to target min C5.
    # Do exact optimization with big-M constraints.
    
    # So let's hard-work on the constraint itself.
    
    # Reformulating easier C5 = max_k int h(x)(1-h(x+k)) = max_k \sum_i h[i] * (1 - h[i+k]) = max_k sum(h[i] - h[i]*h[i+k]) 
    # = max_k (n/2 - sum(h[i] * h[i+k])) where h[i] is binary and sum == n/2.
    
    # We let b_i,k = h[i]*h[i+k] in integer constraints.
    # We want to minimize the max_k \sum b_i,k.
    # So add variables for each such b[i,k], include linearization constraints
    # And add objective minimize some t, s.t. sum b[i,k] <= t for all k
    
    b = {}
    # Add b[i,k] for all valid i,k pairs
    k_vars = []
    for k in range(1, n):
        k_sum = 0
        for i in range(n):
            j = i + k
            if j >= n:
                break
            bi = cp.Variable(boolean=True)
            b[i,k] = bi
            # Linearization of hi* hj. For bi = h[i]*h[i+k] we have:
            # bi <= h[i]
            # bi <= h[i+k]
            # bi >= h[i] + h[i+k] - 1
            # bi >= 0
            constraints.append(bi <= h[i])
            constraints.append(bi <= h[j])
            constraints.append(bi >= h[i] + h[j] - 1)
            constraints.append(bi >= 0)
            k_sum += bi
        k_vars.append(k_sum)  # i.e., sum over i of (bi,k) for fixed k.

    if not k_vars:
        return h.value, np.inf

    # Minimize the maximum of these sums, which may be expressed as:
    # minimize t such that t >= sum b[i,k] for all k=1..n-1
    t = cp.Variable()
    for k_sum in k_vars:
        constraints.append(t >= k_sum)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    sol = prob.solve(solver=solver, time_limit=time_limit)
    if sol is not None:
        h_val = h.value
        if h_val is not None:
            # Most of the binary variables could be left as continuous in solver
            # If this is very well done, we might do a post-processing refinement,
            # but just returning the certifiably optimal binary solution if it's feasible.
            # If not binary, then we can do a post process to make it binary with equality
            # to satisfy constraints.
            
            # Ensure it's a binary solution:
            h_sol = np.round(h_val).astype(float)
            # Normalize to match target sum  
            desired_sum = n / 2.0
            h_sol = h_sol * (desired_sum / np.sum(h_sol))
            h_sol = project_feasible(h_sol, desired_sum)
            c5_val = evaluate(h_sol)
            return h_sol, c5_val
        else:
            return h0, evaluate(h0)
    else:
        return h0, evaluate(h0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None

    initial_h = np.array(initial_h_values)
    n_candidates = [30, 36, 40, 45, 50, 60]
    
    for n in n_candidates:
        if time.time() - t0 > budget_s - 5:
            break
        if n == len(initial_h):
            h = initial_h.copy()
        else:
            # Interpolate initial_h to length n
            x_orig = np.linspace(0, 2, len(initial_h))
            x_new = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_orig, initial_h)
            # Project for sum constraint
            h = project_feasible(h_interp, n / 2.0 * 2.0 / n)
        
        # Binary MILP if possible
        if n <= 36:
            e_time = budget_s - (time.time() - t0) - 5
            if e_time > 0:
                h_b, c5_b = milp_binary(n, e_time)
                if c5_b < best_c5 - 1e-6:
                    best_c5 = c5_b
                    best_h = h_b
                    best_n = n
        
        # Starting from interpolated version
        starts = [h]
        # Add 4 random starting points
        for _ in range(4):
            starts.append(random_feasible(n))
        
        for start_h in starts:
            if time.time() - t0 > budget_s - 5:
                break
            e_time = budget_s - (time.time() - t0) - 5
            if e_time <= 0:
                break
            
            h_opt, val = local_optimize(start_h, max_iter=1000, t_end=time.time() + e_time)
            if val < best_c5 - 1e-6:
                best_c5 = val
                best_h = h_opt
                best_n = n
            
            # Try annealing from the local optimum
            h_ann, val_ann = anneal(h_opt, time.time() + 10)
            if val_ann < best_c5 - 1e-6:
                best_c5 = val_ann
                best_h = h_ann
                best_n = n

    # Optionally upsample to refine
    if best_h is not None and time.time() - t0 < budget_s - 5:
        n_up = min(60, best_n + 10)  # upsample if feasible
        if n_up != best_n:
            x_orig = np.linspace(0, 2, best_n)
            x_new = np.linspace(0, 2, n_up)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_interp = project_feasible(h_interp, n_up / 2.0 * 2.0 / n_up)
            h_refined, val_refined = local_optimize(h_interp, max_iter=500, t_end=time.time() + 10)
            if val_refined < best_c5 - 1e-6:
                best_c5 = val_refined
                best_h = h_refined
                best_n = n_up

    if best_h is None:
        # Fallback to an initial version
        best_h = initial_h
        best_c5 = evaluate(initial_h)
        best_n = len(initial_h)
    return (best_h, best_c5, best_n)
