import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.linalg import solve

# Precomputed constants
M = 8  # Number of Fourier modes
n_low = 256
n_high = 512

# Generate Fourier basis matrix
def _build_fourier_basis(n, M):
    x = np.arange(n)
    # Create the basis matrix for cosine and sine terms (excluding constant)
    basis = np.zeros((n, 2 * M))
    for i in range(1, M + 1):
        basis[:, 2 * (i - 1)] = np.cos(2 * np.pi * i * x / n)  # cosine coefficients
        basis[:, 2 * (i - 1) + 1] = np.sin(2 * np.pi * i * x / n)  # sine coefficients
    return basis

# Precompute
Phi_low = _build_fourier_basis(n_low, M)
Phi_high = _build_fourier_basis(n_high, M)
# Cache the projection matrix
PhiTPhi_low = Phi_low.T @ Phi_low
PhiTPhi_high = Phi_high.T @ Phi_high
PhiT_low = Phi_low.T
PhiT_high = Phi_high.T

# Precompute pseudo-inverse for projection
Pinv_low = solve(PhiTPhi_low, PhiT_low)
Pinv_high = solve(PhiTPhi_high, PhiT_high)

def project_to_feasible(c, Phi, Pinv):
    # h = Phi @ c + 0.5
    h = Phi @ c + 0.5
    # Clamp h to [0,1] and project back
    h_clamped = np.clip(h, 0, 1)
    # Correction: find δc that makes h_clamped = Phi*(c + δc) + 0.5
    # Φ*(c + δc) + 0.5 = h_clamped => δc = Pinv @ (h_clamped - h)
    δc = Pinv @ (h_clamped - h)
    c_new = c + δc
    return c_new

def evaluate_c5(c, Phi, n):
    h = Phi @ c + 0.5
    corr = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / n
    c5 = np.max(corr) * dx
    return c5

def cma_es_step(c, Phi, Pinv, n, μ=15, λ=30, σ=0.2, step_size=0.1, max_iter=100):
    # Initialize population
    pop = [c + σ * np.random.randn(len(c)) for _ in range(λ)]
    scores = []
    for i, ind in enumerate(pop):
        # Project individual to feasible region
        ind = project_to_feasible(ind, Phi, Pinv)
        # Evaluate
        score = evaluate_c5(ind, Phi, n)
        scores.append(score)
    # Sort and select best μ
    indices = np.argsort(scores)[:μ]
    pop_selected = [pop[i] for i in indices]
    scores_selected = [scores[i] for i in indices]
    
    # CMA-ES update
    z = np.random.randn(len(c))
    c = c + step_size * z
    c = project_to_feasible(c, Phi, Pinv)
    return c

def optimize_with_cma_es(n, Phi, Pinv, max_time_s):
    start = time.time()
    c_best = np.zeros(2 * M)
    best_score = np.inf
    
    # Run multiple restarts
    restarts = 15
    for seed in range(restarts):
        np.random.seed(seed)
        c = np.random.randn(2 * M) * 0.2
        c = project_to_feasible(c, Phi, Pinv)
        while time.time() - start < max_time_s * 0.8:
            c = cma_es_step(c, Phi, Pinv, n)
            score = evaluate_c5(c, Phi, n)
            if score < best_score:
                best_score = score
                c_best = c.copy()
        if time.time() - start >= max_time_s * 0.8:
            break
    
    return c_best, best_score

def refine_with_gradients(c, Phi, n, max_iter=200):
    h = Phi @ c + 0.5
    # Find the maximal correlation index
    corr = np.correlate(h, 1 - h, mode='full')
    k_max = np.argmax(corr)
    
    # Compute gradient ∂C5/∂h
    grad_h = np.zeros_like(h)
    for i in range(len(h)):
        # Use wrap-around indexing
        i_k = (i + k_max) % len(h)
        i_negk = (i - k_max) % len(h)
        grad_h[i] = (1 - h[i_k]) - h[i_negk]
    
    # Backpropagate gradient to coefficients
    grad_c = Phi.T @ grad_h
    
    # Take steps with diminishing learning rate
    for t in range(max_iter):
        new_c = c - 0.01 / (1 + t) * grad_c
        new_c = project_to_feasible(new_c, Phi, Pinv)
        c = new_c
        
        # Check for convergence
        h_new = Phi @ c + 0.5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        if np.max(corr_new) < evaluate_c5(c, Phi, n) + 1e-10:
            break
    
    return c

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    
    # Use a lower resolution for initial run
    Phi = Phi_low
    Pinv = Pinv_low
    n = n_low
    
    # Run CMA-ES optimization
    c_opt, best_score = optimize_with_cma_es(n, Phi, Pinv, budget_s)
    
    # Check if we should increase resolution
    if time.time() - start < budget_s * 0.8 and best_score > 0.379:
        # Double resolution
        Phi = Phi_high
        Pinv = Pinv_high
        n = n_high
        # Interpolate best coefficients
        c_opt = c_opt
        # Re-optimize with higher resolution
        c_opt, best_score = optimize_with_cma_es(n, Phi, Pinv, budget_s - (time.time() - start))
    
    # Refine with gradient descent
    c_opt = refine_with_gradients(c_opt, Phi, n)
    
    # Final reconstruction
    h_opt = Phi @ c_opt + 0.5
    # Ensure integra
    integral = np.sum(h_opt) * (2.0 / n)
    # Apply small correction to make sum equal to 1
    err = integral - 1.0
    h_opt = h_opt - err / n
    
    # Final bound
    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    dx = 2.0 / n
    final_c5 = np.max(corr) * dx
    
    return (h_opt.tolist(), float(final_c5), n)
