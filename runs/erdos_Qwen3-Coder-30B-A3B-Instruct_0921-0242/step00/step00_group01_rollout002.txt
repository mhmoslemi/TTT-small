import numpy as np
import time
from scipy import signal

def project_box_affine(h, target_sum):
    """Project h onto box [0,1] and affine constraint sum=h_sum"""
    h = np.clip(h, 0, 1)
    delta = (np.sum(h) - target_sum) / len(h)
    h -= delta
    h = np.clip(h, 0, 1)
    return h

def autocorr(h):
    """Circular autocorrelation of h"""
    n = len(h)
    fft_h = np.fft.fft(h)
    autoc = np.real(np.fft.ifft(np.abs(fft_h)**2))
    return autoc

def compute_C5(h, dx):
    """Compute C5 = max_k int h(x)(1 - h(x+k)) dx"""
    # Use np.correlate with mode="full"
    # h * (1 - h) with full mode gives
    # correlate(h, 1-h, mode="full")
    # But we want h(x)(1 - h(x+k))
    # This is h(x) - h(x)h(x+k), so
    # integrate over all x to get integral of h - integral of h h_k
    # Integral of h is sum(h) * dx
    # The second term is correlation of h and h (circular, with shift k)
    # C5 = max_k { dx * (sum(h) - correlation(h,h,k)) }
    # Max of (sum(h) - correlation) = max of -correlation
    # Which means minimize max k correlation(h,h,k)
    # So we minimize (sum(h) * dx - max_k (correlation(h,h,k) * dx))
    # = (sum(h) * dx) - max_k (correlation(h,h,k) * dx)
    # Which we can compute as (sum(h) * dx) - max_k (V_k * dx) = (sum(h) * dx) - (max_k V_k * dx)
    # where V_k = correlation(h,h,k) = (h * h)_k
    # When h is 1-d vector of len n, V_k = sum_{i=0}^{n-1} h_i * h_{i+k mod n}
    # So, in numpy terms we can compute:
    # correlate(h,h, mode="full") But that's convolution not correlation
    # For correlation h(x) h(x+k), we compute:
    # S = sum(h) * dx
    # Then V_k = correlate(h, h, mode="full")[n-1:] (circular, so need to shift)
    # Actually we only care about values of k=0 to len-1
    # So for full mode correlate(h,h) of length 2n-1, elements at indices n-1 to 2n-2
    # But actually there's an issue with signs...
    # We want correlation of h with (1-h)
    # h(x) * (1 - h(x+k)) = h(x) - h(x)h(x+k)
    # integral of which is sum(h)dx - sum(h_i h_{i+k})dx
    # assume ask for sum of which we want max, and we want minimize max.
    corr = np.correlate(h, 1-h, mode='full')
    corr_full = corr
    corr = corr_full[len(h)-1:]  # Remove initial part that's not valid circular convolution
    # But actually we want the convolution, i.e., when h is doubled, we subtract its own contribution
    # h * (1 - h) = h - h * h
    # double-check our correlation metric
    # V_k = sum_{i} h_i * h_{i+k}  (circular) 
    # So V_k = int_{0}^{2} h(x) * h(x + k) dx = correl(h, h, mode="full")[n+k]  when k in 0..n-1
    # But again for correlation: it's h(x) * h(x + k) so use correlation
    # in scipy correlate(h, h) = sum_{j} h(i) * h(i-j)
    # but we want h(i) * h(i+j) = h(i) * h(i+j) mode circular
    # equivalent to rotate h to left by k -> h[i+k mod n], now correlate with original
    # So we want [h_i * h_{i+k}]_i -> exponential form same as circautocorr
    # So both autocorr(h) and np.correlate(h, h) should be the same for circular correlation
    # My error is definition of integral domain: overlap defined over 0 to 2.
    # We have no absolute values, units of dx
    # Let us reread the integral definition:
    # C_k = int_0^2 h(x) (1 - h(x+k)) dx
    # = int_0^2 h(x) dx  - int_0^2 h(x) h(x+k) dx  
    # = S  - int_0^2 h(x) h(x+k) dx
    # Integrate using samples over [0,2], spacing dx, so dx is the integral step:
    # = S - sum_i h_i * h_{i+k} * dx
    # AE = EV = S * dx - sum_i h_i * h_{i+k} * dx = S * dx - (h * h)_k * dx
    # =
    # = dx * (S - (h * h)_k) where (h * h)_k is the convolution with shift k
    # 
    # So this is dx*(sum(h) - corr(circular)) where corr is circular convolution
    # Actually we normally compute this as:
    # corr_full = np.correlate(h, 1-h, mode="full")
    # So corr_full[0 : 2*n - 1]
    # and we want k = 0 to n-1.  
    # What we need is effective correlation for period 2 (not 1).
    # But it's not a genuine fixed length dependence on k, so we'll use periodic corr:
    # Overall do: compute over full, extract the k=0 to n-1 segment.
    # let's slow but precise way
    # need overlap_k, for each k=0 to n-1
    # we seek max_k over the range [0,n-1].
    # So do
    # overlap_k = dx * (sum(h) - sum_i h_i * h_{i+k mod n}) 
    # The above is inefficient. But numpy supports convolution. 
    # Let's compute exact: for mu = 0..n-1, we get
    # (h * h)[mu] = sum h_i * h_{i+mu mod n} 
    # The correlation is the same.
    # Want overlapping integral of h(x)(1-h(x+k)) for shift k in [0,2]
    # So k is index from 0 to n-1 in [0*2,2*2] and dx=2/n => index is k = kdx / dx = k
    # Tough: we need match the integral over [0,2] 
    # For non-periodic one, overlap = int h(x) h(x+k) dx.
    # But include the 1-h(x+k) part:
    # int h(x)(1-h(x+k)) dx = int h(x) dx - int h(x) h(x+k) dx
    # = S - int h(x) h(x+k) dx. 
    # So this means
    # put h[i] where i = 0 to n-1. index in [0,2].
    # shift by +k =  k/2 * n
    # so when h_i h_{i+k} that's i over i+k mod n
    # different from num.fft algorithm, but separate shift step is okay.
    # Conv(1-h, h) would be convolution of (1-h, h) = norm corr with h
    # It's more like:
    # overlap_k = dx * (sum(h) - dot(h, shift(h, k)))
    # We are using full mode, then for k=0 to n-1
    # _from the end (since lags can be negative, full mode has 2n-1)
    #  corr_lag = 0..n-1 correspond to shifts 0..n-1 
    # Since FFT gives: corr of h with itself at 0 to n-1 in in "...0, h[i]=0... (n),
    # lag indices goes from -n+1 to +n-1 (n entries)
    # full conv starts from index 0 onwards
    # However, using rank 1 routines we do this carefully.
    # Keep the norm from fixed shift.
    # This is tricky: we build h(x) over [0,2] and compute integral.
    # Variable h[i] corresponds to value over [i*dx, (i+1)dx] so overlaps happen
    # at lag k*dx (or k*dx/n in index space), but we get all dists.
    # We'll proceed with FFT method for speed: for now compute C5 from that
    # Autocorr is circular 
    h_np = np.asarray(h)
    corr_full = np.correlate(h_np, 1-h_np, mode='full')
    corr_lags = corr_full[len(h_np)-1:]  # That's k=0 to n-1
    # overwrite with actual correct formula for C5
    h_sum = np.sum(h_np)
    overlaps = dx * (h_sum - corr_lags)  # This is (h * (1-h))[k] = h sum - h*k
    C5 = np.max(overlaps)
    return C5

def evaluate_solution(h, n):
    """Evaluate solution for given h (list of n real numbers in [0,1])"""
    dx = 2.0 / n
    return compute_C5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = []
    best_n = 0
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Strategy support functions:
    # 1. Generate initial h using provided initial_h_values interpolated
    def initialize_h(n):
        # Scale initial_h_values if different size
        init = np.asarray(initial_h_values)
        if len(init) == n:
            return init.copy()
        elif len(init) < n:
            # Interpolate
            x_orig = np.linspace(0, 2, len(init))
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_orig, init)
            h = np.clip(h,0,1)
        else:
            # Downsample
            x_orig = np.linspace(0, 2, len(init))
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_orig, init)
            h = np.clip(h, 0, 1)
        # Renormalize sum
        h = h * target_sum / np.sum(h)
        return h.copy()
    
    # 2. Random initialization: random binary vector with half 1's, half 0's
    def init_random_binary(n):
        h = np.zeros(n)
        indices = np.random.permutation(n)[:n//2]
        h[indices] = 1.0
        # Smooth with small convolution
        kernel = np.ones(11) / 11
        h = np.convolve(h, kernel, mode='same')
        h = np.clip(h, 0, 1)
        # Renormalize
        h = h * target_sum / np.sum(h)
        return h.copy()
    
    # 3. Two-block initialization
    def init_two_block(n, a, b):
        h = np.zeros(n)
        # Convert float boundary locations back to integer indices
        idx_start = int(a / 2 * n)
        idx_end = int(b / 2 * n)
        if idx_end > idx_start:
            # Ensure valid interval
            idx_start = max(0, idx_start)
            idx_end = min(n, idx_end)
            h[idx_start:idx_end] = 1.0
            # Compensate h so that sum is target_sum
            # Let L1 = idx_start, L2 = idx_end - idx_start, L3 = n - idx_end
            # Total integral = 1*L1 + c*L2 + 0*L3 = c*L2 = 1 => c = 1/L2
            # S = sum(h) = L1 + L2*c = L1 + 1
            # Another way: set all h to same constant c, then total = c*n. Want c*n = target_sum.
            # But this mixes the blocks.
            # do it right:
            integral = np.sum(h)
            h = h * target_sum / integral
        return h.copy()
    
    # 4. Gradient ascent with projected step
    def optimize_h(h_start, max_iter=2000, step_sizes=None):
        if step_sizes is None:
            step_sizes = [0.1 * (0.99 ** i) for i in range(max_iter)]
        h = h_start.copy()
        t0 = time.time()
        
        best_eval = float('inf')
        best_h_run = h.copy()
        
        for i in range(max_iter):
            if time.time() - t0 > budget_s * 0.9:
                break
            
            step = step_sizes[min(i, len(step_sizes)-1)]
            
            # Compute all correlations and find worst k
            corr = autocorr(h)
            # Need to minimize worst correlation, so maximize best correlation
            k_worst = np.argmin(corr)  # index k that minimized the correlation    
            # Compute gradient of -corr_k w.r.t. h_i
            # - (h \ast h)_k
            # Derivative wrt h_j of - (h_i h_{i+k}) terms ?
            # For h_j term: if it shows up in h_i h_{i+k} then we get  
            # k = i -i' => i' = i - k 
            # so else -h_{j+k} -h_{j-k} if any but: do rolling
            grad = -(np.roll(h, k_worst) + np.roll(h, -k_worst))
            h_new = h + step * grad
            h_new = project_box_affine(h_new, target_sum)
            h = h_new
            
            # Evaluate current
            current_c5 = evaluate_solution(h, n_points)
            if current_c5 < best_eval:
                best_eval = current_c5
                best_h_run = h.copy()
                
        return best_h_run, best_eval
    
    # Tuned probes for two-block configurations (fast evaluation)
    def test_two_blocks():
        min_c5 = float('inf')
        best_h = None
        
        # Test some two-block search grids
        a_range = np.arange(0.1,1.8,0.05)
        b_range = np.arange(0.2,2.0,0.05)
        for a in a_range:
            for b in b_range:
                if b > a:
                    h = init_two_block(n_points, a, b)
                    try:
                        c5 = evaluate_solution(h, n_points)
                        if c5 < min_c5:
                            min_c5 = c5
                            best_h = h.copy()
                    except Exception:
                        continue
        return best_h, min_c5
    
    # Initialization and two block sweep
    init_h_list = [initialize_h(n_points)]
    init_h_list.append(init_random_binary(n_points))
    
    # Run two-block search
    two_block_h, two_block_c5 = test_two_blocks()
    if two_block_h is not None and two_block_c5 < best_c5:
        best_c5 = two_block_c5
        best_h = two_block_h
        best_n = n_points
        
    # Multi start gradient ascent
    n_starts = 20
    iter_per_start = 1000
    step_schedule = [0.1 * (0.99 ** i) for i in range(iter_per_start)]
    
    for start_id in range(n_starts):
        if time.time() - start_time > budget_s * 0.95:
            break  
        t0 = time.time()
        init_h = init_h_list[start_id % len(init_h_list)]
        final_h, final_c5 = optimize_h(init_h, max_iter=iter_per_start, step_sizes=step_schedule)
        if final_c5 < best_c5: 
            best_c5 = final_c5
            best_h = final_h.copy()
            best_n = n_points
            
        if time.time() - t0 > 0.2:  # update time to avoid premature stop
            break
            
    # Optionally refine with larger n
    # Increase n to 400 and re-optimize
    if time.time() - start_time < budget_s - 50:
        n_big = 400
        dx_big = 2.0 / n_big
        init_h_big = initialize_h(n_big)
        final_h_big, final_c5_big = optimize_h(init_h_big, max_iter=500)
        if final_c5_big < best_c5:
            best_c5 = final_c5_big
            best_h = final_h_big.copy()
            best_n = n_big
    
    # Optionally refine with smaller n 
    if time.time() - start_time < budget_s - 50:
        # Downsample to maybe get better integral form
        n_small = 100
        dx_small = 2.0 / n_small
        # interpolate to smaller grid
        if len(best_h) == n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_small = np.linspace(0, 2, n_small)
            h_small = np.interp(x_small, x_orig, best_h)
            h_small = np.clip(h_small, 0, 1)
            h_small = h_small * (target_sum) / np.sum(h_small)  # Renormalize
            c5_small = evaluate_solution(h_small, n_small)
            if c5_small < best_c5:
                best_c5 = c5_small
                best_h = h_small.copy()
                best_n = n_small
    
    # Final evaluation 
    Final_c5 = evaluate_solution(best_h, best_n)
    
    return (best_h.tolist(), float(Final_c5), int(best_n))
