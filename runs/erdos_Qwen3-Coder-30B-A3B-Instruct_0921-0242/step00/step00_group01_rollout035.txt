import numpy as np
import time
import scipy.signal as signal

def project_box_affine(h, target_sum):
    """Project h onto box [0,1] and affine constraint sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    delta = (np.sum(h) - target_sum) / len(h)
    h -= delta
    h = np.clip(h, 0, 1)
    return h

def autocorr(h):
    """Compute circular autocorrelation of h using FFT"""
    n = len(h)
    fft_h = np.fft.fft(h)
    auto = np.real(np.fft.ifft(np.abs(fft_h)**2))
    return auto

def compute_C5(h, dx):
    """Compute the C5 value from discrete h"""
    # This mimics the full convolution of h with (1-h)
    # But we'll use numpy's correlate with mode='full'
    h_rev = 1 - h
    corr = np.correlate(h, h_rev, mode='full')
    #corr = np.correlate(h, h_rev, mode='full') - np.sum(h)**2  # Subtract S^2 for correct formula
    lags = np.arange(-len(h)+1, len(h))
    max_lag = len(h)-1
    # Only take the parts from 0 to max_lag as valid overlaps
    max_val = -np.inf
    for i, lag in enumerate(lags):
        if 0 <= lag <= max_lag:
            corr_val = corr[i] * dx
            # Actually C5 is the maximum of the overlaps
            # But we want to minimize this max overlap
            # So compute the actual overlap value correctly:
            # S = sum(h) * dx = 1
            # O_k = dx * (sum(h) - sum(h_i * h_{i+k}))
            # So O_k = dx * (1 - (h * h)[k])
            # But we are returning max O_k so:
            val = corr_val  # This is actually 1 - (h * h)[k] 
            max_val = max(max_val, val)
    # The correct way to compute overlap is:
    # O_k = dx * sum(h_i * (1 - h_{i+k})) = dx * (sum(h) - sum(h_i * h_{i+k}))
    # => O_k = dx * (1 - (h * h)[k])  # if we correctly interpret it as circular convolution
    # Correct formula from correlation:
    # overlap = dx * (sum(h) - sum(h[i] * h[i+k]))
    # conv = corr such that corr[k] = sum(h[i] * h[i+k]) 
    # So overlap = dx * (sum(h) - corr) = dx * (1 - corr)
    # Easier to compute actual overlaps for each k directly:
    # Let's redo that per lag k directly:
    # For more preciseness later use real correlate computation
    # Actually let us use opencv or signal.correlate with mode='full'
    corr_full = signal.correlate(h, h_rev, mode='full')
    n = len(h)
    #corr_full = np.correlate(h, h_rev, mode='full')
    overlaps = np.zeros(n)
    # lags go from -n+1 to n-1
    corr_center = n - 1  # index of 0-lag
    for k in range(n):
        idx = 2 * n - 2 - k   # This is negative lag clarity
        # Or directly:
        idx = corr_center + k - (n-1)  # [0,n-1] corresponds to lags [0,n-1]
        idx = corr_center - k  # indices for 0-lag = center
        if k <= n-1:
            idx = corr_center + k
        else:
            idx = corr_center - (k - (n-1))  # k - (n-1) behind 0 lag
        overlaps[k] = corr_full[idx]
        if not (0 <= idx < len(corr_full)):
            overlaps[k] = corr_full[idx]
    # Now we compute: O_k = dx * (sum(h) - overlaps[k])
    # Overlaps are sum of h_i * h_{i+k}, so:
    overlaps = dx * (1.0 - overlaps) 
    # Now the actual overlaps are overlaps, and we want max of these
    return np.max(overlaps)

def compute_C5_exact(h, dx):
    """More direct and exact computation of C5"""
    n = len(h)
    h_rev = 1 - h
    # Direct way using np.correlate with full mode
    # O_k = dx * sum(h[i] * (1 - h[i+k]))
    # =>  O_k = dx * (sum(h) - sum(h[i] * h[i+k]))
    # NP correlate in full mode
    corr = np.correlate(h, h_rev, mode='full')
    # the correlation runs from k=-n+1 to k=n-1
    n_corr = len(corr)
    # k goes from 0 to n-1
    overlaps = np.zeros(n)
    for k in range(n):
        # Corresponding correlation index is (n-1) + k : runs from n-1 to 2*(n-1)
        # or equivalently index n_corr - 1 - (n-1-k) = n - 2 + k
        idx = n - 1 + k
        if idx < 0 or idx >= len(corr):
            overlaps[k] = 0
        else:
            overlaps[k] = corr[idx]
    overlaps = dx * (1.0 - overlaps)
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_C5 = float('inf')
    best_h = None
    best_n = None

    def time_left():
        return budget_s - (time.time() - start_time)

    # 1. Initial setup
    n_points = 200  
    dx = 2.0 / n_points
    
    # 2. Initial candidates
    candidates = []
    
    # 2a. Use the initial values if provided
    try:
        init_h = np.array(initial_h_values)
        if len(init_h) != n_points:
            # Interpolate
            init_h_interp = np.interp(np.linspace(0, 2, n_points), 
                                      np.linspace(0, 2, len(initial_h_values)), 
                                      init_h)
        else:
            init_h_interp = init_h
        init_h_interp = project_box_affine(init_h_interp, n_points/2)
        candidates.append(("initial", init_h_interp))
    except: 
        pass
         

    # 2b. Random binary types
    for i in range(5):
        h = np.zeros(n_points)
        # Set roughly half to 1, and half to 0, then smooth
        idx = np.random.permutation(n_points)[:n_points//2]
        h[idx] = 1.0
        # Add a bit of smoothing
        kernel = np.ones(5) / 5
        h = np.convolve(h, kernel, mode='same')
        h = project_box_affine(h, n_points/2)
        candidates.append((f"rand_bin_{i}", h))
    
    # 2c. Simple block patterns
    for i in range(3):
        h = np.zeros(n_points)
        # Split into two blocks
        a = 0.5 + 0.2 * i  
        b = 1.0 + 0.2 * i
        a = np.clip(a, 0.05, 1.95)
        b = np.clip(b, a+0.05, 2.0)
        # Convert to indices
        ia = int(a / dx)
        ib = int(b / dx)
        ia = min(max(0, ia), n_points-1)
        ib = min(max(ia+1, ib), n_points)
        # h = 1 on [0, a) 
        h[:ia] = 1.0
        # h = c on [a, b)
        # integral = a + c*(b-a) = 1
        # => c = (1-a) / (b-a)
        if (b - a) > 0:
            c = (1.0 - a*dx) / (b*dx - a*dx)
            c = max(0, min(1, c))
            h[ia:ib] = c
        h = project_box_affine(h, n_points/2)
        candidates.append((f"block_2_{i}", h))
    
    # 3. Gradient optimization of each
    # Time allocation: allocate most time for a few optimized runs
    # Let's run a few candidate evaluations each with limited steps
    max_grad_steps = 500

    for name, h0 in candidates:
        if time_left() < 10:
            break
        h = h0.copy()
        C5_prev = float('inf')
        for step in range(max_grad_steps):
            if time_left() < 5:
                break
            # Compute all overlaps using correlation
            n = len(h)
            h_rev = 1 - h
            corr = np.correlate(h, h_rev, mode='full')
            # Correlation index centers at n-1
            # Matches lag = 0 at index n-1
            # So lag k at index n-1+k
            overlaps = np.zeros(n)
            for k in range(n):
                idx = n-1+k
                if 0 <= idx < len(corr):
                    overlaps[k] = corr[idx]
            overlaps = dx * (1.0 - overlaps)
            C5 = np.max(overlaps)
            if C5 < C5_prev:
                C5_prev = C5
            
            # Gradient ascent: we seek to minimize max overlapping, this means  
            # we need max O_k to decrease, which means decrease the worst overlap.
            # The gradient of O_k = dx * (1 - (h*h)[k]) w.r.t. h_j is:
            # dO_k/dh_j = -dx * (h_{j+k} + h_{j-k}) 
            # So the gradient of the max O_k is simply the gradient at k_star
            # where O_{k_star} = max O_k
            
            k_star = np.argmax(overlaps)
            grad = np.zeros(n)
            # Exploit the fact that autocorr gives (h*h)[k]
            # We are trying to decrease O_k = dx * (1 - (h * h)[k])
            # Therefore decreasing (h * h)[k], which means increase its minimize.
            # We want to maximize min_autocorr. So for current minimized index,
            # Have its gradient pointing to increase it.
            # Grad at k* is minus the auto corr gradient for k*. 
            # Element-wise:
            # Let's compute the gradient of -(h * h)[k] w.r.t. h_j
            # i -> [j, j+k]  + [j, j-k]
            # For circle: h_{j+k} and h_{j-k} (mod n)
            for j in range(n):
                j_k = (j + k_star) % n 
                j_msk = (j - k_star) % n
                grad[j] = -(h[j_k] + h[j_msk])  
            
            # Gradient ascent step
            step_size = 0.05 * (1.0 - step / max_grad_steps)
            h_new = h + step_size * grad
            h = project_box_affine(h_new, n_points/2)

        # Evaluate final C5
        final_C5 = compute_C5_exact(h, dx)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = h.copy()
            best_n = n_points

    # 4. Now refine by downsampling to smaller size
    # Try n=100 or smaller
    if best_C5 < 0.4 and time_left() > 50:
        n_small_candidates = [100, 75, 80, 90]
        for n_small in n_small_candidates:
            if time_left() < 30:
                break
            # Downsample best_h
            h_resampled = np.array([np.mean(best_h[i: i+2]) for i in range(0, len(best_h), 2) if i+1 < len(best_h)])
            if len(h_resampled) == n_small:
                h_resampled[0] += (len(best_h) - 2*len(h_resampled))/2
                h_resampled = project_box_affine(h_resampled, n_small/2)
                C5_small = compute_C5_exact(h_resampled, 2.0/n_small)
                if C5_small < best_C5:
                    best_C5 = C5_small
                    best_h = h_resampled
                    best_n = n_small
                if C5_small < 0.379:
                    break

    # Return the best result found
    if best_h is None:
        # Fallback to the initial if nothing found and constraints satisfied
        try:
            init_h = np.array(initial_h_values)
            init_h = project_box_affine(init_h, len(init_h)/2)
            init_C5 = compute_C5_exact(init_h, 2.0 / len(init_h))
            best_C5 = init_C5
            best_h = init_h
            best_n = len(init_h)
        except:
            pass
    
    return (best_h.tolist(), float(best_C5), int(best_n))
