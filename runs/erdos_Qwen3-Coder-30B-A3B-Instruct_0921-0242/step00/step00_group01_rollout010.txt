import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft

def evaluate_c5(h, dx):
    """Evaluate the C5 constant from discretized function h"""
    # Compute autocorrelations using FFT
    H = fft(h)
    A = ifft(np.abs(H)**2).real
    # Adjust for circular correlation
    A = A[:len(h)]
    min_A = np.min(A)
    return dx * (0.5 * len(h) - min_A)

def block_search(n_base=400, delta=0.02):
    """Search for 3-block configurations"""
    best_c5 = float('inf')
    best_h = None
    a_vals = np.arange(0, 2, delta)
    b_vals = np.arange(0.01, 2, delta)
    for a in a_vals:
        for b in b_vals:
            if a >= b:
                continue
            p = (1 - a) / (b - a)
            if 0 <= p <= 1:
                h = np.zeros(n_base)
                h[:int(a * n_base / 2)] = 1
                h[int(a * n_base / 2):int(b * n_base / 2)] = p
                h[int(b * n_base / 2):] = 0
                # Normalize
                h_sum = np.sum(h)
                if h_sum > 0:
                    h *= (n_base / 2) / h_sum
                c5 = evaluate_c5(h, 2.0 / n_base)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    return best_h, best_c5

def de_optimization(n=200, seed=42):
    """Optimize with differential evolution on a Gaussian mixture parameterization"""
    np.random.seed(seed)
    
    def create_h_from_params(params):
        # Parameters are centers and amplitudes of 3 Gaussians (6 values total)
        centers = params[:3]
        amplitudes = params[3:]
        h = np.zeros(n)
        for i in range(3):
            # Create a soft version of the Gaussian
            x_vals = np.linspace(0, 2, n)
            # Use a simple rectangular bump with smoothing
            widths = 0.1  # constant width for simplicity
            mask = (x_vals >= centers[i] - widths/2) & (x_vals <= centers[i] + widths/2)
            h[mask] = np.maximum(h[mask], amplitudes[i])
        # Normalize
        total_mass = np.sum(h) * 2.0 / n
        if total_mass > 0:
            h *= (n / 2) / total_mass
        h = np.clip(h, 0, 1)
        return h
    
    def objective(params):
        h = create_h_from_params(params)
        dx = 2.0 / n
        c5 = evaluate_c5(h, dx)
        return -c5  # Negative because we're minimizing
    
    bounds = [(0, 2)] * 3 + [(0, 1)] * 3  # 3 centers and 3 amplitudes
    res = differential_evolution(objective, bounds, maxiter=100, popsize=30, seed=seed)
    h = create_h_from_params(res.x)
    dx = 2.0 / n
    c5 = evaluate_c5(h, dx)
    return h, c5

def gradient_ascent(h0, n_final=500, budget=300):
    """Projected gradient ascent with multi-resolution grid"""
    start_time = time.time()
    
    # Initialize on a coarser grid for speed
    n = max(150, n_final // 2)
    h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h0)), h0)
    
    # Scale to correct sum
    h = h * (n/2) / np.sum(h)
    
    # Learning rate and decay
    lr = 0.5
    decay = 0.98
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, 2.0 / n)
    
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget and iter_count < 5000:
        # Refine if needed
        if iter_count in [1500, 2500]:
            # Upsample
            n = min(n_final, 2 * n)
            h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h)), h)
            # Scale sum again
            h = h * (n/2) / np.sum(h)
            lr = lr * 0.5
        
        # Evaluate autocorrelations
        dx = 2.0 / n
        H = fft(h)
        A = ifft(np.abs(H)**2).real
        A = A[:n]
        k_star = np.argmin(A)
        
        # Gradient of the minimal autocorrelation
        g = 2 * np.roll(h, k_star)
        
        # Update step
        h_new = h + lr * g
        
        # Projection onto feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum = n/2
        offset = (np.sum(h_new) - n / 2) / n
        h_new = h_new - offset
        h_new = np.clip(h_new, 0, 1)
        
        # Second projection
        offset = (np.sum(h_new) - n / 2) / n
        h_new = h_new - offset
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new C5
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = iter_count
        
        h = h_new
        lr *= decay
        iter_count += 1
        
        # Early stopping
        if iter_count - last_improvement > 200:
            break
    
    if time.time() - start_time < 10:
        # Final upsampling to n_final and polish
        h = np.interp(np.linspace(0, 2, n_final), np.linspace(0, 2, len(h)), h)
        h = h * (n_final/2) / np.sum(h)
        best_c5 = evaluate_c5(h, 2.0 / n_final)
        best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main optimization loop"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Block search (3-blocks)
    h1, c5_1 = block_search(n_base=400, delta=0.02)
    best_h = h1
    best_c5 = c5_1
    
    # Stage 2: DE optimization
    if time.time() - start_time < budget_s * 0.4:
        h2, c5_2 = de_optimization(n=200, seed=seed)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2
    
    # Stage 3: Gradient ascent from best seed (with scheduling)
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 300:
        h3, c5_3 = gradient_ascent(best_h, n_final=300, budget=remaining_time - 100)
        if c5_3 < best_c5:
            best_h = h3
            best_c5 = c5_3
    
    # If still any time left, do fine-grid optimization
    if time.time() - start_time < budget_s - 50:
        h_final, c5_final = gradient_ascent(best_h, n_final=500, budget=50)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    # Final exact C5 evaluation on a fine grid
    final_n = len(best_h)
    dx = 2.0 / final_n
    final_c5 = evaluate_c5(best_h, dx)
    
    return list(best_h), final_c5, final_n
