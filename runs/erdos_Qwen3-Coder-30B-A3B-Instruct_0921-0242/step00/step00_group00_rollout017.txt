import time
import numpy as np
from scipy.sparse import diags
import cvxpy as cp
from cvxpy import SCS
from scipy.ndimage import zoom

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def solve_sdp(n, max_iters=4000, eps=5e-4):
        dx = 2.0 / n
        # Create the shift matrices M_k
        M_list = []
        for k in range(n):
            # M_k[i,j] = 1 if j = i + k and both indices valid
            offsets = [k]
            diagonals = [np.ones(n-k)]
            M_k = diags(diagonals, offsets, shape=(n,n), format='csr')
            M_list.append(M_k)
        
        # Variables
        h = cp.Variable(n)
        X = cp.Variable((n,n), symmetric=True)
        L = cp.Variable()
        
        # Constraints
        constraints = [
            X >> 0,  # X is positive semidefinite
            0 <= h,  # box constraints
            h <= 1,
            cp.sum(h) == n/2,  # integral constraint
        ]
        
        # Add X_{ii} = h_i constraints (enforces that diagonal of X matches h)
        for i in range(n):
            constraints.append(X[i,i] == h[i])
        
        # Add bounds on X_{ij} in terms of h_i and h_j
        for i in range(n):
            for j in range(n):
                if i != j:
                    constraints.append(X[i,j] <= h[i])
                    constraints.append(X[i,j] <= h[j])
                    constraints.append(X[i,j] >= 0)
        
        # Add L constraints
        for M_k in M_list:
            # trace(M_k X) = sum of the diagonal of X shifted by k positions
            # this is equivalent to the sum over valid indices of X[i, i+k] = M_k[i,j] * X[i,j]
            constraints.append(cp.trace(M_k @ X) >= L)
        
        # Objective
        objective = cp.Maximize(L)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=SCS, max_iters=max_iters, eps=eps)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value, X.value, L.value
            else:
                print(f"SDP solve status: {prob.status}")
                return None, None, None
        except Exception as e:
            print(f"SDP solve error: {e}")
            return None, None, None

    def project_to_feasible(h_hat, n):
        # Simple bisection to project onto the capped simplex
        h = np.copy(h_hat)
        h = np.clip(h, 0, 1)
        target_sum = n / 2.0
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-10:
            return h
        
        # Use binary search on the Lagrange multiplier theta
        tol = 1e-10
        low, high = -1.0, 1.0
        while high - low > tol:
            theta = (low + high) / 2.0
            h_test = np.clip(h - theta, 0., 1.)
            test_sum = np.sum(h_test)
            if test_sum > target_sum:
                low = theta
            else:
                high = theta
        theta = (low + high) / 2.0
        h_final = np.clip(h - theta, 0., 1.)
        return h_final
    
    def evaluate_c5(h, dx=2/n):
        # Convolve h with (1-h) for full overlap
        # This is equivalent to max_k of integral h(x)(1-h(x+k)) dx
        # But let's compute it directly using numpy
        h_reverse = h[::-1]
        conv = np.convolve(h, h_reverse, mode="full")
        # The full convolution result has length 2*n - 1
        # The contributions are for lags k = 0, ..., n-1
        # Overlap is: integral h(x)(1-h(x+k)) dx = 1 - integral h(x) h(x+k) dx
        # So C5 = max_k of overlap = max_k of (1 - integral h(x) h(x+k))
        # So C5 = 1 - min_k of integral h(x) h(x+k) 
        # The correct correlation for the difference:
        # Let's precompute the values properly
        total_integral = np.sum(h) * dx  # Should be 1
        diag_index = len(conv) // 2  # This should center around k=0
        correlations = np.array([conv[diag_index + k] for k in range(len(conv)) if (diag_index + k) < len(conv) and (diag_index + k) >= 0])[:n]
        correlations = np.array([conv[diag_index - k] for k in range(len(conv)) if (diag_index - k) >= 0 and (diag_index - k) < len(conv)])[:n]
        # Just compute up to n since proper shift matrix is not used when len is larger than index
        # Correct approach:
        # The correlations for each shift k
        # We want the minimum correlation of h with h(x + k)  
        # i:   h[i] * h[i + k] for valid i, so for k=0: sum over all i, for k=1: sum over i, i+1, ..., n-1: 
        # Correct way:
        corrs = []
        for k in range(n):
            corr = 0
            for i in range(n - k):
                corr += h[i] * h[i + k]
            corrs.append(corr * dx)
        # But really, directly use pointwise evaluated convolution and extract correlation
        corr = np.correlate(h, h, mode='full')
        max_correlate = np.max(corr)
        return 1.0 - max_correlate * dx  # Minimize correlation gives us max overlap
        # Compute convolve h with 1 - h
        inv_h = 1.0 - h
        cross_corr = np.convolve(h, inv_h, mode='full')
        # Extract full for each k from the middle
        mid = len(cross_corr) // 2  # middle is k = 0
        # The actual values for each lag are related
        # Because elements in convolution are of form integral h_i * (1 - h_{i+k}) = integral h_i - integral h_i * h_{i+k}
        # Which is the same as just computing sum h_i for the non-shifted part and sum h_i * h_{i+k} for the shifted part.
        # So for k=-n+1 to n-1, we want to get:
        # For k > 0, if we compute h over [0, 2] and h by shifts of h(x + k),
        # it's the part of h_i h_{i+k} from 0 to n-k (slide over)
        # For k = 0, we want total integral h*(1-h) = sum h - sum h^2:
        # h^2 is the self-correlation of h with itself
        # But we want: sum_0^n h_i (1 - h_{i+k}) 
        # = sum_0^n h_i  - sum_0^{n-k} h_i * h_{i+k} 
        # = 1 - sum_0^{n-k} h_i * h_{i+k} for k >= 0
        # = 1 - sum_{i=0}^{n-1-k} h_i * h_{i+k} but negative k means shifting right, which still is: 
        # For each k from 0 to n-1, the overlap we consider is:
        # sum_{i=0}^{n-1-k} h_i * (1 - h_{i+k}) = sum_{i=0}^{n-1-k} h_i - sum_{i=0}^{n-1-k} h_i h_{i+k}
        # = (n-k) * dx - sum_{i=0}^{n-1-k} h_i h_{i+k} 
        # But about using convolution?
        # We want: sum h_i * h_{i+k} 
        # Only that is equivalent to sum h_i(1 - h_{i+k}), so the other terms cancel
        # Computational:
        # We want max_k int h(x) (1 - h(x + k)) dx
        # = max_k [ int h(x) dx - int h(x)*h(x + k) dx ]
        # = max_k [ 1 - int h(x)*h(x + k) dx ]
        # = 1 - min_k int h(x)*h(x + k) dx
    
        # Since numpy.correlate gives us for negative lags as well, take specific values:
        to_test = []
        # Manually compute the integral h_i * h_{i+k} (for valid overlaps)
        for k in range(n):
            # Summer of h[i] * h[i+k] 
            # over i from 0 to (n-k-1)
            corr = 0
            for i in range(n - k):
                corr += h[i] * h[i + k]
            corr = corr * dx  # scale by dx
            to_test.append(corr)
        # Now C5 = 1 - min of this
        min_corr = min(to_test)
        return 1.0 - min_corr

    def evaluate_c5_shift(h, dx=2/n):
        # Let's use numpy for robustness
        n = len(h)
        all_correlations = []
        for k in range(n):
            corr = 0
            for i in range(n - k):
                corr += h[i] * h[i + k]
            all_correlations.append(corr * dx)
        min_corr = min(all_correlations)
        return 1.0 - min_corr

    def local_optimize(h_init, n, max_iter=500, step_size=0.01):
        h = h_init.copy()
        dx = 2.0 / n
        for t in range(max_iter):
            # The stepsize decreases
            alpha = step_size / np.sqrt(t + 1)
            # Compute the recent k such that sum == 1 - min_k sum(h_i * h_{i+k})
            # We're computing max_k 
            correlations = []
            for k in range(n):
                corr = 0
                for i in range(n-k):
                    corr += h[i] * h[i + k]
                correlations.append(corr * dx)
            k_best = np.argmin(correlations)
            # Gradient for current maximum overlap:
            # We can compute it by direct derivative
            # C5(h) = 1 - \min_k int h(x) h(x + k) dx  
            # Supplier giving back the gradient:
            # For current max k, we are maximizing it. 
            # Gradient of g_k(h) = integral h(x) h(x+k) dx
            # If max_k g_k is being evaluated,
            # Then gradient is the jacobian.
            # So df/d h_i = (if i+k out of bounds, 0) else h_{i+k}
            # But wait there's ambiguity:
            # since we are minimizing first the maximum, then minimizing that.
            # But I'm not using this exactly. I will use approach:
            # Take gradient and do projections.
            # Also recall the integral of h(x) * (1 - h(x+k)) dx = int h(x) dx - int h(x)*h(x+k) dx 
            # = 1 - int h(x)*h(x+k) dx. 
            # So I am seeking to minimize the integral h(x)*h(x+k) dx. 
            # Minimize via gradient descent on the integral of h(x)*h(x+k) over k.
            # C5(k) = 1 - integral h(x) h(x+k) dx
            # But in this problem, the k is actually fixed. So let's go directly to the definition:
            # Evaluate the gradient w.r.t. h
            # C5 = max_k [int h(x) (1 - h(x + k)) dx] = 1 - min_k [int h(x) h(x + k) dx]
            # The minimum k that achieves min of sum h_i * h_{i+k} gives the gradient we need:
            # w.r.t. h_i it's:
            # \partial/(h_i) int h(x)*h(x + k) dx = h_{i+k} + h_{i-k} if valid bounds
            # But for the sum: f(h) = sum_{i=0}^{n-1-k} h_i * h_{i+k}  
            # Thus df/d h_j = sum_{i: i+k = j} h_i + sum_{i: i = j+k} h_i = h_{j-k} + h_{j+k} only when they are valid
            # Want the actual principle: C5 = max_k of 1 - sum h_i h_{i+k} = 1 - min_k sum h_i h_{i+k}  
            # The primal considers gradient for one specific k where the min is.
            # For a fixed k:
            # So df_d h_i = h_{i+k}
            # But we want to decrease it: minimize h_i h_{i+k} over all k. So best is the one that minimizes that.
            # So we can take that k and compute:
            # ∇_{i} int h(x) h(x + k) dx = h_{i + k}
            # This yields subgradient.
            # However there is no strict differentiable function. So we approximate it.
            # Instead we use the fact that if we are at k, we want to minimize sum h_i h_{i + k}  
            # On the objective C5 = 1 - sum h_i h_{i + k}
            # The gradient is - h_{i+k} for the terms that exist  
            # But the source of the constraints is different from what this approach implies.
            # Ergo:
            # Derive expression: 
            # Consider the layers as:
            # for each shift k, the integral h(x) h(x+k) dx  
            # The contribution to C5 = 1 - min of all those sums  
            # so gradient of h_i w.r.t. C5 should be based on whichever k is being minimized:
            # the minimizing k: k^理想 = argmin_k [ \sum h_i * h_{i+k} ]  
            # If that agreement is sorted (in deterministic sense, just take first)`:
            # The effective gradient computation:  
            # If the k is fixed (through the layers)
            # Let's say best k = k_opt. And for the computation done in last iteration of primal, 
            # Then gradient w.r.t. h_i =  - h_{i + k_opt}  (and - h_{i - k_opt} ... for left shift also!)
            # But what if the minimizing k is not unique? Just pick one.
            
            # Compute the best k again (minimum sum h_i h_{i+k})
            # Then compute gradient with respect to that k
            # And apply it.
            all_sums = []
            for k in range(n):
                corr = 0
                for i in range(n - k):
                    corr += h[i] * h[i + k]
                all_sums.append(corr)
            # The index of minimum correlation  
            k_best = np.argmin(all_sums)
            # Gradient = - h_{i+k_best}
            grad = np.zeros(n)
            for i in range(n):
                i_plus_k = i + k_best
                if i_plus_k < n:
                    grad[i] -= h[i_plus_k]
            h = h - alpha * grad
            # Project to feasible set [0,1], sum = n/2 
            h = project_to_feasible(h, n)
        return h
    
    # Multi-resolution approach
    grid_sizes = [120, 200, 300]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n in grid_sizes:
        if time.time() - start_time > budget_s * 0.95:
            break
        print(f"Solving for n={n}")
        # Run SDP
        h_opt, X, L = solve_sdp(n, max_iters=4000, eps=5e-4)
        if h_opt is None:
            print(f"SDP failed for n={n}")
            continue
        # Project to feasible simplex
        h_feas = project_to_feasible(h_opt, n)
        # Evaluate C5
        c5_feas = evaluate_c5_shift(h_feas, 2.0/n)
        print(f"SDP at n={n}, C5={c5_feas:.6f}")
        if c5_feas < best_c5:
            best_c5 = c5_feas
            best_h = h_feas.copy()  # make a copy
            best_n = n
        # Local polishing if acceptable
        if time.time() - start_time > 0.5 and n == grid_sizes[-1]:  # only the last one to save time
            polish_h = local_optimize(h_feas, n, max_iter=500, step_size=0.01)
            polish_c5 = evaluate_c5_shift(polish_h, 2.0/n)
            print(f"Polished at n={n}, C5={polish_c5:.6f}")
            if polish_c5 < best_c5:
                best_c5 = polish_c5
                best_h = polish_h.copy()
                best_n = n
    assert best_n > 0
    return (best_h.tolist(), float(best_c5), int(best_n))
