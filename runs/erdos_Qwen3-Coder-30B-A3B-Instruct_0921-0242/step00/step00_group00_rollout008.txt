import numpy as np
import time
from scipy import fft
from scipy.optimize import minimize
import math

def project_simplex_box(v, total, lo=0., hi=1.):
    """Project vector v onto the simplex with box constraints [lo,hi] and sum = total"""
    v = np.asarray(v)
    n = len(v)
    # Sort values in descending order
    sorted_v = np.sort(v)[::-1]
    # Find threshold
    cumsum = np.cumsum(sorted_v)
    threshold = (cumsum - total) / np.arange(1, n + 1)
    # Find the index of the largest threshold that is <= sorted_v[i]
    mask = sorted_v > threshold
    if not np.any(mask):
        # All elements are projected to 0 or hi, depending on total
        # However, we must handle the case that `total` is outside [0, n*hi]
        # Actually, we're not supposed to get that far with valid inputs
        if total <= 0:
            return np.zeros_like(v) + lo
        elif total >= n * hi:
            return np.zeros_like(v) + hi
        else:
            # just clip to box and rescale
            result = np.clip(v, lo, hi)
            rescale = total / np.sum(result)
            return result * rescale
    # Find the appropriate threshold index
    idx = np.argmax(mask)
    theta = threshold[idx]
    # All values that were not masked are set to (v[i] - theta) and then clipped
    active_mask = v > theta
    result = np.where(active_mask, np.clip(v - theta, lo, hi), lo)
    # Ensure the sum exactly matches total by rescaling if needed
    current_sum = np.sum(result)
    if abs(current_sum - total) > 1e-9:
        # Adjust to fix small numerical errors
        correction = (total - current_sum) / n
        result += correction
        result = np.clip(result, lo, hi)
    return result

def compute_c5(h, dx):
    """Compute the maximum overlap integral using FFT"""
    # h is a numpy array of length n
    # We compute: correlation of h with (1-h)
    h = np.asarray(h)
    one_minus_h = 1.0 - h
    # Full cross-correlation
    corr = np.fft.ifft(np.fft.fft(h) * np.fft.fft(one_minus_h).conj()).real
    # Take only first half of the correlation (the valid values for k >= 0)
    # The full correlation is symmetric around k=0, so we only consider
    # correlations for k = 0,...,n-1 in the forward direction
    # That means copy the first n elements (though we have 2n-1 total)
    c5 = np.max(corr[:len(h)]) * dx
    return c5

def worst_lag(h):
    """Find the index of the worst lag and its value"""
    corr = np.fft.ifft(np.fft.fft(h) * np.fft.fft(1-h).conj()).real
    k_star = np.argmax(corr[:len(h)])
    return k_star, corr[k_star]

def gradient_worst(h, k):
    """Gradient of overlap at lag k w.r.t. h, as approximated by finite differences or direct"""
    n = len(h)
    # The gradient for O_k = sum_i h_i h_{i+k} is:
    # dO_k / dh_i = h_{i+k} + h_{i-k}
    # But it's periodic with period n, so handle indices correctly by %n
    grad = np.zeros_like(h)
    # effective indices for (i+k) and (i-k)
    for i in range(n):
        grad[i] = h[(i + k) % n] + h[(i - k) % n]
    return grad

def optimize(n, time_limit):
    """Main optimization routine"""
    # Start with initial h values
    if hasattr(initial_h_values, '__len__') and len(initial_h_values) == n:
        h = np.array(initial_h_values)
    else:
        # Generate random feasible point
        h = np.random.rand(n)
        h = project_simplex_box(h, n/2, 0, 1)
    dx = 2.0 / n
    best_h = None
    best_c5 = np.inf
    start_time = time.time()
    # Step size will be adaptive
    alpha = 0.1
    patience = 0
    max_patience = 20
    last_improvement = 0

    while time.time() - start_time < time_limit:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = time.time() - start_time
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Adaptive step size decrease
                alpha *= 0.7
                # Reset patience
                patience = 0
                if alpha < 1e-6:
                    break

        # Find worst correlation
        k_star, _ = worst_lag(h)
        grad = gradient_worst(h, k_star)
        # Step in negative gradient direction
        h_new = h - alpha * grad
        # Project back
        h_new = project_simplex_box(h_new, n/2, 0, 1)
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # Set a fixed seed for reproducibility
    np.random.seed(seed)

    # Note: for simplicity, we'll do a few full optimizations only,
    # not a highly adaptive scheme over many different n's.
    # We look for good values at a few reasonable sizes.
    # Start with the current construction and use it as a reasonable baseline.
    
    # Record what we'll test
    config_n = [150, 250, 350]
    best_c5 = np.inf
    best_h = None
    n_points = 0
    start_time = time.time()

    for n in config_n:
        if time.time() - start_time > 0.9 * budget_s:
            break
        # Limit iterations so that total time stays with budget
        time_per_n = min(100, max(5, int(0.2 * budget_s / len(config_n))))
        h, c5 = optimize(n, time_per_n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            n_points = n
            
    # Final refinement of the best one if time allows
    if best_h is not None and time.time() - start_time < 0.9 * budget_s:
        for n_refine in [2*n_points, 3*n_points]:   # Try to refine
            if n_refine > 1000: continue
            if time.time() - start_time > 0.9 * budget_s:
                break
            # Interpolate initial solution (reducing resolution)
            # Alternately, just use the same n if time permits, as we already have a good one
            # For this task, since the problem is at least faster to evaluate at larger n,
            # let's just stick with the n that performed best when it was fast enough.
            
    dx = 2.0 / n_points
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
