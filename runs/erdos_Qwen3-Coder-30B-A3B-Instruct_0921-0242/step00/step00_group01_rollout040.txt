        w = cp.Variable(1, name="w")  # worst overlap

        constraints = []

            for k in range(n):
                idx = (i + k) % n

                # z[i,k] <= b[idx]
                constraints.append(z[i,k] <= b[idx])
                # z[i,k] >= b[i] + b[idx] - 1
                constraints.append(z[i,k] >= b[i] + b[idx] - 1)

        for k in range(n):

        constraints.append(cp.sum(b) == S)

        constraints.append(b[0] == 1)

        for i in range(n//2):

        prob = Problem(objective, constraints)
