import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h):
    """Computes the C5 bound from a step function h"""
    dx = 2.0 / len(h)
    correlations = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlations)
    return max_corr * dx

def binary_hill_climbing(h_bin, n):
    """Improve binary vector by local swaps"""
    h = h_bin.copy()
    A = np.zeros(n-1)  # A[s] = overlap for shift s (0-indexed)
    
    # precompute overlaps
    for s in range(1, n):
        A[s-1] = sum(h[i] * h[i+s] for i in range(n - s))
    
    t = int(min(A))  # current best lower bound
    
    # we'll use a subset of possible swaps to reduce time
    iterations = 0
    max_iter = 1000
    while iterations < max_iter:
        iterations += 1
        # Sample 20 pairs with h[i]=1 and h[j]=0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        from random import choice
        i1 = choice(ones)
        i0 = choice(zeros)
        
        # compute change in overlaps if we swap h[i1] and h[i0]
        old_A = A.copy()
        
        # update A for affected shifts s where i1 or i0 appear
        for s in range(1, n):
            # check if i1 or i0 is involved in shift s
            if i1 + s < n and i0 + s < n:
                A[s-1] += h[i1] - h[i0]  # shift i1,s
                A[s-1] += h[i0] - h[i1]  # shift i0,s
            elif i1 + s < n or i0 + s < n:
                # one term got removed, not both
                if i1 + s < n:
                    A[s-1] -= h[i1]
                if i0 + s < n:
                    A[s-1] += h[i0]
            # check if shift s is now affected by positions i1-s, i0-s
            if i1 >= s and i0 >= s:
                A[s-1] -= h[i1-s]  # was in before
                A[s-1] += h[i0-s]  # now in
            elif i1 >= s or i0 >= s:
                if i1 >= s:
                    A[s-1] += h[i1-s]  # was not in before
                if i0 >= s:
                    A[s-1] -= h[i0-s]  # was in before
                
        new_t = int(min(A))
        if new_t > t:  # accept
            h[i1] = 0
            h[i0] = 1
            t = new_t
        else:  # revert changes
            A = old_A
    
    return h, t

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_bound = 1e9
    best_h = None
    best_n = 0
    remaining_budget = budget_s
    
    # Try n=100 first
    n = 100
    dx = 2.0 / n
    
    print("Starting MILP optimization with n = %d" % n)
    
    # Build the MILP model
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True, name="t")

    # Constraints
    constraints = [
        cp.sum(h) == n // 2,
        t <= 0  # dummy to initialize
    ]

    # Add auxiliary variables for product
    # y[i,s] = h[i]*h[i+s]
    y = {}
    for s in range(1, n):
        for i in range(n - s):
            y[(i, s)] = cp.Variable(boolean=True, name=f"y_{i}_{s}")

    # Enforce associations y[i,s] = h[i]*h[i+s]
    for s in range(1, n):
        for i in range(n - s):
            constraints.extend([
                y[(i,s)] <= h[i],
                y[(i,s)] <= h[i+s],
                y[(i,s)] >= h[i] + h[i+s] - 1
            ])

    # Final constraint: t <= sum of y over all valid shifts
    for s in range(1, n):
        overlap = cp.sum([y[(i, s)] for i in range(n - s)])
        constraints.append(t <= overlap)

    # Objective
    objective = cp.Maximize(t)

    # Compile the problem
    problem = cp.Problem(objective, constraints)

    # Solve
    time_left = remaining_budget
    problem.solve(solver=cp.CBC, verbose=False, max_seconds=time_left // 2)

    # Extract solution
    h_bin = np.round(h.value).astype(int)
    t_optimal = int(t.value if t.value is not None else 0)
    c5_bound = 1 - 2 * t_optimal / n

    print(f"MILP succeeded for n={n}: C5 = {c5_bound:.5f}")
    
    if c5_bound < best_bound:
        best_bound = c5_bound
        best_h = h_bin.copy()
        best_n = n
    
    # Now try hill-climbing improvement
    time_left = remaining_budget - (time.time() - start_time)
    if time_left > 10 and c5_bound < 0.4:
        print("Attempting hill-climbing...")
        h_improved, t_improved = binary_hill_climbing(h_bin, n)
        c5_improved = 1 - 2 * t_improved / n
        
        print(f"After hill-climbing: C5 = {c5_improved:.5f}")
        if c5_improved < best_bound:
            best_bound = c5_improved
            best_h = h_improved.copy()
            best_n = n

    # Try n=120 if time permits
    if remaining_budget - (time.time() - start_time) > 50:
        print("Trying n=120...")
        n = 120
        dx = 2.0 / n

        # Build new MILP model
        h2 = cp.Variable(n, boolean=True)
        t2 = cp.Variable(integer=True, name="t2")

        # Constraints
        constraints2 = [
            cp.sum(h2) == n // 2,
            t2 <= 0  # dummy to initialize
        ]

        # Add auxiliary variables for product
        # y[i,s] = h[i]*h[i+s]
        y2 = {}
        for s in range(1, n):
            for i in range(n - s):
                y2[(i, s)] = cp.Variable(boolean=True, name=f"y_{i}_{s}")

        # Enforce associations y[i,s] = h[i]*h[i+s]
        for s in range(1, n):
            for i in range(n - s):
                constraints2.extend([
                    y2[(i,s)] <= h2[i],
                    y2[(i,s)] <= h2[i+s],
                    y2[(i,s)] >= h2[i] + h2[i+s] - 1
                ])

        # Final constraint: t <= sum of y over all valid shifts
        for s in range(1, n):
            overlap = cp.sum([y2[(i, s)] for i in range(n - s)])
            constraints2.append(t2 <= overlap)

        # Objective
        objective2 = cp.Maximize(t2)

        # Compile the problem
        problem2 = cp.Problem(objective2, constraints2)

        # Solve
        time_left = remaining_budget - (time.time() - start_time)
        problem2.solve(solver=cp.CBC, verbose=False, max_seconds=time_left // 2)

        # Extract solution
        h_bin2 = np.round(h2.value).astype(int)
        t_optimal2 = int(t2.value if t2.value is not None else 0)
        c5_bound2 = 1 - 2 * t_optimal2 / n

        print(f"MILP succeeded for n={n}: C5 = {c5_bound2:.5f}")

        if c5_bound2 < best_bound:
            best_bound = c5_bound2
            best_h = h_bin2.copy()
            best_n = n

        # Hill climbing for 120
        time_left = remaining_budget - (time.time() - start_time)
        if time_left > 10 and c5_bound2 < 0.4:
            print("Attempting hill-climbing on n=120...")
            h_improved2, t_improved2 = binary_hill_climbing(h_bin2, n)
            c5_improved2 = 1 - 2 * t_improved2 / n

            print(f"After hill-climbing: C5 = {c5_improved2:.5f}")

            if c5_improved2 < best_bound:
                best_bound = c5_improved2
                best_h = h_improved2.copy()
                best_n = n

    print(f"Best bound found = {best_bound:.5f}")
    
    # Final upsampling if ready and useful
    if best_h is not None:
        h_final = best_h.copy()
        if best_n > 0:
            # Sample points per original point
            r = min(4, 800 // best_n)  # ensure final <= 800
            if r > 1:
                print("Upsampling...")
                h_final = np.repeat(h_final, r)
                dx = 2.0 / len(h_final)
    
    return (h_final, best_bound, len(h_final))
