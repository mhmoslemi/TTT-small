def autocorr(b):
    """Compute autocorrelation of binary sequence b."""
    A = np.fft.ifft(np.abs(np.fft.fft(b))**2).real
    return A

def generate_legendre_sequence(n):
    """Generate a Legendre sequence of length n."""
    # Find first odd prime >= n
    p = n
    while True:
        if all(p % i != 0 for i in range(2, int(p**0.5)+1)):
            break
        p += 2
    # Generate Legendre sequence
    s = np.array([(i*i) % p < p/2 for i in range(p)], dtype=int)
    # Truncate to n
    if len(s) > n:
        s = s[:n]
    else:
        s = np.pad(s, (0, n-len(s)), constant_values=0)
    # Adjust to balanced
    ones_count = np.sum(s)
    if ones_count > n//2:
        # Remove excess ones
        zero_indices = np.where(s == 0)[0]
        indices_to_flip = zero_indices[:ones_count - n//2]
        s[indices_to_flip] = 1
    elif ones_count < n//2:
        # Add missing ones
        one_indices = np.where(s == 1)[0]
        indices_to_flip = one_indices[:n//2 - ones_count]
        s[indices_to_flip] = 0
    return s

def generate_singer_sequence(n):

        s = np.zeros(n, dtype=int)

        block_size = n // 2  # approximate
        s[:block_size] = 1

def generate_random_balanced(n):
    """Generate random balanced binary sequence."""

def generate_palindrome_pattern(n):
    """Generate palindromic pattern."""

def compute_c5(h):

    dx = 2.0 / len(h)

    max_corr = np.max(correlations[len(h)-1:])
    return max_corr * dx

def local_swap_optimize(b, max_iterations=100000):
    """Perform swap-based optimization on binary sequence b."""
    n = len(b)
    b = b.copy()
    S = n // 2
    dx = 2.0 / n
    prev_best = -1
    cooling_rate = 0.99
    temperature = 1.0
    last_improvement = 0
    start_time = time.time()
    max_time = 60  # seconds
    
    # Precompute shifts to speed up autocorrelation update
    B_shift = np.array([np.roll(b, k) for k in range(n)], dtype=int)
    
    for i in range(max_iterations):
        if time.time() - start_time > max_time:
            break
        # Find current worst lag
        A = autocorr(b)
        m = np.min(A[1:])
        best_k = np.argmin(A[1:]) + 1  # since we skip index 0

        # Try random swap
        ones_list = np.where(b == 1)[0]
        zeros_list = np.where(b == 0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            continue
        i_idx = np.random.choice(ones_list)
        j_idx = np.random.choice(zeros_list)
        # Calculate change in autocorrelation at worst lag
        k = best_k
        delta = -B_shift[k][i_idx] + B_shift[k][j_idx]
        
        # Accept less beneficial swaps with prob. proportional to exp(delta/T)
        if delta > 0:
            b[i_idx] = 0
            b[j_idx] = 1
            last_improvement = i
        elif np.random.rand() < np.exp(delta / temperature):
            b[i_idx] = 0
            b[j_idx] = 1
            last_improvement = i
            
        if i - last_improvement > 5000:
            break
        if i % 5000 == 0 and temperature > 0.001:
            temperature *= cooling_rate
            
    return b

def inflate_binary_vector(b, target_n):
    """Inflate binary vector to target length by duplicating entries."""
    if len(b) >= target_n:
        return b[:target_n]
    b_doubled = np.repeat(b, 2)
    if len(b_doubled) != target_n:
        # Pad if necessary
        pad_width = target_n - len(b_doubled)
        b_doubled = np.pad(b_doubled, (0, pad_width), 'constant', constant_values=0)
    return b_doubled
