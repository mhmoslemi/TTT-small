import cvxpy as cp
from scipy.sparse import diags
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):

    resolutions = [120, 200, 300]
    best_c5 = float('inf')
    best_h = None
    best_n = 0

        h = np.clip(h, 0, 1)

        M = []
        for k in range(n_points):
            diag = np.ones(n_points - k)
            M.append(diags(diag, k).toarray())

        # SDP optimization
        X = cp.Variable((n_points, n_points), symmetric=True)
        h_var = cp.Variable(n_points)
        L = cp.Variable()
        
        constraints = [
            X >> 0,
            cp.diag(X) == h_var,
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points/2,
        ]
        
        for k in range(n_points):
            if k > 0:
                # Define correlation matrix properly, by sparsifying to only relevant indices
                diag_indices = range(n_points - k)
                diag = np.ones(n_points - k)
                M_k = diags(diag, k).toarray()  # sparse matrix
                constraints.append(cp.trace(M_k @ X) >= L)
            else:
                diag_indices = range(n_points)
                diag = np.ones(n_points)
                M_k = diags(diag, 0).toarray()  # Identity
                constraints.append(cp.trace(M_k @ X) >= L)
        
        prob = cp.Problem(cp.Maximize(L), constraints)

                h_proj = project_simplex(h_sol, n_points/2)

                # Compute C5
                corr = np.correlate(h_proj, 1-h_proj, mode='full')
                c5_val = 1.0 - np.min(corr) * dx

                    best_c5 = c5_val
                    best_h = h_proj.copy()
                    best_n = n_points

    if best_h is not None:

            g = g / (np.linalg.norm(g) + 1e-12)

def project_simplex(v, s):
    """
    Project a vector onto the simplex:
    min ||x - v||^2 subject to x >= 0, sum(x) = s
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u > cssv / ind, ind, 0)[-1]
    theta = cssv[rho-1] / rho
    return np.maximum(v - theta, 0)
