import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    # Matches the evaluation used in the test environment
    dx = 2.0 / len(h)
    correlations = correlate(h, 1 - h, mode='full')
    # Take max over all non-zero shifts (excluding the zero shift)
    max_corr = np.max(correlations[len(h):])  # only non-zero shifts
    return max_corr * dx  # this is C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s
    
    # Parameters
    n = 100
    dx = 2.0 / n
    target_bound = 0.3807
    best_h = None
    best_c5 = float('inf')
    
    # MILP solver setup
    def solve_milp(n_points, max_seconds):
        # Variables
        h = cp.Variable(n_points, boolean=True)
        t = cp.Variable(integer=True, lower_bound=0)
        y = cp.Variable((n_points, n_points), boolean=True)  # later handle properly
        y_vars = {}
        # Create y_vars dictionary for non-zero valid indices only
        for s in range(1, n_points):
            for i in range(n_points - s):
                y_vars[(i, s)] = y[i, s]
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h) == n_points // 2)  # exactly half ones
        
        # Product linearization constraints
        # For each shift s and index i where it's valid:
        for s in range(1, n_points):
            for i in range(n_points - s):
                y_var = y_vars[(i, s)]
                constraints.append(y_var <= h[i])
                constraints.append(y_var <= h[i + s])
                constraints.append(y_var >= h[i] + h[i + s] - 1)
        
        # Overlap inequality: t <= Sum over i of y_{i,s} for each s >= 1
        for s in range(1, n_points):
            overlap_sum = cp.sum([y_vars[(i, s)] for i in range(n_points - s)])
            constraints.append(t <= overlap_sum)
        
        # Objective
        objective = cp.Maximize(t)
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        # Solve
        prob.solve(solver=cp.CBC, max_seconds=max_seconds, verbose=False)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = np.round(h.value).astype(int)
            t_val = t.value
            
            # Compute A_s values
            A_s = []
            for s in range(1, n_points):
                A_s.append(np.sum(h_sol[:-s] * h_sol[s:]))
            
            best_A = min(A_s)
            c5 = 1 - 2*best_A / n_points
            return h_sol, c5, t_val
        else:
            # return a feasible solution manually, but check later
            h_sol = np.zeros(n_points, dtype=int)

            return h_sol, c5, best_A

                break

            i0 = np.random.choice(zeros)

            old_A = A_s.copy()
            new_h = h.copy()
            new_h[i1] = 0
            new_h[i0] = 1
            
            updated_A = []
            any_changed = False
            for s in range(1, n):
                old_a = old_A[s-1]
                change = 0
                
                # Which terms need update?

                if i1 - s >= 0 and i1 - s < n:
                    change -= new_h[i1 - s]
                if i1 + s < n:
                    change -= new_h[i1 + s]

                    change += new_h[i0 - s]
