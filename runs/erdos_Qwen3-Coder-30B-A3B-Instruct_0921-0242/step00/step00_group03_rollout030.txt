def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős constant for a given step function h."""
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part (the maximum value of the overlap)
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    """Optimize for an improved upper bound for C₅."""
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Get initial construction
    h0 = initial_h_values()
    
    # Define constraints
    n_points = len(h0)  # 70 points
    dx = 2.0 / n_points
    target_sum = 1.0
    
    # Wrapper to evaluate our function
    def eval_func(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / (np.sum(h) * dx))
        return evaluate_erdos_solution(h, dx)
    
    # Objective function to minimize
    def objective(h):
        h = np.clip(h, 0, 1)
        new_h = h * (target_sum / (np.sum(h) * dx))
        return eval_func(new_h)
    
    # Let's also define a non-optimal but simple strategy with symmetry
    def get_initial():
        # Let's use a better designed initial solution manually
        # Maybe a few "bumps" or spikes to reduce overlap
        h = np.zeros(n_points)
        # Maybe put mass in middle with some gaps
        mid = n_points // 2
        b = 5  # width of bump
        h[max(0, mid - b):min(n_points, mid + b)] = 0.5
        
        # Normalize so integral is 1
        h = h * (target_sum / (np.sum(h) * dx))
        return h
    
    # Minimize using differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        result = differential_evolution(
            objective,
            bounds,
            seed=seed,
            maxiter=200,  # limit iterations for faster runtime
            popsize=10,
            mutation=(0.5, 1),
            recombination=0.7,
            callback=None,
            disp=False
        )
        
        h_opt = np.clip(result.x, 0, 1)
        h_opt = h_opt * (target_sum / (np.sum(h_opt) * dx))
        c5_bound = evaluate_erdos_solution(h_opt, dx)
        
    except Exception as e:
        # If optimization fails, at least return the simple initial solution
        h_opt = get_initial()
        c5_bound = evaluate_erdos_solution(h_opt, dx)
    
    return (h_opt, c5_bound, n_points)
