import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx=2.0 / len(h)):
    """Compute the C5 constant using the correlation-based method."""
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Try different discretization sizes
    for n in [100, 120]:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create the MILP model
        h = cp.Variable(n, boolean=True)
        t = cp.Variable(integer=True, lower_bound=0)
        
        # Constraints
        # Weight constraint
        weight_con = cp.Equality(cp.sum(h), n // 2)
        
        # Auxiliary variables and constraints for auto-correlations
        y_vars = {}
        constraints = [weight_con, t >= 0]
        
        shifts = range(1, n)
        total_y = 0
        for s in shifts:
            for i in range(n - s):
                y = cp.Variable(boolean=True)
                y_vars[(i, s)] = y
                total_y += y
                
                # McCormick constraints
                constraints.append(y <= h[i])
                constraints.append(y <= h[i + s])
                constraints.append(y >= h[i] + h[i + s] - 1)
        
        # Constraints to ensure t <= each overlap
        for s in shifts:
            overlap_sum = cp.sum([y_vars[(i, s)] for i in range(n - s)])
            constraints.append(t <= overlap_sum)

        problem = cp.Problem(cp.Maximize(t), constraints)
        
        # Solve the problem
        solver_time = min(300, (budget_s - (time.time() - start_time)) // 2)
        try:
            problem.solve(solver=cp.CBC, max_seconds=solver_time, verbose=False)
        except:
            continue
            
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_bin = np.round(h.value).astype(int)

            # Compute C5 bound from the MILP solution
            c5_milp = 1 - 2 * t_val / n
            
            if c5_milp < best_c5:
                best_c5 = c5_milp
                best_h = h_bin.copy()
                best_n = n
                
        if best_c5 < 0.3807:
            break
    
    # If we have a feasible solution, do local search improvement
    if best_h is not None and time.time() - start_time < budget_s - 5:
        n = best_n
        dx = 2.0 / n
        
        def compute_overlaps(hv):
            A = []

                A.append(a)

        t_local = min_overlap

            iter_count += 1
            
            # Find candidates where we could swap

            zeros = np.where(h_local == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            # Try a few random swaps
            for _ in range(10):  # Random neighborhood size
                i1 = np.random.choice(ones)
                i0 = np.random.choice(zeros)
                
                # Swap values
                h_new = h_local.copy()
                h_new[i1] = 0
                h_new[i0] = 1
                
                # Recompute all affected auto-correlations efficiently
                current_min = min_overlap
                new_min = float('inf')

                affected_shifts = set()
                for s in range(1, n):
                    if i1 + s < n or i0 + s < n or i1 - s >= 0 or i0 - s >= 0:
                        affected_shifts.add(s)
                
                # These are expensive to compute, so limit to a few
                for s in list(affected_shifts)[:5]:

                if new_min > current_min:
                    min_overlap = new_min
                    h_local = h_new
                    t_local = min_overlap
                        
        c5_final = 1 - 2 * t_local / n
        
        # Maybe run polish step if time allows
        if c5_final > 0.3807 and time.time() - start_time < budget_s - 30:

            def obj_con(h_c):
                dx = 2.0 / len(h_c)
                corr = np.correlate(h_c, 1 - h_c, mode="full") * dx
                return -max(corr)  # We're maximizing the correlation

                return np.sum(h_c) - 1.0
