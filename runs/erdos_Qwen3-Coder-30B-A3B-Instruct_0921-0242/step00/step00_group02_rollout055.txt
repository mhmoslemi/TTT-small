import numpy as np

def project_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum and bounds [0,1]"""
    n = len(h)
    h_clipped = np.clip(h, 0, 1)
    deficit = target_sum - np.sum(h_clipped)
    
    if abs(deficit) < 1e-12:
        return h_clipped
    
    # Create sorted indices
    idx_sorted = np.argsort(h_clipped)[::-1]
    h_sorted = h_clipped[idx_sorted]
    
    # Compute cumulative sum from the largest
    cumsum = np.cumsum(h_sorted)
    threshold = (cumsum - target_sum) / np.arange(1, n + 1)
    
    # Find the index to split
    idx = np.where(h_sorted > threshold)[0]
    if len(idx) == 0:
        # All entries get clamped to 0
        return np.zeros_like(h)
    
    k = idx[-1]
    # Adjust threshold
    tau = threshold[k]
    
    # Adjust the entries
    h_out = h_clipped.copy()
    for i in range(k, n):
        if h_sorted[i] > tau:
            h_out[idx_sorted[i]] = tau
        else:
            h_out[idx_sorted[i]] = 0
    
    # Now, re-adjust those elements that are still in bounds to maintain sum
    for i in range(k):
        h_out[idx_sorted[i]] = h_sorted[i]
    
    # Clipping to [0,1] and final adjustment if necessary
    h_out = np.clip(h_out, 0, 1)
    return h_out

def compute_autocorrelations(h):
    """Compute linear autocorrelations using FFT"""
    n = len(h)
    h_padded = np.concatenate([h, np.zeros(n)])
    H = fft(h_padded)
    correlations = ifft(H * np.conj(H)).real
    # Return r_k for k = 1, 2, ..., n-1
    return correlations[1:n]

def soft_min_alpha(h, alpha):
    r = compute_autocorrelations(h)
    r_shifted = -alpha * r
    # Subtract the maximum to avoid numerical overflow
    r_shifted -= np.max(r_shifted)
    exps = np.exp(r_shifted)
    s = - (1 / alpha) * np.log(np.sum(exps))
    return s

    # But this over-estimates

    # d/dh_i r_k = h_{i+k} + h_{i-k} (when both are in bounds)

    grad = np.zeros(n)

def nesterov_step(h, grad, beta=0.9, step_size=None):
    # Basic Nesterov upgrade
    h_prev = h.copy()
    y = h + beta * (h - h_prev)
    if step_size is None:
        step_size = 1e-3 
    h_new = y - step_size * grad
    return h_new, h_prev

def binary_rounding(h):
    """Binary rounding to get step function"""
    n = len(h)
    # Create indices to sort elements in descending order
    idx = np.argsort(h)[::-1]
    h_rounded = np.zeros(n)
    for i in range(n // 2):
        h_rounded[idx[i]] = 1
    return h_rounded

def swap_refinement(h, target_sum, n_points):
    """Refine binary solution by local swaps to improve worst correlation"""
    # Nicer approach if we can calculate efficient change in r_k with swaps, 
    # but for first implementation we compute new r_k and evaluate improvement
    
    # auto-correlations by expensive method, computed every few swaps
    h = binary_rounding(h)
    if np.abs(np.sum(h) - target_sum) > 1e-8:
        h = project_simplex(h, target_sum) 
        h = binary_rounding(h)

    # Get r_k
    r = compute_autocorrelations(h)
    # Find critical k*
    k_star = np.argmin(r)
    best_h = h.copy()
    best_r_k = r[k_star]
    
    # swap neighborhood search
    # limit the search for efficiency 
    max_swaps = 2000
    iter_swaps = 0
    
    while iter_swaps < max_swaps:
        # Identify critical shift
        # r_k = sum h[i]*h[i+k], 0<=i<n-k, max_k r_k = -min_k -r_k
        r = compute_autocorrelations(h)
        k_star = np.argmin(r)
        
        # For any swap between 1 and 0, compute its impact on r_{k_star}
        # Better to generate pairs carefully
        
        # Find ones and zeros  
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # try to find a good swap quickly - simple strategy: gready
        best_swap = None
        best_impact = 0
        
        for i in ones:
            for j in zeros:
                if abs(i - j) <= 30:  # neighborhood constraint on swaps
                    # Compute change in r_{k_star}
                    old = h[i] * h[i + k_star] + h[j] * h[j + k_star]  # if valid
                    if i + k_star < len(h) and j + k_star < len(h):
                        new = h[j] * h[j + k_star] + h[i] * h[i + k_star]
                        delta = new - old 
                        if delta < best_impact:  # improve r_k_star
                            best_impact = delta
                            best_swap = (i, j)
                        
        if best_swap is None:
            break
            
        i, j = best_swap
        h[i], h[j] = h[j], h[i]
        iter_swaps += 1
        r = compute_autocorrelations(h)
        if r[k_star] < best_r_k:
            best_r_k = r[k_star]
            best_h = h.copy()
            
    return best_h, best_r_k

def evaluate_C5(h, dx=1.0):
    # Convolutions with causality
    correlated = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlated) * dx
    return c5
