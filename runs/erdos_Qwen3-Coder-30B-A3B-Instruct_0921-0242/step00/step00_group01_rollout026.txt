import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft

def evaluate_c5(h):
    """Evaluate the C5 constant for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute autocorrelations using FFT
    H = fft(h)
    A = ifft(np.abs(H)**2).real
    # The overlap integral is dx * (n/2 - min(A))
    return dx * (n/2 - A.min())

def block_family_sweep():
    """Enumerate 3-block and 4-block families and return the best."""
    best_h = None
    best_c5 = float('inf')
    n_base = 400  # Base grid resolution for evaluation
    dx_base = 2.0 / n_base
    # 3-block: [1, p, 0]
    a_vals = np.arange(0.02, 2.0, 0.02)
    b_vals = np.arange(0.02, 2.0, 0.02)
    for a in a_vals:
        for b in b_vals:
            if a >= b: continue
            p = (1.0 - a) / (b - a)
            if 0 <= p <= 1:
                h = np.zeros(n_base)
                h[:int(a/dx_base)] = 1.0
                h[int(a/dx_base):int(b/dx_base)] = p
                # Normalize to sum = n_base/2
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (n_base/2) / h_sum
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    # 4-block: [1, p, q, 0] with balance condition
    a_vals = np.arange(0.04, 2.0, 0.04)
    b_vals = np.arange(0.04, 2.0, 0.04)
    c_vals = np.arange(0.04, 2.0, 0.04)
    for a in a_vals:
        for b in b_vals:
            for c in c_vals:
                if a >= b or b >= c: continue
                # Balance condition makes total middle mass equal to half total
                # a*1 + (b-a)*p + (c-b)*q = 1 and (b-a)*p = (c-b)*q
                p = (1.0 - a * (c-b)) / (b - a + c - b)
                q = (1.0 - a * (b - a) - p * (b - a)) / (c - b)
                if 0 <= p <= 1 and 0 <= q <= 1:
                    h = np.zeros(n_base)
                    h[:int(a/dx_base)] = 1
                    h[int(a/dx_base):int(b/dx_base)] = p
                    h[int(b/dx_base):int(c/dx_base)] = q
                    h_sum = np.sum(h)
                    if h_sum > 0:
                        h = h * (n_base/2) / h_sum
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    return best_h, best_c5

def gaussian_mixture(n, m=3):
    """Create an initial h using Gaussian-shaped bumps."""
    c = np.random.uniform(0, 2, m)
    a = np.random.uniform(0, 1, m)
    ell = 0.1
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for i in range(m):
        h += a[i] * (np.abs(x - c[i]) <= ell)
    # Normalize to sum = n/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n/2) / h_sum
    h = np.clip(h, 0, 1)
    return h

def de_candidate(n, bounds, n_pop=30, maxiter=200):
    """Run differential evolution on a low-dim parameterized space."""
    def de_fitness(params):
        # Each param: 2m for centers and amplitudes
        m = len(params) // 2
        c = params[:m]
        a = np.clip(params[m:], 0, 1)
        # Construct h using Gaussian-like bumps, then normalize
        ell = 0.1
        x = np.linspace(0, 2, n)
        h = np.zeros(n)
        for i in range(m):
            h += a[i] * (np.abs(x - c[i]) <= ell)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n/2) / h_sum
        h = np.clip(h, 0, 1)
        return -evaluate_c5(h)
    # Run differential evolution
    res = differential_evolution(
        de_fitness,
        bounds=bounds,
        maxiter=maxiter,
        popsize=n_pop,
        seed=42,
        disp=False
    )
    # Reconstruct h from best params
    m = len(bounds) // 2
    c = res.x[:m]
    a = np.clip(res.x[m:], 0, 1)
    ell = 0.1
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    for i in range(m):
        h += a[i] * (np.abs(x - c[i]) <= ell)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n/2) / h_sum
    h = np.clip(h, 0, 1)
    return h, -res.fun

def project_to_feasible(h, n):
    """Project h to feasible set: [0,1] and sum = n/2"""
    h = np.clip(h, 0, 1)
    diff = (h.sum() - n/2) / n
    h = h - diff
    h = np.clip(h, 0, 1)
    h = h - (h.sum() - n/2) / n
    h = np.clip(h, 0, 1)
    return h

def projected_gradient_ascent(start_h, n_final, budget_s, early_stop_iter=200):
    """Project gradient ascent with multiresolution approach."""
    t0 = time.time()
    max_iter = 5000  # Roughly time budget after other stages
    n = 150
    h = np.linspace(0, 1, n)  # used earlier, kept for fallback
    if start_h is not None:
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(start_h)), start_h)

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    last_improve = 0
    iter_count = 0

    # Set parameters for gradient ascent
    eta = 0.5
    decay_factor = 0.98
    prev_h = None

    while iter_count < max_iter:
        if time.time() - t0 > budget_s - 2:
            break

        # Compute A_k via FFT
        H = fft(h)
        A = ifft(np.abs(H)**2).real
        k_star = np.argmin(A)

        # Compute gradient
        g = 2 * np.roll(h, k_star)

        # Step
        h_new = h + eta * g
        h_new = project_to_feasible(h_new, n)
        h = h_new.copy()

        iter_count += 1
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improve = 0
        else:
            last_improve += 1

        if last_improve > early_stop_iter:
            break

        # Improve linearly up to n_final
        if n < n_final and iter_count > 1500:
            # Up-sample to higher resolution
            h_up = np.interp(np.linspace(0, 2, n_final), np.linspace(0, 2, n), h)
            h = project_to_feasible(h_up, n_final)
            n = n_final
            eta *= 0.5  # Reduce step size to stabilize
            last_improve = 0

        # Decaying step size
        eta *= decay_factor

    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    t0 = time.time()
    best_c5 = float('inf')
    best_h = None

    # Stage 1: Block family sweep
    if time.time() - t0 < budget_s - 50:
        h_block, c5_block = block_family_sweep()
        if h_block is not None and c5_block < best_c5:
            best_c5 = c5_block
            best_h = h_block.copy()

    # Stage 2: Gaussian mixture optimization
    if time.time() - t0 < budget_s - 50:
        n_de = 200  # sample grid for DE
        bounds = [(0.0, 2.0)] * 3 + [(0.0, 1.0)] * 3  # 3 centers and 3 amplitudes
        h_de, c5_de = de_candidate(n_de, bounds)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()

    # Stage 3: Projected gradient descent with multiresolution
    if time.time() - t0 < budget_s - 50 and best_h is not None:
        n_final = 300
        h_pg, c5_pg = projected_gradient_ascent(best_h, n_final, budget_s - (time.time() - t0))
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
    
    # Post-processing to ensure final n = 500 for best reporting
    n_final = 500
    if best_h is None:
        best_h = gaussian_mixture(n_final, m=3)
    else:
        best_h = np.interp(np.linspace(0, 2, n_final), np.linspace(0, 2, len(best_h)), best_h)
    best_h = project_to_feasible(best_h, n_final)
    final_c5 = evaluate_c5(best_h)

    return list(best_h), final_c5, n_final
