import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all lags
    correlations = np.correlate(h, 1 - h, mode="full")
    # The maximum correlation value times dx gives the C₅ bound
    c5 = np.max(correlations) * dx
    return c5

def initial_h_values():
    """Return an initial construction for h."""
    # Use a simple initial construction: 1/2 on [0,1) and 0 on [1,2)
    n = 47
    h = np.array([0.5] * n)
    dx = 2.0 / n
    # Adjust to make integral exactly 1
    h[0] = 1.0  # Use one point as 1 to compensate
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction
    h_init = initial_h_values()
    n_points = len(h_init)
    dx = 2.0 / n_points
    h_best = h_init.copy()
    c5_best = evaluate_erdos_solution(h_best)
    
    # Define bounds and constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function
    def objective(h):
        # Apply normalization
        h = np.array(h)
        if np.sum(h) * dx != 1.0:
            scale = 1.0 / (np.sum(h) * dx)  # Scale to make integral = 1
            h = h * scale
        return evaluate_erdos_solution(h)
    
    # Equality constraint function: integral = 1
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # Constraints dictionary
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    # Use differential evolution if less than 200 points
    if n_points < 200:
        from scipy.optimize import differential_evolution
        res = differential_evolution(objective, bounds, constraints=constraints, 
                                     seed=seed, maxiter=1000, popsize=15, 
                                     mutation=(0.5, 1.0), recombination=0.7)
        h_opt = res.x
        if np.sum(h_opt) * dx != 1.0:
            scale = 1.0 / (np.sum(h_opt) * dx)
            h_opt = h_opt * scale
        c5_opt = evaluate_erdos_solution(h_opt)
        if c5_opt < c5_best:
            h_best = h_opt
            c5_best = c5_opt

    else:
        # Use SLSQP for larger problems
        # Resort to random search if time remains
        for i in range(20000):
            if time.time() - start_time > budget_s - 1:
                break
            # Random perturbation of the current best
            h_trial = h_best.copy()
            idx = np.random.randint(0, n_points)
            # Add small perturbation
            delta = np.random.normal(0, 0.05)
            h_trial[idx] = max(0, min(1, h_trial[idx] + delta))
            
            # Ensure sum constraint is maintained
            h_trial = h_trial * (1.0 / (np.sum(h_trial) * dx))  # Normalize
            c5_trial = evaluate_erdos_solution(h_trial)
            
            if c5_trial < c5_best:
                h_best = h_trial.copy()
                c5_best = c5_trial
                
    return (h_best, c5_best, n_points)
