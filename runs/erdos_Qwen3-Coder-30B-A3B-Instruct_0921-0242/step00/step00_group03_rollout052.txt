def c5_of_h(h):
    """Computes the C5 bound for a given h."""
    dx = 2.0 / len(h)
    g = 1 - h
    conv = np.correlate(h, g, mode="full") * dx
    return np.max(conv)

def gradient_of_c5(h):
    """Computes gradient of C5 at h, or None if not computable."""
    dx = 2.0 / len(h)
    g = 1 - h
    conv = np.correlate(h, g, mode="full")
    max_idx = np.argmax(conv)
    k_star = max_idx - len(h) + 1  # Convert index to shift k
    grad = np.zeros_like(h)
    try:
        for i in range(len(h)):
            if i + k_star < len(h) and i + k_star >= 0:
                grad[i] += -dx * g[i + k_star]
            if i - k_star >= 0 and i - k_star < len(h):
                grad[i] += dx * h[i - k_star]
        return grad
    except:
        return None

def project_to_feasible(h, target_sum):
    """Projects h to satisfy sum constraint."""
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    if abs(delta) < 1e-10:
        return h
    # Simple projection: adjust by small amount
    # Try to distribute proportional to non-boundary points
    h = np.clip(h, 0, 1)
    if np.sum(h) == 0:
        return h
    h = h + delta * (h > 0) / np.sum(h > 0)
    h = np.clip(h, 0, 1)
    return h
