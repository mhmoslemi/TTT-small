import numpy as np
import time
import scipy.signal

def project_box_affine(h, target_sum):
    """Project h onto the box [0,1] and affine constraint sum = target_sum"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Project onto sum constraint
    delta = (np.sum(h) - target_sum) / len(h)
    h -= delta
    # Clip again
    h = np.clip(h, 0, 1)
    return h

def autocorr(h):
    """Compute circular autocorrelation using FFT for speed"""
    n = len(h)
    fft_h = np.fft.fft(h)
    aut = np.real(np.fft.ifft(np.abs(fft_h)**2))
    return aut

def compute_c5(h, dx):
    """Compute C5 = max_k \int h(x)(1-h(x+k)) dx"""
    h_conv = scipy.signal.correlate(h, 1-h, mode='full')
    # Only take the part from k=0 to max_k=n-1
    k_max = len(h_conv) // 2
    h_conv = h_conv[len(h)-1:]
    return np.max(h_conv) * dx

def gradient_ascent_step(h, dx, step_size):
    """Perform one gradient ascent step for C5 minimization"""
    n = len(h)
    aut = autocorr(h)
    lag = np.argmin(aut)  # Index of minimal autocorrelation
    
    # Build gradient of -aut[lag] w.r.t. h
    grad = -np.roll(h, lag) - np.roll(h, -lag)
    h_new = h + step_size * grad
    
    # Project back to feasible set
    h = project_box_affine(h_new, n/2)
    return h

    else:
        h_interp = initial_h[:n_points]
    candidates.append(h_interp.copy())

    # Smooth

    h_smooth = np.convolve(h_random, kernel, mode='same')
    h_smooth = project_box_affine(h_smooth, n_points / 2)
    candidates.append(h_smooth.copy())

    # Alternating blocks (0.5s, 0s, 0.5s, 0s, etc.)
    h_alternating = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h_alternating[i] = 0.5
    h_alternating = project_box_affine(h_alternating, n_points / 2)
    candidates.append(h_alternating.copy())

    # Two-block construction (similar to an optimal 2-block pattern)
    # Place a=0.5, b=1.5 (for a 1-unit block of 1s and 1-unit block of 0s)
    h_two_block = np.ones(n_points)
    a = 0.5
    b = 1.5
    for i in range(n_points):
        x = i * dx
        if a <= x < b:
            # Calculate value needed for integral = 1
            L1 = a
            L2 = b - a
            L3 = 2.0 - b
            c = (1.0 - L1) / L2 if L2 > 0 else 0
            if 0 <= c <= 1:
                h_two_block[i] = c
            else:
                h_two_block[i] = 0
        elif x >= b:
            h_two_block[i] = 0
    h_two_block = project_box_affine(h_two_block, n_points / 2)
    candidates.append(h_two_block.copy())

    return candidates

def block_search(n_points, dx):
    """Sweep over parameterized two- and three-block functions"""
    best_c5 = float('inf')
    best_h = None
    min_step = 0.02

    # Two-block search
    for a in np.arange(min_step, 2.0 - min_step, min_step):
        for b in np.arange(a + min_step, 2.0 - min_step, min_step):
            h = np.zeros(n_points)
            L1 = a
            L2 = b - a
            L3 = 2.0 - b
            if L2 > 0:
                c = (1.0 - L1) / L2
                if 0 <= c <= 1:
                    for i in range(n_points):
                        x = i * dx
                        if x < a:
                            h[i] = 1.0
                        elif x < b:
                            h[i] = c
                        else:
                            h[i] = 0.0
                    h = project_box_affine(h, n_points / 2)
                    c5 = compute_c5(h, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    return best_h, best_c5
