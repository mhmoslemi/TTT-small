import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def evaluate_c5(h):
    """Evaluate the maximum overlap integral C5 for discrete h."""
    h = np.array(h)
    dx = 2.0 / len(h)
    # Compute all correlations (full mode)
    corr = correlate(h, 1 - h, mode='full')
    # Only the part where k is from -(n-1) to n-1, that yields valid shifted values
    # The middle (n-1) entries (for k=0 to 2*n-3) are the valid correlations
    # The middle entry corresponds to k=0, others correspond to different k values
    corr = corr[len(h)-1: 2*len(h)-1]  # This indexes shifting by k from -(n-1) to n-1
    return max(corr) * dx

def solve_sdp(n_points, time_limit=200):
    """Solve the SDP relaxation for the Erdős problem."""
    dx = 2.0 / n_points
    h = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,  # X is positive semidefinite
        cp.diag(X) == h,  # diagonal elements of X equal h (this is a linear constraint)
        0 <= h, h <= 1,  # Box constraints
        cp.sum(h) == n_points / 2  # sum constraint
    ]

    # Add correlation constraints
    # We'll loop over shifts k from -(n-1) to n-1
    # This corresponds to s = k + n-1
    for s in range(2 * n_points - 1):
        # Create permutation matrix P_s
        P_s = np.zeros((n_points, n_points))
        offset = s - (n_points - 1)
        if offset >= 0:
            # P_s[i, j] = 1 iff j = i + offset (mod n)
            for i in range(n_points - offset):
                P_s[i, i + offset] = 1
        else:
            # offset < 0, j = i + offset for i in range(-offset, n):
            for i in range(-offset, n_points):
                P_s[i, i + offset] = 1
        # Constraint: t >= const - dx * Tr(P_s * X)
        # const = dx * n/2
        const = dx * (n_points / 2)
        constraints.append(t >= const - dx * cp.trace(P_s @ X))

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    start_time = time.time()
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, time_limit=time_limit)
        end_time = time.time()
        if prob.status == cp.OPTIMAL:
            return h.value, t.value, end_time - start_time
        else:
            return None, None, end_time - start_time
    except Exception as e:
        return None, None, time.time() - start_time

def round_feasible(h_sdp, n_points, num_rounds=30):
    """Round SDP solution to obtain feasible discrete h vectors."""
    h_sdp = np.array(h_sdp)
    best_h = None
    best_c5 = np.inf
    
    for _ in range(num_rounds):
        # Random Gaussian vector
        g = np.random.randn(n_points)
        # Construct y from eigen-decomposition of X* (but here we assume h_sdp is extracted)
        # We treat h_sdp itself as a feasible point (x = 1, y = h_sdp).
        # So we simply perturb it slightly for noise.
        y = h_sdp + 0.01 * g
        
        # Clip to [0, 1]
        y = np.clip(y, 0, 1)
        
        # Adjust to match sum constraint
        current_sum = np.sum(y)
        deficit = n_points / 2 - current_sum
        
        # Distribute deficit among elements (uniformly to valid indices)
        if abs(deficit) > 1e-6:
            valid_indices = np.where((y > 0) & (y < 1))[0]
            if len(valid_indices) > 0:
                to_add = deficit / len(valid_indices)
                for i in valid_indices:
                    y[i] = min(y[i] + to_add, 1.0)
        
        # Re-adjust sum exactly
        y = np.clip(y, 0, 1)
        current_sum = np.sum(y)
        diff = n_points / 2 - current_sum
        if abs(diff) > 1e-8:
            # Find indices to distribute the adjustment
            increase_indices = np.where(y < 1)[0]
            decrease_indices = np.where(y > 0)[0]
            for i in range(int(abs(diff))):
                if diff > 0 and len(increase_indices) > 0:
                    idx = increase_indices[i % len(increase_indices)]
                    if y[idx] < 1:
                        y[idx] += 1 if diff >= 1 else diff
                        diff -= 1 if diff >= 1 else diff
                elif diff < 0 and len(decrease_indices) > 0:
                    idx = decrease_indices[i % len(decrease_indices)]
                    if y[idx] > 0:
                        y[idx] -= 1 if abs(diff) >= 1 else abs(diff)
                        diff += 1 if abs(diff) >= 1 else abs(diff)
                else:
                    break
            
        # Make sure sum is exactly n/2
        y = y / np.sum(y) * (n_points / 2)
        
        c5 = evaluate_c5(y)
        if c5 < best_c5:
            best_c5 = c5
            best_h = y.copy()
    
    return best_h, best_c5
