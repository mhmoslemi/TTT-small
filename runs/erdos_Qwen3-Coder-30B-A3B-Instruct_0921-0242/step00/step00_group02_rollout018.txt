def evaluate_erdos_solution(h_values):
    """Compute C5 from discrete h vector."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx


def solve_sdp(n, verbose=False):
    """Solve the SDP relaxation of the Erdős problem."""
    # Define decision variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)

    # Objective: maximize the minimum correlation
    s = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,  # X is positive semidefinite
        cp.diag(X) == h,  # Diagonal of X equals h
        0 <= h,  # h bounded below by 0
        h <= 1,  # h bounded above by 1
        cp.sum(h) == n / 2  # mass constraint
    ]

    # Add correlation constraints
    for k in range(1, n):
        # Sum of X[i, i+k] for i = 0 to n-1-k
        corr_sum = cp.sum([X[i, i + k] for i in range(n - k)])
        constraints.append(corr_sum >= s)

    # Formulate and solve the problem
    prob = cp.Problem(cp.Maximize(s), constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=verbose, eps=1e-4, max_iters=5000)
        # Output debug info if needed
        # print(f"SDP solved with status {prob.status}, optimal value: {prob.value}")
    except Exception as e:
        # print("SDP solving issue:", e) 
        return None, None

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h.value, X.value
    else:
        # Fallback to no solution
        return None, None


def round_from_sdp(X):
    """Round a PSD matrix to binary solution using Gaussian hyperplane rounding"""
    # Compute a factorization of X
    U, s, Vt = np.linalg.svd(X)
    rank_X = np.sum(s > 1e-10)
    # Keep only rank > 0 components; input rank might be noisy
    r = min(rank_X, 10)  # limit to 10
    V = U[:, :r]
    # Draw random Gaussian vector
    g = np.random.randn(r)
    # Project to `z` scores
    z = V @ g
    # Threshold to get binary vector: keep top n/2
    target_count = int(np.ceil(len(z) / 2))
    # Find the correct threshold so we have exactly target_count ones
    sorted_z = np.sort(z)[::-1]
    if target_count >= len(sorted_z):
        tau = -np.inf  # All ones
    else:
        tau = sorted_z[target_count - 1]
    b = (z >= tau).astype(float)
    # If the actual sum is wrong (rare), correct it greedily
    current_sum = np.sum(b)
    deficit = int(current_sum - target_count)
    if deficit < 0:
        # Need more ones: flip the lowest values (smallest z) to 1
        to_flip = np.argpartition(-z, int(-deficit))[:int(-deficit)]
        b[to_flip] = 1
    elif deficit > 0:
        # Need fewer ones: flip the highest values (largest z) to 0
        to_flip = np.argpartition(z, int(deficit))[:int(deficit)]
        b[to_flip] = 0
    return b


def refine_binary_solution(h_opt, n):
    """Refine a binary solution by gradient ascent on correlations"""
    h = h_opt.copy()

    def compute_correlations(h):
        # Use equivalent formulation that leaves indexing clear
        n = len(h)
        correlations = []
        for k in range(1, n):
            total = sum(h[i] * h[i+k] for i in range(n - k))
            correlations.append(total)
        return correlations

    def gradient_h_on_shift_k(h, k):
        # Derivative of sum h[i]*h[i+k] w.r.t. h[i]
        # Each h[i] appears in two terms:
        # i contributes to shift k: term h[i+k]
        # i+k contributes to shift k: term h[i]
        grad = np.zeros_like(h)
        for i in range(min(len(h), len(h) - k)):
            grad[i] += h[i + k]
            grad[i + k] += h[i]
        return grad

    h = h.copy()
    for _ in range(30):  # few iterations to polish
        corr = compute_correlations(h)
        if len(corr) == 0:
            break
        min_k = np.argmin(corr)
        grad = gradient_h_on_shift_k(h, min_k + 1)
        step_size = 0.05
        h += step_size * grad
        # project back to feasible region
        h = np.clip(h, 0, 1)
        # enforce mass = n/2 by scaling or thresholding
        # simple && effective correction mechanism
        h = np.sort(h)[::-1]
        # Ensure sum is correct
        change = (n / 2) - np.sum(h)
        if change > 0:
            h[-1] = min(h[-1] + change, 1.0)
        else:
            h[0] = max(h[0] + change, 0.0)
        h = np.sort(h)[::-1]  # restore sorted order
    return h

def run(seed=42, budget_s=600):
    r = np.random.RandomState(seed)
    start_time = time.time()
    best_C5 = float('inf')
    best_h = None
    best_n = 0
    
    # initialize arrays for different resolutions to use
    resolutions = [42, 38, 30] # [38, 30] may be less the first one is
    for n in resolutions:
        if best_C5 <= 0.3807:
            return best_h, best_C5, best_n  # early exit if won
        if time.time() - start_time > budget_s - 5:
            break  # stop if time's nearly over
        # Define SDC options
        max_time = 30 if n == 42 else 25 if n == 38 else 20  # allocation time based on resolution
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < max_time:
            break

        # Solve SDP
        h_opt, X_opt = solve_sdp(n)
        if h_opt is None or X_opt is None:
            continue
        # Convert to feasible solution (reliably)
        good_solution_found = False
        best_rounded = None
        best_rounded_corr = float('inf')
        rounds = 0
        best_corr_rounded = float('inf')
        while rounds < 300:
            if time.time() - start_time > budget_s - 5:
                break
            rounds += 1
            rounded = round_from_sdp(X_opt)
            # Limited polishing, just verify basic validity rules
            # Check constraints 
            if not (np.all(0 <= rounded) and np.all(rounded <= 1)):
                continue
            if abs(np.sum(rounded) - n / 2) > 1e-8:
                continue

            h = local_swap_polish(rounded, 50)
            CORR = evaluate_erdos_solution(h)
            if CORR < best_rounded_corr:
                best_rounded_corr = CORR
                best_rounded = h.copy()
            if CORR < best_C5:
                best_C5 = CORR
                best_h = h.copy()
                best_n = n
                # if already good, still run a bit for better ones...
                if CORR <= 0.3807:
                    break  # stop early
        # Optional refinement of H by gradient descent on its fractional image
        # useful feedback loop for better x, not current shape but pretty good already
        if best_rounded is not None and time.time() - start_time < budget_s - 5:
            refined = refine_binary_solution(best_rounded, n)
            corr_refined = evaluate_erdos_solution(refined)
            if corr_refined < best_C5:
                best_C5 = corr_refined
                best_h = refined.copy()
                best_n = n
                if corr_refined <= 0.3807:
                    break  # early exit if we found the target

    # Finalize output
    if best_h is not None:
        real_c5 = evaluate_erdos_solution(best_h)
        return best_h, real_c5, best_n
    else:
        # fallback to an initial construction
        initial_n = len(initial_h_values)
        dx = 2.0 / initial_n
        h_vals = initial_h_values.copy()
        corr = evaluate_erdos_solution(h_vals)
        return h_vals, corr, initial_n
