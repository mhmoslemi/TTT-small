import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import math

def evaluate(h):
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, target):
    x = np.array(x, dtype=float)
    x = np.clip(x, 0, 1)
    curr_sum = np.sum(x)
    if abs(curr_sum - target) < 1e-10:
        return x
    diff = target - curr_sum
    # Use iterative water-filling
    x = np.clip(x, 0, 1)
    for _ in range(20):
        if abs(diff) < 1e-10:
            break
        free = 1.0 - x
        # np.count_nonzero(free > 1e-10)
        free_indices = np.where(free > 1e-10)[0]
        if len(free_indices) == 0:
            break
        # Distribute diff uniformly among free entries
        delta = diff / len(free_indices)
        for i in free_indices:
            absorb = min(free[i], -delta if delta < 0 else delta)
            x[i] += absorb
            diff -= absorb
            if abs(diff) < 1e-10:
                break
        if abs(diff) < 1e-10:
            break
    x = np.clip(x, 0, 1)
    return x

def active_shifts(h, eps=1e-9):
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    active = []
    for k in range(len(corr)):
        if abs(corr[k] * dx - c5) < eps:
            active.append(k)
    return active

def gradient(h, ks):
    n = len(h)
    g = np.zeros(n)
    dx = 2.0 / n
    for k in ks:
        # Left shift part
        for i in range(n):
            j = i + k
            if j < n:
                g[i] += 1 - h[j]
            else:
                g[i] += 0
        # Right shift part
        for i in range(n):
            j = i - k
            if j >= 0:
                g[i] += h[j]
            else:
                g[i] += 0
    return g / len(ks)

def line_search(h, g, t_current, budget_s):
    if budget_s < 30:
        step_sizes = [0, 1e-3, 5e-3, 1e-2]
    else:
        step_sizes = [0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1]
    best_h = h.copy()
    best_c = t_current
    for alpha in step_sizes:
        if alpha == 0:
            continue
        h_new = h - alpha * g
        h_new = project_feasible(h_new, 1.0)
        c_new = evaluate(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
    return best_h, best_c

def local_optimize(h0, max_iter, t_end):
    h = h0.copy()
    c = evaluate(h)
    best_h = h.copy()
    best_c = c
    for i in range(max_iter):
        if time.time() > t_end:
            break
        # Compute active shifts
        active = active_shifts(h, 1e-8)
        # Compute gradient
        g = gradient(h, active)
        # Line search
        h_new, c_new = line_search(h, g, c, t_end - time.time())
        h = h_new
        c = c_new
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def anneal(h0, t_end, t0=0.01, decay=0.995, steps=5000):
    h = h0.copy()
    c = evaluate(h)
    best_h = h.copy()
    best_c = c
    T = t0
    for _ in range(steps):
        if time.time() > t_end:
            break
        i, j = np.random.choice(len(h), 2, replace=False)
        delta = np.random.uniform(-0.05, 0.05)
        h_new = h.copy()
        delta = max(-h_new[i], min(delta, 1.0 - h_new[i]))
        delta_j = -delta
        h_new[i] += delta
        h_new[j] += delta_j
        # Project
        h_new = project_feasible(h_new, 1.0)
        c_new = evaluate(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        elif T > 1e-10:
            if np.random.rand() < np.exp(-(c_new - c) / T):
                h = h_new
                c = c_new
        T *= decay
    return best_h, best_c

def random_feasible(n):
    h = np.random.rand(n)
    h = h / np.sum(h) * 1.0  # Scale to sum to 1
    h = project_feasible(h, 1.0)
    return h

def interpolate_and_project(h, n_new):
    n_old = len(h)
    old_x = np.linspace(0, 2, n_old)
    new_x = np.linspace(0, 2, n_new)
    h_interp = np.interp(new_x, old_x, h)
    return project_feasible(h_interp, 1.0)

def milp_binary(n, time_limit):
    try:
        # Build the MILP model
        h = cp.Variable(n, boolean=True)
        # Integrate constraint: sum(h) = n/2
        constraints = [cp.sum(h) == n / 2.0]
        # For each shift k
        for k in range(1, n):
            # Introduce auxiliary variables a_{i,k}
            a = cp.Variable(n, boolean=True)
            # Linear constraints for a[i,k] <= h[i] and a[i,k] <= h[i+k], etc.
            for i in range(n):
                j = i + k
                if j < n:
                    constraints.append(a[i] <= h[i])
                    constraints.append(a[i] <= h[j])
                    # a >= h[i] + h[j] - 1
                    constraints.append(a[i] >= h[i] + h[j] - 1)
                    constraints.append(a[i] >= 0)
                else:
                    # No constraint in this case, but a is 0 if j >= n
                    constraints.append(a[i] == 0)
            # Constraint: sum(a[i,k]) >= (1 - t) / dx
            # Let's use dx = 2/n, so (1 - t)/dx = (1 - t)*n/2
            constraints.append(cp.sum(a) >= 1.0 - 0.0)  # This is a placeholder
        # Minimize t (but we don't have t yet) - just minimize 0.
        # Instead, solve the same model, but check for feasibility
        # We need to ensure feasibility only for a binary problem.
        # For now, we'll keep hardware constraints and just solve the integer feasibility
        prob = cp.Problem(cp.Minimize(0), constraints)
        solver = cp.CBC
        try:
            prob.solve(solver=solver, max_seconds=time_limit)
        except:
            try:
                # fallback to GLPK
                prob.solve(solver=cp.GLPK, max_seconds=time_limit)
            except:
                return None, float("inf")  # failed
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = h.value
            if h_sol is not None:
                return np.array(h_sol, dtype=float), 0.0
        else:
            return None, float("inf")
    except Exception:
        return None, float("inf")
    return None, float("inf")
