        M_list = []

            diag = np.ones(n-k)
            M = sparse.diags(diag, k, shape=(n,n), format='csr')
            M_list.append(M)
        
        # Variables
        X = cp.Variable((n,n), symmetric=True)
        h = cp.Variable(n)
        L = cp.Variable()
        
        # Constraints
        constraints = [
            X >> 0,  # PSD
            0 <= h, h <= 1,  # box constraints
            cp.sum(h) == n/2,  # integral constraint
        ]
        
        # Add constraints from X = h*h^T equality
        for i in range(n):
            constraints.append(X[i,i] == h[i])
        
        # Add constraints from h_i * h_j <= min(h_i, h_j)
        for i in range(n):
            for j in range(i+1, n):
                constraints.append(X[i,j] <= h[i])
                constraints.append(X[i,j] <= h[j])
                constraints.append(X[i,j] >= 0)
        
        # Add correlation constraints
        for k in range(n):
            constraints.append(cp.trace(M_list[k] @ X) >= L)
            
        # Objective
        objective = cp.Maximize(L)
        
        # Solve SDP
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, max_iters=max_iters, eps=eps)
        
        # Extract solution
