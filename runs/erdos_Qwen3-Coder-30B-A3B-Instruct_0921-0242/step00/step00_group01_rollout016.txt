import numpy as np
import scipy.optimize as opt
from math import sqrt
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # Maximum value scaled by dx
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    """Return an initial construction for h"""
    n = 47
    h = np.zeros(n)
    # Simple construction: piecewise constant
    # Split [0,2] into 47 intervals
    # Use a step function that's mostly 0, with 1's in some places to sum to 1
    # The construction below is a variation of a known good initial solution
    # Basic idea: put some mass at 0.5, 1.0, 1.5, and 2.0 arithmetic points
    # But more carefully crafted to stay in [0,1] and integrate to 1
    
    # Start with a simple bump-like function
    start_point = 0
    end_point = 2
    dx = 2.0 / n
    # Standard bump function to integrate to 1
    for i in range(n):
        x = i * dx
        # Clip the function to [0,1], allowing some parts to be more active
        h[i] = max(0, min(1, 1.0 - abs(x - 1.0) / 0.5))
    # Renormalize so that sum * dx = 1
    integral = np.sum(h) * dx
    h = h / integral * 1.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    n_points = 47
    h = initial_h_values()
    dx = 2.0 / n_points
    
    # Track the best solution found
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    print(f"Initial C5 bound: {best_c5}")
    
    # Gradient-based optimization helper
    def objective(h_vec):
        h = h_vec.copy()
        # Re-normalize to constraint
        integral = np.sum(h) * dx
        h = h / integral * 1.0
        c5 = evaluate_erdos_solution(h)
        return c5

    def constraint_func(h_vec):
        h = h_vec.copy()
        integral = np.sum(h) * dx
        return integral - 1.0  # Should equal 0
    
    # Try several optimization approaches
    # Start with a few restarts
    for iter_num in range(50):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation
        h_pert = h.copy() + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        # Renormalize
        integral = np.sum(h_pert) * dx
        h_pert = h_pert / integral * 1.0
        c5_pert = evaluate_erdos_solution(h_pert)
        
        if c5_pert < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_pert
            
        # Simple gradient ascent: move in direction of largest decrease in overlap
        # This helps with escaping local optima
        if iter_num % 10 == 0 and iter_num > 0:
            # Try df optimization with simplex
            try:
                # Use L-BFGS-B
                bounds = [(0, 1)] * n_points
                res = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                                   options={'maxiter': 10, 'ftol': 1e-9})
                if res.success:
                    h_test = res.x
                    h_test = np.clip(h_test, 0, 1)
                    integral = np.sum(h_test) * dx
                    h_test = h_test / integral * 1.0
                    c5_test = evaluate_erdos_solution(h_test)
                    if c5_test < best_c5:
                        best_h = h_test.copy()
                        best_c5 = c5_test
                        h = h_test.copy()
            except:
                pass
        
        h = best_h.copy()  # Reset to best
        
    return (best_h.tolist(), best_c5, n_points)
