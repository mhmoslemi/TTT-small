import numpy as np
import time
from scipy import signal
from scipy.fft import fft, ifft
import math

# ---------------------
# Constants and helpers
# ---------------------

# Golden ratio for seeding
PHI = (1 + math.sqrt(5)) / 2

def autocorr(b):
    """Compute autocorrelation of binary sequence b using FFT."""
    B = fft(b)
    A = ifft(B * np.conj(B)).real
    return A

def min_autocorr(b):
    """Compute minimum non-zero autocorrelation of binary sequence."""
    A = autocorr(b)
    return np.min(A[1:])

def evaluate_erdos_solution(h):
    """Evaluate C5 with the exact method."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    C5 = np.max(corr) * dx
    return C5

def initial_h_values(n_points):
    """Initial construction."""
    h = np.array(initial_h_values)
    if len(h) != n_points:
        # Interpolate or pad to n_points
        old_x = np.linspace(0, 2, len(h), endpoint=False)
        new_x = np.linspace(0, 2, n_points, endpoint=False)
        h = np.interp(new_x, old_x, h)
    # Normalize so sum(h) * dx = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    return h

def get_binary_seed(n):
    """Create balanced binary seed using Singer difference set."""
    # Lengths of Singer difference sets: n = q^2 + q + 1 where q is a prime power
    # For n=183 (q=13), we'll embed it in n=200
    if n == 200:
        # Construct a Singer difference set sequence of length 183
        # This is manually chosen approximate construction for illustration
        # In real implementation would use better combinatorial construction
        b = np.zeros(n, dtype=int)
        # Create a pattern that approximates uniform autocorrelation
        # Taking first 91 ones, then 91 zeros to make it close to balanced
        b[:91] = 1
        # Randomize a bit to increase quality
        np.random.seed(42)
        indices = np.random.choice(n, 91, replace=False)
        b = np.zeros(n, dtype=int)
        b[indices] = 1
        # Ensure balance (exactly n//2 ones)
        if np.sum(b) != n//2:
            # Correct the imbalance
            if np.sum(b) > n//2:
                # Drop some ones
                drop_indices = np.where(b == 1)[0]
                drop_indices = drop_indices[:np.sum(b) - n//2]
                b[drop_indices] = 0
            else:
                # Add some ones
                add_indices = np.where(b == 0)[0]
                add_indices = add_indices[:n//2 - np.sum(b)]
                b[add_indices] = 1
        return b
    elif n % 2 == 0:
        # Quick approximation for other even lengths
        b = np.zeros(n, dtype=int)
        b[:n//2] = 1
        # Randomly shuffle
        np.random.shuffle(b)
        return b
    else:
        raise ValueError("n must be even for binary sequence")

def binary_to_h(b, n):
    """Convert binary vector to h values, adjusting for normalization."""
    dx = 2.0 / n
    h = b.astype(float)
    h = h * (n / 2) / np.sum(h)
    # enforce sum condition
    if abs(np.sum(h) * dx - 1.0) > 1e-9:
        # Scale adjustment
        scale = 1.0 / (np.sum(h) * dx)
        h = h * scale
    return h

def compute_autocorrelations(b):
    """Returns full autocorrelations as numpy array."""
    A = autocorr(b)
    A_real = A.real
    return A_real

def swap_improvement(b, i, j, k):
    """For given b with b[i]=1,b[j]=0, and i,k (i+k mod n)
    compute change in autocorrelation at lag k.
    Return contribution to change in A_k as - b[i+k] + b[j+k]
    """
    n = len(b)
    k = int(k) % n
    contribution = -b[(i + k) % n] + b[(j + k) % n]
    return contribution

def hill_climb(b, max_swaps=20000, temp_init=1.0):
    """Perform hill climbing with simulated annealing using binary swaps."""
    n = len(b)
    # Build lookup: B_shift[i,k] = b[(i + k) % n]
    B_shift = np.zeros((n, n), dtype=int)
    for k in range(n):
        B_shift[:, k] = np.roll(b, k)
    
    A = compute_autocorrelations(b)
    m = np.min(A[1:])  # Min non-zero lag
    k_star = np.argmin(A[1:]) + 1
    best_m = m
    best_b = b.copy()

    swaps = 0
    temp = temp_init
    no_improve = 0
    max_no_improve = 5000
    start = time.time()
    while swaps < max_swaps and time.time() - start < 2.5:
        # Find indices where we can swap (1 -> 0)
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute improvement in A_{k_star}
        delta = swap_improvement(b, i, j, k_star)
        
        # Accept if it improves, otherwise perhaps accept with prob
        if delta > 0:
            # Accept immediately
            b[i] = 0
            b[j] = 1
            # Update A for all k that are affected by this
            for k in range(n):
                A[k] += -B_shift[i, k] + B_shift[j, k]
            A[k_star] += delta
            m = np.min(A[1:])
            if m > best_m:
                best_m = m
                best_b = b.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Perhaps accept with probability
            p_accept = np.exp(delta / temp)
            if np.random.rand() < p_accept:
                b[i] = 0
                b[j] = 1
                for k in range(n):
                    A[k] += -B_shift[i, k] + B_shift[j, k]
                A[k_star] += delta  # important only if k* changes...
                m = np.min(A[1:])
                if m > best_m:
                    best_m = m
                    best_b = b.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
                
        swaps += 1
        temp *= 0.995
        if no_improve > max_no_improve:
            break
    return best_b, best_m


def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_total = time.time()
    
    # First, try the provided initial h_values if they are available and valid size
    try:
        n_points = len(initial_h_values)
        if n_points <= 1000 and n_points >= 40:
            h_start = initial_h_values(n_points)
            c5_initial = evaluate_erdos_solution(h_start)
        else:
            n_points = 200  # default if not properly sized
            h_start = initial_h_values(n_points)
            c5_initial = evaluate_erdos_solution(h_start)
    except Exception:
        n_points = 200
        h_start = initial_h_values(n_points)
        c5_initial = evaluate_erdos_solution(h_start)

    best_c5 = c5_initial
    best_h = h_start.copy()
    best_n = n_points

    # Find the best binary seed
    seeds = []
    for r in range(10):
        np.random.seed(seed + r)
        if r < 2:
            # Legendre-like pattern
            try:
                b = get_binary_seed(200)
            except:
                b = np.random.randint(0, 2, 200)
        else:
            # Pure random
            b = np.random.randint(0, 2, 200)
            # Adjust to get exactly 100 ones
            current_sum = np.sum(b)
            if current_sum > 100:
                zero_indices = np.where(b == 0)[0]
                indices_to_change = np.random.choice(zero_indices, int(current_sum - 100), replace=False)
                b[indices_to_change] = 1
            elif current_sum < 100:
                one_indices = np.where(b == 1)[0]
                indices_to_change = np.random.choice(one_indices, int(100 - current_sum), replace=False)
                b[indices_to_change] = 0
        m = min_autocorr(b)
        if m > 0.9 * 50:
            seeds.append((b, m))
    # Use the best among these seeds
    if seeds:
        best_seed, _ = max(seeds, key=lambda x: x[1])
    else:
        best_seed = np.zeros(200, dtype=int)
        best_seed[:100] = 1
        np.random.shuffle(best_seed)

    # Do local optimization
    if time.time() - start_total > 100:
        return best_h, best_c5, best_n

    # Binary to dense
    h = binary_to_h(best_seed, 200)
    # Do hill climbing
    h_opt, m_opt = hill_climb(best_seed.copy(), max_swaps=15000, temp_init=5.0)
    # Convert back to h
    h_new = binary_to_h(h_opt, 200)
    c5_new = evaluate_erdos_solution(h_new)
    
    if c5_new < best_c5:
        best_c5 = c5_new
        best_h = h_new.copy()
        best_n = 200

    # Then inflate and refine
    if time.time() - start_total > 400:
        return best_h, best_c5, best_n

    # Generate suitable larger grid (e.g. 300)
    n_large = 300
    h_large = np.zeros(n_large)
    h_large[::2] = best_h
    h_large[1::2] = best_h
    # Normalize
    dx_large = 2.0 / n_large
    h_large = h_large * (n_large / 2) / np.sum(h_large)
    # slightly refine
    c5_large = evaluate_erdos_solution(h_large)
    if c5_large < best_c5:
        best_c5 = c5_large
        best_h = h_large.copy()
        best_n = n_large

    # Final validation
    dx = 2.0 / best_n
    actual_sum = np.sum(best_h) * dx
    if abs(actual_sum - 1.0) > 1e-9:
        best_h = best_h / (actual_sum * dx)  # ensure integral = 1
    best_c5 = evaluate_erdos_solution(best_h)

    if time.time() - start_total > 580:
        return best_h, best_c5, best_n

    return best_h, best_c5, best_n
