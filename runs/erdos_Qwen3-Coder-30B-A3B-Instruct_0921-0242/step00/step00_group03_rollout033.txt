import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar
from math import sqrt

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    # Convert to numpy array
    h = np.array(h)
    # Compute all correlations with 1-h using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Extract the relevant correlations (shifts from -(n-1) to n-1)
    n = len(h)
    # Pick out the correlations that correspond to valid shifts from 0 to 2
    start_idx = n - 1
    valid_corr = corr[start_idx:start_idx + 2 * n - 1]
    dx = 2.0 / n
    return max(valid_corr) * dx

def initial_h_values():
    """Return initial step function as a list of values."""
    # Using a simple constructed one from the original problem
    n = 70
    h = np.zeros(n)
    # Make it take values 0 or 1
    # A typical good start: a step function with a large block that's 1
    h[0:n//2] = 1.0
    return h.tolist()

def solve_sdp(n, time_limit=200):
    """Solve the SDP relaxation of the problem."""
    # Check if we can use SDP
    if n > 300:
        return None, np.inf

    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Objective
    objective = cp.Minimize(t)

    # Constraints
    constraints = []

    # Define shift matrices P_s
    P = {}
    for s in range(-n + 1, n):
        P_s = np.zeros((n, n))
        if s >= 0:
            P_s[:-s, s:] = np.eye(n - s)
        else:
            P_s[-s:, :s] = np.eye(n + s)
        P[s] = P_s

    # Box constraints
    constraints += [h >= 0]
    constraints += [h <= 1]

    # Equality constraint
    constraints += [cp.sum(h) == n / 2.0]

    # PSD constraint for X
    constraints += [X >> 0]

    # Diagonal entries of X should equal h
    for i in range(n):
        constraints += [X[i, i] == h[i]]

    # Shift constraints
    dx = 2.0 / n
    constant_term = dx * (n / 2.0)
    for s in range(-n + 1, n):
        constraints += [t >= constant_term - dx * cp.trace(P[s] @ X)]

    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, time_limit=time_limit)
        return h.value, prob.value
    except Exception:
        return None, np.inf

def round_solution(h_sol, n, num_rounds=30):
    """Randomized rounding of the solution."""
    best_h = None
    best_c5 = np.inf
    for round in range(num_rounds):
        # Generate random Gaussian vector
        g = np.random.randn(n)
        # Compute matrix square root of X ≈ h_sol * h_sol^T
        # Use that h_sol is actually the solution used for
        # clipping and scaling to satisfy sum constraint and
        # to get it in [0,1]
        # Here, we apply a different approach:
        # We use the fact that h_sol is already near feasible
        h_rounded = h_sol + 0.01 * np.random.randn(n)
        h_rounded = np.clip(h_rounded, 0, 1)
        total = np.sum(h_rounded)
        scale = (n / 2.0) / total
        h_rounded = h_rounded * scale
        h_rounded = np.clip(h_rounded, 0, 1)
        total = np.sum(h_rounded)
        deficit = (n / 2.0) - total
        if deficit > 0:
            # Distribute deficit to positions that are not 1 or 0
            candidates = np.where((h_rounded > 0) & (h_rounded < 1))[0]
            if len(candidates) == 0:
                break
            increment = deficit / len(candidates)
            for i in candidates:
                h_rounded[i] += increment
        h_rounded = np.clip(h_rounded, 0, 1)
        c5_rounded = evaluate_c5(h_rounded)
        if c5_rounded < best_c5:
            best_c5 = c5_rounded
            best_h = h_rounded.copy()
    return best_h, best_c5

def local_improve(h, max_iter=5000):
    """Improve a solution using local coordinate updates."""
    n = len(h)
    dx = 2.0 / n
    best_h = np.array(h).copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()
    last_improvement = 0

    # Try to make small changes to h at each index
    for _ in range(max_iter):
        if time.time() - start_time > 500.0:
            break
        i = np.random.randint(0, n)
        if best_h[i] == 0 and i == n - 1:
            continue
        if best_h[i] == 1:
            continue
        candidate_h = best_h.copy()
        # Try a small positive increment
        eps = min(0.001, 1 - best_h[i])
        if eps <= 0:
            continue
        candidate_h[i] += eps
        # Now compensate at another index
        j = np.random.randint(0, n)
        if i == j:
            j = (j + 1) % n
        delta = min(eps, best_h[j])  # can't reduce below 0
        if delta <= 0:
            continue
        candidate_h[j] -= delta
        # Check if we stay in box
        if np.any(candidate_h < 0) or np.any(candidate_h > 1):
            continue
        # Make sure sum is same
        new_sum = np.sum(candidate_h)
        correction = (n / 2.0) - new_sum
        # Use the index other than i and j to adjust
        if i != (j + 1) % n:
            idx = (j + 1) % n
        else:
            idx = (i + 1) % n
        corr = min(correction, 1 - candidate_h[idx])
        if corr >= 0:
            candidate_h[idx] += corr
        else:
            corr = max(correction, -candidate_h[idx])
            candidate_h[idx] += corr
        # Recalculate and update
        c5_candidate = evaluate_c5(candidate_h)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = candidate_h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            if last_improvement > 200:
                break
        # Ensure the sum constraint
        best_h = best_h * (n / 2.0) / np.sum(best_h)
        best_h = np.clip(best_h, 0, 1)

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    n_points = 120  # A good balance
    dx = 2.0 / n_points

    # Use SDP to lower bound
    h_sol, lower_bound = solve_sdp(n_points, time_limit=150)
    if h_sol is None:
        # Fallback and try smaller size
        n_points = 90
        dx = 2.0 / n_points
        h_sol, lower_bound = solve_sdp(n_points, time_limit=150)

    if h_sol is not None:
        # Round and refine
        h_rounded, c5_rounded = round_solution(h_sol, n_points, num_rounds=50)
        if h_rounded is not None:
            h_improved, c5_improved = local_improve(h_rounded)
            if c5_improved < best_c5:
                best_h = h_improved
                best_c5 = c5_improved

    # Try initial construction
    h_init = initial_h_values()
    c5_init = evaluate_c5(h_init)
    if c5_init < best_c5:
        best_h = np.array(h_init)
        best_c5 = c5_init

    # Try some permutations of the baseline initially
    if time.time() - start_time < 300:
        for _ in range(10):
            if time.time() - start_time > budget_s - 60:
                break
            h_perm = np.array(h_init)
            # Perturb by moving some element's value
            i = np.random.randint(0, n_points)
            direction = 1 if np.random.rand() > 0.5 else -1
            step = 0.01
            h_perm[i] = np.clip(h_perm[i] + direction * step, 0, 1)
            # Reproject to correct sum
            total = np.sum(h_perm)
            h_perm = h_perm * (n_points / 2.0) / total
            h_perm = np.clip(h_perm, 0, 1)
            c5_perm = evaluate_c5(h_perm)
            if c5_perm < best_c5:
                best_c5 = c5_perm
                best_h = h_perm.copy()

    if best_h is None:
        best_h = np.array(initial_h_values())
        best_c5 = evaluate_c5(best_h)

    # Final refinement
    best_h, best_c5 = local_improve(best_h, max_iter=1000)

    best_h = np.clip(best_h, 0, 1)
    # Adjust to match sum exactly
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Return final result
    return (best_h.tolist(), best_c5, len(best_h))
