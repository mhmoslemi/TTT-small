import numpy as np

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    # Compute the full correlation with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum correlation val represents C5
    return np.max(corr) * (2.0 / n)

def solve_sdp_relaxation(h0, n):
    """Solve the SDP relaxation of the problem."""
    # Use n points, Δx = 2 / n
    dx = 2.0 / n
    
    # Define matrix variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.diag(X) == h,
        X >> 0,
        0 <= h,
        h <= 1,
        cp.sum(h) == n / 2,
        t >= 0  # Ensure t is non-negative
    ]
    
    # Add constraints for each shift
    for k in range(-n+1, n):
        # Create the shift matrix P_s (shift by k positions)
        P = np.zeros((n, n))
        if k >= 0:
            P[k:, :n-k] = np.eye(n-k)
        else:
            P[:n+k, -k:] = np.eye(n+k)
        # Add constraint for this shift
        p = cp.trace(P @ X)
        constraints.append(t >= dx * (0.5 - p))
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=20000)
    except:
        return None, None, None
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h.value, X.value, prob.value
    return None, None, None

                # Distribute the difference evenly
                indices = np.where((y > 0) & (y < 1))[0]

                    # Choose indices to update
                    update_indices = np.random.choice(indices, 
                        min(len(indices), int(abs(diff) * 10)), replace=False)

                        y[update_indices] = np.minimum(y[update_indices] + 0.01, 1)
                    else:
                        y[update_indices] = np.maximum(y[update_indices] - 0.01, 0)
            # Ensure sum is exactly n/2
            scale_factor = (n / 2) / np.sum(y)
            y = y * scale_factor
            # Ensure correct bounds
            y = np.clip(y, 0, 1)
            # Eval C5
            c5 = evaluate_c5(y)
            if c5 < best_c5:
                best_c5 = c5
                best_h = y.copy()

    except:
        return None, np.inf
