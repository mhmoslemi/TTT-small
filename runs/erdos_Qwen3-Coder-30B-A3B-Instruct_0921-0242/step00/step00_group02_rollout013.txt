import time
import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Parameters
    max_m_values = [4, 5, 6, 7, 8]
    n_values = [30, 36, 42, 48, 60]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for m in max_m_values:
        if time.time() - t0 > budget_s - 5:
            break
            
        # Interval set construction
        interval_length = 1.0 / m
        # Phase A: DIRECT-like optimization
        # Use deterministic search with box constraints
        a_lb = np.zeros(m)
        a_ub = np.ones(m) * (2 - interval_length)
        # Add ordering constraints:
        # a[0] >= 0, a[i+1] >= a[i] + interval_length
        
        def overlap_function(a):
            """Compute max overlap for given interval placement"""
            # Compute all interval pairs and their overlaps
            total_o = 0.0
            # Generate breakpoints
            breakpoints = []
            for i in range(m):
                for j in range(m):
                    a_i, a_j = a[i], a[j]
                    # Tent function breakpoints: k = a_i - a_j ± interval_length
                    k1 = (a_i - a_j) + interval_length
                    k2 = (a_i - a_j) - interval_length
                    if 0 <= k1 <= 2:
                        breakpoints.append(k1)
                    if 0 <= k2 <= 2:
                        breakpoints.append(k2)
            # Also include 0,2
            breakpoints.extend([0, 2])
            breakpoints = sorted(set(breakpoints))
            
            max_o = 0.0
            for k in breakpoints:
                o_k = 0.0
                for i in range(m):
                    for j in range(m):
                        a_i, a_j = a[i], a[j]
                        o_ij = max(0, interval_length - abs(a_i - a_j - k))
                        o_k += o_ij
                max_o = max(max_o, o_k)
            return max_o
        
        def project_to_feasible(a):
            """Project point to feasible region"""
            # Compute gaps
            gaps = []
            for i in range(m-1):
                gap = a[i+1] - (a[i] + interval_length)
                gaps.append(max(gap, 0))
            
            # If any gap is negative, redistribute it
            total_deficit = sum(gap for gap in gaps if gap < 0)
            if total_deficit < 0:
                # Add deficit to first gap
                gaps[0] += -total_deficit
                
            # Now ramp up a values from left to right
            a_new = np.copy(a)
            a_new[0] = max(0, a_new[0])
            for i in range(1, m):
                min_pos = a_new[i-1] + interval_length
                a_new[i] = max(a_new[i], min_pos)
            return a_new
            
        # Start with the initial using smoother method: bicgstab-like or similar
        initial_a = np.linspace(0, 2 - m * interval_length, m)
        initial_a = project_to_feasible(initial_a)
        
        # Direct search over feasible space
        log_best = None
        best_a = initial_a.copy()
        best_val = overlap_function(best_a)
        for _ in range(1000):
            if time.time() - t0 > budget_s - 5:
                break
            # Random sample in feasible region
            # Down-project if needed
            trial_a = np.random.uniform(0, 2 - interval_length, m)
            trial_a = project_to_feasible(trial_a)
            
            val = overlap_function(trial_a)
            if val < best_val:
                best_val = val
                best_a = trial_a.copy()
                log_best = (best_a.copy(), best_val)
        # Phase B: Refinement using DE
        bounds = [(0, 2 - interval_length) for _ in range(m)]
        try:
            result = differential_evolution(
                lambda a: overlap_function(a),
                bounds,
                maxiter=1000,
                popsize=10 * m,
                mutation=(0.5, 1),
                recombination=0.7,
                seed=42,
                atol=1e-6,
                x0=best_a
            )
            best_val = result.fun
            best_a = result.x
        except Exception:
            pass
            
        # Discretize
        for n in n_values:
            if time.time() - t0 > budget_s - 5:
                break
                
            dx = 2.0 / n
            h = np.zeros(n)
            for i in range(n):
                x = i * dx
                for j in range(m):
                    if x >= best_a[j] and x <= best_a[j] + interval_length:
                        h[i] = 1
                        break
                        
            # Adjust if mass is wrong
            actual_sum = np.sum(h)
            if abs(actual_sum - n/2) > 1e-10:
                error = n/2 - actual_sum
                if error > 0:  # Need to flip some 0's to 1's
                    # Find positions closest to interval boundaries
                    dists = []
                    for i in range(n):
                        x = i * dx
                        min_dist = float("inf")
                        for j in range(m):
                            dist1 = abs(x - (best_a[j]))
                            dist2 = abs(x - (best_a[j] + interval_length))
                            min_dist = min(min_dist, dist1, dist2)
                        dists.append((min_dist, i))
                    
                    dists.sort()
                    to_flip = min(int(error), len(dists))
                    for i in range(to_flip):
                        pos = dists[i][1]
                        h[pos] = 1
                elif error < 0:
                    # Need flipping some 1's to 0's
                    dists = []
                    for i in range(n):
                        x = i * dx
                        min_dist = float("inf")
                        for j in range(m):
                            dist1 = abs(x - (best_a[j]))
                            dist2 = abs(x - (best_a[j] + interval_length))
                            min_dist = min(min_dist, dist1, dist2)
                        dists.append((min_dist, i))
                    
                    dists.sort()
                    to_flip = min(int(-error), len(dists))
                    for i in range(to_flip):
                        pos = dists[i][1]
                        h[pos] = 0
            
            # Postprocessing: swap refinement
            
            # Compute overlap integral
            # Convolve h with 1-h to get correlation
            h_rev = 1 - h
            corr = correlate(h, h_rev, mode='full')
            # Only valid k values
            max_corr = np.max(corr[len(h)-1:])
            c5 = 1 - max_corr / n
            
            # Only consider if we have a better solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n
                
        if log_best:
            best_a, best_val = log_best
            # If still in time...
            for n in n_values:
                if time.time() - t0 > budget_s - 5:
                    break
                dx = 2.0 / n
                h = np.zeros(n)
                for i in range(n):
                    x = i * dx
                    for j in range(m):
                        if x >= best_a[j] and x <= best_a[j] + interval_length:
                            h[i] = 1
                            break
                            
                actual_sum = np.sum(h)
                if abs(actual_sum - n/2) > 1e-10:
                    error = n/2 - actual_sum
                    if error > 0:  # Need to flip some 0's to 1's
                        dists = []
                        for i in range(n):
                            x = i * dx
                            min_dist = float("inf")
                            for j in range(m):
                                dist1 = abs(x - (best_a[j]))
                                dist2 = abs(x - (best_a[j] + interval_length))
                                min_dist = min(min_dist, dist1, dist2)
                            dists.append((min_dist, i))
                        dists.sort()
                        to_flip = min(int(error), len(dists))
                        for i in range(to_flip):
                            pos = dists[i][1]
                            h[pos] = 1
                    elif error < 0:
                        dists = []
                        for i in range(n):
                            x = i * dx
                            min_dist = float("inf")
                            for j in range(m):
                                dist1 = abs(x - (best_a[j]))
                                dist2 = abs(x - (best_a[j] + interval_length))
                                min_dist = min(min_dist, dist1, dist2)
                            dists.append((min_dist, i))
                        dists.sort()
                        to_flip = min(int(-error), len(dists))
                        for i in range(to_flip):
                            pos = dists[i][1]
                            h[pos] = 0
                            
                # Diffusion-based LB approximation in the continuous regime
                # Compute correlation
                h_rev = 1 - h
                corr = correlate(h, h_rev, mode='full')
                max_corr = np.max(corr[len(h)-1:])
                c5 = 1 - max_corr / n
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n

    # Return final result
    return (best_h, best_c5, best_n)
