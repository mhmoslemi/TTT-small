import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def evaluate_erdos_solution(h):
    """Evaluate C_5 for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # Take only the non-zero shifts part
    shift_idx = range(n - 1, 2 * n - 1)
    overlaps = corr[shift_idx]
    # Normalize and return maximum overlap
    return np.max(overlaps) * dx


def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Try MILP with n=100
    n = 100
    max_seconds = budget_s // 2
    best_h = None
    best_c5 = float('inf')
    h_bin = None
    
    try:
        # Build MILP model
        h = cp.Variable(n, boolean=True)
        t = cp.Variable(integer=True, lb=0)
        
        # Constraints
        constraints = [cp.sum(h) == n // 2, t <= 0]
        
        # Allocate y variables
        y = {}
        A = []
        
        for s in range(1, n):
            # For each shift, create binary variables for the product
            y_s = cp.Variable((n - s), boolean=True)
            # Add constraints for y_s
            for i in range(n - s):
                y[(i, s)] = y_s[i]
                constraints.extend([
                    y_s[i] <= h[i],
                    y_s[i] <= h[i + s],
                    y_s[i] >= h[i] + h[i + s] - 1
                ])
            # Add overlap constraint
            A_s = cp.sum(y_s)
            A.append(A_s)
            constraints.append(t <= A_s)
            
        # Set objective to maximize t
        problem = cp.Problem(cp.Maximize(t), constraints)
        
        # Solve the MILP
        problem.solve(solver=cp.CBC, max_seconds=max_seconds)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_bin = np.round(h.value).astype(int)
            # Compute the actual t from model
            t_val = int(t.value)
            # Convert to actual C5
            c5 = 1 - 2 * t_val / n
            # Evaluate with full integration
            h_full = h_bin.astype(float)
            c5_actual = evaluate_erdos_solution(h_full)
            if c5_actual < best_c5:
                best_h = h_full.copy()
                best_c5 = c5_actual
        else:
            # If solver didn't converge, try a feasible solution
            h_feasible = h_bin if h_bin is not None else np.random.choice(
                [0, 1], size=n, p=[0.5, 0.5]
            )
            h_feasible = np.array(h_feasible, dtype=int)
            # Ensure sum is correct
            ones = np.where(h_feasible == 1)[0]
            zeros = np.where(h_feasible == 0)[0]
            if len(ones) > n // 2:
                # Change some ones to zeros
                idxs = np.random.choice(ones, len(ones) - n//2, replace=False)
                h_feasible[idxs] = 0
            else:
                # Change some zeros to ones
                idxs = np.random.choice(zeros, n//2 - len(ones), replace=False)
                h_feasible[idxs] = 1
            c5_actual = evaluate_erdos_solution(h_feasible)
            if c5_actual < best_c5:
                best_h = h_feasible.copy()
                best_c5 = c5_actual
                
    except Exception:
        pass
    
    # Phase 2: Try n=120 if enough time
    if best_c5 >= 0.3807 and (time.time() - start_time) < (budget_s - 15):
        try:
            n = 120
            max_seconds = (budget_s - (time.time() - start_time)) // 2
            h = cp.Variable(n, boolean=True)
            t = cp.Variable(integer=True, lb=0)
            
            # Constraints
            constraints = [cp.sum(h) == n // 2, t <= 0]
            
            # Allocate y variables
            y = {}
            A = []
            
            for s in range(1, n):
                # For each shift, create binary variables for the product
                y_s = cp.Variable((n - s), boolean=True)
                # Add constraints for y_s
                for i in range(n - s):
                    y[(i, s)] = y_s[i]
                    constraints.extend([
                        y_s[i] <= h[i],
                        y_s[i] <= h[i + s],
                        y_s[i] >= h[i] + h[i + s] - 1
                    ])
                # Add overlap constraint
                A_s = cp.sum(y_s)
                A.append(A_s)
                constraints.append(t <= A_s)
                
            # Set objective to maximize t
            problem = cp.Problem(cp.Maximize(t), constraints)
            
            # Solve the MILP
            problem.solve(solver=cp.CBC, max_seconds=max_seconds)
            
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_bin = np.round(h.value).astype(int)
                # Compute the actual t from model
                t_val = int(t.value)
                # Convert to actual C5
                c5 = 1 - 2 * t_val / n
                # Evaluate with full integration
                h_full = h_bin.astype(float)
                c5_actual = evaluate_erdos_solution(h_full)
                if c5_actual < best_c5:
                    best_h = h_full.copy()
                    best_c5 = c5_actual
            else:
                # If solver didn't converge, try a feasible solution
                h_feasible = h_bin if h_bin is not None else np.random.choice(
                    [0, 1], size=n, p=[0.5, 0.5]
                )
                h_feasible = np.array(h_feasible, dtype=int)
                # Ensure sum is correct
                ones = np.where(h_feasible == 1)[0]
                zeros = np.where(h_feasible == 0)[0]
                if len(ones) > n // 2:
                    # Change some ones to zeros
                    idxs = np.random.choice(ones, len(ones) - n//2, replace=False)
                    h_feasible[idxs] = 0
                else:
                    # Change some zeros to ones
                    idxs = np.random.choice(zeros, n//2 - len(ones), replace=False)
                    h_feasible[idxs] = 1
                c5_actual = evaluate_erdos_solution(h_feasible)
                if c5_actual < best_c5:
                    best_h = h_feasible.copy()
                    best_c5 = c5_actual
                    
        except Exception:
            pass
    
    # Phase 3: Local search improvement
    if best_h is not None and (time.time() - start_time) < (budget_s - 100):
        def compute_overlaps(h, n_points):
            # Compute correlation with 1-h
            corr = np.correlate(h, 1 - h, mode="full")
            # Take the overlapping part
            start = n_points - 1
            end = 2 * n_points - 1
            overlaps = corr[start:end] 
            return overlaps
        
        def improvement_search(h):
            n_points = len(h)
            dx = 2.0 / n_points
            # Compute initial overlaps
            overlaps = compute_overlaps(h, n_points)
            min_o = np.min(overlaps)
            max_o = np.max(overlaps)
            target = min_o
            improvements = 0
            
            # Collect pairs that can potentially be swapped
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                return h, target, improvements
            
            # Try swaps
            num_swaps = 0
            max_swaps = 20000

            while num_swaps < max_swaps:
                # randomly select a pair to swap
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i1 = np.random.choice(ones)
                i0 = np.random.choice(zeros)
                
                # Perform swap
                h[i1], h[i0] = h[i0], h[i1]
                old_min = min_o
                # Update overlaps
                overlaps = compute_overlaps(h, n_points)
                min_o = np.min(overlaps)
                max_o = np.max(overlaps)
                
                if min_o > old_min:
                    improvements += 1
                    target = min_o
                else:
                    # undo swap if not beneficial
                    h[i1], h[i0] = h[i0], h[i1]
                    min_o = old_min
                
                num_swaps += 1
                if time.time() - start_time > budget_s - 5:
                    break
                    
            # Final update of parameters
            overlaps = compute_overlaps(h, n_points)
            min_o = np.min(overlaps)
            return h, min_o, improvements
        
        candidate_h, target, improvements = improvement_search(best_h)
        if improvements > 0 or target < best_c5:
            # Evaluate final result
            final_eval = evaluate_erdos_solution(candidate_h)
            if final_eval < best_c5:
                best_c5 = final_eval
                best_h = candidate_h.copy()
                
    # Phase 4: Continuous polishing if needed
    if best_c5 >= 0.3807 and (time.time() - start_time) < (budget_s - 10):
        def objective(h):
            dx = 2.0 / len(h)
            corr = np.correlate(h, 1-h, mode="full")
            start_idx = len(h) - 1
            end_idx = 2*len(h) - 1
            overlaps = corr[start_idx:end_idx]
            # We minimize the max overlap (opp of maximizing the min)
            return -np.min(overlaps) * dx
        
        def constraint_sum(h):
            return np.sum(h) - len(h) / 2.0  # sum should be exactly n/2
            
        # Format constraints for scipy.optimize
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Initialize with better cold start than random
        initial_h = best_h.reshape(-1)
        bounds = [(0.0, 1.0)] * len(initial_h)
        
        try:
            res = minimize(
                objective,
                initial_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-10}
            )
            
            if res.success:
                # evaluate on new point
                polished_h = res.x
                c5_polished = evaluate_erdos_solution(polished_h)
                if c5_polished < best_c5:
                    best_c5 = c5_polished
                    best_h = polished_h
        except Exception:
            pass
            
    # Phase 5: Upsample if we're approaching target
    final_h = np.array(best_h)
    n_points = len(final_h)
    dx = 2.0 / n_points
    # Since asked to use at most 1000 samples
    # we only do it if final C5 is still above a certain threshold
    if n_points < 800 and best_c5 > 0.3700 and (time.time() - start_time) < (budget_s - 10):
        # Now upsample for better evaluation
        r = 4  # sampling upsample ratio
        if n_points * r <= 800:
            upsampled_h = np.repeat(final_h, r)
            upsampled_c5 = evaluate_erdos_solution(upsampled_h)
            # Just evaluate and keep if within good range
            if upsampled_c5 < best_c5:
                final_h = upsampled_h
                n_points = len(final_h)
            
    return (final_h, best_c5, n_points)
