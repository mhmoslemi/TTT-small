from scipy import sparse
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):

            diags = [1.0]
            M = sparse.diags(diags, offsets, shape=(n, n), format='coo')
            M_list.append(M)
        
        # SDP optimisation
        X = cp.Variable((n, n), symmetric=True)
        h = cp.Variable(n)
        L = cp.Variable()

        constraints.append(X >> 0)  # Positive semidefinite

            constraints.append(X[i,i] == h[i])
            constraints.append(0 <= h[i])
            constraints.append(h[i] <= 1)
