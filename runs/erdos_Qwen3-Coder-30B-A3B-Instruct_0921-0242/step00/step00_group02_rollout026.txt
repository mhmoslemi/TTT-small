import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up resolution levels
    resolutions = [30, 38, 42]
    time_allocations = [80, 120, 150]
    total_time = sum(time_allocations)
    
    best_C5 = float('inf')
    best_h = None
    best_n = 0
    
    stage_times = {}
    for i, (n, alloc_time) in enumerate(zip(resolutions, time_allocations)):
        if time.time() - start_time + alloc_time > budget_s - 5:
            break
            
        stage_start = time.time()
        try:
            # 1. SDP relaxation
            h_opt, X_opt, s_opt = solve_sdp(n)
            stage_times[f"SDP_{n}"] = time.time() - stage_start
            
            # 2. Random hyperplane rounding
            best_binary = None
            best_score = float('inf')
            for _ in range(300):
                binary = round_hyperplane(h_opt, X_opt, n)
                if binary is None: continue
                score = evaluate_C5(binary, n)
                if score < best_score:
                    best_score = score
                    best_binary = binary
            
            # 3. Local polishing (swap heuristic)
            if best_binary is not None:
                polished = swap_polish(best_binary, n)
                polished_score = evaluate_C5(polished, n)
                if polished_score < best_score:
                    best_score = polished_score
                    best_binary = polished
                    
            # 4. Update best solution
            if best_score < best_C5:
                best_C5 = best_score
                best_h = best_binary
                best_n = n
                
            if time.time() - start_time > budget_s - 10:
                break
                
            stage_times[f"rounding_{n}"] = time.time() - stage_start
            
        except Exception as e:
            continue
            
    return (best_h, best_C5, best_n)


def solve_sdp(n):
    # Matrix dimensions
    dx = 2.0 / n
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    
    # Objective: maximize s
    s = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) == h,  # Diagonal elements equal to h_i
        0 <= h,  # Box constraint
        h <= 1,
        cp.sum(h) == n / 2  # Mass constraint
    ]
    
    # Add correlation constraints (positive shift)
    for k in range(1, n):
        # Phi_k(X) = sum_{i=0}^{n-1-k} X_{i, i+k}
        constraint_expr = sum(X[i, i+k] for i in range(n-k))
        constraints.append(constraint_expr >= s)
    
    # Solve SDP
    prob = cp.Problem(cp.Maximize(s), constraints)
    prob.solve(solver=cp.SCS, verbose=False, eps=1e-4, max_iters=1000)
    
    if prob.status == cp.OPTIMAL:
        return h.value, X.value, s.value
    else:
        return None, None, float('inf')


def round_hyperplane(h_opt, X_opt, n):
    try:
        # Eigenvalue decomposition to get factorization
        eigenvals, eigenvects = eigh(X_opt)
        # Filter out small negative eigenvalues
        eigenvals = np.maximum(eigenvals, 0)
        rank = sum(1 for ev in eigenvals if ev > 1e-10)
        if rank <= 0:
            raise Exception("Rank is 0")
            
        # Do SVD on square root of the matrix
        U = eigenvects[:, :rank]  # keep only rank columns
        D = np.diag(np.sqrt(eigenvals[:rank]))
        V = U @ D  # V @ V.T = X_opt
        
        # Sample a random Gaussian vector
        g = np.random.randn(rank)
        z = V @ g
        
        # Threshold to get binary vector
        threshold = np.percentile(z, 100 * (1 - n/2 / n))
        b = np.zeros(n)
        for i in range(n):
            b[i] = 1 if z[i] >= threshold else 0
            
        # If not exactly the right sum, adjust
        current_sum = np.sum(b)
        needed = n // 2
        if current_sum != needed:
            # Find indices where to make changes
            # Convert z back to indices排序
            indices = np.argsort(-z)  # descending order
            if current_sum < needed:
                # need to flip 0s to 1s
                flips = needed - current_sum
                for i in range(flips):
                    if b[indices[i]] == 0:
                        b[indices[i]] = 1
            else:
                # need to flip 1s to 0s
                flips = current_sum - needed
                for i in range(flips):
                    if b[indices[i]] == 1:
                        b[indices[i]] = 0
                        
        # Double check
        if abs(np.sum(b) - n/2) > 1e-9:
            # Got to be really careful; just average positions
            sorted_indices = np.argsort(z)
            change_count = int(n/2 - np.sum(b))
            indices2 = sorted_indices[:abs(change_count)] if change_count < 0 else sorted_indices[abs(change_count):]
            if change_count < 0:
                for i in indices2:
                    b[i] = 1
            else:
                for i in indices2:
                    b[i] = 0
        
        # Final check
        if abs(np.sum(b) - n/2) > 1e-9:
            return None
            
        return b
    except Exception as e:
        return None


def evaluate_C5(binary, n):
    dx = 2.0 / n
    # Compute overlap integral with all shifts
    ones = 1 - binary  # h(x) and (1-h(x+k)) 
    corr = np.correlate(binary, ones, mode='full')  # This will be (2n-1) elements
    max_corr = np.max(corr) * dx  # max is achieved at a certain mode
    return max_corr


def swap_polish(binary, n):
    dx = 2.0 / n
    # Compute correlations for all shifts
    corr = np.correlate(binary, 1-binary, mode='full')
    # Correlation for shift k is corr[n-1+k], where k from -n+1 to n-1
    # Correspondence between shifted index and shift k
    # Shift k means we stride by k: M[k] = sum_i b_i * b_{i+k}
    # So for mode='full', index j corresponds to shift k = j - (n-1)
    # Which is equivalent to push all negative shifts to positive:
    max_len = 2 * n - 1
    correlations = np.zeros(n-1)
    for k in range(1, n):  # shift k from 1 to n-1 (1 based indexing)
        correlations[k-1] = corr[n-1+k]  # or padd [0(n-1), ..., 0, 1, 2, ..., 0(n-1)]
    
    # Find worst shift
    worst_shift_idx = np.argmin(correlations)
    worst_shift = worst_shift_idx + 1
    
    # Examine swaps
    best_swap_gain = 0
    best_i = -1
    best_j = -1
    
    # Limit candidates 
    candidate_pairs = []
    for i in range(n):
        if binary[i] == 1:
            for j in range(n):
                if binary[j] == 0:
                    candidate_pairs.append((i, j))
    
    # Limit to a reasonable amount if needed
    if len(candidate_pairs) > 2000:
        random.shuffle(candidate_pairs)
        candidate_pairs = candidate_pairs[:2000]
    
    for i, j in candidate_pairs:
        # Evaluate change in correlation at worst_shift
        gain = 0
        side_effects = 0
        
        # Recompute means only those terms that involve these indices:
        pure_improvement = 0  # for shift k, how much does recompute of indices i, j change it
        
        # Loop through shift k_i, k_j show how i, j contribute if were in corr_i, corr_j?
        # But actually calculate change directly from affected terms in the k-th mode.
        # Only affects two terms in sum for shift k:
        #          M_k = sum_{i} b_i * b_{i+k}
        # So swapping i,j (i=1, j=0) gives net change in correlation k as:
        # (b_{i+k} from 1 to 0) + (b_{j+k} from 0 to 1) = - b_{i+k} + b_{j+k}
        
        new_corr = 0
        for k in range(1, n):
            # Terms that involve i and j: possibly b_i * b_{i+k} and b_j * b_{j+k}
            term_i = binary[i] * binary[i+k] if i+k < n else 0
            term_j = binary[j] * binary[j+k] if j+k < n else 0
            old_term_i = 1 * binary[i+k] if i+k < n else 0
            old_term_j = 0 * binary[j+k] if j+k < n else 0
            change = (1 - binary[i+k]) if i+k < n else 0  # binary goes from 1 to 0
            change += binary[j+k] if j+k < n else 0  # binary goes from 0 to 1
            
            # Granular update for worst shift's correlation
            if k == worst_shift:
                gain += change
                pure_improvement += change
                
        # If this improves the worst correlation
        if gain > best_swap_gain and pure_improvement > 0:
            best_swap_gain = gain
            best_i = i
            best_j = j
            
    # Perform best swap
    if best_i >= 0:
        binary[best_i] = 0
        binary[best_j] = 1
        
    return binary
