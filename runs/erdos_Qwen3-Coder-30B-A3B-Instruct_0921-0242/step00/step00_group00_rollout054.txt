def project_capped_simplex(v, total):
    """Project onto feasible region: 0 <= h[i] <= 1, sum(h) = total."""
    v = np.asarray(v)
    # Bisection on lambda to satisfy constraint
    def sum_p_lambda(l):
        return np.sum(np.clip(v - l, 0, 1))
    
    l_low, l_high = np.min(v), np.max(v) + 1.0
    for _ in range(100):
        l_mid = (l_low + l_high) / 2
        s = sum_p_lambda(l_mid)
        if s > total:
            l_low = l_mid
        else:
            l_high = l_mid
    lambda_star = (l_low + l_high) / 2
    return np.clip(v - lambda_star, 0, 1)

def auto_corr(h):
    """Compute linear correlation (with zero padding)."""
    # Zero-padded convolution yields nn elements
    _h = np.pad(h, (0, len(h)), mode='constant', constant_values=0)
    return fftconvolve(_h, h[::-1], mode='valid')[:len(h)]

def grad_F(beta, h, O, w):
    """Compute -gradient of smooth surrogate: -div F_beta."""
    # w is already normalized, sum(w) = 1
    # g = sum(w_k grad O_k) where grad O_k is defined appropriately
    # We use fft convolution to evaluate forwards/backwards shifts.
    
    # Compute (w * h) and (w * rev(h)) where w is broadcasted
    # Equivalent to: g_i = \sum_k w_k * (h_{i+k} + h_{i-k})
    h_extend = np.pad(h, (len(h), len(h)), mode='constant', constant_values=0)
    # Reflect h for backward (i-k) terms
    h_backwards = h[::-1]
    h_backwards = np.pad(h_backwards, (0, len(h)-1), mode='constant', constant_values=0)
    # Forw (w * h), Backw (w * h+backward), add
    # Build convolution using fft frequent
    w_h = fftconvolve(w, h_extend, mode='valid')
    # Now we shift the backward
    w_h_backwards = fftconvolve(w, h_backwards, mode='valid')
    g = w_h[:len(h)] + w_h_backwards[len(h):]
    return -g

    h0 = project_capped_simplex(h0, total=n/2)
    h = h0.copy()
    t0 = time.time()
    best_C5 = 1e10
    best_h = h.copy()
    
    # Iteration budget
    max_iter_per_beta = 200
    beta_schedule = np.array(beta_schedule)
    
    # Loop over β
    for i, beta in enumerate(beta_schedule):
        # Inner loop with projected gradient
        iter_no_improvement = 0
        iter_since_checkpoint = 0
        
        for _ in range(max_iter_per_beta):
            if iter_since_checkpoint > 20 or time.time() - t0 > time_limit * 0.95:
                break
            # Compute O_k for current h
            O = auto_corr(h)
            # Compute soft sum exp(β O_k)
            w = np.exp(beta * O)
            w = w / np.sum(w)
            # Compute gradient of surrogate
            g = grad_F(beta, h, O, w)
            # Try line search from α1 = 0.1
            alpha = 0.1
            g_norm_sq = np.sum(g ** 2)
            if g_norm_sq < 1e-15:
                break
            # Armijo condition
            eta = 1e-4
            line_search_failed = True
            while line_search_failed and alpha > 1e-7:
                h_new = h - alpha * g
                h_new = project_capped_simplex(h_new, total=n/2)
                O_new = auto_corr(h_new)
                C5_new = 1 - np.min(O_new)
                if C5_new <= (1 - np.min(O) + eta * alpha * g_norm_sq):
                    h = h_new
                    iter_since_checkpoint = 0
                    if C5_new < best_C5:
                        best_C5 = C5_new
                        best_h = h.copy()
                    line_search_failed = False
                else:
                    alpha *= 0.5
                    iter_since_checkpoint += 1
        if iter_since_checkpoint > 20 or time.time() - t0 > time_limit * 0.95:
            break
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_C5 = 1e10
    best_h = None
    best_n = None
    
    # Test different discretization sizes
    ns = [256, 384]
    best_solved = None
    
    for n in ns:
        if time.time() - t_start > budget_s * 0.9:
            break
        # Define β sequence
        beta_0 = 0.5
        beta_final = 40
        n_levels = 12
        beta_schedule = [beta_0 * (beta_final / beta_0) ** (i / (n_levels - 1)) for i in range(n_levels)]
        
        # Multi restarts
        n_starts = 5
        for i in range(n_starts):
            _i = i + seed * 100
            restart_start = time.time()
            h, C5 = optimize_one_start(n, beta_schedule, time_limit=budget_s * 0.9)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h
                best_n = n
            if time.time() - t_start > budget_s * 0.9:
                break
            if time.time() - restart_start > budget_s * 0.3:
                break
            
        if time.time() - t_start > budget_s * 0.9:
            break

    # Final refinement: exact C5 minimization with projected gradient on worst lags
    if best_h is not None and time.time() - t_start < budget_s * 0.98:
        h = best_h
        n = best_n
        dx = 2 / n
        iters_refine = 300
        for it in range(iters_refine):
            if time.time() - t_start > budget_s * 0.98:
                break
            O = auto_corr(h)
            k_star = np.argmin(O)
            g = O.copy()
            g[:k_star] = 0
            g[k_star + 1:] = 0
            g[k_star] = 1
            g = g - np.sum(g) / len(g)
            # Gradient direction for worst k
            g_dir = np.zeros_like(h)
            # Compute that gradient as corrected version of g (shifted indices)
            # This is tricky: simplified version
            # To compute h_{i+k} and h_{i-k} for gradients:
            g_dir[k_star:] = -h[:-k_star] if k_star > 0 else -h
            g_dir[:-k_star] = -h[k_star:] if k_star > 0 else -h
            alpha = 0.1 / (1 + it ** 0.5)
            h_new = h - alpha * g_dir
            h_new = project_capped_simplex(h_new, total=n/2)
            O_new = auto_corr(h_new)
            C5_new = 1 - np.min(O_new)
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_new.copy()

    # Final evaluation
    dx = 2.0 / len(best_h) if best_h is not None else 2.0 / 256
    if best_h is not None:
        corr = np.convolve(best_h, 1 - best_h, mode="full") * dx
        C5 = np.max(corr)
        return (best_h.tolist(), C5, len(best_h))
    else:
        # Default fallback if we have no good result
        h_default = np.ones(256) * 0.5
        h_default = project_capped_simplex(h_default, total=128)
        corr = np.convolve(h_default, 1 - h_default, mode="full") * dx
        C5 = np.max(corr)
        return (h_default.tolist(), C5, 256)
