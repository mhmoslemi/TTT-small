import numpy as np

# Global constant for initial construction
initial_h_values = None

def project_capped_simplex(v, total=0.5):
    """Projects a vector onto the capped simplex: 0 <= x <= 1, sum(x) = total."""
    v = np.copy(v)
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - total) < 1e-12:
        return v
    # Use bisection to find lambda
    lb, ub = -1.0, 1.0
    for _ in range(100):
        lamb = (lb + ub) / 2
        p = np.clip(v - lamb, 0, 1)
        s = np.sum(p)
        if s > total:
            lb = lamb
        else:
            ub = lamb
    p = np.clip(v - lb, 0, 1)
    return p

def auto_corr(h):
    """Compute zero-padded auto-correlations."""
    n = len(h)
    h2 = np.concatenate([h, np.zeros(n)])
    corr = fftconvolve(h2, h[::-1], mode='full')
    return corr[:n]

def grad_F(beta, h, O, w):
    """Compute gradient of F_beta."""
    # Compute ∇O_k = [h_{i+k}, h_{i-k}]
    n = len(h)
    g = np.zeros_like(h)
    # Forward differences for h_{i+k}
    for k in range(1, n):
        g[:n-k] += w[k] * h[k:]
    # Reverse differences for h_{i-k}
    for k in range(1, n):
        g[k:] += w[k] * h[:n-k]
    return -g

        x_new = np.linspace(0, 2, n)
        h = np.interp(x_new, x_orig, h)
    h = np.clip(h, 0, 1)

            h2 = np.concatenate([h, np.zeros(n)])

            O = corr[:n]

            # Check improvement

            # Armijo line search
            # Compute direction
            h_new = h - alpha * g
            h_new = project_capped_simplex(h_new, total=n/2)
            # Compute new function value
            O_new = np.correlate(h_new, h_new, mode='full')[n-1:]
            f_new = 1.0 - np.min(O_new)
            
            if f_new <= f_old - 1e-4 * alpha * np.linalg.norm(g)**2:
                h = h_new
                if f_new < best_c5:
                    best_c5 = f_new
                    best_h = h.copy()
                iter_since_improvement = 0
                alpha *= 1.2  # Increase step size
            else:
                iter_since_improvement += 1
                alpha *= 0.5  # Reduce step size

                break
        
        # Break early if time is not enough
        if time.time() - t_start > time_limit - 0.3:
            break

    for it in range(200):

            break
        # Compute O
        O = auto_corr(h)
        # Project onto [0,1]
        O = np.clip(O, 0, 1)
        # Minimize via negative gradient of worst lag
        k_star = np.argmin(O)
        g = np.zeros_like(h)

            if i + k_star < len(h):
                g[i] += 1
            if i - k_star >= 0:
                g[i] += 1
        h = h - 0.005 * g
        h = project_capped_simplex(h, total=n/2)
        # Update best
        O_new = auto_corr(h)
        c5_new = 1.0 - np.min(O_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    return best_h, best_c5
