import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist

def decode(x, M):
    """
    Decode flat vector x into segment lengths u and heights a.
    x = (u0, ..., u_{M-1}, a0, v0, ..., v_{M-2})
    """
    u = x[:M]
    a0 = x[M]
    v = x[M+1:]
    
    # Compute heights: a_j = a_{j-1} - v_{j-1}
    a = np.empty(M+1)
    a[0] = a0
    for j in range(1, M):
        a[j] = a[j-1] - v[j-1]
    a[M] = a[M-1] - v[M-1] if M > 0 else 0
    
    # Ensure non-negative heights
    a = np.maximum(a, 0.0)
    
    # Compute breakpoints s
    s = np.concatenate([[0.0], np.cumsum(u)])
    
    # Ensure the last breakpoint is 2
    s[-1] = 2.0
    
    return s, a


def overlap_piecewise(s, a):
    """
    Compute the exact C5 for fixed breakpoints s and heights a.
    """
    M = len(a) - 1
    # All k values where overlaps might change
    K = set()
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0 <= k1 <= 2:
                K.add(k1)
            if 0 <= k2 <= 2:
                K.add(k2)
    K = sorted(K)
    
    # Include 0 and 2
    K = [0.0] + K + [2.0]
    
    # Evaluate F(k) for each k in K
    C5 = 0.0
    for k in K:
        Fk = 0.0
        for j in range(M+1):
            for l in range(M+1):
                # Length of intersection I_j and (I_l - k)
                s_j, s_j1 = s[j], s[j+1]
                s_l, s_l1 = s[l], s[l+1]
                left = max(s_j, s_l - k)
                right = min(s_j1, s_l1 - k)
                if left < right:
                    L = right - left
                    Fk += a[j] * (1 - a[l]) * L
        C5 = max(C5, Fk)
    
    return C5


def fitness(x, M, lambda_penalty=1e6, n_points=128):
    """
    Evaluate the fitness function (penalized)
    """
    try:
        s, a = decode(x, M)
        u = s[1:] - s[:-1]
        
        # Check validity
        if np.any(u <= 0) or np.any(a < 0) or np.any(a > 1):
            return 1e6
        
        integral_violation = np.sum(a[:-1] * u) - 1  # Last a is determined
        if abs(integral_violation) > 1e-8:
            penalty = lambda_penalty * integral_violation ** 2
        else:
            penalty = 0.0
            
        C5 = overlap_piecewise(s, a)
        
        # penalized fitness to be minimized
        return C5 + penalty
    except Exception:
        return 1e6


def project_capped_simplex(v, total):
    """
    Project vector v onto the simplex with sum = total, with each component in [0,1]
    Using binary search on lambda.
    """
    v = np.array(v)
    low, high = 0.0, np.max(v) + 1.0
    
    def Phi(lam):
        projected = np.maximum(v - lam, 0.0)
        return np.sum(projected) - total
    
    # Binary search for lambda
    while high - low > 1e-12:
        mid = (low + high) / 2.0
        if Phi(mid) > 0:
            low = mid
        else:
            high = mid
    
    lamb = (low + high) / 2.0
    result = np.maximum(v - lamb, 0.0)
    
    # Adjust for numerical errors
    if abs(np.sum(result) - total) > 1e-10:
        adj = (total - np.sum(result)) / len(result)
        result += adj
    
    return result

    init_pop = []

    # Clip to bounds

            init_pop[i][j] = np.clip(init_pop[i][j], bounds1[j][0], bounds1[j][1])

    def _fitness_b(x):
        return fitness(x, M1, n_points=n1)

        _fitness_b,
        bounds1,
        seed=seed,
        maxiter=200,
        popsize=15 * len(bounds1),
        disp=False,
        x0=best_A,  # seed from previous best
        callback=lambda x, convergence: None
    )
    
    # Final refinement with subgradient + smooth surrogate
    best_x = res_b.x.copy()
    s, a = decode(best_x, M1)
    
    # Smooth surrogate function
    def smooth_surrogate(x, beta=50.0):
        s, a = decode(x, M1)
        M = len(a) - 1
        K = set()
        for j in range(M+1):
            for l in range(M+1):
                k1 = s[l] - s[j+1]
                k2 = s[l+1] - s[j]
                if 0 <= k1 <= 2:
                    K.add(k1)
                if 0 <= k2 <= 2:
                    K.add(k2)
        K = sorted(K)
        K = [0.0] + K + [2.0]
        F_vals = []
        for k in K:
            Fk = 0.0
            for j in range(M+1):
                for l in range(M+1):
                    s_j, s_j1 = s[j], s[j+1]
                    s_l, s_l1 = s[l], s[l+1]
                    left = max(s_j, s_l - k)
                    right = min(s_j1, s_l1 - k)
                    if left < right:
                        L = right - left
                        Fk += a[j] * (1 - a[l]) * L
            F_vals.append(Fk)
        return 1.0 / beta * np.log(np.sum(np.exp(beta * np.array(F_vals))))

    def _smooth_surrogate(x):
        return smooth_surrogate(x, beta=50.0)

        s, a = decode(x, M1)
        u = s[1:] - s[:-1]

        M = len(a) - 1

            Fk = 0.0
            for j in range(M+1):
                for l in range(M+1):

                    s_l, s_l1 = s[l], s[l+1]
                    left = max(s_j, s_l - k)
                    right = min(s_j1, s_l1 - k)
                    if left < right:
                        L = right - left
                        Fk += a[j] * (1 - a[l]) * L
            if Fk > worst_F:
                worst_F = Fk
                worst_k = k
                
        # Compute subgradient via finite differences:
        eps = 1e-6
        grad = np.zeros_like(final_x)
        base_val = overlap_piecewise(s, a)
        for i in range(len(final_x)):
            x_plus = final_x.copy()
            x_plus[i] += eps
            s_p, a_p = decode(x_plus, M1)
            val_plus = overlap_piecewise(s_p, a_p)
            grad[i] = (val_plus - base_val) / eps
            
        # Step size
        alpha = 1e-3 / (1 + np.sqrt(t))
        final_x = final_x - alpha * grad
        final_x = np.clip(final_x, [b[0] for b in bounds1], [b[1] for b in bounds1])
    
    # Get final function description
    s_final, a_final = decode(final_x, M1)
    
    # Discretize
    h_values = []
    for i in range(n1):
        x = (i + 0.5) * dx1
        j = np.searchsorted(s_final, x, side='right') - 1
        h_values.append(a_final[j])
    
    # Project to integrate to 1
    h_values = project_capped_simplex(h_values, total=n1 / 2)
    
    # Compute final C5
    corr = np.convolve(h_values, 1 - np.array(h_values), mode='full') * dx1
    c5_bound = np.max(corr)
    
    return (h_values, float(c5_bound), n1)
