import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fftpack import fft, ifft

def evaluate_h(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # autocorrelation via FFT
    H = fft(h)
    A = ifft(np.abs(H)**2).real
    # A[0] = sum(h^2), A[1] = sum(h[i]*h[i+1]), etc.
    # we want the min of A[1:] (since A[0] is not a "shift" with nonzero k)
    min_A = np.min(A[1:])
    return dx * (n/2 - min_A)

def block_pattern_3(a, b, n=400):
    """Create 3-block pattern: [1, p, 0] with breakpoints a, b"""
    if a >= b:
        return None
    p = (1 - a) / (b - a) if b > a else 0
    if not (0 <= p <= 1):
        return None
    h = np.zeros(n)
    dx = 2.0 / n
    ia = int(a / dx)
    ib = int(b / dx)
    h[:ia] = 1.0
    h[ia:ib] = p
    return h

def block_search_3():
    """Block enumeration for 3-block patterns"""
    best_h = None
    best_c5 = float('inf')
    n = 400
    dx = 2.0 / n
    step = 0.02
    start_t = time.time()
    for a in np.arange(0, 2.0, step):
        for b in np.arange(a + step, 2.0, step):
            h = block_pattern_3(a, b, n)
            if h is not None:
                c5 = evaluate_h(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            if time.time() - start_t > 75:
                break
        if time.time() - start_t > 75:
            break
    return best_h, best_c5

def block_pattern_4(a, b, c, n=400):
    """Create 4-block pattern: [1, p, q, 0] with breakpoints a,b,c"""
    if a >= b or b >= c:
        return None
    p = (1.0 - a - (c-b)*0.5) / (b - a) if b > a else 0
    q = (1.0 - a - p*(b-a)) / (c-b) if c > b else 0
    if not (0 <= p <= 1 and 0 <= q <= 1):
        return None
    h = np.zeros(n)
    dx = 2.0 / n
    ia = int(a / dx)
    ib = int(b / dx)
    ic = int(c / dx)
    h[:ia] = 1.0
    h[ia:ib] = p
    h[ib:ic] = q
    return h

def block_search_4():
    """Block enumeration for 4-block patterns"""
    best_h = None
    best_c5 = float('inf')
    n = 400
    dx = 2.0 / n
    step = 0.04
    start_t = time.time()
    for a in np.arange(0, 2.0, step):
        for b in np.arange(a + step, 2.0, step):
            for c in np.arange(b + step, 2.0, step):
                h = block_pattern_4(a, b, c, n)
                if h is not None:
                    c5 = evaluate_h(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                if time.time() - start_t > 100:
                    return best_h, best_c5
    return best_h, best_c5

def mixture_of_gaussians(params, n=200):
    """Create a h using mixture of Gaussians"""
    # params: [m, centers, amplitudes, widths] 
    # m = number of bumps (max 5), rest is flattened (m + m + m)
    m = int(params[0])
    centers = params[1:1+m]
    amplitudes = params[1+m:1+2*m]
    widths = [0.1] * m  # fixed for simplicity
    
    h = np.zeros(n)
    dx = 2.0 / n
    for i in range(m):
        c = centers[i]
        a = amplitudes[i]
        w = widths[i]
        # make a box from c-w to c+w
        start_idx = max(0, int((c - w) / dx))
        end_idx = max(0, min(n, int((c + w) / dx)))
        # set the parts in intervals to a
        h[start_idx:end_idx] += a
    
    # normalize to sum = n/2
    total_mass = np.sum(h)
    if total_mass > 0:
        h = h * (n / 2) / total_mass
    
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    return h

def de_optimization():
    """Run differential evolution on mixture of Gaussians"""
    n = 200
    def objective(params):
        h = mixture_of_gaussians(params, n)
        return evaluate_h(h)
    
    # Set bounds based on max 5 bumps
    upper_bounds = []
    lower_bounds = []
    
    # 1. for m, allow 1 to 5 bumps
    lower_bounds.append(1.0)
    upper_bounds.append(5.0)
    
    # 2. 5 centers in [0,2] 
    for i in range(5):
        lower_bounds.append(0.0)
        upper_bounds.append(2.0)
        
    # 3. 5 amplitudes in [0,1]
    for i in range(5):
        lower_bounds.append(0.0)
        upper_bounds.append(1.0)
        
    bounds = list(zip(lower_bounds, upper_bounds))
    
    # Set the seed for reproducibility
    np.random.seed(42)
    
    start_t = time.time()
    res = differential_evolution(objective, 
                                 bounds,
                                 maxiter=400, 
                                 popsize=30,
                                 strategy='best1bin',
                                 seed=42,
                                 workers=1,
                                 updating='immediate')
    
    if time.time() - start_t > 120:
        return None, float('inf')
        
    best_h = mixture_of_gaussians(res.x, n)
    c5 = evaluate_h(best_h)
    return best_h, c5

def project_onto_feasible(h, target_sum=0.5):
    # Clip h to [0,1]
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * target_sum / current_sum
    h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient_ascent(initial_h, max_time=280):
    """Projected gradient ascent for improving h"""
    start_t = time.time()
    h = initial_h.copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    iteration = 0
    last_improvement = 0
    
    # Multi-resolution schedule
    grids = [150, 300] 
    step_sizes = [0.2, 0.05]
    
    for g_idx, new_n in enumerate(grids):
        if time.time() - start_t > max_time:
            break
        if new_n > n:
            new_h = np.interp(np.linspace(0, 2, new_n), np.linspace(0, 2, n), h)
            h = new_h
            n = new_n
            dx = 2.0 / n
        
        eta = step_sizes[g_idx]
        for i in range(1000):  # max iterations per stage
            if time.time() - start_t > max_time:
                break
            
            # compute autocorrelations via FFT
            H = fft(h)
            A = ifft(np.abs(H)**2).real
            k_star = np.argmin(A[1:]) + 1  # ignore A[0] 
            
            # gradient of sum of this is 2*shifted h
            g = np.roll(h, k_star)
            
            h = h + eta * g
            h = project_onto_feasible(h)
            
            # check for improvement
            c5 = evaluate_h(h) 
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = 0
            else:
                last_improvement += 1
            
            # terminate early if no improvement
            if last_improvement > 200:
                break
            
            iteration += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    # Start timing
    start_time = time.time()
    
    # Set seed
    np.random.seed(seed)
    
    # Stage 1: Block-based search
    h1, c5_1 = block_search_3()
    if h1 is None:
        h1, c5_1 = block_search_4()
    
    # Stage 2: DE optimization  
    h2, c5_2 = de_optimization()
    
    # Pick the better
    if c5_1 < c5_2:
        initial_h = h1
        best_c5 = c5_1
    else:
        initial_h = h2 
        best_c5 = c5_2
    
    # Stage 3: gradient ascent
    final_h, final_c5 = projected_gradient_ascent(initial_h, max_time=280)
    
    # Return the best solution seen
    best_h = final_h
    final_c5 = evaluate_h(best_h)
    
    if c5_1 < final_c5:
        best_h = h1 
        final_c5 = c5_1
    if c5_2 < final_c5:
        best_h = h2 
        final_c5 = c5_2
    
    return (best_h.tolist(), final_c5, len(best_h))
