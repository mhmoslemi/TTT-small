```python
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.linalg import solve

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Parameters
    n_points = 256
    n_freqs = 8  # M in the strategy
    total_params = 2 * n_freqs  # 2M coefficients
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Precompute Fourier basis matrix
    def compute_fourier_basis(n, M):
        # Create grid 0 to n-1
        i = np.arange(n)
        # Build the basis matrix
        # For each frequency m from 1 to M
        cos_basis = np.cos(2 * np.pi * np.outer(i, np.arange(1, M+1)) / n)
        sin_basis = np.sin(2 * np.pi * np.outer(i, np.arange(1, M+1)) / n)
        # Resulting shape is (n, 2*M) = (n, 2*M)  
        return np.hstack([cos_basis, sin_basis])
    
    # Precompute for n=256
    Phi = compute_fourier_basis(n_points, n_freqs)
    # Precompute the projection matrix inv(Phi^T * Phi) * Phi^T
    PhiTPhi = Phi.T @ Phi
    PhiTPhi_inv = solve(PhiTPhi, np.eye(total_params))
    proj_mat = PhiTPhi_inv @ Phi.T
    
    def project_to_feasible(c):
        # Compute h = Phi*c + 0.5 (constant offset)
        h = Phi @ c + 0.5
        # clamp to [0,1]
        h = np.clip(h, 0, 1)
        # project to feasible set
        residual = h - (Phi @ c + 0.5)
        delta_c = proj_mat @ residual
        return c + delta_c
    
    def compute_C5(c):
        h = Phi @ c + 0.5
        # compute overlap integral using FFT
        corr = np.convolve(h, 1 - h, mode='full') * dx
        return corr.max()
    
    # Simple (mu,lambda) ES with restarts
    def evolve_with_projection(budget_time):
        best_c = None
        best_C5 = np.inf
        mu = total_params // 2
        lambda_ = 4 + int(3 * np.log(total_params))
        sigma = 0.2
        # CMA-ES inspired but simplified
        restarts = 10  # number of restarts
        lambda_per_restart = lambda_ if lambda_ <= 50 else 50
        lambda_per_restart = max(4, lambda_per_restart)
        remaining_budget = budget_time
        start_time = time.time()
        
        for r in range(restarts): 
            if time.time() - start_time > remaining_budget - 0.5:
                break
            # Initialize population
            population = []
            fitness = []
            for i in range(lambda_per_restart):
                c = np.random.normal(0, sigma, total_params)
                c = project_to_feasible(c)
                population.append(c)
                f = compute_C5(c)
                fitness.append(f)
            
            # Sort by fitness
            sorted_idx = np.argsort(fitness)
            population = [population[i] for i in sorted_idx]
            fitness = [fitness[i] for i in sorted_idx]
            
            # Keep best so far
            if fitness[0] < best_C5:
                best_C5 = fitness[0]
                best_c = population[0].copy()
            
            if best_C5 <= target_bound:
                break
            
            # Update for next generation
            for gen in range(50):  # Allow many generations but time check
                if time.time() - start_time > remaining_budget - 0.5:
                    break
                offspring = []
                # Select best mu
                selected = population[:mu]
                # Generate offspring (simple mutation)
                for i in range(lambda_per_restart):
                    # Choose parent
                    parent = selected[np.random.randint(0, mu)]
                    # Mutate
                    child = parent + np.random.normal(0, sigma, total_params)
                    child = project_to_feasible(child)
                    offspring.append(child)
                
                # Evaluate offspring
                offspring_fitness = []
                for child in offspring:
                    f = compute_C5(child)
                    offspring_fitness.append(f)
                
                # Combine and sort
                all_pop = population + offspring
                all_fit = fitness + offspring_fitness
                sorted_idx = np.argsort(all_fit)
                # Keep top lambda
                population = [all_pop[i] for i in sorted_idx[:lambda_per_restart]]
                fitness = [all_fit[i] for i in sorted_idx[:lambda_per_restart]]
                
                # Update best
                if fitness[0] < best_C5:
                    best_C5 = fitness[0]
                    best_c = population[0].copy()
                
                if best_C5 <= target_bound:
                    break
        return best_c, best_C5
    
    # Run the optimization
    start_time = time.time()
    c_best, best_C5 = evolve_with_projection(budget_s - 0.5)
    
    if best_C5 <= target_bound:
        pass
    else:
        # Try higher resolution if time allows
        if n_points < 512 and time.time() - start_time < (budget_s - 0.5) * 0.8:
            n_points = 512
            Phi_large = compute_fourier_basis(n_points, n_freqs)
            # Create larger projection matrix
            PhiTPhi_large = Phi_large.T @ Phi_large
            PhiTPhi_inv_large = solve(PhiTPhi_large, np.eye(total_params))
            proj_mat_large = PhiTPhi_inv_large @ Phi_large.T
            
            # use same coefficients to initialize the higher res
            Phi_reduced = compute_fourier_basis(n_points//2, n_freqs)
            c_reduced = c_best
            h_reduced = Phi_reduced @ c_reduced + 0.5
            # interpolate to larger grid manually
            h_large = np.zeros(n_points)
            # replicate in higher res
            h_large[::2] = h_reduced  
            # interpolate missing points 
            h_large[1::2] = (h_large[::2][1:] + h_large[::2][:-1]) / 2 
            # compute initial coefficient
            c_best = proj_mat_large @ (h_large - 0.5)
            c_best = project_to_feasible(c_best)
            # re-run optimization at higher resolution
            c_best, best_C5 = evolve_with_projection((budget_s - 0.5) * 0.5)
    
    # Final refinement with projected gradient
    # Create gradient using the index that maximizes overlap  
    # we'll just do a few iterations of projected gradient descent
    c = c_best.copy()
    step_size = 0.01
    max_iter = 200
    for i in range(max_iter):
        h = Phi @ c + 0.5
        # Compute correlation vector  
        corr = np.convolve(h, 1 - h, mode='full') * dx
        max_idx = corr.argmax()
        # max_idx in [0, 2*n-2], which represents k = max_idx - n + 1  
        # we need to compute dC5/dh for this k
        # gradient is (1 - h_{i+k}) - h_{i-k}
        grad = np.zeros_like(h)
        k = max_idx - n_points + 1 
        if k >= 0:
            # Let's handle periodic indices by using full arrays
            # we compute the gradient inline for all entries at once
            # shift h by k and -k 
            h_delayed = np.roll(h, k)
            h_forward = np.roll(h, -k)
            grad = (1 - h_delayed) - h_forward 
        else:
            # k < 0 
            h_delayed = np.roll(h, k)  # negative means delaying
            h_forward = np.roll(h, -k)  # positive delay
            grad = (1 - h_delayed) - h_forward
        # Compute gradient with respect to c
        grad_c = Phi.T @ grad
        # step
        c -= step_size * grad_c
        # project to feasible  
        c = project_to_feasible(c)
        # small decay of step size
        step_size *= 0.995
    
    # Final h computation
    h_opt = Phi @ c + 0.5
    # Ensure integral is exactly 1
    integral = np.sum(h_opt) * dx
    if np.abs(integral - 1) > 1e-12:
        correction = (1 - integral) / n_points
        h_opt += correction
        # clip to [0,1]
        h_opt = np.clip(h_opt, 0, 1)
    # Final evaluation
    final_corr = np.convolve(h_opt, 1 - h_opt, mode='full') * dx
    final_C5 = final_corr.max()
    
    # Return as list, scalar float, int
    return h_opt.tolist(), float(final_C5), n_points
```
