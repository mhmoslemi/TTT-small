import numpy as np
import time
import math
from scipy import signal

def project_box_affine(h, target_sum):
    """Project h onto [0,1] and affine constraint sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    delta = (np.sum(h) - target_sum) / len(h)
    h -= delta
    h = np.clip(h, 0, 1)
    return h

def autocorr(h):
    """Compute circular autocorrelation using FFT."""
    return np.real(np.fft.ifft(np.abs(np.fft.fft(h))**2))

def compute_C5(h, dx):
    """Compute C5 using np.correlate with full mode."""
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the relevant part: only lags from -n+1 to n-1
    n = len(h)
    corr = corr[n-1: 2*n-1]
    return np.max(corr) * dx

def gradient_ascent_step(h, dx, step_size):
    """Do one gradient ascent step on the minimal autocorrelation."""
    aut = autocorr(h)
    lag = np.argmin(aut)
    grad = -(np.roll(h, lag) + np.roll(h, -lag))
    h_new = h + step_size * grad
    h_new = project_box_affine(h_new, len(h) / 2)
    return h_new

def grid_search_two_block(n_points, dx):
    """Try two-block configurations."""
    best_c5 = float('inf')
    best_h = None
    # Try breakpoints a, b on coarse grid
    grid = np.arange(dx, 2 - dx, dx)
    for a in grid:
        for b in grid:
            if b <= a:
                continue
            L1 = a
            L2 = b - a
            L3 = 2 - b
            if L2 <= 0:
                continue
            c = (1 - L1) / L2
            if 0 <= c <= 1:
                h = np.zeros(n_points)
                h[:int(L1/dx)] = 1.0
                h[int(L1/dx):int(b/dx)] = c
                h[int(b/dx):] = 0.0
                c5 = compute_C5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    return best_h, best_c5
