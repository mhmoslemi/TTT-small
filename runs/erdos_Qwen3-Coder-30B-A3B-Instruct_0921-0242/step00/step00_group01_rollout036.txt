import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft

def evaluate_h(h):
    """Evaluate C5 for a given discretized h."""
    n = len(h)
    dx = 2.0 / n
    # Compute autocorrelations via FFT
    H = fft(h)
    A = ifft(np.abs(H)**2).real
    # A_k = sum_i h_i h_{i+k}, k=0, ..., n-1 (circular)
    min_A = np.min(A)
    C5 = dx * (n/2 - min_A)
    return C5

def block_family_3(a, b):
    """Generate 3-block function: [1, p, 0] with breakpoints a, b."""
    # p = (1 - a) / (b - a)
    p = (1 - a) / (b - a)
    if 0 <= p <= 1:
        h = np.ones(400)
        h[400*a//2 : 400*b//2] = p
        h[400*b//2 :] = 0
        return h
    else:
        return None

def block_family_4(a, b, c):
    """Generate 4-block function: [1, p, q, 0] with breakpoints a, b, c."""
    # Solve for p,q: a*1 + (b-a)*p + (c-b)*q = 1 and (b-a)*p = (c-b)*q
    # => p = (1 - a - (c-b)*q) / (b-a)
    # => p = q * (c-b)/(b-a) => q = p*(b-a)/(c-b)
    # => p = (1-a) / (b-a + (c-b) * p*(b-a)/(c-b))
    # => p * (b-a + (c-b) * p*(b-a)/(c-b)) = (1-a)
    # => p * (b-a) + p^2 * (b-a)^2 / (c-b) = (1-a)
    # => p * (b-a) + p^2 * (b-a)^2 / (c-b) - (1-a) = 0
    # Quicker: Balance constraint: integral = 1, and p*(b-a) = q*(c-b)
    # Let p*(b-a) = q*(c-b) = m => p = m/(b-a), q = m/(c-b)
    # Then: a + m + m = 1 => m = (1-a)/2
    # => p = (1-a)/(2*(b-a)), q = (1-a)/(2*(c-b))
    denom1 = b - a
    denom2 = c - b
    if denom1 > 0 and denom2 > 0:
        m = (1 - a) / 2.0
        p = m / denom1
        q = m / denom2
        if 0 <= p <= 1 and 0 <= q <= 1:
            h = np.ones(400)
            h[400*a//2 : 400*b//2] = p
            h[400*b//2 : 400*c//2] = q
            h[400*c//2 :] = 0
            return h
    return None

def sweep_blocks():
    """Enumerate all candidate 3 and 4 block families."""
    best_C5 = float('inf')
    best_h = None
    n_base = 400
    # 3-block sweep
    a_vals = np.arange(0.02, 2.0, 0.02)
    b_vals = np.arange(0.02, 2.0, 0.02)
    for a in a_vals:
        for b in b_vals:
            if a < b:
                h = block_family_3(a, b)
                if h is not None:
                    # Interpolate to 400 points
                    h = h[:n_base]
                    C5 = evaluate_h(h)
                    if C5 < best_C5:
                        best_C5 = C5
                        best_h = h.copy()
    # 4-block sweep
    a_vals = np.arange(0.04, 2.0, 0.04)
    b_vals = np.arange(0.04, 2.0, 0.04)
    c_vals = np.arange(0.04, 2.0, 0.04)
    for a in a_vals:
        for b in b_vals:
            for c in c_vals:
                if a < b < c:
                    h = block_family_4(a, b, c)
                    if h is not None:
                        h = h[:n_base]
                        C5 = evaluate_h(h)
                        if C5 < best_C5:
                            best_C5 = C5
                            best_h = h.copy()
    return best_h, best_C5

def gaussian_bump(h_params, grid_len=200):
    """Generate h using a mixture of Gaussian-like bumps."""
    # h_params = [c1, a1, c2, a2, c3, a3], 0 <= a_j <= 1
    centers = h_params[::2]
    amplitudes = h_params[1::2]
    dx = 2.0 / grid_len
    h = np.zeros(grid_len)
    for c, a in zip(centers, amplitudes):
        if 0 <= a <= 1:
            # Coverage of 0.1 on each side
            left = max(0, int((c - 0.05) / dx))
            right = min(grid_len, int((c + 0.05) / dx))
            h[left:right] += a
    # Normalize to sum = grid_len/2
    total_mass = np.sum(h) * dx
    if total_mass > 0:
        h *= (grid_len / 2) / total_mass
    h = np.clip(h, 0, 1)
    return h

    """Run differential evolution."""
    n = len(start_h)
    dx = 2.0 / n
    def objective(h_params):
        h = gaussian_bump(h_params, n)
        # Force the mean to be compatible
        h_sum = np.sum(h) * dx
        if h_sum > 0:
            h *= (n / 2) / h_sum
        h = np.clip(h, 0, 1)
        C5 = evaluate_h(h)
        return C5

    # DE parameters
    bounds = [(0.0, 2.0)] * 6  # 3 centers in [0,2], 3 amplitudes in [0,1]
    initial_candidate = [0.5, 0.5, 1.0, 0.5, 1.5, 0.5]  # safe start
    try:
        result = differential_evolution(
            objective, bounds,
            maxiter=100, popsize=30,
            seed=42, disp=False, polish=False
        )
        h_opt = gaussian_bump(result.x, n)
        h_opt_sum = np.sum(h_opt) * dx
        if h_opt_sum > 0:
            h_opt *= (n / 2) / h_opt_sum
        h_opt = np.clip(h_opt, 0, 1)
        C5_opt = evaluate_h(h_opt)
        return h_opt, C5_opt
    except Exception:
        return start_h, evaluate_h(start_h)

def project_to_feasible(h, n):
    """Project h onto feasible set."""
    # Clip to [0,1] and rescale
    h = np.clip(h, 0, 1)
    # Rescale mean
    h_sum = np.sum(h)
    if h_sum > 0:
        h *= (n / 2) / h_sum
    h = np.clip(h, 0, 1)
    return h
