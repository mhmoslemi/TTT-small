def decode(x, M):
    n = len(x)
    u = x[:M]
    a0 = x[M]
    v = x[M+1:]
    u = np.concatenate((u, [2.0 - np.sum(u)]))
    a = np.zeros(M+1)
    a[0] = a0
    for i in range(1,M+1):
        if i < M:
            a[i] = max(0.0, a[i-1] - v[i-1])
        else:
            a[i] = max(0.0, a[i-1] - v[i-2])

    s = np.zeros(M+1)
    s[0] = 0.0
    for i in range(M):
        s[i+1] = s[i] + u[i]
    return u, a, s

def overlap_piecewise(s, a):
    M = len(s) - 1
    klist = []
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0.0 <= k1 <= 2.0:
                klist.append(k1)
            if 0.0 <= k2 <= 2.0:
                klist.append(k2)
    klist = sorted(set(klist))
    klist = [0.0] + klist + [2.0]
    max_overlap = 0.0
    for k in klist:
        overlap = 0.0
        for j in range(M+1):
            for l in range(M+1):
                left = max(s[j], s[l] - k)
                right = min(s[j+1], s[l+1] - k)
                if right > left:
                    overlap += a[j] * (1.0 - a[l]) * (right - left)
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def project_capped_simplex(v, total):
    v = np.asarray(v)
    n = len(v)
    lmax = 0.0
    lmin = -100.0
    delta = np.sum(v) - total
    if abs(delta) < 1e-10:
        return v

    while abs(lmax - lmin) > 1e-12:
        l = (lmax + lmin) * 0.5
        vnew = np.clip(v - l, 0.0, 1.0)
        delta_new = np.sum(vnew) - total
        if delta_new > 0:
            lmin = l
        else:
            lmax = l

    vnew = np.clip(v - l, 0.0, 1.0)
    return vnew

def fitness_function(x, M, n_points):
    u, a, s = decode(x, M)
    # Check bounds
    if np.any(u < 0) or np.any(a < 0) or np.any(a > 1):
        return 1e6
    integral_residual = np.sum(a * u) - 1.0
    penalty = 1e6 * integral_residual ** 2
    try:
        c5 = overlap_piecewise(s, a)
        return c5 + penalty
    except:
        return 1e6
