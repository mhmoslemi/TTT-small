import numpy as np
import cvxpy as cp
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    n = len(h)
    # Compute full correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value (normalized by dx = 2/n) gives us C5
    dx = 2.0 / n
    return np.max(corr) * dx

def solve_sdp(n, time_limit=200):
    """Solve SDP relaxation to get lower bound"""
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.diag(X) == h,  # X_ii = h_i (since 0 <= h_i <= 1, so h_i^2 = h_i)
        X >> 0,  # X is PSD
        0 <= h, h <= 1,  # Bounded
        cp.sum(h) == n / 2,  # Sum constraint
        t >= 0,  # t >= 0
    ]

    # Correlation constraints for each shift
    dx = 2.0 / n
    # Generate permutation matrices for each shift and compute traces
    for s in range(-n + 1, n):
        P_s = np.zeros((n, n), dtype=np.float64)
        # Map indices with periodic boundary conditions
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                P_s[i, j] = 1.0
        # Add constraint t >= const - dx * Tr(P_s * X)
        # const = dx * (n / 2)
        const = dx * (n / 2)
        constraints.append(t >= const - dx * cp.trace(P_s @ X))

    # Objective
    objective = cp.Minimize(t)

    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False, time_limit=time_limit,
               eps=1e-4, max_iters=20000)

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        t_relax = prob.value
        h_sol = h.value
        return h_sol, t_relax
    else:
        return None, None

def randomized_rounding(h_sdp, n, rounds=30):
    """Use randomized rounding to produce feasible h vectors"""
    # Perform eigen-decomposition of X* (approximated by h h^T)
    # We directly use the h_sol since X* ≈ h h^T
    h_star = h_sdp.copy()
    best_h = None
    best_c5 = np.inf

    for _ in range(rounds):
        # Sample standard normal vector
        g = np.random.normal(0, 1, n)
        # Round
        y = h_star + g * 0.01  # Small Gaussian jitter
        y = np.clip(y, 0, 1)
        
        # Adjust so sum = n/2
        current_sum = np.sum(y)
        deficit = n / 2 - current_sum
        if abs(deficit) > 1e-10:
            indices = np.where((y > 0) & (y < 1))[0]
            # Distribute deficit
            if len(indices) > 0:
                delta = deficit / len(indices)
                for idx in indices:
                    new_val = y[idx] + delta
                    y[idx] = np.clip(new_val, 0, 1)
        y = y - np.mean(y) + n / (2 * n)  # Approximate adjustment
        
        # Ensure sum constraint with clipping
        current_sum = np.sum(y)
        scale_factor = (n / 2) / current_sum if current_sum != 0 else 1.0
        y = np.clip(y * scale_factor, 0, 1)
        current_sum = np.sum(y)
        # Final adjustment if necessary
        if abs(current_sum - n / 2) > 1e-10:
            idxs = np.argsort(y)
            diff = current_sum - n / 2
            if diff > 0:
                # We need to reduce values
                for i in range(int(diff)):
                    if y[idxs[i]] > 0:
                        y[idxs[i]] -= 1e-8
                        y[idxs[i]] = max(0, y[idxs[i]])
            else:
                # We need to increase values
                for i in range(int(-diff)):
                    if y[idxs[-(i+1)]] < 1:
                        y[idxs[-(i+1)]] += 1e-8
                        y[idxs[-(i+1)]] = min(1, y[idxs[-(i+1)]])

        # Evaluate and store best
        c5 = evaluate_c5(y)
        if c5 < best_c5:
            best_c5 = c5
            best_h = y.copy()

    return best_h, best_c5

def local_refinement(h, max_iter=1000):
    """Coordinate-wise refinement using analytic gradient approximations"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = evaluate_c5(h)
    old_c5 = c5 + 1
    iters = 0
    
    while abs(c5 - old_c5) > 1e-6 and iters < max_iter:
        iters += 1
        old_c5 = c5
        # Iterate through all indices
        for i in range(n):
            # Try small perturbations
            moves = []
            for eps in [-0.002, -0.001, 0.001, 0.002]:
                new_h = h.copy()
                old_val = h[i]
                new_val = old_val + eps
                new_val = np.clip(new_val, 0, 1)
                new_h[i] = new_val
                
                # Adjust other element to preserve sum
                sign = np.sign(eps)
                j = (i + 1) % n
                while abs(new_h[j] - sign * 1e-8) < 1e-8 and abs(new_h[i] - new_val) > 1e-8:
                    j = (j + 1) % n
                new_h[j] = np.clip(new_h[j] - eps, 0, 1)
                
                # Evaluate new max
                new_c5 = evaluate_c5(new_h)
                moves.append((new_c5, eps, new_h))
            
            # Select improving move
            moves.sort()
            new_c5, delta, new_h = moves[0]
            if new_c5 < c5:
                h = new_h
                c5 = new_c5
                
    return h, c5
