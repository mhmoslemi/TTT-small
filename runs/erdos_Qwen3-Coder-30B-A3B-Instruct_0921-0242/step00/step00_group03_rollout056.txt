import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Evaluate the overlap integral for the given h_values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute correlation for all shifts
    corr = correlate(h_values, 1 - h_values, mode='full')
    # Take the maximum over all non-zero shifts
    max_corr = np.max(corr[n-1:])  # Only positive shifts (k > 0)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n = 100  # start with this size
    dx = 2.0 / n
    
    # Create initial binary solution using ``initial_h_values`` if available
    # But since its not provided directly, start with balanced binary vector
    h0 = np.zeros(n, dtype=int)
    h0[:n//2] = 1  # exactly n/2 ones
    np.random.shuffle(h0)
    
    # MILP solver
    def solve_milp(n_points):
        # Variables
        h = cp.Variable(n_points, boolean=True)
        y = cp.Variable((n_points, n_points), boolean=True)
        t = cp.Variable(integer=True, lower_bound=0)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points // 2,  # exactly half ones
        ]
        
        # Linearization constraints for y[i,s] = h[i]*h[i+s]
        for s in range(1, n_points):
            for i in range(n_points - s):
                # y[i,s] <= h[i]
                constraints.append(y[i,s] <= h[i])
                # y[i,s] <= h[i+s]
                constraints.append(y[i,s] <= h[i+s])
                # y[i,s] >= h[i] + h[i+s] - 1
                constraints.append(y[i,s] >= h[i] + h[i+s] - 1)
        
        # Overlap constraints
        for s in range(1, n_points):
            constraints.append(t <= cp.sum(y[:, s]))
        
        # Solve
        prob = cp.Problem(cp.Maximize(t), constraints)
        prob.solve(solver=cp.CBC, max_seconds=300, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value.astype(int), t.value
        else:
            return None, None
    
    h_bin, t = solve_milp(n)
    
    best_h = h_bin.copy() if h_bin is not None else h0.copy()
    best_c5 = 1 - 2 * t / n if t is not None else evaluate_erdos_solution(h0.astype(float))
    
    # Binary hill climbing to improve solution
    def compute_overlaps(h_vec):
        n = len(h_vec)
        # Generate all overlaps
        overlaps = {}
        for s in range(1, n):
            overlap = 0
            for i in range(n - s):
                overlap += h_vec[i] * h_vec[i+s]
            overlaps[s] = overlap
        return overlaps
    
    def update_overlaps(delta_overlaps, overlaps, h_vec, i1, i0):
        for s in range(1, len(h_vec)):
            if i1 + s < len(h_vec):
                delta_overlaps[s] += (h_vec[i0] - h_vec[i1])
            if i0 + s < len(h_vec):
                delta_overlaps[s] += (h_vec[i1] - h_vec[i0])
        return delta_overlaps

    def local_improvement(h_vec, overlaps, t_val):
        t = t_val
        h_temp = h_vec.copy()
        best_t = t
        best_h = h_temp.copy()
        time_left = budget_s - (time.time() - start_time)
        if time_left < 10:
            return h_temp, t
        
        # Try swaps
        best_improvement = 0
        num_swaps = 0
        for _ in range(1000):
            if time.time() - start_time > budget_s - 10:
                break
            ones = np.where(h_temp == 1)[0]
            zeros = np.where(h_temp == 0)[0]
            if len(ones) < 1 or len(zeros) < 1:
                break
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            
            # Simulate swap
            old_vals = [overlaps[s] for s in range(1, len(h_temp))] 
            new_overlaps = old_vals.copy()
            delta = 0
            for s in range(1, len(h_temp)):
                # We need to compute change in overlap due to swapping positions i0 and i1
                if i0+s < len(h_temp):
                    delta += h_temp[i1] - h_temp[i0]
                if i1+s < len(h_temp):
                    delta += h_temp[i0] - h_temp[i1]
                if i0-s >= 0:
                    delta += h_temp[i0] - h_temp[i1]
                if i1-s >= 0:
                    delta += h_temp[i1] - h_temp[i0]
            # Actually compute the actual change in overlap due to swapping indices
            overlap_change = [0]*len(h_temp)
            for s in range(1,len(h_temp)):
                if i0+s<len(h_temp): overlap_change[s] += int(h_temp[i1]) - int(h_temp[i0])
                if i1+s<len(h_temp): overlap_change[s] += int(h_temp[i0]) - int(h_temp[i1])
                if i0-s>=0: overlap_change[s] += int(h_temp[i0]) - int(h_temp[i1])
                if i1-s>=0: overlap_change[s] += int(h_temp[i1]) - int(h_temp[i0])
            
            # New overlaps
            new_min = min([overlaps[s] + overlap_change[s] for s in range(1,len(h_temp))])
            if new_min > t:
                # Accept this swap
                h_temp[i0], h_temp[i1] = h_temp[i1], h_temp[i0]
                t = new_min
                for s in range(1,len(h_temp)):
                    overlaps[s] += overlap_change[s] 
                if t > best_t:
                    best_t = t
                    best_h = h_temp.copy()
                best_improvement = 0
            else:
                best_improvement += 1
            if best_improvement > 100:
                break
        return best_h, best_t

    # If we don't have a good integer solution from MILP, try a heuristic approach
    if h_bin is None:
        # Do a heuristic construction to get initial binary vector
        h_bin = initial_h_values(n) if n <= len(initial_h_values(100)) else np.zeros(n, dtype=int)
        if len(h_bin) < n:
            raise ValueError("initial_h_values not appropriate for size n")
        h_bin = h_bin[:n]

    # locally improve
    overlaps = compute_overlaps(h_bin)
    t = min(overlaps.values())
    h_improved, t = local_improvement(h_bin, overlaps, t)
    h_final = h_improved
    
    # Evaluate final
    c5_bound = evaluate_erdos_solution(h_final.astype(float))
    
    # If time allows and C5 still too high, use SLSQP polish
    time_left = budget_s - (time.time() - start_time)
    if time_left > 30 and c5_bound > 0.3807:
        h_cont = h_final.astype(float) + (np.random.rand(n) - 0.5) * 0.02
        h_cont = np.clip(h_cont, 0.0, 1.0)
        # Ensure dx * sum == 1
        h_cont = h_cont * (1.0 / np.sum(h_cont) * (dx * n) )
        def obj(h):
            return evaluate_erdos_solution(h)
        def constraint_sum(h):
            return np.sum(h) * dx - 1 ## This should be zero
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1)] * n
        res = minimize(obj, h_cont, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            c5_final = obj(res.x)
            if c5_final < c5_bound:
                c5_bound = c5_final
                h_final = res.x.copy()
    
    # Final evaluation after up-sampling to keep output under 1000 points
    r = min(4, 800 // len(h_final))  # Prevent oversampling to > 800 points
    if r > 1:
        h_upsampled = np.repeat(h_final, r)
        c5_bound = evaluate_erdos_solution(h_upsampled)
        h_final = h_upsampled
    
    return h_final, c5_bound, len(h_final)
