import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def c5_of_h(h, dx):
    """Compute C5 for given h using full convolution."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def gradient_of_c5(h, dx):
    """Compute gradient for the active shift."""
    g = 1 - h
    corr = correlate(h, g, mode='full')
    k_star = np.argmax(corr) - len(h) + 1
    grad = np.zeros_like(h)
    n = len(h)
    for i in range(n):
        j = i + k_star
        k = i - k_star
        if 0 <= j < n:
            grad[i] -= dx * h[j]
        if 0 <= k < n:
            grad[i] += dx * g[k]
    return grad

def project_to_constraints(h, n_points, target_sum):
    """Project h onto sum constraint and box constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Distribute proportionally
    weights = np.ones_like(h)
    weights[np.where(h <= 0)] = 0
    weights[np.where(h >= 1)] = 0
    if np.sum(weights) > 0:
        h += diff * weights / np.sum(weights)
        h = np.clip(h, 0, 1)
    return h
