import numpy as np
import time
from scipy import signal
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    def correlation_vector(b):
        return np.correlate(b, b, mode='full')[1:]
    
    def update_correlation(R, i, j, b):
        n = len(b)
        old_min_k = np.argmin(R)
        old_min_val = R[old_min_k]
        
        for k in range(n-1):
            if i-k >= 0 and b[i-k] == 1:
                R[k] -= 1
            if i+k < n and b[i+k] == 1:
                R[k] -= 1
            if j-k >= 0 and b[j-k] == 1:
                R[k] += 1
            if j+k < n and b[j+k] == 1:
                R[k] += 1
                
        return R, old_min_k, old_min_val
    
    def hill_climb(b, R, max_swaps=200):
        n = len(b)
        best_b = b.copy()
        best_R = R.copy()
        best_t = np.min(R)
        
        for _ in range(max_swaps):
            min_k = np.argmin(best_R)
            swaps = []
            for i in range(n):
                if b[i] == 1:
                    for j in range(n):
                        if b[j] == 0:
                            # Estimate the change in R_min_k if we swap i,j
                            new_R = best_R.copy()
                            for k in range(n-1):
                                if i-k >= 0 and best_b[i-k] == 1:
                                    new_R[k] -= 1
                                if i+k < n and best_b[i+k] == 1:
                                    new_R[k] -= 1
                                if j-k >= 0 and best_b[j-k] == 1:
                                    new_R[k] += 1
                                if j+k < n and best_b[j+k] == 1:
                                    new_R[k] += 1
                            new_min_k = np.argmin(new_R)
                            delta = new_R[min_k] - best_R[min_k]
                            swaps.append((delta, i, j, new_R, new_min_k))
            if not swaps:
                break
                
            max_delta, best_i, best_j, new_R, new_min_k = max(swaps, key=lambda x: x[0])
            if max_delta > 0:
                # Apply swap
                best_b[best_i] = 0
                best_b[best_j] = 1
                best_R = new_R
                if new_R[new_min_k] > best_t:
                    best_t = new_R[new_min_k]
            else:
                break
        
        return best_b, best_R, best_t
    
    def tournament_selection(population, fitnesses, tournament_size=3):
        idxs = np.random.choice(len(population), tournament_size)
        best = np.argmax([fitnesses[i] for i in idxs])
        return population[idxs[best]]
    
    def crossover(parent1, parent2, n):
        # Union-based crossover with weight preservation
        support1 = set(np.where(parent1 == 1)[0])
        support2 = set(np.where(parent2 == 1)[0])
        union = support1.union(support2)
        
        if len(union) > n//2:
            # Remove extra elements
            excess = list(union - support1 - support2)
            if len(excess) >= len(union) - n//2:
                remove_count = len(union) - n//2
                to_remove = random.sample(excess, remove_count)
                union = union - set(to_remove)
            else:
                to_remove = list(union)[:(len(union) - n//2)]
                union = union - set(to_remove)
        elif len(union) < n//2:
            # Add missing elements
            complement = set(range(n)) - union
            add_count = n//2 - len(union)
            to_add = random.sample(list(complement), add_count)
            union = union.union(to_add)
            
        child = np.zeros(n, dtype=np.uint8)
        for i in union:
            child[i] = 1
        return child
    
    def mutate(individual, n, mutations=1):
        new_ind = individual.copy()
        ones = np.where(new_ind == 1)[0]
        zeros = np.where(new_ind == 0)[0]
        for _ in range(mutations):
            i = random.choice(ones)
            j = random.choice(zeros)
            new_ind[i] = 0
            new_ind[j] = 1
        return new_ind
    
    def upsample_and_correct(initial_seq, target_len):
        upsample_factor = target_len // len(initial_seq)
        upsampled = np.repeat(initial_seq, upsample_factor)
        if len(upsampled) > target_len:
            upsampled = upsampled[:target_len]
        elif len(upsampled) < target_len:
            upsampled = np.pad(upsampled, (0, target_len - len(upsampled)), 'constant')
        
        # Adjust to correct weight
        current_weight = np.sum(upsampled)
        diff = current_weight - target_len // 2
        
        if diff > 0:
            idxs = np.where(upsampled == 1)[0]
            flips = random.sample(list(idxs), diff)
            for i in flips:
                upsampled[i] = 0
        elif diff < 0:
            idxs = np.where(upsampled == 0)[0]
            flips = random.sample(list(idxs), -diff)
            for i in flips:
                upsampled[i] = 1
                
        return upsampled
    
    def binary_to_float(h):
        h = np.array(h, dtype=float)
        return h / np.sum(h) * 2  # normalize to 1 integral
    
    stages = [30, 36, 42, 48, 60]
    t0 = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Start from initial_h_values if available
    try:
        initial_h = initial_h_values
        initial_h = np.array(initial_h, dtype=float)
        initial_h /= np.sum(initial_h) * 2.0  # scale to integral = 1
        # project to binary
        initial_h_binary = np.zeros(len(initial_h), dtype=np.uint8)
        idxs = np.argsort(initial_h)[::-1]
        initial_h_binary[idxs[:len(initial_h)//2]] = 1
    except:
        initial_h_binary = None
    
    for stage_n in stages:
        if time.time() - t0 > budget_s - 10:
            break
            
        # budget calculation
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 5  # 5s safety margin
        
        if stage_n == 30:
            gens = 100
            pop_size = 30
        elif stage_n == 36:
            gens = 100
            pop_size = 50
        elif stage_n == 42:
            gens = 120
            pop_size = 70
        elif stage_n == 48:
            gens = 120
            pop_size = 80
        else:  # n=60
            gens = 150
            pop_size = 100

        max_time_per_gen = 1  # approximate time per generation in seconds
        estimated_gens = min(gens, int(remaining / max_time_per_gen))
        gens = estimated_gens
        if gens <= 0:
            break

        # Create initial population
        population = []
        fitnesses = []

        # seed with upsampled version or initial_h
        if initial_h_binary is not None and stage_n > 30:
            seed = upsample_and_correct(initial_h_binary, stage_n)
        elif stage_n == 30:
            # random binary
            seed = np.zeros(stage_n, dtype=np.uint8)
            ones = np.random.choice(stage_n, stage_n//2, replace=False)
            seed[ones] = 1
        else:
            seed = upsample_and_correct(population[0] if population else np.zeros(30, dtype=np.uint8), stage_n)
        
        population.append(seed)
        R_of_seed = correlation_vector(seed)
        fitnesses.append(np.min(R_of_seed))
        
        # Add random individuals
        for _ in range(pop_size - 1):
            individual = np.zeros(stage_n, dtype=np.uint8)
            ones = np.random.choice(stage_n, stage_n//2, replace=False)
            individual[ones] = 1
            population.append(individual)
            R = correlation_vector(individual)
            fitnesses.append(np.min(R))

        global_best_idx = np.argmax(fitnesses)
        global_best = population[global_best_idx]
        global_best_R = correlation_vector(global_best)
        global_best_t = np.min(global_best_R)
        
        # Evolutionary loop
        for gen in range(gens):
            if time.time() - t0 > budget_s - 2:
                break
            new_population = []
            new_fitnesses = []

            # Selection, crossover, mutation
            for _ in range(pop_size):
                parent1 = tournament_selection(population, fitnesses)
                if random.random() < 0.8:
                    parent2 = tournament_selection(population, fitnesses)
                    child = crossover(parent1, parent2, stage_n)
                else:
                    child = parent1.copy()
                child = mutate(child, stage_n, 1)  # one mutation
                # local improvement
                child_R = correlation_vector(child)
                improved_child, improved_R, improved_t = hill_climb(child, child_R)
                
                new_population.append(improved_child)
                new_fitnesses.append(improved_t)

            population = new_population
            fitnesses = new_fitnesses

            # update global best
            new_best_idx = np.argmax(fitnesses)
            if fitnesses[new_best_idx] > global_best_t:
                global_best = population[new_best_idx]
                global_best_R = correlation_vector(global_best)
                global_best_t = fitnesses[new_best_idx]

        # Final polishing using exhaustive local search
        polished_h = global_best.copy()
        polished_R = correlation_vector(polished_h)
        polished_t = np.min(polished_R)

        # Exhaustive local search
        max_iter = 1000
        for _ in range(max_iter):
            min_k = np.argmin(polished_R)
            best_delta = 0
            best_i = -1
            best_j = -1
            
            # Try all possible swaps
            ones = np.where(polished_h == 1)[0]
            zeros = np.where(polished_h == 0)[0]
            
            for i in ones:
                for j in zeros:
                    # compute delta in R_min_k if swap i,j
                    temp_R = polished_R.copy()
                    # Apply swap by updating each position
                    for k in range(stage_n - 1):
                        if i-k >= 0 and polished_h[i-k] == 1:
                            temp_R[k] -= 1
                        if i+k < stage_n and polished_h[i+k] == 1:
                            temp_R[k] -= 1
                        if j-k >= 0 and polished_h[j-k] == 1:
                            temp_R[k] += 1
                        if j+k < stage_n and polished_h[j+k] == 1:
                            temp_R[k] += 1
                    new_min_k = np.argmin(temp_R)
                    delta = temp_R[min_k] - polished_R[min_k]
                    
                    if delta > best_delta:
                        best_delta = delta
                        best_i = i
                        best_j = j
                        break
                if best_i >= 0:
                    break
            
            if best_i >= 0 and best_delta > 0:
                polished_h[best_i] = 0
                polished_h[best_j] = 1
                polished_R = temp_R
                polished_t = np.min(polished_R)
            else:
                break

        dx = 2.0 / stage_n
        c5 = 1 - dx * polished_t

        if c5 < best_c5:
            best_c5 = c5
            best_h = polished_h
            best_n = stage_n

        # Check if we are approaching timeout
        if time.time() - t0 > budget_s - 2:
            break

    # Final evaluation with original method
    if best_h is not None:
        dx = 2.0 / best_n
        R = np.correlate(best_h, 1 - best_h, mode='full')
        c5_final = np.max(R) * dx
        return (binary_to_float(best_h), c5_final, best_n)
    else:
        # fallback: return the default from initial if it makes sense
        try:
            h = initial_h_values
            h = np.array(h, dtype=float)
            h /= np.sum(h) * 2.0  # normalize integral to 1
            dx = 2.0 / len(h)
            R = np.correlate(h, 1-h, mode='full')
            c5 = np.max(R) * dx
            return (h, c5, len(h))
        except:
            # simple binary construction
            h = np.zeros(60, dtype=float)
            h[:30] = 1.0
            h = h / np.sum(h) * 2.0
            dx = 2.0 / 60
            R = np.correlate(h, 1-h, mode='full')
            c5 = np.max(R) * dx
            return (h, c5, 60)
