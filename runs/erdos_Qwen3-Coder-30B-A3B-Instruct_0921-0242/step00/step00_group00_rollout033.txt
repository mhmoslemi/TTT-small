import numpy as np
import time
import cma
from scipy.linalg import block_diag
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Problem parameters
    n_points = 256
    dx = 2.0 / n_points
    M = 8  # number of frequencies (total parameters = 2*M)
    n_freq = 2 * M
    
    # Precompute Fourier basis matrix
    i_vals = np.arange(n_points)
    freqs = np.arange(1, M + 1)
    fourier_basis = np.empty((n_points, n_freq))
    for j, m in enumerate(freqs):
        fourier_basis[:, 2*j] = np.cos(2 * np.pi * m * i_vals / n_points)
        fourier_basis[:, 2*j+1] = np.sin(2 * np.pi * m * i_vals / n_points)
    Phi = fourier_basis
    
    # Precompute projection matrix
    PhiTPhi = Phi.T @ Phi
    try:
        # Use SVD for numerical stability
        U, s, Vt = np.linalg.svd(PhiTPhi)
        s_inv = np.where(s > 1e-12, 1.0 / s, 0.0)
        PhiPinv = Vt.T @ np.diag(s_inv) @ U.T
    except:
        # Fallback to pseudo-inverse if SVD fails
        PhiPinv = np.linalg.pinv(PhiTPhi)
        
    def project_to_feasible(c):
        # c: array of size n_freq
        # h = Phi @ c + 0.5
        h = Phi @ c + 0.5
        h_clipped = np.clip(h, 0, 1)
        correction = h_clipped - h
        delta_c = PhiPinv @ correction
        c_proj = c + delta_c
        return c_proj
    
    def evaluate_c5(c):
        # c: array of size n_freq
        h = Phi @ c + 0.5
        # Ensure h is within bounds exactly
        h = np.clip(h, 0, 1)
        # Compute overlap integral
        # Note: in our parametrization we have same sign, so we can use reverse
        # correlation with complementary function
        h_rev = 1.0 - h
        corr = np.correlate(h, h_rev, mode='full')
        corr = corr * dx
        max_overlap = corr.max()
        return max_overlap
    
    def evolve_and_optimize(budget, n_points):
        best_c = None
        best_c5 = float('inf')
        start_time = time.time()
        restarts = 10 if budget > 500 else 5
        if n_points == 512:
            restarts = 8
            
        # Periodically refesh population if needed (from CMA-ES)
        for restart in range(restarts):
            if time.time() - start_time > budget - 0.5:
                break
            np.random.seed(int(seed * 100 + restart))
            
            # Set up solver
            es = cma.CMAEvolutionStrategy(n_freq * [0.0], 0.2, 
                                          {'popsize': 4 + int(3 * np.log(n_freq)), 
                                           'verbose': -1})
            popsize = es.popsize
            
            # Store best individual
            restart_best_c = None
            restart_best_c5 = float('inf')
            
            # Population loop
            while not es.stop() and (time.time() - start_time < budget - 0.5):
                # Generate candidate solutions
                solutions = es.ask()
                cands = []
                for sol in solutions:
                    c = project_to_feasible(np.array(sol))
                    cands.append(c)
                # Evaluate candidates
                values = []
                for c in cands:
                    try:
                        val = evaluate_c5(c)
                        values.append(val)
                    except:
                        values.append(float('inf'))
                # Update population
                es.tell(solutions, values)
                
                # Track best individual
                min_idx = np.argmin(values)
                val = values[min_idx]
                c = cands[min_idx]
                if val < restart_best_c5:
                    restart_best_c5 = val
                    restart_best_c = c.copy()
                
            # Track the overall best
            if restart_best_c5 < best_c5:
                best_c5 = restart_best_c5
                best_c = restart_best_c.copy()
                
        return best_c, best_c5
    
    # Stage 1: 256 points
    c_best, c5_best = evolve_and_optimize(0.4 * budget_s, n_points)
    
    # If not good enough, grow resolution to 512
    if c5_best > 0.3807 and time.time() - start_time < budget_s - 0.5:
        n_points = 512
        dx = 2.0 / n_points
        # New basis matrix (keep same frequency components)
        i_vals = np.arange(n_points)
        freqs = np.arange(1, M + 1)
        fourier_basis = np.empty((n_points, n_freq))
        for j, m in enumerate(freqs):
            fourier_basis[:, 2*j] = np.cos(2 * np.pi * m * i_vals / n_points)
            fourier_basis[:, 2*j+1] = np.sin(2 * np.pi * m * i_vals / n_points)
        Phi = fourier_basis
        # New projection matrix (reuse PhiPinv of small size since M stays same)
        PhiTPhi = Phi.T @ Phi
        try:
            U, s, Vt = np.linalg.svd(PhiTPhi)
            s_inv = np.where(s > 1e-12, 1.0 / s, 0.0)
            PhiPinv = Vt.T @ np.diag(s_inv) @ U.T
        except:
            PhiPinv = np.linalg.pinv(PhiTPhi)
        
        # Make sure to interpolate the frequency domain of previous solution to new space
        # Since we used only M frequencies, increasing n just means extend the same frequencies
        # so i.e., just keep the same coefficients in the higher-resolution matrix
        # We keep c_best (which is the same as previous since we work in low-dim)
        
        # Refine using more restarts on larger grid 
        c_best, c5_best = evolve_and_optimize(0.6 * budget_s, n_points)
        
    # Final refinement using projected gradient descent
    # Compute gradient
    h_final = Phi @ c_best + 0.5
    h_final = np.clip(h_final, 0, 1)

    # Compute max overlap index
    h_rev = 1.0 - h_final
    corr = np.correlate(h_final, h_rev, mode='full')
    max_idx = np.argmax(corr)
    k_star = max_idx - len(h_final) + 1
    
    # Compute gradient w.r.t h_i (at the maximal k)
    # ∂C5/∂h_i = (1 - h_{i+k*}) + h_{i-k*}
    # We compute this by convolution (using FFT)
    grad = np.zeros_like(h_final)
    if k_star >= 0:
        grad[:-k_star] += h_rev[k_star:]
        grad[k_star:] += h_rev[:-k_star]
    else:
        grad[-k_star:] += h_rev[:k_star]
        grad[:k_star] += h_rev[-k_star:]

    max_overlap = corr.max()

    # Gradient descent for more precise optimization
    alpha0 = 0.01
    num_iter = 200
    for i in range(num_iter):
        if time.time() - start_time > budget_s - 0.5:
            break
        alpha = alpha0 / (1 + i/10.)  # diminishing stepsize
        # Get gradient in coefficient space
        grad_coef = Phi.T @ grad
        new_c = c_best - alpha * grad_coef
        new_c = project_to_feasible(new_c)
        
        # Re-evaluate the objective
        new_h = Phi @ new_c + 0.5
        new_h = np.clip(new_h, 0, 1)
        h_rev = 1.0 - new_h
        new_corr = np.correlate(new_h, h_rev, mode='full')
        new_max_overlap = new_corr.max()
        
        if new_max_overlap < max_overlap:
            max_overlap = new_max_overlap
            c_best = new_c.copy()
            h_final = new_h.copy()
        else:
            # No improvement, break
            break
    
    # Final reconstruction of h
    h_final = Phi @ c_best + 0.5
    h_final = np.clip(h_final, 0, 1)
    
    # Fix the integral constraint exactly by correction if needed
    integral = np.sum(h_final) * dx
    diff = 1.0 - integral
    if abs(diff) > 1e-12:
        # Distribute correction uniformly
        correction = diff / n_points
        h_final += correction
        h_final = np.clip(h_final, 0, 1)
    
    # Final evaluation of C5
    h_rev_final = 1.0 - h_final
    final_corr = np.correlate(h_final, h_rev_final, mode='full') * dx
    final_c5 = final_corr.max()
    
    return (h_final.tolist(), float(final_c5), n_points)
