def c5_of_h(h):
    """Compute C5 for a given h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    conv = correlate(h, g, mode='full')

    # The shifts range from -len(h)+1 to len(h)-1
    # We want to transform to actual shift values k in [0, 2]
    # shift indices range from 0 to 2*len(h)-2
    # shift positions = (idx - (len(h)-1)) * dx
    # For k in [0,2], we consider shifts from -len(h)+1 to len(h)-1
    # k = (idx - (len(h)-1)) * dx
    # idx ranges from 0 to 2*len(h)-2
    # So we extract the range where shifts are in interval [0,2]
    # When shift k = 0, idx = len(h)-1
    # When shift k = 2, idx = len(h)-1 + 2/dx = len(h)-1 + 2*(len(h)/2) = len(h)-1 + len(h) = 2*len(h)-1
    # Cool, we want indices from ((len(h)-1) - (2/dx)) to len(h)-1
    # We have:
    # shift = (idx - (len(h)-1)) * dx
    # shift >= 0 => idx >= (len(h)-1)
    # shift <= 2 => idx <= (len(h)-1) + (2/dx) = (len(h)-1) + len(h) = 2*len(h)-1
    # So we want indices from len(h)-1 to 2*len(h)-2
    # Let's check for k in [0, 2], idx ranges from len(h)-1 to 2*len(h)-2
    # But actually we want to check the full range also for direct comparison with neighbor shift values
    # Let's just compute the max over ALL shifts and it will be fine
    # The key is k range covers all integer shift values
    # actually let's align the shift to the center to match the canonical k range [0,2]
    # k = idx - (len(h)-1) * dx
    # for maximum in k=[0,2], we map idx from len(h)-1 to len(h)-1 + 2/dx
    # but the meaningful range for k in [0,2] is the same indices we consider:
    # when k=0, idx = len(h)-1, when k=2, idx = len(h)-1+2/dx  
    # but since 2/dx = 2 * len(h) / 2 = len(h), so we have idx from len(h)-1 to 2*len(h)-1
    # 2*len(h) - 1 - (len(h) - 1) + 1 = len(h) positions, for k = 0 to 2 in dx steps
    # but mathematically we just want all diagonal contributions and take max:
    # then just take max of all conv values
    return np.max(conv) * dx

    integral = np.sum(h) * dx
    h = h * (1.0 / integral)  # Keep the relative shape consistent
    return c5_of_h(h), integral
