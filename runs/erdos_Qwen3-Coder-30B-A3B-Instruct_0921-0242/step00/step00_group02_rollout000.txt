import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h):
    """Compute C₅ = max_k ∫ h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # correlate(h, 1-h, mode='full') evaluates sum_i h[i] * (1-h[i+k]) for all shifts k
    # The max of these values, times dx, is our C₅
    conv = correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def project_feasible(x, target):
    """Project x onto the capped-simplex: 0 ≤ h ≤ 1, sum(h) = target"""
    h = np.clip(x, 0, 1)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Iteratively adjust to match target sum
    # Water-filling approach
    diff = target - s
    while abs(diff) > 1e-10:
        # Determine which entries can accept more mass (are < 1)
        free = h < 1
        if not np.any(free):
            break
        free_count = np.sum(free)
        if free_count == 0:
            break
        add = min(diff / free_count, 1.0 - np.min(h[free]))
        if add <= 0:
            break
        h[free] += add
        diff -= add * free_count
    return np.clip(h, 0, 1)

def active_shifts(h, eps=1e-9):
    """Return all shifts where overlap is within eps of max overlap."""
    dx = 2.0 / len(h)
    conv = correlate(h, 1 - h, mode='full')
    max_val = np.max(conv)
    k_indices = np.where(np.abs(conv - max_val) < eps * max_val)[0]
    # Convert to actual shift values
    n = len(h)
    active = []
    for k_idx in k_indices:
        shift = k_idx - (n - 1)
        active.append(shift)
    return active

def gradient(h, ks):

    # So for each k:
    g = np.zeros(n)

        if k > 0:
            g += (1.0 - h_shift[k:])  # i+k = j, so i = j-k

            g += (1.0 - h_shift[:k])
        else:
            g += 0
    # Better implementation:
    g = np.zeros(n)
    # We are computing sum over k of gradient of sum_i h[i] * (1 - h[i+k])

        if k >= 0:
            # Consider h[i] * (1 - h[i+k]) for i=0..n-1-k

                g[i] += 1 - h_shift[i + k]
        else:
            # For i = -k..n-1, we add h[i] * (1 - h[i+k]) = h[i] * (1 - h[i+k])
            # values with offset i+k from i+k >=0 to i+k <n
            for i in range(-k, n):
                g[i] += 1 - h_shift[i + k]
    # But am avoiding manual looping for performance and clarity.
    # Simplified version, corrected.
    g = np.zeros(n)
    for k in ks:
        # Reflect $h[i] \cdot (1 - h[i+k])$ terms
        k_idx = k + n - 1  # the index in full convolution that corresponds to shift k
        conv_vals = correlate(h, 1 - h, mode='full')
        conv_g = correlate(np.ones_like(h), 1 - h, mode='full') # Should compute partial gradient
        # Does not quite work due to the nature of these.
        # Better:
        k_idx = k + (n - 1)  # shift k expands to [0..n-1-k, k=0: start from 0, k=1: start from 1]
        # Let's compute explicitly:
        tmp_g = np.zeros(n)
        if k >= 0:
            # h[i] multiplies (1 - h[i+k]), i ranges from 0 to n-1-k

                tmp_g[i] += 1 - h[i + k]
        else:
            # h[i] multiplies (1 - h[i+k]), i ranges from -k to n-1
            for i in range(-k, n):
                tmp_g[i] += 1 - h[i + k]
        g += tmp_g
    return -g  # We are minimizing the negative of the largest correlation

def line_search(h, g, t_current, budget):
    """Try several step-sizes and return the best improvement."""
    step_sizes = [0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1]
    best_h = h.copy()
    best_val = t_current
    for step in step_sizes:
        if step == 0:
            new_h = h.copy()
        else:
            new_h = project_feasible(h - step * g, 1.0)
        val = evaluate(new_h)
        if val < best_val:
            best_h = new_h
            best_val = val
    return best_h, best_val

def local_optimize(h0, max_iter, t_end):
    """Perform projected gradient descent with line-search."""
    current_h = h0.copy()
    best_h = current_h.copy()
    best_val = evaluate(current_h)
    dx = 2.0 / len(h0)
    iter_count = 0
    while time.time() < t_end and iter_count < max_iter:
        current_val = evaluate(current_h)
        k_active = active_shifts(current_h)
        if len(k_active) == 0:
            k_active = [0]
        g = gradient(current_h, k_active)
        # https://en.wikipedia.org/wiki/Subgradient_method
        # Take a line search step
        new_h, new_val = line_search(current_h, g, current_val, t_end)
        # Update
        current_h = new_h
        if new_val < best_val:
            best_h = new_h.copy()
            best_val = new_val
        iter_count += 1
    # Return the best solution seen
    return best_h, best_val

def anneal(h0, t_end):
    """Simple simulated annealing to escape local optima."""
    dx = 2.0 / len(h0)
    current_h = h0.copy()
    current_val = evaluate(current_h)
    best_h = current_h.copy()
    best_val = current_val
    T = 0.01
    T_min = 1e-5
    alpha = 0.995
    step_limit = 5000
    step_count = 0
    while time.time() < t_end and step_count < step_limit:
        if T < T_min:
            break

        # Propose a new point by moving mass randomly
        idx1, idx2 = np.random.randint(0, len(h0), size=2)
        delta = np.random.uniform(0, min(current_h[idx1], 1 - current_h[idx2]))
        if delta == 0:
            continue
        old_sum = np.sum(current_h)
        temp_h = current_h.copy()
        temp_h[idx1] -= delta
        temp_h[idx2] += delta

        # Project into feasible set
        temp_h = project_feasible(temp_h, old_sum)

        new_val = evaluate(temp_h)
        if new_val < best_val:
            best_val = new_val
            best_h = temp_h.copy()
        else:
            # Accept with probability e^(-Δ/T)
            if delta_val := (new_val - current_val):
                prob = np.exp(-delta_val / T)
                if np.random.rand() < prob:
                    current_h = temp_h
                    current_val = new_val

        T *= alpha
        step_count += 1

    return best_h, best_val

def random_feasible(n):
    """Sample uniformly random h with mean 1."""
    h = np.random.rand(n)
    h = h / np.sum(h) * 1.0  # Scale to integral = 1
    return project_feasible(h, 1.0)
