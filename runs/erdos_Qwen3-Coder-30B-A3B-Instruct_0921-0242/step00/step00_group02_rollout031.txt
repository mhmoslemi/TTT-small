    sdpa_eps = 1e-4
    sdpa_max_iter = 1000
    n_levels = [30, 38, 42]
    timeallocs = [80, 120, 150]
    total_time = 0

            h = cp.Variable(n)
            X = cp.Variable((n, n), symmetric=True)
            s = cp.Variable()

            # constraints
            constraints = [
                cp.diag(X) == h,
                X >> 0,
                h >= 0,
                h <= 1,
                cp.sum(h) == n / 2
            ]

            # Add moment constraints (shifted correlations)
            # For factors k = 1 to n-1:
            moments = []
            for k in range(1, n):
                # Compute sum over diagonal of X shifted by k
                expr = 0
                for i in range(n - k):
                    expr += X[i, i+k]
                moments.append(expr)
                constraints.append(expr >= s)

            # Objective: maximize s
            prob = cp.Problem(cp.Maximize(s), constraints)
            prob.solve(solver=cp.SCS, eps=sdpa_eps, max_iters=sdpa_max_iter)

            if prob.status != cp.OPTIMAL:
                continue # Skip if infeasible or unbounded

            h_opt = h.value
            X_opt = X.value

            # Now we want to perform randomized hyperplane rounding
            # Factorize: X_opt = V @ V.T (eigendecompose)
            try:
                eigenvals, eigenvecs = eigh(X_opt)
                # Ensure all eigenvalues are non-negative (numerical safety)
                eigenvals = np.maximum(eigenvals, 0)
                # Take all components with non-trivial eigenvalues
                rank = np.sum(eigenvals > 1e-8)
                if rank == 0:
                    continue  # This shouldn't happen but guard against numerical issues
                V = eigenvecs[:, :rank] * np.sqrt(eigenvals[:rank])

                # Sample random vectors and round
                best_rounded = None
                best_rounded_c5 = float('inf')

                # Sample 300 roudings
                for _ in range(300):
                    g = np.random.randn(rank)
                    z = V @ g  # raw scores

                    # Threshold at 1 - n/2 / n to get correct mass
                    quantile = np.quantile(z, 1.0 - n/2 / n)
                    b = (z >= quantile).astype(int)

                    # Force exactly n/2 ones
                    diff = int(n/2 - np.sum(b))
                    if diff != 0:
                        # Flip entries near threshold
                        indices = np.argsort(np.abs(z - quantile))
                        flip_indices = indices if diff > 0 else indices[::-1]
                        candidates = flip_indices[:abs(diff)]
                        b[candidates] = 1 - b[candidates]

                    # Only proceed if mass is correct
                    if np.sum(b) != n/2:
                        continue

                    # Compute actual C5
                    dx = 2.0 / n
                    correlation = []
                    for k in range(1, n):
                        corr = 0.
                        for i in range(n - k):
                            corr += b[i] * b[i+k]
                        correlation.append(corr)
                    c5_conv = 1 - dx * min(correlation) if correlation else 0

                    if c5_conv < best_rounded_c5:
                        best_rounded_c5 = c5_conv
                        best_rounded = b.copy()

                if best_rounded is not None and best_rounded_c5 < best_c5:
                    best_c5 = best_rounded_c5
                    best_h = best_rounded.copy()
                    best_n = n

            except Exception as e:
                pass  # Skip bad factorizations

        except Exception as e:
            pass

        total_time += time.time() - stage_start

        if best_c5 <= 0.3807:
            break  # Early stop if target reached

        if best_c5 > 0.3807:
            # Optional: Ascent on fractional solution before swap polishing
            # We refine h_opt again with projected gradient ascent,
            # as in standard approaches but starting from h_opt.
            h_refined = h_opt.copy()
            dx = 2.0 / n

            def correlate_at_k(k):
                corr = 0
                for i in range(n - k):
                    corr += h_refined[i] * h_refined[i+k]
                return corr

            def worst_correlation(h_vec):
                corrs = [correlate_at_k(k) for k in range(1, n)]
                return min(corrs) if corrs else 0

            def grad_k(k):
                if k == 0:
                    return np.zeros(n)
                grad = np.zeros(n)
                for i in range(min(n-k, n)):
                    grad[i] += h_refined[i+k]  # d/dh_i R_k(h)
                    if i+k < n:
                        grad[i+k] += h_refined[i]
                return grad

            def project_to_box_and_mass(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                total = np.sum(h_vec)
                target_sum = n/2
                if abs(total - target_sum) < 1e-10:
                    return h_vec
                diff = total - target_sum
                # Adjust proportionally to decrease sum slightly
                scale = 1 - diff/(2*total) if abs(total) > 1e-10 else 1
                h_vec *= scale
                h_vec = np.clip(h_vec, 0, 1)
                return h_vec

            def prob_of_correlation():
                return 1 - dx * worst_correlation(h_refined)

            patience = 30

                    corr = correlate_at_k(k)
                    if corr < best_k_corr:
                        best_k_corr = corr
                        k_star = k
                if best_k_corr == 0:
                    break
                g = grad_k(k_star)

                    patience -= 1
                else:
                    patience = 30
                if patience <= 0:

            h_polished = h_refined.copy()

            def get_correlation_vector(h_vec):
                corrs = []
                for k in range(1, n):
                    corr_val = 0.
                    for i in range(n - k):
                        corr_val += h_vec[i] * h_vec[i+k]
                    corrs.append(corr_val)
                return corrs

                corrs = get_correlation_vector(h_vec)
                if len(corrs) == 0:
                    return 0
                dx = 2.0 / n
                return 1 - dx * min(corrs)

            current_c5 = compute_c5(h_polished)
            initial_c5 = current_c5
            max_swaps = 1000

                # Limit to manageable search space
                favor = np.zeros(n)
                for i in range(n):
                    if h_polished[i] == 1:
                        for j in range(n):
                            if j != i and h_polished[j] == 0:
                                # test making swap h[i]=0; h[j]=1
                                # directly increment the correlations involving i and j for k_star
                                score_change = 0
                                # check how change affects k_star shift
                                # [i,i+k],[i+k,i+2k],... and [j,j+k],[j+k,...]
                                # contributions for k_star
                                for d in range(n):
                                    if i == d or d == (i+k_star):
                                        score_change += 1
                                    elif j == d or d == (j+k_star):
                                        score_change -= 1
                                    elif d == i and (i+k_star) < len(h_polished):
                                        score_change += (h_polished[i+k_star] - h_polished[i+k_star])
                                    elif d == j and (j+k_star) < len(h_polished):
                                        score_change -= (h_polished[j+k_star] - h_polished[j+k_star])
                                # Simplified: just do direct check
                                gain = -1  # The initial k_star is penalized by -1 when i->0, j->1
                                if (i+k_star < n and h_polished[i+k_star] == 1) and (j+k_star < n and h_polished[j+k_star] == 0):
                                    gain += 1
                                if (j+k_star < n and h_polished[j+k_star] == 1) and (i+k_star < n and h_polished[i+k_star] == 0):
                                    gain += 1
                                favor[i] += gain

                # Pick best pair
                best_i = np.argmax(favor)
                best_gain = 0
                best_j = -1
                for j in range(n):
                    if h_polished[best_i] == 1 and h_polished[j] == 0:
                        gain = 0
                        # k_star: see how i and j change contribution
                        for d in range(n):
                            if d + k_star < n:
                                if (d == best_i or d+k_star == best_i):
                                    if h_polished[d] == 1:
                                        gain += 1
                                if (d == j or d+k_star == j):
                                    if h_polished[d] == 1:
                                        gain -= 1
                        # More detailed now
                        # under shift k_star:
                        # [best_i,...], [best_i+k_star,...]        : old 1,1
                        # [best_j,...], [best_j+k_star,...]        : old 0,0
                        # new changes:
                        old_contribution = 0
                        new_contribution = 0

                        if best_i + k_star < n:
                            old_contribution += h_polished[best_i] * h_polished[best_i+k_star]
                        if best_j + k_star < n:
                            old_contribution += h_polished[best_j] * h_polished[best_j+k_star]
                        new_contribution = (1 - h_polished[best_j]) * h_polished[best_i] + h_polished[best_i] * (0 if best_i + k_star >= n else h_polished[best_i+k_star])
                        gain = new_contribution - old_contribution
                        if gain > best_gain:
                            best_gain = gain
                            best_j = j

                # Apply swap if beneficial
                if best_gain > 0 and best_j >= 0:
                    h_polished[best_i], h_polished[best_j] = h_polished[best_j], h_polished[best_i]
                    current_c5 = compute_c5(h_polished)
                else:
                    break  # No more improvement

            # Check final result
            final_c5 = compute_c5(h_polished)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_polished.copy()
                best_n = n

        # Early terminate if target met
        if best_c5 <= 0.3807:
            break
