import numpy as np
import time
import cvxpy as cp
from scipy import fft

def project_simplex_box(v, total, lo=0., hi=1.):
    """Project vector v onto the simplex with box constraints [lo, hi] and sum = total."""
    v = np.asarray(v)
    n = len(v)
    # Sort values in descending order
    sorted_v = np.sort(v)[::-1]
    # Find the threshold
    cumsum = np.cumsum(sorted_v)
    temp = (sorted_v - (total - cumsum) / np.arange(1, n + 1))
    rho = np.argmax(temp > 0)
    theta = (total - cumsum[rho - 1]) / (rho + 1) if rho > 0 else (sorted_v[0] - total / n)
    # Compute projected vector
    projected = np.maximum(lo, np.minimum(hi, v - theta))
    # Ensure exact sum
    if abs(np.sum(projected) - total) > 1e-9:
        # If not summing to exact total, adjust via scaling
        projected = projected * total / np.sum(projected)
    return projected

def compute_c5(h, dx):
    """Compute the maximum overlap using FFT."""
    n = len(h)
    # Correlation of h with (1-h)
    H = fft.fft(h)
    OneMinusH = fft.fft(1 - h)
    corr = fft.ifft(H * OneMinusH.conj()).real
    # Compute maximum correlation (shifted)
    corr_full = np.concatenate([corr[n-1:], corr[:n-1]])
    c5 = np.max(corr_full) * dx
    return c5

def worst_lag(h):
    """Return the index and value of the worst (max) overlap."""
    n = len(h)
    # Using convolution (equivalent to correlation of h and (1-h))
    H = fft.fft(h)
    OneMinusH = fft.fft(1 - h)
    corr = fft.ifft(H * OneMinusH.conj()).real
    # Full correlation
    corr_full = np.concatenate([corr[n-1:], corr[:n-1]])
    idx = np.argmax(corr_full)
    val = corr_full[idx]
    return idx, val

def gradient_worst(h, k):
    """Gradient of the overlap for lag k with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # grad[i] = h[i+k] + h[i-k] (wrapping indices)
    for i in range(n):
        grad[i] = h[(i + k) % n] + h[(i - k) % n]
    return grad

def optimize(n, time_limit, seed=42):
    np.random.seed(seed)
    dx = 2.0 / n
    best_h = None
    best_c5 = float('inf')
    
    # Initialize using initial_h_values or random
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        if len(init_h) == n:
            h = init_h.copy()
        else:
            # Interpolate
            old_len = len(init_h)
            x_old = np.linspace(0, 2, old_len)
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_old, init_h)
    else:
        # Random initialization
        h = np.random.rand(n)
        h = h * (n / 2) / np.sum(h)  # Normalize sum to n/2

    h = np.clip(h, 0, 1)  # Clip to [0,1]
    h = project_simplex_box(h, n / 2, 0, 1)
    
    start_time = time.time()
    alpha = 0.01  # step size
    max_iter = int(time_limit * 1000)  # rough estimate
    iter_count = 0
    
    while time.time() - start_time < time_limit and iter_count < max_iter:
        # Find worst lag
        k_max, _ = worst_lag(h)
        
        # Compute gradient
        grad = gradient_worst(h, k_max)
        
        # Take a step
        h_new = h - alpha * grad
        
        # Project
        h_new = project_simplex_box(h_new, n / 2, 0, 1)
        h_new = np.clip(h_new, 0, 1)
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Update h
        h = h_new.copy()
        
        # Adaptive step size
        iter_count += 1
        
        if iter_count % 50 == 0:
            if abs(compute_c5(h, dx) - best_c5) < 1e-8:
                alpha = max(alpha * 0.95, 1e-6)
            else:
                alpha = min(alpha * 1.05, 0.1)
    
    if best_h is None:
        return h, compute_c5(h, dx), n
    return best_h, best_c5, n

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = None
    start_time = time.time()
    
    n_values = [150, 250, 350]
    
    for n in n_values:
        if time.time() - start_time > budget_s * 0.95:
            break
        # Allocate time per run based on how much is left
        time_per_run = min(5.0, budget_s * 0.3)
        h, c5, _ = optimize(n, time_per_run)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
    
    return list(best_h), best_c5, best_n
