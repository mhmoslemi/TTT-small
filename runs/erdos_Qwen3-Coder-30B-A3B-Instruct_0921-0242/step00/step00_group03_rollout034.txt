import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5_bound(h):
    """Evaluate the C5 bound using the provided correlation method."""
    dx = 2.0 / len(h)
    # Compute all correlations with mode='full'
    corr = correlate(h, 1 - h, mode='full')
    # The overlap integral is max over all k (shifts) of int h(x)(1-h(x+k))
    # This is equivalent to max over all k of: sum h(x)(1-h(x+k)) dx
    # Which via correlation becomes: max_k (sum h(x) - sum h(x)h(x+k)) dx
    # Where sum h(x) dx = 1, and sum h(x)h(x+k) dx is the correlation.
    # So C5 = max_k (1 - sum h(x)h(x+k) dx) = 1 - min_k sum h(x)h(x+k) dx
    min_corr = np.min(corr[len(h)-1:])
    c5 = 1 - min_corr * dx
    return c5

def solve_milp(n_points, seed=42, max_seconds=300):
    """Solve the MILP formulation of the binary version."""
    np.random.seed(seed)
    dx = 2.0 / n_points

    # Variables
    h = cp.Variable(n_points, boolean=True)
    t = cp.Variable(integer=True, lower_bound=0)

    # Constraints
    constraints = [cp.sum(h) == n_points // 2]

    # Product linearization
    y_vars = {}
    shift_constraints = []
    for s in range(1, n_points):
        for i in range(n_points - s):
            y = cp.Variable(boolean=True)
            y_vars[(i, s)] = y
            shift_constraints += [
                y <= h[i],
                y <= h[i + s],
                y >= h[i] + h[i + s] - 1
            ]
    
    # Overlap lower bound constraints
    for s in range(1, n_points):
        overlap_sum = 0
        for i in range(n_points - s):
            overlap_sum += y_vars[(i, s)]
        shift_constraints.append(t <= overlap_sum)
    
    constraints += shift_constraints

    # Objective
    objective = cp.Maximize(t)

    # Problem
    problem = cp.Problem(objective, constraints)

    # Solve
    problem.solve(solver=cp.CBC, max_seconds=max_seconds, verbose=False)

    if problem.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURACY]:
        h_val = np.round(h.value).astype(int)
        t_val = int(t.value)
        return h_val, t_val
    else:
        return None, None

def hill_climb(h, n_points, max_iter=10000):
    """Local binary hill-climbing optimization."""
    # Precompute overlaps for current h
    dx = 2.0 / n_points
    corr = correlate(h, h, mode='full')
    min_corr = np.min(corr[n_points-1:])
    t = int((1 - min_corr * dx) * n_points / 2)  # This is the inverted t value in paper
    
    original_h = h.copy()
    t = int(np.min([corr[n_points-1+i] for i in range(1, n_points)]) * dx * n_points / 2)
    
    count = 0
    for iter in range(max_iter):
        # Sample 10 random pairs (1,0) and (0,1) swaps
        positions = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(positions) < 1 or len(zeros) < 1:
            break
        pair_indices = np.random.choice(len(positions), size=min(10, len(positions)), replace=False)
        zero_indices = np.random.choice(len(zeros), size=min(10, len(zeros)), replace=False)
        for i in pair_indices:
            for j in zero_indices:
                s1, s0 = positions[i], zeros[j]
                # Create a new binary vector with these values swapped
                h_new = h.copy()
                h_new[s1] = 0
                h_new[s0] = 1
                corr_new = correlate(h_new, h_new, mode='full')
                new_min_corr = np.min(corr_new[n_points-1:])
                new_t_val = int((1 - new_min_corr * dx) * n_points / 2)
                if new_t_val > t:
                    h = h_new.copy()
                    t = new_t_val
                    count += 1
        if count == 0:
            break
        count = 0
    return h

def polish_continuous(h, n_points):
    """Fine-tune the solution using continuous optimization."""
    dx = 2.0 / n_points
    # Objective function
    def obj(h_vec):
        corr_vec = correlate(h_vec, 1 - h_vec, mode='full')
        min_corr = np.min(corr_vec[n_points-1:])
        return 1 - min_corr * dx

    def constraint_func(h_vec):
        return np.sum(h_vec) * dx - 1.0

    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = [{'type': 'eq', 'fun': constraint_func}]
    
    # Start with perturbed solution
    eps = 0.01
    h_pert = h + np.random.uniform(-eps, eps, n_points)
    h_pert = np.clip(h_pert, 0, 1)
    h_pert = h_pert / (np.sum(h_pert) * dx) * 1.0
    
    # Optimize
    res = minimize(obj, h_pert, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 200, 'ftol': 1e-6})
    
    if res.success:
        return res.x
    return h
