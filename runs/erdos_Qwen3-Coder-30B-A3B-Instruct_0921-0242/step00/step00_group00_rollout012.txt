import time
import numpy as np
from scipy.sparse import diags
import cvxpy as cp
from cvxpy import SCS
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # helper function for projection onto the capped simplex
    def project_simplex(v, s=0.5):
        v = np.asarray(v)
        n = len(v)
        u = np.sort(v)[::-1]
        css = np.cumsum(u) - s
        rho = np.argmax(u > css / (np.arange(n) + 1))
        theta = css[rho] / (rho + 1)
        return np.maximum(v - theta, 0)

    def compute_c5(h, dx):
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx

    def make_shift_matrices(n):
        matrices = {}
        for k in range(n):
            if k == 0:
                # Identity matrix
                matrices[k] = diags(np.ones(n), format='csc')
            else:
                # Diagonal matrix with 1s on the k-th diagonal
                matrices[k] = diags(np.ones(n - k), k, shape=(n, n), format='csc')
        return matrices

    # Check time limit
    def get_remaining_time():
        return budget_s - (time.time() - start_time)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0

        # Create shift matrices
        shift_matrices = make_shift_matrices(n)
        
        # set up SDP problem
        L = cp.Variable()
        X = cp.Variable((n, n), symmetric=True)
        h = cp.Variable(n)
        
        # Constraints
        constraints = []
        
        # matrix trace constraints
        for k in range(n):
            constraints.append(cp.trace(shift_matrices[k] @ X) >= L)
        
        # X is PSD
        constraints.append(X >> 0)
        
        # Diagonal elements
        for i in range(n):
            constraints.append(X[i, i] == h[i])
        
        # Bounding box
        constraints.append(0 <= h)
        constraints.append(h <= 1)
        
        # equality constraint on sum
        constraints.append(cp.sum(h) == n / 2)
        
        # bounding X[i,j] entries
        for i in range(n):
            for j in range(i+1, n):
                constraints.append(X[i, j] <= h[i])
                constraints.append(X[i, j] <= h[j])
                constraints.append(X[i, j] >= 0)
                
        # Objective: maximize L
        prob = cp.Problem(cp.Maximize(L), constraints)
        
        # solve with SCS
        try:
            prob.solve(solver=SCS, eps=5e-4, max_iters=4000)

            if prob.status in ["optimal", "optimal_inaccurate"]:

                # project to feasible simplex

                # compute C5
