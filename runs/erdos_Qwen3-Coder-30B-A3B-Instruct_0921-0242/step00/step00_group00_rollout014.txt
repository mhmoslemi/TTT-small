import numpy as np
import time
from scipy import fft

def project_simplex_box(v, total, lo=0., hi=1.):
    """Project vector v onto box-constrained simplex with sum = total"""
    v = np.asarray(v)
    n = len(v)
    
    # Sort in descending order
    sorted_idx = np.argsort(-v)
    sorted_v = v[sorted_idx]
    
    # Find threshold
    cumsum = np.cumsum(sorted_v)
    threshold = (total - cumsum + sorted_v) / np.arange(1, n + 1)
    theta = np.max(np.concatenate([[0.], threshold]))
    
    # Project and clip
    projected = np.clip(v - theta, lo, hi)
    
    # Normalize to sum = total if not already
    actual_sum = np.sum(projected)
    if abs(actual_sum - total) > 1e-9:
        # Add/subtract residual to all entries
        diff = total - actual_sum
        # Add to smallest values 
        sorted_proj_idx = np.argsort(projected)
        for i in range(n):
            idx = sorted_proj_idx[i]
            add_amount = min(-diff, projected[idx] - lo)
            if add_amount >= 0:
                projected[idx] -= add_amount
                diff += add_amount
                if abs(diff) < 1e-9:
                    break
        if abs(diff) > 1e-9:
            # Compensate by scaling (should be rare)
            projected *= total / actual_sum
            
    return projected

def compute_c5(h, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx using FFT"""
    # Full correlation: correlate h with 1-h
    h = np.asarray(h)
    corr = fft.ifft(fft.fft(h) * fft.fft(1 - h).conj()).real
    return np.max(corr) * dx

def worst_lag(h):
    """Return (index_of_worst_lag, value_of_worst_lag)"""
    h = np.asarray(h)
    corr = fft.ifft(fft.fft(h) * fft.fft(1 - h).conj()).real
    k = np.argmax(corr)
    return k, corr[k]

def gradient_worst(h, k):
    """Compute gradient of correlation at lag k w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = h[(i + k) % n] + h[(i - k) % n]
    return grad

def optimize(n, time_limit):
    """Optimize C5 for given n using projected gradient descent"""
    start_time = time.time()
    dx = 2.0 / n
    total = 1.0
    
    # Initialize
    if 'initial_h_values' in globals() and len(initial_h_values) >= n:
        h = np.array(initial_h_values[:n])
    else:
        # Generate random feasible point
        h = np.random.rand(n)
        h = h / np.sum(h) * total
        h = np.clip(h, 0, 1)
    
    # Project to sum = 1
    h = project_simplex_box(h, total, 0, 1)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    t = 0.01
    step_counter = 0
    while time.time() - start_time < time_limit:
        step_counter += 1
        
        # Find the worst correlation
        k_worst, _ = worst_lag(h)
        
        # Compute gradient
        grad = gradient_worst(h, k_worst)
        
        # Gradient step
        h_new = h - t * grad
        
        # Project back
        h_new = project_simplex_box(h_new, total, 0, 1)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Adapt step size
        if c5_new < compute_c5(h, dx):
            t = min(0.1, 1.1 * t)
        else:
            t *= 0.9
        
        h = h_new
        
        # Occasionally take a different random step to escape local optima
        if step_counter % 1000 == 0 and time.time() - start_time < time_limit:
            # Small randomization
            h = np.clip(h + np.random.randn(n) * 0.001, 0, 1)
            h = project_simplex_box(h, total, 0, 1)
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different resolutions
    n_values = [150, 250, 350]
    best_c5 = float('inf')
    final_h = None
    final_n = None
    
    for n in n_values:
        if time.time() - start_time >= budget_s:
            break
        
        time_left = budget_s - (time.time() - start_time)
        if time_left < 1:
            break
            
        try:
            h, c5 = optimize(n, time_left)
            if c5 < best_c5:
                best_c5 = c5
                final_h = h
                final_n = n
        except Exception as e:
            print(f"Error with n={n}: {e}")
            continue
    
    if final_h is None:
        # Fallback to using initial solution
        h = np.array(initial_h_values[:350]) if 'initial_h_values' in globals() else np.random.rand(350)
        h = (h / np.sum(h)) * 1.0
        h = np.clip(h, 0, 1)
        dx = 2.0 / len(h)
        final_h = h
        final_n = len(h)
        best_c5 = compute_c5(h, dx)
    
    # Final projection in case of slight violations
    final_h = project_simplex_box(final_h, 1.0, 0, 1)
    return (final_h.tolist(), best_c5, final_n)
