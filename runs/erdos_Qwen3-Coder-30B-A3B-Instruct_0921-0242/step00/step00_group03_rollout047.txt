import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize
from math import floor

def evaluate_erdos_solution(h, n_points):
    """Evaluate C5 using np.correlate."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values(n_points):
    """Initial construction from the problem content."""
    h = np.zeros(n_points)
    # Simple construction: evenly distribute 1s
    ones_count = n_points // 2
    for i in range(ones_count):
        h[i] = 1
    return h

def solve_milp_binary(n_points, solver_time_limit=300):
    """Solve the MILP for binary h to maximize minimal autocorrelation."""
    # Create variables
    h = cp.Variable(n_points, boolean=True)
    y = cp.Variable((n_points, n_points - 1), boolean=True)
    t = cp.Variable(integer=True, lower_bound=0)
    
    # Constraints
    constraints = [
        cp.sum(h) == n_points // 2,
    ]
    
    # Linearization of h_i * h_{i+s}
    for s in range(1, n_points):
        for i in range(n_points - s):
            # y[i, s-1] = h[i] * h[i+s]
            constraints.append(y[i, s-1] <= h[i])
            constraints.append(y[i, s-1] <= h[i+s])
            constraints.append(y[i, s-1] >= h[i] + h[i+s] - 1)
            
    # t <= sum_{i=0}^{n-1-s} y[i, s-1] for all s
    for s in range(1, n_points):
        constraints.append(t <= cp.sum(y[:, s-1][:n_points - s]))
        
    # Objective
    objective = cp.Maximize(t)
    
    problem = cp.Problem(objective, constraints)
    
    start_time = time.time()
    try:
        problem.solve(solver=cp.CBC, max_seconds=solver_time_limit, verbose=False)
        elapsed = time.time() - start_time
    except Exception as e:
        print(f"MILP solve error: {e}")
        elapsed = time.time() - start_time
    
    if problem.status == cp.OPTIMAL or problem.status == cp.OPTIMAL_INACCURATE:
        h_sol = np.round(h.value).astype(int)
        t_sol = t.value
        return h_sol, t_sol, elapsed
    else:
        # Fallback to existing solution if MILP fails
        h_sol = np.round(h.value).astype(int)
        if h_sol is None:
            h_sol = initial_h_values(n_points)
        t_sol = 0
        return h_sol, t_sol, elapsed

def improve_with_hill_climbing(h_bin, n_points, max_time=100):
    """Improve binary solution with hill-climbing swaps."""
    # Compute initial overlaps
    h = h_bin.copy()
    dx = 2.0 / n_points
    A_s = np.zeros(n_points - 1)
    t = float('inf')
    for s in range(1, n_points):
        corr = np.sum(h[:-s] * h[s:])
        A_s[s-1] = corr
        t = min(t, corr)
    t = int(t)
    
    # Start optimizing
    start_time = time.time()
    while time.time() - start_time < max_time:
        improved = False
        # Try swaps
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Try 20 random swaps
        for _ in range(20):
            if time.time() - start_time > max_time - 5:
                break
            i1 = np.random.choice(ones)
            i0 = np.random.choice(zeros)
            # Temporarily swap
            h[i1], h[i0] = 0, 1
            # Compute new overlaps affected
            new_t = t
            changes = []
            for s in range(1, n_points):
                if (i1 + s < n_points or i0 + s < n_points) and \
                   (i1 - s >= 0 or i0 - s >= 0):
                    # Check if this shift changes
                    old_val = h_bin[i1] * h_bin[i1+s] + h_bin[i0] * h_bin[i0+s]
                    new_val = h[i1] * h[i1+s] + h[i0] * h[i0+s]
                    if old_val != new_val:
                        changes.append((s, old_val, new_val))
            # Compute new overlap for affected shifts
            old_t = t
            for s, old_val, new_val in changes:
                A_s[s-1] = A_s[s-1] - old_val + new_val
                new_t = min(new_t, A_s[s-1])
            if new_t > old_t:
                t = int(new_t)
                improved = True
            else:
                # Revert
                h[i1], h[i0] = 1, 0
        if not improved:
            break
    # Final A_s and t
    for s in range(1, n_points):
        A_s[s-1] = np.sum(h[:-s] * h[s:])
    t = min(A_s)
    t = int(t)
    return h, t

def polish_with_slsqp(h_bin, n_points, max_time=30):
    """Use SLSQP to improve continuous version of h."""
    h_init = h_bin.astype(float)
    dx = 2.0 / n_points
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Add tiny random noise
    h_init += np.random.uniform(-0.01, 0.01, n_points)
    h_init = np.clip(h_init, 0, 1)
    
    start_time = time.time()
    try:
        res = minimize(objective, h_init, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 1000, 'ftol': 1e-9},
                       tol=1e-9)
        effective_time = time.time() - start_time
        if res.success and effective_time < max_time:
            h_result = res.x.copy()
        else:
            h_result = h_init.copy()
    except Exception as e:
        # Default to h_init if optimization fails
        h_result = h_init.copy()
    return h_result

def upsample_solution(h_orig, n, factor=3):
    """Replicate each element to increase resolution."""
    assert n * factor <= 1000
    h_new = np.repeat(h_orig, factor)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = 1.0
    best_h = None
    n_points = 100  # Start with 100 points in binary version
    
    # Try with 100 points first
    print(f"Solving with n={n_points}...")
    h_bin, t, elapsed = solve_milp_binary(n_points, solver_time_limit=budget_s / 2)
    print(f"MILP finished in {elapsed:.2f}s with t={t}")
    
    # Local improvement
    time_left = budget_s - (time.time() - start_time)
    if time_left > 30:
        print(f"Improving with hill climbing...")
        h_improve, t_improve = improve_with_hill_climbing(h_bin, n_points, max_time=time_left - 30)
        print(f"After hill climbing t={t_improve}, before it was {t}")
        t = t_improve
        h_bin = h_improve.copy()
    else:
        h_improve = h_bin.copy()
    if t == 0:
        h_improve = h_bin.copy()  # Fallback

    # Compute final binary c5
    n_points_bin = n_points
    dx_bin = 2.0 / n_points_bin
    corr_bin = np.correlate(h_improve, 1 - h_improve, mode='full')
    c5_bin = np.max(corr_bin) * dx_bin
    print(f"Binary c5 = {c5_bin}")
    
    # Try larger N if time allows
    if time.time() - start_time < budget_s - 100 and n_points == 100:
        print("Trying larger n=120...")
        n_points = 120
        h_bin_large, t_large, _ = solve_milp_binary(n_points, 150)
        if t_large > t:
            print(f"Improving to n={n_points}, t={t_large}")
            # Improve this one
            time_left = budget_s - (time.time() - start_time)
            if time_left > 30:
                h_improve_l, t_improve_l = improve_with_hill_climbing(h_bin_large,
                    n_points, max_time=time_left - 30)
                print(f"After hill climbing t={t_improve_l} (was {t_large})")
                if t_improve_l > t_large:
                    t_large = t_improve_l
                    h_bin_large = h_improve_l
            t = t_large
            h_bin = h_bin_large
            n_points_bin = n_points

    # Try local improvement again to ensure best binary solution
    time_left = budget_s - (time.time() - start_time)
    if time_left > 30:
        print("Final improvement with hill climbing...")
        h_improve_final, t_final = improve_with_hill_climbing(h_bin, n_points_bin,
            max_time=time_left - 30)
        t = t_final
        h_bin = h_improve_final

    # Continuous polishing (only once time allows)
    c5_final = 1.0 # fallback
    h_poly = h_bin.copy()
    if 1 - 2 * t / n_points_bin < 0.3807:
        time_left = budget_s - (time.time() - start_time)
        if time_left > 30:
            print("Polishing with SLSQP...")
            h_poly = polish_with_slsqp(h_bin, n_points_bin, max_time=30)
            c5_final = evaluate_erdos_solution(h_poly, n_points_bin)

    # Upsample solution if still below target
    if 1 - 2 * t / n_points_bin > 0.3807:
        n_upsample = min(1000, n_points_bin * 4)
        factor = n_upsample // n_points_bin
        if factor > 1:
            h_final = upsample_solution(h_bin, n_points_bin, factor)
        else:
            h_final = h_bin.copy()
        c5_final = evaluate_erdos_solution(h_final, n_upsample)
    else:
        # Already low enough
        h_final = h_poly.copy()
        c5_final = 1 - 2 * t / n_points_bin
    
    # Final evaluation of best result
    assert len(h_final) > 0
    if c5_final < 1 - 2 * t / n_points_bin:
        # Use polygon result
        best_c5 = c5_final
        best_h = h_final
        n_points_return = len(h_final)
    else:
        # Use binary
        best_c5 = 1 - 2 * t / n_points_bin
        best_h = h_bin
        n_points_return = n_points_bin

    return (best_h, best_c5, n_points_return)
