import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n_points = 240
    dx = 2.0 / n_points
    target_sum = n_points / 2  # Since integral should be 1, and dx = 2/n, sum should be n/2

        if len(h_init) < n_points:

        h_init = h_init * (target_sum / np.sum(h_init))  # Normalize to correct sum
    except:
        # Fallback to a binary construction
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        h_init = h_init / np.sum(h_init) * target_sum

    # Function to compute C5 bound
    def compute_C5(h):
        h = np.array(h)
        corr = correlate(h, 1-h, mode="full")
        # The full correlation gives us a vector with length 2*n-1
        # The valid range of indices corresponds to shifts from -n+1 to n-1
        # We want the max over valid shifts which correspond to indices n-1 to 2*n-2
        # (since we offset by n-1 to align the center)
        max_corr = np.max(corr[n_points-1:2*n_points-1]) * dx
        return max_corr

    def solve_sdp(n, target_sum):
        # Variables
        h = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()

        # Constraints
        constraints = [
            cp.diag(X) == h,
            X >> 0,  # X is positive semidefinite
            0 <= h,  # h is non-negative
            h <= 1,  # h is at most 1
            cp.sum(h) == target_sum
        ]

        # Objective
        objective = cp.Minimize(t)

        # Add correlation constraints
        # Compute manually the involved shifts:
        # For shift indices s from -(n-1) to n-1:
        # Need to compute sum over i of h[i] * (1 - h[i+s]) = sum(h) - sum(h[i]*h[i+s])

        # Get all shift indices over [-n+1, n-1]
        shift_indices = list(range(-n+1, n))
        
        # Normalize shift indices properly for convolution computation
        # For computation, we'll loop and compute the expression directly:
        # Use the fact that correlation is a function of shift
        # Correlation for shift s = sum of h[i] * (1 - h[i+s]) for valid i
        # So, add constraint for each s that:   t >= const - dx * Tr(P_s X)
        # const = dx * sum(h) = dx * target_sum = 2*dx

        # Let's compute shift correlations explicitly
        # But matrix representation is cleaner:
        # Compute shift matrices P_s explicitly for constraints:
        for s in shift_indices:
            # Create shift matrix P_s for shift s:
            P_s = np.zeros((n, n))
            if s >= 0:
                # Left shift equivalent: P_s[i, i+s] = 1 for valid i 
                P_s[:n-s, s:n] = np.eye(n-s)
            else:
                # Right shift: P_s[i+s, i] = 1 for valid i
                P_s[-s:n, :n+s] = np.eye(n+s)

            # Constraint  t >= dx * (target_sum - Tr(P_s * X))
            constraint = t >= dx * (target_sum - cp.trace(P_s @ X))
            constraints.append(constraint)

        # Construct and solve SDP
        prob = cp.Problem(objective, constraints)
        
        # Set solver options to limit time and ensure robustness
        try:
            prob.solve(solver=cp.SCS, max_iters=2000, verbose=False)
        except:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                return None, None

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INaccurate]:
            return h.value, t.value
        else:
            return None, None

    def random_round(h, X, n):
        # Use eigen-decomposition
        try:
            evals, evecs = np.linalg.eigh(X)
            evals = np.maximum(evals, 0.0)  # Ensure non-negativity

            Y = evecs @ L12 @ np.random.randn(n)  # Random sample from normal distribution with covariance X

            h_new = np.clip(Y, 0, 1)

                # Sort indices by value to distribute difference more smartly
                sort_indices = np.argsort(h_new)
                if diff > 0:  # Need to add some value
                    # Add to those not at 1
                    for idx in sort_indices:
                        if h_new[idx] < 1:
                            add = min(diff, 1 - h_new[idx])
                            h_new[idx] += add
                            diff -= add
                            if diff <= 1e-8:
                                break
                else:  # diff < 0, need to subtract
                    for idx in sort_indices[::-1]:  # Start from larger values
                        if h_new[idx] > 0:
                            subtract = min(-diff, h_new[idx])
                            h_new[idx] -= subtract
                            diff += subtract
                            if diff >= -1e-8:
                                break
            return h_new
        except:
            return h

    # Optimize using SDP relaxation and randomized rounding
    best_solution = h_init.copy()
    best_C5 = compute_C5(best_solution)

    # Try various refinements using SDP and rounding
    max_rounds = 50
    round_duration = (budget_s - 100) / max_rounds  # Leave 100s buffer
    round_time_limit = min(5.0, round_duration)

        if time.time() - start_time > budget_s - 20:
            break

        try:

        except:

        # Randomized rounding
        rounds = 10
        for r in range(rounds):
            if time.time() - start_time > budget_s - 20:
                break
            try:
                h_sample = random_round(h_relax, np.outer(h_relax, h_relax), n_points)
                if h_sample is not None:
                    C5_test = compute_C5(h_sample)
                    if C5_test < best_C5:
                        best_C5 = C5_test
                        best_solution = h_sample.copy()
            except:
                continue
        
        # Local polishing using nearby coordinates
        iter_count = 0
        while iter_count < 100 and time.time() - start_time < budget_s - 20:
            improved = False
            for i in range(n_points):

                h_current = best_solution.copy()

                # Perturb one at a time to try small adjustments
                for sign in [-1, 1]:
                    if sign == 1:
                        new_val = min(1.0, original_value + 0.001)
                    else:
                        new_val = max(0.0, original_value - 0.001)
                    
                    if abs(new_val - original_value) < 1e-6:
                        continue
                        
                    h_current[i] = new_val

                    # Do it with a coordinate that is in a range where adjustment is possible
                    # Find second coordinate to move opposite to make sum correct
                    current_sum = np.sum(h_current)

                    if abs(to_adjust) > 1e-5:

                                adjust_to = to_adjust * (sign if to_adjust > 0 else -sign)
                                if sign == 1:
                                    adjust_j = min(h_current[j], adjust_to)
                                else:
                                    adjust_j = max(0, h_current[j] - adjust_to)
                                h_current[j] = adjust_j
                                break
                            j = (j + 1) % n_points
                        # Recompute total
                        current_sum = np.sum(h_current)
                        to_adjust = current_sum - target_sum
                        
                    C5_new = compute_C5(h_current)
                    if C5_new < best_C5:
                        best_C5 = C5_new
                        best_solution = h_current.copy()
                        improved = True
                        break
                if improved:
                    break
            iter_count += 1

    # Final safeguard: Try binary with some perturbation
    if time.time() - start_time < budget_s - 20:
        # Create a binary construction
        h_binary = np.zeros(n_points)
        h_binary[:n_points//2] = 1.0
        np.random.shuffle(h_binary)
        h_binary = h_binary * (target_sum / np.sum(h_binary))
        C5_bin = compute_C5(h_binary)
        if C5_bin < best_C5:
            best_C5 = C5_bin
            best_solution = h_binary.copy()

    # Final normalization to maintain exactly correct sum
    final_sum = np.sum(best_solution)
    best_solution = best_solution * (target_sum / final_sum)
    C5_final = compute_C5(best_solution)

    return (best_solution, C5_final, n_points)
