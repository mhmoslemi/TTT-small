def run(seed=42, budget_s=600, **kwargs):

    best_c5 = float('inf')
    best_h = None

            initial_solutions.append(h_init)

def project_simplex(h):
    """Project h onto the feasible set: 0 ≤ h_i ≤ 1 and sum(h) = n/2"""
    n = len(h)
    h = np.clip(h, 0, 1)
    deficit = n / 2.0 - np.sum(h)
    free_indices = np.where((h > 0) & (h < 1))[0]
    
    if np.abs(deficit) < 1e-12:
        return h
    if len(free_indices) == 0:
        return h
    
    iteration = 0
    max_iter = n
    
    while np.abs(deficit) > 1e-12 and len(free_indices) > 0 and iteration < max_iter:
        delta = deficit / len(free_indices)
        h[free_indices] += delta
        # Clamp and remove out-of-bounds indices
        out_of_bounds = np.where((h[free_indices] <= 0) | (h[free_indices] >= 1))[0]
        if len(out_of_bounds) > 0:
            indices_to_remove = free_indices[out_of_bounds]
            h[indices_to_remove] = np.clip(h[indices_to_remove], 0, 1)
            free_indices = np.setdiff1d(free_indices, indices_to_remove)
            deficit = n / 2.0 - np.sum(h)
            if len(free_indices) == 0:
                break
        else:
            deficit = n / 2.0 - np.sum(h)
        iteration += 1
    return h

def compute_autocorrelations(h, n):
    """Compute linear autocorrelations up to lag n-1"""
    # Pad with zeros to avoid circular correlations
    h_padded = np.concatenate([h, np.zeros(n)])
    H = fft(h_padded)
    r = np.real(ifft(H * np.conj(H)))
    # Return only the linear autocorrelations (excluding the circular part)
    return r[:n]

def compute_soft_min_gradient(h, n, alpha=1.0):
    """Compute the gradient of the soft minimum of the autocorrelations"""
    r = compute_autocorrelations(h, n)
    # Apply log-sum-exp trick to avoid overflow
    r_alpha = -alpha * r[1:]  # skip r[0] because it is autocorrelation of h with itself
    max_r = np.max(r_alpha)
    exp_r = np.exp(r_alpha - max_r)
    sum_exp = np.sum(exp_r)
    weights = exp_r / sum_exp
    
    # gradient components
    # ∇S_α = Σ w_k · ∂r_k/∂h
    
    # ∂r_k/∂h_i = h_{i+k} + h_{i-k} (or 0 when out of bounds)
    # For explicit computation, perform FFT-based convolution
    h_padded = np.concatenate([h, np.zeros(n)])
    h_rev = np.flip(h)
    h_rev_padded = np.concatenate([h_rev, np.zeros(n)])
    
    # Convolution-like operations
    # For k=1,...,n-1, w_k·∂r_k/∂h; the derivative is sum of shifted versions
    # The gradient is (weights * h) + (reverse(weights) * h) where * is linear convolution

    w_padded = np.concatenate([weights, np.zeros(n)])
    w_rev = np.flip(weights)
    w_rev_padded = np.concatenate([w_rev, np.zeros(n)])
    
    # Convolve using FFT
    W = fft(w_padded)
    H = fft(h_padded)
    W_rev = fft(w_rev_padded)
    
    grad1 = np.real(ifft(W * H))
    grad2 = np.real(ifft(W_rev * H))
    
    # Extract linear convolution (exclude padding and wrap-around)
    grad = grad1[:n] + grad2[:n]
    
    return grad

def line_search_step(h, grad, n, max_alpha=1.0, maximum_evals=10):
    """Simple line search"""
    # Evaluate function at h + s * grad
    step_size = 1.0
    for _ in range(maximum_evals):
        h_new = h - step_size * grad
        h_new = project_simplex(h_new)
        
        # Approximate the function value
        r = compute_autocorrelations(h_new, n)
        # Avoid numerical issues
        if np.all(r > -1e10):
            min_r = np.min(r[1:])
            # Call soft minimum with large alpha even for the approximation
            # alpha = 1000. for numerical stability
            alpha = 1000.0
            r_alpha = -alpha * r[1:]
            max_r = np.max(r_alpha)
            exp_r = np.exp(r_alpha - max_r)
            sum_exp = np.sum(exp_r)
            softmin = -(1.0 / alpha) * (np.log(sum_exp) + max_r)
            return h_new, step_size
        step_size /= 2.0
    return h, 0.0


def minimize_with_homotopy(h_start, n, max_iter=1000, timeout_s=300):
    """Minimize using homotopy with soft minimum surrogate"""
    # allocate buffers
    h_prev = h_start.copy()  # Past iterate for Nesterov momentum
    h = h_start.copy()
    alpha = 0.5  # Start with small alpha
    alpha_max = 80.0
    alpha_factor = 2.0
    best_h = h.copy()
    best_min_r = float('inf')
    t0 = time.time()
    
    # We'll compute the exact function only for final decision.
    for iter_n in range(max_iter):
        if time.time() - t0 > timeout_s - 1:
            break
            
        # Compute gradient of soft min surrogate
        grad = compute_soft_min_gradient(h, n, alpha)
        
        # Nesterov-style momentum
        beta = 0.9  # momentum
        y = h + beta * (h - h_prev)
        h_prev = h
        
        # Apply line search
        h_new, _ = line_search_step(y, grad, n)
        
        # Update h
        h = h_new
        
        # Logging to decide best
        r = compute_autocorrelations(h, n)
        min_r = np.min(r[1:])
        
        if min_r < best_min_r:
            best_min_r = min_r
            best_h = h.copy()
            
        # Anneal alpha
        if iter_n % 20 == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_factor, alpha_max)
            
    return best_h

def round_binary(h):
    """Round to binary values without violating sum constraint"""
    n = len(h)
    # Sort indices by value
    sorted_indices = np.argsort(h)[::-1]
    
    # Set top n/2 values to 1, rest to 0
    h_new = np.zeros_like(h)
    count = int(n / 2)
    for i in range(count):
        h_new[sorted_indices[i]] = 1.0
    
    return h_new
