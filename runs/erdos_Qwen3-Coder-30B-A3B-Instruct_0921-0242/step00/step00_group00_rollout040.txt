def evaluate_h(h, dx):
    """Evaluate C5 for given h (discretized function)."""
    n = len(h)
    # convert to full convolution form
    h_extended = np.concatenate([h, np.zeros(n)])
    ones_extended = np.concatenate([np.ones(n), np.zeros(n)])
    corr = np.correlate(h_extended, ones_extended - h, mode='full')
    # Take the max across all lags
    return np.max(corr) * dx

def project_to_capped_simplex(v, target_sum):
    """Project vector v onto simplex sum = target_sum with 0 <= v_i <= 1.
    Uses bisection on the Lagrange multiplier."""
    v = np.asarray(v)
    n = len(v)
    if target_sum <= 0:
        return np.zeros_like(v)
    # Compute projection
    # We determine theta such that sum(max(0, min(1, v_i - theta))) = target_sum
    # Use Newton bisection method
    low, high = np.min(v - 1), np.max(v)
    for _ in range(50):
        theta = (low + high) / 2.0
        proj = np.maximum(0, np.minimum(1, v - theta))
        s = np.sum(proj)
        if s > target_sum:
            low = theta
        else:
            high = theta
    theta = (low + high) / 2.0
    proj = np.maximum(0, np.minimum(1, v - theta))
    return proj

def solve_sdp(n, target_sum, max_time=350):
    t0 = time.time()
    dx = 2.0 / n
    # Build shift matrices
    shift_matrices = []
    for k in range(n):
        m = diags([1]*max(0,n-k), k, shape=(n,n), format='lil')
        shift_matrices.append(m)
    # Decision variables
    h = cp.Variable(n)
    X = cp.Variable((n,n), symmetric=True)
    # Objective: maximize L
    L = cp.Variable()
    # Constraints
    constraints = [
        X >> 0,  # X positive semidefinite
        0 <= h,  
        h <= 1,
        cp.sum(h) == target_sum,
        X @ cp.diag(np.ones(n)) == h,  # diagonal elements of X = h
        X >= 0,
    ]
    # Add constraints for X_ij <= min(h_i,h_j)
    for i in range(n):
        for j in range(i+1,n):
            constraints.append(X[i,j] <= h[i])
            constraints.append(X[i,j] <= h[j])
    # Add autecorrelation constraints for max k
    for k in range(n):
        constraints.append(cp.trace(shift_matrices[k] @ X) >= L)
    # Solve SDP
    problem = cp.Problem(cp.Maximize(L), constraints)
    problem.solve(solver=cp.SCS, max_iters=1000, eps=1e-4)
    if problem.status != cp.OPTIMAL:
        print(f"Warning: SDP not optimal. Status: {problem.status}")
    
    # Get solution
    h_sol = h.value
    X_sol = X.value
    
    # Eigenvalue decomposition
    eigenvals, eigenvecs = np.linalg.eigh(X_sol)
    if np.all(eigenvals >= -1e-6):
        # take sqrt of positive eigenvalues, reconstruct
        idx = eigenvals > 1e-6
        v = np.sqrt(np.maximum(0, eigenvals[idx])) * eigenvecs[:, idx]
        v = np.sum(v, axis=1)
    else:
        # fallback to svd method
        # reshape h_sol
        v = h_sol.copy()
    
    # Project to feasible simplex
    h_feas = project_to_capped_simplex(v, target_sum)
    
    end_time = time.time()
    if end_time - t0 > max_time:
        print("SDP time limit reached")
    return h_feas, end_time - t0

def local_polish(h, max_iter=500):
    """Local improvement via alternating minimization"""
    n = len(h)
    dx = 2.0 / n
    # Inner loop: gradient decent on the main objective
    for it in range(max_iter):
        # evaluate current values
        h_extended = np.concatenate([h, np.zeros(n)])
        ones_extended = np.concatenate([np.ones(n), np.zeros(n)])
        corr = np.correlate(h_extended, ones_extended - h, mode='full')
        # k* for which we get maximum overlap
        max_pos = np.argmax(corr)
        k_star = max_pos - (n - 1)
        # place k_star in valid range [-n+1, n-1]
        if k_star < -(n - 1):
            k_star = -(n - 1)
        elif k_star > n - 1:
            k_star = n - 1
        
        # Compute subgradient (for k_star)
        # subgrad is (1 - h_{i+k_star}) - h_{i-k_star}
        grad = np.zeros(n)
        k = k_star
        for i in range(n):
            hi = h[i]
            if i + k < n and i + k >= 0:
                grad[i] -= h[i + k]
            if i - k >= 0 and i - k < n:
                grad[i] -= h[i - k]
            if i + k < n and i + k >= 0:
                grad[i] += 1
                
        # Gradient step with diminishing step size
        alpha = 1.0 / (1.0 + it)  # diminish step magnitude
        h -= alpha * grad
        # Project back to feasible region
        h = project_to_capped_simplex(h, n / 2)
    return h

        h_sdp_fine, _ = solve_sdp(n_fine, n_fine / 2, max_time=budget_s * 0.4)
        if time.time() - start_time > 0.5:
            pass
            
        C5_sdp_fine = evaluate_h(h_sdp_fine, dx_fine)
        if C5_sdp_fine < best_C5:
            best_h = h_sdp_fine.copy()
            best_C5 = C5_sdp_fine
            best_n = n_fine
    except Exception as e:
        print(f"Error in fine SDP: {e}")
        
    # Local polishing
    try:
        if best_h is not None:
            t_polish = time.time()
            # Improve with local polish
            h_improved = local_polish(best_h.copy())
            c5_improved = evaluate_h(h_improved, 2.0 / best_n)
            if c5_improved < best_C5:
                best_h = h_improved.copy()
                best_C5 = c5_improved
    except Exception as e:
        print(f"Error in polishing: {e}")
    
    if best_h is None:
        # Fallback
        best_h = np.array(initial_h_values()[:100])  # Check availability
        best_h = best_h / np.sum(best_h) * 50
        best_h = np.clip(best_h, 0, 1)
        best_C5 = evaluate_h(best_h, 2.0 / len(best_h))
        best_n = len(best_h)
    
    return (best_h.tolist(), float(best_C5), int(best_n))
