import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def decode(x, M):
    """Decode the parameter vector x into segment lengths u and heights a."""
    u = x[:M]
    a0 = x[M]
    v = x[M+1:]
    a = np.empty(M+1)
    a[0] = a0
    for j in range(1, M):
        a[j] = a[j-1] - v[j-1]
    a[M] = a[M-1] - v[M-1]
    a = np.maximum(a, 0)
    s = np.concatenate([[0], np.cumsum(u)])
    return u, a, s

def overlap_piecewise(s, a):
    """Compute the exact max overlap C5 for a piecewise constant h."""
    M = len(a) - 1
    K = []
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0 <= k1 <= 2:
                K.append(k1)
            if 0 <= k2 <= 2:
                K.append(k2)
    K = sorted(set(K + [0, 2]))
    F_vals = []
    for k in K:
        F_k = 0
        for j in range(M+1):
            for l in range(M+1):
                s_j, s_jp1 = s[j], s[j+1]
                s_l, s_lp1 = s[l], s[l+1]
                overlap_len = max(0, min(s_jp1, s_lp1 - k) - max(s_j, s_l - k))
                F_k += a[j] * (1 - a[l]) * overlap_len
        F_vals.append(F_k)
    return max(F_vals)

def project_capped_simplex(v, total):
    """Project v onto the capped simplex where sum = total."""
    v = np.array(v)
    if np.abs(np.sum(v) - total) < 1e-10:
        return v
    v = np.maximum(v, 0)
    v_sum = np.sum(v)
    if v_sum > 0:
        d = (total - v_sum) / len(v)
        v += d
        # Ensure non-negativity
        v = np.maximum(v, 0)
    return v

def fitness_func(x, M, n_points, lambda_reg=1e6):
    u, a, s = decode(x, M)
    # Check all constraints
    if np.any(u < 0) or np.any(a < 0) or np.any(a > 1):
        return 1e6
    # Check integral constraint
    integral = np.sum(a * u)
    # Check non-negativity (from u and a)
    r = integral - 1
    penalty = lambda_reg * r**2
    C5 = overlap_piecewise(s, a)
    return C5 + penalty

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    time_start = time.time()
    M_init = 4
    n_points = 128
    # Phase A
    M = M_init
    n_var = M + 1 + (M - 1)
    bounds = []
    for _ in range(M):
        bounds.append((0, 2))
    bounds.append((0, 1))
    for _ in range(M - 1):
        bounds.append((0, 1))
    DE_res = differential_evolution(
        fitness_func,
        bounds,
        args=(M, n_points),
        seed=seed,
        maxiter=150,
        popsize=15 * n_var,
        mutation=(0.5, 1),
        recombination=0.7,
        disp=False,
        tol=1e-6,
    )
    best_x = DE_res.x
    best_C5 = DE_res.fun
    best_x_A = best_x
    best_C5_A = best_C5
    x_A = best_x.copy()
    tc = time.time()
    if tc - time_start > 0.8 * budget_s:
        # Too little time left for B, early exit
        M_final = M
        x_final = x_A
    else:
        # Phase B
        M = 6
        n_var = M + 1 + (M - 1)
        bounds = []
        for _ in range(M):
            bounds.append((0, 2))
        bounds.append((0, 1))
        for _ in range(M - 1):
            bounds.append((0, 1))
        # Initialize DE around best from phase A
        popsize = 15 * n_var
        population = []
        for _ in range(popsize):
            x1 = best_x + np.random.normal(0, 0.1, len(best_x))
            population.append(x1)
        population[0] = best_x.copy()
        DE_res = differential_evolution(
            fitness_func,
            bounds,
            args=(M, n_points),
            seed=seed,
            maxiter=100,
            popsize=popsize,
            mutation=(0.5, 1),
            recombination=0.7,
            disp=False,
            tol=1e-6
        )
        best_x = DE_res.x
        best_C5 = DE_res.fun
        x_B = best_x.copy()
        # Keep the best from A or B
        if best_C5 < best_C5_A:
            x_final = x_B
            M_final = M
        else:
            x_final = x_A
            M_final = M_init
    # Phase C: Refinement with local optimizers
    tc = time.time()
    if tc - time_start >= 0.9 * budget_s:
        # too little time left, skip to discretization
        pass
    else:
        def smooth_func(x, M, beta=50.0):
            u, a, s = decode(x, M)
            if np.any(u < 0) or np.any(a < 0) or np.any(a > 1):
                return 1e6
            # Check integral constraint
            integral = np.sum(a * u)
            r = integral - 1
            penalty = 1e6 * r**2
            K = []
            for j in range(M+1):
                for l in range(M+1):
                    k1 = s[l] - s[j+1]
                    k2 = s[l+1] - s[j]
                    if 0 <= k1 <= 2:
                        K.append(k1)
                    if 0 <= k2 <= 2:
                        K.append(k2)
            K = sorted(set(K + [0, 2]))
            F_vals = []
            for k in K:
                F_k = 0
                for j in range(M+1):
                    for l in range(M+1):
                        s_j, s_jp1 = s[j], s[j+1]
                        s_l, s_lp1 = s[l], s[l+1]
                        overlap_len = max(0, min(s_jp1, s_lp1 - k) - max(s_j, s_l - k))
                        F_k += a[j] * (1 - a[l]) * overlap_len
                F_vals.append(F_k)
            reg_term = np.log(np.sum(np.exp(beta * np.array(F_vals)))) / beta
            return reg_term + penalty
        bounds = []
        for _ in range(M_final):
            bounds.append((0, 2))
        bounds.append((0, 1))
        for _ in range(M_final - 1):
            bounds.append((0, 1))
        # Perform local optimization with SLSQP
        x0 = x_final.copy()
        res = minimize(
            smooth_func,
            x0,
            args=(M_final,),
            method='SLSQP',
            bounds=bounds,
            options={'maxiter': 100},
            tol=1e-8
        )
        if not res.success:
            res = minimize(
                smooth_func,
                x0,
                args=(M_final,),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 100},
                tol=1e-8
            )
        x_final = res.x
    # Final discretization
    u, a, s = decode(x_final, M_final)
    n_points = 256
    dx = 2.0 / n_points
    h = np.empty(n_points)
    for i in range(n_points):
        x = (i + 0.5) * dx
        j = np.searchsorted(s, x, side="right") - 1
        h[i] = a[j]
    total = n_points / 2
    h = project_capped_simplex(h, total)
    corr = np.convolve(h, 1 - h, mode='full') * dx
    final_C5 = np.max(corr)
    return (h.tolist(), float(final_C5), n_points)
