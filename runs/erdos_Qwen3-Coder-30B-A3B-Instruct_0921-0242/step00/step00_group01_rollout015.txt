import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Hyperparameters
    n0 = 120  # Base grid size (even)
    dx = 2.0 / n0
    S = n0 / 2  # Required sum
    target_n = 360  # Final grid size to upscale to
    
    # Timeout management
    def remaining_time():
        return max(0, budget_s - (time.time() - start_time))
    
    # Phase 0: Prepare initial solution if available
    try:
        original_h = np.array(initial_h_values)
        h0 = np.array([original_h[i] for i in range(min(len(original_h), n0))])
        if len(h0) < n0:
            h0 = np.pad(h0, (0, n0 - len(h0)), 'constant', constant_values=0)
        # Convert to binary if needed
        h_bin = (h0 > 0.5).astype(int)
        h0 = h_bin
    except:
        # Default binary construction
        h0 = np.zeros(n0)
        h0[:n0//2] = 1
        # Ensure sum = n/2 (exact)
        ones_count = int(S)
        h0[:ones_count] = 1
        h0[ones_count:] = 0
    
    # Phase 1: Solve exact MILP for binary optimization (if possible)
    w_value = None
    binary_solution = None
    try:
        if remaining_time() > 30:
            # Build the MILP model using CVXPY
            b = Variable(n0, boolean=True)
            w = Variable()
            
            # Constraints for z = b[i] * b[i+k]
            # Add constraints for z[i,k] using big-M method
            
            # Define product variables z[i,k] = b[i]*b[i+k]
            z_vars = []
            z_dict = {}
            prob_constraints = []

            for i in range(n0):
                row = []
                for k in range(n0):
                    z_var = Variable(boolean=True)
                    z_dict[(i, k)] = z_var
                    
                    # Big-M constraints for z[i,k] = b[i] * b[i+k] (circular)
                    z_var_constr = [
                        z_var <= b[i],
                        z_var <= b[(i + k) % n0],
                        z_var >= b[i] + b[(i + k) % n0] - 1
                    ]
                    prob_constraints.extend(z_var_constr)
                    row.append(z_var)
                z_vars.append(row)

            # Overlap constraints: w >= S - sum(z[i,k])
            for k in range(n0):
                # Compute sum over i of z[i,k]
                term = sum([z_dict[(i, k)] for i in range(n0)])
                prob_constraints.append(w >= S - term)
            
            # Mass constraint
            prob_constraints.append(sum(b) == S)
            
            # Symmetry breaking
            prob_constraints.append(b[0] == 1)
            # Lexicographic order: first half non-increasing
            for i in range(0, n0 // 2 - 1):
                prob_constraints.append(b[i] >= b[i+1])

            # Objective: minimize w
            prob = Problem(Minimize(w), prob_constraints)
            
            # Set solver options for time limit
            solver_time = min(remaining_time(), 300)
            prob.solve(solver=CPLEX, time_limit=solver_time, verbose=False)
            
            if prob.status in ['optimal', 'optimal_inaccurate']:
                w_value = w.value
                binary_solution = b.value
            else:
                binary_solution = None
    except Exception as e:
        # Fallback to greedy if MILP fails
        binary_solution = None
    
    # Phase 3b: If we still don't have solution, use fast greedy approach
    if binary_solution is None:
        # Greedy balanced construction
        def greedy_balanced():
            # Start with empty set
            # h is now a boolean vector
            h = np.zeros(n0, dtype=int)
            target = int(S)
            used_indices = set()
            
            while len(used_indices) < target:
                best_i = -1
                min_increase = float('inf')
                
                for i in range(n0):
                    if i in used_indices:
                        continue
                    # Simulate adding this 1 to h
                    current_autocorrelations = []
                    for k in range(n0):
                        acc = 0
                        for j in range(n0):
                            # Circular shift
                            shifted_j = (j + k) % n0
                            if h[j] == 1 and h[shifted_j] == 1:
                                acc += 1
                        current_autocorrelations.append(acc)
                    
                    # Compute k that would see max increase in overlap with newest 1
                    # Total auto-correlation at each lag after inserting 1 at i
                    effect = []
                    for k in range(n0):
                        shift = (i + k) % n0
                        # Count adjustments to overlaps due to 1 inserted at position i
                        # Two cases: i directly contributes to Autocorrelation at lag 0
                        # and i contributes to all overlaps involving itself
                        count = 0
                        for j in range(n0):
                            # Check if both h[j] and h[j-k] are 1 in old h
                            # When inserting 1 at i, 1 starts contributing to overlaps where lag
                            # makes sense, this is tricky due to circular nature.
                            # Just compute the difference manually for simplicity:
                            # What happens when inserting 1 at i?
                            # Existing overlaps from point j to point j+k with high bits
                            # do not change. What changes: # of 1s that participate in old₀ = 
                            # self-overlap brings in 1 more pair (1,1) if i references i.
                            # But spatial overlap must be recomputed mentally per data point:
                            # Effect at lag k = count of j such that h[j]=1 and h[j+k]=1.
                            # By placing 1 at i, for k that avoids i, 
                            # there is no change unless j==i or j+k==i
                            # So effect is when:
                            #  1. j = i : 1 now contributes to overlap lag (i-i) = 0 (if index j also has 1)
                            #  2. j-k = i => j = i + k  
                            if j == i:
                                count += h[0]
                            elif j == (i + k) % n0:
                                count += 1  
                            # But this makes interpretation hard.
                            # Faster to just recompute all auto-correlations iteratively.
                            # This is not scalable, we will update greedy approach algorithmically:
                            # Let's just compute future overlaps directly – think symmetrically
                            
                            # Instead directly pass these to validation functions (less error-prone)
                    # Instead brute-force estimate best i to add based on minimal increase
                    new_h = h.copy()
                    new_h[i] = 1
                    new_auto = []
                    for k in range(n0):
                        acc = 0
                        for j in range(n0):
                            if new_h[j] == 1 and new_h[(j + k) % n0] == 1:
                                acc += 1
                        new_auto.append(acc)
                    
                    # How much each autocorrelation changed
                    # Current bounds on how far they are from worst case
                    improvement = min(new_auto)
                    if improvement < min_increase:
                        min_increase = improvement
                        best_i = i
                    
                if best_i >= 0:
                    h[best_i] = 1
                    used_indices.add(best_i)
            return h
        
        h0 = greedy_balanced()
        binary_solution = h0
    
    # Convert back to regular list if needed
    if isinstance(binary_solution, np.ndarray):
        h0 = binary_solution.astype(int)
    else:
        h0 = binary_solution
    
    # Post-processing: Swap refinement
    maxswaps = 200
    current_h = h0.copy()
    
    def eval_autocorr(h_arr):
        # Compute circular autocorrelation
        corr = correlate(h_arr, h_arr, mode='full')
        cross = correlate(h_arr, 1 - h_arr, mode='full')
        # Only return values at positive lags from middle
        n = len(h_arr)
        # Take from index n-1 to end
        pos_lags = cross[n-1:]
        return pos_lags
    
    # Iterative improvement via swaps
    old_auto = eval_autocorr(current_h)
    min_auto = np.min(old_auto)
    swaps = 0
    while swaps < maxswaps:
        improved = False
        for p in range(n0):
            if current_h[p] != 1:
                continue
            for q in range(n0):
                if current_h[q] != 0:
                    continue
                
                # Try swapping p and q
                trial_h = current_h.copy()
                trial_h[p] = 0
                trial_h[q] = 1
                
                new_auto = eval_autocorr(trial_h)
                new_min_auto = np.min(new_auto)
                
                if new_min_auto > min_auto:
                    current_h[:] = trial_h[:]
                    min_auto = new_min_auto
                    improved = True
                    swaps += 1
                    break
            if improved:
                break
        
        if not improved:
            break
        swaps += 1
    
    # Enforce final integral constraint
    actual_sum = np.sum(current_h)
    if abs(actual_sum - S) > 1e-9:
        print(f"Warning: Actual sum {actual_sum} differs from expected S={S}")
    
    # Final phase: upscale to desired size
    # Compute final h vector
    m = target_n // n0  # factor to repeat each element
    final_h = []
    for i in range(n0):
        final_h.extend([int(current_h[i])] * m)
    
    # Recompute final C5 after duplication to ensure accuracy
    final_h = np.array(final_h)
    final_auto = []
    for k in range(target_n):
        mc = 0
        for i in range(target_n):
            if final_h[i] == 1 and final_h[(i+k) % target_n] == 1:
                mc += 1
        final_auto.append(mc)
    
    best_overlap = np.min(final_auto)  # This gives the final rough C5 estimate
    C5_final = best_overlap * (2.0 / target_n)  # multiply by dx = 2/N
    
    return (final_h.tolist(), C5_final, len(final_h))
