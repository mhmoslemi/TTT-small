import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def project_to_F(h, n):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Compute current deficit/excess
    deficit = n/2 - h.sum()
    # Indices that are not at a bound
    free = np.where((h>0) & (h<1))[0]
    while np.abs(deficit) > 1e-12 and free.size>0:
        # Distribute equally over free indices
        delta = deficit / free.size
        # Apply delta, respecting bounds, and recompute deficit
        h[free] += delta
        # Clamp again and update free set
        h = np.clip(h, 0.0, 1.0)
        deficit = n/2 - h.sum()
        free = np.where((h>0) & (h<1))[0]
    return h

def evaluate_fitness(h, dx, n):
    # Compute g = 1 - h
    g = 1 - h
    # Zero-pad both to length L = 2*n (next power of 2)
    L = 2 * n
    h_pad = np.zeros(L)
    g_pad = np.zeros(L)
    h_pad[:n] = h
    g_pad[:n] = g
    # Compute FFTs
    H = fft(h_pad)
    G = fft(g_pad)
    # Compute correlation
    conv = ifft(H * np.conj(G)).real * dx
    # Return max of full correlation
    return conv.max(), conv

def local_improve(h, g, dx, n, c, loc_steps=30):
    # Find the worst shift
    k_star = np.argmax(c)
    # Generate candidates for pairwise exchange
    h_new = h.copy()
    c_new = c.copy()
    for _ in range(loc_steps):
        # Find indices with non-zero and non-one entries
        indices_nonzero = np.where(h_new > 0)[0]
        indices_nonone = np.where(h_new < 1)[0]
        if len(indices_nonzero) == 0 or len(indices_nonone) == 0:
            break
        i1 = np.random.choice(indices_nonzero)
        i0 = np.random.choice(indices_nonone)
        # Compute epsilon (min of moves possible)
        delta = min(h_new[i1], 1 - h_new[i0], 0.05)
        if delta < 1e-10:  # Prevent min is zero
            break
            
        # Exchange
        h_new[i1] -= delta
        h_new[i0] += delta
        
        # Update correlation appropriately
        # Update only the correlations where shifts involve i1 or i0
        for k in range(-n+1, n):
            # Note: we have to be careful with indexing
            i1p = i1 + k
            i0p = i0 + k
            # i1p and i0p can be beyond bounds but that's fine for zero-padding
            # But when indexing into h, we're okay because zero-padding
            # corresponds to how the FFT handles the convolution (zero-padding)
            
            # The change in the correlation for shift k
            if 0 <= i1p < n:
                c_new[k + n - 1] -= delta * dx * g[i1p]
            if 0 <= i0p < n:
                c_new[k + n - 1] += delta * dx * g[i0p]
                
        # We only recompute max of the correlations for efficiency
        if c_new.max() < c.max():
            c = c_new.copy()
        else:  # Revert changes if none improved
            h_new[i1] += delta
            h_new[i0] -= delta
            break  # early exit
    return h_new, c

def initialize_population(n, pop_size, initial_h_values):
    population = []
    # Interpolated seed
    h_seed = np.interp(np.linspace(0, 2, n),
                       np.linspace(0, 2, len(initial_h_values)),
                       initial_h_values)
    h_seed = project_to_F(h_seed, n)
    c5_seed, _ = evaluate_fitness(h_seed, 2.0/n, n)
    population.append((h_seed, c5_seed))
    
    # Random continuous
    for _ in range(pop_size//3 - 1):
        h = np.random.rand(n)
        h = project_to_F(h, n)
        c5, _ = evaluate_fitness(h, 2.0/n, n)
        population.append((h, c5))

    # Binary block
    h = np.zeros(n)
    h[:n//2] = 1.0
    np.random.shuffle(h)
    h = project_to_F(h, n)
    c5, _ = evaluate_fitness(h, 2.0/n, n)
    population.append((h, c5))

    # Gaussian perturbed
    h = np.full(n, 0.5)
    h += np.random.normal(0.0, 0.05, n)
    h = project_to_F(h, n)
    c5, _ = evaluate_fitness(h, 2.0/n, n)
    population.append((h, c5))

    return population

def crossover(p1, p2):
    alpha = np.random.rand()
    child = alpha * p1 + (1 - alpha) * p2
    return child

def mutate(h, n, sigma):
    # Turn to integer for number of changes
    m = max(1, int(0.02 * n))
    indices = np.random.choice(n, m, replace=False)
    noise = np.random.normal(0, sigma, m)
    h_mut = h.copy()
    h_mut[indices] += noise
    h_mut = project_to_F(h_mut, n)
    return h_mut
