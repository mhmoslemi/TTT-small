import numpy as np
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up base parameters
    n0 = 120  # base grid size
    dx = 2.0 / n0
    S = n0 / 2  # sum constraint (equal to 1 when normalized)
    
    # Select solver - prefer SCIP if available, fall back to CBC
    solver = None
    try:
        solver = "SCIP"
        # Implicitly check if available by just defining a small model
        x = Variable(n0, boolean=True)
        prob = Problem(Minimize(0), [x == 0])
        prob.solve(solver=solver, verbose=False, max_seconds=budget_s*0.9)
        solver = "SCIP"
    except:
        try:
            solver = "CBC"
            x = Variable(n0, boolean=True)
            prob = Problem(Minimize(0), [x == 0])
            prob.solve(solver=solver, verbose=False, max_seconds=1)
            solver = "CBC"
        except:
            solver = None
    
    # Initialize variables for MILP model
    b = Variable(n0, boolean=True)  # binary variables for h[i]
    w = Variable()  # worst overlap value
    z = Variable((n0, n0), boolean=True)  # products b[i]*b[i+k]
    
    # Constraints
    constraints = []
    
    # Product constraints z[i,k] = b[i]*b[i+k]
    for i in range(n0):
        for k in range(n0):
            idx = (i + k) % n0
            constraints.append(z[i,k] <= b[i])
            constraints.append(z[i,k] <= b[idx])
            constraints.append(z[i,k] >= b[i] + b[idx] - 1)
    
    # Overlap constraints
    for k in range(n0):
        sum_z_k = sum(z[i,k] for i in range(n0))
        constraints.append(w >= S - sum_z_k)
    
    # Mass constraint
    constraints.append(sum(b) == S)
    
    # Symmetry breaking
    constraints.append(b[0] == 1)
    
    # Bounds
    constraints.append(w >= 0)
    constraints.append(w <= S)
    
    # Objective
    prob = Problem(Minimize(w), constraints)
    
    # Run optimization
    try:
        remaining_time = max(1, budget_s - (time.time() - start_time))
        prob.solve(solver=solver, verbose=False, max_seconds=remaining_time)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            # Extract binary solution
            b_sol = [int(round(b[i].value)) for i in range(n0)]
        else:
            # Fallback to greedy
            b_sol = None
    except:
        # Fallback to greedy
        b_sol = None
    
    # If no solution from MILP, use greedy (exact method)
    if b_sol is None:
        h_greedy = greedy_construction(n0)
        b_sol = [int(x > 0.5) for x in h_greedy]
        
        # Apply heuristic refinement
        b_sol = refine_by_swapping(b_sol)
    
    # Upscale solution to target grid
    target_n = min(1000, 360)  # Prefer 360 for speed
    m = target_n // n0
    h_final = []
    for j in range(target_n):
        h_final.append(b_sol[j // m])
    
    # Final evaluation
    C5 = calculate_C5(h_final)
    return (list(h_final), float(C5), target_n)

def greedy_construction(n):
    """Construct a binary sequence using a greedy algorithm."""
    # Start with an empty set S of indices that will be set to 1
    S = set()
    target_sum = n / 2
    
    while len(S) < target_sum:
        best_i = -1
        min_increase = float('inf')
        
        # Try adding each index that's not currently in S
        for i in range(n):
            if i in S:
                continue
                
            # Calculate increase in overlap if we add i to S
            current_O = [0] * n  # Store current overlaps
            for k in range(n):
                if k < len(S):
                    current_O[k] = sum(S.intersection({(i+k)%n}))
                    
            max_gain = min(current_O)  # Want to minimize the minimum overlap
            
            if max_gain < min_increase:
                min_increase = max_gain
                best_i = i
                
        if best_i == -1:
            break
        S.add(best_i)
    
    h = [1 if i in S else 0 for i in range(n)]
    return h

def calculate_C5(h):
    """Calculate C5 using the overlap integral formula."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute shifts
    overlaps = []
    for k in range(n):
        overlap = 0
        for i in range(n):
            overlap += h[i] * (1 - h[(i+k)%n])
        overlaps.append(overlap * dx)
    
    return max(overlaps)

def refine_by_swapping(b):
    """Refine binary solution with local swaps to improve the worst overlap."""
    n = len(b)
    # Precompute all overlaps
    def compute_autocorrelations():
        A = [0]*n
        for k in range(n):
            for i in range(n):
                A[k] += b[i] * b[(i+k)%n]
        return A
    
    def compute_worst_index(A):
        # Return index of worst autocorrelation
        return np.argmin(A)
    
    def evaluate_swap(i, j):
        # If we swap b[i] and b[j], how much does worst autocorrelation improve?
        old_A = compute_autocorrelations()
        old_worst = compute_worst_index(old_A)
        new_A = [0]*n
        for k in range(n):
            new_A[k] = old_A[k] - b[i]*b[(i+k)%n] - b[j]*b[(j+k)%n] + b[j]*b[(i+k)%n] + b[i]*b[(j+k)%n]
        new_worst = compute_worst_index(new_A)
        return old_A[old_worst] - new_A[new_worst]
    
    max_iter = 200
    for _ in range(max_iter):
        A = compute_autocorrelations()
        worst_index = compute_worst_index(A)
        best_i, best_j = -1, -1
        best_delta = 0
        for i in range(n):
            if b[i] == 0:
                continue
            for j in range(n):
                if b[j] == 1:
                    continue
                delta = evaluate_swap(i, j)
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                    
        if best_i != -1:
            b[best_i], b[best_j] = b[best_j], b[best_i]
        else:
            break
    return b
