import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def evaluate(h):
    """Compute C₅ = max_k ∫ h(x)(1 - h(x+k)) dx."""
    dx = 2.0 / len(h)
    result = correlate(h, 1 - h, mode="full")
    return np.max(result) * dx

def project_feasible(x, target):
    """Project x onto the capped-simplex: {0 ≤ h ≤ 1, sum(h) = target}"""
    h = np.array(x, dtype=float)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    deficit = target - current_sum
    if abs(deficit) < 1e-12:
        return h
    # Iteratively distribute deficit among zeros
    # This is a simple water-filling strategy
    for _ in range(50):
        if abs(deficit) < 1e-12:
            break
        free_indices = np.where(h < 1)[0]
        if len(free_indices) == 0:
            break
        distribute_amount = min(deficit, (1 - h[free_indices]).sum())
        if distribute_amount <= 1e-12:
            break
        # Distribute evenly
        num_free = len(free_indices)
        delta = distribute_amount / num_free
        for idx in free_indices:
            h[idx] = min(1, h[idx] + delta)
        deficit -= distribute_amount
    return h

def active_shifts(h, eps=1e-9):
    """Find all active k for which overlap is within eps of max."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    indices = np.where(np.abs(corr - max_val) < eps * max_val)[0]
    # Converts full-correlation indices to shift indices
    n = len(h)
    ks = indices - (n - 1)
    return ks

def gradient(h, ks):
    """Compute sub-gradient of max overlap with respect to active shifts."""
    dx = 2.0 / len(h)
    n = len(h)
    g = np.zeros_like(h)
    # For each active k, compute contribution to gradient
    for k in ks:
        if k >= 0:
            # h[i] * (1 - h[i + k]) 
            # grad wrt h[i] is (1 - h[i + k]) if i + k < n else 0
            for i in range(n):
                if i + k < n:
                    g[i] += 1 - h[i + k]
                else:
                    pass # out of bounds
        else:
            # h[i] * (1 - h[i - k]) 
            # grad wrt h[i] is (1 - h[i - k]) if i - k >= 0 else 0
            for i in range(n):
                if i - k >= 0:
                    g[i] += 1 - h[i - k]
                else:
                    pass # out of bounds
    g = g / len(ks) * dx
    return g

def line_search(h, g, t_current, budget_s):
    """Try several step sizes for projected descent."""
    dx = 2.0 / len(h)
    # step sizes to try
    alphas = [0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1]
    best_h = h.copy()
    best_val = t_current
    t0 = time.time()
    
    for alpha in alphas:
        if time.time() - t0 > budget_s - 1:
            break
        if alpha == 0:
            t_val = t_current
            h_new = h.copy()
        else:
            h_new = project_feasible(h - alpha * g, 1.0)
            t_val = evaluate(h_new)
        if t_val < best_val:
            best_val = t_val
            best_h = h_new.copy()
    
    return best_h, best_val

def local_optimize(h0, max_iter, t_end):
    """Projected gradient descent with line search."""
    h = h0.copy()
    current_val = evaluate(h)
    best_h = h.copy()
    best_val = current_val
    dx = 2.0 / len(h)

    for i in range(max_iter):
        if time.time() > t_end:
            break
        ks = active_shifts(h)
        grad = gradient(h, ks)
        h_new, val_new = line_search(h, grad, current_val, t_end)
        if val_new < best_val - 1e-10:
            best_h = h_new.copy()
            best_val = val_new
        h = h_new
        current_val = val_new
    return best_h, best_val

def anneal(h0, t_end):
    """Simple simulated annealing."""
    h = h0.copy()
    t_current = evaluate(h)
    best_h = h.copy()
    best_val = t_current
    T = 0.01
    decay = 0.995
    max_iter = 5000

    t0 = time.time()
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        # Random move
        i,j = np.random.randint(0, len(h), 2)
        delta = np.random.uniform(-0.1, 0.1)
        new_h = h.copy()
        if 0 <= j < len(h):  # j is held constant, we move from i to j
            # take some amount from i and add to j
            if new_h[i] >= abs(delta):
                new_h[i] -= abs(delta)
                new_h[j] = min(1, new_h[j] + abs(delta))
                new_h = project_feasible(new_h, 1.0)
                t_val = evaluate(new_h)
                if t_val < best_val:
                    best_val = t_val
                    best_h = new_h.copy()
                elif np.random.rand() < np.exp(-(t_val - t_current)/T):
                    h = new_h.copy()
                    t_current = t_val
            if t_current > t_val:
                h = new_h.copy()
                t_current = t_val
        T *= decay
        if T < 1e-6:
            break

    return best_h, best_val

def random_feasible(n):
    """Generate a random feasible step function."""
    x = np.random.random(n)
    return project_feasible(x, 1.0)

def interpolate_h(h0, n_new):
    """Interpolate h0 to size n_new using linear interpolation."""
    n_old = len(h0)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h0)
    return project_feasible(h_new, 1.0)

def milp_binary(n, time_limit):
    """Solve binary version with cvxpy."""
    try:
        import cvxpy as cp
        # Create variables
        h = cp.Variable(n, boolean=True)
        dx = 2.0 / n
        
        # Constraints
        constr = [cp.sum(h) == n // 2]
        # Objective: minimize max_{k} sum h[i] * h[i+k] * dx_upper

        # For computing the max overlap, we need this.
        # But CVXPY does not directly support max over all k.
        # So, we compute bound on max overlap with auxiliary variables.
        # Let a[i,k] be binary and indicate h[i]*h[i+k] = 1 (where valid).
        # So minimize sum(a[i,k] * dx)
        # where a[i,k] <= h[i], a[i,k] <= h[i+k], a[i,k] >= h[i]+h[i+k]-1, a[i,k] >= 0
        # and we bound sum(a[i,k]) >= (1-t)/dx for each k, to make max overlap <= t
        # But let's just try a simple bounding approach:

        # Throughout, we directly model:
        # max_k sum_i h[i] h[i+k] dx <= t
        
        # Brute-force this by adding constraints for each k
        # Let’s define aux[binary] variables for each valid i,k pairs,
        # and impose linearization constraints.
        # Easy way: we make a min model. The Max is lower-bounded by T.
        # Let’s re-do using a one-by-one shift and introduce auxiliary variables for that,

        # Instead, let's add constraint for each shift:
        # introduce A_k and add constraint sum(h[i] * h[i+k]) - A_k dx <= 0 
        # Then minimize max(1 - A_k) over all k.

        # Make one extra variable for the worst-case.
        # But to directly do max of multiple constraints using MILP:
        # introduce variable T such that
        # for each k: 1 - sum(h[i]*h[i+k]) <= T
        # minimize T

        # That can be done more easily in CVXPY with single constraint.
        # use a trick to write the final objective (minimize max overlap).
        # An equivalent problem:
        # let objective be max {k} sum(h[i]*h[i+k]) * dx, then we want to minimize it
        # which is same as minimize t  such that t >= sum(h[i]*h[i+k]) * dx for all k

        # Much easier approach for this:
        # Set up variables to approximate the correlation on the integers.
        t = cp.Variable()
        
        constraints = []
        for k in range(1, n):
            # For shift k, calculate sum(h[i] * h[i+k])
            # Need indexing bounds. Exclude [0, k) for backwards and [n-k,n) for forwards.
            sum_ = 0
            for i in range(n):
                if i + k < n:
                    sum_ += h[i] * h[i + k]  # This is binary, so valid
            constraints.append(t >= (1 - sum_ * dx) )  # To minimize max overlap.
        
        # Make sure to also add the integral constraint
        constraints.append(cp.sum(h) == n // 2)

        prob = cp.Problem(cp.Minimize(t), constraints)
        # Try with CBC solver if available
        try:
            prob.solve(solver=cp.CBC, timelimit=time_limit)
        except:
            pass  # fallback
        if prob.status == cp.OPTIMAL:
            return np.array(h.value).astype(int), prob.value
    except Exception:
        # If cvxpy or solver is missing, return None
        pass
    return None, None

            h_start = seed_h
        else:
            # interpolate to target size
            h_start = interpolate_h(seed_h, n)

        # Local Optimization
        max_iter = 2000
        t_end = t0 + budget_s - 5
        start_points = [h_start]
        for _ in range(3):
            st = random_feasible(n)
            start_points.append(st)

        local_best_h, local_best_val = None, np.inf

        for h_init in start_points:
            if time.time() - t0 > budget_s - 5:
                break
            h_opt, val_opt = local_optimize(h_init, max_iter, t_end)
            if val_opt < local_best_val:
                local_best_h, local_best_val = h_opt, val_opt

        if local_best_val < best_val - 1e-7:
            best_h = local_best_h
            best_val = local_best_val
            best_n = n

        # After local optimization, run annealing
        if time.time() - t0 < budget_s - 5 and local_best_h is not None:
            h_ann, val_ann = anneal(local_best_h, t_end)
            if val_ann < best_val - 1e-7:
                best_h = h_ann
                best_val = val_ann
                best_n = n

    if best_h is None:
        # Fallback to seed
        best_h = seed_h
        best_val = evaluate(best_h)
        best_n = n_base
    return best_h, best_val, best_n
