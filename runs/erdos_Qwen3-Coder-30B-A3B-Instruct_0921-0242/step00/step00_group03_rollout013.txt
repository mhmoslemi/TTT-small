import numpy as np
import time
from scipy.fftpack import fft, ifft
from scipy.optimize import minimize
from math import sqrt

def project_to_F(h, n):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Compute current deficit/excess
    deficit = n/2 - h.sum()
    # Indices that are not at a bound
    free = np.where((h>0) & (h<1))[0]
    while np.abs(deficit) > 1e-12 and free.size>0:
        # Distribute equally over free indices
        delta = deficit / free.size
        # Apply delta, respecting bounds, and recompute deficit
        h[free] += delta
        # Clamp again and update free set
        h = np.clip(h, 0.0, 1.0)
        deficit = n/2 - h.sum()
        free = np.where((h>0) & (h<1))[0]
    return h

def fast_conv(h, g, dx):
    L = 2 * len(h)
    h_pad = np.pad(h, (0, L - len(h)), 'constant')
    g_pad = np.pad(g, (0, L - len(g)), 'constant')
    H = fft(h_pad)
    G = fft(g_pad)
    conv = ifft(H * np.conj(G)).real * dx
    return conv

def evaluate_C5(h, dx):
    g = 1 - h
    conv_full = fast_conv(h, g, dx)
    c5 = conv_full.max()
    return c5, conv_full

        # Compute the change in correlations for all k affected by the swap
        # Affected shifts: all k such that i1+k or i0+k is in valid range
        # We will update the change incrementally for affected k values
        
        # Increase i0, decrease i1 by epsilon
        h_new = h.copy()
        h_new[i1] -= eps
        h_new[i0] += eps

        # Incrementally update correlation for all affected k
        for k in range(-len(h) + 1, len(h)):
            # Compute new correlations using only updated terms
            # Handle the case when k is outside the valid range
            if k >= 0:
                # i1+k is valid, i0+k is valid
                i1_plus_k = i1 + k
                i0_plus_k = i0 + k
                if 0 <= i1_plus_k < len(h) and 0 <= i0_plus_k < len(h):
                    delta = eps * dx * (g[i1_plus_k] - g[i0_plus_k])
                    conv_new[k + len(h) - 1] -= delta
                elif 0 <= i1_plus_k < len(h):
                    delta = eps * dx * (g[i1_plus_k])
                    conv_new[k + len(h) - 1] -= delta
                elif 0 <= i0_plus_k < len(h):
                    delta = eps * dx * (-g[i0_plus_k])
                    conv_new[k + len(h) - 1] -= delta
            else:
                # k < 0, i1+k or i0+k might be negative
                i1_plus_k = i1 + k
                i0_plus_k = i0 + k
                # Only deal with the actual indices that are valid
                if 0 <= i1_plus_k < len(h) and 0 <= i0_plus_k < len(h):
                    delta = eps * dx * (g[i1_plus_k] - g[i0_plus_k])
                    conv_new[k + len(h) - 1] -= delta
                elif 0 <= i1_plus_k < len(h):
                    delta = eps * dx * (g[i1_plus_k])
                    conv_new[k + len(h) - 1] -= delta
                elif 0 <= i0_plus_k < len(h):
                    delta = eps * dx * (-g[i0_plus_k])
                    conv_new[k + len(h) - 1] -= delta
        
        # Check if the change improves the correlation
        new_max = conv_new.max()
        if new_max < conv_full.max():
            h = h_new
            conv_full = conv_new
        else:
            break

    return h, conv_full

def initialise_population(n, pop_size, initial_h):
    # Give some diversity to the initial population
    pop = []
    # Interpolated initial segment
    h_interpolated = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h)), initial_h)
    h_interpolated = project_to_F(h_interpolated, n)
    c5_interpolated, _ = evaluate_C5(h_interpolated, 2.0 / n)
    pop.append((h_interpolated, c5_interpolated))

    # Random vectors
    for _ in range(pop_size - 1):
        h_random = np.random.rand(n)
        h_random = project_to_F(h_random, n)
        c5_random, _ = evaluate_C5(h_random, 2.0 / n)
        pop.append((h_random, c5_random))

    # Sort list for easier access
    pop.sort(key=lambda x: x[1])
    return pop

def crossover(p1, p2, n):
    alpha = np.random.rand()
    child = alpha * p1 + (1 - alpha) * p2
    child = project_to_F(child, n)
    return child

def mutate(h, n, sigma):
    # Number of coordinates to mutate
    m = max(1, int(0.02 * n))
    indices = np.random.choice(n, m, replace=False)
    noise = np.random.normal(0, sigma, m)
    h_mut = h.copy()
    h_mut[indices] += noise
    h_mut = project_to_F(h_mut, n)
    return h_mut
