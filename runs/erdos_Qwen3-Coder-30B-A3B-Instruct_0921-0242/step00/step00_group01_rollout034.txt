def project_box_affine(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Project onto affine constraint
    delta = (np.sum(h) - target_sum) / len(h)
    h -= delta
    h = np.clip(h, 0, 1)
    
    return h

def autocorr(h):
    """Circular autocorrelation of h using FFT"""
    n = len(h)
    # Using FFT-based circular autocorrelation
    H = np.fft.fft(h)
    autocorr_vals = np.real(np.fft.ifft(H * np.conj(H)))
    return autocorr_vals

def compute_c5(h, dx):
    """Compute the C5 value as max of overlap integral"""
    n = len(h)
    # Use full convolution to compute O_k = dx * sum(h_i * (1 - h_{i+k}))
    # This is equivalent to: dx * (sum(h_i) - sum(h_i * h_{i+k}))
    total = np.sum(h)
    auto = autocorr(h)
    overlaps = dx * (total - auto)
    return np.max(overlaps)

def gradient_ascent_step(h, dx, step_size):
    """Perform one gradient ascent step for minimizing C5"""
    n = len(h)
    auto = autocorr(h)
    # We want to maximize the minimum autocorrelation 
    # so we descent the negative of that
    worst_lag = np.argmin(auto) 
    
    # Gradient for the worst_autocorrelation w.r.t. each h_i
    grad = np.zeros(n)
    # Shift h right and left by worst_lag to get terms of h_{i + k} + h_{i - k}
    shifted_right = np.roll(h, worst_lag)
    shifted_left = np.roll(h, -worst_lag)
    grad = -(shifted_right + shifted_left)
    
    # Gradient step
    h_new = h + step_size * grad
    # Project back
    h_new = project_box_affine(h_new, n/2)
    
    return h_new
