import numpy as np
from scipy import fftpack
import time
import random

def autocorr(b):
    """Compute cyclic autocorrelation of binary sequence b."""
    B = np.fft.fft(b)
    A = np.fft.ifft(B * np.conj(B)).real
    return A

def evaluate_solution(h):
    """Evaluate the C5 bound for a given h."""
    dx = 2.0 / len(h)
    C5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
    return C5

def initialize_seeds(n=200):
    """Generate several candidate binary sequences."""
    seeds = []
    
    # 1. Legendre sequence (quadratic residue)
    p = 211  # prime >= n
    b = np.zeros(n, dtype=int)
    for i in range(n):
        if pow(i, (p-1)//2, p) == 1:
            b[i] = 1
    
    # Ensure balance
    ones_count = np.sum(b)
    if ones_count > n//2:
        # Remove excess ones
        to_remove = ones_count - n//2
        indices = np.where(b == 1)[0]
        np.random.shuffle(indices)
        for i in range(to_remove):
            b[indices[i]] = 0
    elif ones_count < n//2:
        to_add = n//2 - ones_count
        indices = np.where(b == 0)[0]
        np.random.shuffle(indices)
        for i in range(to_add):
            b[indices[i]] = 1
    
    # compute min autocorrelation
    A = autocorr(b)
    m = np.min(A[1:])
    seeds.append((b.copy(), m))
    
    # 2. Random balanced binary
    b = np.zeros(n, dtype=int)
    ones = np.random.permutation(n)
    b[ones[:n//2]] = 1
    A = autocorr(b)
    m = np.min(A[1:])
    seeds.append((b.copy(), m))
    
    return seeds

def swap_improve(seed_b, max_iter=30000, seed=42):
    """Improve a binary seed using swap-based optimization."""
    np.random.seed(seed)
    b = seed_b.copy()
    n = len(b)
    S = n // 2
    
    # Pre-compute shifts for efficiency
    shifts = {}
    for k in range(n):
        shifts[k] = np.roll(b, k)
    
    # Compute initial autocorrelations
    A = autocorr(b)
    m = np.min(A[1:])
    worst_k = np.argmin(A[1:]) + 1
    
    iter_count = 0
    last_improve = 0
    
    temperature = 1.0
    cool_rate = 0.99
    
    while iter_count < max_iter and last_improve < 5000:
        # Find indices of 1s and 0s
        ones = np.where(b == 1)[0]  
        zeros = np.where(b == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Effect of swap at worst lag
        old_worst_val = A[worst_k]
        Delta = 0
        if b[i] != b[j]:
            # Compute delta for the current worst k
            Delta = (-shifts[worst_k][i] + shifts[worst_k][j])
            # Modify A according to this change
            A_delta = np.zeros(n)
            for k in range(n):
                if (i + k) % n != (j + k) % n:
                    A_delta[k] = -shifts[k][i] + shifts[k][j]
        else:
            Delta = 0
        
        if Delta > 0:
            # Definitely accept
            b[i], b[j] = b[j], b[i]
            # Update shifts if needed
            if Delta > 0:
                # Update correlation approximation
                A[worst_k] += Delta
                if A[worst_k] < m:
                    # Update the worst auto correlation value
                    m = A[worst_k]
                    worst_k = np.argmin(A[1:]) + 1
                else:
                    # Update all values, efficient to compute exact only when needed
                    A = autocorr(b)
                    m = np.min(A[1:])
                    worst_k = np.argmin(A[1:]) + 1
            last_improve = 0
        else:
            # Accept with probability exp(Delta / T)
            if np.random.rand() < np.exp(Delta / temperature):
                b[i], b[j] = b[j], b[i]
                A = autocorr(b)
                m = np.min(A[1:])
                worst_k = np.argmin(A[1:]) + 1
                last_improve = 0
            else:
                last_improve += 1
                
        temperature = max(0.001, temperature * cool_rate)
        iter_count += 1
        
    return b, m

def expand_vector(b, target_length):
    """Fold binary vector to a larger resolution."""
    b_expanded = np.zeros(target_length)
    step_size = target_length // len(b)
    if step_size == 0:
        return b
    for i in range(len(b)):
        for k in range(step_size):
            b_expanded[i * step_size + k] = b[i]
    return b_expanded

def interpolate_and_refine(b, n_final):
    """Interpolate and refine a binary solution."""
    b_expanded = expand_vector(b, n_final)
    
    # Optimization step 1: fine tune the transitions
    # Make some entries float (in (0,1) range) to improve the bound
    # We will do this for a small number of transitions
    b_float = b_expanded.copy()
    
    # Seemingly around 5% entries should be modified to float
    num_tune = max(1, int(0.05 * n_final))
    indices = np.random.choice(n_final, num_tune, replace=False)
    for idx in indices:
        if idx > 0 and idx < n_final-1:
            # Modify by a small amount, ensuring integral constraint
            if b_float[idx] != 0.5:
                b_float[idx] = 0.5
        else:
            b_float[idx] = 0.5
    
    # Use a gradient step for fine tuning
    dx = 2.0 / n_final
    current_C5 = evaluate_solution(b_float)
    # Perform a jiggle (approximate node adjustment)
    # Try small adjustments to see if we can reduce C5
    best_h = b_float.copy()
    best_C5 = current_C5
    
    for _ in range(20):
        trial_h = b_float.copy()
        idx = np.random.choice(n_final)
        delta = np.random.uniform(-0.1, 0.1)
        old_val = trial_h[idx]
        trial_h[idx] = np.clip(old_val + delta, 0, 1)
        # Project to satisfy sum constraint
        old_sum = np.sum(b_float) * dx
        new_sum = np.sum(trial_h) * dx
        scale_factor = (old_sum - 1) / (new_sum - old_sum) if abs(new_sum - old_sum) > 1e-10 else 0
        if abs(scale_factor) < 1:
            # adjust other points to rebalance
            trial_h = trial_h * (1 - scale_factor) + 0.5 * scale_factor
        c_val = evaluate_solution(trial_h)
        if c_val < best_C5:
            best_C5 = c_val
            best_h = trial_h.copy()
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Phase 1: Generate seeds (0-60s)
    n_seeds = 200
    seeds = initialize_seeds(n_seeds)
    best_seed = None
    best_m = -1
    
    # Compute autocorrelations for seeds and track best
    for i, (seed_b, _) in enumerate(seeds):
        if time.time() - start_time > 50:
            break
        A = autocorr(seed_b)
        m = np.min(A[1:])
        if m > best_m:
            best_m = m
            best_seed = seed_b.copy()
    
    # Phase 2: Swap-based optimization (60-360s)
    if time.time() - start_time > 300:
        return None, 1.0, 100
        
    visited_seeds = []
    best_b = None
    best_C5 = 1000.0
    n_best = 0
    
    for i, (b_seed, _) in enumerate(seeds):
        if time.time() - start_time > 350:
            break
            
        b_opt, m_opt = swap_improve(b_seed.copy(), max_iter=20000, seed=seed+i)
        A_opt = autocorr(b_opt)
        m_final = np.min(A_opt[1:])
        dx = 2.0 / len(b_opt)
        C5_opt = dx * (len(b_opt)/2 - m_final)
        
        if C5_opt < best_C5:
            best_C5 = C5_opt
            best_b = b_opt.copy()
            n_best = len(b_opt)
    
    # Phase 3: Scale up and refine (360-540s)
    if time.time() - start_time > 500:
        pass
    else:
        # Duplicate and refine
        n_target = min(350, n_seeds*2)
        b_final, C5_refined = interpolate_and_refine(best_b, n_target)
        best_C5 = min(best_C5, C5_refined)
        best_b = b_final
        n_best = n_target
    
    # Phase 4: Final check (540-590s)
    if time.time() - start_time > 580:
        pass
    else:
        dx = 2.0 / len(best_b)
        final_C5 = evaluate_solution(best_b)
        if final_C5 < best_C5:
            best_C5 = final_C5
    
    # Ensure integral constraint
    total_sum = np.sum(best_b) * (2.0 / len(best_b))
    if abs(total_sum - 1.0) > 1e-9:
        print(f"Warning: Sum constraint violation {total_sum} (should be 1.0)")
        
    return tuple(best_b), best_C5, len(best_b)
