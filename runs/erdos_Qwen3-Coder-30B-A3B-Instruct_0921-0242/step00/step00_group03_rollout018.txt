import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate initial solution from provided `initial_h_values` if available
    try:
        h_initial = np.asarray(initial_h_values)
        if len(h_initial) == n_points:
            h_current = h_initial.copy()
        else:
            # Resample if not matching
            h_current = np.interp(np.linspace(0, 2, n_points), 
                                    np.linspace(0, 2, len(initial_h_values)), 
                                    initial_h_values)
    except:
        h_current = np.zeros(n_points)
        h_current[:n_points//2] = 1  # Binary initial solution
        np.random.shuffle(h_current)
    
    c5_best = evaluate_erdos_solution(h_current)
    h_best = h_current.copy()
    
    # MILP
    def solve_milp(n):
        if n > 120:
            return None, None
        # Binary variables h_i
        h = cp.Variable(n, boolean=True)
        # Auxiliary variables y_{i,s}
        y = {}
        A_s = {}
        # For each shift s = 1,...,n-1
        for s in range(1, n):
            y[s] = cp.Variable((n - s), boolean=True)
            A_s[s] = cp.sum(y[s])
        
        # Variables for lower bound on all overlaps (min shift correlation)
        t = cp.Variable(integer=True, lower_bound=0)
        
        # Constraints
        constraints = []
        # Exact weight constraint
        constraints.append(cp.sum(h) == n // 2)
        
        # Product linearisation
        for s in range(1, n):
            for i in range(n - s):
                # y_{i,s} <= h_i, h_{i+s}
                constraints.append(y[s][i] <= h[i])
                constraints.append(y[s][i] <= h[i+s])
                # y_{i,s} >= h_i + h_{i+s} - 1
                constraints.append(y[s][i] >= h[i] + h[i+s] - 1)
        
        # Lower bounds on overlaps
        for s in range(1, n):
            constraints.append(t <= A_s[s])
        
        # Objective
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solver settings
        solver_time_limit = min(300, budget_s - (time.time() - start_time) - 10)
        if solver_time_limit < 10:
            return None, None
        
        try:
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=solver_time_limit)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = np.round(h.value).astype(int)
                t_opt = t.value
                # Return integer solution and t value
                return h_opt, t_opt
            else:
                return None, None
        except:
            return None, None
    
    # Try MILP with n=100
    h_opt, t_opt = solve_milp(100)
    if h_opt is not None:
        # Use this as starting point
        h_current = h_opt
        c5_current = 1 - 2 * t_opt / 100.0
        if c5_current < c5_best:
            c5_best = c5_current
            h_best = h_current.copy()

    # Try MILP with n=120 if time allows
    if time.time() - start_time < budget_s - 10:
        h_opt, t_opt = solve_milp(120)
        if h_opt is not None:
            h_current = h_opt
            c5_current = 1 - 2 * t_opt / 120.0
            if c5_current < c5_best:
                c5_best = c5_current
                h_best = h_current.copy()
    
    # Convert to continuous representation with dx = 2/n
    h_continuous = h_best.astype(float)
    
    # Binary hill-climbing local improvement
    def compute_all_overlaps(h_vec):
        # Compute full autocorrelations allowing for shifts s = 1..n-1
        # return list of Corr(h,h,s) for s=1 to n-1
        corr = []
        for s in range(1, len(h_vec)):
            if s <= len(h_vec) - 1:
                # h[i] * h[i+s] for i = 0 .. (n-1-s)
                acc = sum(h_vec[i] * h_vec[i+s] for i in range(len(h_vec)-s))
                corr.append(acc)
            else:
                corr.append(0)
        return corr
    
    def compute_min_overlap(h_vec):
        corr = compute_all_overlaps(h_vec)
        return min(corr) if corr else 0
    
    # Local swapping (hill climbing)
    max_iters = 2000
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 5:
            break
        # Pick two indices: one with 1 and one with 0
        ones = np.where(h_continuous == 1)[0]
        zeros = np.where(h_continuous == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        
        # Simulate swap
        h_test = h_continuous.copy()
        h_test[i1] = 0
        h_test[i0] = 1
        t_before = compute_min_overlap(h_current)
        t_after = compute_min_overlap(h_test)
        
        if t_after > t_before:
            h_current = h_test.copy()
            c5_current = 1 - 2 * t_after / len(h_current)
            if c5_current < c5_best:
                c5_best = c5_current
                h_best = h_current.copy()
    
    # Optional: SLSQP polishing if close to target
    if c5_best <= 0.3807 and time.time() - start_time < budget_s - 30:
        def obj_continuous(h_array):
            # Normalize
            h_array = h_array * (1.0 / np.sum(h_array) * n_points * dx)
            corr = np.correlate(h_array, 1 - h_array, mode="full")
            corr = corr[len(corr)//2+1:]  # non-zero shifts only
            return 1.0 - np.min(corr) * dx
        
        def constraint_sum(h_array):
            return np.sum(h_array) * dx - 1.0
        
        bounds = [(0, 1) for _ in range(len(h_best))]
        constraints = {"type": "eq", "fun": constraint_sum}
        
        # Add some noise to start
        h_init = h_best + np.random.normal(0, 0.01, len(h_best))
        h_init = np.clip(h_init, 0, 1)
        
        result = minimize(obj_continuous, h_init, method="SLSQP", 
                          bounds=bounds, constraints=constraints, 
                          options={"maxiter": 100})
        h_final = result.x
        h_final = h_final * (1/np.sum(h_final) * n_points * dx)
        c5_final = evaluate_erdos_solution(h_final)
        if c5_final < c5_best:
            c5_best = c5_final
            h_best = h_final.copy()
    
    # Upsample if not already done
    final_n = n_points
    h_upsampled = h_best.copy()
    if final_n <= 800:
        final_n = 800
        repeat_factor = final_n // len(h_best)
        h_upsampled = np.repeat(h_best, repeat_factor)
        c5_best = evaluate_erdos_solution(h_upsampled)
    
    h_rescaled = h_upsampled / np.sum(h_upsampled) * 2
    
    return (h_rescaled, c5_best, len(h_rescaled))
