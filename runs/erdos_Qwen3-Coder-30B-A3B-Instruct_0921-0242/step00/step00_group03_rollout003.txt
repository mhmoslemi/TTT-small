import numpy as np
import cvxpy as cp
import time
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h array."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation with 1-h (full mode)
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum of these is the C5 bound
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    """Return initial construction: modified binary solution."""
    n = 70
    h = np.zeros(n)
    # Fill with 1s and 0s to make sum = n/2 = 35
    h[:35] = 1
    return h

def create_binary_milp(n, max_time_s=None):
    """Create and solve the binary MILP formulation for max-min overlap."""
    # Initialize variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True, lower_bound=0)

    # Objective to maximize min overlap
    objective = cp.Maximize(t)

    constraints = []
    
    # Sum constraint: exactly n/2 ones
    constraints.append(cp.sum(h) == n // 2)

    # Linearization of products y[i,s] = h[i]*h[i+s]
    y = {}
    for s in range(1, n):
        for i in range(n - s):
            y[(i, s)] = cp.Variable(boolean=True)
            # McCormick envelopes
            constraints.append(y[(i, s)] <= h[i])
            constraints.append(y[(i, s)] <= h[i + s])
            constraints.append(y[(i, s)] >= h[i] + h[i + s] - 1)
            # Add to overlap sum
            # We'll add it later as part of constraints for shift s

    # Bounds on t based on overlaps
    for s in range(1, n):
        overlap = cp.sum([y[(i, s)] for i in range(n - s)])
        constraints.append(t <= overlap)

    # Create problem
    problem = cp.Problem(objective, constraints)

    # Solve with time limit
    if max_time_s is not None:
        solver_options = {'max_seconds': max_time_s}
    else:
        solver_options = {}
        
    try:
        problem.solve(solver=cp.CBC, **solver_options, verbose=False)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.round(h.value).astype(int)
            if np.sum(h_sol) != n // 2:
                # Check if we have a feasible but not valid solution
                h_sol = np.zeros(n, dtype=int)
                h_sol[:n//2] = 1
            # Return solutions and slack
            t_value = int(t.value)
            return h_sol, t_value, problem.status
        else:
            # Fallback to initialization
            h_sol = np.zeros(n, dtype=int)
            h_sol[:n//2] = 1
            return h_sol, 0, problem.status
    except Exception as e:
        # Fallback
        h_sol = np.zeros(n, dtype=int)
        h_sol[:n//2] = 1
        return h_sol, 0, cp.INFEASIBLE

def binary_hill_climbing(h_bin, n, max_time_s):
    """Run hill-climbing to locally improve the binary solution."""
    # Reconstruct array
    h = h_bin.copy()
    # Compute initial overlaps and minimum
    A = {}
    # Precompute overlaps for each shift
    t_min = float('inf')
    for s in range(1, n):
        A_s = 0
        for i in range(n - s):
            A_s += h[i] * h[i + s]
        A[s] = A_s
        t_min = min(t_min, A_s)
    t_min = int(t_min)

    start_time = time.time()
    while time.time() - start_time < max_time_s:
        # Select random pair to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        
        # Temporarily change values
        h[i1] = 0
        h[i0] = 1
        
        # Update A_s for affected shifts
        old_min = t_min
        updated = set()
        new_min = float('inf')
        
        # Compute new overlaps for all shifts affected by the swap
        for s in range(1, n):
            # Check if indices i1 or i0 or i1+s or i0+s are involved
            affected = False
            if i1 < n - s and i0 < n - s:
                # Update only in case there's overlap (axes of overlap)
                if i1 + s < n and (i1 + s == i0 or i0 + s == i1):
                    affected = True
            if i1 < n - s and i0 < n - s:
                if i1 == i0 + s:
                    # i1 is at i0+s
                    affected = True
                if i0 == i1 + s:
                    # i0 is at i1+s
                    affected = True
            # Simplified: just compute all overlaps near affected indices
            if not affected:
                if abs(i1 - i0) <= 2*s or abs(i1 + s - i0) <= 2*s:
                    affected = True
            if affected:
                new_A_s = 0
                for i in range(n - s):
                    new_A_s += h[i] * h[i + s]
                A[s] = new_A_s
                new_min = min(new_min, new_A_s)
            else:
                # This is a strict subset of the values
                # Optimization: do not update A if out of bounds
                # Just trust that it is computed properly after
                pass
                
        new_min = min(A.values()) if A else 0
        if new_min > old_min:
            t_min = int(new_min)
        else:
            # Undo
            h[i1] = 1
            h[i0] = 0

    # Return final vector and t value
    return h, t_min

def polish_continuous(h, max_time_s):
    """Refine the solution using continuous optimization."""
    n = len(h)
    dx = 2.0 / n
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec)

    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0},
        {'type': 'ineq', 'fun': lambda x: x},
        {'type': 'ineq', 'fun': lambda x: 1.0 - x}
    ]

    # Add slight noise
    h_noisy = np.clip(h + np.random.randn(n) * 0.01, 0, 1)
    
    # Run SLSQP
    from scipy.optimize import minimize

    start_time = time.time()
    while time.time() - start_time < max_time_s:
        try:
            res = minimize(objective, h_noisy, method='SLSQP', constraints=cons, bounds=[(0, 1)]*n, options={'maxiter': 50})
            if res.success:
                return res.x, objective(res.x)
            else:
                break
        except:
            break

    # Default return
    return h, evaluate_erdos_solution(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # First test n = 100
    n = 100
    if budget_s < 400:
        n = 80
        
    # Solve MILP
    remaining_budg = budget_s - (time.time() - start_time)
    # Ensure at least a small time for next phases
    solver_time = min(remaining_budg // 2, 300)
    
    h_bin, t_min, status = create_binary_milp(n, solver_time)
    if status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        # fallback solution
        h_bin = initial_h_values()
        t_min = 0
        
    # Local binary search
    remaining_budg = budget_s - (time.time() - start_time)
    if remaining_budg > 50:
        hill_time = min(remaining_budg // 2, 100)
        h_bin, t_min = binary_hill_climbing(h_bin, n, hill_time)
    
    # Evaluate best solution found so far
    c5_bound = 1 - 2 * t_min / n

    # Optional continuous polishing if time permits and C5 is close to target
    remaining_budg = budget_s - (time.time() - start_time)
    if remaining_budg > 30 and c5_bound > 0.38:
        h_cont, polished = polish_continuous(h_bin, 30)
        if polished < c5_bound:
            h_bin = h_cont
            c5_bound = polished

    # Try larger n = 120 if we're under budget
    if budget_s - (time.time() - start_time) > 100:
        n2 = 120
        # Re-solve MILP with larger resolution
        remaining_budg = budget_s - (time.time() - start_time)
        solver_time = min(remaining_budg // 2, 150)
        h_bin2, t_min2, status = create_binary_milp(n2, solver_time)
        if status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Check if new t_min is better
            c5_2 = 1 - 2 * t_min2 / n2
            if c5_2 < c5_bound:
                c5_bound = c5_2
                h_bin = h_bin2
                n = n2

    # Final upsampling if we're still far off target
    if n <= 150:
        upsample = 3 if n <= 100 else 2
        h_upsampled = np.repeat(h_bin, upsample)
        n_point = len(h_upsampled)
        dx_up = 2.0 / n_point
        final_c5 = evaluate_erdos_solution(h_upsampled)
        return (h_upsampled, final_c5, n_point)
    else:
        dx = 2.0 / n
        final_c5 = evaluate_erdos_solution(h_bin)
        return (h_bin, final_c5, n)
