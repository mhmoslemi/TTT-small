```python
import time
import numpy as np
from scipy import signal
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdos overlap constant C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using full convolution
    overlap = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is dx * max_k overlap[k]
    max_overlap = np.max(overlap)
    return max_overlap * dx

def initial_h_values():
    """Return the initial construction from which to start."""
    # This is inherited from environment; typically a step function
    # for n=47 points yielding an initial C5 ≈ 0.38092
    h = np.array([0.0]*24 + [1.0]*23)  # simple 0/1 construction
    # normalize to integral of 1
    h = h * 2.0 / np.sum(h)
    return h.tolist()

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Parameters
    n0 = 120  # Base grid size for exact optimization
    target_n = 360  # Final output size
    dx0 = 2.0 / n0
    
    # Model parameters
    S = n0 / 2  # Required sum (integral constraint)
    
    # ========== Construct Model ==========
    print(f"Constructing MILP model with n0={n0}")
    
    # Variables
    b = cp.Variable(n0, boolean=True)
    w = cp.Variable()
    
    # Define Z matrix elements, but use sparse constraints
    
    # Objective: minimize w
    objective = cp.Minimize(w)
    
    # Constraints
    constraints = []
    
    # Integral constraint
    constraints.append(cp.sum(b) == S)
    
    # Fix first entry for symmetry breaking
    constraints.append(b[0] == 1)
    
    # Create products z[i,k] = b[i] * b[(i+k) % n0]
    # Precompute all constraints now (all between 0 and n0-1)
    
    # Add the three linear constraints for each z[i,k]
    # The constraint sets: z[i,k] <= b[i], z[i,k] <= b[i+k], z[i,k] >= b[i] + b[i+k] - 1
    
    # For efficient handling, we'll build a list of constraints by iterating over pairs
    z_vars = {}
    for i in range(n0):
        for k in range(n0):
            z_vars[(i, k)] = cp.Variable(boolean=True)
    
    # Add exact product constraints
    for i in range(n0):
        for k in range(n0):
            z_vars[(i, k)] = cp.Variable(boolean=True)
            
    for i in range(n0):
        for k in range(n0):
            ik = (i, k)
            j = (i + k) % n0
            z = z_vars[ik]
            
            # z <= b[i]
            constraints.append(z <= b[i])
            # z <= b[j]
            constraints.append(z <= b[j])
            # z >= b[i] + b[j] - 1
            constraints.append(z >= b[i] + b[j] - 1)
    
    # Add overlap constraints
    for k in range(n0):
        some_sum = 0
        for i in range(n0):
            some_sum += z_vars[(i, k)]
        # w >= S - some_sum  -->  some_sum >= S - w
        # i.e. more compactly:
        constraints.append(some_sum >= S - w)
    
    # Normalize the bound
    constraints.append(w >= 0)
    constraints.append(w <= S)
    
    # ========== Solve ==========
    print(f"Creating and solving model...")
    prob = cp.Problem(objective, constraints)
    
    # time remaining
    time_left = budget_s - (time.time() - start_time)
    if time_left < 1.0:
        # Fall back to a greedy solution now.
        return get_greedy_solution(n0, target_n, start_time, budget_s)
    
    try:
        # Use default solver (=CBC) with time limit
        prob.solve(solver=cp.CBC, maximum_seconds=time_left)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            bl = b.value
            w_opt = w.value
            print(f"Optimal solution found with w = {w_opt}, C5 = {w_opt*dx0}")
        else:
            # fallback to greedy
            print(f"Solver status: {prob.status}; using greedy fallback")
            return get_greedy_solution(n0, target_n, start_time, budget_s)
    except Exception as e:
        print(f"MIP Solver error: {e}; using greedy fallback.")
        return get_greedy_solution(n0, target_n, start_time, budget_s)
    
    # ========== Post-process (refinement only if needed) ==========
    if bl is None:
        return get_greedy_solution(n0, target_n, start_time, budget_s)
    
    # Convert b to list of floats (0 or 1) and refine
    b_bin = [float(x) for x in bl]
    
    # Refinement using swaps to improve worst autocorrelation
    # Only if condition holds or time is available
    
    time_left = budget_s - (time.time() - start_time)
    if time_left > 2:
        b_bin = refine_solution(b_bin, n0)
    
    # ========== Up-sampling (duplication) ==========
    m = target_n // n0
    h_final = []
    for j in range(target_n):
        h_final.append(b_bin[j // m])
    
    # Verify integral constraint
    assert abs(np.sum(h_final) * (2.0 / target_n) - 1.0) < 1e-9
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_final)
    
    return h_final, final_c5, target_n

def get_greedy_solution(n0, target_n, start_time, budget_s):
    """Greedy version as fallback when MIP doesn't converge."""
    # The basic greedy: start from empty SET, at each step
    # pick a position that minimizes the change in worst overlap
    # If not time-heavy, expand step.
    # Fallback with local improvement.
    
    # Build initial empty set
    b = [0] * n0
    count = 0  # number of 1s
    target = n0 // 2
    used = set()
    
    while count < target and (time.time() - start_time) < 0.8 * budget_s:
        gains = []  # delta奂 if we flip this bit
        
        # compute overlaps for current b
        overlaps = compute_overlaps(b, n0)
        min_k = np.argmin(overlaps)
        best_score = overlaps[min_k]
        
        # try to toggle each position
        for i in range(n0):
            if i in used:
                continue
            orig = b[i]
            b[i] = 1 - orig
            overlaps_t = compute_overlaps(b, n0)
            new_score = overlaps_t[min_k]
            b[i] = orig  # reset
            
            gains.append((new_score - best_score, i))
        
        if not gains:
            break # early exit
        gains.sort()
        i_t = gains[0][1]
        b[i_t] = 1 - b[i_t]
        used.add(i_t)
        count += (1 if b[i_t] == 1 else -1)
    
    # Refinement with binary swaps
    b_bin = b[:]
    time_left = budget_s - (time.time() - start_time)
    if time_left > 0.1:
        b_bin = refine_solution(b_bin, n0)
    
    # Duplication
    m = target_n // n0
    h_final = []
    for j in range(target_n):
        h_final.append(b_bin[j // m])
    
    final_c5 = evaluate_erdos_solution(h_final)
    
    return h_final, final_c5, target_n

def compute_overlaps(b, n):
    """Efficiently compute all overlaps O_k."""
    h = np.array(b)
    A = np.correlate(h, h, mode='full')
    S = np.sum(h)
    O = S - A
    return O[n//2:n//2+n]  # return just for non-negative lags

def refine_solution(b, n):
    """Perform deterministic refinement to improve worst overlap."""
    MAX_SWAPS = 200
    
    # Make sure it's working with numpy array
    b = np.array(b, dtype=int)
    best = b.copy()
    overlaps = compute_overlaps(best, n)
    current_min = np.min(overlaps)
    
    changed = True
    swaps = 0
    
    while changed and swaps < MAX_SWAPS:
        changed = False
        for p in range(n):
            if best[p] == 0:
                continue
            for q in range(n):
                if best[q] == 1:
                    continue
                # Swap p and q
                best[p], best[q] = 0, 1
                new_overlaps = compute_overlaps(best, n)
                new_min = np.min(new_overlaps)
                if new_min > current_min:
                    current_min = new_min
                    changed = True
                # Restore
                best[p], best[q] = 1, 0
                
        swaps += 1
    return best.tolist()
```
