def evaluate_erdos_solution(h_values, n_points):
    # computes C5 = max_k int h(x)(1 - h(x+k)) dx
    dx = 2.0 / n_points
    h = np.array(h_values)
    # compute all correlations (full mode)
    corr = correlate(h, 1 - h, mode='full')
    # take only the necessary part (k from -n+1 to n-1)
    # but we care only about shifts from 0 to n-1 (this is the shift used in the definition)
    # for shift k, correlation is at index n-1+k = n-1+(n-1-k) = 2*(n-1)-k (in full mode)
    # to get correlation for shift k -> index is n-1-k (for positive k)
    # but correlate with mode full gives a long vector. Let's recompute properly.
    # correlation at shift k is:
    #   sum_{i=0}^{n-1} h[i] * (1 - h[(i+k)%n])
    #   = sum h[i] - sum h[i]*h[(i+k)%n] = n - sum h[i]*h[(i+k)%n]
    # which we compute efficiently with circular autocorrelation
    # result should be [0, n-1, n-2, ..., 2, 1, 0, 1, 2, ..., n-2, n-1]
    # indices [0, ..., n-1, n, ..., 2*n-2]
    # for shift k in [0, n-1], which is what we're interested in,
    # -> we should take corr[n-1-k] for k in [0, n-1]
    # Actually let me just use c = correlate(h, 1-h, mode='full') and get c[n-1:n-1+n]
    # which is indices [n-1, ..., n-1+n-1] = [n-1, ..., 2*n-2]
    # we say:
    # shift k (in range [0, n-1]) corresponds to index in corr
    # that is 2*n-2 - k = n-1 + (n-1-k) = n-1 - k + n-1
    # No, let's reindex:
    # if shift is k in [0, n-1], we want the k-th element of result from numpy
    # i.e., we want correlate(i, j, mode='full') with the i shifted by -k
    # so we want indices [n-1 - k, ..., n-1 + (n-1-k)] = [n-1 - k, 2*n-2-k]
    #    so at 0 is k = n-1, at n-1 is k = 0  
    # CONFIRMED:
    # corr[full_index] = sum_{i=0}^{n-1} h[i] * (1 - h [(i + full_index - (n-1)) % n])
    # which becomes:
    # sum_{i=0}^{n-1} h[i] * (1 - h [(i + full_index - n + 1) % n])
    # so for k=0: full_index = n-1, h[i] (1-h[i+0]) = h[i](1-h[i]) = h[i] - h[i]^2
    # for k=1: full_index = n-2, h[i] (1-h[i+1]) 
    # and sum_{i=0}^{n-1} h[i] (1-h[(i+2*n-2 - (n-1)) % n]) = h[i](1-h[i-1]) ...
    # for k=0   -> full_index = n-1, shift k = 0
    # for k=n-1 -> full_index = n-1 - (n-1) = 0, shift k = n-1
    # So from full correlation:
    #     shift k = 0..n-1 -> full index = 2(n-1)-k,  i.e., indices = [n-1, n, ..., 2*n-2]
    # or:  at shift k = 0..n-1 we take correspondng entry corr[2*(n-1) - k]
    # So:
    # for k in range(n):
    #    c = corr[n-1+k]
    #    O_k = n - c :  integral of h* (1-h shifted by k)
    # Or (equivalently):
    # for k in range(n):
    #   corr_pool = corr[2*(n-1) - k]  = sum h[i] h[i+k] (circular)
    #   O_k = n - corr_pool =  sum h[i] (1 - h[i+k])
    # This is correct. So we take corr[n-1] through corr[2*n-2]
    # i.e., corr[n-1:n]
    corr_for_shifts = corr[n-1:n-1+n]
    max_overlap = np.max(corr_for_shifts)  # this is the overlaps at all shifts
    # return the overlap integral (to assign to C5 value)
    return max_overlap * dx

    #   sum_{i} z[i,k] <= S - w --- for all k
    #  
    # So we introduce z[i,k] auxiliary binary vars
    # add linearization constraints: 
    # z[i,k] <= b[i]
    # z[i,k] <= b[i+k]
    # z[i,k] >= b[i] + b[i+k] - 1
    
    # to keep memory manageable, generate constraints on the fly.
    # Use a model-based approach: 
    def build_constraints():
        # w is a scalar, where w >= S - sum b[i]*b[i+k] for all k
        constraints = []
        # add mass constraint
        constraints.append(sum(b) == S)

        # O_k = O_{n0-k}
