import numpy as np
import cvxpy as cp
from scipy.sparse import diags
from scipy.sparse.linalg import eigs
import time
from cvxpy import SCS

def project_to_simplex(v, target_sum):
    """Project vector v onto the capped simplex with sum = target_sum."""
    v = np.asarray(v)
    low, high = 0.0, 1.0
    theta_lo, theta_hi = 0.0, 1.0

    def proj_sum(theta):
        return np.sum(np.clip(v - theta, low, high)) - target_sum

    # Bisection to find the right theta
    for _ in range(50):
        theta = (theta_lo + theta_hi) / 2
        s = proj_sum(theta)
        if s > 0:
            theta_lo = theta
        else:
            theta_hi = theta

    return np.clip(v - theta, low, high)

def solve_sdp(n, max_iters=4000):
    """Solve the semidefinite program to find a lower bound on C5."""
    dx = 2.0 / n
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)

    constraints = [
        X >> 0,  # X is positive semidefinite
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2,
    ]
    
    for i in range(n):
        constraints.append(X[i, i] == h[i])
        
    # Additional constraints
    for i in range(n):
        for j in range(n):
            if i != j:
                constraints.append(X[i, j] <= h[i])
                constraints.append(X[i, j] <= h[j])
                constraints.append(X[i, j] >= 0)

    # Compute all shift matrices M_k
    Mk = []
    for k in range(n):
        if k == 0:
            Mk.append(diags(np.ones(n), format='coo'))
        else:
            if k < n:
                d = np.ones(n - k)
                Mk.append(diags(d, k, shape=(n, n), format='coo'))
            else:
                Mk.append(diags(np.zeros(0), k, shape=(n, n), format='coo'))

    L = cp.Variable()
    for k in range(n):
        Mk_sparse = Mk[k]
        # Trace(Mk * X) = sum of diagonal elements of Mk * X
        # which is just sum over valid indices of X[i, i+k]
        trace = cp.sum(cp.multiply(Mk_sparse, X))  # Using element-wise product
        constraints.append(trace >= L)
    
    objective = cp.Maximize(L)
    problem = cp.Problem(objective, constraints)
    
    try:
        problem.solve(solver=SCS, max_iters=max_iters, eps=5e-4)
    except:
        return None, None

    if problem.status == cp.OPTIMAL:
        X_sol = X.value
        h_sol = h.value
        
        # Now due to sparse structure, we recompute
        # Find dominant eigenpair
        try:  # Use sparse matrix for eigenvectors
            evals, evecs = eigs(X_sol, k=1, which='LM')
            v = np.real(evecs[:, 0])
            l = np.real(evals[0])
            if l > 0:
                h_feas = np.sqrt(l) * v
                h_feas = project_to_simplex(h_feas, n / 2)
                return h_feas, 1 - L.value
            else:
                return None, None
        except:
            return None, None
    else:
        return None, None

def gradient_polish(h, n_points, max_iter=500):
    """Perform projected gradient descent to polish the solution."""
    h = np.array(h)
    dx = 2.0 / n_points
    max_iter = min(max_iter, 1000)

    for t in range(max_iter):
        # Compute optimal lag k*
        k_star = 0
        max_overlap = -np.inf
        for k in range(n_points):
            # Compute ∫ h(x)(1 - h(x + k))
            overlap = 0.0
            if k == 0:
                overlap = np.dot(h, 1 - h)
            else:
                # Convolution manually computed and truncated
                h_shift = np.zeros(n_points)
                if k < n_points:
                    h_shift[k:] = h[:-k]
                else:
                    h_shift[:] = 0.0
                overlap = np.dot(h, 1 - h_shift)

            if overlap > max_overlap:
                max_overlap = overlap
                k_star = k
        
        # Construct the gradient vector g
        g = np.zeros(n_points)
        if k_star < n_points:
            # Non-negative lag -- safe to shift
            if k_star == 0:
                g = 1 - h  # Not shift
                # How the LaTeX was interpreted: -g = 1 - h for k=0 (should be just 1-h after correction to replace g with -g afterward to match formulation).
                # We're using -g := (1 - h) for this case
                # But what we want is partial derivative of integral w.r.t. h_i = 1 - h_i+k
                # Which means g_i = h_{i-k} - (1-h_{i+k})
                # Let's compute properly:
                g = np.zeros_like(h)
                g[:-k_star] = h[k_star:]  # g[i] = h[i + k_star]
                g[k_star:] = 1.0 - h[:-k_star]  # g[i+k_star] = 1 - h[i]
        # Now re-do the correct derivation
        # Let’s define g correctly for lags
        g = np.zeros(n_points)
        if k_star == 0:
            h_sk = np.zeros(n_points)
            h_sk[:] = h
            g = 1 - h_sk  # Influence of shift by 0
        else:
            # h_{k_star} -> gets used by h_i in the term...
            # So left part h_i * (1 - h_{i+k_star}) means
            # Derivative w.r.t. h_i : 1 - h_{i+k_star}
            # Do we effectively have g_i = 1 - h_{i+k_star} ?
            # Fixing the notation:
            g_left = np.zeros(n_points)
            g_right = np.zeros(n_points)
            g_left[k_star:] = 1.0
            g_right[k_star:] = -h[:-k_star]
            g = g_left + g_right

        # Project the gradient step into feasible set
        step = 0.1 / np.sqrt(t + 1)
        h_new = h - step * g
        h_new = project_to_simplex(h_new, n_points / 2)
        
        h = h_new.copy()
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # First pass: Coarse grid
    n_coarse = 120
    if time.time() - start_time + 10 < budget_s:
        h_sol_coarse, c5_coarse = solve_sdp(n_coarse)
        if h_sol_coarse is not None and c5_coarse is not None:
            if c5_coarse < best_c5:
                best_c5 = c5_coarse
                best_h = h_sol_coarse.copy()
                best_n = n_coarse
    
    # Medium pass
    n_medium = 200
    if time.time() - start_time + 10 < budget_s:
        h_sol_medium, c5_medium = solve_sdp(n_medium)
        if h_sol_medium is not None and c5_medium is not None:
            if c5_medium < best_c5:
                best_c5 = c5_medium
                best_h = h_sol_medium.copy()
                best_n = n_medium
    
    # Fine pass
    n_fine = 300  # If we still have time
    if time.time() - start_time + 10 < budget_s:
        h_sol_fine, c5_fine = solve_sdp(n_fine)
        if h_sol_fine is not None and c5_fine is not None:
            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_sol_fine.copy()
                best_n = n_fine
    
    # Apply gradient polishing if we have a valid candidate
    if best_h is not None and best_c5 < float('inf'):
        # Interpolate to finest resolution if necessary
        if best_n > 120:
            h_interp = np.interp(np.linspace(0, 2, 300), np.linspace(0, 2, best_n), best_h)
            best_h = h_interp
        
        polished_h = gradient_polish(best_h, 300)
        # Perfoming local evaluation
        dx = 2.0 / 300.0
        overlaps = np.correlate(polished_h, 1 - polished_h, mode='full') * dx
        final_c5 = np.max(overlaps)
        
        return (polished_h.tolist(), float(final_c5), 300)
    else:
        return (np.ones(300) * 0.5, 0.5, 300)
