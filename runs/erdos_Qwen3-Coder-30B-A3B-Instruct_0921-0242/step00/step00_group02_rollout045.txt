import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import math

# Helper functions

def evaluate(h):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Max over all shifts
    return np.max(corr) * dx

def project_feasible(x, target):
    """Project x onto {0 ≤ h ≤ 1, sum h = target} using water-filling"""
    # Ensure bounds
    h = np.clip(x, 0, 1)
    # Waterfill to get exact sum
    diff = target - np.sum(h)
    n = len(h)
    # Repeat water-filling at most 10 times
    for _ in range(10):
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to raise values
            free = h < 1
            if not np.any(free):
                break
            num_free = np.sum(free)
            add = diff / num_free
            if add > 1 - np.min(h[free]):
                # Up to 1
                h[free] = 1
                diff -= np.sum(1 - h[free])
            else:
                h[free] += add
                diff = 0
        else:
            # Diff < 0, need to lower values
            free = h > 0
            if not np.any(free):
                break
            num_free = np.sum(free)
            sub = -diff / num_free
            if sub > np.min(h[free]):
                # Down to 0
                h[free] = 0
                diff += np.sum(h[free])
            else:
                h[free] -= sub
                diff = 0
    return h

def active_shifts(h, eps=1e-9):
    """Return shift indices k where overlap is within eps of current C5"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    ks = []
    for i in range(len(corr)):
        if abs(corr[i] - max_overlap) < eps:
            ks.append(i)
    return ks

def gradient(h, ks):
    """Calculate sub-gradient for active shifts"""
    dx = 2.0 / len(h)
    g = np.zeros_like(h)
    n = len(h)
    # For each active shift k
    for k in ks:
        # d/dh_i of the overlap at shift k
        # which is sum_j h[j] * (1 - h[j+k])
        # So d/dh_i = (1 - h[i+k]) if k >= 0 and i+k < n else 0
        # and   d/dh_i = -h[i-k] if k <= 0 and i-k >= 0 else 0
        # So d/dh_i = (1 - h[i+k]) * (i+k < n and k >= 0) - h[i-k] * (i-k >= 0 and k <= 0)
        for i in range(n):
            if i + k < n and i + k >= 0:
                g[i] += (1 - h[i + k])
            if i - k >= 0 and i - k < n:
                g[i] -= h[i - k]
    return g / len(ks)

def line_search(h, g, t_current, budget):
    """Try a few step sizes and take the best descent step within budget"""
    dx = 2.0 / len(h)
    step_sizes = [0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1]
    best_h = None
    best_val = t_current
    t0 = time.time()
    for alpha in step_sizes:
        if time.time() - t0 > budget:
            break
        if alpha == 0:
            h_new = h.copy()
        else:
            h_new = h - alpha * g
        h_new = project_feasible(h_new, 1.0)  # target is the integral constraint
        val = evaluate(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
    if best_h is None:
        return h, t_current
    return best_h, best_val

def local_optimize(h0, max_iter, t_end):
    """Projected gradient descent with line search"""
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate(h)
    dx = 2.0 / len(h)
    
    # Relax maximum iterations
    iter = 0
    while iter < max_iter and time.time() < t_end:
        ks = active_shifts(h, 1e-9)  # strict equality with max overlap
        g = gradient(h, ks)
        h, val = line_search(h, g, evaluate(h), 0.1)  # short line search
        if val < best_val:
            best_h = h.copy()
            best_val = val
        iter += 1
    return best_h, best_val

def anneal(h0, t_end):
    """Simple simulated annealing around a current solution"""
    h = h0.copy()
    current_val = evaluate(h)
    best_h = h.copy()
    best_val = current_val
    t0 = 0.01
    t = t0
    decay_rate = 0.995
    
    tries = 0
    while time.time() < t_end and tries < 5000:
        # Pick two indices at random
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        if i == j:
            continue
        delta = np.random.rand() * 0.05  # max 0.05 shift
        if np.random.rand() < 0.5:
            delta = -delta
        # Store old h and compute new h
        old_i = h[i]
        old_j = h[j]
        new_i = old_i + delta
        new_j = old_j - delta
        new_h = h.copy()
        new_h[i] = new_i
        new_h[j] = new_j
        new_h = project_feasible(new_h, 1.0)
        new_val = evaluate(new_h)
        # Acceptance probability
        if new_val < current_val or np.random.rand() < np.exp(-(new_val - current_val) / t):
            h = new_h
            current_val = new_val
            if current_val < best_val:
                best_h = h.copy()
                best_val = current_val
        t *= decay_rate
        tries += 1
    return best_h, best_val

def random_feasible(n):
    """Generate a random feasible vector"""
    h = np.random.rand(n)
    # Scale so that integral = 1.0
    dx = 2.0 / n
    h = h * (1.0 / np.sum(h) * dx)  # such that sum(h)*dx = 1
    h = project_feasible(h, 1.0)
    return h

def build_initial(n, initial_h_values):
    """Interpolate or build initial h for a given n"""
    if n == len(initial_h_values):
        return initial_h_values.copy()
    if n < len(initial_h_values):
        # Downsample
        ratio = len(initial_h_values) / n
        h_new = np.zeros(n)
        for i in range(n):
            start = int(i * ratio)
            end = int((i+1) * ratio)
            if end > len(initial_h_values):
                end = len(initial_h_values)
            h_new[i] = np.mean(initial_h_values[start:end])
    else:
        # Upsample
        h_new = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    h_new = project_feasible(h_new, 1.0)
    return h_new
