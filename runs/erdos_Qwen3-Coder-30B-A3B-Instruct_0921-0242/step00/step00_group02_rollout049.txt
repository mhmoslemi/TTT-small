    def solve_sdp(n):
        # SDP formulation
        X = cp.Variable((n, n), symmetric=True)
        h = cp.Variable(n)
        s = cp.Variable()
        dx = 2.0 / n
        
        constraints = [
            cp.diag(X) == h,
            X >> 0,
            0 <= h,
            h <= 1,
            cp.sum(h) == n / 2,
            s <= 0  # will update constraints
        ]
        
        # Add correlation constraints
        for k in range(1, n):
            # sum of X[i,i+k] for i=0..n-1-k
            expr = cp.sum([X[i, i+k] for i in range(n-k)])
            constraints.append(expr >= s)
        
        objective = cp.Maximize(s)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, eps=1e-4, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_opt = h.value
                X_opt = X.value
                s_opt = s.value
                return h_opt, X_opt, s_opt
            else:
                return None, None, None
        except:
            return None, None, None

                X = eigvecs @ np.diag(eigvals) @ eigvecs.T

                if z[i] >= threshold:
                    h_round[i] = 1.0
                else:
                    h_round[i] = 0.0

            if not np.isclose(np.sum(h_round), n/2):
                # Fix by flipping entries
                diff = int(n/2 - np.sum(h_round))

                    # Flip smallest entries to 1
                    indices = np.where(h_round == 0)[0]

                    indices = np.where(h_round == 1)[0]
                    indices = indices[np.argsort(-z[indices])][:abs(diff)]
                    for i in indices:
                        h_round[i] = 0.0
            assert np.isclose(np.sum(h_round), n/2), "Mass constraint failed after rounding"

            score = evaluate_c5(h_round, dx)

                best_score = score
                best_h = h_round.copy()
                
        return best_h, best_score

        overlaps = np.correlate(h, 1 - h, mode="full")
        # Multiplied by dx to get proper scaling
        overlaps *= dx
        # Returns max of all shifts
        return np.max(overlaps)
