def c5_of_h(h):
    """Compute C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full convolution
    conv = np.correlate(h, g, mode='full')
    # The full convolution has 2*len(h) - 1 elements
    # The shifts are from -(len(h)-1) to (len(h)-1)
    # But we're only interested in the overlap integral
    # which is the maximum of the convolution values
    result = np.max(conv) * dx
    return result

        starts.append(h_initial.copy())

        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        starts.append(h.copy())

    for _ in range(5):
        h = np.zeros(n_points)
        indices = np.random.permutation(n_points)[:n_points//2]
        h[indices] = 1.0
        h = h * (n_points / 2) / np.sum(h)
        starts.append(h.copy())
    
    # 4. Smoothed block structure
    for _ in range(5):
        h = np.zeros(n_points)
        # Place a block of height 1 in a random interval of length 1 (half the domain width)
        start_idx = np.random.randint(0, n_points//2)
        end_idx = min(start_idx + n_points//2, n_points)
        h[start_idx:end_idx] = 1.0
        # Add small random noise
        if not np.any(h > 0):
            h[0] = 1
        h = h * (n_points / 2) / np.sum(h)
        starts.append(h.copy())
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points

    # Optimization parameters
    tolerance = 1e-9
    max_iter = 300

    for i, h_start in enumerate(starts):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Fix h_start to be in [0,1] and sum to n_points/2
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Define constraints and bounds
        bounds = [(0, 1)] * n_points
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2,
                'jac': lambda h: np.ones_like(h)}

        # Optimization function
        def obj(h):
            return c5_of_h(h)

        try:
            result = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'ftol': tolerance, 'maxiter': max_iter, 'disp': False})
            h_opt = result.x

            c5_opt = c5_of_h(h_opt)
