import numpy as np

def project_to_F(h, n):
    """Project h onto the feasible set with exact sum n/2 and bounds [0,1]"""
    h = np.clip(h, 0.0, 1.0)
    deficit = n/2 - h.sum()
    free = np.where((h > 0) & (h < 1))[0]
    while np.abs(deficit) > 1e-12 and free.size > 0:
        delta = deficit / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        deficit = n/2 - h.sum()
        free = np.where((h > 0) & (h < 1))[0]
    return h

def evaluate_fitness(h, dx):
    """Evaluate C5 using FFT for full correlation"""
    n = len(h)
    g = 1 - h
    # Zero pad to next power of 2
    L = 2 * n
    h_pad = np.pad(h, (0, L - n), 'constant')
    g_pad = np.pad(g, (0, L - n), 'constant')
    H = fftpack.fft(h_pad)
    G = fftpack.fft(g_pad)
    conv = np.real(fftpack.ifft(H * np.conj(G))) * dx
    conv = conv[:2*n-1]
    return np.max(conv), conv

def local_search(h, g, conv, dx, n, loc_steps=30):
    """Apply memetic local search to refine h with pairwise exchanges"""
    h = h.copy()
    current_max = np.max(conv)
    k_star = np.argmax(conv)

    shift = k_star - (n - 1)

        idxs_geq1 = np.where(h >= 1 - 1e-12)[0]
        idxs_leq0 = np.where(h <= 1e-12)[0]
        
        if len(idxs_geq1) == 0 or len(idxs_leq0) == 0:
            break
            
        i1 = np.random.choice(idxs_geq1)
        i0 = np.random.choice(idxs_leq0)
        
        # Determine step size
        eps = min(h[i1], 1.0 - h[i0], 0.05)
        if eps < 1e-6:
            break
            
        # Perform exchange
        h[i1] -= eps
        h[i0] += eps
        
        # Recalculate the effect on correlation
        # Only update the part that depends on i1 and i0
        # Reset the correlation properly using indices
        # Since we only care about max, compute it again
        g_new = 1 - h
        L = 2 * n
        h_pad = np.pad(h, (0, L - n), 'constant')
        g_pad = np.pad(g_new, (0, L - n), 'constant')
        H = fftpack.fft(h_pad)
        G = fftpack.fft(g_pad)
        new_conv = np.real(fftpack.ifft(H * np.conj(G))) * dx
        new_conv = new_conv[:2*n-1]
        new_max = np.max(new_conv)
        
        if new_max < current_max - 1e-12:
            current_max = new_max
            conv = new_conv
        else:
            # Revert
            h[i1] += eps
            h[i0] -= eps
            
    return h, conv

def generate_population(initial_h, n, pop_size):
    """Generate diverse initial population"""
    pop = []
    
    # Interpolated seed
    if len(initial_h) != n:
        # Linear interpolation
        orig_x = np.linspace(0, 2, len(initial_h))
        new_x = np.linspace(0, 2, n)
        h_seed = np.interp(new_x, orig_x, initial_h)
    else:
        h_seed = initial_h
    
    h_seed = project_to_F(h_seed, n)
    c5_seed, _ = evaluate_fitness(h_seed, 2.0 / n)
    pop.append((h_seed, c5_seed))

    # Random continuous
    for _ in range(pop_size // 3):
        h_random = np.random.rand(n)
        h_random = project_to_F(h_random, n)
        c5_random, _ = evaluate_fitness(h_random, 2.0 / n)
        pop.append((h_random, c5_random))

    # Binary block
    h_block = np.zeros(n)
    h_block[:n//2] = 1.0
    np.random.shuffle(h_block)
    h_block = project_to_F(h_block, n)
    c5_block, _ = evaluate_fitness(h_block, 2.0 / n)
    pop.append((h_block, c5_block))

    # Gaussian perturbed
    h_gauss = np.full(n, 0.5)
    h_gauss += np.random.normal(0, 0.05, n)
    h_gauss = project_to_F(h_gauss, n)
    c5_gauss, _ = evaluate_fitness(h_gauss, 2.0 / n)
    pop.append((h_gauss, c5_gauss))

    # Shuffle to avoid bias
    np.random.shuffle(pop)
    
    return pop

def crossover(parent1, parent2, n):
    """Blend crossover with projection"""
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return project_to_F(child, n)

def mutate(h, n, sigma):
    """Add Gaussian noise to a random subset of coordinates"""
    m = max(1, int(0.02 * n))
    indices = np.random.choice(n, m, replace=False)
    noise = np.random.normal(0, sigma, m)
    h = h.copy()
    h[indices] += noise
    return project_to_F(h, n)
