import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize_scalar
import time

def project_capped_simplex(v, total=0.5):
    """Project vector v onto the capped simplex: 0 <= v_i <= 1, sum v_i = total"""
    v = np.array(v)
    low, high = 0.0, 1.0
    # Binary search on lambda
    def sum_proj(lam):
        return np.sum(np.clip(v - lam, low, high))
    
    # This needs lambda such that sum_proj(lambda) = total
    # Since sum_proj is decreasing in lambda, we use bisection
    lambda_min = np.min(v) - high
    lambda_max = np.max(v) - low
    eps = 1e-12
    while lambda_max - lambda_min > eps:
        lam = (lambda_min + lambda_max) / 2
        s = sum_proj(lam)
        if s > total:
            lambda_min = lam
        else:
            lambda_max = lam
    lam = (lambda_min + lambda_max) / 2
    return np.clip(v - lam, low, high)

def auto_corr(h):
    """Compute auto-correlation O_k(h) using FFT for zero-padded convolution."""
    n = len(h)
    # Zero-pad to length 2*n
    h_pad = np.pad(h, (0, n), mode='constant')
    # Compute correlation as convolution with reversed h
    # O_k = sum_i h[i] * h[i+k]
    # This is equivalent to conv(h, reverse(h), mode='full')[n:]
    # But since we want the a_k = sum_{i} h[i+k] * h[i] we compute
    # the cross-correlation h * rev(h)
    rev_h = h[::-1]
    rev_h_pad = np.pad(rev_h, (0, n), mode='constant')
    corr = fftconvolve(h_pad, rev_h_pad, mode='full')
    # Take the last n elements (this is O_k for k=0,1,...,n-1)
    return corr[n:]

def grad_F(beta, h, O, w):
    """Compute gradient of F_beta(h). w are current soft-max weights."""
    n = len(h)
    # g_i = sum_k w_k * (h_{i+k} + h_{i-k}) (with appropriate boundaries)
    # Compute w * h and w * rev(h) (for backward terms)
    # This corresponds to running two convolutions:
    # g = conv(w, h) + conv_rev(w, h)
    # But need care in proper zero-padded handling...
    w_pad = np.pad(w, (0, n), mode='constant')
    h_pad = np.pad(h, (0, n), mode='constant')
    # h_conv = conv(w, h)  (convolution for forward terms)
    h_conv = fftconvolve(w_pad, h_pad, mode='full')
    # for backward terms, w_rev * h_rev
    h_rev = h[::-1]
    w_rev = w[::-1]
    w_rev_pad = np.pad(w_rev, (0, n), mode='constant')
    h_rev_pad = np.pad(h_rev, (0, n), mode='constant')
    h_rev_conv = fftconvolve(w_rev_pad, h_rev_pad, mode='full')
    
    # Combine to get full gradient
    g_full = h_conv + h_rev_conv
    g = g_full[n:2*n]  # This gives correct gradient lengths
    
    return -g

def compute_F_beta(beta, h, O):
    """Compute the smooth surrogate F_beta(h)"""
    exp_O = np.exp(beta * O)
    sum_exp = np.sum(exp_O)
    if sum_exp == 0:
        return 0
    log_sum_exp = np.log(sum_exp)
    return 1.0 - (1.0 / beta) * log_sum_exp

    elif len(h0) > n:
        h0 = h0[:n]
    h0 = project_capped_simplex(h0, total=n/2)
    
    # Initialize h
    h = h0.copy()
    
    best_C5 = float('inf')
    best_h = h.copy()
    
    # Inner loop: continuation
    for i, beta in enumerate(beta_schedule):
        if time.time() - start_time > time_limit - 0.5:
            break
        # Inner optimization loop with warm start
        max_iter = 5000
        for it in range(max_iter):
            if time.time() - start_time > time_limit - 0.5:
                break
            # Compute O_k(h)
            O = auto_corr(h)
            
            # Compute current F_beta(h)
            F_val = compute_F_beta(beta, h, O)
            
            # Compute weights w_k = exp(beta * O_k) / Z
            exp_O = np.exp(beta * O)
            Z = np.sum(exp_O)
            w = exp_O / Z
            
            # Compute gradient
            g = grad_F(beta, h, O, w)
            
            # Gradient descent with line search
            alpha = 0.1
            max_backtrack = 5
            for step_i in range(max_backtrack):
                h_new = h - alpha * g
                h_new = project_capped_simplex(h_new, total=n/2)
                O_new = auto_corr(h_new)
                F_new = compute_F_beta(beta, h_new, O_new)
                # Check Armijo condition
                # F_new <= F - eta * alpha * norm(g)^2
                eta = 1e-4
                norm_sq = np.linalg.norm(g)**2
                if F_new <= F_val - eta * alpha * norm_sq:
                    break
                else:
                    alpha /= 2.0
                
            # Update h
            h = h_new
            # Record progress
            current_min_O = np.min(O)
            current_C5 = 1.0 - current_min_O
            if current_C5 < best_C5:

                best_h = h.copy()

    max_iter_final = 500

        O_grad = np.zeros_like(best_h)
        # Forward term: h_{i+k*} 
        if min_idx > 0:
            O_grad[min_idx:] += best_h[:-min_idx]  # h_{i+k*}
        # Backward term: h_{i-k*}
        if min_idx > 0:
            O_grad[:-min_idx] += best_h[min_idx:]  # h_{i-k*}
        # But that's not quite correct.
        # When computing auto_correlation, O_new = sum h[i] * h[i+k*], so
        # gradient of O_new w.r.t. h[i] is h[i+k*] (forward) and h[i-k*] (backward)
        # So g[i] = h[i+k*] if i+k* < len and h[i-k*] if i-k* >= 0
        k_star = min_idx
        g_exact = np.zeros(len(best_h))

            if i-k_star >= 0:
                g_exact[i] += best_h[i-k_star]

        step_size = 0.01 / (1.0 + it * 0.002)
        h_new = best_h - step_size * g_exact
        h_new = project_capped_simplex(h_new, total=n/2)
        # Evaluate actual new C5
        O_new = auto_corr(h_new)
        new_min = np.min(O_new)
        new_C5 = 1.0 - new_min
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
        else:
            if new_C5 > 0.999 * best_C5:  # Small improvement condition
                pass
        # Update individual O for next step
        O = O_new
        min_idx = np.argmin(O)
                
    return best_h, best_C5
