def evaluate(h):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    # Compute all overlaps via correlation
    corr = correlate(h, 1 - h, mode='full')
    # The full correlation includes negative shifts, but we want all k
    # We only care about positive shifts in [0,2], which are indices [len(h)-1, 2*len(h)-2]
    # The actual shifts correspond to k = 0 to 2, from index offset len(h)-1 to 2*len(h)-2
    max_k = len(h) - 1  # max k in [0,2] when discretized
    # Extract the part from shift 0 to max shift allowed
    overlap = corr[len(h) - 1:2*len(h) - 1]  # this is suboptimal indexing
    # Extract the section that corresponds to k ∈ [0,2] essentially
    # This is a bit tricky - let's track indices starting at 0 to get k from 0 to whatever
    # So correlation at index i = len(h)-1 + k, so k = i - (len(h)-1)
    shift_indices = np.arange(len(h)-1, 2*len(h)-1)
    # Now extract overlaps for k=0 to k=2 valid values:
    valid_indices = shift_indices <= len(h) - 1  # index for k from 0 to len(h)-1
    # To properly extract: k should be 0 to 2 only. Assume len(h) is large enough
    # Look at overlap for shifts 0,1,2,... up to 2*len(h) - (len(h)-1) - 1 = len(h) 
    # So overlaps at indices [len(h)-1, ..., 2*len(h)-2]
    # But we only want up to max_k = 2 but that implies we want full period up to k < 2
    # Which actually maps to: indices from (len(h)-1) to (len(h)-1 + 2*len(h)/2 -1) = if len(h) = 45:
    # indices from 44 to 88, i.e. overlaps for k=0 to 44 = 45 values, and some are duplicates we want 0 concurrent with k=0
   
    # Correct indexing:
    # The mode='full' gives a result of 2*n-1. So the range of shifts k is from -n+1 to n-1.
    # We want the actual effect of shifting by k in [0,2] 
    # len(h) points — dx = 2/n. So k in [0,2] => shift k/dx in [0, n]. 
    # But we want valid shifts for 0 < k <= 2, so index range: [n-1, 2n-2] corresponds to k=0..n-1, not k=0..2!
    # What we want: the overlap values for small k in [0,2] 
    # With this we need to project k back to indices or directly select off k<2
    # It's easier to set up and see actual values.
    
    # Revisit: mirroring the definition clearly:
    # The shifts k in [0,2] have corresponding indices in the non-zero overlaps shifted:
    # The negative shift indices of full correlation are from 1 .. n-1
    # Positive overlap indices correspond to k from 0 to n-1  
    # k from 0 to 2 => index range is from n-1 to n+2-1 = n+1 
    overlaps = corr[int(len(h)-1):int(len(h)+2)]  # k = 0 to 2 => indices = len(h)-1 to len(h)+1
    return np.max(overlaps) * dx

def project_feasible(x, target_sum=None):
    """Project x onto {0 ≤ h ≤ 1, sum h = target_sum}"""
    if target_sum is None:
        target_sum = np.sum(x)
    
    x = np.asarray(x).copy()
    # Water filling algorithm to project onto capped-simplex
    d = len(x)
    x = np.minimum(x, 1.0)
    s = np.sum(x)
    diff = target_sum - s
    
    if diff > 0:
        # Need to increase
        for iter in range(50):
            if abs(diff) < 1e-10:
                break
            active = x < 1.0
            if not np.any(active):
                break
            free = np.sum(active)
            incr = diff / free
            if incr >= 1.0 - np.min(x[active]):
                # Can't fill all, so fill them all to 1.
                x[active] = 1.0
                diff = target_sum - np.sum(x)
                break
            else:
                x[active] += incr
            diff = target_sum - np.sum(x)
    elif diff < 0:
        # Need to decrease
        for iter in range(50):
            if abs(diff) < 1e-10:
                break
            active = x > 0.0
            if not np.any(active):
                break
            free = np.sum(active)
            decr = -diff / free
            if decr >= np.min(x[active]):
                # Can't decrease all, so decrease them all to 0.
                x[active] = 0.0
                diff = target_sum - np.sum(x)
                break
            else:
                x[active] -= decr
            diff = target_sum - np.sum(x)
            
    return x

def active_shifts(h, eps=1e-9):
    """Return indices k where overlap is within eps of current max"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_val = np.max(corr[len(h) - 1:len(h) + 1])
    active_ks = []
    for i in range(len(h) - 1, len(h) + 2):  # k=0,1,2 case
        if abs(corr[i] - max_val) < eps:
            active_ks.append(i)
    return active_ks

def gradient(h, active_ks):
    """Compute sub-gradient of max-overlap objective"""
    dx = 2.0 / len(h)
    g = np.zeros_like(h)
    
    # Get maximum over active shifts
    all_ks = active_ks
    n = len(h)
    for k in all_ks:
        k_paid = k - (n - 1)
        # e.g., k=0 -> k_paid = -44, => h[i+k_paid] = h[i-44] (outside array => 0)
        # k = n-1 => k_paid = 0 => h[i]
        # k = n => k_paid = 1 => h[i+1]
        # k = 2*n-2 => k_paid = n-1 => h[i+n-1]
        for i in range(n):
            # For skew of k, h[i] contributes (1 - h[i+k]) where k is positive shift from i
            
            # Before: for overlap at index k of corr == h shift by k:
            # corr[i] = sum(h[j] * (1 - h[j-i])) for all j
            # We treat corr as axis such that each element corresponds to shift k being
            # used. Index corresponds to k+ len(h)-1.
            
            # So if k corresponds to index k_start = k + len(h) - 1
            # We are looking at pattern h[i] * (1 - h[i+(k-(n-1))] = h[i] * (1 - h[i+k-n+1])
            # i.e., bit k paid = k - (n-1) = k-n+1
            
            if k_paid < 0:
                g[i] += h[i+k_paid]  # This assumes negative indexing in Python
            elif k_paid >= 0 and i+k_paid < n:
                g[i] += h[i+k_paid] 
            else:
                pass
            # always add (1-scalar Contribution) 
            # Actually better computation:
            g[i] -= 1.0 if i+k_paid < n and i+k_paid >= 0 else 0.0
            g[i] -= 1.0 if (i-k_paid) < n and (i-k_paid) >= 0 else 0.0  # i - k_paid
    return g

def line_search(h, g, t_current, budget):
    """Perform line search for step size"""
    alpha_values = [0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1]
    best_h = h.copy()
    best_val = t_current
    
    for alpha in alpha_values:
        if abs(alpha) < 1e-8:
            h_test = h.copy()
        else:
            h_test = h - alpha * g
            if np.any(h_test > 1.0) or np.any(h_test < 0.0):
                h_test = project_feasible(h_test)
            else:
                h_test = h_test
        val = evaluate(h_test)
        if val < best_val:
            best_val = val
            best_h = h_test.copy()
    
    return best_h, best_val

def local_optimize(h0, max_iter=2000, t_end=None):
    """Projected gradient descent with line search"""
    h = h0.copy()
    t_current = evaluate(h)
    best_h = h.copy()
    best_val = t_current
    last_grad = time.time()

    for i in range(max_iter):
        if t_end and time.time() > t_end:
            break
        if time.time() - last_grad > 0.5:
            # update gradient only occasionally
            ks = active_shifts(h)
            g = gradient(h, ks)
            last_grad = time.time()
        # line search
        h_new, t_new = line_search(h, g, t_current, t_end)
        if t_new < best_val:
            best_val = t_new
            best_h = h_new.copy()
        h = h_new
        t_current = t_new

    return best_h, best_val

def anneal(h_start, t_end, temp0=0.01, decay=0.995, steps=5000):
    """Simulated annealing"""
    h = h_start.copy()
    t_current = evaluate(h)
    best_h = h.copy()
    best_val = t_current
    
    T = temp0
    for _ in range(steps):
        if time.time() > t_end:
            break
            
        # propose move
        i = random.randint(0, len(h)-1)
        j = random.randint(0, len(h)-1)
        delta = random.uniform(-0.05, 0.05)
        
        h_next = h.copy()
        if h_next[i] >= delta and h_next[j] + delta <= 1:
            h_next[i] -= delta
            h_next[j] += delta
            h_next = project_feasible(h_next)
            
            t_new = evaluate(h_next)
            if t_new < best_val:
                best_val = t_new
                best_h = h_next.copy()
            elif random.random() < np.exp(-(t_new - t_current) / T):
                h = h_next
                t_current = t_new
                
        T *= decay
        
    return best_h, best_val

def random_feasible(n):
    """Generate random feasible vector"""
    x = np.random.rand(n)
    x = x / np.sum(x) * 2
    x = project_feasible(x, 1.0)
    return x

def interpolate_h(h_old, n_new):
    """Linearly interpolate or repeat samples to new resolution"""
    # Linear interpolation
    old_indices = np.linspace(0, 1, len(h_old))
    new_indices = np.linspace(0, 1, n_new)
    h_new = np.interp(new_indices, old_indices, h_old)
    h_new = project_feasible(h_new, 1.0)
    return h_new
