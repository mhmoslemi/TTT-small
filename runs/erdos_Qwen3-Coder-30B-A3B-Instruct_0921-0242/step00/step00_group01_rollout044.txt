import numpy as np
import time
from scipy import fftpack

def autocorr(b):
    """Compute cyclic autocorrelation of binary sequence b."""
    n = len(b)
    # Zero-pad to avoid wraparound effects in FFT
    padded = np.zeros(2*n - 1)
    padded[:n] = b
    fft_vals = fftpack.fft(padded)
    autocorr_vals = fftpack.ifft(fft_vals * np.conj(fft_vals)).real
    return autocorr_vals[:n]

def evaluate_erdos_solution(h):
    """Use the provided evaluation function for final check."""
    dx = 2.0 / len(h)
    result = np.correlate(h, 1-h, mode="full") * dx
    return max(result)

def create_legendre_sequence(n):
    """Create a Legendre sequence of length n."""
    p = 211  # A prime >= n
    s = np.zeros(p, dtype=int)
    for i in range(p):
        if pow(i, (p-1)//2, p) == 1:  # Quadratic residue?
            s[i] = 1
    # Truncate to n
    b = s[:n].copy()
    # Balance the ones and zeros
    ones_count = np.sum(b)
    if ones_count > n//2:
        # Drop some ones
        drop_indices = np.where(b == 1)[0][:ones_count - n//2]
        b[drop_indices] = 0
    elif ones_count < n//2:
        # Flip some zeros to ones
        flip_indices = np.where(b == 0)[0][:n//2 - ones_count]
        b[flip_indices] = 1
    return b

def create_singer_sequence(n):
    """Create a Singer difference set sequence of length n."""
    # Use q=13 for n=183, then pad/embed to achieve n=200
    q = 13
    p = q**2 + q + 1  # = 183
    if n > p:
        # Pad to p, then embed properly
        b = np.zeros(p, dtype=int)
        # Singer set = one block of 1s
        ones_start = 0  # Typical for this construction
        ones_length = (p-3)//4 + 1  # Approximate
        b[ones_start:ones_start+ones_length] = 1
        # Embed it correctly in n=200 by padding and shifting to create balanced binary
        b_large = np.zeros(200, dtype=int)
        # Make it more balanced manually
        b_large[:p] = b
        for _ in range(200 - p):
            # cyclically rotate to spread ones
            b_large = np.roll(b_large, 1)
            # flip an element if needed
            ones_count = np.sum(b_large)
            if ones_count == 100:
                break
            if ones_count > 100:
                # find a one, flip it
                idx = np.where(b_large == 1)[0][0]
                b_large[idx] = 0
            elif ones_count < 100:
                # find a zero, flip it
                idx = np.where(b_large == 0)[0][0]
                b_large[idx] = 1
        result = b_large[:n]
    else:
        # Just embed the p=183 version in a large vector if n < 183?
        # Simpler: Just generate a seed and leave it
        result = np.zeros(n, dtype=int)
    return result

def create_random_balanced(n):
    """Create a random balanced binary vector."""
    indices = np.arange(n)
    np.random.shuffle(indices)
    b = np.zeros(n, dtype=int)
    b[indices[:n//2]] = 1
    return b

def create_palindromic_block(n):
    """Create a palindromic block pattern."""
    b = np.zeros(n, dtype=int)
    mid = n // 2
    b[:mid] = 1
    # Make it palindromic
    b[mid:] = np.flip(b[:mid])
    # Ensure it's balanced (should be already if n is even)
    if np.sum(b) != n // 2:
        # Adjust: flip some digits to balance
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        extra_ones = np.sum(b) - n // 2
        if extra_ones > 0:
            # Must flip extra_ones one to zeros
            for i in range(extra_ones):
                # Find a one to flip to 0
                j = ones[i]
                b[j] = 0
        else:
            # Need to flip -extra_ones zeros to ones
            for i in range(-extra_ones):
                j = zeros[i]
                b[j] = 1
    return b

def get_binary_seeds(n=200):
    """Generate several binary seed vectors."""
    seeds = []
    seeds.append(create_legendre_sequence(n))
    seeds.append(create_singer_sequence(n))
    seeds.append(create_random_balanced(n))
    seeds.append(create_palindromic_block(n))
    
    # Add 3 random balanced sequences
    for _ in range(3):
        seeds.append(create_random_balanced(n))

    results = []
    for i, b in enumerate(seeds):
        ac = autocorr(b)
        m = np.min(ac[1:])  # Ignore lag=0
        S = np.sum(b)
        C5 = (S - m) * 2.0 / n  # dx = 2/n
        results.append((C5, b, m, i))
    results.sort()
    return [(C5, b.copy(), m) for C5, b, m, _ in results[:10]]

def swap_improve(b, dx):
    """Optimize the binary vector `b` using swap-based hill climbing."""
    n = len(b)
    S = np.sum(b)
    
    # Precompute shifts
    B_shift = np.zeros((n, n), dtype=int)  # B_shift[k,i] = b[(i+k)%n]
    for k in range(n):
        B_shift[k] = np.roll(b, k)
    
    # Compute initial autocorrelations
    A = autocorr(b)
    m = np.min(A[1:])
    if m == 0:
        minimize_idx = np.argmin(A)
        if minimize_idx == 0:
            j = 1
        else:
            j = minimize_idx
        while A[j] == 0:  # need non-zero since we modify
            j += 1
    k_min = np.argmin(A[1:]) + 1  # lag corresponding to min
    m_min = A[k_min]
    
    best_m = m_min
    best_b = b.copy()
    
    # Initialize temperatures
    temp = 10
    cooldown = 0.99
    max_swaps = 30000
    
    # Track convergence
    no_improvement = 0
    last_accept = 0
    
    for i in range(max_swaps):
        # Sample positions to swap
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break

        i_idx = np.random.choice(ones)
        j_idx = np.random.choice(zeros)
        
        # Compute the effect of this swap on k_min
        k = k_min
        delta = - B_shift[k][i_idx] + B_shift[k][j_idx]
        
        # If swap improves minimium correlation
        if delta > 0:
            # Perform the swap
            b[i_idx], b[j_idx] = b[j_idx], b[i_idx]  # In-place
            
            # Update shift matrix for this pair
            # Only need to recompute row k since that's the only one impacted
            B_shift[k] = np.roll(b, k)
            for other_k in range(n):
                if other_k != k:
                    B_shift[other_k] = np.roll(b, other_k)

            # Recompute full autocorrelations
            A = autocorr(b)
            m_new = np.min(A[1:])
            # Update best solution
            if m_new > best_m:
                best_m = m_new
                best_b = b.copy()
            m_min = m_new
            # Find new worst lag
            k_min = np.argmin(A[1:]) + 1
            no_improvement = 0
            last_accept = i
        else:
            # Accept based on simulated annealing
            if np.random.rand() < np.exp(delta / temp):
                # Perform the swap
                b[i_idx], b[j_idx] = b[j_idx], b[i_idx]
                
                # Update shift matrix for this pair
                B_shift[k] = np.roll(b, k)
                for other_k in range(n):
                    if other_k != k:
                        B_shift[other_k] = np.roll(b, other_k)

                A = autocorr(b)
                m_new = np.min(A[1:])
                if m_new > best_m:
                    best_m = m_new
                    best_b = b.copy()
                m_min = m_new
                k_min = np.argmin(A[1:]) + 1
                no_improvement = 0
                last_accept = i
            else:
                no_improvement += 1
                
        # Decay temperature
        if i % 500 == 0:
            temp *= cooldown
        if no_improvement > 5000:
            break
            
    # Return best so far
    return (best_b, best_m)

def inflate_to_n(b, target_n):
    """Duplicate entries and pad to target_n."""
    expanded = np.repeat(b, 2)  # Duplicate each element
    if len(expanded) > target_n:
        return expanded[:target_n]
    elif len(expanded) < target_n:
        old_len = len(expanded)
        expanded = np.resize(expanded, target_n)
        # Replicate half of the prior expansion (for n=200 to 300, which will be handled by scaling)
        # For now, just pad with zeros at the end (this results in a valid vector)
        # For a better unifying approach interpolate with integer values including fractional parts, but simpler as fallback
        assert target_n >= len(b)
        # Do not change the sum of values here where constraints stay as needed
        return expanded
    else:
        return expanded.copy()

def interpolate_b_to_n(b, target_n):
    """Interpolation step (useful for fine-tuning). This is a minimal step."""
    n = len(b)
    if n == target_n:
        return b.copy()
    elif target_n > n:
        # Simple duplicate + pad
        x = np.repeat(b, 2)
        if len(x) == target_n:
            return x
        elif len(x) > target_n:
            return x[:target_n]
        else:
            return np.resize(x, target_n)
    else:
        # Downscale
        return b[::2]

def run(seed=42, budget_s=600, **kwargs):

    start = time.time()

        if C5_seed < best_C5:
            best_C5 = C5_seed

            break

        except Exception as e:

    # Begin Stage 3: Scale up if needed
    final_n = 350  # Final size
    inflated = inflate_to_n(best_b, final_n)
    # Further polishing?
    inflated_b, inflated_m = swap_improve(inflated, 2.0 / final_n)
    
    # Final evaluation
    dx_final = 2.0 / final_n

    # Apply final polishing if time allows
    if time.time() - start < budget_s - 20:
        # We can do a few gradient-like manipulations, but keeping it simple.
        # Just compute exact corr with np.correlate
        final_C5_exact = evaluate_erdos_solution(inflated_b)
        actual_sum = np.sum(inflated_b) * dx_final
        
        # Final check: ensure sum constraint holds to tolerance

            pass

    # Convert to step function h
