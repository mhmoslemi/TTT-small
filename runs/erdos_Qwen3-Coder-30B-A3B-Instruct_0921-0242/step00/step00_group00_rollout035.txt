import numpy as np
from scipy.signal import fftconvolve
import time

def project_capped_simplex(v, total=0.5):
    """Project vector v onto the capped simplex: 0 <= v[i] <= 1, sum(v) = total"""
    # Use bisection to find the Lagrange multiplier lambda
    v = np.array(v, dtype=float)
    v_max = 1.0
    v_min = 0.0
    
    # Ensure v is within [0, 1]
    v = np.clip(v, v_min, v_max)
    
    # Bisect over lambda to find correct projection
    low, high = -1.0, 1.0
    iter_count = 0
    while high - low > 1e-12 and iter_count < 50:
        iter_count += 1
        lam = (low + high) / 2.0
        proj = np.clip(v - lam, v_min, v_max)
        proj_sum = np.sum(proj)
        if proj_sum > total:
            low = lam
        else:
            high = lam
    lam = (low + high) / 2.0
    return np.clip(v - lam, v_min, v_max)

def auto_corr(h):
    """Compute zero-padded auto-correlation of h."""
    n = len(h)
    h_pad = np.pad(h, (0, n), 'constant', constant_values=0)
    # Use FFT for convolution: h * h then extract valid part
    corr = fftconvolve(h_pad, h, mode='full')
    return corr[n-1:2*n-1]

def grad_F(beta, h, O, w):
    """Compute gradient of smooth max of auto-correlations."""
    # w is weights, O_k values, future gradient computed as:
    # g_i = sum_k w_k * (h_{i+k} + h_{i-k})
    n = len(h)
    g = np.zeros(n)
    
    # Forward component: h_{i+k} for k=0..n-1, i from 0 to n-k-1
    w_h = fftconvolve(w, h, mode='full')
    g += w_h[n-1:2*n-1]
    
    # Backward component: h_{i-k} for i from k to n-1
    # reorder to compute properly
    w_h_rev = fftconvolve(w[::-1], h[::-1], mode='full')
    # accumulate g_{i} += w_k * h_{i-k} (i-k >= 0) = w_k * h_{i-k}
    for k in range(n):
        if k < n:
            g[k] += w_h_rev[2*(n-1) - k]
    
    return -g

def optimize_one_start(n, beta_schedule, time_limit, seed):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize h from initial_h_values, or random
    # if initial_h_values is long enough, downsample it
    try:
        ini_h = np.array(initial_h_values)
        if len(ini_h) >= n:
            # downsample
            idx = np.linspace(0, len(ini_h)-1, n, dtype=int)
            h = ini_h[idx]
        else:
            # upsample
            h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(ini_h)), ini_h)
        h = np.clip(h, 0, 1)
    except:
        # construct random sample
        u = np.random.exponential(1.0, n)
        h = (u / np.sum(u)) * (n / 2)
        h = np.clip(h, 0, 1)
    
    h = project_capped_simplex(h, n/2)
    
    dx = 2.0 / n
    best_C5 = np.inf
    best_h = h.copy()
    
    # continuation parameters
    max_iter_per_beta = 1000
    inner_time_budget = time_limit * 0.7 / len(beta_schedule)
    
    for beta_idx, beta in enumerate(beta_schedule):
        if time.time() - start_time > time_limit - 0.5:
            break
        # Inner loop: projected gradient descent
        iter_count = 0
        for iter_num in range(max_iter_per_beta):
            if time.time() - start_time > time_limit - 0.5:
                break
            # compute auto-correlations
            O = auto_corr(h)
            # compute smooth max approximation 
            log_sum = np.log(np.sum(np.exp(beta * O)))
            F_beta = 1 - (1 / beta) * log_sum
            # compute weights
            w = np.exp(beta * O) / np.sum(np.exp(beta * O))
            
            # gradient
            g = grad_F(beta, h, O, w)
            # line search parameters
            alpha = 0.1
            eta = 1e-4
            max_backtrack = 10
            accepted = False
            g_norm = np.linalg.norm(g)**2
            
            for _ in range(max_backtrack):
                h_new = h - alpha * g
                h_new = project_capped_simplex(h_new, n/2)
                
                O_new = auto_corr(h_new)
                F_new = 1 - (1 / beta) * np.log(np.sum(np.exp(beta * O_new)))
                armijo_condition = (F_new <= F_beta - eta * alpha * g_norm)
                if armijo_condition:
                    h = h_new
                    if F_new < best_C5:
                        best_C5 = F_new
                        best_h = h.copy()
                    accepted = True
                    break
                alpha *= 0.5
            
            if not accepted:
                h = h_new
                
            iter_count += 1
            if iter_count > 20 and F_beta < 1e-7:
                break
    
    # Refinement with exact C5
    # use simple gradient descent on the true min_k O_k
    # to polish the solution
    max_iter_refine = 500
    alpha_base = 0.01
    
    for iter_refine in range(max_iter_refine):
        if time.time() - start_time > time_limit - 0.5:
            break
        # compute O
        O = auto_corr(h)
        # find the worst lag k_star
        k_star = np.argmin(O)
        # compute true gradient
        g_true = np.zeros(len(h))
        # O_k_star = sum_{i} h_i * h_{i+b} for b=k_star
        # gradient g_i = h_{i+k_star} if i+k_star < n else 0
        #           + h_{i-k_star} if i-k_star >= 0 else 0
        for i in range(len(h)):
            if i + k_star < len(h):
                g_true[i] += h[i + k_star]
            if i - k_star >= 0:
                g_true[i] += h[i - k_star]

        # update with direction -g_true
        alpha = alpha_base / np.sqrt(iter_refine + 1)
        h_new = h - alpha * g_true
        h_new = project_capped_simplex(h_new, n/2)
        
        # compute current C5
        O_new = auto_corr(h_new)
        curr_C5 = 1 - np.min(O_new)
        if curr_C5 < best_C5:
            best_C5 = curr_C5
            best_h = h_new.copy()
        h = h_new

    # Final analysis
    corr = np.convolve(best_h, 1 - best_h, mode='full') * dx
    final_C5 = np.max(corr)
    
    return best_h, final_C5, n
