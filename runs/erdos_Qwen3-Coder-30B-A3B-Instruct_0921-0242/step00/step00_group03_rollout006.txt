import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute convolution of h with (1-h)
    conv = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    # The maximum value of the convolution scaled by dx
    max_conv = np.max(conv) * dx
    return max_conv

def initial_h_values():
    """Return initial construction for the problem."""
    n = 70

    # Simple construction based on the idea of maximizing overlap
    # We distribute the mass so that it minimizes the C5 constant
    # This particular construction is from the known good solution
    h[10:30] = 1.0
    h[40:60] = 1.0
    # Normalize to make the integral equal to 1
    h = h * 2.0 / np.sum(h)
    return h

def objective(h_values):
    """Objective function to minimize (negative of the bound)."""
    return evaluate_h(h_values)
