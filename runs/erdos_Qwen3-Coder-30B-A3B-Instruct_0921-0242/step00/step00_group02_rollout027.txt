import numpy as np
import time
from scipy import signal

def correlation_vector(b):
    """Compute the autocorrelation vector R_k for binary sequence b."""
    R = np.correlate(b, b, mode='full')
    return R[len(b)-1:]  # Keep only the k=1, ..., n-1 correlations

def update_correlation_delta(R, i, j, b):
    """Update R vector in-place for a swap at positions (i, j).
    b: binary vector, R: current autocorrelation, i,j: swapped positions."""
    # From i to j: i becomes 0, j becomes 1
    # For shifts k where the change matters:
    n = len(b)
    delta_R = np.zeros(n-1, dtype=int)
    for k in range(1, n):
        if i - k >= 0 and b[i-k] == 1:
            delta_R[k-1] -= 1
        if i + k < n and b[i+k] == 1:
            delta_R[k-1] -= 1
        if j - k >= 0 and b[j-k] == 1:
            delta_R[k-1] += 1
        if j + k < n and b[j+k] == 1:
            delta_R[k-1] += 1
    R += delta_R
    return delta_R

def local_hill_climb(b, R, max_swaps=200, escape_prob=0.05):
    """Local hill climb to improve min correlation."""
    n = len(b)
    b = b.copy()
    R = R.copy()
    last_improvement = 0
    while True:
        k_star = np.argmin(R)
        min_correlation = R[k_star]
        best_delta = 0
        best_swap = None
        
        # Sample a subset of swaps
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        swaps = [(i, j) for i in ones for j in zeros]
        if len(swaps) > max_swaps:
            # Random sampling
            np.random.shuffle(swaps)
            swaps = swaps[:max_swaps]
        
        for i, j in swaps:
            delta_R = np.zeros(n-1, dtype=int)
            for k in range(1, n):
                if i - k >= 0 and b[i-k] == 1:
                    delta_R[k-1] -= 1
                if i + k < n and b[i+k] == 1:
                    delta_R[k-1] -= 1
                if j - k >= 0 and b[j-k] == 1:
                    delta_R[k-1] += 1
                if j + k < n and b[j+k] == 1:
                    delta_R[k-1] += 1
            new_k_star = np.argmin(R + delta_R)
            new_min = R[new_k_star]
            change = new_min - min_correlation
            if change > best_delta:
                best_delta = change
                best_swap = (i, j)
        
        if best_swap is not None:
            # make the swap
            i, j = best_swap
            b[i], b[j] = b[j], b[i]
            update_correlation_delta(R, i, j, b)
            last_improvement = 0  # reset counter
        else:
            # Try a random non-improving swap with probability escape_prob
            if escape_prob > 0 and np.random.random() < escape_prob:
                np.random.shuffle(swaps)
                for i, j in swaps:
                    if b[i] == 1 and b[j] == 0:
                        b[i], b[j] = b[j], b[i]
                        update_correlation_delta(R, i, j, b)
                        last_improvement = 0
                        break
            else:
                last_improvement += 1
                if last_improvement > 5:
                    break
    return b, R

def tournament_selection(pop, t_size=3):
    """Select best individual from tournament."""
    tour_indices = np.random.choice(len(pop), t_size, replace=False)
    tour_scores = [np.min(correlation_vector(ind)) for ind in pop[tour_indices]]
    winner_idx = tour_indices[np.argmax(tour_scores)]
    return pop[winner_idx]

def crossover(parent1, parent2, n):
    """Create child using union method with weight correction."""
    support1 = set(np.where(parent1 == 1)[0])
    support2 = set(np.where(parent2 == 1)[0])
    union_support = support1.union(support2)
    
    child = np.zeros(n, dtype=np.uint8)
    if len(union_support) > n//2:
        # Too many 1's: remove extras randomly
        extras = list(union_support - support1 - support2)
        too_many = len(union_support) - n//2
        remove_indices = np.random.choice(extras, too_many, replace=False)
        union_support = union_support - set(remove_indices)
    elif len(union_support) < n//2:
        # Add missing ones
        complement = set(range(n)) - union_support
        needed = n//2 - len(union_support)
        add_indices = np.random.choice(list(complement), needed, replace=False)
        union_support = union_support.union(set(add_indices))
    
    child[list(union_support)] = 1
    return child

def mutate(individual, num_swaps=1):
    """Apply num_swaps random swaps."""
    n = len(individual)
    ones = np.where(individual == 1)[0]
    zeros = np.where(individual == 0)[0]
    for _ in range(num_swaps):
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            individual[i], individual[j] = individual[j], individual[i]
    return individual

def binary_to_float(b, n_points):
    """Convert binary to floating point with integral constraint."""
    h_float = b.astype(float)
    h_float *= n_points / (2 * np.sum(b))
    return h_float

def upsample_binary(b, new_n):
    """Upsample binary vector, truncate and fix weight."""
    old_n = len(b)
    ratio = new_n // old_n
    b_upsampled = np.repeat(b, ratio)
    if len(b_upsampled) > new_n:
        b_upsampled = b_upsampled[:new_n]
    elif len(b_upsampled) < new_n:
        b_upsampled = np.pad(b_upsampled, (0, new_n - len(b_upsampled)), mode='constant')
        
    # Fix weight if necessary by flipping a few zeros to ones
    current_weight = np.sum(b_upsampled)
    desired_weight = new_n // 2
    diff = desired_weight - current_weight
    
    if diff != 0:
        zero_indices = np.where(b_upsampled == 0)[0]
        if len(zero_indices) < abs(diff):
            ones_indices = np.where(b_upsampled == 1)[0]
            # Flip enough ones to make up for zeros needed
            flip_indices = np.random.choice(ones_indices, abs(diff), replace=False)
            b_upsampled[flip_indices] = 0
            b_upsampled[zero_indices[:abs(diff)]] = 1
        else:
            flip_indices = np.random.choice(zero_indices, abs(diff), replace=False)
            b_upsampled[flip_indices] = 1
            
    return b_upsampled
