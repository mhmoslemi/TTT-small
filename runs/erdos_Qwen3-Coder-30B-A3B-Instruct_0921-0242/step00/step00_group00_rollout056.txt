```python
import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.special import logsumexp

def decode(x, M):
    """Decode the optimization vector into segment lengths u and heights a."""
    u = x[:M]
    a0 = x[M]
    v = x[M+1:]
    a = np.empty(M+1)
    a[0] = a0
    for j in range(1, M):
        a[j] = a[j-1] - v[j-1]
    a[M] = a[M-1] - v[M-1]
    s = np.concatenate([[0], np.cumsum(u)])
    return u, a, s

def overlap_piecewise(s, a):
    """Compute exact C5 for piecewise constant h with given breakpoints and heights."""
    M = len(a) - 1
    K = []
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0 <= k1 <= 2:
                K.append(k1)
            if 0 <= k2 <= 2:
                K.append(k2)
    K = sorted(set(K))
    K = [0] + K + [2]
    
    C5 = 0.0
    for k in K:
        Fk = 0.0
        for j in range(M+1):
            for l in range(M+1):
                s_j, s_j1 = s[j], s[j+1]
                s_l, s_l1 = s[l], s[l+1]
                # Length of intersection: max(0, min(s_j1, s_l1-k) - max(s_j, s_l-k))
                len_int = max(0, min(s_j1, s_l1 - k) - max(s_j, s_l - k))
                Fk += a[j] * (1 - a[l]) * len_int
        C5 = max(C5, Fk)
    return C5

def project_capped_simplex(v, total):
    """Project vector v onto the capped simplex with sum = total."""
    v = np.array(v)
    if np.isclose(np.sum(v), total):
        return v
    # Sort in descending order 
    sorted_v = np.sort(v)[::-1]
    # Find the threshold for the projection
    cumsum = np.cumsum(sorted_v)
    n = len(v)
    rho = -1
    for i in range(n):
        if sorted_v[i] > (cumsum[i] - total) / (i + 1):
            rho = i
        else:
            break
    if rho < 0:
        # All elements are too small, make them equal
        val = total / n
        return np.full_like(v, val)
    else:
        # Use threshold
        threshold = (cumsum[rho] - total) / (rho + 1)
        ret = np.maximum(v - threshold, 0)
        return ret

def evaluate_fitness(x, M, n_points, integral_weight=1e6):
    """Fitness function for DE."""
    try:
        u, a, s = decode(x, M)
        # Check constraints
        if np.any(u < 0) or np.any(u > 2) or np.any(a < 0) or np.any(a > 1):
            return 1e6
        integral_sum = np.sum(u * a)
        integral_error = abs(integral_sum - 1)
        C5 = overlap_piecewise(s, a)
        if integral_error > 1e-8:
            return C5 + integral_weight * integral_error ** 2
        return C5
    except Exception:
        return 1e6

def run(seed=42, budget_s=600, **_):
    np.random.seed(seed)
    start = time.time()
    M_list = [4, 6]
    best_C5 = float('inf')
    best_x = None
    
    n_points = 256
    
    # Phase A: DE with M=4
    if time.time() - start > budget_s - 0.5:
        return [0.5]*n_points, 0.5, n_points
    
    M = M_list[0]
    bounds = []
    # u variables: 
    for _ in range(M):
        bounds.append((0, 2))
    # a0:
    bounds.append((0, 1))
    # v variables (drops): 
    for _ in range(M - 1):
        bounds.append((0, 1))
    
    constraint_integral = lambda x: np.sum(decode(x, M)[0] * decode(x, M)[1]) - 1
    
    # run DE
    resA = differential_evolution(
        lambda x: evaluate_fitness(x, M, n_points),
        bounds,
        seed=seed,
        maxiter=100,
        popsize=15*M,
        disp=False
    )
    if resA.fun < best_C5:
        best_C5 = resA.fun
        best_x = resA.x
    
    # Phase B: DE with M=6
    if time.time() - start > budget_s - 0.5:
        return [0.5]*n_points, 0.5, n_points
        
    M = M_list[1]
    bounds = []
    for _ in range(M):
        bounds.append((0, 2))
    bounds.append((0, 1))
    for _ in range(M - 1):
        bounds.append((0, 1))
    
    # Initialize population from previous result (nearby)
    def init_population():
        pop = [resA.x + np.random.normal(0, 0.1, len(bounds)) for _ in range(15*M)]
        # Clip to valid bounds
        for i in range(len(pop)):
            for j in range(len(bounds)):
                pop[i][j] = max(bounds[j][0], min(bounds[j][1], pop[i][j]))
        return np.array(pop)
    
    resB = differential_evolution(
        lambda x: evaluate_fitness(x, M, n_points),
        bounds,
        seed=seed + 10,
        maxiter=100,
        popsize=15*M,
        init=init_population,
        disp=False
    ) 
    if resB.fun < best_C5:
        best_C5 = resB.fun
        best_x = resB.x

    # Phase C: Local refinement
    if time.time() - start > budget_s - 0.5:
        return [0.5]*n_points, 0.5, n_points
        
    def refine_objective(x):
        u, a, s = decode(x, M)
        # Use logsumexp proxy to smooth the max
        K = []
        for j in range(M+1):
            for l in range(M+1):
                k1 = s[l] - s[j+1]
                k2 = s[l+1] - s[j]
                if 0 <= k1 <= 2:
                    K.append(k1)
                if 0 <= k2 <= 2:
                    K.append(k2)
        K = sorted(set(K))
        K = [0] + K + [2]
        full_values = []
        for k in K:
            Fk = 0.0
            for j in range(M+1):
                for l in range(M+1):
                    s_j, s_j1 = s[j], s[j+1]
                    s_l, s_l1 = s[l], s[l+1]
                    len_int = max(0, min(s_j1, s_l1 - k) - max(s_j, s_l - k))
                    Fk += a[j] * (1 - a[l]) * len_int
            full_values.append(Fk)

        beta = 50
        # logsumexp
        proxy_value = logsumexp(full_values, b=np.ones_like(full_values)) / beta
        return proxy_value

    # Constraint for integral
    def integral_constraint(x):
        u, a, s = decode(x, M)
        return np.sum(u * a) - 1

    # Use SLSQP
    cons = {'type': 'eq', 'fun': integral_constraint}
    bounds_slsqp = [(0, 2)] * M + [(0, 1)] + [(0, 1)] * (M-1)
    res = minimize(
        refine_objective,
        best_x,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=cons,
        options={'maxiter': 200},
        callback=lambda x: None
    )

    # If not converged, or if not better, fallback to the best from DE
    final_x = res.x if res.success and res.fun < best_C5 else best_x

    # Get final solution
    u, a, s = decode(final_x, M)

    # Convert to step function
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        x = (i + 0.5) * dx
        j = 0
        left = s[0]
        right = s[1]
        while x >= right and j+1 < len(s)-1:
            j += 1
            left = right
            right = s[j+1]
        h[i] = a[j]
    
    # Apply capped-simplex projection to ensure integral is exactly 1
    h = project_capped_simplex(h, n_points/2)

    # Final evaluation
    corr = np.convolve(h, 1 - h, mode='full') * dx
    final_C5 = np.max(corr)
    return (h.tolist(), float(final_C5), n_points)
```
