import numpy as np
import time
from scipy import fftpack

def autocorr(b):
    """Compute the autocorrelation of a binary sequence using FFT."""
    n = len(b)
    B = np.fft.fft(b)
    A = np.fft.ifft(B * np.conj(B)).real
    return A

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using the exact computation."""
    dx = 2.0 / len(h)
    full_corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(full_corr) * dx
    return c5

def compute_min_autocorr(b):
    """Compute the minimal non-zero autocorrelation of binary sequence b."""
    A = autocorr(b)
    # Ignore lag 0
    return np.min(A[1:])

def swap_improvement(b, i, j, A, k_star):
    """Compute the change in autocorrelation at lag k_star by swapping b[i] and b[j]."""
    delta = 0
    for k in range(len(b)):
        delta += (b[(j+k)%len(b)] - b[(i+k)%len(b)]) * (b[(i+k)%len(b)] - b[(j+k)%len(b)])
        if k == k_star:
            delta = -b[(i+k)%len(b)] + b[(j+k)%len(b)]
            break
    return delta

def local_search_binary(b, max_iter=10000, T0=1.0, cooling=0.99, accept_every=1000):
    """Perform local search on binary vector to maximize minimal autocorrelation."""
    n = len(b)
    S = n // 2  # sum of b should be n/2
    A = autocorr(b)
    m = np.min(A[1:])
    k_star = np.argmin(A[1:]) + 1

    # Precompute shift matrices to speed up delta computation
    B_shift = np.array([np.roll(b, k) for k in range(n)])
    
    T = T0
    last_improved = 0
    best_b = b.copy()
    best_m = m
    best_A = A.copy()
    
    accepted = 0
    for i_iter in range(max_iter):
        # Sample a 1 and a 0
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Precompute contribution at k_star of swapping
        old_i_k = B_shift[k_star, i]
        old_j_k = B_shift[k_star, j]
        delta = old_j_k - old_i_k
        # Now update A for all lags
        new_A = A.copy()
        for k in range(n):
            if k == k_star:
                diff = delta
            else:
                diff = B_shift[k, j] - B_shift[k, i]
            new_A[k] = A[k] + diff
            # In exactly one case we update the worst
            if new_A[k] < new_A[k_star] and k != k_star:
                k_star = k
                # Verify new m is <= to new_A[k_star]
                if k_star == 0:
                    continue
            
        m_new = np.min(new_A[1:])
        r = np.random.random()
        if m_new > m or (m_new == m and r < np.exp(delta / T)):
            b[i], b[j] = b[j], b[i]  # swap
            A = new_A
            m = m_new
            if m > best_m:
                best_b = b.copy()
                best_m = m
                best_A = A.copy()
            last_improved = i_iter
        if i_iter % accept_every == 0 and i_iter > 0:
            T *= cooling
            
        if i_iter - last_improved > 5000:
            break
    return best_b, best_m, best_A

def seed_from_legendre(n):
    """Create a balanced binary vector based on quadratic residue sequence."""
    p = 211  # Next odd prime > n
    s = [0] * p
    for i in range(p):
        if (i * i) % p < p // 2:
            s[i] = 1
    s = s[:n]
    if sum(s) > n // 2:
        # Too many ones
        to_flip = sorted(range(n), key=lambda i: (1-s[i]) * (i % 2 + 1))[:sum(s) - n//2]
        for i in to_flip:
            s[i] = 1 - s[i]
    elif sum(s) < n // 2:
        # Too few ones
        to_flip = sorted(range(n), key=lambda i: s[i] * (i % 2 + 1))[:n//2 - sum(s)]
        for i in to_flip:
            s[i] = 1 - s[i]
    return np.array(s)

def expand_to_longer(n_target, b):
    """Expand the binary vector by duplicating its entries."""
    if len(b) % 2 != 0:
        b = np.pad(b, (0, 1), 'constant', constant_values=0)
    b = np.repeat(b, 2)
    if len(b) > n_target:
        return b[:n_target]
    else:
        return b
