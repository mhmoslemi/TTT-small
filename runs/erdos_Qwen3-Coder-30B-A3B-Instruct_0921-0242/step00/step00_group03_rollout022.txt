import numpy as np
import scipy.optimize as opt
import time
from scipy.interpolate import interp1d

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given step function h"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation: this computes all shifts
    corr = np.correlate(h, g, mode='full') * dx
    # The correlation is of length 2*n - 1, centered at index n-1
    # The shifts k range from -(n-1)*dx to (n-1)*dx in steps of dx
    # We want the maximum over all k, which is max of correlation values
    return np.max(corr)

def gradient_of_c5(h):
    """Compute gradient of C5 with respect to h (analytically)"""
    dx = 2.0 / len(h)
    g = 1.0 - h
    # Full correlation
    corr = np.correlate(h, g, mode='full')
    # Find which shift gives the maximum overlap
    max_idx = np.argmax(corr)
    # Shift in units of dx; k = idx - (n-1)
    k = max_idx - (len(h) - 1)
    
    # Gradient is a vector with entries:
    # For shift k, d/dh_i = dx * (1 - h_{i+k}) - dx * h_{i-k}
    # indices are wrapped around
    grad = np.zeros_like(h)
    
    # The indices that contribute to gradient for shift k
    # For each i, we need heuristics about automating this:
    # Index i in h contributes to term h_i * (1 - h_{i+k}) if k >= 0
    # and to term h_i * h_{i-k} if k < 0
    for i in range(len(h)):
        # Contribution from term: h_i * (1 - h_{i+k})
        i_plus_k = (i + k) % len(h)
        # Contribution from term: h_i * h_{i-k}
        i_minus_k = (i - k) % len(h) if k != 0 else 0 # Avoid division by 0 or doubled term
        grad[i] = dx * (1.0 - h[i_plus_k]) - dx * h[i_minus_k]
    return grad

def project_to_feasible(h):
    """Ensure sum(h) = n/2 and h in [0,1]^n"""
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    
    # Project into feasible set with bounds [0,1] and sum = n/2
    # Simple 1D projection (delta distributed uniformly)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    deficit = n/2 - sum_h
    if abs(deficit) < 1e-10:
        return h
    # Distribute deficit
    # Roughly: we spread deficit to all points that are not at boundary
    # and not particularly sensitive to boundary value
    # Use a uniform adjustment
    adjustment = deficit / n
    h += adjustment
    h = np.clip(h, 0, 1)
    return h
