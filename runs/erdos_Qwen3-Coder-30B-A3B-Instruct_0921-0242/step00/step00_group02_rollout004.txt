import cvxpy as cp
from scipy.linalg import svd

import time

        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = max(corr) * dx
        return c5

    def solve_sdp(n):
        X = cp.Variable((n, n), symmetric=True)
        h = cp.Variable(n)
        s = cp.Variable()

        # Constraints
        constraints = [
            cp.diag(X) == h,
            X >> 0,
            0 <= h,
            h <= 1,
            cp.sum(h) == n / 2
        ]

        # Add correlation constraints
        for k in range(1, n):
            # Sum of X_{i,i+k} for i = 0 to n-k-1
            corr_sum = cp.sum([X[i, i + k] for i in range(n - k)])
            constraints.append(corr_sum >= s)

        prob = cp.Problem(cp.Maximize(s), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4, max_iters=5000)
        except:
            return None
        if prob.status != cp.OPTIMAL:
            return None

        h_opt = h.value
        X_opt = X.value

        return h_opt, X_opt

            if abs(np.sum(b) - n/2) > 1e-6:
                continue
            c5 = evaluate_h(b)
            if c5 < best_c5:
                best_c5 = c5
                best_h = b.copy()
        return best_h

            return corr

            delta = 0

                    if idx - k_idx >= 0 and idx < n and idx - k_idx < n and idx >= 0:

                    if idx + k_idx < n and idx >= 0 and idx + k_idx >= 0 and idx < n:

                            delta += h[idx] - h[idx + k_idx]
            return delta

            h[i], h[j] = h[j], h[i]
            return h
