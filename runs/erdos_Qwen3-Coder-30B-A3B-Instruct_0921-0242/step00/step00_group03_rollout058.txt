import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from math import ceil

def project_to_F(h, n):
    """Project h onto the feasible set {h : sum(h) = n/2, 0 <= h[i] <= 1}"""
    h = np.clip(h, 0.0, 1.0)
    deficit = n / 2 - h.sum()
    free = np.where((h > 0) & (h < 1))[0]
    while np.abs(deficit) > 1e-12 and free.size > 0:
        delta = deficit / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        deficit = n / 2 - h.sum()
        free = np.where((h > 0) & (h < 1))[0]
    return h

def evaluate_fast(h, dx, n):
    """Fast evaluation using FFT for full convolution"""
    g = 1 - h
    L = 2 * n
    h_pad = np.pad(h, (0, L - n), 'constant')
    g_pad = np.pad(g, (0, L - n), 'constant')
    H = fft(h_pad)
    G = fft(g_pad)
    conv = ifft(H * np.conj(G)).real * dx
    C5 = conv.max()
    return C5, conv

def local_search(h, g, conv, dx, n, loc_steps=30):
    """Apply pairwise exchanges to improve the worst correlation"""
    h = h.copy()
    c = conv.copy()
    for _ in range(loc_steps):
        k_star = np.argmax(c)
        # Find indices that contribute to this shift
        # This is approximate; we do a full sweep with O(n) update
        idx1 = np.random.randint(0, n)
        idx2 = np.random.randint(0, n)
        if h[idx1] <= 0.0 or h[idx2] >= 1.0:
            continue
        # Determine step size
        step = min(h[idx1], 1.0 - h[idx2], 0.05)
        if step <= 1e-10:
            continue
        # Compute effect on correlations
        h_new = h.copy()
        h_new[idx1] -= step
        h_new[idx2] += step
        # Updates to correlation array
        c_new = c.copy()
        # The change in correlation due to swapping these indices
        for k in range(1, n):
            # If either index is part of this correlation
            # 1. Index idx1 gets -step in h[idx1+k] or -step in h[idx1-k]
            # 2. Index idx2 gets +step in h[idx2+k] or +step in h[idx2-k]
            if idx1 + k < n:
                c_new[k] -= step * dx * g[idx1]
            if idx2 + k < n:
                c_new[k] += step * dx * g[idx2]
            if idx1 - k >= 0:
                c_new[-k] -= step * dx * g[idx1]
            if idx2 - k >= 0:
                c_new[-k] += step * dx * g[idx2]
        # Special case for k=0
        c_new[0] -= step * dx * (h[idx1] - h[idx2])  # Only change because of the full-shift
        # Direct computation
        new_correlations = np.zeros_like(c_new)
        for k in range(len(c_new)):
            if k == 0:
                # No shift
                new_correlations[k] = 0
            else:
                if k > 0:
                    # Forward shift
                    shift_start = k
                    shift_end = min(shift_start + n, 2*n)
                    # h[shift_start:shift_end] * g[:shift_end-shift_start]
                    if shift_start < n and (shift_end - shift_start) > 0:
                        # h[shift_start:] * g[:n-shift_start]
                        new_correlations[k] = np.sum(h[shift_start:] * g[:n-shift_start]) * dx
                    else:
                        new_correlations[k] = 0
                else:
                    # Backward shift
                    shift_start = -k
                    shift_end = min(shift_start + n, 2*n)
                    if shift_start < n and (shift_end - shift_start) > 0:
                        # h[:shift_end] * g[shift_start:shift_start + (shift_end - shift_start)]
                        new_correlations[k] = np.sum(h[:shift_end] * g[shift_start:shift_start + (shift_end - shift_start)]) * dx
                    else:
                        new_correlations[k] = 0
        # Add direct O(n) update
        old_conv = c
        new_conv = new_correlations
        # Just one update for now:
        if new_conv.max() < c.max():
            # Accept
            h = h_new
            c = new_conv
    return h, c.max()

def make_initial_pop(n, initial_h_values, pop_size=30):
    """Generate a diverse initial population"""
    pop = []
    # Seed 1: Interpolate initial values to length n
    source = initial_h_values
    if len(source) > n:
        # downsample
        downsample_factor = len(source) // n
        step = len(source) // n
        interpolated = np.array([source[i] for i in range(0, len(source), step)])
        if len(interpolated) > n:
            interpolated = interpolated[:n]
        elif len(interpolated) < n:
            # pad
            interpolated = np.pad(interpolated, (0, n - len(interpolated)), 'constant', constant_values=(0.0,))
    else:
        # upsample
        interpolated = np.interp(np.linspace(0, len(source), n), np.arange(len(source)), source)
    h = project_to_F(interpolated, n)
    c5, _ = evaluate_fast(h, 2.0/n, n)
    pop.append((h, c5))
    
    # Seed 2: Random feasible
    for _ in range(pop_size // 4):
        r = np.random.uniform(0, 1, n)
        h = project_to_F(r, n)
        c5, _ = evaluate_fast(h, 2.0/n, n)
        pop.append((h, c5))
    
    # Seed 3: Block binary
    block = np.array([1.0] * (n // 2) + [0.0] * (n - n // 2))
    np.random.shuffle(block)
    h = project_to_F(block, n)
    c5, _ = evaluate_fast(h, 2.0/n, n)
    pop.append((h, c5))

    # Seed 4: Gaussian perturbed
    base = np.full(n, 0.5)
    noise = np.random.normal(0, 0.1, n)
    h = project_to_F(base + noise, n)
    c5, _ = evaluate_fast(h, 2.0/n, n)
    pop.append((h, c5))
    
    # Populate rest with random feasible
    while len(pop) < pop_size:
        r = np.random.uniform(0, 1, n)
        h = project_to_F(r, n)
        c5, _ = evaluate_fast(h, 2.0/n, n)
        pop.append((h, c5))
    
    return pop

def tournament_selection(pop, tournament_size=3):
    """Pick an individual via tournament selection"""
    selected = np.random.choice(len(pop), tournament_size)
    best_idx = min(selected, key=lambda i: pop[i][1])
    return pop[best_idx][0].copy()

def crossover(p1, p2):
    """Blend crossover (arithmetic blend)"""
    alpha = np.random.uniform(0, 1)
    child = alpha * p1 + (1 - alpha) * p2
    return child

def mutate(h, n, sigma):
    """Add Gaussian noise to some coordinates"""
    mask = np.random.choice(n, size=max(1, int(0.02 * n)), replace=False)
    noise = np.random.normal(0, sigma, len(mask))
    h = h.copy()
    h[mask] += noise
    return h
