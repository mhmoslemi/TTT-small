import numpy as np

def autocorr(b):
    """Compute autocorrelation of binary sequence b."""
    n = len(b)
    B = np.fft.fft(b)
    ac = np.fft.ifft(B * np.conj(B)).real
    return ac

def score_autocorr(b):
    """Return min autocorrelation (excluding lag 0)."""
    ac = autocorr(b)
    return np.min(ac[1:])

def make_seed_binary(b_v, n_points):
    """
    Convert a continuous function into a binary one.
    """
    if len(b_v) != n_points:
        raise ValueError("b_v must have length n_points")
    # Flatten and round values near 0.5 to either 0 or 1 more carefully
    mean_val = np.mean(b_v)
    # Start from initial h (likely near continuous)
    v_binary = (b_v > 0.5).astype(int)
    # Adjust to ensure sum = n_points/2 exactly
    current_sum = np.sum(v_binary)
    diff = n_points // 2 - current_sum
    if diff > 0:
        # Add ones elsewhere
        zero_indices = np.where(v_binary == 0)[0]
        if len(zero_indices) < diff:
            # not enough zeros to add ones; this shouldn't happen in practice
            diff = len(zero_indices)
        flip_indices = np.random.choice(zero_indices, diff, replace=False)
        v_binary[flip_indices] = 1
    elif diff < 0:
        # Remove ones
        one_indices = np.where(v_binary == 1)[0]
        if len(one_indices) < -diff:
            # shouldn't happen
            diff = -len(one_indices)
        flip_indices = np.random.choice(one_indices, -diff, replace=False)
        v_binary[flip_indices] = 0
    return v_binary

def swap_improvement(b, i, j, A):
    """Compute change in worst autocorrelation if we swap positions i and j.
    Assumes b[i] = 1, b[j] = 0.
    """
    old_min_id = np.argmin(A[1:]) + 1
    old_min = A[old_min_id]
    delta = 0  # byte-wise swap middle
    
    diff = []
    for k in range(len(A)):
        if k == 0:
            continue
        off_i = (i + k) % len(b)
        off_j = (j + k) % len(b)
        delta_k = -b[off_i] + b[off_j]
        diff.append(delta_k)
    new_A = A.copy()
    new_A[1:] += np.array(diff)
    new_min = np.min(new_A[1:])
    return new_min - old_min

def simple_hill_climb(b, max_iter=100000):
    """Perform a local hill climb optimizing the autocorrelation."""
    A = np.fft.ifft(np.fft.fft(b) * np.conj(np.fft.fft(b))).real
    best_A = A.copy()
    best_b = b.copy()
    min_A = np.min(A[1:])
    iteration = 0
    tried = 0
    accepted = 0
    last_improved = 0

    one_indices = np.where(b == 1)[0]
    zero_indices = np.where(b == 0)[0]
    n = len(b)
    
    # Precompute shifts
    B_shift = {}
    for k in range(n):
        B_shift[k] = np.roll(b, k)

    while tried < max_iter and ((accepted == 0) or (iteration - last_improved < 5000)):
        tried += 1
        
        if len(one_indices) == 0 or len(zero_indices) == 0:
            break
        i = np.random.choice(one_indices)
        j = np.random.choice(zero_indices)
        
        # There's a pattern in positions that gives how A_k changes
        # e.g. for lag k, A_k = sum_{j} b_j b_{j+k}
        delta_A_k = []
        for k in range(n):
            if k == 0:
                delta_A_k.append(0)
                continue
            new_raw = B_shift[k][i] * (1 - B_shift[k][j])
            old_raw = B_shift[k][i] * B_shift[k][j]
            delta_A_k.append(new_raw - old_raw)
            
        new_A = A - delta_A_k
        new_min = np.min(new_A[1:])
        
        if new_min > min_A:
            # Improving move
            A = new_A
            min_A = new_min
            last_improved = iteration
            # update b
            b[i], b[j] = 0, 1
            # update indices
            one_indices = np.where(b == 1)[0]
            zero_indices = np.where(b == 0)[0]
            accepted += 1
        elif np.random.uniform() < 0.001:  # Periodic compromise per 1000 trials
            A = new_A
            min_A = new_min
            b[i], b[j] = 0, 1
            one_indices = np.where(b == 1)[0]
            zero_indices = np.where(b == 0)[0]
            accepted += 1
            
        iteration += 1
    return b, min_A, accepted

def generate_seeds(n_points, seed=42):
    """
    Generate candidate balanced binary vectors based on combinatorial methods.
    """
    np.random.seed(seed)
    # Make a large vector n_points=200 to have good seeds
    n_large = 200
    seeds = []

    # 1. Quadratic residue (Legendre)
    try:
        p = 211  # Large enough prime
        qr = ([1 if ((i*i) % p) == 0 else 0 for i in range(p)])
        qr_sum = sum(qr)
        qr = qr[:n_large]  # Already in range
        if qr_sum > n_large // 2:
            # too many 1s
            indices = np.arange(n_large)[np.array(qr) == 1]
            drop = qr_sum - n_large // 2
            to_drop = np.random.choice(indices, drop, replace=False)
            for i in to_drop:
                qr[i] = 0
        else:
            # too few
            indices = np.arange(n_large)[np.array(qr) == 0]
            add = n_large // 2 - qr_sum
            to_add = np.random.choice(indices, add, replace=False)
            for i in to_add:
                qr[i] = 1
        qr = np.array(qr)
        seeds.append(("Legendre", qr))
    except Exception as e:
        pass

    # 2. Singer difference set - Use length-specific construction 
    # Try n=183 (q=13), pad to 200
    try:
        b = np.zeros(183, dtype=int)
        b[0:61] = 1
        b[42:103] = 1
        b[83:144] = 1
        
        b = np.roll(b, 10)
        b = np.pad(b, (0, 17))
        b = b[:n_large]
        b = (b > 0).astype(int)
        seeds.append(("Singer", b))
    except Exception as e:
        pass

    # 3. Random balanced binary (adaptive)
    try:
        b = np.zeros(n_large, dtype=int)
        b[:n_large//2] = 1
        np.random.shuffle(b)
        seeds.append(("Random", b))
    except Exception as e:
        pass

    # Filter to keep only good seeds based on initial scoring
    kept_seeds = []
    for name, seed_b in seeds:
        if len(seed_b) != n_large:
            continue
        # Adjust to exact length n_points (reducing resolution)
        if n_large != n_points:
            # Downsample
            new_b = seed_b[:n_points]
            if len(new_b) < n_points:
                new_b = np.pad(new_b, (0, n_points - len(new_b)))
            # Often length mismatch happens, so:
            projected = np.linspace(0, n_large, n_points, dtype=int)
            projected = seed_b[projected]
            new_b = (projected > 0.5).astype(int)
        else:
            new_b = seed_b
        score = score_autocorr(new_b)
        if score > 0.9 * n_points / 4:
            kept_seeds.append((name, new_b, score))
    
    return kept_seeds

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    remaining_time = budget_s

    # 1. Generate seeds
    available_seeds = generate_seeds(400, seed=seed)
    if len(available_seeds) == 0:
        h_start = initial_h_values()
        h_start = np.array(h_start)
        h_start = (h_start > 0.5).astype(int) * 1.0
        assert int(h_start.sum()) == len(h_start) // 2
        target_n = 400
        h = h_start
    else:
        # Use best seed from here
        best_seed = max(available_seeds, key=lambda x: x[2])
        h = best_seed[1]
        target_n = len(h)

    # Ensure it has correct sum
    while h.sum() != target_n // 2:
        # Adjust equal numbers of 0s and 1s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > target_n // 2:
            # Remove some ones
            to_remove = len(ones) - target_n // 2
            idx = np.random.choice(ones, to_remove, replace=False)
            h[idx] = 0
        elif len(ones) < target_n // 2:
            to_add = target_n // 2 - len(ones)
            idx = np.random.choice(zeros, to_add, replace=False)
            h[idx] = 1

    # 2. Apply hill climb local search
    time_up_to_local = time.time() - start_time
    remaining_time -= time_up_to_local

    best_h = h.copy()
    best_score = score_autocorr(h)
    best_c5 = 0.0

    # Local search
    if remaining_time > 10:
        try:
            # Fast hill climbing
            local_h, local_score, _ = simple_hill_climb(h, max_iter=int(50000*(remaining_time / 120)))
            # Check if better
            if local_score > best_score:
                best_h = local_h
                best_score = local_score
        except Exception as e:
            pass

    # 3. Duplicate and adjust if needed
    final_h = best_h.copy()
    if len(final_h)*2 <= 1000:
        # Duplication
        final_h = np.repeat(final_h, 2)
    target_n = len(final_h)
    
    # Quick fine-tuning if time left
    if remaining_time > 10:
        try:
            fine_h, _, _ = simple_hill_climb(final_h, max_iter=20000)
            if score_autocorr(fine_h) > best_score:
                final_h = fine_h
        except Exception as e:
            pass

    # Final evaluation using the full numpy correlate
    dx = 2.0 / target_n
    # compute the final score
    cross_corr_full = np.correlate(final_h, 1 - final_h, mode='full')
    c5_best = dx * (target_n / 2 - np.min(cross_corr_full[target_n // 2:]))
    
    result_h = final_h.astype(float)
    # Ensure sum = 1 exactly
    s = np.sum(result_h) * dx
    if abs(s - 1.0) > 1e-9:
        ratio = 1.0 / s
        result_h *= ratio
    
    return (result_h, c5_best, len(result_h))
