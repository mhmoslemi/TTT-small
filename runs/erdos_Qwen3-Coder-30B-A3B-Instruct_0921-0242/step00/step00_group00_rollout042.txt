import numpy as np
import time
from scipy.linalg import cholesky
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Key constants
    M = 8  # Number of frequencies (excluding constant term)
    n_points_1 = 256
    n_points_2 = 512
    max_restarts = 15
    restart_budget = 0.4 * budget_s / max_restarts
    max_refine_iter = 200

    # Precompute the Fourier matrix
    def build_fourier_basis(n):
        # Create the matrix Phi: [1, cos(2pi m i/n), sin(2pi m i/n)] for m=1..M and i=0..n-1
        Phi = np.zeros((n, 2 * M))
        for i in range(n):
            for m in range(1, M+1):
                Phi[i, m-1] = np.cos(2 * np.pi * m * i / n)
                Phi[i, m-1 + M] = np.sin(2 * np.pi * m * i / n)
        return Phi

    # Fourier basis for 256 points
    Phi_256 = build_fourier_basis(n_points_1)
    P = np.dot(Phi_256.T, Phi_256)
    P_inv = np.linalg.inv(P)
    P_inv_phiT = np.dot(P_inv, Phi_256.T)

    def project_feasible(c):
        # c = [a1,...,aM, b1,...,bM] (2M vector)
        h = np.dot(Phi_256, c) + 0.5
        h_clamped = np.clip(h, 0, 1)
        delta = h_clamped - h
        c_corr = np.dot(P_inv_phiT, delta)
        return c + c_corr
    
    def evaluate_objective(c):
        h = np.dot(Phi_256, c) + 0.5
        # Ensure h is in [0, 1] (it should be due to projection)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr

    def cma_es_restart(c0, n_pop, sigma, iters, time_limit):
        c_best = c0.copy()
        best_f = evaluate_objective(c_best)
        t0 = time.time()
        
        for t in range(iters):
            if time.time() - t0 > time_limit:
                break
                
            # Sample population
            pop = np.random.randn(n_pop, 2*M) * sigma
            for i in range(n_pop):
                pop[i] += c_best
                pop[i] = project_feasible(pop[i])
                
            # Evaluate
            values = [evaluate_objective(x) for x in pop]
            idx = np.argsort(values)
            pop = pop[idx]
            values = np.array(values)[idx]
            
            # Keep top μ
            mu = n_pop // 2
            pop_top = pop[:mu]
            values_top = values[:mu]
            
            # Update mean
            c_new = np.mean(pop_top, axis=0)
            c_new = project_feasible(c_new)

            if values_top[0] < best_f:
                best_f = values_top[0]
                c_best = c_new.copy()
        
        return c_best, best_f

    # Use larger resolution if needed
    def run_multi_resolution():
        c_best = np.zeros(2*M)
        best_score = float('inf')

        n_points = n_points_1
        Phi = Phi_256
        P = np.dot(Phi.T, Phi)
        P_inv = np.linalg.inv(P)
        P_inv_phiT = np.dot(P_inv, Phi.T)

        for _ in range(2):  # Run on both resolutions
            if time.time() - start_time > budget_s - 0.5:
                break
                
            # Multi-restart CMA-ES
            n_pop = 4 + int(3 * np.log(2*M))
            all_scores = []
            all_cands = []
            for k in range(max_restarts):
                if time.time() - start_time > budget_s - 0.5:
                    break
                    
                np.random.seed(seed * 10000 + k)
                c0 = np.random.randn(2*M) * 0.2
                c_opt, f_opt = cma_es_restart(c0, n_pop, 0.2, 200, restart_budget)
                all_scores.append(f_opt)
                all_cands.append(c_opt)
            
            best_idx = np.argmin(all_scores)
            if all_scores[best_idx] < best_score:
                best_score = all_scores[best_idx]
                c_best = all_cands[best_idx].copy()
                
            # If we achieved goal, no need to go further
            if best_score <= 0.379:
                break
                
            # Move to higher resolution
            n_points = n_points_2
            # Phi is now the higher resolution one
            # We keep coefficients unchanged, just increase resolution
            Phi = build_fourier_basis(n_points)
            P = np.dot(Phi.T, Phi)
            P_inv = np.linalg.inv(P)
            P_inv_phiT = np.dot(P_inv, Phi.T)
    
        return c_best, n_points, best_score

    # Run on both resolutions
    c_final, n_points, score = run_multi_resolution()

    # Local refinement via projected gradient descent
    def project_feasible(c, Phi, P_inv_phiT):
        h = np.dot(Phi, c) + 0.5
        h_clamped = np.clip(h, 0, 1)
        delta = h_clamped - h
        c_corr = np.dot(P_inv_phiT, delta)
        return c + c_corr

    def compute_grad(c, Phi, n):
        h = np.dot(Phi, c) + 0.5
        corr = np.correlate(h, 1-h, mode='full')
        k_star = np.argmax(corr)
        # Construct gradient w.r.t. h
        grad_h = np.zeros(n)
        # Find the positions in the convolved array where it's maximum
        # So, for the full convolution, i-th element of correlation
        # is sum over all j where j+k_star = i : h[j] * (1 - h[j-k_star])
        # Alternatively, treat it as matrix system in FFT
        # Gradient of C5 = sum_i (1 - h_{i+k*}) * delta_{i, j} + (h_{i-k*}) * delta_{i, j}
        # So we just take 1 - h shifted by k* and h shifted by -k*
        pi = np.zeros(n)
        for i in range(n):
            j = (i + k_star) % n
            pi[i] = 1 - h[j]
            j = (i - k_star) % n
            pi[i] += h[j]
        grad_h = pi
        grad_c = np.dot(Phi.T, grad_h)
        return grad_c

    # Refinement loop
    Phi = build_fourier_basis(n_points)
    P = np.dot(Phi.T, Phi)
    P_inv = np.linalg.inv(P)
    P_inv_phiT = np.dot(P_inv, Phi.T)

    for t in range(max_refine_iter):
        if time.time() - start_time > budget_s - 0.5:
            break
        grad_c = compute_grad(c_final, Phi, n_points)
        alpha = 0.1 / (1 + t * 0.01)
        c_final -= alpha * grad_c
        c_final = project_feasible(c_final, Phi, P_inv_phiT)

    # Final reconstruction
    h_final = np.dot(Phi, c_final) + 0.5
    h_final = np.clip(h_final, 0, 1)
    corr = np.correlate(h_final, 1 - h_final, mode='full')
    c5_bound = np.max(corr)
    
    return (h_final.tolist(), float(c5_bound), n_points)
