import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h_values"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h)
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Take maximum value (corresponding to the worst overlap)
    max_corr = np.max(corr)
    return max_corr * dx

def initial_h_values():
    """Return initial h function as specified by the problem"""
    n = 70
    h = np.zeros(n)
    # Simple construction: step function with 3 segments
    # First segment: 0 to 1/3, value 1
    # Second segment: 1/3 to 2/3, value 0  
    # Third segment: 2/3 to 1, value 1
    h[:n//3] = 1.0
    h[2*n//3:] = 1.0
    # Adjust for integral constraint
    integral=h.sum() * 2.0 / n
    h *= (1.0 / integral)
    return h

def objective(h_values):
    """Objective function to minimize for C5"""
    return evaluate_erdos_solution(h_values)

def bounds(n_points):
    """Return bounds for optimization"""
    return [(0, 1) for _ in range(n_points)]

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1""" 
    n_points = len(h_values)
    dx = 2.0 / n_points
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    """Find an improved upper bound for C5"""
    np.random.seed(seed)
    
    # Start with the initial construction
    h0 = initial_h_values()
    n_points = len(h0)
    
    # Define the optimization constraints
    eq_constraint = {'type': 'eq', 'fun': constraint_eq}
    
    # Optimization parameters
    maxiter = 1000  # Something reasonable for given runtime
    solution_time = 0.0
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    # Perform optimization using differential evolution
    # This is a derivative-free method which is robust to local optima
    try:
        # Define bounds for each variable
        bounds_list = bounds(n_points)
        
        # Run differential evolution with Manhattan distance to avoid early local minimum
        result = differential_evolution(
            objective, 
            bounds_list, 
            args=(),
            maxiter=300,  # Reasonable iterations within budget
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=seed,
            callback=lambda x, convergence: None,
            disp=False
        )
        
        # If successful, check if we got a better solution
        if not result.failed:
            h_opt = result.x
            c5_opt = result.fun
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except Exception:
        pass
    
    # Fine-tune with local optimization if still time
    if time.time() - start_time < budget_s * 0.9:
        try:
            result = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds_list,
                constraints=eq_constraint,
                options={'maxiter': 1000,
                         'ftol': 1e-12,
                         'gtol': 1e-12},
                callback=lambda x: None
            )
            
            if result.success:
                c5_new = result.fun
                if c5_new < best_c5:
                    best_h = result.x
                    best_c5 = c5_new
                    
        except Exception:
            pass
            
    # Final validation and ensure constraints
    dx = 2.0 / n_points
    integral_check = np.sum(best_h) * dx
    # Clamp values to [0, 1]
    best_h = np.clip(best_h, 0, 1)
    
    # Re-normalize to get exact integral
    best_h *= (1.0 / integral_check)
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
