```python
import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
from scipy.interpolate import interp1d

def compute_c5(h, dx):
    n = len(h)
    # Compute cross-correlation using FFT
    H = fft(h)
    auto_corr = ifft(np.abs(H)**2).real
    # The maximum cross-correlation (i.e. minimum overlap)
    min_auto_corr = np.min(auto_corr)
    return dx * (0.5 * n - min_auto_corr)

def evaluate_h(h, dx):
    """Evaluate using the discrete correlation directly."""
    n = len(h)
    result = []
    for k in range(n):
        corr = 0.0
        for i in range(n):
            corr += h[i] * (1 - h[(i + k) % n])
        result.append(corr)
    return np.max(result) * dx

def block_family_enum(n_points=400, delta=0.02):
    """Enumerate 3-block families."""
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Sweep breakpoints a,b where 0 < a < b < 2
    for a in np.arange(delta, 2, delta):
        for b in np.arange(a + delta, 2, delta):
            p = (1 - a) / (b - a)
            if 0 <= p <= 1:
                h = np.zeros(n_points)
                start_a = int(a / dx)
                end_a = int(b / dx)
                h[:start_a] = 1.0
                h[start_a:end_a] = p
                h[end_a:] = 0.0
                # Normalize to sum to n_points/2
                h = h * (n_points / 2) / np.sum(h)
                c5 = evaluate_h(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    return best_h, best_c5

def four_block_family_enum(n_points=400, delta=0.04):
    """Enumerate 4-block families with two parameters p, q."""
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')

    # Sweep a, b, c
    for a in np.arange(delta, 2, delta):
        for b in np.arange(a + delta, 2, delta):
            for c in np.arange(b + delta, 2, delta):
                # For a three-block state: [1, p, 0]
                # But we consider [1, p, q, 0] with integral constraint:
                # a*1 + (b-a)*p + (c-b)*q + (2-c)*0 = 1
                # Also we impose a balance condition on p and q, e.g., (b-a)*p = (c-b)*q (equal masses)
                # => q = p * (b-a)/(c-b)
                p = (1 - a) / (b - a) 
                if p < 0:
                    continue
                q = p * (b - a) / (c - b)
                if q < 0:
                    continue
                if 0 <= q <= 1:
                    h = np.zeros(n_points)
                    start_a = int(a / dx)
                    start_b = int(b / dx)
                    start_c = int(c / dx)
                    h[:start_a] = 1.0
                    h[start_a:start_b] = p
                    h[start_b:start_c] = q
                    h[start_c:] = 0.0
                    h = h * (n_points / 2) / np.sum(h)
                    c5 = evaluate_h(h, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    return best_h, best_c5

def de_stage(n_points=200, budget_s=120):
    """Stochastic search using Differential Evolution."""
    dx = 2.0 / n_points
    start_time = time.time()
    
    def objective(h):
        # Keep tract of integral
        norm = np.sum(h) * dx
        if norm == 0:
            return float('inf')
        h = h * 0.5 / norm
        return -compute_c5(h, dx)
    
    bounds = [(0.0, 1.0)] * n_points
    res = differential_evolution(
        objective,
        bounds,
        seed=42,
        maxiter=1000,
        popsize=30,
        disp=False,
        callback=lambda x, convergence: time.time() - start_time > budget_s,
        tol=1e-6
    )
    
    final_h = res.x * 0.5 / (np.sum(res.x) * dx)
    final_h = np.clip(final_h, 0, 1)
    return final_h, -res.fun

def project_to_feasible(h, n):
    """Project h onto feasibility conditions."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Correct sum to n/2
    mu = np.mean(h)
    h = h - mu + 0.5 * n / n
    h = np.clip(h, 0, 1)
    h = h * (0.5 * n) / np.sum(h)
    return h

def gradient_ascent_stage_start(h0, n_points, budget_s):
    dx = 2.0 / n_points
    start_time = time.time()
    h = h0.copy()
    h = project_to_feasible(h, n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    last_improvement = 0.0
    iterations_since_improvement = 0
    kstar = 0
    step_size = 0.5
    for i in range(10000):
        if time.time() - start_time > budget_s:
            break
        # Evaluate auto-correlation using FFT
        H = fft(h)
        auto_corr = ifft(np.abs(H)**2).real
        kstar = np.argmin(auto_corr)
        grad = 2 * np.roll(h, kstar)
        h = h + step_size * grad
        h = project_to_feasible(h, n_points)
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            iterations_since_improvement = 0
        else:
            iterations_since_improvement += 1
        if iterations_since_improvement > 200:
            break
        step_size *= 0.98
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    dx = 2.0 / 400
    
    # Stage 1: Block family enumeration (3-block)
    h1, c5_1 = block_family_enum(n_points=400, delta=0.02)
    if h1 is None:
        h1 = np.zeros(400)
        h1[:200] = 1.0
        h1 = h1 * 0.5 / 200

    # Stage 2: Differential Evolution
    h2, c5_2 = de_stage(n_points=200, budget_s=120)
    if h2 is None:
        h2 = np.zeros(200)
        h2[:100] = 1.0
        h2 = h2 * 0.5 / 100

    # Select best initial vector for gradient stage
    if c5_1 < c5_2:
        h_start = h1
        c5_ref = c5_1
    else:
        h_start = h2
        c5_ref = c5_2

    # Interpolate to 400 points for high-resolution analysis
    x_orig = np.linspace(0, 2, 400)
    x_refine = np.linspace(0, 2, 400)
    f = interp1d(x_orig, h_start, kind='linear', fill_value='extrapolate')
    h_start = f(x_refine)

    # Stage 3: Gradient ascent with multi-resolution
    # Start with fine resolution
    h_final, c5_final = gradient_ascent_stage_start(h_start, 400, budget_s - (time.time() - start_time) - 50)
    
    final_n_points = 400
    final_h = h_final
    
    # Final evaluation
    true_c5 = compute_c5(final_h, 2.0/len(final_h))
    
    return list(final_h), true_c5, final_n_points
```
