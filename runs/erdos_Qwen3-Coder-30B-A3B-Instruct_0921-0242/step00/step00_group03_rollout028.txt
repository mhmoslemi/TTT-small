```python
import time
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from scipy import signal

# Constants and helpers
n_points = 240
dx = 2.0 / n_points
initial_h_values = None

def project_to_F(h):
    """Project h onto the feasible set with exact sum n/2 and bounds [0,1]"""
    h = np.clip(h, 0.0, 1.0)
    deficit = n_points / 2 - h.sum()
    free = np.where((h > 0) & (h < 1))[0]
    while np.abs(deficit) > 1e-12 and free.size > 0:
        delta = deficit / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        deficit = n_points / 2 - h.sum()
        free = np.where((h > 0) & (h < 1))[0]
    return h

def compute_C5_fft(h):
    """Compute C5 using FFT - O(n log n)"""
    # Pad to next power of 2 for FFT efficiency
    L = 2 * n_points
    h_padded = np.pad(h, (0, L - n_points), 'constant')
    g_padded = np.pad(1 - h, (0, L - n_points), 'constant')
    
    H = fft(h_padded)
    G = fft(g_padded)
    conv = ifft(H * np.conj(G)).real * dx
    
    # Take full convolution, return maximum
    c5 = np.max(conv)
    return c5, conv

def local_improvement(h, conv_full):
    """Perform pairwise exchanges to decrease the maximum correlation"""
    h = h.copy()
    conv = conv_full.copy()
    loc_steps = 30
    for _ in range(loc_steps):
        # Find worst shift
        k_star = np.argmax(conv)
        # Try to improve by swapping two elements
        i1 = np.random.randint(n_points)
        i0 = np.random.randint(n_points)
        if h[i1] > 0 and h[i0] < 1:
            eps = min(h[i1], 1 - h[i0], 0.05)
            if eps > 1e-10:
                h[i1] -= eps
                h[i0] += eps
                # Recompute only affected correlation entries
                g = 1 - h
                for k in range(-n_points + 1, n_points):
                    idx1 = (i1 + k) % n_points
                    idx0 = (i0 + k) % n_points
                    conv[k + n_points - 1] -= eps * dx * (g[idx1] - g[idx0])
                # Ensure no negative values due to numerical误差
                conv = np.maximum(conv, 0)
                new_max = np.max(conv)
                if new_max < np.max(conv_full):
                    conv_full = conv.copy()
                else:
                    h[i1] += eps
                    h[i0] -= eps
    return h, np.max(conv_full)

def create_initial_population():
    """Create diverse initial population"""
    pop_size = 30
    population = []
    
    # Seed from initial_h_values (if available)
    if initial_h_values is not None:
        h_seed = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        h_seed = project_to_F(h_seed)
        c5, _ = compute_C5_fft(h_seed)
        population.append((h_seed, c5))
    
    # Random initializations
    for _ in range(pop_size - len(population)):
        # Uniform random
        h = np.random.uniform(0, 1, n_points)
        h = project_to_F(h)
        c5, _ = compute_C5_fft(h)
        population.append((h, c5))
    
    return population

def crossover(p1, p2, alpha):
    """Blend crossover of two parents"""
    child = alpha * p1 + (1 - alpha) * p2
    return project_to_F(child)

def mutate(h, sigma):
    """Mutation for continuous representation"""
    h = h.copy()
    m = max(1, int(0.02 * n_points))
    indices = np.random.choice(n_points, m, replace=False)
    noises = np.random.normal(0, sigma, m)
    h[indices] += noises
    return project_to_F(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety_margin = 15
    
    # Initial population
    population = create_initial_population()
    
    # Adaptive parameters
    sigma = 0.2
    elite_size = 2
    pop_size = 30
    max_generations = 2000
    generation_time_estimate = 0.35
    
    best_h = None
    best_C5 = np.inf
    
    # Main EA loop
    for gen in range(max_generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s - safety_margin:
            break
            
        # Tournament selection
        def tournament_selection():
            idxs = np.random.choice(len(population), 3, replace=False)
            best_idx = idxs[np.argmin([population[i][1] for i in idxs])]
            return population[best_idx][0]

        # Generate offspring
        new_population = population[:elite_size]  # Elitism
        while len(new_population) < pop_size:
            p1 = tournament_selection()
            p2 = tournament_selection()
            alpha = np.random.uniform(0, 1)
            child = crossover(p1, p2, alpha)
            child = mutate(child, sigma)
            
            # Local improvement
            _, conv_full = compute_C5_fft(child)
            improved_h, improved_C5 = local_improvement(child, conv_full)
            
            new_population.append((improved_h, improved_C5))
            
        # Merge and sort
        combined = population + new_population
        combined.sort(key=lambda x: x[1])
        population = combined[:pop_size]
        
        # Update best
        current_best_h, current_best_C5 = population[0]
        if current_best_C5 < best_C5:
            best_C5 = current_best_C5
            best_h = current_best_h.copy()
            
        # Update mutation sigma and occasional restart
        sigma = max(0.02, sigma * 0.995)
        if sigma < 0.02 and np.random.rand() < 0.05:
            sigma = 0.1
            for i in range(pop_size // 2):
                h = np.random.uniform(0, 1, n_points)
                population[i] = (project_to_F(h), np.inf)
    
    # Final polishing with SLSQP if we have a good solution
    if best_h is not None:
        h_final = best_h.copy()
        def obj(h):
            c5, _ = compute_C5_fft(h)
            return c5
            
        def jac(h):
            _, conv = compute_C5_fft(h)
            k_star = np.argmax(conv)
            g = 1 - h
            # Gradient for active shift
            grad = np.zeros_like(h)
            for i in range(n_points):
                # Use indices with appropriate boundary conditions (zero padding)
                k_idx = k_star
                j1 = (i + k_idx) % n_points
                j2 = (i - k_idx) % n_points
                grad[i] = dx * (g[j1] - h[j2])
            return grad
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        try:
            res = minimize(obj, h_final, method='SLSQP', jac=jac, bounds=[(0, 1)]*n_points,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = project_to_F(res.x)
        except:
            pass

    # Final evaluation
    c5_bound, _ = compute_C5_fft(h_final)
    h_final = project_to_F(h_final)
    return (h_final, c5_bound, n_points)
```
