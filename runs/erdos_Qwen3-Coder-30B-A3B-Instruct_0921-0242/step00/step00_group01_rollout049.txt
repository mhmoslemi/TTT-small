import numpy as np
import time
import math
from scipy import fftpack

def project_box_affine(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Project onto affine constraint
    delta = (np.sum(h) - target_sum) / len(h)
    h -= delta
    # Clip again
    h = np.clip(h, 0, 1)
    return h

def autocorr(h):
    # Compute circular autocorrelation using FFT
    H = fftpack.fft(h)
    autoc = fftpack.ifft(np.conj(H) * H).real
    return autoc

def compute_C5(h, dx):
    # compute max of overlaps with all lags
    n = len(h)
    # overlap with lag k is int h(x) (1 - h(x + k))
    # using correlation of h with (1-h) = int h - int h(x) h(x + k)
    # so overlap_k = dx * ( sum(h) - sum(h_i * h_{i+k}) )
    h1m = 1 - h
    corr = np.correlate(h, h1m, mode='full')
    # the overlap value for lag k (as integer)
    # is corr[n-1+k] * dx, but we don't want to loop
    # full corr is length 2n - 1
    # lag index from -n+1 to n-1, nbeg to n-1 in corr 
    # so we do the middle part of size n
    overlaps = np.zeros(n)
    for k in range(n):
        overlaps[k] = corr[n - 1 + k] * dx  # -k to k range
    return np.max(overlaps)

def evaluate_step_function(h, dx):
    return compute_C5(h, dx)

def initialize_candidates(n_points, seed=42):
    np.random.seed(seed)
    init_h = np.array(initial_h_values)
    # linearly interpolate to n_points
    x0 = np.linspace(0, 2, len(init_h))
    x_new = np.linspace(0, 2, n_points)
    h_interp = np.interp(x_new, x0, init_h)
    candidates = [h_interp]
    
    # add random binary candidates (1s, 0s)
    for _ in range(4):
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, n_points//2, replace=False)
        h[indices] = 1
        # smooth with some convolution
        kernel = np.array([0.25, 0.5, 0.25])
        h = np.convolve(h, kernel, mode='same')
        h = project_box_affine(h, n_points / 2)
        candidates.append(h)
    
    # deterministic block patterns
    # two-block: first region 1, rest 0
    h1_block = np.ones(n_points)
    h1_block[n_points//2:] = 0
    h1_block = project_box_affine(h1_block, n_points / 2)
    candidates.append(h1_block)
    
    # two-block uniform
    h2_block = np.ones(n_points)
    h2_block[n_points//4:] = 0
    h2_block = project_box_affine(h2_block, n_points / 2)
    candidates.append(h2_block)
    
    return candidates

def gradient_ascent_step(h, step_size, dx):
    # compute auto-correlations
    # negative will make it maximise
    aut = autocorr(h)
    # find k that minimizes aut (worst lag for overlap)
    k = np.argmin(aut)
    # gradient of -aut_k w.r.t. h_i is -(h_{i-k} + h_{i+k})
    grad = -(np.roll(h, k) + np.roll(h, -k))
    # gradient ascent
    h_new = h + step_size * grad
    h_new = project_box_affine(h_new, len(h) / 2)
    return h_new

def gradient_descent(h0, max_iter=1000, dx=0.01):
    h = h0.copy()
    step_size = 0.01
    prev_c5 = float('inf')
    
    for i in range(max_iter):
        # adaptive step size
        if i > 0 and i % 100 == 0:
            step_size *= 0.9
        
        h = gradient_ascent_step(h, step_size, dx)
        
        # compute C5 and check progress
        c5 = evaluate_step_function(h, dx)
        if c5 < prev_c5:
            prev_c5 = c5
        else:
            if i > 500 and abs(c5 - prev_c5) < 1e-8:
                break
        
        if c5 < 0.379:  # early stopping if target hit
            break
            
    return h, c5

def evaluate_block_pattern(a, b, n_points, dx):
    # two-block pattern: h=1 on [0,a], h=c on (a,b), h=0 else
    if a >= b:
        return None, float('inf')
    L1 = a
    L2 = b - a
    L3 = 2 - b
    
    if L2 <= 0:
        return None, float('inf')
    
    c = (1 - L1) / L2
    if c < 0 or c > 1:
        return None, float('inf')
    
    h = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    h[x <= a] = 1.0
    h[(x > a) & (x <= b)] = c
    h[x > b] = 0.0
    
    # ensure integral = 1
    h = project_box_affine(h, n_points / 2)
    c5 = evaluate_step_function(h, dx)
    return h, c5

def block_search(n_points, dx, time_budget, start_time):
    # Two-block pattern sweep
    # a in [best_c5, 2]
    # b in [a, 2]
    best_h = None
    best_c5 = float('inf')

    # Sweep from 0.05 to 1.95 in steps of 0.05
    step = 0.05
    for a in np.arange(step, 2, step):
        if time.time() - start_time > time_budget - 10:
            break
        for b in np.arange(a + step, min(2, a + 1.9), step):
            if time.time() - start_time > time_budget - 10:
                break
            h, c5 = evaluate_block_pattern(a, b, n_points, dx)
            if h is not None and c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # three-block 
    # h = 1 on [0,a]
    # h = p on (a,b] 
    # h = q on (b,c]
    # h = 0 on (c, 2]
    
    # approximate discrete 3-block: 1, p, q, 0
    best_h3 = None
    best_c5_3 = float('inf')
    for a in np.arange(step, 1.8, step):
        for b in np.arange(a + step, min(1.9, a + 1.8), step):
            for c in np.arange(b + step, 2, step):
                # L1=a, L2=b-a, L3=c-b, L4=2-c
                L1 = a
                L2 = b - a
                L3 = c - b
                L4 = 2 - c
                
                if L3 <= 0 or L4 <= 0:
                    continue
                
                # integral = L1*1 + L2*p + L3*q = 1
                # 1 = L1 + L2*p + L3*q
                # p = (1 - L1 - L3*q) / L2
                # bound 0 <= p <= 1, so 0 <= (1 - L1 - L3*q) / L2 <= 1
                # 0 <= 1 - L1 - L3*q <= L2
                # L1 <= 1 - L3*q <= L1 + L2
                
                # When L2=0, must have L1=L1+L2=1 and L3=0, so L1=1, L3=0 means q=0
                # But let's solve the constraints analytically:
                # first, L1 + L2*p + L3*q = 1
                # and 0 ≤ p ≤ 1, 0 ≤ q ≤ 1
                q_min = max(0, (1 - L1 - L2) / L3)  # from p ≤ 1
                q_max = min(1, (1 - L1) / L3)       # from p ≥ 0
                
                if q_min > q_max:
                    continue
                    
                # try middle point
                if q_min < q_max:
                    q = (q_min + q_max) / 2
                    p = (1 - L1 - L3*q) / L2
                else:
                    q = q_min
                    p = (1 - L1 - L3*q) / L2
                
                if not (0 <= p <= 1 and 0 <= q <= 1):
                    continue
                
                # define h
                h = np.zeros(n_points)
                x = np.linspace(0, 2, n_points)
                h[x <= a] = 1.0
                h[(x > a) & (x <= b)] = p
                h[(x > b) & (x <= c)] = q
                h[x > c] = 0.0
                
                # ensure integral = 1
                h = project_box_affine(h, n_points / 2)
                c5 = evaluate_step_function(h, dx)
                if c5 < best_c5_3:
                    best_c5_3 = c5
                    best_h3 = h.copy()
                    
    if best_h3 is not None and best_c5_3 < best_c5:
        return best_h3, best_c5_3
    else:
        return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 200
    dx = 2.0 / n_points
    
    # initial candidates
    candidates = initialize_candidates(n_points, seed)
    print(f"Initial candidates: {len(candidates)} generated.")
    
    # block search
    h_block, c5_block = block_search(n_points, dx, budget_s, start_time)
    if h_block is not None and c5_block < best_c5:
        best_c5 = c5_block
        best_h = h_block
    print(f"Block search result: C5 = {c5_block}")
    
    # gradient descent
    attempts = 0
    max_attempts = 20  # at most 20 gradient runs
    for i, h0 in enumerate(candidates):
        if attempts >= max_attempts:
            break
        if time.time() - start_time > budget_s - 10:
            break
            
        try:
            h, c5 = gradient_descent(h0, max_iter=1000, dx=dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                print(f"Got new best C5 at attempt {i}: {c5}")
            attempts += 1
        except Exception as e:
            print(f"Failed gradient descent on candidate {i}: {e}")
            continue
    
    # refine (if needed)
    if best_h is not None and best_c5 > 0.379:
        # try to refine with a higher resolution grid
        high_res_n = 400
        dx_hr = 2.0 / high_res_n
        x_hr = np.linspace(0, 2, high_res_n)
        x_lr = np.linspace(0, 2, n_points)
        h_lr_interpolated = np.interp(x_hr, x_lr, best_h)
        h_refined, c5_refined = gradient_descent(h_lr_interpolated, max_iter=500, dx=dx_hr)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined
            print(f"Refined C5 to {c5_refined}")
            
        if best_c5 > 0.379:  # if still too large, downsample to 100
            sampled_indices = np.arange(0, high_res_n, 2)  # every 2 points
            h_downsampled = h_refined[sampled_indices].copy()
            h_downsampled = project_box_affine(h_downsampled, 100 / 2)
            c5_down = evaluate_step_function(h_downsampled, 2.0/100)  
            if c5_down < best_c5:
                best_c5 = c5_down
                best_h = h_downsampled
                
    if best_h is None:
        # fall back to the one block result (already computed)
        best_h = h_block
        best_c5 = c5_block
    
    # final check: recompute the exact C5 for best h
    if best_h is not None:
        final_c5 = evaluate_step_function(best_h, dx)
        print(f"Final computed C5 = {final_c5}")
        return (best_h.tolist(), float(final_c5), len(best_h))

        # fallback to initial
        fallback_h = np.array(initial_h_values)
        fallback_h = project_box_affine(fallback_h, 2 / dx)
        final_c5 = evaluate_step_function(fallback_h, 2.0 / len(fallback_h))
        return (fallback_h.tolist(), final_c5, len(fallback_h))
