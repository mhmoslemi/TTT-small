```python
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.linalg import solve
from scipy.optimize import minimize

# Constants
N_START = 256
N_REFINE = 512
M = 8  # Number of frequencies per side (a_m, b_m)
N_PARAMS = 2 * M  # Total number of Fourier coefficients
SIGMA0 = 0.2
RESTARTS = 15
BUDGET_PER_RESTART = 35  # seconds
MAX_ITER_REFINE = 200

# Generate Fourier basis matrix
def build_fourier_basis(n, m):
    """Build a Fourier basis matrix for n points with m frequencies (not counting DC)"""
    x = np.arange(n) / n
    phim = np.zeros((n, 2 * m))
    for i in range(m):
        phim[:, 2*i] = np.cos(2 * np.pi * (i+1) * x)
        phim[:, 2*i+1] = np.sin(2 * np.pi * (i+1) * x)
    return phim

# Precomputed data
PHI_START = build_fourier_basis(N_START, M)
PHI_REFINE = build_fourier_basis(N_REFINE, M)
PHI_T_PHI_INV = np.linalg.inv(PHI_START.T @ PHI_START)

def project_to_feasible(c):
    """
    Projects coefficient vector c to the feasible set:
    0 <= h[i] <= 1 where h = PHI @ c + 0.5
    """
    h = PHI_START @ c + 0.5
    h_clipped = np.clip(h, 0, 1)
    delta = h_clipped - h
    delta_c = PHI_T_PHI_INV @ (PHI_START.T @ delta)
    c_proj = c + delta_c
    return c_proj

def evaluate_c5(c):
    """Evaluate the C5 bound for coefficient vector c"""
    h = PHI_START @ c + 0.5
    # Convolution
    h_rev = np.flip(h)
    corr = np.convolve(h, 1 - h_rev, mode='full')
    corr = corr * 2.0 / N_START  # dx = 2/N_START
    max_corr = np.max(corr)
    return max_corr

def optimize_cma_es(budget_per_restart):
    """Run CMA-ES multi-start optimization"""
    best_c = None
    best_value = float('inf')
    start_time = time.time()
    
    # Population size
    mu = 10  # Half population size
    lambda_val = 2 * mu
    sigma = SIGMA0
    mean = np.zeros(N_PARAMS)  # Initial mean
    
    while True:
        if time.time() - start_time > budget_per_restart - 1:  # Reserve time for final processing
            break
            
        # Sample new population
        population = []
        for _ in range(lambda_val):
            sample = np.random.multivariate_normal(mean, sigma ** 2 * np.eye(N_PARAMS))
            sample = project_to_feasible(sample)
            population.append(sample)
            
        # Evaluate population
        values = []
        for c in population:
            val = evaluate_c5(c)
            values.append(val)
            
        # Sort by value
        sorted_indices = np.argsort(values)
        top_indices = sorted_indices[:mu]
        
        # Update mean and sigma
        top_pop = [population[i] for i in top_indices]
        top_vals = [values[i] for i in top_indices]
        
        mean = np.mean(top_pop, axis=0)
        # Project mean back to feasibility
        mean = project_to_feasible(mean)
        
        # Update sigma adaptive step
        sigma = sigma * 0.995 if np.std(top_vals) < 0.05 else sigma
        
        # Track best
        min_val = min(top_vals)
        if min_val < best_value:
            best_value = min_val
            best_c = top_pop[np.argmin(top_vals)].copy()
            
    return best_c, best_value

def gradient_refinement(c, max_iter=MAX_ITER_REFINE):
    """Refine solution with gradient descent"""
    current_c = c.copy()
    h = PHI_START @ current_c + 0.5
    step_size = 0.01
    
    for i in range(max_iter):
        # Find k that maximizes the overlap
        corr = np.convolve(h, 1 - np.flip(h), mode='full')
        max_idx = np.argmax(corr)
        offset = max_idx - (len(h) - 1)
        
        # Compute gradient direction at optimal offset
        grad = np.zeros(len(h))
        for j in range(len(h)):
            # Positive contribution from h[j] (1 - h[j+offset])
            if j + offset >= 0 and j + offset < len(h):
                grad[j] += -1.0
            # Negative contribution from h[j+offset] (1 - h[j])
            if j - offset >= 0 and j - offset < len(h):
                grad[j] += 1.0
                
        # Project gradient to Fourier coefficients
        grad_coeff = PHI_START.T @ grad
        
        # Update
        current_c = current_c - step_size * grad_coeff
        current_c = project_to_feasible(current_c)
        
        # Recompute h
        h = PHI_START @ current_c + 0.5
        step_size *= 0.995  # Decaying step size
    
    return current_c

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial run at resolution N_START
    print(f"Starting CMA-ES optimization on {N_START} points")
    best_c = None
    best_value = float('inf')
    
    # Loop over restarts
    for i in range(RESTARTS):
        if time.time() - start_time > budget_s - 2:
            break
            
        # New random seed for each restart
        np.random.seed(seed * 1000 + i)
        
        c, val = optimize_cma_es(BUDGET_PER_RESTART)
        if val < best_value:
            best_value = val
            best_c = c.copy()
            
        print(f"Restart {i}: Best C5 = {val:.5f}")
            
    # Refinement stage at N_REFINE (if time allows)
    if time.time() - start_time < budget_s - 5 and N_REFINE > N_START:
        print("Refining at higher resolution")
        # Interpolate to higher resolution - take same coefficients
        c_coarse = best_c
        # Expand to higher resolution  
        c_fine = c_coarse  # Same coefficients; direction is respected by interpolation
        refined_c = gradient_refinement(c_fine)
        final_c = refined_c.copy()
        final_value = evaluate_c5(final_c)
        print(f"Refined C5 = {final_value:.5f}")
        final_h = PHI_REFINE @ final_c + 0.5
        final_h = np.clip(final_h, 0, 1)
    else:
        print("Reconstructing at original resolution")
        final_h = PHI_START @ best_c + 0.5
        final_h = np.clip(final_h, 0, 1)
        final_value = evaluate_c5(best_c)
        final_c = best_c
        
    # Final validation 
    integral = np.sum(final_h) * 2.0 / len(final_h)
    error = abs(integral - 1.0)
    if error > 1e-10:
        # Adjust to satisfy integral constraint
        correction = (1.0 - integral) / len(final_h)
        final_h = np.clip(final_h + correction, 0, 1)
        final_value = evaluate_c5(final_c)  # Reevaluate
    

    # Output final result
    h_vals = final_h.tolist()
    return (h_vals, float(final_value), len(final_h))
```
