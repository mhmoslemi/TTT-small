import numpy as np
import time
from scipy import signal

def project_simplex_box(v, total, lo=0., hi=1.):
    # Project vector v onto the simplex with box constraints [lo, hi]
    # Using the algorithm from "Projection Onto A Simplex" by Duchi et al.
    v = np.asarray(v)
    n = len(v)
    
    # Sort values in descending order
    u = np.sort(v)[::-1]
    # Compute cumulative sums
    su = np.cumsum(u)
    # Find threshold index
    rho = np.where(u > (su - total) / np.arange(1, n + 1))[0]
    if len(rho) == 0:
        # All elements should be at the upper bound
        theta = (su[-1] - total) / n
        result = np.clip(v - theta, lo, hi)
        return result
    else:
        # rho is nonempty, take the last element
        rho = rho[-1]
        theta = (su[rho] - total) / (rho + 1)
        result = np.clip(v - theta, lo, hi)
        return result

def compute_c5(h, dx):
    # Compute correlation using FFT
    h = np.array(h)
    correlation = signal.correlate(h, 1-h, mode="full")
    correlation = correlation[len(h)-1:]  # take the right half
    return np.max(correlation) * dx

def worst_lag(h):
    # Returns the index k with max correlation, and the correlation value
    h = np.array(h)
    correlation = signal.correlate(h, 1-h, mode="full")
    correlation = correlation[len(h)-1:]  # take the right half
    k_best = np.argmax(correlation)
    return k_best, correlation[k_best]

def gradient_worst(h, k):
    # Gradient of overlap for lag k at each position i
    h = np.array(h)
    n = len(h)
    grad = np.zeros(n)
    # FOR direction k: i -> i +k
    # BACKWARD "i -> i-k"
    for i in range(n):
        j1 = (i + k) % n
        j2 = (i - k) % n
        grad[i] = h[j1] + h[j2]
    return grad
