import numpy as np

# Available solvers
SOLVERS = ["SCS", "MOSEK", "ECOS", "GUROBI", "GLPK", "CBC"]
SOLVER = None
for solver in SOLVERS:
    try:
        cp.problems.problem.Problem.solve(cp.Variable(1), solver=solver)
        SOLVER = solver
        break
    except:
        continue
if not SOLVER:
    SOLVER = "SCS"  # fallback to SCS

def solve_sdp(n):
    """Solve the SDP relaxation for the given n."""
    # Variable definition
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    s = cp.Variable()

    # Constraints
    constraints = [
        cp.diag(X) == h,
        X >> 0,
        0 <= h,
        h <= 1,
        cp.sum(h) == n / 2
    ]

    # Add correlation constraints
    for k in range(1, n):
        # Sum of X_{i,i+k} over valid i
        expr = cp.sum([X[i, i+k] for i in range(n-k)])
        constraints.append(expr >= s)

    # Objective
    objective = cp.Maximize(s)

    # Problem
    prob = cp.Problem(objective, constraints)

    # Solve
    try:
        prob.solve(solver=SOLVER, eps=1e-4, max_iters=5000)
        return h.value, X.value, s.value
    except Exception as e:
        return None, None, None

def round_fractional(h, X, n):
    """Rounding via Gaussian hyperplane."""
    # Factorize
    try:
        U, V = eigh(X)
        r = np.sum(U > 1e-10)
        V = V[:, -r:]
        U = U[-r:]
        V = V @ np.diag(np.sqrt(U))
    except:
        # fallback to simpler method if eigh fails
        V = np.random.randn(n, 1)
        V = V / np.linalg.norm(V, axis=0, keepdims=True)

    # Sample 300 hyperplanes
    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / n

    for _ in range(300):
        g = np.random.randn(V.shape[1])
        z = V @ g
        # Quantile for mass constraint
        threshold = np.percentile(z, 100*(1 - n/2/n))
        b = (z >= threshold).astype(float)
        # Ensure correct mass (n/2)
        if np.sum(b) > n/2:
            # Need to set some 1s to 0
            indices_to_flip = np.argsort(z)[:int(np.sum(b) - n/2)]
            b[indices_to_flip] = 0
        elif np.sum(b) < n/2:
            # Need to set some 0s to 1
            indices_to_flip = np.argsort(z)[-int(n/2 - np.sum(b)):]
            b[indices_to_flip] = 1
        assert abs(np.sum(b) - n/2) < 1e-9

        # Compute objective
        c5 = compute_C5(b, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = b.copy()
    return best_h, best_c5

def compute_C5(h, dx):
    """Compute the C5 value."""
    n = len(h)
    r = np.correlate(h, 1-h, mode="full")
    return 1 - dx * np.min(r[n-1:])

def local_refinement(h, max_iter=30):
    """Project gradient ascent for refinement."""
    n = len(h)
    dx = 2.0 / n
    iter_count = 0
    while iter_count < max_iter:
        # Compute worst k
        r = np.correlate(h, 1-h, mode="full")
        min_k = np.argmin(r[n-1:])
        # Compute gradient of R_{min_k}(h)
        # dR_k/dh_i = h_{i+k} + h_{i-k} (when valid)
        grad = np.zeros(n)
        if min_k > 0:
            for i in range(min_k, n):
                grad[i] += h[i - min_k]
        if min_k < n-1:
            for i in range(n-min_k-1):
                grad[i] += h[i + min_k]
        
        # Gradient ascent step
        step_size = 0.01
        h_new = h + step_size * grad
        # Project to [0,1]^n and then to correct sum
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        diff = (n/2) - current_sum
        if abs(diff) > 1e-10:
            # Adjust by adding diff to largest elements
            indices = np.argsort(h_new)
            for i in reversed(indices):
                if diff >= 0:
                    add = min(1-h_new[i], diff)
                    h_new[i] += add
                    diff -= add
                else:
                    add = min(h_new[i], -diff)
                    h_new[i] -= add
                    diff += add
                if abs(diff) < 1e-10:
                    break
        # Ensure exact mass
        h_new = h_new * (n/2) / np.sum(h_new)
        assert abs(np.sum(h_new) - n/2) < 1e-9

        # Check improvement
        c5_new = compute_C5(h_new, dx)
        if c5_new < compute_C5(h, dx):
            h = h_new
            iter_count += 1
        else:
            break

    return h

def swap_polish(h, max_iter=500):
    """Polish a binary solution via pair swaps."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    r = np.correlate(h, 1-h, mode="full")
    min_r_k = np.argmin(r[n-1:])
    r_all = r[n-1:]
    iter_count = 0
    improvement = True
    while improvement and iter_count < max_iter:
        improvement = False
        best_improvement = -1e-15
        best_i, best_j = -1, -1
        for i in range(n):
            if h[i] == 1.0:
                for j in range(n):
                    if h[j] == 0.0:
                        # Estimate change in r[min_r_k]
                        # Count overlap with current min_r_k
                        change = 0.0
                        # Look at positions that are k distant
                        k = min_r_k
                        # Increment h[i] -> h[i+1] and decrement h[j] -> h[j-1] if applicable
                        # All indices affect only shifts involving i,j
                        # But manually check 2k around each such k different from min_r_k
                        for shift in range(1, n):
                            if i+shift < n:
                                if h[i] == 1 and h[i+shift] == 1 and h[j] == 0 and h[j+shift] == 0:
                                    change += 1
                                elif h[i] == 0 and h[i+shift] == 0 and h[j] == 1 and h[j+shift] == 1:
                                    change -= 1
                        # Check min_r_k alone directly
                        actual_change = 0.
                        # We must compute delta in R_{min_k} by swapping i,j
                        # Compute how this affects the number of double-ones at shift min_k
                        prev_val = 0
                        if i+min_k < n and i >= 0:
                            prev_val += h[i] * h[i+min_k]
                        if j+min_k < n and j >= 0:
                            prev_val += h[j] * h[j+min_k]
                            
                        new_val = 0
                        if i+min_k < n and i >= 0:
                            new_val += (1-h[i]) * h[i+min_k]
                        if j+min_k < n and j >= 0:
                            new_val += h[j] * (1-h[j+min_k])
                            
                        actual_change = new_val - prev_val
                        if actual_change > best_improvement:
                            best_improvement = actual_change
                            best_i, best_j = i, j
        if best_improvement > 1e-12:
            h[best_i] = 0
            h[best_j] = 1
            improvement = True
            # Update r_all based on swap and new min_r_k
            r = np.correlate(h, 1-h, mode="full")
            r_all = r[n-1:]
            min_r_k = np.argmin(r_all)
            iter_count += 1
        else:
            break
    return h
