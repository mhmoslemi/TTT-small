        dx = 2.0 / len(h)

            corr_val = np.correlate(h, 1 - h, mode='full')[len(h)-1+k]
            corr.append(corr_val * dx)
        return np.array(corr)

        # Define problem
        X = cp.Variable((n, n), symmetric=True)

        constraints = [
            X >> 0,  # Positive semi-definite
            cp.diag(X) == h,  # Diagonal constraint
            0 <= h, h <= 1,  # Box constraints
            cp.sum(h) == n / 2,  # Mass constraint
        ]

        # Add correlation constraints
        for k in range(1, n):
            # Sum of shifts
            expr = cp.sum([X[i, i+k] for i in range(n-k)])
            constraints.append(expr >= s)

        # Objective
        prob = cp.Maximize(s)
        prob = cp.Problem(prob, constraints)

        # Solve with protections
        try:
            prob.solve(solver=cp.SCS, eps=1e-4, max_iters=2000)
