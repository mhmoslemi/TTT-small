    def solve_sdp(n):
        """Solve SDP relaxation for given n"""
        # Create decision variables
        h = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        s = cp.Variable()
        
        # Constraints
        constraints = [
            X >> 0,  # PSD
            cp.diag(X) == h,  # Diag constraints
            0 <= h, h <= 1,  # Box constraints
            cp.sum(h) == n / 2,  # Mass constraint
        ]
        
        # Add correlation constraints for all shifts k
        dx = 2.0 / n
        for k in range(1, n):
            # sum(h_i * h_{i+k}) for i = 0 to n-1-k
            corr_sum = 0
            for i in range(n - k):
                corr_sum += X[i, i + k]
            constraints.append(corr_sum >= s)
        
        # Objective
        prob = cp.Problem(cp.Maximize(s), constraints)
        
        # Solve with SCS solver
        prob.solve(solver=cp.SCS, eps=1e-4, max_iters=1000)
        
        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return None, None
            
        return h.value, X.value
    
    def round_hyperplane(h_opt, X_opt, n, num_rounds=300):

            X_safe = X_opt + 1e-6 * np.eye(n)
            # Perform eigen-decomposition

                    # Prevent error in sampling

                    # Binary thresholding
                    # Get the (n/2)th largest value
                    idx = int(n/2)
                    if idx < len(sorted_z):
                        threshold = sorted_z[idx]
                    else:
                        threshold = np.min(z) - 1e-6
                h_bin = (z >= threshold).astype(float)
                
                # Adjust if sum is not exactly n/2
                current_sum = np.sum(h_bin)
                if abs(current_sum - n/2) > 1e-6:
                    # Adjust by flipping the closest points
                    diff = int(n/2 - current_sum)
                    # Flip to reduce sum
                    if diff < 0:
                        # Need to decrease sum
                        # Find top diff values to flip
                        flip_indices = np.argpartition(z, -abs(diff))[-abs(diff):]
                    else:
                        # Need to increase sum
                        # Find lowest diff values to flip
                        flip_indices = np.argpartition(z, abs(diff))[:abs(diff)]
                    
                    # Adjust
                    for idx in flip_indices:
                        h_bin[idx] = 1 - h_bin[idx]
                        
                # Recompute objective
                dx = 2.0 / n
                h = h_bin
                correlations = np.correlate(h, 1 - h, mode="full")
                max_correlation = np.max(correlations)
                current_C5 = max_correlation * dx
                
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h_bin = h_bin.copy()
                    
            return best_h_bin, best_C5
        except Exception:
            # Fallback in case of issues
            return None, float('inf')
