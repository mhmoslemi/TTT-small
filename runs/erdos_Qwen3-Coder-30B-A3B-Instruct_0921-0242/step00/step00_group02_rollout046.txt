import time

def project_simplex(h, target_sum):
    """
    Project h onto the simplex: 0 <= h_i <= 1, sum(h) = target_sum
    Using bounded simplex projection algorithm
    """
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    deficit = target_sum - current_sum

    if abs(deficit) < 1e-12:
        return h

    # Free set
    free = np.where((h > 0) & (h < 1))[0]
    
    while abs(deficit) > 1e-12 and len(free) > 0:
        if deficit > 0:
            delta = deficit / len(free)
            h[free] += delta
        else:
            delta = deficit / len(free)
            h[free] += delta

        # clamp and update free set
        for i in free:
            if h[i] >= 1:
                h[i] = 1
            elif h[i] <= 0:
                h[i] = 0

        free = np.where((h > 0) & (h < 1))[0]
        current_sum = np.sum(h)
        deficit = target_sum - current_sum

    return h

def compute_correlations(h):
    """Linear autocorrelations of h, r_k = sum h_i * h_{i+k} for k=1,..,n-1"""
    n = len(h)
    H = fft(np.pad(h, (0, n), 'constant'))
    r_full = np.real(ifft(H * np.conj(H)))
    r = r_full[1:n]
    return r

def soft_min_grad(h, alpha):
    """
    Compute gradient of S_alpha(h) = -(1/alpha)*log(sum_i(exp(-alpha * r_i)))
    """
    n = len(h)
    # compute r_k for all k from 1 to n-1
    r = compute_correlations(h)
    # prevent numerical overflow
    max_r = np.max(r)
    if max_r > 0:
        r_scaled = r - max_r
    else:
        r_scaled = r

    exp_term = np.exp(-alpha * r_scaled)
    exp_sum = np.sum(exp_term)

    if exp_sum <= 0:
        return np.zeros_like(h)

    weights = exp_term / exp_sum

    # differentiate ∂r_k/∂h_i to get ∇S_alpha
    v = np.zeros(2*n - 1)
    w = np.zeros_like(h)

    for i, w_k in enumerate(weights):
        # compute convolution of w_k * shift(h)
        k = i + 1
        v[:n] += w_k * h
        v[k:k+n] += w_k * h
        
    grad = v[:n]
    return grad

def line_search(h, grad, alpha, step_size=None, max_iter=100):
    """Backtracking line search for S_alpha(h)"""
    if step_size is None:
        step_size = 0.1
        
    def S_alpha(h):
        r = compute_correlations(h)
        max_r = np.max(r) if len(r) > 0 else 0
        if max_r <= 0:
            return 0
        r_scaled = r - max_r
        exp_term = np.exp(-alpha * r_scaled)
        exp_sum = np.sum(exp_term)
        if exp_sum <= 0:
            return -np.inf
        return -(1/alpha) * np.log(exp_sum)

    current_value = S_alpha(h)
    for _ in range(max_iter):
        h_new = h - step_size * grad
        h_new = project_simplex(h_new, len(h) / 2)
        new_value = S_alpha(h_new)
        if new_value > current_value + 1e-8:
            return step_size, h_new
        step_size *= 0.5
    return step_size, h

def projected_gradient_ascent(h0, alpha_start=0.5, alpha_max=80, factor=2, max_iter=500):
    """
    Gradient ascent with projected updates using soft minimum surrogate
    """
    h = project_simplex(h0, len(h0) / 2)
    best_h = h.copy()
    prev_h = None
    best_correlation = np.inf
    alpha = alpha_start
    iter_count = 0
    last_improvement = float('inf')
    
    while alpha <= alpha_max and iter_count < max_iter:
        improvement_count = 0
        for _ in range(50):
            iter_count += 1
            grad = soft_min_grad(h, alpha)
            step_size, h_trial = line_search(h, grad, alpha)
            h = project_simplex(h_trial, len(h) / 2)
            
            # compute actual objective
            r = compute_correlations(h)
            corr_min = np.min(r) if len(r) > 0 else 0
            if corr_min < best_correlation:
                best_correlation = corr_min
                best_h = h.copy()
                improvement_count = 0
            else:
                improvement_count += 1
            
            # Check for local convergence
            if improvement_count > 10:
                break

        # Increase α
        alpha *= factor
        if best_correlation <= 1e-6:
            break
            
    return best_h, best_correlation

def round_to_step_function(h, dx):
    """Convert to step function by rounding and local swaps"""
    n = len(h)
    target_sum = n / 2
    # sort indices by h values
    sorted_indices = np.argsort(h)[::-1]
    
    # Greedy rounding
    h_round = np.zeros(n)
    selected = set()
    for i in sorted_indices:
        if i in selected:
            continue
        h_round[i] = 1
        selected.add(i)
        if sum(h_round) >= target_sum:
            break
    
    # Adjust if needed
    while sum(h_round) < target_sum:
        for i in sorted_indices:
            if i not in selected:
                h_round[i] = 1
                selected.add(i)
                if sum(h_round) >= target_sum:
                    break
        if sum(h_round) < target_sum:
            break
    
    if sum(h_round) > target_sum:
        # Remove elements from lowest to highest
        for i in sorted_indices:
            if i in selected and sum(h_round) > target_sum:
                h_round[i] = 0
                selected.remove(i)
    
    # Local swap refinement
    def get_correlations(h):
        r = compute_correlations(h)
        return r

    r = get_correlations(h_round)
    min_k = np.argmin(r) + 1  # k=1, so +1
    C5_current = 1 - min(r) * dx
    
    max_iters = 30
    for it in range(max_iters):
        # Compute current r_k values
        r = get_correlations(h_round)
        min_r = np.min(r)
        min_k = int(np.argmin(r)) + 1
        if min_r > 0.47 or C5_current <= 0.3807:
            break
        best_change = 0
        best_swap = None
        best_r = r.copy()
        num_swaps = 0
        # Try 100 random swaps
        while num_swaps < 100:
            num_swaps += 1
            # Pick a position with h=1 and a position with h=0
            ones = np.where(h_round == 1)[0]
            zeros = np.where(h_round == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            if i == j:
                continue
            # Compute change
            # contribution to r_{min_k} in positions i, i+min_k, j, j+min_k
            orig_r = r[min_k - 1]
            new_r = r[min_k - 1]
            # The modification to r_{min_k}
            # h_i h_{i+min_k}, possibly zero
            # h_j h_{j+min_k}, possibly zero
            if i + min_k < len(h_round):
                new_r += h_round[i] * h_round[i + min_k]
                if i + min_k < len(h_round) and i >= min_k:
                    new_r -= h_round[i] * h_round[i - min_k]
            if j + min_k < len(h_round):
                new_r += h_round[j] * h_round[j + min_k]
                if j + min_k < len(h_round) and j >= min_k:
                    new_r -= h_round[j] * h_round[j - min_k]
            change = orig_r - new_r
            # Change in the first term leads to reduction of overlap
            if change > best_change:
                best_change = change
                best_swap = (i, j)
                best_r = r.copy()
        if best_swap is not None and best_change > 1e-8:
            i, j = best_swap
            h_round[i] = 0
            h_round[j] = 1
            # update r
            r = get_correlations(h_round)
            C5_current = 1 - min(r) * dx
        else:
            break
    
    return h_round
