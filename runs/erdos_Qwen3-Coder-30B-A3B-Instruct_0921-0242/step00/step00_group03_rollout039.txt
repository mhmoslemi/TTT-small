import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from math import floor

def evaluate_c5(h):
    """Evaluate the C5 constant using full correlation."""
    h = np.array(h)
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    c5 = max(corr) * dx
    return c5

def initial_h_values():
    """Provide initial construction for h."""
    n = 70
    h = np.zeros(n)
    mid = n // 2
    h[:mid] = 1.0
    return h

def solve_sdp_relaxation(n, max_time=200):
    """Solve semidefinite relaxation for small n."""
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints (reformulate to avoid the rounding over linear)
    # X = h * h^T, but we enforce X[i,i] = h[i] and relax rank condition
    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) == h,  # Enforce diagonal entries
        cp.sum(h) == n / 2,  # Sum constraint
        0 <= h, h <= 1  # Box constraints
    ]

    dx = 2.0 / n
    for s in range(-n + 1, n):
        # Build shift matrix P
        P = np.zeros((n, n))
        if s >= 0:
            for i in range(n - s):
                P[i + s, i] = 1
        else:
            for i in range(-s, n):
                P[i - s, i] = 1
        # c_s = dx * (1.T @ h - h.T @ P @ h) = dx * (n/2 - Tr(P @ X))
        # So impose t >= dx * (n/2 - Tr(P @ X)) => t + dx * Tr(P @ X) >= dx * n/2
        constraints.append(t + dx * cp.trace(P @ X) >= dx * n / 2)

    # Objective
    objective = cp.Minimize(t)

    prob = cp.Problem(objective, constraints)
    start_time = time.time()
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, time_limit=max_time)
    except:
        try:
            prob.solve(solver=cp.ECOS, max_iters=20000, time_limit=max_time)
        except:
            return None, None

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        hval = h.value
        tval = t.value
        return hval, tval
    return None, None


def round_and_refine(h_sdp, n, n_rounds=30):
    """Randomized rounding with local optimization."""
    # Eigen decomposition to approximate h as outer product
    h_sdp = np.clip(h_sdp, 0, 1)
    X_sdp = np.outer(h_sdp, h_sdp)
    
    # Rounding loop
    best_h = None
    best_c5 = float('inf')
    for _ in range(n_rounds):
        # Sample random vector
        g = np.random.randn(n)
        y = np.linalg.eigh(X_sdp)[1] @ np.diag(np.sqrt(np.linalg.eigh(X_sdp)[0])) @ g
        # Clip to [0,1]
        y_clip = np.clip(y, 0, 1)
        # Renormalize to sum = n/2
        current_sum = np.sum(y_clip)
        deficit = n / 2 - current_sum
        idxs = np.where((y_clip > 0) & (y_clip < 1))[0]
        if len(idxs) > 0:
            change_per = deficit / len(idxs)
            for i in idxs:
                y_clip[i] = min(1.0, max(0.0, y_clip[i] + change_per))
        # Final check
        y_clip = np.clip(y_clip, 0, 1)
        new_sum = np.sum(y_clip)
        if abs(new_sum - n/2) > 1e-6:
            # Adjust uniformly
            diff = n/2 - new_sum
            if diff > 0:
                sort_idx = np.argsort(y_clip)
                for i in range(int(diff * n)):
                    y_clip[sort_idx[i]] = min(1.0, y_clip[sort_idx[i]] + 1.0)
            elif diff < 0:
                sort_idx = np.argsort(y_clip)[::-1]
                for i in range(int(-diff * n)):
                    y_clip[sort_idx[i]] = max(0.0, y_clip[sort_idx[i]] - 1.0)
        h_candidate = np.clip(y_clip, 0, 1)
        c5_val = evaluate_c5(h_candidate)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_candidate.copy()
    
    if best_h is None:
        return None, float('inf')
    return best_h, best_c5

def local_coordinate_descent(h_start, max_iter=1000, tol=1e-6, max_time=100):
    """Apply simple coordinate descent random walk."""
    h = np.array(h_start)
    dx = 2.0 / len(h)
    N = len(h)
    start_t = time.time()
    for i in range(max_iter):
        if time.time() - start_t > max_time:
            break
        # Pick an index at random
        idx = np.random.randint(N)
        # Try small changes to h[idx]
        original = h[idx]
        old_c5 = evaluate_c5(h)
        best_h = h.copy()
        best_c5 = old_c5
        # Try small +/- moves
        for delta in [-0.01, 0.01]:
            h_try = h.copy()
            h_try[idx] += delta
            if 0 <= h_try[idx] <= 1:
                # Adjust others for sum constraint
                remaining = -(delta)
                # Find closest index that can take the change
                j = (idx + 1) % N
                while abs(remaining) > 1e-10:
                    if h_try[j] + remaining <= 1 and h_try[j] + remaining >= 0:
                        h_try[j] += remaining
                        remaining = 0
                    else:
                        if h_try[j] + remaining > 1:
                            h_try[j] = 1.0
                            remaining -= 1
                        if h_try[j] + remaining < 0:
                            h_try[j] = 0.0
                            remaining += 1
                        j = (j + 1) % N
                        if j == idx:
                            break  # Should not happen in valid case
                h_try = np.clip(h_try, 0, 1)
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
        if best_c5 < old_c5:
            h = best_h
            if abs(old_c5 - best_c5) < tol:
                break
    return h, evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    """Main run function meeting interface requirements."""
    np.random.seed(seed)
    start_time = time.time()
    best_solution = None
    best_c5 = float('inf')
    n_points = 240  # Try this

    # Generate good starting point
    initial = initial_h_values()
    init_h = np.resize(initial, n_points)
    init_h = init_h / np.sum(init_h) * (n_points / 2)
    
    # Start SDP relaxation
    h_sdp, c5_lb = solve_sdp_relaxation(n_points, max_time=200)
    
    if h_sdp is None:
        # Fallback to simpler method
        h_sdp = init_h

    # Now use rounding loop
    h_rounded, c5_rounded = round_and_refine(h_sdp, n_points, n_rounds=30)
    
    if h_rounded is not None and c5_rounded < best_c5:
        best_c5 = c5_rounded
        best_solution = h_rounded.copy()
    
    # Local refinement
    if time.time() - start_time > budget_s - 30:
        pass  # No time, use what we have
    else:
        refined_h, refined_c5 = local_coordinate_descent(best_solution, max_time=100)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_solution = refined_h.copy()

    return (best_solution, best_c5, n_points)
