import numpy as np
import time
from scipy import fft

def project_simplex_box(v, total, lo=0., hi=1.):
    """Project vector v onto the simplex with box constraints [lo, hi] and sum = total."""
    v = np.array(v)
    v = np.clip(v, lo, hi)
    s = np.sum(v)
    if np.abs(s - total) < 1e-9:
        return v
    # Simple algorithm: sort, find threshold, then project
    # We want sum(min(hi, max(0, vi - theta))) = total
    # Try to make it a simplex projection with box constraints.
    # Sort in descending order to handle projection efficiently
    order = np.argsort(v)[::-1]
    v_sorted = v[order]
    cumsum = np.cumsum(v_sorted)
    n = len(v_sorted)
    # Find threshold that makes cumulative sum match desired total
    theta = 0.0
    for i in range(n):
        if i == 0:
            sum_i = v_sorted[i]
        else:
            sum_i = cumsum[i] - cumsum[i-1]
        # Now if we project v_sorted[:i+1] to value theta, their sum is:
        # sum(max(0, v_sorted[j] - theta) for j in range(i+1))
        # This is sum(v_sorted[j] - theta) for j in range(i+1) where v_sorted[j] > theta.
        # This is (i+1)*theta - sum_i if sum_i > (i+1)*theta, or 0 otherwise.
        # But we have an upper bound of 1, so compute properly...
        
        # We want to scale to achieve the target sum
        diff = total - s
        remain = 0.0
        for j in range(i+1):
            if v_sorted[j] > theta:
                remain += v_sorted[j] - theta
        if remain < diff:
            theta = (cumsum[i] - total) / (i+1)
            if theta < lo: theta = lo
        else:
            break
    
    # Apply the final projection
    result = np.zeros_like(v)
    for i in range(n):
        result[order[i]] = max(lo, min(hi, v[order[i]] - theta))
    
    # Renormalize to ensure exact sum
    s_new = np.sum(result)
    if np.abs(s_new - total) > 1e-9:
        scale = total / s_new
        result *= scale
    return result

def compute_c5(h, dx):
    """Compute C5 using efficient FFT correlation."""
    h = np.asarray(h)
    h_conj = 1 - h
    corr = fft.ifft(fft.fft(h) * fft.fft(h_conj).conj()).real
    corr = corr[:len(h)]  # Full correlation, take first half only lag k = 0 to n-1
    corr *= dx
    return np.max(corr)

def worst_lag(h):
    """Return (index of worst lag, value of worst overlap)."""
    h = np.asarray(h)
    h_conj = 1 - h
    corr = fft.ifft(fft.fft(h) * fft.fft(h_conj).conj()).real
    max_id = np.argmax(corr)
    return max_id, corr[max_id]

def gradient_worst(h, k):
    """Return gradient of the k-th correlation term."""
    n = len(h)
    grad = np.zeros(n)

    return grad
