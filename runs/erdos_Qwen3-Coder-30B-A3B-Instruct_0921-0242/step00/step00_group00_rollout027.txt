```python
import numpy as np
import time
import scipy.fft as fft
from scipy.linalg import lstsq

# Global constants
M = 8  # Number of Fourier modes (excluding constant term)
N_POINTS_BASE = 256  # Base resolution
N_POINTS_HIGH = 512  # High resolution
POP_SIZE = 4 + int(np.floor(3 * np.log(2 * M)))  # Population size for CMA-ES
MU = POP_SIZE // 2  # Number of parents
SIGMA0 = 0.2  # Initial step size for CMA-ES
MAX_ITER = 1000  # Max iterations for gradient descent
GRAD_STEP_SIZE = 0.01  # Initial gradient step size

def compute_fourier_basis(n, m):
    """Compute the Fourier basis matrix Φ of size (n, 2*m)"""
    i_indices = np.arange(n)
    phi = np.zeros((n, 2 * m))
    for j in range(m):
        # Cosine terms (index 2*j)
        phi[:, 2*j] = np.cos(2 * np.pi * (j+1) * i_indices / n)
        # Sine terms (index 2*j+1)
        phi[:, 2*j+1] = np.sin(2 * np.pi * (j+1) * i_indices / n)
    return phi

def project_onto_feasible_space(c, phi, phi_tphi_inv):
    """
    Project c onto the feasible set {c | 0 <= Φc + 0.5 <= 1}
    Returns corrected c
    """
    h = phi @ c + 0.5
    h_clipped = np.clip(h, 0, 1)
    correction = phi_tphi_inv @ (h_clipped - h)
    return c + correction

def compute_c5(c, phi):
    """Evaluate C5 for given coefficient vector c"""
    h = phi @ c + 0.5
    corr = np.correlate(h, 1-h, mode='full')
    c5 = corr.max()
    return c5

def compute_gradient_c5_at_k(c, phi, k):
    """Compute the gradient of C5 with respect to h at fixed lag k"""
    h = phi @ c + 0.5
    grad_h = np.zeros_like(h)
    # for i in range(len(h)):
    #     grad_h[i] = -h[(i+k)%len(h)] - h[(i-k)%len(h)]
    # Simplified computation assuming 0 extend
    h_shifted = np.roll(h, k)
    grad_h = 1 - h_shifted
    grad_h[(k+1):] += h[:-k-1]
    grad_h[:k] += h[-k:]
    return grad_h

def projected_gradient_descent(c, phi, phi_tphi_inv, max_iter=MAX_ITER):
    """Perform projected gradient descent to refine c"""
    for t in range(max_iter):
        # Find the best lag k that maximizes the overlap
        h = phi @ c + 0.5
        corr = np.correlate(h, 1-h, mode='full')
        k_best = np.argmax(corr)
        # Compute gradient with respect to h induced by k_best
        grad_h = compute_gradient_c5_at_k(c, phi, k_best)
        # Transfer gradient to coefficients
        grad_c = phi.T @ grad_h
        # Step with diminishing step size
        alpha = GRAD_STEP_SIZE / (1 + t)
        c_new = c - alpha * grad_c
        # Project back
        c = project_onto_feasible_space(c_new, phi, phi_tphi_inv)
    return c

def cma_es_restart(seed, phi, phi_tphi_inv, n_points, budget_s):
    """Run one restart of CMA-ES optimization"""
    np.random.seed(seed)
    N_VARS = 2 * M
    # Initialize population
    population = np.random.normal(0, SIGMA0, (POP_SIZE, N_VARS))
    # Main loop
    for gen in range(MAX_ITER):
        # Project each individual onto feasibility
        proj_pop = np.array([project_onto_feasible_space(c, phi, phi_tphi_inv) for c in population])
        # Evaluate objectives
        fitness = np.array([compute_c5(c, phi) for c in proj_pop])
        # Select best individuals
        indices = np.argsort(fitness)[:MU]
        parents = proj_pop[indices]
        # Update mean and covariance matrix (simple version for efficiency)
        mu = np.mean(parents, axis=0)
        # Update population
        population = mu + SIGMA0 * np.random.normal(0, 1, (POP_SIZE, N_VARS))
        # Project the mean back once per generation
        mu = project_onto_feasible_space(mu, phi, phi_tphi_inv)
        # Re-inject a fraction of random individuals to maintain diversity
        num_random = max(1, int(0.2 * POP_SIZE))
        for i in range(num_random):
            individual = np.random.normal(0, SIGMA0, N_VARS)
            population[i] = project_onto_feasible_space(individual, phi, phi_tphi_inv)
    # Final refine using projected gradient decent
    best_c = project_onto_feasible_space(mu, phi, phi_tphi_inv)
    best_c = projected_gradient_descent(best_c, phi, phi_tphi_inv)
    return best_c

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    # Precompute key matrices
    phi_base = compute_fourier_basis(N_POINTS_BASE, M)
    phi_tphi_inv_base = np.linalg.pinv(phi_base.T @ phi_base) @ phi_base.T
    best_c = None
    best_c5 = float('inf')
    
    # First batch on low resolution
    total_runs = 10
    time_per_batch = 0.4 * budget_s / total_runs
    for k in range(total_runs):
        c = cma_es_restart(seed + k, phi_base, phi_tphi_inv_base, N_POINTS_BASE, time_per_batch)
        c5 = compute_c5(c, phi_base)
        if c5 < best_c5:
            best_c5 = c5
            best_c = c.copy()
        if time.time() - start_time > budget_s - 0.5:
            break
    
    # If time remains and C5 is still above target, try high resolution
    if (best_c5 > 0.3807) and (time.time() - start_time < budget_s - 0.5):
        # Reconstruct high resolution basis
        phi_high = compute_fourier_basis(N_POINTS_HIGH, M)
        # Copy the same coefficients to high resolution (interpolation)
        # Note: assuming phi_high and phi_base have same structure
        # We directly use best_c interpolated to 512 points using same M
        # basis of 8 modes each on [0,2] and [0,1] respectively.
        # For exact conversion, fewer new basis need to be added:
        # phi_high keeps same coefficients in matching dimensions
        use_c = best_c
        # Now refine with higher resolution
        best_c = project_onto_feasible_space(use_c, phi_high, np.linalg.pinv(phi_high.T @ phi_high) @ phi_high.T)
        best_c = projected_gradient_descent(best_c, phi_high, np.linalg.pinv(phi_high.T @ phi_high) @ phi_high.T, 200)
        best_c5 = compute_c5(best_c, phi_high)
    
    # Final reconstruction
    if best_c is not None:
        if len(best_c) == 2*M:
            h = phi_high @ best_c + 0.5 if 'phi_high' in locals() else phi_base @ best_c + 0.5
        else:
            raise ValueError("Invalid final coefficient dimension")
    else:
        # Use a basic symmetric solution if nothing found
        h = np.ones(N_POINTS_BASE) * 0.5

    # Ensure integral matches
    integral_error = np.sum(h) * 2 / len(h) - 1  # dx = 2 / n_points
    if abs(integral_error) > 1e-12:
        h -= integral_error / len(h)  # distribute error across all pixels
    # Clamp
    h = np.clip(h, 0, 1)

    # Final evaluation with correct N_POINTS
    n_points = N_POINTS_HIGH if 'phi_high' in locals() else N_POINTS_BASE
    h_eval = phi_high @ best_c + 0.5 if 'phi_high' in locals() else phi_base @ best_c + 0.5
    corr_eval = np.correlate(h_eval, 1-h_eval, mode='full')
    c5_bound = float(corr_eval.max())

    return (h.tolist(), c5_bound, n_points)
```
