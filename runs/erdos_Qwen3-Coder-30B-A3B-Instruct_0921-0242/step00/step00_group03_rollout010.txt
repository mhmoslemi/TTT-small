import numpy as np
import cvxpy as cp
import time
from scipy import linalg
from scipy.optimize import minimize_scalar
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 240
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_time = start_time + budget_s - 30
    
    # Start with initial construction
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = h_base[:n_points]
        if len(h_base) < n_points:
            h_base = np.pad(h_base, (0, n_points - len(h_base)), 'constant', constant_values=0)
        h_base = np.clip(h_base, 0, 1)
    else:
        # Generate an initial binary-like function
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
        h_base = np.clip(h_base, 0, 1)
    
    # Normalize and ensure sum constraint
    current_sum = np.sum(h_base)
    h_base = h_base * target_sum / current_sum
    h_base = np.clip(h_base, 0, 1)
    
    best_h = h_base.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # SDP relaxation
    def solve_sdp(n, target_sum, dx):
        # SDP relaxation for the problem
        h = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            cp.diag(X) == h,
            X >> 0,  # X is positive semidefinite
            cp.sum(h) == target_sum,
            0 <= h,
            h <= 1,
        ]
        
        # Add max constraint for each shift
        # We'll generate all shift constraints
        # But to reduce computation, we sample some
        max_shift = 2 * n - 1
        shift_indices = [i - (n - 1) for i in range(max_shift)]
        for s in shift_indices[::10]:  # Sparse sampling to reduce constraint count
            # Build shift matrix P_s
            P_s = np.zeros((n, n))
            if s >= 0:
                P_s[s:, :n-s] = np.eye(n - s)
            else:
                P_s[:n+s, -s:] = np.eye(n + s)
            
            # We compute sum(h_i * h_{i+s})
            # But in SDP form it's Tr(P_s X)
            constraints.append(t >= 0.5 * dx - dx * cp.trace(P_s @ X))
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            # Try solving with time limit
            prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value, prob.value, True
        except:
            pass
        return h_base, float('inf'), False
    
    # Try SDP relaxation
    try:
        sdp_h, sdp_value, success = solve_sdp(n_points, target_sum, dx)
        if success:
            # Round to get a feasible solution
            sdp_h = np.clip(sdp_h, 0, 1)
            roundings = 50
            for _ in range(roundings):
                if time.time() > max_time:
                    break
                # Randomized rounding
                g = np.random.normal(0, 1, n_points)
                U, V = linalg.eigh(np.outer(sdp_h, sdp_h))
                y = U @ np.diag(np.sqrt(np.maximum(V, 0))) @ g
                y = np.clip(y, 0, 1)
                current_sum = np.sum(y)
                if current_sum > 0:
                    y = y * target_sum / current_sum
                y = np.clip(y, 0, 1)
                c5_bound = evaluate_erdos_solution(y, dx)
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = y.copy()
    except Exception as e:
        pass
    
    # Local refinement by coordinate descent
    def refine_h(h, dx, max_iter=100):
        h = h.copy()
        current_sum = np.sum(h)
        for _ in range(max_iter):
            if time.time() > max_time:
                break
            improvement = False
            for i in range(n_points):
                if time.time() > max_time:
                    break
                # Try a small perturbation
                h_new = h.copy()
                delta = np.random.uniform(-0.1, 0.1)
                h_new[i] = np.clip(h[i] + delta, 0, 1)
                new_sum = current_sum + delta
                new_sum = np.clip(new_sum, 0, n_points)
                if new_sum > 0:
                    h_new = h_new * target_sum / new_sum
                c5 = evaluate_erdos_solution(h_new, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    h = h_new.copy()
                    improvement = True
            if not improvement:
                break
        return h, best_c5
    
    # Run refine
    best_h, best_c5 = refine_h(best_h, dx)
    
    # Final correction to ensure the sum is correct
    current_sum = np.sum(best_h)
    if not abs(current_sum - target_sum) < 1e-10:
        best_h = best_h * target_sum / current_sum 
    
    # Double-check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    """
    Evaluate the C5 bound for given h
    """
    h2 = 1 - h
    corr = np.correlate(h, h2, mode='full')
    # Take only the positive shifts
    corr = corr[len(h)-1:]
    max_val = np.max(corr) * dx
    return max_val
