def project_to_F(h, n):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Compute current deficit/excess
    deficit = n/2 - h.sum()
    # Indices that are not at a bound
    free = np.where((h>0) & (h<1))[0]
    while np.abs(deficit) > 1e-12 and free.size>0:
        # Distribute equally over free indices
        delta = deficit / free.size
        # Apply delta, respecting bounds, and recompute deficit
        h[free] += delta
        # Clamp again and update free set
        h = np.clip(h, 0.0, 1.0)
        deficit = n/2 - h.sum()
        free = np.where((h>0) & (h<1))[0]
    return h

def evaluate_C5(h, dx):
    # Fast FFT-based evaluation
    n = len(h)
    L = 2 * n
    h_pad = np.pad(h, (0, L - n), 'constant', constant_values=0)
    g_pad = np.pad(1 - h, (0, L - n), 'constant', constant_values=0)
    H = fft(h_pad)
    G = fft(g_pad)
    conv = ifft(H * np.conj(G)).real * dx
    # mask out the first n-1 and last n-1 entries (they are not valid correlation values)
    # valid range: from n-1 to 2*n-2 (inclusive) corresponds to shifts from -(n-1) to n-1
    # we are interested in max over k!=0, which excludes k=0
    # k=0 correlation is at index n-1 (zero shift)
    # So we want max over all indices except n-1 (corresponding to k=0)
    c5 = np.max(conv[0:n-1])  # k=-n+1 to -1
    c5 = max(c5, np.max(conv[n+1:]))  # k=1 to n-1
    return c5

def fast_conv(h, g, dx):
    n = len(h)
    L = 2 * n
    h_pad = np.pad(h, (0, L - n), 'constant', constant_values=0)
    g_pad = np.pad(g, (0, L - n), 'constant', constant_values=0)
    H = fft(h_pad)
    G = fft(g_pad)
    conv = ifft(H * np.conj(G)).real * dx
    return conv

def local_improve(h, dx, max_steps=30):
    """Improve h by swapping entries to reduce maximum correlation"""
    n = len(h)
    g = 1 - h
    c = fast_conv(h, g, dx)
    # Find the maximum correlation (worst shift)
    max_correlation = np.max(c[0:n-1])
    max_correlation = max(max_correlation, np.max(c[n+1:]))
    best_h = h.copy()
    best_c5 = evaluate_C5(h, dx)
    # Indices we can decrease / increase
    # Need to avoid permuting values that are 0 or 1 to make swaps meaningful
    idx_nonzero = np.where(h > 1e-10)[0]
    idx_nonone = np.where(h < 1 - 1e-10)[0]
    swap_count = 0
    for _ in range(max_steps):
        if len(idx_nonzero) == 0 or len(idx_nonone) == 0:
            break
        # Choose two indices randomly - one to decrease and one to increase
        i1 = random.choice(idx_nonzero)
        i0 = random.choice(idx_nonone)
        # Compute change in correlation if we swap h[i1] and h[i0]
        epsilon = min(h[i1], 1 - h[i0], 0.05)
        if epsilon < 1e-8:
            continue
        # Incremental update vector for correlation change due to the swap
        # Only affected by shifts where i1 and i0 participate
        # For shifts k, c_new = c_old - epsilon * dx * (g[i1 - k] - g[i0 - k])
        new_c = c.copy()
        for k in range(n):
            # new contribution from i1
            if i1 - k >= 0 and i1 - k < n:
                new_c[k] -= epsilon * dx * g[i1 - k]
            if i1 + k < n:
                new_c[2*n - 2 - (k + 1)] -= epsilon * dx * g[i1 + k]
            # new contribution from i0
            if i0 - k >= 0 and i0 - k < n:
                new_c[k] += epsilon * dx * g[i0 - k]
            if i0 + k < n:
                new_c[2*n - 2 - (k + 1)] += epsilon * dx * g[i0 + k]
        
        # compute new maximum
        max_correlation_new = np.max(new_c[0:n-1])
        max_correlation_new = max(max_correlation_new, np.max(new_c[n+1:]))
        if max_correlation_new < best_c5 - 1e-12:
            h[i1] -= epsilon
            h[i0] += epsilon
            c = new_c
            best_c5 = max_correlation_new
            best_h = h.copy()
            swap_count += 1
        else:
            # If we're not helping much, be slightly more risky (try to break local minima)
            # we still want to perform the swap for exploration or if we think it works locally
            pass # get out of here, but don't change h
        # Possibly break if no improvement after several attempts
        if swap_count >= 5:
            break
    # final check
    return best_h

def crossover(p1, p2, alpha):
    # Blend crossover with projection
    child = alpha * p1 + (1 - alpha) * p2
    return project_to_F(child, len(p1))

    noise = np.random.normal(0, sigma, size=m)
    h[indices] += noise
    return project_to_F(h, n)
