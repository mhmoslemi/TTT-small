import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    safety_margin = 5
    
    # Candidate discretisations
    candidates = [30, 42, 48, 60, 78]
    n_starts = 4  # Random restarts per n
    
    def soft_min(alpha, r):
        # Compute soft minimum using log-sum-exp trick for numerical stability
        r_alpha = -alpha * r
        max_r_alpha = np.max(r_alpha)
        exp_terms = np.exp(r_alpha - max_r_alpha)
        log_sum_exp = max_r_alpha + np.log(np.sum(exp_terms))
        return -log_sum_exp / alpha
    
    def compute_correlations_and_gradient(h, alpha=1.0):
        n = len(h)
        padded_h = np.pad(h, (0, n), mode='constant')
        H = fft(padded_h)
        r = np.real(ifft(H * np.conj(H)))[1:n]  # linear correlations from 1 to n-1
        r = r[:n-1]  # Remove extra terms from padding
        
        # Compute weights
        r_alpha = -alpha * r
        max_r_alpha = np.max(r_alpha)
        exp_terms = np.exp(r_alpha - max_r_alpha)
        w = exp_terms / np.sum(exp_terms)
        
        # Compute gradient using convolution
        # ∇S_α = w * h + reverse(w) * h where * is linear convolution
        w_rev = w[::-1]
        grad = np.convolve(w, h, mode='full')[:n] + np.convolve(w_rev, h, mode='full')[:n]
        
        return r, w, grad
    
    def project_simplex(h, target_sum):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        
        if abs(delta) < 1e-12:
            return h
        
        # Identical to the paper: bounded simplex projection algorithm
        F = np.where((h > 0) & (h < 1))[0]
        if len(F) == 0:
            if current_sum < target_sum:
                h = np.ones_like(h) * (target_sum / len(h))
            else:
                h = np.zeros_like(h)
            return h
        
        if len(F) == 0:
            H = np.zeros_like(h)
            return H
        
        # Sort indices by h_i in descending order
        sorted_indices = np.argsort(-h[F])
        F_sorted = F[sorted_indices]
        
        # Set-up for projection
        # greedily subtract shortage greedily from free set
        for i_idx in range(len(F_sorted)):
            i = F_sorted[i_idx]
            diff = h[i] - 0
            if delta >= diff:
                delta -= diff
                h[i] = 0
                continue
            else:
                h[i] -= delta
                delta = 0
                break
        
        # Now add delta to the rest
        if delta > 0:
            free_count = np.sum(h[F] > 0)
            if free_count > 0:
                delta_per_item = delta / free_count
                for i_idx in range(len(F_sorted)):
                    i = F_sorted[i_idx]
                    if h[i] > 0:
                        current = h[i]
                        h[i] = min(1.0, current + delta_per_item)
        
        return h
    
    def backtracking_line_search(h, g, alpha, step_size=1.0):
        # Simple decrease condition
        # For function value of soft minimum, we compute it during iteration
        # Simple backtracking
        c1 = 0.5
        rho = 0.5
        # Check if direct update is improving or not, if it's better:
        # We check numeric feasibility within the given constraints
        while True:
            h_new = h - step_size * g
            # projection
            h_proj = project_simplex(h_new, len(h) / 2)
            r_new, _, _ = compute_correlations_and_gradient(h_proj, alpha)
            f_new = soft_min(alpha, r_new)
            f_old = soft_min(alpha, compute_correlations_and_gradient(h, alpha)[0])
            if f_new > f_old + c1 * step_size * np.dot(g, h_proj - h):
                return step_size
            step_size *= rho
            if step_size < 1e-10:
                return step_size
    
    def Homotopy_Ascend(h0, alpha_start=0.5, alpha_max=80.0, factor=2.0, max_iter=500, min_improvement=1e-8):
        # return final h after ascending
        h = h0.copy()
        alpha = alpha_start
        prev_obj = None
        
        for iteration in range(max_iter):
            # Get gradient with current alpha
            r, _, grad = compute_correlations_and_gradient(h, alpha)
            obj = soft_min(alpha, r)
            
            if prev_obj is not None:
                if obj - prev_obj < min_improvement:
                    break  # no further gain
            prev_obj = obj
            
            # Nesterov-like step
            beta = 0.5   # typical value
            y = h + beta * (h - h)  # h_prev = h -> y = h + beta * 0 = h
            _, _, grad_at_y = compute_correlations_and_gradient(y, alpha)
            
            # backtracking line search
            step_size = backtracking_line_search(h, -grad_at_y, alpha, step_size=1.0)
            
            h_new = h - step_size * grad_at_y
            h = project_simplex(h_new, len(h) / 2)
            
            # Possibly increase alpha
            if iteration > 0 and iteration % 20 == 0:
                alpha = min(alpha * factor, alpha_max)
        
        return h
    
    for n in candidates:
        if time.time() - t0 > budget_s - safety_margin:
            break
        target_sum = n / 2.0
        h_init = None
        
        # Deterministic start: interpolate initial_h_values to n points
        if 'initial_h_values' in globals():
            h_raw = initial_h_values
            x_floor = np.linspace(0, 2, len(h_raw))
            x_new = np.linspace(0, 2, n)
            h_init = np.interp(x_new, x_floor, h_raw)
        else:
            h_init = np.random.rand(n)  # fallback
            
        # Ensure sum == target_sum
        h_init = h_init * target_sum / np.sum(h_init)
        h_init = project_simplex(h_init, target_sum)
        
        # Multiple random starts
        starts = [h_init.copy()]
        for _ in range(n_starts - 1):
            h_rand = np.random.rand(n)
            h_rand = h_rand * target_sum / np.sum(h_rand)
            h_rand = project_simplex(h_rand, target_sum)
            starts.append(h_rand.copy())
        
        for i, h_start in enumerate(starts):
            if time.time() - t0 > budget_s - safety_margin:
                break
                
            h_opt = Homotopy_Ascend(h_start)
            
            # Now we compute the true C5 (not the soft version) 
            dx = 2.0 / n
            # Use np.correlate with mode='full' and extract appropriate parts
            h_concat = np.concatenate([h_opt, np.zeros(n)])
            corr = np.correlate(h_concat, 1 - h_opt, mode='full')[:]
            corr = corr[n:]   # k from 1 to n-1
            min_r = np.min(corr) * dx
            c5_est = 1.0 - min_r
            
            # Update best
            if c5_est < best_c5:
                best_c5 = c5_est
                best_h = h_opt.copy()
                best_n = n

            if best_c5 <= 0.3807:
                break
                
    if best_h is None:
        # return default or raise error
        h0 = np.ones(42) / 2
        h0 = project_simplex(h0, 21.0)
        dx = 2.0 / 42
        corr = np.correlate(np.concatenate([h0, np.zeros(42)]), 1 - h0, mode='full')[42:]
        min_r = np.min(corr) * dx
        best_c5 = 1.0 - min_r
        best_h = h0
        best_n = 42
    
    return (best_h, best_c5, best_n)
