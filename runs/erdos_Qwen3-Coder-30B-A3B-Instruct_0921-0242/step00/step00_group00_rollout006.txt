import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.linalg import solve

# Global parameters
N_POINTS = 256
M_FREQS = 8  # Number of frequency pairs (a_m, b_m)
N_VARS = 2 * M_FREQS  # Total number of optimization variables

# Precompute the Fourier basis matrix
def build_fourier_basis(n, m_freqs):
    # Create the basis matrix Φ of size (n, 2*m_freqs)
    # Column j = [cos(2π*j*i/n), sin(2π*j*i/n)] for j = 1..m_freqs
    # and cosine terms indexed by 0..m_freqs-1, sine by m_freqs..2*m_freqs-1
    
    # Compute the matrix entries
    i_vals = np.arange(n)
    base = np.arange(1, m_freqs + 1)
    
    # Build basis matrices for cosine and sine
    c = np.cos(2 * np.pi * np.outer(i_vals, base) / n)
    s = np.sin(2 * np.pi * np.outer(i_vals, base) / n)
    
    # Combine into Φ
    phi = np.hstack([c, s])  # Shape (n, 2*m_freqs)
    return phi

def project_to_feasible(c, phi, phi_t_phi_inv_phi_t):
    """
    Project c to feasibility: 0 <= h[i] <= 1 where h = phi @ c + 0.5
    Returns the corrected coefficient vector.
    """
    # Compute h = phi @ c + 0.5
    h = phi @ c + 0.5
    
    # Clamp h to [0, 1]
    h_clamped = np.clip(h, 0, 1)
    
    # Compute the error: (h_clamped - h)
    error = h_clamped - h
    
    # Solve for correction δc such that Φ @ δc = error (least squares)
    # δc = inv(Φ^T @ Φ) @ Φ^T @ error
    delta_c = phi_t_phi_inv_phi_t @ error
    
    # Update c
    c_new = c + delta_c
    
    return c_new

def compute_c5(c, phi):
    """
    Compute overlap integral C5 for coefficient vector c.
    """
    # Reconstruct h
    h = phi @ c + 0.5
    dx = 2.0 / len(h)
    
    # Compute overlap: integral of h(x)(1 - h(x+k))dx for all k
    # Use FFT-based convolution
    h1 = 1 - h
    correlation = np.convolve(h, h1, mode='full') * dx
    
    # The maximum value is on the middle k-values (the full correlation)  
    c5_bound = np.max(correlation)
    
    return c5_bound

def optimize_on_basis(seed, phi, phi_t_phi_inv_phi_t, max_time):
    np.random.seed(seed)
    start_time = time.time()
    best_c = None
    best_c5 = float('inf')
    
    # Population parameters
    lambda_pop = 4 + int(3 * np.log(N_VARS))
    mu_pop = lambda_pop // 2
    
    # Sample initial population in coefficient space
    population = []
    for _ in range(lambda_pop):
        c = np.random.randn(N_VARS) * 0.2
        c = project_to_feasible(c, phi, phi_t_phi_inv_phi_t)
        population.append(c)
    
    # CMA-ES-like evolutionary optimization
    for gen in range(50000):
        if time.time() - start_time > max_time - 0.5:
            break
        # Evaluate each individual
        fitness_scores = []
        for c in population:
            try:
                c5 = compute_c5(c, phi)
                fitness_scores.append((c5, c))
            except Exception:
                fitness_scores.append((float('inf'), c))
                
        # Sort by fitness (ascending)
        fitness_scores.sort(key=lambda x: x[0])
        
        # Keep best
        if fitness_scores[0][0] < best_c5:
            best_c5 = fitness_scores[0][0]
            best_c = fitness_scores[0][1].copy()
        
        if best_c5 < 0.3807:
            break
            
        # Select
        selected = [c for _, c in fitness_scores[:mu_pop]]
        
        # Generate offspring
        offspring = []
        for _ in range(lambda_pop):
            # Mutate (simple version of CMA-ES mutation)
            parent = selected[np.random.randint(0, mu_pop)]
            child = parent + np.random.randn(N_VARS) * 0.2
            child = project_to_feasible(child, phi, phi_t_phi_inv_phi_t)
            offspring.append(child)
        
        population = offspring
          
    return best_c, best_c5

def gradient_refinement(c, phi, max_iter=200):
    """
    Apply projected gradient descent to refine c
    """
    h_current = phi @ c + 0.5
    dx = 2.0 / len(h_current)
    
    # Get maximizing lag k*
    h1 = 1 - h_current
    correlation = np.convolve(h_current, h1, mode='full') * dx
    k_max = np.argmax(correlation) - len(h_current) + 1
    
    for t in range(max_iter):
        # Compute gradient at each point
        # ∂C5/∂h_i = (1 - h_{i+k*}) - h_{i-k*}
        grad = np.zeros_like(h_current)
        
        for i in range(len(h_current)):
            idx1 = (i + k_max) % len(h_current)
            idx2 = (i - k_max) % len(h_current)
            grad[i] = (1 - h_current[idx1]) - h_current[idx2]
        
        # Project gradient back to coefficient space: ∇_c = Φ^T @ ∇_h
        grad_coeffs = phi.T @ grad
        
        # Step size with decay
        alpha = 1.0 / (1.0 + t)
        c_new = c - alpha * grad_coeffs
        
        # Project to feasible polytope
        c_new = project_to_feasible(c_new, phi, phi_t_phi_inv_phi_t)
        
        c = c_new
        
        # Check convergence
        h_new = phi @ c + 0.5
        if np.allclose(h_current, h_new, atol=1e-10):
            break
        h_current = h_new
    
    return c

def run(seed=42, budget_s=600, **kwargs):
    # Precompute basis matrix
    phi = build_fourier_basis(N_POINTS, M_FREQS)
    phi_t_phi_inv_phi_t = solve(phi.T @ phi, phi.T)  # Pseudo-inverse only used in projection
    
    # CMA-ES batch of restarts
    start_time = time.time()
    best_total_c = None
    best_total_c5 = float('inf')
    
    # First run on lower resolution
    restarts = 10
    run_time_per_batch = (budget_s - 0.5) / restarts
    for i in range(restarts):
        seed_i = seed + i
        c, c5 = optimize_on_basis(seed_i, phi, phi_t_phi_inv_phi_t, run_time_per_batch)
        if c5 < best_total_c5:
            best_total_c = c
            best_total_c5 = c5
    
    if best_total_c5 < 0.3807:
        # Early exit
        h_final = phi @ best_total_c + 0.5
        if np.abs(np.sum(h_final)*2.0/N_POINTS - 1) > 1e-12:
            # Correct slight integral drift
            residual = np.sum(h_final) * 2.0 / N_POINTS - 1
            correction = -residual / N_POINTS
            h_final += correction
            h_final = np.clip(h_final, 0, 1)
                
        c5_final = compute_c5(best_total_c, phi)
        return (h_final.tolist(), float(c5_final), N_POINTS)
    
    # If not good enough, run on higher resolution
    N_POINTS_HIGH = 512
    phi_high = build_fourier_basis(N_POINTS_HIGH, M_FREQS)
    # Use coefficients from low res but interpolate to high res
    c_high = best_total_c 
    phi_t_phi_inv_phi_t_high = solve(phi_high.T @ phi_high, phi_high.T)

    # Batch restarts for higher resolution  
    for i in range(restarts):
        seed_i = seed + i + 100  # avoid reused seeds
        c, c5 = optimize_on_basis(seed_i, phi_high, phi_t_phi_inv_phi_t_high, run_time_per_batch)
        if c5 < best_total_c5:
            best_total_c = c
            best_total_c5 = c5
    
    if time.time() - start_time > budget_s - 0.5:
        pass # early termination for final refinement

    # Final gradient refinement on most recent high-res best
    best_total_c = gradient_refinement(best_total_c, phi_high)
    
    h_final = phi_high @ best_total_c + 0.5
    
    # Integral correction
    integral = np.sum(h_final) * 2.0 / N_POINTS_HIGH
    if abs(integral - 1.0) > 1e-12:
        correction = - (integral - 1.0) / N_POINTS_HIGH
        h_final += correction
        h_final = np.clip(h_final, 0, 1)
    
    c5_final = compute_c5(best_total_c, phi_high)
    
    return (h_final.tolist(), float(c5_final), N_POINTS_HIGH)
