import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize_scalar
import time

def project_capped_simplex(v, total=None):
    """
    Project v onto the capped simplex:
    0 <= v[i] <= 1, sum(v) = total.
    Uses bisection on Lagrange multiplier.
    """
    if total is None:
        total = np.sum(v)
    v = np.asarray(v)
    n = len(v)
    
    def project_with_lambda(lam):
        """Project v - lam onto [0,1] and check sum."""
        p = np.clip(v - lam, 0, 1)
        return np.sum(p) - total
    
    # Find lambda such that sum(p) = total
    low = np.min(v) - 1
    high = np.max(v)
    eps = 1e-12
    while high - low > eps:
        mid = (low + high) / 2
        val = project_with_lambda(mid)
        if val > 0:
            low = mid
        else:
            high = mid
    
    lambda_opt = (low + high) / 2
    return np.clip(v - lambda_opt, 0, 1)

def auto_corr(h):
    """Compute zero-padded auto-correlations O_k = sum h[i] h[i+k]"""
    n = len(h)
    # Compute O_k using fft
    h_padded = np.concatenate((h, np.zeros(n)))
    corr = fftconvolve(h_padded, h[::-1], mode='valid')
    return corr[:n]

def grad_F(beta, h, O, w):
    """
    Compute gradient of F_beta = 1 - (1/beta) * log sum_k w_k * O_k
    The gradient is -g where g = sum_k w_k * grad O_k
    """
    n = len(h)
    g = np.zeros(n)
    # Gradient of O_k(h) wrt h_i is h_{i+k} + h_{i-k} (with convention that out-of-bounds values are 0)
    # So we're going to compute this as a convolution:
    # g_i = sum_k w_k (h_{i+k} + h_{i-k}) = w * h (shifted) + w * h (reversed shift)
    # Handle carefully so we only index in valid ranges.
    w_padded = np.concatenate((w, np.zeros(n)))
    h_padded = np.concatenate((h, np.zeros(n)))
    # This gives the contribution from h_{i+k}
    forward_part = fftconvolve(w_padded, h_padded, mode='valid')[:n]
    # For h_{i-k}, we process it as h_{k-i} and use the same trick
    # More directly:
    # g_i = sum_k w_k * [h_{i+k} * (if i+k<n) + h_{i-k} * (if i-k>=0)]
    # = sum_{k: i+k<n} w_k * h_{i+k} + sum_{k: i-k>=0} w_k * h_{i-k}
    # = sum_{k: i+k<n} w_k * h_{i+k} + sum_{k: i>=k} w_{i-k} * h_k
    # = sum_{k: i+k<n} w_k * h_{i+k} + sum_{j: j<=i} w_{i-j} * h_j
    # = F_k * h + R_k * h  (as forward_part + reverse_part)
    # But be careful:
    reverse_h = h[::-1]
    reverse_w = w[::-1]
    reverse_part = fftconvolve(reverse_w, reverse_h, mode='full')[:n][::-1]
    g = forward_part + reverse_part
    return -g

            alpha = 0.1
            step = 0

            h = h_new.copy()
            iter_count += 1

def run(seed=42, budget_s=600, **kwargs):
    # Constants
    time_budget = budget_s
    start_time = time.time()
    
    # Discretization
    n_values = [256, 384]
    n_best = 256
    best_C5 = 1.0
    best_h = None
    
    # Beta schedule
    beta_0 = 0.5
    beta_final = 40
    N_level = 12
    beta_schedule = [beta_0 * (beta_final / beta_0)**(i / (N_level - 1)) for i in range(N_level)]
    
    # Multi-restart
    num_seeds = 5
    init_seeds = np.arange(1000, 1000 + num_seeds)
    
    for n in n_values:
        if time.time() - start_time > 0.8 * time_budget:
            break
        # For each size, multiple starts
        for seed in init_seeds:
            if time.time() - start_time > 0.9 * time_budget:
                break
            h, C5 = optimize_one_start(n, beta_schedule, time_budget, seed)
            if C5 < best_C5:
                best_C5 = C5
                n_best = n
                best_h = h.copy()
    
    # Final refinement
    if best_h is not None and time.time() - start_time < time_budget * 0.9:
        # Use no smooth, just exact C5 minimization
        h = best_h.copy()
        max_iter = 500
        for _ in range(max_iter):
            if time.time() - start_time > time_budget * 0.95:
                break
            # Find worst lag
            O = auto_corr(h)
            k_worst = np.argmin(O)
            # Gradient is -∇O_{k_worst}(h)
            # Gradient computation is simpler: 
            # ∇O_k(h)_i = h_{i+k} + h_{i-k} if both indices valid
            g = np.zeros_like(h)
            if k_worst < len(h):
                g[k_worst:] += h[:-k_worst]
            if k_worst > 0:
                g[:-k_worst] += h[k_worst:]
            
            # Line search
            alpha = 0.01
            for _ in range(10):
                h_new = h - alpha * g
                h_new = project_capped_simplex(h_new, total=n_best/2)
                C5_new = 1.0 - np.min(auto_corr(h_new))
                if C5_new < best_C5:
                    best_C5 = C5_new
                    h = h_new.copy()
                    break
                else:
                    alpha *= 0.5
    
    # Return best solution
    return (best_h.tolist(), best_C5, n_best)
