def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)

    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value is what we want
    max_overlap = np.max(conv) * dx
    return max_overlap
