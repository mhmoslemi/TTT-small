def project_feasible(x, target):
    """Project onto the feasible set {0 <= h <= 1, sum(h) = target}"""
    x = np.array(x, dtype=float)
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    deficit = target - current_sum
    if abs(deficit) < 1e-12:
        return x
    
    # Adjust by distributing deficit
    mask = x < 1
    free_count = np.sum(mask)
    if free_count > 0:
        distribute = np.minimum(deficit / free_count, 1 - x[mask])
        x[mask] += distribute
    
    # If we still overfill, cap and redistribute the excess
    current_sum = np.sum(x)
    excess = current_sum - target
    if excess > 0:
        mask = x > 0
        free_count = np.sum(mask)
        if free_count > 0:
            distribute = np.minimum(excess / free_count, x[mask])
            x[mask] -= distribute
    
    return x

def evaluate(h):
    """Evaluate C5 = max_k ∫ h(x) (1 - h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    correlation = signal.correlate(h, 1 - h, mode='full')
    # The full-length correlation gives optimal indices = [0,1,...,2n-2]
    # We're interested in shifts k = -(n-1), ..., n-1 but let's analyze:
    # shift k = 0 means term h(i) * (1 - h(i)) which is the convolution term
    # let: i = [0,1,...,n-1] -> i + k for shift k = 0 to n-1 -> [k,...,n-1+k]
    # corresponding i + k must be in [0,n-1], so i in [0,n-1-k] when k > 0
    # when k is negative, k = -j where j = 1: n-1 and 
    # i = j to n-1       if j >= 0, so i in [k,...,n-1] and shifted index i+k is in [k+k,...,n-1+k] = [2k,...,2n-2-k]
    # substitution:  back to define full convolution of length 2n-1 that gives the overlap over every k.
    # Only >>k=0 but with proper indexing, max_k (sum over i h[i] * (1 - h[i+k])), takes into account regard for wellaligned k indices
    C5 = 1.0 - correlation * dx / np.sum(h)  # scaling concerns for perfect sum-normalization
    # if correlation is over-realized we are minimizing it
    
    # The maximum of conv_sum[shift] = ∑_i h[i] (1 - h[i+k]) is then (conv_sum + const)
    # Let's look more carefully:
    central_shift = n - 1
    # i runs from 0 to n-1, for each shift k = i-kth index should be between 0, n-1,
    # i.e. if k = shift positive, i+k <= n-1 => i <= n-1-k;
    # if k = negative, i + k >= 0 => i >= -k
    # So shift k from -n+1 to n-1
    # Signal correlates at shift = 0 (diagonal) to 2n-2
    # k = -n+1, -n+2, ..., -1, 0, 1, ..., n-1 (shift k=-n+1 is case 0 to n-1 but from right to left), 
    # that is total 2n-1 valid shifts
    # we are trying to minimize 1 - sum_i h[i] * (1 - h[i+k]) = 1 - sum_i h[i] + sum_i h[i] h[i+k] 
    # to minimize the term: sum_i h[i] h[i+k] as maximization of the sum.
    # The correlation from scipy gives h[i], h[i+k] where mean scales differently, but we know:
    # Let's try an approach with array indexing corrected so that it is inner product/overlap:
    # correlation = [0, 1, ..., 2n-2]
    # k values maps to indices from 0 to 2n-2 with 0 being k=-n+1,  1 -> -n+2, ..., n-1 -> 0 ... etc
    # i.k. for k from -n+1 to n-1, shift k: steady up to length n-k when k+1 to n-1, 
    # but for subtractive k: shift right (positive change to index), 
    # so for k=-n+1 k=-n+2, the shift means h[i] instead of h[n-1+k], etc.
    
    # Let us directly calculate overlap overlap(k) = \sum_i h[i] (1 - h[i+k])
    # Efficient approach:
    # compute (h * (1-h)) overlaps for all k
    # Better: use only k=0.. n-1 and truncated results so that shifts coincide with valid indices
    # OK, just take the actual full conv, but if we interpret it as standard k = shift from -n+1 to n-1,
    # as that's the domain where we're considering distributed i: h[i], h[i+k] overlapping,
    # then that's really compute the shift. normalization: we are minimizing 1 - sum(h[i] (1 - h[i+k])) = 
    # 1 - (sum h + sum h[i] h[i+k]) = 1 - (1 + sum h[i] h[i+k]) = - sum h[i] h[i+k]
    # So need to maximize sum h[i] h[i+k]  via convolution following and taking max of good range.
    # DEMONSTRATION:  
    # Within SCIPY convention:
    # For signals x and y, correlate(x, y) at index j corresponds to shift k = j-n+1 for 
    # x[i+k] = y[i] if that's valid. So for aligning overlap function it's true some way.
    # As k can range from -n+1 to n-1 (since range of i [0,n-1], so shift k in [-n+1, n-1])
    # Therefore, the linear result or signal.convolve(h, 1-h, mode='full') gives result 
    # Indexes from 0 to 2n-2, and shift k from left to right is index j = 0 to 2n-2; 
    # with k = j - (n-1). 
    # The overlap =  sum h[i] (1 - h[i+k]) = sum h[i] - sum h[i] h[i+k] = 1 - sum h[i] h[i+k]
    # We want min  (1 - sum h[i]h[i+k]) = max (sum h[i] h[i+k])
    
    # So maximize the correlation of h with (1-h) (which is our kernel)
    # Find maximum value (sum h[i] h[i+k]) max  k over shift
    # Hints: the factor of dx and sum(h)=1 are interrelated
    correl = signal.correlate(h, 1-h, mode='full')
    max_correl = np.max(correl)  
    C5_max = 1.0 - max_correl*dx
    return C5_max 

def active_shifts(h, eps=1e-9):
    """Return all shift indices k for which overlap is near the max"""
    n = len(h)
    dx = 2.0 / n
    correl = signal.correlate(h, 1-h, mode='full')
    max_correl = np.max(correl)
    active = []
    for i in range(2*n - 1):
        if abs(correl[i] - max_correl) < eps:
            k = i - (n-1)
            active.append(k)
    return active

def gradient(h, ks):
    """Compute the sub-gradient for active shifts"""
    n = len(h)
    g = np.zeros_like(h)
    for k in ks:
        # gradient[i] := (1 - h[i+k]) when i+k is valid,
        # same for the other term,
        for i in range(n):
            val = 0.0 
            if i+k >= 0 and i+k < n:
                val += (1.0 - h[i+k])
            if i-k >= 0 and i-k < n:
                val += (-h[i-k])
            g[i] += val
    return g / len(ks)  # Average over active shifts

def line_search(h, g, t_current, max_step=1e-2):
    """Try various alphabetic steps and choose one that decreases objective"""
    # Try 0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1
    steps = [0.0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1]
    best_h = None
    best_val = t_current
    
    for alpha in steps:
        if alpha == 0:
            new_h = h.copy()
        else:
            new_h = h - alpha * g
            new_h = project_feasible(new_h, np.sum(h))
        
        new_val = evaluate(new_h)

        if new_val < best_val:
            best_val = new_val
            best_h = new_h
            
    if best_h is None:
        return h, t_current
    return best_h, best_val

def local_optimize(h0, max_iter, t_end):
    """Projected gradient descent with line search"""
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate(h)
    start = time.time()
    
    for _ in range(max_iter):
        if time.time() + 1.0 > t_end:
            break
        active = active_shifts(h)
        grad = gradient(h, active)
        h, val = line_search(h, grad, best_val)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val

def anneal(h0, t_end, T0=0.01, alpha=0.995, steps=5000):
    """Simulated annealing to escape local minima"""
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate(h)
    
    T = T0
    for _ in range(steps):
        if time.time() + 1.0 > t_end:
            break
        # Pick two indices
        i, j = np.random.randint(0, len(h), 2)
        delta = np.random.uniform(-0.01, 0.01)
        if h[i] - delta < 0 or h[j] + delta < 0 or h[i] + delta > 1 or h[j] - delta > 1:
            continue
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_h = project_feasible(new_h, np.sum(h))
        new_val = evaluate(new_h)
        delta_obj = new_val - best_val
        
        if delta_obj < 0 or np.random.rand() < np.exp(-delta_obj / T):
            h = new_h
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        T *= alpha
    
    return best_h, best_val

def random_feasible(n):
    """Generate a random feasible vector"""
    x = np.random.uniform(0, 1, n)
    # Scale to make sum = 1
    total = np.sum(x)
    x /= total * 2 / n
    # Project to [0, 1] and enforce sum
    x = project_feasible(x, 1.0)
    return x

def interpolate_vector(h_orig, n_new):
    """Linear interpolation."""
    n_orig = len(h_orig)
    x_orig = np.arange(n_orig) * 2.0 / n_orig
    x_new = np.arange(n_new) * 2.0 / n_new
    new_h = np.interp(x_new, x_orig, h_orig)
    return new_h

def milp_binary(n, time_limit):
    """Mixed Integer Programming formulation for exact binary solutions, if solver allows"""
    try:
        # Attempt using CBC if available
        solver = cp.CBC
        # Fall back to GLPK if needed
        try:
            solver = cp.CBC
        except:
            try:
                solver = cp.GLPK
            except:
                solver = None
                return None, None
    except:
        solver = None
        return None, None
    
    if solver is None:
        return None, None

    # The problem: 
    # min t s.t. for all shifts k=1...n-1:
    #          sum_{i} h[i] h[i+k] >= 1/(dx) - t * dx   where dx = 2./n
    # h binary variables
    # sum(h) = n/2
    
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # dx = 2/n
    dx = 2.0 / n
    
    constraints = [
        cp.sum(h) == n // 2,
    ]
    
    for k in range(1, n):
        # aux variables a[i,k] = h[i] * h[i+k] 
        # Instead of having integer product, we write pairwise linear constraints:
        # a <= h[i]    (1)
        # a <= h[i+k] (2)  
        # a >= h[i] + h[i+k] - 1 (3)
        # a >= 0     (4)
        # Then we constrain sum(a[i,k]) >= 1/dx - t * dx
        
        a = []
        constr_l = []

        # Create variables and constraints to represent the product h[i]*h[i+k] 
        # Two indices: i from 0 to n-1, and i+k = i+k 
        # We have to not have out-of-bounds, so check that i+k < n, for i < (n-k)
        # Also for increasing index, i+k <= n-1 => i <= n-1-k
        # i runs from 0 to max(n-1-k, 0) when summing over a[i,k]   
        
        # Precompute valid indices
        i_list = []
        for i in range(n):
            if i+k < n:
                i_list.append(i)
            else:
                break
    
        for i in i_list:
            a_i = cp.Variable()
            # Constraint (1)
            constr_l.append(a_i <= h[i])
            # Constraint (2)  
            constr_l.append(a_i <= h[i+k])
            # Constraint (3)
            constr_l.append(a_i >= h[i] + h[i+k] - 1)
            # Constraint (4)
            constr_l.append(a_i >= 0)
            a.append(a_i)
        
        constr_l.append(cp.sum(a) >= 1/dx - t*dx)
        constraints.extend(constr_l)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    prob.solve(solver=solver, verbose=False, timelimit=time_limit)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
        h_sol = h.value.round().astype(int)
        bound = t.value
        return h_sol, bound
    
    return None, None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h, best_val, best_n = None, float('inf'), None

    # List of possible sizes to try
    candidate_sizes = [30, 36, 40, 45, 50, 60]
    
    # Use the original initial construction as starting point
    initial_h_new = interpolate_vector(initial_h_values, 45)
    
    # Track which size we're working on
    for n in candidate_sizes:
        if time.time() - t0 > budget_s - 5:
            break
        # Interpolate or reset.
        if n == 45:
            h0 = initial_h_new
        else:
            h0 = interpolate_vector(initial_h_values, n)
            h0 = project_feasible(h0, 1.0)
        
        # If SMALL size, try exact binary MILP
        if n <= 36:
            if time.time() - t0 > budget_s - 5:
                break
            try:
                h_binary, bound = milp_binary(n, min(30, budget_s - (time.time() - t0) - 5))
                if h_binary is not None:
                    h_binary = project_feasible(h_binary, 1.0)
                    val = evaluate(h_binary)
                    if val < best_val - 1e-6:
                        best_val = val
                        best_h = h_binary
                        best_n = n
            except:
                pass

        # Multi-start local optimization
        starts = [h0]
        num_random = min(5, 5 if n > 40 else 3)
        for i in range(num_random):
            starts.append(random_feasible(n))
            
        for i_start, h_s in enumerate(starts):
            if time.time() - t0 > budget_s - 5:
                break
            # Run local optimization with fixed time budget
            t_end = t0 + budget_s - 5
            max_iter = 2000
            
            h_opt, val_opt = local_optimize(h_s, max_iter, t_end)
            if val_opt < best_val - 1e-6:
                best_val = val_opt
                best_h = h_opt
                best_n = n
                
            # Try annealing from this best
            try:
                h_ann, val_ann = anneal(h_opt, t_end)
                if val_ann < best_val - 1e-6:
                    best_val = val_ann
                    best_h = h_ann
                    best_n = n
            except:
                pass
    
    # Refine from coarse if strong candidate
    if best_h is not None and best_n < 60:
        n_upsample = min(60, 2*best_n)
        if n_upsample in candidate_sizes:
            h_upsample = interpolate_vector(best_h, n_upsample)
            h_upsample = project_feasible(h_upsample, 1.0)
            t_end = t0 + budget_s - 5
            h_fine, val_fine = local_optimize(h_upsample, 1000, t_end)
            if val_fine < best_val - 1e-6:
                best_val = val_fine
                best_h = h_fine
                best_n = n_upsample
                
    if best_h is None:
        # fallback to original
        best_h = initial_h_new
        best_val = evaluate(initial_h_new)
        best_n = 45
    
    return (best_h, best_val, best_n)
