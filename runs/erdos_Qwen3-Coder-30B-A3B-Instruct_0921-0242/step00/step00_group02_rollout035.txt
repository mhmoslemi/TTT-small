import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist
import math

def initial_h_values():
    # Initial construction from the problem, 45 samples
    n = 45
    dx = 2.0 / n
    h = np.zeros(n)
    # Simple staggered construction
    for i in range(n):
        x = i * dx
        if 0 <= x < 1:
            h[i] = 1
        elif 1 <= x < 2:
            h[i] = 0
    # Adjust for integral = 1
    h = h * 2.0  # Scale up to have integral = 2; then normalize by 2
    h = h / 2.0
    return h

def evaluate_overlap(h):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    # use np.correlate with full mode and scale
    correl = np.correlate(h, 1 - h, mode="full")
    dx = 2.0 / len(h)
    # ensure correct size: corresponds to shifts in [0,2]
    C5 = np.max(correl) * dx
    return C5

def build_interval_set(a_list, m, ell):
    """Build intervals from list of endpoints"""
    return [(a_list[i], a_list[i] + ell) for i in range(m)]

def compute_overlap_function_O(a_list, m, ell):
    """Compute O(k) = sum over all pairs of intervals of intersecting length"""
    # Precompute all interval pairs
    intervals = build_interval_set(a_list, m, ell)
    # Compute breakpoints
    breaks = set()
    for i in range(m):
        for j in range(m):
            a_i, b_i = intervals[i]
            a_j, b_j = intervals[j]
            # These are the k values where overlap changes
            k1 = a_i - a_j - ell
            k2 = a_i - a_j + ell
            break1 = max(0, min(2, k1))
            break2 = max(0, min(2, k2))
            break1 = max(0, min(2, k1))  
            break2 = max(0, min(2, k2))  
            breaks.add(break1)
            breaks.add(break2)
    break_list = sorted(list(breaks))
    max_k = 0
    for k in break_list:
        # Compute O(k) = sum over all pairs
        O_k = 0
        for i in range(m):
            for j in range(m):
                a_i, b_i = intervals[i]
                a_j, b_j = intervals[j]
                # Length of interval intersection [a_i,b_i] and [a_j+k,b_j+k]
                left = max(a_i, a_j + k)
                right = min(b_i, b_j + k)
                if right > left:
                    O_k += right - left
        max_k = max(max_k, O_k)
    return max_k

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    t0 = time.time()
    best_C5 = float('inf')
    best_h = None
    best_n = None
    
    # Define search parameters
    m_list = [4, 5, 6, 7, 8]
    n_list = [30, 36, 42, 48, 60]
    
    def constraint_bounds(m):
        # Returns lower/upper bounds [a1,a2,...,am] and [b1,b2,...,bm] or others
        # For intervals with fixed length ell = 1/m, placed in [0...2]
        ell = 1.0 / m
        # We have a1 >= 0, a2 >= a1 + ell, ..., am >= a_{m-1} + ell,
        # and am + ell <= 2 => am <= 2 - ell
        # So a1 >= 0, a2 >= ell, ..., a_m <= 2 - (m-1)*ell
        lower = np.array([0] * m)
        upper = np.array([2 - (m - 1) * ell] + [2] * (m - 1))
        for i in range(1, m):
            # a_{i+1} >= a_i + ell => a_i <= a_{i+1} - ell
            upper[i] = min(upper[i], upper[i - 1] - ell)
        return lower, upper

    def compute_interval_overlap(a_list, m):
        """fast overlap evaluation for given intervals"""
        ell = 1.0 / m
        intervals = [(a_list[i], a_list[i] + ell) for i in range(m)]
        max_val = 0
        # We check the breakpoints at endpoints of intervals
        # Evaluate at k = 0, 2 and interval gaps
        # In particular, for k = 0: overlap is just sum of intervals
        base_0 = sum(b - a for a, b in intervals)
        base_2 = sum(b - a for a, b in intervals) 
        breaks = set()
        # generate all 2*m^2 breakpoints
        for i in range(m):
            for j in range(m):
                a_i, b_i = intervals[i]
                a_j, b_j = intervals[j]
                # At displacement k = a_i - a_j - ell
                k1 = a_i - a_j - ell
                k2 = a_i - a_j + ell
                breaks.add(k1)
                breaks.add(k2)
        # filter valid points
        breaks = [k for k in breaks if 0 <= k <= 2]
        breaks = [0] + breaks + [2]
        breaks = sorted(list(set(breaks)))
        
        for k in breaks:
            O_k = 0
            for i in range(m):
                for j in range(m):
                    a_i, b_i = intervals[i]
                    a_j, b_j = intervals[j]
                    left = max(a_i, a_j + k)
                    right = min(b_i, b_j + k)
                    if right > left:
                        O_k += right - left
            max_val = max(max_val, O_k)
        return max_val
    
    def optimize_intervals(m):
        # Return optimized a_list and its overlap value
        ell = 1.0 / m
        lower, upper = constraint_bounds(m)
        def objective(a_list):
            if any(a_list < lower) or any(a_list > upper):
                return float('inf')
            return compute_interval_overlap(a_list, m)
        
        # use DE with population size=20*m, mutation=0.8, crossover=0.9
        result = differential_evolution(objective, list(zip(lower, upper)), 
                                         popsize=20, mutation=0.8, recombination=0.9,
                                         seed=seed, maxiter=2000, disp=False)
        return result.x, result.fun

    def interval_to_discrete(a_list, m, n):
        """
        Convert interval set to discrete binary vector h.
        """
        dx = 2.0 / n
        h = np.zeros(n)
        ell = 1.0 / m
        intervals = build_interval_set(a_list, m, ell)
        for i in range(n):
            x = i * dx
            for a, b in intervals:
                if a <= x < b:
                    h[i] = 1
                    break
        # Make sure sum is exactly n/2
        cur_sum = int(np.sum(h))
        diff = (n//2) - cur_sum
        if diff != 0:
            # Flip some nearest boundary points
            dists = []
            for i in range(n):
                x = i * dx
                min_dist = float('inf')
                for a, b in intervals:
                    min_dist = min(min_dist, abs(x - a), abs(x - b))
                dists.append((min_dist, i))
            dists.sort()
            # Flip diff points among the nearest to an interval boundary
            for i in range(abs(diff)):
                idx = dists[i][1]
                h[idx] = 1 - h[idx]
        return h
    
    def improve_binary(h):
        """Perform simple local search with swaps to improve C5"""
        # we compute exact correlations
        # C5 = max_k int h(x)(1-h(x+k)) dx
        # Here it's max_k (sum_i h[i]*h[i+k])
        best_c5 = evaluate_overlap(h)
        best_h = h.copy()
        # attempt small swaps to improve
        for _ in range(200):
            # Find best corresponding k*
            correlations = np.correlate(h, h, mode="full")
            k_star = np.argmin(correlations)
            if k_star >= len(h):  # adjust
                k_star = k_star - len(h)  # Allow indexing from 0
            # Find approximate k_star from indices
            # we compute again for clear assignment
            correlations = np.correlate(h, h, mode="full") 
            k_star = np.argmin(correlations)
            if k_star >= len(h):
                k_star = k_star - len(h)
            if k_star < 0:
                k_star = len(h) + k_star
                
            # try local swaps
            best_delta = 0
            best_i, best_j = -1, -1
            for _ in range(500):  # random sample of possible swaps
                i = np.random.randint(0, len(h))
                j = np.random.randint(0, len(h))
                if i == j:
                    continue
                # compute delta R_{k*}
                # R_{k*} = sum h[i] h[i+k*]
                # assume FFT or fast convoluion for quick update
                # Here just update directly as less efficient but fine
                if h[i] == 0 or h[j] == 1:  # only one of them is 0 or 1
                    d1 = 0
                    if i+k_star < len(h) and i+k_star >= 0 and h[i+k_star] == 1:
                        d1 += 1
                    if j+k_star < len(h) and j+k_star >= 0 and h[j+k_star] == 1:
                        d1 -= 1
                    if i-k_star < len(h) and i-k_star >= 0 and h[i-k_star] == 1:
                        d1 += 1
                    if j-k_star < len(h) and j-k_star >= 0 and h[j-k_star] == 1:
                        d1 -= 1
                    # flip
                    d1 = -d1  # sets for when 1 -> 0
                    if j == i + k_star:
                        d1 -= 1
                    if j == i - k_star:
                        d1 -= 1
                    if i == j + k_star:
                        d1 += 1
                    if i == j - k_star:
                        d1 += 1
                    delta = d1
                else:
                    d1 = 0
                    d2 = 0
                    if i+k_star < len(h) and i+k_star >= 0 and h[i+k_star] == 1:
                        d1 -= 1
                    if j+k_star < len(h) and j+k_star >= 0 and h[j+k_star] == 1:
                        d1 += 1
                    if i-k_star < len(h) and i-k_star >= 0 and h[i-k_star] == 1:
                        d1 -= 1
                    if j-k_star < len(h) and j-k_star >= 0 and h[j-k_star] == 1:
                        d1 += 1
                    if j+k_star < len(h) and j+k_star >= 0 and h[j+k_star] == 1:
                        d2 -= 1
                    if i+k_star < len(h) and i+k_star >= 0 and h[i+k_star] == 1:
                        d2 += 1
                    if j-k_star < len(h) and j-k_star >= 0 and h[j-k_star] == 1:
                        d2 -= 1
                    if i-k_star < len(h) and i-k_star >= 0 and h[i-k_star] == 1:
                        d2 += 1
                    delta = d1 + d2  # compensate mutual recalculations
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
            if best_delta > 0:
                temp = h[best_i]
                h[best_i] = h[best_j]
                h[best_j] = temp
                new_c5 = evaluate_overlap(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                else:
                    h[best_i] = temp  # revert
            else:
                break
        return best_h, best_c5

    for m in m_list:
        if time.time() - t0 > budget_s - 5:
            break
        try:
            a_opt, min_overlap = optimize_intervals(m)
            for n in n_list:
                if time.time() - t0 > budget_s - 5:
                    break
                h = interval_to_discrete(a_opt, m, n)
                h_refined, c5 = improve_binary(h)
                if c5 < best_C5:
                    best_C5 = c5
                    best_h = h_refined.copy()
                    best_n = n
                # early check
                if best_C5 < 0.379:
                    break
        except Exception as e:
            continue

    if best_h is None:
        h = initial_h_values()
        best_C5 = evaluate_overlap(h)
        best_n = len(h)
        best_h = h

    return (best_h, best_C5, best_n)
