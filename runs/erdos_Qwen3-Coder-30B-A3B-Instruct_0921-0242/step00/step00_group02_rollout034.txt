import numpy as np
import time
from scipy.optimize import differential_evolution
from math import floor, ceil

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Parameters
    max_m = 8
    min_m = 4
    n_values = [30, 36, 42, 48, 60]
    time_per_m = (budget_s - 5) // (max_m - min_m + 1)
    
    # Generate intervals
    for m in range(min_m, max_m + 1):
        if time.time() - t0 > budget_s - 5:
            break
            
        # Phase A: DIRECT-style deterministic global search
        # Initial point (simplified for now)
        a0 = np.array([i * 1.0 / m for i in range(m)])
        a0 = np.clip(a0, 0, 2 - 1.0 / m)
        
        a_best, max_overlap_best = direct_optimize_intervals(a0, m, time.time() - t0, time_per_m)
        
        if max_overlap_best is None:
            continue
            
        # Phase B: Differential Evolution refinement
        bounds = [(0, 2 - 1.0 / m)] * m
        res = differential_evolution(
            lambda a: max_overlap_func(a, m),
            bounds,
            maxiter=2000,
            popsize=10 * m,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            callback=lambda x, convergence: None,
            disp=False
        )
        
        if res.success:
            a_opt = res.x
            max_overlap_opt = res.fun
        else:
            a_opt = a_best
            max_overlap_opt = max_overlap_best
            
        # Discretize and refine
        for n in n_values:
            if time.time() - t0 > budget_s - 5:
                break
                
            h_discrete = intervals_to_vector(a_opt, m, n, 1.0 / n)
            
            # Local binary swaps refinement
            h_final, c5_final = binary_refinement(h_discrete, n)
            
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final
                best_n = n

                break
                
    return (best_h, best_c5, best_n)

def direct_optimize_intervals(a0, m, elapsed, time_left):
    """Simplified deterministic search. In real implementation: use DIRECT-style branch and bound."""
    # For demonstration, return a simplified value
    a_best = a0.copy()
    max_overlap_best = max_overlap_func(a0, m)
    return a_best, max_overlap_best

def max_overlap_func(a, m):
    """Compute max_k overlap O(k) = sum_ij o_ij(k) for given interval endpoints"""
    ell = 1.0 / m
    # Generate interval endpoints
    intervals = []
    for i in range(m):
        intervals.append((a[i], a[i] + ell))
    
    # Collect all breakpoints
    breakpoints = set()
    for idx1 in range(m):
        for idx2 in range(m):
            a1, a2 = intervals[idx1][0], intervals[idx2][0]
            # Breakpoints for o_ij(k) = max(0, ell - |(a1 - a2) - k|)
            # Breakpoints at k = (a1 - a2) ± ell
            k1, k2 = (a1 - a2) - ell, (a1 - a2) + ell
            if 0 <= k1 <= 2:
                breakpoints.add(k1)
            if 0 <= k2 <= 2:
                breakpoints.add(k2)
    breakpoints.add(0)
    breakpoints.add(2)
    
    # Evaluate O(k) at breakpoints (and neighbors to avoid ties)
    eps = 1e-12
    eval_points = list(breakpoints)
    for k in list(breakpoints):
        eval_points.extend([k - eps, k + eps])
    eval_points = list(set(eval_points))
    eval_points = [k for k in eval_points if 0 <= k <= 2]
    
    max_overlap = 0
    for k in eval_points:
        # Compute O(k) = sum_i,j o_ij(k)
        overlap = 0
        for i in range(m):
            for j in range(m):
                a1, a2 = intervals[i][0], intervals[j][0]
                o_ij = max(0, ell - abs((a1 - a2) - k))
                overlap += o_ij
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def intervals_to_vector(a, m, n, dx):
    """Convert interval positions to vector h"""
    ell = 1.0 / m
    h = np.zeros(n)
    x_vals = np.array([j * dx for j in range(n)])
    
    # For each point in the grid, check if it's inside any interval
    for j in range(n):
        x = x_vals[j]
        # Check if x belongs to any interval [a[i], a[i]+ell]
        for i in range(m):
            if a[i] <= x <= a[i] + ell:
                h[j] = 1
                break
    total = np.sum(h)
    target = n / 2
    diff = int(target - total)
    
    # Adjust if necessary
    if diff > 0:
        # Add 1s in places closest to the boundaries
        dist_to_bound = []
        for j in range(n):
            x = j * dx
            min_dist = float('inf')
            for i in range(m):
                min_dist = min(min_dist, abs(x - a[i]), abs(x - (a[i] + ell)))
            dist_to_bound.append(min_dist)
        indices = np.argsort(dist_to_bound)[:diff]
        for idx in indices:
            h[idx] = 1
    elif diff < 0:
        # Remove 1s from points closest to boundary
        dist_to_bound = []
        for j in range(n):
            x = j * dx
            min_dist = float('inf')
            for i in range(m):
                min_dist = min(min_dist, abs(x - a[i]), abs(x - (a[i] + ell)))
            dist_to_bound.append(min_dist)
        indices = np.argsort(dist_to_bound)[-diff:]
        for idx in indices:
            h[idx] = 0
            
    return h

def binary_refinement(h, n):
    """Perform hill climbing swaps using correlation overlap"""
    h = h.copy().astype(np.int64)
    # Compute all correlations for all shifts
    c = np.correlate(h, h, mode='full')
    kmin = np.argmin(c)
    min_val = c[kmin]
    c5 = 1 - min_val / (n / 2)
    # Try small number of swaps
    max_iter = 200
    for i in range(max_iter):
        if time.time() - start_time > 600 - 5:
            break
        improvement = False
        # Generate some swaps
        swap_pairs = []
        for _ in range(500):
            i1, i2 = np.random.choice(n, 2, replace=False)
            if i1 > i2:
                i1, i2 = i2, i1
            if h[i1] == 1 and h[i2] == 0:
                swap_pairs.append((i1, i2))
        for k1, k2 in swap_pairs:
            # Compute change to correlation at kmin
            if k1 + kmin >= 0 and k1 + kmin < len(c) and \
               k2 - kmin >= 0 and k2 - kmin < len(c):
                delta = 0
                # Determine which terms change
                if k1 + kmin < len(c):
                    delta -= h[k1 + kmin]
                if k2 - kmin < len(c):
                    delta -= h[k2 - kmin]
                # Check if candidate swap increases correlation
                if delta + c[kmin] > c[kmin]:
                    h[k1] = 0
                    h[k2] = 1
                    c[kmin] += delta
                    if c[kmin] < min_val:
                        min_val = c[kmin]
                        c5 = 1 - min_val / (n / 2)
                    improvement = True
        if not improvement:
            break
    return h, c5
