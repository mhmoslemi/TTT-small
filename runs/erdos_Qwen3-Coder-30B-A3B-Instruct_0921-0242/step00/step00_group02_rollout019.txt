import numpy as np
import time
import random
from math import ceil
from scipy.signal import correlate

def correlation_vector(b):
    """Compute correlation vector R_k for binary sequence b"""
    r = correlate(b, b, mode='full')
    return r[len(b)-1:]

def update_correlation(R, i, j, b):
    """Update R when swapping b[i] and b[j] (assuming they are 1 and 0)"""
    # For each shift k, update only those pairs that involve i or j
    n = len(b)
    delta = np.zeros(len(R))
    
    for k in range(1, n):
        # Remove i's connections to i-k and i+k  
        if i - k >= 0 and b[i - k] == 1:
            delta[k-1] -= 1
        if i + k < n and b[i + k] == 1:
            delta[k-1] -= 1
            
        # Add j's connections to j-k and j+k 
        if j - k >= 0 and b[j - k] == 1:
            delta[k-1] += 1
        if j + k < n and b[j + k] == 1:
            delta[k-1] += 1
    
    # Update R in place
    R += delta
    return R

def find_worst_correlation(R):
    """Find the index of the minimal correlation (1-indexed)"""
    min_val = np.min(R)
    # Find all indices with minimum value (any one is fine)
    min_indices = np.where(R == min_val)[0]
    return min_indices[0] + 1, min_val  # returning 1-based index and value

def hill_climb(b, R, max_iter=1000, escape_prob=0.05):
    """Local search to improve worst correlation"""
    n = len(b)
    b = b.copy()
    R = R.copy()
    
    for iteration in range(max_iter):
        k_star, _ = find_worst_correlation(R)
        # Find all valid swaps around the worst shift
        swaps = []
        valid_positions = np.where(b == 1)[0]
        for i in valid_positions:
            # near elements to include in neighborhood
            start = max(0, i - 5)
            end = min(n, i + 6)
            neighbor_positions = np.arange(start, end)
            zero_positions = np.where((b == 0) & (np.isin(np.arange(n), neighbor_positions)))[0]
            for j in zero_positions:
                # Compute delta for swap (i,j): this involves updating R according to special rule
                delta = np.zeros(len(R))
                for k in range(1, n):
                    # Remove i's connections to i-k and i+k  
                    if i - k >= 0 and b[i - k] == 1:
                        delta[k-1] -= 1
                    if i + k < n and b[i + k] == 1:
                        delta[k-1] -= 1
                    # Add j's connections to j-k and j+k 
                    if j - k >= 0 and b[j - k] == 1:
                        delta[k-1] += 1
                    if j + k < n and b[j + k] == 1:
                        delta[k-1] += 1
                # Capture the change in correlation at the critical shift only
                delta_k = delta[k_star - 1]  # zero-indexed
                swaps.append((i, j, delta_k))
        
        if not swaps:
            break  # Stagnant

        # Sort by delta descending and pick the best
        swaps.sort(key=lambda s: s[2], reverse=True)
        i, j, delta = swaps[0]  # Best improving
        if delta > 0 or random.random() < escape_prob:
            # Apply swap
            b[i], b[j] = b[j], b[i]
            update_correlation(R, i, j, b)
            if delta <= 0 and random.random() < escape_prob:
                # Allow some non-improving moves
                pass  # keep going
        
    return b, R

def crossover(parent1, parent2):
    """Uniform crossover preserving weight"""
    n = len(parent1)
    # Take union of the 1s
    support = set(np.where(parent1 == 1)[0]) | set(np.where(parent2 == 1)[0])
    # fix weight and create child
    if len(support) > n // 2:
        # remove extras
        excess = len(support) - n // 2
        to_remove = random.sample(list(support), excess)
        for idx in to_remove:
            support.remove(idx)
    if len(support) < n // 2:
        # add some
        complement = set(range(n)) - support
        need = n // 2 - len(support)
        to_add = random.sample(list(complement), need)
        for idx in to_add:
            support.add(idx)
    child = np.zeros(n, dtype=np.uint8)
    for idx in support:
        child[idx] = 1
    return child

def mutate(b, strength=1):
    """Apply mutation by swapping 'strength' pairs"""
    n = len(b)
    idx1 = np.where(b == 1)[0]
    idx0 = np.where(b == 0)[0]
    for _ in range(strength):
        if len(idx1) == 0 or len(idx0) == 0:
            break
        i = random.choice(idx1)
        j = random.choice(idx0)
        b[i], b[j] = b[j], b[i]
        idx1 = np.delete(idx1, np.argwhere(idx1 == i))
        idx0 = np.delete(idx0, np.argwhere(idx0 == j))
        idx1 = np.append(idx1, j)
        idx0 = np.append(idx0, i)
    return b

def get_initial_seed(n, initial_h_values=None):
    """Seed optimization stage"""
    if initial_h_values is None:
        # Fallback: just make a uniform binary pattern
        h = np.zeros(n, dtype=np.uint8)
        h[:n//2] = 1
        return h
    else:
        # Use initial_h_values
        h = np.array(initial_h_values)
        return h

def upsampling(seed_h, target_n):
    """Upsample a binary pattern to larger size"""
    source_len = len(seed_h)
    ratio = ceil(target_n / source_len)
    expanded = np.repeat(seed_h, ratio)
    expanded = expanded[:target_n]
    # fix the weight if needed
    current_sum = expanded.sum()
    if current_sum > target_n // 2:
        # flip excess 1s to 0s
        flip_indices = np.where(expanded == 1)[0][:current_sum - target_n // 2]
        expanded[flip_indices] = 0
    elif current_sum < target_n // 2:
        # flip excess 0s to 1s
        flip_indices = np.where(expanded == 0)[0][:target_n // 2 - current_sum]
        expanded[flip_indices] = 1
    return expanded

def optimize_stage(n, population_size=30, generations=300, seed_n=None, seed=None, time_left=None):
    """Run optimized memetic search on a given size"""
    if seed_n is not None:
        # Up sampling
        seed = upsampling(seed, n)
        # Ensure weight
        while seed.sum() < n // 2:
            zero_pos = np.where(seed == 0)[0]
            choice = random.choice(zero_pos)
            seed[choice] = 1
        while seed.sum() > n // 2:
            one_pos = np.where(seed == 1)[0]
            choice = random.choice(one_pos)
            seed[choice] = 0
    
    # Initialize population
    P = max(30, 2 * n)
    pop = []
    R_pop = []
    
    if seed is not None:
        pop.append(seed)
        r = correlation_vector(seed)
        R_pop.append(r)
    else:
        pop.append(get_initial_seed(n))
        r = correlation_vector(pop[-1])
        R_pop.append(r)
    
    for i in range(P - 1):
        # Proportional sampling from original initial_h_values
        temp_h = get_initial_seed(n)
        pop.append(temp_h)
        r = correlation_vector(temp_h)
        R_pop.append(r)
    
    # Track best solution
    best_idx = 0
    best_t = np.min(R_pop[0])
    
    for g in range(generations):
        if time_left and time_left() < 10:
            break
        
        # Tournament selection
        def select_parent(pop, R_pop):
            chosen = []
            for _ in range(3):
                idx = random.randint(0, P - 1)
                chosen.append((idx, R_pop[idx]))
            best_idx = min(chosen, key=lambda x: np.min(x[1]))[0]
            return pop[best_idx].copy(), R_pop[best_idx].copy()
        
        new_pop = []
        new_R = []
        
        # Elitism: first always include current best
        new_pop.append(pop[best_idx])
        new_R.append(R_pop[best_idx])
        
        for _ in range(P - 1):
            parent1, R1 = select_parent(pop, R_pop)
            parent2, R2 = select_parent(pop, R_pop)
            
            if random.random() < 0.8:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            child = mutate(child, 1)  # small mutation
            
            # Local improvement
            child, R_child = hill_climb(child, R_child)
            
            new_pop.append(child)
            new_R.append(R_child)
        
        # Replace population with new generation
        pop = new_pop
        R_pop = new_R

        # Update best
        for i in range(P):
            t = np.min(R_pop[i])
            if t > best_t:
                best_t = t
                best_idx = i
                
    # Return best individual from last population
    return pop[best_idx], best_t

def run(seed=42, budget_s=600, **kwargs):
    print("Running optimization with budget", budget_s, "seconds")
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    # Stage sizes to try
    stages = [30, 36, 42, 48, 60]
    time_by_stage = {}
    total_allocated = 0
    
    # Precompute rough time estimates
    for i, n in enumerate(stages):
        # Stage 1 takes more time as smaller search
        time_by_stage[n] = max(10, 200 - i*40)
        if total_allocated + time_by_stage[n] > (budget_s - 5):
            break 
        total_allocated += time_by_stage[n]
    
    # Make sure we have time for final polishing
    stages = [n for n in stages if n in time_by_stage]
    
    # Enhanced time check wrapper
    def time_left():
        return budget_s - (time.time() - start_time) - 5
    
    # Keep track of the best candidate
    best_candidate = None
    best_t = -1
    
    # First seed
    initial_seed = None
    try:
        initial_seed = get_initial_seed(60)  # fallback
        # Use full scale initial condition if somehow provided
        if 'initial_h_values' in globals():
            initial_seed = np.array(initial_h_values)[:]
    except:
        pass
    
    # Now process stages in order
    old_seed = initial_seed
    for idx, n in enumerate(stages):
        # time limit earlier stages to be quick, later to be more thorough
        stage_budget = min(time_by_stage[n], budget_s)
        if time_left() < 10:
            break
        
        print(f"Processing stage n={n}, time left: {time_left():.2f}s")
        
        # Early stopping check
        if not old_seed is None:
            stage_seed = downsampling(old_seed, n)
        else:
            stage_seed = None
    
        try:
            if time_left() < 30:
                print("Not enough time for this stage, stopping early")
                break
            # Run with appropriate seed from previous step, budget for this stage
            s_size = max(20, 2*n)
            gen_count = max(50, 200 - 5*(idx - 1)) if idx < len(stages)-1 else 200
            h, t = optimize_stage(
                n=n, 
                population_size=s_size,
                generations=gen_count,
                seed_n=None, seed=stage_seed,
                time_left=time_left
            )
            if t > best_t:
                best_t = t
                best_candidate = h
        except Exception as e:
            print(f"Stage {n} failed: {e}")
            continue
            
        # For next iteration, use current best but mutate if needed
        if best_candidate is not None:
            old_seed = best_candidate
            print(f"Stage {n}: min_correlation = {t:.3f}")
    
    # Final polishing: exhaustive hill climbing in worst shift neighborhood
    if best_candidate is not None:
        h = best_candidate.copy()
        R = correlation_vector(h)
        # keep running local search until no improvement
        # use only improvement-aware swaps for speed
        improvement = True
        while improvement:
            improvement = False
            best_delta = 0
            best_ij = None
            k_star, _ = find_worst_correlation(R)
            
            one_pos = np.where(h == 1)[0]
            for i in one_pos:
                # linear sweep, can make this more efficient but fine for angle shrink
                for j in np.where(h == 0)[0]:
                    delta = 0
                    # compute change in R[k] when swapping i and j, specifically k=k_star
                    # this can be computed by adding counts if 1s are involved
                    for k in range(1, len(h)):
                        if i-k>=0 and h[i-k]==1:
                            delta += (-1 if k==k_star else 0)
                        if i+k<len(h) and h[i+k]==1:
                            delta += (-1 if k==k_star else 0)
                        if j-k>=0 and h[j-k]==1:
                            delta += (+1 if k==k_star else 0)
                        if j+k<len(h) and h[j+k]==1:
                            delta += (+1 if k==k_star else 0)
                    if delta > best_delta:
                        best_delta = delta
                        best_ij = (i, j)
            
            if best_ij:
                i, j = best_ij
                if R[k_star-1] + best_delta > R[k_star-1]:
                    h[i], h[j] = h[j], h[i]
                    update_correlation(R, i, j, h)
                    improvement = True
                    print(f"Polish improved by {best_delta} (k={k_star})")
                else:
                    break
    
    # Final evaluation
    if best_candidate is not None:
        dx = 2.0 / len(best_candidate)
        # Compute C5 via full correlation (should be accurate)
        R_full = correlate(best_candidate, 1 - best_candidate, mode='full')
        c5 = 1 - dx * np.min(R_full[len(best_candidate)-1:])
        return (best_candidate.astype(float), c5, len(best_candidate))
    
    # Return fallback if nothing worked
    h = np.zeros(60)
    h[:30] = 1

    R_full = correlate(h, 1 - h, mode='full')
    c5 = 1 - dx * np.min(R_full[len(h)-1:])
    return (h, c5, 60)
