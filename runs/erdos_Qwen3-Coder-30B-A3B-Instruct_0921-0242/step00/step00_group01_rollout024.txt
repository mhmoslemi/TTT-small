def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral using convolution
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value is what we want
    max_val = np.max(conv) * dx
    return max_val
