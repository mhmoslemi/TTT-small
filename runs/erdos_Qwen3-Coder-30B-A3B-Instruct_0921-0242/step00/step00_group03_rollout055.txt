import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate C5 bound for given h."""
    n = len(h)
    dx = 2.0 / n
    # compute all correlations with 1-h
    corr = correlate(h, 1 - h, mode="full")
    # indices from -n+1 to n-1 (inclusive)
    max_k = np.argmax(corr)
    max_corr = corr[max_k]
    return max_corr * dx

def initial_h_values():
    # Return the initial solution from the problem
    n = 70
    h = np.zeros(n)
    # Create a piecewise constant function that works well in experiments
    # This function has a slightly more complex structure than the basic 1/2
    # but improved empirically in optimizing the overlap integral
    # Let's try something more sophisticated.
    # Try: place a symmetric "tent" shaped function
    for i in range(n):
        x = i * 2.0 / n
        if x < 0.5:
            h[i] = 2 * x
        elif x < 1.5:
            h[i] = 1.0
        else:
            h[i] = 2.0 - 2 * x
    # Normalize to have sum = n/2
    h = h * (n / 2) / np.sum(h)
    return h

def solve_sdp(n):
    # Create SDP relaxation to get a lower bound
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Define the correlation terms
    # We'll copy the formulation as described:
    # This is a simplified model:
    # Correlation over all valid shifts
    # We'll construct the PSD constraints based on the equality:
    # X = h * h^T, and the diagonal = h
    # But actually, let's build it directly from:
    # For each shift s, we want:
    # c_s = dx * (sum(h) - sum(h_i * h_{i+s}))
    # but we're interested in max_s c_s

    # For this SDP relaxation, we do not need to enumerate all s due to its structure.
    # Here we construct: a relaxed SDP with constraints based on variable relationships

    # Constraints:
    # 1. Diagonal of X equals h
    diag_cons = [X[i,i] == h[i] for i in range(n)]
    # 2. X positive semi-definite (relaxing rank=1)
    psd_cons = [X >> 0]
    # 3. Linear sum constraint
    sum_cons = [cp.sum(h) == n / 2]
    # 4. Box constraints
    box_cons = [0 <= h, h <= 1]
    
    # Convert constraints into system of linear constraints:
    # The objective is to minimize t such that t >= c_s for all s
    # This effectively becomes: for all s, Tr(P_s * X) <= const - t / dx
    # where const = n / 2 is the fixed!
    # That's how we build the layer: for two indices s,t:
    # here comes a critical extension:
    # We enumerate all valid shifts for n=240, this is manageable.

    dx = 2.0 / n
    
    # shift matrix for one specific shift (simplified)
    # Actually, let's build one matrix constraint for an individual shift:
    # Just approximate with a few key ones for feasible start.

    num_constraints = 0
    constraints = []
    constraints += diag_cons + psd_cons
    constraints += sum_cons
    constraints += box_cons

    # Let's approach it by sampling some constraints
    # Instead of full set of correlations,
    # Let's just do 50 random indexes and compute correlation
    # Better yet, use a fundamental observation:
    # If we can precompute and wrap constraints, it becomes easier.

    # Right now, perform a direct SDP approach:
    # minimize t
    # s.t. t >= const - v_s for all s
    # which means constraint: v_s <= t where v_s = Tr(P_s * X)
    # But we don't form P_s matrices.

    # For now, this is isotropic to the heuristic / slow route.
    # Assuming our continuous relaxation in variability is good enough,
    # let's use a moderate-sized problem.
    
    # Use fewer but well-chosen constraints for small SDP
    all_shifts = list(range(-n+1, n))
    num_sample_shifts = min(50, len(all_shifts))
    sampled_shifts = np.random.choice(all_shifts, size=num_sample_shifts, replace=False)
    
    # For evaluation: just pick a few.
    constraints = diag_cons + psd_cons + sum_cons + box_cons
    
    # for some shifts our goal is lim t s.t. t >= sum(h_i (1-h_{i+s}))
    # thus we need [sum(h_i h_{i+s})] <= [sum h_i - t*dx]*1/dx = (n/2 - t*dx)/dx = (n/2)/dx - t
    # *sum(h_i h_{i+s}) = Tr(P_s X) <= (n/2)/dx - t = n*(0.5)/dx - t
    # => solve t s.t. (n/2)/dx - t >= Tr(P_s X)

    # Pick 20 random shifts:
    for s_i in range(min(20, len(sampled_shifts))):
        s = sampled_shifts[s_i]
        if s >= 0:
            P = np.zeros((n, n))
            for i in range(n - s):
                P[i + s, i] = 1
        else:
            P = np.zeros((n, n))
            for i in range(-s, n):
                P[i, i + s] = 1
        # Track sum h_i h_{i+s} = Tr(P X)
        Tr_term = cp.trace(P @ X)
        # constant term = dx * sum(h) = dx * n/2
        const = dx * n / 2
        # constraint: Tr(P X) <= const - t*d
        # Cosntraint: t >= const - Tr(P X) / dx
        constraints.append(t >= const - Tr_term / dx)
        num_constraints += 1
    
    # Objective
    problem = cp.Problem(cp.Minimize(t), constraints)
    try:
        # Use ECOS solver by default for quick execution
        # Setting a timeout for solver
        problem.solve(solver=cp.ECOS, verbose=False, max_iters=5000)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return problem.value, h.value
        else:
            print("SDP did not solve to optimality.")
            return None, None
    except Exception as e:
        print(f"SDP failed: {e}")
        return None, None

def round_solution(h_val, n):

    # Sample a random vector, project (if needed), clip and rescale to [0,1], and sum = n/2

        # Reconstruct approximate vector via Cholesky
        # Since we used very coarse approximations in relaxation to save time,
        # we use modulated version (essentially take h_val and get into set)
        x_star = np.clip(h_val, 0, 1)
        sum_val = np.sum(x_star)
        if abs(sum_val - n/2) < 1e-3:  # Already good
            prob_h = x_star
        else:
            # bring to correct sum via linear shifting
            offset = n/2 - sum_val
            if abs(offset) < 1e-8:
                prob_h = x_star
            else:
                # make a form that distributes offset proportionally
                # simple: add (offset/n) to all
                prob_h = np.clip(x_star + offset/n, 0, 1)

        current_sum = np.sum(prob_h)
        diff = n/2 - current_sum
        if abs(diff) > 1e-6:
            # Adjust uniformly deviate to balance
            d = diff / n if abs(diff) > 0 else 0
            prob_h = np.clip(prob_h + d, 0, 1)
            # Must ensure exactly sum=n/2

                indices = np.flatnonzero(np.logical_and(prob_h > 0, prob_h < 1))
                if len(indices) > 0:
                    # Distribute diff among such indices (effectively)

                    idxs = indices[:min(len(indices), int(abs(diff * 100)))]
                    for i in idxs:
                        if diff > 0:

        # Evaluate C5

        if c5 < best_c5:
            best_c5 = c5
            best_h = prob_h.copy()
    return best_h, best_c5

def local_refine_h(h_init, max_iter=50):
    # Perform coordinate descent local search
    h = h_init.copy()
    n = len(h)
    dx = 2.0 / n
    # Memoize correlation values:
    old_corr = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(old_corr)
    max_corr = old_corr[max_k]
    best_h = h.copy()
    best_corr = max_corr
    for _ in range(max_iter):
        improved = False
        # Try moving one index i up | down
        for i in range(n):
            old_h_i = h[i]
            # Simple exploration with small steps
            step = 0.02
            # Try + and - steps
            if h[i] + step <= 1:
                h[i] += step
            if h[i] - step >= 0:
                h[i] -= step
            # A faster lexicographic order:
            # Try to see if small single moves improve the maximize of overlapping
            # current h values
            corr = correlate(h, 1 - h, mode="full")
            max_k = np.argmax(corr)
            new_corr = corr[max_k]
            if new_corr < best_corr:
                # Accept improvement
                best_corr = new_corr
                best_h = h.copy()
                improved = True
            # Set it back to adjust next one
            h[i] = old_h_i
        if not improved:
            break
    return best_h, best_corr
