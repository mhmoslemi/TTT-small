import numpy as np
import cvxpy as cp
from scipy.sparse import diags
from scipy.sparse.linalg import eigs
import time

def project_to_unit_simplex(h, target_sum):
    """Projects h onto the simplex {h: 0 <= h_i <= 1, sum(h) = target_sum}"""
    h = np.array(h)
    theta = 0.0
    low, high = 0.0, 1.0
    
    while high - low > 1e-10:
        theta = (low + high) / 2
        h_proj = np.clip(h - theta, 0, 1)
        if np.sum(h_proj) > target_sum:
            low = theta
        else:
            high = theta
    
    return np.clip(h - theta, 0, 1)

def solve_sdp(n, max_time, sdp_solver='SCS'):
    """Solve the SDP relaxation of the Erdos problem."""
    dx = 2.0 / n
    num_lags = n
    
    # Create shift matrices
    M = []
    for k in range(num_lags):
        offset = k
        diag = np.ones(n - k)
        M.append(diags(diag, offset=offset, shape=(n, n), format='coo'))
    
    # Decision variables
    X = cp.Variable((n, n), symmetric=True)
    h = cp.Variable(n)
    L = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) == h,  # X[i,i] = h[i]
        0 <= h, h <= 1,  # Box constraints
        cp.sum(h) == n / 2,  # Sum constraint
    ]
    
    # Add product constraints
    for i in range(n):
        for j in range(i, n):
            constraints.append(X[i, j] <= h[i])
            constraints.append(X[i, j] <= h[j])
            constraints.append(X[i, j] >= 0)
    
    # Add correlation constraints
    for k in range(num_lags):
        # Use sparse matrix trick to compute trace efficiently
        M_sparse = M[k]
        if M_sparse.nnz > 0:
            trace_expr = cp.trace(M_sparse @ X)
            constraints.append(trace_expr >= L)
    
    # Objective
    objective = cp.Maximize(L)
    prob = cp.Problem(objective, constraints)
    
    # Solve the problem with limited time
    start_time = time.time()
    try:
        prob.solve(solver=sdp_solver, verbose=False, 
                   eps=5e-4, max_iters=4000)
    except:
        pass
    
    elapsed = time.time() - start_time
    if elapsed > max_time or prob.status == cp.INFEASIBLE:
        return None, None, None
    
    # Extract solution if valid
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_optimal = h.value
        X_optimal = X.value
        L_optimal = L.value
        return h_optimal, X_optimal, L_optimal
    
    return None, None, None

def local_optimize(h, n):
    """Simple local optimization via gradient descent on the objective."""
    if h is None:
        return None, None
    
    dx = 2.0 / n
    h = np.array(h)
    h = np.clip(h, 0, 1)
    h = project_to_unit_simplex(h, n / 2)
    
    # Local gradient-based optimization loop
    best_h = h.copy()
    best_C5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    
    for k in range(1000):
        # Compute correlation, find worst-case lag k*
        correlations = []
        for lag in range(n):
            corr = np.sum(h[:-lag] * h[lag:]) if lag < len(h) else 0
            correlations.append(corr)
        
        k_star = np.argmin(correlations)
        C5_val = 1.0 - correlations[k_star]
        
        # Compute subgradient of C5 at k*
        # g[i] = (1 - h[i + k_star]) - h[i - k_star]
        g = np.zeros_like(h)
        g_ext = np.pad(h, (0, k_star), mode='constant', constant_values=0)
        g += -(g_ext[k_star:])  # -h[i+k*]
        g_ext = np.pad(h, (k_star, 0), mode='constant', constant_values=0)
        g += -g_ext[:-k_star]   # -h[i-k*]
        g += 1
        
        # Update with projected gradient descent step
        alpha = 1e-3 / (1 + k**0.5)
        h_new = h - alpha * g
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_unit_simplex(h_new, n / 2)
        
        h = h_new
        new_C5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
        if new_C5 < best_C5:
            best_h = h.copy()
            best_C5 = new_C5
            
    return best_h.tolist(), best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_C5 = np.inf
    n_points = None
    
    # Batch configurations for different sizes
    n_list = [120, 200, 256]  # Start with coarse,
    
    for idx, n in enumerate(n_list):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        if idx == 0:
            max_sdp_time = min(remaining_time * 0.5, 300.0)
        else:
            max_sdp_time = remaining_time * 0.2
        
        # Solve SDP
        h_sdp, _, L_optimal = solve_sdp(n, max_sdp_time)
        if h_sdp is None:
            continue
        
        # Try to smooth the SDP solution via eigenvalue rounding
        try:
            V = eigs(np.atleast_2d(h_sdp @ h_sdp.T), k=1, which="LM")[1].flatten()
            h_rounded = np.abs(V) * np.sqrt(np.sum(h_sdp**2))
            h_rounded = h_rounded / np.sqrt(np.sum(h_rounded**2)) * n / 2
            h_projected = project_to_unit_simplex(h_rounded, n / 2)
            h_projected = np.clip(h_projected, 0, 1)
        except:
            # Fallback to identity
            h_projected = np.clip(h_sdp, 0, 1)
            h_projected = project_to_unit_simplex(h_projected, n / 2)
        
        # Polish with local optimization
        h_polished, c5_polished = local_optimize(h_projected.tolist(), n)
        if h_polished is not None and c5_polished < best_C5:
            best_h = h_polished
            best_C5 = c5_polished
            n_points = n
            
    return (best_h, best_C5, n_points) if best_h is not None else ([], np.inf, 0)
