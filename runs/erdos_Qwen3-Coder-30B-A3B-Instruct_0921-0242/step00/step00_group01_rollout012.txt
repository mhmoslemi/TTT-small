import numpy as np
from scipy import fftpack
import time

def autocorr(b):
    """Compute cyclic autocorrelation of binary vector b."""
    n = len(b)
    # Use FFT for fast computation
    B = fftpack.fft(b)
    A = fftpack.ifft(B * np.conj(B)).real
    return A

def evaluate_c5(h, dx):
    """Evaluate C5 from the discrete step function h."""
    # Compute correlation using numpy correlate (full mode)
    corr = np.correlate(h, 1 - h, mode="full")
    # Return maximum correlation scaled by dx
    return np.max(corr) * dx

def examine_seed(b):
    """Compute the minimal autocorrelation of a binary sequence b."""
    A = autocorr(b)
    # Exclude the zero lag (self-correlation)
    m = np.min(A[1:])
    return m

def swap_improvement(b, i, j, A, k_star):
    """Assess change in autocorrelation at k_star when swapping b[i] and b[j]."""
    # b_i was 1, b_j was 0
    delta = -b[(i + k_star) % len(b)] + b[(j + k_star) % len(b)]
    return delta

def quick_swap_optimize(b, max_time_s=2.0):
    """Fast swap-based local optimization on binary vector b."""
    n = len(b)
    S = n // 2  # Target sum
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    ones_set = set(ones)
    zeros_set = set(zeros)

    if len(ones) != S:
        raise ValueError(f"Sum of b should be {S}, got {len(ones)}")

    A = autocorr(b)
    m = np.min(A[1:])  # Current worst correlation
    k_star = np.argmin(A[1:]) + 1  # Index of worst lag
    best_m = m
    best_b = b.copy()

    swap_count = 0
    temperature = 1.0
    T_decay = 0.99
    tol = 1e-9

    start_time = time.time()
    while time.time() - start_time < max_time_s:
        # Select a random 1 and 0
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)

            delta = swap_improvement(b, i, j, A, k_star)

            # Decide whether to accept the swap
            if delta > 0:
                b[i], b[j] = b[j], b[i]
                ones_set.remove(i)
                zeros_set.add(i)
                ones_set.add(j)
                zeros_set.remove(j)
                A = autocorr(b)
                m_new = np.min(A[1:])
                k_star_new = np.argmin(A[1:]) + 1
                if m_new > best_m:
                    best_m = m_new
                    best_b = b.copy()
                m = m_new
                k_star = k_star_new
                swap_count = 0  # Reset counter
            else:
                # Accept with probability
                if np.random.rand() < np.exp(delta / (temperature + tol)):
                    b[i], b[j] = b[j], b[i]
                    ones_set.remove(i)
                    zeros_set.add(i)
                    ones_set.add(j)
                    zeros_set.remove(j)
                    A = autocorr(b)
                    m_new = np.min(A[1:])
                    k_star_new = np.argmin(A[1:]) + 1
                    m = m_new
                    k_star = k_star_new
                    swap_count = 0  # Reset counter
                else:
                    swap_count += 1
                    if swap_count > 5000:
                        break

            # Cool down temperature
            if swap_count % 1000 == 0:
                temperature = max(1e-10, T_decay * temperature)
        else:
            break

    return best_b, best_m

def expand_binary(b, target_n):
    """Expand a binary sequence by duplicating elements."""
    n = len(b)
    c = np.zeros(target_n, dtype=int)
    for i in range(target_n):
        c[i] = b[i % n]
    return c

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    # Generate seeds
    n_seeds = 100
    n = 200
    dx = 2.0 / n
    seeds = []

    # Generate Legendre sequence
    p = 211  # A prime >= n
    legendre = np.zeros(n, dtype=int)
    for i in range(n):
        legendre[i] = 1 if pow(i, (p - 1) // 2, p) == 1 else 0
    # Ensure balance
    sum_legendre = np.sum(legendre)
    if sum_legendre > n // 2:
        # Remove excess ones
        ones = np.where(legendre == 1)[0]
        excess = sum_legendre - n // 2
        for idx in ones[:excess]:
            legendre[idx] = 0
    elif sum_legendre < n // 2:
        # Add ones (select low magnitude entries?)
        zeros = np.where(legendre == 0)[0]
        deficit = n // 2 - sum_legendre
        for idx in zeros[:deficit]:
            legendre[idx] = 1
    seeds.append(("legendre", legendre))

    # Generate random binary sequence
    for _ in range(10):
        b = np.random.randint(0, 2, n)
        b = b.astype(int)
        sum_b = np.sum(b)
        if sum_b != n // 2:
            # Adjust by flipping elements
            excess = sum_b - n // 2
            if excess > 0:
                # Remove some ones
                ones = np.where(b == 1)[0]
                for i in range(excess):
                    if i < len(ones):
                        b[ones[i]] = 0
            else:
                # Add some zeros (flip to 1) - but that's hard with 0s...
                # instead just match the sum by flipping but averaging the change
                zeros = np.where(b == 0)[0]
                for i in range(-excess):
                    if i < len(zeros):
                        b[zeros[i]] = 1
        seeds.append(("random", b))

    # Compute autocorrelation for seeds
    candidates = []
    for name, b in seeds:
        m = examine_seed(b)
        candidates.append((m, name, b))

    # Sort by minimal autocorrelation
    candidates.sort(reverse=True)

    # Keep top candidates
    top_seeds = [b for m, name, b in candidates[:10]]

    # Optimize each seed
    best_m = -1
    best_h = None
    best_C5 = float('inf')
    best_n = 0

    # Run optimization on top seeds
    for seed_idx, b in enumerate(top_seeds):
        if time.time() - start_time > budget_s - 5:
            break

        try:
            optimized_b, optimized_m = quick_swap_optimize(b, max_time_s=20.0)
            # Re-evaluate minimum autocorrelation
            A_opt = autocorr(optimized_b)
            m_opt = np.min(A_opt[1:])
            if m_opt > best_m:
                best_m = m_opt
                best_h = optimized_b
            if m_opt > 0:
                C5 = dx * (n // 2 - m_opt)
                if C5 < best_C5:
                    best_C5 = C5
                    best_n = n
        except Exception as e:
            continue

    # Try to refine via expansion and additional fine-tuning
    if best_h is None:
        # Fall back to best seed
        b = candidates[0][2]
        best_h = b

    # Final expansion to ~350 (or larger but under 1000)
    n_final = min(400, 1000)
    dx_final = 2.0 / n_final
    h_expanded = expand_binary(best_h, n_final)

    # Re-optimize a bit longer with higher resolution if enough time
    if time.time() - start_time < budget_s - 20:
        try:
            h_expanded, _ = quick_swap_optimize(h_expanded, max_time_s=10)
        except Exception as e:
            pass

    # Final evaluation using older-style correlation
    C5_eval = evaluate_c5(h_expanded, dx_final)

    flag = abs(np.sum(h_expanded) * dx_final - 1.0) < 1e-9
    while not flag and len(h_expanded) < 1000:
        # Padding or refinement?
        h_expanded = np.append(h_expanded, [0.])
        dx_final = 2.0 / len(h_expanded)
        C5_eval = evaluate_c5(h_expanded, dx_final)
        flag = abs(np.sum(h_expanded) * dx_final - 1.0) < 1e-9

    if not flag:
        # Final attempt on first few ones
        h_expanded = h_expanded[:len(h_expanded)-1]
        dx_final = 2.0 / len(h_expanded)
        C5_eval = evaluate_c5(h_expanded, dx_final)

    return (h_expanded, C5_eval, len(h_expanded))
