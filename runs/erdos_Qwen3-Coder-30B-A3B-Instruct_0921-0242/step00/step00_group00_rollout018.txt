import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.linalg import solve

# Global variables to store the basis matrices
Phi = None
Phi_T_Phi_inv = None

def init_global_basis(n, M):
    global Phi, Phi_T_Phi_inv
    # Create the Fourier basis matrix Phi
    x = np.arange(n)
    Phi = np.zeros((n, 2 * M))
    for i in range(M):
        Phi[:, 2*i] = np.cos(2 * np.pi * (i+1) * x / n)
        Phi[:, 2*i + 1] = np.sin(2 * np.pi * (i+1) * x / n)
    
    # Compute (Phi^T Phi)^{-1} Phi^T for efficient projection later
    Phi_T_Phi = Phi.T @ Phi
    Phi_T_Phi_inv = solve(Phi_T_Phi, Phi.T)

def project_to_feasible(c):
    """Project c onto the feasible set defined by 0 <= h <= 1 """
    global Phi, Phi_T_Phi_inv
    h = Phi @ c + 0.5  # Add constant offset 0.5
    h_clipped = np.clip(h, 0, 1)
    delta = h_clipped - h
    correction = Phi_T_Phi_inv @ delta
    return c + correction

def evaluate_objective(c, n):
    """Evaluate the overlap integral C5 from coefficients c"""
    global Phi
    h = Phi @ c + 0.5
    # full convolution with 1-h
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx = 2 / n
    dx = 2.0 / n
    corr *= dx
    return np.max(corr)

def compute_gradient(c, n, k_star):
    """Compute the gradient of C5 with respect to c """
    global Phi
    h = Phi @ c + 0.5
    grad = np.zeros_like(c)
    # For gradient ∂C5/∂h_i = (1 - h_{i+k*}) - h_{i-k*}
    # This is equivalent to doing the convolution with [1, -1] at the right indices
    # But easier to just compute with shifts used by 1-d corr
    grad_h = np.zeros_like(h)
    # Indices of maximum overlap
    ncorr = 2 * n - 1
    # Coefficients also in the corr vector [0, ncorr-1]
    # if k_star = ncorr/2 (central index), then i+k* = i+n-1
    # We shift indexes wrt to the full conv - middle is n-1
    # Index in full_corr corresponding to k_star (offset from middle)
    # y = corr[2*n - 2 - k_star] if searching in corr
    # but actually just do a diff conv to get lead term, need i+k and i-k
    
    h_shifted_plus = np.roll(h, -k_star)
    h_shifted_minus = np.roll(h, k_star)
    grad_h = (1 - h_shifted_plus) - h_shifted_minus
    grad = Phi.T @ grad_h
    return grad

# CMA-ES common parameters
def run_cma_es_batch(c0, n, M, seed, budget_s, total_budget_s):
    np.random.seed(seed)
    # Population size
    lambda_ = 4 + int(3 * np.log(2 * M))
    mu = lambda_ // 2
    sigma = 0.2
    c_m = 0.01
    c_c = 0.1
    c_s = 0.2
    
    # initialize
    population = np.random.randn(lambda_, 2 * M) * sigma
    population[0] = c0
    best_c = c0.copy()
    best_C5 = evaluate_objective(best_c, n)
    sigma = 0.2
    
    # Dictate how much budget is left for current run
    # Constellation plots and structured mutation?
    
    start_time = time.time()
    budget_per_run = 0.4 * total_budget_s
    end_time = start_time + budget_per_run
    
    use_time = lambda: time.time() < end_time
    
    # iteration limit just for safety, no stop condition
    iter_limit = 10000
    
    gen = 0
    while gen < iter_limit and use_time():
        gen += 1
        
        # Evaluation
        fitness = np.zeros(lambda_)
        for i in range(lambda_):
            c = population[i]
            c = project_to_feasible(c)
            fitness[i] = evaluate_objective(c, n)
        
        # Sort by fitness
        indices = np.argsort(fitness)
        best_idx = indices[0]
        if fitness[best_idx] < best_C5:
            best_c = population[best_idx].copy()
            best_C5 = fitness[best_idx]
        
        # Select parents
        parents = population[indices[:mu]]
        mean = np.mean(parents, axis=0)
        
        # Update mean to keep within feasible set
        mean = project_to_feasible(mean)
        population[0] = mean
        
        # Reproduce
        # Normal mutation
        for i in range(1, lambda_):
            # Careful about very small vars - jump in right direction
            # Use same as in other work, varied sigma - e.g. hsig = sum( (y_k / sigma) ** 2 ) > 0.5 or something like that
            # Here simplified.
            population[i] = mean + sigma * np.random.randn(2 * M)
            
        # Adaptive
        # var drop factor
        sigma *= 0.99
        if sigma < 1e-6:
            sigma = 1e-6
            
        # Compute best C5 so far across runs
        # After exp, that's the best one so far
        # This is sufficient since we are only looking to improve
            
    return best_c, best_C5

def run_refinement(c, n, M, num_iter=200):
    """Run projected gradient descent to polish the solution"""
    global Phi

    # Get k_star for init gradient computation
    h = Phi @ c + 0.5
    corr = np.correlate(h, 1 - h, mode='full')
    k_star = np.argmax(corr)
    
    # Start
    for i in range(num_iter):
        grad = compute_gradient(c, n, k_star)
        step = 1.0 / (1 + i) * 0.001
        c = c - step * grad
        c = project_to_feasible(c)
        
        # Update k_star if necessary
        h = Phi @ c + 0.5
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
    
    return c

def project_integral(h):
    # h is originally mostly in [0,1], so it's a simple rescaling
    total = np.sum(h)
    dx = 2.0 / len(h)
    err = total * dx - 1.0
    # Undo err from the mean only correct over the shape so we keep h low, increase where low
    # nudge each entry down by err/n such that total sum is 1 exactly (but tolerate bounds)
    factor = 1.0 - err / len(h)
    h *= factor
    h = np.clip(h, 0, 1)
    # should stay in [0,1] + small impact
    # worst case: we rescaled up all values a bit, clipped them then bring back
    # No real control or guarantee of exact integrality
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 256
    M = 8
    
    # Precompute basis for 256 points
    init_global_basis(n, M)
    print(f"Precomputed basis for n={n}, M={M}")
    
    # Start with zero coefficients:
    c0 = np.zeros(2 * M)
    
    # Run batch of evolutionary strategies with different seeds
    best_c = c0.copy()
    best_C5 = 100
    
    run_end_time = start_time + 0.4 * budget_s
    
    restarts = 10
    for i in range(restarts):
        if time.time() > run_end_time:
            break
        this_seed = seed + i
        c, C5 = run_cma_es_batch(c0, n, M, this_seed, budget_s, budget_s)
        if C5 < best_C5:
            best_C5 = C5
            best_c = c.copy()
            
    print(f"After EAs: best_C5 = {best_C5}")
    
    # If time left AND hit target early - do refinement or double resolution or break early
    
    # [[Refinement]]
    # Refine with projected gradient descent to ensure best point
    best_c = run_refinement(best_c, n, M)
    h_final = Phi @ best_c + 0.5
    # double-check final C5
    corr = np.correlate(h_final, 1 - h_final, mode='full')
    best_C5 = np.max(corr) * (2.0 / n)
    
    print(f"Refined: best_C5 = {best_C5}")
    # Try 512 if enough time
    if time.time() < start_time + 0.9 * budget_s:
        print("Duplicating resolution...")
        # Increase resolution to 512
        n_large = 512
        # Set up new basis - keep M same
        init_global_basis(n_large, M)
        # new c = [old_c, 0, 0, ...] to retain same shape
        c_large = np.zeros(2 * M)
        c_large[:] = best_c[:]  # Copy coefficients
        # Project + final eval
        
        h_final = Phi @ c_large + 0.5
        
        # Evaluate final
        dx = 2.0 / n_large
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        best_C5 = np.max(corr) * dx
        
        print(f"Large resolution: C5 = {best_C5}")
        
        # Final refinements
        best_c = run_refinement(c_large, n_large, M, 100)
        h_final = Phi @ best_c + 0.5
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        best_C5 = np.max(corr) * (2.0 / n_large)
        n = n_large
        
    # Rebuild final h (exact)
    h = Phi @ best_c + 0.5
    # ensure integral is exactly 1.0
    h = project_integral(h)
    # recompute final C5
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    final_C5 = np.max(corr) * dx
    
    print(f"Final best C5: {final_C5}")
    
    return (h.tolist(), float(final_C5), n)
