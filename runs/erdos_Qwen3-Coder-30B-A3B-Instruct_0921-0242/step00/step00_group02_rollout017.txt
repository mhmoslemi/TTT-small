import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
import time
from math import floor

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    correlations = np.correlate(h, 1 - h, mode="full")
    # Extract the valid correlations for shifts k=1 to n-1
    valid_correlations = correlations[n-1:]
    min_correlation = np.min(valid_correlations)
    c5 = 1 - dx * min_correlation
    return c5

def solve_sdp(n):
    """Solve the SDP relaxation for the given n"""
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    
    # Constraints
    constraints = []
    
    # Box constraints
    constraints += [h >= 0, h <= 1]
    
    # Mass constraint
    constraints += [cp.sum(h) == n / 2]
    
    # PSD linking constraints (X = h * h^T unless rank = 1)
    constraints += [X == h @ h.T]
    constraints += [X >> 0]
    
    # Diagonal constraints
    constraints += [cp.diag(X) == h]
    
    # Objective (minimize worst correlation)
    # We'll maximize s in the constraint sum(h_i * h_{i+k}) >= s
    s = cp.Variable(1)
    
    # Use the assumption that we have n-1 shifts (k from 1 to n-1)
    # Create matrix of correlation values for each shift
    # But in this highly constrained approach, we focus on inner constraints
    
    # Maximizing s such that all Phi_k(X) >= s
    # For now, we'll do a simpler optimization:
    # We directly enforce the constraints via system, then solve for s
    
    # Build the sets of inequalities
    P = []
    for k in range(1, n):
        # Build vector that represents sum over i of X(i, i+k)
        row = []
        for i in range(n - k):
            row.append(X[i, i + k])
        Phi_k = cp.sum(row)
        P.append(Phi_k >= s)
    
    # Combine all constraints
    constraints += P
    
    # Objective: maximize s
    objective = cp.Maximize(s)
    
    prob = cp.Problem(objective, constraints)
    
    try:
        # Try using MOSEK if available, otherwise SCS
        prob.solve(solver=cp.MOSEK, verbose=False, feastol=1e-4, abstol=1e-4, reltol=1e-4, max_iters=3000)
    except:
        try:
            prob.solve(solver=cp.SCS, verbose=False, eps=1e-4, max_iter=5000)
        except:
            # Last resort: use ECOS if available
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                # If nothing works, fall back to a simple feasible solution start
                pass
    
    if prob.status != cp.OPTIMAL:
        # Fallback when SDP doesn't converge
        return np.ones(n) * 0.5
    
    h_opt = h.value
    if h_opt is None:
        return np.ones(n) * 0.5

    return h_opt

def hyperplane_rounding(X, n):
    """Perform Gaussian hyperplane rounding to convert fractional solution to binary"""
    # Reconstruct V from X
    # Use eigen decomposition for numerical stability
    try:
        _, V = eigh(X)
        V = V[:, ::-1]  # Sort by decreasing eigenvalues
        V = V[:, :V.shape[1]]
        r = V.shape[1]
    except:
        V = np.random.randn(n, 10)
        r = 10
    
    # If rank is very low, add noise to make it higher
    if r <= 5:
        V += 1e-6 * np.random.randn(n, 10)
        _, V = eigh(V @ V.T)
        V = V[:, ::-1]
        V = V[:, :min(10, n)]
        r = min(10, n)
    
    scores = np.zeros(n)
    # We do this multiple times and take the best for performance
    best_score = float("inf")
    best_b = None
    num_samples = 300
    
    for _ in range(num_samples):
        g = np.random.randn(r)
        z = V @ g  # This simulates the Gaussian projection
        # Sort to find threshold
        indices = np.argsort(z)[::-1]
        b = np.zeros(n)
        b[indices[:n//2]] = 1
        b = np.round(b).astype(int)
        
        if np.sum(b) == n // 2:
            score = evaluate_erdos_solution(b)
            if score < best_score:
                best_score = score
                best_b = b
    return best_b

def swap_polish(h, max_swaps=500):
    """Perform local swap-based polishing to improve solution"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    # Correlation vector
    correlation = np.correlate(h, 1 - h, mode="full")
    valid_correlations = correlation[n-1:]
    k_star = np.argmin(valid_correlations)
    
    corr_vector = valid_correlations.copy()
    last_corr = min(corr_vector)
    
    for _ in range(max_swaps):
        improving = False
        # Try a few pairs to minimize k_star
        candidates = []
        count = 0
        for i in range(n - k_star):
            if h[i] == 1 and h[i + k_star] == 0:
                candidates.append((i, i + k_star))
                count += 1
                if count >= 1000:
                    break
        # Shuffle
        import random
        random.shuffle(candidates)
        for i, j in candidates[:200]:
            # Compute how correlation would change with this swap
            pre = h[i] * h[i + k_star] + h[j] * h[j + k_star]
            post = h[i] * h[i + k_star] + h[j] * h[j + k_star]
            old_val = h[i] * h[j + k_star] + h[j] * h[i + k_star]
            new_val = h[i] * h[j + k_star] + h[j] * h[i + k_star]
            
            # This simplifies - let's recompute properly:
            new_corr = corr_vector.copy()
            old_corr = corr_vector[k_star]
            changes = h[j] * h[i + k_star] - h[i] * h[i + k_star] + h[i] * h[j + k_star] - h[j] * h[j + k_star]
            
            if changes < 0:
                # Actually equivalent to what we're looking at, but use raw change in k_star's correlation
                change = h[i] * h[j + k_star] + h[j] * h[i + k_star] - h[i] * h[i + k_star] - h[j] * h[j + k_star]
                if change < 0:
                    # We are now MORE affected (k_star correlation got worse) - skip
                    continue
                
                # Swap them!
                h[i], h[j] = h[j], h[i]
                corr_vector[k_star] = corr_vector[k_star] + change
                improving = True
                break
    
        if not improving:
            break
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    time_start = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Stages: n=30, 38, 42 (in descending size order for better performance)
    n_values = [30, 38, 42]
    
    for n in n_values:
        if time.time() - time_start > budget_s - 5:
            break
            
        # Solve SDP relaxation
        h_sdp = solve_sdp(n)
        
        # Convert to binary via rounding
        h_binary = hyperplane_rounding(np.outer(h_sdp, h_sdp), n)
        
        if h_binary is None:
            continue
            
        # Local optimization
        h_optimized = swap_polish(h_binary, 500)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h_optimized)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_optimized
            best_n = n
            
        if time.time() - time_start > budget_s - 5:
            break
        
    return (best_h, best_c5, best_n)
