import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist

# Global constants
MAX_M = 7

TIME_A = 0.30

def decode(x, M):
    """Decode the parameter vector x into segment lengths u and heights a."""
    u = x[:M]
    a0 = x[M]
    v = x[M+1:]
    
    # Reconstruct a
    a = np.empty(M + 1)
    a[0] = a0
    for j in range(1, M):
        a[j] = a[j-1] - v[j-1]
    a[M] = a[M-1] - v[M-2] if M > 1 else 0.0
    a[M] = max(0.0, a[M])
    
    # Reconstruct s
    s = np.empty(M + 1)
    s[0] = 0.0
    for j in range(M):
        s[j+1] = s[j] + u[j]
    
    return u, a, s

def overlap_piecewise(s, a):
    """Compute exact C5 value for piecewise-constant function h defined by (s,a)"""
    M = len(a) - 1
    # Collect all breakpoints
    K = set()
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0 <= k1 <= 2:
                K.add(k1)
            if 0 <= k2 <= 2:
                K.add(k2)
    K = sorted(list(K)) + [0.0, 2.0]
    K = [k for k in K if 0 <= k <= 2]
    
    # Compute F(k) for each k in K
    F_values = []
    for k in K:
        F = 0.0
        for j in range(M+1):
            for l in range(M+1):
                idx1 = max(s[j], s[l] - k)
                idx2 = min(s[j+1], s[l+1] - k)
                if idx2 > idx1:
                    L = idx2 - idx1
                    F += a[j] * (1 - a[l]) * L
        F_values.append(F)

    return max(F_values)

def fit_to_integral(u, a, target=1.0):
    """Adjust the heights to exactly match the integral constraint"""
    integral = np.sum(a * u)
    return a * (target / integral) if integral > 0 else a

def fitness_function(x, M, target=1.0):
    """Fitness function for differential evolution"""
    try:
        u, a, s = decode(x, M)
        # Ensure all a are valid
        if np.any(a < 0) or np.any(a > 1):
            return 1e6
        
        # Adjust a to satisfy integral
        a = fit_to_integral(u, a, target)
        
        # Compute penalty for violating integral
        integral = np.sum(a * u)
        integral_penalty = 1e6 * (integral - target)**2 if abs(integral - target) > 1e-8 else 0.0

        # Compute C5
        c5 = overlap_piecewise(s, a)
        
        # Combine penalties
        total = c5 + integral_penalty
        
        if np.isnan(total) or total > 1e6:
            return 1e6
        return total
    except Exception as e:
        return 1e6

def generate_initial_population(n_points, M, bounds):
    """Generate an initial population for DE"""
    pop = []
    for _ in range(n_points):
        x = []
        # Sample segment lengths (Dirichlet-style)
        for i in range(M):
            x.append(np.random.uniform(bounds[0][i], bounds[0][i] + 0.5))  # lengths
        # Sample initial height
        x.append(np.random.uniform(0, 1))
        # Sample height drops
        for i in range(M-1):
            x.append(np.random.uniform(0, 1))
        pop.append(x)
    return pop

def project_capped_simplex(v, total):
    """Project vector v to simplex with sum=total, preserving values in [0,1]"""
    v = np.array(v)
    n = len(v)
    
    # Normalize
    v_sum = np.sum(v)
    if v_sum <= 0:
        return np.ones(n) * (total / n)
    
    v_scaled = v * total / v_sum
    v_proj = np.clip(v_scaled, 0, 1)

    # Adjust if needed
    v_proj_sum = np.sum(v_proj)
    if abs(v_proj_sum - total) < 1e-8:
        return v_proj

    diff = total - v_proj_sum
    if abs(diff) < 1e-8:
        return v_proj

    # Simple adjustment: scale and clamp
    if diff > 0:
        # Add to lower values
        mask = v_proj > 0
        if np.sum(mask) > 0:
            add = diff / np.sum(mask)
            v_proj = np.clip(v_proj + add, 0, 1)
    else:
        # Remove from higher values
        mask = v_proj < 1
        if np.sum(mask) > 0:
            sub = -diff / np.sum(mask)
            v_proj = np.clip(v_proj - sub, 0, 1)
    
    return v_proj

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 256
    dx = 2.0 / n_points

    M_vals = [4, 6]
    best_c5 = float('inf')
    best_solution = None
    
    for M in M_vals:
        if time.time() - start_time > budget_s - 0.5:
            break
            
        # Define parameter dimensions
        dims = M + 1 + (M - 1) if M > 1 else M + 1  # u, a0, v
        bounds = [(0, 2.0)] * M + [(0, 1.0)] + [(0, 1.0)] * (M - 1 if M > 1 else 0)  
        
        # Initialize with default solution
        x0 = [1.0] * M + [0.5] + [0.1] * (M - 1 if M > 1 else 0)  
        
        # DE
        res = differential_evolution(
            fitness_function,
            bounds,
            args=(M, 1.0),
            maxiter=int(1000 / (M + 1)),
            popsize=int(15 * dims),
            seed=seed,
            disp=False,
            polish=False,
            timeout=budget_s - (time.time() - start_time) - 0.5
        )
        
        u, a, s = decode(res.x, M)
        a = fit_to_integral(u, a, 1.0)
        c5 = overlap_piecewise(s, a)
        
        if c5 < best_c5:
            best_c5 = c5
            best_solution = (u.copy(), a.copy(), s.copy(), M)
            
    # If no solution was found, use simple strategy
    if best_solution is None:
        # Just sample.
        M = 5
        u = np.array([0.5] * M)
        u[-1] = 2.0 - np.sum(u[:-1])
        a = np.array([0.5] * (M + 1))
        a[-1] = max(0, a[-2] - 0.1)
        s = np.concatenate([[0.0], np.cumsum(u)])
        best_c5 = overlap_piecewise(s, a)
        best_solution = (u.copy(), a.copy(), s.copy(), M)

    u, a, s, M = best_solution
    
    # Local refinement using smooth approximation
    def smooth_obj(x):
        try:
            u, a, s = decode(x, M)
            if np.any(a < 0) or np.any(a > 1):
                return 1e6
            integral = np.sum(a * u)
            if abs(integral - 1.0) > 1e-8:
                penalty = 1e6 * (integral - 1.0)**2
            else:
                penalty = 0.0
            a = fit_to_integral(u, a, 1.0)
            
            # Use log-sum-exp approximation of max
            K = set()
            for j in range(M+1):
                for l in range(M+1):
                    k1 = s[l] - s[j+1]
                    k2 = s[l+1] - s[j]
                    if 0 <= k1 <= 2:
                        K.add(k1)
                    if 0 <= k2 <= 2:
                        K.add(k2)
            K = sorted(list(K)) + [0.0, 2.0]
            K = [k for k in K if 0 <= k <= 2]
            
            # Use log-sum-exp
            beta = 50.0
            log_sum = -np.inf
            for k in K:
                F = 0.0
                for j in range(M+1):
                    for l in range(M+1):
                        idx1 = max(s[j], s[l] - k)
                        idx2 = min(s[j+1], s[l+1] - k)
                        if idx2 > idx1:
                            L = idx2 - idx1
                            F += a[j] * (1 - a[l]) * L
                log_sum = np.logaddexp(log_sum, beta * F)
            smooth_c5 = log_sum / beta
            
            return smooth_c5 + penalty
        except Exception as e:
            return 1e6
            
    # Optimize with SLSQP using best candidate
    try:
        bounds_ls = [(0, 2.0)] * M + [(0, 1.0)] + [(0, 1.0)] * (M - 1 if M > 1 else 0)
        
        res_ls = minimize(
            smooth_obj,
            res.x,
            method='SLSQP',
            bounds=bounds_ls,
            options={'maxiter': 500},
            constraints={'type': 'eq', 'fun': lambda x: np.sum(decode(x, M)[1] * decode(x, M)[0]) - 1.0}
        )
        if res_ls.success:
            u, a, s = decode(res_ls.x, M)
            a = fit_to_integral(u, a, 1.0)
            c5_local = overlap_piecewise(s, a)
        else:
            c5_local = best_c5
    except:
        c5_local = best_c5

    # Final conversion to discrete h
    h_values = np.zeros(n_points)
    for i in range(n_points):
        x = (i + 0.5) * dx
        for j in range(len(s) - 1):
            if s[j] <= x < s[j+1]:
                h_values[i] = a[j]
                break
        else:
            h_values[i] = a[-1]

    # Project to ensure integral constraint
    current_sum = np.sum(h_values) * dx
    delta = current_sum - 1.0
    if abs(delta) > 1e-4:
        h_values -= delta / n_points
        h_values = np.clip(h_values, 0, 1)

    # Final check 
    corr = np.convolve(h_values, 1 - h_values, mode='full') * dx
    final_c5_bound = np.max(corr)
    
    return (h_values.tolist(), float(final_c5_bound), n_points)
