import numpy as np

def evaluate_overlap(h):
    """Evaluate maximum overlap integral using full correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution of h with (1-h)
    conv = signal.correlate(h, 1 - h, mode='full')
    # The full convolution has 2*n-1 elements; max is in center region
    # But we're only interested in valid shifts in [0, 2], which corresponds
    # to shifts from -(n-1) to (n-1) in the full convolution
    # The maximum valid shift is 2 in x units = 2/dx samples
    # So indices from (n-1) - 2/dx to (n-1) + 2/dx
    max_shift_idx = int(2.0 / dx)  # actual number of indices to consider from center
    start_idx = n - 1 - max_shift_idx
    end_idx = n - 1 + max_shift_idx
    # 'full' mode means we have indices 0 to 2*n - 2
    # The valid range is [start_idx, end_idx] which corresponds to shifts k in [0, 2]
    # But we are only truncating the result properly in case shifting was too large
    valid_conv_vals = conv[start_idx:end_idx+1]
    return np.max(valid_conv_vals) * dx

def solve_sdp_relaxation(n):
    """Solve the SDP relaxation with relaxation"""
    dx = 2.0 / n
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    A = np.eye(n)
    # Constraints:
    # X >= 0 (positive semidefinite)  
    # diag(X) = h (diagonal elements must be h_i)
    # sum(h) = n/2 (average = 0.5)
    # 0 <= h <= 1
    constraints = [
        X >> 0,                           # X is PSD
        cp.diag(X) == h,                  # diagonal elements
        cp.sum(h) == n / 2,               # sum constraint
        h >= 0, h <= 1                    # box constraints
    ]
    # Convolution terms: for each possible shift s=-n+1,...,n-1:
    # c_s = dx * (sum(h) - sum(h_i * h_{i+s}))
    # suppose h_{i+s} not in range, then is 0 -> correct
    for s in range(-n+1, n):
        # construct shift matrix P_s
        P = np.zeros((n, n))
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                P[i, j] = 1.0
        # term = Tr(P_s * X)
        term = cp.trace(P @ X)
        # c_s = dx * (1 - term) 
        # add constraint (you want to minimize max c_s)
        # Replace c_s = dx * (sum(h) - Tr(P_s * X)), so
        # = dx * (n/2 - Tr(P_s X))
        # later: minimize t s.t. t >= c_s for all s 
        constraints.append(t >= dx * (n / 2 - term)) # sum=1, so n/2 for the sum of 1-h, and dx coefficient
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    print("Solving SDP...")
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=20000, eps=1e-6)
        print(f"SDP status: {prob.status}")
        if prob.status == cp.OPTIMAL:
            print(f"Relaxed lower bound: {prob.value}")
            return h.value, X.value, prob.value
        else:
            print(f"Solve failed, status: {prob.status}")
            return None, None, None
    except Exception as e:
        print(f"Solving SDP failed due to exception: {e}")
        return None, None, None

def randomize_round_h(X_star, h_star, n, dx):
    """Random rounding of the SDP solution to feasible binary-like h"""
    # Eigen-decompose X* = UΛU^T
    # Sample y = U Λ^(1/2) g
    try:
        U, Lambda, _ = np.linalg.svd(X_star)
        Lambda = np.maximum(Lambda, 0)  # numerical positive
        Lambda_sqrt = np.sqrt(Lambda)
        g = np.random.standard_normal(n)
        y = U @ (Lambda_sqrt * g)  # Gaussian sample with covariance X*

        h_candidate = y.copy()
        # Clip to [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to exact sum n/2
        current_sum = np.sum(h_candidate)
        delta = n/2 - current_sum
        if abs(delta) > 1e-4:
            # Distribute delta
            indices = np.where((h_candidate > 0) & (h_candidate < 1))[0]
            if len(indices) > 0:
                # Distribute along indices
                ratio = delta / len(indices)
                for i in indices:
                    new_val = h_candidate[i] + ratio
                    h_candidate[i] = np.clip(new_val, 0, 1)
        
        # Final check and reproject if needed
        final_sum = np.sum(h_candidate)
        h_final = h_candidate.copy()
        if abs(final_sum - n/2) > 1e-5:
            # use projection interior
            # Set up a quick simple projection with box constraints
            # Use rescaling
            if final_sum > 0:
                h_final = h_candidate * (n/2) / final_sum
                h_final = np.clip(h_final, 0, 1)
                
        # Correct sum by clipping and renormalizing
        h_final = np.clip(h_final, 0, 1)  
        h_final = h_final * (n/2) / np.sum(h_final)
        
        return h_final
    except Exception as e: 
        print("Rounding failed:", e)
        return h_star.copy()

        idx = i % n

            candidate_vals.append(min(old_val + dx, 1.0))
        if old_val > dx:
            candidate_vals.append(max(old_val - dx, 0.0))
        
        best_val = old_val

        target_sum = n/2

        for val in candidate_vals:
            if val == old_val:
                continue

            diff = val - old_val
            # assume we can adjust just one other element to compensate for the sum constraint
            # total diff = diff + delta -> we need delta = (target_sum - (old_sum + diff)) - (old_sum + diff) = (target_sum - (old_sum + diff)) - (old_sum + diff) = target_sum - old_sum - 2*diff 
            new_h = current_h.copy()
            new_h[idx] = val
            # adjust other element to keep sum
            # try updating 0 index to nearest one (ish)
            cov_idx = (idx + 1) % n

            new_h[cov_idx] = max(0, min(1, new_h[cov_idx])) # keep bounded
            
            new_c5 = compute_C5(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        current_h = best_h.copy()
    return best_h, best_C5
