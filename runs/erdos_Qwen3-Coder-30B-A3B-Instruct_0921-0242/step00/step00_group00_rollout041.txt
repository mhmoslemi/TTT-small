import time
import numpy as np
from scipy.linalg import block_diag, lstsq
from scipy.fft import fftconvolve

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    M = 8  # Number of frequency modes
    n = 256  # Initial resolution
    dx = 2.0 / n
    n_points = n
    
    # Precompute Fourier basis matrix
    i_vals = np.arange(n)
    fourier_basis = np.zeros((n, 2 * M))
    for m in range(1, M + 1):
        fourier_basis[:, m - 1] = np.cos(2 * np.pi * m * i_vals / n)
        fourier_basis[:, M + m - 1] = np.sin(2 * np.pi * m * i_vals / n)
    
    # Precompute (Φ^T Φ)^{-1} Φ^T to avoid recomputation
    FTF = fourier_basis.T @ fourier_basis
    if np.linalg.det(FTF) != 0:
        Pinv = np.linalg.inv(FTF) @ fourier_basis.T
    else:  # Fallback if singular
        Pinv = np.linalg.pinv(FTF) @ fourier_basis.T
    
    def project_to_feasible(c):
        """Project coefficient vector c into feasible set"""
        h = fourier_basis @ c + 0.5  # Actually this gives h[i] = 0.5 + Φ@c
        h_clipped = np.clip(h, 0, 1)
        correction = h_clipped - h
        delta_c = Pinv @ correction
        return c + delta_c
    
    def evaluate_objective(c):
        """Evaluate C5 for given coefficient vector"""
        h = fourier_basis @ c + 0.5
        corr = fftconvolve(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    # CMA-ES
    class CMA_ES:
        def __init__(self, n_params, sigma=0.2):
            self.n_params = n_params
            self.sigma = sigma
            self.mu = n_params // 2
            self.lam = n_params * 4
            self.weights = np.log(self.mu + 0.5) - np.log(np.arange(1, self.mu + 1))
            self.weights = self.weights / np.sum(self.weights)
            self.mu_eff = 1.0 / np.sum(self.weights ** 2)
            self.c_cov = (4 + self.mu_eff / n_params) / (n_params + 4 + 2 * self.mu_eff)
            self.c_sig = (self.mu_eff + 2) / (n_params + self.mu_eff + 5)
            self.d_sig = 1 + 2 * np.max([0, np.sqrt((self.mu_eff - 1) / (n_params + 1)) - 1]) + self.c_sig
            self.c_1 = 2 / ((n_params + 1.3) ** 2 + self.mu_eff)
            self.c_mu = 2 * (self.mu_eff - 2 + 1 / self.mu_eff) / ((n_params + 2) ** 2 + self.mu_eff)
            self.A = np.eye(n_params)
            self.mean = np.zeros(n_params)
            self.sigma = sigma
            self.best_cand = None
            self.best_fitness = float('inf')
    
        def sample(self):
            return np.random.randn(self.n_params) * self.sigma + self.mean
            
        def update(self, fitnesses, samples):
            # Sort samples by fitness
            idx = np.argsort(fitnesses)
            samples = np.array(samples)[idx]
            
            # Update mean
            z = samples[:self.mu] - self.mean
            self.mean = self.mean + self.c_sig * self.sigma * (self.weights @ z)
            
            # Update covariance matrix
            z = samples - self.mean
            self.A = (1 - self.c_cov) * self.A + self.c_1 * z[0] @ z[0].T
            for i in range(1, self.mu):
                self.A += self.c_mu * self.weights[i] * z[i] @ z[i].T
            
            # Update step size
            self.sigma = self.sigma * np.exp(self.c_sig / self.d_sig * (np.linalg.norm(z[0]) - 1))
    
    # Run optimization
    best_c = None
    best_fitness = float('inf')
    
    # Single restart
    es = CMA_ES(n_params=2*M, sigma=0.2)
    restart_count = 0
    max_restarts = 15
    
    while restart_count < max_restarts and (time.time() - start_time + 0.5) < budget_s:
        for _ in range(200):  # Iterations per restart
            if (time.time() - start_time + 0.5) >= budget_s:
                break
            samples = []
            fitnesses = []
            for _ in range(es.lam):
                c = es.sample()
                c = project_to_feasible(c)
                fitness = evaluate_objective(c)
                samples.append(c)
                fitnesses.append(fitness)
            es.update(fitnesses, samples)
            
            if np.min(fitnesses) < best_fitness:
                best_fitness = np.min(fitnesses)
                best_c = samples[np.argmin(fitnesses)].copy()
                
        restart_count += 1
        
        # Check if we've already achieved the target
        if best_fitness <= 0.3807:
            break
    
    # Refinement using gradient descent if needed
    if best_c is not None and (time.time() - start_time + 0.5) < budget_s:
        # Use projected gradient descent to refine
        max_iter = 200
        for it in range(max_iter):
            if (time.time() - start_time + 0.5) >= budget_s:
                break
            h = fourier_basis @ best_c + 0.5
            # Find k that maximizes the overlap
            corr = fftconvolve(h, 1 - h, mode='full') * dx
            k_max = np.argmax(corr)
            # Compute gradient of correlation with respect to h
            grad_h = np.zeros_like(h)
            n_h = len(h)
            if k_max > 0:
                for i in range(n_h):
                    grad_h[i] = -(1 - h[(i + k_max) % n_h])
            if k_max < n_h:
                for i in range(n_h):
                    grad_h[i] -= h[(i - k_max) % n_h]
            # Map gradient to coefficients
            grad_c = fourier_basis.T @ grad_h
            # Update
            alpha = 0.1 / (1 + it)
            best_c -= alpha * grad_c
            best_c = project_to_feasible(best_c)
            
            # Evaluate new fitness
            new_fitness = evaluate_objective(best_c)
            if new_fitness < best_fitness:
                best_fitness = new_fitness
    
    # Final reconstruction
    if best_c is not None:
        h = fourier_basis @ best_c + 0.5
        # Ensure integral constraint
        integral_error = np.sum(h) * dx - 1
        if abs(integral_error) > 1e-12:
            # Apply small uniform correction
            h += integral_error / n
            h = np.clip(h, 0, 1)
        corr = fftconvolve(h, 1 - h, mode='full') * dx
        c5_bound = np.max(corr)
        h_vals = h.tolist()
    else:
        # Fallback to initial construction
        h_vals = [0.5] * n
        corr = fftconvolve(h_vals, [0.5] * n, mode='full') * dx
        c5_bound = np.max(corr)
        
    return (h_vals, float(c5_bound), n_points)
