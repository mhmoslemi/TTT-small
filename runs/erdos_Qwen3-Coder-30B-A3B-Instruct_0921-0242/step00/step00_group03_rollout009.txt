def evaluate_erdos_solution(h_values, n_points):
    """Original evaluation function as described."""
    dx = 2.0 / n_points
    k_max = n_points - 1
    conv = np.correlate(h_values, 1 - h_values, mode='full')
    c5 = np.max(conv) * dx
    return c5

def c5_of_h(h):
    """Compute the maximum overlap integral C5 for a given h."""
    dx = 2.0 / len(h)
    g = 1 - h
    conv = np.correlate(h, g, mode='full') * dx
    return np.max(conv)

def project_to_feasible(h, n_points):
    """Adjust h to satisfy integral constraint with exact sum = n_points / 2."""
    current_sum = np.sum(h)
    deficit = n_points / 2 - current_sum
    if abs(deficit) < 1e-12:
        return h.copy()
    # Try to distribute deficit slightly
    # Ensure h lies between [0,1] and sum to n_points / 2
    h = np.clip(h, 0, 1)
    delta = deficit / (n_points - 2) if n_points > 2 else 0
    for i in range(1, n_points - 1):
        if abs(h[i]) > 1e-10 and abs(h[i] - 1) > 1e-10:
            h[i] += delta
    # Renormalize carefully
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, active_k):
    """Compute the gradient of c5 at h for active shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    # Find the shift that gives max
    # Using the correlation convolution result
    # But for efficiency, we compute the gradient
    # d/dh_i [ \int h(x)(1-h(x+k)) dx ] = dx * (1 - h[i+k]) - dx * h[i-k]
    # indices wrapped appropriately
    if abs(active_k) <= n - 1:
        # Equivalent to convolving with 1-h
        for i in range(n):
            j_plus = (i + active_k) % n
            j_minus = (i - active_k) % n
            grad[i] = dx * (1 - h[j_plus]) - dx * h[j_minus]
    else:
        # Out of range, set to zero
        grad.fill(0)
    return grad

def get_active_shift(h):
    """Find the shift that gives the highest convolution value."""
    # Do the full convolution first
    g = 1 - h
    conv = np.correlate(h, g, mode='full')
    idx = np.argmax(conv)
    n = len(h)
    k = idx - (n - 1)  # shift values are from -(n-1) to n-1
    return k

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 300
    dx = 2.0 / n_points
    
    # Setup constraints
    bounds = [(0, 1)] * n_points
    eq_constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2,
                    'jac': lambda h: np.ones_like(h)}
    
    start_points = []

    # Start 1: Interpolated from initial_h_values
    h_init = initial_h_values()
    if len(h_init) < n_points:
        x_init = np.linspace(0, 2, len(h_init))
        x_target = np.linspace(0, 2, n_points)
        f = interp1d(x_init, h_init, kind='linear', fill_value='extrapolate')
        h_interp = f(x_target)
        h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
        start_points.append(h_interp)
    else:
        h_interp = h_init[:n_points] if len(h_init) > n_points else h_init
        h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
        start_points.append(h_interp)
    
    # Start 2: Random
    for _ in range(3): # 3 random starts
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        start_points.append(h_rand)
    
    # Start 3: Binary-like
    h_binary = np.zeros(n_points)
    h_binary[:n_points//2] = 1
    np.random.shuffle(h_binary)
    h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
    start_points.append(h_binary)

    # Start 4: Smooth block
    h_block = np.zeros(n_points)
    mid = n_points // 2
    width = n_points // 2
    h_block[mid - width//2 : mid + width//2] = 1
    h_block = h_block * (n_points / 2) / np.sum(h_block)
    start_points.append(h_block)

    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points

    num_starts = len(start_points)
    runs_per_start = 100

    # Take time for experimentations
    # Simple loop over starts
    for i, h_start in enumerate(start_points):
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Running several optimizations with different random restarts
        for j in range(runs_per_start):
            if time.time() - start_time > budget_s - 0.5:
                break
            try:
                h_current = np.copy(h_start) + np.random.normal(0, 0.01, n_points)
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)
                
                # Find active k once
                active_k = get_active_shift(h_current)
                
                def obj_func(h):
                    return c5_of_h(h)
                
                def obj_grad(h):
                    try:
                        k_active = get_active_shift(h)
                        return compute_gradient(h, k_active)
                    except:
                        return None  # fallback to finite diff

                result = opt.minimize(obj_func, h_current,
                                      method='SLSQP',
                                      bounds=bounds,
                                      constraints=eq_constraint,
                                      options={'maxiter': 500, 'ftol': 1e-9, 'disp': False},
                                      jac=obj_grad)
                
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_test = c5_of_h(h_opt)
                
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_opt.copy()

            except Exception as e:
                continue
            
            if time.time() - start_time > budget_s - 0.5:
                break
        
        if time.time() - start_time > budget_s - 0.5:
            break

    # Second refinement stage with more focused local search
    if best_h is not None and budget_s - (time.time() - start_time) > 1.0:
        h_final = best_h.copy()

        # Try hill climbing
        for _ in range(500):
            if time.time() - start_time > budget_s - 0.5:
                break
            h_prop = h_final.copy()
            idxs = np.random.choice(n_points, min(5, n_points // 5), replace=False)
            for i in idxs:
                perturbation = np.random.uniform(-0.01, 0.01)
                h_prop[i] += perturbation
                h_prop[i] = np.clip(h_prop[i], 0, 1)
            h_prop = h_prop * (n_points / 2) / np.sum(h_prop)
            c5_prop = c5_of_h(h_prop)
            if c5_prop < best_c5:
                best_c5 = c5_prop
                h_final = h_prop.copy()
        
        best_h = h_final

    if best_h is None:
        # Fallback default if we have no good solution
        best_h = initial_h_values()
        if len(best_h) < n_points:
            x_init = np.linspace(0, 2, len(best_h))
            x_target = np.linspace(0, 2, n_points)
            f = interp1d(x_init, best_h, kind='linear', fill_value='extrapolate')
            best_h = f(x_target)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
        elif len(best_h) > n_points:
            best_h = best_h[:n_points]
        else:
            pass  # same length

        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)

    return (best_h, best_c5, best_n_points)
