import time
import numpy as np
from scipy.linalg import eigh
from cvxpy import *
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_values = [30, 38, 42]
    max_rounds = 300
    max_swaps = 500
    target_c5 = 0.3807
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in n_values:
        if time.time() - start_time > budget_s - 5:
            break
        try:
            # Solve SDP relaxation
            s, h_opt = solve_sdp(n, start_time, budget_s)
            if s is None:
                continue
                
            # Rounding step
            best_binary = None
            best_round_c5 = float('inf')
            
            for _ in range(max_rounds):
                if time.time() - start_time > budget_s - 5:
                    break
                binary_h = random_hyperplane_rounding(h_opt, n)
                if binary_h is None:
                    continue
                c5 = compute_c5(binary_h, n)
                if c5 < best_round_c5:
                    best_round_c5 = c5
                    best_binary = binary_h
                
            if best_round_c5 < best_c5:
                best_c5 = best_round_c5
                best_h = best_binary
                best_n = n
            
            # Local polishing
            if best_binary is not None:
                polished_h = local_swap_polish(best_binary, n, max_swaps, start_time, budget_s)
                polished_c5 = compute_c5(polished_h, n)
                if polished_c5 < best_c5:
                    best_c5 = polished_c5
                    best_h = polished_h
                    best_n = n

            if best_c5 <= target_c5:
                break
                
        except Exception as e:
            continue
    
    # Final check to ensure proper constraints
    if best_h is not None:
        dx = 2.0 / best_n
        assert abs(np.sum(best_h) * dx - 1.0) < 1e-9
        assert np.all((best_h >= 0) & (best_h <= 1))
    else:
        best_h = initial_h_values
        best_n = len(initial_h_values)
        best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, best_n)

def solve_sdp(n, start_time, budget_s):
    # Matrix constraints for correlation
    X = cp.Variable((n, n), symmetric=True)
    h = cp.Variable(n)
    s = cp.Variable()

    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) == h,  # Diagonal elements
        0 <= h, h <= 1,  # Binary constraints
        cp.sum(h) == n/2  # Mass constraint
    ]

    # Compute shift correlations
    for k in range(1, n):
        if time.time() - start_time > budget_s - 5:
            return None, None
        sum_term = cp.sum([X[i, i+k] for i in range(n-k)])
        constraints.append(sum_term >= s) 

    objective = cp.Maximize(s)
    prob = cp.Problem(objective, constraints)
    
    # Solver settings
    prob.solve(solver=cp.SCS, eps=1e-4, max_iters=5000, verbose=False)
    
    if prob.status != cp.OPTIMAL:
        return None, None
        
    return s.value, h.value

def random_hyperplane_rounding(h_opt, n):
    # Apply rounding using Gaussian hyperplane technique
    # Compute one possible factorization
    try:
        # Use eigenvalue decomposition
        V = eigh(np.outer(h_opt, h_opt), subset_by_index=[n-1, n-1])[1][:, 0]
        V = V.reshape(-1, 1)
        r = 1
        
        # Make a random Gaussian vector
        g = np.random.randn(r)
        z = V.T @ g
        
        # Threshold to get binary vector
        threshold = np.quantile(z, 1 - n/2 / n)
        binary_h = (z >= threshold).astype(float)
        
        # Make sure the correct mass
        ones_count = int(n/2)
        if np.sum(binary_h) != ones_count:
            # Adjust the mass
            sorted_indices = np.argsort(z)
            to_flip = ones_count - np.sum(binary_h)
            if to_flip > 0:
                # Flip from 0 to 1
                for idx in sorted_indices[:to_flip]:
                    binary_h[idx] = 1
            elif to_flip < 0:
                # Flip from 1 to 0
                for idx in sorted_indices[-to_flip:]:
                    binary_h[idx] = 0
        
        return binary_h
    except:
        return None

def compute_c5(binary_h, n):
    dx = 2.0 / n
    # Compute correlations for all shifts
    correlations = []
    for k in range(1, n):
        corr = np.correlate(binary_h, 1 - binary_h, mode="full")[k:n]
        correlations.append(np.sum(corr) * dx)
    return 1 - np.min(correlations) if correlations else float('inf')

def local_swap_polish(h, n, max_swaps, start_time, budget_s):
    dx = 2.0 / n
    # Precompute all correlations
    correlations = []
    for k in range(1, n):
        corr = np.correlate(h, 1 - h, mode="full")[k:n]
        correlations.append(np.sum(corr) * dx)
    current_correlations = correlations[:]
    min_corr_idx = np.argmin(current_correlations) + 1  # Shift index (1 to n-1)

    for _ in range(max_swaps):
        if time.time() - start_time > budget_s - 5:
            break
        improved = False
        
        # Check for potentially improving swaps
        # Consider only a few thousand candidates instead of full set
        candidate_pairs = []
        count = 0
        for i in range(n):
            for j in range(i+1, n):
                if h[i] == 1 and h[j] == 0:
                    candidate_pairs.append((i, j))
                    count += 1
                    if count >= 2000:
                        break
            if count >= 2000:
                break
            
        best_pair = None
        best_delta = 0
        for i, j in candidate_pairs:
            delta = 0
            for k in range(1, n):
                if k == min_corr_idx:
                    # Compute the impact of swapping i (1) and j (0)
                    if (i + k < n) and (j + k < n):
                        if h[i + k] == 1:
                            delta += 1
                        if h[j + k] == 0:
                            delta -= 1
                        if h[i] == 1 and (i + k < n) and h[i + k] == 0:
                            delta -= 1
                        if h[j] == 0 and (j + k < n) and h[j + k] == 1:
                            delta += 1
            if delta > best_delta:
                best_delta = delta
                best_pair = (i, j)
        
        if best_pair is None:
            break
            
        i, j = best_pair
        # Perform swap
        h[i], h[j] = 0, 1
        # Update the correlation of the critical shift
        for k in range(1, n):
            if k == min_corr_idx:
                if i + k < n and j + k < n:
                    # If both were 1, reduce corr by 1.
                    # If both were 0, no effect.
                    # If i was 1, j was 0, then reduce by 1 (i's contribution removed)
                    # If j was 1, i was 0, then increase by 1 (j's contribution added)
                    if h[i + k] == 1 and (j + k < n) and h[j + k] == 0:
                        current_correlations[k - 1] -= 1
                    elif h[j + k] == 1 and (i + k < n) and h[i + k] == 0:
                        current_correlations[k - 1] += 1
                        
        # Update the min shift index
        min_corr_idx = np.argmin(current_correlations) + 1
        improved = True
        
        if not improved:
            break
    
    return h
