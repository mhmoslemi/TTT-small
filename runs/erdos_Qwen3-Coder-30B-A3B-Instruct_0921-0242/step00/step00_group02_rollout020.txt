import numpy as np
import time
import random
from scipy.signal import correlate

def correlation_vector(b):
    """Compute the correlation vector R_k for binary sequence b."""
    return correlate(b, b, mode='full')[1:]

def update_correlation(R, i, j, b):
    """Update R vector in place after swapping b[i] and b[j]."""
    n = len(b)
    # For each shift k, check the affected correlations
    for k in range(1, n):
        # Subtract contribution from old i and j
        if i - k >= 0 and b[i - k] == 1:
            R[k-1] -= 1
        if i + k < n and b[i + k] == 1:
            R[k-1] -= 1
        # Add contribution from new i and j
        if j - k >= 0 and b[j - k] == 1:
            R[k-1] += 1
        if j + k < n and b[j + k] == 1:
            R[k-1] += 1

def evaluate_candidate(b, dx):
    """Evaluate the C5 bound for binary sequence b."""
    R = correlation_vector(b)
    min_R = min(R)
    return 1 - dx * min_R

def local_improvement(b, R, max_iter=1000):
    """Locally improve b using hill climbing on the lowest correlation."""
    n = len(b)
    best_R = R.copy()
    best_b = b.copy()
    best_t = min(R)
    for _ in range(max_iter):
        k_star = np.argmin(best_R) + 1
        max_delta = 0
        best_swap = None
        # Enumerate all swaps between 1 and 0 positions
        ones = [i for i in range(n) if best_b[i] == 1]
        zeros = [i for i in range(n) if best_b[i] == 0]
        for i in ones:
            for j in zeros:
                # Compute delta change to R[k_star]
                delta = 0
                # Subtract old contributions
                if i - k_star >= 0 and best_b[i - k_star] == 1:
                    delta -= 1
                if i + k_star < n and best_b[i + k_star] == 1:
                    delta -= 1
                if j - k_star >= 0 and best_b[j - k_star] == 1:
                    delta += 1
                if j + k_star < n and best_b[j + k_star] == 1:
                    delta += 1
                # Only consider improving swaps
                if delta > max_delta:
                    max_delta = delta
                    best_swap = (i, j)
        if best_swap is None or max_delta <= 0:
            break
        i, j = best_swap
        best_b[i] = 0
        best_b[j] = 1
        update_correlation(best_R, i, j, best_b)
        new_t = min(best_R)
        if new_t > best_t:
            best_t = new_t
    return best_b, best_R

def generate_random_binary_sequence(n):
    """Generate a random binary sequence of length n with weight n//2."""
    ones = random.sample(range(n), n//2)
    b = np.zeros(n, dtype=int)
    b[ones] = 1
    return b

def crossover(p1, p2):
    """Cross two binary sequences."""
    n = len(p1)
    union = set()
    for i in range(n):
        if p1[i] == 1 or p2[i] == 1:
            union.add(i)
    # It might be too big or too small, so we adjust
    if len(union) > n//2:
        to_remove = list(union)[n//2:]
        for i in to_remove:
            union.remove(i)
    elif len(union) < n//2:
        to_add = set()
        for i in range(n):
            if i not in union:
                to_add.add(i)
        add_list = random.sample(list(to_add), n//2 - len(union))
        union.update(add_list)

    child = np.zeros(n, dtype=int)
    for i in union:
        child[i] = 1
    return child

def mutate(b, mu=2):
    """Mutate a binary sequence by swapping mu positions."""
    n = len(b)
    ones = [i for i in range(n) if b[i] == 1]
    zeros = [i for i in range(n) if b[i] == 0]
    if len(ones) < mu or len(zeros) < mu:
        # Just do one swap
        i = random.choice(ones)
        j = random.choice(zeros)
        b[i] = 0
        b[j] = 1
    else:
        for _ in range(mu):
            i = random.choice(ones)
            j = random.choice(zeros)
            b[i] = 0
            b[j] = 1
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)

def exhaustive_search_on_best(b):
    """Do exhaustive search on the best state to refine further."""
    n = len(b)
    dx = 2.0 / n
    R = correlation_vector(b)
    k_star = np.argmin(R) + 1
    iterations = 0
    while iterations < 1000:
        max_delta = 0
        best_swap = None
        ones = [i for i in range(n) if b[i] == 1]
        zeros = [i for i in range(n) if b[i] == 0]
        for i in ones:
            for j in zeros:
                delta = 0
                if i - k_star >= 0 and b[i - k_star] == 1:
                    delta -= 1
                if i + k_star < n and b[i + k_star] == 1:
                    delta -= 1
                if j - k_star >= 0 and b[j - k_star] == 1:
                    delta += 1
                if j + k_star < n and b[j + k_star] == 1:
                    delta += 1
                if delta > max_delta:
                    max_delta = delta
                    best_swap = (i, j)
        if best_swap is None or max_delta <= 0:
            break
        i, j = best_swap
        b[i] = 0
        b[j] = 1
        update_correlation(R, i, j, b)
        iterations += 1
    return b
