    # Variables: b[i] are binary; z[i,k] are products; w is auxiliary variable
    b = Variable(n0, boolean=True)
    z = Variable((n0, n0), boolean=True)  # z[i,k] = b[i] * b[(i+k)%n0]
    w = Variable()  # auxiliary variable for the minimum overlap
    
    # Constraints for z[i,k] = b[i] * b[(i+k)%n0] (exact linearization)
    constraints = []
    for i in range(n0):
        for k in range(n0):
            constraints += [
                z[i,k] <= b[i],
                z[i,k] <= b[(i+k) % n0],
                z[i,k] >= b[i] + b[(i+k) % n0] - 1
            ]
    
    # Overlap constraints: w >= S - sum(z[i,k]) for all k
    for k in range(n0):
        # Total overlap for lag k: O_k = S - sum_{i} z[i,k] = sum_{i} b[i]*b[i+k] (circular)
        constraints.append(w >= S - sum(z[i,k] for i in range(n0)))
    
    # Mass constraint: sum(b[i]) = S
    constraints.append(sum(b) == S)
    
    # Symmetry breaking (fix b[0] = 1 and enforce lexicographic order)
    constraints.append(b[0] == 1)
