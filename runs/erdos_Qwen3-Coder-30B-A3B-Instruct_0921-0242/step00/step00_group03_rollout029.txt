import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Only consider the valid overlap integrals (k from 0 to 2)
    # The full correlation has length 2*n-1
    # Valid k values range from 0 to 2, which are indices n-1 to 2*n-2
    # But we want only those from interval [0,2], so it's from index n-1 to n+1 (exclusive)
    # Actually, with a shift of k, we are computing integral of h(x)(1-h(x+k)) for k in [0,2]
    # The correlation represents shifts from -n+1 to n-1
    # So k = shift + (n-1), and k = 0 to 2 means shift = -n+1 to -n+3
    # For valid range of k, we want correlations with indices corresponding to shifts -n+1,...,n-1 with effective k in [0,2]
    # Actually, working in standard interpretation: 
    # When valid k (shift) is 0 to 2 (dx step), valid indices of correlation must be k/dx
    # But to approach the problem cleanly:
    # Let's recompute properly, use indexing for valid shifts
    # convolve using numpy: we want all shifts k from 0 to 2 in steps of dx
    shifted_corr = corr[n-1:]  # this gives k from 0 to 2 in terms of dx
    
    # The C5 is the maximum of all correlations, multiplied by dx
    c5 = np.max(shifted_corr) * dx
    return c5
