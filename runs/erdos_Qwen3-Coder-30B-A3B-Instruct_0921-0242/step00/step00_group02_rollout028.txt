import numpy as np
import time
import random
from scipy.optimize import differential_evolution
from scipy.spatial.distance import pdist

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    best_C5 = float('inf')
    best_h = None
    best_n = None
    
    # Try different number of intervals
    m_values = [4, 5, 6, 7, 8]
    n_values = [30, 36, 42, 48, 60]
    
    for m in m_values:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Phase A: DIRECT-like global optimization
        best_a, best_O_max = optimize_intervals_direct(m, start_time, budget_s)
        
        # Phase B: Differential Evolution refinement
        a_refined = refine_intervals_de(best_a, m, start_time, budget_s)
        
        # Convert to discrete step function
        for n in n_values:
            if time.time() - start_time > budget_s - 5:
                break
            h_discrete = intervals_to_discrete(a_refined, m, n)
            
            # Post-processing swap improvement
            h_improved = swap_improvement(h_discrete, start_time, budget_s)
            
            # Evaluate C5
            C5_val = evaluate_c5(h_improved)
            
            if C5_val < best_C5:
                best_C5 = C5_val
                best_h = h_improved
                best_n = n
                
            if best_C5 < 0.379:
                break
                
        if best_C5 < 0.379:
            break
    
    return (best_h, best_C5, best_n)

def intervals_to_discrete(a, m, n):
    # Get interval endpoints
    dx = 2.0 / n
    h = np.zeros(n)
    interval_length = 1.0 / m
    
    # Determine which grid points are inside intervals
    for i in range(n):
        x = i * dx
        for j in range(m):
            if a[j] <= x <= a[j] + interval_length:
                h[i] = 1
                break
    
    # Adjust to ensure sum = n / 2
    current_sum = np.sum(h)
    if current_sum != n / 2:
        diff = int(n / 2 - current_sum)
        # Find indices to flip
        if diff > 0:  # Need to add ones
            # Find indices farthest from interval boundaries
            distances = []
            for i in range(n):
                min_dist = float('inf')
                for j in range(m):
                    left = a[j]
                    right = a[j] + interval_length
                    dist = min(abs(i*dx - left), abs(i*dx - right))
                    if dist < min_dist:
                        min_dist = dist
                distances.append((min_dist, i))
            
            distances.sort()
            for i in range(min(diff, len(distances))):
                idx = distances[i][1]
                if h[idx] == 0:
                    h[idx] = 1
        elif diff < 0:  # Need to remove ones
            # Find indices closest to interval boundaries
            distances = []
            for i in range(n):
                min_dist = float('inf')
                for j in range(m):
                    left = a[j]
                    right = a[j] + interval_length
                    dist = min(abs(i*dx - left), abs(i*dx - right))
                    if dist < min_dist:
                        min_dist = dist
                distances.append((min_dist, i))
            
            distances.sort()
            for i in range(min(-diff, len(distances))):
                idx = distances[i][1]
                if h[idx] == 1:
                    h[idx] = 0
    
    return h

def optimize_intervals_direct(m, start_time, budget_s):
    # Initialize with the initial construction if available
    best_a = None
    best_O_max = float('inf')
    
    # Try some initial points based on problem structure
    for _ in range(50):
        if time.time() - start_time > budget_s - 5:
            return best_a, best_O_max
        # Warm-start with a structured solution
        a = np.linspace(0, 2 - 1.0/m, m)
        # Slightly randomize
        for i in range(m):
            a[i] += (random.random() - 0.5) * 0.1
        # Enforce correct ordering and boundary constraints
        a = enforce_constraints(a, m)
        
        # Evaluate O(k)
        O_max = evaluate_O_max(a, m)
        if O_max < best_O_max:
            best_O_max = O_max
            best_a = a.copy()
            
    return best_a, best_O_max

def evaluate_O_max(a, m):
    # Compute all breakpoints
    interval_length = 1.0 / m
    breakpoints = set()
    
    for i in range(m):
        for j in range(m):
            # For intervals [a[i], a[i]+l] and [a[j], a[j]+l]
            breakpoints.add(a[i] - a[j] - interval_length)  # k = (a[i] - a[j]) - l
            breakpoints.add(a[i] - a[j] + interval_length)  # k = (a[i] - a[j]) + l
    
    # Keep only those in [0,2]
    breakpoints = [k for k in breakpoints if 0 <= k <= 2]
    
    # Evaluate O(k) at all critical points
    max_O = 0
    for k in breakpoints:
        O_k = evaluate_O_k(a, m, k)
        max_O = max(max_O, O_k)
    
    # Also check k=0 and k=2
    max_O = max(max_O, evaluate_O_k(a, m, 0))
    max_O = max(max_O, evaluate_O_k(a, m, 2))
    
    return max_O

def evaluate_O_k(a, m, k):
    interval_length = 1.0 / m
    total = 0
    
    for i in range(m):
        for j in range(m):
            left1 = a[i]
            right1 = a[i] + interval_length
            left2 = a[j] + k
            right2 = a[j] + interval_length + k
            
            # Overlap of two intervals
            overlap = max(0, min(right1, right2) - max(left1, left2))
            total += overlap
    
    return total

def enforce_constraints(a, m):
    # Project onto ordered simplex
    a = np.array(a)
    
    # Order it
    a_sorted = np.sort(a)
    
    # Ensure bounds
    interval_length = 1.0 / m
    if a_sorted[0] < 0:
        a_sorted[0] = 0
    if a_sorted[-1] + interval_length > 2:
        shift = (2 - interval_length) - a_sorted[-1]
        a_sorted += shift
    
    # Ensure ordering and gap constraints
    gap = 0
    for i in range(m-1):
        if a_sorted[i+1] < a_sorted[i] + interval_length:
            gap = max(0, a_sorted[i] + interval_length - a_sorted[i+1])
            for j in range(i+1, m):
                a_sorted[j] += gap
    
    # Final project back if needed
    if a_sorted[0] < 0:
        shift = -a_sorted[0]
        a_sorted += shift
    if a_sorted[-1] + interval_length > 2:
        shift = (2 - interval_length) - (a_sorted[-1] + interval_length)
        a_sorted += shift
        
    return np.array(a_sorted)

def refine_intervals_de(a_initial, m, start_time, budget_s):
    def objective(a):
        if time.time() - start_time > budget_s - 10:
            raise RuntimeError("Time limit reached")
        return evaluate_O_max(a, m)
    
    # Make bounds for differential evolution
    bounds = [(0, 2 - 1.0/m)] * m
    history = []
    
    # Ensure bounds are valid
    a_initial = enforce_constraints(a_initial, m)
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=1000,
            popsize=10*m,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=42,
            disp=False
        )
        return result.x
    except:
        return a_initial
    
def swap_improvement(h, start_time, budget_s):
    # Work with the already discretized h
    max_swaps = 200
    swaps_done = 0
    
    while swaps_done < max_swaps and time.time() - start_time < budget_s - 5:
        # Compute correlations
        R = np.correlate(h, h, mode='full')
        R = R[len(R)//2:]  # Take positive lags
        k_opt = np.argmin(R)
        
        # Find best single swap
        best_delta = 0
        best_i, best_j = -1, -1
        
        # Try a few candidate swaps
        candidate_indices = []
        for i in range(len(h)):
            for j in range(len(h)):
                if h[i] == 1 and h[j] == 0:
                    candidate_indices.append((i, j))
        
        # Shuffle to diversify
        random.shuffle(candidate_indices)
        for i, j in candidate_indices[:500]:
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute change in R[k_opt]
            delta = 0
            
            # i becomes 0, j becomes 1
            
            # Contributions to sum when i is removed (in cross-correlations)
            # and when j is added
            # Change in R[k_opt] = 
            #   contribution from i matches at k_opt
            #   contribution from j matches at k_opt
            #   cross-terms
            
            # For i: subtract if it was part of correlation
            # For j: add if it contributes to correlation
            
            # Actual exact implementation
            delta_R = 0
            for k in range(len(h)):
                # k_opt is what we care about
                # We shift t by k and t+k to see if positions match
                # If i is at position during correlation, and i+k is also at position...
                old_contribution = int(i + k < len(h) and h[i + k] == 1) + int(j - k >= 0 and h[j - k] == 1)
                new_contribution = int(i + k < len(h) and h[i + k] == 0) + int(j - k >= 0 and h[j - k] == 0)
                delta_R += (new_contribution - old_contribution)
            
            if delta_R < best_delta:
                best_delta = delta_R
                best_i, best_j = i, j
                
        if best_i != -1 and best_j != -1:
            h[best_i] = 0
            h[best_j] = 1
            swaps_done += 1
        else:
            break
            
    return h

def evaluate_c5(h):
    # c5 = max_k int h(x)(1 - h(x + k)) dx
    # Since h is discretized, we can use np.correlate
    h_flip = 1 - h
    corr = np.correlate(h, h_flip, mode='full')
    corr = corr[len(corr)//2:]  # Take positive lags
    return np.max(corr) * 2 / len(h)
