import numpy as np
import scipy.fft as fft
from scipy.optimize import differential_evolution
import time
from math import floor, ceil

def initial_h_values():
    """Provide an initial reasonable construction"""
    n = 47
    h = np.zeros(n)
    # Based on previous known constructions
    h[0:20] = 1.0
    h[20:30] = 0.5
    h[30:47] = 0.0
    return h

def evaluate_C5(h_vals, dx=2.0/len(h_vals)):
    """Evaluate C5 using FFT for fast correlation"""
    n = len(h_vals)
    h_fft = fft.fft(h_vals)
    autocorr = fft.ifft(np.conj(h_fft) * h_fft).real
    # Circular autocorrelation values (k = 0 to n-1)
    A = autocorr[:n]
    return dx * (n / 2.0 - np.min(A))

def block_family_search(n_points=400):
    """Analytic block search for 3-block patterns."""
    h_best = None
    C5_best = float('inf')
    
    dx = 2.0 / n_points
    a_vals = np.arange(0, 2, 0.02)
    b_vals = np.arange(0.02, 2, 0.02)
    
    for a in a_vals:
        for b in b_vals:
            if a >= b:
                continue
            # Solve p from a*1 + (b-a)*p = 1
            p = (1.0 - a) / (b - a)
            if 0 <= p <= 1:
                h = np.zeros(n_points)
                h[:int(a/dx)] = 1.0
                h[int(a/dx):int(b/dx)] = p
                # Rescale to ensure integral = 1
                total = np.sum(h) * dx
                if total != 0:
                    h = h / total * 1.0
                C5 = evaluate_C5(h, dx)
                if C5 < C5_best:
                    C5_best = C5
                    h_best = h.copy()
    return h_best, C5_best

def four_block_pattern(a, b, c, n_points):
    """Construct 4-block pattern: [1, p, q, 0]"""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:int(a/dx)] = 1.0
    p = (1.0 - a) / (b - a)
    q = (1.0 - a - (b-a)*p) / (c - b)
    if 0 <= p <= 1 and 0 <= q <= 1:
        h[int(a/dx):int(b/dx)] = p
        h[int(b/dx):int(c/dx)] = q
        # Adjust to meet integral exactly
        total = np.sum(h) * dx
        if total > 0:
            h = h / total * 1.0
        return h
    return None

def four_block_search(n_points=400):
    """Search 4-block patterns"""
    h_best = None
    C5_best = float('inf')
    
    dx = 2.0 / n_points
    a_vals = np.arange(0.01, 2, 0.04)
    b_vals = np.arange(0.04, 2, 0.04)
    c_vals = np.arange(0.08, 2, 0.04)
    
    for a in a_vals:
        for b in b_vals:
            for c in c_vals:
                if a < b < c:
                    h = four_block_pattern(a, b, c, n_points)
                    if h is not None:
                        C5 = evaluate_C5(h, dx)
                        if C5 < C5_best:
                            C5_best = C5
                            h_best = h.copy()
    return h_best, C5_best

def stochastic_search(n_points=200, budget=120):
    """Low-dimensional stochastic search using mixture of indicators."""
    def individual_to_h(individual, n_points):
        c1, a1, c2, a2, c3, a3 = individual
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for c, a in zip([c1, c2, c3], [a1, a2, a3]):
            left = max(0, int((c - 0.1) / dx))
            right = min(n_points, int((c + 0.1) / dx))
            h[left:right] = np.maximum(h[left:right], a)
        # Normalize
        total = np.sum(h) * dx
        if total > 0:
            h = h / total * 1.0
        h = np.clip(h, 0, 1)
        return h
    
    def fitness(individual):
        h = individual_to_h(individual, n_points)
        C5 = evaluate_C5(h, 2.0 / n_points)
        return -C5
    
    bounds = [(0.0, 2.0), (0.0, 1.0), (0.0, 2.0), (0.0, 1.0), (0.0, 2.0), (0.0, 1.0)]
    result = differential_evolution(fitness, bounds, seed=42, maxiter=300, popsize=30)
    h_opt = individual_to_h(result.x, n_points)
    C5_opt = evaluate_C5(h_opt, 2.0 / n_points)
    return h_opt, C5_opt

def project_to_feasible(h, target_sum):
    """Project h to lie in [0,1] and have sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
    return h

def gradient_ascent(h_start, n_points, max_iters=3000, time_limit=300):
    """Projected gradient ascent to maximize minimal autocorrelation."""
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_current = h_start.copy()
    h_current = project_to_feasible(h_current, target_sum)
    
    best_h = h_current.copy()
    best_C5 = evaluate_C5(h_current, dx)
    
    # Multi-resolution schedule
    res_levels = [(150, 800), (300, 400), (500, 300)]
    current_res = 0
    ns = [150, 300, 500]
    stepsizes = [0.5, 0.1, 0.01]
    max_iters_per_level = []
    for i, (n, iters) in enumerate(res_levels):
        if n == n_points:
            current_res = i
            max_iters_per_level = [iters]
            break
    if not max_iters_per_level:
        max_iters_per_level = [iters for _, iters in res_levels]
    
    stepsize = stepsizes[current_res]
    loss_history = []
    
    start_time = time.time()
    for iteration in range(max_iters):
        # Check if time is exceeded
        if time.time() - start_time > time_limit:
            break
            
        # Compute all autocorrelations via fft
        h_fft = fft.fft(h_current)
        autocorr = fft.ifft(np.conj(h_fft) * h_fft).real[:n_points]
        
        # Find minimal autocorrelation index
        k_min = np.argmin(autocorr)
        
        # Gradient descent step
        g = 2 * np.roll(h_current, k_min)
        h_current = h_current + stepsize * g

        # Projection
        h_current = project_to_feasible(h_current, target_sum)

        # Evaluate loss
        C5_current = evaluate_C5(h_current, dx)
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h_current.copy()
        
        loss_history.append(C5_current)
        
        # Early stopping
        if len(loss_history) >= 200 and abs(loss_history[-1] - loss_history[-200]) < 1e-6:
            break
            
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Stage 1: Block searches
    h1, C5_1 = block_family_search()
    h2, C5_2 = four_block_search()
    
    if h1 is not None and h2 is not None:
        h_start = h1 if C5_1 < C5_2 else h2
        C5_start = min(C5_1, C5_2)
    elif h1 is not None:
        h_start = h1
        C5_start = C5_1
    else:
        h_start = h2
        C5_start = C5_2
    
    # Stage 2: Stochastic search
    if time.time() - t0 < budget_s - 150:
        h_stoch, C5_stoch = stochastic_search()
        if C5_stoch < C5_start:
            h_start = h_stoch
            C5_start = C5_stoch
        t0 = time.time()
    
    # Stage 3: Gradient ascent
    if time.time() - t0 < budget_s - 50:
        # Find resolution that fits our design
        h_final, C5_final = gradient_ascent(h_start, 300, max_iters=2500, time_limit=budget_s - (time.time() - t0))
    else:
        h_final = h_start
        C5_final = C5_start
    
    return list(h_final), C5_final, len(h_final)
