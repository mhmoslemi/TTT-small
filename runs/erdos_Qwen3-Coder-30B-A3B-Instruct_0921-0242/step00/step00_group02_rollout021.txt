# Helper functions for interval-based optimization
def compute_overlap_intervals(a, ell, k_vals):
    """Compute overlap function O(k) for given interval centers and shifts."""
    m = len(a)
    O_vals = np.zeros_like(k_vals)
    for i in range(m):
        for j in range(m):
            # tent function contribution from intervals i and j
            diff = a[i] - a[j]
            for k_idx, k in enumerate(k_vals):
                dist = abs(diff - k)
                if dist <= ell:
                    O_vals[k_idx] += ell - dist
    return O_vals

def generate_breakpoints(a, ell):
    """Generate all breakpoints for O(k)"""
    m = len(a)
    breakpoints = set()
    for i in range(m):
        for j in range(m):
            diff = a[i] - a[j]
            breakpoints.add(diff - ell)
            breakpoints.add(diff + ell)
    # Filter to [0, 2]
    valid = [k for k in breakpoints if 0 <= k <= 2]
    return sorted(set(valid + [0, 2]))

def evaluate_O(a, ell):
    """Evaluate maximum overlap for given interval pattern."""
    breakpoints = generate_breakpoints(a, ell)
    if not breakpoints:
        return 0.0

    k_vals = np.array(breakpoints)
    O_vals = compute_overlap_intervals(a, ell, k_vals)
    return np.max(O_vals)

def project_to_feasible(a):
    """Project given interval positions onto feasible region with ordering constraints."""
    a = np.array(a)
    m = len(a)
    if m <= 1:
        return a if 0 <= a[0] <= 1 else np.array([0.0])
    # Ensure gaps are non-negative and total length is 1
    # Start with ordered positions
    a = np.sort(a)
    # Try to adjust the positions
    ell = 1.0 / m
    minsep = ell
    for i in range(m - 1):
        gap = a[i+1] - a[i] - ell
        if gap < 0:
            # Move all subsequent intervals to the right
            adjustment = -gap
            a[i+1:] += adjustment
    # Ensure fit in [0, 2]
    offset = max(0, a[0])
    a -= offset
    if a[-1] + ell > 2:
        adjustment = (a[-1] + ell - 2)
        a -= adjustment
    # Final check
    if a[-1] + ell > 2:
        # Fallback adjustment
        a[-1] = 2 - ell
    # Ensure position constraints still hold
    for i in range(m - 1):
        if a[i+1] < a[i] + ell:
            a[i+1] = a[i] + ell
    return a

def solve_m_intervals(m, t0, budget_s):
    """Solve for m intervals using a two-phase approach."""

    # Start with a reasonable initial guess
    ell = 1.0 / m
    initial_a = np.linspace(0, 2 - ell, m)
    best_a = initial_a.copy()
    best_val = 1000.0
    best_seed_a = None
    best_seed_val = 1000.0

    def objective(a_vec):
        # Project to feasible and compute overlap
        a_proj = project_to_feasible(a_vec)
        val = evaluate_O(a_proj, ell)
        return val

    # Run DE optimization
    bounds = [(0, 2 - ell)] * m
    try:

        result = differential_evolution(objective, bounds, maxiter=maxiter, seed=42)

def discretize_vector(a, ell, n_points):
    """Convert interval pattern to binary approximation."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        for j in range(len(a)):
            if a[j] <= x <= a[j] + ell:
                h[i] = 1
                break
    # Adjust for integral constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) < 0.5:
        if abs(current_sum - target_sum) > 1e-6:
            diff = int(round(target_sum - current_sum))
            # Perform small corrections
            if diff > 0:
                # Turn some 0s into 1s
                for i in range(n_points):

                        # Check if it's close to a boundary
                        found = False

                                h[i] = 1
                                diff -= 1
                                found = True
                                break
                        if not found:
                            h[i] = 1
                            diff -= 1
                        if diff <= 0:
                            break
            else:
                # Turn some 1s into 0s
                for i in range(n_points):
                    if h[i] == 1:
                        x = i * dx
                        # Check if it's close to a boundary
                        found = False
                        for j in range(len(a)):
                            if abs(x - a[j]) < dx * 0.5 or abs(x - (a[j] + ell)) < dx * 0.5:
                                h[i] = 0
                                diff += 1
                                found = True
                                break
                        if not found:
                            h[i] = 0
                            diff += 1
                        if diff >= 0:
                            break
    # Ensure constraint
    current_sum = np.sum(h)
    if not abs(current_sum - target_sum) < 1.0:
        # Resort to greedy flip near boundaries
        distances = []
        for i in range(n_points):
            x = i * dx
            dist = float('inf')

                d1 = abs(x - a[j])
                d2 = abs(x - (a[j] + ell))
                dist = min(dist, d1, d2)
            distances.append((dist, i))
        distances.sort()
        needed = int(round(target_sum - current_sum))
        if needed > 0:
            for _, i in distances[:needed]:
                h[i] = 1
        elif needed < 0:
            for _, i in distances[:abs(needed)]:
                h[i] = 0
    return h

def binary_refinement(h, n_points):
    """Perform local swaps to reduce max overlap."""
    h = np.array(h, dtype=int)
    # Compute correlations
    corr = np.correlate(h, h, mode='full')
    corr = corr[len(corr)//2:]  # Only positive lags
    # Find critical k
    k_star = np.argmin(corr)
    # Early termination check
    if k_star == 0:
        k_star = 1
    # Enumerate swaps
    best_gain = 0
    best_i, best_j = -1, -1
    n_swaps = 0
    max_swaps = 200
    for _ in range(max_swaps):
        # Random pair of indices (i,j)
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue
        # Check if they are valid for swapping
        if (h[i] == 0) == (h[j] == 1):
            continue
        # Compute change in kernel (use cheaper update)
        delta = 0
        # Adjustments for safe integer math
        if i + k_star < n_points and h[i + k_star]:
            delta -= 1
        if j + k_star < n_points and h[j + k_star]:
            delta += 1
        if i >= k_star and h[i - k_star]:
            delta -= 1
        if j >= k_star and h[j - k_star]:
            delta += 1
        # Apply inverse logic: we really want to minimize corr
        gain = delta  # inverse sign for gain in optimization
        if gain > best_gain:
            best_gain = gain
            best_i, best_j = i, j
        n_swaps += 1
        if n_swaps > 500:  # Prevent unfair long loops
            break
    if best_gain > 0 and best_i >= 0:
        # Perform the swap
        h[best_i], h[best_j] = h[best_j], h[best_i]
    # Return back to float range
    return h.astype(float)
