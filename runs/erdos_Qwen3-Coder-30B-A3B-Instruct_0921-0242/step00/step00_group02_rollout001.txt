import numpy as np
import time

def project_simplex(h, target_sum):
    """Project vector h onto the simplex with given target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-12:
        return h

    free = np.where((h > 0) & (h < 1))[0]
    if len(free) == 0:
        return h

    free_sorted = np.argsort(-h[free])
    free_order = free[free_sorted]
    free_list = list(free_order)
    
    # Sort lengths for in-order processing, and use a list as a set:
    # keep the set of indices with 0 < h[i] < 1
    F = set(free_order)
    
    while abs(diff) > 1e-12 and len(F) > 0:
        while len(free_list) > 0 and free_list[-1] not in F:
            free_list.pop()
        
        if len(free_list) == 0:
            break
            
        step = diff / len(F)
        for i in list(F):
            h[i] += step
            if h[i] >= 1:
                h[i] = 1
                F.discard(i)
            elif h[i] <= 0:
                h[i] = 0
                F.discard(i)
        diff = target_sum - np.sum(h)
    
    return h


def soft_min_gradient(r, w, alpha):
    """Compute gradient of soft minimum via convolution."""
    w = np.array(w, dtype=np.float64)
    g = np.zeros_like(r)
    for k in range(len(r)):
        g += w[k] * np.concatenate((r[k:], [0] * k))
    return g


def corr_at_shift(h, k):
    """Compute correlation r_k(h) for shift k."""
    n = len(h)
    if k >= n:
        return 0.0
    return sum(h[i] * h[i + k] for i in range(n - k))


def get_autocorrelations(h):
    """Compute linear autocorrelations r_k for k>=1."""
    n = len(h)
    h_padded = np.pad(h, (0, n), 'constant')
    H = fft(h_padded)
    r = np.real(ifft(H * np.conj(H)))
    return r[1:n]


def compute_objective(h, dx):
    """Compute C5 from h."""
    r = get_autocorrelations(h)
    min_r = np.min(r)
    return 1.0 - min_r * dx


def line_search(h, grad, func, alpha, max_iter=10):
    """Simple backtracking line search. Returns optimal step size."""
    # Function values
    f_prev = func(h)
    step_size = 1.0
    eta = 0.5
    c = 1e-4
    
    # Gradient descent
    for _ in range(max_iter):
        h_new = h - step_size * grad
        h_new = project_simplex(h_new, len(h) / 2.0)
        f_new = func(h_new)
        if f_new <= f_prev - c * step_size * np.dot(grad, grad):
            return step_size
        step_size *= eta
    return step_size


def optimize_single_h(h0, alpha, max_iter, dx):

    h = h0.copy()
    best_h = h.copy()
    best_val = compute_objective(h, dx)
    
    # Initial gradient
    r = get_autocorrelations(h)
    r = list(r)
    
    w = [exp(-alpha * r_k) for r_k in r]
    total_w = sum(w)
    if total_w > 0:
        w = [wi / total_w for wi in w]
    else:
        w = [1.0 / len(w)] * len(w)
    
    prev_h = h.copy()
    y = h.copy()
    beta = 0.0
    
    for iter_num in range(max_iter):
        # Update y
        if iter_num > 0:
            beta = (iter_num - 1) / (iter_num + 2)
            y = h + beta * (h - prev_h)
        else:
            y = h.copy()
        
        # Compute gradient w.r.t. soft min
        g = soft_min_gradient(r, w, alpha)
        
        # Line search
        step_size = line_search(y, g, lambda x: compute_objective(
            project_simplex(x, len(x)/2), dx), alpha)
        h = project_simplex(y - step_size * g, len(h) / 2)
        
        # Update variables
        prev_h = h0
        h0 = h.copy()
        
        # Update r and w
        r = get_autocorrelations(h)
        r = list(r)
        
        if abs(np.max(r)) < 1e-12:
            break
        
        w = [exp(-alpha * r_k) for r_k in r]
        total_w = sum(w)
        if total_w > 0:
            w = [wi / total_w for wi in w]
        else:
            w = [1.0 / len(w)] * len(w)
        
        # Update best
        f_new = compute_objective(h, dx)
        if f_new < best_val:
            best_h = h.copy()
            best_val = f_new
    
    return best_h, best_val
