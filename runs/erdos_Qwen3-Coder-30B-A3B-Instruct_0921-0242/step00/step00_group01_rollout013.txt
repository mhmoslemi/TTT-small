import numpy as np
import time
from scipy import signal

def autocorr(b):
    """Compute cyclic autocorrelation of binary sequence b"""
    n = len(b)
    # FFT-based computation
    B = np.fft.fft(b)
    A = np.fft.ifft(B * np.conj(B)).real
    return A

def evaluate_c5(h):
    """Evaluate C5 for a given h using np.correlate"""
    n = len(h)
    dx = 2.0 / n
    # Using full mode to get all lags
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the overlapping part (shifts from -n+1 to n-1)
    mid = len(corr) // 2
    overlaps = corr[mid - n + 1 : mid + n - 1]
    # We want maximum of ∫ h(x)(1-h(x+k)) dx
    max_overlap = np.max(overlaps) * dx
    return max_overlap

def evaluate_c5_with_constraints(h, dx=2.0/47):
    """Evaluate C5 with proper constraints"""
    # Note: already have constraints h_i ∈ [0,1]
    # ∫ h dx = 1, so sum(h)*dx = 1 → sum(h) = 1/dx = 47/2 = 23.5 
    # This is implicit in the construction
    # We'll use correlate as needed
    corr = np.correlate(h, 1 - h, mode="full")
    shifts = len(corr) // 2
    overlaps = corr[shifts - len(h) + 1 : shifts + len(h) - 1]
    return np.max(overlaps) * dx

# Here we define a function to generate binary seeds from known constructions
def legendre_sequence(n):
    """Generate a Legendre sequence of length n (truncated from a prime length)"""
    # Use prime p such that p ≥ n and p is odd
    p = n
    while True:
        if p > n and p % 2 == 1 and is_prime(p):
            break
        p += 2
    # Generate Legendre sequence of size p
    s = np.zeros(p, dtype=int)
    for i in range(p):
        s[i] = 1 if quadratic_residue(i, p) else 0
    # Truncate or pad to n
    if p > n:
        s = s[:n]
    else:
        s = np.pad(s, (0, n-p), constant_values=0)
    # Balance it: make near equal 0s and 1s
    ones = np.sum(s)
    if ones > n//2:
        # Flip some 1s to 0s
        flip_indices = np.where(s == 1)[0][:ones - n//2]
        s[flip_indices] = 0
    elif ones < n//2:
        # Flip some 0s to 1s
        flip_indices = np.where(s == 0)[0][:n//2 - ones]
        s[flip_indices] = 1
    return s

def is_prime(p):
    if p < 2:
        return False
    if p == 2:
        return True
    if p % 2 == 0:
        return False
    for i in range(3, int(p**0.5)+1, 2):
        if p % i == 0:
            return False
    return True

def quadratic_residue(a, p):
    if a == 0:
        return True
    return pow(a, (p-1)//2, p) == 1

def singer_diff_set(n):
    """Generate Singer difference set for appropriate n = q^2 + q + 1"""
    q = 1
    while True:
        candidate = q*q + q + 1
        if candidate == n:
            return _singer_ds(q)
        elif candidate > n:
            break
        q += 1
    return None  # Not suitable length

def _singer_ds(q):
    # Standard construction for Singer set of order q
    # We generate a sequence of 0s and 1s with perfect difference set properties
    # Just to get a good head start, this is an approximation
    # For the rigorous implementation, it's actual construction
    # Here we provide a simpler approach
    n = q*q + q + 1
    b = np.zeros(n, dtype=int)
    # Place ones according to a known pattern for difference set
    # This is an approximation for a real construction
    for i in range(n):
        b[i] = 0
    # Take a segment, make a shift to make it balanced, simplest
    # Here we use a heuristic method to make a good starting binary vector
    # Not quite the mathematical design but serves as reasonable seed
    # Other future work could refine this more
    # Return with 1 in n//2 locations
    idxs = np.random.choice(n, size=n//2, replace=False)
    b[idxs] = 1
    # Randomly skip some redundant logic
    return b

def flip_bad_positions(b, n):
    """Make sure b is balanced"""
    ones = np.sum(b)
    if ones > n//2:
        z = np.where(b == 1)[0][:ones - n//2]
        b[z] = 0
    elif ones < n//2:
        z = np.where(b == 0)[0][:n//2 - ones]
        b[z] = 1
    return b

def improve_binary_autocorr(b, swaps=30000):
    """Improve a binary sequence through local swaps"""
    n = len(b)
    # Precompute B_shift[k][i] = b[(i+k)%n] (array of shifted versions)
    B_shift = np.zeros((n, n), dtype=int)
    for k in range(n):
        B_shift[k] = np.roll(b, k)
    
    # Compute initial autocorrelations
    A = autocorr(b)
    m = np.min(A[1:])  # Min non-lag-0
    best_m = m
    best_b = b.copy()
    
    # Track last improvement
    last_improve = 0
    it = 0
    
    # Swap Kernel: compute change in A_k of (i,j) swap
    def delta(k, i, j):
        return -B_shift[k][i] + B_shift[k][j]
    
    # Set temperature for simulated annealing
    temp = 1.0
    decay = 0.999
    
    # Allow 30k swaps or until no improvement for a while
    while it < swaps and last_improve < 5000:
        # Get indices of 1s and 0s
        one_indices = np.where(b == 1)[0]
        zero_indices = np.where(b == 0)[0]
        
        if len(one_indices) == 0 or len(zero_indices) == 0:
            break
            
        i = np.random.choice(one_indices) 
        j = np.random.choice(zero_indices)
        
        # Find current worst k
        k_star = np.argmin(A[1:]) + 1
        delta_m = delta(k_star, i, j)
        
        if delta_m > 0:  # Improving swap
            # Perform swap
            b[i] = 0
            b[j] = 1
            # Update A incrementally
            for k in range(n):
                if k != k_star:
                    A[k] += delta(k, i, j)
                A[k_star] += delta_m  # Must be updated separately
            
            m = np.min(A[1:])
            if m > best_m:
                best_m = m
                best_b = b.copy()
                last_improve = 0  # Reset
        else:
            # Accept based on probability
            if np.random.rand() < np.exp(delta_m / temp):
                b[i] = 0
                b[j] = 1
                for k in range(n):
                    if k != k_star:
                        A[k] += delta(k, i, j)
                    A[k_star] += delta_m
                m = np.min(A[1:])
        it += 1
        last_improve += 1
        temp *= decay
    
    return best_b, best_m

def inflate_binary(b, new_len):
    """Duplicate each element to scale up to new_len"""
    if new_len % len(b) != 0:
        raise ValueError("new_len must be multiple of len(b)")
    return np.repeat(b, new_len // len(b))

def generate_initial_h_values(n):
    """Generate good starting construction as a seed,
    probably customized version of initial_h_values 
    """
    # We will start with a Legendre sequence like the suggestion
    # Need to generate a good seed for n points over [0,2]
    h = np.zeros(n)
    dx = 2.0 / n
    
    # Use a marked Legendre or Singer approach
    # This in practice reused code but simplified here...
    # Generate slightly different approach
    # Approx simple version
    b = legendre_sequence(n)
    b = flip_bad_positions(b, n)
    
    # Convert this binary back to h(x) 
    h = b.astype(float)
    # Now rescale so sum(h)*dx = 1
    if sum(h) * dx != 1.0:
        h = h * (1.0 / (sum(h) * dx))
    return h

        target_len = min(350, best_n*2)

        sum_h = np.sum(h)
        h = h * (1.0 / (sum_h * 2.0 / target_len))

        # Just a sanity check - don't do too much
        for attempt in range(1000):
            i1, i2 = np.random.randint(0, len(h), 2)
            if h[i1] > 0 and h[i2] < 1:

                orig_sum = np.sum(h)
                delta = min(h[i1], 1.0 - h[i2])
                if delta > 0:
                    h[i1] -= delta
                    h[i2] += delta
                # Renormalize to preserve sum
                new_sum = np.sum(h) 
                if abs(new_sum - orig_sum) > 1e-10:
                    h *= (1.0 / new_sum)
        
        # Final check
        dx2 = 2.0 / len(h)
        eval_C5 = evaluate_c5_with_constraints(h, dx2)

    if best_C5 >= 1e9:
        # Fallback to simplest one

        # Note sum is (47/2) * (2/47) = 1 
        best_h = h_simple
        best_C5 = evaluate_c5(h_simple)
        best_n = 47
    
    return (best_h, best_C5, best_n)
