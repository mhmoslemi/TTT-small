import numpy as np
import time
import math

def project_box_affine(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Project onto affine constraint
    delta = (np.sum(h) - target_sum) / len(h)
    h -= delta
    h = np.clip(h, 0, 1)
    return h

def autocorr(h):
    # Circular autocorrelation using FFT
    H = np.fft.fft(h)
    autocorr_vals = np.real(np.fft.ifft(H * np.conj(H)))
    return autocorr_vals

def compute_C5(h, dx):
    # Use numpy correlate with full mode
    # C5 = max_k int h(x)(1 - h(x+k)) dx = max_k (int h(x) dx - int h(x)h(x+k) dx)
    # = max_k (1 - int h(x) h(x+k) dx) = max_k (1 - overlap_k)
    # This is max_k dx * (1 - (h conv h)_k) = max_k dx * (1 - autocorr_k) 
    # But we use: max_k (dx * (E[h] - autocorr_k))
    # E[h] = 1 since integral = 1
    # So max_k (dx * (1 - autocorr_k)) = dx * max_k (1 - autocorr_k) = dx * (1 - min_k autocorr_k)
    # But that's not quite right with our overlap definition 
    # Actually it's: max_k int h(x)(1 - h(x+k)) dx  
    # Which equals: max_k ( int h(x) dx - int h(x) h(x+k) dx ) = max_k (1 - int h(x) h(x+k) dx)
    # So we want max_k dx * (1 - autocorr_k) where autocorr_k = (h conv h)_k
    # Therefore we want to minimize the maximum of (dx * (1 - autocorr_k))
    # Which equals minimizing max_k (1 - autocorr_k) = minimizing min_k autocorr_k 
    aut = autocorr(h)
    # More directly and correctly: 
    # overlap_k = dx * (sum(h) - sum(h_i * h_{i+k})) 
    # This is dx * (1 - autocorr_k)
    # And we want to minimize max (dx * (1 - autocorr_k)) = dx * max (1 - autocorr_k)  
    # Which is equivalent to minimizing min autocorr_k
    # Now, if the interval h(x) is not a simple constant, this isn't the right formula either.
    # So, use the direct approach via np.correlate:
    n = len(h)
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    # The last n-1 elements are the valid shifts for k = 1..n-1
    # But the problem is that using a continuous formulation, we should include 
    # all shifts up to n-1, but we're looking at a discrete shift k = 0..n-1
    # We only take the last n-1 elements (for shifts 1 to n-1) in our max
    # For k=0, overlap = dx * (sum(h) - sum(h_i * h_{i+0})) = dx * (sum(h) - sum(h)) = 0
    # But we consider periods k=0.1..n-1  
    # Let's be precise about the problem:
    # C5 = max_k int h(x)(1-h(x+k)) dx for k in [0,2] 
    # The equivalent discrete case is to evaluate all k such that we consider 
    # the whole shift k*2/n over the discrete grid.  
    # But let us define overlaps correctly from full convolution
    overlaps = overlaps[len(h)-1:]  # take only k=1,2,...,n-1
    if len(overlaps) > 0:
        return np.max(overlaps)
    else:
        return 0.0

def evaluate_candidate(h, n_points):
    dx = 2.0 / n_points
    C5 = compute_C5(h, dx)
    return C5

def gradient_ascent_step(h, dx, step_size=0.01):
    # Autocorrelation
    autocorr_vals = autocorr(h)
    # Since we want to maximize minimum autocorrelation, we take negative.
    # minimize max -autocorr_vals,  which means minimize max (autocorr_vals)
    # We will use descent in Autocorr for modern viewpoint and match code below.
    # We will do gradient ascent on min_k autocorr_k
    # Which means ascent on -max_k -autocorr_k = max_k autocorr_k as the wrong sign
    # If we think of objective as max_k (dx*(1 - autocorr_k)), 
    # then the subgradient for the k that produces max is (here for k=0 at least example)
    # we set gradient as dO/dh_j = dx *(1 - derivative of h_i h_{i+k} w.r.t. h_j)
    # Derivative of h_i * h_{i+k} w.r.t. h_j is h_{i+k} if j==i else h_{i-k} if j==i+k else 0.
    # So for k=0: dh_i * h_{i+0}/dh_j = h_i if j==i+k=0, i.e. j==i.  
    # n.b. It's easier just to project onto the gradient of the maximum from the cost function.
    #     We minimize max (1 - autocorr_k)  == minimize max -autocorr_k == maximize min autocorr_k
    # The subgradient of max_k (-autocorr_k) is subgradient of -autocorr_{k*}
    # where k* = argmin autocorr_k. 
    # Gradient of -autocorr_{k*} w.r.t. h is: -d/dautocorr_{k*} dh_j = - (h_{j + k*} + h_{j - k*})  
    # Since pos_k = argmin_autocorr -->
    # grad = - (h + roll(h, k*)) - (h + roll(h, -k*)) = -(2 * h + roll(h, k*) + roll(h, -k*))
    # But we compute derivative correctly using: 
    # I think the hand-crafted gradient of the subgradient of aut[lag] 
    # w.r.t. h[j] is   roll(h, lag)[j] + roll(h, -lag)[j] = h[j+lag] + h[j-lag]  
    # We want to ascent wrt to min_k autocorr_k
    k_min = np.argmin(autocorr_vals)
    grad = - (np.roll(h, k_min) + np.roll(h, -k_min))  # because gradient of a_i * a_{i+k} w.r.t. a_j is a_{j+k} if j==i else a_{j-k} if j==i+k, etc.
    
    # Ascent
    h_new = h + step_size * grad
    # Project back to feasible region
    h_new = project_box_affine(h_new, len(h) / 2)
    return h_new
