def evaluate(h):
    """Compute C₅ = max_k ∫ h(x)(1-h(x+k)) dx"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, target):
    """Project a real vector x onto [0,1] with sum = target."""
    h = np.clip(x, 0, 1)
    current_sum = np.sum(h)
    diff = target - current_sum
    if abs(diff) < 1e-12:
        return h
    # Iteratively adjust to hit target sum
    # Start with the largest entries to add mass to
    sorted_idx = np.argsort(-h)
    for i in range(10):  # Precision limit
        if abs(diff) < 1e-12:
            break
        adjust = min(diff, 1 - h[sorted_idx[0]])
        if abs(adjust) < 1e-12:
            break
        h[sorted_idx[0]] += adjust
        diff -= adjust
        sorted_idx = sorted_idx[1:]
    return h

def active_shifts(h, eps=1e-9):
    """Identify active shifts k where overlap is within eps of max."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    # Real indices of the correlations (from -n+1 to n-1)
    ks = np.arange(-len(h) + 1, len(h), dtype=int)  # size 2n-1
    active = ks[np.abs(corr - max_corr) <= eps]
    return active

def gradient(h, ks):
    """Computes subgradient of the max-overlap at h with respect to k-active shifts."""
    # Length of h
    n = len(h)
    g = np.zeros(n)
    dx = 2.0 / n
    # Consider all fixed indices i and corresponding shifts k in ks
    for i in range(n):
        temp = 0
        count = 0
        for k in ks:
            if k >= 0:
                j = i + k
                if 0 <= j < n:
                    temp += (1 - h[j])
                    count += 1
            else:
                j = i + k
                if 0 <= j < n:
                    temp += h[j]
                    count += 1
        if count > 0:
            g[i] = temp / count
    return g

def line_search(h, g, t_current, budget):
    # Evaluate g^T h
    # but also test step sizes
    step_sizes = [0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1]
    best_h = h
    best_eval = t_current
    dx = 2.0 / len(h)
    for alpha in step_sizes:
        if alpha == 0:
            test_h = h.copy()
        else:
            test_h = project_feasible(h - alpha * g, 1.0)
        eval_val = evaluate(test_h)
        if eval_val < best_eval:
            best_eval = eval_val
            best_h = test_h
    return best_h, best_eval

def local_optimize(h0, max_iter, t_end):
    """Projected gradient descent with line search."""
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate(h)
    dx = 2.0 / len(h)
    elapsed = time.time() - t_end + 600  # Hack for first step
    t0 = time.time()
    for i in range(max_iter):
        if time.time() - t0 > 580:
            break
        eps = 1e-9
        ak = active_shifts(h, eps=eps)
        grad = gradient(h, ak)
        h_new, eval_new = line_search(h, grad, evaluate(h), t_end)
        if eval_new < best_val:
            best_val = eval_new
            best_h = h_new
        h = h_new
    return best_h, best_val

def anneal(h0, t_end, T0=0.01, decay=0.995):
    """Simulated Annealing for escaping local minima."""
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate(h)
    T = T0
    dx = 2.0 / len(h)
    t0 = time.time()
    steps = 0
    while steps < 5000 and time.time() - t0 < 595:
        i, j = np.random.choice(len(h), size=2, replace=False)
        delta = np.random.rand() * 0.02
        delta = delta if np.random.rand() > 0.5 else -delta
        old_h_i = h[i]
        old_h_j = h[j]
        # New values
        new_h_i = old_h_i + delta
        new_h_j = old_h_j - delta
        # Project to feasible set
        test_h = h.copy()
        test_h[i] = new_h_i
        test_h[j] = new_h_j
        test_h = project_feasible(test_h, 1.0)
        new_val = evaluate(test_h)
        if new_val < best_val:
            best_h = test_h.copy()
            best_val = new_val
        if new_val < evaluate(h):
            # Accept
            h = test_h
        else:
            # Maybe accept with probability
            prob = np.exp(- (new_val - evaluate(h)) / T)
            if np.random.rand() < prob:
                h = test_h
        T *= decay
        steps += 1
    return best_h, best_val

def random_feasible(n):
    """Generate a random feasible h in [0,1] with integral=1."""
    x = np.random.rand(n)
    # Project to sum = 1
    x = project_feasible(x, 1.0)
    return x

def interpolate_and_project(h_old, n_new):
    """Interpolate h_old to length n_new, then project."""
    old_len = len(h_old)
    # Linear interpolation using numpy's interp
    x_old = np.linspace(0, 2, old_len)
    x_new = np.linspace(0, 2, n_new)
    h_interp = np.interp(x_new, x_old, h_old)
    # Project to feasible bounds and sum = 1
    h_new = project_feasible(h_interp, 1.0)
    return h_new
