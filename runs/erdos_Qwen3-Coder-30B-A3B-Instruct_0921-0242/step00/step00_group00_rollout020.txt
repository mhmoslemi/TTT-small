import numpy as np
import time
from scipy import signal
from math import sqrt

# Helper function to project a vector onto the box-constrained simplex
def project_simplex_box(v, total, lo=0., hi=1.):
    """Project v onto {x : lo <= x_i <= hi, sum(x) = total}"""
    v = np.array(v)
    n = len(v)
    
    # First, clamp to [lo, hi]
    v = np.clip(v, lo, hi)
    s = np.sum(v)
    
    # If already feasible
    if abs(s - total) < 1e-9:
        return v
    
    # Otherwise, adjust by a constant
    if s > total:
        # Reduce by constant
        diff = s - total
        # Distribute reduction
        v -= diff / n
        np.clip(v, lo, hi, out=v)
    else:
        # Increase by constant
        diff = total - s
        # Distribute increase
        v += diff / n
        np.clip(v, lo, hi, out=v)
    
    return v

# Efficient computation of C5 using FFT
def compute_c5(h, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx using FFT."""
    h = np.array(h)
    h1 = 1 - h
    # Full correlation (equivalent to cross-correlation of h with 1-h)
    corr = signal.correlate(h, h1, mode='full')
    # Take real part (can be slightly complex due to FFT roundoff)
    corr = np.real(corr)
    # Normalize by dx, then take max over all lags
    max_val = np.max(corr) * dx
    return max_val

# Compute the index of worst lag and its correlation value
def worst_lag(h):
    """Find the lag k that maximizes sum h_i * h_{i+k}."""
    h = np.array(h)
    h1 = 1 - h
    corr = signal.correlate(h, h1, mode='full')
    corr = np.real(corr)
    k_max = np.argmax(corr)
    return k_max, corr[k_max]

# Compute gradient of overlap at lag k w.r.t. h
def gradient_worst(h, k):
    """Gradient of sum_{i} h_i * h_{i+k} w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    # i -> i + k, i -> i - k
    # Circular indices
    for i in range(n):
        j1 = (i + k) % n
        j2 = (i - k) % n
        grad[i] = h[j1] + h[j2]
    return grad

def optimize(n, time_limit, seed=42):
    np.random.seed(seed)
    start_time = time.time()
    
    dx = 2.0 / n
    
    # Initialize from initial_h_values if n matches
    h = initial_h_values if len(initial_h_values) == n else np.random.dirichlet(np.ones(n)) * 2
    h = np.clip(h, 0, 1)
    h = project_simplex_box(h, n / 2)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    alpha = 0.01  # step size
    step_count = 0
    max_steps = int(time_limit * 2000 / 600) if time_limit > 0 else 2000
    
    for iteration in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        
        k_worst, _ = worst_lag(h)
        
        grad = gradient_worst(h, k_worst) 
        # Gradient descent direction to reduce overlap (negative)
        h_new = h - alpha * grad
        h_new = project_simplex_box(h_new, n / 2)
        
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Accept the update if it improves
        if c5_new <= compute_c5(h, dx):
            h = h_new.copy()
            step_count += 1
        else:
            # Sometimes, accept step with reduced alpha
            alpha *= 0.95
            if alpha < 1e-6:
                alpha = 1e-6
        
        if step_count % 50 == 0:
            if time.time() - start_time > time_limit * 0.95:
                break
    
    return best_h, best_c5, n

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    n_list = [150, 200, 250, 300, 350, 400]
    
    for n in n_list:
        if time.time() - start_time > budget_s * 0.9:
            break
        time_left = budget_s - (time.time() - start_time)
        # Limit time for each n
        time_per_n = min(time_left * 0.3, 150.0)  # at most 150s per n
        
        h, c5, _ = optimize(n, time_per_n, seed=seed)
        
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n = n
    
    # Final refinement at best resolution
    if best_n is not None:
        h_refine, c5_refine, _ = optimize(best_n, 100.0, seed=seed)
        if c5_refine < best_c5:
            best_h = h_refine
            best_c5 = c5_refine
            
    dx = 2.0 / len(best_h)
    final_c5 = compute_c5(best_h, dx)
    
    return list(best_h), final_c5, len(best_h)
