def solve_sdp(n, start_time, budget_s):
    """
    Solve the SDP relaxation for a given n.
    """
    # Time limit for SDP solving
    sdp_time = min(30, budget_s - (time.time() - start_time) - 5)
    
    # Variable matrix dimensions
    m = n
    
    # Define optimization variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    s = cp.Variable()
    
    # Constraints based on SDP relaxation
    constraints = [
        cp.diag(X) == h,
        X >> 0,  # Positive semidefinite
        0 <= h, h <= 1,
        cp.sum(h) == n / 2
    ]
    
    # Linear constraints for correlation bounds
    # We will try to maximize s such that for all k=1..n-1, sum of off-diagonal terms >= s
    # This is equivalent to Phi_k(X) >= s for all k
    
    max_k = n - 1
    for k in range(1, max_k + 1):
        # Sum of diagonal elements of shifted matrix
        expr = sum(X[i, i+k] for i in range(n-k))
        constraints.append(expr >= s)
        
    # Objective
    objective = cp.Maximize(s)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with SCS or MOSEK if available
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, max_iters=2000, scale=10, verbose=False)
    except:
        # Fall back to an approximate solution
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            # Final resort - create a very rough initial solution
            h_val = np.ones(n) / 2
            h_val[0] = 0.01
            h_val[1] = 0.99
            h_val[2] = 0.99
            h_val[3] = 0.01
            h_val[4] = 0.01
            h_val[5] = 0.99
            h_val[6] = 0.99
            h_val[7] = 0.01
            h_val[8] = 0.01
            h_val[9] = 0.99
            h_val[10] = 1.0
            if len(h_val) > 10:
                h_val[10:] = 0.0
            h_val = h_val[:n]
            h_val = h_val * (n / 2) / np.sum(h_val)
            c5 = compute_c5(h_val)
            return h_val, c5, n

    # Extract solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = h.value
        # For small n five draws for best result
        best_h = None
        best_c5 = float('inf')
        
        # Rounding: Use 300 samples instead of many as round trip is safe
        for _ in range(300):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Generate random Gaussian vector for rounding
            g = np.random.normal(0, 1, n)
            z = h_sol  # Simplified approach
            # Try several quantile steps to sample best
            z_ab = np.argsort(z)
            l = n//2
            
            # Sample uniformly from lower half
            tau = z[z_ab[n-l]]
            
            # Thresholding
            h_sampled = (z >= tau).astype(float)
            
            # Ensure mass constraint
            if np.sum(h_sampled) > n / 2:
                # Pick entries that are too high and set to zero
                to_set_zero = np.where(h_sampled == 1)[0]
                to_set_zero = to_set_zero[np.argsort(z[to_set_zero])]
                for i in to_set_zero[n//2:]:
                    h_sampled[i] = 0
            elif np.sum(h_sampled) < n / 2:
                to_set_one = np.where(h_sampled == 0)[0]
                to_set_one = to_set_one[np.argsort(z[to_set_one])]
                for i in to_set_one[n//2:]:
                    h_sampled[i] = 1
                    
            c5 = compute_c5(h_sampled)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_sampled
                
        if best_h is not None:
            return best_h, best_c5, n
    
    # On failure return either:
    # 1. The fractional initial h(0.5) (just to progress)
    h_val = np.ones(n) / 2
    h_val = h_val * (n / 2) / np.sum(h_val)
    c5 = compute_c5(h_val)
    return h_val, c5, n


def compute_c5(h):
    """
    Compute the overlap integral for step function h
    """
    n = len(h)
    dx = 2.0 / n
    h = np.array(h)
    # Compute convolution of h and (1-h) using correlation with full mode
    conv = np.correlate(h, 1-h, mode="full")
    max_correlation = np.max(conv)
    c5 = max_correlation * dx
    return c5


def refine_fractional(h_original, n, start_time, budget_s):
    """
    Refine solution using gradient ascent
    """
    def obj(h):
        # Minimization objective is -min_k R_k(h)
        dx = 2.0 / n
        h_array = np.array(h)
        # Convolution part
        conv = np.correlate(h_array, 1-h_array, mode="full")
        max_correlation = np.max(conv)
        return -max_correlation * dx

    def grad(h):
        # Approximate gradient of max correlation
        h_array = np.array(h)
        dx = 2.0 / n
        # This is rather heuristical for now - simple centered difference ∇
        # We're doing a rough gradient using h and variations
        eps = 1e-5
        grad_f = np.zeros(n)
        for i in range(n):
            h_plus = h_array.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_minus = h_array.copy()
            h_minus[i] -= eps
            if h_minus[i] < 0:
                h_minus[i] = 0
            val_plus = obj(h_plus)
            val_minus = obj(h_minus)
            grad_f[i] = (val_plus - val_minus) / (2 * eps)
        return grad_f

    h_ref = h_original.copy()
    # Project onto simplex to maintain mass constraint
    def project_simplex(h_in):
        h_out = h_in.copy()
        # Sum constraint enforcement
        s = np.sum(h_out)
        c = s - n/2
        h_out = h_out - c / n
        # Box constraints
        h_out = np.clip(h_out, 0, 1)
        return h_out

    # Simple projected gradient descent
    for _ in range(50):
        if time.time() - start_time > budget_s - 5:
            return h_ref, compute_c5(h_ref), n
        g = -grad(h_ref)
        lr = 0.1
        h_ref += lr * g
        h_ref = project_simplex(h_ref)

    # Final computation
    return h_ref, compute_c5(h_ref), n


def local_swap_optimization(h, n, start_time, budget_s):
    """
    Perform local swaps to improve solution
    """
    dx = 2.0 / n
    h = np.array(h)
    h = h.copy()
    
    # Calculate all correlations for initial, and then only update
    # the correlation binst that could change.
    def compute_all_correlations(h):
        corrs = []
        for k in range(1, n):
            c = 0.0
            for i in range(n - k):
                c += h[i] * h[i + k]
            corrs.append(c)
        return np.array(corrs)

    corrs = compute_all_correlations(h)
    k_star = np.argmin(corrs)
    min_corr = corrs[k_star]

    swaps_left = 500
    while swaps_left > 0 and time.time() - start_time < budget_s - 5:
        # Pick two indices, i=1, j=0
        i, j = -1, -1
        max_change = -float('inf')
        
        # Check various pair pairs close to threshold with impact
        for i1 in range(n):
            if h[i1] != 1:
                continue
            for j1 in range(n):
                if h[j1] != 0:
                    continue
                    
                # How does this pair affect corrs[k_star]?
                # For dynamic update, check all hits potentially affected
                # Simplified logic here:
                change = 0.0
                # Assume fixed shift updating
                added = h[i1] * h[j1]  # Once
                for k in range(1, n):
                    if i1 + k >= n or j1 + k >= n:
                        continue
                    if i1 + k == j1:  # Only one index flips here
                        added += 1  # We now have 1 instead of 0 at i1 * h[j1+k]
                    if i1 == j1 + k:
                        added -= 1  # Phyically switched in center of convolution
                    change += added

                if change > max_change:
                    max_change = change
                    i, j = i1, j1
                
        if max_change <= 0:
            break

        # Make swap
        h[i], h[j] = h[j], h[i]
        
        # Update only the affected size (very approximate)
        # In full version, we update the actual correlations
        # but for efficiency here assume we recalculate all
        corrs = compute_all_correlations(h)
        k_star = np.argmin(corrs)
        min_corr = corrs[k_star]
        
        swaps_left -= 1
            
    return h, compute_c5(h), n
