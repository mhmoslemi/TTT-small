import time
import numpy as np
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    # Parameters
    n0 = 120  # initial grid size
    dx = 2.0 / n0
    S = n0 / 2  # target sum
    
    # Precomputed initial solution (if available)
    try:
        h0 = np.array(initial_h_values)
        if len(h0) == 47:
            # upscale to n0 by repeating elements
            h0 = np.repeat(h0, n0 // len(h0))
            h0 = np.pad(h0, (0, n0 - len(h0)), 'constant')
    except:
        h0 = None

    # Binary integer program formulation
    # Variables
    b = cp.Variable(n0, boolean=True)
    w = cp.Variable(1, 0, S)  # objective value
    z = cp.Variable((n0, n0), boolean=True)  # products b[i] * b[i+k]

    # Constraints
    constraints = []

    # Product linearisation
    for i in range(n0):
        for k in range(n0):
            idx = (i + k) % n0
            constraints.append(z[i, k] <= b[i])
            constraints.append(z[i, k] <= b[idx])
            constraints.append(z[i, k] >= b[i] + b[idx] - 1)

    # Overlap constraints
    for k in range(n0):
        constraints.append(w >= S - cp.sum(z[:, k]))

    # Mass constraint
    constraints.append(cp.sum(b) == S)

    # Symmetry breaking
    constraints.append(b[0] == 1)
    
    # Objective
    objective = cp.Minimize(w)

    # Build problem
    prob = cp.Problem(objective, constraints)

    # Solve with solver and remaining time limit
    elapsed = time.time() - start
    remaining = max(1, budget_s - elapsed)
    
    # Use GLPK or SCIP if available; otherwise fallback to CBC
    solver = cp.CBC
    try:
        solver = cp.SCIP
    except:
        try:
            solver = cp.GLPK
        except:
            solver = cp.CBC
    
    try:
        prob.solve(solver=solver, verbose=False, max_seconds=remaining)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            b_opt = np.round(b.value).astype(int)
            w_opt = w.value[0]
        else:
            raise Exception("MILP solver did not find an optimal solution")
    except Exception as e:
        # fallback to greedy construction
        print(f"MILP failure: {e}, falling back to greedy")
        b_opt, w_opt = greedy_balanced(n0, S, start, budget_s)
    
    # Compute C5 from optimal w
    c5_bound = w_opt * dx
    
    # Post-process with pairwise swaps
    b_opt, c5_bound = local_swap_improvement(b_opt, c5_bound, n0, dx, start, budget_s)
    
    # Up-scale to final size (360)
    N = 360
    h_final = np.repeat(b_opt, N // n0)
    c5_final = c5_bound
    
    return (h_final.tolist(), float(c5_final), len(h_final))

def greedy_balanced(n0, S, start, budget_s):
    # Simple greedy construction
    # Start with an empty set
    b = np.zeros(n0, dtype=int)
    selected = set()
    s = 0
    
    remaining = budget_s - (time.time() - start)
    if remaining < 0.1:
        # Quits early if insufficient time left
        return b, 0.1  # Dummy value, may be better to raise an error
    
    while s < S:
        best_i = -1
        min_delta = float('inf')
        for i in range(n0):
            if i in selected:
                continue
            # Estimate change in overlaps if we add this point
            # Count overlap increases for all shifts k
            num_1s = s + 1
            # For each position in neighbors, compute change in overlaps
            max_increase = 0
            for k in range(n0):
                # Check how many existing 1s are at lag k from i
                increase = 0
                for j in selected:
                    if (j + k) % n0 == i:
                        increase += 1
                if increase > max_increase:
                    max_increase = increase
            # We want to minimize maximal hit
            if max_increase < min_delta:
                min_delta = max_increase
                best_i = i
        if best_i >= 0:
            selected.add(best_i)
            b[best_i] = 1
            s += 1
        else:
            # We're done
            break
        # Early exit if time is running out
        if time.time() - start > budget_s - 0.1:
            break
    
    # compute tight empirical C5
    b = np.array(b).astype(float)
    dx = 2.0 / n0
    c5_empirical = 0.0
    
    # compute all correlations
    for k in range(n0):
        acc = 0.0
        for i in range(n0):
            j = (i - k) % n0
            acc += b[i] * b[j]
        c5 = n0 - acc   # this is overlap for shift k
        if c5 > c5_empirical:
            c5_empirical = c5
        
    c5_bound = c5_empirical * dx
    return b.astype(int), c5_bound

def local_swap_improvement(b, current_c5, n0, dx, start, budget_s):
    # One-pass local search using swaps
    # Correlation structure: A_k = sum_i b[i]*b[i+k]
    # Overlap = S - A_k
    
    remaining = budget_s - (time.time() - start)
    if remaining < 0.2:
        return b, current_c5
    
    # compute current overlaps
    def compute_autocorrelations(b_vec):
        n = len(b_vec)
        A = np.zeros(n)
        b = b_vec.astype(float)
        for k in range(n):
            acc = 0.0
            for i in range(n):
                j = (i - k) % n
                acc += b[i] * b[j]
            A[k] = acc
        return A
    
    iter_limit = 200
    for _ in range(iter_limit):
        A = compute_autocorrelations(b)
        min_k = np.argmin(A)
        worst_autocorr = A[min_k]
        improved = False
        
        # Try all swaps between 1 and 0 positions
        for i in range(n0):
            if b[i] == 1:
                for j in range(n0):
                    if b[j] == 0:
                        # Simulate swap effect
                        # This affects correlations around min_k
                        delta = 0.0
                        
                        # first compute prior value at min_k contribution of i and j
                        prior_i_contrib = 0.0
                        prior_j_contrib = 0.0
                        for k in range(n0):
                            # compute what the term of this correlation is before swap
                            pi = (i - k) % n0
                            pj = (j - k) % n0
                            if b[pi] == 1: prior_i_contrib += 1.0
                            if b[pj] == 1: prior_j_contrib += 1.0
                        
                        # compute new
                        new_i_contrib = 0.0
                        new_j_contrib = 0.0
                        for k in range(n0):
                            pi = (i - k) % n0
                            pj = (j - k) % n0
                            if b[pi] == 0 and b[j] == 1:
                                new_i_contrib += 1.0
                            if b[i] == 0 and b[pj] == 1:
                                new_j_contrib += 1.0
                        
                        # difference
                        old = A[min_k]  # or compute old contribution
                        # Instead, manually compute new value:
                        # For simplicity, apply change manually
                        # This part is tricky to do in full lazy fashion.
                        # Let's compute the best local swap directly.
                        
                        # Compute reversely:
                        # Updated A_k values
                        new_A = np.zeros(n0)
                        for k in range(n0):
                            new_A[k] = A[k] - (b[i] * b[(i - k) % n0] + 
                                                b[j] * b[(j - k) % n0]) + \
                                       (b[j] * b[(i - k) % n0] + 
                                        b[i] * b[(j - k) % n0])
                        
                        new_worst = np.min(new_A)
                        if new_worst < worst_autocorr:
                            # This swap improves
                            b[i] = 0
                            b[j] = 1
                            worst_autocorr = new_worst
                            improved = True
                            break
                if improved:
                    break
                
        if not improved:
            break
    
    # recompute final C5
    A = compute_autocorrelations(b)
    c5_bound = np.max(A) * dx  # but overlaps = n - A
    min_overlap = n0 - np.min(A)  # Maximum overlap
    c5_bound = min_overlap * dx
    
    return b, c5_bound
