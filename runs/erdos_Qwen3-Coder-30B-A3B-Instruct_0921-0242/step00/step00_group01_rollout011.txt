import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy import signal

def evaluate_h(h):
    """Evaluate the C5 constant for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The integral is dx times the maximum negative of this correlation
    # So we maximize the minimum autocorrelation (which is always negative here)
    C5 = dx * (n/2 - np.min(corr))
    return C5

def evaluate_h_precise(h, n_evaluate=1000):
    """Evaluate C5 with a fine grid for accuracy."""
    h_fine = np.interp(np.linspace(0, 2, n_evaluate), np.linspace(0, 2, len(h)), h)
    return evaluate_h(h_fine)

def block_search():
    """Enumerate 3-block families and return best."""
    best_C5 = float('inf')
    best_h = None
    n_points = 400
    dx = 2.0 / n_points
    a_range = np.arange(0, 2, 0.02)
    b_range = np.arange(0.02, 2, 0.02)
    
    for a in a_range:
        for b in b_range:
            if a >= b:
                continue
            p = (1 - a) / (b - a)
            if 0 <= p <= 1:
                h = np.zeros(n_points)
                h[:int(a / dx)] = 1.0
                h[int(a / dx):int(b / dx)] = p
                # Normalize to sum = n_points / 2
                h_sum = np.sum(h)
                h = h * (n_points / 2) / h_sum
                C5 = evaluate_h(h)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h.copy()
    
    return best_h, best_C5

def gen_mixture_h(c, a, l=0.1):
    """Generate a 3-block mixture."""
    h = np.zeros(200)
    for i in range(len(c)):
        left = max(0, int((c[i] - l) * 100))
        right = min(200, int((c[i] + l) * 100))
        h[left:right] += a[i]
    # Clip and normalize
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * 100 / h_sum
    return h

def de_stage():
    """Run differential evolution search."""
    def objective(x):
        # x is [c1, c2, c3, a1, a2, a3]
        c = [x[0], x[1], x[2]]
        a = [x[3], x[4], x[5]]
        h = gen_mixture_h(c, a)
        # Stretch to 200 points for evaluation
        h = np.interp(np.linspace(0, 2, 200), np.linspace(0, 2, 200), h)
        return evaluate_h(h)

    bounds = [(0.0, 2.0), (0.0, 2.0), (0.0, 2.0),
              (0.0, 1.0), (0.0, 1.0), (0.0, 1.0)]
    
    result = differential_evolution(objective, bounds, seed=42, maxiter=1000, popsize=30)
    
    # Compute final h
    c = [result.x[0], result.x[1], result.x[2]]
    a = [result.x[3], result.x[4], result.x[5]]
    h = gen_mixture_h(c, a)
    h = np.interp(np.linspace(0, 2, 200), np.linspace(0, 2, 200), h)
    C5 = evaluate_h(h)
    return h, C5

def project_step(h, n_points, dx):
    """Project step after gradient update."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    # Adjust to make sum equal to n/2
    offset = (h_sum - n_points / 2) / n_points
    h = h - offset
    h = np.clip(h, 0, 1)
    return h

def gradient_ascent_step(h, kstar):
    """Perform one gradient ascent step."""
    g = np.roll(h, kstar)
    h_new = h + 0.5 * g
    return project_step(h_new, len(h), 2.0 / len(h))

def gradient_ascent(h_start, max_time):
    """Projected gradient ascent with multi-grid resolution."""
    start_time = time.time()
    h = h_start.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_h(h)
    
    # Multi-resolution schedule
    r = 1  # Initial refinement level
    n_levels = [150, 300, 500]
    
    for level in range(len(n_levels)):
        if time.time() - start_time > max_time - 5:
            break
        n = n_levels[level]
        dx = 2.0 / n
        if level > 0:
            h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h)), h)
            h = project_step(h, n, dx)
        
        iter_count = 0
        for i in range(5000):  # Max iterations per level
            if time.time() - start_time > max_time - 10:
                break
                
            if iter_count % 50 == 0:
                C5_candidate = evaluate_h(h)
                if C5_candidate < best_C5:
                    best_C5 = C5_candidate
                    best_h = h.copy()
            
            # Auto-correlation via FFT
            A = np.fft.ifft(np.abs(np.fft.fft(h))**2).real[:n]
            kstar = np.argmin(A)
            
            # Gradient ascent step, decreasing step size as iterations go on
            step_size = 0.5 * (0.98 ** i)
            h = h + step_size * np.roll(h, kstar)
            h = project_step(h, n, dx)
            
            iter_count += 1
    
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Block enumeration
    h1, C5_1 = block_search()
    if time.time() - start_time > budget_s - 50:
        return list(h1), C5_1, len(h1)
        
    # Stage 2: Differential Evolution
    h2, C5_2 = de_stage()
    if time.time() - start_time > budget_s - 50:
        return list(h2), C5_2, len(h2)
        
    # Stage 3: Gradient Ascent
    # Select the better of h1 and h2 as starting point
    h_start = h1 if C5_1 < C5_2 else h2
    h_final, C5_final = gradient_ascent(h_start, budget_s)
    
    return list(h_final), C5_final, len(h_final)
