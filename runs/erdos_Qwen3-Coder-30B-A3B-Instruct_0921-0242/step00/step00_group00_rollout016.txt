import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist

def decode(x, M):
    """Decode the flat vector x into segment lengths u and heights a."""
    u = x[:M]
    a0 = x[M]
    v = x[M+1:]
    
    # Compute a_j from a0 and v_j
    a = np.empty(M+1)
    a[0] = a0
    for j in range(1, M):
        a[j] = a[j-1] - v[j-1]
    a[M] = a[M-1] - v[M-1]
    
    # Compute breakpoints s_j
    s = np.empty(M+1)
    s[0] = 0
    for j in range(M):
        s[j+1] = s[j] + u[j]
    
    return u, a, s

def overlap_piecewise(s, a):
    """Compute the exact overlap C5 for a piecewise constant function."""
    M = len(a) - 1
    K = []
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0 <= k1 <= 2:
                K.append(k1)
            if 0 <= k2 <= 2:
                K.append(k2)
    K = sorted(set(K + [0, 2]))
    
    def len_intersect(j, l, k):
        left = max(s[j], s[l] - k)
        right = min(s[j+1], s[l+1] - k)
        return max(0, right - left)
    
    def F(k):
        total = 0
        for j in range(M+1):
            for l in range(M+1):
                total += a[j] * (1 - a[l]) * len_intersect(j, l, k)
        return total
    
    F_vals = [F(k) for k in K]
    return max(F_vals)

def project_capped_simplex(v, total):
    """Project v onto the capped simplex sum(v) = total."""
    v = np.array(v)
    high = 1.0
    low = 0.0
    s = v.sum()
    if abs(s - total) < 1e-12:
        return v
    if s < total:
        d = total - s
        v += d / len(v)
        return np.clip(v, low, high)
    elif s > total:
        # Use binary search on lambda to find the right projection
        def xs(x):
            return np.clip(v - x, low, high).sum()
        def func(x):
            return xs(x) - total
        if s < total + 1e-8:
            return np.clip(v, low, high)
        # Binary search for lambda
        lb = 0.0
        ub = np.max(v) - low
        while ub - lb > 1e-10:
            x = (lb + ub) / 2
            if func(x) > 0:
                lb = x
            else:
                ub = x
        lam = ub
        v = np.clip(v - lam, low, high)
        # Fix numeric drift
        v *= total / v.sum()
        return v
    else:
        return v

def fitness(x, M, n_points, lambda_integral=1e6):
    u, a, s = decode(x, M)
    if np.any(a < 0) or np.any(a > 1) or np.any(u <= 0) or np.any(u > 2):
        return 1e6
    integral = np.sum(a[:-1] * u)
    if abs(integral - 1) > 1e-8:
        penalty = lambda_integral * (integral - 1)**2
    else:
        penalty = 0
    C5 = overlap_piecewise(s, a)
    return C5 + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase A: Coarse model with M=4 (n_points=128)
    M = 4
    n_points = 128
    dx = 2.0 / n_points
    
    bounds = [(0, 2)] * M + [(0, 1)] + [(0, 1)] * (M-1)
    de_args = {
        'bounds': bounds,
        'strategy': 'best1bin',
        'mutation': (0.5, 1),
        'recombination': 0.7,
        'polish': False,
        'maxiter': 200,
        'popsize': 15 * len(bounds),
        'seed': seed + 1
    }

    de_result = differential_evolution(fitness, bounds, args=(M, n_points), **de_args)
    bestx_A = de_result.x
    best_C5_A = de_result.fun
    u_A, a_A, s_A = decode(bestx_A, M)
    
    # Phase B: Finer model with M=6 (n_points=256)
    M = 6
    n_points = 256
    dx = 2.0 / n_points
    
    bounds = [(0, 2)] * M + [(0, 1)] + [(0, 1)] * (M-1)
    
    # Initialize DE population around best from Phase A
    init_pop = []
    init_pop.append(bestx_A)
    for _ in range(de_args['popsize'] - 1):
        noise = np.random.normal(0, 0.1, len(bounds))
        new_x = bestx_A + noise
        # Re-project to bounds manually
        new_x = np.clip(new_x, [b[0] for b in bounds], [b[1] for b in bounds])
        init_pop.append(new_x)
    
    de_args['init'] = init_pop

    de_result = differential_evolution(fitness, bounds, args=(M, n_points), **de_args)
    bestx_B = de_result.x
    best_C5_B = de_result.fun
    
    # Phase C: Local refinement using multiple methods
    # 1. Smoothed log-sum-exp optimization
    def smooth_fitness(x, beta=50):
        u, a, s = decode(x, M)
        if np.any(a < 0) or np.any(a > 1) or np.any(u <= 0) or np.any(u > 2):
            return 1e6
        integral = np.sum(a[:-1] * u)
        if abs(integral - 1) > 1e-8:
            penalty = 1e6 * (integral - 1)**2
        else:
            penalty = 0

        K = []
        for j in range(M+1):
            for l in range(M+1):
                k1 = s[l] - s[j+1]
                k2 = s[l+1] - s[j]
                if 0 <= k1 <= 2:
                    K.append(k1)
                if 0 <= k2 <= 2:
                    K.append(k2)
        K = sorted(set(K + [0, 2]))

        def len_intersect(j, l, k):
            left = max(s[j], s[l] - k)
            right = min(s[j+1], s[l+1] - k)
            return max(0, right - left)

        def F(k):
            total = 0
            for j in range(M+1):
                for l in range(M+1):
                    total += a[j] * (1 - a[l]) * len_intersect(j, l, k)
            return total

        F_vals = [F(k) for k in K]
        logsumexp = np.log(np.sum(np.exp(beta * np.array(F_vals)))) / beta
        return logsumexp + penalty

    # Optimize using SLSQP
    x0 = bestx_B
    bounds_slsqp = [(b[0], b[1]) for b in bounds]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(decode(x, M)[1][:-1] * decode(x, M)[0]) - 1}

    try:
        result_slsqp = minimize(smooth_fitness, x0, method='SLSQP', bounds=bounds_slsqp, constraints=cons, options={'maxiter': 200})
        if result_slsqp.success:
            refined_x = result_slsqp.x
        else:
            refined_x = bestx_B
    except:
        refined_x = bestx_B

    # 2. Subgradient descent
    u, a, s = decode(refined_x, M)
    def F(k):
        total = 0
        for j in range(M+1):
            for l in range(M+1):
                def len_intersect(j, l, k):
                    left = max(s[j], s[l] - k)
                    right = min(s[j+1], s[l+1] - k)
                    return max(0, right - left)
                total += a[j] * (1 - a[l]) * len_intersect(j, l, k)
        return total

    K = []
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0 <= k1 <= 2:
                K.append(k1)
            if 0 <= k2 <= 2:
                K.append(k2)
    K = sorted(set(K + [0, 2]))

    F_vals = [F(k) for k in K]
    max_idx = np.argmax(F_vals)
    k_opt = K[max_idx]
    
    # Subgradient-based descent
    max_iter = 100
    alpha0 = 1e-3
    for t in range(max_iter):
        if time.time() - start_time > budget_s - 0.5:
            break
        # Approximate gradient based on local geometry
        g = np.zeros_like(refined_x)
        # Simple gradient descent, we identify a single relevant shift
        a_old = a.copy()
        u_old = u.copy()
        # For an approximate gradient, we just shift slightly
        step = 1e-4
        while True:
            u_test = u_old.copy()
            a_test = a_old.copy()
            u_test[0] = max(0.01, u_test[0] - step)
            a_test[0] -= step / 10  # arbitrary scaling
            a_test = np.clip(a_test, 0, 1)
            if np.any(u_test <= 0) or np.any(u_test > 2):
                break
            s_test = np.concatenate([[0], np.cumsum(u_test)])
            
            C5_test = overlap_piecewise(s_test, a_test)
            C5_ref = overlap_piecewise(s, a)
            grad = (C5_test - C5_ref) / step
            if abs(grad) < 1e-6:
                break
            alpha = alpha0 / np.sqrt(t+1)
            u = u_test.copy()
            a = a_test.copy()
            s = s_test.copy()
            break

    # Final discretization
    last_x = refined_x
    u, a, s = decode(last_x, M)
    h_values = []
    for i in range(n_points):
        x = (i + 0.5) * dx
        for j in range(M):
            if s[j] <= x < s[j+1]:
                h_values.append(a[j])
                break
        else:
            h_values.append(a[M])
    h_values = np.array(h_values)
    
    # Final projection to enforce integral = 1
    h_values = project_capped_simplex(h_values, total=n_points/2)
    
    # Compute final C5
    corr = np.convolve(h_values, 1-h_values, mode='full') * dx
    c5_bound = np.max(corr)
    
    return (h_values.tolist(), float(c5_bound), n_points)
