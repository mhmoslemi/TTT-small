import numpy as np

def evaluate_h(h, dx):
    """Evaluate C5 bound for given h"""
    # Compute the overlap integral for all possible shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # We're interested in the maximum value among all shifts
    # The full correlation has length 2*n-1, we want the middle n elements
    # which correspond to shifts from -n+1 to n-1
    # The maximum corresponds to k from 0 to 2n-2, but we really want
    # the shift k from 0 to n-1 (since shift k and shift 2n-2-k are equivalent)
    # Actually, let's just take the maximum over all the correlation values
    # that are valid for the overlap integral
    max_correlation = np.max(correlation)
    return max_correlation * dx
