import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate C5 for a given h function"""
    # Compute the overlap integral using convolution
    n = len(h)
    dx = 2.0 / n
    # Compute the full convolution of h and (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # The maximum value of the convolution scaled by dx
    c5 = np.max(conv) * dx
    return c5
