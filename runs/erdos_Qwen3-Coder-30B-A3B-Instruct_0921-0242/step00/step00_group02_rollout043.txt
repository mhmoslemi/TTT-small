import numpy as np
import time
import math
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist
import random

# Constants
MAX_M = 8
MIN_M = 4
N_VALUES = [30, 36, 42, 48, 60]
TOL = 1e-4
MAX_DIRECT_NODES = 5000
MAX_DE_GEN = 2000
MAX_SWAP_ITERS = 200
MAX_DE_RESTART = 3

def initial_h_values():
    return np.array([1.0] * 22 + [0.0] * 23)  # n=45 as in initial problem

# === Helper functions for geometric interval optimization ===
def compute_overlap_at_shift(a, ell, k):
    """Compute overlap at a given shift k for a set of intervals"""
    total_overlap = 0.0
    for i in range(len(a)):
        for j in range(len(a)):
            diff = a[i] - a[j]
            overlap = max(0.0, ell - abs(k - diff))
            total_overlap += overlap
    return total_overlap

def generate_breakpoints(a, ell):
    """Generate all breakpoint candidates for O(k)"""
    breakpoints = set()
    for i in range(len(a)):
        for j in range(len(a)):
            diff = a[i] - a[j]
            breakpoints.add(diff - ell)
            breakpoints.add(diff + ell)
    return sorted(breakpoints)

def compute_max_overlap(a, ell):
    """Compute max overlap over all shifts in [0,2]"""
    breakpoints = generate_breakpoints(a, ell)
    max_overlap = 0.0
    for k in breakpoints:
        if 0 <= k <= 2:
            overlap = compute_overlap_at_shift(a, ell, k)
            max_overlap = max(max_overlap, overlap)
    # Also consider the endpoints
    max_overlap = max(max_overlap, compute_overlap_at_shift(a, ell, 0))
    max_overlap = max(max_overlap, compute_overlap_at_shift(a, ell, 2))
    return max_overlap

# === DIRECT-style branch-and-bound solver (simplified) ===
def lipschitz_lower_bound(a_center, a_widths, ell, interval_count):
    """Compute a lower bound using Lipschitz continuity"""
    # Simplified lower bound estimation
    # We use max overlap viewing as a piecewise linear function
    # Lipschitz constant is about interval_count, but let's use a simplified version
    max_self_overlap = interval_count * ell  # self overlaps are constant
    estimated_max = max_self_overlap
    return estimated_max - interval_count * sum(a_widths)

def optimize_intervals_direct(m, elapsed_time, budget_s, best_so_far):
    """Phase A: Deterministic Lipschitz branch-and-bound"""
    start_time = time.time()
    ell = 1.0 / m
    best_solution = None
    best_value = float('inf')
    interval_bounds = [(0, 2 - ell * m)]  # Initial box
    node_count = 0
    stack = [([0] * m, [2.0 / m] * m)]  # (a_vals, widths)

    while stack and node_count < MAX_DIRECT_NODES and time.time() - start_time < 0.8 * budget_s - elapsed_time:
        a, widths = stack.pop()
        node_count += 1
        center = [a[i] + widths[i] / 2 for i in range(m)]
        if not isValid(center, ell): continue  # feasibility check
        try:
            value = compute_max_overlap(center, ell)
            if value < best_value:
                best_value = value
                best_solution = center
        except:
            continue
        
        # Bisect the node with largest width
        max_width_idx = np.argmax(widths)
        if widths[max_width_idx] > 1e-6:
            new_widths = widths[:]
            new_widths[max_width_idx] /= 2
            a1 = a[:]
            a2 = a[:]
            a1[max_width_idx] += new_widths[max_width_idx]
            stack.append((a1, new_widths))
            stack.append((a2, new_widths))

    return best_solution, best_value

def isValid(a, ell):
    """Check if interval configuration is valid"""
    for i in range(len(a) - 1):
        if a[i + 1] < a[i] + ell:
            return False
    if a[-1] + ell > 2:
        return False
    return True

# === DE refinement (Phase B) ===
def optimize_intervals_de(a_init, m, ell, elapsed_time, budget_s):
    """Phase B: Stochastic refinement using Differential Evolution"""
    call_count = 0
    initial = np.array(a_init)
    bounds = [(0.0, 2.0 - ell * m) for _ in range(m)]
    min_M = float('inf')
    best_a = None
    restarts = 0

    while restarts <= MAX_DE_RESTART and (time.time() - elapsed_time < 0.8 * budget_s):
        def objective(x):
            nonlocal call_count
            call_count += 1
            if call_count > MAX_DE_GEN:
                return float('inf')
            x = np.clip(x, 0.0, 2.0 - ell * m)
            # Enforce ordering constraints by projection
            x = enforce_ordering(x, ell)
            try:
                val = compute_max_overlap(x, ell)
                return val
            except:
                return float('inf')

        result = differential_evolution(objective, bounds, maxiter=1000, popsize=min(20, 5*m),
                                        seed=int(time.time() * 1000) % 1000000, 
                                        disp=False, polish=False)
        if result.success and result.fun < min_M:
            min_M = result.fun
            best_a = result.x
        restarts += 1

    if best_a is None:
        best_a = a_init 
    return best_a, min_M

def enforce_ordering(a, ell):
    """Project to a valid interval sequence"""
    a = np.array(a).copy()
    total = sum(a[:-1]) + a[-1]
    for i in range(1, len(a)):
        if i == len(a) - 1:
            a[i] = max(a[i], a[i-1] + ell)  # Last interval must satisfy
        else:
            a[i] = max(a[i], a[i-1] + ell)  # Ensure gaps
    return a

# === Conversion to discrete vector ===
def discretize_intervals_to_h(a, ell, n):
    """Convert interval positions to discrete h values"""
    dx = 2.0 / n
    h = np.zeros(n)
    for i in range(n):
        x = i * dx
        for j in range(len(a)):
            left = a[j]
            right = a[j] + ell
            if left <= x <= right:
                h[i] = 1.0
                break
    if abs(np.sum(h) - n / 2) > 1:
        # Distribute over boundary points to achieve sum=n/2
        adjust_points = []
        for i in range(n):
            x = i * dx
            closest = float('inf')
            for j in range(len(a)):
                left = a[j]
                right = a[j] + ell
                dist = min(abs(x - left), abs(x - right))
                closest = min(closest, dist)
            adjust_points.append((closest, i))
        adjust_points.sort()
        needed = int(n / 2) - int(np.sum(h))
        if needed > 0:
            for i in range(needed):
                idx = adjust_points[i][1]
                h[idx] = 1.0
        elif needed < 0:
            for i in range(abs(needed)):
                idx = adjust_points[-(i+1)][1]
                h[idx] = 0.0
    return h

# === Binary swap polishing ===
def refine_with_swaps(h, min_k):
    """Heuristic local refinement via binary swaps"""
    h = h.copy()
    overlaps_original = np.correlate(h, 1 - h, mode='full')
    max_orig = np.max(overlaps_original)
    swaps_total = 0
    for _ in range(MAX_SWAP_ITERS):
        best_improvement = float('-inf')
        best_i, best_j = -1, -1
        for _ in range(200):  # sample pair of positions
            i = random.randint(0, len(h) - 1)
            j = random.randint(0, len(h) - 1)
            if i == j:
                continue
            if h[i] == h[j]:
                continue
            # compute change in overlap at critical shift k*
            # For now: just use a crude check based on >= improvement
            h_copy = h.copy()
            h_copy[i], h_copy[j] = h_copy[j], h_copy[i]
            overlaps_new = np.correlate(h_copy, 1 - h_copy, mode='full')
            max_new = np.max(overlaps_new)
            if max_new < max_orig:
                improvement = max_orig - max_new
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i, best_j = i, j
        if best_improvement > 1e-6:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            max_orig = best_improvement
            swaps_total += 1
        else:
            break
    return h

def evaluate_erdos_solution(h):
    # The same as in the original problem
    return np.max(np.correlate(h, 1 - h, mode="full")) * 2 / len(h)

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None

    for m in range(MIN_M, MAX_M + 1):
        if (time.time() - start_time) > budget_s - 5:
            break
        ell = 1.0 / m
        print(f"Optimizing m={m} intervals...")
        # Phase A: Direct-style search
        a_init, value = optimize_intervals_direct(m, start_time, budget_s, best_c5)
        if a_init is None or value >= best_c5:
            continue
        # Phase B: DE refinement
        a_refined, value_de = optimize_intervals_de(a_init, m, ell, start_time, budget_s)
        print(f"  M={m}: refined overlap = {value_de}")
        # Convert to discrete vector
        for n in N_VALUES:
            if (time.time() - start_time) > budget_s - 5:
                break
            h_discrete = discretize_intervals_to_h(a_refined, ell, n)
            h_final = refine_with_swaps(h_discrete, None)  # No special handling for i=5 case -- consider shifts
            derived_c5 = evaluate_erdos_solution(h_final)
            print(f"    n={n}: C5 = {derived_c5}")
            if derived_c5 < best_c5:
                best_c5 = derived_c5
                best_h = h_final.copy()
                best_n = n
            if derived_c5 < 0.379:
                break
        if best_c5 < 0.379:
            break

    if best_h is None:
        # fallback to initial solution if nothing works
        h0 = initial_h_values()
        best_h = h0
        best_c5 = evaluate_erdos_solution(best_h)
        best_n = len(best_h)
    return (best_h, best_c5, best_n)
