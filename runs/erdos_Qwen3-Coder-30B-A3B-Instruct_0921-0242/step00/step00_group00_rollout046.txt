import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.special import logsumexp

def decode(x, M):
    """Decodes the flattened parameter vector x into segment lengths u and heights a."""
    u = x[:M]
    a0 = x[M]
    v = x[M+1:]
    a = np.empty(M+1)
    a[0] = a0
    for j in range(1, M):
        a[j] = a[j-1] - v[j-1]
    a[M] = a[M-1] - v[M-1]
    
    # Ensure non-negativity
    a = np.maximum(a, 0.0)
    
    # Compute the segments
    s = np.concatenate([[0.0], np.cumsum(u)])
    s[M] = 2.0  # Fix last breakpoint to be 2
    
    return s, a

def overlap_piecewise(s, a):
    """Computes the overlap integral C5 for a given piecewise-constant function."""
    M = len(a) - 1
    # Compute all breakpoints in [0, 2]
    K = []
    for j in range(M+1):
        for l in range(M+1):
            k1 = s[l] - s[j+1]
            k2 = s[l+1] - s[j]
            if 0 <= k1 <= 2:
                K.append(k1)
            if 0 <= k2 <= 2:
                K.append(k2)
    K = sorted(set(K) | {0.0, 2.0})
    
    # Evaluate F(k) for each k in K
    F = np.empty(len(K))
    for i, k in enumerate(K):
        F[i] = 0.0
        for j in range(M+1):
            for l in range(M+1):
                term = a[j] * (1 - a[l])
                # Compute intersection length L_{jl}(k)
                left = max(s[j], s[l] - k)
                right = min(s[j+1], s[l+1] - k)
                if right > left:
                    L = right - left
                else:
                    L = 0.0
                F[i] += term * L
    return np.max(F)

def project_capped_simplex(v, total):
    """Projects a vector onto the capped simplex."""
    v = np.array(v)
    n = len(v)
    sorted_indices = np.argsort(v)[::-1]
    sorted_v = v[sorted_indices]
    
    # Compute the cumulative sums
    cumsum = np.cumsum(sorted_v)
    # Compute the thresholds
    threshold = (cumsum - total) / np.arange(1, n+1)
    
    # Find the largest index where threshold <= sorted_v
    mask = sorted_v > threshold
    if not np.any(mask):
        # All elements are zero
        return np.zeros(n)
    k = np.argmax(mask)  # This is now the largest invalid position
    
    # Alpha is threshold[k]
    alpha = threshold[k]
    # Project back to original order
    result = np.zeros(n)
    for i in range(n):
        if v[i] > alpha:
            result[i] = v[i] - alpha
        else:
            result[i] = 0.0
    return result

def evaluate(x, M):
    """Objective function for differential evolution."""
    s, a = decode(x, M)
    
    # Check bounds, penalize if violated
    if np.any(a < 0) or np.any(a > 1):
        return 1e6
    
    # Evaluate integral
    integral = np.sum(a[:-1] * np.diff(s))
    r = integral - 1.0
    
    if abs(r) > 1e-8:
        penalty = 1e6 * r**2
    else:
        penalty = 0
        
    # Evaluate overlap
    c5 = overlap_piecewise(s, a)
    return c5 + penalty

def discretize_h(s, a, n_points):
    """Generate discrete h values from continuous s and a."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    for i in range(n_points):
        x = (i + 0.5) * dx
        # Find which segment x belongs to
        j = np.searchsorted(s, x, side="right") - 1
        # Handle boundary case for x=2
        if x >= s[-1]:
            j = len(s) - 2
        h[i] = a[j]
    
    # Apply capped simplex projection to maintain integral = 1
    h = project_capped_simplex(h, total=n_points / 2)
    
    # Compute overlap using full convolution
    corr = np.convolve(h, 1 - h, mode='full') * dx
    c5_bound = np.max(corr)
    
    return h.tolist(), c5_bound
