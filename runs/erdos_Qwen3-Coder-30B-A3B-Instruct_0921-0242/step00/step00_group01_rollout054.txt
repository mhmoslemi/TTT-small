import numpy as np
import time
import scipy.signal as signal

def project_box_affine(h, target_sum):
    """Project h onto box [0,1] and affine constraint sum(h)=target_sum."""
    h = np.clip(h, 0, 1)
    delta = (np.sum(h) - target_sum) / len(h)
    h -= delta
    h = np.clip(h, 0, 1)
    return h

def autocorr(h):
    """Circular autocorrelation of h using FFT."""
    H = np.fft.fft(h)
    autocorr_vals = np.real(np.fft.ifft(H * np.conj(H)))
    return autocorr_vals

def evaluate_c5(h, dx):
    """Evaluate C5 using full convolution."""
    # h is a list of values on [0, 2]
    # Evaluate all overlaps: int h(x)(1 - h(x+k)) dx for k=0..n-1
    # Use full mode correlation
    conv = np.correlate(h, 1 - h, mode="full")
    # Correlation is over +k shifts, we need max of all positive lags
    # since we stack h and 1-h we need to extract -n+1 to n-1 lags
    # but due to symmetry and full mode, last n-1 elements + first n elements
    # are the lags. Get lags 0 to 2*n-2 and max kicks in at indices n-1 to 2*n-2
    # Let's get this right:
    # correlation of length 2*n-1, we want from index n-1 to 2*n-2, corresponding to lags 0 to n-1
    # But actually in correlation, the last n-1 are positive lags, so we must subtract n-1 to get shift.
    # Just take last n elements (shifts 0 to n-1) after computing full correlation
    return np.max(conv[-len(h):]) * dx

def evaluate_c5_approx(h, dx):
    """Approximate C5 using autocorrelation."""
    # We computed all overlaps, but we want max_k int h(x)(1-h(x+k)) dx = max_k [ S - (h*h)_k ] * dx
    # where S = sum(h) * dx, and (h*h)_k = sum h_i h_{i+k}
    # So max_k [ S - (h*h)_k ] = S - min_k (h*h)_k
    # So C5 = S - min_k (h*h)_k if S is constant
    # But we're using DX, and we know the sum constraint is integral = 1, so sum(h) = n/2 
    # If we use h = array of size n with range [0,2], dx = 2/n.
    # Then sum(h)*dx = (n/2)*dx = (n/2)*(2/n) = 1.
    c5_val = 0
    S = np.sum(h) * dx
    auto = autocorr(h)  # length n
    # confusion in indexing: auto[k] is correlation at lag k (0 to n-1)
    # so minimum auto is what we want to maximize (as in problem setup)
    min_auto = np.min(auto)
    c5_val = S - min_auto  # This is not correct for the given integral, let's recompute
    # recompute from scratch:
    # Compute overlaps like before but manually
    # To use correlation correctly, we compute integral for all lag k 
    # int h(x)(1-h(x+k)) dx
    # This is max of 
    # int h(x)*dx - int h(x)h(x+k) dx = S - int h(x)h(x+k) dx
    # So max_k of ( S - correlation_k ) = ( S - min_k of correlation_k )
    # min auto corresponds to biggest decrease of overlap, so biggest C5
    # Actually, expected C5 = S - min(auto) where S is total sum * dx  
    # But what's S exactly? If h is on [0,2] with n points, dx = 2/n
    # Conventional will be 1 = sum(h)*dx = sum(h) * 2/n -> sum(h) = n/2
    S = np.sum(h) * dx  # should be 1
    min_auto = np.min(auto)
    return S - min_auto

def gradient_ascent_step(h, dx, step_size):
    """Perform one gradient ascent step to improve min(h*h)."""
    auto = autocorr(h)
    lag = np.argmin(auto)  # min auto = worst lag
    # gradient of -auto at k is (h_{k+l} + h_{k-l}) 
    # Let's compute gradient for lag=lag of \sum h_i h_{i+k}
    # Which for gradient of auto_k is (h_{l+k} + h_{l-k}) 
    grad = -(np.roll(h, lag) + np.roll(h, -lag))
    h_new = h + step_size * grad
    h_new = project_box_affine(h_new, np.sum(h) * dx)  # ensure mean is same  
    return h_new

    n = 200
    dx = 2.0 / n
    best_c5 = float("inf")
    best_h = None
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Initialize candidates:
    candidates = []

        h_interp = np.interp(x_new, x_orig, init)
        candidates.append(("initial", h_interp))
    else:  
        # Fallback initialization
        h1 = np.ones(n) * 0.5
        candidates.append(("half", h1))

    # 2. Random binary-like
    h2 = np.zeros(n)
    h2[:n//2] = 1
    h2 = np.random.permutation(h2)
    h2 = np.convolve(h2, np.ones(5) / 5, mode='same')  # smooth
    candidates.append(("random_smooth", h2))
    # 3. Alternating blocks
    h3 = np.zeros(n)
    for i in range(0, n, 20):
        h3[i:i+10] = 1
    h3 = np.convolve(h3, np.ones(5) / 5, mode='same')  # smooth 
    candidates.append(("alternating", h3))

    # 4. Random with fixed sum
    for i in range(5):
        h = np.random.rand(n)
        h = project_box_affine(h, n/2 * dx)  
        candidates.append(("random_{}".format(i), h))
        
    # 5. Block-based search could be included but run in fixed time

    # Set up gradient ascent for each
    max_iters = 1500
    step_size = 0.05
    patience = 100
    
    for candidate_name, h_in in candidates:
        if time_left() < 10: 
            break
        h = h_in.copy() 
        old_c5 = float("inf")
        for it in range(max_iters):
            if time_left() < 1:
                break  
            h = gradient_ascent_step(h, dx, step_size)
            if it % 10 == 0:
                c5 = evaluate_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5 
                    best_h = h.copy()
                if abs(c5 - old_c5) < 1e-6:
                    patience -= 1
                    if patience == 0:
                        break
                else:
                    patience = 100  
                old_c5 = c5

    # Downsample if needed
    if best_h is None or time_left() < 30:
        # fall back to original if no step taken, or not enough time
        if "initial_h_values" in globals():
            init = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(init))
            x_new = np.linspace(0, 2, 100)
            best_h = np.interp(x_new, x_orig, init)
        else:
            best_h = np.ones(100) * 0.5
        best_c5 = evaluate_c5(best_h, 2.0/100) 
    
    # smaller sample size for final result
    n_small = 100
    dx_small = 2.0 / n_small
    # Use simplest compatible h with good quality  
    if len(best_h) != n_small:
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_small)
        h_final = np.interp(x_new, x_orig, best_h)
    else:
        h_final = best_h.copy()
    final_c5 = evaluate_c5(h_final, dx_small)
    
    return (h_final.tolist(), float(final_c5), n_small)
