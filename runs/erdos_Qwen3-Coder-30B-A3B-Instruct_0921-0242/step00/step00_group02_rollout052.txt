import numpy as np
import scipy.optimize as opt
import time
import math
from math import sqrt, floor, ceil

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Parameters
    max_m = 8
    target_n_values = [30, 36, 42, 48, 60]
    phases = ['DIRECT', 'DE', 'discretize', 'polish']
    
    best_C5 = float('inf')
    best_h = None
    best_n = None
    
    # test m = 4 to 8
    for m in range(4, max_m + 1):
        if time.time() - t0 > budget_s - 5:
            break
            
        target_L = 1.0 / m
        # search space bounds
        max_gap = 2 - m * target_L
        bounds = [(0, max_gap)] * (m-1)
        
        # Phase A: DIRECT global search
        best_a = None
        best_M = float('inf')
        initial_box = [(0, max_gap)] * (m-1)
        n_box = 1
        boxes = [initial_box]
        
        def compute_Ok(a):
            # compute max_k O(k) for given a
            # a[0],...,a[m-1] are the left endpoints
            # length of each interval is target_L
            # total length is 1
            # we compute all breakpoints
            Ks = []
            for i in range(m):
                for j in range(m):
                    # interval i starts at a[i] and ends at a[i]+target_L
                    # interval j starts at a[j] and ends at a[j]+target_L
                    # break at k = |a[i]-a[j]| ± target_L
                    k1 = abs(a[i] - a[j]) - target_L
                    k2 = abs(a[i] - a[j]) + target_L
                    if k1 >= 0 and k1 <= 2:
                        Ks.append(k1)
                    if k2 >= 0 and k2 <= 2:
                        Ks.append(k2)
            Ks = list(set(Ks))
            Ks.sort()
            # compute the value over all breakpoints
            max_O = 0
            for k in Ks:
                O = 0
                for i in range(m):
                    for j in range(m):
                        # overlap between interval i and j at shift k
                        a_i = a[i]
                        a_j = a[j]
                        o_ij = max(0, target_L - abs((a_i - a_j) - k))
                        O += o_ij
                max_O = max(max_O, O)
            # also check k = 0 and k = 2
            if len(Ks) > 0 and Ks[0] != 0:
                O = 0
                for i in range(m):
                    for j in range(m):
                        a_i = a[i]
                        a_j = a[j]
                        o_ij = max(0, target_L - abs((a_i - a_j) - 0))
                        O += o_ij
                max_O = max(max_O, O)
            if len(Ks) > 0 and Ks[-1] != 2:
                O = 0
                for i in range(m):
                    for j in range(m):
                        a_i = a[i]
                        a_j = a[j]
                        o_ij = max(0, target_L - abs((a_i - a_j) - 2))
                        O += o_ij
                max_O = max(max_O, O)
            return max_O
        
        # Run a limited DIRECT-style search
        direct_budget = 2000
        trial_time = 0
        for _ in range(direct_budget):
            if time.time() - t0 > budget_s - 5:
                break
            if not boxes: break
            # pick the box with minimal lower bound
            best_box_idx = -1
            best_lower = float('inf')
            for i, box in enumerate(boxes):
                center = [(low + high) / 2 for low, high in box]
                l = m
                # get max O by computing at the center
                O_center = compute_Ok(center)
                lower_bound = O_center - l * max([high - low for low, high in box]) / 2
                if lower_bound < best_lower:
                    best_lower = lower_bound
                    best_box_idx = i
            if best_box_idx == -1: break
            # get box
            box = boxes.pop(best_box_idx)
            # if it's too small, process it directly
            max_width = max([high - low for low, high in box])
            if max_width < 1e-6:
                center = [(low + high) / 2 for low, high in box]
                O = compute_Ok(center)
                if O < best_M - 1e-10:
                    best_M = O
                    best_a = center
                continue
            # otherwise divide it
            longest_dim = max(enumerate(box), key=lambda x: x[1][1] - x[1][0])[0]
            low, high = box[longest_dim]
            mid = (low + high) / 2
            box1 = [list(box[i]) for i in range(m-1)]
            box2 = [list(box[i]) for i in range(m-1)]
            box1[longest_dim] = (low, mid)
            box2[longest_dim] = (mid, high)
            boxes.extend([box1, box2])
            n_box = len(boxes)
        # Phase B: DE refinement
        if best_a is not None:
            if time.time() - t0 > budget_s - 5:
                continue
            def obj(a_vec):
                return compute_Ok(a_vec)
            
            # DE parameters
            popsize = 10 * (m-1)
            mut = 0.8
            cross = 0.9
            maxiter = 1000
            restarts = 3
            
            # initial population
            initial_pop = []
            for i in range(min(popsize, 50)):
                # sample near the best found point with some noise
                a_perturbed = [0] * (m-1)
                for j in range(m-1):
                    if best_a[j] > 0:
                        a_perturbed[j] = np.clip(best_a[j] + np.random.normal(0, 0.001), 0, max_gap)
                    else:
                        a_perturbed[j] = 0
                initial_pop.append(a_perturbed)
            if len(initial_pop) < popsize:
                for i in range(popsize - len(initial_pop)):
                    a_rand = np.random.uniform(0, max_gap, (m-1))
                    initial_pop.append(a_rand)
            
            # DE core
            pop = initial_pop
            best_pop = None
            best_val = float('inf')
            for restart in range(restarts):
                if time.time() - t0 > budget_s - 5:
                    break
                last_best = best_val
                for iter in range(maxiter):
                    if time.time() - t0 > budget_s - 5:
                        break
                    new_pop = []
                    for i in range(len(pop)):
                        # select 3 different individuals
                        while True:
                            idxs = []
                            while len(idxs) < 3:
                                idx = np.random.randint(0, len(pop))
                                if idx not in idxs:
                                    idxs.append(idx)
                            if idxs[0] != i and idxs[1] != i and idxs[2] != i:
                                break
                        a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
                        # Mutation
                        mutant = [a[j] + mut * (b[j] - c[j]) for j in range(m-1)]
                        # check bounds
                        for j in range(m-1):
                            mutant[j] = np.clip(mutant[j], 0, max_gap)
                        # Crossover
                        trial = []
                        for j in range(m-1):
                            if np.random.random() < cross or j == 0:
                                trial.append(mutant[j])
                            else:
                                trial.append(pop[i][j])
                        # Projection
                        # make sure all are in order
                        trial = [max(0, trial[0])]
                        for j in range(1, len(trial)):
                            trial[j] = max(trial[j-1] + target_L, trial[j])
                        if len(trial) > 1:
                            max_allowed = 2 - m * target_L
                            if trial[-1] > max_allowed:
                                # decrease last few by equal amounts
                                diff = trial[-1] - max_allowed
                                for j in range(len(trial)-1, -1, -1):
                                    if diff <= 0: break
                                    reduce = min(diff, trial[j] - (trial[j-1] if j > 0 else 0))
                                    trial[j] -= reduce
                                    diff -= reduce
                        # Evaluate
                        val = obj(trial)
                        if val < best_val - 1e-10:
                            best_pop = trial[:]  # copy
                            best_val = val
                        if val < obj(pop[i]):
                            new_pop.append(trial)
                        else:
                            new_pop.append(pop[i])
                    pop = new_pop
                if best_val >= last_best * 0.999 and restart == 0:
                    break  # likely not improving
            # best_a is updated
            best_a = best_pop
        # Compute discretizations and polish
        if best_a is not None:
            # compute the original intervals
            a = [0] + best_a
            intervals = [(a[i], a[i] + target_L) for i in range(m)]
            
            # ensure ordering
            for i in range(1, m):
                a[i] = max(a[i], a[i-1] + target_L)
            
            # try discretizations
            for n in target_n_values:
                if time.time() - t0 > budget_s - 5:
                    break
                dx = 2.0 / n
                h = [0] * n
                # assign h[i] based on if x[i] is in any interval
                for i in range(n):
                    x = i * dx
                    for a_i, b_i in intervals:
                        if a_i <= x <= b_i:
                            h[i] = 1
                            break
                
                # Ensure integral constraint
                s = sum(h)
                if s != n // 2:
                    # adjust the h values to fix the integral
                    diff = n // 2 - s
                    if diff > 0:  # need to increase
                        # find 0s closest to interval boundaries
                        to_flip = []
                        for j in range(n):
                            x = j * dx
                            dist = float('inf')
                            for a_i, b_i in intervals:
                                d = min(abs(x - a_i), abs(x - b_i))
                                dist = min(dist, d)
                            to_flip.append((dist, j))
                        to_flip.sort()
                        for _, j in to_flip[:diff]:
                            h[j] = 1
                    elif diff < 0:  # need to decrease
                        to_flip = []
                        for j in range(n):
                            x = j * dx
                            dist = float('inf')
                            for a_i, b_i in intervals:
                                d = min(abs(x - a_i), abs(x - b_i))
                                dist = min(dist, d)
                            to_flip.append((dist, j))
                        to_flip.sort()
                        for _, j in to_flip[:abs(diff)]:
                            h[j] = 0
                
                # Compute C5
                corr = np.correlate(h, [1 - x for x in h], mode='full')
                c5 = max(corr) * dx
                
                # polish with swaps
                def score(i, k_star):
                    # compute the change in R[k_star] if flip h[i]
                    # R[k] = \sum_j h[j] * h[j+k]
                    # initial h[i] = 1, new h[i] = 0
                    # change = 
                    if i + k_star < len(h):
                        change = -h[i + k_star]  # affected by h[i+k_star]
                    else:
                        change = 0
                    if i >= k_star:
                        change -= h[i - k_star]
                    else:
                        change -= 0
                    return change
                
                c5_old = c5
                swap_budget = 200
                for _ in range(swap_budget):
                    if time.time() - t0 > budget_s - 5: break
                    # find the k* that minimizes correlation
                    corr = np.correlate(h, [1 - x for x in h], mode='full')
                    k_star = np.argmax(corr)
                    if k_star >= n:
                        k_star = k_star - n
                    change = 0
                    best_i = -1
                    for i in range(n):
                        # check if swap is beneficial
                        # we flip h[i] to 0 and see the effect on R[k*]
                        old_val = h[i]
                        h[i] = 1 - old_val  # flip
                        new_corr = np.correlate(h, [1 - x for x in h], mode='full')
                        new_c5 = max(new_corr) * dx
                        if new_c5 < c5:
                            # this was a good swap
                            c5 = new_c5
                            best_i = i
                        h[i] = old_val  # revert
                    # apply swap if any
                    if best_i != -1:
                        h[best_i] = 1 - h[best_i]
                        if c5 < c5_old - 1e-12:
                            c5_old = c5
                        else:
                            break
                    else:
                        break
                
                if c5 < best_C5 - 1e-10:
                    best_C5 = c5
                    best_h = h[:]
                    best_n = n
    
    return (best_h, best_C5, best_n)
