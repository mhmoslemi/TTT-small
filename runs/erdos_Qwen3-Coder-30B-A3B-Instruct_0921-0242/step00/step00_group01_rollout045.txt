import time
import numpy as np
from scipy import signal
import cvxpy as cp
from cvxpy import Minimize, Problem

def initial_h_values():
    # Return a reasonable starting point
    # Load the initial construction as an array of 0s and 1s
    # We predefine the values for the initial construction
    n = 47
    h = [0] * n
    for i in range(n):
        if (i % 4) < 2:
            h[i] = 1
    # normalize the sum to be n/2 = 23.5
    # To make it exactly 23.5, we can flip some values
    current_sum = sum(h)
    if current_sum < 23.5:
        for i in range(len(h)):
            if h[i] == 0:
                h[i] = 1
                current_sum += 1
                if current_sum == 23.5:
                    break
    elif current_sum > 23.5:
        for i in range(len(h)):
            if h[i] == 1:
                h[i] = 0
                current_sum -= 1
                if current_sum == 23.5:
                    break
    return h

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    c5 = max(np.correlate(h, 1-h, mode="full")) * dx
    return c5

def greedy_balanced_binary(n, S):
    """Construct binary sequence using a greedy approach to maximize min autocorrelation"""
    b = [0] * n
    selected = set()
    # Start with assigning 1 to the first element (symmetry breaking)
    b[0] = 1
    selected.add(0)
    
    # We will iteratively assign 1's to elements in a way to minimize worst overlap
    while len(selected) < S:
        best_i = -1
        min_delta = float('inf')
        for i in range(n):
            if i in selected:
                continue
            # Simulate adding this 1 and measure the maximum change in overlaps
            current_overlaps = compute_overlaps(b, n, selected, i)
            # Find the worst overlap after adding this 1
            worst_incr = min(current_overlaps) if current_overlaps else 0
            # Lower the worst overlap counter by creating worst effect possible
            delta = -worst_incr
            if delta < min_delta:
                min_delta = delta
                best_i = i
        if best_i == -1:
            raise RuntimeError("System error in greedy construction")
        selected.add(best_i)
        b[best_i] = 1
    return b

def compute_overlaps(b, n, selected, new_idx):
    # Current overlaps are stored as O_k = S - sum b[i]*b[i+k]
    overlapping_pairs = np.zeros(n)
    for k in range(n):
        for i in range(n):
            if b[i] and b[(i+k)%n]:
                overlapping_pairs[k] += 1
    # Compute changes if new_idx is added
    new_overlaps = [0] * n
    for k in range(n):
        # number of new pairs added at lag k due to new_idx
        # Check neighbors: (new_idx-k)%n and (new_idx+k)%n
        count = 0
        if (new_idx - k) % n in selected:
            count += 1
        if (new_idx + k) % n in selected:
            count += 1
        new_overlaps[k] = overlapping_pairs[k] + count
    # The new overlaps in the context of increased counts
    # O_k = S - Σ_i b[i]*b[i+k]
    # So we compute for each k the new O_k value
    # Since we are adding 1 at position new_idx,
    # The new O_k = cumul_O_k + number of old 1's at lag-k and lag+k
    # But that's only for pairs, so we calculate the difference in the direct overlap counts
    return [new_overlaps[k] for k in range(n)]

def swap_improvements(b, n, iterations=200):
    """Apply swaps to improve worst autocorrelation in binary sequence"""
    # Compute initial overlaps
    A = [0] * n
    for k in range(n):
        for i in range(n):
            if b[i] and b[(i + k) % n]:
                A[k] += 1
    worst_k = np.argmin(A)
    count = 0
    while count < iterations:
        improved = False
        for p in range(n):
            if not b[p]:
                continue
            for q in range(n):
                if b[q]:
                    continue
                # Compute change to overlap at worst_k if we swap b[p] and b[q]
                old_val = A[worst_k]
                # How many 1s are in positions (p+k)%n and (q+k)%n and (p-k)%n and (q-k)%n?
                # Count old: p at (p+k) and q at (q+k)
                old_p_pos_plus_k = b[(p + worst_k) % n]
                old_q_pos_plus_k = b[(q + worst_k) % n]
                new_p_pos_plus_k = b[(q + worst_k) % n]  # after swap
                new_q_pos_plus_k = b[(p + worst_k) % n]
                
                # Check the change in overlap at k
                delta_k = (new_p_pos_plus_k + new_q_pos_plus_k) - (old_p_pos_plus_k + old_q_pos_plus_k)
                
                if delta_k > 0:
                    # Check if we can improve A[worst_k]
                    # We also need to update all A[k'] differently
                    A = [0] * n
                    for k in range(n):
                        for i in range(n):
                            if b[i] and b[(i + k) % n]:
                                A[k] += 1
                    worst_k = np.argmin(A)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        count += 1
    return b

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start = time.time()
    
    # Use a coarse grid for MILP first, n0=120
    n0 = 120
    dx = 2.0 / n0
    S = n0 // 2  # Target count
    
    best_h = None
    best_c5 = float('inf')
    
    b_opt = None
    w_opt = float('inf')
    
    # Try to solve with MILP using CBC or other MIP solvers
    try:
        # Create MILP model using cvxpy
        # Variables: b[0], ..., b[n0-1] binary, w (real)
        b = cp.Variable(n0, boolean=True)
        w = cp.Variable()
        
        # Objective: minimize w
        objective = Minimize(w)

        # Constraints:
        # 1. sum(b) = S
        constraints = [cp.sum(b) == S, w >= 0, w <= S]
        
        # 2. Define z[i,k] = b[i]*b[(i+k)%n]
        # For each k, the constraint is hard to make exactly
        # So compute each line separately:
        # For each k, let y_k[i] = b[i]*b[i+k], enforce y_k[i] <= b[i], y_k[i] <= b[i+k], y_k[i] >= b[i] + b[i+k] - 1
        
        # Fast matrix computation of overlap counts for all k
        z = {}
        for k in range(n0):
            z[k] = [cp.Variable(n0, boolean=True) for _ in range(n0)]
            # For each i, enforce z[k][i] == b[i] * b[(i+k)%n]
            for i in range(n0):
                z[k][i].upper_bound = 1
                z[k][i].lower_bound = 0
                constraints.append(z[k][i] <= b[i])
                constraints.append(z[k][i] <= b[(i+k)%n0])
                constraints.append(z[k][i] >= b[i] + b[(i+k)%n0] - 1)
        
        # For each k, add constraint: w >= S - sum(z[k])
        for k in range(n0):
            constraints.append(w >= S - cp.sum(z[k]))
        
        # Symmetry breaking
        constraints.append(b[0] == 1)
        
        # The main optimization problem
        prob = Problem(objective, constraints)
        
        # Set timeout
        remaining = budget_s - (time.time() - start)
        if remaining < 20:
            raise RuntimeError("Too little time left for MIP")
        
        # Now we choose to solve with the available solver
        # SCIP is good choice, use CBC if not available (last resort)
        # For this implementation we will test only the adaptive path
        # Solve directly with limits
        try:
            prob.solve(solver='CBC', max_seconds=remaining, verbose=False)
            if prob.status == 'optimal':
                b_opt = b.value
                w_opt = w.value
        except Exception as e:
            print(f"CVXPY failed to solve MILP with error: {e}")

    except Exception as _e:
        pass
    
    if b_opt is not None and w_opt < best_c5:
        best_h = [1 if x >= 0.5 else 0 for x in b_opt]
        best_c5 = w_opt * dx
    
    # If not solved via MILP, fallback to greedy + swaps
    if best_h is None:
        try: 
            start_greedy = time.time()
            # Use the provided init_h if available
            h_init = initial_h_values()
            if len(h_init) != n0:
                # Normalize h_init to have sum = S
                total = sum(h_init)
                h_init = [x * S / total for x in h_init]
                # Convert to binary with rounding toward target
                b = [1 if (x >= 0.5) else 0 for x in h_init]
                # Adjust exact count
                current = sum(b)
                if current < S:
                    # Add 1's to lowest indices
                    for i in range(n0):
                        if b[i] == 0:
                            b[i] = 1
                            current += 1
                            if current == S:
                                break
                elif current > S:
                    # Remove 1's from highest indices
                    for i in range(n0-1, -1, -1):
                        if b[i] == 1:
                            b[i] = 0
                            current -= 1
                            if current == S:
                                break
            else:
                # Don't force normalization if same size
                b = [max(0, min(1, round(x))) for x in h_init]
                # Ensure exactly S ones
                current = sum(b)
                to_change = abs(current - S)
                if current < S:
                    sgn = 1
                else:
                    sgn = -1
                for i in range(n0):
                    if b[i] == (1 if sgn == 1 else 0) and to_change > 0:
                        b[i] = 1 - b[i]
                        to_change -= 1
                        if to_change == 0:
                            break
            # Ensure exactly S ones
            if sum(b) != S:
                raise ValueError(f"Sum is not {S}")

            # Apply greedy logic (simplified: create from seed)
            try:
                # Greedy accurate loop
                greedy_b = greedy_balanced_binary(n0, S)
                # Measure initial quality of greedy_b
                A_init = [0] * n0
                for k in range(n0):
                    for i in range(n0):
                        if greedy_b[i] and greedy_b[(i+k)%n0]:
                            A_init[k] += 1
                greedy_c5 = (S - min(A_init)) * dx
                if greedy_c5 < best_c5:
                    best_c5 = greedy_c5
                    best_h = greedy_b
            except:
                pass
                
            # If fallback, see if swaps help
            try:
                improved_b = swap_improvements(best_h, n0)
                A_ims = [0] * n0
                for k in range(n0):
                    for i in range(n0):
                        if improved_b[i] and improved_b[(i+k)%n0]:
                            A_ims[k] += 1
                swap_c5 = (S - min(A_ims)) * dx
                if swap_c5 < best_c5:
                    best_c5 = swap_c5
                    best_h = improved_b
            except:
                pass
        except Exception as _e:
            pass
    
    # If still no solution, create a simple binary pattern
    if best_h is None:
        # Simple "staggered" binary pattern
        best_h = [i % 2 for i in range(n0)]
        # Adjust to sum = S
        current_sum = sum(best_h)
        if current_sum < S:
            # Change some 0s to 1s
            for i in range(n0):
                if best_h[i] == 0 and current_sum < S:
                    best_h[i] = 1
                    current_sum += 1
        elif current_sum > S:
            # Change some 1s to 0s
            for i in range(n0-1, -1, -1):
                if best_h[i] == 1 and current_sum > S:
                    best_h[i] = 0
                    current_sum -= 1
        # Determine overlap
        A = [0] * n0
        for k in range(n0):
            for i in range(n0):
                if best_h[i] and best_h[(i+k)%n0]:
                    A[k] += 1
        best_c5 = (S - min(A)) * dx

    # Duplication step (go to target size like N = 360)
    target_n = 360 # or max 1000
    n_multiple = target_n // n0  # integer division
    h_duplicated = [best_h[i // n_multiple] for i in range(target_n)]
    
    # Final check on integrity of h_duplicated
    check_sum = sum(h_duplicated)
    if abs(check_sum - target_n/2) > 1e-8:
        # Re-adjust to target sum
        h_duplicated = [int(i > 0.5) for i in h_duplicated]
        adjust_sum = sum(h_duplicated)
        if adjust_sum < target_n/2:
            for i in range(target_n):
                if h_duplicated[i] == 0:
                    h_duplicated[i] = 1
                    adjust_sum += 1
                    if adjust_sum >= target_n/2:
                        break
        elif adjust_sum > target_n/2:
            for i in range(target_n-1, -1, -1):
                if h_duplicated[i] == 1:
                    h_duplicated[i] = 0
                    adjust_sum -= 1
                    if adjust_sum <= target_n/2:
                        break

    # Recompute final value
    final_c5 = evaluate_erdos_solution(h_duplicated)
    final_c5 = max(0, final_c5)  # Enforce non-negative

    return (h_duplicated, final_c5, len(h_duplicated))
