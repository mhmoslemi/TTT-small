import numpy as np
import time
from scipy import signal
import math

def project_simplex_box(v, total, lo=0., hi=1.):
    """Project vector v onto box-constrained simplex with sum = total"""
    v = np.asarray(v)
    # Sort values in descending order
    v_sorted = np.sort(v)[::-1]
    # Find the threshold
    cumsum = np.cumsum(v_sorted)
    n = len(v)
    theta = (cumsum - total) / np.arange(1, n + 1)
    # Find largest index such that v_sorted[i] > theta[i]
    idx = -1
    for i in range(n):
        if v_sorted[i] > theta[i]:
            idx = i
    if idx == -1:
        # All projected to 0 or 1
        result = np.clip(v, lo, hi)
        result = result * (total / np.sum(result))
    else:
        # Apply threshold
        th = theta[idx]
        result = np.clip(v - th, lo, hi)
    return result

def compute_c5(h, dx):
    """Compute C5 using FFT for correlation"""
    h = np.asarray(h)
    corr = np.real(signal.fftconvolve(h, np.flip(1 - h), mode='full'))
    # Take the non-repeated ones (full convolution)
    # correlation length is 2*n - 1
    n = len(h)
    # take the middle elements (the valid ones that correspond to
    # "full" mode with the full range) of the correlation
    # but we only want the valid range which is from 0 to 2*n-2
    # but the maximum computed from the valid lags 0 to n-1
    # but shift to get the first element of full convolution in middle
    # Let's take the odd part of convolution
    # Convolution with 'full' creates 2n-1 elements
    # last n-1 elements are from h[0] at lag 1 to h[n-1] at lag n-1
    # first n-1 are from h[n-1] at lag -n+1 to h[0] at lag -1
    # middle element is at lag 0 (which is 0 shift condition)
    # The important part is to get the max of correlation: 
    # unnegate the negative lags (stored at the beginning) 
    # and the positive.
    # Shift so that lag 0 is in the middle of the full convolution
    mid = len(corr) // 2
    corr_data = corr[mid - n + 1 : mid + n]
    max_val = np.max(corr_data) * dx
    return max_val

def worst_lag(h):
    """Return the worst lag and its overlap value"""
    h = np.asarray(h)
    n = len(h)
    corr = np.real(signal.fftconvolve(h, np.flip(h), mode='full'))
    # Again, we want to focus on lags 0 to n-1, even if some others are also there.
    # Usual for a function, lag k >= 0 means h[i] * h[i+k]
    # But for correlation center, we take the middle of full conv
    mid = len(corr) // 2
    # The actual useful lags range from 0 to n (from -n+1 to n-1)
    # The mode='full' puts all the entries where lag goes from -n+1 to n-1
    # So the overlap values for lags k=0 to n-1 range from index mid - (n-1) to mid (which is mid + (n-1) - 1) 
    corr_data = corr[mid - (n-1) : mid + n]
    max_o = np.max(corr_data)
    k_max = np.argmax(corr_data)
    # Turn k_max into the correct lag
    lag = k_max - (n - 1)
    # If lag < 0 then we need to shift 
    # But `k_max` indexes into `-n+1` to `n-1`, i.e., lags from `-n+1` (from index 0) to `n-1` (index 2n-2).
    # So we convert the index k_max to lag
    # lag = k_max - (n - 1) to select from 0 to n-1 offsets.
    # actually make sure to compute overlap for k = 0 to n-1, just for the valid shifts
    # indices 0 to 2n-2 for full convolution
    # let's do it via loop instead for clarity
    # compute the correlation with just h and 1-h over lags 0 to n-1 properly
    corr_all = []
    for k in range(n):
        # Compute sum h[i] * (1 - h[i+k]) 
        # this is looped and assumes wrapping of indices?
        overlap_k = 0.
        for i in range(n):
            j = (i + k) % n
            overlap_k += h[i] * (1 - h[j])
        corr_all.append(overlap_k)
    val = np.max(corr_all)
    k_max = np.argmax(corr_all)
    return k_max, val

def gradient_worst(h, k):
    """Compute gradient of overlap k w.r.t. h_i"""
    h = np.asarray(h)
    n = len(h)
    grad = np.zeros_like(h)
    for i in range(n):
        j = (i + k) % n
        grad[i] += h[j]  # gradient w.r.t. h[i] involving h[i+k]
        j = (i - k) % n
        grad[i] += h[j]  # gradient w.r.t. h[i] involving h[i-k]
    return grad

def optimize(n, time_limit, seed=42):
    np.random.seed(seed)
    dx = 2.0 / n
    # Initialize h
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values).copy()
    else:
        # Random initialization within constraints
        h = np.random.rand(n)
        # Make sure sum = n/2 (dx = 2/n, so dx*sum = 1)
        h *= (n / 2) / np.sum(h)
    h = np.clip(h, 0, 1)

    # Project to simplex with constraint sum = n/2
    h = project_simplex_box(h, n / 2.0, 0, 1)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    start_time = time.time()
    step_size = 0.01
    max_it = 100000
    count = 0
    
    for i in range(max_it):
        if time.time() - start_time > time_limit:
            break
        count += 1
        k_max, overlap_val = worst_lag(h)
        # Compute gradient of this overlap
        g = gradient_worst(h, k_max)
        # Gradient step (we're minimizing max_overlap -> reduce overlap)
        h_new = h - step_size * g
        # Project onto box and simplex
        h_new = project_simplex_box(h_new, n / 2.0, 0, 1)
        # Evaluate new value
        new_c5 = compute_c5(h_new, dx)
        # Accept if better or accept with small probability (simulated annealing)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new.copy()
            step_size *= 1.05  # adaptive, increase if improving
        else:
            step_size *= 0.99  # adaptive, reduce otherwise
            h = h_new.copy()
    return best_h, best_c5

def run(seed=42, budget_s=600):
    # Enumerate possible sizes
    sizes = [150, 250, 350]  # All less than 1000
    # Budget allocation
    # Allocate more budget to each size than to number of runs (for better fill)
    n_tests = 100
    time_per_iter = float(budget_s) / (n_tests * 3)
    run_start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    for n in sizes:
        if time.time() - run_start > budget_s:
            break
        h, c5 = optimize(n, time_per_iter, seed + n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n
    # Do final refinement with a finer grid, but don't use too much time
    if time.time() - run_start < budget_s - 10:
        if best_n is not None:
            n_refine = min(800, best_n * 2)
            if n_refine < 1000:
                _, c5_after = optimize(n_refine, budget_s - (time.time() - run_start) - 5, seed + 1000)
                if c5_after < best_c5:
                    best_c5 = c5_after
                    # Redo the full optimization to get the right h values (for size = n_refine)
                    # But if we're under time already, short-cut suboptimal final refinement
        return (
            best_h.tolist(),
            best_c5,
            best_n
        )
    return (
        best_h.tolist() if best_h is not None else [0.5]*150,
        best_c5,
        best_n
    )
