import numpy as np

def project_capped_simplex(v, total=1.0):
    """Project vector v onto the capped simplex: 0 <= v[i] <= 1, sum(v) = total."""
    v = np.asarray(v)
    n = len(v)
    # bisection on lambda
    lbd = 0.0
    ubd = 1.0
    while True:
        if ubd - lbd < 1e-15:
            break
        lmb = (lbd + ubd) / 2.0
        projected = np.clip(v - lmb, 0.0, 1.0)
        sum_proj = np.sum(projected)
        if sum_proj >= total:
            lbd = lmb
        else:
            ubd = lmb
    lmb = (lbd + ubd) / 2.0
    return np.clip(v - lmb, 0.0, 1.0)

def auto_corr(h):
    """Compute the auto-correlation O_k = sum_i h[i] * h[i+k] for all k."""
    n = len(h)
    # Use zero-padded fft
    h_pad = np.pad(h, (0, n), mode='constant')
    corr = fftconvolve(h_pad, h[::-1], mode='valid')
    return corr[:n]

def grad_F(beta, h, O, w):
    """Compute the gradient of the smooth surrogate F_beta(h)."""
    n = len(h)
    # w is a probability distribution
    # g_k = sum_i w_k * (h_{i+k} + h_{i-k})
    # Do this via fft. For even convolution: h*r, where r is reversed h
    w_pad = np.pad(w, (0, n), mode='constant')
    h_pad = np.pad(h, (0, n), mode='constant')
    # Go through the 2 parts:
    # One from h_{i+k}, the other from h_{i-k}
    # Would be (w * h_reverse) + (w_reverse * h) = 
    g1 = fftconvolve(w_pad, h_pad[::-1], mode='valid')[:n]
    g2 = fftconvolve(w_pad[::-1], h_pad, mode='valid')[n:]
    # Symmetrize
    g = g1 + g2
    # Can do with proper zero padding: 2n then trunc.
    hlo = np.pad(h, (n, 0), mode='constant')
    hhi = np.pad(h, (0, n), mode='constant')
    g = fftconvolve(w, hlo, mode='valid')[:n]
    g += fftconvolve(w[::-1], hhi, mode='valid')[n:]
    return -g

def evaluate_C5(h, dx):
    """Evaluate C5 from the current h."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def optimize_one_start(n, beta_schedule, time_limit, seed):
    """Optimize with continuation method. Return best h and C5."""
    np.random.seed(seed)
    # Init
    h = np.random.exponential(1.0, n)
    h = (h / np.sum(h)) * (n / 2.0)
    h = np.clip(h, 0, 1)
    h = project_capped_simplex(h, total=n/2)
    best_C5 = evaluate_C5(h, 2.0 / n)
    best_h = h.copy()
    start_t = time.time()
    
    # For each beta
    for beta_level, beta in enumerate(beta_schedule):
        if time.time() - start_t > time_limit * 0.7:
            break
        # Inner loop
        for it in range(5000):
            if time.time() - start_t > time_limit * 0.7:
                break
                
            # Compute all O_k
            O = auto_corr(h)
            # Compute w_k = exp(beta * O_k) / sum exp(beta * O_k)
            O_beta = beta * O
            exp_O = np.exp(O_beta)
            w = exp_O / np.sum(exp_O)
            # Compute gradient of F_beta
            g = grad_F(beta, h, O, w)
            
            # Line search
            alpha = 0.1
            iter_max = 20
            for _ in range(iter_max):
                h_new = h - alpha * g
                h_new = project_capped_simplex(h_new, total=n/2)
                C5_new = evaluate_C5(h_new, 2.0 / n)
                C5_old = evaluate_C5(h, 2.0 / n)
                if C5_new <= C5_old - 1e-4 * alpha * np.linalg.norm(g)**2:
                    h = h_new
                    break
                alpha /= 2.0
                
            # Update best
            C5_new = evaluate_C5(h, 2.0 / n)
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
    return best_h, best_C5
