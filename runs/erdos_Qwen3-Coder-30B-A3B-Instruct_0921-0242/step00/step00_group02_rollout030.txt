import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar


def project_simplex(h, target_sum):
    """Project h onto the simplex {x: sum(x) = target_sum, 0 <= x <= 1}."""
    n = len(h)
    h_clipped = np.clip(h, 0, 1)
    deficit = target_sum - np.sum(h_clipped)
    F = np.where((h_clipped > 0) & (h_clipped < 1))[0]
    if len(F) == 0:
        return h_clipped

    h_proj = h_clipped.copy()
    while abs(deficit) > 1e-12 and len(F) > 0:
        delta = deficit / len(F)
        for i in F[:]:
            h_proj[i] += delta
            if h_proj[i] <= 0:
                h_proj[i] = 0
                F = F[F != i]
            elif h_proj[i] >= 1:
                h_proj[i] = 1
                F = F[F != i]
        deficit = target_sum - np.sum(h_proj)
    return h_proj


def autocorrelations(h):
    """Compute linear autocorrelations r_k(h) for k=1 to n-1."""
    n = len(h)
    # Zero-pad to avoid circular effects
    h_extended = np.zeros(2 * n)
    h_extended[:n] = h
    H = fft(h_extended)
    corr = ifft(H * np.conj(H)).real
    # Return only non-circular autocorrelations
    r = corr[1:n]
    return r


def soft_min_grad(h, alpha):
    """Compute gradient of soft minimum of autocorrelations."""
    n = len(h)
    # Compute autocorrelations
    r = autocorrelations(h)
    # Soft weights using log-sum-exp trick
    r_scaled = -alpha * r
    r_max = np.max(r_scaled)
    exp_r = np.exp(r_scaled - r_max)
    w = exp_r / np.sum(exp_r)
    # Gradient using convolutions
    h_padded = np.zeros(2 * n)
    h_padded[:n] = h
    w_padded = np.zeros(2 * n)
    w_padded[:n] = w
    w_reversed = w_padded[:n][::-1]
    w_reversed_padded = np.zeros(2 * n)
    w_reversed_padded[:n] = w_reversed
    # Convolution: (w * h) + (reverse(w) * h)
    grad = ifft(fft(w_padded) * fft(h_padded)).real
    grad_rev = ifft(fft(w_reversed_padded) * fft(h_padded)).real
    gradient = grad[:n] + grad_rev[:n]
    return gradient


def line_search(h, grad, alpha, step_size=1.0, max_iter=10):
    """Backtracking line search for increasing S_alpha."""
    def eval_alpha(t):
        h_new = project_simplex(h + t * grad, len(h) / 2)
        return -np.log(np.sum(np.exp(-alpha * autocorrelations(h_new))) / len(autocorrelations(h_new)))

    # binary search for good step_size.
    low, high = 0.0, step_size
    for _ in range(max_iter):
        mid = (low + high) / 2
        val = eval_alpha(mid)
        if np.isnan(val) or val > eval_alpha(0):
            high = mid
        else:
            low = mid
    return low


def projected_gradient_ascent(h0, alpha, max_iter=500, tol=1e-8, beta=0.9):
    """Perform projected gradient ascent on S_alpha."""
    h = h0.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_r = np.min(autocorrelations(h))
    last_improvement = 0

    for i in range(max_iter):
        # Accelerated step
        y = h + beta * (h - h_prev)
        g = soft_min_grad(y, alpha)
        step_size = line_search(y, g, alpha)
        h_new = project_simplex(y + step_size * g, len(h) / 2)
        h_prev = h.copy()
        h = h_new.copy()

        r = np.min(autocorrelations(h))
        if r > best_r:
            best_r = r
            best_h = h.copy()

        if i > 20:
            if abs(last_improvement) < tol:
                break
        last_improvement = r - best_r
    return best_h, best_r


def round_to_step_function(h):
    """Greedy rounding to binary so that sum==target_sum."""
    n = len(h)
    target_sum = n / 2
    indices = np.argsort(h)[::-1]
    h_new = np.zeros(n)
    for i in indices:
        if np.sum(h_new) < target_sum:
            h_new[i] = 1
        else:
            break
    return h_new
