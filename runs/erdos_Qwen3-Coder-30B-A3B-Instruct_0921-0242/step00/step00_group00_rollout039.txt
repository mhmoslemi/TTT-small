import cvxpy as cp
from scipy.sparse import diags
import time

def run(seed=42, budget_s=600, **kwargs):

    eps = 5e-4
    max_iters = 4000
    n_list = [120, 200, 300]
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in n_list:
        # Skip if too little time left
        if time.time() - start_time > budget_s - 30:
            break
            
        # Create shift matrices
        M_list = []
        dx = 2.0 / n
        
        for k in range(n):
            diag = np.ones(n - k)
            if k == 0:
                M = diags(diag, 0, shape=(n, n), format='csr')
            else:
                M = diags(diag, k, shape=(n, n), format='csr')
            M_list.append(M)
        
        # SDP formulation
        X = cp.Variable((n, n), symmetric=True)
        h = cp.Variable(n)
        L = cp.Variable()
        
        constraints = [
            X >> 0,  # Positive semidefinite
            0 <= h, h <= 1,  # Box constraints
            cp.sum(h) == n / 2,  # Sum constraint
            X[cp.arange(n), cp.arange(n)] == h,  # Diagonal constraint
        ]
        
        # Add X_ij <= min(h_i, h_j) constraints
        for i in range(n):
            for j in range(i+1, n):
                constraints.append(X[i,j] <= h[i])
                constraints.append(X[i,j] <= h[j])
                constraints.append(X[i,j] >= 0)
        
        # Add the main constraints
        for M in M_list:
            constraints.append(cp.trace(M @ X) >= L)
            
        prob = cp.Problem(cp.Maximize(L), constraints)
        
        # Solve the SDP
        try:
            prob.solve(solver=cp.SCS, eps=eps, max_iters=max_iters, verbose=False)
        except:
            continue

        if prob.status != cp.OPTIMAL:

                h_candidate = np.sqrt(lambda_1) * np.abs(v)

                    # Bisection

                        return np.sum(np.clip(v - theta, 0, 1)) - target_sum

                            a = mid
                        else:
                            b = mid
                    theta_star = (a + b) / 2
                    return np.clip(v - theta_star, 0, 1)

                conv = np.convolve(h_feas, 1 - h_feas, mode='full')
