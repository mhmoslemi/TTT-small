import numpy as np
import time

def evaluate(h):
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The overlap integral C_5 is max_k of each correlation value
    return np.max(corr) * dx

def project_feasible(x, target):
    # Project onto {0 <= h <= 1, sum(h) = target}
    # Water-filling approach
    x = np.array(x, dtype=float)
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target) < 1e-10:
        return x
    # Iteratively adjust
    tol = 1e-10
    max_iter = 100
    for _ in range(max_iter):
        deficit = target - current_sum
        if abs(deficit) < tol:
            break
        # Find free indices
        free = (x < 1) & (x > 0)
        free_count = np.sum(free)
        if free_count == 0:
            break
        # distribute uniformly
        delta = deficit / free_count
        if delta > 0:
            # Increase x[i] where possible
            x[free] = np.minimum(1.0, x[free] + delta)
        else:
            # Decrease x[i] where possible
            x[free] = np.maximum(0.0, x[free] + delta)
        current_sum = np.sum(x)
    return x

def active_shifts(h, eps=1e-9):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    # full correlation has 2*n - 1 elements indexed from -(n-1) to (n-1)
    # indices in corr are 0 to 2n-2
    active = []
    for i in range(len(corr)):
        if corr[i] >= max_val - eps:
            active.append(i - (len(h) - 1))
    return active

def gradient(h, ks):
    # compute gradient of maximum overlap over active shifts ks
    n = len(h)
    g = np.zeros(n)
    dx = 2.0 / n
    for k in ks:
        for i in range(n):
            i_plus_k = i + k
            i_minus_k = i - k
            # Set out-of-band values to 0
            term1 = 1 - h[i_plus_k] if 0 <= i_plus_k < n else 0
            term2 = h[i_minus_k] if 0 <= i_minus_k < n else 0
            g[i] += (term1 - term2)
    g /= len(ks)
    return g

def line_search(h, g, t_current, budget_s, t0):
    steps = [0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1]
    best_h = h.copy()
    best_val = t_current
    dx = 2.0 / len(h)
    for alpha in steps:
        h_new = h - alpha * g
        h_new = project_feasible(h_new, 1.0)
        new_val = evaluate(h_new)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
    return best_h, best_val

def local_optimize(h0, max_iter, t_end):
    # Project to ensure initial feasibility
    dx = 2.0 / len(h0)
    h = project_feasible(h0, 1.0)
    best_h = h.copy()
    best_val = evaluate(h)
    t_start = time.time()
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        ks = active_shifts(h)
        g = gradient(h, ks)
        h_new, new_val = line_search(h, g, evaluate(h), 600, t_start)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        h = h_new.copy()
        # Note: we could implement a convergence test here but not required
    return best_h, best_val

def anneal(h0, t_end, T0=0.01, decay=0.995, num_steps=5000):
    h = h0.copy()
    current_val = evaluate(h)
    T = T0
    dx = 2.0 / len(h)
    for _ in range(num_steps):
        if time.time() > t_end:
            break
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        if i == j:
            continue
        delta = np.random.uniform(-0.1, 0.1)
        h_save = h.copy()
        old_val = current_val
        h[i] += delta
        h[j] -= delta
        h = project_feasible(h, 1.0)
        new_val = evaluate(h)
        if new_val < old_val:
            current_val = new_val
        else:
            # Accept with probability depending on T
            if np.random.rand() < np.exp(-(new_val - old_val) / T):
                current_val = new_val
            else:
                h = h_save
        T *= decay
    return h, current_val

def random_feasible(n):
    x = np.random.rand(n)
    x = x / np.sum(x) * 1.0  # sum will be 1.0
    x = project_feasible(x, 1.0)
    return x

    # Create the auxiliary variables for the MILP linearization
    # For each k = 1...n-1, define auxiliary a[i,k] = h[i]*h[i+k]
    # and add constraints
    dx = 2.0 / n
    # We want max_k sum_i h[i]*h[i+k] ≤ (1 - t) / dx
    # So to minimize t (subject to sum_i h[i]*h[i+k] ≥ (1-t)/dx for each k)
    t = cp.Variable()
    # Add auxiliary variables and constraints per k
    constraints = [t >= 0]
    # Loop over all possible shifts k in [1, n-1]
    shift_constrs = []
    for k in range(1, n):
        # Create auxiliary variable per i
        # Note: for each k, we should find that a[i,k] = h[i]*h[i+k] for valid i+k
        # A i^th helper to compute l[k][i] = h[i]*h[i+k] l is 1 if both are 1 else 0
        # For j such that i+k = j, let a[i,k] be a constraint that represents the product
        # We do the standard linearization: a >= h[i] + h[i+k] - 1, a <= h[i], a <= h[i+k]
        vars_a = {}
        for i in range(n):
            if i + k < n:
                a = cp.Variable(boolean=True)
                vars_a[i] = a
                constraints.append(a >= h[i] + h[i + k] - 1)
                constraints.append(a <= h[i])
                constraints.append(a <= h[i + k])
                # And constraint to force sum(i,k) >= (1-t)/dx
                shift_constrs.append(a >= 0)
        if vars_a:
            # Sum over valid i of a[i,k] to use in optimization
            # To simplify, just add the constraint for k that sum of a[i,k] >= (1-t)/dx
            sum_a = cp.sum(list(vars_a.values()))
            # Note the right side is corrected: 
            # 't' here assumes we want a linear constraint for using in obj minimization
            # In the discrete case we don't want to do that directly with an equality
            # Actually, since the max is over shifts and we want to minimize max_overlap,
            # This needs epigraph formulation for "min max_k ..." effectively:
            # There's a fixed target, we want to know min always sum over shift >= (1-t)/dx
            # But we don't know target yet.
            # A cleaner approach:
            # Define for each k: s_k = sum_{i=0}^{n+k-1} h[i] * h[i+k]
            # We minimize t such that for all k, s_k >= (1-t)/dx
            # i.e., min t subject to s_k >= (1-t)/dx = 1/dx - t/dx
            # i.e., min t such that s_k + t/dx >= 1/dx for all k
            # Therefore: 
            # Z = s_k + t/dx
            # minimize sum_k (Z - 1/dx) = sum_k Z_k  - n * (1/dx) and Z_k >= 1/dx
            # This becomes a convex problem via epigraph and we want to minimize t directly
            # So, hard way: expand all of them in terms of binary products (but no expressions)
            # Let's view this as: 
            # sum_{i,k} h[i]*h[i+k] >= (1-t)/dx  
            # This is exact with MILP but we note that there are <= 2*n many terms:
            # So construct auxiliary sums for all k, where k running from 1 to n-1
            # but the variable setup remains simple for construction.
            # From a numerical point of view, defining it properly:
            # min t such that 1 - t/dx <= s_k for all k.
            # But let us do it cleaner: let's just make this
            # Ine: sum_k (sum_i h[i]*h[i+k]) >= n*(1-t)/dx  # Since max constraint where we want to make the worst value smaller
            constraints.append(sum_a >= (1.0 - t) / dx)
    # Return the problem with constraints
    # Actually, let's simplify to optimize the main min t such that for _all_ shifts,
    # the sums are large enough.
    # git better formulation: For any shift k:
    # Let A_k = \sum_i h[i] * h[i+k]. Constraint: A_k >= (1-t)/dx
    # Minimize t subject to A_k >= 1/dx - t/dx
    # This gives: A_k + t/dx >= 1/dx 
    # As: t + d*A_k >= d => add t as variable
    # This will end up using target := (1 - t)/dx
    problem = cp.Problem(cp.Minimize(t), constraints)
    try:
        problem.solve(solver=solver, verbose=False)
        if problem.status in ['optimal', 'optimal_inaccurate']:  # Gets returned sometimes in CVXOPT

            C5_bound = 1.0 - np.sum(h_sol) * dx + t.value * dx
            return h_sol, C5_bound
        else:
            return None, None
    except:
        return None, None
