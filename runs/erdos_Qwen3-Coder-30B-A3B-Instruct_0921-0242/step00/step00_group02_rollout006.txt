import numpy as np
import time
import random
from scipy.signal import correlate

def correlation_vector(b):
    """Compute the autocorrelation vector R_k for all k >= 1."""
    # b is a binary vector: 0s and 1s
    # Full correlation gives us n + (n-1) + ... + 1 terms
    # So we start from k=1 (the first actual correlation shift)
    R = correlate(b, b, mode='full')
    # R[0] is the correlation at shift 0, skip it.
    # The rest is R_1, ..., R_{n-1} where R_k = \sum_i b_i b_{i+k}
    # We want [R_1, ..., R_{n-1}] which is R[n:] (n elements)
    return R[len(b):]

def update_correlation_delta(R, b, i, j):
    """
    Update the correlation vector R after swapping b[i] and b[j].
    This is O(n) in runtime.
    """
    old_R_min = min(R)
    old_k_star = np.argmin(R)
    delta_R = np.zeros_like(R)
    n = len(b)
    
    # For each shift k, calculate the change in R_k due to the swap
    for k in range(1, n):
        # Subtract 1 if b[i-k] == 1 (i-k would now be 0)
        if i - k >= 0 and b[i - k] == 1:
            delta_R[k - 1] -= 1
        # Subtract 1 if b[i+k] == 1 (i+k would now be 0)
        if i + k < n and b[i + k] == 1:
            delta_R[k - 1] -= 1
        # Add 1 if b[j-k] == 1 (j-k would now be 1)
        if j - k >= 0 and b[j - k] == 1:
            delta_R[k - 1] += 1
        # Add 1 if b[j+k] == 1 (j+k would now be 1)
        if j + k < n and b[j + k] == 1:
            delta_R[k - 1] += 1
    # Apply the change
    R += delta_R
    # Return the new minimum and its index
    new_R_min = min(R)
    new_k_star = np.argmin(R)
    return new_R_min, new_k_star

def hill_climb_single_step(b, R, swap_ensemble_size=200, escape_prob=0.05):
    """Single step of hill climbing to maximize t = min(R_k)"""
    n = len(b)
    t = min(R)
    t_star_idx = np.argmin(R)
    
    candidates = []
    # Try some random swaps
    for _ in range(swap_ensemble_size):
        # Try to find a good or random swap
        while True:
            i = random.randint(0, n - 1)
            j = random.randint(0, n - 1)
            if b[i] == 1 and b[j] == 0:
                break
        # Compute change in R_star under this swap
        temp_R = R.copy()
        delta_R = np.zeros_like(R)
        # For each shift k
        for k in range(1, n):
            # Subtract 1 if b[i-k] == 1 (i-k would now be 0)
            if i - k >= 0 and b[i - k] == 1:
                delta_R[k - 1] -= 1
            # Subtract 1 if b[i+k] == 1 (i+k would now be 0)
            if i + k < n and b[i + k] == 1:
                delta_R[k - 1] -= 1
            # Add 1 if b[j-k] == 1 (j-k would now be 1)
            if j - k >= 0 and b[j - k] == 1:
                delta_R[k - 1] += 1
            # Add 1 if b[j+k] == 1 (j+k would now be 1)
            if j + k < n and b[j + k] == 1:
                delta_R[k - 1] += 1
        temp_R += delta_R
        delta_t = min(temp_R) - t
        if delta_t > 0:
            candidates.append((i, j, delta_t))
    
    if candidates:
        best_i, best_j, _ = max(candidates, key=lambda x: x[2])
    else:
        if random.random() < escape_prob:
            # Random move to escape
            i = random.randint(0, n - 1)
            j = random.randint(0, n - 1)
            while b[i] == 0 or b[j] == 1:
                i = random.randint(0, n - 1)
                j = random.randint(0, n - 1)
                # Reduce the chance this will be called
            best_i, best_j = i, j
        else:
            return False  # No gain achievable
   
    # Perform the swap and update R in place
    b[best_i], b[best_j] = b[best_j], b[best_i]
    # Update R efficiently
    delta_R = np.zeros_like(R)
    for k in range(1, n):
        # Subtract 1 if b[best_i-k] == 1 (best_i-k would now be 0)
        if best_i - k >= 0 and b[best_i - k] == 1:
            delta_R[k - 1] -= 1
        # Subtract 1 if b[best_i+k] == 1 (best_i+k would now be 0)
        if best_i + k < n and b[best_i + k] == 1:
            delta_R[k - 1] -= 1
        # Add 1 if b[best_j-k] == 1 (best_j-k would now be 1)
        if best_j - k >= 0 and b[best_j - k] == 1:
            delta_R[k - 1] += 1
        # Add 1 if b[best_j+k] == 1 (best_j+k would now be 1)
        if best_j + k < n and b[best_j + k] == 1:
            delta_R[k - 1] += 1
    R += delta_R
    return True

def generate_random_binary_vector(n, weight):
    indices = np.random.permutation(n)[:weight]
    v = np.zeros(n, dtype=np.uint8)
    v[indices] = 1
    return v

def binary_mutation(b, num_swaps=1):
    n = len(b)
    for _ in range(num_swaps):
        i = random.randint(0, n - 1)
        j = random.randint(0, n - 1)
        while b[i] == 0 or b[j] == 1:
            i = random.randint(0, n - 1)
            j = random.randint(0, n - 1)
        b[i], b[j] = b[j], b[i]
    return b

def crossover(parent1, parent2):
    #union of support sets
    support1 = np.where(parent1 == 1)[0]
    support2 = np.where(parent2 == 1)[0]
    # Ensure we can handle non-uniform support
    union_support = np.union1d(support1, support2)
    if len(union_support) > len(parent1) // 2:
        # Remove excess randomly from union
        diff = len(union_support) - len(parent1) // 2
        indices_to_remove = np.random.choice(union_support, diff, replace=False)
        union_support = np.setdiff1d(union_support, indices_to_remove)
    elif len(union_support) < len(parent1) // 2:
        # Add more indices from complement
        complement = np.where(parent1 == 0)[0]
        diff = len(parent1) // 2 - len(union_support)
        indices_to_add = np.random.choice(complement, diff, replace=False)
        union_support = np.union1d(union_support, indices_to_add)
    # Ensure 0s and 1s
    child = np.zeros(len(parent1), dtype=np.uint8)
    child[union_support] = 1
    return child

def iterate_population(population, R_list, fitness_list, elite_percent=0.1, keep_elite=True):
    # Tournament selection
    p = len(population)
    new_pop = [] if not keep_elite else [population[np.argmax(fitness_list)].copy()]
    new_R = [] if not keep_elite else [R_list[np.argmax(fitness_list)].copy()]
    new_fitness = [] if not keep_elite else [np.max(fitness_list)]

    # Select and operate on pairs
    num_selections = (p - len(new_pop)) // 2
    
    for _ in range(num_selections):
        # Select parents
        parents = []
        for _ in range(2):
            # Tournament selection
            tournament_size = 3
            best_idx = random.randint(0, len(fitness_list)-1)
            for _ in range(tournament_size - 1):
                idx = random.randint(0, len(fitness_list)-1)
                if fitness_list[idx] > fitness_list[best_idx]:
                    best_idx = idx
            parents.append(population[best_idx])
            
        # Crossover with probability 0.8
        child = parents[0].copy() if random.random() > 0.8 else crossover(parents[0], parents[1])
        # Mutation: swap 1-2 times
        num_muts = random.randint(1, 2)
        child = binary_mutation(child, num_muts)
        # Local improvement
        R = correlation_vector(child)
        t = min(R)
        improvement_iterations = 0
        while hill_climb_single_step(child, R) and improvement_iterations < 25:
            improvement_iterations += 1
        t = min(R)
        new_pop.append(child)
        new_R.append(R)
        new_fitness.append(t)
    return new_pop, new_R, new_fitness
