import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
import math

def evaluate(h):
    """Compute C₅ = max_k ∫ h(x)(1 - h(x+k)) dx"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, target):
    """Project x onto the feasible set {0<=h<=1, sum(h) = target}"""
    x = np.array(x, dtype=float)
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target) < 1e-10:
        return x
    diff = target - current_sum
    if diff > 0:
        # Add to zeros or low values
        free = np.where(x < 1)[0]
        if len(free) > 0:
            # Distribute among free indices
            add = diff / len(free)
            for i in free:
                x[i] = min(1.0, x[i] + add)
    else:
        # Remove from 1s or high values
        free = np.where(x > 0)[0]
        if len(free) > 0:
            remove = -diff / len(free)
            for i in free:
                x[i] = max(0.0, x[i] - remove)
    return x

def active_shifts(h, eps=1e-9):
    """Return all shift indices k where overlap is within eps of current C₅"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    active = []
    for k in range(len(corr)):
        if abs(corr[k] - max_val) < eps:
            active.append(k)
    return active

def gradient(h, ks):
    """Compute subgradient for active shifts"""
    n = len(h)
    g = np.zeros(n)
    dx = 2.0 / n
    for k in ks:
        for i in range(n):
            i_plus_k = i + k - (n - 1)  # Symmetric shift handling
            i_minus_k = i - k
            if i_plus_k >= 0 and i_plus_k < n:
                g[i] += (1 - h[i_plus_k])
            if i_minus_k >= 0 and i_minus_k < n:
                g[i] += h[i_minus_k]
    return g / (len(ks) * dx) if len(ks) > 0 else g

def line_search(h, g, t_current, budget_s, t_start):
    """Try various step sizes"""
    candidates = [0.0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1]
    best_h = h.copy()
    best_val = t_current
    for alpha in candidates:
        new_h = h - alpha * g
        new_h = project_feasible(new_h, 1.0)
        val = evaluate(new_h)
        if val < best_val:
            best_val = val
            best_h = new_h
    return best_h, best_val

def local_optimize(h0, max_iter, t_end):
    """Run projected gradient descent with line search"""
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate(h)
    t_start = time.time()
    for i in range(max_iter):
        elapsed = time.time() - t_start
        if elapsed > (t_end - t_start - 0.5):
            break
        ks = active_shifts(h)
        g = gradient(h, ks)
        h_new, val_new = line_search(h, g, evaluate(h), 600, t_end)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        if abs(evaluate(h_new) - evaluate(h)) < 1e-10:
            break
        h = h_new
    return best_h, best_val

def anneal(h0, t_end):
    """Simple simulated annealing"""
    h = h0.copy()
    current_val = evaluate(h)
    best_h = h.copy()
    best_val = current_val
    t_start = time.time()
    T = 0.01
    alpha = 0.995
    steps = 0
    while steps < 5000 and time.time() - t_start < (t_end - t_start - 1.0):
        steps += 1
        i, j = np.random.randint(0, len(h), 2)
        delta = np.random.uniform(-0.05, 0.05)
        if delta > 0 and h[i] + delta > 1.0:
            delta = 1.0 - h[i]
        if delta < 0 and h[j] - delta > 1.0:
            delta = h[j]
        if delta == 0:
            continue
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h = project_feasible(new_h, 1.0)
        new_val = evaluate(new_h)
        if new_val < current_val or np.random.rand() < np.exp(-(new_val - current_val) / T):
            h = new_h
            current_val = new_val
            if current_val < best_val:
                best_val = current_val
                best_h = h.copy()
        T *= alpha
    return best_h, best_val

def random_feasible(n):
    """Generate a random feasible h"""
    x = np.random.rand(n)
    # Scale to target sum
    x = x * (n / np.sum(x))
    # Project to feasible set
    x = project_feasible(x, 1.0)
    return x

def milp_binary(n, time_limit):
    """Mixed-integer program for binary h values"""
    try:
        # Define variables
        h = cp.Variable(n, boolean=True)
        t = cp.Variable()
        dx = 2.0 / n
        
        # Constraints
        constraints = [
            cp.sum(h) == n / 2,
            t >= 0
        ]
        
        # Add constraints for each shift k
        for k in range(1, n):
            product_vars = []
            for i in range(n):
                j = i + k
                if j < n:
                    # Introduce auxiliary variable for h[i]*h[j]
                    a = cp.Variable()
                    constraints.extend([
                        a <= h[i],
                        a <= h[j],
                        a >= h[i] + h[j] - 1,
                        a >= 0
                    ])
                    product_vars.append(a)
            if product_vars:
                constraints.append(cp.sum(product_vars) >= (1 - t) / dx)
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        solver = cp.CBC if cp.CBC in cp.installed_solvers() else cp.GLPK
        prob.solve(solver=solver, verbose=False, time_limit=time_limit)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            solution = h.value.astype(float) if h.value is not None else None
            bound = t.value if isinstance(t.value, (int, float)) else 0.0
            return solution, bound
        
    except Exception:
        pass
    return None, float('inf')
