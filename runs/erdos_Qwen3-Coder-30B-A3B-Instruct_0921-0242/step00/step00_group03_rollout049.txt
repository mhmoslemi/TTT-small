import numpy as np
import scipy.fft as fft
import time
from scipy.optimize import minimize

def project_to_F(h, n):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Compute current deficit/excess
    deficit = n/2 - h.sum()
    # Indices that are not at a bound
    free = np.where((h>0) & (h<1))[0]
    while np.abs(deficit) > 1e-12 and free.size>0:
        # Distribute equally over free indices
        delta = deficit / free.size
        # Apply delta, respecting bounds, and recompute deficit
        h[free] += delta
        # Clamp again and update free set
        h = np.clip(h, 0.0, 1.0)
        deficit = n/2 - h.sum()
        free = np.where((h>0) & (h<1))[0]
    return h

def compute_C5_fast(h, dx):
    n = len(h)
    # Zero-pad both h and g to length L = 2*n
    L = 2 * n
    h_pad = np.pad(h, (0, L - n), 'constant')
    g_pad = np.pad(1 - h, (0, L - n), 'constant')
    # Compute FFT
    H = fft.fft(h_pad)
    G = fft.fft(g_pad)
    # Compute convolution
    conv = fft.ifft(H * np.conj(G)).real * dx
    # Take max over all shifts
    C5 = conv.max()
    return C5, conv

def local_polish(h, c, dx, n, loc_steps=30):
    # Make a copy to modify
    h = h.copy()
    for _ in range(loc_steps):
        # Find index of worst shift
        k_star = np.argmax(c)
        # Try to reduce max correlation by coordinate swaps
        # Find indices with full support
        i1 = np.random.choice(np.where(h > 0)[0])
        i0 = np.random.choice(np.where(h < 1)[0])
        # Compute the change in correlations
        epsilon = min(h[i1], 1 - h[i0], 0.05)
        if epsilon <= 0:
            break
        
        # Compute the changes in correlation
        new_c = c.copy()
        # Update changes only for indices that are affected
        for k in range(n):
            i1k = i1 + k
            i0k = i0 + k
            if 0 <= i1k < n:
                new_c[k] -= epsilon * dx * (1 - h[i1k])  # truncated
            if 0 <= i0k < n:
                new_c[k] -= epsilon * dx * h[i0k]  # truncated
        # When full convolution loop is expensive
        # Estimate induced change in max
        c_max_new = new_c.max()
        if c_max_new < c.max():
            h[i1] -= epsilon
            h[i0] += epsilon
            c = new_c
        else:
            break
    return h, c.max()

def initialize_population(n, pop_size, initial_h_values):
    pop = []
    
    # Interpolated seed from initial_h_values
    if len(initial_h_values) < n:
        h_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        h_interp = initial_h_values[:n]
    h_interp = project_to_F(h_interp, n)
    C5, _ = compute_C5_fast(h_interp, 2.0 / n)
    pop.append((h_interp, C5))
    
    # Random continuous
    for _ in range((pop_size - 1) // 3):
        h = np.random.rand(n)
        h = project_to_F(h, n)
        C5, _ = compute_C5_fast(h, 2.0 / n)
        pop.append((h, C5))
        
    # Binary block
    h_block = np.zeros(n)
    h_block[:n//2] = 1.0
    np.random.shuffle(h_block)
    h_block = project_to_F(h_block, n)
    C5, _ = compute_C5_fast(h_block, 2.0 / n)
    pop.append((h_block, C5))
    
    # Gaussian perturbed from uniform
    h_gauss = np.full(n, 0.5) + np.random.normal(0, 0.1, n)
    h_gauss = project_to_F(h_gauss, n)
    C5, _ = compute_C5_fast(h_gauss, 2.0 / n)
    pop.append((h_gauss, C5))
    
    return pop

def select_parents(pop, tournament_size=3):
    tournament = np.random.choice(len(pop), tournament_size)
    winner = tournament[np.argmin([pop[i][1] for i in tournament])]
    return pop[winner][0]

def crossover(p1, p2):
    alpha = np.random.rand()
    child = alpha * p1 + (1 - alpha) * p2
    return child

def mutate(h, n, sigma, mu=0):
    m = max(1, int(0.02 * n))
    indices = np.random.choice(n, m, replace=False)
    noise = np.random.normal(mu, sigma, m)
    h[indices] += noise
    return h

def adaptive_mutation_sigma(generation, sigma_start=0.2, decay_rate=0.995):
    sigma = sigma_start * (decay_rate ** generation)
    if sigma < 0.02:
        sigma = 0.1
    return sigma
