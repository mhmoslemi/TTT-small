import numpy as np
import time
import random
from scipy import signal

def correlation_vector(b):
    """Compute the correlation vector R_k for a binary sequence b."""
    n = len(b)
    # Compute full convolution of b with itself
    conv = signal.correlate(b, b, mode='full')
    # Extract correlation values R_1 to R_{n-1}
    R = conv[n-1:n-1+n-1]
    return R

def update_correlation(R, i, j, b):
    """Update the correlation vector R after swapping positions i and j (b[i]=1, b[j]=0)."""
    n = len(b)
    # For each shift k, compute changes based on the swap (i,j)
    for k in range(1, n):
        # Compute if contribution of pairs with shift k is affected
        
        # If we're removing 1 from i and adding 1 to j in shift k computation:
        if i - k >= 0 and b[i - k] == 1:
            R[k - 1] -= 1
        if i + k < n and b[i + k] == 1:
            R[k - 1] -= 1
        if j - k >= 0 and b[j - k] == 1:
            R[k - 1] += 1
        if j + k < n and b[j + k] == 1:
            R[k - 1] += 1

def evaluate_solution(h):
    """Evaluate the solution with the given h."""
    n = len(h)
    dx = 2.0 / n
    h_vec = np.array(h, dtype=np.float64)
    correlations = np.correlate(h_vec, 1 - h_vec, mode="full")
    c5 = max(correlations) * dx
    return c5

def hill_climb_binary(b, R, max_iters=1000):
    """Perform hill climbing on a binary sequence."""
    n = len(b)
    t_best = min(R)
    max_improvement = 0
    
    for _ in range(max_iters):
        k_star = np.argmin(R) + 1  # shift indices start at 1
        best_delta = 0
        best_i, best_j = None, None

        # Look for swaps that improve R_{k_star}
        # Use a neighborhood: only consider swaps involving positions within distance 10
        # of any 1-position in the critical shift k*
        positions_ones = []
        for i in range(n):
            if b[i] == 1:
                positions_ones.append(i)
        
        # Get candidate positions within typical neighborhood of critical shift
        candidates = set()
        for i in positions_ones:
            for k in range(1, min(k_star + 10, n)):
                if i >= k and b[i - k] == 1:
                    candidates.add(i - k)
                if i + k < n and b[i + k] == 1:
                    candidates.add(i + k)
        
        # Also include nearby 1's and 0's
        for i in range(n):
            if b[i] == 1:
                # Look at nearby zeros
                for j in range(max(0, i-10), min(n, i+10)):
                    if b[j] == 0 and (i, j) not in candidates:
                        candidates.add((i, j))
            else:
                for i_pos in positions_ones:
                    if abs(i_pos - i) <= 10:  # vicinity of 1s
                        candidates.add((i_pos, i))
        
        candidates_list = list(candidates)
        if not isinstance(candidates_list[0], tuple):
            candidates_list = [(i, j) for i in positions_ones for j in range(n) if b[j] == 0 and abs(i-j) <= 10]

        # Limit to a random sample for efficiency
        if len(candidates_list) > 500:
            random.shuffle(candidates_list)
            candidates_list = candidates_list[:500]

        for i, j in candidates_list:
            if b[i] != 1 or b[j] != 0:
                continue
                
            # Compute delta change in R_{k_star}
            delta = 0
            # Recompute the change in R_k_star without full recalculation
            if i - k_star >= 0 and b[i - k_star] == 1:
                delta -= 1
            if i + k_star < n  and b[i + k_star] == 1:
                delta -= 1
            if j - k_star >= 0 and b[j - k_star] == 1:
                delta += 1
            if j + k_star < n and b[j + k_star] == 1:
                delta += 1
            
            # Handle other shifts (recommended to keep recalculating only computed shifts)
            # But here for simplicity we only focus on improving the critical shift
            if delta > best_delta:
                best_delta = delta
                best_i, best_j = i, j

        if best_i is not None:
            # Perform the swap
            b[best_i], b[best_j] = 0, 1
            update_correlation(R, best_i, best_j, b)
            new_t = min(R)
            if new_t > t_best:
                t_best = new_t
                max_improvement = max(max_improvement, best_delta)
            else:
                # Stopping condition
                break
        else:
            break
            
    return b, R, t_best

def crossover(parent1, parent2, n):
    """Create offspring by intersection of supports (union with adjustment)"""
    set1 = set(np.where(parent1 == 1)[0])
    set2 = set(np.where(parent2 == 1)[0])
    union_set = set1.union(set2)
    
    # Adjust to exactly n/2 ones if needed
    num_ones = len(union_set)
    if num_ones == n // 2:
        child = np.zeros(n, dtype=np.uint8)
        child[list(union_set)] = 1
    elif num_ones > n // 2:
        excess = num_ones - n // 2
        to_remove = random.sample(list(union_set), excess)
        child = parent1.copy()
        child[to_remove] = 0
        ones = np.where(child == 1)[0]
        while len(ones) < n // 2:
            zero_positions = np.where(child == 0)[0]
            if len(zero_positions) == 0:
                break
            rand_zero = random.choice(zero_positions)
            child[rand_zero] = 1
            ones = np.where(child == 1)[0]
    else:  # Too few ones
        needed = n // 2 - num_ones
        # Get complement positions
        comp = set(np.arange(n)) - union_set
        add_pos = random.sample(list(comp), min(needed, len(comp)))
        child = parent1.copy()
        child[add_pos] = 1

    return child

def mutate(individual, num_swaps=1):
    """Perform several random swaps to mutate an individual."""
    n = len(individual)
    indices = np.nonzero(individual)[0]
    zeros = np.nonzero(1 - individual)[0]
    for _ in range(num_swaps):
        if len(indices) == 0 or len(zeros) == 0:
            break
        i, j = np.random.choice(indices), np.random.choice(zeros)
        individual[i], individual[j] = 0, 1
        indices = np.nonzero(individual)[0]
        zeros = np.nonzero(1 - individual)[0]
    return individual

def project_to_weight(arr, target_weight):
    """Ensure array has exactly target_weight 1's."""
    n = len(arr)
    current_ones = np.sum(arr)
    diff = current_ones - target_weight
    if diff > 0:
        # Remove diff 1's
        ones = np.where(arr == 1)[0]
        to_remove = np.random.choice(ones, diff, replace=False)
        arr[to_remove] = 0
    elif diff < 0:
        # Add -diff 1's
        zeros = np.where(arr == 0)[0]
        to_add = np.random.choice(zeros, -diff, replace=False)
        arr[to_add] = 1
    return arr

def upsample_and_correct(small_seq, target_len):
    """Upsample binary sequence to target length and correct weight."""
    ratio = target_len // len(small_seq)
    if ratio == 0:
        ratio = 1
    expanded = np.repeat(small_seq, ratio)
    if len(expanded) > target_len:
        expanded = expanded[:target_len]
    elif len(expanded) < target_len:
        expanded = np.pad(expanded, (0, target_len - len(expanded)))
        
    # Fix the weight to exactly target_len/2
    fixed = project_to_weight(expanded, target_len // 2)
    return fixed

def time_left(t0, budget_s):
    """ return remaining time in seconds """
    return budget_s - (time.time() - t0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    t0 = time.time()
    
    # Strategy:  multi-scale progression with EA (memetic algorithm)
    # Time allocated: 40 s approx to 100 s/level (n=30, 36, 42, 48, 60)
    # Start from coarse to fine

    # initial_h_values is a global provided variable
    # Convert to binary if not already
    global initial_h_values
    init_len = len(initial_h_values)
    init_as_binary = np.zeros(init_len, dtype=np.uint8)
    # Place most significant values as 1s
    sorted_indices = np.argsort(initial_h_values)[::-1]
    num_ones = init_len // 2
    init_as_binary[sorted_indices[:num_ones]] = 1

    results = []
    n_levels = [30, 36, 42, 48, 60]
    t_budget_per_level = [30, 60, 90, 90, 120]
    
    for level, (n, level_budget) in enumerate(zip(n_levels, t_budget_per_level)):
        if time_left(t0, budget_s) < 5:
            break
            
        #----------------=======
        # Stage 1: Generate or use elite seed
        #-----------------------
        if n == 30:
            elite_seq = init_as_binary if level == 0 else None
            n_pop = max(30, 2 * n)
        elif level > 0:
            # Upsample previous best
            prev_seq = results[-1][0]  # binary vector
            elite_seq = upsample_and_correct(prev_seq, n)
            n_pop = max(30, min(100, 2 * n))

        #-----------------------
        # Generate population
        #-----------------------
        population = []
        if elite_seq is not None:
            population.append(elite_seq.copy())
        else:
            # Random seed
            seed_ind = np.zeros(n, dtype=np.uint8)
            ones = np.random.choice(n, n // 2, replace=False)
            seed_ind[ones] = 1
            population.append(seed_ind)

        # Add rest of population via random binary vectors
        for _ in range(n_pop - len(population)):
            individual = np.zeros(n, dtype=np.uint8)
            ones = np.random.choice(n, n // 2, replace=False)
            individual[ones] = 1
            population.append(individual)

        #=======================
        # Main Evolutionary Loop
        #=======================
        best_overall = -1
        best_sequence = None

        # Limit to fewer generations to respect time budget
        max_gens = int(level_budget * 150 / 50)  # approx // 3*n
        num_gens = 0
        gen_t0 = time.time()
        
        try:
            for gen in range(max_gens):
                if time_left(t0, budget_s) < 5:
                    break
                if time_left(gen_t0, level_budget) < 3:
                    break
                    
                num_gens += 1
                
                # Evaluate population
                pop_correlations = [correlation_vector(ind) for ind in population]
                t_values = [min(R) for R in pop_correlations]
                best_idx = np.argmax(t_values)
                if t_values[best_idx] > best_overall:
                    best_overall = t_values[best_idx]
                    best_sequence = population[best_idx].copy()

                # Tournament selection
                selected = []
                for _ in range(n_pop):
                    tourn = []
                    for _ in range(3):
                        idx = np.random.randint(0, n_pop)
                        tourn.append((idx, t_values[idx]))
                    chosen = max(tourn, key=lambda x: x[1])
                    selected.append(population[chosen[0]])

                # Crossover and mutation
                new_pop = []

                for i in range(0, n_pop, 2):
                    p1, p2 = selected[i], selected[i+1] if i+1 < n_pop else selected[0]
                    if np.random.rand() < 0.8:
                        child = crossover(p1, p2, n)
                    else:
                        child = p1.copy()
                    # Mutation
                    child = mutate(child, num_swaps=random.randint(1, 2))
                    # Fix weight
                    child = project_to_weight(child, n // 2)
                    # Local optimization
                    R_child = correlation_vector(child)
                    child, _, _ = hill_climb_binary(child, R_child)
                    new_pop.append(child)

                # Replace with worse individuals
                for i in range(n_pop):
                    if t_values[i] == min(t_values):
                        population[i] = new_pop[i]
                        
                # Occasionally inject random individual (diversify)
                if gen % 20 == 0:
                    ind = np.zeros(n, dtype=np.uint8)
                    ones = np.random.choice(n, n // 2, replace=False)
                    ind[ones] = 1
                    population[0] = ind

        except Exception as e:
            print("Exception in generation loop:", e)
            pass

        # Final best from stage
        if best_sequence is None:
            # Fallback: take the last pop
            pop_correlations = [correlation_vector(ind) for ind in population]
            t_values = [min(R) for R in pop_correlations]
            i_best = np.argmax(t_values)
            best_sequence = population[i_best]
        
        results.append((best_sequence, best_overall))

    #=====================
    # Final polishing with exhaustive search on best sequence
    #=====================
    final_seq = results[-1][0]
    
    # Get the best result so far
    best_t = None
    num_iter = 0
    while True:
        R = correlation_vector(final_seq)
        k_star = np.argmin(R) + 1  # 1-based indexing
        min_val = min(R)
        
        print(f"Polishing iteration {num_iter}, min R = {min_val}, k_star = {k_star}")
        
        # Enumerate all possible swaps for improvement at k_star
        gamma = 0.05  # Escape probability
        ones = np.where(final_seq == 1)[0]
        zeros = np.where(final_seq == 0)[0]
        
        best_delta = 0
        best_i, best_j = None, None
        found = False
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                delta = 0
                # Compute delta in R_k_star
                if i - k_star >= 0 and final_seq[i - k_star] == 1:
                    delta -= 1
                if i + k_star < len(final_seq) and final_seq[i + k_star] == 1:
                    delta -= 1
                if j - k_star >= 0 and final_seq[j - k_star] == 1:
                    delta += 1
                if j + k_star < len(final_seq) and final_seq[j + k_star] == 1:
                    delta += 1
                if delta > best_delta and min_val + delta > min_val:
                    best_delta = delta
                    best_i, best_j = i, j
                    found = True
                    
        if not found and np.random.rand() < gamma:  # Escape; sometimes do random swaps
            if len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                best_i, best_j = i, j
                
        # Apply change if any
        if found or (np.random.rand() < gamma and best_i is not None):
            final_seq[best_i], final_seq[best_j] = 0, 1
            update_correlation(R, best_i, best_j, final_seq)
            new_min = min(R)
            print("  Applied change:", best_i, "->", best_j, "New min:", new_min)
            if new_min <= min_val or not found:
                break  # No improvement or escaping done
        else:
            print("No improving swap found.")
            break
            
        num_iter += 1
        if num_iter > 5000:
            break

    # Final evaluation
    n_final = len(final_seq)
    dx = 2.0 / n_final
    final_conv = np.correlate(final_seq, 1 - final_seq, mode="full")
    c5_final = max(final_conv) * dx
    
    # Output as floating point array
    float_h = final_seq.astype(float)

    return float_h, c5_final, n_final
