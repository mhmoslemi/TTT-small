import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft

def evaluate_overlap(h, dx):
    n = len(h)
    # Compute the autocorrelation using FFT
    H = fft(h)
    A = ifft(np.conj(H) * H).real
    # A is now the autocorrelation at all lags
    # Since h is real, we should have A[k] = A[n-k] for k > 0, so we just use first n//2 + 1 values
    # But since we want the actual integration, A[0] corresponds to k=0, etc.
    # For C5 = max_k int h(x)(1-h(x+k)) dx, we compute:
    # int h(x)(1-h(x+k)) dx = int h(x) dx - int h(x)h(x+k) dx = 1 - A[k]
    # So C5 = max_k (1 - A[k]) = 1 - min_k A[k]
    # We have a small issue here with the indexing: we're looking at a circle
    # If we compute A[k] as sum h[i] h[i+k], we are summing over all valid i
    # That corresponds to a circular correlation, but the index ranges need to be handled carefully
    
    # Re-compute A as a proper circular correlation
    # Implementation: for k in [0, ..., n-1], A[k] = sum_i h[i] h[(i+k)%n]
    # This is exactly what FFT-based methods compute for autocorrelation if we handle indexing carefully
    
    # Since H^* H gives us this, and we're taking the real part
    # We know that all A[k] = A[n-k] by symmetry, so we consider only first half
    # To avoid numerical issues, use just the first half
    nA = len(A)
    A = A[:nA//2+1]  # Get only non-redundant elements from circular autocorrelation
    
    # The actual overlap integral on the continuous problem
    # reflects as 1 - A_k -> C5 = max_k(1 - A_k) = 1 - min_k A_k
    min_autocorrelation = A.min()
    return 1.0 - min_autocorrelation

def evaluate_overlap_fast(h, dx):
    # Simplified version that mimics the logic while taking shortcut for speed
    # A_k is computed as the circular autocorrelation
    # Since this is more accurate than the simpler block-computations, we'll use it
    n = len(h)
    # We'll do the circular correlation explicitly in the code with fast FFT
    # F = fft(h)
    # A = ifft(conj(F) * F).real
    H = fft(h)
    A = ifft(np.conj(H) * H).real
    # Cut down to first n//2 + 1, for easier indexing and since circular autocorrelation
    # for a circular shift k is same as -k, so A[k] = A[n-k] for non-trivial k >= 1
    A = A[:n//2 + 1]
    # Since we are looking for overlap integral = 1 - min A_k
    # min A_k is simply minimum of A (assuming normalized so that total mass = 1)
    return 1.0 - np.min(A)

def block_pattern_3(a, b, n_base=400):
    # A 3-block pattern: h = [1, p, 0] between [0, a, b, 2]
    # Solve p* (b-a) + a = 1 => p = (1-a)/(b-a)
    if b <= a or (b-a) == 0:
        return None
    p = (1 - a) / (b - a)
    if p < 0 or p > 1:
        return None
    # populate h on [0,2]
    dx = 2.0 / n_base
    h = np.zeros(n_base)
    start_a = int(np.round(a/dx))
    start_b = int(np.round(b/dx))
    h[:start_a] = 1.0
    h[start_a:start_b] = p
    # The last part remains 0, implicitly
    return h

def block_pattern_4(a, b, c, n_base=400):
    # A 4-block pattern: h = [1, p, q, 0] between [0, a, b, c, 2]
    # We solve integral constraint: a*1 + (b-a)*p + (c-b)*q = 1
    # Set  p = q, for simplicity, for a fixed value of q
    # a + (b-a)*q + (c-b)*q = 1  => a + q*(b-a + c-b) = 1  => a + (c-a)*q = 1
    # So q = (1-a)/(c-a)
    if c <= b or b <= a or (c-a) == 0:
        return None
    q = (1 - a) / (c - a)
    if q < 0 or q > 1:
        return None
    p = q
    dx = 2.0 / n_base
    h = np.zeros(n_base)
    start_a = int(np.round(a/dx))
    start_b = int(np.round(b/dx))
    start_c = int(np.round(c/dx))
    h[:start_a] = 1.0
    h[start_a:start_b] = p
    h[start_b:start_c] = q
    # The rest is 0
    return h

def block_sweep(n_base=400):
    best_h = None
    best_C5 = float('inf')
    dx = 2.0 / n_base
    step = 0.02  # Grid resolution for block positions
    for a in np.arange(0., 1.99, step):
        for b in np.arange(a + 0.001, 2., step):
            h = block_pattern_3(a, b, n_base=n_base)
            if h is None:
                continue
            C5 = evaluate_overlap_fast(h, dx)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
    # Try 4-block too
    step_f = 0.04
    for a in np.arange(0., 1.95, step_f):
        for b in np.arange(a + 0.001, 1.95, step_f):
            for c in np.arange(b + 0.001, 2., step_f):
                h = block_pattern_4(a, b, c, n_base=n_base)
                if h is None:
                    continue
                C5 = evaluate_overlap_fast(h, dx)
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h.copy()
    return best_h, best_C5

def mixture_gaussian_h(x, centers, amplitudes, sigma=0.1):
    h = np.zeros_like(x)
    for c, a in zip(centers, amplitudes):
        h += a * (np.abs(x - c) <= sigma)
    h = np.clip(h, 0, 1)
    # Rescale so the integral is 1
    return h

def de_fitness(params, n):
    # Simple version
    # Find n peaks which are mixture of Gausians
    # x = np.linspace(0, 2, n)
    centers = np.array(params[0::2])
    amplitudes = np.array(params[1::2])
    # Clip to legal range
    centers = np.clip(centers, 0, 2)
    amplitudes = np.clip(amplitudes, 0, 1)
    # Rescale to mass = 1.0
    h = mixture_gaussian_h(np.linspace(0, 2, n), centers, amplitudes)
    return -evaluate_overlap_fast(h, 2.0 / n)

def de_optimization(n=200, max_time=120):
    np.random.seed(42)
    start_time = time.time()
    bounds = [(0, 2)] * (n//4) + [(0, 1)] * (n//4)  # 3 centers and 3 amplitudes for total 6 dimensions
    best_found = float('inf')
    best_h = None

    # Try with fewer components, e.g., 3
    n_components = 3
    bounds = [(-0.01, 2.01)] * n_components + [(-0.01, 1.01)] * n_components

    def fitness(params):
        # params = [c1, c2, c3, a1, a2, a3], with constraints:
        centers = params[:n_components]
        amplitudes = params[n_components:]
        x = np.linspace(0, 2, n)
        h = mixture_gaussian_h(x, centers, amplitudes)
        # Enforce sum = 1 by normalization
        mass = np.sum(h) * (2.0 / n)
        if mass > 0:
            h = h / mass * n / 2.0
        return -evaluate_overlap_fast(h, 2.0 / n)

    result = differential_evolution(fitness, bounds, seed=42, maxiter=1000, popsize=30, disp=False)
    if result.success or True:
        centers = result.x[:n_components]
        amplitudes = result.x[n_components:]
        x = np.linspace(0, 2, n)
        h = mixture_gaussian_h(x, centers, amplitudes)
        mass = np.sum(h) * (2.0 / n)
        if mass > 0:
            h = h / mass * n / 2.0
        C5 = evaluate_overlap_fast(h, 2.0 / n)
        if C5 < best_found:
            best_found = C5
            best_h = h.copy()
    return best_h, best_found

def project_h(h, n_half):
    # Clip outside [0,1]
    h = np.clip(h, 0, 1)
    # Project to sum = n_half
    h_sum = np.sum(h)
    h = h - (h_sum - n_half) / n_half
    # Clip again to enforce box constraint
    h = np.clip(h, 0, 1)
    return h

def gradient_ascent_optimization(h_init, dx, max_time=300, step_start=0.5, step_decay=0.98):
    # Apply gradient ascent to minimize C5
    # Let h be the vector we're optimizing
    # We want to maximize min_k A_k := min_k h^T R_k h
    # Then C5 = 1 - min_k A_k = 1 - \min_k h^T R_k h
    # So minimize C5 = max_k (1 - h^T R_k h) = max_k (1 - A_k)

    # That means let's minimize f(h) = 1 - min_k h^T R_k h

    # This is non-convex but still doable.
    # Gradient of f(h) w.r.t. h: we will treat the kernel R_{k*} such that A_{k*} = max_k h^T R_k h
    # Gradient w.r.t. f = 1 - A_{k*} is g = -grad A_{k*} = -2 R_{k*} h

    # So: we will compute gradient g = -2 * roll(h, k*) and solve gradient ascent
    # with steps at decreasing size (geometrically)
    best_h = h_init.copy()
    best_C5 = evaluate_overlap_fast(best_h, dx)
    start_time = time.time()
    n = len(h_init)
    n_half = n / 2.0
    k_start = 0

    step = step_start
    last_improved = 0
    iteration = 0
    while time.time() < start_time + max_time:
        iteration += 1
        if iteration % 1000 == 0:
            print(f"iteration {iteration} C5 = {best_C5}")

        # Compute autocorrelations efficiently via FFT
        H = fft(h_init)
        A = ifft(np.conj(H) * H).real
        A = A[:n//2 + 1]

        # Find which k gives the minimum auto-corr
        k_min = np.argmin(A)
        k_min = k_min if k_min < n//2 + 1 else n - k_min
        # To apply gradient: g = -2 * roll(h_init, k_min) 
        # So we compute h_new = h_init + step * g = h_init - step * 2 * roll(h_init, k_min)
        g = np.roll(h_init, k_min)  # This is R_k h
        g = -2.0 * g  # Since : - grad h^T R_k h = -2 R_k h

        # Update
        h_new = h_init + step * g

        # Project h_new onto feasible region:
        # 1. clip to [0,1]
        h_new = np.clip(h_new, 0, 1)

        # 2. force sum = n_half
        h_sum = np.sum(h_new)
        offset = (h_sum - n_half) / n
        h_new -= offset
        # 3. Clip again
        h_new = np.clip(h_new, 0, 1)

        # Evaluate current loss and progress
        C5 = evaluate_overlap_fast(h_new, dx)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h_new.copy()
            last_improved = iteration
        # Update
        h_init = h_new.copy()
        step *= step_decay

        if iteration - last_improved > 200 and step < 1e-8:
            print("converged")
            break
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    dx = 2.0 / 400
    h1, c5_1 = block_sweep(n_base=400)
    if time.time() - start_time > budget_s:
        return h1.tolist(), c5_1, 400
    
    h2, c5_2 = de_optimization(n=200, max_time=120)
    if time.time() - start_time > budget_s:
        best_h = h1 if c5_1 < c5_2 else h2
        best_c5 = min(c5_1, c5_2)
        return best_h.tolist(), best_c5, 400

    # Choose the best of h1 or h2 as seed
    if c5_1 < c5_2:
        h_seed = h1
        c5_seed = c5_1
    else:
        h_seed = h2
        c5_seed = c5_2

    # now do gradient ascent on 300 points
    n = 300
    dx = 2.0 / n
    x = np.linspace(0, 2, n)
    # Interpolate h_seed to higher resolution
    h_interp = np.interp(x, np.linspace(0, 2, 400), h_seed)
    h_opt, c5_opt = gradient_ascent_optimization(h_interp, dx, max_time=300)
    if time.time() - start_time > budget_s:
        return h_opt.tolist(), c5_opt, n

    # If we had time, do final evaluation on 500 points to see if we can just slightly further improve
    if n < 500:
        x = np.linspace(0, 2, 500)
        h_interp = np.interp(x, np.linspace(0, 2, n), h_opt)
        dx = 2.0 / 500
        final_c5 = evaluate_overlap_fast(h_interp, dx)
        if final_c5 < c5_opt:
            h_opt = h_interp
            c5_opt = final_c5
    return h_opt.tolist(), c5_opt, len(h_opt)
