import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from scipy.sparse import csc_matrix

def project_to_F(h, n):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Compute current deficit/excess
    deficit = n/2 - h.sum()
    # Indices that are not at a bound
    free = np.where((h>0) & (h<1))[0]
    while np.abs(deficit) > 1e-12 and free.size>0:
        # Distribute equally over free indices
        delta = deficit / free.size
        # Apply delta, respecting bounds, and recompute deficit
        h[free] += delta
        # Clamp again and update free set
        h = np.clip(h, 0.0, 1.0)
        deficit = n/2 - h.sum()
        free = np.where((h>0) & (h<1))[0]
    return h

def fast_convolution(h, g, dx):
    # zero-pad to at least 2*n
    n = len(h)
    L = 2 * n
    h_pad = np.zeros(L)
    g_pad = np.zeros(L)
    h_pad[:n] = h
    g_pad[:n] = g
    
    H = fft(h_pad)
    G = fft(g_pad)
    conv = ifft(H * np.conj(G)).real * dx
    
    return conv

def evaluate_fitness(h, dx):
    g = 1 - h
    conv = fast_convolution(h, g, dx)
    # ignore the zero shift (which is always 1)
    C5 = np.max(conv[1:])
    return C5, conv

def local_improve(h, conv, dx, loc_steps=30):
    h_new = h.copy()
    conv_new = conv.copy()
    n = len(h)
    
    # Find worst shift
    k_best = np.argmax(conv[1:]) + 1
    g = 1 - h_new
    n_iter = 0
    for _ in range(loc_steps):
        if n_iter >= 5:
            break  # break if no improvement for 5 steps
        # Choose indices
        i1 = np.random.choice(np.where(h_new > 0)[0])
        i0 = np.random.choice(np.where(h_new < 1)[0])
        
        # Compute epsilon
        eps = min(h_new[i1], 1 - h_new[i0], 0.05)
        if eps <= 1e-10:
            continue
            
        # Proposed change
        i1_shifted = (i1 + k_best) % n
        i0_shifted = (i0 + k_best) % n
        delta = eps * dx * (g[i1_shifted] - g[i0_shifted])
        if conv_new[k_best] - delta >= conv_new[k_best]:
            # Accept the change
            h_new[i1] -= eps
            h_new[i0] += eps
            # Update correlations in place
            for k in range(1, n):
                i1_k = (i1 + k) % n
                i0_k = (i0 + k) % n
                conv_new[k] -= eps * dx * (g[i1_k] - g[i0_k])
            # Recalculate max for next iteration
            conv_new[k_best] = np.max(conv_new[1:])
            n_iter = 0
        else:
            n_iter += 1
    return h_new, conv_new

def initialize_population(n, pop_size, initial_h_values):
    pop = []
    # seed 1: interpolated from initial_h_values
    seed_h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    h1 = project_to_F(seed_h, n)
    c5_1, _ = evaluate_fitness(h1, 2.0/n)
    pop.append((h1, c5_1))
    
    # seed 2: binary block
    h2 = np.zeros(n)
    h2[:n//2] = 1
    h2 = project_to_F(h2, n)
    c5_2, _ = evaluate_fitness(h2, 2.0/n)
    pop.append((h2, c5_2))
    
    # seed 3: random continuous
    h3 = np.random.rand(n)
    h3 = project_to_F(h3, n)
    c5_3, _ = evaluate_fitness(h3, 2.0/n)
    pop.append((h3, c5_3))
    
    # seed 4: Gaussian perturbed
    h4 = np.full(n, 0.5) + np.random.normal(0, 0.05, n)
    h4 = project_to_F(h4, n)
    c5_4, _ = evaluate_fitness(h4, 2.0/n)
    pop.append((h4, c5_4))
    
    # remaining populations from random
    for _ in range(pop_size - 4):
        h = np.random.rand(n)
        h = project_to_F(h, n)
        c5, _ = evaluate_fitness(h, 2.0/n)
        pop.append((h, c5))
    
    return pop

def crossover(p1, p2):
    alpha = np.random.rand()
    child = alpha * p1 + (1 - alpha) * p2
    return project_to_F(child, len(p1))

def mutate(h, sigma):
    n = len(h)
    h = h.copy()
    m = max(1, int(0.02 * n))  # number of positions to mutate
    indices = np.random.choice(n, m, replace=False)
    noise = np.random.normal(0, sigma, m)
    h[indices] += noise
    return project_to_F(h, n)
