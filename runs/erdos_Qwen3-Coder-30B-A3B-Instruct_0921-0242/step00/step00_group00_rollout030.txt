import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize_scalar
import time
import math

def project_capped_simplex(v, total):
    """Project v onto the capped simplex: 0 <= v[i] <= 1, sum(v) = total"""
    if np.sum(v) == total:
        return v.copy()
    v = np.maximum(0, np.minimum(1, v))
    if np.sum(v) < total:
        # Need to increase values
        # Binary search on lambda
        def shrink(l):
            return np.sum(np.maximum(0, v - l))
        low, high = 0, 1
        for _ in range(50):
            mid = (low + high) / 2
            if shrink(mid) > total:
                low = mid
            else:
                high = mid
        l = (low + high) / 2
        v = np.maximum(0, v - l)
    elif np.sum(v) > total:
        # Need to decrease values
        def shrink(l):
            return np.sum(np.maximum(0, v - l))
        low, high = 0, 1
        for _ in range(50):
            mid = (low + high) / 2
            if shrink(mid) < total:
                high = mid
            else:
                low = mid
        l = (low + high) / 2
        v = np.maximum(0, v - l)
    v = v * total / np.sum(v)
    return v

def auto_corr(h):
    """Compute zero-padded auto-correlation of h"""
    n = len(h)
    # Zero-pad to 2*n
    h_pad = np.pad(h, (0, n), mode='constant')
    # Convolve with flipped h to get auto-correlations
    corr = fftconvolve(h_pad, h[::-1], mode='valid')
    return corr[:n]

def grad_F(beta, h, O, w):
    """Compute negative gradient of log-sum-exp of auto-correlations."""
    n = len(h)
    # Using FFT to compute convolutions for efficiency
    # Construct the right-hand side vector
    g = np.zeros(n)
    # w is a probability-like vector (w_k representing weight of O_k)
    # Compute 2*conv(w,h) - w[0]*h[0] - w[-1]*h[-1] to handle edge cases
    # The simple form using fftconvolve
    w_pad = np.pad(w, (0, n), mode='constant')
    h_pad = np.pad(h, (0, n), mode='constant')
    g = fftconvolve(w_pad, h_pad, mode='valid')
    g = g[:n]  # Taking first n elements only

    # Handle direct indices in gradient
    g_direct = np.zeros_like(g)
    # g[i] = \sum_k w[k] * (h[i+k] (if i+k < n) + h[i-k] (if i-k >= 0))
    # Merge both directions into one computation
    # Use for simplicity an iteration or simpler expression
    # What we do: for each k, we compute gradient contribution
    # Since we compute FFT of w * h then, fftshift or similar required,
    # we carefully perform the operations to ensure correctness.
    # Let's compute manually the necessary value
    for i in range(n):
        g[i] = 0
        for k in range(n):
            if i + k < n:
                g[i] += w[k] * h[i + k]
            if i - k >= 0:
                g[i] += w[k] * h[i - k]
    
    return -g

def update_beta_schedule(beta_min=0.5, beta_max=40, N=12):
    return np.logspace(np.log10(beta_min), np.log10(beta_max), N)

def optimize_one_start(n, beta_schedule, time_limit):
    # Initialize with interpolation of the fixed initial which we have
    # we assume it is valid to go from previous
    # initialize from something similar to available
    # probably best to avoid hardcoded assumption
    # we will первую инициализацию числовую: каноническая начальная точка
    h0 = np.ones(n) * 0.5
    total = n/2
    h0 = h0 / np.sum(h0) * total
    h0 = project_capped_simplex(h0, total)
    
    # Initial state
    h_last = h0.copy()
    best_h = h0.copy()
    best_C5 = 1e10
    
    start_time = time.time()
    
    for i, beta in enumerate(beta_schedule):
        # Inner loop with line search for projected gradient descent
        h = h_last.copy()
        last_O = auto_corr(h)
        last_C5 = 1 - np.min(last_O)
        if last_C5 < best_C5:
            best_C5 = last_C5
            best_h = h.copy()
            
        # Compute weights
        exp_O = np.exp(beta * last_O)
        w = exp_O / np.sum(exp_O)
        
        # EA approach to encourage convergence
        converged = False
        iter_i = 0
        time_check = False
        
        while not converged and time.time() - start_time < time_limit * 0.7:
            iter_i += 1
            # Compute gradient using w and O
            g = grad_F(beta, h, last_O, w)
            
            # Backtracking line search
            alpha = 0.1
            max_iter_btl = 10
            for _ in range(max_iter_btl):
                h_new = h - alpha * g
                h_new = project_capped_simplex(h_new, total)
                O_new = auto_corr(h_new)
                C5_new = 1 - np.min(O_new)
                # Armijo condition
                if C5_new <= last_C5 - 1e-4 * alpha * np.dot(g, g):
                    break
                else:
                    alpha *= 0.5
            
            # Updated h
            h = h_new
            
            # Update E
            last_O = O_new
            last_C5 = C5_new
            
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
                
            if iter_i > 50 and abs(C5_new - last_C5) < 1e-7:
                converged = True
                
        h_last = h.copy()
        if time.time() - start_time > time_limit * 0.7:
            break
    
    # Refinement to exact C5 with same steps as gradient method but exact non-smooth
    # Use as a refinement phase
    refinement_steps = 100
    for i in range(refinement_steps):
        if time.time() - start_time > time_limit * 0.9:
            break
        O = auto_corr(h)
        min_k = np.argmin(O)
        # Gradient of C5 at this lag
        # ∇C5 = ∇O_{k*} (negative sign for minimization)
        g = np.zeros_like(h)
        k_star = min_k
        # Modify only this specific case as a gradient
        for i in range(len(h)):
            if i + k_star < len(h):
                g[i] += h[i + k_star]
            if i - k_star >= 0:
                g[i] += h[i - k_star]
        g = -g
        
        # Apply line search again
        alpha = 0.01 / np.sqrt(i + 1)
        h_new = h - alpha * g
        h_new = project_capped_simplex(h_new, total)
        O_new = auto_corr(h_new)
        C5_new = 1 - np.min(O_new)
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    # Use specified discretization sizes
    nvals = [256, 384]
    best_C5 = 1e10
    best_h = None
    best_n = None
    time_budget = budget_s * 0.95  # Let 5% for final evaluation and cleanup
    start_time = time.time()
    
    # Set seed for reproducibility
    np.random.seed(seed)
    
    for n in nvals:
        if time.time() - start_time > time_budget:
            break
        # Run multiple restarts
        num_restarts = 5
        for restart in range(num_restarts):
            if time.time() - start_time > time_budget:
                break
            beta_schedule = update_beta_schedule()
            h, C5 = optimize_one_start(n, beta_schedule, time_budget)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h
                best_n = n
    
    # Final check with exact computation
    if best_h is not None and np.sum(best_h) > 0:
        dx = 2.0 / best_n
        corr = np.convolve(best_h, 1 - best_h, mode='full')
        exact_C5 = np.max(corr) * dx
        if exact_C5 < best_C5:
            best_C5 = exact_C5
    
    return (best_h, best_C5, best_n)
