import numpy as np
from scipy.optimize import minimize
import time
from scipy.interpolate import interp1d

def c5_of_h(h, dx):
    """Compute C5 for a given h"""
    g = 1 - h
    # Full convolution
    conv = np.correlate(h, g, mode='full')
    # Convert to shifts
    shifts = np.arange(-len(h)+1, len(h))
    # Find maximum
    max_idx = np.argmax(conv)
    return conv[max_idx] * dx

def compute_gradient(h, dx):
    """Compute gradient of C5 with respect to h"""
    g = 1 - h
    conv = np.correlate(h, g, mode='full')
    max_idx = np.argmax(conv)
    k_star = max_idx - (len(h) - 1)
    
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # Terms from the convolution derivative
        # In the convolution sum: \sum_j h[i-j] * (1 - h[j+k_star])
        # Derivative w.r.t. h[i]: 1 - h[i+k_star] + h[i-k_star]
        # But we need to be careful with indexing due to padding
        
        # j = i - k_star
        j1 = i - k_star
        if 0 <= j1 < len(h):
            grad[i] += -dx * h[j1]
        # j = i + k_star 
        j2 = i + k_star
        if 0 <= j2 < len(h):
            grad[i] += dx * h[j2]
        else:
            grad[i] += 0
            
    return grad

def project_onto_constraints(h, target_sum):
    """Project h onto the constraint sum(h) = target_sum with h[i] \in [0,1]"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Distribute the difference towards interior indices
    # Find indices that are not at bounds
    idxs = np.where((h > 0) & (h < 1))[0]
    if len(idxs) == 0:
        # All are at bounds, just set the mean
        return np.full_like(h, target_sum / len(h))
    # Distribute diff among interior points
    delta = diff / len(idxs)
    h[idxs] += delta
    
    # Refine projection
    h = np.clip(h, 0, 1)
    # Fix any over/under adjustments using back-and-forth
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Distribute the residual evenly to interiorIndices that are not near bounds
        adjusted = False
        while abs(diff) > 1e-10:
            idxs = np.where((h > 0) & (h < 1))[0]
            if len(idxs) == 0:
                break
            delta = diff / len(idxs)
            h[idxs] += delta
            h = np.clip(h, 0, 1)
            diff = target_sum - np.sum(h)
            if abs(diff) < 1e-10:
                break            
    return h

        # But keep it in the function signature for robustness.
        return compute_gradient(h, dx)
    
    feasible_h = None
    best_c5 = float('inf')

        h_interp = np.clip(h_interp, 0, 1)
        init_h = project_onto_constraints(h_interp, target_sum)
    else:
        init_h = project_onto_constraints(init_h, target_sum)
        
    # Several starts
    all_starts = [init_h]
    
    # Binary-like construction
    binary_h = np.zeros(n_points)
    binary_h[:n_points // 2] = 1
    np.random.shuffle(binary_h)
    binary_h = project_onto_constraints(binary_h, target_sum)
    all_starts.append(binary_h)
    
    # Smooth block + noise
    smooth_h = np.zeros(n_points)
    block_start = n_points // 4
    block_end = 3 * n_points // 4
    smooth_h[block_start:block_end] = 1.0
    noise = np.random.normal(0, 0.05, n_points)
    smooth_h += noise
    smooth_h = np.clip(smooth_h, 0, 1)
    smooth_h = project_onto_constraints(smooth_h, target_sum)
    all_starts.append(smooth_h)
    
    # Random uniform
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = project_onto_constraints(h_rand, target_sum)
        all_starts.append(h_rand)
    
    # Run optimizers
    constr = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum, 
              'jac': lambda h: np.ones_like(h)}
    bounds = [(0.0, 1.0)] * n_points

    for i, h_start in enumerate(all_starts):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 2.0:
            break
            
        try:
            res = minimize(obj, h_start, method='SLSQP',
                           bounds=bounds, constraints=constr,
                           jac=jac, options={'ftol': 1e-9, 'maxiter': 300, 'disp': False})
            
            h_final = res.x if res.success else h_start
            
            # Post-process to enforce sum
            h_final = project_onto_constraints(h_final, target_sum)
            
            # Final call to compute C5
            c5_final = obj(h_final)
            
            if c5_final < best_c5:
                best_c5 = c5_final
                feasible_h = h_final.copy()
        
        except Exception:
            continue

    # Local refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 1.0 and feasible_h is not None:
        # 100 local hill climbing steps
        base_h = feasible_h.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 0.5:
                break
            # Mutate a small part
            idxs = np.random.choice(n_points, min(10, n_points//10), replace=False)
            new_h = base_h.copy()
            perturb = np.random.rand(len(idxs)) * 0.02 - 0.01
            new_h[idxs] += perturb
            # Clip and correct
            new_h = np.clip(new_h, 0, 1)
            new_h = project_onto_constraints(new_h, target_sum)
            new_c5 = c5_of_h(new_h, dx)
            if new_c5 < best_c5:
                base_h = new_h
                best_c5 = new_c5
        feasible_h = base_h
    
    # Final evaluation
    if feasible_h is not None:
        final_c5 = c5_of_h(feasible_h, dx)
        return (feasible_h.tolist(), final_c5, n_points)
    else:
        # Default fallback to initial solution
        return (init_h.tolist(), c5_of_h(init_h, dx), n_points)
