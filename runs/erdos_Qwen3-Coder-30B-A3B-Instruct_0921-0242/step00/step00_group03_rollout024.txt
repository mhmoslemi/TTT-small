import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def evaluate_c5(h, dx):
    # compute full correlation
    corr_full = np.correlate(h, 1 - h, mode='full')
    # multiply by dx to get integral approximation
    corr_full *= dx
    # return maximum of all shifts
    return np.max(corr_full)

def solve_sdp(n_points, max_time=200):
    # define variables
    h = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()

    # constraints
    constraints = [
        cp.diag(X) == h,  # X_{ii} = h_i
        X >> 0,  # X is positive semidefinite
        0 <= h, h <= 1,  # bounds on h
        cp.sum(h) == n_points / 2  # integral constraint
    ]
    
    # for each shift, we define a constraint: t >= const - dx * Tr(P_s X)
    dx = 2.0 / n_points
    shift_range = range(-(n_points - 1), n_points)
    
    for k in shift_range:
        # create shift permutation matrix
        P_s = np.zeros((n_points, n_points))
        if k >= 0:
            P_s[k:, :n_points - k] = np.eye(n_points - k)
        else:
            P_s[:n_points + k, -k:] = np.eye(n_points + k)
        
        # constant term: dx * sum(h) = dx * n/2
        constant = dx * (n_points / 2)
        # constraint: t >= constant - dx * Tr(P_s X)
        constraint = t >= constant - dx * cp.trace(P_s @ X)
        constraints.append(constraint)
    
    # objective
    objective = cp.Minimize(t)
    
    # build problem
    prob = cp.Problem(objective, constraints)
    
    # attempt to solve within limit
    start = time.time()
    try:
        prob.solve(solver=cp.SCS, max_iters=20000, verbose=False)
        end = time.time()
        if end - start > max_time:
            return None, None
            
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # return the relaxed solution's h and t
            return h.value, prob.value
        return None, None
    except:
        return None, None

def randomized_rounding(X_star, h_star, n_points, dx, n_rounds=30):
    h_best = None
    c5_best = float('inf')
    
    for _ in range(n_rounds):
        # sample gaussian vector
        g = np.random.randn(n_points)
        # transform: y = U Λ^{1/2} g
        U, Lambda = np.linalg.eigh(X_star)
        y = U @ np.diag(np.sqrt(np.maximum(Lambda, 0))) @ g
        # clip to [0,1]
        y = np.clip(y, 0, 1)
        # project to correct sum
        current_sum = np.sum(y)
        deficit = n_points / 2 - current_sum
        if abs(deficit) > 1e-8:
            # adjust by distributing deficit
            # find indices with 0 < y < 1 for adjustment
            valid_idx = np.where((y > 0) & (y < 1))[0]
            if len(valid_idx) > 0:
                # simple distribution
                idx = valid_idx[:int(np.ceil(abs(deficit)))]
                if deficit > 0:
                    y[idx] = np.minimum(y[idx] + deficit / len(idx), 1.0)
                else:
                    y[idx] = np.maximum(y[idx] + deficit / len(idx), 0.0)
        h = np.clip(y, 0, 1)
        
        # evaluate c5
        c5 = evaluate_c5(h, dx)
        if c5 < c5_best:
            h_best = h.copy()
            c5_best = c5
            
    return h_best, c5_best

def local_refinement(h_init, dx, max_passes=1000, budget_s=150):
    start = time.time()
    h = h_init.copy()
    old_c5 = evaluate_c5(h, dx)
    passes = 0
    
    while passes < max_passes:
        improved = False
        # pass over indices
        for i in range(len(h)):
            # compute how the max would change if we slightly modify h[i]
            # change is epsilon in (-0.01, 0.01)
            old_val = h[i]
            best_c5 = old_c5
            best_delta = 0
            for delta in np.linspace(-0.03, 0.03, 20):
                if old_val + delta < 0 or old_val + delta > 1:
                    continue
                h[i] = old_val + delta
                new_c5 = evaluate_c5(h, dx)
                if new_c5 < best_c5:
                    best_delta = delta
                    best_c5 = new_c5
                    
            # update if improvement
            if best_delta != 0:
                h[i] += best_delta
                improved = True
                old_c5 = best_c5
            # restore
            h[i] = old_val
            
        if not improved:
            break
        passes += 1
        
        if time.time() - start > budget_s:
            break
            
    # recompute final c5 after local refinement
    c5_final = evaluate_c5(h, dx)
    
    # force the sum constraint exactly
    current_sum = np.sum(h)
    scale = (len(h) / 2) / current_sum
    h *= scale
    c5_final = evaluate_c5(h, dx)
    
    return h, c5_final
