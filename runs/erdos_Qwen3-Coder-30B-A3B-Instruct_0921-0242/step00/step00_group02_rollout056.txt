def project_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum"""
    h = np.clip(h, 0, 1)
    deficit = target_sum - np.sum(h)
    if abs(deficit) < 1e-12:
        return h
    
    # Create a list of (value, index) pairs and sort by value
    indexed_h = list(enumerate(h))
    indexed_h.sort(key=lambda x: x[1])
    
    # Greedily adjust entries
    F = set(range(len(h)))
    for i, _ in indexed_h:
        if deficit == 0 or len(F) == 0:
            break
        if i in F:
            # Compute how much we can add
            can_add = 1.0 - h[i]
            add = min(deficit, can_add)
            h[i] += add
            deficit -= add
            if h[i] >= 1.0:
                F.discard(i)
    
    # Adjust remaining deficit (this should not happen if deficiency is small enough)
    if abs(deficit) > 1e-9:
        # If still deficit, we add it equally to all free entries
        if len(F) > 0:
            delta = deficit / len(F)
            for i in list(F):
                h[i] += delta
                if h[i] >= 1.0:
                    h[i] = 1.0
                    F.discard(i)
    
    return h

def compute_autocorrelations(h):
    """Compute linear autocorrelations r_k = sum h[i] * h[i+k] for k=1,...,n-1"""
    n = len(h)
    h_padded = np.pad(h, (0, n), mode='constant')
    H = fft(h_padded)
    r = np.real(ifft(H * np.conj(H)))
    return r[1:n]  # r[1] to r[n-1]

def soft_minimum(h, alpha):
    """Compute S_alpha = - (1/alpha) * log(sum(exp(-alpha * r_k)))"""
    r = compute_autocorrelations(h)
    r = r[r > 0]  # avoid minus infinity later
    if len(r) == 0:
        return 0.0, np.zeros_like(h)
    if np.any(np.isinf(r)):
        return 0.0, np.zeros_like(h)
    exp_minus_alpha_r = np.exp(-alpha * r)
    if np.any(np.isnan(exp_minus_alpha_r)):
        return 0.0, np.zeros_like(h)
    sum_exp = np.sum(exp_minus_alpha_r)
    if sum_exp == 0.0:
        return 0.0, np.zeros_like(h)
    weight = exp_minus_alpha_r / sum_exp
    return -np.log(sum_exp) / alpha, weight

def step_size_line_search(h, grad, alpha, target_sum):
    """Perform backtracking line search for step size"""
    rate = 1.0
    eta = 0.1
    t = 1.0
    while True:
        h_new = h - rate * grad
        h_new = project_simplex(h_new, target_sum)
        val_new = -soft_minimum(h_new, alpha)[0]
        val_old = -soft_minimum(h, alpha)[0]
        if val_new <= val_old - 0.5 * t * rate * np.dot(grad, grad):
            # Accept the step
            return rate
        rate *= eta
        if rate < 1e-8:
            return 1e-8

def projected_gradient_ascent(h, alpha, max_iter=500, tol=1e-8):
    """Projected gradient ascent method."""
    n = len(h)
    target_sum = n / 2.0
    h_best = h.copy()
    best_value = -soft_minimum(h, alpha)[0]
    
    h_prev = h.copy()
    beta = 0.9
    for it in range(max_iter):
        # 1. Compute soft minimum value
        r_val, weights = soft_minimum(h, alpha)
        cur_val = -r_val
        
        # print(f"iter {it}: soft_min = {r_val}, cur_val = {cur_val}")
        
        # 2. Compute gradient
        grad = gradient_soft_minimum(h, weights)
        
        # 3. Backtracking line search
        rate = step_size_line_search(h, grad, alpha, target_sum)
        
        # 4. Nesterov-like update
        y = h + beta * (h - h_prev)
        h_new = project_simplex(y - rate * grad, target_sum)
        
        # keep for next iteration
        h_prev = h.copy()
        h = h_new.copy()
        
        if cur_val > best_value:
            best_value = cur_val
            h_best = h.copy()
            
        if it > 20:
            diff = best_value - (-soft_minimum(h, alpha)[0])
            if diff < 0.0001 and abs(diff) < tol:
                break
                
    return h_best

def do_homotopy_interpolation(h_start, max_alpha=80, alpha_factor=2.0):
    """Run the homotopy scheme with decreasing alpha."""
    n = int(len(h_start) / 2) * 2  # ensure even for splitting
    h = h_start.copy()  # make a copy local
    
    # small fix to ensure the starting point has the exact target sum
    target_sum = n / 2.0
    h = project_simplex(h, target_sum)
    
    # schedule
    current_alpha = 0.5
    h_best = h.copy()
    best_obj = -soft_minimum(h, np.inf)[0]  # this gives min q_k
    
    # run multiple steps with increasing alpha
    while current_alpha < max_alpha:
        # print(f"Starting with alpha = {current_alpha}")
        h = projected_gradient_ascent(h, current_alpha, max_iter=500)
        current_alpha *= alpha_factor
        # evaluate current objective
        r_vals = compute_autocorrelations(h)
        if len(r_vals) > 0:
            value = -np.min(r_vals)
        else:
            value = 0.0
        if value < best_obj:
            best_obj = value
            h_best = h.copy()
        # stop early if safe
        if best_obj < 1e-7:
            break

    return h_best, 1.0 - best_obj

def interpolate_initial_to_length(h_start, new_len):
    """Interpolate initial h array to new length and return."""
    ratio = float(new_len) / float(len(h_start))  # downsample or upsample factor
    if ratio == 1.0:
        return h_start.copy()
    h_new = np.interp(np.linspace(0, len(h_start)-1, new_len), 
                      np.arange(len(h_start)), h_start)
    return np.array(h_new)

def round_to_step_function(h, target_len, dx=1.0):
    """Round h to a binary step function"""
    # Sort by values to decide top half
    sorted_indices = np.argsort(h)[::-1]
    h_binary = np.zeros_like(h)
    num_ones = int(target_len / 2)
    h_binary[sorted_indices[:num_ones]] = 1.0
    
    # Ensure correct sum
    diff = np.sum(h_binary) - (target_len / 2)
    if abs(diff) > 1e-6:
        # adjust by swapping 1s and 0s
        if diff > 0:
            indices_with_one = np.argwhere(h_binary == 1).flatten()
            indices_with_zero = np.argwhere(h_binary == 0).flatten()
            # changing 1s to 0s in indices_with_one.
            for i in range(int(diff)):
                if len(indices_with_one) > 0 and len(indices_with_zero) > 0:
                    idx_1 = indices_with_one[0]
                    idx_0 = indices_with_zero[-1]
                    h_binary[idx_1] = 0
                    h_binary[idx_0] = 1
                    indices_with_one = np.delete(indices_with_one, 0)
                    indices_with_zero = np.delete(indices_with_zero, -1)
    
    return h_binary

def local_swap_refinement(h, dx=1.0):
    """Local binary swap refinement"""
    # Recalculate worst shift
    r = compute_autocorrelations(h)
    if len(r) == 0:
        return h
    k_star = np.argmin(r)
    k_star = k_star  # 0-based

    # Get the indices involved in the k_star shift
    # indices where h[i] = 1 and h[i+k_star] = 0 are candidates for swapping
    h_int = round_to_step_function(h, len(h), dx)
    m = len(h_int)
    assert len(h_int) == m
    assert np.sum(h_int) == m/2
    
    # Preparation: identify candidates
    candidates = []
    for i in range(m):
        if h_int[i] > 0.5 and i + k_star < m and h_int[i+k_star] < 0.5:
            candidates.append(i)
    
    # Base r(k_star) value
    base_r = 0
    for i in range(m - k_star):
        base_r += h_int[i] * h_int[i + k_star]
    
    improvement_found = True
    while improvement_found:
        improvement_found = False
        max_delta = -float('inf')
        best_pair = None
        # Take manageable number of candidate swaps
        n_candidates = min(300, len(candidates)*10)
        indices_from = np.random.permutation(candidates)[:n_candidates]
        
        for i in indices_from:
            j = i + k_star
            if j < m and h_int[j] < 0.5:
                # Compute Δr_k_star = - (h_int[i] * h_int[j] - h_int[i+1] * h_int[j+1])
                # We want Δ = new_r - old_r = -1*1 + 1*1 = 0 originally if both valid
                # Impact: Save calculation exactly here; just main-
                # Change of correlation we care about: i,k and i+k
                # When swap i,j:
                # old_i = h_int[i]*h_int[i+k_star]
                # old_j = h_int[j]*h_int[j+k_star]
                # new_i = h_int[j]*h_int[i+k_star]
                # new_j = h_int[i]*h_int[j+k_star]
                # Δ = (h_int[i]*h_int[i+k_star]) - (h_int[j]*h_int[i+k_star]) 
                #   - (h_int[j]*h_int[j+k_star]) + (h_int[i]*h_int[j+k_star])
                #   = (h_int[i] - h_int[j]) * (h_int[i+k_star] - h_int[j+k_star])
                delta = (h_int[i] - h_int[j]) * (h_int[i + k_star] - h_int[j + k_star])
                if delta > max_delta:
                    max_delta = delta
                    best_pair = (i, j)
        
        if best_pair is not None and max_delta > 0:
            improvement_found = True
            i, j = best_pair
            h_int[i], h_int[j] = h_int[j], h_int[i]
            # Recalculate correlations and possibly update k_star
            r_new = compute_autocorrelations(h_int)
            if len(r_new) > 0:
                k_star = np.argmin(r_new)
                # Update as appropriate
    return h_int
