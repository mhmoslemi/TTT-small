import numpy as np
import time
from scipy.signal import correlate

def correlation_vector(b):
    """Compute the autocorrelation vector for binary sequence b."""
    R = correlate(b, b, mode='full')
    # Remove the middle element (correlation at shift 0)
    R = np.concatenate([R[:len(b)-1], R[len(b)+1:]])
    return R

def update_correlation(R, i, j, b):
    """
    Update R after swapping b[i] and b[j].
    Precondition: b[i] = 1, b[j] = 0.
    """
    n = len(b)
    old_min_k = np.argmin(R)
    old_min_val = R[old_min_k]

    # Process each shift k
    for k in range(1, n):
        # i-k >= 0 and b[i-k] == 1: remove contribution of (i-k,i)
        if i - k >= 0 and b[i - k] == 1:
            R[k - 1] -= 1
        # i+k < n and b[i+k] == 1: remove contribution of (i,i+k)
        if i + k < n and b[i + k] == 1:
            R[k - 1] -= 1
        # j-k >= 0 and b[j-k] == 1: add contribution of (j-k,j)
        if j - k >= 0 and b[j - k] == 1:
            R[k - 1] += 1
        # j+k < n and b[j+k] == 1: add contribution of (j,j+k)
        if j + k < n and b[j + k] == 1:
            R[k - 1] += 1

    new_min_k = np.argmin(R)
    new_min_val = R[new_min_k]

    return new_min_k, new_min_val, R

def local_hill_climb(b, R):
    """
    Perform hill climbing on the binary sequence b with precomputed R.
    """
    n = len(b)
    improved = True
    iterations = 0
    max_iters = 1000  # sanity check

    while improved and iterations < max_iters:
        improved = False
        iterations += 1

        # Find critical shift
        min_k = np.argmin(R)
        min_val = R[min_k]

        # Find best swap to improve min_val
        best_delta = 0
        best_swap_i = -1
        best_swap_j = -1

        # Enumerate possible swaps
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        best_swap = None
        best_delta = 0

        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                # Compute delta for this swap
                new_R = R.copy()
                # Simulate the swap in the binary vector
                temp_b = b.copy()
                temp_b[i], temp_b[j] = temp_b[j], temp_b[i]

                # Manually compute change in each entry of R
                changed = []
                for k in range(1, n):
                    contrib = 0
                    # Check contributions to R[k-1] (shift k)
                    # Changes in i-j connections as before
                    # i-k >= 0 and b[i-k] == 1
                    if i - k >= 0 and b[i - k] == 1:
                        contrib -= 1
                    # i+k < n and b[i+k] == 1
                    if i + k < n and b[i + k] == 1:
                        contrib -= 1
                    # j-k >= 0 and b[j-k] == 1
                    if j - k >= 0 and b[j - k] == 1:
                        contrib += 1
                    # j+k < n and b[j+k] == 1
                    if j + k < n and b[j + k] == 1:
                        contrib += 1

                    # Apply swap in temp_b to see if it affects R[k-1]
                    # i-k >= 0 and temp_b[i-k] == 1
                    if i - k >= 0 and temp_b[i - k] == 1:
                        contrib += 1
                    # i+k < n and temp_b[i+k] == 1
                    if i + k < n and temp_b[i + k] == 1:
                        contrib += 1
                    # j-k >= 0 and temp_b[j-k] == 1
                    if j - k >= 0 and temp_b[j - k] == 1:
                        contrib -= 1
                    # j+k < n and temp_b[j+k] == 1
                    if j + k < n and temp_b[j + k] == 1:
                        contrib -= 1

                    new_R[k - 1] = R[k - 1] + contrib
                    if R[k - 1] != new_R[k - 1]:
                        changed.append(k - 1)

                new_min = np.min(new_R)  # new value of min R_k
                delta = new_min - min_val

                if delta > best_delta or (delta == best_delta and best_delta >= 0):
                    best_delta = delta
                    best_swap_i = i
                    best_swap_j = j
                    # Update the actual R
                    R = new_R
                else:
                    # Revert change to temp for next i,j
                    pass
        # Now we know what swap improved min_k the most
        if best_delta > 0:
            improved = True
            b[best_swap_i], b[best_swap_j] = b[best_swap_j], b[best_swap_i]
            # Update R based on real swaps
            R = correlation_vector(b)
            min_val += best_delta
        else:
            break

    return b, R

def create_population(n, n_pop, weight, elite_b=None):
    """Create a population of binary sequences."""
    pop = []
    if elite_b is not None:
        pop.append(elite_b.copy())
    for _ in range(n_pop - len(pop)):
        temp = np.zeros(n, dtype=np.uint8)
        ones = np.random.choice(n, size=weight, replace=False)
        temp[ones] = 1
        pop.append(temp)
    return np.array(pop)

def crossover(parent1, parent2, n, weight):
    """Crossover with weight preservation."""
    # Union of support sets
    union_indices = np.unique(np.concatenate([np.where(parent1 == 1)[0], np.where(parent2 == 1)[0]]))
    # If union size > weight:
    if len(union_indices) > weight:
        # Randomly remove enough indices
        to_remove = len(union_indices) - weight
        indices_to_remove = np.random.choice(union_indices, size=to_remove, replace=False)
        union_indices = np.setdiff1d(union_indices, indices_to_remove)
    elif len(union_indices) < weight:
        # Search for 0s in either to add
        complement = np.where((parent1 == 0) & (parent2 == 0))[0]
        num_needed = weight - len(union_indices)
        indices_to_add = np.random.choice(complement, size=min(num_needed, len(complement)), replace=False)
        union_indices = np.concatenate([union_indices, indices_to_add])
    # Actual child
    child = np.zeros(n, dtype=np.uint8)
    child[union_indices] = 1
    return child

def mutate(individual, n, weight):
    """Mutation: swap 1 random 1-bit with 1 random 0-bit."""
    ones = np.where(individual == 1)[0]
    zeros = np.where(individual == 0)[0]
    num_swaps = np.random.choice([1, 2])
    
    for _ in range(num_swaps):
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        individual[i], individual[j] = individual[j], individual[i]
    return individual

def memetic_algorithm(n, max_gen, budget_s, seed=42):
    """
    Run a memetic algorithm to find the best solution.
    """
    np.random.seed(seed)

    # Parameters
    n_pop = max(30, 2 * n)
    p_c = 0.8
    mu = 1  # mutations per offspring
    diversification_freq = 20
    tournament_size = 3

    # Initialize population
    if 'initial_h_values' in globals():
        elite = globals()['initial_h_values']
        # Interpolate to n
        temp_h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(elite)), elite)
        elite_b = (temp_h > 0.5).astype(np.uint8)
        # Adjust weight to exactly n//2
        current_weight = np.sum(elite_b)
        ones = np.where(elite_b == 1)[0]
        zeros = np.where(elite_b == 0)[0]
        # Flip some bits
        if current_weight > n // 2:
            # We need to flip some 1 → 0
            to_flip = current_weight - n // 2
            flip_indices = np.random.choice(ones, size=to_flip, replace=False)
            elite_b[flip_indices] = 0
        elif current_weight < n // 2:
            to_flip = n // 2 - current_weight
            flip_indices = np.random.choice(zeros, size=to_flip, replace=False)
            elite_b[flip_indices] = 1
    else:
        elite_b = None

    pop = create_population(n, n_pop, n // 2, elite_b)

    # Fitness tracking (inverted min R for higher values)
    fitnesses = np.zeros(n_pop)
    tvals = np.zeros(n_pop)
    R_vecs = np.zeros((n_pop, n - 1))

    # Initial evaluations
    eval_start = time.time()
    for i, ind in enumerate(pop):
        R = correlation_vector(ind)
        R_vecs[i] = R
        tval = np.min(R)
        tvals[i] = tval
        fitnesses[i] = tval

    best_t = np.max(tvals)
    best_ind = pop[np.argmax(tvals)].copy()

    above_time = 0
    below_time = 0

    if n < 50:
        max_gen = max(100, int(n * 10))
    else:
        max_gen = max(100, int(n * 5))

    for gen in range(max_gen):
        if time.time() - eval_start > budget_s - 5:
            break

        # Selection: Tournament
        selected = []
        for _ in range(n_pop):
            idxs = np.random.choice(n_pop, size=tournament_size)
            chosen = np.argmax(fitnesses[idxs])
            selected.append(pop[idxs[chosen]].copy())

        # Crossover + Mutation
        new_pop = []

        # Half pairs for crossover
        for i in range(0, n_pop, 2):
            if i + 1 < n_pop:
                p1 = selected[i]
                p2 = selected[i + 1]

                if np.random.rand() < p_c:
                    child = crossover(p1, p2, n, n // 2)
                else:
                    child = p1.copy()

                child = mutate(child, n, n // 2)
                # Local improvement
                # Before local improvement, compute R
                R = correlation_vector(child)
                improved_child, R = local_hill_climb(child, R)
                new_pop.append(improved_child)
            else:
                # If odd number, just copy parent
                new_pop.append(selected[i])

        # Replace old population
        pop = new_pop

        # Update fitnesses
        best_from_pop = -np.inf
        for i, pop_ind in enumerate(pop):
            R = correlation_vector(pop_ind)
            t = np.min(R)
            fitnesses[i] = t
            if t > best_from_pop:
                best_from_pop = t
                best_ind = pop_ind.copy()
        if best_from_pop > best_t:
            best_t = best_from_pop

    # Final polishing with exhaustive search on critical element
    final_R = correlation_vector(best_ind)
    min_k = np.argmin(final_R)
    final_t = np.min(final_R)
    final_best_r = final_R.copy()

    # Perform exhaustive search
    best_swap_improves = True
    rounds = 0
    max_rounds = 100
    while best_swap_improves and rounds < max_rounds:
        best_swap_improves = False
        ones = np.where(best_ind == 1)[0]
        zeros = np.where(best_ind == 0)[0]
        best_delta = 0
        best_i = -1
        best_j = -1

        for i in ones:
            for j in zeros:
                temp_R = final_R.copy()
                changed = []
                delta = 0
                # Update temp_R to account for the swap of i and j
                for k in range(1, n):
                    # Original contribution of this shift
                    # in calculation of this entry
                    # i-k, i+k, j-k, j+k
                    contrib = 0
                    if i - k >= 0 and best_ind[i - k] == 1:
                        contrib -= 1  # existed before the swap
                    if i + k < n and best_ind[i + k] == 1:
                        contrib -= 1
                    if j - k >= 0 and best_ind[j - k] == 1:
                        contrib += 1
                    if j + k < n and best_ind[j + k] == 1:
                        contrib += 1

                    # Undo this in temp_R
                    temp_R[k - 1] += contrib

                    # Recalculate with newly swapped values
                    contrib = 0
                    if i - k >= 0 and best_ind[i - k] == 1 and i != j:
                        contrib -= 1
                    if i + k < n and best_ind[i + k] == 1 and i != j:
                        contrib -= 1
                    if j - k >= 0 and best_ind[j - k] == 1:
                        contrib += 1
                    if j + k < n and best_ind[j + k] == 1:
                        contrib += 1
                    temp_R[k - 1] += contrib
                delta = np.min(temp_R) - final_t
                if delta > best_delta or (delta == best_delta and best_delta > 0):
                    best_delta = delta
                    best_i = i
                    best_j = j
        if best_delta > 0:
            # Apply the swap
            best_ind[best_i], best_ind[best_j] = best_ind[best_j], best_ind[best_i]
            # Recalculate final_R recursively to propagate the effects
            # Simpler: recompute R
            final_R = correlation_vector(best_ind)
            final_t += best_delta
            best_swap_improves = True
        rounds += 1

    dx = 2.0 / n
    final_c5 = 1.0 - dx * final_t
    return best_ind.astype(float), final_c5, n

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    best_c5 = 1.0
    best_h = None
    best_n = 0
    sizes = [30, 36, 42, 48, 60]

    # Adjust budget for stages
    total_budget = budget_s - 5  # Safety
    budget_per_stage = {}
    # Each requires ~120s for n=60, others less
    stage_budgets = [30, 60, 90, 120, 180]
    for i, n in enumerate(sizes):
        budget_per_stage[n] = min(stage_budgets[i] * (n / 60.0), 120.0)

    for n in sizes:
        if time.time() - start_time > total_budget:
            break

        remaining = budget_s - (time.time() - start_time)
        if remaining <= 5:
            break
        allocated = min(budget_per_stage[n], remaining - 5)

        h_vals, c5_val, n_used = memetic_algorithm(n, 5000, allocated, seed + n)

        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_vals
            best_n = n_used

        # Break if very close to target
        if best_c5 <= 0.379:
            break

    return best_h, best_c5, best_n
