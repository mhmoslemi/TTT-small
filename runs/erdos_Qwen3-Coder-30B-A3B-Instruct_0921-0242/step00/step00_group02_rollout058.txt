import numpy as np
import time
import random
from bisect import bisect_left

def correlation_vector(b):
    """Compute the autocorrelation vector R where R[k] = sum b[i] * b[i+k] for k >= 1"""
    R_full = np.correlate(b, b, mode='full')
    R = R_full[len(b)-1:]  # Remove the first element (k=0)
    return R

def incremental_update(R, i, j, n):
    """Update R after swapping b[i]=1 with b[j]=0.
       Returns the change in R_k for each k."""
    changes = np.zeros(n-1, dtype=int)
    for k in range(1, n):
        if i >= k and b[i - k] == 1:
            changes[k-1] -= 1
        if i + k < n and b[i + k] == 1:
            changes[k-1] -= 1
        if j >= k and b[j - k] == 1:
            changes[k-1] += 1
        if j + k < n and b[j + k] == 1:
            changes[k-1] += 1
    return changes

def hill_climb(b, R, max_iters=1000):
    """Local hill climb to increase min(R)"""
    n = len(b)
    best_t = np.min(R)
    best_b = b.copy()
    best_R = R.copy()

    for iteration in range(max_iters):
        # Find critical shift
        k_star = np.argmin(best_R)
        t_star = best_R[k_star]
        
        # Enumerate all possible swaps
        ones = np.where(best_b == 1)[0]
        zeros = np.where(best_b == 0)[0]
        
        best_swap = None
        best_delta = 0
        
        # For moderately large problems, sample a small neighborhood
        if len(ones) * len(zeros) > 2000:
            # Sample swaps rather than enumerate all
            try:
                for _ in range(200):
                    i = random.choice(ones)
                    j = random.choice(zeros)
                    delta_R = incremental_update(best_R, i, j, n)
                    new_t = min(best_R[k_star] + delta_R[k_star], np.min(best_R + delta_R))
                    # We want to improve the min(R_k) so the delta change in min(R_k) should be positive
                    delta = delta_R[k_star]
                    if delta > best_delta:
                        best_delta = delta
                        best_swap = (i, j)
                        if delta > 0:
                            break  # Early exit if we find an improving swap
            except:
                pass
        else:
            # Enumerate all swaps
            for i in ones:
                for j in zeros:
                    delta_R = incremental_update(best_R, i, j, n)
                    delta = delta_R[k_star]
                    if delta > best_delta:
                        best_delta = delta
                        best_swap = (i, j)
                        if delta > 0:
                            break  # Early exit if we find an improving swap

        if best_delta <= 0:
            break  # No improvement found, stop

        # Apply the best swap
        i, j = best_swap
        best_b[i], best_b[j] = 0, 1
        delta_R = incremental_update(best_R, i, j, n)
        best_R += delta_R
        # Update best_t to actual value
        best_t = np.min(best_R)

    return best_b, best_R, best_t

def crossover(parent1, parent2, n):
    """Uniform crossover that preserves exact weight"""
    # Take union and adjust weight
    p1_ones = set(np.where(parent1 == 1)[0])
    p2_ones = set(np.where(parent2 == 1)[0])
    union = sorted(list(p1_ones.union(p2_ones)))
    
    if len(union) > n//2:
        # Remove excess
        diff = len(union) - n//2
        for _ in range(diff):
            idx = random.choice(union)
            union.remove(idx)
    elif len(union) < n//2:
        # Add more
        comp = list(set(range(n)) - set(union))
        diff = n//2 - len(union)
        for _ in range(diff):
            idx = random.choice(comp)
            union.append(idx)
            comp.remove(idx)
    
    child = np.zeros(n, dtype=np.uint8)
    for idx in union:
        child[idx] = 1
    return child

def mutate(b, mutations=1):
    """Apply mutations to a binary sequence"""
    n = len(b)
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    new_b = b.copy()
    for _ in range(mutations):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = random.choice(ones)
        j = random.choice(zeros)
        new_b[i], new_b[j] = 0, 1
        ones = np.where(new_b == 1)[0]
        zeros = np.where(new_b == 0)[0]
    return new_b
