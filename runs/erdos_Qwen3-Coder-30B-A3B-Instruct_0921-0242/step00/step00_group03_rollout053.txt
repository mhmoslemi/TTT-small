import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 bound for given step function h"""
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # The correlation is indexed from -n+1 to n-1
    # So the corresponding k values are from -1 to 1 in steps of dx
    max_c5 = np.max(corr) * dx
    return max_c5

def gradient_c5(h):
    """Return gradient of C5 with respect to h, for the active shift"""
    dx = 2.0 / len(h)
    g = 1 - h
    corr = correlate(h, g, mode="full")
    # Find the active shift (max correlation index)
    active_idx = np.argmax(corr)
    n = len(h)
    # Convert index to shift k = (idx - n + 1) * dx
    shift = (active_idx - n + 1) * dx
    
    # Gradient w.r.t. h[i]: for shift k=shift
    grad = np.zeros_like(h)
    for i in range(n):
        # h[i] contributes to correlation with g[i+k]
        j = i + int(shift / dx)  # shifted index
        if 0 <= j < n:
            grad[i] += -dx * g[j]  # negative from h[i] term
        # g[i] contributes to correlation with h[i-k]
        j = i - int(shift / dx)
        if 0 <= j < n:
            grad[i] += dx * h[j]  # positive from g term
    
    return grad

def project_to_feasible(h, target_sum=1.0):
    """Project h to satisfy sum constraint, clipping to [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute difference uniformly
    diff = target_sum - current_sum
    n = len(h)
    if n > 0:
        h += diff / n
    return np.clip(h, 0, 1)
