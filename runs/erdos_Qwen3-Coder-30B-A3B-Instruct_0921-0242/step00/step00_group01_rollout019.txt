import numpy as np
import time
from scipy import fft

def project_box_affine(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Project onto affine constraint
    delta = (np.sum(h) - target_sum) / len(h)
    h -= delta
    # Clip again to maintain box constraint
    h = np.clip(h, 0, 1)
    return h

def autocorr(h):
    # Compute circular autocorrelation using FFT
    H = fft.fft(h)
    aut = fft.ifft(H * np.conj(H))
    return np.real(aut)

def evaluate_c5(h, dx):
    # Evaluate the overlap integral for all lags
    h_flipped = np.flip(h)
    # Compute cross-correlation (which is what we want)
    overlap = np.correlate(h, h_flipped, mode="full")
    # This gives us the sum h_i * h_{i+k} for all k
    # We want h_i * (1 - h_{i+k}) = h_i - h_i * h_{i+k}
    # So we use:
    # overlap[k] = sum_i h_i * h_{i+k}
    # Our overlap integral = dx * ( sum h_i - overlap[k] )
    # = dx * ( size * mean_h - overlap[k] ) = dx * ( n/2 - overlap[k] )
    max_overlap = np.max(overlap)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):

    best_c5 = float('inf')

    candidates = []

        candidates.append(h_initial)
    
    # 2. Random binary with smoothing
    h_rand = np.zeros(n_points)
    h_rand[:n_points//2] = 1.0
    np.random.shuffle(h_rand)
    h_rand = np.convolve(h_rand, np.ones(5)/5, mode='same')
    h_rand = project_box_affine(h_rand, target_sum)
    candidates.append(h_rand)
    
    # 3. Alternating pattern
    h_alt = np.zeros(n_points)
    for i in range(n_points):
        h_alt[i] = 0.5 if i % 2 == 0 else 0.0
    h_alt = project_box_affine(h_alt, target_sum)
    candidates.append(h_alt)
    
    # 4. Smooth step function (initially 0, then increasing to 1, then 0)
    h_step = np.zeros(n_points)
    mid = n_points // 2
    half = n_points // 4
    h_step[:half] = 0.0
    h_step[half:mid] = np.linspace(0, 1, mid - half)
    h_step[mid:mid+half] = np.linspace(1, 0, half)
    h_step[mid+half:] = 0.0
    h_step = project_box_affine(h_step, target_sum)
    candidates.append(h_step)
    
    t = time.time()
    
    # Local search with multistart
    num_starts = 20
    max_iter = 1000
    step_schedule = np.linspace(0.1, 0.001, max_iter)
    
    for i in range(num_starts):
        idx = i % len(candidates)
        h = candidates[idx].copy()
        
        # Random perturbation
        if i > 0:
            h += np.random.normal(0, 0.01, n_points)
            h = project_box_affine(h, target_sum)
        
        for it in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute autocorrelation
            aut = autocorr(h)  
            # Maximize minimum correlation (equivalently, minimize the largest value of (n/2 - aut))
            k_min = np.argmin(aut)  # This is the index of the minimum value
            
            # Gradient ascent on the worse case (min of autocorrelation)
            grad = - (np.roll(h, k_min) + np.roll(h, -k_min))
            
            # Update step size
            step = step_schedule[it]
            h_new = h + step * grad
            h = project_box_affine(h_new, target_sum)
            
            # Evaluate objective (this is the derivative of the objective)
            # We already update the value in each direction
            
        # Evaluate final solution
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Refinement: attempt higher resolution
    if best_h is not None and time.time() - start_time < budget_s - 50:
        n_points_fine = 400
        dx_fine = 2.0 / n_points_fine
        h_fine = np.interp(np.linspace(0, 2, n_points_fine),

        # Do a few optimization iterations on fine grid
        max_iter_refine = 200
        step_schedule_fine = np.linspace(0.05, 0.0005, max_iter_refine)
        
        for i in range(max_iter_refine):
            if time.time() - start_time > budget_s - 10:
                break
            aut = autocorr(h_fine)
            k_min = np.argmin(aut)
            grad = - (np.roll(h_fine, k_min) + np.roll(h_fine, -k_min))
            step = step_schedule_fine[i]
            h_fine = h_fine + step * grad
            h_fine = project_box_affine(h_fine, n_points_fine / 2)
        
        c5_fine = evaluate_c5(h_fine, dx_fine)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine
    
    # Final evaluation using numpy correlate with full mode as in task
    if best_h is not None:

        return (best_h.tolist(), float(best_c5), len(best_h))
    
    # Failed, return the initial solution
    if 'initial_h_values' in globals():
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                             np.linspace(0, 2, len(initial_h_values)), 
                             initial_h_values)
        dx = 2.0 / n_points
        c5_val = evaluate_c5(h_interp, dx)
        return (h_interp.tolist(), float(c5_val), n_points)
    else:
        # Just return a valid candidate
        h_rand = np.zeros(n_points)
        h_rand[:n_points//2] = 1.0
        np.random.shuffle(h_rand)
        h_rand = project_box_affine(h_rand, target_sum)
        c5_val = evaluate_c5(h_rand, dx)
        return (h_rand.tolist(), float(c5_val), n_points)
