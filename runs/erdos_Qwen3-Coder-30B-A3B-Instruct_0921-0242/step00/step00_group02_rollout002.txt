        return np.correlate(h, 1-h, mode="full")

    def evaluate_c5(h):
        dx = 2.0 / len(h)
        corr = correlate(h)
        return 1.0 - np.min(corr[1:]) * dx

    def project_simplex(h, target_sum):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        delta = target_sum - current_sum
        if abs(delta) < 1e-12:
            return h
        
        # Find free variables
        free = np.where((h > 0) & (h < 1))[0]
        if len(free) == 0:
            return h
        
        iteration_count = 0
        while abs(delta) > 1e-12 and len(free) > 0 and iteration_count < len(h):
            step = delta / len(free)
            h[free] += step
            # Remove entries that go out of bounds
            out_of_bounds = np.where((h[free] <= 0) | (h[free] >= 1))[0]
            if len(out_of_bounds) > 0:
                to_remove = free[out_of_bounds]
                free = np.setdiff1d(free, to_remove)
                # adjust delta
                delta = target_sum - np.sum(h)
            else:
                delta = target_sum - np.sum(h)
                iteration_count += 1
                
        return h

    def soft_min_gradient(h, alpha):
        # Compute r_k for k = 1 to n-1
        n = len(h)
        pad_h = np.concatenate([h, np.zeros(n)])
        H = fft.fft(pad_h)
        r = np.real(fft.ifft(H * np.conj(H)))[:n]
        r = r[1:]  # only k >= 1

        # Compute soft weights
        r_alpha = -alpha * r
        r_alpha -= np.max(r_alpha)
        exp_r_alpha = np.exp(r_alpha)
        weight = exp_r_alpha / np.sum(exp_r_alpha)

        # Compute gradient
        # Each component of gradient is a weighted sum of two shifted copies
        temp = np.zeros(2*n - 1)
        temp[:n] = weight
        weight_conv = fft.ifft(fft.fft(temp) * fft.fft(h)).real[:n]
        
        temp = np.zeros(2*n - 1)
        temp[-n:] = weight[::-1]
        weight_conv += fft.ifft(fft.fft(temp) * fft.fft(h[::-1])).real[:n][::-1]

        return weight_conv
