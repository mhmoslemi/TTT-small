import scipy.fft as fft

# Global constants
n_points = 240
dx = 2.0 / n_points
n_half = n_points // 2

def project_to_F(h):
    """Project h to the feasible set {h | 0<=h<=1, sum(h) = n/2}"""
    h = np.clip(h, 0.0, 1.0)
    deficit = n_half - h.sum()
    free = np.where((h>0) & (h<1))[0]
    while np.abs(deficit) > 1e-12 and free.size>0:
        delta = deficit / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        deficit = n_half - h.sum()
        free = np.where((h>0) & (h<1))[0]
    return h

def compute_C5(h):
    """Compute C5 using FFT for O(n log n) correlation computation"""
    # pad to fft_len for full convolution
    h_pad = np.zeros(fft_len)
    g_pad = np.zeros(fft_len)
    h_pad[:n_points] = h
    g_pad[:n_points] = 1 - h
    H = fft.fft(h_pad)
    G = fft.fft(g_pad)
    conv = fft.ifft(H * np.conj(G)).real
    # cut to the full correlation range (symmetric)
    conv = conv[:n_points]  # indices -n+1 to n-1
    # shift the indices back to be close to 0
    c5_val = conv.max() * dx
    return c5_val, conv

def local_improve(h, c):
    """Apply local pairwise exchanges to improve C5."""
    # Find worst shift
    k_star = np.argmax(c)
    # Local optimization using pairwise coordinate swaps
    loc_steps = 30
    for _ in range(loc_steps):
        # Find indices with non-bound values
        i1 = np.random.choice(np.where(h > 1e-6)[0])
        i0 = np.random.choice(np.where(h < 1 - 1e-6)[0])
        # Compute change in correlation
        eps = min(h[i1], 1-h[i0], 0.05)
        h_new = h.copy()
        h_new[i1] -= eps
        h_new[i0] += eps
        # Incremental update of correlation
        # The affected correlations are those where index i1 or i0 contribute
        # This is approximated by computing gofthe change
        c_new = c.copy()
        for k in range(n_points):
            ik = (i1 + k) % n_points
            jk = (i0 + k) % n_points
            # compute the effect of these changes on the shift k
            c_new[k] -= eps * dx * (1 - h[ik])  # i1 contribution
            c_new[k] -= eps * dx * h[jk]        # i0 contribution

        new_max = c_new.max()
        if new_max < c.max():
            h = h_new.copy()
            c = c_new.copy()
            # update k_star if necessary
            k_star = np.argmax(c)
        else:
            # Try again
            pass
    return h, c

def crossover(p1, p2):
    """Blend crossover"""
    alpha = np.random.rand()
    child = alpha * p1 + (1 - alpha) * p2
    return project_to_F(child)

def mutate(h, sigma):
    """Add Gaussian noise to a random subset of entries"""
    m = max(1, int(0.02 * n_points))
    indices = np.random.choice(n_points, m, replace=False)
    noise = np.random.normal(0, sigma, m)
    h_new = h.copy()
    h_new[indices] += noise
    return project_to_F(h_new)

def evaluate_erdos_solution(h):
    return compute_C5(h)[0]

        c5, _ = compute_C5(h)
        return c5

        _, c = compute_C5(h)

            # as we assume periodic extension with enough padding
            ipk = (i + k_star) % n_points

            grad[i] = dx * ((1 - h[ipk]) - h[imk])
