def evaluate_erdos_solution(h_values, n_points):
    """Evaluate the Erdős constant for given h_values."""
    dx = 2.0 / n_points
    # Compute all correlations
    correlations = np.correlate(h_values, 1 - h_values, mode="full")
    # Extract the relevant part
    middle = len(correlations) // 2
    # Only consider non-zero lags
    correlations = correlations[middle + 1:]
    # Return maximum value times dx
    return np.max(correlations) * dx

def autocorr(b):
    """Compute full autocorrelation using FFT, returning real part."""
    return np.real(np.fft.ifft(np.abs(np.fft.fft(b))**2))

def swap_improvement(h, b, k_star, m, dx, n_points, T=1.0):
    """Perform simulated annealing swap-based improvement."""
    # Precompute shift matrices for performance
    B_shift = np.zeros((n_points, n_points), dtype=int)
    for k in range(n_points):
        B_shift[k] = np.roll(b, k)
    
    # Identify indices with 1s and 0s
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return h, b, m
    
    no_improvement_count = 0
    max_improvement = 0
    attempts = 0
    starting_time = time.time()
    # Allocate time budget: 60% of total time for swaps
    swap_time_budget = 0.6 * 600
    
    # Make swaps for finite time
    while True:
        if time.time() - starting_time > swap_time_budget:
            break
        if no_improvement_count > 5000:
            break
            
        # Choose two random indices, one from each set
        i_idx = np.random.randint(len(ones))
        j_idx = np.random.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        
        # Compute delta change to the worst lag k_star
        delta = -B_shift[k_star, i] + B_shift[k_star, j]
        
        # Accept improvements unconditionally
        if delta > 0:
            # Apply the swap
            b[i], b[j] = 0, 1
            # Update h
            h[i], h[j] = 0.0, 1.0
            
            # Update the correlation of k_star
            m_new = m + delta
            m = m_new
            no_improvement_count = 0
            max_improvement = max(max_improvement, delta)
        else:
            # Accept with probability exp(delta/T)
            if np.random.rand() < np.exp(delta / T):
                # Apply the swap
                b[i], b[j] = 0, 1
                # Update h
                h[i], h[j] = 0.0, 1.0
                
                # Update the correlation of k_star
                m_new = m + delta
                m = m_new
                no_improvement_count = 0
                max_improvement = max(max_improvement, delta)
            else:
                no_improvement_count += 1
        
        # Reduce temperature slightly (simulated annealing)
        T *= 0.99999
        attempts += 1
    
    return h, b, m

def generate_seeds(n):
    """Generate several balanced binary seeds - Legendre, Singer, etc."""
    seeds = []
    
    # Generate Legendre sequence
    # Use large prime >= n
    prime = 211 if n < 211 else n+2
    while prime < n*2:
        if (prime % 2 == 1) and all(prime % i != 0 for i in range(3, int(prime**0.5)+1, 2)):
            break
        prime += 2
    
    legendre = np.zeros(n, dtype=int)
    for i in range(n):
        legendre[i] = 1 if pow(i, (prime-1)//2, prime) == 1 else 0
    
    # Adjust balance to be exactly n/2 1s
    ones_count = np.sum(legendre)
    if ones_count > n//2:
        indices_to_flip = np.where(legendre == 1)[0][:ones_count - n//2]
        legendre[indices_to_flip] = 0
    elif ones_count < n//2:
        indices_to_flip = np.where(legendre == 0)[0][:n//2 - ones_count]
        legendre[indices_to_flip] = 1
    
    # Append to seeds
    seeds.append(("Legendre", legendre.copy()))
    
    # More seeds could be added here if allowed, but for simplicity stick with one good one
    
    return seeds
