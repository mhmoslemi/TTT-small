import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Evaluate C5 for a given step function h."""
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The full correlation has length 2*n-1, corresponding to shifts from -(n-1) to (n-1)
    # We're interested in the maximum correlation (maximum C5)
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, active_shift_idx):
    """Compute gradient for the active shift index."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Determine shift value k = (idx - (n-1)) * dx
    k_idx = active_shift_idx - (n - 1)
    k = k_idx * dx
    
    # Gradient computation for each index i
    # ∂C5/∂h_i = dx * (1 - h_{i+k}) - dx * h_{i-k}
    # Use periodic wrapping
    for i in range(n):
        # Contributions from the term -dx * h_{i+k}
        j_plus = (i + k_idx) % n
        grad[i] -= dx * h[j_plus]
        
        # Contributions from the term dx * h_{i-k}
        j_minus = (i - k_idx) % n
        grad[i] += dx * h[j_minus]
    
    return grad

def c5_of_h(h):
    """Recompute C5 of h vector."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n):
    """Project h to satisfy sum constraint."""
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    
    delta = target_sum - current_sum
    # Distribute adjustment to all entries
    h = h + delta / n
    h = np.clip(h, 0, 1)
    return h
