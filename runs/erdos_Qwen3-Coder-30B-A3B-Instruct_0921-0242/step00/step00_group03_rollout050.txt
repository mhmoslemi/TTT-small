def project_to_F(h, n):
    """
    Project h onto the feasible simplex with sum = n/2 and bounds [0,1]
    """
    h = np.clip(h, 0.0, 1.0)
    deficit = n / 2 - h.sum()
    free = np.where((h > 0) & (h < 1))[0]
    while np.abs(deficit) > 1e-12 and free.size > 0:
        delta = deficit / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
        deficit = n / 2 - h.sum()
        free = np.where((h > 0) & (h < 1))[0]
    return h

def fast_correlation(h):
    """
    Fast correlation using FFT
    """
    dx = 2.0 / len(h)
    g = 1 - h
    n = len(h)
    L = 2*n
    h_pad = np.pad(h, (0, L - n), 'constant')
    g_pad = np.pad(g, (0, L - n), 'constant')
    H = fft(h_pad)
    G = fft(g_pad)
    conv = ifft(H * np.conj(G)).real * dx
    return conv

def fast_C5(h):
    """
    Compute C_5 using fast correlation
    """
    conv = fast_correlation(h)
    return conv.max()

def local_improve(h, loc_steps=30, dx=2.0/len(h)):
    """
    Local polishing using pairwise exchanges on worst correlation
    """
    c = fast_correlation(h)
    k_star = np.argmax(c)
    # Shift k* is from -n+1 to n-1, but we need 0-based index for array bounds
    # k_star represents offset, convert to shift index
    #   conv is length 2*n-1, indexed from 0 to 2*n-2
    #   shift = k_star - (n-1) => k_star = shift + (n-1)
    # So actual shift = k_star - (n - 1)
    # When shift goes from -n+1 to n-1 (inclusive), k_star goes from 0 to 2*n-2.
    shift = k_star - (len(h) - 1)

    # For penalty calculation, we need to avoid the problematic cases
    for _ in range(loc_steps):
        # Look for approximately balanced indices where swapping can help
        non_zero = np.where(h > 0)[0]
        non_one = np.where(h < 1)[0]
        if len(non_zero) == 0 or len(non_one) == 0:
            break
        i1 = np.random.choice(non_zero)
        i0 = np.random.choice(non_one)

        # Compute potential improvement
        eps = min(h[i1], 1 - h[i0], 0.05)
        if eps < 1e-6:
            break
        # Update h inplace
        h[i1] -= eps
        h[i0] += eps

        # Incremental update of the convolution to avoid recomputation
        c = c - eps * dx * (np.roll(1-h, shift) - np.roll(h, -shift))
        # Otherwise would use a more complex update logic ...
        # For simplicity, just recompute after large strides:
        # Though for low step counts and translations, the cost is low.
        # Re-evaluate current value
        new_c = fast_correlation(h)
        if new_c.max() < c.max():
            c = new_c
        else:
            # revert
            h[i1] += eps
            h[i0] -= eps

    return h, fast_C5(h)

def crossover(p1, p2):
    """
    Blend crossover between two parents
    """
    alpha = np.random.rand()
    child = alpha * p1 + (1 - alpha) * p2
    return child

def mutate(h, sigma):
    """
    Mutation with Gaussian noise
    """
    n = len(h)
    m = max(1, int(0.02 * n))
    indices = np.random.choice(n, m, replace=False)
    noise = np.random.normal(0, sigma, m)
    h[indices] += noise
    return h

def tournament_selection(pop, k=3):
    """
    Tournament selection with k participants
    """
    chosen = np.random.choice(len(pop), k, replace=False)
    best = min(chosen, key=lambda i: pop[i][1]) # Minimize C5 bound
    return pop[best]

        h_block[:n_points//2] = 1.0
        h_block = project_to_F(h_block, n_points)
        pop.append((h_block.copy(), fast_C5(h_block)))
        
        # Uniform + noise
        h_uni = np.full(n_points, 0.5)
        h_uni += np.random.normal(0, 0.05, n_points)
        h_uni = project_to_F(h_uni, n_points)
        pop.append((h_uni.copy(), fast_C5(h_uni)))
        
        # Add random individuals to complete population
        for _ in range(pop_size - 5 - 2 - 1 - 1): # 5+2+1+1 = 9
            h = np.random.random(n_points)
            h = project_to_F(h, n_points)
            pop.append((h.copy(), fast_C5(h)))
            
        return pop
    
    # Start the optimization process
    start_time = time.time()
    
    pop = initialize_population()
    h_best = min(pop, key=lambda x: x[1])[0]
    C5_best = fast_C5(h_best)

        for _ in range(pop_size):
            parent1 = tournament_selection(pop)
            parent2 = tournament_selection(pop)
            child = crossover(parent1[0], parent2[0])
            child = mutate(child, sigma)
            child = project_to_F(child, n_points)
            
            # Apply memetic local refinement
            child, child_c5 = local_improve(child, loc_steps)
            next_pop.append((child, child_c5))
        
        # Merge with current population and remove worst
        combined_pop = pop + next_pop
        combined_pop.sort(key=lambda x: x[1])
        pop = combined_pop[:pop_size]

        if pop[0][1] < C5_best:

                h = np.random.random(n_points)

    def slsqp_objective(h):
        return fast_C5(h)

        return np.sum(h) - n_points/2

    # Manually compute gradient for the active shift:
    # dJ/dh_i = Δx * (1 - h_{i+k}) - h_{i-k}
    def compute_gradient(h, shift, dx):
        g = 1 - h 
        grad = np.zeros_like(h)
        # Compute gradient term by term
        for i in range(n_points):
            second_term = (h[(i-shift) % n_points] if (i - shift) >= 0 and (i - shift) < n_points else 0)

    h_refined += np.random.normal(0, 1e-4, n_points)
