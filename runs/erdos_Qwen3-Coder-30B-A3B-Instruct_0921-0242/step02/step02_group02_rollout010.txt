def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def logsumexp(x, beta=5):
    """Log-sum-exp smooth approximation of max"""
    x = np.array(x)
    return 1/beta * np.log(np.sum(np.exp(beta * x)))

def smooth_max_overlap(h, beta=10):
    """Smooth approximation to max overlap"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1-h, mode='full')
    corr_scaled = corr * dx
    return logsumexp(corr_scaled, beta)

def smooth_gradient(h, k, beta=10):
    """Gradient of smooth max overlap w.r.t. h_i"""
    dx = 2.0 / len(h)
    # Assuming shift k is relative inside bounds
    n = len(h)
    grad = np.zeros_like(h)
    # Compute partial derivative ∂c_k/∂h_i
    for i in range(n):
        # c_k = dx * (n/2 - sum(h_i * h_{i-k}))
        # ∂c_k/∂h_i = -dx * h_{i-k}
        idx = (i - k) % n
        grad[i] -= dx * h[idx]
        # Also ∂c_k/∂h_i = dx * h_{i+k} (from the second part of 1-h_{i+k})
        # Actually this is only correct for the definition used in evaluate_c5
        # Need to compute correct terms for all necessary h terms
        # But for now, using a refined formulation
        # ∫h(x)(1-h(x+k)) dx = ∫h(x)dx - ∫h(x)h(x+k)dx  
        # So gradient is darboux (∂c_k/∂h_i = dx * h_{i+k})
        idx_flip = (i + k) % n
        grad[i] += dx * h[idx_flip]
    return grad

def compute_gradients(h):
    """Return all gradients for translation k from 0 to n-1"""
    n = len(h)
    grads = np.zeros((n, n))
    dx = 2.0 / n
    for k in range(n):
        # Compute gradient vector v_k
        # ∂c_k/∂h_i = dx * (h_{i+k} - h_{i-k})
        grad = np.zeros_like(h)
        for i in range(n):
            idx1 = (i + k) % n
            idx2 = (i - k) % n
            grad[i] = dx * (h[idx1] - h[idx2])
        grads[k] = grad
    return grads

        corr = correlate(h, 1-h, mode="full")
        k_star = np.argmax(corr)
        if k_star > len(h):
            k_star = k_star - len(h)
        else:
            k_star = k_star - len(h) + 1
            
        grad = smooth_gradient(h, k_star, beta=10)
        # Subgradient descent step
        h_new = h - step_size * grad
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n)
        
        # Stopping criterion
        if np.allclose(h, h_new, atol=1e-5):
            break
        h = h_new
    
    return h

def solve_sdp(n):
    """Solve the SDP relaxation of the autocorrelation problem"""
    dx = 2.0 / n
    # Construct circulant matrices C_k for k = 0,...,n-1
    C_list = []
    for k in range(n):
        C = np.zeros((n, n))
        for i in range(n):
            j = (i - k) % n  # circular indexing
            C[i, j] = 1
        C_list.append(C)
    
    # Variables for relaxation
    X = cp.Variable((n, n), symmetric=True)
    h = cp.Variable(n)
    gamma = cp.Variable()
    
    # Constrain (h_i)^2 = h_i (binary constraint as relaxation fails anyway)
    # Instead, by the SDP, X = hh^T with X >= 0
    
    # Expectation constraints; from C_k* h = ∑ h_i h_{i-k}
    # We want minimized max of  dx(n/2 - h^T C_k h)
    # => minimize gamma such that h^T C_k h <= (n/2 - gamma/dx)
    
    constraints = [X == h @ h.T,
                   X >> 0,  # Positive semidefinite
                   h >= 0,
                   h <= 1,
                   cp_sum(h) == n/2
                   ]
    
    for k in range(n):
        constraints.append(h.T @ C_list[k] @ h >= n/2 - gamma/dx)
        
    prob = cp.Problem(Minimize(gamma), constraints)
    
    try:
        prob.solve(solver='SCIPY', verbose=False)
        h_opt = h.value
        if h_opt is not None:
            return h_opt
    except:
        return None
