import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    h_copy = np.clip(h_copy, 0, 1)
    return h_copy

def fast_correlation_update(corr, h, i, j):
    """Update correlation array after swapping h[i] and h[j]."""
    # Add contribution from i -> j
    # Only relevant shifts where i and j are within the valid range
    n = len(h)
    for s in range(n - 1):
        if i - s >= 0 and j < n:
            corr[s] += (h[i] - h[j]) * (1 - h[i - s]) - (h[j] - h[i]) * (1 - h[j + s])
        if j - s >= 0 and i < n:
            corr[s] += (h[j] - h[i]) * (1 - h[j - s]) - (h[i] - h[j]) * (1 - h[i + s])
    return corr

def AnnealBinary(h, n_points, time_limit):
    """Perform simulated annealing on binary vector to minimize max correlation."""
    T0 = 0.02 * n_points
    alpha = 0.997
    max_moves = 50000
    
    # Compute initial correlations
    h_view = h.astype(int)
    corr = np.zeros(n_points)
    for s in range(n_points):
        corr[s] = np.sum(h_view[:-s] * (1 - h_view[s:])) if s < n_points else 0

    best_corr = np.max(corr)
    best_h = h_view.copy()
    current_corr = best_corr
    T = T0
    move_count = 0
    last_improvement = 0
    tabu_list = set()
    
    while T > 1e-5 and move_count < max_moves:
        if time.time() > time_limit:
            break
        move_count += 1
        if move_count - last_improvement > 5000:
            break  # Early-stop heuristic

        # Pick two indices to swap
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        while h_view[i] == 0 or h_view[j] == 1:  # Ensure we can swap
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
        
        # Check tabu
        key = (i, j) if i < j else (j, i)
        if key in tabu_list:
            continue
            
        # Update correlation
        old_corr_max = current_corr
        new_corr = corr.copy()
        # Direct update of correlations (update is in-place)
        # Additive changes for the possibly affected shifts
        for s in range(n_points):
            if i >= s and j < n_points:
                new_corr[s] += (1 - h_view[i]) * h_view[i - s] - (h_view[i] - h_view[j]) * (1 - h_view[i - s]) 
            if j >= s and i < n_points:
                new_corr[s] += (h_view[j] - h_view[i]) * (1 - h_view[j - s]) - (1 - h_view[j]) * h_view[j + s] if s > 0 else 0

        new_max = np.max(new_corr)
        
        # Accept or reject
        if new_max <= old_corr_max:
            # Accept
            corr = new_corr.copy()
            current_corr = new_max
            h_view[i], h_view[j] = h_view[j], h_view[i]
            if current_corr < best_corr:
                best_corr = current_corr
                best_h = h_view.copy()
                last_improvement = move_count
        else:
            # Accept with probability
            prob = np.exp(-(new_max - old_corr_max) / T)
            if np.random.rand() < prob:
                corr = new_corr.copy()
                current_corr = new_max
                h_view[i], h_view[j] = h_view[j], h_view[i]
                if current_corr < best_corr:
                    best_corr = current_corr
                    best_h = h_view.copy()
                    last_improvement = move_count
        
        # Update temperature and tabu
        T *= alpha
        tabu_list.add(key)
        if len(tabu_list) > 100:
            tabu_list.pop()
    
    return best_h, best_corr

def Polishing(h_binary, n_points):
    """Perform continuous refinement on binary solution."""
    # Convert to float
    h = h_binary.astype(float)
    dx = 2.0 / n_points
    
    # Apply several rounds of smooth max approximation
    lambda_val = 5.0
    eta = 0.1 * dx
    n_steps = 30
    
    for step in range(n_steps):
        # Compute correlations
        corr = np.zeros(n_points)
        for s in range(n_points):
            if s == 0:
                # This is a different form; center mistakedly used all points, but should only be ...[:len(h)-s] * (1-h[s:])
                corr[s] = np.sum(h[:-s] * (1 - h[s:])) if s < n_points else 0
            else:
                corr[s] = np.sum(h[:-s] * (1 - h[s:])) if s < len(h) else 0
                
        # Smooth approximation of max
        # Phi = log(sum(exp(lambda * corr)))/lambda
        # Gradient = [exp(lambda*corr_i) / sum(exp(lambda*corr))] * (1 - h_i - shift(h_i+s)) etc.
        # Rather compute constant multipliers and approximate gradient per-sample
        
        # For numerical stability, shift coefficients
        corr_shifted = corr - np.max(corr)
        exp_corr = np.exp(lambda_val * corr_shifted)
        sum_exp = np.sum(exp_corr)
        phi = np.log(sum_exp) / lambda_val

        # Compute gradient based on function f_i = sum(corr - corr_s * (1-h_{i+s}) * delta_1)
        # Simple direct gradient estimations
            
        # Gradient for each sample, assume small perturbations
        # Gradient of i-th entry w.r.t to the simplified sum of cross-correlations
        grad = np.zeros(n_points)
        for i in range(n_points):
            grad[i] = 0.0
            for s in range(1, n_points):
                if i + s < n_points:
                    grad[i] += (1 - h[i + s])  # derivative from point i
                if i - s >= 0:
                    grad[i] += h[i - s]  # derivative can also originate from point (i - s)

        # Final stretch
        h_new = h - eta * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_feasible(h_new, n_points)
        h = h_new.copy()
        
        # Decaying step size
        if step % 10 == 0:
            eta /= 2.0
            
        # Increase lambda if needed 
        if lambda_val < 30.0:
            lambda_val += 5.0 / 10
        else:
            lambda_val = 30.0
            
    return h, evaluate_c5(h)
