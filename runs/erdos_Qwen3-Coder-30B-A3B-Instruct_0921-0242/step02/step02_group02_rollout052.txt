import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.fft import rfft, irfft

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling approach for adjusting sum
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def build_cyclic_shift_matrix(k, n):
    """Build a sparse cyclic shift matrix P_k, where (P_k)[i,j] = 1 if j = (i-k) mod n."""
    P = np.zeros((n, n))
    for i in range(n):
        j = (i - k) % n
        P[i, j] = 1
    return P

def subgradient_refine(h, dx, budget_s):
    """Refine h using projected subgradients."""
    start_time = time.time()
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute full correlation
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_star = np.argmax(corr)
        if k_star > len(current_h):
            k_star = k_star - len(current_h)
        else:
            k_star = k_star - len(current_h) + 1
        
        # Compute subgradient
        grad = np.zeros_like(current_h)
        for i in range(len(current_h)):
            left_idx = (i + k_star) % len(current_h)
            right_idx = (i - k_star) % len(current_h)
            grad[i] = dx * (current_h[left_idx] - current_h[right_idx])
        
        # Line search
        alpha = 1.0
        step_size = 1e-5
        max_iter = 10
        for _ in range(max_iter):
            h_try = current_h - alpha * grad
            h_try = project_to_feasible(h_try, len(current_h))
            c5_try = evaluate_c5(h_try)
            if c5_try < current_c5:
                current_h = h_try
                current_c5 = c5_try
                break
            alpha *= 0.5
            
        if alpha < 1e-8:
            break
            
    return current_h, current_c5

def upsample_and_project(h0, n2):
    """Upsample h0 to resolution n2 with linear interpolation."""
    n = len(h0)
    dx = 2.0 / n
    dx2 = 2.0 / n2
    
    # Create upsampled grid
    x_orig = np.linspace(0, 2, n, endpoint=False)
    x_up = np.linspace(0, 2, n2, endpoint=False)
    
    # Interpolate
    h_up = np.interp(x_up, x_orig, h0, period=2)
    
    # Project sum constraint
    h_up = h_up * (n2 / 2) / np.sum(h_up)
    
    # Clip to feasible set
    h_up = np.clip(h_up, 0, 1)
    
    return project_to_feasible(h_up, n2)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    # Choose smaller n for faster SDP solving, try n=180
    n = 180
    dx = 2.0 / n

    # Pre-compute shift matrices for a sparse assembly of constraints
    P = [build_cyclic_shift_matrix(k, n) for k in range(n)]
    
    # SDP formulation
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()
    
    diag_constraints = [cp.diag(X) >= 0, cp.diag(X) <= 1]
    sum_constraint = cp.sum(cp.diag(X)) == n/2
    auto_constraints = [cp.trace(P[k] @ X) >= t for k in range(n)]
    
    prob = cp.Problem(cp.Maximize(t), diag_constraints + [sum_constraint] + auto_constraints)
    
    try:
        prob.solve(solver='SCS', max_iters=25000, eps=1e-5)
        X_opt = X.value
    except Exception as e:
        print(f"SDP solver issue: {e}")
        # Fall back to approximate routine if solver fails
        return None, None, None
    
    # Extract candidate vector using dominant eigenpair
    eigvals, eigvecs = np.linalg.eigh(X_opt)
    lead = eigvecs[:, -1] * np.sqrt(eigvals[-1])
    
    # Ensure the lead vector is feasible
    h0 = project_to_feasible(lead, n)
    
    # Try multiple random draws
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    num_random = min(15, 20)
    for _ in range(num_random):
        g = np.random.randn(n) @ np.linalg.cholesky(X_opt)
        h = project_to_feasible(np.clip(g, 0, 1), n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Local refinement with sub-gradient descent
    refine_time = min(0.15 * budget_s, budget_s - (time.time() - start_time) - 1.0)
    if refine_time > 0.1:
        best_h, best_c5 = subgradient_refine(best_h, dx, refine_time)

    # If enough time, upsample
    if time.time() - start_time < 0.85 * budget_s and n < 500:
        n2 = 2*n
        h_up = upsample_and_project(best_h, n2)
        refine_time2 = min(0.1 * budget_s, budget_s - (time.time() - start_time) - 1.0)
        if refine_time2 > 0.1:
            best_h, best_c5 = subgradient_refine(h_up, 2.0/n2, refine_time2)

    return (best_h, best_c5, len(best_h))
