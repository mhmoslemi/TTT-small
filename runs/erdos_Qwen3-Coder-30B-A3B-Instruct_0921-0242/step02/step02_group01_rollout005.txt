import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def smooth_max(h, p=64):
    """Compute smooth approximation to max over shifts."""
    dx = 2.0 / len(h)
    # Compute C(k) for all k
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    # k=0 -> full correlation [0:2n-1] -> corr[0] is self-correlation
    # To associate with C(k), we take k=0 ... n-1 from corr[1:n]
    # Also, corr values at indices 0 to n-1 correspond to shifts 0 to n-1
    # shift 0: no shift, so add dx * (1 - sum(h**2)) 
    # But we compute fft of h * conj(1-h) -> let's get correct indexing
    # corr[i] = \sum_j h[j] * (1-h[j+i]) 
    # Take values corr[1] to corr[n]
    overlap = corr.real[:len(h)]
    C_vals = 1 - dx * overlap  # C(k) = 1 - dx * sum h[i] * h[i+k]
    
    # We want max C_vals, using smooth replacement with p-norm
    # Avoid numerical issues
    C_vals = np.maximum(C_vals, 1e-12)
    smooth_val = (np.sum(C_vals ** p)) ** (1/p)
    return smooth_val

def compute_gradient(h, p=64):
    """Compute gradient of smooth max with respect to h."""
    n = len(h)
    dx = 2.0 / n
    # Compute full auto-correlation of h
    h_fft = fft(h, 2 * n)
    h2_fft = fft(h * h, 2 * n)
    corr_hh = ifft(h_fft * np.conj(h_fft))
    corr_h1mh = ifft(h_fft * np.conj(1 - h))
    C_vals = np.zeros(n)
    for k in range(n):
        # Just copy what was direct computation
        # Each shift will generate contribution to gradient
        C_vals[k] = 1 - dx * corr_h1mh.real[k]
    
    # Compute smooth max value with p
    smooth_val = np.sum(C_vals ** p) ** (1/p)
    if smooth_val < 1e-15:
        smooth_val = 1e-15
    
    # Compute gradient of smooth max
    # ∂S_p = (1/S_p^(p-1)) * \sum_k w_k * ∂C_k/h_i where w_k = C_k^(p-1)
    grad = np.zeros(n)
    C_p1 = np.power(np.maximum(C_vals, 1e-12), p - 1)
    S_p_off = smooth_val ** (p - 1)
    
    for i in range(n):
        grad_term = 0.0
        for k in range(n):
            # ∂ C(k)/∂ h_i = dx * [ (1-h_{i+k}) - h_{i-k} ]
            h_iplusk = h[(i + k) % n] if i + k < n else 0
            h_imink = h[(i - k) % n] if i - k >= 0 else 0
            partial = dx * ( (1 - h_iplusk) - h_imink )
            grad_term += C_p1[k] * partial
        
        grad[i] = grad_term / S_p_off
    
    return grad

def refit_sum_constraint(h, n):
    """Project h to satisfy sum(h) = n/2 using waterfilling."""
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def evaluate_C5(h):
    """Exactly compute C5 (true max)."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation (excluding shift 0)
    max_corr = np.max(corr.real[1:]) * dx
    return max_corr

def generate_seeds(n, num_seeds=30):
    """Generate diverse feasible starting points."""
    seeds = []
    
    # 1. uniform half-filled
    h_uniform = np.ones(n) * 0.5
    seeds.append(project_to_feasible(h_uniform, n))
    
    # 2. from a Dirichlet distribution (note: not really dirichlet, we approach with uniform samples)
    # We'll make a sample with sum = n/2, each uniform [0,1]
    for _ in range(num_seeds // 5):
        h_dir = np.random.rand(n)
        h_dir = h_dir * (n/2) / np.sum(h_dir)
        seeds.append(project_to_feasible(h_dir, n))

        h_interp = h_interp * (n/2) / np.sum(h_interp)
        seeds.append(project_to_feasible(h_interp, n))

    # 4. binary patterns (φ pattern) - approximated
    # Simple binary with exactly half ones
    for _ in range(num_seeds // 5):
        h_bin = np.ones(n)
        h_bin[:n//2] = 0
        np.random.shuffle(h_bin)
        h_bin = h_bin * (n/2) / np.sum(h_bin)
        seeds.append(project_to_feasible(h_bin, n))

    # 5. baseline seed (all 0.5 filled with sum constraint)
    if len(seeds) < num_seeds:
        h_baseline = np.ones(n) * 0.5
        seeds.append(project_to_feasible(h_baseline, n))

    return seeds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup basic variables
    n = 350
    max_iter_per_p = 150
    p_vals = [8, 16, 32, 64, 128, 256]
    
    # Generate seeds
    seeds = generate_seeds(n)
    best_c5 = float('inf')
    best_h = None
    
    # Optimisation time checking
    remaining_budget = budget_s * 0.6  # Reserve first ~60% for complex optimization
    start_opt = time.time()
    
    for p in p_vals:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate each seed
        current_best = float('inf')
        current_best_h = None
        
        for h_start in seeds[:len(seeds) // 2]:  # Use only best half at later stages
            if time.time() - start_time > budget_s * 0.9:
                break
                
            # This is a proxy for our subgradient descent with smooth max
            def obj(h):
                # add penalty for sum constraint (softly)
                h = np.clip(h, 0, 1)
                sum_err = np.sum(h) - n/2
                penalty = 1e4 * sum_err ** 2
                return smooth_max(h, p) + penalty
            
            def grad(h):
                # This is gradient w.r.t. h of smooth surrogate
                h = np.clip(h, 0, 1)
                return compute_gradient(h, p) + 2 * 1e4 * sum_err
            
            # Run optimization
            result = minimize(obj, h_start, method='L-BFGS-B', jac=grad, bounds=[(0,1)]*n, 
                              options={'maxiter': max_iter_per_p})
            # After optimization, we may need to re-impose constraints
            h_opt = project_to_feasible(result.x, n)
            # Evaluate true C5 for final result
            c5_trial = evaluate_C5(h_opt)
            if c5_trial < current_best:
                current_best = c5_trial
                current_best_h = h_opt.copy()
        
        # After trying all seeds on this p, keep best result from this stage
        if current_best < best_c5:
            best_c5 = current_best
            best_h = current_best_h.copy()
            
        # Early termination based on evaluated best
        if time.time() - start_time > budget_s * 0.9:
            break
    
    if best_h is None:
        best_h = np.ones(n) * 0.5
        best_c5 = float('inf')
    
    # Worst shift refinement (deterministic ascent on worst shift)
    if time.time() - start_time < budget_s * 0.95:
        h_refined = best_h.copy()
        for _ in range(200):  # Limited iterations to stay under time
            if time.time() - start_time > budget_s * 0.95:
                break
            # Compute C_k for all k
            dx = 2.0 / len(h_refined)
            h_fft = fft(h_refined, 2 * len(h_refined))
            ih_fft = fft(1 - h_refined, 2 * len(h_refined))
            corr = ifft(h_fft * np.conj(ih_fft))
            C_vals = np.zeros(len(h_refined))
            for k in range(len(h_refined)):
                C_vals[k] = 1 - dx * corr.real[k]
            
            # Find worst shift
            k_worst = np.argmax(C_vals[1:]) + 1  # ignore shift 0
            # Compute gradient for this k
            grad = np.zeros(len(h_refined))
            for i in range(len(h_refined)):
                # Compute ∂C(k)/∂h_i = dx * [(1 - h_{i+k}) - h_{i-k}]
                h_iplusk = h_refined[(i + k_worst) % len(h_refined)]
                h_imink = h_refined[(i - k_worst) % len(h_refined)]
                grad[i] = dx * ((1 - h_iplusk) - h_imink)
            # Line search
            alpha = 0.1
            max_alpha = 1.0
            best_alpha = 0.0
            while alpha <= max_alpha:
                h_test = h_refined - alpha * grad
                h_test = project_to_feasible(h_test, n)
                c5_test = evaluate_C5(h_test)
                if c5_test <= evaluate_C5(h_refined):
                    best_alpha = alpha
                    break
                alpha *= 0.5
            if best_alpha > 0:
                h_refined = h_refined - best_alpha * grad
                h_refined = project_to_feasible(h_refined, n)
            else:
                break
    
    # Final evaluation of C5
    final_c5 = evaluate_C5(h_refined)
    
    return (h_refined, final_c5, n)
