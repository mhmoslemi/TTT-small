import numpy as np

def project_box_sum(y, total):
    """Project y onto {x : 0 ≤ x[i] ≤ 1, sum x = total}."""
    a, b = 0.0, 1.0
    y = np.asarray(y)
    n = len(y)
    
    def sum_clipped(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    
    # Bisection to find lambda such that sum_clipped(lambda) = total
    for _ in range(40):
        lam = (a + b) / 2
        s = sum_clipped(lam)
        if s > total:
            a = lam
        else:
            b = lam
    lam = (a + b) / 2
    x = np.clip(y - lam, 0, 1)
    return x

def softmax_weights(C, beta):
    """Compute softmax weights from vector C with temperature beta."""
    C = np.asarray(C)
    C_centered = C - np.max(C)
    w = np.exp(beta * C_centered)
    w = w / np.sum(w)
    return w

def gradient_from_weights(h, w):
    """Compute gradient from weights w."""
    n = len(h)
    # w is 2n-1 long; h is n
    # w[0] corresponds to shift -(n-1)
    # w[n-1] corresponds to shift 0
    # w[2n-2] corresponds to shift n-1
    
    # Compute conv(w, h) assuming full mode
    # Give the full convolution result, then we'll extract the middle part
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    
    # Now take the (n,) slice where shifts are from -n+1 to n-1
    # i.e., indices n-1 to n-1+n
    grad = -(2.0 / len(h)) * (conv1[n-1:n-1+len(h)] + conv2[n-1:n-1+len(h)])
    return grad

def evaluate_c5(h):
    """Evaluate C5 constant for given h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

            if c5 < prev_best:
                step_init = min(step_init * 1.1, 2.0)
            else:
