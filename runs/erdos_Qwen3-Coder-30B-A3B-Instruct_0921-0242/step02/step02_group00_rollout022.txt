import numpy as np
import time
from scipy.signal import correlate

def project_box_sum(y, target_sum):
    """Project y onto the set {x | 0 ≤ x ≤ 1, sum(x) = target_sum} using water filling."""
    y = np.asarray(y)
    n = len(y)
    
    def S(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    
    # Binary search for lambda
    low, high = np.min(y) - 1.0, np.max(y) + 1.0
    eps = 1e-12
    while high - low > eps:
        mid = (low + high) / 2
        if S(mid) > target_sum:
            low = mid
        else:
            high = mid
    
    lam = (low + high) / 2
    x = np.clip(y - lam, 0, 1)
    return x

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def evaluate_autocorr(h):
    """Return the full autocorrelation vector."""
    return correlate(h, h, mode='full')

def update_autocorr_delta(γ, h, i, j, Δ, n):
    """Update γ by transferring Δ from i to j."""
    # Temporary h vector after removing Δ from i
    ĥ = h.copy()
    ĥ[i] -= Δ
    
    Δγ = np.zeros_like(γ)
    
    # Update for each shift s
    for s in range(-n+1, n):
        idx = s + n - 1
        # Removal of Δ from i
        if 0 <= i + s < n:
            Δγ[idx] -= Δ * ĥ[i + s]
        if 0 <= i - s < n and s != 0:
            Δγ[idx] -= Δ * ĥ[i - s]
        # Addition of Δ to j
        if 0 <= j + s < n:
            Δγ[idx] += Δ * ĥ[j + s]
        if 0 <= j - s < n and s != 0:
            Δγ[idx] += Δ * ĥ[j - s]
    # Self term correction
    Δγ[n - 1] += -2 * h[i] * Δ + Δ * Δ + 2 * h[j] * Δ + Δ * Δ
    
    return Δγ

def compute_worst_shift_set(C, C5, ε=1e-5):
    """Returns indices where C is within ε of C5."""
    return np.where(np.abs(C - C5) < ε)[0]

def compute_score(h, worst_shifts, n):
    """Compute score for each index based on contributions to worst shifts."""
    score = np.zeros_like(h)
    for s in worst_shifts:
        # s is in range -(n-1) to (n-1)
        # we map s to index in [0, 2n-2]
        idx = s + n - 1
        # add contribution from h[p+s] and h[p-s]
        # pixel p contributes when 0 <= p+s < n and 0 <= p-s < n
        for p in range(n):
            if 0 <= p + s < n:
                score[p] += h[p + s]
            if 0 <= p - s < n and s != 0:
                score[p] += h[p - s]
    return score
