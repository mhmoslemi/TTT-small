import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import bisect

def project_box_sum(y, total):
    """Project y onto the set {x : 0 <= x_i <= 1, sum x_i = total}"""
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if abs(s - total) < 1e-10:
        return y
    # Use bisection on lambda to find the right shift
    def sum_with_lambda(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    if s < total:
        # we need to increase values 
        a, b = 0, 1
        while sum_with_lambda(b) < total:
            b *= 2
    else:
        a, b = -1, 0
        while sum_with_lambda(a) > total:
            a *= 2
    lambda_val = bisect(lambda lam: sum_with_lambda(lam) - total, a, b, xtol=1e-12)
    y = np.clip(y - lambda_val, 0, 1)
    return y

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def generate_legendre_code(n):
    """Generate a Legendre code of length n."""
    # We'll use the prime 257 which gives us a good length (256)
    p = 257
    if n >= p-1:
        # We don't want the full period for consistency with previous code
        n = p - 1
    legendre = [0] * n
    for i in range(1, n+1):
        # Legendre symbol (i | 257)
        symbol = pow(i, (p-1)//2, p)
        if symbol == 1:
            legendre[i-1] = 1
        else:
            legendre[i-1] = 0
    return np.array(legendre)

def generate_m_sequence(n):
    """Generate a maximal length m-sequence of length n."""
    # We'll construct an LFSR of order 8
    # The primitive polynomial is x^8 + x^4 + x^3 + x^2 + 1
    lfsr = np.zeros(8, dtype=int)
    lfsr[0] = 1
    tap = [4, 3, 2, 0]
    sequence = []
    for _ in range(n-1):
        feedback = 0
        for i in tap:
            feedback ^= lfsr[i]
        sequence.append(lfsr[7])
        for i in reversed(range(7)):
            lfsr[i+1] = lfsr[i]
        lfsr[0] = feedback
    # Append a zero to make length n
    sequence.append(0)
    return np.array(sequence)

def generate_random_balanced(n):
    """Generate a random balanced binary string."""
    # Generate roughly n/2 ones
    idxs = np.random.choice(n, size=n//2, replace=False)
    s = np.zeros(n)
    s[idxs] = 1
    return s

def compute_autocorrelation(h):
    """Compute the aperiodic autocorrelation of h"""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), 'constant')
    h_conj = np.pad(h[::-1], (0, n-1), 'constant')
    corr = correlate(h_padded, h_conj, mode='valid')
    return corr[:n]

def update_autocorr_delta(γ, h, i, j):
    """Given current γ and h, compute the change in γ due to swapping h[i] and h[j]
    Assumes h[i] is 1 and h[j] is 0 for now.
    """
    n = len(h)
    # γ_k = \sum_i h[i] h[i+k]
    # When we swap position i (from 1 to 0) and j (from 0 to 1)
    # Change is due to shifts of i+k, i-k in first part and j+k, j-k in second part
    Δγ = np.zeros(n)
    for k in range(-n+1, n):
        # The correlation shifts are effectively concats and overlap checks
        if 0 <= (i + k) < n and 0 <= (i - k) < n:
            Δγ[k + n - 1] += -h[i] * h[i + k] - h[i] * h[i - k]
        if 0 <= (j + k) < n and 0 <= (j - k) < n:
            Δγ[k + n - 1] += h[j] * h[j + k] + h[j] * h[j - k]
    return Δγ

def binary_hill_climb(h0, max_iters=10000):
    """Hill-climb from a binary h0 to reduce maximum correlation."""
    h = h0.copy().astype(float)
    γ = compute_autocorrelation(h)
    best_h = h.copy()
    best_C5 = evaluate_c5(h)
    
    for it in range(max_iters):
        # Pick a random index where h is 1 and another where it is 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # compute the change
        Δγ = update_autocorr_delta(γ, h, i, j)
        new_γ = γ + Δγ
        # ... compute the new C5, which will be max term in (1 - 2/n * γ_k)
        new_C5 = 1 - (2/len(h)) * np.max(new_γ)
        if new_C5 < best_C5:
            # accept the swap
            h[i] = 0
            h[j] = 1
            γ = new_γ
            best_C5 = new_C5
            best_h = h.copy()
    
    return best_h, best_C5

def spectral_alternating_projection(h, max_iter=400, alpha_start=0.7, alpha_increase=0.02):
    """Improve h using SAP to flatten autocorrelation."""
    n = len(h)
    h = h.astype(float)
    α = alpha_start
    best_C5 = evaluate_c5(h)
    best_h = h.copy()
    
    for iter in range(max_iter):
        H = fft(h)
        P = np.abs(H)**2
        γ = ifft(P).real
        avg = (np.sum(γ) - γ[0]) / (n - 1)
        γ_target = (1 - α) * γ + α * avg
        γ_target[0] = γ[0]
        A_target = np.sqrt(np.maximum(fft(γ_target), 0))
        H_new = A_target * np.exp(1j * np.angle(H))
        h_new = ifft(H_new).real
        h_new = project_box_sum(h_new, n/2)
        C5 = evaluate_c5(h_new)
        if C5 < best_C5:
            best_h = h_new.copy()
            best_C5 = C5
        # update for next iteration
        h = h_new
        α = min(α + alpha_increase, 0.99)
        if iter > 20 and abs(C5 - best_C5) < 1e-6:
            break
    return best_h, best_C5

def projected_gradient_descent(h, max_iter=300, beta_start=5, eta_init=1.0):
    """Refine h via projected gradient descent on the soft max of C_k."""
    h = h.astype(float)
    best_h = h.copy()
    best_C5 = evaluate_c5(h)
    β = beta_start
    eta = eta_init
    
    for it in range(max_iter):
        # Compute all C_k values
        dx = 2.0 / len(h)
        corr = correlate(h, 1-h, mode='full')  # length 2n-1
        C_vals = 1 - (2/len(h)) * corr  # corresponds to 1 - 2/n * γ_k for k in range(-n+1, n)
        C_vals = C_vals * dx
        
        # Soft max of C_vals to get weights
        w = np.exp(β * C_vals)
        w /= np.sum(w)
        
        # Compute ∇ = -(2/n) * [conv(w, h) + conv(w[::-1], h)]
        # Using fast FFT-based convolution
        conv_w_h = ifft(fft(w) * fft(h)).real
        conv_w_rev_h = ifft(fft(w[::-1]) * fft(h)).real
        grad = -(2/len(h)) * (conv_w_h + conv_w_rev_h)
        
        # Do line search
        h_trial = project_box_sum(h - eta * grad, len(h)/2)
        C5_trial = evaluate_c5(h_trial)
        if C5_trial < best_C5:
            h = h_trial
            best_C5 = C5_trial
            best_h = h.copy()
        else:
            eta *= 0.5
            
        # Update β to sharpen the approximation
        β *= 1.2
        
        if abs(C5_trial - best_C5) < 1e-7:
            break
    
    return best_h, best_C5
