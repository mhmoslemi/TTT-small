import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Compute the C5 constant for the given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def compute_autocorrelations(x, n):
    """Compute the autocorrelations R_k for shift k."""
    x_extended = np.concatenate([x, x])  # for circular convolution
    R = np.zeros(n)
    for k in range(1, n):
        corr = np.correlate(x_extended[:n], x_extended[k:k+n], mode='valid')
        R[k] = corr[0]
    return R

def generate_binary_candidates(X, v, n, R=30):
    """Generate binary candidates via random hyperplane rounding."""
    candidates = []
    # Use principal component v
    for _ in range(R):
        g = np.random.randn(n)
        z = v @ g
        # Sort indices by z values and set top n/2 to 1
        indices = np.argsort(z)[::-1][:n//2]
        x = np.zeros(n)
        x[indices] = 1
        candidates.append(x)
    # Also use deterministic rounding
    indices = np.argsort(v)[::-1][:n//2]
    x = np.zeros(n)
    x[indices] = 1
    candidates.append(x)
    return candidates

def find_worst_shift(R):
    """Find the shift with the smallest autocorrelation."""
    R = R[1:]  # exclude self-correlation (k=0)
    min_idx = np.argmin(R)
    return min_idx + 1

def update_autocorrelation(R, k, x, i, j):
    """Efficiently update the autocorrelation vector after swapping x[i] and x[j]."""
    # Remove old contributions
    for shift in range(1, len(R)):
        R[shift] -= x[i] * x[(i + shift) % len(x)] + x[(i - shift) % len(x)] * x[i]
        R[shift] += x[j] * x[(j + shift) % len(x)] + x[(j - shift) % len(x)] * x[j]
    return R

def greedy_swap_refinement(x, n, max_iterations=5000):
    """Refine binary vector through greedy swaps aiming at increasing minimum autocorrelation."""
    # Precompute R vector
    R = compute_autocorrelations(x, n)
    min_val = np.min(R[1:])  # 0 index is self correlation, ignore it
    
    iterations = 0
    while iterations < max_iterations:
        k_star = find_worst_shift(R)
        if iterations % 100 == 0:
            min_val = np.min(R[1:])
    
        # Compute contribution scores for 1s and 0s
        c_vals = []
        for i in range(n):
            if x[i] == 1:
                score = x[(i + k_star) % n] + x[(i - k_star) % n]
                c_vals.append((score, i))
        c_vals.sort(reverse=True)
        d_vals = []
        for j in range(n):
            if x[j] == 0:
                score = x[(j + k_star) % n] + x[(j - k_star) % n]
                d_vals.append((score, j))
        d_vals.sort()
        
        if not c_vals or not d_vals:
            break
        
        c_max, i = c_vals[0]
        d_min, j = d_vals[0]
        if d_min > c_max:
            # Perform the swap
            x[i] = 0
            x[j] = 1
            # Update R vector
            R = update_autocorrelation(R, k_star, x, i, j)
            iterations += 1
        else:
            break
    
    return x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200  # Use 200 points - moderate resolution
    
    # SDP relaxation step
    elapsed_time = time.time() - start_time
    sdp_budget = budget_s * 0.3
    
    # Prepare the SDP relaxation for max-min binary quadratic problem
    # Variables: X (n x n positive semidefinite), t (scalar)
    # Objective: maximize t
    # Constraints:
    # 1. trace(A_k * X) >= t for all shifts k=1, ..., n-1
    # 2. diags(X) = x, where sum(x) = n/2 => sum(diag(X)) = n/2
    # 3. 0 <= X_{ii} <= 1 for all i
    # 4. X_{ij} <= X_{ii}, X_{ij} <= X_{jj} (optional bounds)
    
    # Initialize X as a matrix variable with n rows and columns
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()
    objective = cp.Maximize(t)
    constraints = [X >> 0]  # X positive semidefinite
    constraints.append(cp.sum(X.diagonal()) == n_points / 2)
    
    # Add diagonal bounds
    for i in range(n_points):
        constraints.append(X[i, i] <= 1)
        constraints.append(X[i, i] >= 0)

    # Add pairwise bounds
    for i in range(n_points):
        for j in range(i+1, n_points):
            constraints.append(X[i, j] <= X[i, i])
            constraints.append(X[i, j] <= X[j, j])
    
    # Compute autocorrelation matrix coefficients for all k values
    A = {}
    for k in range(1, n_points):
        A_k = np.zeros((n_points, n_points))
        for i in range(n_points):
            A_k[i, (i+k)%n_points] = 1
        A[k] = A_k
    
    # Add heuristic bounds
    for k in range(1, n_points):
        # trace(A_k * X) >= t
        constraints.append(cp.trace(A_k @ X) >= t)  # Fixed
    
    # Solve SDP
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=1000, eps=1e-4, time_limit=sdp_budget)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            X_val = X.value
            t_opt = t.value
        else:
            # fallback: use random binary vectors
            return random_search_fallback(n_points, start_time, budget_s)
    except Exception as e:
        print("SDP failed with error:", str(e))
        # fallback: use random binary vectors
        return random_search_fallback(n_points, start_time, budget_s)

    # Round to binary
    v = np.squeeze(np.linalg.eigvals(X_val[:n_points, :n_points]))  # Approximate principal vector
    v = v / np.linalg.norm(v)
    
    # Generate binary candidates using different methods
    candidates = generate_binary_candidates(X_val, v, n_points, R=30)
    
    # Evaluate all candidates
    best_candidate = None
    best_min_corr = -float('inf')
    
    for candidate in candidates:
        R = compute_autocorrelations(candidate, n_points)
        min_corr = np.min(R[1:])  # ignore k=0
        if min_corr > best_min_corr:
            best_min_corr = min_corr
            best_candidate = candidate.copy()
    
    if best_candidate is None:
        return random_search_fallback(n_points, start_time, budget_s)
    
    # Apply greedy swap refinement
    elapsed_time = time.time() - start_time
    swap_budget = budget_s * 0.3
    
    if elapsed_time < swap_budget:
        optimized = greedy_swap_refinement(best_candidate, n_points)
        # Extra evaluation
        final_R = compute_autocorrelations(optimized, n_points)
        final_min_corr = np.min(final_R[1:])
        x_final = optimized
    else:
        x_final = best_candidate
    
    # Optional polishing step
    elapsed_time = time.time() - start_time
    polish_budget = budget_s * 0.1
    
    if elapsed_time < polish_budget:
        h_current = x_final
        # Method: sub-gradient descent on real-valued version of the problem
        c5_current = evaluate_erdos_solution(h_current)
        max_it = 1000
        step_size = 0.01
        
        for i in range(max_it):
            # Find worst shift
            dx = 2.0 / len(h_current)
            h_extended = np.concatenate([h_current, h_current])
            corr = np.zeros(len(h_current))
            for k in range(1, len(h_current)):
                corr[k] = np.correlate(h_extended[:len(h_current)], h_extended[k:k+len(h_current)], mode='valid')[0]
            
            k_star = np.argmin(corr[1:]) + 1  # k=0 is self correlation
            
            # Gradient computation
            g = np.zeros_like(h_current)
            for i in range(len(h_current)):
                g[i] = dx * ((1 - h_current[(i + k_star) % len(h_current)]) - h_current[(i - k_star) % len(h_current)])
            
            # Take step
            h_new = h_current - step_size * g
            # Project back onto [0,1]^n
            h_new = np.clip(h_new, 0, 1)
            # Enforce sum = 1
            current_sum = np.sum(h_new)
            diff = 1 - current_sum
            h_new += diff / len(h_new)
            h_new = np.clip(h_new, 0, 1)
            
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
            else:
                step_size *= 0.95
                if step_size < 1e-6:
                    break
    
    # Final evaluation
    final_h = h_current
    final_c5 = evaluate_erdos_solution(final_h)
    
    # Convert to desired format
    h_values = final_h
    c5_bound = final_c5
    n_points_final = len(h_values)
    
    return (h_values, c5_bound, n_points_final)

def random_search_fallback(n, start_time, budget_s):
    max_iterations = 10000
    best_c5 = float('inf')
    best_h = None
    it = 0
    
    while it < max_iterations and (time.time() - start_time < budget_s):
        h = np.random.rand(n)
        h = h * (n / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        it += 1
    
    # Greedy refinement
    if best_h is not None:
        refined = greedy_swap_refinement(best_h.copy(), n, 1000)
        final_c5 = evaluate_erdos_solution(refined)
        return (refined, final_c5, n)
    
    # Fallback
    h_fallback = np.ones(n) * 0.5
    return (h_fallback, float('inf'), n)
