import numpy as np
import time
from scipy.signal import fftconvolve
from scipy.fftpack import fft, ifft

def project_to_box_plus_sum(h, n):
    """Project h onto {0 ≤ h_i ≤ 1, sum(h) = n/2} using water-filling."""
    # Sort in descending order
    sorted_h = np.sort(h)[::-1]
    # Cumulative sums
    cumsum = np.cumsum(sorted_h)
    # Find largest k such that sum(sorted_h[:k]) <= n/2
    k = 0
    for i in range(len(sorted_h)):
        if cumsum[i] <= n/2:
            k = i + 1
        else:
            break
    # If sum(sorted_h[:k]) < n/2, we need to increase values to total n/2
    # Compute the threshold lambda
    if k == 0:
        # All values should be 0 or 1
        # But we need sum = n/2, so we compute lambda as such
        # Actually, set all values to the same
        val = n/2 / len(h)
        h_proj = np.clip(np.full_like(h, val), 0, 1)
    elif cumsum[k-1] < n/2:
        # The threshold lambda is chosen so that the sum of clipped values is n/2
        S = cumsum[k-1]
        lambda_ = (n/2 - S) / (len(h) - k) if len(h) > k else 0.0
        lambda_ = max(0, lambda_)
        h_proj = np.clip(sorted_h - lambda_, 0, 1)
        # Re-sort to original order
        h_proj = h_proj[np.argsort(np.argsort(-h))]
    else:
        # Technology of sort + cumsum allows us to compute sum(sorted[:k]) = n/2
        # In this case, other values already at boundary
        h_proj = np.clip(sorted_h, 0, 1)
        # Re-sort to original order once more
        h_proj = h_proj[np.argsort(np.argsort(-h))]
    return h_proj

def compute_autocorrelation(h, n):
    """Compute aperiodic autocorrelation via FFT."""
    n_fft = 2**(int(np.ceil(np.log2(2*n))))
    h_padded = np.pad(h, (0, n_fft - n), 'constant')
    h_fft = fft(h_padded)
    corr_fft = h_fft * np.conj(h_fft)
    corr = ifft(corr_fft).real
    # Take first 2*n-1 entries (aperiodic)
    return corr[:2*n-1]

def compute_C5_from_autocorr(gamma, n):
    """Compute C5 = max_s [1 - (2/n)*gamma_s]"""
    C = 1.0 - (2.0 / n) * gamma
    return np.max(C)

def solve_with_admm(initial_h, n_points, max_time=0.55):
    """Solve using ADMM."""
    start_time = time.time()
    # Parameters
    rho = 1.0
    max_admm_iter = 1000
    rel_tol = 1e-7
    bad_count = 0
    t_prev = float('inf')
    
    # Initial values
    z = project_to_box_plus_sum(initial_h, n_points)
    h = z.copy()
    u = np.zeros_like(z)
    t = compute_C5_from_autocorr(compute_autocorrelation(h, n_points), n_points)
    best_t = t
    best_z = z.copy()
    
    # Precompute FFT of ones vector (this will be constant)
    ones = np.ones(n_points)
    n_fft = 2**(int(np.ceil(np.log2(2*n_points))))
    ones_padded = np.pad(ones, (0, n_fft - n_points), 'constant')
    ones_fft = fft(ones_padded)
    
    for i in range(max_admm_iter):
        if time.time() - start_time > max_time:
            break
            
        # h-update: minimize (ρ/2)*||h - (z - u)||^2 s.t. 1 - (2/n)(h∘h)_s <= t
        # First, solve: h = (z - u) - (2/(rho*n)) * (h∘1) * μ, where μ is chosen so that constraint is tight
        # This is equivalent to solving: h = w / (1 + μ * ones_fft)
        # where w = F(z - u) and we want the best μ for the constraint at hand
        # We solve this by bisection on μ to find smallest μ that makes:
        # 1 - (2/n) * (F^{-1} ( w / (1 + μ * ones_fft) ) ∘ h) <= t
        
        w = fft(np.pad(z - u, (0, n_fft - n_points), 'constant'))
        
        # Bisection for μ
        mu_lo = 0.0
        mu_hi = 1e10  # Arbitrary large value
        
        # Additional safeguard to reduce bisection range
        # After computing a few values, we adjust search to best range
        h_test = ifft(w / (1 + mu_lo * ones_fft)).real[:n_points]
        t_lo = compute_C5_from_autocorr(compute_autocorrelation(h_test, n_points), n_points)
        if t_lo > t:
            mu_hi = 10000  # fallback
        else:
            mu_hi = 1000000000
            
        for _ in range(30):  # Bisection steps (more than enough)
            if mu_hi - mu_lo < 1e-12:
                break
                
            mu_trial = (mu_lo + mu_hi) / 2
            h_trial = ifft(w / (1 + mu_trial * ones_fft)).real[:n_points]
            t_trial = compute_C5_from_autocorr(compute_autocorrelation(h_trial, n_points), n_points)
            if t_trial <= t:
                mu_hi = mu_trial
            else:
                mu_lo = mu_trial
                
        mu_final = (mu_lo + mu_hi) / 2
        h = ifft(w / (1 + mu_final * ones_fft)).real[:n_points]
        
        # z-update: project h + u onto box+sum constraints
        z = project_to_box_plus_sum(h + u, n_points)
        
        # Dual update
        u += h - z
        
        # t-update
        gamma = compute_autocorrelation(h, n_points)
        t = compute_C5_from_autocorr(gamma, n_points)
        
        # Bookkeeping
        if t < best_t:
            best_t = t
            best_z = z.copy()
            
        # Check convergence of t
        if abs(t_prev - t) / (abs(t) + 1e-10) < rel_tol:
            bad_count += 1
        else:
            bad_count = 0
        t_prev = t
            
        if bad_count >= 5:
            break

    return best_z, best_t

def nesterov_refinement(h, n_points, max_iter=200):
    """Polish using Nesterov accelerated gradient on smooth surrogate function."""
    alpha = 1e-3  # Step size
    beta = 5.0    # Smoothing parameter
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute the ones vector for FFT-based shift operations
    n_fft = 2**(int(np.ceil(np.log2(2*n_points))))
    ones = np.ones(n_points)
    ones_padded = np.pad(ones, (0, n_fft - n_points), 'constant')
    ones_fft = fft(ones_padded)
    
    h_prev = h.copy()
    h_current = h.copy()
    
    for iter_nest in range(max_iter):
        if iter_nest % 20 == 0:
            beta *= 1.3
        
        # Compute more sophisticated weights based on current h
        gamma = compute_autocorrelation(h_current, n_points)
        C = 1.0 - (2.0/n_points) * gamma
        # Smooth estimate of max
        exp_C = np.exp(beta * (C - np.max(C)))
        weights = exp_C / np.sum(exp_C)
        
        # Compute gradient
        # ∇F_β = -(2/n) Σ w_s·(shift_{+s}(h) + shift_{-s}(h))
        # We use FFT for this computation
        h_fft = fft(np.pad(h_current, (0, n_fft - n_points), 'constant'))
        grad_fft = np.zeros_like(h_fft)
        for s in range(len(C)):
            w_s = weights[s]
            shift_s =  np.eye(len(h_current), k=s)
            shift_neg = np.eye(len(h_current), k=-s)
            h_shifted = shift_s @ h_current
            h_neg_shifted = shift_neg @ h_current
            result = h_shifted + h_neg_shifted
            grad_fft += w_s * fft(result)
        grad = ifft(grad_fft).real[:n_points]
        grad = -(2.0 / n_points) * grad
        
        # Nesterov step
        k = iter_nest + 1
        y = h_current + ((k - 1) / (k + 2)) * (h_current - h_prev)
        h_trial = y - alpha * grad
        h_trial = project_to_box_plus_sum(h_trial, n_points)
        
        # Evaluate
        gamma_trial = compute_autocorrelation(h_trial, n_points)
        C_trial = 1.0 - (2.0 / n_points) * gamma_trial
        c5_trial = np.max(C_trial)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            
        # Update projections
        h_prev = h_current.copy()
        h_current = h_trial.copy()

    return best_h, best_c5

def evaluate_c5(h):
    """Compute C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr
