import numpy as np

def project_box_sum(y, total):
    """Project y onto {x : 0 <= x[i] <= 1, sum x = total}"""
    lb, ub = 0.0, 1.0
    y = np.asarray(y)
    n = len(y)
    
    # bisection to find lambda
    def sum_clipped(lam):
        return np.sum(np.clip(y - lam, lb, ub))
    
    low, high = np.min(y) - ub, np.max(y) - lb
    for _ in range(40):
        lamb = (low + high) / 2.0
        s = sum_clipped(lamb)
        if s > total:
            low = lamb
        else:
            high = lamb
    lam = (low + high) / 2.0
    
    x = np.clip(y - lam, lb, ub)
    return x

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def softmax_weights(beta, C):
    """Compute softmax weights from C values."""
    C = np.asarray(C)
    C_shifted = C - np.max(C)  # for numerical stability
    exp_C = np.exp(beta * C_shifted)
    return exp_C / np.sum(exp_C)

def gradient_from_weights(w, h):
    """Compute gradient from softmax weights."""
    n = len(h)
    # Compute convolutions
    w_full = np.concatenate([w[::-1], np.zeros(n-1)])
    h_full = np.concatenate([h, np.zeros(n-1)])
    conv1 = np.correlate(w_full, h_full, mode='valid')[:n]
    conv2 = np.correlate(w_full[::-1], h_full, mode='valid')[:n]
    return -(2.0 / n) * (conv1 + conv2)
