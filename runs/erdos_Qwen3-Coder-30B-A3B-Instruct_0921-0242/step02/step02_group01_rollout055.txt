import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Compute exact C5 using np.correlate"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def smooth_max_surrogate(h, p=128):
    """Compute smooth max surrogate of the overlap function"""
    dx = 2.0 / len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    corr_vals = corr.real[:len(h)]  # First n values
    overlap = corr_vals * dx
    C_vec = 1 - overlap
    # Smooth max: (sum(C^p))^(1/p)
    # Add small epsilon to avoid numerical issues
    return np.power(np.sum(np.power(np.maximum(C_vec, 1e-12), p)), 1.0 / p)

def gradient_smooth_max(h, p=128):

def estimate_sum_error(h, n):
    """Get the sum error."""
    return np.abs(np.sum(h) - n/2)

def get_seed_set(n):
    """Create diverse initialization strategies."""
    seeds = []
    # Seed 1: uniform
    h1 = np.ones(n) * 0.5
    seeds.append(h1)
    # Seed 2: random Dirichlet sample
    h2 = np.random.dirichlet(np.ones(n)) * (n / 2)
    seeds.append(h2)
    # Seed 3: interpolate initial_h_values if available
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) >= n:
            h3 = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(orig_h)), orig_h)
            h3 = h3 * (n / 2) / np.sum(h3)
            seeds.append(h3)
    # Seed 4: binary with indeterminate switches 
    # Construct on grid with 0 or 1 at most points, 
    #   make just a few fluctuations, or balanced pattern
    h4 = np.random.rand(n)
    h4 = np.clip(h4, 0, 1)
    h4 = project_to_feasible(h4, n)
    seeds.append(h4)
    # Seed 5: smooth with Gaussian blur from uniform
    h5 = np.random.rand(n)
    # Do convolution with some Gaussian kernel to smooth it
    from scipy import signal
    # Gaussian kernel to blur it: with center at 0.5, width 0.1
    # Justified in separation of scales
    kernel = np.exp(-np.linspace(-2.5, 2.5, 31)**2 / (2 * 0.1**2))
    kernel = kernel / np.sum(kernel)
    h5 = signal.convolve(h5, kernel, mode="same")
    h5 = project_to_feasible(h5, n)
    seeds.append(h5)
    return seeds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 350  # num points for refinement
    dx = 2.0 / n_points

    # Generate preliminary seeds:
    seeds = get_seed_set(n_points)
    print(f"Generated {len(seeds)} seeds for optimization.")

    best_h = None
    best_c5 = np.inf

    # Use L-BFGS-B to optimize smooth surrogate through homotopy
    p_vals = [8, 16, 32, 64, 128, 256]  # homotopy parameter trajectory

    # Optimization via smooth approximations  
    for p in p_vals:
        if time.time() - start_time > budget_s * 0.95:
            break
        print(f"Optimizing with p={p}")
        best_at_p = np.inf
        best_h_at_p = None
        for i, seed_h in enumerate(seeds):
            if time.time() - start_time > budget_s * 0.95: 
                break
            h_initial = project_to_feasible(seed_h.copy(), n_points)
            # Optimize with standard scipy minimize (L-BFGS-B)
            # Construct the objective and its gradient for this p  
            def obj(h):
                return smooth_max_surrogate(h, p)
            
            def grad(h):
                # For current iteration with specific p, compute gradient properly
                # Alternative methods?
                return gradient_smooth_max(h, p)
            
            # Bound constraints: [0,1] 
            bounds = [(0,1) for _ in range(n_points)]
            # Penalty approach for sum constraint in scipy.optimize.minimize
            # Since we can define a regularized version, penalty or projection
            # But let's stick to what works: set up equality:
            # Lamdba regularizer + target sum
            def penalty_func(h):
                # Add penalty for deviation from sum constraint using simple distinctive
                penalty_weight = 5000.
                sum_dev = np.sum(h) - n_points / 2
                return penalty_weight * sum_dev ** 2  
            
            def combined_obj(h):
                return obj(h) + penalty_func(h)
                
            def combined_grad(h):
                return grad(h) + 2 * 5000 * (np.sum(h) - n_points / 2)  # Simple penalty gradient
            
            # Repeat 3 times with small maxiter per L-BFGS call, to make good progress quickly with varied restarts.
            for i in range(3):
                if time.time() - start_time > budget_s * 0.95:
                    break
                try:
                    sol = minimize(combined_obj, h_initial, method='L-BFGS-B', jac=combined_grad, bounds=bounds, 
                                   options={'maxiter': 150, 'ftol': 1e-8})
                    h_opt = sol.x
                    h_opt = project_to_feasible(h_opt, n_points)
                    
                    c5_val = evaluate_c5_exact(h_opt)
                    if c5_val < best_at_p:
                        best_at_p = c5_val
                        best_h_at_p = h_opt.copy()
                        
                except Exception as e:
                    print(f"Error in L-BFGS iteration: {e}")
                    continue
                
                if time.time() - start_time > budget_s * 0.95:
                    break
            # Response channel update: however still affordable for this iteration unless time expires
        if best_h_at_p is not None and best_at_p < best_c5:
            best_h = best_h_at_p.copy()
            best_c5 = best_at_p
            
    # Now with final high-smooth solution (p=256 by default), do refined deterministic worst shift correction
    if best_h is not None:
        h_opt = best_h.copy()
        # Worst shift refinement:
        prev_c5 = np.inf
        print("Beginning worst-shift refinement with projected subgradient ascent")
        for t in range(100):  # for reasonable run time and final tuning
            if time.time() - start_time > budget_s * 0.98:
                break
            corr = correlate(h_opt, 1 - h_opt, mode="full")
            max_shift = np.argmax(corr)
            # ignoring shift 0:
            max_shift = max_shift if max_shift != len(h_opt) else max_shift - 1
            
            # Compute gradient for this shift  
            g = np.zeros_like(h_opt)
            for i in range(len(h_opt)):
                # Forward shift: h_{i+k} 
                idx_fwd = (i + max_shift) % len(h_opt)
                # Backward shift: h_{i-k} 
                idx_bwd = (i - max_shift) % len(h_opt) 
                g[i] = dx * ((1 - h_opt[idx_fwd]) - h_opt[idx_bwd])
            
            # Backtracking line search
            alpha = 0.1 * dx
            h_test = h_opt - alpha * g
            h_test = project_to_feasible(h_test, n_points)
            c5_new = evaluate_c5_exact(h_test)
            
            # Accept step if gain improves
            while c5_new >= best_c5 and alpha > 1e-10:
                alpha /= 2.0
                h_test = h_opt - alpha * g
                h_test = project_to_feasible(h_test, n_points)
                c5_new = evaluate_c5_exact(h_test)
                
            if c5_new >= best_c5:  # shouldn't happen, but safety 
                break
            
            h_opt = h_test
            best_c5 = c5_new
            
            if prev_c5 - best_c5 < 1e-6:
                break
            prev_c5 = best_c5
    
    return (h_opt, best_c5, n_points)
