import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Projects h to feasible set: 0 <= h_i <= 1, sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Use bisection to find lambda for the water-filling projection
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Computes the C5 constant."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def autocorr(h):
    """Computes periodic autocorrelation of h using FFT."""
    n = len(h)
    H = fft(h, n)
    ac = np.fft.irfft(H * np.conj(H), n)
    return np.real_if_close(ac)

def evaluate_c5_from_autocorr(autocorr_vals):
    """Computes C5 directly from autocorrelation."""
    n = len(autocorr_vals)
    dx = 2.0 / n
    c = dx * (n/2 - autocorr_vals)
    return np.max(c)

def binary_anneal(n, initial_h, time_limit):
    """Binary simulated annealing with exact autocorrelation updates."""
    dx = 2.0 / n
    T0 = 5 * dx
    alpha = 0.9997
    T = T0
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    
    h = initial_h.copy()
    ac = autocorr(h)
    c = dx * (n/2 - ac)
    current_c5 = np.max(c)
    
    start_time = time.time()
    
    # Precompute indices for efficient swapping
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    step_count = 0
    stagnation_steps = 0
    last_improvement = 0
    
    while time.time() - start_time < time_limit:
        if step_count - last_improvement > 200000:
            # Large jump to escape stagnation
            new_ones = np.random.choice(n, size=n//4, replace=False)
            h = np.zeros(n)
            h[new_ones] = 1
            h = project_to_feasible(h, n)
            ac = autocorr(h)
            c = dx * (n/2 - ac)
            current_c5 = np.max(c)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            last_improvement = step_count
        
        # Select a random 1 and 0 for swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        delta_ac = np.zeros(n)
        
        # Compute delta in autocorrelation exactly
        # Remove the contributions of h_i and h_j to ac
        delta_ac[i] += 2 * h[i]  # contribution of h_i * h_i (self)
        delta_ac[(i - np.arange(n)) % n] -= h[j]  # contribution of h_i * h_{i-k}
        delta_ac[(j - np.arange(n)) % n] -= h[i]  # contribution of h_j * h_{j-k}
        
        # Add back the effect of h'_i and h'_j
        delta_ac[i] += -2 * h[i]
        delta_ac[(i - np.arange(n)) % n] += h[i]  # h'_i * h_{i-k} instead of h_i * h_{i-k}
        delta_ac[(j - np.arange(n)) % n] += h[j]  # h'_j * h_{j-k} instead of h_j * h_{j-k}
        
        # Apply changes
        ac_new = ac + delta_ac
        c_new = dx * (n/2 - ac_new)
        new_c5 = np.max(c_new)
        
        # Accept or reject
        if new_c5 <= current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            h[i] = 0
            h[j] = 1
            ac = ac_new
            c = c_new
            current_c5 = new_c5
            
            # Update indices after swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                last_improvement = step_count
        
        step_count += 1
        if step_count % 1000 == 0:
            T = max(T * alpha, dx * 1e-4)
    
    return best_h, best_c5

def smooth_boundaries(h_bin, ramp_width=2):
    """Convert binary pattern to a smooth step function."""
    n = len(h_bin)
    h_smooth = h_bin.copy()
    
    # Identify boundaries (0 -> 1 or 1 -> 0)
    diff = np.diff(h_bin, prepend=h_bin[-1])
    boundaries = np.where(np.abs(diff) > 0.5)[0]
    
    for b in boundaries:
        # Determine whether this is a 0 -> 1 or 1 -> 0 boundary
        if h_bin[b] == 0 and h_bin[(b-1) % n] == 1:
            # 1 -> 0; smooth from right to left
            for r in range(1, min(ramp_width + 1, b + 1)):
                idx = (b - r) % n
                h_smooth[idx] = (r) / (ramp_width + 1)
        elif h_bin[b] == 1 and h_bin[(b-1) % n] == 0:
            # 0 -> 1; smooth from left to right
            for r in range(1, min(ramp_width + 1, n - b)):
                idx = (b + r) % n
                h_smooth[idx] = (r) / (ramp_width + 1)
    
    return project_to_feasible(h_smooth, n)

def proj_grad_refine(h_init, time_limit):
    """Projected gradient descent on the max shift."""
    h = h_init.copy()
    dx = 2.0 / len(h)
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    step_count = 0
    while time.time() - start_time < time_limit and step_count < 10000:
        # Compute the worst shift
        corr = correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        
        # Compute subgradient
        if k_star > len(h) - 1:
            k_star = k_star - len(h)  # Convert to negative index
        g = -dx * (np.roll(h, k_star) + np.roll(h, -k_star))
        
        # Backtracking line search
        eta = 0.5
        max_iter = 100
        for _ in range(max_iter):
            h_trial = h - eta * g
            h_trial = project_to_feasible(h_trial, len(h))
            trial_c5 = evaluate_c5(h_trial)
            
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = h_trial.copy()
            
            if trial_c5 < evaluate_c5(h):
                break
            else:
                eta /= 2.0
        
        # Simple update with stepsize
        h = h - eta * g
        h = project_to_feasible(h, len(h))
        step_count += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n0 = 256  # Power-of-two for fast FFT
    dx0 = 2.0 / n0
    time_left = budget_s - (time.time() - start_time)
    
    # Stage 1: Binary start
    h_bin = np.zeros(n0)
    ones = np.random.choice(n0, size=n0//2, replace=False)
    h_bin[ones] = 1
    h_bin = project_to_feasible(h_bin, n0)
    
    # Binary annealing (55% of budget)
    time_for_annealing = 0.55 * budget_s
    h_opt_bin, c5_bin = binary_anneal(n0, h_bin, time_for_annealing)
    
    # Stage 2: Boundary smoothing and refinement
    time_for_refinement = 0.25 * budget_s
    h_smooth = smooth_boundaries(h_opt_bin, ramp_width=2)
    h_refined, c5_refined = proj_grad_refine(h_smooth, time_for_refinement)
    
    # Stage 3: Upsampling and final polishing
    time_for_upsampling = 0.15 * budget_s
    n1 = 2 * n0
    h_upsampled = np.interp(np.linspace(0, 2, n1), np.linspace(0, 2, n0), h_refined)
    h_upsampled = project_to_feasible(h_upsampled, n1)
    
    # Do short gradient polish at higher resolution
    for _ in range(500):  # About 500 steps at high resolution
        if time.time() - start_time > budget_s - 0.5:
            break
        corr = correlate(h_upsampled, 1 - h_upsampled, mode='full')
        k_star = np.argmax(corr)
        if k_star > len(h_upsampled) - 1:
            k_star = k_star - len(h_upsampled)
        g = -dx0 * (np.roll(h_upsampled, k_star) + np.roll(h_upsampled, -k_star))
        eta = 0.25
        h_trial = h_upsampled - eta * g
        h_trial = project_to_feasible(h_trial, n1)
        if evaluate_c5(h_trial) < evaluate_c5(h_upsampled):
            h_upsampled = h_trial
    
    return (h_upsampled, evaluate_c5(h_upsampled), n1)
