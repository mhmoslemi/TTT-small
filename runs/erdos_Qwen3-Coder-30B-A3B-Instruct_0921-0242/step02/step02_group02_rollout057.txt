import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cvxpy as cp
from cvxpy import Minimize, Variable, sum as cvxsum, norm
import warnings

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2, h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    if abs(diff) < 1e-10:
        return h
    h = h + diff / n
    h = np.clip(h, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def solve_sdp(n):
    """Solve SDP relaxation to get target autocorrelation."""
    try:
        # Create circulant matrices
        C = []
        for k in range(n):
            Ck = np.zeros((n, n))
            for i in range(n):
                Ck[i, (i - k) % n] = 1
            C.append(Ck)

        # SDP variables
        X = Variable((n, n), symmetric=True)
        h = Variable(n)

        # Objective
        gamma = Variable()

        # Constraints
        constraints = [X == h @ h.T]
        for k in range(n):
            constraints.append(cp.trace(C[k] @ X) >= n/2 - gamma / (2.0 / n))

        constraints.append(X >> 0)  # PSD
        constraints.append(0 <= h)
        constraints.append(h <= 1)
        constraints.append(cp.sum(h) == n/2)

        # Solve
        prob = cp.Problem(Minimize(gamma), constraints)
        prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)

        # Get solution
        h_sol = h.value
        if h_sol is not None:
            return h_sol
        else:
            # Fallback to uniform
            return np.ones(n) * 0.5

    except Exception as e:
        # Fallback
        warnings.warn("SDP failed, using uniform.")
        return np.ones(n) * 0.5

def binary_model_from_sdp(n, target_h):
    """Construct binary pattern from target vector."""
    try:
        # Adjust target to be more binary-like
        # Apply some squashing to sharpen the transitions
        target = np.clip(target_h, 0.01, 0.99)
        scores = np.log(target / (1 - target))
        # Soft thresholding to approximate 0/-1 encoding
        binary_pattern = (scores > 0).astype(float)
        return binary_pattern
    except:
        return np.ones(n) * 0.5

def solve_binary_mip(n, target_autocorr):
    """Solve binary pattern matching target autocorrelation."""
    try:
        # Generate binary integer program
        b = Variable(n, boolean=True)
        
        # Linearized correlations
        s = Variable(n)  # s_k = sum_i b_i b_{i-k}
        gamma = Variable()

        # Constraints
        constraints = [cp.sum(b) == n/2]

        # Correlation linearization via McCormick
        for k in range(n):
            # b_i * b_{i-k}
            z = Variable(n)
            z_constr = [
                z <= b,
                z <= b[np.roll(np.arange(n), k)],
                z >= b + b[np.roll(np.arange(n), k)] - 1,
                z >= 0
            ]
            
            # s_k = sum_i z_{i,k}
            s[k] = cp.sum(z)
            constraints.append((2.0/n) * (n/2 - s[k]) <= gamma)

        # Minimize gamma
        prob = cp.Problem(Minimize(gamma), constraints)
        prob.solve(solver=cp.CBC, verbose=False, timelimit=15)

        # Extract solution
        if b.value is not None:
            return np.round(b.value).astype(float)
        else:
            return np.ones(n) * 0.5
            
    except Exception as e:
        # Fallback to projected SDP
        warnings.warn("Binary MIP failed, using projected SDP.")
        h_sol = solve_sdp(n)
        binary_pattern = (h_sol > 0.5).astype(float)
        return binary_pattern

def smooth_ramp_pattern(binary_pattern, width=3):
    """Convert binary pattern to continuous with ramps."""
    n = len(binary_pattern)
    h = np.zeros(n)
    
    # For each bit transition, apply smooth ramp
    for i in range(n):
        # Consider it's a step location (change from 0 to 1 or 1 to 0)
        if i == 0:
            prev = binary_pattern[-1]
        else:
            prev = binary_pattern[i - 1]
        if i == n - 1:
            next_val = binary_pattern[0]
        else:
            next_val = binary_pattern[i + 1]

        # Smooth ramp centered at transition
        if binary_pattern[i] == 0 and next_val == 1:
            # Transition: 0 -> 1
            for j in range(max(0, i - width//2), min(n, i + width//2 + 1)):
                dist = abs(j - i)
                h[j] = max(h[j], 1.0 - 2.0 * dist / width)
        elif binary_pattern[i] == 1 and next_val == 0:
            # Transition: 1 -> 0
            for j in range(max(0, i - width//2), min(n, i + width//2 + 1)):
                dist = abs(j - i)
                h[j] = max(h[j], 2.0 * dist / width - 1.0)
        else:
            # No transition
            h[i] = binary_pattern[i]
            
    h = np.clip(h, 0, 1)
    return h

def compute_grad(h, k):
    """Compute gradient for shift k."""
    dx = 2.0 / len(h)
    g = np.zeros_like(h)
    for i in range(len(h)):
        # h_{i + k}
        left_idx = (i + k) % len(h)
        g[i] -= dx * h[left_idx]
        # h_{i - k}
        right_idx = (i - k) % len(h)
        g[i] += dx * h[right_idx]
    return g

def lse_grad(h, beta=10.0):
    """Compute gradient of log-sum-exp approximation."""
    n = len(h)
    dx = 2.0 / n
    
    # Evaluate all c_k
    corr = correlate(h, 1 - h, mode='full')
    cks = corr * dx
    
    # softmax weights
    exp_cks = np.exp(beta * cks)
    w = exp_cks / np.sum(exp_cks)
    
    # weighted average of the individual gradient vectors
    g = np.zeros_like(h)
    for k in range(n):
        if k == 0:
            dk = 0
        else:
            dk = k
        grad_k = compute_grad(h, dk)
        g += w[k] * grad_k
        
    return g

def lbfgsb_refine(h0, max_iter=2000, beta_start=5.0):
    """Refine h using projected L-BFGS-B on log-sum-exp surrogate."""
    
    def lse_objective(h):
        dx = 2.0 / len(h)
        corr = correlate(h, 1 - h, mode='full')
        cks = corr * dx
        beta = beta_start
        return (1.0/beta) * np.log(np.sum(np.exp(beta * cks)))
    
    def objective_and_grad(h):
        return lse_objective(h), lse_grad(h)
    
    # Final solver
    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * len(h0)
    options = {'maxiter': max_iter, 'disp': False, 'ftol': 1e-7, 'gtol': 1e-7}
    
    # Project on simplex
    h = project_to_feasible(h0, len(h0))
    
    # Minimize
    result = minimize(objective_and_grad, h, method='L-BFGS-B', bounds=bounds, jac=True, options=options)
    
    final_h = result.x
    final_h = project_to_feasible(final_h, len(final_h))
    
    return final_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    np.seterr(divide='ignore', invalid='ignore')
    start_time = time.time()
    
    # Given constraints for 600s budget
    n = 300
    
    # Step 1: Solve SDP for target autocorrelation
    target_h = solve_sdp(n)
    
    # Step 2: Discretize to binary pattern  
    binary_pattern = binary_model_from_sdp(n, target_h)
    
    # Step 3: Solve binary integer program if possible
    binary_sol = solve_binary_mip(n, target_h)
    
    # Step 4: Smooth with ramps
    h0 = smooth_ramp_pattern(binary_sol, width=4)
    
    # Step 5: Final continuous refinement
    h_final = lbfgsb_refine(h0, beta_start=10.0)
    
    # Final projection to ensure feasibility
    h_final = project_to_feasible(h_final, n)
    
    # Evaluate final value
    c5_value = evaluate_c5(h_final)
    
    return (h_final, c5_value, n)
