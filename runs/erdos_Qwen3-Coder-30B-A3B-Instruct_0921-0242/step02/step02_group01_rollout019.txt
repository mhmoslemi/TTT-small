import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Computes the exact C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Use full correlation mode for exact computation
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def compute_smooth_max(h, p=200):
    """Compute smooth max approximation of overlaps."""
    dx = 2.0 / len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    overlap = corr.real[:len(h)] * dx
    C_vec = 1.0 - overlap
    return np.power(np.sum(np.power(C_vec, p)), 1.0 / p)

def compute_gradient_smooth(h, p=200):
    """Compute gradient of smooth max approximation."""
    dx = 2.0 / len(h)
    n = len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    overlap = corr.real[:len(h)] * dx
    C_vec = 1.0 - overlap
    
    # Smooth gradient
    if np.any(C_vec <= 0):
        # Handle numerical issues
        C_vec = np.maximum(C_vec, 1e-12)
    
    w = np.power(C_vec, p - 1)
    S_p = np.power(np.sum(np.power(C_vec, p)), 1.0 / p)
    
    if S_p == 0:
        return np.zeros(n)
        
    # Gradient w.r.t. each h_i
    grad = np.zeros(n)
    for i in range(n):
        # Compute dC(k)/dh_i for k = 0 to n-1
        # For each k, we have C(k) = dx * sum(h_i * h_{i+k})
        # So dC(k)/dh_i = dx * h_{i+k} or dx * h_{i-k} depending on the shift
        
        # Direct approach using FFTs for each shift
        # Without computing all shifts explicitly, we compute directly:
        # For position i, the derivative of C(k) = dx * sum(h_i * h_{i+k})
        # comes from d/dh_i (h_i * h_{i+k}) = h_{i+k} 
        # and d/dh_i (h_i * h_{i-k}) = h_{i-k}
        term1 = 0.0
        term2 = 0.0
        
        # Compute forward shift: h_{i+k}
        k = 0
        for shift in range(1, n):
            # This constructs h_i * h_{i+k} term-by-term
            # dC(k)/dh_i involves h_{i+k} when we're at i
            j = (i + shift) % n
            term1 += (1 - h[j])
            
        # Compute backward shift: h_{i-k}  
        for shift in range(1, n):
            j = (i - shift) % n
            term2 += h[j]
            
        grad[i] = dx * (term1 - term2)
    
    return (1.0 / np.power(S_p, p - 1)) * np.sum(w * grad)

def compute_gradient_exact(h):
    """Compute exact gradient of the true max function."""
    dx = 2.0 / len(h)
    n = len(h)
    
    # Evaluate all correlations
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    overlap = corr.real[:len(h)] * dx
    C_vec = 1.0 - overlap
    
    # Find maximum
    max_idx = np.argmax(C_vec)
    
    # Compute gradient for that particular shift  
    grad = np.zeros(n)
    for i in range(n):
        # If the overlap at shift k=0 corresponds to C_vec[i] , 
        # we compute for i-th component 
        # dC(k)/dh_i where k = max_idx
        k = max_idx
        
        j_fwd = (i + k) % n
        j_bwd = (i - k) % n
        grad[i] = dx * ( (1 - h[j_fwd]) - h[j_bwd] )
    
    return grad

def backtrack_line_search(h, grad, current_c5, max_iter=20):
    """Simple line search to find a good step size."""
    alpha = 0.1
    dx = 2.0 / len(h)
    for i in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5_exact(h_new)
        if c5_new < current_c5:
            return h_new, c5_new, alpha
        alpha *= 0.5
    
    return h, current_c5, alpha

def generate_seeds(n_points, num_seeds=30):
    """Generate diverse feasible starting points."""
    seeds = []
    # Uniform values
    uniform_h = np.ones(n_points) * 0.5
    seeds.append(project_to_feasible(uniform_h, n_points))
    
    # Random Dirichlet
    for _ in range(min(5, num_seeds // 6)):
        h_dir = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
        seeds.append(project_to_feasible(h_dir, n_points))
        
    # Interpolated initial values if available
    if 'initial_h_values' in globals():
        y_orig = initial_h_values
        x_orig = np.linspace(0, 2, len(y_orig))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_orig, y_orig)
        seeds.append(project_to_feasible(h_interp, n_points))
    
    # Binary patterns
    for _ in range(min(5, num_seeds // 6)):
        h_bin = np.zeros(n_points)
        ones_idx = np.random.choice(n_points, size=n_points//2, replace=False)
        h_bin[ones_idx] = 1.0
        h_bin = project_to_feasible(h_bin, n_points)
        # Optionally smooth it
        if n_points > 100:
            h_bin = np.convolve(h_bin, np.ones(5)/5, mode='same')
        seeds.append(project_to_feasible(h_bin, n_points))
    
    # Random normal distributions
    for _ in range(min(10, num_seeds // 6)):
        h_gaussian = np.random.normal(0.5, 0.1, n_points)
        seeds.append(project_to_feasible(h_gaussian, n_points))
    
    # Fill the rest with random samples
    while len(seeds) < num_seeds:
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        seeds.append(project_to_feasible(h_rand, n_points))
    
    return seeds[:num_seeds]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 350
    max_iter = 100
    p_vals = [8, 16, 32, 64, 128, 256]
    
    best_solution = None
    best_c5 = float('inf')
    
    # Generate diverse initial points
    seeds = generate_seeds(n_points, num_seeds=30)
    
    # Timestamp checks
    elapsed = time.time() - start_time
    
    # Main homotopy loop
    for p in p_vals:
        if elapsed + 5 > budget_s:
            break
            
        for i, h_start in enumerate(seeds):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Use penalty to enforce sum constraint
            def objective(h):
                dx = 2.0 / len(h)
                penalty = 1e4 * (np.sum(h) - n_points / 2)**2
                smooth_val = compute_smooth_max(h, p)
                return smooth_val + penalty
            
            def gradient(h):
                dx = 2.0 / len(h)
                grad_smooth = compute_gradient_smooth(h, p)
                grad_pen = 2 * 1e4 * (np.sum(h) - n_points / 2)
                return grad_smooth + grad_pen
            
            try:
                res = minimize(
                    objective, 
                    h_start, 
                    method='L-BFGS-B', 
                    bounds=[(0, 1)] * n_points,
                    jac=gradient,
                    options={'maxiter': max_iter}
                )
                if not res.success:
                    continue
                    
                h_opt = project_to_feasible(res.x, n_points)
                c5_val = evaluate_c5_exact(h_opt)
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_solution = h_opt.copy()
                    
            except Exception as e:
                continue
            
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.95:
                break
                
        # Update seeds for next iteration with best solution so far
        if best_solution is not None:
            seeds = [best_solution] + [np.random.rand(n_points) * 0.5 
                                         for _ in range(20)]
            seeds = [project_to_feasible(h, n_points) for h in seeds]
    
    # Final deterministic worst-shift refinement
    if best_solution is not None and time.time() - start_time < budget_s * 0.99:
        h_refined = best_solution.copy()
        c5_refined = evaluate_c5_exact(h_refined)
        
        # Refinement loop 
        for i in range(100):
            if time.time() - start_time > budget_s * 0.99:
                break
                
            # Find worst shift
            dx = 2.0 / len(h_refined)
            h_fft = fft(h_refined, 2 * len(h_refined))
            ih_fft = fft(1 - h_refined, 2 * len(h_refined))
            corr = ifft(h_fft * np.conj(ih_fft))
            overlap = corr.real[:len(h_refined)] * dx
            C_vec = 1.0 - overlap
            k_worst = np.argmax(C_vec)
            
            # Compute gradient for this specific shift
            grad = np.zeros_like(h_refined)
            for j in range(len(h_refined)):
                # Process shift: we're differencing terms h_{k} and h_{j-k}
                k_plus = (j + k_worst) % len(h_refined)
                k_minus = (j - k_worst) % len(h_refined)
                grad[j] = dx * ((1 - h_refined[k_plus]) - h_refined[k_minus])
                
            # Line search
            h_new, c5_new, _ = backtrack_line_search(h_refined, grad, c5_refined)
            
            if c5_new < c5_refined:
                h_refined = h_new
                c5_refined = c5_new
            else:
                break
                
        # Final check
        best_solution = h_refined
        best_c5 = c5_refined
    
    # Optional stochastic polishing if close to time limit
    if time.time() - start_time < budget_s * 0.99 and best_solution is not None:
        h_final = best_solution.copy()
        c5_final = evaluate_c5_exact(h_final)
        current_time = time.time()
        time_limit = budget_s * 0.99
        
        while time.time() - current_time < 10 and time.time() < time_limit:
            # Pair swap
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            delta = min(0.01, h_final[i], 1 - h_final[j])
            if delta <= 0:
                continue
                
            h_test = h_final.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, n_points)
            c5_test = evaluate_c5_exact(h_test)
            
            if c5_test < c5_final:
                h_final = h_test
                c5_final = c5_test
                
        best_solution = h_final
        best_c5 = c5_final
    
    # Return final result
    if best_solution is not None:
        return (best_solution, best_c5, n_points)
    else:
        # Provide fallback
        fallback = np.ones(n_points) * 0.5
        fallback = project_to_feasible(fallback, n_points)
        c5_fallback = evaluate_c5_exact(fallback)
        return (fallback, c5_fallback, n_points)
