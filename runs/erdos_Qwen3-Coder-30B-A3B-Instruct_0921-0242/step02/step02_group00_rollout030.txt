import numpy as np
import time

def project_box_sum(y, total):
    """Project y onto {x : 0 <= x[i] <= 1, sum x = total}."""
    y = np.asarray(y)
    n = len(y)
    # Bisection on lambda to find the right value
    # S(lambda) = sum_i clip(y[i] - lambda, 0, 1)
    # Want S(lambda) = total
    # S is decreasing in lambda, so we can bisection
    
    # Compute sorted y
    y_sorted = np.sort(y)[::-1]  # descending
    # Compute cumulative sums
    cumsum = np.cumsum(y_sorted)
    # Find where the cumulative sum exceeds total
    # The projection can be expressed as:
    # x[i] = clip(y[i] - lambda, 0, 1)
    # We need sum(x) = total
    # sum_i clip(y[i] - lambda, 0, 1) = total
    # Let's find lambda such that the projection gives sum = total
    
    # Find the index k such that the sum of the largest k values is >= total
    # and sum of larger k-1 values < total, then lambda = (sum(y[:k]) - total) / k
    
    # Binary search on lambda from 0 to max(y) - min(y) (more precisely)
    # Do a simpler approach:  
    # lambda = (sum(y) - total) / n  starting value is reasonable unless we need to be more careful.
    # We'll do a binary search
    # Actually use the standard projection algorithm:
    # That of finding lambda such that sum_i clip(y_i - lambda) = total with bounds [0,1].
    # Begin:
    # Let s = sum(y) = n*mean(y)
    # If s == total, then y is feasible.
    # We process elements sorted descending.
    
    def sum_clip(y, l):
        return np.sum(np.clip(y - l, 0, 1))
    
    if np.sum(np.clip(y, 0, 1)) < total:
        # if even the full [0,1] clip gives less than total, 
        # we can't satisfy the constraint
        # return y = ones
        z = np.ones_like(y) * (total / n)
        return z
    
    if np.sum(np.clip(y, 0, 1)) == total:
        # Already feasible
        return np.clip(y, 0, 1)
    
    # Binary search
    # lambda in [0, sum(y) - total + 1] approximately 
    # But we should be more precise.
    # Very conservative: 
    # The minimum lambda for sum=total will be when we reduce to 0 first and still get some positive value from remaining
    
    # better way: compute the largest possible lambda that still gives sum>0, and smallest lambda that gives sum=total
    
    # Using bracket
    l_lo = np.min(y) - 1.0 
    l_hi = np.max(y) + 1.0
    
    # We should look at the range more carefully:
    # lambda that makes sum = 0 (i.e., all clipped to 0)  
    # means for those y[i] <= lambda, all become 0.
    # So first find lambda such that sum of boxes with y > lambda is total
    # Let lambda = (sum(y_satisfying) - total) / k
    # This is not quite right either. Conceptual approach:
    
    # For any lambda, sum(x) = sum_i clip(y[i] - lambda, 0, 1) = sum_i max(0, min(1, y[i]-lambda))
    # So sums correspond to piecewise constant functions.
    # Prefer a brouter force but mathematically correct approach using sorting.
    
    # Sort in descending:
    y_sorted = np.sort(y)[::-1] 
    y_cumsum = np.cumsum(y_sorted)
    # Find first position where y_cumsum - lambda*j >= total
    # That lambda = (sum pass - total) / j where j is first index with this property
    
    # Try solving: for largest (say, k) we get y_sorted[:k] such that 
    # sum_i(y_sorted[i] - lambda) >= total where lambda >= y_sorted[k] and all previous <= lambda
    # So sum_i(y_sorted[i] - lambda) >= total
    # k*lambda <= k*mean(y) - total
    # lambda <= mean(y) - total/k
    # But we also have lambda >= y_sorted[k-1] (for k>0) to ensure clipping to <= y_sorted[k-1]
    # But we also have that lambda cannot be higher than y[k] (or any element of the vector) to get 0 from all latter elements.
    # Let’s rather write it numerically. We'll binary search.
    
    # A method: compute the piecewise linear function f(lambda) = sum_i clamp(y[i], lambda, 1)
    # We need to find lambda such that f(lambda) = total
    
    # Model possible values of the total by the way we partition
    def S_lambda(lam):
        # Computes sum(max(0, min(1, y[i] - lam)))  
        return np.sum(np.clip(y - lam, 0, 1))
    
    # Widen search interval for safety
    try_mins = 0.0 if np.min(y) <= 0 else np.min(y)
    try_maxs = 1.0 if np.max(y) >= 1 else np.max(y)
    
    low = try_mins - 0.5
    high = try_maxs + 0.5
    
    # Ensure bracketing
    s_low = S_lambda(low)
    s_high = S_lambda(high)
    if s_low<total and s_high>total:
        pass
    elif s_low > total:
        # if lambda is too small then sum is too big, reverse bracket
        # Swap
        low, high = high, low
        s_low, s_high = s_high, s_low
    
    # We need to find lambda such that S_lambda(lambda) = total
    # Binary search on lambda
    # In practice, limit number of iterations to 40
    eps = 1e-10
    k = 0
    while k < 60 and abs(high - low) > eps:
        mid = (low + high) / 2
        smid = S_lambda(mid)
        if smid >= total:
            low = mid
        else:
            high = mid
        k += 1
        
    if k == 60:
        # fallback - use a simple projection, but just make sure sum matches
        z = y.copy()
        z = np.clip(z, 0, 1)
        z = z * total / np.sum(z)
        return z
    
    lambda_sol = (low + high) / 2
    
    x = np.clip(y - lambda_sol, 0, 1)
    
    return x


def softmax_weights(C, beta):
    # Compute w = softmax(beta*C) where the maximum is numerically stabilized
    # C is a vector of length 2n-1 
    C = np.asarray(C)
    max_C = np.max(C)
    exp_C = np.exp(beta * (C - max_C))
    w = exp_C / np.sum(exp_C) 
    return w

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr
