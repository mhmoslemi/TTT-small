import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Computes the exact C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def smooth_max(h, p=128):
    """Compute smooth max approximation of overlap values"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations using fft
    h_fft = fft(h, 2 * n)
    ih_fft = fft(1 - h, 2 * n)
    corr = ifft(h_fft * np.conj(ih_fft))
    corr_vals = corr.real[:n] * dx
    C_vals = 1 - corr_vals  # C(k) = 1 - corr(k)
    # Smooth approximation to max: (sum(C^p))^(1/p)
    # but we'll work with log-sum-exp for numerical stability
    # log(sum(exp(p * log(C))) / p) to compute smooth max
    C_vals = np.maximum(C_vals, 1e-12)  # Avoid log(0)
    log_C = np.log(C_vals)
    log_sum_exp = np.logaddexp.reduce(p * log_C) / p
    return np.exp(log_sum_exp)

def compute_smooth_gradient(h, p=128):
    """Compute gradient of smooth max surrogate w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation terms for each shift
    h_fft = fft(h, 2 * n)
    ih_fft = fft(1 - h, 2 * n)
    corr = ifft(h_fft * np.conj(ih_fft))
    corr_vals = corr.real[:n] * dx
    C_vals = 1 - corr_vals  # C(k) = 1 - corr(k)
    
    # Compute gradient entries for each shift k
    G = np.zeros((n, n))  # Gradient matrix [n x n]
    for k in range(n):
        # Build shift matrices
        shift_forward = np.roll(h, -k)  # h_{i+k} 
        shift_backward = np.roll(h, k)  # h_{i-k}
        # Gradient for shift k: dx * [ (1 - h_{i+k}) - h_{i-k} ]
        G[:, k] = dx * ((1 - shift_forward) - shift_backward)
    
    # Now compute weighted sum:
    C_vals = np.maximum(C_vals, 1e-12)
    w = C_vals ** (p - 1)
    # Weighted mean gradient
    w = w / np.sum(w)
    grad = np.sum(w[:, None] * G, axis=0)  # This gives ∇S_p
    
    return grad

def compute_exact_gradients(h):
    """Compute exact gradient of max(k)C(k) = max(k) [1 - corr_k]"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations using numpy correlate
    corr = correlate(h, 1 - h, mode='full')
    corr_vals = corr[:n] * dx  # Only first n terms for overlap
    C_vals = 1 - corr_vals  # C(k) = 1 - corr(k)
    max_idx = np.argmax(C_vals)
    # We're taking the gradient of the largest C_k term
    # Direct analytical derivative of that specific C(k)
    # ∂C(k)/∂h_i = dx * [ (1 - h_{i+k})  -  h_{i-k} ]
    # but this is expensive for all k's -> use DFT version
    
    # Use the full DFT approach:
    shift_forward = np.roll(h, -max_idx)  # h_{i+k}
    shift_backward = np.roll(h, max_idx)  # h_{i-k}
    grad = dx * ((1 - shift_forward) - shift_backward)
    
    return grad

def refine_worst_shift(h, max_iter=100):
    """Refine solution with projected sub-gradient ascent on the worst shift."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    c5_old = evaluate_c5_exact(h)
    
    for _ in range(max_iter):
        # Compute all correlations to find worst shift
        corr = correlate(h, 1 - h, mode='full')
        corr_vals = corr[:n] * dx
        idx_max = np.argmax(corr_vals)
        # Now compute the exact gradient for that shift
        shift_forward = np.roll(h, -idx_max)
        shift_backward = np.roll(h, idx_max)
        grad = dx * ((1 - shift_forward) - shift_backward)
        
        # Backtracking line search
        alpha = 0.1 * dx
        for _ in range(20):
            h_try = h - alpha * grad
            h_try = project_to_feasible(h_try, n)
            c5_try = evaluate_c5_exact(h_try)
            if c5_try < c5_old:
                h = h_try
                c5_old = c5_try
                break
            alpha *= 0.5
        else:
            break  # No descent step made
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 350
    dx = 2.0 / n_points
    
    # Check pre-defined initial solution
    if 'initial_h_values' in globals():
        h0 = initial_h_values
        if len(h0) != n_points:
            x_orig = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, n_points)
            h0 = np.interp(x_new, x_orig, h0)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        h0 = project_to_feasible(h0, n_points)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Generate diverse start points
    start_points = [h0]
    # Random Dirichlet
    for _ in range(5):
        h_dir = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
        start_points.append(project_to_feasible(h_dir, n_points))
    # Blurred binary
    for _ in range(5):
        h_bin = np.zeros(n_points)
        indices = np.random.choice(n_points, n_points // 2, replace=False)
        h_bin[indices] = 1.0
        # Smooth a bit with Gaussian
        kernel = np.exp(-0.5 * np.linspace(-2, 2, 21)**2)
        h_bin = np.convolve(h_bin, kernel, mode='same')
        h_bin = h_bin / np.sum(h_bin) * (n_points / 2)
        start_points.append(project_to_feasible(h_bin, n_points))
    
    best_c5 = float('inf')
    best_h = None
    
    # Homotopy loop: p=8, 16, 32, 64, 128, 256
    p_vals = [8, 16, 32, 64, 128, 256]
    for p in p_vals:
        if time.time() - start_time > 0.9 * budget_s:
            break
        
        best_seed_c5 = float('inf')
        best_seed_h = h0
        
        # Process all seeds for this p
        for i, h_seed in enumerate(start_points):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Use penalty for sum constraint
            def objective(h):
                h = np.clip(h, 0, 1)
                # Smooth max surrogate
                s_p = smooth_max(h, p)
                # Penalty for sum constraint
                penalty = 1e4 * (np.sum(h) - n_points / 2) ** 2
                return s_p + penalty
            
            def grad(h):
                h = np.clip(h, 0, 1)
                g = compute_smooth_gradient(h, p)
                # Gradient of penalty
                g += 2 * 1e4 * (np.sum(h) - n_points / 2)
                return g
            
            # L-BFGS-B optimization
            try:
                res = minimize(objective, h_seed, method='L-BFGS-B', 
                              bounds=[(0, 1)] * n_points,
                              jac=grad, options={'maxiter': 200})
                h_opt = res.x
                h_opt = project_to_feasible(h_opt, n_points)
                c5 = evaluate_c5_exact(h_opt)
                if c5 < best_seed_c5:
                    best_seed_c5 = c5
                    best_seed_h = h_opt.copy()
            except Exception as e:
                continue
        
        if best_seed_c5 < best_c5:
            best_c5 = best_seed_c5
            best_h = best_seed_h.copy()
        # Warm start for the next p
        if p != p_vals[-1]:
            h0 = best_seed_h.copy()
    
    # Deterministic worst-shift refinement
    if best_h is not None and (time.time() - start_time) < 0.98 * budget_s:
        best_h = refine_worst_shift(best_h)
        best_c5 = evaluate_c5_exact(best_h)
    
    # Optional stochastic polishing
    if best_h is not None and (time.time() - start_time) < 0.995 * budget_s:
        init_c5 = best_c5
        for _ in range(10000):
            if time.time() - start_time > 0.995 * budget_s:
                break
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            if best_h[i] <= 0 or best_h[j] >= 1:
                continue
            delta = min(0.01, best_h[i], 1 - best_h[j])
            h_test = best_h.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, n_points)
            c5_test = evaluate_c5_exact(h_test)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
    
    if best_h is not None:
        return (best_h, best_c5, n_points)
    else:
        fallback = np.ones(n_points) * 0.5
        fallback = project_to_feasible(fallback, n_points)
        return (fallback, float('inf'), n_points)
