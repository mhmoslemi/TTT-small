import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft
from scipy.spatial.distance import cdist

def project_to_feasible(h, target_sum):
    """Ensure sum(h) = target_sum with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    max_corr = np.max(corr.real) * dx
    return max_corr

def build_lp_data(h0):
    """Build LP data A, b from current iterate h0."""
    n = len(h0)
    # Compute auto-correlation
    h_fft = fft(h0, 2 * n)
    c = ifft(h_fft * np.conj(h_fft)).real
    
    # Create matrix W where each row corresponds to a shift k
    # W[k, i] = h0[(i+k)%n] + h0[(i-k)%n]
    W = np.zeros((n, n))
    for k in range(n):
        W[k, :] = h0[(np.arange(n) + k) % n] + h0[(np.arange(n) - k) % n]
    
    A = -W
    b = -n / 2 + c
    
    return A, b

def solve_lp(A, b, n):
    """Solve the LP min t s.t. A*h + t*1 <= b, 0<=h<=1, sum(h)=n/2, t>=0"""
    # Variables: h[0], ..., h[n-1], t
    c = np.zeros(n + 1)
    c[-1] = 1  # minimize t
    
    # Inequality: A * h <= b  →  A * h - t * 1 <= b
    # Use the last (n) rows of A for this constraint (the linear terms)
    # And 1 row for t: [0 ... 0 1] ...
    # We will form the matrix [A, -1] and vector b as A'; b'
    rows = A.shape[0]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :-1] = 1  # sum h = n/2
    
    bounds = [(0, 1) for _ in range(n)] + [(0, None)]
    
    # Add the constraint that sum(h) = n/2
    # We solve the LP: 
    # minimize t,
    # s.t. A * h + 1 * t <= b, 0 <= h <= 1, sum(h) = n/2, t >= 0
    
    # We don't want to eliminate equality constraint. 
    # So let's make one matrix for all constraints, with A rows for inequality.
    
    # Let's discretize the problem slightly differently:
    # Instead of making bottom row 0s, simplify:
    # Minimize t subject to [A, 1] * [h; t] <= b, with the sum constraint
        
    # Instead, we'll think of:
    # Constraints:
    # 1. A * h + t * 1 <= b -> A * h <= b - t * 1 
    # 2. 0 <= h <= 1  
    # 3. sum(h) = n/2  
    # 4. t >= 0
    
    # The constraint matrix will be:
    # [A, -1]  <=  b
    # [1,  0]  =  n/2
    
    # So we construct:
    A_full = np.zeros((rows + 1, n + 1))
    A_full[:rows, :n] = A
    A_full[:rows, -1] = -1
    A_full[-1, :-1] = 1
    
    b_full = np.zeros(rows + 1)
    b_full[:rows] = b
    b_full[-1] = n / 2
    
    # Solve the LP
    res = linprog(c, A_ub=A_full, b_ub=b_full, bounds=bounds, method='highs')
    
    if not res.success:
        raise ValueError("LP did not converge")
        
    h_new = res.x[:-1]
    t_new = res.x[-1]
    
    return h_new, t_new


def ccp_loop(h0, max_iter=30, tol=1e-8):
    """Perform CCP iterations."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        try:
            A, b = build_lp_data(h)
            h_new, t_new = solve_lp(A, b, len(h))
        except Exception as e:
            # If there's trouble, just break
            break
        
        h_new = project_to_feasible(h_new, len(h) / 2)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        if np.max(np.abs(h_new - h)) < tol:
            break
            
        h = h_new
    
    return best_h, best_c5

def project_to_distribution(h, target_sum):
    """Ensure sum(h) = target_sum with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
        
    # Water-filling approach
    diff = target_sum - current_sum
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use appropriate grid size
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Genreate multiple starting points strategies
    seeds = []
    
    # 1. Uniform distribution
    h_uniform = np.ones(n_points) * 0.5
    h_uniform = project_to_feasible(h_uniform, target_sum)
    seeds.append(("uniform", h_uniform.copy()))
    
    # 2. Random Dirichlet (uniform)
    h_dirichlet = np.random.dirichlet([1.0] * n_points)
    h_dirichlet = h_dirichlet * (target_sum / np.sum(h_dirichlet))
    h_dirichlet = project_to_feasible(h_dirichlet, target_sum)
    seeds.append(("dirichlet", h_dirichlet.copy()))
    
    # 3. Binary-like with randomness
    h_binary = np.zeros(n_points)
    h_binary[:n_points // 2] = 1
    np.random.shuffle(h_binary)
    h_binary = h_binary * (target_sum / np.sum(h_binary))
    h_binary = project_to_feasible(h_binary, target_sum)
    seeds.append(("binary", h_binary.copy()))
    
    # 4. Interpolate initial values if available

                             initial_h_values)
        h_interp = h_interp * (target_sum / np.sum(h_interp))
        h_interp = project_to_feasible(h_interp, target_sum)
        seeds.append(("interpolated", h_interp.copy()))
    
    # 5. Random small perturbation of uniform
    h_rand = np.random.rand(n_points) * 0.1 + 0.45
    h_rand = h_rand * (target_sum / np.sum(h_rand))
    h_rand = project_to_feasible(h_rand, target_sum)
    seeds.append(("random_pert", h_rand.copy()))
    
    best_h = None
    best_c5 = float('inf')
    
    # Set total time for CRS loops
    remaining_time = budget_s * 0.7
    
    # Run CCP from each seed
    for name, h0 in seeds:
        if time.time() - start_time > remaining_time:
            break
            
        try:
            h_final, c5_final = ccp_loop(h0, max_iter=30)
            
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
                
        except Exception as e:
            continue
    
    # Post-process with gradient descent
    if best_h is not None and time.time() - start_time < budget_s * 0.9:
        h_refined = best_h.copy()
        orig_c5 = evaluate_c5(h_refined)
        for step in range(200):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Find worst shift
            corr = ifft(fft(h_refined, 2 * len(h_refined)) * np.conj(fft(1 - h_refined, 2 * len(h_refined))))
            # Ignore shift 0 (self-correlation)
            corr_vals = corr.real[1: len(h_refined)]
            worst_shift = np.argmin(corr_vals) + 1
            
            # Compute subgradient
            grad = np.zeros_like(h_refined)
            for i in range(len(h_refined)):
                left_idx = (i + worst_shift) % len(h_refined)
                right_idx = (i - worst_shift) % len(h_refined)
                grad[i] = dx * ((1 - h_refined[left_idx]) - h_refined[right_idx])
            
            # Update with small step
            alpha = 0.01 / (1 + step / 100)
            h_test = h_refined - alpha * grad
            h_test = project_to_feasible(h_test, target_sum)
            new_c5 = evaluate_c5(h_test)
            
            if new_c5 < orig_c5:
                h_refined = h_test
                orig_c5 = new_c5
    
    # Final polishing with simulated annealing
    if time.time() - start_time < budget_s * 0.99:
        h_final = h_refined.copy()
        c5_final = evaluate_c5(h_final)
        
        temp = 0.1
        decay = 0.99995
        max_steps = 50000
        
        for i in range(max_steps):
            if time.time() - start_time > budget_s * 0.99:
                break
                
            # Pick two indices
            i1 = np.random.randint(0, n_points)
            i2 = np.random.randint(0, n_points)
            
            if h_final[i1] <= 0 or h_final[i2] >= 1:
                continue
                
            delta = np.minimum(0.01, h_final[i1], 1 - h_final[i2])
            
            # Test move
            h_test = h_final.copy()
            h_test[i1] -= delta
            h_test[i2] += delta
            h_test = project_to_feasible(h_test, target_sum)
            c5_test = evaluate_c5(h_test)
            
            if c5_test < c5_final:
                h_final = h_test
                c5_final = c5_test
            else:
                prob = np.exp(-(c5_test - c5_final) / temp)
                if np.random.rand() < prob:
                    h_final = h_test
                    c5_final = c5_test
                    
            temp *= decay
    
    # Final upscale if time allows (if the final is small enough)
    if time.time() - start_time < budget_s and n_points == 400:
        h_up = np.interp(np.linspace(0, 2, 800), np.linspace(0, 2, 400), h_final)
        h_up = h_up * (400 / np.sum(h_up))
        h_up = project_to_feasible(h_up, 400)
        try:
            A_up, b_up = build_lp_data(h_up)
            h_up, t_up = solve_lp(A_up, b_up, 800)
            h_up = project_to_feasible(h_up, 400)
            c5_up = evaluate_c5(h_up)
            h_final = h_up
            c5_final = c5_up
        except:
            pass  # No need to fail
    
    return (h_final, c5_final, n_points)
