from scipy.fft import fft, ifft
from scipy.interpolate import interp1d

def project_to_feasible(h, n):
    """Project h to satisfy sum(h) = n/2 and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-12:
        return h
    # Water-filling approach
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Evaluate the C5 constant for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Use FFT for correlation
    H = fft(h)
    A = ifft(H * np.conj(H)).real
    # Take the full correlation
    fullcorr = np.concatenate([A[-(n-1):], A])
    # Shift the correlation back to start from 0
    corr = fullcorr[:n]
    max_corr = np.max(corr)
    # C5 = 1 - min autocorrelation
    c5 = 1 - (max_corr * dx)
    return c5

def basis_evaluation(n0, M, freqs):
    """Evaluate cosine/sine basis functions on a grid."""
    x = np.linspace(0, 2, n0, endpoint=False)
    basis = np.zeros((n0, 2*M))
    for i in range(M):
        basis[:, 2*i] = np.cos((i+1) * np.pi * x)  # cos
        basis[:, 2*i + 1] = np.sin((i+1) * np.pi * x)  # sin
    return basis

def compute_h_from_coeffs(c, basis, n0):
    """Reconstruct h from coefficients c using basis."""
    M = len(c) // 2
    a0 = 0.5  # Integral constraint
    h_raw = a0 + basis @ c
    h_clipped = np.clip(h_raw, 0, 1)
    scale = 1.0 / np.mean(h_clipped)
    h = h_clipped * scale
    return h

def evaluate_fitness(c, basis, n0):
    """Evaluate fitness given coefficients."""
    h = compute_h_from_coeffs(c, basis, n0)
    return evaluate_c5(h)

def linearise_and_solve_lp(h, k_star, n):
    """Linearize the constraint for worst shift k* and solve LP."""
    # The constraint is: A(k) >= t, where A(k) = h^T P_k h
    # Gradient of A(k) w.r.t h is 2 * roll(h, k)
    grad = 2 * np.roll(h, k_star)
    
    # If k_star is 0, we don't need to roll
    if k_star == 0:
        grad = 2 * h
    else:
        grad = 2 * np.roll(h, k_star)
    
    # Explicitly compute A(k) for k_star - use FFT since it's faster
    H = fft(h)
    A = ifft(H * np.conj(H)).real
    A_val = A[k_star] if k_star < len(A) else A[k_star - len(A)]
    
    # Build constraints matrix for LP
    # constraints: 
    # grad^T (h - h_prev) >= (A_prev(k) - t) 
    # so: grad^T h >= (A_prev(k) - t) + grad^T h_prev
    # which is: -grad^T h <= t - A_prev(k) - grad^T h_prev
    # We want to maximize t, so we want:
    # 0 <= -grad^T h + t - A_prev(k) - grad^T h_prev = t - A_prev(k) - grad^T h_prev - grad^T h
    
    c_LP = np.zeros(n + 1)  # variable t -> coefficient 1
    c_LP[-1] = 1.0  # maximize -t, i.e., minimize t
    
    # matrix of constraint inequalities: A_ub @ [h, t] <= b_ub
    # we express   A_prev(k) - t - grad^T h_prev <= -grad^T h
    # So we get     -grad^T h + t >= A_prev(k) - grad^T h_prev
    # That is:         t - grad^T h >= A_prev(k) - grad^T h_prev
    # As:             A_prev(k) - grad^T h_prev - t + grad^T h <= 0
    # Then:                         grad^T h - t <= A_prev(k) - grad^T h_prev
    # Equals:                       -grad^T h + t >= -A_prev(k) + grad^T h_prev
    
    # Actual Python construction:
    A_ub = np.zeros((1, n + 1))
    A_ub[0, :n] = -grad
    A_ub[0, -1] = 1.0  # t coefficient contributes to constraint
    b_ub = A_val - np.dot(grad, h) 
    
    # Add constraints for bounds
    # 0 <= h <= 1
    bounds_constr = np.zeros((2*n, n + 1))
    for i in range(n):
        bounds_constr[2*i, i] = -1
        bounds_constr[2*i, -1] = 0
        bounds_constr[2*i + 1, i] = 1
        bounds_constr[2*i + 1, -1] = 0
        
    bounds_rh = np.zeros(2*n)
    bounds_rh[:n] = -0
    bounds_rh[n:] = 1

    # Add sum constraint
    total_constr = np.zeros((1, n + 1))
    total_constr[0, :n] = 1
    total_constr[0, -1] = 0  # No t in sum constraint
    total_rh = np.array([n / 2])
    
    # Combine all constraints
    A_ub = np.vstack([A_ub, bounds_constr, total_constr])
    b_ub = np.hstack([b_ub, bounds_rh, total_rh])
    
    # Solve LP using high precision
    try:
        res = linprog(c_LP, A_ub=A_ub, b_ub=b_ub, method='highs')
        if res.success:
            return res.x[:-1], res.x[-1]  # h_new, t_new (the value)
        else:
            # Let's just return the old h
            return h.copy(), A_val
    except Exception:
        return h.copy(), A_val

def subgradient_step(h, n, alpha=0.05):
    """Perform a simple sub-gradient step."""
    # We will compute the worst shift once
    dx = 2.0 / n
    H = fft(h)
    A = ifft(H * np.conj(H)).real
    k_star = np.argmax(A)
    # The constraint on A[k] is min(A) == t where we want to maximize t
    # Gradient of A[-k] w.r.t. h
    # This is best approximated by rolling h in reverse direction
    new_h = np.roll(h, -k_star)
    # during that, we can also split and mass redistribution
    h_new = np.where(h > 0.5, 0.5, h - alpha * (new_h - h))
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new * (n / 2) / np.sum(h_new)
    return h_new
