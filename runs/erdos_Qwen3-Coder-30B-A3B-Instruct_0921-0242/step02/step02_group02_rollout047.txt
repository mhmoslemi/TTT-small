import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def project_to_feasible(h, n):
    """Project h to feasible set: 0 <= h[i] <= 1, sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    if abs(diff) < 1e-10:
        return h
    
    # Use binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        h_trial = np.clip(h + mid, 0, 1)
        if np.sum(h_trial) < n/2:
            ub = mid
        else:
            lb = mid
    h = np.clip(h + lb, 0, 1)
    h = h * (n/2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Evaluate C5 for the given step function h"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    c = corr * dx
    return np.max(c)

def compute_autocorr(h):
    """Compute exact autocorrelation using FFT"""
    n = len(h)
    H = rfft(h, n)
    ac = irfft(H * np.conj(H), n)
    return np.real_if_close(ac)

def binary_anneal(start_h, time_limit):
    """Binary simulated annealing to maximize minimum autocorrelation"""
    start_time = time.time()
    n = len(start_h)
    dx = 2.0 / n

    h = start_h.copy()
    ac = compute_autocorr(h)
    c = dx * (n/2 - ac)
    current_best = np.max(c)
    best_h = h.copy()
    best_c = current_best

    # Annealing parameters
    T0 = 5 * dx
    alpha = 0.9997
    T = T0

    stagnation_count = 0
    last_improvement = 0
    
    while time.time() - start_time < time_limit:
        # Flip a 1 to 0 and a 0 to 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Make the change
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        # Update autocorrelation efficiently
        delta = np.zeros(n)
        # For shift k = (i-j) % n and (j-i) % n
        delta[(i - j) % n] += -1.0
        delta[(j - i) % n] += -1.0
        delta[0] += 2.0  # Self-correlation
        
        # Account for the elements being that can be non-zero
        # in the correlation calculation.
        delta += h[i] - h[j]
        
        # Since this is a local change, only the affected correlations change.
        ac_new = ac + delta
        
        # Compute new c vector
        c_new = dx * (n/2 - ac_new)
        new_max = np.max(c_new)
        
        # Accept or reject
        if new_max < current_best or np.random.rand() < np.exp(-(new_max - current_best)/T):
            h = h_new
            ac = ac_new
            current_best = new_max
            if new_max < best_c:
                best_c = new_max
                best_h = h.copy()
        
        # Temperature update
        T = T * alpha
        
        # Stagnation handling
        if new_max < current_best:
            last_improvement = time.time() - start_time
        elif time.time() - start_time - last_improvement > 20:
            # Reset many bits
            mask = np.random.rand(n) < 0.05
            h_reset = np.where(mask, 1-h, h)
            h = h_reset
            ac = compute_autocorr(h)
            c = dx * (n/2 - ac)
            current_best = np.max(c)
            last_improvement = time.time() - start_time
        
        # Prevent numerical issues
        T = max(T, dx * 1e-4)
    
    return best_h, best_c

def smooth_boundaries(h_bin, ramp_width=2):
    """Insert fractional ramps at boundaries"""
    n = len(h_bin)
    h = h_bin.astype(float)
    
    # Find boundaries
    boundaries = []
    for i in range(n):
        # Check for 0->1 or 1->0 transitions
        left_val = h_bin[i]
        right_val = h_bin[(i + 1) % n]
        if left_val != right_val:
            boundaries.append((i, right_val == 0, right_val == 1))
    
    # Insert ramps
    for b, is_0_to_1, is_1_to_0 in boundaries:
        for r in range(1, ramp_width + 1):
            idx = (b + r) % n
            h[idx] = r / (ramp_width + 1.0)
            
            # Also modify reverse side for 1->0 transitions
            if is_1_to_0:
                idx_rev = (b - r) % n
                h[idx_rev] = (ramp_width - r + 1) / (ramp_width + 1.0)
    
    # Project to final feasible set
    h = project_to_feasible(h, n)
    return h

def proj_grad_refine(h_initial, time_limit):
    """Project gradient descent to minimize max overlap"""
    start_time = time.time()
    n = len(h_initial)
    dx = 2.0 / n
    h = h_initial.copy()
    
    # Initial evaluation
    ac = compute_autocorr(h)
    c = dx * (n/2 - ac)
    current_c5 = np.max(c)
    best_h = h.copy()
    best_c5 = current_c5
    
    step_count = 0
    max_steps = 8000  # Max iterations to prevent runaway
    max_fractional_swap_iter = 3000
    
    while time.time() - start_time < time_limit and step_count < max_steps:
        if step_count % 100 == 0:
            # Save the best during training
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
        # Find worst correlation index
        k_star = np.argmax(c)
        
        # Sub-gradient of C_kstar
        # ∇f = -dx * (h_{i+k*} + h_{i-k*})
        gradient = np.zeros(n)
        # Forward shift
        h_shift_fwd = np.roll(h, k_star)
        # Backward shift
        h_shift_bwd = np.roll(h, -k_star)
        
        gradient = -dx * (h_shift_fwd + h_shift_bwd)
        
        # Backtracking line search for step size
        eta = 0.5
        for _ in range(20):  # max steps in backtracking
            h_test = h - eta * gradient
            h_test = project_to_feasible(h_test, n)
            ac_test = compute_autocorr(h_test)
            c_test = dx * (n/2 - ac_test)
            test_c5 = np.max(c_test)
            
            if test_c5 < current_c5:
                h = h_test
                ac = ac_test
                c = c_test
                current_c5 = test_c5
                break
            else:
                eta /= 2.0

        # Periodic fractional swap on rare iterations
        if step_count % max_fractional_swap_iter == 0:
            i = np.random.randint(n)
            j = np.random.randint(n)
            d = np.random.uniform(0.001, min(h[i], 1 - h[j], 0.01))
            h[i] -= d
            h[j] += d
            h = project_to_feasible(h, n)
            # Recompute
            ac = compute_autocorr(h)
            c = dx * (n/2 - ac)
            current_c5 = np.max(c)
        
        step_count += 1
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Binary optimization 
    n = 256
    dx = 2.0 / n
    
    # Start binary pattern
    h_bin = np.random.rand(n) > 0.5
    h_bin = h_bin.astype(int)
    h_bin = h_bin * (n // 2) // np.sum(h_bin)
    
    # Ensure exact sum
    h_bin_sum = np.sum(h_bin)
    if h_bin_sum < n // 2:
        # Add ones to reach sum = n//2
        to_add = n // 2 - h_bin_sum
        indices = np.random.choice(n, to_add, replace=False)
        for idx in indices:
            h_bin[idx] = 1
    elif h_bin_sum > n // 2:
        # Remove ones to reach sum = n//2
        to_remove = h_bin_sum - n // 2
        indices = np.random.choice(np.where(h_bin == 1)[0], to_remove, replace=False)
        for idx in indices:
            h_bin[idx] = 0
    h_bin = h_bin.astype(float)

    remaining_time = 0.55 * budget_s
    h_bin, c5_bin = binary_anneal(h_bin, remaining_time)
    
    # Stage 2: Boundary smoothing and gradient refinement
    h_smooth = smooth_boundaries(h_bin, ramp_width=2)
    remaining_time = 0.25 * budget_s
    h_refined, c5_refined = proj_grad_refine(h_smooth, remaining_time)
    
    # Stage 3: Upsampling and final polish
    n_up = min(800, 2 * n)
    h_up = np.interp(np.linspace(0, 2, n_up), np.linspace(0, 2, n), h_refined)
    h_up = project_to_feasible(h_up, n_up)
    
    # Final polish with fewer steps for remaining time
    remaining_time = 0.15 * budget_s
    h_final, c5_final = proj_grad_refine(h_up, remaining_time)
    
    # Track the best over all stages
    if c5_final < c5_refined:
        best_h = h_final
        best_c5 = c5_final
        final_n = n_up
    else:
        best_h = h_refined
        best_c5 = c5_refined
        final_n = n
    
    # Final evaluation
    # clamping to prevent numerical error
    c5_final = evaluate_c5(best_h)
    
    return (best_h, c5_final, final_n)
