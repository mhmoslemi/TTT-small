import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def build_cyclic_shift_matrix(k, n):
    """Builds the cyclic shift matrix P_k where (P_k)[i, j] = 1 iff j = (i - k) mod n """
    P = np.zeros((n, n))
    for i in range(n):
        j = (i - k) % n
        P[i, j] = 1
    return P

def project_to_feasible(h, n):
    """Project h to satisfy 0 <= h[i] <= 1 and sum(h) = n/2 """
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach  
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Find the point where we should start clipping
    idx = np.argmax(cumsum >= n/2)
    if cumsum[idx] < n/2:
        # Should add more to satisfy sum
        add_val = (n/2 - cumsum[-1]) / (n - idx)
        h = np.clip(h + add_val, 0, 1)
    else:
        # We should clip something
        lambda_val = (cumsum[idx] - n/2) / (idx + 1)
        h = np.clip(h - lambda_val, 0, 1)
    
    # Fixed point iterations to enforce sum
    for _ in range(20):
        current_sum_new = np.sum(h)
        if abs(current_sum_new - n/2) < 1e-10:
            break
        delta = (n/2 - current_sum_new) / n
        h = np.clip(h + delta, 0, 1)
        
    return h

def evaluate_c5(h):
    """Evaluates C5 = max_k int h(x)(1-h(x+k)) dx """
    n = len(h)
    dx = 2.0 / n
    # Use FFT for better efficiency
    H = rfft(h)
    auto = irfft(np.abs(H)**2, n)
    c = dx * (n/2 - auto)
    return c.max()

def subgradient_refine(h, dx, remaining_time):
    """Refine h using subgradient steps """
    start_time = time.time()
    h = h.copy()
    last_c5 = evaluate_c5(h)
    
    while time.time() - start_time < remaining_time - 1.0:
        # Calculate all correlations
        H = rfft(h)
        auto = irfft(np.abs(H)**2, len(h))
        c = dx * (len(h)/2 - auto)
        k_star = np.argmax(c)
        
        if k_star > len(h)//2:
            k_star = k_star - len(h)
        else:
            k_star = k_star - len(h) + 1

        # Gradient based on shift k*
        grad = np.zeros_like(h)
        # Prevent index out of bounds
        if k_star < 0:
            for i in range(len(h)):
                grad[i] = -h[(i + k_star) % len(h)] + h[(i - k_star) % len(h)]
        else:
            for i in range(len(h)):
                grad[i] = h[(i + k_star) % len(h)] - h[(i - k_star) % len(h)]

        grad = dx * grad
        
        # Line search
        step_size = 1.0
        max_iter = 10
        for _ in range(max_iter):
            h_trial = h - step_size * grad
            h_trial = project_to_feasible(h_trial, len(h))
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < last_c5:
                h = h_trial
                last_c5 = c5_trial
                break
            step_size /= 2.0

        # Early termination check
        if abs(step_size - 1.0) > 0.98:
            break

    return h, last_c5

def upsample_and_project(h_in, nout):
    """Upsample h_in to nout points and re-project to sum to nout/2."""
    n_in = len(h_in)
    dx_in = 2.0 / n_in
    dx_out = 2.0 / nout
    
    # Interpolate
    x_in = np.linspace(0, 2, n_in, endpoint=False)
    x_out = np.linspace(0, 2, nout, endpoint=False)
    
    h_out = np.interp(x_out, x_in, h_in, period=2.0)
    
    # Scale to meet integral constraint
    total_new = np.sum(h_out) * dx_out
    factor = (nout / 2) / total_new
    h_out *= factor
    
    # Clip to [0,1] and reproject
    h_out = np.clip(h_out, 0, 1)
    h_out = project_to_feasible(h_out, nout)
    
    return h_out

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: SDP and extraction
    n = 180
    dx = 2.0 / n
    P = [build_cyclic_shift_matrix(k, n) for k in range(n)]

    X = cp.Variable((n,n), PSD=True)
    t = cp.Variable()

    constraints = [
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n/2,
    ] + [cp.trace(P[k] @ X) >= t for k in range(n)]
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver='SCS', max_iters=25000, eps=1e-5)

    X_opt = X.value
    if X_opt is None:
        raise RuntimeError("SDP solver failed")

    # Spectral factorization
    eigvals, eigvecs = np.linalg.eigh(X_opt)
    lead = eigvecs[:, -1] * np.sqrt(eigvals[-1])
    h0 = project_to_feasible(lead, n)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Try random draws from the solution space
    for _ in range(15):
        g = np.random.randn(n) @ np.linalg.cholesky(X_opt)
        h = project_to_feasible(np.clip(g, 0, 1), n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 2: Subgradient refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 1.0:
        best_h, best_c5 = subgradient_refine(best_h, dx, remaining)

    # Phase 3: Upsample and refine
    if time.time() - start_time < 0.9 * budget_s:
        n_up = 2 * n
        h_up = upsample_and_project(best_h, n_up)
        dx_up = 2.0 / n_up
        remaining = budget_s - (time.time() - start_time)
        best_h, best_c5 = subgradient_refine(h_up, dx_up, remaining)

    return (best_h, best_c5, len(best_h))
