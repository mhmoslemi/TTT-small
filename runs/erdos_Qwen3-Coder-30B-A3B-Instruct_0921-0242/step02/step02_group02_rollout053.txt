def build_cyclic_shift_matrix(k, n):
    """Build the cyclic shift matrix P_k where P_k[i,j] = 1 if j = (i-k) mod n."""
    P = np.zeros((n, n))
    for i in range(n):
        j = (i - k) % n
        P[i, j] = 1
    return P

def project_to_feasible(h, n):

    if abs(diff) < 1e-10:
        return h

    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # FFT-based auto-correlation
    H = rfft(h)
    auto = irfft(np.abs(H)**2, n)
    c = dx * (n/2 - auto)
    return c.max()

        # Compute full correlation

        # Compute subgradient
        grad = np.zeros_like(h_copy)
        # shift by -k_star
        shifted_neg = np.roll(h_copy, k_star)
        shifted_pos = np.roll(h_copy, -k_star)

                c5 = c5_new
                break
            alpha /= 2.0
            
        if alpha < 1e-8 or abs(c5_new - c5) < 1e-6:
            break
            
    return h_copy, c5

def upsample_and_project(h_coarse, n_new):
    """Linearly interpolate and project to new resolution."""

    x_orig = np.linspace(0, 2, n_orig, endpoint=False)

    # Periodic interpolation
    h_interp = np.interp(x_new, x_orig, np.concatenate([h_coarse, h_coarse[:1]]), period=2)
    
    # Re-project to sum constraint
    h_new = h_interp * (n_new / 2) / np.sum(h_interp)
    h_new = project_to_feasible(h_new, n_new)
    
    return h_new
