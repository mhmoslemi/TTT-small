import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Computes the exact C5 constant using full correlation."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1-h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def smooth_max_surrogate(h, p=128):
    """Smooth surrogate of the max operator."""
    dx = 2.0 / len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    corr_vals = corr.real[:len(h)] * dx
    C_vec = 1 - corr_vals
    return np.sum(C_vec ** p) ** (1/p)

def compute_gradient(h, p=128):
    """Compute gradient of smooth max surrogate."""
    dx = 2.0 / len(h)
    n = len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    corr_vals = corr.real[:len(h)] * dx
    C_vec = 1 - corr_vals

    # Prepare some useful shifts using FFT
    h_shifted = np.zeros((n, 2*n))
    for k in range(n):
        h_shifted[k] = np.roll(h, k)
    
    # Compute dC/dh analytically using shifts
    grad = np.zeros(n)
    w = C_vec ** (p - 1)
    S_p = np.sum(C_vec ** p) ** (1/p)
    if S_p == 0:
        return np.zeros(n)
    
    for i in range(n):
        for k in range(n):
            # contribute to the gradient based on shift k
            h_k_forward = h_shifted[k, i]
            h_k_backward = h_shifted[k, (i - k) % n]
            grad[i] += w[k] * (dx * (1 - h_k_forward) - dx * h_k_backward)
        
    grad = grad / (S_p ** (p - 1))
    return grad

def get_seeds(n, num_seeds=30):
    """Generate diverse feasible starting points."""
    seeds = []
    
    # uniform constant
    h_base = np.ones(n) * 0.5
    seeds.append(project_to_feasible(h_base, n))
    
    # random Dirichlet
    for _ in range(num_seeds // 5):
        h = np.random.dirichlet(np.ones(n)) * (n/2)
        h = project_to_feasible(h, n)
        seeds.append(h)
    
    # interpolated from initial_h_values if available
    if 'initial_h_values' in globals():
        orig = initial_h_values
        x_orig = np.linspace(0, 2, len(orig))
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_orig, orig)
        h_interp = h_interp * (n/2) / np.sum(h_interp)
        h_interp = project_to_feasible(h_interp, n)
        seeds.append(h_interp)
    
    # binary pattern with exactly n/2 ones
    for _ in range(num_seeds // 10):
        h = np.zeros(n)
        ones = np.random.permutation(n)[:n//2]
        h[ones] = 1.0
        h = project_to_feasible(h, n)
        seeds.append(h)
    
    # random with sum constraint
    for _ in range(num_seeds - len(seeds)):
        h = np.random.rand(n)
        h = h * (n/2) / np.sum(h)
        h = project_to_feasible(h, n)
        seeds.append(h)
    
    # ensure all are unique instances
    uniq = []
    seen = set()
    for h in seeds:
        key = tuple(np.round(h, decimals=4))
        if key not in seen:
            seen.add(key)
            uniq.append(h.copy())
        if len(uniq) >= num_seeds:
            break
    return uniq[:num_seeds]

def optimize_with_continuation(h_start, n, budget_s, start_time):
    """Optimize using homotopy on p in L-BFGS-B."""
    p_vals = [8, 16, 32, 64, 128, 256]
    best_h = h_start.copy()
    best_c5 = evaluate_c5_exact(best_h)
    h_current = h_start.copy()
    
    # Each L-BFGS step is bounded to avoid long runtimes
    max_iter_per_p = min(200, int(budget_s * 0.4 / len(p_vals)))
    
    for p in p_vals:
        if time.time() - start_time > budget_s * 0.95:
            break
                        
        def obj(h):
            return smooth_max_surrogate(h, p)
        
        def grad(h):
            return compute_gradient(h, p)
        
        # bounds
        bounds = [(0, 1) for _ in range(n)]
        res = minimize(obj, h_current.copy(), method='L-BFGS-B', jac=grad,
                      bounds=bounds, options={'maxiter': max_iter_per_p})
        
        if res.success:
            h_test = project_to_feasible(res.x, n)
            c5_test = evaluate_c5_exact(h_test)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
            h_current = h_test
        else:
            # fallback to current h_current
            pass
            
    return best_h, best_c5

def worst_shift_refinement(h, budget_s, start_time):
    """Refine by descending on the worst shift."""
    dx = 2.0 / len(h)
    max_iter = 100
    h_refined = h.copy()
    
    # Get initial max shift
    corr = correlate(h_refined, 1 - h_refined, mode="full")
    best_shift = np.argmax(corr)
    corr_vals = corr[best_shift] * dx
    best_c5 = corr_vals
    i = 0
    
    while i < max_iter and (time.time() - start_time) < budget_s * 0.9:
        if time.time() - start_time > budget_s * 0.9:
            break
        i += 1
        
        # Compute gradient for the worst shift
        worst_k = best_shift
        grad = np.zeros_like(h_refined)
        
        # Loop over all indices to compute gradient
        for j in range(len(h_refined)):
            left_idx = (j + worst_k) % len(h_refined)
            right_idx = (j - worst_k) % len(h_refined)
            grad[j] = dx * ((1 - h_refined[left_idx]) - h_refined[right_idx])
        
        # Backtracking line search
        alpha = 0.1 * dx
        while True:
            h_new = h_refined - alpha * grad
            h_new = project_to_feasible(h_new, len(h_new))
            
            c5_new = evaluate_c5_exact(h_new)
            if c5_new < best_c5:
                h_refined = h_new
                best_c5 = c5_new
                break
            else:
                alpha *= 0.5
                if alpha < 1e-10:
                    break
        
        # Recompute worst shift for next iteration
        corr = correlate(h_refined, 1 - h_refined, mode="full")
        best_shift = np.argmax(corr)
        best_c5 = corr[best_shift] * dx
    
    return h_refined, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup and seeding
    n_points = 350
    seeds = get_seeds(n_points, 30)
    
    if time.time() - start_time > budget_s * 0.1:
        return (np.ones(n_points) * 0.5, float('inf'), n_points)
    
    best_h = None
    best_c5 = float('inf')
    
    for h_seed in seeds:
        if time.time() - start_time > budget_s * 0.9:
            break
        
        try:
            h_opt, c5_opt = optimize_with_continuation(h_seed, n_points, budget_s, start_time)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
        except Exception as e:
            continue
    
    # Worst shift refinement as final pass
    if best_h is not None and (time.time() - start_time) < budget_s * 0.95:
        try:
            best_h, best_c5 = worst_shift_refinement(best_h, budget_s, start_time)
        except Exception as e:
            pass
    
    # Final check to ensure we still receive a solution
    if best_h is not None:
        return (best_h, best_c5, n_points)
    else:
        return (np.ones(n_points) * 0.5, float('inf'), n_points)
