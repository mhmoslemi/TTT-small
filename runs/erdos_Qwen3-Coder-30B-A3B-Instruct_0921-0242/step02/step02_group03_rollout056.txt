import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def build_trapezoid(t, H, n):
    """Construct trapezoidal h from parameters t (ramp width) and H (height)."""
    dx = 2.0 / n
    h = np.zeros(n)
    L = 1.0 / H - t
    a = (2.0 - 2.0 * t - L) / 2.0
    
    if a < 0 or L < 0:
        return h  # Infeasible, return zeros

    # Left ramp
    left_start = int(a / dx)
    left_end = int((a + t) / dx)
    if left_start < left_end:
        for i in range(left_start, left_end):
            x = (i + 0.5) * dx
            h[i] = H * (x - a) / t

    # Flat top
    top_start = left_end
    top_end = int((a + t + L) / dx)
    if top_start < top_end:
        for i in range(top_start, top_end):
            h[i] = H

    # Right ramp
    right_start = top_end
    right_end = int((a + 2 * t + L) / dx)
    if right_start < right_end:
        for i in range(right_start, right_end):
            x = (i + 0.5) * dx
            h[i] = H * (a + 2 * t + L - x) / t

    return h

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    h_copy = np.clip(h_copy, 0, 1)
    return h_copy

def evaluate_c5_fft(h):
    """Fast evaluation using FFT."""
    n = len(h)
    dx = 2.0 / n
    N = 2 * n
    Hc = fft(h, N)
    Gc = fft(1 - h, N)
    corr = ifft(Hc * np.conj(Gc)).real[:2*n-1]
    return np.max(corr) * dx

def memetic_descent(h, n_points, max_steps=20):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute full correlation to find worst shift
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5_fft(h_trial)
            current_c5 = evaluate_c5_fft(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def random_feasible(n):
    """Generate a random feasible h."""
    h = np.random.rand(n)
    return project_to_feasible(h, n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    n = 800
    dx = 2.0 / n
    best_val = float('inf')
    best_h = None
    
    # ---- Coarse 2D parametric search ----
    H_values = np.linspace(0.5, 1.0, 41)
    for H in H_values:
        t_max = min(1.0 / H, 2.0 - 1.0 / H)
        if t_max <= 0:
            continue
        t_values = np.linspace(0.0, t_max, 81)
        for t in t_values:
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val, best_h = val, h.copy()

    # ---- Refinement around best point ----
    if best_h is not None:
        # Invert the construction to get parameters from best_h
        # Approximate: Linear search around the best curve
        t_max = 1.0 / 0.5 if 0.5 < 1.0 else 2.0 - 1.0 / 0.5
        t_start = max(0, t_max * 0.7)
        t_end = min(t_max, t_max * 1.3)
        
        H_best = 0.75  # estimate
        t_range = np.linspace(t_start, t_end, 41)
        for t in t_range:
            H_max = min(1.0, 2.0 - 2.0 * t) if 2.0 * t < 2.0 else 1.0
            if H_max <= 0:
                continue
            H_vals = np.linspace(0.5, H_max, 41)
            for H in H_vals:
                h = build_trapezoid(t, H, n)
                h = project_to_feasible(h, n)
                val = evaluate_c5_fft(h)
                if val < best_val:
                    best_val, best_h = val, h.copy()

    # ---- Optional local optimiser (Nelder-Mead) ----
    # Simple parameter to point
    def Phi(x):
        t, H = x
        h = build_trapezoid(t, H, n)
        h = project_to_feasible(h, n)
        return evaluate_c5_fft(h)

    try:
        # Use a simple search around best parameters as initial guess
        init_t = 0.3
        init_H = 0.7
        res = minimize(Phi, [init_t, init_H], method='Nelder-Mead', bounds=[(0, 1.5), (0.5, 1.0)])
        if res.success:
            t_opt, H_opt = res.x
            h_opt = build_trapezoid(t_opt, H_opt, n)
            h_opt = project_to_feasible(h_opt, n)
            val_opt = evaluate_c5_fft(h_opt)
            if val_opt < best_val:
                best_val, best_h = val_opt, h_opt.copy()
    except:
        pass

    # ---- Polishing with memetic descent ----
    if best_h is not None and time.time() < time_limit - 5:
        best_h = memetic_descent(best_h, n, max_steps=20)
        best_val = evaluate_c5_fft(best_h)

    # ---- Fallback random memetic runs if time permits ----
    while time.time() < time_limit - 5:
        try:
            h0 = random_feasible(n)
            h_opt = memetic_descent(h0, n, max_steps=30)
            val = evaluate_c5_fft(h_opt)
            if val < best_val:
                best_val, best_h = val, h_opt.copy()
        except:
            pass

    # Ensure final result is correct
    if best_h is None:
        best_h = np.ones(n) * 0.5
        best_val = evaluate_c5_fft(best_h)

    return (best_h, best_val, n)
