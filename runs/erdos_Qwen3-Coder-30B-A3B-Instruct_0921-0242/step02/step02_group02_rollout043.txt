import numpy as np
import time
from scipy.optimize import differential_evolution, linprog
from scipy.signal import correlate
from scipy.interpolate import interp1d

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def build_basis(n0, M):
    """Build cosine/sine basis on grid of size n0."""
    x = np.linspace(0, 2, n0, endpoint=False)
    basis = []
    for m in range(1, M+1):
        basis.append(np.cos(m * np.pi * x))
        basis.append(np.sin(m * np.pi * x))
    return np.array(basis)

def sample_from_coeffs(coeffs, basis, n0):
    """Sample function from Fourier coefficients."""
    a0 = 0.5
    # Build the function
    h_raw = a0 + basis.T @ coeffs
    # Clip to [0,1]
    h_clip = np.clip(h_raw, 0, 1)
    # Renormalize to integrate to 1
    scale = 1 / np.trapz(h_clip, dx=2/n0)
    h = h_clip * scale
    return h

def fitness(c, basis, n0, maxevals):
    """Fitness function for DE."""
    h = sample_from_coeffs(c, basis, n0)
    h = project_to_feasible(h, n0)
    return evaluate_c5(h)

def linearise_and_solve_lp(h, n0, k_active):
    """Solve the LP for SLP refinement."""
    # Build matrices for the LP
    # Objective: maximize t
    c = np.zeros(n0 + 1)  # h[i] + t for i=0..n0-1 + t
    c[-1] = -1  # maximize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # A_ub[i] @ (h, t) >= A_val[i] for i=0..k_active-1
    A_ub = []
    b_ub = []
    
    # Add each active constraint
    # For each k, the constraint is:
    # h^T P_k h >= t
    # where P_k is the shift matrix (left-shift)
    # gradient h^T (P_k + P_k^T) = 2 * roll(h, k)
    # we linearize at h_current
    # A(k) >= A_current(k) + (P_k+P_k^T)*h_current @ (h-h_current) >= t
    # which is: (P_k + P_k^T)*h_current @ h - (P_k+P_k^T)*h_current @ h_current >= t

    # Objective value is h^T A(h)
    # Gradient is 2*roll(h, k) for a shift k
    for k in k_active:
        if k > n0//2:
            k = k - n0
        # We linearize A(k) = h^T P_k h around current h
        # gradient of A(k) is 2 * (roll(h,k) + roll(h,-k))
        # but we need to make this agnostic to indexing
        shift_k = np.roll(h, k)
        shift_neg_k = np.roll(h, -k)
        grad = 2 * (shift_k + shift_neg_k)
        # Now (h^T P_k h) = sum(h * roll(h,k)) = sum(h * shift_k)
        # Linearization at h_current: (sum(h_current * shift_k) + grad @ (h-h_current))
        A_current_k = np.sum(h * shift_k)
        l = grad - A_current_k  # This is actually wrong in original idea, let's fix:
        l_row = np.zeros(n0 + 1)
        l_row[:n0] = grad
        l_row[-1] = -1
        A_ub.append(l_row)
        b_ub.append(-A_current_k)
    
    # Bounds on h: [0,1] for all i, sum h = n0/2.
    bounds = [(0, 1) for _ in range(n0)] + [(None, None)]  # Last variable t is unbounded from below (but we maximize)
    # Or better: t >= A(k) for all k, so t can only be determined by the system
    
    bounds = [(0, 1) for _ in range(n0)] + [(-np.inf, np.inf)]
    
    # variables: h[0]..h[n0-1], t
    A_eq = np.zeros((1, n0 + 1))
    A_eq[0, :n0] = 1
    b_eq = [n0 / 2]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if result.success:
            h_new = np.clip(result.x[:n0], 0, 1)
            t_new = result.x[-1]
            return np.array(h_new), t_new
        else:
            # fallback to current h
            return h.copy(), -np.inf
    except:
        return h.copy(), -np.inf

def possible_shifts(h, n):
    """Return possible worst shifts as indices close to minimum."""
    # Actually the full correlation, so we can find the minimum A(k)
    # Let's go the safer way of evaluating correlation, but keep track of shifts
    corr = correlate(h, 1 - h, mode='full')
    # We need the nonnegative indices (i shift from 0 to n)
    if len(corr) > n:
        # full mode, first n are [0..n-1], last n are [-n+1..0], middle is n (center). 
        # The first n elements are the non-negative shifts: shift=0 appears at index n-1
        # but note that we are looking for k = i - (n-1) for i = n-1,...,2n-2
        # i.e. we match h[i] * h[i-k] to get A(k) = sum(h[i] * h[i-k]) = 
        #  = sum(h[i-k] * h[i])
        # so corr[i] = sum h[j] * h[j - i + n - 1] when we keep i = n-1..2n-2
        
        # Let's extract positive shifts in range 0 to n-1
        # We get full size, we'll just take indices from n-1 to 2n-2
        shift_0_idx = n-1
        # Correlation at shift k=-m will appear at index shift_0_idx + m = n-1+m = (n+n-1) - m
        # for k=-m, we want shift_0_idx + m = (2*n-1) - m = 2*n - 1 - m
        # So for shift k=-m, index is (2*n-1) - m
        # But, shift k=m corresponds to index (2*n-1) - (n-1) + m = n + m - 1 
        # Wait no. Let's just extract the positive ones:
        corrs = corr[shift_0_idx:] * (2.0 / n)  # Adjust by dx
        # Now actual shift k is from 0 to n-1, 
        # but the index where we find A(k) is n-1 + k with n-1 <= (n-1+k) <= 2n-2
        # Therefore, we should take corrs with indices k from 0 to n-1 (adjusted of n-1)
        # and we want to look at the k=0..n-1 corresponding correlations
        # But we want to sort from big to small, so as well sort the indexes first or look at the largest negative_vals in the middle 
        # Let's just extract all positive shifts from index n-1 to 2n-2 (i.e. shifts 0 to n-1)
        corr_positive = corr[n-1:]
        # Get actual shifts and find the worst two
        # We want k such that corr[i] = A(k) where i-n+1 = k
        k_list = np.arange(len(corr_positive))
        # take 5 worst (lowest A)
        worst_k = np.argsort(corr_positive)[:5]
        return worst_k
    else:
        # Often this case is not relevant, but just in case we retrocompute it
        # redistribute corr values to represent the same data
        # grouping A(k) = sum h[i] * h[i - k] for i from 0 to n-1: let k vary 
        # here we already computed the correlation as a function of offset
        # let's just process [-n+1 to n-1]
        # index from 0 to 2n-2
        shift_indices = np.array(range(len(corr))) - (n - 1)  # shifts from -n+1 to n-1
        shift_positive = shift_indices[shift_indices >= 0]
        corr_positive = corr[shift_indices >= 0]
        worst_k = np.argsort(corr_positive)[:5]
        return worst_k

def subgradient_step(h, n0):
    """A basic up-down subgradient step for polishing."""
    k_star = np.argmax(correlate(h, 1 - h, mode="full")) - (n0 - 1)
    if k_star < 0:
        k_star += n0
    
    # To move mass from points with high h to points with low h at shift k*
    # Move small increments
    dx = 2.0 / n0
    # Build the shift matrix M such that the i-th column is M[:,i] = h roll(i)
    # For each index, we shift h by k_star to compute its effect on the correlation
    # At shift k, the part of correlation h[i] * (1 - h[i-k]) contributes to A(k)
    # The gradient of A(k) w.r.t. h[i] is 2*(h[i-k]) (since both h and (1-h) are involved)
    # actually the full gradient at shift k of A(k) wrt h is 2 * roll(h, k) + 2 * roll(h, -k) = 2 * (shift_k + shift_neg_k)
    # But actually, starting the shift from i to i-k:
    # A(k) = sum(h[i] * (1 - h[i-k])) = sum(h[i]) - sum(h[i] * h[i-k])
    # So the gradient dA/dh[i] = -h[i-k] + 1
    # The torque gradient is dA/dh[i] = 1 if i-k < 0 else 0 (because of the structure)
    # Let's simulate the gradient from the definition: 
    # Is increment at h[i] by epsilon?
    #  -> that affects correlation by iq_shifted = h[i] * (1 - h[i-k])  
    #  -> when we change that:
    #  -> dA/dh[i] = 1 - h[i-k] (from sum(1 - h[i-k] * h[i]))
    # Actually interpret:
    #  A(k) = sum_{i=0}^{n-1} h[i] * (1 - h[i-k])
    #  dA/dh[i] = (1 - h[i-k]) . 
    #    If h[i] is being made small (non-increasing), then we compensate some jump 
    #    from the later h values that have large 素
    # Thus we take the combination of the exact projection idea used before
    
    # In fact, a much simpler version:
    # find the value k such that on A(k) is minimal.
    # i.e., the pair of values h[i] and h[i-k] that are particularly responsible for that low value.
    # we now want to swap some values to push up the minimum part. 

    # Let's try a more significant idea:
    # Get the actual k_star where the corruption is minimal
    # A(k) = sum where i from 0 to n-1 of h[i] * (1 - h[i-k])
    # So the Jacobian of each summand is [h[i-k], 1].
    # So J(k) = [1 - h[i-k]] for all i (the h[i-k] terms contribute to negation reaction)
    active_shifts = possible_shifts(h, n0)
    # Do slow step simplicis
    # ensure sum equals n0/2, keep individual elements in [0,1]
    remaining_i = np.argsort(h, kind="stable")
    h_out = h.copy()
    # Let's make a change in the mixing that incurs small change
    
    # Try the simplest gradient step idea on the max nonpositive correlation
    # Consider only indices where you move by k from a sml
    # So the sign of 1−h[i−k] is important
    # Now we know A(k) is minimized at some k
    shift_to_win = k_star
    win_indices = []  # w/i = i such that i-shift-to-win is valid
    lose_indices = []
    win_values = []
    lose_values = []
    # Using the approximation that if corr is worse at some k, that is due to some pairs h[i], h[i-k]
    # Learn about dominant values that we can move 
    for i in range(n0):
        if i >= shift_to_win and (i - shift_to_win) >= 0:
            # moving from position (i - shift_to_win) to i increases A(k) by truncating 
            # because h[i-shift_to_win] contributes as [1 - h[i]] to value of the correlation.
            losing_idx = i - shift_to_win
            if h[losing_idx] > 0.01:  # avoid nearby zero
                lose_indices.append(losing_idx)
                lose_values.append(h[losing_idx])
            if h[i] < 0.99:
                win_indices.append(i)
                win_values.append(h[i])
    
    # Move small amounts from entries with larger h values to those with smaller h values
    delta = 0.005  # batch change
    
    # To solve a balancing problem:
    # take a few of them and accumulate / subtract
    # This is a pretty stiff way of changing things, applied to the worst shift
    
    # Let's do a hybrid approach: 
    # Try swapping small amounts from i which has high vals to i-shift_to_win which has low values to improve the tightness.
    
    diff = float(np.sum(h[lose_indices]) - np.sum(h[win_indices]))
    
    if diff > 0:
        # Clearly something is wrong with arbitrary assignment
        # Let's try different posturing
        # Alternate simplification based on less unavailable we need to filter
        # Mark those indices we can work with
        def indices_for_heap(func):
            # return indices from the domain of func that satisfy some Property
            return [i for i in range(n0) if func(h[i])]
        indices_to_modify = [i for i in range(n0) if 0.01 < h[i] < 0.99]
        # Shuffle indices_to_modify slightly
        np.random.shuffle(indices_to_modify)
        
        # Now, implement pseudo subgradient using the natural partition: i and i-shift_to_win  
        transfer = delta
        # While still more capacity 
        for i in indices_to_modify:
            if transfer <= 0.0001:
                break
            if i >= shift_to_win:  # i and i-shift_to_win exist
                # Pass amount from i to i-shift_to_win if we aim to increase A(k)
                # by increasing the influence on h[i-shift_to_win] * (1-h[i])
                # possible increase if h[i] > h[i-shift_to_win] so we should enhance that term.
                if h[i] > h[i - shift_to_win]:
                    # take that bit
                    should_take = min(delta, h[i])
                    h_out[i] -= should_take
                    h_out[i - shift_to_win] += should_take
                    transfer -= should_take
                
                # Also, enhance inverse: 
                if h[i-shift_to_win] > h[i]:
                    # transfer from i-shift_to_win to i
                    # prompt some swapping of for "upper until" data correlation 
                    should_take = min(delta, h[i-shift_to_win])
                    h_out[i-shift_to_win] -= should_take
                    h_out[i] += should_take
                    transfer -= should_take
                else:
                    # otherwise, take benefits proportional to the        changes caused
                    pass
                    
        # Re-normalize/handle boundary conditions
        h_out = project_to_feasible(h_out, n0)
        return h_out
    
    # If transfer fails, return original (fallback)
    return h.copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Fixed parameters
    n0 = 250  # low-resolution samples
    M = 12    # truncation order for basis
    n_points = 500  # final resolution
    
    # Precompute basis functions
    basis = build_basis(n0, M)
    
    # Define bounds for coefficients: symmetric around 0, auto HedgeClip.
    bounds = [(-0.5, 0.5) for _ in range(2*M)]
    
    # DE parameters
    popsize = 15 * M * 2  # e.g., 360
    maxiter = min(200, int(3000 / popsize))  # limit it to something that fits budget
    
    # Run DE globally 
    de_start = time.time()
    result = differential_evolution(
        fitness,
        bounds,
        args=(basis, n0, maxiter),
        maxiter=maxiter,
        popsize=popsize,
        seed=seed,
        disp=False
    )
    
    # Find top 5 distinct candidates
    candidates = [result.x]
    candidates_h = [sample_from_coeffs(result.x, basis, n0)]
    candidates_h = [project_to_feasible(h, n0) for h in candidates_h]
    candidates_c5 = [evaluate_c5(h) for h in candidates_h]
    
    # Add more candidates
    num_new = 4
    for _ in range(num_new):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Try a random different coefficient vector and check if it's different enough
        new_c = np.random.uniform(-0.5, 0.5, size=2*M)
        if np.allclose(new_c, result.x, atol=1e-3):
            continue
            
        new_h = sample_from_coeffs(new_c, basis, n0)
        new_h = project_to_feasible(new_h, n0)
        new_c5 = evaluate_c5(new_h)
        
        # Add if significantly different
        is_distinct = True
        for c in candidates:
            if np.allclose(new_c, c, atol=1e-3):
                is_distinct = False
                break
        if is_distinct:
            candidates.append(new_c)
            candidates_h.append(new_h)
            candidates_c5.append(new_c5)
    
    # Now refine each
    best_candidate = candidates[0]
    best_h = candidates_h[0]
    best_c5 = candidates_c5[0]
    
    for c, h, c5 in zip(candidates, candidates_h, candidates_c5):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Apply SLP refinement for 100 iterations or until converges
        h_refined = h.copy()
        c5_refined = c5
        for _ in range(200):
            if time.time() - start_time > budget_s - 0.5:
                break
                
            # Find worst active shifts
            k_active = possible_shifts(h_refined, n0)
            if len(k_active) == 0:
                k_active = [0]
            # Do the linearized refine step
            try:
                h_new, t_new = linearise_and_solve_lp(h_refined, n0, k_active)
                # Evaluate actual C5, not E(k) here, but we use A(k) = 1 - C5
                corr_new = correlate(h_new, 1-h_new, mode='full') * (2.0/n0)
                c5_actual = np.max(corr_new)
                if c5_actual < c5_refined:
                    h_refined = h_new.copy()
                    c5_refined = c5_actual
                else:
                    # Use first direction
                    alpha = 0.5  # backtrack
                    h_trial = alpha * h_refined + (1 - alpha) * h_new 
                    h_trial = project_to_feasible(h_trial, n0)
                    c5_trial = evaluate_c5(h_trial)
                    if c5_trial < c5_refined:
                        h_refined = h_trial
                        c5_refined = c5_trial
            except:
                break
        
        # Update best
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined
            best_candidate = c
    
    # Final upsampling
    if time.time() - start_time > budget_s - 0.5:
        return (best_h[:n_points], best_c5, n_points)
    
    # Resample at higher resolution
    x_low = np.linspace(0, 2, n0, endpoint=False)
    x_high = np.linspace(0, 2, n_points, endpoint=False)
    
    # Ensure shape_a0=0.5, use full basis generated earlier
    h_periodic = sample_from_coeffs(best_candidate, basis, n0)
    h_periodic = project_to_feasible(h_periodic, n0)
    
    # Get continuous fit from the coefficients used
    # First, define the continuous function as in the BMC formulas in the later notes
    
    # Simple approach: interpolate using linear periodic
    # This works well enough for smooth shapes from bases
    
    # Step: ensure coarse h is generated right
    # Possibly trust that sampling taught about a fixed point first
    # try interpolation with cubic (`kind=cubic` in interp1d)
    # For now, just use plain unclipped
    interpolated_f = interp1d(x_low, h_periodic, kind='linear', periodic=True)
    h_interp = interpolated_f(x_high)
    
    # Normalize & project to sum = n_points / 2
    h_final = project_to_feasible(h_interp, n_points)
    
    # Final sub-gradient polishing
    max_steps = 2000
    for i in range(max_steps):
        if time.time() - start_time > budget_s - 0.5:
            break
        h_final = subgradient_step(h_final, n_points)
