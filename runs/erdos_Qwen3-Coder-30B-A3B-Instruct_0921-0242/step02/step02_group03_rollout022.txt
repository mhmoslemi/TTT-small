import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def build_trapezoid(t, H, n):
    """Build a trapezoidal function h with parameters t (ramp length) and H (height)."""
    dx = 2.0 / n
    h = np.zeros(n)
    L = 1.0 / H - t
    a = (2.0 - 2.0 * t - L) / 2.0
    
    if L < 0 or a < 0:
        return h
    
    # Indices for ramps and top
    left_ramp_start = int(a / dx)
    left_ramp_end = int((a + t) / dx)
    top_start = left_ramp_end
    top_end = int((a + t + L) / dx)
    right_ramp_start = top_end
    right_ramp_end = int((a + 2 * t + L) / dx)
    
    # Fill ramps
    if t > 0:
        for i in range(left_ramp_start, left_ramp_end):
            x = (i + 0.5) * dx
            h[i] = H * (x - a) / t
            
        for i in range(right_ramp_start, right_ramp_end):
            x = (i + 0.5) * dx
            h[i] = H * (a + 2 * t + L - x) / t
    
    # Fill top
    for i in range(top_start, top_end):
        h[i] = H
    
    return h

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    h_copy = np.clip(h_copy, 0, 1)
    return h_copy

def evaluate_c5_fft(h):
    """Fast FFT-based C5 evaluation."""
    dx = 2.0 / len(h)
    N = 2 * len(h)
    Hc = fft(h, N)
    Gc = fft(1 - h, N)
    corr = np.real(ifft(Hc * np.conj(Gc)))[:2*len(h)-1]
    return np.max(corr) * dx

def memetic_descent(h, n_points, max_steps=20):
    """Refine h using projected gradient descent."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = np.real(ifft(corr_freq))[:n_points]
        corr_vals = corr_vals * dx
        
        # Find worst shift (non-zero shift)
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5_fft(h_trial)
            current_c5 = evaluate_c5_fft(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def random_feasible(n):
    """Generate a random feasible function."""
    h = np.random.rand(n)
    return project_to_feasible(h, n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    # ---- 2-D parametric search ----
    n = 800
    dx = 2.0 / n
    best_val = float('inf')
    best_h = None
    
    # Coarse grid search
    H_vals = np.linspace(0.5, 1.0, 41)
    for H in H_vals:
        t_max = min(1.0/H, 2.0 - 1.0/H)
        if t_max <= 0:
            continue
        t_vals = np.linspace(0.0, t_max, 81)
        for t in t_vals:
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
    
    # ---- refinement around the best pair ----
    # Reconstruct best H and t from best_h
    def reconstruct_params(h):
        # Estimate:
        area = np.sum(h) * dx
        if np.abs(area - 1.0) > 1e-3:
            return 0.0, 0.0  # Invalid
        # Find min/max h values
        h_nonzero = h[h > 1e-6]
        if len(h_nonzero) == 0:
            return 0.0, 0.0
        H = np.max(h_nonzero)
        # Find the region where H occurs 
        flat_indices = np.where(np.abs(h - H) < 1e-3)[0]
        if len(flat_indices) == 0:
            L = 0.0
        else:
            left_idx = flat_indices[0]
            right_idx = flat_indices[-1]  
            L = (right_idx - left_idx + 1) * dx

        L = max(0, L)
        t = (1.0/H - L) if H > 0 else 0.0
        a = (2.0 - 2.0*t - L)/2.0
        if a < 0 or t < 0:
            return 0.0, 0.0
        return t, H

    t_best, H_best = reconstruct_params(best_h)
    # Refinement around best point
    step_t = 0.01
    step_H = 0.025

    refined_t_vals = np.arange(max(0.0, t_best - 2*step_t), min(2.0, t_best + 2*step_t + step_t/2), step_t)
    refined_H_vals = np.arange(max(0.5, H_best - 2*step_H), min(1.0, H_best + 2*step_H + step_H/2), step_H)
    
    for H in refined_H_vals:
        t_max = min(1.0/H, 2.0 - 1.0/H)
        if t_max <= 0:
            continue
        for t in refined_t_vals:
            if t > t_max:
                continue
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
    
    # ---- optional local optimiser (Nelder-Mead) ----
    # This is still a step towards finding better local solution if we are there
    # But the gradient is likely already pretty accurate now
    
    # ---- polishing with memetic descent ----
    best_h = memetic_descent(best_h, n, max_steps=20)
    best_val = evaluate_c5_fft(best_h)
    
    # ---- fallback random memetic runs if time permits ----
    remaining_time = time_limit - time.time()
    if remaining_time > 10.0:
        num_iter = min(10, int(remaining_time / 5))
        for _ in range(num_iter):
            if time.time() >= time_limit:
                break
            h0 = random_feasible(n)
            h_opt = memetic_descent(h0, n, max_steps=30)
            val = evaluate_c5_fft(h_opt)
            if val < best_val:
                best_val, best_h = val, h_opt.copy()
    
    return (best_h, best_val, n)
