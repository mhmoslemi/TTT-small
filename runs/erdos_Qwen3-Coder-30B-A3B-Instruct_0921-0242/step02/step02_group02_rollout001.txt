import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n_points):
    """Ensure that sum(h) = n_points / 2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n_points / 2 - current_sum

    if abs(diff) < 1e-10:
        return h

    # Water-filling approach
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)

    # Binary search for lambda to adjust to target sum
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum < n_points / 2:
            ub = mid
        else:
            lb = mid

    h = np.clip(h + lb, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def binary_anneal(n0, time_limit):
    """Binary simulated annealing for maximizing minimal autocorrelation."""
    np.random.seed(42)
    n = n0
    dx = 2.0 / n
    T0 = 5 * dx
    alpha = 0.9997
    T = T0
    max_swaps = 1000000  # controls number of swaps

    h_bin = np.random.rand(n) > 0.5
    h_bin = h_bin.astype(float)
    h_bin = project_to_feasible(h_bin, n)

    # Compute initial autocorrelation using FFT
    H = np.fft.rfft(h_bin, n)
    ac = np.fft.irfft(H * np.conj(H), n)
    ac = np.real_if_close(ac)

    c = dx * (n / 2 - ac)
    best_c = np.max(c)
    best_h = h_bin.copy()

    swaps = 0
    stagnation = 0
    start_time = time.time()

    while swaps < max_swaps and time.time() - start_time < time_limit:
        # Pick two indices to swap
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin < 0.5)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]

        # Compute change in autocorrelation
        h_shift_i = np.roll(h_bin, i)
        h_shift_j = np.roll(h_bin, j)
        delta = -h_shift_i + h_shift_j
        self_terms = np.zeros_like(ac)
        self_terms[0] += 2
        self_terms[i - j] -= 1
        self_terms[j - i] -= 1

        ac_new = ac + delta + self_terms

        c_new = dx * (n / 2 - ac_new)
        new_best = np.max(c_new)

        if new_best <= best_c or np.random.rand() < np.exp(-(new_best - best_c) / T):
            h_bin[i] = 0
            h_bin[j] = 1
            ac = ac_new
            c = c_new
            if new_best < best_c:
                best_c = new_best
                best_h = h_bin.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        swaps += 1

        # Large jump if stagnating
        if stagnation > 200000:
            # Randomly reassign 5% of ones
            num_to_change = n // 20
            ones_to_change = np.random.choice(np.where(h_bin > 0.5)[0], num_to_change, replace=False)
            zeros_to_change = np.random.choice(np.where(h_bin < 0.5)[0], num_to_change, replace=False)
            h_bin[ones_to_change] = 0
            h_bin[zeros_to_change] = 1
            h_bin = project_to_feasible(h_bin, n)
            H = np.fft.rfft(h_bin, n)
            ac = np.fft.irfft(H * np.conj(H), n)
            ac = np.real_if_close(ac)
            c = dx * (n / 2 - ac)
            best_c = np.max(c)
            best_h = h_bin.copy()
            stagnation = 0

        # Cooling
        if swaps % 1000 == 0:
            T = max(T * alpha, 0.0001)

    return best_h, best_c

def smooth_boundaries(h_bin, ramp_width=2):
    """Inserts linear ramps at 0/1 boundaries."""
    h_cont = h_bin.copy()
    n = len(h_bin)

    # Scan for boundaries from 0 to 1
    for i in range(n):
        if h_bin[i] < 0.5 and h_bin[(i + 1) % n] > 0.5:
            # Insert ramp
            for r in range(1, ramp_width + 1):
                idx = (i + r) % n
                if idx < n:
                    h_cont[idx] = r / (ramp_width + 1)

    # Project back to feasible set
    h_cont = project_to_feasible(h_cont, n)
    return h_cont

def proj_grad_refine(h_cont, time_limit):
    """Projected gradient descent to maximize minimal autocorrelation."""
    n = len(h_cont)
    dx = 2.0 / n
    start_time = time.time()

    best_h = h_cont.copy()
    current_c = evaluate_c5(h_cont)
    best_c = current_c

    def compute_subgradient(h, k_star):
        # Compute subgradient of max_k c_k
        grad = np.zeros_like(h)
        shifted_l = np.roll(h, k_star)
        shifted_r = np.roll(h, -k_star)
        # Note: -dx because we want to minimize
        grad = -dx * (shifted_l + shifted_r)
        return grad

    def backtracking_line_search(h, g, max_iter=10):
        alpha = 0.5
        for _ in range(max_iter):
            h_new = h + alpha * g
            h_new = project_to_feasible(h_new, n)
            c_new = evaluate_c5(h_new)
            if c_new < current_c:
                return h_new, c_new
            alpha *= 0.5
        return h, current_c

    # Main loop
    step = 0
    while time.time() - start_time < time_limit:
        # Find worst shift
        corr = correlate(h_cont, 1 - h_cont, mode="full")
        k_star = np.argmax(corr)
        # Normalize index
        if k_star > len(h_cont):
            k_star = k_star - len(h_cont)
        else:
            k_star = k_star - len(h_cont) + 1

        # Compute subgradient
        grad = compute_subgradient(h_cont, k_star)
        h_new, c_new = backtracking_line_search(h_cont, grad)
        h_cont = h_new
        current_c = c_new

        if c_new < best_c:
            best_c = c_new
            best_h = h_cont.copy()

        step += 1

        # Occasionally move a small amount to escape plateaus
        if step % 100 == 0:
            # Fractional swaps
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            delta = 0.001
            h_trial = h_cont.copy()
            if h_trial[i] > delta and h_trial[j] < 1 - delta:
                h_trial[i] -= delta
                h_trial[j] += delta
                h_trial = project_to_feasible(h_trial, n)
                trial_c = evaluate_c5(h_trial)
                if trial_c < current_c:
                    h_cont = h_trial
                    current_c = trial_c
                    if trial_c < best_c:
                        best_c = trial_c
                        best_h = h_cont.copy()

    return best_h, best_c

def upscale_and_refine(h_cont, n_target, time_limit):
    """Upscale resolution and refine."""
    n = len(h_cont)
    start_time = time.time()
    dx = 2.0 / n

    # Interpolate to higher resolution
    h_upscaled = np.interp(np.linspace(0, 2, n_target), np.linspace(0, 2, n), h_cont)
    h_upscaled = project_to_feasible(h_upscaled, n_target)

    # Short gradient refinement
    def compute_subgradient(h, k_star):
        grad = np.zeros_like(h)
        shifted_l = np.roll(h, k_star)
        shifted_r = np.roll(h, -k_star)
        grad = -dx * (shifted_l + shifted_r)
        return grad

    def backtracking_line_search(h, g, max_iter=5):
        alpha = 0.5
        for _ in range(max_iter):
            h_new = h + alpha * g
            h_new = project_to_feasible(h_new, n_target)
            c_new = evaluate_c5(h_new)
            if c_new < evaluate_c5(h):
                return h_new, c_new
            alpha *= 0.5
        return h, evaluate_c5(h)

    # Gradient refinements
    for _ in range(500):
        if time.time() - start_time > time_limit:
            break

        corr = correlate(h_upscaled, 1 - h_upscaled, mode="full")
        k_star = np.argmax(corr)
        if k_star > len(h_upscaled):
            k_star = k_star - len(h_upscaled)
        else:
            k_star = k_star - len(h_upscaled) + 1

        grad = compute_subgradient(h_upscaled, k_star)
        h_upscaled, c_new = backtracking_line_search(h_upscaled, grad)

        if c_new < evaluate_c5(h_upscaled):
            h_upscaled = h_upscaled
        else:
            break

    return h_upscaled, evaluate_c5(h_upscaled)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Stage 1: Binary optimization (shared for all trials)
    n0 = 256  # must be power of 2 for FFT
    time_limit = 0.55 * budget_s
    h_bin, c5_bin = binary_anneal(n0, time_limit)

    # Stage 2: Smoothing
    h_smooth = smooth_boundaries(h_bin, ramp_width=2)
    time_limit = 0.25 * budget_s
    h_smooth, c5_smooth = proj_grad_refine(h_smooth, time_limit)

    # Stage 3: Upsampling
    n_upscaled = 2 * n0
    time_limit = 0.15 * budget_s
    h_final, c5_final = upscale_and_refine(h_smooth, n_upscaled, time_limit)

    return (h_final, c5_final, n_upscaled)
