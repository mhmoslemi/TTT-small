```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Computes the true C5 constant using np.correlate."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def smooth_max_gradient(h, p=256):
    """Compute gradient of smooth max approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute auto-correlation using FFT
    h_fft = fft(h, 2 * n)
    h2_fft = fft(1 - h, 2 * n)
    corr = ifft(h_fft * np.conj(h2_fft))
    
    # Extract the first n values (proper overlap)
    overlap = corr.real[:n] * dx
    
    # Compute C(k) = 1 - overlap(k)
    C_vec = 1.0 - overlap
    
    # Apply smooth max surrogate
    C_vec_p = C_vec ** p
    
    # Compute sum for normalization
    sum_C_p = np.sum(C_vec_p)
    
    # Avoid divide by zero
    if sum_C_p < 1e-15:
        grad = np.zeros_like(h)
    else:
        # Gradient components
        # w_k = C(k)^{p-1}  (but we can do it more efficiently)
        w_vec = C_vec_p / sum_C_p
        
        # Compute gradient for each shift
        grad = np.zeros_like(h)
        
        # For each shift k, compute partial derivative
        for k in range(n):
            # Forward shift: h_{i+k}
            h_forw = np.roll(h, -k)
            # Backward shift: h_{i-k}
            h_back = np.roll(h, k)
            
            # Contribution to gradient
            temp_grad = dx * (1 - h_forw - h_back)
            # Weighted by smooth max
            weighted_grad = w_vec[k] * temp_grad
            grad += weighted_grad
        
        # Normalize the gradient
        grad = grad / (sum_C_p ** (1 - 1/p))
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 350
    dx = 2.0 / n_points
    lambda_sum = 1e4  # Penalty weight for sum constraint
    
    # Generate diverse seeds
    seeds = []
    
    # Seed 1: uniform half-filled
    h0 = np.ones(n_points) * 0.5
    seeds.append(project_to_feasible(h0, n_points))
    
    # Seed 2: random Dirichlet
    h_dir = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
    seeds.append(project_to_feasible(h_dir, n_points))
    
    # Seed 3: initial_h_values if available
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) > n_points:
            x_orig = np.linspace(0, 2, len(orig_h))
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_orig, orig_h)
        else:
            h_interp = orig_h
        h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
        seeds.append(project_to_feasible(h_interp, n_points))
    
    # Seeds 4-30: random with sum constraint
    for _ in range(27):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        seeds.append(project_to_feasible(h, n_points))
    
    best_h = None
    best_c5 = float('inf')
    
    # Homotopy on smoothness parameter p
    p_vals = [8, 16, 32, 64, 128, 256]
    
    for p in p_vals:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i, seed_h in enumerate(seeds):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Optimization function with penalty
            def opt_func(h_vec):
                h = h_vec
                # Add penalty term for sum constraint
                sum_penalty = lambda_sum * (np.sum(h) - n_points/2) ** 2
                # Compute the smooth max
                dx = 2.0 / n_points
                h_fft = fft(h, 2 * n_points)
                h2_fft = fft(1 - h, 2 * n_points)
                corr = ifft(h_fft * np.conj(h2_fft))
                overlap = corr.real[:n_points] * dx
                C_vec = 1.0 - overlap
                smooth_max = np.sum(C_vec ** p) ** (1/p)
                return smooth_max + sum_penalty
            
            # Gradient function
            def grad_func(h_vec):
                h = h_vec
                grad = smooth_max_gradient(h, p)
                # Add gradient of penalty term
                sum_grad = 2 * lambda_sum * (np.sum(h) - n_points/2)
                grad += sum_grad
                return grad
            
            # Optimization using L-BFGS-B
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(opt_func, seed_h, method='L-BFGS-B', 
                              bounds=bounds, jac=grad_func, 
                              options={'maxiter': 150})
            
            if result.success:
                h_opt = result.x
                c5_opt = evaluate_c5_exact(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
    
    # Worst-shift refinement
    if best_h is not None and time.time() - start_time < budget_s * 0.99:
        h_refined = best_h.copy()
        max_iter = 100
        for _ in range(max_iter):
            if time.time() - start_time > budget_s * 0.99:
                break
                
            # Compute exact overlaps
            corr = correlate(h_refined, 1 - h_refined, mode="full")
            max_shift = np.argmax(corr) - (len(h_refined) - 1)  # Convert to k
            if max_shift == 0:
                # This shouldn't happen but be safe
                max_shift = 1
            
            # Compute gradient for the worst shift
            h_forw = np.roll(h_refined, -max_shift)
            h_back = np.roll(h_refined, max_shift)
            
            grad = dx * (1 - h_forw - h_back)
            
            # Backtracking line search
            step_size = 0.1 * dx
            while step_size > 1e-8:
                h_new = h_refined - step_size * grad
                h_new = project_to_feasible(h_new, n_points)
                
                c5_new = evaluate_c5_exact(h_new)
                c5_old = evaluate_c5_exact(h_refined)
                
                if c5_new < c5_old:
                    h_refined = h_new
                    break
                else:
                    step_size *= 0.5
            else:
                break
    
    # Final polishing - small random exchanges
    if time.time() - start_time < budget_s * 0.995:
        h_final = h_refined.copy()
        c5_final = evaluate_c5_exact(h_final)
        count = 0
        max_count = 5000
        while count < max_count and time.time() - start_time < budget_s * 0.995:
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            if h_final[i] <= 0 or h_final[j] >= 1:
                count += 1
                continue
            delta = np.random.uniform(0, min(h_final[i], 1 - h_final[j], 0.01))
            h_test = h_final.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, n_points)
            c5_test = evaluate_c5_exact(h_test)
            if c5_test < c5_final:
                h_final = h_test
                c5_final = c5_test
            count += 1
    else:
        h_final = h_refined
        c5_final = evaluate_c5_exact(h_refined)
        
    return (h_final, c5_final, n_points)
```
