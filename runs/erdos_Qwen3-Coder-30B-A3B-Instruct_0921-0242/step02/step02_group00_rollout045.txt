def aperiodic_autocorr(h):
    """Compute aperiodic autocorrelation using FFT."""
    n = len(h)
    # pad to avoid wraparound
    h_pad = np.pad(h, (0, n-1), mode='constant')
    # compute correlation via FFT
    corr = fftconvolve(h_pad, h[::-1], mode='valid')
    # return just the positive shifts (0 to n-1)
    return corr[:n]

def softmax(x, beta=1.0):
    """Compute softmax with temperature beta."""
    x_scaled = beta * x
    x_scaled -= np.max(x_scaled)  # for numerical stability
    exp_x = np.exp(x_scaled)
    return exp_x / np.sum(exp_x)

def conv(a, b):
    """Compute aperiodic linear convolution a*b, length >= len(a)."""
    return fftconvolve(a, b, mode='full')[:len(a)]

def compute_C5(h):
    """Compute the overlap constant C5."""
    n = len(h)
    dx = 2.0 / n
    gamma = aperiodic_autocorr(h)
    C = 1 - (2/n) * gamma
    return np.max(C)

def score_guided_swap(h):
    """Perform a deterministic score-guided swap as heuristic escape."""
    n = len(h)
    eps = 1e-5
    c5 = compute_C5(h)
    C = 1 - (2/n) * aperiodic_autocorr(h)
    worst_indices = np.where(abs(C - c5) < eps)[0]
    
    if len(worst_indices) == 0:
        return h
    
    # Compute scores per index
    scores = np.zeros(n)
    for i in range(n):
        score = 0
        for s in worst_indices:
            if 0 <= i + s < n:
                score += h[i + s]
            if 0 <= i - s < n:
                score += h[i - s]
        scores[i] = score
    
    # pick one with high score as source and one with low score as sink
    idx_source = np.argmax(scores)
    idx_target = np.argmin(scores)
    
    if idx_source == idx_target:
        return h
    
    # move a tiny amount
    delta = min(0.01, h[idx_source], 1 - h[idx_target])
    if delta <= 0:
        return h
        
    h_copy = h.copy()
    h_copy[idx_source] -= delta
    h_copy[idx_target] += delta
    h_copy = project_box_sum(h_copy, total=n/2)
    
    return h_copy
