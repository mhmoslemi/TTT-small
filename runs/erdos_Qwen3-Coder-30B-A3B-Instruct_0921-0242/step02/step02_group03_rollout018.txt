import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, maximum as cp_max
import math

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    if diff > 0:
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    h_copy = np.clip(h_copy, 0, 1)
    return h_copy

def build_binary_initial(n_points):
    """Construct a symmetric and unimodal binary pattern."""
    half = n_points // 2
    b_left = np.zeros(half + 1, dtype=int)
    
    if len(initial_h_values) >= n_points:
        h_vals = initial_h_values[:n_points]
    else:
        x_orig = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h_vals = np.interp(x_new, x_orig, initial_h_values)
    
    # Threshold and adjust count to n/2
    h_vals = np.clip(h_vals, 0, 1)
    b_vals = (h_vals > 0.5).astype(int)
    current_count = np.sum(b_vals)
    
    diff = n_points//2 - current_count
    if diff > 0:
        # Need to add more 1's
        indices = np.argsort(h_vals)
        for i in range(n_points - 1, -1, -1):
            if diff <= 0:
                break
            if b_vals[i] == 0:
                b_vals[i] = 1
                diff -= 1
    elif diff < 0:
        # Need to remove some 1's  
        indices = np.argsort(h_vals)[::-1]
        for i in indices:
            if diff >= 0:
                break
            if b_vals[i] == 1:
                b_vals[i] = 0
                diff += 1
    
    # Enforce symmetry and unimodality
    b_left = b_vals[:half+1]
    b_left = np.sort(b_left)[::-1]  # non-increasing   
    
    # Mirror
    b_right = b_left[-2::-1]  # exclude middle
    b = np.concatenate([b_left, b_right])
    
    return b

def compute_full_correlations(b, n_points):
    """Full correlation for all shifts using FFT."""
    # Pad to at least doubled size for linear convolution
    pad_len = 2 * n_points
    b_padded = np.pad(b, (0, pad_len - n_points), mode='constant', constant_values=0)
    one_minus_b_padded = np.pad(1 - b, (0, pad_len - n_points), mode='constant', constant_values=0)
    
    # FFT convolution-like: C = F^{-1}(F(b)*F(1-b))
    corr_freq = fft(b_padded) * np.conj(fft(one_minus_b_padded))
    corr_vals = ifft(corr_freq).real[:n_points]
    
    dx = 2.0 / n_points
    corr_vals = corr_vals * dx
    
    return corr_vals

def solve_binary_milp_cutting_plane(n_points, time_limit):
    """Solve the binary MILP with cutting plane method for shifts."""
    # Build left half variables
    half = n_points // 2
    b_left = Variable(half + 1, boolean=True)
    
    # Mirror variables to full
    b = cp.hstack([b_left, b_left[-2::-1]])  # symmetric: [0..n/2] + [n/2-1..0]
    
    # Area constraint
    area_constr = cp_sum(b) == n_points / 2
    
    # Define auxiliary variables for each shift (we'll identify in cutting plane loop)
    shift_constraints = []
    y_vars = {}
    
    # Initial shift set
    S0 = [0, 1, 2, n_points // 2]
    
    # Movement limit
    max_iters = 30
    shift_set = list(S0)
    best_b = None
    min_ws = float('inf')
    
    # Constraints
    base_constraints = [area_constr]
    
    elapsed = 0
    
    # Use the best binary solution so far as warm-start
    # We'll construct it from the initial_h_values
    binary_template = build_binary_initial(n_points)
    
    # Start time for solver loop
    start_solver = time.time()
    remaining_solver_time = time_limit - 1.0  # Reserve last second for post-processing
    solver_seed = 42
    rand = np.random.RandomState(solver_seed)
    
    for i_iter in range(max_iters):
        if elapsed >= time_limit - 3.0:
            break
        
        shift_set = list(set(shift_set))  # dedup
        if len(shift_set) == 0:
            break
        
        # Constraints for current set of shifts
        constraints = base_constraints.copy()
        t = Variable()
        
        # Build y matrix just for the current shift_set
        y_matrix = {}
        for k in shift_set:
            y_matrix[k] = Variable(n_points - k, boolean=True)
            # Might need to re-sort all of these y vars
        
        # Constraint: for shift k, sum y ik ≥ n/2 - t/dx
        dx = 2.0 / n_points
        for k in shift_set:
            y_list = y_matrix[k]
            constr = cp_sum(y_list) >= n_points / 2 - t / dx
            constraints.append(constr)
            
            # Linearization constraints for y[i,k] = b[i] * b[i+k]
            for i in range(n_points - k):
                # y[i,k] ≤ b[i]  
                # y[i,k] ≤ b[i+k]
                # y[i,k] ≥ b[i] + b[i+k] - 1
                # y[i,k] ≥ 0
                constraints.append(y_list[i] <= b[i])
                constraints.append(y_list[i] <= b[i+k])
                constraints.append(y_list[i] >= b[i] + b[i+k] - 1)
                constraints.append(y_list[i] >= 0)
                
        # Epigraph constraints
        constraints.append(t >= 0)
        
        # Objective: minimize t
        obj = Minimize(t)
        prob = Problem(obj, constraints)
        
        # Warm start
        # Best b_left is from binary_template we constructed
        if i_iter == 0:
            # Set initial lower half, enforcing sum = n/2
            # Convert binary pattern to left part
            b_true = binary_template
            b_left_vals = b_true[:half+1]
            b_left.value = b_left_vals
        
        try:
            # set a timeout for the solver:  a fixed sub time
            solver_timeout = min(remaining_solver_time / (max_iters - i_iter), 20.0)
            # Ensure solver doesn't exceed total time for MILP stack
            sol = prob.solve(solver=cp.CBC, verbose=False, maximumSeconds=solver_timeout)
            
            # Evaluate if we found a new best
            if sol is not None and sol < min_ws:
                # Extract the optimal b (left half)
                b_opt_left = b_left.value
                if b_opt_left is not None and np.all(b_opt_left >= 0) and np.all(b_opt_left <= 1):
                    # construct full b from left half
                    b_opt = np.append(b_opt_left, b_opt_left[-2::-1])
                    corr_full = compute_full_correlations(b_opt, n_points)
                    worst_shift_idx = np.argmax(corr_full)
                    worst_val = corr_full[worst_shift_idx]
                    
                    if worst_val < min_ws:
                        min_ws = worst_val
                        best_b = b_opt.copy()
                
                elapsed = time.time() - start_solver
                remaining_solver_time = time_limit - elapsed - 5.0
                if remaining_solver_time <= 0:
                    break

        except Exception as e:
            # fallback or just proceed
            elapsed = time.time() - start_solver
            remaining_solver_time = time_limit - elapsed - 5.0
            if remaining_solver_time < 0:
                break
            pass
        
        # Compute the full correlations for *all* shifts to find worst
        # propagate better bounds
        if best_b is not None:
            corr_all = compute_full_correlations(best_b, n_points)
            worst_shift = np.argmax(corr_all)
            worst_val = corr_all[worst_shift]
            
            if worst_shift not in shift_set:
                shift_set.append(worst_shift)
        else:
            break
        
        elapsed = time.time() - start_solver
        remaining_solver_time = time_limit - elapsed - 5.0
    
    return best_b, min_ws if min_ws < float('inf') else 0.5

def memetic_descent(h, n_points, max_steps=20):
    """Refine h using projected gradient descent."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for step in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]
        corr_vals *= dx
        
        # Find the worst shift (non-zero)
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                break
            else:
                alpha *= 0.5

    return h_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    n_points = 200  # smaller size for MILP solution
    dx = 2.0 / n_points
    
    h_best = None
    c5_best = float('inf')
    
    # Try binary MILP cutting plane first
    try:
        b_sol, c5_milp = solve_binary_milp_cutting_plane(n_points, budget_s * 0.9)
        h_milp = b_sol.astype(float)
        c5_milp_eval = evaluate_c5(h_milp)
        if c5_milp_eval < c5_best:
            c5_best = c5_milp_eval
            h_best = h_milp.copy()
    except Exception as e:
        pass
    
    # Fall back to memetic refinement
    if h_best is None:
        if len(initial_h_values) >= n_points:
            h_initial = initial_h_values[:n_points]
        else:
            # Interpolate initial
            x_orig = np.linspace(0, 2, len(initial_h_values))
            x_new = np.linspace(0, 2, n_points)
            h_initial = np.interp(x_new, x_orig, initial_h_values)
        h_initial = project_to_feasible(h_initial, n_points)
        h_best = h_initial
        c5_best = evaluate_c5(h_best)
    
    # Now cache initial structure for next refinement
    try:
        # Refine with memetic descent if possible
        h_refined = memetic_descent(h_best, n_points, max_steps=20)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < c5_best:
            h_best = h_refined
            c5_best = c5_refined
    except Exception as e:
        pass
    
    return (h_best, c5_best, n_points)
