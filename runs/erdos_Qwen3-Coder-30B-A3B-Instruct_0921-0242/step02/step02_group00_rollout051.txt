import numpy as np
import time
from scipy.signal import correlate, fftconvolve
import math
from cvxpy import *

def aperiodic_autocorr(h, dx=1.0):
    """Compute aperiodic autocorrelation via FFT."""
    n = len(h)
    # Zero pad to avoid wraparound
    h_padded = np.pad(h, (0, n-1), mode='constant')
    h_conj = np.pad(h[::-1], (0, n-1), mode='constant')
    corr = fftconvolve(h_padded, h_conj, mode='valid')
    # Extract the first n elements (correlation at shifts 0,1,...,n-1)
    gamma = corr[:n]
    return gamma

def softmax(beta, x):
    """Numerically stable softmax."""
    x_center = x - np.max(x)
    exp_x = np.exp(beta * x_center)
    return exp_x / np.sum(exp_x)

def conv(a, b):
    """Linear convolution that mimics numpy's mode='full' and zero padding."""
    # No padding needed for zero-aperiodic conv
    result = fftconvolve(a, b, mode='full')[:len(a)+len(b)-1]
    return result

def project_box_sum(h, total):
    """Project onto the set {h: 0 ≤ h_i ≤ 1, sum h_i = total}."""
    # clip to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = total - current_sum

    if abs(diff) < 1e-12:
        return h

    # Sort h in descending order to find where to adjust
    idx_sorted = np.argsort(h)[::-1]
    h_sorted = h[idx_sorted]

    # Find where to shift (if we adjust all in the same direction, we must be smart)
    # We do waterfilling
    h_out = h.copy()
    # Create a ranking of indices by values
    low_idx = np.argsort(h)[::-1]  # indexes from high to low
    low_vals = h[low_idx]

    # Waterfilling: compute cumulative sum of shifted values
    cumsum = np.cumsum(low_vals)
    # We're adding the same amount (diff/n) everywhere
    # We want nothing to go above 1, and the sum to be desired_total
    target_sum = total

    # Adjust
    if diff > 0:
        # Add diff to elements, but keep bounded by 1
        for i in range(len(h)):
            if h_out[low_idx[i]] < 1:
                available = 1 - h_out[low_idx[i]]
                act_add = min(available, diff)
                h_out[low_idx[i]] += act_add
                diff -= act_add
                if diff <= 0:
                    break
    else:
        # We're reducing; should distribute evenly, but if we're below zero we need to handle
        for i in range(len(h)):
            # This is slightly tricky to do right
            pass

    # For clearer correctness:
    # Final projection via root-finding
    # Clone input
    h_temp = h.copy()
    h_temp = np.clip(h_temp, 0, 1)
    current_sum = np.sum(h_temp)
    if abs(current_sum - total) < 1e-12:
        return h_temp

    # Feasible set assumes that the sum is the desired total
    # So we will do a dynamic projection (more precise approach)
    # Use binary search on lambda such that solution to:
    # sum(clip(y[i]-lambda, 0, 1)) = total
    def sum_clipped(x, lmbda):
        return np.sum(np.clip(x - lmbda, 0, 1))

    # Binary search on lambda
    tol = 1e-12
    iter_max = 50
    lower, upper = -1.0, 1.0

    while lower <= upper and iter_max > 0:
        iter_max -= 1
        lmb = (lower + upper) * 0.5
        s = sum_clipped(h_temp, lmb)
        if abs(s - total) < tol:
            break
        elif s > total:
            lower = lmb
        else:
            upper = lmb

    # Apply final projection
    # We can be more careful
    # Better method: find lambda such that sum(clip(h_temp - lambda, 0, 1)) = total
    # Binary search over lambda
    def objective(lmbda):
        return np.sum(np.clip(h_temp - lmbda, 0, 1)) - total

    # Initial bounds
    l = -2.0
    r = 2.0
    for _ in range(50):
        m = (r + l) / 2
        val = objective(m)
        if abs(val) < 1e-10:
            break
        elif val > 0:
            r = m
        else:
            l = m

    final_lmb = (l + r) / 2
    h_out = np.clip(h_temp - final_lmb, 0, 1)
    return h_out

def score_guided_swap(h, epsilon=1e-5):
    """Perform a deterministic swap if some shift is near optimal."""
    n = len(h)
    # Compute correlations (c_s = 1 - (2/n)*gamma_s)
    gamma = aperiodic_autocorr(h)
    # compute C_s = 1 - (2/n)*gamma_s
    C = 1.0 - (2.0 / n) * gamma
    # Identify worst shifts within epsilon of best
    max_C = np.max(C)
    worst_indices = np.where(abs(C - max_C) < epsilon)[0]

    if len(worst_indices) == 0:
        return h

    # Compute scores for each index
    score = np.zeros(n)
    for s in worst_indices:
        # score_i = sum_{s} (h_{i+s} + h_{i-s})
        for i in range(n):
            if s <= i:
                score[i] += h[i-s]
            if i+s < n:
                score[i] += h[i+s]
    
    # Find index with largest score (good candidate to increase) 
    # and index with smallest score (good candidate to decrease)
    elite_source = np.argmax(score)
    elite_target = np.argmin(score)
    move = min(0.01, h[elite_source], 1.0 - h[elite_target])
    if move < 1e-10:
        return h

    h_new = h.copy()
    h_new[elite_source] -= move
    h_new[elite_target] += move
    return project_box_sum(h_new, len(h)/2)

def evaluate_c5(h):
    """Evaluate the C5 constant for a function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1-h, mode='full')
    # scale and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n1 = 120
    half = n1 // 2
    num_shifts = n1  # only non-negative shifts (because the negative ones are same as positive by |preludes|

    # Phase I: Binary MILP solution for small n₁  
    b = Variable(n1, boolean=True)
    a = Variable((n1, n1), boolean=True)  # a[i,s] = b[i] * b[i+s] for s≥0 and 0 ≤ i < n1-s
    gamma = Variable(n1)  # autocorrelations, 0 ≤ s < n1
    L = Variable()  # lower bound on all gamma_s

    constraints = [sum(b) == half, L >= 0]
    for s in range(n1):  # Only non-negative shifts for symmetry reduction and simplicity
        if s == 0:
            constraints += [gamma[s] == sum(b)]
        else:
            # Define gamma[s] = \sum_{i=0}^{n1-1-s} a[i,s]
            range_end = n1 - s
            constraints += [gamma[s] == sum(a[i, s] for i in range(range_end))] 

        # McCormick constraints for a[i,s] = b[i]*b[i+s]
        for i in range(range_end):
            # a[i,s] ≤ b[i]; a[i,s] ≤ b[i+s]
            constraints.append(a[i, s] <= b[i])
            constraints.append(a[i, s] <= b[i+s])
            # a[i,s] ≥ b[i] + b[i+s] - 1
            constraints.append(a[i, s] >= b[i] + b[i+s] - 1)

    # Enforce lower bound
    for s in range(n1):
        # gamma[s] ≥ L
        constraints.append(gamma[s] >= L)

    objective = Maximize(L)
    prob = Problem(objective, constraints)

    # Solve using CBC
    time_limit = 0.2 * budget_s
    prob.solve(solver=CPLEX, verbose=False, timeLimit=time_limit, gapRel=5e-3)

    if prob.status in ["optimal", "optimal_inaccurate"]:
        b_star = np.round(b.value).astype(int)
    else:
        if b.value is not None:
            b_star = np.round(b.value).astype(int)
        else:
            b_star = np.zeros(n1, dtype=int)
            
    # Phase II: Upsample to target resolution
    n2 = 300
    x1 = np.linspace(0, 2, n1, endpoint=False)
    x2 = np.linspace(0, 2, n2, endpoint=False)
    h0 = np.interp(x2, x1, b_star)
    h0 = project_box_sum(h0, n2/2)

    # Optional Gaussian blur
    h = h0.copy()
    # Smooth briefly
    kernel = np.exp(-np.linspace(-2.5, 2.5, 5)**2/2)
    kernel /= kernel.sum()
    h = np.convolve(h, kernel, mode='same')
    h = project_box_sum(h, n2/2)

    # Phase III: gradient descent with smooth max
    # Softmax approximation to C5 = max_{s=0..n-1} C_s
    beta = 2.0
    eta = 1.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    max_iter = 50000
    steps = 0

    def step_grad_descent(h, beta, eta=eta):
        gamma = aperiodic_autocorr(h)
        C = 1 - (2.0 / n2) * gamma
        w = softmax(beta, C)
        # Convolution of w with h (both are length n2)
        # this is approx w * h with zero padding, corresponding to linear conv: 
        w_fft = np.fft.fft(w, 2*n2-1)  # zero pad to prevent wraparound
        h_fft = np.fft.fft(h, 2*n2-1)
        conv1 = np.real(np.fft.ifft(w_fft * h_fft))[:n2]  # w * h
        # For w_rev * h, we reverse w first
        conv2 = np.real(np.fft.ifft(np.conj(w_fft) * h_fft))[:n2]  # w_rev * h

        grad = -(2.0 / n2) * (conv1 + conv2)
        
        # line search
        alpha = eta
        h_trial = h - alpha * grad
        h_trial = project_box_sum(h_trial, n2/2)
        c5_trial = evaluate_c5(h_trial)
        iter_ls = 0
        while c5_trial > evaluate_c5(h) and alpha > 1e-10 and iter_ls < 10:
            alpha *= 0.5
            h_trial = h - alpha * grad
            h_trial = project_box_sum(h_trial, n2/2)
            c5_trial = evaluate_c5(h_trial)
            iter_ls += 1
        return h_trial, c5_trial

    while time.time() - start_time < 0.85 * budget_s:
        h, c5 = step_grad_descent(h, beta, eta)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        steps += 1
        if steps % 150 == 0:
            beta = min(beta * 1.5, 200)
        if steps % 200 == 0:
            h = score_guided_swap(h, epsilon=1e-5)
    
    # Phase IV: final polishing
    while time.time() - start_time < budget_s - 0.1 * budget_s:
        # Identify worst shifts
        gamma = aperiodic_autocorr(best_h)
        C = 1 - (2.0 / n2) * gamma
        max_C = np.max(C)
        worst_indices = np.where(abs(C - max_C) < 1e-5)[0]
        # Compute a subgradient
        subgrad = np.zeros(n2)
        for s in worst_indices:
            if s == 0:
                passes = best_h   # Forward pass of h
            else:
                # shift_s is forward s shift
                # Original shift pattern was h_{i+s}, and we want to handle that correctly;
                # NumPy arrays are zero indexed, shifted i in Python index corresponds to i-in,
                # but for shift s, it is h[i+s] (non-cyclic)
                passes = np.zeros_like(best_h)
                if s < n2:
                    passes[s:] = best_h[:-s]

            # we want to compute shift_{+s}(h) + shift_{-s}(h)
            rev_shift = np.zeros_like(best_h)
            if s < n2:
                rev_shift[:-s] = best_h[s:]

            subgrad += passes + rev_shift

        subgrad *= -(2.0 / n2) / len(worst_indices)
        e = 1.0 / np.sqrt(steps + 10)
        h_new = project_box_sum(best_h - e * subgrad, n2/2)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        steps += 1

    return (best_h, best_c5, n2)
