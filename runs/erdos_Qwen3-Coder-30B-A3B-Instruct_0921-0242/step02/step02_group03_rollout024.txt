import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cma

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def memetic_descent(h, n_points, max_steps=30):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        # The maximum is at shift 0, so we need to find the min among non-zero shifts
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        # Gradient wrt h[i] is dx * (1 - h[i + worst_shift_idx] - h[i - worst_shift_idx])
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def build_symmetric_h(p, q, n_points):
    """Builds a symmetric piecewise constant h function."""
    # m is number of segments in one half (left or right)
    m = len(p)
    
    # Normalize widths
    w = np.exp(p)
    w = w / np.sum(w)
    
    # Normalize heights
    a = 1.0 / (1.0 + np.exp(-q))
    
    # Build the left half using widths and heights
    h_half = np.zeros(n_points // 2)
    cumulative_width = 0.0
    for i in range(m):
        start_idx = int(cumulative_width * n_points // 2)
        width_in_points = int(w[i] * n_points // 2) 
        end_idx = min(start_idx + width_in_points, n_points // 2)
        if end_idx > start_idx:
            h_half[start_idx:end_idx] = a[i]
        cumulative_width += w[i]
    
    # Mirror to get the full function
    h_full = np.concatenate([h_half, h_half[::-1]])
    
    return h_full

def cma_es_strategy(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 400
    m = 7  # segments in one half
    d = 2 * m  # total parameters: m widths + m heights
    
    # Initialize CMA-ES
    initial_mean = np.zeros(d)
    # Initialize p (widths) to make middle blocks larger
    initial_mean[:m] = np.ones(m) * 0.2
    initial_mean[m:] = np.ones(m) * 0.0  # heights start at 0 (logistic is 0.5)
    initial_mean[m + m//2 - 1:m + m//2 + 2] = 5.0  # middle heights are high
    initial_mean[:m//3] = 1.0  # initial width is good
    
    # Set up CMA-ES with appropriate parameters
    es = cma.CMAEvolutionStrategy(initial_mean, 0.3, {'popsize': 4 + int(3 * np.log(d)), 'verbose': -1})

    best_c5 = float('inf')
    best_h = None
    n_evals = 0
    
    G_noimp = 80
    restarts = 0
    max_restarts = 10
    time_limit = start_time + budget_s - 30  # Leave 30s for polishing
    
    # Keep track of the last improvement generation
    last_improvement = 0
    improvement_threshold = 1e-5
    
    while es.stop() is False and time.time() < time_limit and restarts <= max_restarts:
        # Generate offspring
        X = es.ask()
        
        # Evaluate each individual
        fitnesses = []
        for x in X:
            p = x[:m]
            q = x[m:]
            h = build_symmetric_h(p, q, n_points)
            h = project_to_feasible(h, n_points)
            c5 = evaluate_c5(h)
            fitnesses.append(c5)
            n_evals += 1
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = es.iteration
                
        # Tell CMA-ES
        es.tell(X, fitnesses)
        
        # Check for restart condition
        if es.iteration - last_improvement >= G_noimp:
            if restarts >= max_restarts:
                break

            # Restart with larger population
            new_popsize = min(es.popsize * 2, 100)
            es = cma.CMAEvolutionStrategy(es.result.xbest, 0.3, {
                'popsize': new_popsize,
                'verbose': -1
            })
            restarts += 1
            last_improvement = es.iteration
            
    # If no CMA-ES run was successful, return the initial solution
    if best_h is None:
        # Build initial h
        p_init = np.ones(m) * 0.2
        p_init[m//3:m//3*2] = 1.0  # Make middle segment wider
        q_init = np.ones(m) * 0.0
        q_init[m//2-1:m//2+2] = 5.0  # Make middle segments tall
        best_h = build_symmetric_h(p_init, q_init, n_points)
        best_h = project_to_feasible(best_h, n_points)
        best_c5 = evaluate_c5(best_h)
    
    # Polish with memetic descent
    if time.time() < time_limit:
        refined_h = memetic_descent(best_h, n_points, max_steps=25)
        refined_c5 = evaluate_c5(refined_h)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    return best_h, best_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    h_best, c5_best, n_points = cma_es_strategy(seed, budget_s)
    return (h_best, c5_best, n_points)
