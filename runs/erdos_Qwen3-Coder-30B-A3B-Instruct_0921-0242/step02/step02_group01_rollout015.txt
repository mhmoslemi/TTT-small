import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.linalg import toeplitz

def project_to_feasible(h, n):
    """Project h to feasible set with sum = n/2 and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def compute_c5(h):
    """Evaluate C5 from h using FFT-based correlation."""
    dx = 2.0 / len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    max_corr = np.max(corr.real) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 256   # Even number for spectral methods
    dx = 2.0 / n_points
    
    # Phase 1: SDP bisection for feasibility
    t_low, t_high = 0.0, 1.0
    t_best = 0.0
    best_r = None
    time_bisection = 0.35 * budget_s
    iter_count = 0
    
    while time.time() - start_time < time_bisection:
        iter_count += 1
        t_trial = (t_low + t_high) / 2.0
        # Create variables
        r = cp.Variable(n_points)
        # Constraints
        constraints = [
            r[0] == 1.0,
            r[1:] >= t_trial,
            r <= 1.0,
            cp.positive_semidefinite(toeplitz(r))
        ]
        # Dummy objective
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                t_best = t_trial
                best_r = r.value.copy()
            else:
                t_high = t_trial
        except:
            t_high = t_trial
    
    # Phase 2: Spectral factorisation from best_r
    if best_r is None:
        # Fallback to initial h
        if 'initial_h_values' in globals():
            h0 = initial_h_values[:n_points]
        else:
            h0 = np.ones(n_points) * 0.5
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Power spectrum
        S = np.fft.fft(best_r)
        S = np.maximum(S, 0)
        H = np.sqrt(S)
        g = np.real(np.fft.ifft(H))
        # Project to feasible
        h0 = project_to_feasible(g, n_points)
    
    # Phase 3: deterministic worst shift refinement
    # num iterations
    max_iter_refine = 200
    time_refine = 0.35 * budget_s
    elapsed_refine = 0

    while elapsed_refine < time_refine and time.time() - start_time < budget_s * 0.7:
        # Evaluate correlations
        h = h0.copy()
        h_fft = fft(h, 2 * n_points)
        ih_fft = fft(1 - h, 2 * n_points)
        corr = ifft(h_fft * np.conj(ih_fft))
        Ck = np.real(corr) * dx

        # Worst shift
        k_star = np.argmax(Ck[1:]) + 1  # exclude k=0
        # Gradient
        g = np.zeros(n_points)
        for i in range(n_points):
            left_idx = (i + k_star) % n_points
            right_idx = (i - k_star) % n_points
            g[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Step size
        t = elapsed_refine / time_refine * 100 + 1
        alpha = 0.5 * dx / np.sqrt(t)

        # Update
        h_new = h - alpha * g
        h0 = project_to_feasible(h_new, n_points)
        
        elapsed_refine = time.time() - (start_time + 0.35 * budget_s)
    
    # Final polishing with stochastic pair transfers
    # 10% of available time
    final_time = 0.1 * budget_s
    stop_time = start_time + 0.9 * budget_s + final_time

    h_final = h0.copy()
    c5_final = compute_c5(h_final)

    while time.time() < stop_time:
        # Random pair transfer
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        if h_final[i] <= 0 or h_final[j] >= 1:
            continue
        delta = min(0.01, h_final[i], 1 - h_final[j])
        h_test = h_final.copy()
        h_test[i] -= delta
        h_test[j] += delta
        h_test = project_to_feasible(h_test, n_points)
        c5_test = compute_c5(h_test)
        if c5_test < c5_final:
            c5_final = c5_test
            h_final = h_test

    return (h_final, c5_final, n_points)
