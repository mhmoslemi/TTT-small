import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cma

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def memetic_descent(h, n_points, max_steps=25):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        # The maximum is at shift 0, so we need to find the min among non-zero shifts
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def build_symmetric_h(p, q, n):
    """Build a symmetric piecewise-constant h from p and q parameters."""
    M = len(p)
    # Transform parameters to actual widths and heights
    w = np.exp(p) / np.sum(np.exp(p))  # Softmax
    a = 1.0 / (1.0 + np.exp(-q))       # Logistic
    
    # Build left half
    h_half = np.zeros(n // 2)
    cumulative_width = 0.0
    for i in range(M):
        width_i = int(np.round(w[i] * (n // 2)))
        if i == M - 1:  # Last segment
            width_i = (n // 2) - int(cumulative_width)
        start_idx = int(cumulative_width)
        h_half[start_idx:start_idx + width_i] = a[i]
        cumulative_width += width_i
    
    # Mirror to full
    h_full = np.concatenate([h_half, h_half[::-1]])
    
    return h_full

def c5_objective(params, n=400):
    """Objective function for CMA-ES optimization."""
    p, q = params[:len(params)//2], params[len(params)//2:]
    h = build_symmetric_h(p, q, n)
    h = project_to_feasible(h, n)
    return evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 30  # Leave 30s for polishing
    
    # Problem setup
    n = 400
    M = 7  # Number of segments on half interval
    
    # Initial parameters (p and q)
    init_p = np.array([0.0] * M)
    init_p[M//2-1:M//2+2] = [0.3, 0.3, 0.3]
    init_p = np.log(init_p)  # Use log-transformed initial widths
    
    init_q = np.array([0.0] * M)
    init_q[M//2-1:M//2+2] = 5.0  # High for center, low for edges
    init_q[:M//3] = -5.0
    init_q[2*M//3:] = -5.0
    
    # Combine into single vector
    init_params = np.concatenate([init_p, init_q])
    
    # Prepare for CMA-ES
    D = 2 * M
    σ0 = 0.3
    popsize = 4 + int(3 * np.log(D))
    
    opts = cma.CMAOptions()
    opts.set('verbose', -1)  # Quiet
    opts.set('popsize', popsize)
    opts.set('CMA_stds', np.ones(D))
    
    # Run CMA-ES
    es = cma.CMAEvolutionStrategy(init_params, σ0, opts)
    best_c5 = float('inf')
    best_h = None
    
    # Track no improvement generations
    no_improvement_count = 0
    G_noimp = 80
    restart_count = 0
    max_restarts = 10
    
    while es.should_stop() == False and time.time() < time_limit:
        # Generate population
        X = es.ask()
        fitnesses = []
        
        for x in X:
            try:
                fit = c5_objective(x, n)
                fitnesses.append(fit)
                if fit < best_c5:
                    best_c5 = fit
                    # Save corresponding h
                    p, q = x[:len(x)//2], x[len(x)//2:]
                    h_candidate = build_symmetric_h(p, q, n)
                    h_candidate = project_to_feasible(h_candidate, n)
                    best_h = h_candidate.copy()
            except Exception:
                fitnesses.append(float('inf'))
        
        es.tell(X, fitnesses)
        
        if len(fitnesses) > 0 and min(fitnesses) < best_c5 - 1e-5:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= G_noimp and restart_count < max_restarts:
            # Restart
            no_improvement_count = 0
            restart_count += 1
            es = cma.CMAEvolutionStrategy(init_params, σ0, opts)
            if best_h is not None:
                es.mean = init_params  # Not really, just keep it simple

    # Postprocessing: polish with memetic descent
    if best_h is None:
        # Fallback
        init_h = np.full(n, 0.5)
        init_h = project_to_feasible(init_h, n)
        best_h = init_h
        best_c5 = evaluate_c5(best_h)
    
    # Polish with memetic descent
    polished_h = memetic_descent(best_h, n, max_steps=25)
    polished_c5 = evaluate_c5(polished_h)
    
    if polished_c5 < best_c5:
        best_h = polished_h
        best_c5 = polished_c5
    
    return (best_h, best_c5, n)
