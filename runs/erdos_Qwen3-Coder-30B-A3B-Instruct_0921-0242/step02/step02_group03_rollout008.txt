import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def compute_correlation_changes(h, i, j, corr_array):
    """Efficiently compute the change in correlation when swapping h[i] and h[j]."""
    n = len(h)
    delta = np.zeros(n)
    for s in range(n):
        # When we swap h[i] with h[j], positions affected are:
        # - h[i] loses its influence on correlation with h[i+s] and h[i-s] (if valid)
        # - h[j] loses its influence on correlation with h[j+s] and h[j-s] (if valid)
        if i + s < n:
            delta[s] -= (1 - h[i + s]) + h[i + s]
        if j + s < n:
            delta[s] += (1 - h[j + s]) + h[j + s]
        if i >= s:
            delta[s] -= h[i - s] + (1 - h[i - s])
        if j >= s:
            delta[s] += h[j - s] + (1 - h[j - s])
    return delta

def anneal_binary(h, n, time_limit):
    """Simulated annealing on binary vectors to minimize maximum cross-correlation."""
    T0 = 0.02 * n
    alpha = 0.997
    max_moves = 60000
    T = T0
    move_count = 0
    current_h = h.copy()
    current_corr = correlate(current_h, 1 - current_h, mode='full')
    current_max = np.max(current_corr)
    
    best_h = current_h.copy()
    best_max = current_max
    
    tabu_list = []
    tabu_size = 100
    
    while T > 1e-5 and move_count < max_moves and time.time() < time_limit:
        # Find 1s and 0s for candidate swap
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Pick a 1 and a 0 at random to swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Check if this swap is in tabu list
        if (i, j) in tabu_list or (j, i) in tabu_list:
            continue
            
        # Compute the change in correlations due to swapping i and j
        delta = compute_correlation_changes(current_h, i, j, current_corr)
        new_corr = current_corr + delta
        new_max = np.max(new_corr)
        
        # Acceptance probability
        if new_max <= current_max or np.random.rand() < np.exp(-(new_max - current_max) / T):
            current_h[i] = 0
            current_h[j] = 1
            current_corr = new_corr
            current_max = new_max
            
            if new_max < best_max:
                best_h = current_h.copy()
                best_max = new_max
                
            # Add to tabu list
            tabu_list.append((i, j))
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        else:
            # Minor movement - no change made, but still reduce temperature
            pass
            
        T *= alpha
        move_count += 1

    return best_h, best_max

def polish_continuous(h_binary, n):
    """Polish a binary solution with continuous updates."""
    dx = 2.0 / n
    h = h_binary.astype(float)
    
    # Smooth max surrogate parameter
    lambda_val = 5.0
    eta = 0.1 * dx
    max_iter = 30
    
    for iteration in range(max_iter):
        # Compute both h and (1-h)
        h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
        
        # Full correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n]
        
        # Compute smooth max using log-sum-exp
        # Since log(exp(lambda * corr) + ...) = lambda * max(corr) + error,
        # we approximate by scaling factor. To make smooth max, we use logsumexp.
        # But since we have fixed fixed bounds of h in [0,1] and want to maximize
        # smooth minimum of corr_vals (not maximum), let's instead 
        # internally compute: s = sum(exp(lambda * c_s)), so that log(s)/lambda -> max c_s.
        # in order to compute the gradient wrt h_i, just use auto-diff manipulation:
        # for c_s = sum_i h_i (1 - h_{i+s})
        # when differentiating with respect to h_i we get:
        # d/dh_i = (1 - h_{i+s}) + sum_{t s.t. i-t=s} h_t
        # Let's compute it directly using shifts

        # To get outer product-like derivative efficiently:
        # gradient = partial_derivative of sum exp(lambda * c_s)  
        # = sum_s lambda * exp(lambda * c_s) * d c_s / d h_i
        # d c_s / d h_i = (1 - h_{i+s}) - (h_{i-s} + ... )
        # So construct the gradient per i by summing for all s such that the term involves h_i
        
        # Do not use logsumexp directly due to numerical issues; instead, compute gradient directly
        # Outline:
        #  1. compute correlation array for given h,
        #  2. compute gradient of that corr w.r.t. each h[i]  using shifts,
        #  3. update h by gradient descent,
        #  4. project back to feasible set and renomalize if sum changed.

        # Compute the gradient for each h_i
        grad = np.zeros_like(h)
        for i in range(n):
            # contributions for correlations involving h[i]
            # when shifting i forward
            for s in range(0, n - i):  # s = 0..(n-i-1) means i+s < n
                grad[i] += (1 - h[i + s]) * lambda_val * np.exp(lambda_val * corr_vals[s])
            # when shifting backward  
            for s in range(1, i + 1):  # s = 1..i means i-s >= 0
                grad[i] -= h[i - s] * lambda_val * np.exp(lambda_val * corr_vals[s])
        
        # Normalize gradient for numerical stability
        norm = np.sqrt(np.sum(grad**2))
        if norm > 1e-12:
            grad /= norm
            
        # Gradient step
        h_new = h - eta * grad
        
        # Project back to feasible bounds [0,1] and renormalize
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = project_to_feasible(h_new, n)
        
        # Check max C5 after each update step
        h = h_new
        
        # Increase lambda slightly for sharper max approx.
        lambda_val = min(30.0, lambda_val + 0.1)
        
        # Early stopping based on small changes or loss of improvement
        if iteration % 5 == 0:
            max_corr = evaluate_c5(h)
            if max_corr > 0.99 * (evaluate_c5(h_binary) if iteration == 0 else max_corr):
                break
                
        eta *= 0.93  # Gradually decrease learning rate (clipping may make it very small; this avoids it)
        
    return h, evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    # Select odd n for better symmetry properties and resolution
    n = 221
    dx = 2.0 / n
    h_best = None
    c5_best = float('inf')

    num_restarts = 12
    for restart in range(num_restarts):
        if time.time() >= time_limit - 10:
            break
            
        # Start with either a block pattern or random binary vector
        if restart % 2 == 0:
            # Block pattern with jitter
            h = np.zeros(n, dtype=np.int8)
            mid = n // 2
            start = mid - n // 4
            end = mid + n // 4
            h[start:end] = 1
            
            # Add some random jitters to increase diversity
            for _ in range(10):
                idx1 = np.random.randint(0, n)
                idx2 = np.random.randint(0, n)
                if h[idx1] != h[idx2]:
                    h[idx1], h[idx2] = h[idx2], h[idx1]
        else:
            # Random binary vector  
            h = np.random.randint(0, 2, size=n).astype(np.int8)
            # Ensure exact half ones
            while np.sum(h) != n // 2:
                if np.sum(h) < n // 2:
                    idx = np.where(h == 0)[0]
                    h[idx[0]] = 1
                else:
                    idx = np.where(h == 1)[0]
                    h[idx[0]] = 0

        # Core loop for this restart
        # Run simulated Annealing (adjust time limit for this trial)
        trial_limit = time_limit - (time.time() - start_time)
        if trial_limit < 10:
            break
        h_opt, max_corr = anneal_binary(h, n, time.time() + 10)
        
        # Optional polishing
        h_polished, c5 = polish_continuous(h_opt, n)
        
        if c5 < c5_best:
            c5_best = c5
            h_best = h_polished.copy()

    if h_best is None:
        h_best = np.ones(n) * 0.5
        c5_best = evaluate_c5(h_best)
        
    return (h_best, c5_best, n)
