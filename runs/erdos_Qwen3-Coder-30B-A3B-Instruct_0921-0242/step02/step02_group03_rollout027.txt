def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    if diff > 0:
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    h_copy = np.clip(h_copy, 0, 1)
    return h_copy

def memetic_descent(h, n_points, max_steps=20):
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Find the worst shift using a full correlation
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]
        corr_vals = corr_vals * dx

        # Find the minimum among non-zero shifts
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
    
    return h_current

    # Build a neutral initial binary solution from the continuous one
    h_binary = (h_initial > 0.5).astype(int)
    current_count = np.sum(h_binary)
    
    h_binary = project_to_feasible(h_binary, n_points)
    binary_pattern = h_binary > 0.5  # Ensure binary
    
    # Warm start with binary version of best continuous solution
    binary_template = binary_pattern.copy()
    
    # If not enough 1s, raise some of the 0s to 1s
    # If too many 1s, lower some 1s to 0s
    actual_count = np.sum(binary_template)
    if actual_count > n_points // 2:
        # Need to remove some 1s

            # Start from the rightmost 1s and convert them to 0 (more local change)

                binary_template[idx] = False
    elif actual_count < n_points // 2:
        # Need to add 1s
        indices_zeros = np.where(~binary_template)[0]
        if len(indices_zeros) > 0:
            # Convert some 0s to 1s from the left

        else:
            prob = np.exp((old_cost - new_cost) / temp)

            current_c5 = new_c5

    # Final polishing with memetic descent once more

        best_h = memetic_descent(best_h, n_points, max_steps=10)
