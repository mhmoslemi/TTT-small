import numpy as np
import time
from scipy.optimize import differential_evolution, linprog
from scipy.fft import rfft, irfft
from scipy.interpolate import interp1d

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    
    # Water-filling approach
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorrelation(h):
    """Compute the autocorrelation using FFT for efficiency."""
    n = len(h)
    H = rfft(h)
    A = irfft(H * np.conj(H), n)
    return A

def basis_evaluation(x, M):
    """Evaluate a Fourier basis for x."""
    cos_basis = [np.cos(m * np.pi * x) for m in range(1, M+1)]
    sin_basis = [np.sin(m * np.pi * x) for m in range(1, M+1)]
    return np.array([1.0] + cos_basis + sin_basis)

def get_basis_functions(n0, M):
    """Generate the basis functions on the interval [0, 2]."""
    x = np.linspace(0, 2, n0, endpoint=False)
    basis = np.zeros((n0, 2*M + 1))
    basis[:, 0] = 1.0  # constant term for a0 = 0.5 is not used here due to later scaling
    for i in range(1, M+1):
        basis[:, 2*i-1] = np.cos(i * np.pi * x)
        basis[:, 2*i] = np.sin(i * np.pi * x)
    return basis

def fitness_function(c, basis, n0):
    """Evaluate the fitness of the candidate coefficients."""
    # Reconstruct function with the given coefficients
    # c = [a1, b1, a2, b2, ..., aM, bM]
    M = len(c) // 2
    # add a0 = 0.5 for the offset term
    h_raw = basis @ c  # only the cosine/sine components
    h_raw = h_raw + 0.5  # a0
    h = np.clip(h_raw, 0, 1)
    h = project_to_feasible(h, n0)
    return evaluate_c5(h)

def linearise_and_solve_lp(h, A_vals, k_star, n0, debug=False):
    """Solve the linearized LP subproblem at current h."""
    # The constraint is A(k) >= t for all k
    # A(k) = h^T P_k h
    # grad A(k) = 2 * roll(h, k)
    
    # we are solving max t s.t.
    # h^T P_k h >= t for all k in active set
    
    # For the LP: t <= A(k) - grad A(k)^T (h-h0),
    # i.e., t + grad A(k)^T h >= A(k) - grad A(k)^T h0
    
    # We linearize at h0 = h, so the constraint becomes:
    # t + grad A(k)^T h >= A(k)
    # i.e., t >= A(k) - grad A(k)^T h
    
    # we use only the worst k's (since the minimum should be at a few)
    # Use largest few negative residuals or for now just sort by closeness to A(k) = t
    
    # we've already done A vals for all k's by FFT earlier using current h
    # this should be correct for h
    # TODO: we need just the worst cases; let's take 10 worst shifts
    # Define constraint set
    n_k = len(A_vals)
    active_k_idx = np.argsort(A_vals)[:10]  # but this would be the minima, we want max buffer
    # h need not be used in the computation of A_vals?
    # actually, let's directly use the method for solving linearized can take shift k_idx with the worst A(k)
    
    # Solvers handle this by rolling
    
    # We want to find k_star such that A(k_star) = t = min_k A(k)
    # Try expanding around those shifts that give buffer tightest equality constraints
    worst_k_indices = np.argsort(A_vals)[:10]  # we want the lowest first
    # because we're trying to make A(k) >= t for all k, so it's a min constraint, not max
    
    # So solve:
    # max t
    # s.t. h^T P_k h >= t ; for some k's
    # also h >= 0, h <= 1, sum h = n0/2
    
    # Current value at k_star:
    t = A_vals[k_star]
    
    # Linear constraint set:
    # Let g_k = (P_k + P_k.T) h. 
    # Then dA(k)/dh = g_k
    # And the constraint is h^T P_k h >= t
    # Linearization around h_old is:
    # (h_old^T P_k h_old) + (h_old^T P_k + P_k^T h_old)^T (h - h_old) >= t
    # => t <= (h_old^T P_k h_old) - (h_old^T P_k + P_k^T h_old)^T h_old + (h_old^T P_k + P_k^T h_old)^T h
    
    # Simpler simpler way to compute gradient at k: it's just 2 * roll(h, k)
    # Now the constraint can be written as:
    # (h_old^T P_k h_old) + g_k^T (h - h_old) >= t
    # (h_old^T P_k h_old) - g_k^T h_old + g_k^T h >= t
    # g_k^T h >= t - (h_old^T P_k h_old) + g_k^T h_old
    
    # Matrix form (as convex cone):
    # A_L < x < b_L  for general inequality constraints Ax <= b
    # So we have (g_k^T h - (h_old^T P_k h_old) + g_k^T h_old) >= t
    # let's construct the problem
    # get the constraint matrix for these k's
    
    # Acoeff = 1 for objective t; rows for each fringe
    rows = []
    rows_b = []
    
    # add 0 <= h_i <= 1
    # inequality: -h_i <= 0, h_i <= 1
    for i in range(n0):
        rows.append([-1 if i == j else 0 for j in range(n0)] + [0])
        rows_b.append(0)
        rows.append([1 if i == j else 0 for j in range(n0)] + [0])
        rows_b.append(1)
    
    # sum h = n0/2  =>  sum h_i = 1 (after normalization, scaled so sum = 1 )
    rows.append([1]*n0 + [0])
    rows_b.append(n0/2)
    
    # constraint for the shifted autocorrelations
    for k in worst_k_indices:
        # gradient at k of: sum(h_i * h_ij)
        # roll vector h by k and take [i] * [i-k] = h_i * h_{i-k}
        # so d(A(k))/d h_i = h_{i-k} + h_{i+k}
        # Let's compute as shift columns in P_k
        row = [0]*n0 + [1]  # t has coefficient 1
        # For shift k, r_i = h_{i-k} + h_{i+k}
        # (interpret cyclically)
        for i in range(n0):
            idx_l = (i - k) % n0
            idx_r = (i + k) % n0
            row[i] = h[idx_l] + h[idx_r]
        row = np.array(row)
        rows.append(row)
        rows_b.append(A_vals[k] - h @ ( np.roll(h, k) + np.roll(h, -k) ) / 2.0)
    
    # Make matrix A_ub x <= b_ub
    A_ub = np.array(rows)
    b_ub = np.array(rows_b)
    
    # The objective, maximize t, i.e. minimize -t => minimize c^T x = -1*t + 0 for h
    c = np.zeros(n0+1)
    c[-1] = -1.0  # t coefficient negative
    
    # Bounds via variable bounds
    bounds = [(0.0, 1.0) for _ in range(n0)] + [(None, None)]  # h bounds, t unbounded
	
    try:
        sol = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if sol.success:
            h_new = sol.x[:n0]
            t_new = sol.x[n0]
            return h_new, t_new
        else:
            # fallback to old h if infeasible
            return h.copy(), A_vals[k_star]
    except:
        return h.copy(), A_vals[k_star]

def subgradient_step(h, k_star, n0):
    """Simple subgradient-like step to improve current best."""
    dx = 2.0 / n0
    # just use the aut correlation matrix
    A = compute_autocorrelation(h)  # assumes h has 0 padding for internal wrapping
    max_A = np.max(A)
    # try to move mass from where h(x) is high to low for shift k_star
    # gradient = 2 * (roll(h, k_star) + roll(h, -k_star))
    grad_k = 2 * (np.roll(h, k_star) + np.roll(h, -k_star))
    # Move mass in the direction of -grad_k
    h_new = h - 0.01 * grad_k
    h_new = project_to_feasible(h_new, n0)
    return h_new

    bounds = [(-0.5, 0.5)] * (2*M)

        return fitness_function(c, basis, n0)
    
    # Run DE
    result = differential_evolution(de_fitness, bounds, seed=seed, maxiter=2000)
    
    # Collect top 5 distinct candidates
    candidates = [result.x]  # best result so far
    # Collect nearby solutions
    unique_cands = set()

    for _ in range(sample_size):
        if time.time() - start_time > 0.45 * budget_s:
            break
        # Mutate base vector
        c_trial = result.x + np.random.normal(0, 0.05, len(result.x))
        c_trial = np.clip(c_trial, -0.5, 0.5)
        # Add to candidates
        if tuple(c_trial) not in unique_cands:
            unique_cands.add(tuple(c_trial))
            candidates.append(c_trial)
        if len(candidates) >= 5:
            break
    
    # Evaluate all the candidates
    candidate_evals = []
    for c in candidates:
        if time.time() - start_time > 0.5 * budget_s:
            break
        val = fitness_function(c, basis, n0)
        candidate_evals.append((c, val))
    
    # Sort by lowest fitness
    candidate_evals.sort(key=lambda x: x[1])
    
    # Top 5
    final_candidates = candidate_evals[:5]
    
    # Stage D: Sequential Linear Programming
    final_h = None
    min_c5 = np.inf
    
    for _, c in final_candidates:
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # construct h with basis
        h = 0.5 + basis @ c  # a0 = 0.5
