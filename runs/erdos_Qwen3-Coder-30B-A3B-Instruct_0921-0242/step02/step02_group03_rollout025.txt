import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cma

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def memetic_descent(h, n_points, max_steps=30):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        # The maximum is at shift 0, so we need to find the max among non-zero shifts
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmax(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        # Gradient wrt h[i] is dx * (1 - h[i + worst_shift_idx] - h[i - worst_shift_idx])
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def build_h_from_params(p, q, n):
    """Build h from parameters using piecewise constant symmetric function."""
    M = len(p)
    # Softmax for widths (p_i -> w_i)
    w = np.exp(p)
    w = w / np.sum(w)
    
    # Logistic for heights (q_i -> a_i)
    a = 1.0 / (1.0 + np.exp(-q))
    
    # Build left half
    h_half = np.zeros(n//2)
    cumulative = 0.0
    for i in range(M):
        start_idx = int(np.round(cumulative * (n//2)))
        end_idx = int(np.round((cumulative + w[i]) * (n//2)))
        if start_idx >= end_idx:  # Avoid empty segments
            end_idx = start_idx + 1
        h_half[start_idx:end_idx] = a[i]
        cumulative += w[i]
    
    # Construct full function with symmetry
    h_full = np.concatenate([h_half, h_half[::-1]])
    return h_full

def objective_cma(pq):
    """Objective for CMA-ES optimization."""
    # pq = [p1, p2, ..., pM, q1, q2, ..., qM]
    p = pq[:M]
    q = pq[M:]
    
    # Build the function
    h = build_h_from_params(p, q, n_points)
    # Project to ensure integral constraint
    h = project_to_feasible(h, n_points)
    
    # Evaluate C5
    c5 = evaluate_c5(h)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 30  # Leave 30s for final polishing
    
    # Parameters
    n_points = 400
    M = 7  # Number of segments in each half (total 2*M parameters)
    
    # Initialize mean vector - start with a good symmetric shape
    p_mean = np.zeros(M)
    q_mean = np.zeros(M)
    
    # Distribute width: middle segments should be wider
    # Distribute heights: central segments higher, outer segments lower
    p_mean[M//2 - 1:M//2 + 2] = np.log(0.3/3)  # Middle segments get more weight
    p_mean[:M//2 - 1] = np.log(0.1/(M//2 - 1))  # Left side
    p_mean[M//2 + 2:] = np.log(0.1/(M//2 - 1))  # Right side
    
    q_mean[M//2 - 1:M//2 + 2] = 5.0  # High height: middle
    q_mean[:M//2 - 1] = -5.0  # Low height: left
    q_mean[M//2 + 2:] = -5.0  # Low height: right
    
    # Initial parameters for CMA-ES
    mean = np.concatenate([p_mean, q_mean])
    sigma = 0.3
    bounds = []  # Unbounded
    options = {
        'maxiter': 1000,
        'verbose': -9
    }
    
    # Run CMA-ES optimization
    es = cma.CMAEvolutionStrategy(mean, sigma, options)
    best_c5 = float('inf')
    best_h = None
    
    # Keep track of generations without improvement
    generations_no_improvement = 0
    best_generation = 0
    
    try:
        while not es.stop() and time.time() < time_limit:
            # Generate offspring
            X = es.ask()
            fitness = []
            
            for x in X:
                f = objective_cma(x)
                fitness.append(f)
            
            # Tell the CMA-ES about fitnesses
            es.tell(X, fitness)
            
            # Keep best ever
            current_best = np.min(fitness)
            if current_best < best_c5:
                best_c5 = current_best
                idx = np.argmin(fitness)
                best_params = X[idx]
                p_best = best_params[:M]
                q_best = best_params[M:]
                h_temp = build_h_from_params(p_best, q_best, n_points)
                h_temp = project_to_feasible(h_temp, n_points)
                best_h = h_temp.copy()
                generations_no_improvement = 0
                best_generation = es.generation
            else:
                generations_no_improvement += 1
                
            # Some restarting policy
            if generations_no_improvement > 80 and es.generation - best_generation > 20:
                # Restart
                generations_no_improvement = 0
                es = cma.CMAEvolutionStrategy(mean, sigma, options)
                
    except Exception as e:
        pass  # May happen if time is up
    
    if best_h is None:
        # Fallback to simple solution
        h_temp = np.ones(n_points) * 0.5
        h_temp = project_to_feasible(h_temp, n_points)
        best_h = h_temp
        best_c5 = evaluate_c5(h_temp)
    
    # Polishing with memetic descent
    try:
        polished_h = memetic_descent(best_h, n_points, max_steps=25)
        polished_c5 = evaluate_c5(polished_h)
        
        if polished_c5 < best_c5:
            best_h = polished_h
            best_c5 = polished_c5
    except Exception:
        pass
        
    return (best_h, best_c5, n_points)
