import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def build_trapezoid(t, H, n):
    """Construct a trapezoidal function h(x) with given parameters."""
    dx = 2.0 / n
    h = np.zeros(n)
    
    L = 1.0 / H - t
    a = (2.0 - 2.0 * t - L) / 2.0
    
    if L < 0 or a < 0:
        return h
    
    # Indices for ramps and top
    i_a = int(a / dx)
    i_t = int(t / dx)
    i_L = int(L / dx)
    
    # Left ramp
    if i_t > 0:
        i_start = i_a
        i_end = min(i_a + i_t, n)
        if i_start < i_end:
            x = np.arange(i_start, i_end) * dx + dx / 2
            h[i_start:i_end] = H * (x - a) / t
    
    # Flat top
    if i_L > 0:
        i_start = i_a + i_t
        i_end = min(i_start + i_L, n)
        if i_start < i_end:
            h[i_start:i_end] = H
    
    # Right ramp
    if i_t > 0:
        i_start = i_a + i_t + i_L
        i_end = min(i_start + i_t, n)
        if i_start < i_end:
            x = np.arange(i_start, i_end) * dx + dx / 2
            h[i_start:i_end] = H * (a + 2.0 * t + L - x) / t
    
    return h

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT correlation."""
    n = len(h)
    dx = 2.0 / n
    
    # Zero padding for linear convolution
    N = 2 * n
    Hc = fft(h, N)
    Gc = fft(1 - h, N)
    corr = ifft(Hc * np.conj(Gc)).real[:2*n-1]
    
    # Scale by dx and find the maximum
    return np.max(corr) * dx

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    h_copy = np.clip(h_copy, 0, 1)
    return h_copy

def memetic_descent(h, n_points, max_steps=30):
    """Refine h using projected gradient descent."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute correlation using FFT
        N = 2 * n_points
        Hc = fft(h_current, N)
        Gc = fft(1 - h_current, N)
        corr = ifft(Hc * np.conj(Gc)).real[:2*n_points-1]
        
        # Get shift index for the max correlation value (excluding shift 0)
        non_zero_shifts = np.concatenate([corr[1:], corr[:n_points-1]])
        worst_shift_idx = np.argmax(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5_fft(h_trial)
            current_c5 = evaluate_c5_fft(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    # Use a fine resolution for parameter search
    n = 800
    dx = 2.0 / n
    best_val = float('inf')
    best_h = None
    
    # Coarse grid search
    H_vals = np.linspace(0.5, 1.0, 41)
    
    for H in H_vals:
        t_max = min(1.0/H, 2.0 - 1.0/H)
        if t_max <= 0:
            continue
        t_vals = np.linspace(0.0, t_max, 81)
        for t in t_vals:
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    
    # Refinement around best point
    H_refine = np.linspace(max(0.5, H_vals[np.argmin([abs(h - best_val) for h in H_vals])]-0.025),
                           min(1.0, H_vals[np.argmin([abs(h - best_val) for h in H_vals])]+0.025), 41)
    best_h_refine = None
    best_val_refine = float('inf')
    for H in H_refine:
        t_max = min(1.0/H, 2.0 - 1.0/H)
        if t_max <= 0:
            continue
        t_vals = np.linspace(max(0.0, 0.01),
                             min(t_max, 2.0 - 0.01), 41)
        for t in t_vals:
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val_refine:
                best_val_refine = val
                best_h_refine = h.copy()

    if best_h_refine is not None:
        best_h = best_h_refine
        best_val = best_val_refine
    
    # Local optimization with Nelder-Mead
    def obj(params):
        t, H = params
        h = build_trapezoid(t, H, n)
        h = project_to_feasible(h, n)
        return evaluate_c5_fft(h)
    
    # Start from a good initial guess if refining worked
    if best_h is not None:
        # Extract t, H from current solution
        # Very quick estimate
        t, H = 0.0, 0.5
        try:
            # Find mean height
            H_guess = np.mean(best_h)
            # Find length of uniform part
            flat_part = np.where(best_h > 0.1)[0]
            if len(flat_part) > 1:
                L_guess = (flat_part[-1] - flat_part[0]) * dx
                t_guess = (1.0/H_guess - L_guess) / 2.0
                t = t_guess
                H = H_guess
        except:
            pass

        # Nelder-Mead optimization to refine
        try:
            result = minimize(obj, [t, H], method='Nelder-Mead', options={'maxfev': 50, 'disp': False})
            if not result.success:
                pass
            else:
                t_new, H_new = result.x
                h_opt = build_trapezoid(t_new, H_new, n)
                h_opt = project_to_feasible(h_opt, n)
                new_val = evaluate_c5_fft(h_opt)
                if new_val < best_val:
                    best_val = new_val
                    best_h = h_opt
        except:
            pass

    # Polish with memetic descent
    # Possibly optimize more
    if best_h is not None and (time.time() - start_time) < time_limit - 1.0:
        h_opt = memetic_descent(best_h, n, max_steps=20)
        c5_opt = evaluate_c5_fft(h_opt)
        if c5_opt < best_val:
            best_val = c5_opt
            best_h = h_opt.copy()

    # Fallback random restarts
    while (time.time() - start_time) < time_limit - 5.0:
        h_restart = np.random.rand(n)
        h_restart = project_to_feasible(h_restart, n)
        h_opt = memetic_descent(h_restart, n, max_steps=30)
        c5_opt = evaluate_c5_fft(h_opt)
        if c5_opt < best_val:
            best_val = c5_opt
            best_h = h_opt.copy()
    
    # Final evaluation to make sure
    if best_h is not None:
        final_c5 = evaluate_c5_fft(best_h)
        if final_c5 < best_val:
            best_val = final_c5
    
    return (best_h, best_val, n)
