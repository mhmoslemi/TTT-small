import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft
from scipy.interpolate import interp1d
from scipy.signal import correlate

def build_cyclic_shift_matrix(k, n):
    """Builds the cyclic shift matrix P_k such that (P_k x)_i = x_{i-k mod n}"""
    P = np.zeros((n, n))
    for i in range(n):
        j = (i - k) % n
        P[i, j] = 1
    return P

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def subgradient_refine(h, dx, remaining_time):
    """Refine h using subgradient descent."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    max_iter = 2000
    for _ in range(max_iter):
        if time.time() - start_time > remaining_time - 1.0:
            break
        # Compute gradient
        corr = correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        if k_star > len(h):
            k_star = k_star - len(h)
        else:
            k_star = k_star - len(h) + 1
        
        # Construct gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            left_idx = (i + k_star) % len(h)
            grad[i] -= dx * h[left_idx]
            right_idx = (i - k_star) % len(h)
            grad[i] += dx * h[right_idx]
        
        # Line search
        alpha = 1.0
        h_trial = h - alpha * grad
        h_trial = project_to_feasible(h_trial, len(h))
        c5_trial = evaluate_c5(h_trial)
        
        while c5_trial >= best_c5 and alpha > 1e-8:
            alpha /= 2.0
            h_trial = h - alpha * grad
            h_trial = project_to_feasible(h_trial, len(h))
            c5_trial = evaluate_c5(h_trial)
        
        if c5_trial < best_c5:
            h = h_trial
            best_c5 = c5_trial
            best_h = h.copy()
    
    return best_h, best_c5

def upsample_and_project(h, n2):
    """Upsample h to resolution n2 using linear periodic interpolation."""
    n1 = len(h)
    x1 = np.linspace(0, 2, n1, endpoint=False)
    x2 = np.linspace(0, 2, n2, endpoint=False)
    
    f = interp1d(x1, h, kind='linear', fill_value='periodic')
    h2 = f(x2)
    h2 = project_to_feasible(h2, n2)
    return h2

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    # SDP phase
    n = 180
    dx = 2.0 / n
    P = [build_cyclic_shift_matrix(k, n) for k in range(n)]

    # SDP problem formulation
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()

    constraints = [
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n/2,
    ] + [cp.trace(P[k] @ X) >= t for k in range(n)]

    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver='SCS', max_iters=25000, eps=1e-5)

    # Post-processing
    X_opt = X.value
    if X_opt is None:
        # Fallback to random
        h = np.random.rand(n)
        h = project_to_feasible(h, n)
        return h, evaluate_c5(h), n

    eigvals, eigvecs = np.linalg.eigh(X_opt)
    lead = eigvecs[:, -1] * np.sqrt(eigvals[-1])
    h0 = project_to_feasible(lead, n)

    # Keep best of few random draws
    best_h, best_c5 = h0, evaluate_c5(h0)
    for _ in range(15):
        if time.time() - start_time > budget_s - 10:
            break
        g = np.random.randn(n) @ np.linalg.cholesky(X_opt)
        h = project_to_feasible(np.clip(g, 0, 1), n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Sub-gradient polishing
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        best_h, best_c5 = subgradient_refine(best_h, dx, remaining_time)

    # Upsample if time remains
    if time.time() - start_time < 0.85 * budget_s:
        n2 = 2 * n
        h_up = upsample_and_project(best_h, n2)
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 5:
            best_h, best_c5 = subgradient_refine(h_up, 2.0 / n2, remaining_time)

    return best_h, best_c5, len(best_h)
