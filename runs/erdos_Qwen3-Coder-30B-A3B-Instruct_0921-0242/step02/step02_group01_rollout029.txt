import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft
from scipy.stats import dirichlet

def project_to_feasible(h, target_sum):
    """Ensure that sum(h) = target_sum, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach to adjust the sum
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def build_lp_data(h0):
    """Build the LP data for the CCP method."""
    n = len(h0)
    # Compute the auto-correlation of h0
    c = ifft(fft(h0, 2 * n) * np.conj(fft(h0, 2 * n))).real
    # The auto-correlations, indexed by k = 0, ..., n-1
    C0 = c[:n]
    # Build the matrix W where each row corresponds to a shift k
    W = np.zeros((n, n))  # Each row = h0[(i+k)%n] + h0[(i-k)%n]
    for k in range(n):
        W[k, :] = h0[(np.arange(n) + k) % n] + h0[(np.arange(n) - k) % n]
    A = -W  # A is negative because we want <= form
    b = -n/2 + C0  # RHS of the constraints
    return A, b

def solve_ccp_step(h0, A, b):
    """Solve one iteration of the CCP method."""
    n = len(h0)
    # Bounds and constraints for LP
    bounds = [(0, 1) for _ in range(n)]
    # Equality constraint: sum = n/2
    A_eq = np.array([np.ones(n)])
    b_eq = np.array([n/2])
    # We track t in the objective min t
    c = np.zeros(n + 1)
    c[-1] = 1  # t is in the last position
    # Inequality constraint: A * h + t * 1 <= b
    A_ub = np.column_stack([A, np.ones(A.shape[0])])  # Add the column for t
    b_ub = b
    # Solve linear program
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        h_new = res.x[:-1]  # Extract h part
        t_new = res.x[-1]   # Extract t
        return h_new, t_new
    else:
        # fallback to original h
        return h0.copy(), evaluate_c5(h0)

def generate_seeds(n, target_sum):
    """Generate different starting points."""
    seeds = []
    # 1. Uniform
    h_uniform = np.ones(n) * 0.5
    h_uniform = project_to_feasible(h_uniform, target_sum)
    seeds.append(h_uniform)
    
    # 2. Random - Dirichlet(1)
    h_dirichlet = dirichlet.rvs(np.ones(n), size=1)[0]
    h_dirichlet = h_dirichlet * target_sum / np.sum(h_dirichlet)
    h_dirichlet = project_to_feasible(h_dirichlet, target_sum)
    seeds.append(h_dirichlet)
    
    # 3. Binary random
    h_binary = np.zeros(n)
    h_binary[:int(target_sum)] = 1
    np.random.shuffle(h_binary)
    h_binary = h_binary * target_sum / np.sum(h_binary)
    h_binary = project_to_feasible(h_binary, target_sum)
    seeds.append(h_binary)
    
    # 4. Interpolation of existing initial_h_values if available
    if 'initial_h_values' in globals():
        h_orig = initial_h_values
        x_orig = np.linspace(0, 2, len(h_orig))
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_orig, h_orig)
        h_interp = h_interp * target_sum / np.sum(h_interp)
        h_interp = project_to_feasible(h_interp, target_sum)
        seeds.append(h_interp)
    
    # 5. Sobol-like quasi-random, starts from [0.5]*n and slightly perturbed
    np.random.seed(42)
    h_sobol = 0.5 + 0.1 * np.random.rand(n) - 0.05
    h_sobol = h_sobol * target_sum / np.sum(h_sobol)
    h_sobol = project_to_feasible(h_sobol, target_sum)
    seeds.append(h_sobol)
    
    return seeds

def projected_gradient_descent(h0, max_iter=200):
    """Refine h0 via gradient descent on true objective."""
    h = h0.copy()
    dx = 2.0 / len(h)
    for i in range(max_iter):
        # Compute worst shift k
        h_fft = fft(h, 2 * len(h))
        ih_fft = fft(1 - h, 2 * len(h))
        corr = ifft(h_fft * np.conj(ih_fft)).real
        # Exclude shift 0
        corr_vals = corr[1: len(h)]
        k_star = np.argmin(corr_vals) + 1
        # Compute sub-gradient
        grad = np.zeros_like(h)
        for j in range(len(h)):
            left_idx = (j + k_star) % len(h)
            right_idx = (j - k_star) % len(h)
            grad[j] = dx * ((1 - h[left_idx]) - h[right_idx])
        # Take small step
        step = 1e-3
        h_new = h - step * grad
        h_new = project_to_feasible(h_new, len(h) / 2)
        h = h_new
    return h

def simulated_annealing(h0, max_iter=5000):
    """Polish h0 using simulated annealing."""
    h = h0.copy()
    dx = 2.0 / len(h)
    c5 = evaluate_c5(h)
    T = 1.0
    for i in range(max_iter):
        # Temperature decay
        if i % 100 == 0:
            T *= 0.99
        # Random swap
        i1, i2 = np.random.randint(0, len(h), 2)
        if h[i1] <= 0 or h[i2] >= 1:
            continue
        delta = np.random.uniform(0, min(h[i1], 1 - h[i2], 0.01))
        
        h_test = h.copy()
        h_test[i1] -= delta
        h_test[i2] += delta
        h_test = project_to_feasible(h_test, len(h) / 2)
        
        c5_test = evaluate_c5(h_test)
        
        if c5_test < c5:
            h = h_test
            c5 = c5_test
        else:
            # Accept with probability
            if T > 0:
                prob = np.exp(-(c5_test - c5) / T)
                if np.random.rand() < prob:
                    h = h_test
                    c5 = c5_test
    return h

def upscale_and_refine(h0, target_n):
    """Increase resolution and refine."""
    if len(h0) == target_n:
        return h0
    x = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, target_n)
    h_up = np.interp(x_new, x, h0)
    h_up = h_up * target_n / (2 * np.sum(h_up))
    h_up = project_to_feasible(h_up, target_n / 2)
    # Apply a few CCP iterations
    for _ in range(5):
        A, b = build_lp_data(h_up)
        h_up, _ = solve_ccp_step(h_up, A, b)
        h_up = project_to_feasible(h_up, target_n / 2)
    # Final gradient step
    h_up = projected_gradient_descent(h_up, 20)
    return h_up

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 400
    dx = 2.0 / n
    target_sum = n / 2
    
    # Generate starting points
    seeds = generate_seeds(n, target_sum)
    best_c5 = float('inf')
    best_h = None

    # Run CCP method with multiple seeds
    # Use about 60% of time for this part
    time_limit = start_time + 0.6 * budget_s
    for i, h0 in enumerate(seeds):
        if time.time() > time_limit:
            break
        h = h0.copy()
        max_iter = 30
        for _ in range(max_iter):
            if time.time() > time_limit:
                break
            A, b = build_lp_data(h)
            h_new, t = solve_ccp_step(h, A, b)
            if evaluate_c5(h_new) < evaluate_c5(h):
                h = h_new
            else:
                break  # Convergence detection
        if evaluate_c5(h) < best_c5:
            best_c5 = evaluate_c5(h)
            best_h = h.copy()
    
    if best_h is None:
        best_h = np.ones(n) * 0.5
        best_c5 = evaluate_c5(best_h)
    
    # Refinement with gradient descent
    if time.time() < start_time + 0.7 * budget_s:
        best_h = projected_gradient_descent(best_h, 200)
        best_c5 = evaluate_c5(best_h)

    # Simulated Annealing for polishing
    final_time = start_time + 0.95 * budget_s
    if time.time() < final_time:
        best_h = simulated_annealing(best_h, 2000)
        best_c5 = evaluate_c5(best_h)

    # Optional upscaling
    if time.time() < start_time + budget_s - 30:
        best_h = upscale_and_refine(best_h, 800)
        best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, n)
