import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_box_sum(y, total):
    """Project vector y onto the set {x : 0 <= x[i] <= 1, sum x = total}."""
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if abs(s - total) < 1e-10:
        return y
    
    # Bisection on lambda
    lam_lo, lam_hi = -1.0, 2.0
    for _ in range(40):
        lam = (lam_lo + lam_hi) / 2
        z = np.clip(y - lam, 0, 1)
        if np.sum(z) > total:
            lam_lo = lam
        else:
            lam_hi = lam
    lam = (lam_lo + lam_hi) / 2
    return np.clip(y - lam, 0, 1)

def compute_autocorrelation(h):
    """Compute cyclic autocorrelation of h."""
    H = fft(h)
    P = np.abs(H)**2
    gamma = ifft(P).real
    return gamma

def build_target_autocorrelation(gamma, alpha):
    """Build target autocorrelation with soft averaging."""
    n = len(gamma)
    avg = (np.sum(gamma) - gamma[0]) / (n - 1)
    gamma_target = gamma.copy()
    gamma_target[1:] = (1 - alpha) * gamma[1:] + alpha * avg
    return gamma_target

def evaluate_c5(h):
    """Evaluate C5 for h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def generate_legendre_seed(n):
    """Generate Legendre symbol seed of length n."""
    # Use prime p = 257
    p = 257
    # Generate Legendre symbols for k = 1 to p-1 and map to {0,1}
    legendre = np.array([ ((k % p) ** 2) % p for k in range(1, p)] )
    legendre = (legendre != 0).astype(int)
    if len(legendre) > n:
        legendre = legendre[:n]
    elif len(legendre) < n:
        legendre = np.pad(legendre, (0, n - len(legendre)), 'constant')
    return legendre

def generate_msequence_seed(n):
    """Generate maximal length sequence (m-sequence) of length n."""
    # LFSR parameters for m-sequence of length 255 (x^8 + x^4 + x^3 + x^2 + 1)
    # This generates a primitive polynomial sequence
    L = 8
    taps = [4, 3, 2, 1]
    state = [1] + [0] * (L-1)
    seq = []
    for i in range(n-1):
        output = state[0]
        seq.append(output)
        new_bit = 0
        for t in taps:
            new_bit ^= state[t-1]
        state = state[1:] + [new_bit]
    # Append one zero to make length n
    seq = np.array(seq + [0])
    return seq

def generate_random_balanced_seed(n, num_samples=50):
    """Generate random balanced seed vectors."""
    seeds = []
    for _ in range(num_samples):
        h = np.zeros(n)
        indices = np.random.choice(n, n//2, replace=False)
        h[indices] = 1
        seeds.append(h)
    return seeds

def binary_hill_climb(h0, max_swaps=3*n):
    """Improve a binary h using hill-climbing swaps."""
    n = len(h0)
    h = h0.astype(float)
    gamma = compute_autocorrelation(h)
    current_c5 = evaluate_c5(h)
    
    # Precompute gamma[i+k] and gamma[i-k] for all i,k
    gamma_shifted = [np.roll(gamma, k) for k in range(1, n)]
    
    swaps = 0
    improvements = 0
    
    start_time = time.time()
    
    while swaps < max_swaps and time.time() - start_time < 0.4 * 600:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h[i] != 1 or h[j] != 0:
            continue
        # Compute Δγ_k for all k
        # Updated gamma at k = i + k:
        # old: h[i]*h[i+k] + h[j]*h[j+k]
        # new: h[j]*h[i+k] + h[i]*h[j+k]
        delta_gamma = np.zeros(n)
        for k in range(n):
            # Simulate the shift effect
            k1 = (k - i) % n
            k2 = (k - j) % n
            delta_gamma[k] = -(h[i]*gamma[k1] + h[i]*gamma[k1]) \
                           + (h[j]*gamma[k2] + h[i]*gamma[k2])
        # Adjust for boundary cases with the length n wrap
        new_gamma = gamma + delta_gamma
        # Compute new C5 
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_c5 = max(1 - (2/n) * new_gamma[k] for k in range(len(new_gamma)) )
        if new_c5 < current_c5:
            h = h_new
            gamma = new_gamma
            current_c5 = new_c5
            improvements += 1
        swaps += 1
    
    return h, current_c5

def spectral_alternating_projection(h0, alpha_start=0.7, max_iter=100, target_c5=0.3807):
    """Perform SAP to improve h."""
    h = h0.astype(float)
    n = len(h)
    alpha = alpha_start
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iter in range(max_iter):
        H = fft(h)
        P = np.abs(H)**2
        gamma = ifft(P).real
        
        gamma_target = build_target_autocorrelation(gamma, alpha)
        P_target = fft(gamma_target)
        P_target = np.maximum(P_target, 0)
        A_target = np.sqrt(P_target)
        H_new = A_target * np.exp(1j * np.angle(H))
        h_new = ifft(H_new).real
        
        h_new = project_box_sum(h_new, n/2)
        
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        
        if c5 <= target_c5:
            break
            
        h = h_new
        alpha = min(alpha + 0.02, 0.99)
        
    return best_h, best_c5

def projected_gradient_polish(h0, max_iter=300):
    """Perform gradient descent polish on h."""
    h = h0.copy()
    n = len(h)
    c5 = evaluate_c5(h)
    beta = 5.0
    eta = 1.0
    
    for i in range(max_iter):
        # Compute C_k values
        corr = correlate(h, 1 - h, mode='full')
        C = 1.0 - (2.0 / n) * corr
        
        # Softmax weights
        w = np.exp(beta * C)
        w = w / np.sum(w)
        
        # Compute gradient
        h_fft = fft(h)
        w_fft = fft(w)
        grad_fft = w_fft * h_fft + np.conj(w_fft) * h_fft
        
        grad = ifft(grad_fft).real
        grad = - (2.0 / n) * grad
        
        # Backtracking line search
        h_new = h - eta * grad
        h_new = project_box_sum(h_new, n/2)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            eta *= 0.5
            if eta < 1e-6:
                break
        
        # Reduce beta
        if i % 20 == 0:
            beta *= 1.2
            
    return h, c5
