```python
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.signal import correlate

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2 with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes C5 from a step function h."""
    dx = 2.0 / len(h)
    # Full correlation: we zero-pad to length 2n for linear correlation
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # shift and real
    corr = np.real(corr)
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 256
    dx = 2.0 / n
    
    # Phase 1: Feasible SDP Bisection on t
    t_low, t_high = 0.0, 1.0
    eps = 2e-4
    best_feasible_r = None
    time_limit = 0.35 * budget_s
    
    while time.time() - start_time < time_limit and (t_high - t_low) > eps:
        t_mid = (t_low + t_high) / 2
        
        # Define SDP feasibility problem
        r = cp.Variable(n)
        
        # Constraints
        constraints = [
            r[0] == 1,
            r[1:] >= t_mid,
            r <= 1
        ]
        
        # Toeplitz PSD matrix: T(r) must be positive semi-definite
        T = toeplitz(r)
        constraints.append(T >> 0)
        
        # Dummy objective (feastibility check)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        # Solve
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status in ['optimal', 'optimal_inaccurate']:
            # Feasible
            t_low = t_mid
            best_feasible_r = r.value
        else:
            t_high = t_mid
            
        if time.time() - start_time > time_limit:
            break
                
    if best_feasible_r is None:
        # Fallback to uniform
        r = np.ones(n)
        r[0] = 1
    else:
        r = best_feasible_r.copy()
    
    # Phase 2: Spectral Factorization and Projection
    # Get power spectrum
    S = np.fft.fft(r)
    S = np.maximum(S, 0)
    H = np.sqrt(S)
    g = np.real(np.fft.ifft(H))
    
    # Project via water-filling
    h = project_to_feasible(g, n)
    
    # Phase 3: Worst-shift gradient descent
    max_iter = 250
    alpha0 = 0.5 * dx
    current_c5 = evaluate_c5(h)
    
    for t in range(max_iter):
        if time.time() - start_time > 0.7 * budget_s:
            break
        # Compute overlap values
        h_fft = fft(h, 2 * n)
        ih_fft = fft(1 - h, 2 * n)
        corr = ifft(h_fft * np.conj(ih_fft))
        corr = corr.real
        Ck = corr * dx
        k_star = np.argmax(Ck[1:]) + 1
        
        # Gradient
        grad = np.zeros(n)
        for i in range(n):
            left_idx = (i + k_star) % n
            right_idx = (i - k_star) % n
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Update with step size decay
        alpha = alpha0 / np.sqrt(t + 1)
        h_guess = h - alpha * grad
        h = project_to_feasible(h_guess, n)
        
        new_c5 = evaluate_c5(h)
        if abs(new_c5 - current_c5) < 1e-6:
            break
        current_c5 = new_c5
    
    # Phase 4: Stochastic Pair Transfer
    num_moves = 2000
    for _ in range(num_moves):
        if time.time() - start_time > 0.95 * budget_s:
            break
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h[i] <= 0 or h[j] >= 1:
            continue
        delta = min(0.01, h[i], 1 - h[j])
        if delta <= 0:
            continue
        
        h_test = h.copy()
        h_test[i] -= delta
        h_test[j] += delta
        h_test = project_to_feasible(h_test, n)
        c5_test = evaluate_c5(h_test)
        if c5_test < current_c5:
            h = h_test
            current_c5 = c5_test
    
    # Phase 5: Optionally upsample and refine
    if time.time() - start_time < 0.99 * budget_s:
        n_up = 2 * n
        dx_up = 2.0 / n_up
        x_orig = np.linspace(0, 2, n)
        x_up = np.linspace(0, 2, n_up)
        h_up = np.interp(x_up, x_orig, h)
        h_up = project_to_feasible(h_up, n_up)
        
        # Short local refinement on higher grid
        for _ in range(50):
            if time.time() - start_time > 0.995 * budget_s:
                break
            corr = ifft(fft(h_up, 2 * n_up) * np.conj(fft(1-h_up, 2 * n_up)))
            corr = corr.real
            Ck = corr * dx_up
            k_star = np.argmax(Ck[1:]) + 1
            
            grad = np.zeros(n_up)
            for i in range(n_up):
                left_idx = (i + k_star) % n_up
                right_idx = (i - k_star) % n_up
                grad[i] = dx_up * ((1 - h_up[left_idx]) - h_up[right_idx])
            
            h_up = h_up - 0.1 * dx_up * grad
            h_up = project_to_feasible(h_up, n_up)
        
        h = h_up
        current_c5 = evaluate_c5(h)
    
    return (h, current_c5, len(h))
```
