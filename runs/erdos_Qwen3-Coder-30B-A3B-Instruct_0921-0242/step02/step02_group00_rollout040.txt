import numpy as np
import time
import math

def project_box_sum(y, total):
    """Project y onto {x : 0 ≤ x[i] ≤ 1, Σ x = total}."""
    # Use bisection to find lambda
    low, high = -1.0, 2.0
    for _ in range(40):
        lam = (low + high) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = np.sum(x)
        if s > total:
            low = lam
        else:
            high = lam
    lam = (low + high) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    return x

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def softmax_weights(C, beta):
    """Compute softmax weights given C and beta."""
    C_shifted = C - np.max(C)
    exp_C = np.exp(beta * C_shifted)
    return exp_C / np.sum(exp_C)

def gradient_from_weights(h, w):
    """Compute gradient from weight vector w."""
    n = len(h)
    # Convolve w with h (forward and reverse)
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    # Take the middle part of length n
    grad = -(2.0 / n) * (conv1[n-1:n-1+n] + conv2[n-1:n-1+n])
    return grad

    # Optimisation parameters
    beta_start = 2.0
    beta_max = 200.0
    beta_factor = 1.6
    max_iter_beta = 2000
    step_init = 1.0

    iteration = 0
    
    # Time management

    while time_remaining() > 1.0 and beta <= beta_max:
        for it in range(max_iter_beta):

                break
                
            iteration += 1
            
            # Compute autocorrelation
            gamma = np.correlate(h_current, h_current, mode='full')
            C = 1.0 - (2.0 / n) * gamma
            
            # Compute softmax weights
            w = softmax_weights(C, beta)
            
            # Compute gradient
            grad = gradient_from_weights(h_current, w)
            
            # Backtracking line search
            alpha = step_init
            h_trial = None

            if 'last_c5' in locals() and c5_current < last_c5:
