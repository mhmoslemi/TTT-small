import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_feasible(h, n):
    """Project h to feasible set: sum(h) = n/2, h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-12:
        return h
    # Water filling
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    # Find index to cut
    cut_idx = np.searchsorted(cumsum, n/2, side='right')
    if cut_idx == 0:
        return h * (n/2) / current_sum
    # Water level
    total = cumsum[cut_idx - 1]
    level = (n/2 - total) / (n - cut_idx)
    level = max(0, min(1, level))
    h = np.where(np.arange(n) < cut_idx, h_sorted[:cut_idx], level)
    # permute back to original order
    idx_sorted = np.argsort(h_sorted)[::-1]
    h_new = np.empty_like(h)
    h_new[idx_sorted] = h
    return h_new

def evaluate_c5(h):
    """Return C5 = max_k correlate(h, 1-h)[k] * dx."""
    dx = 2.0 / len(h)
    # Full correlation via FFT
    c = np.real(ifft(fft(h) * fft(1 - h).conj()))
    # mode='full' => c has length 2*n-1
    # k ranges from -(n-1) to (n-1)
    # max is at indices (n-1) + k where k in [-n+1, n-1]
    # max over all shifted correlations
    # take max of correlation at all lags
    max_corr = np.max(c) * dx
    return max_corr

def gradient_at_worst(h):
    """Return subgradient of C5 at h."""
    n = len(h)
    dx = 2.0 / n
    # compute correlation so we can find k* = argmax c
    c = np.real(ifft(fft(h) * fft(1 - h).conj()))
    k_star = np.argmax(c)
    grad = np.zeros_like(h)
    for i in range(n):
        # cyclic indices
        left = (i - k_star) % n
        right = (i + k_star) % n
        grad[i] = dx * (h[right] - h[left])
    return grad

def line_search(h, grad, eta0=0.5):
    """Backtracking line search to find step size."""
    dx = 2.0 / len(h)
    eta = eta0
    max_iter = 30
    for _ in range(max_iter):
        h_new = h - eta * grad
        h_new = project_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        if c5_new < c5_old:
            return h_new, c5_new
        eta /= 2.0
    return h, evaluate_c5(h)

def mutate(v, mode='gaussian', params=None):
    """Generate new vector from v via mutation strategies."""
    n = len(v)
    v_new = v.copy()
    if mode == 'gaussian':
        # perturb 5-10 elements
        n_pert = np.random.randint(5, 11)
        indices = np.random.choice(n, size=n_pert, replace=False)
        sigma = params.get('sigma', 0.2)
        v_new[indices] += np.random.normal(0, sigma, n_pert)
        v_new = project_feasible(v_new, n)
    elif mode == 'swap':
        # swap small amount between indices
        i = np.random.randint(n)
        j = np.random.randint(n)
        delta = params.get('delta', 0.01)
        if v[i] > 0.01 and v[j] < 0.99:
            d = min(delta, v[i], 1 - v[j])
            v_new[i] -= d
            v_new[j] += d
            v_new = project_feasible(v_new, n)
    elif mode == 'dct':
        # modify dct coefficient
        dct_h = np.fft.fftshift(np.fft.fft(v))
        k = np.random.randint(1, len(dct_h) // 3)
        step = params.get('step', 0.01)
        dct_h[k] += np.random.normal(0, step)
        v_new = np.real(np.fft.ifft(np.fft.fftshift(dct_h)))
        v_new = project_feasible(v_new, n)
    else:
        raise ValueError(f"Unknown mutation mode: {mode}")
    return v_new

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    t_start = time.time()
    safety_margin = 0.5
    assert budget_s > safety_margin

    # phase 1: generate initial pool
    M = 2000
    n_points = 250
    dx = 2.0 / n_points
    pool = []
    for _ in range(M):
        u = np.random.rand(n_points)
        h = project_feasible(u, n_points)
        pool.append(h)
    # Evaluate all
    vals = [evaluate_c5(h) for h in pool]
    pool = [h for h, val in zip(pool, vals) if not np.isnan(val)]
    vals = [val for val in vals if not np.isnan(val)]
    sorted_indices = np.argsort(vals)
    pool = [pool[i] for i in sorted_indices[:30]]
    vals = [vals[i] for i in sorted_indices[:30]]
    best_c5 = vals[0]
    best_h = pool[0]

    # Peak performance time (200s)
    # PSO-style global search
    t_elapsed = time.time() - t_start
    while t_elapsed < 0.25 * budget_s:
        if time.time() - t_start > 0.25 * budget_s - safety_margin:
            break
        # pick parent uniformly from low vals
        idx_parent = np.random.randint(0, min(len(pool), 30))
        parent = pool[idx_parent]
        mode = np.random.choice(['gaussian', 'swap', 'dct'])
        h_new = mutate(parent, mode=mode, params={'sigma': 0.2, 'step': 0.02})
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # replace worst if improves
        worst_idx = np.argmax(vals)
        if c5_new < vals[worst_idx]:
            pool[worst_idx] = h_new
            vals[worst_idx] = c5_new
        t_elapsed = time.time() - t_start

    # Phase 2: refine all pool members
    t_elapsed = time.time() - t_start
    for i in range(min(5, len(pool))):
        if time.time() - t_start > 0.7 * budget_s - safety_margin:
            break
        h = pool[i]
        c5 = vals[i]
        # minimize the worst correlation entry
        for _ in range(2000):
            if time.time() - t_start > 0.7 * budget_s - safety_margin:
                break
            grad = gradient_at_worst(h)
            h_new, c5_new = line_search(h, grad, eta0=0.5)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                break

    # Phase 3: fine tuning with swaps
    h = best_h
    for _ in range(10000):
        if time.time() - t_start > budget_s - safety_margin:
            break
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        if h[i] > 0.01 and h[j] < 0.99:
            d = np.random.uniform(1e-4, 0.01)
            d = min(d, h[i], 1 - h[j])
            h_trial = h.copy()
            h_trial[i] -= d
            h_trial[j] += d
            h_trial = project_feasible(h_trial, n_points)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
            elif c5_trial < evaluate_c5(h):
                h = h_trial
    # Final check on current state
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
