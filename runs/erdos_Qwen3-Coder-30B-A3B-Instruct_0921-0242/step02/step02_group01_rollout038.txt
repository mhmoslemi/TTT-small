import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Computes the exact C5 constant using np.correlate."""
    dx = 2.0 / len(h)
    # Full correlation as in the original problem
    corr = correlate(h, 1 - h, mode="full")
    # We ignore the first value (correlation with shift 0),
    # take the max of the relevant (valid) correlations
    max_corr = np.max(corr[len(h)-1:])
    return max_corr * dx

def smooth_max_surrogate(h, p=256):
    """Compute the smooth max approximation S_p(h) = (sum(C(k,h)^p))^(1/p)."""
    dx = 2.0 / len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    overlap = corr.real[:len(h)] * dx
    C_vec = 1 - overlap  # overlaps, C(k,h)
    
    # Since we want max of C(k,h), we are computing smooth max
    # S_p = (sum(C^p))^(1/p)
    # we use a small epsilon to avoid numerical issues with 0^p
    eps = 1e-12
    C_vec = np.maximum(C_vec, eps)
    S_p = np.power(np.sum(np.power(C_vec, p)), 1/p)
    return S_p

def compute_gradient_smooth_surrogate(h, p=256):
    """Return the gradient of S_p with respect to each h_i."""
    dx = 2.0 / len(h)
    n = len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    overlap = corr.real[:len(h)] * dx
    C_vec = 1 - overlap  # overlaps, C(k,h)
    
    # Prevent log from being undefined
    eps = 1e-12
    C_vec = np.maximum(C_vec, eps)
    if p < 1000:
        # Optimized evaluation with early exit if needed
        S_p = np.power(np.sum(np.power(C_vec, p)), 1/p)
        if S_p < 1e-15:
            return np.zeros_like(h)
            
        # w_k = C(k)^{p-1}
        w = np.power(C_vec, p-1)
        
        # Now compute the total gradient for each i
        grad = np.zeros_like(h)
        for k in range(n):
            # we compute ∂C(k)/∂h_i = dx * [ (1 - h_{i+k}) - h_{i-k} ]
            # where indices are treated circularly
            if k == 0:
                pterm = 0  # no shift, it's the auto-correlation
            else:
                # We compute h_{i+k}, h_{i-k}
                term1 = 0  # default is 0 if boundary not touched
                term2 = 0
                i_k = (np.arange(n) + k) % n
                i_km = (np.arange(n) - k) % n
                term1 = np.sum(h[i_k] * (1 - h[i_k]))  # h_{i+k} term
                term2 = np.sum(h[i_km] * h[i_km])    # h_{i-k} term
                term1 = np.sum((1 - h[i_k]) * (1 - h[i_k]))  # correction for -h_{i-k}
                term2 = np.sum(h[i_km] * h[i_km])
                # Actually, one needs to be more careful. Let's recompute:
                # Each C_k = dx * (sum h_i - sum h_i * h_{i+k})
                # ∂C_k/∂h_i = dx * ( (1 - h_{i+k}) - h_{i-k} ) where indices are circ shifted
                grad1 = np.zeros_like(h)
                grad2 = np.zeros_like(h)
                # now compute all terms: ∂C(k)/∂h_i for each i
                for i in range(n):
                    # (∂/∂h_i) \int h(x) (1-h(x+k)) dx
                    left_idx = (i + k) % n
                    right_idx = (i - k) % n
                    grad1[i] = (1 - h[left_idx])
                    grad2[i] = h[right_idx]
                # Matrix
                grad_i = dx * (grad1 - grad2)
                # Add to total gradient with weight w_k
                grad += w[k] * grad_i
        
        # Final gradient
        if S_p < 1e-12:
            return np.zeros_like(h)
        grad /= np.power(S_p, p - 1)
        return grad
    else:
        # p is large enough, can use log-mean approximation
        # S_p ≈ max(C_k)
        # ∂S_p/∂h_i ≈ (1/S_p) * [sum_k w_k * ∂C_k/∂h_i]
        # But in this limit, we often need the maximum exact gradient,
        # we use the full formula to be precise
        S_p = np.power(np.sum(np.power(C_vec, p)), 1/p)
        # less common case that requires the full gradient, repeated below
        w = np.power(C_vec, p-1)
        # For each i, compute ∂C(k)/∂h_i
        grad = np.zeros_like(h)
        for k in range(n):
            # grad = sum_k w_k * (1-h_{i+k} - h_{i-k})
            # But only contribute to indices in loop.
            # Let’s effectively create the sum here efficiently:
            # walk through all shifts k and build up gradient for each component
            if k == 0:
                continue
            # Left shift: h[i + k] for all i
            left_term = np.roll(h, -k)
            # Right shift: h[i - k] for all i
            right_term = np.roll(h, k)
            grad += w[k] * dx * (1 - left_term - right_term)
        # Divide by S_p^(p-1) (if S_p is not too small)
        if S_p > 1e-10:
            grad /= np.power(S_p, p - 1)
        return grad

def add_sum_penalty(h, lambda_sum=1e4):
    """Add a squared penalty to satisfy Σ h = n/2."""
    current_sum = np.sum(h)
    diff = current_sum - len(h) / 2
    penalty = 0.5 * lambda_sum * diff * diff
    return penalty

def grad_sum_penalty(h, lambda_sum=1e4):
    """Gradient of the squared sum penalty."""
    current_sum = np.sum(h)
    diff = current_sum - len(h) / 2
    return lambda_sum * diff * np.ones_like(h)

def wrapped_obj_grad(h, p=256, lambda_sum=1e4):
    """Wrapper that returns the value and gradient for scipy.optimize."""
    # Ensure h is valid first (to avoid contradictions or numerical issues with gradient)
    h = np.clip(h, 0, 1)
    n = len(h)
    val = smooth_max_surrogate(h, p)
    val += add_sum_penalty(h, lambda_sum)
    grad = compute_gradient_smooth_surrogate(h, p)
    grad += grad_sum_penalty(h, lambda_sum)
    # Ensure all values are in [0,1], clipping again
    h = np.clip(h, 0, 1)
    grad = np.clip(grad, -10000, 10000)  # prevent overly large gradients
    return val, grad

def generate_seeds(n, ns=20):
    """Generate a diverse set of initial h vectors."""
    seeds = []
    # 1. Uniform vector
    seeds.append(np.ones(n) * 0.5)
    # 2. Random Dirichlet
    seeds.append(np.random.dirichlet(np.ones(n)) * (n/2))
    # 3. Interpolate initial_h_values if available (shrink if needed)
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) == n:
            seeds.append(orig_h.copy())
        else:
            # interpolate
            x_orig = np.linspace(0, 2, len(orig_h))
            x_new = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_orig, orig_h)
            seeds.append(h_interp)
    # 4. Some random binary patterns with smoothing
    for _ in range(ns - 3):
        # pick exactly n/2 ones
        h = np.zeros(n)
        idxs = np.random.permutation(n)[:n//2]
        h[idxs] = 1.0
        h = np.convolve(h, np.exp(-np.arange(10) * 0.3), mode='same')
        h = h[:n]
        h = np.clip(h, 0, 1)
        seeds.append(h)
    candidates = []
    for h in seeds:
        h = project_to_feasible(h, n)
        candidates.append(h)
    return candidates

def run(seed=42, budget_s=600, **kwargs):
    """
    Run the full optimization pipeline to find a good h to minimize C5.
    Returns (h, C5, n_points)
    """
    np.random.seed(seed)
    start_time = time.time()
    
    n = 350  # default fine enough
    # Timeout check: must not exceed at least 10% margin to avoid crash
    remaining = budget_s * 0.9
    
    # Generate seeds
    if time.time() - start_time > remaining * 0.1:
        return np.ones(n) * 0.5, float('inf'), n
    seeds = generate_seeds(n, ns=25)
   best_h = None
    best_c5 = float('inf')
    
    # Homotopy parameters
    p_vals = [8, 16, 32, 64, 128, 256, 512]
    
    # Decision tree to use just a subset if needed to save time
    homotopy_steps = len(p_vals)
    
    # Start with no optimization, only the first gaussian pass:
    t_remaining = budget_s * 0.95 - (time.time() - start_time)
    
    # we refine each seed through p values, keep best h, c5
    for i, p in enumerate(p_vals[:homotopy_steps]):
        if time.time() - start_time > remaining:
            break
        t_run = max(0.1, t_remaining / max(1, homotopy_steps - i))
        if t_run < 0.01:
            break
        for idx, seed in enumerate(seeds):
            if time.time() - start_time > remaining:
                break
            try:
                # Optimization with scipy (L-BFGS-B is good for box bounds)
                res = minimize(
                    fun=lambda h: smooth_max_surrogate(h, p),
                    x0=seed,
                    method='L-BFGS-B',
                    jac=lambda h: compute_gradient_smooth_surrogate(h, p),
                    bounds=[(0, 1)] * n,
                    options={
                        'maxiter': min(150, int(t_run * 50)),
                        'ftol': 1e-8,
                        'gtol': 1e-7,
                    }
                )
                # If it fails, skip this seed
                if res.success:
                    h_sol = project_to_feasible(res.x, n)
                    c5 = evaluate_c5_exact(h_sol)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_sol.copy()
                else:
                    # Try again with just penalty method and gradient
                    h_sol = project_to_feasible(seed, n)
                    for _ in range(20):
                        grad = compute_gradient_smooth_surrogate(h_sol, p)
                        h_sol = h_sol - 0.01 * grad * 2.0 / n
                        h_sol = project_to_feasible(h_sol, n)
                        if np.allclose(h_sol, seed, atol=1e-3):
                            break
                    c5 = evaluate_c5_exact(h_sol)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_sol.copy()
            except Exception:
                continue  # skip if something bad happens
        # Early stopping if no improvement was seen recently
        if best_c5 < 0.3808 and i > 2:
            break
    
    # If nothing found, return default
    if best_h is None:
        best_h = np.ones(n) * 0.5
        best_c5 = evaluate_c5_exact(best_h)
    
    # Refine with worst shift ascent:
    h_refined = best_h.copy()
    c5_refined = evaluate_c5_exact(h_refined)
    t_remaining = budget_s - (time.time() - start_time)
    
    if t_remaining > 2.0:
        for iter_worst in range(100):
            if time.time() - start_time > budget_s * 0.98:
                break
            # Find worst shift
            corr = correlate(h_refined, 1-h_refined, mode='full')
            max_shift = np.argmax(corr[len(h_refined)-1:])
            max_shift = max_shift - len(h_refined)+1  # convert to shift index
            if max_shift < 0:
                max_shift += len(h_refined)
            
            # Compute exact gradient for this shift
            grad = np.zeros_like(h_refined)
            dx = 2.0 / len(h_refined)
            k = max_shift
            left_idx = (np.arange(n) + k) % n
            right_idx = (np.arange(n) - k) % n
            grad = dx * ((1 - h_refined[left_idx]) - h_refined[right_idx])
            
            # Backtracking line search
            alpha = 0.1 * dx
            for ls_it in range(20):
                h_trial = h_refined - alpha * grad
                h_trial = project_to_feasible(h_trial, n)
                c5_trial = evaluate_c5_exact(h_trial)
                if c5_trial < c5_refined:
                    h_refined = h_trial
                    c5_refined = c5_trial
                    break
                else:
                    alpha *= 0.5
            # convergence check
            if abs(c5_refined - evaluate_c5_exact(h_refined)) < 1e-8:
                break

    print(f"Final C5 estimate found: {c5_refined:.6f} with n={n}")
    return (h_refined, c5_refined, n)
