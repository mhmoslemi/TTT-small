import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def project_to_sum_constraint(h, target_sum):
    """Project h to have sum = target_sum with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    
    return h

def compute_correlation(h):
    """Compute all correlations using FFT."""
    n = len(h)
    dx = 2.0 / n
    # Use real FFT for efficiency
    h_rfft = rfft(h)
    ih_rfft = rfft(1 - h)
    corr_fft = h_rfft * np.conj(ih_rfft)
    corr = irfft(corr_fft)
    # Return the first n values (correlations with shifts 0 to n-1)
    corr_vals = np.real_if_close(corr[:n]) * dx
    # Ignore shift 0 since it's always 0 (1-h(x) - h(x) = 0)  
    return corr_vals[1:]

def find_worst_shift(h):
    """Find shift that maximizes correlation."""
    corr = compute_correlation(h)
    max_k = np.argmax(corr)
    return max_k

def gradient_worst_shift(h, worst_shift):
    """Compute gradient based on the worst shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    
    # For shift k, gradient of h_i * (1 - h_{i+k}) is (1 - h_{i+k}) - h_{i-k}
    for i in range(n):
        left_idx = (i + worst_shift) % n
        right_idx = (i - worst_shift) % n
        grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
    
    return grad

def gradient_descent_on_worst_shift(h, max_iter=200, alpha0=0.01):
    """Perform gradient descent on the worst shift."""
    h = h.copy()
    for t in range(max_iter):
        if t == 0:
            alpha = alpha0
        else:
            alpha = alpha0 / np.sqrt(t)
            
        worst_shift = find_worst_shift(h)
        grad = gradient_worst_shift(h, worst_shift)
        h_new = h - alpha * grad
        h_new = project_to_sum_constraint(h_new, len(h)/2)
        
        # Check if improvement
        new_c5 = np.max(compute_correlation(h_new)[1:])
        old_c5 = np.max(compute_correlation(h)[1:])
        
        if new_c5 < old_c5:
            h = h_new
        else:
            break
    return h

def evaluate_c5(h):
    """Evaluates the C5 constant for a given h."""
    corr = compute_correlation(h)
    return np.max(corr[1:])  # skip shift 0

def de_step(pop, fitness, F=0.6, CR=0.9):
    """DE operation: mutation and crossover."""
    n, P = len(pop), len(pop)
    new_pop = []
    
    for i in range(P):
        # Select three distinct indices
        idxs = np.random.choice(P, 3, replace=False)
        a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
        # Mutation
        donor = a + F * (b - c)
        donor = np.clip(donor, 0, 1)
        # Enforce sum constraint
        donor = project_to_sum_constraint(donor, len(donor)/2)
        # Crossover
        trial = donor.copy()
        jrand = np.random.randint(len(donor))
        for j in range(len(donor)):
            if np.random.random() < CR or j == jrand:
                trial[j] = donor[j]
            else:
                trial[j] = pop[i][j]
                
        new_pop.append(trial)
    return new_pop

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    n_points = 400
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Initialize population
    population = []
    
    # Add the initial solution
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) > n_points:
            h_start = orig_h[:n_points]
        else:
            x_orig = np.linspace(0, 2, len(orig_h))
            x_new = np.linspace(0, 2, n_points)
            h_start = np.interp(x_new, x_orig, orig_h)
    else:
        h_start = np.ones(n_points) * 0.5
    
    h_start = project_to_sum_constraint(h_start, target_sum)
    population.append(h_start)
    
    # Add random individuals
    for _ in range(39):
        h = np.random.rand(n_points)
        h = project_to_sum_constraint(h, target_sum)
        population.append(h)
    
    # DE parameters
    P = len(population)
    F = 0.6
    CR = 0.9
    max_generations = 5000  # Allow up to 5000 evals
    
    best_h = None
    best_c5 = float('inf')
    
    # Run DE
    generation = 0
    while generation < max_generations and (time.time() - start_time < 0.75 * budget_s):
        # Evaluate each individual
        fitness = []
        for i in range(P):
            if time.time() - start_time > 0.75 * budget_s:
                break
            h = population[i]
            c5 = evaluate_c5(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        if time.time() - start_time > 0.75 * budget_s:
            break
        
        # Differential evolution step
        new_pop = de_step(population, fitness, F, CR)
        population = new_pop
        
        # Sometimes refine best
        if generation > 0 and generation % 200 == 0:
            if best_h is not None:
                refined_h = gradient_descent_on_worst_shift(best_h, max_iter=200)
                refined_c5 = evaluate_c5(refined_h)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h
        
        generation += 1
    
    # Early exit if no improvement or time limit
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        best_c5 = 0.5
    
    # Refine with gradient descent
    if time.time() - start_time < 0.9 * budget_s:
        refined_h = gradient_descent_on_worst_shift(best_h, max_iter=200)
        refined_c5 = evaluate_c5(refined_h)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    # Final stochastic polish 
    if time.time() - start_time < 0.99 * budget_s:
        h_final = best_h.copy()
        current_c5 = best_c5
        temp = 0.1
        decay = 0.9997
        for _ in range(20000):
            if time.time() - start_time > 0.99 * budget_s:
                break
            i, j = np.random.randint(0, n_points, 2)
            if h_final[i] <= 0 or h_final[j] >= 1:
                continue
            delta = np.random.uniform(0, min(h_final[i], 1 - h_final[j], 0.01))
            h_test = h_final.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_sum_constraint(h_test, target_sum)
            test_c5 = evaluate_c5(h_test)
            if test_c5 < current_c5:
                h_final = h_test
                current_c5 = test_c5
            else:
                # Accept with probability
                prob = np.exp(-(test_c5 - current_c5) / temp)
                if np.random.random() < prob:
                    h_final = h_test
                    current_c5 = test_c5
            temp *= decay
    
    # Final check and interpolation 
    if time.time() - start_time < budget_s:
        # Try fine resolution
        n_fine = 800
        if n_fine <= 1000:
            x_fine = np.linspace(0, 2, n_fine)
            x_coarse = np.linspace(0, 2, n_points)
            h_fine = np.interp(x_fine, x_coarse, h_final)
            h_fine = project_to_sum_constraint(h_fine, n_fine / 2)
            c5_fine = evaluate_c5(h_fine)
            if c5_fine < current_c5:
                h_final = h_fine
                current_c5 = c5_fine
    
    return (h_final, current_c5, n_points)
