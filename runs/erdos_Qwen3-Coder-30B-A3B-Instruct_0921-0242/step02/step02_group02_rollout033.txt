import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_feasible(v, n):
    """Project v onto the set {h : sum(h) = n/2, 0 <= h[i] <= 1}"""
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    if abs(current_sum - n/2) < 1e-10:
        return v
    
    # Water filling approach
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    
    # Find index where we need to add water
    # We want to adjust the largest components to make the sum correct
    # System: we set all components beyond index i to the same level,
    # sum of first i is fixed in sums.
    target = n / 2.0
    diff = target - current_sum
    if diff > 0:
        # Need to increase sum
        # Add up to diff to largest elements
        v = np.clip(v + diff / len(v), 0, 1)
    else:
        # Need to decrease sum
        # Reduce same amount from highest elements
        v = np.clip(v + diff / len(v), 0, 1)
    
    # Correct exact sum
    v = v * (n / 2.0) / np.sum(v)
    return v

def evaluate_c5(h):
    """Evaluate the C5 constant: max_k integral h(x)(1 - h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    
    # Compute correlation with (1-h) using FFT
    # v_correlation = correlate(h, 1 - h, mode='full')
    # Here we compute via FFT for efficiency
    f_h = fft(h)
    f_one_minus_h = fft(1 - h)
    # Cross correlation: conv(h, 1-h) = ifft(fft(h) * conj(fft(1-h)))
    v_correlation = ifft(f_h * f_one_minus_h.conj()).real
    max_corr = np.max(v_correlation) * dx
    return max_corr

def gradient_at_worst(h):
    """Compute gradient of objective w.r.t. the worst correlation"""
    n = len(h)
    dx = 2.0 / n
    
    # Compute full correlation
    f_h = fft(h)
    f_one_minus_h = fft(1 - h)
    v_correlation = ifft(f_h * f_one_minus_h.conj()).real
    
    # Find the index of the maximum correlation
    k_star = np.argmax(v_correlation)
    
    # Convert from full correlation index to actual offset
    # For full mode, convolution with length n gives 2*n-1 outputs.
    # indices from 0 to 2*n-2.
    if k_star < n:
        # Normal part
        shift = k_star
    else:
        # Wraparound
        shift = k_star - (2 * n - 2)
    
    # Construct gradient for the k* correlation term
    grad = np.zeros(n)
    for i in range(n):
        # Gradient w.r.t. h_i of correlation at shift k*:
        # d/dh_i (sum_j h[j]*h[i+j]) = h[i+k*] - h[i-k*]
        # But we are computing int h(x)*(1 - h(x+k)) dx
        # So it's - sum_j h[j]*h[i-j+k*] = -h[i-k*]+h[i+k*]
        # Actually no, integrate over h(x+k): 
        # int h(x)(1-h(x+k)) dx -> finite diff:
        # There's k* shift of h(x) with respect to h(x+k) 
        # Expression is: int h(x) * ( 1 - h(x+k) ) dx
        # So correlation is h * (1-h) with shift k*
        # Gradient w.r.t. h_i gives
        # - (h[i+k*] - h[i-k*]) = h[i-k*] - h[i+k*]
        # indices are modulo n
        idx_l = (i - shift) % n
        idx_r = (i + shift) % n
        # Note: the cross correlation is h * (1 - h) with shift k*
        # This really means the gradient is: (-) *(h[i+k*] - h[i-k*])
        # But in this field we use negative for inversion or easier formula.
        # Actually let's be precise:
        # For correlation v[k] = sum over i of h[i] * (1 - h[i+k])
        # Gradient w.r.t. h[i]  = sum over j of delta[i,j] * (1 - h[j+k])
        # If j==i, that's (1 - h[i+k]) 
        # If j==i-k, it's -h[i]
        # Let's recompute from scratch properly:
        # Full convolution f(h) = sum_i h[i] * h[i+k]
        # So gradient at i is (h[i+k] - h[i-k]) 
        # However, in our cross:
        # v[k] = int h(x) * (1-h(x-k)) dx
        # Has gradient at i: (1-h[i-k]) - (-h[i+k]) = (1 - h[i-k]) + h[i+k]
        # But that's still not right.
        # Let’s approach from direct:
        # Correlation integral is int_x h(x) * (1 - h(x+k)) dx
        # This leads to v[k] for shift k in convolution sense.
        # The gradient w.r.t. h[i] gives:
        # d/dh[i] int h(x)(1 - h(x+k)) dx
        # Integrate by parts, handling point derivatives:
        # You can consider the deviation:
        #     h(x) → h(x) + epsilon * delta_x[i]
        # The change in integral is:
        # int (epsilon * delta_x[i]) * (1 - h(x+k) ) dx = epsilon * (1 - h[i+k])
        # And also for h(x+k) term when h[i-k] contributes:
        # int h(x) * (-epsilon * delta_x[i-k]) dx = - epsilon * h[i-k]
        # So the gradient is (1 - h[i+k]) - h[i-k] = 1 - h[i+k] - h[i-k]
        # Which makes sense: we compute a cross-correlation
        # But we are trying to minimize.
        # So we want to reduce gradient.
        # Mix appropriate: gradient = h[i-k] + h[i+k] -1
        grad[i] = h[idx_l] + h[idx_r] - 1.0
    
    return grad, k_star

def line_search(h, grad, eta0=0.5):
    """Backtracking line search to find good step size"""
    dx = 2.0 / len(h)
    eta = eta0
    for _ in range(20):
        h_new = h - eta * grad
        h_new = project_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        if c5_new < c5_old:
            return h_new, c5_new
        eta /= 2.0
    return h, evaluate_c5(h)

def mutate(v, mode='gaussian', params=None):
    """Create child by mutation of parent v"""
    n = len(v)
    v = v.copy()
    
    if mode == 'gaussian':
        # Perturb small subset
        indices = np.random.choice(n, size=np.random.randint(5,11), replace=False)
        sigma = 0.2 if params is None else params.get('sigma', 0.2)
        for i in indices:
            v[i] += np.random.normal(0, sigma)
        v = project_feasible(v, n)
    elif mode == 'swap_mass':
        # Move a small amount from high to low
        h_i = np.random.randint(0, n)
        h_j = np.random.randint(0, n)
        while v[h_i] <= 0.01:
            h_i = np.random.randint(0, n)
        while v[h_j] >= 0.99:
            h_j = np.random.randint(0, n)
        delta = np.random.uniform(0.001, 0.01)
        v[h_i] -= delta
        v[h_j] += delta
        v = project_feasible(v, n)
    elif mode == 'dct':
        # Apply a low-frequency DCT tweak
        if params is None:
            lowf_idx = np.random.randint(0, 10)
        else:
            lowf_idx = params.get('lowfreq_idx', 0)
        v_dct = np.fft.dct(v)
        v_dct[lowf_idx] += np.random.normal(0, 0.05)  # Add tweak to low freq
        v_new = np.fft.idct(v_dct)
        v = project_feasible(v_new, n)
    else:
        raise ValueError("Unknown mutation mode")
    
    return v

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')
    
    n = 300
    dx = 2.0 / n

    # Phase 1: Generate random pool
    M = 2000
    pool = []
    for _ in range(M):
        h = np.random.rand(n)
        h = project_feasible(h, n)
        c5 = evaluate_c5(h)
        pool.append((h, c5))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Sort pool by c5 (ascending)
    pool.sort(key=lambda x: x[1])
    
    # Keep top K candidates
    K = 30
    pool = pool[:K]
    
    # Phase 2: Mutation loop
    max_iter = 10000
    t_elapsed = time.time() - t_start
    iter_count = 0
    
    while t_elapsed < 0.35 * budget_s and iter_count < max_iter:
        t_elapsed = time.time() - t_start
        if t_elapsed > 0.35 * budget_s:
            break
        
        parent_idx = np.random.randint(0, len(pool))
        parent_h = pool[parent_idx][0]
        
        # Decide mutation type
        mode = np.random.choice(['gaussian', 'swap_mass', 'dct'], p=[0.6, 0.3, 0.1])
        child_h = mutate(parent_h, mode)
        
        c5_child = evaluate_c5(child_h)
        
        # If better than worst in pool, replace it
        worst_idx = 0
        for i in range(len(pool)):
            if pool[i][1] > pool[worst_idx][1]:
                worst_idx = i
        
        if c5_child < pool[worst_idx][1]:
            pool[worst_idx] = (child_h, c5_child)
            pool.sort(key=lambda x: x[1])
            if c5_child < best_c5:
                best_c5 = c5_child
                best_h = child_h.copy()
        
        iter_count += 1
    
    # Phase 3: Local refinement of top candidates
    t_elapsed = time.time() - t_start
    for h, c5 in pool[:5]:  # Top 5
        if t_elapsed > 0.75 * budget_s:
            break
        h_refined = h.copy()
        c5_refined = c5
        for _ in range(2000):
            t_elapsed = time.time() - t_start
            if t_elapsed > 0.75 * budget_s:
                break
            grad, k_star = gradient_at_worst(h_refined)
            h_new, c5_new = line_search(h_refined, grad)
            if c5_new < c5_refined:
                h_refined = h_new
                c5_refined = c5_new
            else:
                break
    
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined
    
    # Phase 4: Final polishing with swaps  
    t_elapsed = time.time() - t_start
    if t_elapsed < budget_s - 0.1:
        h_final = best_h.copy()
        for _ in range(10000):
            i, j = np.random.randint(0, n, 2)
            if h_final[i] <= 0.01 or h_final[j] >= 0.99:
                continue
            delta = np.random.uniform(0.0001, 0.001)
            h_trial = h_final.copy()
            h_trial[i] -= delta
            h_trial[j] += delta
            h_trial = project_feasible(h_trial, n)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                h_final = h_trial
                best_c5 = c5_trial
        
        best_h = h_final
    
    return (best_h.tolist(), best_c5, n)
