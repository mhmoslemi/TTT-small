import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_feasible(v, n):
    """Project v to the feasible set: sum(v)=n/2, 0<=v<=1."""
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return v
    
    # Water-filling to adjust sum
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    n_excess = np.searchsorted(cumsum, n/2, side='right')
    if n_excess == 0:
        # All elements need to be raised
        level = n/2 / n
    else:
        # Some elements are set to 1, others are adjusted
        level = (n/2 - cumsum[n_excess-1]) / (n - n_excess)
        level = max(level, 0)
        level = min(level, 1)
        
    # Compute projected vector
    v = np.clip(v, 0, 1)
    v_new = np.minimum(v, level)
    v_new = np.maximum(v_new, 0)
    v_new = v_new + (n/2 - np.sum(v_new)) / n
    v_new = np.clip(v_new, 0, 1)
    
    # Adjust one more time to exactly match target
    v_new = v_new * (n/2) / np.sum(v_new)
    return v_new

def evaluate_c5(h):
    """Evaluate C5 using FFT for full correlation."""
    dx = 2.0 / len(h)
    # Compute full correlation via FFT
    f_h = fft(h)
    f_1mh = fft(1 - h)
    # Correlate: h * (1-h).conj()
    c = ifft(f_h * f_1mh.conj())
    c = np.real(c)
    # Find maximum over indices
    max_c = np.max(c) * dx
    return max_c

def gradient_at_worst(h):
    """Compute gradient of max correlation with respect to h."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation
    c = correlate(h, 1-h, mode='full')
    k_star = np.argmax(c)
    # k_star from 0 to 2*n-2
    shift = k_star - (n - 1)
    
    grad = np.zeros_like(h)
    for i in range(n):
        i_plus = (i + shift) % n
        i_minus = (i - shift) % n
        grad[i] = dx * (h[i_plus] - h[i_minus])
    return grad

def line_search(h, grad, eta0=0.5):
    """Line search with backtracking."""
    eta = eta0
    dx = 2.0 / len(h)
    c0 = evaluate_c5(h)
    for _ in range(10):
        h_new = h - eta * grad
        h_new = project_feasible(h_new, len(h))
        c_new = evaluate_c5(h_new)
        if c_new < c0:
            return h_new, c_new
        eta *= 0.5
    return h, c0

def mutate_gaussian(v, sigma=0.2):
    """Gaussian mutation on random subset."""
    n = len(v)
    indices = np.random.choice(n, size=np.random.randint(5, 10), replace=False)
    delta = np.random.normal(0, sigma, len(indices))
    v_new = v.copy()
    v_new[indices] += delta
    return project_feasible(v_new, len(v))

def mutate_swap(v, delta=0.01):
    """Swap mass between values."""
    v_new = v.copy()
    i = np.random.choice(np.where(v_new > delta)[0])
    j = np.random.choice(np.where(v_new < 1-delta)[0])
    d = min(delta, v_new[i], 1 - v_new[j])
    v_new[i] -= d
    v_new[j] += d
    return v_new

def mutate_dct(v, mode=0):
    """Modify a low-frequency DCT coefficient."""
    n = len(v)
    # Compute DCT
    dct = np.fft.fftshift(np.fft.fft(v))[:n//2]
    # Modify one coefficient
    idx = np.random.randint(1, min(10, n//2))  # Avoid DC
    change = np.random.normal(0, 0.02)
    dct[idx] += change
    # Inverse transform
    v_dct = np.zeros(n)
    v_dct[:n//2] = dct
    v_dct[n//2:] = dct[::-1]
    v_recovered = np.real(np.fft.ifft(np.fft.fftshift(v_dct)))
    return project_feasible(v_recovered, n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n0 = 250
    n1 = 500
    n2 = 800
    n = n0
    t_left = budget_s
    dx = 2.0 / n
    
    # Pool based global search
    M = 2000  # global pool size
    K = 30    # top candidates saved
    
    # Initial random pool
    pool = []
    for _ in range(M):
        v = np.random.rand(n)
        pool.append(project_feasible(v, n))
    
    # Evaluate pool
    scores = []
    for i, v in enumerate(pool):
        c5 = evaluate_c5(v)
        scores.append(c5)
    
    # Sort and keep top K
    idxs = np.argsort(scores)[:K]
    candidates = [pool[i] for i in idxs]
    scores = [scores[i] for i in idxs]
    
    # Store best
    best_score = min(scores)
    best_h = candidates[0].copy()
    
    # Mutation phase
    t_elapsed = time.time() - start_time
    mutations = 0
    sigma = 0.2
    while t_elapsed < 0.25 * budget_s and mutations < 50000:
        if time.time() - start_time > 0.25 * budget_s:
            break
        
        # Choose parent from pool
        parent = np.random.choice(candidates)
        
        # Select mutation type
        mode = np.random.randint(0, 3)
        if mode == 0:
            child = mutate_gaussian(parent, sigma)
        elif mode == 1:
            child = mutate_swap(parent)
        elif mode == 2:
            child = mutate_dct(parent)
        
        # Eval and potentially replace
        c5_child = evaluate_c5(child)
        if c5_child < scores[-1]:  # replace worst if better
            candidates[-1] = child
            scores[-1] = c5_child
            idxs = np.argsort(scores)
            candidates = [candidates[i] for i in idxs]
            scores = [scores[i] for i in idxs]
            if c5_child < best_score:
                best_score = c5_child
                best_h = child.copy()
        
        mutations += 1
        t_elapsed = time.time() - start_time
        sigma = max(0.01, sigma * 0.995)
    
    # Local refinement on top candidates
    for i in range(min(5, len(candidates))):
        h_local = candidates[i].copy()
        c5_local = scores[i]
        t_refine = time.time() - start_time
        
        for _ in range(2000):
            if time.time() - start_time > 0.45 * budget_s:
                break
            
            # Gradient
            grad = gradient_at_worst(h_local)
            h_new, c5_new = line_search(h_local, grad)
            
            if c5_new < c5_local:
                h_local = h_new
                c5_local = c5_new
                if c5_local < best_score:
                    best_score = c5_local
                    best_h = h_local.copy()
            else:
                break  # No improvement, stop

    # Pairwise swaps
    t_polish = time.time() - start_time
    h_best = best_h.copy()
    c5_best = best_score
    
    for _ in range(10000):
        if time.time() - start_time > 0.55 * budget_s:
            break
        # Random pair
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h_best[i] < 1e-3 or h_best[j] > 1 - 1e-3:
            continue
        if h_best[i] > 1 - 1e-3 or h_best[j] < 1e-3:
            continue
            
        # Swap small amount
        delta = np.random.uniform(1e-4, 1e-2)
        h_trial = h_best.copy()
        h_trial[i] -= delta
        h_trial[j] += delta
        h_trial = project_feasible(h_trial, n)
        
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_best:
            h_best = h_trial
            c5_best = c5_trial
            if c5_best < best_score:
                best_score = c5_best
                best_h = h_best.copy()
    
    # Sizing strategy
    if n == n0 and best_score < 0.3807:
        # Try also n1
        h_upsampled = np.interp(np.linspace(0, 2, n1), np.linspace(0, 2, n0), best_h)
        h_upsampled = project_feasible(h_upsampled, n1)
        c5_up = evaluate_c5(h_upsampled)
        if c5_up < best_score:
            best_score = c5_up
            best_h = h_upsampled
            n = n1
        
        # Refine further
        h_refined = best_h.copy()
        c5_ref = best_score
        for _ in range(1000):
            if time.time() - start_time > 0.9 * budget_s:
                break
            grad = gradient_at_worst(h_refined)
            h_new, c5_new = line_search(h_refined, grad)
            if c5_new < c5_ref:
                h_refined = h_new
                c5_ref = c5_new
                if c5_ref < best_score:
                    best_score = c5_ref
                    best_h = h_refined.copy()
            else:
                break
        best_h = h_refined
    
    return (best_h.tolist(), best_score, n)
