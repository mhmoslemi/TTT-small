import numpy as np

from scipy.signal import correlate

def project_box_sum(y, target_sum):
    """Project y onto the set {x : 0 ≤ x ≤ 1, sum(x) = target_sum} using water filling."""
    n = len(y)
    y = np.array(y)
    y_sorted = np.sort(y)[::-1]
    
    # Find lambda such that sum(clip(y - lambda, 0, 1)) == target_sum
    # Use bisection on lambda
    
    # Compute cumulative sums from largest values
    cumsum = np.cumsum(y_sorted)
    # These are the sums k sorted values could be set to
    lambda_min = y_sorted[-1] - 1  # worst case: all at 0 or 1
    lambda_max = y_sorted[0]
    
    def S(lam):
        # This is the total value of clipped elements: sum_i(max(0, y_i - lam))
        clipped = np.maximum(y - lam, 0)
        return np.sum(clipped)
    
    # Binary search to find the right lambda
    low, high = lambda_min, lambda_max
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2
        s = S(mid)
        if abs(s - target_sum) < tol:
            break
        elif s > target_sum:
            low = mid
        else:
            high = mid
    
    # Construct x: normal truncated
    x = np.maximum(y - (low + high)/2, 0)
    x = np.minimum(x, 1)
    # Adjust to match exact sum
    diff = target_sum - np.sum(x)
    
    # Distribute diff
    if abs(diff) > 1e-10:
        # Need to adjust
        # Find indices where x < 1, increase them by diff/num
        where_under = np.where(x < 1)[0]
        if len(where_under) > 0:
            adjust = min(diff, 1 - x[where_under[0]])
            x[where_under[0]] += adjust
            diff -= adjust
            
        # If there's still diff, adjust others
        if abs(diff) > 1e-10:
            iter = 0
            for i in range(0, len(x)):
                if diff <= 1e-10:
                    break
                if x[i] < 1:
                    adjust = min(diff, 1 - x[i])
                    x[i] += adjust
                    diff -= adjust
    return x

def compute_autocorrelation(h):
    """Compute full autocorrelation of h."""
    return correlate(h, h, mode='full')

def evaluate_c5(h):
    """Evaluate C5 for a given h vector."""
    dx = 2.0 / len(h)
    # Full autocorrelation of h with h itself
    corr = correlate(h, h, mode='full')
    # Compute the overlap integral as max of 1 - (1 - h(x+k)) * h(x) which is same as 
    # max of 1 - (h(x) * h(x+k)) which is max of 1 - correlation.
    # So we use 1 - gamma_min
    min_corr = np.min(corr)
    max_overlap = 1 - min_corr * dx
    return max_overlap

def find_worst_shifts(C, delta=1e-5):
    """Find all indices s such that C[s] is within delta of max(C)."""
    max_C = np.max(C)
    S_w = np.where(np.abs(C - max_C) < delta)[0]
    return S_w

def update_autocorrelation_incremental(h, γ, i, j, Δ, n):
    """Update γ due to transfer of Δ from i to j in O(n)."""
    # Precondition: h[i] >= Δ, h[j] <= 1 - Δ
    # Don't need to recompute h[] since we transfer from i to j in-place
    # but we want to compute γ_new = γ + Δγ
    
    Δγ = np.zeros_like(γ)
    
    # We compute changes in γ for one transfer from i to j
    # 'h' is passed unchanged; puzzle out what to compute
    
    # Remove Δ from h[i]:
    # Contribution to γ[s] = h[i+s]*h[i] for s = 0..n-1 (on right)
    # Only s = 0..n-1 such that both i+s and i are within bounds
    # i+s must be in [0, n), i-k must be in [0, n)
    
    for s in range(-(n - 1), n):
        idx = s + n - 1
        # Term 1 : -Δ * h[i+s] (or 0 if out of bounds)
        if 0 <= i + s < n:
            Δγ[idx] -= Δ * h[i+s]
        
        # Term 2 : -Δ * h[i-s] (or 0 if out of bounds, also exclude s=0 case in first term)
        if 0 <= i - s < n and s != 0:
            Δγ[idx] -= Δ * h[i-s]
            
        # Term 3: +Δ * h[j+s] (or 0 if out of bounds)
        if 0 <= j + s < n:
            Δγ[idx] += Δ * h[j+s]
            
        # Term 4: +Δ * h[j-s] (or 0 if out of bounds, also exclude s=0 cases)
        if 0 <= j - s < n and s != 0:
            Δγ[idx] += Δ * h[j-s]
        
        # Final adjustment: self-term s=0, due to quadratic tightening changes  
        if s == 0:
            Δγ[idx] += 2 * h[i] * Δ - Δ*Δ + 2 * h[j] * Δ + Δ*Δ
            # Simplify: 2*(h[i] - h[j]) * Δ
            
    return Δγ

def move_generator(h, γ, S_w, n, η=1e-8):
    """
    Generate move from index i to j based on weighted selection.
    """
    # Score for each index: sum over shifts in S_w of neighbors' values
    score = np.zeros(n)
    for s in S_w:
        # Add to forward neighbors
        offset = s
        if offset >= 0:
            for i in range(n - offset):
                score[i] += h[i + offset]
        else:
            for i in range(-offset, n):
                score[i] += h[i + offset]
        
        # Add to backward neighbors
        offset = -s
        if offset >= 0:
            for i in range(n - offset):
                score[i] += h[i + offset]
        else:
            for i in range(-offset, n):
                score[i] += h[i + offset]
    
    # Score for index i
    P = np.array([score[j] for j in range(n)])
    P[~(h > η)] = 0.0  # No score for indices that are nearly zero
    P /= np.sum(P) + 1e-12   # Normalize to sum 1
    
    # Sample i based on these weights
    i = np.random.choice(np.arange(n), p=P)
    
    # Sample j from bottom 10% of indices with small values, weighted inverse score.
    mask = h < 1 - η  # Must not be at upper bound
    weights = np.array([1.0 / (score[k] + 1e-12) if score[k] > 1e-12 else 1e12
                       for k in range(n)])
    weights[~mask] = 0.0
    weights = np.clip(weights, 0, 1e12)
    weights = np.power(weights, 1.0)
    weights[~mask] = 0.0
    weights /= np.sum(weights) + 1e-12
    
    # Sample with probability of weights
    j = np.random.choice(np.arange(n), p=weights)
    
    return i, j

def local_subgradient_descent(h, max_iter=500):
    """Local refinement of h to reduce max overlap."""
    # Get overlapping shifts (lowest autocorrelation entries)
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, h, mode='full')
    min_corr = np.min(corr)
    max_C = 1 - min_corr * dx
    
    gamma = corr.copy()
    x = h.copy()
    stepsize = 0.01  # Elasticity of weights, should decouple from x[i] scaling
    
    i = 0
    while i < max_iter and stepsize > 1e-8:
        # Recompute from scratch to find worst shifts (this round)
        corr = correlate(x, x, mode='full')
        max_C = 1 - np.min(corr) * dx
        # We also find such shift indices that C = max_C
        S_w = find_worst_shifts(1 - corr * dx)

        # Subgradient
        g = np.zeros_like(x)
        for s in S_w:
            # Centralize on index s (that's shift s from 0)
            idx_s = s + n - 1
            # Change in shift s of x is
            # (x[s+1] + x[s+2] + ...) + (-x[0] - x[1] - ...) if s>0, absorptively 
            # (x[-1] + x[-2] + ... ) + x[0] if s<0 
            # But better parsed using segmentation:
            # First term is sum of left side (at indices 0..s)
            # Other term (from right) is sum across to s, with sign flipped.
            # directly recompute장애인:1e-100  
            start = max(0, -s)
            end = min(n, n - s)
            for j in range(start, end):
                g[j] -= x[j]

            # Same for the other term
            start = max(0, s)
            end = min(n, n + s)
            for j in range(start, end):
                g[j] += x[j] 
                
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 0:
            g = g / g_norm
            
        # Backtracking line search
        alpha = 1.0
        iter = 0
        while iter < 10:
            trial = x - alpha * g
            trial = np.clip(trial, 0, 1)
            # Project it back to sum = n/2
            trial = project_box_sum(trial, n/2)
            trial_C5 = evaluate_c5(trial)
            if trial_C5 < max_C or iter == 0:
                x = trial
                break
            else:
                alpha *= 0.5
                iter += 1
        i += 1

    return x, evaluate_c5(x)
