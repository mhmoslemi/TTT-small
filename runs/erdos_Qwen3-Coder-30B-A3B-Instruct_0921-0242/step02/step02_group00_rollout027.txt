import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import bisect

def project_to_box_plus_sum(h, n):
    """Project h to the set {0 <= h <= 1, sum(h) = n/2} using water-filling."""
    # Sort values in descending order
    sorted_h = np.sort(h)[::-1]
    # Compute cumulative sums from the right
    cumsum = np.cumsum(sorted_h[::-1])[::-1]
    # Find the threshold lambda
    idx = np.where(sorted_h * (np.arange(len(sorted_h)) + 1) >= cumsum - n/2)[0]
    if len(idx) == 0:
        lambda_star = 0.0
    else:
        k = np.max(idx)
        lambda_star = (cumsum[k] - n/2) / (k + 1)
    # Generate the projected values
    result = np.clip(h - lambda_star, 0, 1)
    # Ensure that sum is exactly n/2
    current_sum = np.sum(result)
    diff = n/2 - current_sum
    if abs(diff) > 1e-10:
        # Distribute the difference to preserve order
        indices = np.argsort(result)[::-1]  # indices of elements in descending value order
        i = 0
        while abs(diff) > 1e-10 and i < len(indices):
            idx = indices[i]
            add = min(diff, 1 - result[idx])
            result[idx] += add
            diff -= add
            i += 1
    return result

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    # Index 0 of corr corresponds to shift -(n-1)
    # We want indices n-1 to 2*(n-1) for shifts 0 to n-1
    max_corr = np.max(corr)
    return max_corr * dx

def compute_autocorrelation(h):
    """Compute aperiodic autocorrelation of h directly in time."""
    n = len(h)
    # For aperiodic autocorrelation, we don't pad
    # Use ffts for efficiency
    nfft = 2**int(np.ceil(np.log2(2*n - 1)))
    z = np.zeros(nfft)
    z[:n] = h
    Hz = fft(z)
    aut = ifft(Hz * np.conj(Hz))
    return np.real(aut[:2*n-1])

def compute_c_values(h):
    """Compute all C_s values from h."""
    n = len(h)
    gamma = compute_autocorrelation(h)[:2*n-1]
    C = 1.0 - (2.0/n) * gamma
    return C

def admm_update_h(z, u, t_prev, rho, n):
    """Update h given current z, u, t using bisection on mu."""
    # FFT of z - u for efficiency
    Hz_u = fft(np.concatenate([z - u, [0]*n]))
    ones_vec = np.ones(n)
    ones_fft = fft(np.concatenate([ones_vec, [0]*(2*n-1)]))
    
    # The optimization is: h = (z - u - mu * convolution(h, ones)) / (1 + mu * ones_fft)
    # Or iterating in freq domain: h = (z - u) / (1 + mu * ones_fft)
    # Let's solve in frequency:
    def eval_mu(mu):
        # h in frequency domain
        denom = 1.0 + mu * ones_fft
        if np.allclose(denom, 0.0):
            h_freq = np.zeros_like(denom)
        else:
            h_freq = Hz_u / denom
        h = np.real(ifft(h_freq))
        # truncate to first n entries
        h = h[:n]
        C = compute_c_values(h)
        return np.max(C) - t_prev
        
    # Attempt bisection to find mu such that eval_mu(mu) <= 0
    # Start with mu = 0.0 (this should correspond to zero constraint violation)
    # Since we want the smallest such mu, we bisection to find the root of eval_mu(mu)
    # We want eval_mu(mu) to be close to zero, not always < 0
    
    # Try to find an upper bound first:
    mu_lo = 0.0
    mu_hi = 1000.0
    # Heuristic: try increasing values of mu until we find one with a negative eval
    try:
        mult = 1.0
        while True:
            mu_try = mu_lo + mult
            val = eval_mu(mu_try)
            if val <= 0:
                mu_hi = mu_try
                break
            mult *= 2.0
    except:
        pass  # Just use fallback in case of failure
    # Now do standard bisection on [mu_lo, mu_hi] where eval_mu(m) <= 0 for final mu
    try:
        mu = bisect(eval_mu, mu_lo, mu_hi, xtol=1e-12)
    except:
        # fallback to a tuned value if bisection fails
        mu = 1e-6 if mu_lo == 0.0 else mu_lo + 0.1
    # Now compute final h
    denom = 1.0 + mu * ones_fft
    h_freq = Hz_u / denom
    return np.real(ifft(h_freq))[:n]

def admm_step(z, h, u, t, rho, n):
    """Perform one ADMM step."""
    # Step 1: Update h
    h_new = admm_update_h(z, u, t, rho, n)
    # Step 2: Update z via projection
    z_new = project_to_box_plus_sum(h_new + u, n)
    # Step 3: Update dual variable
    u_new = u + (h_new - z_new)
    # Step 4: Update t from new h
    C = compute_c_values(h_new)
    t_new = np.max(C)
    return h_new, z_new, u_new, t_new
