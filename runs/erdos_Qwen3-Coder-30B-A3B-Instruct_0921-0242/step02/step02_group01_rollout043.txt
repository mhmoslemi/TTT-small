import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.linalg import toeplitz
import cvxpy as cp

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 256  # Use a reasonable size for SDP
    dx = 2.0 / n
    t_low, t_high = 0.0, 1.0
    epsilon = 2e-4  # Bisection tolerance
    max_sdp_iter = 15  # Maximum bisection iterations
    
    # Time budgets
    sdp_budget = 0.35 * budget_s
    refine_budget = 0.35 * budget_s
    polish_budget = 0.15 * budget_s
    upscale_budget = budget_s - sdp_budget - refine_budget - polish_budget
    
    # Phase 1: SDP bisection to find feasible r_k
    r_feasible = None
    while t_low < t_high - epsilon and (time.time() - start_time) < sdp_budget:
        t_mid = (t_low + t_high) / 2
        
        # Build the SDP
        r = cp.Variable(n)
        constraints = [
            r[0] == 1,
            r[1:] >= t_mid,
            r <= 1,
            cp.PSD(toeplitz(r))
        ]
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                r_feasible = r.value
                t_low = t_mid
            else:
                t_high = t_mid
        except Exception:
            t_high = t_mid
    
    if r_feasible is None:
        # Fallback if SDP failed
        r_feasible = np.ones(n)
        r_feasible[0] = 1.0
        r_feasible[1:] = 0.0
    
    # Phase 2: Spectral factorization & projection into h
    # Power spectrum:
    S = np.fft.fft(r_feasible)
    # Clamp negative values to zero
    S = np.maximum(S, 0.0)
    H = np.sqrt(S)
    g = np.real(np.fft.ifft(H))
    h = project_to_feasible(g, n)

    # Phase 3: Local refinement via sub-gradient descent
    if (time.time() - start_time) < refine_budget:
        max_iter = min(250, int((refine_budget - (time.time() - start_time)) / 0.005))
        alpha0 = 0.5 * dx
        for t in range(max_iter):
            if (time.time() - start_time) >= refine_budget:
                break
            
            # Compute correlations
            h_fft = fft(h, 2 * n)
            ih_fft = fft(1 - h, 2 * n)
            corr = ifft(h_fft * np.conj(ih_fft))
            corr_vals = corr.real[1:n]  # Exclude k=0
            worst_k = np.argmax(corr_vals)
            # Gradient
            grad = np.zeros_like(h)
            for i in range(n):
                left_idx = (i + worst_k) % n
                right_idx = (i - worst_k) % n
                grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
            
            # Update
            alpha = alpha0 / np.sqrt(t + 1)
            h = h - alpha * grad
            h = project_to_feasible(h, n)
            
            # Early stop
            if t > 10 and np.abs(evaluate_c5(h) - evaluate_c5(h)) < 1e-6:
                break
    
    # Phase 4: Stochastic polishing
    if (time.time() - start_time) < polish_budget:
        max_trials = 10000
        max_time = polish_budget
        deadline = time.time() + max_time
        current_c5 = evaluate_c5(h)
        
        for _ in range(max_trials):
            if time.time() > deadline:
                break
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if h[i] <= 0 or h[j] >= 1:
                continue
            delta = np.random.uniform(0, min(h[i], 1 - h[j], 0.01))
            
            h_trial = h.copy()
            h_trial[i] -= delta
            h_trial[j] += delta
            h_trial = project_to_feasible(h_trial, n)
            c5_trial = evaluate_c5(h_trial)
            
            if c5_trial < current_c5:
                h = h_trial
                current_c5 = c5_trial
    
    # Phase 5: Optional upscaling and final refinement
    if (time.time() - start_time) < budget_s * 0.99:
        n_upscale = min(1024, n * 2)
        x_orig = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_upscale)
        h_upscaled = np.interp(x_new, x_orig, h)
        h_upscaled = h_upscaled * (n_upscale / 2) / np.sum(h_upscaled)
        h_upscaled = project_to_feasible(h_upscaled, n_upscale)
        h_final = h_upscaled
        
        c5_final = evaluate_c5(h_final)
        # Fine tune for 50 iters
        for _ in range(50):
            if (time.time() - start_time) > budget_s * 0.99:
                break
            # Compute worst shift
            corr = ifft(fft(h_final, 2 * len(h_final)) * np.conj(fft(1 - h_final, 2 * len(h_final))))
            corr_vals = corr.real[1:n_upscale]  # exclude k=0
            worst_k = np.argmax(corr_vals)
            # Gradient
            grad = np.zeros_like(h_final)
            for i in range(n_upscale):
                left_idx = (i + worst_k) % n_upscale
                right_idx = (i - worst_k) % n_upscale
                grad[i] = dx * ((1 - h_final[left_idx]) - h_final[right_idx])
            # Update
            alpha = 0.1 * dx
            h_final = h_final - alpha * grad
            h_final = project_to_feasible(h_final, n_upscale)
            
            c5_final = evaluate_c5(h_final)
                            

    # Final evaluation
    final_c5 = evaluate_c5(h_final)
    return (h_final, final_c5, n_upscale)
