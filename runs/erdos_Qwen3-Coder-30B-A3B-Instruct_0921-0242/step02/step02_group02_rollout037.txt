import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    scaling = (n / 2) / current_sum
    h = h * scaling
    return h

def compute_spectral_lower_bound(n):
    """Compute lower bound using SDP relaxation."""
    dx = 2.0 / n
    # Create circulant matrix for each shift
    C = []
    for k in range(n):
        mat = np.zeros((n, n))
        for i in range(n):
            mat[i, (i - k) % n] = 1
        C.append(mat)
    
    # SDP relaxation to compute the lower bound
    h = cp.Variable(n)
    gamma = cp.Variable()
    
    # Constraint: h >= 0, h <= 1
    constraints = [h >= 0, h <= 1]
    
    # Constraint: sum(h) = n/2
    constraints.append(cp.sum(h) == n / 2)
    
    # SDP relaxations
    for k in range(n):
        # Slacked constraint: h^T C_k h >= (n/2 - gamma/dx)
        constraints.append(cp.quad_form(h, C[k]) >= (n / 2 - gamma / dx))
    
    # Objective: minimize gamma
    objective = cp.Minimize(gamma)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=CPLEX, verbose=False, max_seconds=30)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return gamma.value, h.value
    except:
        pass
    
    # Fallback: just a guess
    return 0.5, np.ones(n) * 0.5

def generate_binary_pattern_from_spectral(n, target_autocorr=None):
    # If target_autocorr is not given, compute from SDP
    if target_autocorr is None:
        lower_bound, hvals = compute_spectral_lower_bound(n)
        if hvals is not None:
            hvals = project_to_feasible(hvals, n)
            target_autocorr = correlate(hvals, 1-hvals, mode='full') * (2.0/n)
            target_autocorr = target_autocorr[n-1:]
        else:
            target_autocorr = np.ones(n) * 0.5
    
    # Use integer programming to match target autocorrelation
    try:
        h_bin = cp.Variable(n, boolean=True)
        gamma = cp.Variable()
        constraints = [cp.sum(h_bin) == n / 2]
        
        # Introduce auxiliary variables s_k for each shift
        s_k = [cp.Variable() for _ in range(n)]
        z_ik = [[cp.Variable(boolean=True) for _ in range(n)] for _ in range(n)]
        
        for k in range(n):
            # Linearization of b_i b_{i-k}
            for i in range(n):
                # z_ik[i][k] <= b_i, z_ik[i][k] <= b_{i-k}
                constraints.append(z_ik[i][k] <= h_bin[i])
                constraints.append(z_ik[i][k] <= h_bin[(i-k) % n])
                # z_ik[i][k] >= b_i + b_{i-k} - 1
                constraints.append(z_ik[i][k] >= h_bin[i] + h_bin[(i-k) % n] - 1)
                # z_ik[i][k] >= 0
                constraints.append(z_ik[i][k] >= 0)
                
                # s_k = sum z_ik[i][k]
                constraints.append(s_k[k] == cp.sum(z_ik[i][k]))
            
            # Constraining: dx*(n/2 - s_k) <= gamma
            constraints.append(dx * (n / 2 - s_k[k]) <= gamma)
        
        objective = cp.Minimize(gamma)
        prob = cp.Problem(objective, constraints)
        
        prob.solve(solver=CPLEX, verbose=False, max_seconds=15)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h_bin.value
        else:
            return None
    except:
        pass
    
    # Fallback using threshold rounding
    try:
        if target_autocorr is not None:
            if np.allclose(target_autocorr, np.mean(target_autocorr)):
                # Uniform target autocorrelation
                goal = 0.5
                return (np.random.rand(n) > goal).astype(float)
    except:
        pass
    
    # Final fallback to 0.5
    return np.ones(n) * 0.5

def smooth_ramp(h, width=2):
    """Convert binary to continuous using smooth ramps."""
    new_h = h.copy()
    for i in range(len(h)):
        if h[i] == 0 and i != 0:
            if h[i-1] == 1:
                for j in range(max(0, i-width), i):
                    if j % len(h) != i:
                        ratio = 1 - (i-j) / width
                        new_h[j] = max(0, ratio)
        elif h[i] == 1 and i != 0:
            if h[i-1] == 0:
                for j in range(max(0, i-width), i):
                    if j % len(h) != i:
                        ratio = 1 - (i-j) / width
                        new_h[j] = min(1, ratio)
    return new_h

def lse_surrogate(h, beta=5):
    """Compute log-sum-exp approximation of the maximum overlap."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1-h, mode='full') * dx
    k_max = np.argmax(corr)
    return np.log(np.sum(np.exp(beta * corr))) / beta

def projected_lbfgsb_optimize(h, max_iter=2000, beta_start=5):
    """Optimize step function using projected L-BFGS-B with log-sum-exp surrogate."""
    n = len(h)
    dx = 2.0 / n
    
    def gradient(h, beta):
        corr = correlate(h, 1-h, mode='full') * dx
        indices = np.argsort(-corr)
        # Softmax weights
        exps = np.exp(beta * corr)
        weights = exps / np.sum(exps)
        # Compute gradient based on weights
        grad = np.zeros_like(h)
        for i in range(n):
            for k in range(n):
                if weights[k] > 0:
                    left_idx = (i - k) % n
                    right_idx = (i + k) % n
                    grad[i] += weights[k] * dx * (h[right_idx] - h[left_idx])
        return grad
    
    def obj(h, beta):
        return lse_surrogate(h, beta)
    
    def proj(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scaling = (n / 2) / current_sum
        h = h * scaling
        return h
    
    h = proj(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    beta = beta_start
    for it in range(max_iter):
        grad = gradient(h, beta)
        # Step size estimation (Armijo rule)
        step_size = 1.0
        while True:
            h_trial = h - step_size * grad
            h_trial = proj(h_trial)
            trial_val = obj(h_trial, beta)
            if trial_val < obj(h, beta) - 0.5 * step_size * np.sum(grad**2):
                h = h_trial
                break
            step_size *= 0.5
            if step_size < 1e-10:
                break
        
        if it % 10 == 0:
            c5_val = evaluate_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
        
        # Increase beta
        if beta < 80:
            beta *= 2
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Allocation: 200 for spectral, 300 for binary phase
    if budget_s > 400:
        n_points = 300
    else:
        n_points = 200
    
    # Phase 1: Spectral relaxation to generate target autocorrelation
    lower_bound, hvals = compute_spectral_lower_bound(n_points)
    if hvals is not None:
        hvals = project_to_feasible(hvals, n_points)
        target_autocorr = correlate(hvals, 1-hvals, mode='full') * (2.0/n_points)
        target_autocorr = target_autocorr[n_points-1:]
    else:
        target_autocorr = None
    
    # Phase 2: Binary integer program
    h_bin = generate_binary_pattern_from_spectral(n_points, target_autocorr)
    if h_bin is None or isinstance(h_bin, np.ndarray):
        pass
    else:
        # Fallback to uniform
        h_bin = np.ones(n_points) * 0.5
    
    # Convert binary to continuous using smooth ramp
    h = smooth_ramp(h_bin, width=3)
    h = project_to_feasible(h, n_points)
    
    # Phase 3: Continuous refinement with L-BFGS-B
    h_opt, c5_bound = projected_lbfgsb_optimize(h, max_iter=min(2000, max(500, int(budget_s * 0.2))))
    
    # Final polishing with exact max-overlap objective
    t_remaining = budget_s - (time.time() - start_time)
    if t_remaining > 2.0:
        # Subgradient method
        h_best = h_opt.copy()
        c5_best = c5_bound
        
        def subgradient_step(h, k_star):
            # compute gradient of c_k_star wrt h
            dx = 2.0 / len(h)
            grad = np.zeros_like(h)
            for i in range(len(h)):
                left_idx = (i - k_star) % len(h)
                right_idx = (i + k_star) % len(h)
                grad[i] = dx * (h[right_idx] - h[left_idx])
            
            # compute step size
            norm2 = np.sum(grad**2)
            if norm2 == 0:
                return h
            eta = c5_best / norm2
            h_new = h - eta * grad
            h_new = np.clip(h_new, 0, 1)
            current_sum = np.sum(h_new)
            scaling = (len(h_new) / 2) / current_sum
            h_new = h_new * scaling
            return h_new
        
        for _ in range(200):
            corr = correlate(h_best, 1-h_best, mode='full') * (2.0 / len(h_best))
            k_star = np.argmax(corr)
            h_trial = subgradient_step(h_best, k_star)
            trial_c5 = evaluate_c5(h_trial)
            if trial_c5 < c5_best:
                h_best = h_trial
                c5_best = trial_c5
            if time.time() - start_time > budget_s - 0.5:
                break
        h_opt, c5_bound = h_best, c5_best
    
    return (h_opt, c5_bound, n_points)
