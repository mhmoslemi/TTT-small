import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import bisect

def project_box_sum(y, target_sum):
    """Water-filling projection to keep sum = target_sum and bounds [0,1]"""
    def S(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    
    # find lambda such that S(lambda) = target_sum
    low, high = np.min(y), np.max(y) + (target_sum - np.sum(np.clip(y, 0, 1)))
    try:
        lambda_val = bisect(lambda l: S(l) - target_sum, low, high, xtol=1e-12)
    except:
        lambda_val = (low + high) / 2  # fallback
    x = np.clip(y - lambda_val, 0, 1)
    return x

def evaluate_c5(h):
    """Computes the C5 constant"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorrelation(h):
    """Compute full autocorrelation of h with itself"""
    return correlate(h, h, mode='full')

def update_autocorr(γ, h, i, j, Δ):
    """Efficiently update autocorrelation after transferring Δ from i to j"""
    n = len(h)
    γ_new = γ.copy()
    ĥ = h.copy()  # temporary vector after removing Δ from i
    ĥ[i] = 0
    
    for s in range(-n+1, n):
        idx = s + n - 1
        # removal of Δ from i
        if 0 <= i+s < n:
            γ_new[idx] -= Δ * ĥ[i+s]
        if 0 <= i-s < n and s != 0:
            γ_new[idx] -= Δ * ĥ[i-s]
        # addition of Δ to j
        if 0 <= j+s < n:
            γ_new[idx] += Δ * ĥ[j+s]
        if 0 <= j-s < n and s != 0:
            γ_new[idx] += Δ * ĥ[j-s]
    # self-term (s = 0)
    γ_new[n - 1] += -2 * h[i] * Δ + Δ * Δ + 2 * h[j] * Δ + Δ * Δ
    return γ_new

def get_worst_shifts(C, C5, epsilon=1e-5):
    """Return indices of shifts where C[s] >= C5 - epsilon"""
    return np.where(C >= C5 - epsilon)[0]

def get_scores(h, S_w, n):
    """Calculate score for each index based on contribution to worst shifts."""
    scores = np.zeros(n)
    for s in S_w:
        # s corresponds to index s + n - 1 in full correl array
        shift_idx = s + n - 1
        if s >= 0:
            # h[i] * h[i+s]
            for i in range(n - s):
                scores[i] += h[i+s]
            # h[i+s] * h[i]
            for i in range(s, n):
                scores[i] += h[i-s]
        else:
            # s < 0
            for i in range(-s, n):
                scores[i] += h[i+s]
            for i in range(n + s):
                scores[i] += h[i-s]
    return scores
