import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_box_plus_sum(h, n):
    """Project h onto the set {0 ≤ h ≤ 1, sum(h) = n/2} using water filling."""
    # This is a simplified version - more precise implementation might be needed
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    # Simple approach: adjust by adding diff/n to each element
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def smooth_max_surrogate(h, beta, n):
    """Compute smooth max of C_s values with parameter beta."""
    # Compute all C_s values
    C_s = np.zeros(2*n - 1)
    h_conv = np.convolve(h, h, mode='full')
    h_rev_conv = np.convolve(h, 1 - h, mode='full')
    
    # Reverse the h vector for second convolution 
    # Because C_s = 1 - (2/n)*gamma_s, and gamma_s is autocorr
    for s in range(2*n - 1):
        C_s[s] = 1.0 - (2.0/n) * h_conv[s]
    
    # Smooth max with parameter beta (nominal values)
    max_C = np.max(C_s)
    shifted_C = C_s - max_C
    exp_C = np.exp(beta * shifted_C)
    w = exp_C / np.sum(exp_C)
    return w

        if s >= 0:

            grad += w[s+n-1] * (shifted_h + np.roll(h, s))
        
    return - (2.0/n) * grad

def fft_convolve_1d(a, b):
    """Perform FFT-based convolution of 1D arrays."""
    nA = len(a)
    nB = len(b)
    n_fft = 2**int(np.ceil(np.log2(nA + nB - 1)))
    A = fft(a, n_fft)
    B = fft(b, n_fft)
    return ifft(A * B).real[:nA + nB - 1]
