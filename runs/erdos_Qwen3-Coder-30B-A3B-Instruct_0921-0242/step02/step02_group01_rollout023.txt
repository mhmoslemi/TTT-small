import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def project_to_feasible(h, n_points):
    """Ensure sum(h) = n_points/2 with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n_points / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n_points
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # Take real part and maximum of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def build_lp_data(h0):
    """Build LP data from h0."""
    n = len(h0)
    # Auto-correlation
    c0 = ifft(fft(h0) * np.conj(fft(h0))).real
    C0 = c0  # Already circular auto-correlation

    # Build W matrix for shifts
    W = np.zeros((n, n))
    for k in range(n):
        left = np.roll(h0, k)
        right = np.roll(h0, -k)
        W[k, :] = left + right

    A = -W
    b = -n/2 + C0

    return A, b

def solve_lp(A, b, n_points):
    """Solve the LP problem for the given A, b."""
    # Variables: h[0]..h[n-1], t
    t_var = np.ones(1)  # t term in the constraint (A*h + t*1 <= b)
    c = np.zeros(n_points + 1)
    c[-1] = 1  # minimize t
    A_ub = np.column_stack([A, t_var])
    b_ub = b
    bounds = [(0, 1)] * n_points + [(0, None)]  # h in [0,1], t in [0,inf)
    # Add sum constraint
    A_eq = np.zeros((1, n_points + 1))
    A_eq[0, :n_points] = 1  # sum h = n/2
    b_eq = [n_points / 2]
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        h_new = np.clip(res.x[:-1], 0, 1)
        t_new = res.x[-1]
        return h_new, t_new
    else:
        raise ValueError("LP solver failed.")

def ccp_loop(h0, max_iter=30, epsilon=1e-8):
    """Run CCP iterations."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        A, b = build_lp_data(h)
        try:
            h_new, t_new = solve_lp(A, b, len(h))
        except:
            break
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if np.max(np.abs(h_new - h)) < epsilon:
            break
        h = h_new.copy()
    
    return best_h, best_c5

def generate_seeds(n_points):
    """Generate different starting points."""
    seeds = []
    
    # Uniform
    seeds.append(np.ones(n_points) * 0.5)
    
    # Random Dirichlet
    h = np.random.dirichlet(np.ones(n_points))
    h = h * (n_points / 2) / np.sum(h)
    seeds.append(h)
    
    # Binary random
    h = np.zeros(n_points)
    h[:n_points//2] = 1
    np.random.shuffle(h)
    h = h + np.random.uniform(-0.05, 0.05, n_points)
    h = project_to_feasible(h, n_points)
    seeds.append(h)
    
    # Interpolate initial_h_values if available
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h_orig)
        h = project_to_feasible(h, n_points)
        seeds.append(h)
    
    # Sobol-like: quasi-random in simplex
    h = np.random.rand(n_points)
    h = h * (n_points / 2) / np.sum(h)
    h = project_to_feasible(h, n_points)
    seeds.append(h)
    
    return seeds

def projected_gradient_descent(h, max_iter=200):
    """Refine with projected gradient descent."""
    dx = 2.0 / len(h)
    h = h.copy()
    alpha0 = 0.01
    for t in range(max_iter):
        # Find worst shift
        h_fft = fft(h, 2 * len(h))
        ih_fft = fft(1 - h, 2 * len(h))
        corr = ifft(h_fft * np.conj(ih_fft))
        corr_vals = corr.real[1:len(h)]
        worst_shift = np.argmin(corr_vals) + 1
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            left_idx = (i + worst_shift) % len(h)
            right_idx = (i - worst_shift) % len(h)
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        # Update
        alpha = alpha0 / np.sqrt(t + 1)
        h = h - alpha * grad
        h = project_to_feasible(h, len(h))
    return h

def simulated_annealing(h, max_time=120):
    """Polish with simulated annealing."""
    dx = 2.0 / len(h)
    h = h.copy()
    c5 = evaluate_c5(h)
    T = 0.1
    T_min = 1e-6
    alpha = 0.95
    start_time = time.time()
    while T > T_min and (time.time() - start_time) < max_time:
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        if h[i] <= 0 or h[j] >= 1:
            continue
        delta = np.random.uniform(0, min(h[i], 1 - h[j], 0.01))
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            prob = np.exp(-(c5_new - c5) / T)
            if np.random.rand() < prob:
                h = h_new
                c5 = c5_new
        T *= alpha
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 400
    max_ccp_iter = 30
    
    # Multi-start: generate seeds
    seeds = generate_seeds(n_points)
    best_h = None
    best_c5 = float('inf')
    
    for i, h in enumerate(seeds):
        if time.time() - start_time > 0.7 * budget_s:
            break
        try:
            h_best, c5_best = ccp_loop(h, max_iter=max_ccp_iter)
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best.copy()
        except:
            continue
    
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_c5(best_h)
    
    # Gradient descent refinement
    if time.time() - start_time < 0.9 * budget_s:
        h_refined = projected_gradient_descent(best_h)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Simulated Annealing
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        best_h = simulated_annealing(best_h, min(120, remaining_time - 10))
        best_c5 = evaluate_c5(best_h)
    
    # Optional upscale
    if time.time() - start_time < 0.99 * budget_s and n_points == 400:
        n_up = 800
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_up)
        h_up = np.interp(x_new, x_orig, best_h)
        h_up = project_to_feasible(h_up, n_up)
        h_up, c5_up = ccp_loop(h_up, max_iter=5)
        h_up = projected_gradient_descent(h_up, 50)
        if evaluate_c5(h_up) < best_c5:
            best_h = h_up
            best_c5 = evaluate_c5(h_up)
    
    return (best_h, best_c5, n_points)
