import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_box_sum(h, n_points):
    """Project h to the feasible set {0≤h≤1, sum(h)=n/2} using water-filling."""
    h = np.array(h, dtype=float)
    target_sum = n_points / 2.0

    # Sort in descending order to find the water level
    sorted_h = np.sort(h)[::-1]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_h)
    n = len(sorted_h)
    
    # Find the threshold
    threshold = 0.0
    for i in range(n):
        if i == n - 1 or sorted_h[i] - (target_sum - cumsum[i]) / (n - i) > 0:
            threshold = sorted_h[i] - (target_sum - cumsum[i]) / (n - i)
            break
    
    # Clip and scale
    h_proj = np.clip(h - threshold, 0, 1)
    # Scale to meet the sum constraint
    current_sum = np.sum(h_proj)
    scale = target_sum / current_sum
    h_proj *= scale
    
    return h_proj

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_auto_correlation(h):
    """Compute aperiodic autocorrelation via FFT."""
    n = len(h)
    n_fft = 2**int(np.ceil(np.log2(2*n)))
    h_padded = np.pad(h, (0, n_fft - n), 'constant', constant_values=0)
    h_fft = fft(h_padded)
    autocorr_fft = h_fft * np.conj(h_fft)
    autocorr = ifft(autocorr_fft).real[:2*n-1]
    
    return autocorr

def admm_h_update(z, u, t, n_points, rho=1.0, max_iter=30):

    n = n_points

    ones_padded = np.pad(np.ones(n), (0, n_fft - n), 'constant', constant_values=0)

    # Compute shifted version of z - u
    z_u = z - u
    z_u_padded = np.pad(z_u, (0, n_fft - n), 'constant', constant_values=0)
    z_u_fft = fft(z_u_padded)
    
    # Compute initial feasible h
    h = z_u  # Start with z-u (will be corrected)
    
    # Initial bisection bounds for μ
    min_mu = 0.0
    max_mu = 1e10
    
    # Create a vector of 1s for convolution
    ones_vec = np.ones(n)
    
    for _ in range(max_iter):  # Bisection iterations
        mu = (min_mu + max_mu) / 2.0
        
        # h = (z-u) / (1 + μ * 1)  in freq domain
        denominators = 1.0 + mu * ones_fft
        h_fft = z_u_fft / denominators
        h = ifft(h_fft).real[:n]  # Back to time domain
        
        # Modify h (typically a frequency domain operation) to ensure
        # it satisfies h * 1 ≤ t (i.e., the constraints)
        # This is a specific way to compute this based on the constrained
        # optimization problem. We solve the quadratic subproblem directly.
        # But we just check feasibility now.
        # Compute autocorrelation and then max of C_s = 1 - (2/n) * γ_s
        γ = compute_auto_correlation(h)
        max_C = 1.0 - (2.0/n_points) * np.max(γ)
        
        if max_C <= t:
            # This μ is feasible, try smaller
            max_mu = mu
        else:
            # This μ is not feasible, try larger
            min_mu = mu
    
    # Final chosen μ is max_mu
    mu = max_mu
    denominators = 1.0 + mu * ones_fft
    h_fft = z_u_fft / denominators
    h = ifft(h_fft).real[:n]
    
    return h

def admm_z_update(h, u, n_points):
    """Compute z-update by Euclidean projection."""
    z = h + u
    return project_to_box_sum(z, n_points)
