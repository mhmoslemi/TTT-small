import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def build_cyclic_shift_matrix(k, n):
    """Build a sparse cyclic shift matrix P_k where (P_k)_{ij} = 1 if j = (i-k) mod n."""
    P = np.zeros((n, n))
    for i in range(n):
        j = (i - k) % n
        P[i, j] = 1
    return P

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def subgradient_refine(h, dx, budget_s):
    """Refine h with a few gradient descent steps."""
    start = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    max_iter = 2000

    for i in range(max_iter):
        if time.time() - start > budget_s - 0.5:
            break
        corr = correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        if k_star > len(h):
            k_star = k_star - len(h)
        else:
            k_star = k_star - len(h) + 1

        grad = np.zeros_like(h)
        # Compute derivative with respect to k_star
        for j in range(len(h)):
            right_idx = (j + k_star) % len(h)
            left_idx = (j - k_star) % len(h)
            grad[j] = dx * (h[left_idx] - h[right_idx])
        grad = -grad

        alpha = 1.0
        max_backtrack = 10
        for _ in range(max_backtrack):
            h_new = h - alpha * grad
            h_new = project_to_feasible(h_new, len(h))
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                break
            alpha /= 2.0

        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new

    return best_h, best_c5

def upsample_and_project(h_coarse, n_fine):
    """Interpolate h, compute new integral, project onto feasible set."""
    n_coarse = len(h_coarse)
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse, period=2)
    h_fine = project_to_feasible(h_fine, n_fine)
    return h_fine

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    n = 180
    dx = 2.0 / n
    P = [build_cyclic_shift_matrix(k, n) for k in range(n)]

    # SDP
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()
    constraints = [
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n/2,
    ] + [cp.trace(P[k] @ X) >= t for k in range(n)]
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver='SCS', max_iters=25000, eps=1e-5)

    # Extract solution
    X_opt = X.value
    eigvals, eigvecs = np.linalg.eigh(X_opt)
    lead = eigvecs[:, -1] * np.sqrt(eigvals[-1])
    h0 = project_to_feasible(lead, n)

    # Randomized refinement
    best_h = h0
    best_c5 = evaluate_c5(h0)
    for _ in range(15):
        if time.time() - start_time > 0.8 * budget_s:
            break
        g = np.random.randn(n) @ np.linalg.cholesky(X_opt)
        h = project_to_feasible(np.clip(g, 0, 1), n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Local refinement
    remaining_time = 0.5 * budget_s
    best_h, best_c5 = subgradient_refine(best_h, dx, remaining_time)

    # Upsample if possible
    if time.time() - start_time < 0.85 * budget_s:
        n2 = 2 * n
        h_up = upsample_and_project(best_h, n2)
        remaining_time = budget_s - (time.time() - start_time)
        best_h, best_c5 = subgradient_refine(h_up, 2.0/n2, remaining_time)

    return best_h, best_c5, len(best_h)
