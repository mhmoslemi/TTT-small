import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import bisect

def project_box_sum(y, target_sum, lower=0, upper=1):
    """Project y onto the set {x : lower ≤ x ≤ upper, sum(x) = target_sum}."""
    def S(lam):
        return np.sum(np.clip(y - lam, lower, upper))
    
    # Binary search for lambda
    lam_lo, lam_hi = np.min(y) - (target_sum - np.sum(np.clip(y, lower, upper))), np.max(y)
    tol = 1e-12
    while lam_hi - lam_lo > tol:
        lam = (lam_lo + lam_hi) / 2
        s = S(lam)
        if s > target_sum:
            lam_lo = lam
        else:
            lam_hi = lam
    lam = (lam_lo + lam_hi) / 2
    return np.clip(y - lam, lower, upper)

def compute_autocorrelation(h):
    """Compute full autocorrelation of h with itself"""
    return correlate(h, h, mode='full')

def compute_C5_from_autocorr(gamma, n):
    """Compute C5 from autcorrelation gamma"""
    gamma_min = np.min(gamma)
    return 1.0 - (2.0 / n) * gamma_min

def compute_worst_shifts(C, eps=1e-5):
    """Find indices where C is within epsilon of max C"""
    max_C = np.max(C)
    return np.where(np.abs(C - max_C) <= eps)[0]

def compute_score(h, shifts, n):
    """Compute score for each index based on worst shifts"""
    score = np.zeros_like(h)
    for s in shifts:
        # Compute indices corresponding to shift s
        # In full correlation, shift s corresponds to index s + n - 1
        # Positive shift: h[i] contributes to h[i-s]
        # Negative shift: h[i] contributes to h[i+s]
        s_idx = s + n - 1
        if s > 0:
            score[:-s] += h[s:]
        elif s < 0:
            score[-s:] += h[:-s]
        else:
            # self-correlation
            score += h
    return score

def perturb(h, i, j, delta, n):
    """Perturb h by moving delta from i to j, keeping sum constant"""
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def update_autocorr_incremental(h, gamma, i, j, delta, n):
    """Update gamma incrementally for a move from i to j, delta amount"""
    gamma_new = gamma.copy()
    # Temporarily zero out i
    h_tmp = h.copy()
    h_tmp[i] = 0
    h_tmp[j] = 0
    
    # Update each shift entry in O(n)
    for s in range(-n+1, n):
        idx = s + n - 1
        # Remove contribution from h[i]
        if 0 <= i+s < n:
            gamma_new[idx] -= delta * h_tmp[i+s]
        if 0 <= i-s < n and s != 0:
            gamma_new[idx] -= delta * h_tmp[i-s]
        # Add contribution from h[j]
        if 0 <= j+s < n:
            gamma_new[idx] += delta * h_tmp[j+s]
        if 0 <= j-s < n and s != 0:
            gamma_new[idx] += delta * h_tmp[j-s]
    
    # Handle self-term s=0 case separately with quadratic correction
    gamma_new[n-1] = gamma_new[n-1] - 2*h[i]*delta + delta*delta + 2*h[j]*delta + delta*delta
    
    return gamma_new

def get_move(h, worst_shifts, h_min=1e-8, h_max=1-1e-8, score_impl=None):
    """Generate a move (i, j, delta) guided by current worst shifts"""
    n = len(h)
    if score_impl is None:
        score = compute_score(h, worst_shifts, n)
    else:
        score = score_impl

    # Source: indices with high score and beyond threshold
    candidates_i = np.where((h > h_min) & (score > np.percentile(score, 90)))[0]
    if len(candidates_i) == 0:
        candidates_i = np.where(h > h_min)[0]

    # Target: indices with low score and below threshold
    candidates_j = np.where((h < h_max) & (score < np.percentile(score, 10)))[0]
    if len(candidates_j) == 0:
        candidates_j = np.where(h < h_max)[0]

    if len(candidates_i) == 0 or len(candidates_j) == 0:
        # Fallback to random for robustness
        idx = np.random.randint(0, n)
        i = idx if h[idx] > h_min else np.random.choice(np.where(h > h_min)[0])
        j = np.random.choice(np.where(h < h_max)[0])
        delta = np.random.uniform(0, 0.1)
        return i, j, delta

    i = np.random.choice(candidates_i)
    j = np.random.choice(candidates_j)
    delta = np.random.uniform(0, min(h[i], 1 - h[j], 0.1))
    return i, j, delta

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    start_time = time.time()
    n = 300
    dx = 2.0 / n
    eps = 1e-5

    # Initialize
    if 'initial_h_values' in globals() and len(initial_h_values) >= n:
        h_init = np.array(initial_h_values[:n])
    else:
        h_init = np.random.rand(n)
    h_init = h_init * (n / 2) / np.sum(h_init)
    h_init = np.clip(h_init, 0, 1)
    h = h_init.copy()
    gamma = compute_autocorrelation(h)
    C = 1.0 - (2.0 / n) * gamma
    C5 = np.max(C)
    best_C5 = C5
    best_h = h.copy()

    # Phase A: Random seeding
    n_seeds = 5
    h_seeds = []
    C5_seeds = []

    for r in range(n_seeds):
        if time.time() - start_time > 0.05 * budget_s:
            break
        h_new = np.random.rand(n)
        h_new = h_new * (n / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        gamma_new = compute_autocorrelation(h_new)
        C_new = 1.0 - (2.0 / n) * gamma_new
        C5_new = np.max(C_new)
        h_seeds.append(h_new)
        C5_seeds.append(C5_new)

    if len(h_seeds) > 0:
        best_seed_idx = np.argmin(C5_seeds)
        h = h_seeds[best_seed_idx].copy()
        C5 = C5_seeds[best_seed_idx]
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
        gamma = compute_autocorrelation(h)
        C = 1.0 - (2.0 / n) * gamma

    # Main SA phases
    # Phase B: Coarse and rapid exploration
    T0 = 0.3 * (np.max(C) - np.min(C))
    alpha = 0.9995  # For ~200k steps to decay T to ~1e-4

    # Use a variable delta_max
    delta_max = 0.2
    steps_per_phase = int(0.3 * budget_s / (2.0 / 600.0))  # Approximately
    step_count = 0
    target_accept = 0.4

    T = T0
    acc_count = 0
    total_accepted = 0
    step_limit = 150000  # Cap for time management
    decay_time = 5000

    while step_count < step_limit and time.time() - start_time < 0.7 * budget_s:
        step_count += 1

        # Periodic full recomputation
        if step_count % 5000 == 0:
            gamma = compute_autocorrelation(h)
            C = 1.0 - (2.0 / n) * gamma
            C5 = np.max(C)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()

        # Compute worst shifts
        worst_shifts = compute_worst_shifts(C, eps)

        # Compute score vector
        score = compute_score(h, worst_shifts, n)

        # Generate a move
        i, j, delta = get_move(h, worst_shifts, score_impl=score)
        delta = min(delta, delta_max)

        # Create trial h
        h_trial = h.copy()
        h_trial[i] -= delta
        h_trial[j] += delta
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = project_box_sum(h_trial, n/2)

        # Update gamma incrementally
        gamma_trial = update_autocorr_incremental(h, gamma, i, j, delta, n)
        C_trial = 1.0 - (2.0 / n) * gamma_trial
        C5_trial = np.max(C_trial)

        # Accept with probability
        if C5_trial < C5:
            h = h_trial
            C = C_trial
            gamma = gamma_trial
            C5 = C5_trial
            acc_count += 1
            total_accepted += 1
        else:
            prob = np.exp(-(C5_trial - C5) / T)
            if np.random.rand() < prob:
                h = h_trial
                C = C_trial
                gamma = gamma_trial
                C5 = C5_trial
                acc_count += 1
                total_accepted += 1

        # Adjust temperature
        if step_count % decay_time == 0:
            accept_ratio = acc_count / decay_time
            if accept_ratio < 0.25:
                T *= 1.2  # Reheat
            else:
                T *= alpha

        # Check for improvement
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()

        # Gradually decrease delta
        delta_max *= 0.9999

        # Occasionally recompute all
        if step_count % 10000 == 0:
            gamma = compute_autocorrelation(h)
            C = 1.0 - (2.0 / n) * gamma
            C5 = np.max(C)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()

    # Phase C: Focused optimization with smaller deltas
    delta_max = 0.05
    last_update = 0

    steps_per_phase = int(0.45 * budget_s / (2.0 / 600.0))

    for i in range(steps_per_phase):
        if time.time() - start_time > 0.9 * budget_s:
            break

        # Recompute
        if i % 5000 == 0:
            gamma = compute_autocorrelation(h)
            C = 1.0 - (2.0 / n) * gamma
            C5 = np.max(C)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()

        # Compute worst shifts
        worst_shifts = compute_worst_shifts(C, eps)
        score = compute_score(h, worst_shifts, n)

        # Move selection
        i, j, delta = get_move(h, worst_shifts, score_impl=score)
        delta = min(delta, delta_max)

        h_trial = h.copy()
        h_trial[i] -= delta
        h_trial[j] += delta
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = project_box_sum(h_trial, n/2)

        # Incremental update
        gamma_trial = update_autocorr_incremental(h, gamma, i, j, delta, n)
        C_trial = 1.0 - (2.0 / n) * gamma_trial
        C5_trial = np.max(C_trial)

        # Accept with SA
        if C5_trial < C5:
            h = h_trial
            C = C_trial
            gamma = gamma_trial
            C5 = C5_trial
            total_accepted += 1
        else:
            prob = np.exp(-(C5_trial - C5) / T)
            if np.random.rand() < prob:
                h = h_trial
                C = C_trial
                gamma = gamma_trial
                C5 = C5_trial
                total_accepted += 1

        # Adjust delta and decay
        if i % 5000 == 0:
            delta_max *= 0.999

    # Phase E: Exact final polishing
    if time.time() - start_time < 0.95 * budget_s:
        # Subgradient polishing on the exact max objective
        for _ in range(100):
            if time.time() - start_time > 0.98 * budget_s:
                break
            # Compute worst shifts
            worst_shifts = compute_worst_shifts(C, eps)
            if len(worst_shifts) == 0:
                break

            # Gradient estimation via worst shifts
            # g = -(2/n) * mean(shift_{+s}(h) + shift_{-s}(h)) for s in worst_shifts
            g = np.zeros(n)
            for s in worst_shifts:
                # Positive shift
                if s > 0:  # h[i] * h[i-s] contribution
                    g[:-s] -= (2.0 / n) * h[s:]
                elif s < 0:  # h[i] * h[i+s] contribution
                    g[-s:] -= (2.0 / n) * h[:-s]
                else:  # self term, exact derivative
                    g += (2.0 / n) * h

            # Line search for step direction
            h_trial = h - 0.01 * g
            h_trial = project_box_sum(h_trial, n/2)

            # Evaluate new C
            gamma_trial = compute_autocorrelation(h_trial)
            C_trial = 1.0 - (2.0 / n) * gamma_trial
            C5_trial = np.max(C_trial)

            if C5_trial < C5:
                h = h_trial
                C = C_trial
                gamma = gamma_trial
                C5 = C5_trial
                if C5 < best_C5:
                    best_C5 = C5
                    best_h = h.copy()

    # Paper solution with exact evaluation
    gamma_final = compute_autocorrelation(best_h)
    C_final = 1.0 - (2.0 / n) * gamma_final
    final_C5 = np.max(C_final)
    final_h = best_h

    return (final_h, final_C5, n)
