import numpy as np
import cvxpy as cp
import time

def full_to_sym(h_full):
    """Convert full array to symmetric version (first half + middle)"""
    n = len(h_full)
    if n % 2 == 0:
        return np.concatenate([h_full[:n//2], [h_full[n//2-1]]])
    else:
        return h_full[:(n+1)//2]

def sym_to_full(x_sym):
    """Convert symmetric x back to full array with mirror symmetry"""
    n = 2 * (len(x_sym) - 1)
    if n % 2 == 0:
        return np.concatenate([x_sym[:-1], x_sym[::-1]])
    else:
        return np.concatenate([x_sym[:-1], x_sym[::-1][1:]])

def correlation_vals(h_full):
    """Compute correlation values (positive shifts only)"""
    n = len(h_full)
    corr = np.correlate(h_full, 1-h_full, mode='full')
    # index 0 corresponds to shift -(n-1); we keep non-negative shifts
    return corr[n-1:] * (2.0/n)   # dx = 2/n

def compute_shift_constraint(h_sym, shift, dx):
    """Compute linear constraint coefficient matrix for given shift"""
    n_sym = len(h_sym)
    n_full = 2 * (n_sym - 1)
    # For shift s in [0, n-1], indexes of h[i] and h[i+s] in the full vector
    # indices are 0 to n-1
    indices_i = np.arange(n_full - shift)
    indices_j = indices_i + shift
    # Compute linear constraint
    return dx * np.sum(h_sym[indices_i] * (1.0 - h_sym[indices_j]))

def solve_lp(shifts, h0_sym, n, dx):
    """Solve LP for given shifts with symmetry constraint"""
    # Variables: h_sym vector (size n_sym = n//2 +1) and t
    n_sym = n // 2 + 1
    h_sym = cp.Variable(n_sym)
    t = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Bounds
    constraints.extend([0 <= h_sym, h_sym <= 1])
    
    # Integral constraint (sum = n/2)
    constraints.append(cp.sum(h_sym) == 1.0)
    
    # Shift Constraints (max over all shifts)
    # For each shift, we add the constraint
    for s in shifts:
        # This is the correlation of shift s, symmetric
        # We compute h_{i} + h_{i+s} over full domain (after reflecting)
        if s == 0:
            expr = dx * cp.sum(cp.multiply(h_sym, 1 - h_sym))
        else:
            # Indices for shift s:
            i_indices = np.arange(0, n - s)
            j_indices = i_indices + s
            terms = []
            # handle overlap correctly
            for i, j in zip(i_indices, j_indices):
                # Get symmetric indices
                i_sym = i if i < n_sym - 1 else 2*(n_sym-1) - i
                j_sym = j if j < n_sym - 1 else 2*(n_sym-1) - j
                terms.append(h_sym[i_sym] * (1 - h_sym[j_sym]))
            expr = dx * cp.sum(terms)
        constraints.append(expr <= t)
    
    # Optimize
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Attempt to solve with ECOS
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_full_opt = sym_to_full(h_sym.value)
            t_sol = t.value
            return h_full_opt, t_sol
    except:
        pass
    return None, None
