import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cvxpy as cp
from cvxpy import Semidef, Variable, Minimize, Problem, trace

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def evaluate_binary_c5(x):
    """Evaluate C5 for binary vector x (corrected for dx = 2/n)"""
    n = len(x)
    dx = 2.0 / n
    # Compute the autocorrelation for all shifts
    xfft = fft(x, 2 * n)
    ixf = fft(1 - x, 2 * n)
    cor = ifft(xfft * np.conj(ixf))
    cor_vals = cor.real[1:n]  # Exclude shift 0
    max_corr = np.max(cor_vals) * dx
    return max_corr

def compute_autocorrelations_binary(x):
    """Returns full autocorrelation vector for binary x"""
    n = len(x)
    xfft = fft(x, 2 * n)
    ixf = fft(1 - x, 2 * n)
    cor = ifft(xfft * np.conj(ixf))
    return cor.real

def build_sdp_relaxation(n, t, A_k_matrices):
    """Build SDP relaxation to max-min autocorrelation for binary vectors"""
    X = Semidef(n)
    constraints = []
    
    # Diagonal bounds
    for i in range(n):
        constraints.append(X[i,i] >= 0)
        constraints.append(X[i,i] <= 1)
        
    # Cardinality constraint
    constraints.append(cp.sum(X.diagonal()) == n/2)
    
    # Autocorrelation constraints
    for k in range(1, n):
        constraints.append(trace(A_k_matrices[k] @ X) >= t)
        
    # Pairwise bounds to tighten relaxation
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(X[i,j] >= 0)
            constraints.append(X[i,j] <= X[i,i])
            constraints.append(X[i,j] <= X[j,j])
    
    # Objective
    prob = Problem(Maximize(t), constraints)
    return prob, X

def compute_shift_matrix(n, k):
    """Construct matrix A_k with 1s at positions (i, (i+k)%n)"""
    A = np.zeros((n,n))
    for i in range(n):
        j = (i + k) % n
        A[i,j] = 1
    return A

def round_sdp_solution(X_solution, n):
    """Round the SDP solution to binary using random hyperplane rounding"""
    # Compute eigenvalues and eigenvectors
    eigenvals, eigenvecs = np.linalg.eigh(X_solution)
    # Take principal component (largest eigenvalue)
    v = np.sqrt(eigenvals[-1]) * eigenvecs[:,-1]
    
    rounds = []
    # Rounding by random hyperplane with multiple samples
    for rr in range(10):
        # Generate random vector
        g = np.random.randn(n)
        # Project to principal component
        z = eigenvecs @ (eigenvals ** 0.5 * g)
        # Threshold to get binary vector
        x = np.zeros(n)
        idx = np.argsort(z)[::-1][:n//2]
        x[idx] = 1.0
        rounds.append(x)
    
    # Deterministic rounding as backup
    idx = np.argsort(v)[::-1][:n//2]
    x_det = np.zeros(n)
    x_det[idx] = 1.0
    rounds.append(x_det)
    
    return rounds

def local_swap_improvement(x0, n, max_time=10.0):
    """Greedy local search with swaps to improve the minimum autocorrelation"""
    start_time = time.time()
    x = x0.copy()
    X = np.diag(x)  # Note: X is not a full SDP matrix here, just for precompute
    
    # Compute initial correlations
    cor_vals = compute_autocorrelations_binary(x)
    min_cor = np.min(cor_vals)
    worst_shift = np.argmin(cor_vals) + 1
    
    max_swaps = 2000
    swaps = 0
    
    while swaps < max_swaps and time.time() - start_time < max_time:
        # Compute contribution of all 1s and 0s to the worst shift
        c_i = np.zeros(n)  # contribution for each i where x_i = 1
        d_j = np.zeros(n)  # contribution for each j where x_j = 0
        
        for i in range(n):
            if x[i] == 1:
                c_i[i] = x[(i + worst_shift) % n] + x[(i - worst_shift) % n]
        
        for j in range(n):
            if x[j] == 0:
                d_j[j] = x[(j + worst_shift) % n] + x[(j - worst_shift) % n]
        
        if np.max(c_i) <= np.min(d_j):
            break  # No improvement possible
            
        i_star = np.argmax(c_i)
        j_star = np.argmin(d_j)
        
        if d_j[j_star] > c_i[i_star]:
            # Perform swap
            x[i_star] = 0
            x[j_star] = 1
            swaps += 1
            
            # Efficiently update correlations using shifting
            # Update only the affected shifts
            for k in range(1, n):
                # Computing change in R_k for correct shift k
                old_contrib = x[(i_star+k)%n]*x[i_star] + x[(i_star-k)%n]*x[i_star]
                new_contrib = x[(j_star+k)%n]*x[j_star] + x[(j_star-k)%n]*x[j_star]
                cor_vals[k-1] += (new_contrib - old_contrib)
                
            # update min_cor
            if min_cor == cor_vals[worst_shift-1]:
                # has changed, recompute
                min_cor = np.min(cor_vals)
                if min_cor == cor_vals[worst_shift-1]:  # Still equal
                    worst_shift = np.argmin(cor_vals) + 1
                else:
                    min_cor = np.min(cor_vals)
                    worst_shift = np.argmin(cor_vals) + 1
            else:
                min_cor = np.min(cor_vals)  
        else:
            break
    
    return x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed) 
    start_time = time.time()

    # Initial setup 
    n = 200
    dx = 2.0 / n

    # Generate A_k matrices for all shifts k 
    A_k_matrices = {}
    for k in range(1, n):
        A_k_matrices[k] = compute_shift_matrix(n, k)

    # Approximate 180s for SDP solving + rounding 
    sdp_time = 0.3 * budget_s
    sdp_start = time.time()

    # Build and solve SDP relaxation
    t = cp.Variable()
    prob, X = build_sdp_relaxation(n, t, A_k_matrices)
    
    prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
    
    time_used = time.time() - sdp_start
    sdp_time = time_used
    
    # If solver fails return a baseline
    if prob.status != 'optimal': 
        print("SDP did not converge; fallback to basic search")
        X_sol = np.eye(n) * 0.5
        t_opt = 0.0
    else:
        X_sol = X.value
        t_opt = t.value
        print(f"SDP solution t = {t_opt:.4f}")

    # Rounding and Selection (up to 60s remaining)
    round_start = time.time()
    candidates = round_sdp_solution(X_sol, n)
    
    best_cand = candidates[0]
    best_value = evaluate_binary_c5(best_cand)
    
    for x in candidates[1:]:
        val = evaluate_binary_c5(x)
        if val < best_value:
            best_value = val
            best_cand = x
    
    current_cand = best_cand.copy()
    round_time = time.time() - round_start
    
    # Parallel local swap improvement (180s)
    local_start = time.time()
    improved = local_swap_improvement(current_cand, n, max_time=150)
    c5_improved = evaluate_binary_c5(improved)
    
    print(f"Initial C5 estimate: {best_value:.6f}")
    print(f"Improved C5 estimate: {c5_improved:.6f}")
    final_cand = improved
    final_value = c5_improved
    
    # Optional polishing (60s)
    polish_start = time.time()
    
    # Convert to continuous approximation using the binary vector
    h_final = final_cand.astype(float)
    h_final = h_final * (n / 2) / np.sum(h_final)  # Adjust
    h_final = project_to_feasible(h_final, n)
    c5_final = evaluate_c5(h_final)
    
    # Run continuous minimization steps for final refinement
    max_stages = 50
    step_size = 0.2 * dx
    
    for i in range(max_stages):
        if time.time() - polish_start > 59.0 or time.time() - start_time > 0.95 * budget_s:
            break
            
        # Compute worst shift
        corr_vals = compute_autocorrelations_binary(h_final)
        k_star = np.argmin(corr_vals) + 1  # shift index not 0

        # Compute gradient
        g = np.zeros_like(h_final)
        for i in range(n):
            g[i] = dx * ((1 - h_final[(i + k_star) % n]) - h_final[(i - k_star) % n])
        
        # Stepping
        h_new = h_final - step_size * g
        h_new = project_to_feasible(h_new, n)
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5_final:
            h_final = h_new
            c5_final = c5_new
            step_size *= 0.99
        else:
            step_size *= 0.9
        
        # Optional early termination on small improvement
        if abs(c5_final - c5_new) / c5_new < 5e-6:
            break
    
    # Final return
    if np.isnan(c5_final) or np.isinf(c5_final):
        return (np.ones(n) * 0.5, 0.5, n)
    else:
        return (h_final, c5_final, n)
