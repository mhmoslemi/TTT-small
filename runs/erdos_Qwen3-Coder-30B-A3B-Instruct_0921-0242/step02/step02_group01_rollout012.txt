import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from sobol_seq import i4_sobol_generate

def project_to_feasible(h, target_sum):
    """Ensure that sum(h) = target_sum, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum

    if abs(diff) < 1e-10:
        return h

    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def build_lp_data(h0):
    """Build the linear programming data for the current iterate h0."""
    n = len(h0)
    # Compute auto-correlation C_k = sum h_i h_{i+k}
    c_fft = fft(h0) * np.conj(fft(h0))
    c = ifft(c_fft).real
    C0 = c[:n//2+1]  # Only first half due to symmetry

    # Build W matrix where each row corresponds to a shift
    W = np.zeros((n//2, n))
    for k in range(n//2):
        W[k, :] = h0[(np.arange(n) + k) % n] + h0[(np.arange(n) - k) % n]

    A = -W  
    b = -n/2 + C0[:n//2]

    return A, b

def solve_ccp_step(A, b, target_sum):
    """Solve the CCP linear program."""
    n = A.shape[1]
    c = np.ones(n + 1)  # Minimize t
    c[-1] = 0  # t coefficient
    # A @ [h; t] <= b
    # constraint sum: np.ones(n) @ h == target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = np.array([target_sum])

    # Construct the constraint matrix
    A_ub = np.hstack([A, np.ones((A.shape[0], 1))])  # Add column of ones for t
    b_ub = b

    bounds = [(0, 1)] * n + [(0, None)]  # h in [0,1], t >= 0

    # Solve using high performance method
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        h_new = res.x[:-1]  # extract h
        t_new = res.x[-1]   # extract t
        return h_new, t_new
    else:
        raise RuntimeError("Optimizer failed to solve LP")

def generate_seeds(n, num_seeds):
    """Generate seed h vectors."""
    seeds = []
    # 1. Uniform
    h = np.ones(n) * 0.5
    h = project_to_feasible(h, n/2)
    seeds.append(h.copy())

    # 2. Random Dirichlet
    if num_seeds > 1:
        r = np.random.dirichlet([1]*n)
        h = r * (n/2) / np.sum(r)
        h = project_to_feasible(h, n/2)
        seeds.append(h.copy())

    # 3. Binary random
    if num_seeds > 2:
        h = np.zeros(n)
        h[:n//2] = 1
        np.random.shuffle(h)
        # Add noise and reproject
        h += np.random.normal(0, 0.01, n)
        h = project_to_feasible(h, n/2)
        seeds.append(h.copy())

    # 4. Interpolated initial values
    if num_seeds > 3 and 'initial_h_values' in globals():
        h_orig = initial_h_values
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h_orig)), h_orig)
        h = project_to_feasible(h, n/2)
        seeds.append(h.copy())

    # 5. Sobol sample
    if num_seeds > 4:
        try:
            # Generate Sobol samples in the unit simplex
            Sobol_samples = i4_sobol_generate(n, num_seeds - 4)
            # Transform to simplex
            for i in range(num_seeds - 4):
                # cumulative sum trick to put in simplex
                s = np.sort(Sobol_samples[i, :])
                s = np.append(s, 1.0)
                x = np.diff(s)
                h = x * (n/2) / np.sum(x)
                h = project_to_feasible(h, n/2)
                seeds.append(h.copy())
        except:
            pass

    if len(seeds) < num_seeds:
        # Fill with random uniform
        for _ in range(num_seeds - len(seeds)):
            h = np.random.rand(n)
            h = h * (n/2) / np.sum(h)
            h = project_to_feasible(h, n/2)
            seeds.append(h.copy())

    return seeds[:num_seeds]

def refine_with_gradient(h, max_iter=200):
    """Projected gradient descent refinement."""
    dx = 2.0 / len(h)
    for _ in range(max_iter):
        # Find worst shift
        h_fft = fft(h, 2 * len(h))
        ih_fft = fft(1 - h, 2 * len(h))
        corr = ifft(h_fft * np.conj(ih_fft))
        corr_vals = corr.real[1:len(h)]
        k_star = np.argmin(corr_vals) + 1
        
        # Compute gradient 
        grad = np.zeros_like(h)
        for i in range(len(h)):
            left_idx = (i + k_star) % len(h)
            right_idx = (i - k_star) % len(h)
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Take step
        alpha = 0.1 / np.sqrt(_ + 1)
        h = h - alpha * grad
        
        # Project back to feasible set
        h = project_to_feasible(h, len(h)/2)
    
    return h

def anneal_sequential(h, time_budget, start_time):
    """Simulated-annealing based polishing."""
    dx = 2.0 / len(h)
    T = 0.1
    T_min = 1e-6
    decay_rate = 0.999
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    while time.time() - start_time < time_budget:
        T *= decay_rate
        if T < T_min:
            break
            
        # Pick indices to swap
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        if h[i] <= 0 or h[j] >= 1:
            continue
            
        # Try move
        delta = min(0.01, h[i], 1 - h[j])
        h_test = h.copy()
        h_test[i] -= delta
        h_test[j] += delta
        h_test = project_to_feasible(h_test, len(h) / 2)
        c5_test = evaluate_c5(h_test)
        
        if c5_test < current_c5 or np.random.rand() < np.exp(-(c5_test - current_c5) / T):
            h = h_test
            current_c5 = c5_test
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 400
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    best_c5 = float('inf')
    best_h = None
    
    # Multi-start with different seeds
    num_seeds = 5
    seeds = generate_seeds(n_points, num_seeds)
    
    max_ccp_iter = 30
    time_per_seed = budget_s * 0.7 / num_seeds
    for i, start_h in enumerate(seeds):
        h = start_h.copy()
        for iter_count in range(max_ccp_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
            
            # Build LP data
            A, b = build_lp_data(h)
            
            # Solve
            try:
                h_new, t_new = solve_ccp_step(A, b, target_sum)
                h = project_to_feasible(h_new, target_sum)
            except:
                break
            
            # Check convergence or stop too early
            if iter_count > 0 and np.max(np.abs(h - prev_h)) < 1e-8:
                break
            prev_h = h.copy()
            
            # Evaluate true C5
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Refine with gradient descent
    if best_h is not None and time.time() - start_time < budget_s * 0.95:
        h_refined = refine_with_gradient(best_h, 200)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Polish with simulated-annealing
    if best_h is not None and time.time() - start_time < budget_s * 0.995:
        h_final, c5_final = anneal_sequential(best_h, budget_s, start_time)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Optional upscale and final refinement
    if best_h is not None and time.time() - start_time < budget_s * 0.999:
        # Upsample to 800 points
        n_up = 800
        h_up = np.interp(np.linspace(0, 2, n_up), np.linspace(0, 2, len(best_h)), best_h)
        h_up = project_to_feasible(h_up, n_up / 2)
        
        # CCP pass with 5 iterations
        for _ in range(5):
            if time.time() - start_time > budget_s * 0.999:
                break
            A, b = build_lp_data(h_up)
            try:
                h_up_new, t_new = solve_ccp_step(A, b, n_up / 2)
                h_up = project_to_feasible(h_up_new, n_up / 2)
            except:
                break
            
        # Gradient refinement
        h_up = refine_with_gradient(h_up, 100)
        c5_up = evaluate_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up
    
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, n_points)
