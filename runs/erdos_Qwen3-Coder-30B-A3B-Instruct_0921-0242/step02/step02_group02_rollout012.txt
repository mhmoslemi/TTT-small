import numpy as np
import time
from scipy.optimize import differential_evolution, linprog
from scipy.fft import fft, ifft
from scipy.interpolate import interp1d

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def basis_evaluation(n_points, m_max):
    """Constructs cosine and sine basis functions modulo 2."""
    x = np.linspace(0, 2, n_points, endpoint=False)
    basis_cos = []
    basis_sin = []
    for m in range(1, m_max + 1):
        basis_cos.append(np.cos(m * np.pi * x))
        basis_sin.append(np.sin(m * np.pi * x))
    return np.array(basis_cos), np.array(basis_sin)

def construct_h_from_coeffs(c, basis_cos, basis_sin, n_points):
    """Constructs h(x) from coefficients."""
    # a0 = 0.5 because the integral must be 1
    a0 = 0.5
    am = c[::2]
    bm = c[1::2]

    # Build h(x) as a linear combination of basis functions
    h_raw = a0 + basis_cos.T @ am + basis_sin.T @ bm
    h_raw = np.clip(h_raw, 0, 1)

    # Scale to enforce integral = n_points / 2
    integral = np.trapz(h_raw, dx=2/n_points)
    if integral > 0:
        scale = (n_points / 2) / integral
        h_raw = h_raw * scale
    else:
        h_raw = np.ones_like(h_raw) * (n_points / 2) / n_points

    return h_raw

def fitness_function(c, basis_cos, basis_sin, n_points):
    """Fitness function used by DE."""
    h = construct_h_from_coeffs(c, basis_cos, basis_sin, n_points)
    return evaluate_c5(h)

def linearise_and_solve_lp(h, k_star, n_points):
    """Solve the linearised LP subproblem."""
    # Solve the LP: maximize t subject to:
    # A(k) >= t for k in a few active shifts
    # 0 <= h_i <= 1
    # sum h_i = n_points / 2
    
    # Get all gradients and at least k_star
    active_shifts = [k_star]
    # Add worst few shifts
    A = np.correlate(h, h, mode='full')
    # For shift k, gradient is 2*roll(h, k) or 2*roll(h, -k)
    A = A[n_points-1:]
    if len(A) > 8:
        worst_k = np.argsort(A)[:8]
        active_shifts = sorted(set(active_shifts + list(worst_k)))
    
    # Build constraint matrix
    A_ub = []
    b_ub = []
    for k in active_shifts:
        grad = 2 * np.roll(h, k)
        row = grad
        A_ub.append(-row)
        b_ub.append(-A[k])
    
    # Bounds on h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective: maximize t -> minimize -t
    # So the linear program to solve is:
    # minimize c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds(x)
    # We expand x to include t at the end
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # We set up the full LP with t, but we dynamically add slacks
    # So we work with h and t variables separately
    
    # Virtual constraint matrix for h bounds and integral constraint
    A_eq = np.zeros((2, n_points + 1))
    A_eq[0, :n_points] = 1  # integral constraint
    A_eq[1, :n_points] = 0  # invalid
    A_eq[0, n_points] = 0  # t not part of integral
    b_eq = [n_points/2, 0]
    
    # Do the calculation with linear search for t (slower but safe)
    # For now, we'll simplify and solve for without explicit variable t
    
    # Instead we use direct optimization with solvers
    # This section uses `linprog` in standard form, so we adapt:
    
    # Let us declare the final simpler problem:
    # We optimize t directly with maximum A(k) >= t.
    # Constraints are additional ones (if too overloaded, we limit number)
    
    # Improved formulation
    identity = np.eye(n_points)
    A_in = np.zeros((len(active_shifts) + 1, n_points + 1))
    b_in = []
    for i, k in enumerate(active_shifts):
        # Gradient for constraint i is 2*roll(h, k)
        grad = 2 * np.roll(h, k)
        A_in[i, :n_points] = -grad  # constraint: -grad^T * h <= -t => grad^T * h >= t
        # t appear with coeff -1 in inequality, in addition to h we add last col
        A_in[i, n_points] = 1  
        b_in.append(-np.dot(grad, h))
    
    # Force integral constraint, 0 <= h_i <= 1
    A_in[len(active_shifts)] = [1]*n_points + [0]
    b_in.append(n_points/2)
    
    # Minimize -t -> maximize t
    c = [0]*n_points + [-1]
    
    try:
        res = linprog(c, A_ub=A_in, b_ub=b_in, bounds=[(0, 1)]*n_points + [None], method='highs')
        if res.success:
            t = -res.fun
            if np.isnan(t):
                t = 0
            return t
        else:
            return 0
    except:
        return 0

def solve_with_slp(h, max_iter=300):
    """Sequential Linear Programming refinement."""
    n_points = len(h)
    h_new = h.copy()
    
    for _ in range(max_iter):
        # Compute A, get k where it's minimal
        A = np.correlate(h_new, h_new, mode='full')[n_points-1:]
        k_star = np.argmin(A)
        t_new = A[k_star]
        
        if t_new <= 0.0001:
            break

        # Solve linearized problem to get improvement direction
        t_approx = linearise_and_solve_lp(h_new, k_star, n_points)
        if t_approx < t_new:
            # This is just a fallback to avoid worse solutions
            break

        # Now try to compute a better h
        try:
            old_h = h_new.copy()
            # Build shifted gradients
            grad = 2 * np.roll(h_new, k_star)
            # Simple line search - just move towards min in direction of gradient
            step_size = 0.01
            h_trial = h_new + step_size * grad
            # Project to feasible
            h_trial = project_to_feasible(h_trial, n_points)
            # Evaluate new C5
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < evaluate_c5(h_new):
                h_new = h_trial
            else:
                # Damped step
                alpha = 0.5
                while alpha > 0.001:
                    h_trial = (1 - alpha) * h_new + alpha * h_new
                    h_trial = project_to_feasible(h_trial, n_points)
                    c5_trial = evaluate_c5(h_trial)
                    if c5_trial < evaluate_c5(h_new):
                        h_new = h_trial
                        break
                    alpha *= 0.5
                    
        except Exception:
            break
    
    return h_new
