import numpy as np
import time
from scipy.signal import correlate
from scipy.ndimage import gaussian_filter1d
import cvxpy as cp

def aperiodic_autocorr(h):
    """Compute aperiodic autocorrelation using FFT."""
    n = len(h)
    # Pad to double length for aperiodic
    h_padded = np.pad(h, (0, n-1), mode='constant')
    corr = correlate(h_padded, h, mode='valid')
    return corr[:n]

def softmax(x, beta=1.0):
    """Numerically stable softmax with temperature beta."""
    x = np.asarray(x)
    x_shifted = x - np.max(x)
    exp_x = np.exp(beta * x_shifted)
    return exp_x / np.sum(exp_x)

def conv(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    n_total = 2 * n - 1
    A = np.fft.fft(a, n_total)
    B = np.fft.fft(b, n_total)
    return np.fft.ifft(A * B).real[:n]

def project_box_sum(h, total):
    """Project onto box + sum constraints using waterfilling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - total) < 1e-10:
        return h
    
    # Waterfilling algorithm
    sorted_h = np.sort(h)[::-1]
    total_diff = total - current_sum
    
    if total_diff > 0:  # Need to increase
        # Find where to cut the top
        cumsum = np.cumsum(sorted_h)
        for i in range(len(sorted_h)):
            if (i + 1) * sorted_h[i] - cumsum[i] >= total_diff:
                lambda_val = (total_diff + cumsum[i]) / (i + 1)
                break
        else:
            lambda_val = sorted_h[-1] + total_diff / len(sorted_h)
    else:  # Need to decrease
        lambda_val = sorted_h[-1] + total_diff / len(sorted_h)
    
    h_proj = np.clip(h - lambda_val, 0, 1)
    return h_proj

def score_guided_swap(h, eps=1e-5):
    """Performs a deterministic swap based on score-based heuristic."""
    n = len(h)
    eval_c = 1 - 2 * aperiodic_autocorr(h) / n
    max_c = np.max(eval_c)
    
    # Get worst indices
    worst_indices = np.where(np.abs(eval_c - max_c) < eps)[0]
    
    if len(worst_indices) == 0:
        return h
    
    # Compute scores (score for each index as potential source)
    scores = np.zeros(n)
    for s in worst_indices:
        # Shifted indices (wrap)
        forward = (np.arange(n) + s) % n
        backward = (np.arange(n) - s) % n
        scores += h[forward] + h[backward]
    
    # Pick a large-score source and a small-score target
    src_idx = np.argmax(scores)
    weights = np.ones(n)
    weights[src_idx] = 0
    tgt_idx = np.argmin(scores + weights * 1e10)  # Large penalty for src
    delta = min(0.01, h[src_idx], 1 - h[tgt_idx])
    
    if delta < 1e-8:
        return h
    
    h[src_idx] -= delta
    h[tgt_idx] += delta
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # --- Phase I: Binary MILP --- 
    n1 = 120
    half = n1 // 2
    
    # Create variables for binary solution
    b = cp.Variable(n1, boolean=True)
    
    # Variables for quadratic terms
    a = cp.Variable((n1, n1), boolean=True)
    
    # Compute γ_s directly from sum of products
    γ = cp.Variable(n1)
    L = cp.Variable()
    
    # Constraints for binary variables
    constraints = [cp.sum(b) == half]
    
    # McCormick constraints for a[i,s]
    for i in range(n1):
        for s in range(n1 - i):
            # a_{i,s} <= b_i
            constraints.append(a[i, s] <= b[i])
            # a_{i,s} <= b_{i+s}
            constraints.append(a[i, s] <= b[i + s])
            # a_{i,s} >= b_i + b_{i+s} - 1
            constraints.append(a[i, s] >= b[i] + b[i + s] - 1)
    
    # Convert autocorrelations from products
    for s in range(n1):
        # Only sum over valid i where i+s < n1, for proper aperiodic definition
        row_sum = cp.sum([a[i, s] for i in range(n1 - s)])
        constraints.append(γ[s] == row_sum)
    
    # Outcome lower bounds on all γ_s:
    constraints += [γ[s] >= L for s in range(n1)]
    
    # Objective
    objective = cp.Maximize(L)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    prob.solve(solver=cp.CBC, time_limit=0.2 * budget_s, mip_gap=5e-3)
    
    # Extract best binary solution
    b_star = np.round(b.value).astype(int)
    
    # --- Phase II: Upscaling and Smoothing ---
    n2 = 300
    x1 = np.linspace(0, 2, n1, endpoint=False)
    x2 = np.linspace(0, 2, n2, endpoint=False)
    h0 = np.interp(x2, x1, b_star)
    
    # Ensure the sum constraint
    h0 = project_box_sum(h0, total=n2/2)
    
    # Optional smoothing
    h0 = gaussian_filter1d(h0, sigma=1.0)
    h0 = project_box_sum(h0, total=n2/2)
    
    # Initial descent
    h = h0.copy()
    c5 = 1 - 2 * np.max(aperiodic_autocorr(h)) / n2
    
    # --- Phase III: Smooth-max projected gradient descent ---
    beta = 2.0
    eta = 1.0
    best_h = h.copy()
    best_c5 = c5
    
    iteration = 0
    while (time.time() - start_time) < 0.85 * budget_s:
        if iteration % 150 == 0:
            beta = min(beta * 1.5, 200)
        
        # Autocorrelation for current h
        γ = aperiodic_autocorr(h)
        C = 1 - (2 / n2) * γ
        w = softmax(C, beta=beta)
        
        # Gradient computation
        conv1 = conv(w, h)
        conv2 = conv(w[::-1], h)
        grad = -(2 / n2) * (conv1 + conv2)
        
        # Backtracking line search
        alpha = eta
        h_trial = h - alpha * grad
        h_trial = project_box_sum(h_trial, total=n2/2)
        
        while True:
            c5_trial = 1 - 2 * np.max(aperiodic_autocorr(h_trial)) / n2
            if c5_trial <= c5:
                break
            alpha *= 0.5
            if alpha < 1e-8:
                break
            h_trial = h - alpha * grad
            h_trial = project_box_sum(h_trial, total=n2/2)
        
        h = h_trial
        c5 = 1 - 2 * np.max(aperiodic_autocorr(h)) / n2
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Occasionally perform score guided swaps
        if iteration % 200 == 0:
            h = score_guided_swap(h)
            c5 = 1 - 2 * np.max(aperiodic_autocorr(h)) / n2
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        iteration += 1
    
    # --- Phase IV: Final Sub-gradient Polishing ---
    while (time.time() - start_time) < budget_s - 0.1 * budget_s:
        γ = aperiodic_autocorr(h)  
        C = 1 - (2 / n2) * γ
        max_c = np.max(C)
        indices = np.where(np.abs(C - max_c) < 1e-5)[0]
        
        if len(indices) == 0:
            break
            
        # Subgradient computation
        sub_grad = np.zeros_like(h)
        for s in indices:
            # Forward & backward shift
            h_shifted = np.roll(h, shift=-s)
            h_rev_shifted = np.roll(h, shift=s)
            sub_grad += h_shifted + h_rev_shifted
        
        sub_grad = -(2 / n2) * sub_grad / len(indices)
        
        # Take a step with diminishing learning rate
        eta = 1.0 / np.sqrt(iteration + 1)
        h_new = h - eta * sub_grad
        h_new = project_box_sum(h_new, total=n2/2)
        
        c5_new = 1 - 2 * np.max(aperiodic_autocorr(h_new)) / n2
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        iteration += 1
    
    return (best_h, best_c5, n2)
