import numpy as np
import time
from scipy.signal import correlate

from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def smooth_max(h, p):
    """Smooth approximation of max using Lp norm."""
    dx = 2.0 / len(h)
    # Compute overlaps
    H = fft(h, 2 * len(h))
    IH = fft(1 - h, 2 * len(h))
    corr = ifft(H * np.conj(IH))
    overlaps = corr.real[:len(h)] * dx
    
    # Avoid numerical issues at zero
    overlaps = np.maximum(overlaps, 1e-12)
    
    # Lp norm
    sum_p = np.sum(overlaps ** p)
    return sum_p ** (1.0 / p)

def compute_gradient_simple(h, corr_vals, p):
    """Approximate gradient using a combinatorial approach."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Note: corr_vals = [C(0), C(1), ..., C(n-1)] where C(k) = dx * sum h_i * h_{i+k}
    # But C(k) = 1 - dx * sum h_i * h_{i+k}
    # So we need derivation: dC(k)/dh_i = -dx * (h_{i+k} + h_{i-k}) (as indices cyclic)
    # And dS_p/dh_i = 1/S_p^(p-1) * sum_k w_k * dC(k)/dh_i where w_k = (1 - C(k))^(p-1) 
    
    # But we instead construct a more direct approach for numerical stability:
    # Use a vectorized approach by handling each contribution to gradient.
    
    # If we take all overlaps C(k) = 1 - dx * sum h_i * h_{(i+k) % n}
    # then for C(k):
    # dC(k)/dh_i = -dx * h_{(i+k)%n} 
    # dC(k)/dh_i = 0 if there is a roll
    # But same for h_{i-k}: we should consider the directional case properly
    
    # Kept simple: have warnings in eval vs using our approach
    # We proceed with full correlate approach but encoded optimization

    # instead compute sorted ones:
    # Compute matrix G_{k,i} of dC_k/dh_i = -dx * (h_{i+k} + h_{i-k}) 
    # Gradient is: dS/dh_i = sum_k w_k * G_{k,i} = sum_k w_k * (h_{i+k} + h_{i-k})
    
    w = np.power(1 - corr_vals, p - 1)
    norm_factor = np.sum(w) ** (1 - 1.0/p)  # S_p^(p-1)
    
    # But can be simplified:
    # For all k invert the operation, and stack the two components trick:
    
    # Instead of constructs, we just code the following:
    # If sum_k w_k * (h_{k+i} + h_{i-k})
    # = sum_k w_k h_{k+i} + sum_k w_k h_{i-k}
    # = sum_k w_k h_mod(k+i) + sum_k w_k h_mod(i-k)

    # To compute this in efficient scratch loop on h gradients:
    # For each i in range(n):
    # we added  w_k * h_{(i+k)%n}  +  w_k * h_{(i-k)%n} = \sum -dx w_k (h_{i+k} + h_{i-k})
    
    # For simplicity, 2 steps:
    # First we compute indices: 
    # test forward case:
    h_back = np.roll(h, 1)
    h_forward = np.roll(h, -1)
    
    # Precompute for the all shifts 
    for i in range(n):
        corr_index = i % n
        # For the derivative wrt h[i], sum over all k:
        # We're off by a factor dx
        # That is: 
        temp_grad = 0
        # Add integrand for each k:
        for k in range(n):
            temp_grad += w[k] * (-dx) * (h[(i+k)%n] + h[(i-k)%n]) # by circular-shift
        grad[i] = temp_grad / norm_factor
    
    return grad

def refine_exact_worst_shift(h, max_iter=100):
    """Apply exact worst shift refinement as described."""
    dx = 2.0 / len(h)
    for it in range(max_iter):
        # Compute actual overlaps
        corr = correlate(h, 1 - h, mode="full")
        max_corr_idx = np.argmax(corr)
        # We have: max corr at `max_corr_idx` in 0..2n-2
        # This is k = max_corr_idx - n 
        # or if max_corr_idx >= n  then k = max_corr_idx - n  
        # or if k < 0 then k = max_corr_idx  
        c0 = 0 if max_corr_idx < n else max_corr_idx - n
        if max_corr_idx < (n):
            k_worst = max_corr_idx
        else:
            k_worst = max_corr_idx - n
            
        # Alternative way to get k_worst
        # Unroll the logic (should be even).
        # print(f"max on full correlation at index {max_corr_idx}, k =  {k_worst}")  
        # Gradient for this shift k
        gradient_shift = np.zeros_like(h)
        if k_worst != 0:
            # For k=0, h[i] * h[i] term  
            h_shifted = np.roll(h, k_worst)
            h_shifted_rev = np.roll(h, -k_worst)
            gradient_shift = dx * (-h_shifted - h_shifted_rev)
        else:
            gradient_shift = dx * (-h - h)  # For (i+k) = (i+0) case
            gradient_shift = -2.0 * dx * h   
        
        # Update step with line search  
        step_size = 0.1 * dx
        # Update h in direction:
        h_new = h - step_size * gradient_shift
        h_new = project_to_feasible(h_new, len(h))
        
        # Test improvement:
        c5_new = evaluate_c5_exact(h_new)
        c5_old = evaluate_c5_exact(h)
        
        # If improvement, accept step.
        if c5_new < c5_old - 1e-12:
            h = h_new
        else:
            # Backtrack
            step_size *= 0.5
            h_new = h - step_size * gradient_shift
            h_new = project_to_feasible(h_new, len(h))
            c5_new = evaluate_c5_exact(h_new)
            if c5_new < c5_old - 1e-12:
                h = h_new
            else:
                break
                
    return h

def generate_seeds(n, num_seeds=30):
    """Generate a variety of feasible initial points."""
    seeds = []
    # Uniform vector
    seeds.append(np.ones(n) * 0.5)
    
    # Random Dirichlet
    seeds.append(np.random.dirichlet(np.ones(n)) * n / 2)
    
    # Interpolated initial_h_values if available
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        x_orig = np.linspace(0, 2, len(orig_h))
        x_new = np.linspace(0, 2, n)
        h_interped = np.interp(x_new, x_orig, orig_h)
        h_interped = h_interped * (n / 2) / np.sum(h_interped)
        h_interped = project_to_feasible(h_interped, n)
        seeds.append(h_interped)
    
    # Alternate binary patterns (approx of sum = n/2)  
    h_binary = np.zeros(n)
    for i in range(n // 2):
        h_binary[i] = 1.0
    h_binary = project_to_feasible(h_binary, n)
    seeds.append(h_binary)

    # Inject noise into binary
    h_noise = h_binary + 0.01 * np.random.randn(n)
    h_noise = project_to_feasible(h_noise, n)
    seeds.append(h_noise)
    
    # Fill some
    h_fill = np.full(n, 0.2)
    h_fill[5:10] = 0.8
    h_fill = project_to_feasible(h_fill, n)
    seeds.append(h_fill)
    
    # Random initializes
    for _ in range(num_seeds - len(seeds)):
        h = np.random.rand(n)
        h = h * (n / 2) / np.sum(h)
        h = project_to_feasible(h, n)
        seeds.append(h)
    
    return seeds

def compute_smooth_surrogate(h, p):
    dx = 2.0 / len(h)
    H = fft(h, 2 * len(h))
    IH = fft(1 - h, 2 * len(h))
    corr = ifft(H * np.conj(IH))
    overlaps = corr.real[:len(h)] * dx
    return np.mean(overlaps ** p) ** (1.0 / p)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 350
    if n > 1000:
        n = 1000
    dx = 2.0 / n

    # Generate several starting points  
    seeds = generate_seeds(n, num_seeds=30)
    
    best_h = None
    best_c5 = np.inf

    # Homotopy along p parameter
    p_vals = [8, 16, 32, 64, 128, 256]
    
    # Iterate each start with p progression
    for i_seed, h0 in enumerate(seeds):
        if time.time() - start_time > budget_s * 0.7:
            break
        h_cur = h0.copy()
        best_in_seed = None
        best_c5_in_seed = np.inf
        
        for p in p_vals:
            if time.time() - start_time > budget_s * 0.7:
                break
                
            # Simple callback to keep track of time
            opt_kwargs = {
                "maxiter": 150,
                "method": "L-BFGS-B",
                "bounds": [(0,1)] * n,
                "options": {"ftol": 1e-9, "gtol": 1e-9}
            }
            
            def obj_and_grad(h):
                h = np.clip(h, 0, 1)
                # Compute overlaps again (once per function call)
                H = fft(h, 2 * n)
                IH = fft(1 - h, 2 * n)
                corr = ifft(H * np.conj(IH))
                overlaps = corr.real[:n] * dx
                S = np.sum(overlaps ** p) ** (1.0 / p)
                # Gradient via finite difference
                grad = np.zeros_like(h)
                step = 1e-4
                for i in range(n):
                    h_temp = h.copy()
                    h_temp[i] += step
                    h_temp = np.clip(h_temp, 0, 1)
                    H2 = fft(h_temp, 2 * n)
                    IH2 = fft(1 - h_temp, 2 * n)
                    corr2 = ifft(H2 * np.conj(IH2))
                    overlaps2 = corr2.real[:n] * dx
                    S2 = np.sum(overlaps2 ** p) ** (1.0 / p)
                    grad[i] = (S2 - S) / step
                
                return S, grad
                
            try:
                result = minimize(obj_and_grad, h_cur, jac=True, 
                                bounds=[(0,1)]*n, method='L-BFGS-B', 
                                options={'maxiter': 100})
                if result.success:
                    # Project to feasible and recompute
                    h_result = project_to_feasible(result.x, n)
                    c5 = evaluate_c5_exact(h_result)
                    if c5 < best_c5_in_seed:
                        best_c5_in_seed = c5
                        best_in_seed = h_result
                    
                    h_cur = h_result  # Use as seed for next p
                else:
                    h_cur = h_cur  # Just continue with current
            except Exception as e:
                h_cur = h_cur  # fallback
                
        if best_in_seed is not None and best_c5_in_seed < best_c5:
            best_c5 = best_c5_in_seed
            best_h = best_in_seed.copy()
    
    if best_h is None:
        # fallback with initial seed
        if 'initial_h_values' in globals():
            orig_h = initial_h_values  
        else:
            orig_h = np.ones(300) * 0.5
            
        n_points = min(350, len(orig_h))
        if len(orig_h) > n_points:
            h_start = orig_h[:n_points]
        else:
            x_orig = np.linspace(0, 2, len(orig_h))
            x_new = np.linspace(0, 2, n_points)
            h_start = np.interp(x_new, x_orig, orig_h)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        h_start = project_to_feasible(h_start, n_points)
        best_h = h_start
        best_c5 = evaluate_c5_exact(best_h)

    # Now apply deterministic worst-shift refinement
    if time.time() - start_time < budget_s * 0.9:
        h_optim = refine_exact_worst_shift(best_h, 100)
        c5_optim = evaluate_c5_exact(h_optim)
        if c5_optim < best_c5:
            best_c5 = c5_optim
            best_h = h_optim

    # Optional stochastic polishing
    if time.time() - start_time < budget_s * 0.98:
        h_final = best_h.copy()
        c5_final = best_c5
        T0 = 1.0
        T = T0
        for _ in range(5000):
            if time.time() - start_time > budget_s * 0.98:
                break
                
            i, j = np.random.randint(0, n), np.random.randint(0, n)
            if h_final[i] <= 0 or h_final[j] >= 1:
                continue
            delta = np.random.uniform(0, min(h_final[i], 1 - h_final[j], 0.01))
            if delta < 1e-5:
                continue
                
            h_try = h_final.copy()
            h_try[i] -= delta
            h_try[j] += delta
            h_try = project_to_feasible(h_try, n)
            c5_try = evaluate_c5_exact(h_try)
            if c5_try < c5_final:
                c5_final = c5_try
                h_final = h_try
            else:
                acc_prob = np.exp(-(c5_try - c5_final) / T)
                if np.random.rand() < acc_prob:
                    c5_final = c5_try
                    h_final = h_try
            T *= 0.999
            
        best_c5 = c5_final
        best_h = h_final

    return (best_h, best_c5, n)
