import numpy as np
import time

def project_box_sum(y, total):
    """Project y onto the set {x : 0 <= x <= 1, sum(x) = total}."""
    y = np.asarray(y)
    n = len(y)
    # Sort y in descending order to compute the bisection efficiently
    sorted_y = np.sort(y)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_y)
    # Find the threshold lambda
    # We want to find lambda such that sum(clamp(y_i - lambda, 0, 1)) = total
    # Lambda is between 0 and 1, and the lambda that satisfies the constraint
    # makes some prefix of sorted_y > lambda and remaining <= lambda
    # So we find the largest k such that:
    # sum_1^k (y[i] - lambda) + sum_{k+1}^n 0 >= total, where the first k are > lambda
    # or sum_1^k y[i] - k * lambda >= total (since 0 <= lambda <= y[i] in the first k)
    # sum_1^k y[i] >= total + k * lambda -- lambda <= (sum_1^k y[i] - total) / k
    # (Sum of y_i's that go through the line in the first k elements)
    # But we also want sum_1^k y[i] - k * lambda <= total
    # Therefore: sum_1^k y[i] - total <= k * lambda <= sum_1^k y[i] - total -- contradiction
    # Let's reframe:
    # For a λ, let k be such that y[k] >= λ > y[k+1]. 
    # Then for i = 0,...,k we keep (y[i] - λ) and for i = k+1,...,n-1 we keep 0.
    # So sum = [sum y_i - k*λ] where sum is from i=0 to k
    # We want sum = total --> λ = (sum y_i - total) / k
    # This is satisfied if y_k >= λ > y_{k+1}  (k can be -1 if all are negative)
    # Let's compute with bisection:
    def evaluate_sum_at_lambda(lam):
        # return sum of max(0, y[i] - lam)
        result = np.sum(np.maximum(0, y - lam))
        return result
    
    # Binary search for the lambda that results in the total
    low, high = 0.0, 1.0
    if evaluate_sum_at_lambda(high) < total:
        # Too high lambda, pick lower among possible lambda
        return np.clip(y, 0, 1) * (total / np.sum(np.clip(y, 0, 1)))
    # Note: the case where all y are bigger than lambda is already handled by
    # clipping. If all smaller, it's dealt with by clamp being valid.
    # We now make sure lambda is in range where we have a valid solution
    
    # Simple implementation for a strict version
    # We compute exact lambda where the sum equals total via bisection
    # It's more informational to ensure sorting and fewer explicit checks
    from scipy.optimize import bisect
    try:
        lambda_star = bisect(lambda lam: evaluate_sum_at_lambda(lam) - total, 0.0, 1.0, xtol=1e-12)
        result = y - lambda_star
        result = np.clip(result, 0, 1)
        return result
    except:
        # Fall back to numerical approximation using bisect-like idea
        low, high = 0.0, 1.0
        for _ in range(40):
            mid = (low + high) / 2
            mid_sum = evaluate_sum_at_lambda(mid)
            if mid_sum >= total:
                low = mid
            else:
                high = mid
        lambda_star = (low + high) / 2
        result = y - lambda_star
        result = np.clip(result, 0, 1)
        return result

def softmax_weights(x, beta):
    """Compute softmax weights with temperature beta."""
    x = np.asarray(x)
    x_max = np.max(x)
    # Subtract max to avoid numerical overflow in exp
    exp_x = np.exp(beta * (x - x_max))
    return exp_x / np.sum(exp_x)

def gradient_from_weights(h, w):
    """
    Compute the gradient from weights w.
    Gradient is -(2/n) * (conv(w, h) + conv(w[::-1], h))[slice]
    """
    n = len(h)
    # Only compute the relevant part: start from index n-1 to n-1+n (so [n-1:2n-1]) 
    # Now we need the full convolution results for both w and w[::-1]
    # Convolution of w with h in 'full' mode gives length 2*n-1
    # To align the indices properly, we compute first left convolution then right
    # Alternative: Use correlate with zero padding at both ends
    # This is faster directly via causal convolution then reversing
    # Because we're primarily interested in what is located from index n-1 to 2*n-2
    
    # Order of steps:
    # convolve(w, h, mode='full') gives [n-1 + len(w) -1] elements
    # convolve(w[::-1], h, mode='full') does the same reverse sum
    # But not quite; better to explicitly slice what we need
    
    # This amounts to computing two full-length correlations, then using slice
    # Full convolution can be done by expanding tail of w
    # Just use scipy.signal.correlate or numpy.convolve directly
    
    # Since both make full-length output and index shift is fixed, just compute and slice
    from scipy.signal import correlate
    conv1 = correlate(h, w, mode='full')
    conv2 = correlate(h, w[::-1], mode='full')
    # The proper indexing we want is to extract center slice:
    # 1: convolve w with h
    # 2: convolve w[::-1] with h (reversed)
    # Now convolution of w with h across 2n-1 points, slices n-1 to n-1+n are [n-1:n+n-1] = [n-1:2n-1]
    
    # Actually, first define the valid subset:
    start_idx = 0
    end_idx = 2 * n - 1
    # Since we want center values, indices are n-1 through n-1 + n
    # extract the n elements from the middle 
    # cor1 = correlate(h, w, mode='full')[n-1:2*n-1] 
    # cor2 = correlate(h, w[::-1], mode='full')[n-1:2*n-1] 
    # But now they match the full contract:
    # w on [0,:len(w)) not indexed directly!
    # Let's simplify: use true full mode, just make sure we're looking at the right part
    
    # Direct way:
    # Take properly aligned parts for both: elements [n-1 : n-1+n] 
    x = np.zeros(n)
    # When convolving w with h, we are interested in only 1st n elements starting at shift 0
    # Index 0 of conv1 is alignment at shift 0 hence slice [n-1:n-1+n] = [n-1:2n-1] 

    cor1_slice = correlate(h, w, mode='full')[n-1:n-1+n]
    cor2_slice = correlate(h, w[::-1], mode='full')[n-1:n-1+n]
    return -(2.0 / len(h)) * (cor1_slice + cor2_slice)

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

        x_new = np.linspace(0, 2, n_points)
        h_interpolated = np.interp(x_new, x_orig, orig_h)
    else:
        h_interpolated = orig_h
    
    # Scale h to meet integral constraint
    h_normalized = h_interpolated * (total_mass / np.sum(h_interpolated))

    # Final projection to satisfy box + equal sum constraints
    h = project_box_sum(h_normalized, total_mass)

    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    # Optimisation parameters
    beta_start = 2.0
    beta_max = 200.0
    beta_factor = 1.6
    max_iter_beta = 2000
    step_init = 1.0

    beta = beta_start
    iterations_since_improvement = 0
    max_stagnation = 500

    while time.time() - start_time < 0.94 * budget_s:
        if beta > beta_max:
            break

        # Inner loop for fixed beta
        for iter_beta in range(max_iter_beta):
            if time.time() - start_time >= 0.94 * budget_s:
                break

            # Compute γ_s = corr(h,h) over all shifts
            gamma = np.correlate(h, h, mode='full')
            C = 1.0 - (2.0 / n_points) * gamma
            w = softmax_weights(beta * C, beta)
            grad = gradient_from_weights(h, w)

            # Backtracking line search with feasible update
            alpha = step_init
            h_trial = h - alpha * grad
            h_trial = project_box_sum(h_trial, total_mass)
            c5_trial = evaluate_c5(h_trial)
            c5_current = evaluate_c5(h)

            # Perform line search until acceptable improvement or very small α
            while alpha > 1e-10 and c5_trial >= c5_current:
                alpha *= 0.5
                h_trial = h - alpha * grad
                h_trial = project_box_sum(h_trial, total_mass)
                c5_trial = evaluate_c5(h_trial)

            h = h_trial
            c5_value = c5_trial

            if c5_value < best_c5:
                best_c5 = c5_value
                best_h = h.copy()
                iterations_since_improvement = 0
            else:
                iterations_since_improvement += 1

            # Adapt step size slightly
            adapt_factor = 1.1 if c5_value < c5_current else 0.9
            step_init = np.clip(step_init * adapt_factor, 1e-5, 2.0)

            # Check for.restart
            if iterations_since_improvement > max_stagnation:
                # Random restart
                h = np.random.rand(n_points)
                h = project_box_sum(h, total_mass)
                iterations_since_improvement = 0
            elif iterations_since_improvement % 100 == 0:
                # Optional periodic hybrid:
                noise_scale = 0.02
                h += np.random.uniform(-noise_scale, noise_scale, n_points)
                h = project_box_sum(h, total_mass)

        # Increase β for next outer loop
        beta = min(beta * beta_factor, beta_max)

    # Occasionally (after time safety margin) run final hill-climb
    if time.time() - start_time < budget_s - 0.5:
        try:
            # Short final refinement with small smoothness regularization
            h_final = best_h.copy()
            c5_final = best_c5
            # Simple steps for local polish based on neighbor fixes:
            for _ in range(1000):
                if time.time() - start_time > budget_s - 0.05:
                    break
                i_left, i_right = np.random.randint(0, n_points, 2)
                if i_left == i_right:
                    continue
                delta = np.random.uniform(0, 0.01)
                if h_final[i_left] >= delta and h_final[i_right] < 1 - delta:
                    h_trial = h_final.copy()
                    h_trial[i_left] -= delta
                    h_trial[i_right] += delta
                    h_trial = project_box_sum(h_trial, total_mass)
                    c5_trial = evaluate_c5(h_trial)
                    if c5_trial < c5_final:
                        h_final = h_trial
                        c5_final = c5_trial
            best_h = h_final
            best_c5 = c5_final
        except Exception:
            pass

    return (best_h, best_c5, n_points)
