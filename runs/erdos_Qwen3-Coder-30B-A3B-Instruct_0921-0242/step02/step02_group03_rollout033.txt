import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate
import cma

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def build_symmetric_h(p, q, n_points):
    """Build a symmetric piecewise constant h from parameters p and q."""
    M = len(p)
    # Transform widths with softmax
    w = np.exp(p)
    w = w / np.sum(w)
    
    # Transform heights with logistic
    a = 1.0 / (1.0 + np.exp(-q))
    
    # Construct left half
    h_half = np.zeros(n_points // 2)
    cum_w = np.concatenate([[0], np.cumsum(w)])
    
    for i in range(M):
        start_idx = int(np.round(cum_w[i] * (n_points // 2)))
        end_idx = int(np.round(cum_w[i+1] * (n_points // 2)))
        if start_idx < end_idx:
            h_half[start_idx:end_idx] = a[i]
    
    # Mirror to create full h
    h_full = np.concatenate([h_half, h_half[::-1]])
    return h_full

def objective_function(x, n_points):
    """Objective function for CMA-ES optimization."""
    # x is [p1, ..., pM, q1, ..., qM]
    M = len(x) // 2
    p = x[:M]
    q = x[M:]
    
    h = build_symmetric_h(p, q, n_points)
    h = project_to_feasible(h, n_points)
    
    return evaluate_c5(h)

def memetic_descent(h, n_points, max_steps=25):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        # The maximum is at shift 0, so we need to find the min among non-zero shifts
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        # Gradient wrt h[i] is dx * (1 - h[i + worst_shift_idx] - h[i - worst_shift_idx])
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    n_points = 400
    dx = 2.0 / n_points
    M = 7  # Number of segments
    
    # Define initial parameters for p and q
    # Initialize p so that middle blocks are wider
    p_init = np.zeros(M)
    p_init[M//2 - 1:M//2 + 2] = 1.0  # Middle 3 blocks wider
    p_init += np.random.normal(0, 0.1, M)
    
    # Initialize q with high values in the middle
    q_init = np.zeros(M)
    q_init[M//2 - 1:M//2 + 2] = 5.0  # Middle 3 blocks have high height
    q_init[:M//2 - 1] = -5.0  # Outer blocks have low height
    q_init[M//2 + 2:] = -5.0
    
    # Combine into a single vector
    x0 = np.concatenate([p_init, q_init])
    
    # Run CMA-ES optimization
    es = cma.CMAEvolutionStrategy(x0, 0.3, {'popsize': 4 + int(3 * np.log(2*M))})
    
    best_c5 = float('inf')
    best_h = None
    
    G_noimp = 80
    restart_count = 0
    max_restarts = 10
    
    while es.stop() is False and (time.time() - start_time) < (budget_s - 30):
        if restart_count > max_restarts:
            break
            
        # Get a sample of individuals
        X = es.ask()
        
        # Evaluate all individuals
        evals = []
        for x in X:
            try:
                c5 = objective_function(x, n_points)
                evals.append(c5)
            except:
                evals.append(float('inf'))
        
        # Sort by evaluation
        indices = np.argsort(evals)
        best_idx = indices[0]
        best_c5_current = evals[best_idx]
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            # Evaluate the best individual to get actual h
            p_best = X[best_idx][:M]
            q_best = X[best_idx][M:]
            h_temp = build_symmetric_h(p_best, q_best, n_points)
            h_temp = project_to_feasible(h_temp, n_points)
            best_h = h_temp.copy()
        
        # Tell CMA-ES
        es.tell(X, evals)
        
        # Check for restarts
        if es.countiter % G_noimp == 0:
            if es.countiter > 0:
                # Only restart if there's no improvement
                if es.best.f - best_c5 > 1e-5:
                    es = cma.CMAEvolutionStrategy(x0, 0.3, {'popsize': 2*es.popsize}) # restart with larger population
                    restart_count += 1
    
    # If we found a solution, polish it
    if best_h is not None:
        polished_h = memetic_descent(best_h, n_points, max_steps=25)
        polished_c5 = evaluate_c5(polished_h)
        if polished_c5 < best_c5:
            best_h = polished_h
            best_c5 = polished_c5
    
    if best_h is None:
        # Fallback to a simple solution if nothing was found
        h_initial = np.ones(n_points) * 0.5
        h_initial = project_to_feasible(h_initial, n_points)
        best_h = h_initial
        best_c5 = evaluate_c5(best_h)
        
    return (best_h, best_c5, n_points)
