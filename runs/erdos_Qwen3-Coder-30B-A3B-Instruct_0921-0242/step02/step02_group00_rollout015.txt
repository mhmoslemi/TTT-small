import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import bisect

def project_box_sum(y, total):
    """Project y onto the set {x : 0 <= x[i] <= 1, sum(x) = total}."""
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if abs(s - total) < 1e-10:
        return y
    if s > total:
        # Need to decrease values
        def sum_func(lam):
            return np.sum(np.clip(y - lam, 0, 1)) - total
        lam_min = 0.0
        lam_max = np.max(y)
        lam = bisect(sum_func, lam_min, lam_max, xtol=1e-12)
        y = np.clip(y - lam, 0, 1)
    else:
        # Need to increase values
        def sum_func(lam):
            return np.sum(np.clip(y + lam, 0, 1)) - total
        lam_min = 0.0
        lam_max = 1.0 - np.min(y)
        lam = bisect(sum_func, lam_min, lam_max, xtol=1e-12)
        y = np.clip(y + lam, 0, 1)
    return y

def compute_autocorrelation(h):
    """Compute aperiodic autocorrelation using FFT."""
    H = fft(h)
    P = np.abs(H)**2
    gamma = ifft(P).real
    return gamma

def evaluate_c5(h):
    """Evaluate C5 for given h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def generate_legendre_seed(n):
    """Generate a Legendre sequence with length n."""
    # Use prime p = 257, which gives a period of 256
    p = 257
    if n >= p-1:
        # Generate Legendre sequence of length p-1, drop last
        l = np.array([(k * k) % p for k in range(1, p)])
        # Legendre symbol: 1 if quadratic residue, 0 if zero, -1 if non-residue
        # For p=257, k=0 case is skipped
        leg = np.zeros(p - 1)
        for i in range(1, p):
            if i in l:
                leg[i-1] = 1
            else:
                leg[i-1] = 0
        h = leg[:n]
        return h
    else:
        # Shorter length, still use prime of that size or smaller
        # For now return balanced
        h = np.zeros(n)
        h[:n//2] = 1
        return h

def generate_m_sequence_seed(n):
    """Generate an m-sequence of length n."""
    # Generate maximal length sequence (order 8 LFSR) -> length 255
    # But we use smaller length by truncating
    # Direct implementation of an LFSR generating 255-length sequence
    # But for shorter, we simulate n steps
    if n <= 127:  # use a random construction which very roughly mimics low autocorrelation
        h = np.zeros(n, dtype=int)
        indices = np.random.choice(n, size=n//2, replace=False)
        h[indices] = 1
        return h
    else:
        # Use a standard de Bruijn sequence
        # Generate a 255-bit m-sequence
        LFSR = np.ones(8, dtype=int)
        # We simulate the LFSR shift with taps at positions 7, 6, 4, 3
        sequence = [1]  # start with 1
        for i in range(254):
            new_bit = (LFSR[7] + LFSR[5] + LFSR[3] + LFSR[2]) % 2
            sequence.append(new_bit)
            # Shift register
            LFSR[1:8] = LFSR[0:7]
            LFSR[0] = new_bit
        h = np.array(sequence[:n], dtype=int)
        return h

def generate_random_balanced_seed(n):
    """Generate a random balanced binary sequence."""
    h = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1
    return h

def binary_hill_climb(initial_h, max_swaps=None):
    """Improve a binary seed by swapping 1's with 0's."""
    if max_swaps is None:
        max_swaps = 3 * len(initial_h)
    h = initial_h.copy().astype(float)
    gamma = compute_autocorrelation(h)  # Precomputed autocorrelation
    n = len(h)
    
    last_improvement = 0
    for it in range(max_swaps):
        # Pick a random 1 and 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute the change in each auto-correlation
        gamma_delta = np.zeros_like(gamma)
        for k in range(-n+1, n):
            # Adjust for list indexing
            shift = k + n - 1  # k=-(n-1) to n-1 becomes 0 to 2*n-2
            if shift < 0 or shift >= 2*n-1:
                continue
            # FFT version correction:
            # If we swap h[i]=1 with h[j]=0:
            # We have h[i]*h[i+k] + h[i]*h[i-k] and h[j]*h[j+k] + h[j]*h[j-k]
            # which simplifies to: - h[i+k] - h[i-k] + h[j+k] + h[j-k]
            # Bring range to 0..2n-2
            kr1 = (i+k) % n
            kr2 = (i-k) % n
            kr3 = (j+k) % n
            kr4 = (j-k) % n
            gamma_delta[shift] = -h[kr1] - h[kr2] + h[kr3] + h[kr4]
        
        # Compute change in max C5 = max_k (1-(2/n)*gamma_k)
        # So we minimize max_k gamma_k
        # Change ΔC5 = max_k ((2/n) * Δγ_k)  
        # More precise idea: the change in the max term is what matters.
        # To simplify, consider only the indices involving the changes:
        # But internally  we use the update rule of correlation
        new_gamma = gamma + gamma_delta
        old_max = np.max(gamma)
        new_max = np.max(new_gamma)
        
        if new_max < old_max:
            h[i] = 0
            h[j] = 1
            gamma = new_gamma
            last_improvement = it
        elif it - last_improvement > 2*n:
            break  # stalemate
            
    return h

def spectral_alternating_projection(h_bin, max_iter=400, alpha_start=0.7):
    """Perform Spectral Alternating Projection to smooth the autocorrelation."""
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.astype(float)
    alpha = alpha_start
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        # Compute current gamma
        H = fft(h)
        P = np.abs(H)**2
        gamma = ifft(P).real
        
        # Compute average of non-zero lags
        avg = (np.sum(gamma) - gamma[0]) / (n - 1) if n > 1 else 0
        
        # Build target autocorrelation
        gamma_target = (1 - alpha) * gamma + alpha * avg
        gamma_target[0] = gamma[0]  # preserve zero lag
        
        # Apply target
        P_target = fft(gamma_target)
        P_target = np.maximum(P_target, 0)  # Handle numerical errors
        A_target = np.sqrt(P_target)
        
        # Form new H and go back to time domain
        H_new = A_target * np.exp(1j * np.angle(H))
        h_new = ifft(H_new).real
        
        # Project to feasible set T
        h = project_box_sum(h_new, n/2)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Check if this is still decreasing
        if iteration % 10 == 0:
            if iteration > 0 and iteration % 20 == 0:
                alpha = min(alpha + 0.02, 0.99)
        
        # Early termination when small improvements after 20 iterations
        if iteration > 20 and np.abs(best_c5 - evaluate_c5(best_h)) < 1e-6:
            break
            
    return best_h, best_c5

def projected_gradient_polish(h_init, max_iter=300, beta_start=5.0):
    """Do a final polishing with projected gradient descent."""
    h = h_init.copy()
    n = len(h)
    beta = beta_start
    eta = 1.0
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for iteration in range(max_iter):
        # Compute gamma
        gamma = compute_autocorrelation(h)
        # C_k = 1 - (2/n) gamma_k
        C = 1 - (2/n) * gamma
        
        # Soft max weights
        exp_C = np.exp(beta * C)
        w = exp_C / np.sum(exp_C)
        
        # Gradient: - (2/n)( conv(w, h) + conv(w[::-1], h) )
        # Use FFT for speed
        W = fft(w)
        H = fft(h)
        conv1 = ifft(W * H).real  # 1d
        w_rev = w[::-1]
        W_rev = fft(w_rev)
        conv2 = ifft(W_rev * H).real
        
        # Subtract and scale
        grad = -(2/n) * (conv1 + conv2)
        
        # Line search with eta
        while True:
            h_new = h - eta * grad
            h_new = project_box_sum(h_new, n/2)
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5 or eta < 1e-10:
                break
            eta *= 0.5
        
        h = h_new
        
        new_c5 = evaluate_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        if iteration % 20 == 0:
            beta *= 1.2
            eta = min(eta, 1.0)
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 256
    dx = 2.0 / n
    
    # 1. Generate seeds
    seeds = []
    # Optionally use initial_h_values (if present)
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) >= n:
            h = orig_h[:n]
        else:
            x_orig = np.linspace(0, 2, len(orig_h))
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_orig, orig_h)
        h_projected = project_box_sum(h, n/2)
        seeds.append(("initial", h_projected))
    
    # Legendre
    leg = generate_legendre_seed(n)
    leg = project_box_sum(leg, n/2)
    seeds.append(("legendre", leg))
    
    # m-sequence
    m_seq = generate_m_sequence_seed(n)
    m_seq = project_box_sum(m_seq, n/2)
    seeds.append(("mseq", m_seq))
    
    # Random balanced
    for i in range(5):
        rand = generate_random_balanced_seed(n)
        rand = project_box_sum(rand, n/2)  
        seeds.append((f"rand_{i}", rand))
    
    # 2. Binary hill climbing to find best binary seed
    best_binary_h = None
    best_binary_c5 = float('inf')
    
    input_remaining = budget_s - (time.time() - start_time)
    
    for name, h_seed in seeds:
        # Early stop
        elapsed = time.time() - start_time
        if elapsed > 0.65 * budget_s:
            break
            
        try:
            h_bin = binary_hill_climb(h_seed)
            c5_bin = evaluate_c5(h_bin)
            if c5_bin < best_binary_c5:
                best_binary_c5 = c5_bin
                best_binary_h = h_bin.copy()
        except Exception:
            continue
    
    if best_binary_h is None:
        # fall back to initial if no success
        if 'initial_h_values' in globals():
            h_fallback = initial_h_values
            if len(h_fallback) >= n:
                h_fallback = h_fallback[:n]
            else:
                x_orig = np.linspace(0, 2, len(h_fallback))
                x_new = np.linspace(0, 2, n)
                h_fallback = np.interp(x_new, x_orig, h_fallback)
            h_fallback = project_box_sum(h_fallback, n/2) 
            best_binary_h = h_fallback
            best_binary_c5 = evaluate_c5(h_fallback)
        else:
            best_binary_h = np.full(n, 0.5)  # fallback
            best_binary_c5 = 1.0
    
    # 3. SAP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        try:
            h_sap, c5_sap = spectral_alternating_projection(best_binary_h, max_iter=300)
            best_c5 = c5_sap
            best_h = h_sap
        except Exception:
            best_h = best_binary_h
            best_c5 = best_binary_c5
    else:
        best_h = best_binary_h
        best_c5 = best_binary_c5
        
    # 4. Final polishing
    if time.time() - start_time < budget_s - 5:
        try:
            polish_h, polish_c5 = projected_gradient_polish(best_h, max_iter=200, beta_start=5.0)
            if polish_c5 < best_c5:
                best_h = polish_h
                best_c5 = polish_c5
        except Exception:
            pass
    
    return (best_h, best_c5, n)
