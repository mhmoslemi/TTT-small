import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from sobol_seq import i4_sobol_generate

def project_to_feasible(h, target_sum):
    """Ensure that sum(h) = target_sum, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def build_lp_data(h0):
    """Build LP data for solving the CCP subproblem."""
    n = len(h0)
    dx = 2.0 / n
    c0 = ifft(fft(h0, 2 * n) * np.conj(fft(h0, 2 * n))).real
    
    # Build the matrix W where row k is h0[(i+k)%n] + h0[(i-k)%n]
    W = np.zeros((n, n))
    for k in range(n):
        W[k, :] = h0[(np.arange(n) + k) % n] + h0[(np.arange(n) - k) % n]
    
    A = -W
    b = -n/2 + c0
    return A, b

def solve_ccp_step(h0):
    """Solve one CCP step to get next h."""
    A, b = build_lp_data(h0)
    n = len(h0)
    target_sum = n / 2
    
    # Define the LP
    c = np.zeros(n + 1)  # h variables + t variable
    c[-1] = 1  # minimize t
    
    # Inequality constraints: A * h + t * 1 <= b
    # Or equivalently: -A * h - t * 1 >= -b
    A_ub = np.column_stack([A, np.ones(n)])
    b_ub = -b
    
    # Bound constraints on h[0..n-1]
    bounds = [(0, 1) for _ in range(n)] + [(0, None)]  # h in [0,1], t >= 0

    # Equality constraint: sum(h) = n/2
    A_eq = np.ones((1, n + 1))
    A_eq[0, -1] = 0  # t coefficient is 0
    b_eq = [target_sum]
    
    # Solve
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        h_new = result.x[:-1]
        t_new = result.x[-1]
        return h_new, t_new
    else:
        # Fall back to a projection method if LP fails
        return project_to_feasible(h0, target_sum), evaluate_c5(h0)

def generate_seeds(n, target_sum):
    """Generate multiple starting points for the CCP algorithm."""
    seeds = []
    
    # 1. Uniform
    h_uniform = np.ones(n) * 0.5
    h_uniform = project_to_feasible(h_uniform, target_sum)
    seeds.append(h_uniform.copy())

    # 2. Random Dirichlet
    h_dirichlet = np.random.dirichlet(np.ones(n))
    h_dirichlet = h_dirichlet * target_sum / np.sum(h_dirichlet)
    h_dirichlet = project_to_feasible(h_dirichlet, target_sum)
    seeds.append(h_dirichlet.copy())

    # 3. Binary + smoothing
    h_binary = np.zeros(n)
    h_binary[:int(target_sum)] = 1.0
    np.random.shuffle(h_binary)
    # Add small random noise to smooth
    h_binary += np.random.uniform(0, 0.05, n)
    h_binary = project_to_feasible(h_binary, target_sum)
    seeds.append(h_binary.copy())

    # 4. Interpolated from initial_h_values if available
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        x_orig = np.linspace(0, 2, len(orig_h))
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_orig, orig_h)
        h_interp = h_interp * target_sum / np.sum(h_interp)
        h_interp = project_to_feasible(h_interp, target_sum)
        seeds.append(h_interp.copy())

    # 5. Sobol sample in simplex
    sobol_sample = i4_sobol_generate(n, 1).flatten()
    h_sobol = sobol_sample / np.sum(sobol_sample) * target_sum
    h_sobol = project_to_feasible(h_sobol, target_sum)
    seeds.append(h_sobol.copy())
    
    return seeds

def projected_gradient_descent(h_init, max_iter=200):
    """Projected gradient descent on the true objective."""
    h = h_init.copy()
    dx = 2.0 / len(h)
    
    for t in range(max_iter):
        # Compute full correlation to find worst shift
        h_fft = fft(h, 2 * len(h))
        ih_fft = fft(1 - h, 2 * len(h))
        corr = ifft(h_fft * np.conj(ih_fft))
        corr_vals = corr.real[1:]  # Skip shift 0
        worst_shift = np.argmin(corr_vals) + 1
        
        # Compute subgradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            left_idx = (i + worst_shift) % len(h)
            right_idx = (i - worst_shift) % len(h)
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Perform update with diminishing step size
        alpha = 1.0 / np.sqrt(t + 1)
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h) / 2)
        
        # Early stopping if change is small
        if np.max(np.abs(h_new - h)) < 1e-8:
            break
            
        h = h_new
    
    return h

def simulated_annealing(h_init, max_time, start_time):
    """Simulated Annealing polishing step."""
    h = h_init.copy()
    c5_current = evaluate_c5(h)
    dx = 2.0 / len(h)
    
    temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-6
    max_iter = 100000

    for _ in range(max_iter):
        if time.time() - start_time > max_time * 0.995:
            break
            
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        
        if h[i] <= 0 or h[j] >= 1:
            continue
            
        delta = np.random.uniform(0, min(0.01, h[i], 1 - h[j]))
        h_test = h.copy()
        h_test[i] -= delta
        h_test[j] += delta
        h_test = project_to_feasible(h_test, len(h) / 2)
        
        c5_test = evaluate_c5(h_test)
        
        if c5_test < c5_current:
            h = h_test
            c5_current = c5_test
        else:
            # Accept with probability
            prob = np.exp(-(c5_test - c5_current) / temp)
            if np.random.rand() < prob:
                h = h_test
                c5_current = c5_test
        
        temp = max(temp * cooling_rate, min_temp)

    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple seeds
    seeds = generate_seeds(n_points, target_sum)
    
    # Run CCCP on each seed
    best_c5 = float('inf')
    best_h = None
    
    for i, seed_h in enumerate(seeds):
        if time.time() - start_time > budget_s * 0.7:
            break
            
        h_current = seed_h.copy()
        prev_c5 = evaluate_c5(h_current)
        
        max_iter = 30  # damping iterative loop
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s * 0.7:
                break
                
            h_new, _ = solve_ccp_step(h_current)
            h_new = project_to_feasible(h_new, target_sum)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
            if abs(c5_new - prev_c5) < 1e-8:
                break
                
            h_current = h_new
            prev_c5 = c5_new
    
    # Refine the best solution found so far with projected GD
    if best_h is not None and time.time() - start_time < budget_s * 0.9:
        best_h_gd = projected_gradient_descent(best_h)
        c5_gd = evaluate_c5(best_h_gd)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = best_h_gd.copy()
    
    # Polish with Simulated Annealing
    if best_h is not None and time.time() - start_time < budget_s * 0.995:
        best_h_sa = simulated_annealing(best_h, budget_s, start_time)
        c5_sa = evaluate_c5(best_h_sa)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = best_h_sa.copy()
    
    # Optional upscale and re-refine
    if best_h is not None and time.time() - start_time < budget_s - 10:
        # Interpolate to higher resolution
        high_res = 800
        x_orig = np.linspace(0, 2, n_points)
        x_high = np.linspace(0, 2, high_res)
        h_high = np.interp(x_high, x_orig, best_h)
        h_high = h_high * (high_res / 2) / np.sum(h_high)
        h_high = project_to_feasible(h_high, high_res / 2)
        
        # Run one short CCP pass, then few gradient steps
        h_refine = h_high.copy()
        for _ in range(5):
            h_new, _ = solve_ccp_step(h_refine)
            h_new = project_to_feasible(h_new, high_res / 2)
            h_refine = h_new.copy()
        
        # Projected GD to further smooth the result
        h_refine = projected_gradient_descent(h_refine, max_iter=100)
        c5_refined = evaluate_c5(h_refine)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refine.copy()
    
    if best_h is None:
        # Fallback in case of failure
        best_h = np.ones(n_points) * 0.5
        best_h = project_to_feasible(best_h, target_sum)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, n_points)
