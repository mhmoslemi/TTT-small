import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar

def project_feasible(h, n):
    """Project h to feasible set: sum(h) = n/2, h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    # Water-filling to get sum = n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    # Find where to cut (where we have enough to add/subtract)
    excess = current_sum - n/2
    if excess > 0:
        # Need to decrease sum. Assume we reduce all by a constant amount
        # up to some point.
        diff = n/2 - current_sum
        h = np.clip(h + diff/n, 0, 1)
        scale = (n/2) / np.sum(h)
        h = h * scale
    elif excess < 0:
        # Need to increase sum. This means we can add to the lowest values.
        h = np.clip(h + (-excess)/n, 0, 1)
        scale = (n/2) / np.sum(h)
        h = h * scale
    return h

def evaluate_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation using FFT for speed
    corr = np.real(ifft(fft(h) * fft(1 - h).conj()))
    # Maximize over circular shift indices [0, 2n-1]
    max_corr = np.max(corr) * dx
    return max_corr

def gradient_at_worst(h):
    """Get gradient of maximum correlation at k* = argmax_{k} integral h(x)(1-h(x+k)) dx."""
    n = len(h)
    dx = 2.0 / n
    # xcorr = correlate(h, 1-h, mode='full')
    xcorr = np.real(ifft(fft(h) * fft(1 - h).conj()))
    k_star = np.argmax(xcorr)
    # k_star is in 0..2n-2 corresponding to circular shifts
    # k = k_star - n + 1 -> range is [-n+1, n-1]
    k = k_star - n + 1  # if k_star >= n then k positive; else negative
    # Original formula is d/dh_i of max_k \sum_j h_j (1 - h_{j+k}), so
    # d/dh_i of the correlation with shift k = sum_j d/dh_i [h_j (1 - h_{j+k})] = 0 if j != i, 1 if j = i, -1 if j = i+k mod n
    # Actually h_i appears in two correlations: for j such that j == i or j == i-k, so:
    # df/dh_i = h_{i+k} - h_{i-k}
    grad = np.zeros_like(h)
    for i in range(n):
        i_plus_k = (i + k) % n
        i_minus_k = (i - k) % n
        grad[i] = h[i_plus_k] - h[i_minus_k]
    return grad * dx

def line_search(h, grad, eta0=0.5):
    """Backtracking line search for a feasible directional step."""
    h_new = h - eta0 * grad
    h_new = project_feasible(h_new, len(h))
    eta = eta0
    while eta > 1e-10:
        h_trial = h - eta * grad
        h_trial = project_feasible(h_trial, len(h))
        if evaluate_c5(h_trial) < evaluate_c5(h):
            return h_trial
        eta /= 2.0
    return h

def mutate_with_dct(h, sigma=0.2):
    """Modifies a low-frequency DCT mode."""
    # Discrete cosine transform
    H = np.fft.rfft(h)
    # Modify first few coefficients slightly
    for i in range(min(5, len(H))):
        H[i] += np.random.normal(0, sigma)
    h_new = np.fft.irfft(H, len(h))
    h_new = project_feasible(h_new, len(h))
    return h_new

def mutate_swap_mass(h, delta=0.01):
    """Swap small amounts of mass between indices."""
    h_new = h.copy()
    # Select two indices
    indices = np.where((h_new > 1e-6) & (h_new < 1 - 1e-6))[0]
    if len(indices) < 2:
        return h_new
    i = np.random.choice(indices)
    j = np.random.choice(indices)
    while j == i:
        j = np.random.choice(indices)
    # Move delta from i to j, respecting bounds
    move = min(delta, h_new[i], 1 - h_new[j])
    h_new[i] -= move
    h_new[j] += move
    h_new = project_feasible(h_new, len(h_new))
    return h_new

def mutate_gaussian(h, sigma=0.2):
    """Add Gaussian noise to a random few elements."""
    n = len(h)
    # Pick 5 to 10 elements randomly
    num_to_modify = np.random.randint(5, 11)
    indices = np.random.choice(n, num_to_modify, replace=False)
    h_new = h.copy()
    delta = np.random.normal(0, sigma, num_to_modify)
    # Add without bounds for now
    h_new[indices] += delta
    h_new = project_feasible(h_new, len(h_new))
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Adaptive n selection
    n_points = 250  # Start small for speed
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Initial pool: random samples
    pool_size = 2000
    M = 2000
    pool = []
    pool_c5 = []
    
    for _ in range(M):
        if time.time() - start_time > 0.15 * budget_s:
            break
        u = np.random.rand(n_points)
        h = project_feasible(u, n_points)
        c5 = evaluate_c5(h)
        pool.append(h)
        pool_c5.append(c5)
    
    # Sort and keep top K
    top_k = 30
    top_idx = np.argsort(pool_c5)[:top_k]
    pool = [pool[i] for i in top_idx]
    pool_c5 = [pool_c5[i] for i in top_idx]

    # Mutation loop
    mutation_iter = 0
    max_mutation_iter = 2000
    sigma_current = 0.2
    
    t1 = time.time()
    while (time.time() - start_time < 0.4 * budget_s) and mutation_iter < max_mutation_iter:
        if time.time() - start_time > 0.39 * budget_s:
            break
        parent_idx = np.random.choice(len(pool))
        # Mutate
        mode = np.random.choice(['gaussian', 'swap', 'dct'])
        if mode == 'gaussian':
            h_child = mutate_gaussian(pool[parent_idx], sigma_current)
        elif mode == 'swap':
            h_child = mutate_swap_mass(pool[parent_idx], 0.01)
        else: # dct
            h_child = mutate_with_dct(pool[parent_idx], sigma_current)
        
        c5_child = evaluate_c5(h_child)
        if c5_child < pool_c5[-1]:  # replace worst
            pool[-1] = h_child
            pool_c5[-1] = c5_child
            # Keep sorted
            idx_sorted = np.argsort(pool_c5)
            pool = [pool[i] for i in idx_sorted]
            pool_c5 = [pool_c5[i] for i in idx_sorted]

        mutation_iter += 1
        if mutation_iter % 50 == 0:
            sigma_current *= 0.99
            if sigma_current < 1e-4:
                sigma_current = 1e-4

    # Local refinement of top 5 (or what we have)
    candidates = pool[:min(5, len(pool))]
    local_refinement_iter = 0
    t2 = time.time()
    for h in candidates:
        if time.time() - start_time > 0.75 * budget_s - 0.1:
            break
        h_refined = h.copy()
        c5_current = evaluate_c5(h_refined)
        
        # Use a few thousand steps
        step_limit = 2000
        for _ in range(step_limit):
            if time.time() - start_time > 0.75 * budget_s - 0.1:
                break
            grad = gradient_at_worst(h_refined)
            h_new = line_search(h_refined, grad, eta0=0.5)
            if evaluate_c5(h_new) < c5_current:
                h_refined = h_new
                c5_current = evaluate_c5(h_refined)
                if c5_current < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_current
            else:
                break
        if c5_current < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_current
        local_refinement_iter += 1

    # Final polishing with expensive "pairwise swaps"
    if best_h is not None and time.time() - start_time < budget_s - 0.5:
        h_be = best_h.copy()
        c5_be = best_c5
        # Do many swaps
        total_swaps = 10000
        for i in range(total_swaps):
            if time.time() - start_time > budget_s - 0.5:
                break
            # Select two indices for swapping
            indices = np.where((h_be > 1e-6) & (h_be < 1 - 1e-6))[0]
            if len(indices) < 2:
                break
            a = np.random.choice(indices)
            b = np.random.choice(indices)
            while a == b:
                b = np.random.choice(indices)
            # Small amount to move
            delta = np.random.uniform(1e-4, 0.01)
            if h_be[a] >= delta and (1 - h_be[b]) >= delta:
                # Transfer
                delta = min(delta, h_be[a], 1 - h_be[b])
                h_trial = h_be.copy()
                h_trial[a] -= delta
                h_trial[b] += delta
                h_trial = project_feasible(h_trial, n_points)
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < c5_be:
                    h_be = h_trial
                    c5_be = c5_trial
        if c5_be < best_c5:
            best_h = h_be
            best_c5 = c5_be

    # Upscale to higher resolution if time allows
    if time.time() - start_time < 0.9 * budget_s and n_points < 500:
        n_points = 500
        dx = 2.0 / n_points
        h_up = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_up = project_feasible(h_up, n_points)
        c5_up = evaluate_c5(h_up)
        if c5_up < best_c5:
            best_h = h_up
            best_c5 = c5_up

        # Refine a bit
        if time.time() - start_time < 0.95 * budget_s:
            h_refined = h_up.copy()
            c5_current = c5_up
            for _ in range(1000):
                if time.time() - start_time > 0.95 * budget_s:
                    break
                grad = gradient_at_worst(h_refined)
                h_new = line_search(h_refined, grad, eta0=0.5)
                if evaluate_c5(h_new) < c5_current:
                    h_refined = h_new
                    c5_current = evaluate_c5(h_refined)
                else:
                    break
            if c5_current < best_c5:
                best_h = h_refined
                best_c5 = c5_current

    return (best_h, best_c5, n_points)
