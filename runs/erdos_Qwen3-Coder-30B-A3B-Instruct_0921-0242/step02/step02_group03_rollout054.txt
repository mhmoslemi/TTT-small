import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def compute_correlations(h):
    """Compute all correlations of h with (1-h)."""
    dx = 2.0 / len(h)
    h_padded = np.pad(h, (0, len(h)-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, len(h)-1), mode='constant', constant_values=0)
    
    # Using FFT for efficient cross-correlation
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    corr_vals = corr_vals[:len(h)]  # Take only the non-cyclic part
    # Normalize by dx
    corr_vals = corr_vals * dx
    return corr_vals

def update_correlations_fast(corr_vals, h, i, j):
    """Update correlations after swapping h[i] and h[j]."""
    # Compute change in the correlation sum
    n = len(h)
    # For each shift s, compute change in correlation due to swapping positions i and j
    delta = np.zeros(n)
    for s in range(n):
        if i + s < n and j + s < n:
            delta[s] += (1 - h[i]) - h[i+s]  # contribution of pos i removed
            delta[s] += (1 - h[j]) - h[j+s]  # contribution of pos j added
        if i - s >= 0 and j - s >= 0:
            delta[s] += (1 - h[i]) - h[i-s]  # contribution of pos i removed
            delta[s] += (1 - h[j]) - h[j-s]  # contribution of pos j added
    return corr_vals + delta

def block_initialization(n):
    """Initialize with a block pattern."""
    h = np.zeros(n, dtype=int)
    half = n // 2
    h[:half] = 1
    np.random.shuffle(h)
    return h

def random_initialization(n):
    """Initialize randomly with exactly n/2 ones."""
    h = np.zeros(n, dtype=int)
    h[:n//2] = 1
    np.random.shuffle(h)
    return h

def simulated_annealing_binary(h0, n, max_moves=60000, time_limit=None):
    """Simulated Annealing for binary sequences."""
    T0 = 0.02 * n
    alpha = 0.997
    T = T0
    
    h = h0.copy()
    corr_vals = compute_correlations(h.astype(float))
    current_max = np.max(corr_vals)
    best_h = h.copy()
    best_max = current_max

    moves = 0
    no_improvement = 0
    
    while moves < max_moves:
        if time_limit and time.time() > time_limit:
            break
            
        # Pick two random positions to swap
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        
        if h[i] == h[j]:  # No change if same value
            moves += 1
            continue

        # Compute new correlation values efficiently
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]  # Swap
        
        new_corr_vals = update_correlations_fast(corr_vals, h, i, j)
        new_max = np.max(new_corr_vals)
        
        # Acceptance probability
        if new_max <= current_max:
            accept = True
        else:
            accept = np.random.rand() < np.exp(-(new_max - current_max) / max(T, 1e-10))
        
        if accept:
            h = h_new
            corr_vals = new_corr_vals
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1

        moves += 1
        T *= alpha
        
        if no_improvement > 5000:
            # Early stopping if no improvement
            break

    return best_h, best_max

def polish_continuous(h_binary, n_points, max_iter=30):
    """Refine with continuous optimization."""
    h = h_binary.astype(float)
    dx = 2.0 / n_points
    lamb = 5.0  # initial smooth max parameter
    
    for i in range(max_iter):
        # Use smooth max surrogate
        corr_vals = compute_correlations(h)
        # Smooth max
        smooth_max = np.log(np.sum(np.exp(lamb * corr_vals))) / lamb
        # Compute gradient: d/dh_i of c_s = h_i*(1-h_{i+s}) + ...
        grad = np.zeros(n_points)
        # Compute gradient for each s
        for s in range(n_points):
            if s == 0:
                continue
            # grad[i] += (1-h[i+s]) - (1-h[i-s]) for valid indices
            # Approach: Use the simple derivative w.r.t each h at all positions
            
            # The gradient of h_i*(1-h_{i+s}) w.r.t h_i is 1-h_{i+s}
            # And w.r.t h_{i+s} is -h_i
            # We just fill in the contributions per shift
            for i in range(n_points - s):
                grad[i] += 1 - h[i+s]
                if i - s >= 0:
                    grad[i-s] -= h[i]
        
        grad = grad * dx
        
        # Update step
        eta = 0.1 * dx * (0.5 ** (i // 10))
        h -= eta * grad
        # Project to feasible set
        h = np.clip(h, 0, 1)
        # Project to sum constraint
        h = project_to_feasible(h, n_points)
        
        # Stop early if not changing much
        if i > 10 and np.max(np.abs(grad)) < 1e-5:
            break
            
        # Increase smooth max parameter
        lamb = min(lamb + 5, 30.0)
              
    return h, evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 10  # Leave some margin for cleanup
    
    n_points = 221  # Odd number, fine resolution, manageable for FFT and improvements
    h_best = None
    c5_best = float('inf')
    
    num_trials = 10  # Number of independent SA trials

    for trial in range(num_trials):
        if time.time() > time_limit:
            break
        
        # Choose initialization type
        if np.random.rand() < 0.5:
            h_init = block_initialization(n_points)
        else:
            h_init = random_initialization(n_points)
            
        # Run simulated annealing
        h_local, val_local = simulated_annealing_binary(
            h_init, n_points,
            max_moves=50000,
            time_limit=time_limit
        )
        
        # Polish with continuous refinement
        h_polished, val_polished = polish_continuous(h_local, n_points)
        
        if val_polished < c5_best:
            c5_best = val_polished
            h_best = h_polished.copy()
    
    # Check final best
    if h_best is None:
        # Default fallback
        h_default = np.ones(n_points) * 0.5
        c5_default = evaluate_c5(h_default)
        h_best = h_default
        c5_best = c5_default
    
    return (h_best, c5_best, n_points)
