import numpy as np
import time

def project_box_sum(y, total):
    """Project y onto {x : 0 <= x[i] <= 1, sum(x) = total}"""
    # This uses the 'water-filling' method - find lambda such that
    # sum(clip(y-lambda, 0, 1)) = total
    y = np.asarray(y)
    low = np.zeros_like(y)
    high = np.ones_like(y)
    # Compute all possible lambda values by sorting - no need to be exhaustive
    # because with convexity, the solution is at one of the breakpoints
    # Projecting onto box [0,1] w/ sum total is equivalent to sorting,
    # truncating the highest values to 1, lowest to 0, and adjusting the rest
    # to make the sum total
    # Use binary search for lambda
    def project_with_lambda(l):
        return np.clip(y - l, 0, 1)
    
    def sum_with_lambda(l):
        return np.sum(project_with_lambda(l))
    
    # bisection
    l_low, l_high = np.min(y - 1), np.max(y)
    for _ in range(50):
        l_mid = (l_low + l_high) / 2
        s_mid = sum_with_lambda(l_mid)
        if s_mid > total:
            l_low = l_mid
        else:
            l_high = l_mid
    l = (l_low + l_high) / 2
    
    return np.clip(y - l, 0, 1)

def evaluate_c5(h):
    """Evaluate C5 constant for given h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def softmax_weights(beta_C):
    """Compute softmax weights from beta * C values."""
    C = beta_C
    # for numerical stability
    C = C - np.max(C)
    exp_C = np.exp(C)
    return exp_C / np.sum(exp_C)

def gradient_from_weights(w, h):
    """Compute gradient from weights."""
    n = len(h)
    # Considering that we have the weights of each shift, reversed in index
    # w is of length 2n-1
    w_extend = np.zeros(2*n - 1)
    w_extend[n-1:n-1+n] = w
    # Shifted convolution result
    conv1 = np.correlate(w_extend, h, mode='full')
    conv2 = np.correlate(w_extend[::-1], h, mode='full')
    # No need to reconstruct the whole result, we just take the correct slice
    result = -(2.0/n) * (conv1[n-1:n-1+n] + conv2[n-1:n-1+n])
    return result
