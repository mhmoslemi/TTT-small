import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def project_to_sum_constraint(h, target_sum):
    """Project h to sum to target_sum with h in [0, 1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT."""
    n = len(h)
    dx = 2.0 / n
    h_fft = rfft(h)
    ih_fft = rfft(1 - h)
    corr = irfft(h_fft * np.conj(ih_fft))
    # Take positive shifts, exclude 0
    max_corr = np.max(corr[1:n]) * dx
    return max_corr

def gradient_worst_shift(h):
    """Compute gradient with respect to the worst shift."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations efficiently via FFT
    h_fft = rfft(h)
    ih_fft = rfft(1 - h)
    corr = irfft(h_fft * np.conj(ih_fft))
    # Ignore shift 0
    max_idx = np.argmax(corr[1:n]) + 1
    grad = np.zeros(n)
    for i in range(n):
        left_idx = (i + max_idx) % n
        right_idx = (i - max_idx) % n
        grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
    return grad

def gradient_descent_worst_shift(h, max_iter=200, alpha0=0.1):
    """Projected gradient descent on the worst shift."""
    dx = 2.0 / len(h)
    h = project_to_sum_constraint(h, len(h) / 2)
    for t in range(max_iter):
        if t > 0 and t % 10 == 0:
            alpha = alpha0 / np.sqrt(t)
        else:
            alpha = alpha0
        grad = gradient_worst_shift(h)
        h_new = h - alpha * grad
        h_new = project_to_sum_constraint(h_new, len(h) / 2)
        if np.allclose(h, h_new, rtol=1e-7):
            break
        h = h_new
    return h

def de_step(population, fitnesses, F=0.6, CR=0.9):
    """Single DE step."""
    new_pop = []
    n = len(population)
    for i in range(n):
        a, b, c = np.random.choice(n, 3, replace=False)
        donor = population[a] + F * (population[b] - population[c])
        donor = np.clip(donor, 0, 1)
        donor = project_to_sum_constraint(donor, len(donor) / 2)
        trial = np.where(np.random.rand(len(donor)) < CR, donor, population[i])
        trial = project_to_sum_constraint(trial, len(trial) / 2)
        new_pop.append(trial)
    return new_pop

    h_start = np.interp(x_new, x_orig, orig_h)
    h_start = project_to_sum_constraint(h_start, target_sum)
    
    P = 40
    population = [h_start.copy()]
    for _ in range(P - 1):
        h = np.random.rand(n_points)
        h = project_to_sum_constraint(h, target_sum)
        population.append(h)
    
    # DE main loop
    best_h = None
    best_c5 = float('inf')
    last_improvement = time.time()
    max_gen = 5000
    gen_step = 0

    while gen_step < max_gen and (time.time() - start_time < budget_s * 0.75):
        fit_vals = [evaluate_c5_fft(h) for h in population]
        best_idx = np.argmin(fit_vals)
        if fit_vals[best_idx] < best_c5:
            best_c5 = fit_vals[best_idx]
            best_h = population[best_idx].copy()
            last_improvement = time.time()

        if time.time() - last_improvement > 10:
            # Re-seed
            for i in range(P // 5):
                idx = np.random.randint(P)
                h = np.random.rand(n_points)
                h = project_to_sum_constraint(h, target_sum)
                population[idx] = h

        refined_h = gradient_descent_worst_shift(best_h, max_iter=200)

        return (h_start, evaluate_c5_fft(h_start), n_points)
