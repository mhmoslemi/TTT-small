import numpy as np
import time
from scipy.signal import correlate
from scipy.ndimage import gaussian_filter1d
import cvxpy as cp

def project_box_sum(h, total):
    """Project h onto the set {h: 0≤h≤1, sum(h)=total} using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - total) < 1e-10:
        return h
    
    # Water-filling approach
    # Sort in descending order
    h_sorted = np.sort(h)[::-1]
    # Find the threshold l
    cumsum = np.cumsum(h_sorted)
    n = len(h_sorted)
    # We want to find the largest k such that
    # sum(h_sorted[:k]) - k*l >= total
    # This is equivalent to sum(h_sorted[:k]) - total >= k*l
    # So l <= (sum(h_sorted[:k]) - total) / k
    # s = sum(h_sorted[:k]) - total
    # We seek min_l s/k >= 0, so the largest k such that s/k >= 0
    kmax = 0
    for k in range(1, n + 1):
        if cumsum[k - 1] - total >= 0:
            if (cumsum[k - 1] - total) / k <= h_sorted[k - 1]:
                kmax = k
            else:
                break
    
    if kmax == 0:
        l = 0.0
    else:
        l = (cumsum[kmax - 1] - total) / kmax
        
    result = np.clip(h - l, 0, 1)
    return result


def aperiodic_autocorr(h):
    """Compute aperiodic autocorrelations using FFT for efficiency."""
    n = len(h)
    h_padded = np.pad(h, (0, n - 1), mode='constant', constant_values=0)
    h_rev_padded = np.pad(h[::-1], (0, n - 1), mode='constant', constant_values=0)
    fft_h = np.fft.fft(h_padded)
    fft_h_rev = np.fft.fft(h_rev_padded)
    corr_fft = np.fft.ifft(fft_h * np.conj(fft_h_rev))
    # Return only the valid autocorrelations for shifts 0..n-1
    corr = np.real(corr_fft[:n])
    return corr


def softmax(v, beta=1.0):
    """Numerically stable softmax with temperature beta."""
    v = np.asarray(v)
    v_max = np.max(v)
    exp_v = np.exp(beta * (v - v_max))
    return exp_v / np.sum(exp_v)


def conv(h, w):
    """Convolution using FFT."""
    n = len(h)
    h_pad = np.pad(h, (0, n - 1), mode='constant', constant_values=0)
    w_pad = np.pad(w, (0, n - 1), mode='constant', constant_values=0)
    conv_fft = np.fft.ifft(np.fft.fft(h_pad) * np.fft.fft(w_pad))
    return np.real(conv_fft[:n])


def score_guided_swap(h, eps=1e-5):
    """Simple score-guided swap."""
    n = len(h)
    c5_values = np.array([1.0 - (2.0 / n) * gamma for gamma in aperiodic_autocorr(h)])
    worst_indices = np.where(np.abs(c5_values - np.max(c5_values)) < eps)[0]
    if len(worst_indices) == 0:
        worst_indices = [0]
    # Select a random worst shift
    s = worst_indices[np.random.randint(0, len(worst_indices))]
    if s == 0:
        s = 1
    
    # Find the indices involved in that shift
    indices = [i for i in range(n) if (i + s < n or i - s >= 0)]
    
    # Quick score for indices based on neighbors (agreement with best shifts)
    scores = np.array([np.sum([h[(i + s) % n] + h[(i - s) % n] for s in worst_indices if s != 0]) if i < n else 0 for i in range(n)])
    
    # Choose a source with high score
    src_indices = np.where(scores > np.percentile(scores, 60))[0]
    if len(src_indices) == 0:
        src_indices = list(range(n))
    i = np.random.choice(src_indices)
    
    # Choose a target with low score
    target_indices = np.where(scores < np.percentile(scores, 40))[0]
    if len(target_indices) == 0:
        target_indices = list(range(n))
    j = np.random.choice(target_indices)
    
    # Perform swap
    delta = min(0.01, h[i], 1.0 - h[j])
    if delta < 1e-6:
        return h
    h[i] -= delta
    h[j] += delta
    h = project_box_sum(h, total=n / 2)
    return h


def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Phase I: Solve binary MILP
    n1 = 120
    half_n1 = n1 // 2
    x1 = np.linspace(0, 2, n1, endpoint=False)
    
    # Define CVXPY variables for binary optimization
    b = cp.Variable(n1, boolean=True)
    a = cp.Variable((n1, n1), boolean=True)
    L = cp.Variable()
    
    constraints = [cp.sum(b) == half_n1]
    
    # Constraint for product: a[i,s] = b[i]*b[i+s]
    for i in range(n1):
        for s in range(n1 - i):
            constraints += [
                a[i, s] <= b[i],
                a[i, s] <= b[i + s],
                a[i, s] >= b[i] + b[i + s] - 1
            ]
            
    # Autocorrelation constraints
    for s in range(n1):
        # Sum over valid i values where i+s < n1
        terms = [a[i, s] for i in range(n1 - s)]
        constraints += [cp.sum(terms) >= L]
    
    # The objective is to maximize L
    objective = cp.Maximize(L)
    
    prob = cp.Problem(objective, constraints)
    
    # Set a generous time limit and tolerance
    try:
        prob.solve(solver=cp.CBC, time_limit=0.2 * budget_s, relative_gap=5e-3)
    except Exception:
        # Fall back to solver if not available or error occurs
        pass
    
    # Get binary solution
    b_val = np.round(b.value).astype(int)
    if b_val is None:
        # fallback: use sinusoidal
        b_val = np.zeros(n1)
        b_val[list(range(0, half_n1, 2))] = 1
        b_val = np.roll(b_val, 3)
    
    # Phase II: Upsample to target resolution
    n2 = 300
    x2 = np.linspace(0, 2, n2, endpoint=False)
    h0 = np.interp(x2, x1, b_val)
    h0 = project_box_sum(h0, total=n2 / 2)
    
    # Optional smoothing
    h0 = gaussian_filter1d(h0, sigma=1.0)
    h0 = project_box_sum(h0, total=n2 / 2)

    # Phase III: Smoothed gradient descent with projected updates
    h = h0.copy()
    beta = 2.0
    eta = 1.0
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < 0.85 * budget_s:
        # Compute aperiodic autocorrelation via FFT
        gamma = aperiodic_autocorr(h)
        C = 1.0 - (2.0 / n2) * gamma
        w = softmax(beta * C)
        
        # Compute gradient efficiently with FFT
        conv_w_h = conv(w, h)
        conv_w_rev_h = conv(w[::-1], h)
        grad = -(2.0 / n2) * (conv_w_h + conv_w_rev_h)
        
        # Backtracking line search
        alpha = eta
        h_trial = project_box_sum(h - alpha * grad, total=n2 / 2)
        while True:
            c5_trial = evaluate_c5(h_trial)
            if c5_trial <= best_c5 or alpha < 1e-8:
                break
            alpha *= 0.5
            h_trial = project_box_sum(h - alpha * grad, total=n2 / 2)
            
        h = h_trial
        c5 = evaluate_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Schedule β
        if iteration % 150 == 0:
            beta = min(beta * 1.5, 200)
            
        # Occasionally perform swaps
        if iteration % 200 == 0:
            h = score_guided_swap(h)
        
        iteration += 1
        
        # Early break if no improvement
        if iteration - last_improvement > 500 and (time.time() - start_time > 0.75 * budget_s):
            break

    # Phase IV: Final polishing with exact subgradient
    h_polish = best_h.copy()
    step = 1
    while time.time() - start_time < budget_s - 0.1 * budget_s:
        c5_vals = np.array([1.0 - (2.0 / n2) * gamma for gamma in aperiodic_autocorr(h_polish)])
        worst_indices = np.where(np.abs(c5_vals - np.max(c5_vals)) < 1e-5)[0]
        g = np.zeros(n2)
        for s in worst_indices:
            # Padding to assume periodicity for indexing
            shifted_h = np.roll(h_polish, -s)
            g += shifted_h
            shifted_h_rev = np.roll(h_polish, s)
            g += shifted_h_rev
        g *= -2.0 / n2 / len(worst_indices)
        eta = 1.0 / np.sqrt(step)
        h_new = project_box_sum(h_polish - eta * g, total=n2 / 2)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_polish = h_new
        step += 1

    return (best_h, best_c5, n2)
