def project_to_box_plus_sum(h, target_sum):
    """Project h onto the set {0 <= h_i <= 1, sum(h) = target_sum} using water-filling."""
    # Sort in descending order to apply water-filling
    sorted_indices = np.argsort(h)[::-1]
    sorted_h = h[sorted_indices]
    
    # Find the water level
    n = len(h)
    cumsum = np.cumsum(sorted_h)
    # Find where cumulative sum exceeds target_sum + i - 1
    # We try to find the largest k such that sum of first k elements is <= target_sum
    # But we must respect the constraints that each element is at most 1
    
    # Water-filling algorithm
    total = 0.0
    water_level = 0.0
    for i in range(n):
        # If we set all elements from 0 to i to at most water_level,
        # sum over those elements is enough
        water_level = (target_sum - total) / (n - i)
        if water_level > 1:
            water_level = 1.0
        # Check if setting elements with index 0, ..., i to this level is valid
        # We can fit up to (n - i) elements to this level
        # Contribution so far is target_sum - total
        # Actually just compare the previous amount with k = i-neighbors (except last)
        # and live with situation of hitting levels on first iteration:
        # If total + (n - i) * water_level >= target_sum, then that is fine
        # Instead of checking elements then, just adjust water_level
        # If we know the water level, compute adjusted values:
        # The first i+1 elements are at water_level, and we need to trim
        # If water_level > 1, it should never happen, but let's not write funny down to 0
        
        # Check how many can be filled to current level
        # This is "how many" in [0..n) at most can be set to water_level?
        i = 0
        while i < n and cumsum[i] < target_sum:
            i += 1
        if i == 0:
            pass
        else:
            # Let's take simpler approach:
            # always project to [0,1], but using careful water-filling algorithm
            # Fix a key property: if tail summation is equal to (target_sum - prefix)
            # Then we can make those to ceiling, then set rest at minimum
            # curve-ball of iterations:
            # sum(sorted_h[:i+1]) = prev_sum (non-inclusive), therefore, 
            # for next itelement, the target minus prev_sum is large enough:
            # We build prefix sum
            # So we move while we haven't completed the fully processed elements
            # So let us just do correct bisection-solution
            break
    
    # Clear-cut solution using full algorithm:
    
    # Fallback to iterative method
    def compute_water_level():
        low = 0.0
        high = 1.0
        epsilon = 1e-12
        for _ in range(100):
            mid = (low + high) / 2.0
            # Compute how much sum would be if set to mid
            proposed = np.clip(sorted_h - mid, 0, 1)
            if np.sum(proposed) > target_sum:
                low = mid
            else:
                high = mid
        return low

    water_level = compute_water_level()
    # Now set all elements as well as possible to water_level
    # But must be <= 1 and have sum = target_sum
    # Use water-filling again:
    z = np.clip(sorted_h - water_level, 0, 1)
    z_rest_sum = np.sum(z)
    # Need to adjust z to target_sum -> distribute rest
    if z_rest_sum > target_sum:
        # need to reduce some elements
        reduction = z_rest_sum - target_sum
        # Reduce from top (larger elements) until sum is correct
        # In worst, all are reduced by min(1, reduction)
        for i in range(n):
            if reduction <= 0:
                break
            to_reduce = min(z[i], reduction)
            z[i] -= to_reduce
            reduction -= to_reduce
    elif z_rest_sum < target_sum:
        # distribute the excess amount
        extra = target_sum - z_rest_sum
        # Distribute to smallest elements (outer loop handles that too)
        for i in reversed(range(n)):
            if extra <= 0:
                break
            # can add at most (1 - z[i]) 
            add = min(1 - z[i], extra)
            z[i] += add
            extra -= add
    return z[sorted_indices.argsort()]

def evaluate_c5(h):
    """Evaluates the C5 constant for a given h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorr(h):
    """Compute aperiodic autocorrelation using FFT."""
    n = len(h)
    nfft = 2 ** int(np.ceil(np.log2(2 * n)))
    # Zero pad h
    h_padded = np.zeros(nfft)
    h_padded[:n] = h
    # Compute FFT
    H = fft(h_padded)
    # Autocorrelation in frequency domain: conj(H) * H
    R = ifft(np.conj(H) * H)
    # Real part, first 2n-1 elements
    r = np.real(R[:2*n-1])
    return r

def admm_loop(g0, n_iterations=500):
    """Main ADMM solver for optimizing C5."""
    # Initialize variables
    rho = 1.0
    z = g0.copy()  # z and h are the same at start
    h = g0.copy()
    u = np.zeros_like(g0)
    t = evaluate_c5(z)
    
    # Precompute
    n = len(g0)
    nfft = 2 ** int(np.ceil(np.log2(2 * n)))
    ones = np.ones(n)
    ones_padded = np.zeros(nfft)
    ones_padded[:n] = ones
    ones_fft = fft(ones_padded)
    
    # Precompute
    h_prev = h.copy()
    
    # Keep track of best solution
    best_t = t
    best_z = z.copy()
    
    # Loop
    last_changes = []
    
    for iteration in range(n_iterations):
        # h-update: solve h = (z - u) - mu (h * 1)
        # Rough estimate of up to 30 bisection steps
        mu_low = 0.0
        mu_high = 1e10
        mu = mu_low
        
        # bisection for mu
        for _ in range(30):
            mu = (mu_low + mu_high) / 2.0
            # Virtual h = (z - u) / (1 + mu * ones_fft)
            z_minus_u = z - u
            z_minus_u_padded = np.zeros(nfft)
            z_minus_u_padded[:n] = z_minus_u
            H = fft(z_minus_u_padded)
            H_new = H / (1.0 + mu * ones_fft)
            h_trial = np.real(ifft(H_new))[:n]
            
            # Check the constraint
            R = compute_autocorr(h_trial)
            C = 1.0 - (2.0 / n) * R
            max_C_s = np.max(C)
            
            if max_C_s <= t:
                mu_high = mu
            else:
                mu_low = mu
        
        # Re-compute with final mu
        mu = mu_high
        z_minus_u = z - u
        z_minus_u_padded = np.zeros(nfft)
        z_minus_u_padded[:n] = z_minus_u
        H = fft(z_minus_u_padded)
        H_new = H / (1.0 + mu * ones_fft)
        h = np.real(ifft(H_new))[:n]
        
        # z-update: project h + u onto box plus sum constraints
        y = h + u
        z = project_to_box_plus_sum(y, n / 2.0)
        
        # dual update
        u = u + (h - z)
        
        # t-update
        t = evaluate_c5(h)
        
        # Keep best solution
        if t < best_t:
            best_t = t
            best_z = z.copy()
        
        # Convergence check
        if len(last_changes) < 5:
            last_changes.append(t)
        else:
            last_changes = last_changes[1:] + [t]
            rel_change = abs(last_changes[-1] - last_changes[0]) / (abs(last_changes[0]) + 1e-12)
            if rel_change < 1e-7:
                break
    
    return best_z, best_t

def nesterov_refinement(z, max_iter=200):
    """Refine solution with Nesterov acceleration."""
    n = len(z)
    best_z = z.copy()
    best_t = evaluate_c5(z)
    h = z.copy()
    
    # Water-filling: precompute scaled "solution"
    # For this subproblem we'll just slightly adjust the approach
    # Do Backtracking line-search + Nesterov acceleration
    
    def compute_C(h):
        n = len(h)
        dx = 2.0 / n
        corr = correlate(h, 1 - h, mode='full')
        corr *= dx  # this is actually the overlap
        return corr
    
    # smooth max surrogate
    def smooth_max(C, beta):
        # For numerical stability
        # shift to avoid overflow
        C_shifted = C - np.max(C) 
        exp_C = np.exp(beta * C_shifted)
        exp_sum = np.sum(exp_C)
        weights = exp_C / exp_sum
        return weights
    
    # gradient smooth max
    def grad_smooth_max(h, beta, C):
        # simplified:
        # Compute 2/n * ( \sum_s w_s * (shift_s(h) + shift_-s(h)) )
        # i.e.,  (shift_s(h) + shift_-s(h)) x w_s
        # we make the 'dual form'
        
        weights = smooth_max(C, beta)
        # eval the stacked h vectors:
        h_conc = np.concatenate((h, [0], h))  # Add zeros on boundaries
        # This is i -> i + s and i -> i - s 
        gr = np.zeros_like(h)
        n_corr = 2*n - 1  # total length in shift-s space
        for s in range(1, n):
            # left[i] = h[i-s] 
            # right[i] = h[i+s] 
            left = np.concatenate(([1e-12] * s, h[:-s] if s != 0 else h))[:n]
            right = np.concatenate((h[s:] if s != 0 else h, [1e-12] * s))[:n]
            
            gr += weights[n-1+s] * (left + right)  # one index shifted form the G_kernel: s in [n-1, n-1+shifts]
            
        return 2.0 / n * gr
    
    # Now do the acceleration loop:
    # initial:
    # beta start at 5
    beta = 5.0
    k = 0
    y = h.copy()
    scheme_step = 0.1
    
    for iter in range(max_iter):
        k += 1
        try:
            C = compute_C(h)
            g = grad_smooth_max(h, beta, C)
            y = h + ((k - 1) / (k + 2)) * (h - h_prev)
            # try much smaller delta
            eta = 1.0
            seen_fail = 0
            while seen_fail < 5:
                h_trial = y - eta * g
                h_trial = project_to_box_plus_sum(h_trial, n / 2.0)
                t_trial = evaluate_c5(h_trial)
                if t_trial < best_t:
                    best_t = t_trial
                    best_z = h_trial.copy()
                if t_trial < (1 - 1e-10) * evaluate_c5(h):
                    h = h_trial
                    break
                else:
                    eta *= 0.5
                    seen_fail += 1
        except Exception:
            pass

        # Update exponent
        if beta < 200:
            beta *= 1.1
            
        h_prev = h.copy()
        
    return best_z, best_t
