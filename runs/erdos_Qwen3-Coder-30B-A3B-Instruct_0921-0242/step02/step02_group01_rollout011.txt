import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from sobol_seq import i4_sobol_generate

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def build_lp_data(h0):
    """Builds the linear programming data from current iterate h0."""
    n = len(h0)
    # Compute the circular auto-correlation of h0
    h_fft = fft(h0, 2 * n)
    corr = ifft(h_fft * np.conj(h_fft))
    c = corr.real[:n]

    # Build the matrix W where row k corresponds to shift k
    W = np.zeros((n, n))
    for k in range(n):
        left_idx = (np.arange(n) + k) % n
        right_idx = (np.arange(n) - k) % n
        W[k, :] = h0[left_idx] + h0[right_idx]
    
    # LP data
    A = -W
    b = -n/2 + c
    
    return A, b

def solve_lp(A, b, n):
    """Solve the LP using scipy linprog."""
    # Variables: h[0] ... h[n-1], t
    # t is added to the constraints
    c = np.ones(n + 1)  # we minimize t (last variable)
    c[-1] = 1.0
    # A * x <= b
    # 0 <= h <= 1
    # sum(h) = n/2  =>  sum(h) - n/2 = 0
    # t >= 0
    
    # Add the constraint ∑h = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = np.array([n/2])
    bounds = [(0, 1) for _ in range(n)] + [(0, None)]
    
    # Solve
    # Note: linprog minimizes c^T * x
    result = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        h_new = result.x[:n]
        t_new = result.x[-1]
        return h_new, t_new
    else:
        # fallback
        h_new = np.ones(n) / 2
        return h_new, float('inf')

def create_seeds(n):
    """Create multiple seeds for multi-start approach."""
    seeds = []
    
    # 1. Uniform
    h_uniform = np.ones(n) * 0.5
    seeds.append(("uniform", h_uniform.copy()))
    
    # 2. Random Dirichlet (uniform on simplex)
    h_dirichlet = np.random.dirichlet(np.ones(n))
    h_dirichlet = h_dirichlet * (n/2) / np.sum(h_dirichlet)
    seeds.append(("dirichlet", h_dirichlet.copy()))
    
    # 3. Binary random (uniform ones and zeros, then perturb)
    binary = np.zeros(n)
    binary[:n//2] = 1
    np.random.shuffle(binary)
    h_binary = binary.copy()
    h_binary += np.random.rand(n) * 0.01
    h_binary = h_binary * (n/2) / np.sum(h_binary)
    seeds.append(("binary", h_binary.copy()))
    
    # 4. Interpolation of initial_h_values (if available and size matches)
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) == n:
            h_interp = orig_h.copy()
        else:
            x_orig = np.linspace(0, 2, len(orig_h))
            x_new = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_orig, orig_h)
        h_interp = h_interp * (n/2) / np.sum(h_interp)
        seeds.append(("interpolated", h_interp.copy()))
    
    # 5. Sobol low-discrepancy sample
    sobol_points = i4_sobol_generate(n, 1).flatten()
    h_sobol = sobol_points / np.sum(sobol_points) * (n/2)
    seeds.append(("sobol", h_sobol.copy()))
    
    return seeds

def projected_gradient_descent(h_start, max_iter=200):
    """Projected gradient descent to refine solution."""
    h = h_start.copy()
    dx = 2.0 / len(h)
    for i in range(max_iter):
        # Find worst shift
        corr = ifft(fft(h, 2*len(h)) * np.conj(fft(1-h, 2*len(h))))
        corr_vals = corr.real[1:len(h)]
        worst_shift = np.argmin(corr_vals) + 1
        
        # Compute subgradient
        grad = np.zeros_like(h)
        left_idx = (np.arange(len(h)) + worst_shift) % len(h)
        right_idx = (np.arange(len(h)) - worst_shift) % len(h)
        grad = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Gradient step
        alpha = 0.01 / np.sqrt(i+1)
        h_new = h - alpha * grad
        
        # Project to [0,1] and normalize sum
        h_new = project_to_feasible(h_new, len(h))
        h = h_new.copy()
    
    return h

def simulated_annealing(h_start, max_iter=10000):
    """Simulated annealing polish using pair-wise moves."""
    h = h_start.copy()
    dx = 2.0 / len(h)
    c5 = evaluate_c5(h)
    T = 0.01
    decay = 0.999
    for i in range(max_iter):
        # Pick two indices
        i1, i2 = np.random.randint(0, len(h), 2)
        if h[i1] <= 0 or h[i2] >= 1:
            continue
        delta = np.random.uniform(0, min(h[i1], 1 - h[i2], 0.01))
        h_new = h.copy()
        h_new[i1] -= delta
        h_new[i2] += delta
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5 or np.random.rand() < np.exp(-(c5_new - c5) / T):
            h = h_new
            c5 = c5_new
        T *= decay
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution
    n_points = 400
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    
    # Generate multiple seeds
    seeds = create_seeds(n_points)
    
    # Iterate with CCP for each seed
    MAX_ITER = 30
    for name, h0 in seeds:
        if time.time() - start_time > budget_s * 0.8:
            break
        h = h0.copy()
        for iteration in range(MAX_ITER):
            if time.time() - start_time > budget_s * 0.8:
                break
            # Build LP data
            A, b = build_lp_data(h)
            # Solve LP
            h_new, t_new = solve_lp(A, b, n_points)
            # Project to feasible set
            h_new = project_to_feasible(h_new, n_points)
            # Compute true C5
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            # Convergence check
            if np.max(np.abs(h_new - h)) < 1e-8:
                break
            h = h_new.copy()
    
    # Polishing steps
    if best_h is not None and time.time() - start_time < budget_s * 0.95:
        # Gradient descent refinement
        h_refined = projected_gradient_descent(best_h, max_iter=200)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Simulated annealing polishing
    if time.time() - start_time < budget_s * 0.99:
        h_anneal = simulated_annealing(best_h, max_iter=10000)
        c5_anneal = evaluate_c5(h_anneal)
        if c5_anneal < best_c5:
            best_c5 = c5_anneal
            best_h = h_anneal.copy()
    
    # Return result
    if best_h is not None:
        return (best_h, best_c5, n_points)
    else:
        h_uniform = np.ones(n_points) * 0.5
        return (h_uniform, float('inf'), n_points)
