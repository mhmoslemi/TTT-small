import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def proj_sum_box(x, target_sum):
    """Project x onto the feasible set: 0 <= x <= 1, sum(x) = target_sum."""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return x
    x = x + diff / len(x)
    x = np.clip(x, 0, 1)
    return x

def evaluate_c5_fft(h):
    """Compute C5 using FFT for efficiency."""
    n = len(h)
    dx = 2.0 / n
    # Convolution of h with (1-h) using FFTs
    h_fft = rfft(h)
    ih_fft = rfft(1 - h)
    conv = irfft(h_fft * np.conj(ih_fft))
    # The correlation values for shifts 1 to n-1 (ignore shift 0)
    corr_vals = conv[:n-1]
    # Include shifts from negative to positive (modular)
    corr_vals = np.concatenate([corr_vals, conv[n-1:]])
    # Return the maximum value scaled by dx
    return np.max(corr_vals) * dx

def eval_gradient_worst_shift(h, k_worst):
    """Compute gradient of C5 w.r.t. h for the worst shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        left_idx = (i + k_worst) % n
        right_idx = (i - k_worst) % n
        grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
    return grad

def gradient_worst_shift_descent(h, max_iter=200, alpha0=0.1, decay=0.5):
    """Run gradient descent on the current worst shift."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = evaluate_c5_fft(h)
    for t in range(max_iter):
        # Find current worst shift
        corr_fft = irfft(rfft(h) * np.conj(rfft(1 - h)))
        corr_vals = corr_fft[:n-1]
        corr_vals = np.concatenate([corr_vals, corr_fft[n-1:]])
        k_worst = np.argmax(corr_vals)
        # Compute gradient
        grad = eval_gradient_worst_shift(h, k_worst)
        # Update with decayed learning rate
        alpha = alpha0 / (np.sqrt(t + 1))
        h_new = h - alpha * grad
        # Project
        h_new = proj_sum_box(h_new, n / 2)
        c5_new = evaluate_c5_fft(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
        else:
            # Accept with small probability for escape
            if np.random.rand() < 0.01:
                h = h_new
    return h

def de_step(pop, F=0.6, CR=0.9):
    """One DE generation step."""
    # Population size
    n_pop = len(pop)
    n_dim = len(pop[0])
    next_pop = []
    for i in range(n_pop):
        # Select three distinct indices for mutation
        a, b, c = np.random.choice(n_pop, 3, replace=False)
        v = pop[a] + F * (pop[b] - pop[c])
        v = np.clip(v, 0, 1)
        # Mutate
        u = np.zeros(n_dim)
        j_rand = np.random.randint(n_dim)
        for j in range(n_dim):
            if np.random.rand() < CR or j == j_rand:
                u[j] = v[j]
            else:
                u[j] = pop[i][j]
        # Enforce sum constraint
        u = proj_sum_box(u, n_dim / 2)
        next_pop.append(u)
    return next_pop

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 400
    n = n_points
    dx = 2.0 / n
    target_sum = n / 2
    
    # Initialize population
    pop = []
    if 'initial_h_values' in globals():
        h_init = initial_h_values
        h_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h_init)), h_init)
        h_interp = proj_sum_box(h_interp, target_sum)
        pop.append(h_interp)
    else:
        pop.append(np.ones(n) * 0.5)
    
    # Add random vectors to fill pop
    while len(pop) < 40:
        h = np.random.rand(n)
        h = proj_sum_box(h, target_sum)
        pop.append(h)
    
    best_h = pop[0].copy()
    best_c5 = evaluate_c5_fft(best_h)
    
    # DE parameters
    F = 0.6
    CR = 0.9
    max_iter = int(4000)  # Floor(budget / (0.001 * 40))
    reseed_freq = 200
    
    last_improvement = time.time()
    
    for gen in range(max_iter):
        if time.time() - start_time > budget_s * 0.70:
            break
            
        # Perform one DE step
        pop = de_step(pop, F, CR)
        
        # Evaluate and update best
        for i in range(len(pop)):
            if time.time() - start_time > budget_s * 0.70:
                break
            h = pop[i]
            c5 = evaluate_c5_fft(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = time.time()
        
        # Periodically refine best_h using gradient descent
        if gen % 200 == 0 and gen > 0:
            if time.time() - start_time < budget_s * 0.90:
                h_refined = gradient_worst_shift_descent(best_h, max_iter=200)
                c5_refined = evaluate_c5_fft(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
                    last_improvement = time.time()
        # Early exit if no progress
        if time.time() - last_improvement > 5.0:
            # Possibly re-seed population
            for i in range(4):
                h = np.random.rand(n)
                h = proj_sum_box(h, target_sum)
                pop[i] = h
            last_improvement = time.time()
    
    # Final optimization
    if time.time() - start_time < budget_s * 0.95:
        # Gradient descent on best
        best_h = gradient_worst_shift_descent(best_h, max_iter=200)
        best_c5 = evaluate_c5_fft(best_h)
    
    # Polishing with "pair swap" simulated annealing
    if time.time() - start_time < budget_s * 0.99:
        h_final = best_h.copy()
        c5_final = evaluate_c5_fft(h_final)
        T = 0.1
        T_min = 1e-5
        alpha = 0.99
        max_iter_swap = 5000
        iter_count = 0
        
        while T > T_min and iter_count < max_iter_swap:
            if time.time() - start_time > budget_s * 0.99:
                break
            iter_count += 1
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.uniform(0, min(h_final[i], 1 - h_final[j], 0.01))
            h_test = h_final.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = proj_sum_box(h_test, target_sum)
            c5_test = evaluate_c5_fft(h_test)
            if c5_test < c5_final:
                h_final = h_test
                c5_final = c5_test
            else:
                prob = np.exp(-(c5_test - c5_final) / T)
                if np.random.rand() < prob:
                    h_final = h_test
                    c5_final = c5_test
            T *= alpha
            
        best_h = h_final
        best_c5 = c5_final
    
    # Optional refinement at higher resolution
    if time.time() - start_time < budget_s * 0.999:
        h_high = np.interp(np.linspace(0, 2, 800), np.linspace(0, 2, n_points), best_h)
        h_high = proj_sum_box(h_high, 400)
        h_high = gradient_worst_shift_descent(h_high, max_iter=100)
        c5_high = evaluate_c5_fft(h_high)
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h = h_high
            n_points = 800
    
    return (best_h, best_c5, n_points)
