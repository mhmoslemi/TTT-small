import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_box_plus_sum(h, n):
    """Euclidean projection onto box [0,1]^n intersected with sum = n/2."""
    # Water-filling algorithm
    y = h.copy()
    sum_y = np.sum(y)
    if abs(sum_y - n/2) < 1e-12:
        return y
    
    # Compute the water level
    # Sort y in descending order
    sorted_y = np.sort(y)[::-1]
    # Find the water level
    cumsum = np.cumsum(sorted_y)
    n_above = np.arange(1, len(sorted_y)+1)
    below_threshold = sorted_y * n_above - cumsum + n/2
    threshold_idx = np.argmax(below_threshold)
    # The water level is the threshold at which sum is exactly n/2
    # We want the minimum of lambda such that sum(max(y_i - lambda, 0)) = n/2
    if threshold_idx == 0:
        lambda_val = sorted_y[0] - (sum_y - n/2) / n
    else:
        lambda_val = (cumsum[threshold_idx-1] - n/2 + sorted_y[threshold_idx] * threshold_idx) / threshold_idx
    # Apply thresholding
    z = np.maximum(y - lambda_val, 0)
    return z

def autocorr(h):
    """Compute aperiodic autocorrelation of h using FFT."""
    n = len(h)
    n_fft = 2**int(np.ceil(np.log2(2*n - 1)))
    h_pad = np.pad(h, (0, n_fft - n), 'constant')
    H = fft(h_pad)
    return ifft(H * np.conj(H)).real[:2*n - 1]

def evaluate_c5(h):
    """Evaluate C5 = max_s [1 - (2/n)*γ_s] where γ_s is the autocorrelation."""
    n = len(h)
    gamma = autocorr(h)
    C = 1 - (2 / n) * gamma
    return np.max(C)

def solve_with_admm(initial_h, n, max_iter=500, time_limit_s=350):
    """Solve with ADMM, returning best solution found."""
    start_time = time.time()
    
    # Initial setup
    rho = 1.0
    z = project_to_box_plus_sum(initial_h, n)
    h = z.copy()
    u = np.zeros_like(z)
    t = evaluate_c5(z)
    
    # Precompute FFT of ones vector
    ones = np.ones(n)
    ones_fft = fft(np.pad(ones, (0, 2*n - 1 - n), 'constant'))
    
    best_z = z.copy()
    best_t = t
    no_improvement_count = 0
    
    # ADMM iteration
    for k in range(max_iter):
        if time.time() - start_time > time_limit_s:
            break
            
        # h-update (solve quadratic with convex inequality constraints)
        # h = argmin_h (rho/2)||h - (z - u)||^2 s.t. 1 - (2/n) (h * h)_s <= t
        # This means solving h = (z - u) - (2/(rho*n)) * (h * h) * mu
        # This is equivalent to h = (z - u) - mu * (h * 1) where * is convolution
        # Using FFT
        z_minus_u = z - u
        z_minus_u_fft = fft(np.pad(z_minus_u, (0, 2*n - 1 - n), 'constant'))
        
        # Bisection on mu to find feasible h
        mu_lo, mu_hi = 0.0, 1e12
        mu = mu_lo
        for _ in range(30):
            if mu_hi - mu_lo < 1e-12:
                break
            mu = (mu_lo + mu_hi) / 2
            
            # h_fft = (z_minus_u_fft) / (1 + mu * ones_fft)
            h_fft = z_minus_u_fft / (1 + mu * ones_fft)
            h_trial = ifft(h_fft).real[:n]

            # Evaluate objective
            gamma = autocorr(h_trial)
            C = 1 - (2 / n) * gamma
            max_C = np.max(C)
            
            if max_C <= t:
                mu_hi = mu
            else:
                mu_lo = mu
                
        # Use final mu to compute h
        mu = mu_lo
        h_fft = z_minus_u_fft / (1 + mu * ones_fft)
        h = ifft(h_fft).real[:n]
        
        # Apply projection
        z = project_to_box_plus_sum(h + u, n)
        
        # Update dual variable
        u += h - z
        
        # Update t (max overlap)
        gamma = autocorr(h)
        C = 1 - (2 / n) * gamma
        t = np.max(C)
        
        # Update best solution
        if t < best_t:
            best_t = t
            best_z = z.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 5:
                break
    
    return best_z, best_t

            else:

                grad += w[n-1+s] * ifft(H * S).real
        grad = -(2/n) * grad
    
        # Backtracking line search for step size
        eta = 1.0
        y = h + ((k-1)/(k+2)) * (h - h_prev)

            beta = min(beta * 1.3, 200.0)
    
    return best_h, best_t
