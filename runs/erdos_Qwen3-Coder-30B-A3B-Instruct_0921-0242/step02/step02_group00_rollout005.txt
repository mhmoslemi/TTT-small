import numpy as np
import time
from scipy.signal import correlate
from scipy.ndimage import gaussian_filter1d
import cvxpy as cp

def aperiodic_autocorr(h):
    """Compute aperiodic autocorrelation using FFT for efficiency."""
    n = len(h)
    # zero-pad to avoid circular effects
    h_padded = np.pad(h, (0, n-1), mode='constant')
    corr = correlate(h_padded, h, mode='valid')
    return corr[:n]

def softmax(β, x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(β * x_shifted)
    return exp_x / np.sum(exp_x)

def convolve_fft(a, b):
    """Convolution using FFT - zero-padded aperiodic."""
    n = len(a)
    a_padded = np.pad(a, (0, n-1), mode='constant')
    b_padded = np.pad(b, (0, n-1), mode='constant')
    return np.real(np.fft.ifft(np.fft.fft(a_padded) * np.fft.fft(b_padded)))[:n]

def project_box_sum(h, total):
    """Project h onto [0,1] and scale to have sum = total."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - total) < 1e-10:
        return h
    diff = total - current_sum
    # Try to distribute diff evenly
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Evaluate C5 constant."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def score_guided_swap(h, ε=1e-5):
    """Perform score-guided swaps to escape plateaus."""
    n = len(h)
    # compute C_s values
    γ = aperiodic_autocorr(h)
    C = 1 - (2.0 / n) * γ

    worst_indices = np.where(np.abs(C - np.max(C)) < ε)[0]

    # compute scores
    scores = np.zeros(n)
    for i in range(n):
        for s in worst_indices:
            if 0 <= i + s < n:
                scores[i] += h[i + s]
            if 0 <= i - s < n:
                scores[i] += h[i - s]

    # pick source (high score) and target (low score)
    src = np.argmax(scores)
    target = np.argmin(scores)

    # move small amount if possible
    delta = min(0.01, h[src], 1 - h[target])
    if delta < 1e-8:
        return h

    h[src] -= delta
    h[target] += delta

    return project_box_sum(h, n / 2)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Phase I: binary MILP seed
    n1 = 120  # smaller for efficacy
    half = n1 // 2

    # Create MILP model
    b = cp.Variable(n1, boolean=True)
    a = cp.Variable((n1, n1), boolean=True)
    γ = cp.Variable(n1)
    L = cp.Variable()

    constraints = [
        cp.sum(b) == half,
        L >= 0
    ]
    
    # McCormick constraints for a[i,s]
    for i in range(n1):
        for s in range(n1 - i):  # s such that i+s < n1
            constraints.append(a[i, s] <= b[i])
            constraints.append(a[i, s] <= b[i + s])
            constraints.append(a[i, s] >= b[i] + b[i + s] - 1)
    
    # Define γ_s
    for s in range(n1):
        constraints.append(γ[s] == cp.sum(a[:, s]))
    
    # Lower bound on all γ
    for s in range(n1):
        constraints.append(γ[s] >= L)
    
    # Objective
    prob = cp.Problem(cp.Maximize(L), constraints)

    # Set solver parameters
    prob.solve(solver=cp.CBC, time_limit=0.2 * budget_s, relative_gap=5e-3)

    # Extract boolean solution
    b_star = np.round(b.value).astype(int)
    if np.sum(b_star) != half:
        # fallback to greedy if infeasible
        b_star = np.zeros(n1)
        indices = np.argsort(b.value)[::-1][:half]
        b_star[indices] = 1

    # Phase II: Upscale to target
    n2 = 300
    x1 = np.linspace(0, 2, n1, endpoint=False)
    x2 = np.linspace(0, 2, n2, endpoint=False)
    h = np.interp(x2, x1, b_star)
    h = project_box_sum(h, n2 / 2)

    # Optional smoothing
    h = gaussian_filter1d(h, sigma=1)
    h = project_box_sum(h, n2 / 2)

    # Phase III: Smooth-max projected gradient descent
    β = 2.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    iteration = 0

    while time.time() - start_time < 0.85 * budget_s:
        γ = aperiodic_autocorr(h)
        C = 1 - (2.0 / n2) * γ
        w = softmax(β, C)

        # Compute gradients via FFT
        conv1 = convolve_fft(w, h)
        conv2 = convolve_fft(w[::-1], h)
        grad = -(2.0 / n2) * (conv1 + conv2)

        # Backtracking line search
        α = 1.0
        h_trial = project_box_sum(h - α * grad, n2 / 2)
        c5_trial = evaluate_c5(h_trial)
        while c5_trial > evaluate_c5(h) and α > 1e-8:
            α *= 0.5
            h_trial = project_box_sum(h - α * grad, n2 / 2)
            c5_trial = evaluate_c5(h_trial)

        h = h_trial
        c5 = evaluate_c5(h)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        if iteration % 150 == 0:
            β = min(β * 1.5, 200)

        if iteration % 200 == 0:
            h = score_guided_swap(h)

        iteration += 1

    # Phase IV: Final sub-gradient polishing
    step = 1
    η0 = 0.1

    while time.time() - start_time < budget_s - 0.1 * budget_s:
        if step > 200:
            break

        γ = aperiodic_autocorr(best_h)
        C = 1 - (2.0 / n2) * γ
        worst_indices = np.where(np.abs(C - np.max(C)) < 1e-5)[0]
        g = np.zeros(n2)
        for s in worst_indices:
            g += np.roll(best_h, -s)
            g += np.roll(best_h, s)
        g = -(2.0 / n2) * g / len(worst_indices)

        η = η0 / np.sqrt(step)
        h_new = project_box_sum(best_h - η * g, n2 / 2)
        c5_new = evaluate_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        step += 1

    return (best_h, best_c5, n2)
