def project_box_sum(y, total):
    """Project y onto {x : 0 ≤ x[i] ≤ 1, sum x = total}"""
    y = np.asarray(y)
    lo, hi = 0.0, 1.0
    L, U = 0.0, 0.0
    for i in range(len(y)):
        L += min(y[i], lo)
        U += max(y[i], hi)
    if total <= L:
        return np.full_like(y, lo)
    if total >= U:
        return np.full_like(y, hi)
    # Binary search for lambda
    lambda_lo, lambda_hi = -1.0, 1.0
    for _ in range(100):
        lam = 0.5 * (lambda_lo + lambda_hi)
        s = 0.0
        for i in range(len(y)):
            s += max(lo, min(hi, y[i] - lam))
        if s < total:
            lambda_hi = lam
        else:
            lambda_lo = lam
    lam = 0.5 * (lambda_lo + lambda_hi)
    x = np.zeros_like(y)
    for i in range(len(y)):
        x[i] = max(lo, min(hi, y[i] - lam))
    return x

def softmax_weights(C, beta):
    """Compute softmax weights w_s = exp(β C_s) / sum exp(β C_t)"""
    C = np.asarray(C)
    C_max = np.max(C)
    exp_C = np.exp(beta * (C - C_max))
    return exp_C / np.sum(exp_C)

def gradient_from_weights(h, w):
    """Compute gradient for smooth max of correlations"""
    n = len(h)
    # For each s, grad_s = - (2/n) * (shift_s(h) + shift_{-s}(h))
    # We compute it via convolutions
    # conv1 = convolve(w, h) on full mode -> length 2n - 1
    # conv2 = convolve(w[::-1], h) on full mode -> length 2n - 1
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    # take central part of size n
    grad = -(2.0 / n) * (conv1[n-1:n-1+n] + conv2[n-1:n-1+n])
    return grad

def evaluate_c5(h):
    """Evaluate C5 using np.correlate properly."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

        # Do max_iter_β iterations for current β
        iter_this_β = 0

            # Compute γ_s = sum h[i] * h[i+s], then C_s = 1 - (2/n) * γ_s
            gamma = np.correlate(h, h, mode='full')

            # Get weights

                    h = h_trial
                    val = val_trial
                    break

                    alpha *= 0.5
                    failed_count += 1

                        alpha = 1e-10
                        break
            
            # Update best

                best_val = val
                best_h = h.copy()
                last_improvement_iter = iteration

                step = min(step * 1.1, 2.0)
            else:
                step *= 0.9
            
            iteration += 1
            iter_this_β += 1

                elapsed = time.time() - t0
                if elapsed > 0.95 * budget_s:

            # Check for stagnation
            if iteration - last_improvement_iter > 500:
                # Random restart
                h = np.random.rand(n)
                h = project_box_sum(h, total_mass)
                last_improvement_iter = iteration
                step = step_init

        # Increase β
        β = min(β * β_factor, β_max)
    
    return best_h, best_val
