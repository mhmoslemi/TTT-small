import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """Evaluate C5 using FFT for efficiency"""
    n = len(h_values)
    dx = 2.0 / n
    h_fft = fft(h_values, 2 * n)
    ih_fft = fft(1 - h_values, 2 * n)
    corr = ifft(h_fft * np.conj(ih_fft))
    max_corr = np.max(corr.real) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def compute_autocorrelations_binary(x):
    """Compute all autocorrelations for binary vector x using FFT."""
    n = len(x)
    if n <= 0:
        return np.array([])
    # x is binary, x^2 = x, so we can convolve with itself
    x_fft = fft(x, 2*n)
    # Autocorrelation of x is x * x^T (circular conv)
    autocorr = ifft(x_fft * np.conj(x_fft)).real
    return autocorr[:n]

def compute_worst_shift_autocorr(x):
    """Return the index of the worst shift and its autocorrelation."""
    n = len(x)
    R = compute_autocorrelations_binary(x)
    # Exclude shift 0 (should be n/2 for uniform)
    if len(R) > 1:
        R = R[1:]  # Remove the first element (shift 0)
        k = np.argmin(R)
        return k + 1, R[k]
    return 0, 0

def update_autocorrelations_for_swap(R, x, i, j, n):
    """Update autocorrelations after swapping x[i] and x[j]."""
    # Original contributions
    orig_i = [x[(i + k) % n] + x[(i - k) % n] for k in range(1, n)]
    orig_j = [x[(j + k) % n] + x[(j - k) % n] for k in range(1, n)]
    
    # New contributions
    new_i = [x[(j + k) % n] + x[(j - k) % n] for k in range(1, n)]
    new_j = [x[(i + k) % n] + x[(i - k) % n] for k in range(1, n)]
    
    # Update R for all shifts
    for k in range(n - 1):
        R[k] = R[k] - orig_i[k] + new_i[k]
        R[k] = R[k] - orig_j[k] + new_j[k]
    
    return R

def swap_based_local_improvement(x, max_iter=2000, timeout_s=100):
    """Improves binary vector by swapping 1s and 0s to increase min autocorr."""
    start = time.time()
    n = len(x)
    
    # Pre-compute all autocorrelations at start
    R = compute_autocorrelations_binary(x)
    R = R[1:]  # Exclude shift 0
    
    # Good to go
    iter_count = 0
    while iter_count < max_iter and (time.time() - start < timeout_s):
        # Find worst shift
        k_star = np.argmin(R)
        # What are the contributions for 1s and 0s?
        contrib_1s = []
        contrib_0s = []
        for i in range(n):
            if x[i] == 1:
                score = x[(i + k_star + 1) % n] + x[(i - k_star + n) % n]
                contrib_1s.append((i, score))
            else:
                score = x[(i + k_star + 1) % n] + x[(i - k_star + n) % n]
                contrib_0s.append((i, score))
        
        if not contrib_1s or not contrib_0s:
            break
            
        best_1_pos, max_score = max(contrib_1s, key=lambda a: a[1])
        worst_0_pos, min_score = min(contrib_0s, key=lambda a: a[1])

        if min_score <= max_score:
            # Swap is beneficial
            x[best_1_pos] = 0
            x[worst_0_pos] = 1
            
            # Update R efficiently
            R = update_autocorrelations_for_swap(R, x, best_1_pos, worst_0_pos, n)
        else:
            # No beneficial swap left; stop
            break
            
        iter_count += 1
        
    return x

def projected_subgradient_descent(h, max_iter=50, timeout_s=30):
    """Iteratively improve fractional solution."""
    start = time.time()
    dx = 2.0 / len(h)
    iter_count = 0
    while iter_count < max_iter and (time.time() - start < timeout_s):
        # Find worst shift
        h_fft = fft(h, 2 * len(h))
        ih_fft = fft(1 - h, 2 * len(h))
        corr = ifft(h_fft * np.conj(ih_fft))
        corr_vals = corr.real[1:]
        k_star = np.argmin(corr_vals) + 1

        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            left_idx = (i + k_star) % len(h)
            right_idx = (i - k_star) % len(h)
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])

        # Step size
        alpha = 0.5 * dx / np.sqrt(iter_count + 1)

        # Update h
        h_new = h - alpha * grad

        # Project to [0,1] and re-enforce sum constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_feasible(h_new, len(h))

        # Evaluate improvement
        c5_new = evaluate_erdos_solution(h_new)
        c5_old = evaluate_erdos_solution(h)

        if c5_new < c5_old:
            h = h_new
        iter_count += 1

    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n = 200 to balance problem size with computation time
    n = 200
    dx = 2.0 / n
    
    # Step 1: SDP-based rounding
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Objective
    objective = cp.Maximize(t)

    # Constraints
    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n / 2
    ]

    # Autocorrelation constraint for each shift k = 1 to n-1
    for k in range(1, n):
        A_k = np.zeros((n, n))
        for i in range(n):
            A_k[i,(i+k)%n] = 1
        constraints.append(cp.trace(A_k @ X) >= t)

    # Add bounds on off-diagonal elements for tighter relaxation
    for i in range(n):
        for j in range(i+1, n):
            constraints.extend([
                X[i,j] >= 0,
                X[i,j] <= X[i,i],
                X[i,j] <= X[j,j]
            ])

    prob = cp.Problem(objective, constraints)
    
    # Use SCS solver with time limit
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            X_val = X.value
            t_val = t.value
            print(f"SDP solver status: {prob.status}, bound: {t_val}")
        else:
            raise Exception("SDP did not converge")
    except Exception as e:
        print(f"SDP failed: {e}")
        # Defer to random binary initialization
        h_start = np.random.rand(n)
        h_start = project_to_feasible(h_start, n)
        x = (h_start > 0.5).astype(int) 
    else:
        # Rounding
        # Principle component rounding (Principal eigenvector of X)
        w, v = np.linalg.eigh(X_val)
        idx = np.argmax(w)
        v = v[:, idx]
        v = np.abs(v)  # Could apply sign eigenvector
        
        # Hyperplane rounding: sample g ~ N(0,I) and compute v * g
        candidates = []
        for _ in range(30):
            g = np.random.normal(0, 1, n)
            z = v * g
            perm_idx = np.argsort(z)[::-1][:n//2]
            x = np.zeros(n)
            x[perm_idx] = 1
            candidates.append(x)
        
        # Add deterministic rounding
        # Project to [0, 1] and sort
        v_sort = np.argsort(v)[::-1][:n//2]
        x_det = np.zeros(n)
        x_det[v_sort] = 1
        candidates.append(x_det)
        
        # Evaluate all candidates
        best_x = None
        best_r = -np.inf
        for x_curr in candidates:
            R = compute_autocorrelations_binary(x_curr)[1:]
            min_r = np.min(R)
            if min_r > best_r:
                best_r = min_r
                best_x = x_curr
        
        if best_x is None:
            # fallback
            x = np.random.rand(n) > 0.5
            x = project_to_feasible(x, n)
        else:
            x = best_x

    # Step 2: Refine with greedy swap
    x = swap_based_local_improvement(x, max_iter=2000, timeout_s=180)

    # Step 3: Final polishing
    h_final = x.astype(float)
    h_final = project_to_feasible(h_final, n)
    c5_final = evaluate_erdos_solution(h_final)

    # Run additional swaps for small amount of time
    # Try a few greedy steps more in case there's still room
    x = h_final
    prev_c5 = float('inf')
    while (time.time() - start_time) < 0.9 * budget_s:
        old_x = x.copy()
        x = swap_based_local_improvement(x, max_iter=100, timeout_s=5)
        new_c5 = evaluate_erdos_solution(x)
        if abs(new_c5 - prev_c5) < 1e-8:
            break  # no improvement
        prev_c5 = new_c5

    h_final = x
    c5_final = evaluate_erdos_solution(h_final)

    return (h_final, c5_final, n)
