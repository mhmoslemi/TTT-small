import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft
from scipy.spatial.distance import cdist
from scipy.stats import dirichlet

def project_to_feasible(h, target_sum):
    """Ensure that sum(h) = target_sum, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    
    # Adjust sum to exact target
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        idx = np.argsort(h)[::-1]
        for i in idx:
            if diff <= 0:
                break
            can_add = min(diff, 1 - h[i])
            h[i] += can_add
            diff -= can_add
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def generate_seeds(n_points):
    """Generate diverse starting points."""
    seeds = []
    
    # 1. Uniform
    h_uniform = np.ones(n_points) * 0.5
    seeds.append(project_to_feasible(h_uniform, n_points / 2))
    
    # 2. Dirichlet (random)
    h_dir = dirichlet.rvs([1] * n_points, size=1)[0]
    h_dir = h_dir * (n_points / 2) / np.sum(h_dir)
    seeds.append(project_to_feasible(h_dir, n_points / 2))
    
    # 3. Binary with noise
    h_bin = np.zeros(n_points)
    h_bin[:n_points // 2] = 1
    np.random.shuffle(h_bin)
    h_bin += np.random.uniform(-0.05, 0.05, n_points)

    # 4. Interpolate initial values

        h_interp = np.interp(np.linspace(0, 2, n_points),

        h_interp = project_to_feasible(h_interp, n_points / 2)
        seeds.append(h_interp)
    
    # 5. Sobol-like (quasi-random)
    indices = np.random.permutation(n_points)
    h_sobol = np.zeros(n_points)
    h_sobol[indices[:n_points // 2]] = 1
    h_sobol = h_sobol + np.random.uniform(-0.02, 0.02, n_points)
    seeds.append(project_to_feasible(h_sobol, n_points / 2))
    
    return seeds

def build_lp_data(h0):
    """Build data for linear program from h0."""
    n = len(h0)
    # Compute auto correlation
    c = ifft(fft(h0) * np.conj(fft(h0))).real
    c0 = c
    
    # Build matrix W
    W = np.zeros((n - 1, n))
    for k in range(1, n):
        W[k - 1, :] = h0[(np.arange(n) + k) % n] + h0[(np.arange(n) - k) % n]
    
    # LP data
    A = -W
    b = -n/2 + c0[1:]
    
    return A, b

def solve_subproblem(A, b, n_points):
    """Solve LP subproblem."""
    # Build the full LP
    # Minimize t subject to A h + t*1 <= b, 0 <= h <= 1, sum(h) = n/2, t >= 0
    
    c_full = np.zeros(n_points + 1)
    c_full[-1] = 1.0  # minimize t
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, n_points + 1))
    A_eq[0, :n_points] = 1.0
    b_eq = np.array([n_points / 2.0])
    
    # Inequality constraint: A h + t*1 <= b
    A_ub = np.hstack([A, np.ones((A.shape[0], 1))])
    b_ub = b
    
    # Bound constraints: [0, 1] for h, t >= 0
    bounds = [(0.0, 1.0)] * n_points + [(0.0, None)]

    result = linprog(c_full, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs', tol=1e-8)
    
    if result.success:
        h = result.x[:-1]
        t = result.x[-1]
        return h, t
    else:
        # fallback to a utility-based approximation
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points / 2)
        return h, evaluate_c5(h)

def projected_gradient_descent(h, max_iter=200):
    """Projected gradient descent based on worst shift."""
    dx = 2.0 / len(h)
    for t in range(max_iter):
        if time.time() > start_time + 0.995 * budget_s:
            break
        
        # Compute full correlation
        corr = ifft(fft(h, 2 * len(h)) * np.conj(fft(1 - h, 2 * len(h)))).real
        # Remove k=0 (self-correlation)
        corr_max = np.max(corr[1:])
        worst_shift = np.argmax(corr[1:]) + 1
        
        # Compute gradient
        grad = np.zeros_like(h)
        grad[1:] = (1 - h[:-1])
        grad[0] = (1 - h[-1])
        grad[:-1] -= h[1:]
        grad[-1] -= h[0]
        
        # Projected update (follow 1/sqrt(t) schedule)
        alpha = 0.1 / np.sqrt(t + 1)
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h) / 2)
        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h):
            h = h_new.copy()
    
    return h

def simulated_annealing(h, max_iter=1000):
    """Polish with simulated annealing."""
    dx = 2.0 / len(h)
    current_c5 = evaluate_c5(h)
    
    T = 0.1  # Initial temperature
    Tmin = 1e-6  # Minimum temperature
    alpha = 0.99  # Cooling rate
    
    for _ in range(max_iter):
        if time.time() > start_time + 0.999 * budget_s:
            break
        i = np.random.randint(len(h))
        j = np.random.randint(len(h))
        
        if h[i] <= 0 or h[j] >= 1:
            continue
        delta = np.random.uniform(0, min(h[i], 1 - h[j], 0.01))
        h_test = h.copy()
        h_test[i] -= delta
        h_test[j] += delta
        h_test = project_to_feasible(h_test, len(h) / 2)
        c5_test = evaluate_c5(h_test)
        
        if c5_test <= current_c5:
            h = h_test
            current_c5 = c5_test
        else:
            if np.random.rand() < np.exp(-(c5_test - current_c5) / T):
                h = h_test
                current_c5 = c5_test
        
        T = max(Tmin, T * alpha)
    
    return h
