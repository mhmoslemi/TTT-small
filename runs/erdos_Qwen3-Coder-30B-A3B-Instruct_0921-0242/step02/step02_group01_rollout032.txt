import numpy as np
import time
from scipy.linalg import eigh
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cvxpy as cp
from cvxpy import trace, reshape

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    if abs(diff) < 1e-10:
        return h
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h):
    """Compute C5 using FFT-based correlation."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # Take real part, ignore index 0
    max_corr = np.max(corr.real[1:])
    return max_corr * dx

def compute_autocorrelation(h):
    """Compute the full autocorrelation vector"""
    dx = 2.0 / len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    return corr.real

def evaluate_solution(h):
    """Evaluate a binary solution for the min autocorrelation."""
    dx = 2.0 / len(h)
    corr_vals = compute_autocorrelation(h)
    min_corr = np.min(corr_vals[1:])  # exclude shift 0
    return 1 - min_corr * dx

def solve_sdp_relaxation(n, time_budget):
    """Solve SDP relaxation to find lower bound on minimum autocorrelation."""
    start_time = time.time()
    try:
        # Create variables
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            X >> 0,  # Positive semidefinite
            cp.diag(X) >= 0,
            cp.diag(X) <= 1,
            cp.sum(cp.diag(X)) == n/2,
        ]
        
        # Autocorrelation constraints for all shifts k=1 to n-1
        for k in range(1, n):
            A_k = np.zeros((n, n))
            for i in range(n):
                A_k[i, (i+k)%n] = 1
            constraints.append(trace(A_k @ X) >= t)
        
        # Pairwise bounds for tighter relaxation
        for i in range(n):
            for j in range(i+1, n):
                constraints.append(X[i,j] >= 0)
                constraints.append(X[i,j] <= X[i,i])
                constraints.append(X[i,j] <= X[j,j])
        
        # Objective
        objective = cp.Maximize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve with early stopping
        solver_time = time_budget * 0.3
        prob.solve(solver=cp.SCS, verbose=False, time_limit=solver_time)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            t_val = t.value
            X_val = X.value
            return X_val, t_val
        else:
            return None, None
    except Exception:
        return None, None

def binarize_solution(X_val, n, num_rounds=30):
    """Use random hyperplane rounding to get binary vectors."""
    if X_val is None:
        return None
    
    # Eigen-decomposition
    try:
        eigvals, eigvecs = eigh(X_val)
        idx = np.argsort(eigvals)[::-1]
        v = np.sqrt(eigvals[idx[0]]) * eigvecs[:, idx[0]]
    except:
        # fallback: deterministic rounding
        diag_vals = np.diag(X_val)
        v = diag_vals
    
    best_x = None
    best_min_corr = -np.inf
    
    # Random hyperplane rounding
    for _ in range(num_rounds):
        g = np.random.normal(0, 1, n)
        z = np.dot(eigvecs, np.sqrt(np.maximum(eigvals, 0)) * g)
        x = np.zeros(n)
        x[np.argsort(z)[::-1][:n//2]] = 1
        corr_vals = compute_autocorrelation(x)
        min_corr = np.min(corr_vals[1:])
        if min_corr > best_min_corr:
            best_min_corr = min_corr
            best_x = x.copy()
    
    # deterministic rounding as backup
    if best_x is None:
        x = np.zeros(n)
        x[np.argsort(v)[::-1][:n//2]] = 1
        corr_vals = compute_autocorrelation(x)
        min_corr = np.min(corr_vals[1:])
        best_x = x
        best_min_corr = min_corr
    
    return best_x

def greedy_swap_refinement(h, time_budget):
    """Refine a binary solution by greedy swapping."""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    
    # Fast correlation computation using FFT
    h_fft = fft(h, 2 * n)
    ih_fft = fft(1 - h, 2 * n)
    corr = ifft(h_fft * np.conj(ih_fft)).real
    corr = np.concatenate([corr[1:], corr[:1]])  # move shift 0 to end
    
    # find k* = argmin corr_k
    k_star = np.argmin(corr[1:]) + 1  # exclude 0
    
    max_swaps = int(time_budget * 500)
    swaps = 0
    
    while swaps < max_swaps:
        if time.time() - start_time > time_budget * 0.999:
            break
        
        # Precompute correlation contributions for k_star
        contrib_1 = []  # for 1s
        contrib_0 = []  # for 0s
        
        for i in range(n):
            if h[i] == 1:
                contrib_1.append(h[(i + k_star) % n] + h[(i - k_star) % n])
            else:
                contrib_0.append(h[(i + k_star) % n] + h[(i - k_star) % n])
        
        if not contrib_1 or not contrib_0:
            break
            
        max_contrib_1 = max(contrib_1)
        min_contrib_0 = min(contrib_0)
        
        if min_contrib_0 <= max_contrib_1:
            break  # no improving move

        # Find indices for swapping
        i_best = -1
        j_best = -1

        for i in range(n):
            if h[i] == 1 and (h[(i + k_star) % n] + h[(i - k_star) % n]) == max_contrib_1:
                i_best = i
                break

        for j in range(n):
            if h[j] == 0 and (h[(j + k_star) % n] + h[(j - k_star) % n]) == min_contrib_0:
                j_best = j
                break

        if i_best == -1 or j_best == -1:
            break
        
        # Perform swap and update correlations
        h[i_best] = 0
        h[j_best] = 1
        
        swaps += 1
        
        # Re-compute the correlation vector for this shift
        h_fft = fft(h, 2 * n)
        ih_fft = fft(1 - h, 2 * n)
        corr = ifft(h_fft * np.conj(ih_fft)).real
        corr = np.concatenate([corr[1:], corr[:1]])  # move shift 0 to end
        
        k_star = np.argmin(corr[1:]) + 1  # recompute worst shift

    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # pick reasonable size
    n = 200
    
    # SDP relaxation
    X_val, t_val = solve_sdp_relaxation(n, budget_s)
    
    if X_val is None or t_val is None:
        # fallback to a simple random heuristic if SDP failed
        h_best = np.zeros(n)
        h_best[:n//2] = 1
        np.random.shuffle(h_best)
        h_best = project_to_feasible(h_best, n)
        for _ in range(1000):
            if time.time() - start_time > budget_s * 0.95:
                break
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if h_best[i] == 0 or h_best[j] == 1:
                continue
            h_test = h_best.copy()
            h_test[i] = 0
            h_test[j] = 1
            h_test = project_to_feasible(h_test, n)
            if compute_c5(h_test) < compute_c5(h_best):
                h_best = h_test
        return h_best, compute_c5(h_best), n
    
    # Binarization
    h_guess = binarize_solution(X_val, n)
    if h_guess is None:
        h_guess = np.ones(n) * 0.5
        h_guess[:n//2] = 0
    
    # Greedy swap
    h_refined = greedy_swap_refinement(h_guess, budget_s * 0.3)
    
    # Optional polishing with projected gradient
    if time.time() - start_time < budget_s * 0.98:
        h_final = h_refined.copy()
        # compute worst shift
        corr_vals = compute_autocorrelation(h_final)
        k_star = np.argmin(corr_vals[1:]) + 1
        
        num_steps = 300
        for i in range(num_steps):
            if time.time() - start_time > budget_s * 0.98:
                break
                
            # compute gradient
            grad = np.zeros(n)
            for idx in range(n):
                grad[idx] = (1 - h_final[(idx + k_star) % n]) - h_final[(idx - k_star) % n]
            
            alpha = 0.5 * (2.0 / n) / np.sqrt(i + 1)
            h_new = h_final - alpha * grad
            h_new = project_to_feasible(h_new, n)
            
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h_final)
            if c5_new < c5_old:
                h_final = h_new
                # recompute worst shift
                corr_vals = compute_autocorrelation(h_final)
                k_star = np.argmin(corr_vals[1:]) + 1
    
    # Final evaluation
    c5_bound = compute_c5(h_final)
    return h_final, c5_bound, n
