import numpy as np
import time

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def evaluate_autocorr(h):
    """Evaluate the autocorrelations efficiently using FFT."""
    n = len(h)
    H = np.fft.rfft(h, n)
    autocorr = np.fft.irfft(H * np.conj(H), n)
    return np.real_if_close(autocorr)

def binary_anneal(h_bin, time_limit):
    """Binary Annealing: maximize minimal autocorrelation using simulated annealing."""
    start_time = time.time()
    n = len(h_bin)
    dx = 2.0 / n
    # Current autocorrelations
    ac = evaluate_autocorr(h_bin)
    c = dx * (n/2 - ac)
    best_c = c.max()
    best_h = h_bin.copy()

    # Temperature schedule
    T0 = 5 * dx
    T = T0
    decay = 0.9997
    swap_count = 0
    stagnation_count = 0
    max_stagnation = 200000

    while time.time() - start_time < time_limit:
        if stagnation_count > max_stagnation:
            # Large jump reset
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            n_ones = len(ones)
            n_zeros = len(zeros)
            # Replace 5% of ones
            n_to_replace = max(1, int(0.05 * n_ones))
            to_zero = np.random.choice(ones, n_to_replace, replace=False)
            to_one = np.random.choice(zeros, n_to_replace, replace=False)
            h_new = h_bin.copy()
            h_new[to_zero] = 0
            h_new[to_one] = 1
            h_bin = h_new
            ac = evaluate_autocorr(h_bin)
            c = dx * (n/2 - ac)
            best_c = c.max()
            best_h = h_bin.copy()
            stagnation_count = 0
            continue

        i = np.random.randint(0, n)
        if h_bin[i] == 0:
            continue
        j = np.random.randint(0, n)
        if h_bin[j] == 1:
            continue

        # Swap h[i] and h[j] (0/1)
        h_trial = h_bin.copy()
        h_trial[i] = 0
        h_trial[j] = 1

        # Update autocorrelation
        # shift vectors for both elements
        delta = np.zeros(n)
        delta[i] = -1
        delta[j] = 1
        delta = np.roll(delta, i-j)
        delta[0] += 2 # for shift 0
        # Add self corrections
        if i != j:
            delta[i-j] -= 1
            delta[j-i] -= 1
        else:
            delta[0] -= 1

        ac_new = ac + delta
        c_new = dx * (n/2 - ac_new)
        new_bound = c_new.max()

        if new_bound <= best_c:
            h_bin = h_trial
            ac = ac_new
            c = c_new
            best_c = new_bound
            best_h = h_bin.copy()
            stagnation_count = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp((best_c - new_bound) / T):
                h_bin = h_trial
                ac = ac_new
                c = c_new
                stagnation_count = 0
            else:
                stagnation_count += 1

        # decrease temperature
        swap_count += 1
        if swap_count % 1000 == 0:
            T = max(T * decay, dx * 1e-4)

    return best_h, best_c

def smooth_boundaries(h_bin, ramp_width=2):
    """Convert a binary pattern into a smooth step function."""
    n = len(h_bin)
    h_cont = h_bin.astype(float)
    # Scan for transitions and smooth them
    for i in range(n):
        if h_bin[i] == 0 and h_bin[(i-1) % n] == 1:
            # This is a 1 -> 0 transition, smooth it
            for r in range(1, ramp_width+1):
                j = (i + r) % n
                h_cont[j] = float(r) / (ramp_width + 1)
        elif h_bin[i] == 1 and h_bin[(i-1) % n] == 0:
            # This is a 0 -> 1 transition,
            for r in range(1, ramp_width+1):
                j = (i - r) % n
                h_cont[j] = float(r) / (ramp_width + 1)
    return h_cont

def proj_grad_refine(h, time_limit, use_logsumexp=False):
    """Projected gradient descent on the max overlap."""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n

    best_h = h.copy()
    best_c = evaluate_c5(h)
    if use_logsumexp:
        gamma = 10.0
        def logsumexp_autocorr(h_val):
            ac = evaluate_autocorr(h_val)
            c_vals = dx * (n/2 - ac)
            return 1/gamma * np.log(np.sum(np.exp(gamma * c_vals))),
        def grad_logsumexp(h_val):
            ac = evaluate_autocorr(h_val)
            c_vals = dx * (n/2 - ac)
            exp_vals = np.exp(gamma * c_vals)
            exp_sum = np.sum(exp_vals)
            soft_c = exp_vals / exp_sum
            # Gradient of -sum_(i,k) h_i h_{i-k} = -sum_k( h_{i+k} + h_{i-k} ) * soft_c(k)
            g = np.zeros_like(h_val)
            for k in range(n):
                g += -dx * soft_c[k] * (np.roll(h_val, k) + np.roll(h_val, -k))
            return g
    else:
        def grad_max(h_val):
            ac = evaluate_autocorr(h_val)
            c_vals = dx * (n/2 - ac)
            k_star = np.argmax(c_vals)
            g = -dx * (np.roll(h_val, k_star) + np.roll(h_val, -k_star))
            return g

    # Initialization
    current_h = h.copy()
    current_c = evaluate_c5(h)

    def backtracking_line_search(g, eta0=0.5, decay=0.5, max_iter=20):
        eta = eta0
        for _ in range(max_iter):
            h_trial = current_h - eta * g
            h_trial = project_to_feasible(h_trial, n)
            trial_c = evaluate_c5(h_trial)
            if trial_c < current_c:
                return h_trial, trial_c, eta
            eta *= decay
        return current_h, current_c, eta

    iteration = 0
    while time.time() - start_time < time_limit:
        if use_logsumexp:
            grad = grad_logsumexp(current_h)
        else:
            grad = grad_max(current_h)
        h_trial, c_trial, _ = backtracking_line_search(grad)
        if c_trial < current_c:
            current_h = h_trial
            current_c = c_trial
            if current_c < best_c:
                best_c = current_c
                best_h = current_h.copy()
        iteration += 1

        if iteration % 100 == 0:
            # Also perform pairwise fractional swaps
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            while current_h[i] <= 0.001:
                i = np.random.randint(0, n)
            while current_h[j] >= 0.999:
                j = np.random.randint(0, n)
            delta = np.random.uniform(0.001, min(current_h[i], 1 - current_h[j], 0.01))
            h_trial = current_h.copy()
            h_trial[i] -= delta
            h_trial[j] += delta
            h_trial = project_to_feasible(h_trial, n)
            trial_c = evaluate_c5(h_trial)
            if trial_c < current_c:
                current_h = h_trial
                current_c = trial_c
                if current_c < best_c:
                    best_c = current_c
                    best_h = current_h.copy()

    return best_h, best_c

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    # Phase 1: Initialization with binary pattern and initial C5 bound
    n0 = 256  # Seems optimal
    dx0 = 2.0 / n0
    # Intial h via the user-provided initial_h_values padded/truncated to n0
    h0_bin = initial_h_values[:n0].copy()
    # Convert to binary by threshold at 0.5
    h0_bin = (h0_bin > 0.5).astype(int)
    # Ensure weight is exactly n0/2
    ones_indices = np.where(h0_bin == 1)[0]
    zeros_indices = np.where(h0_bin == 0)[0]
    current_weight = len(ones_indices)
    target_weight = n0 // 2
    if current_weight > target_weight:
        # Remove excess ones
        excess = current_weight - target_weight
        inds_to_zero = np.random.choice(ones_indices, size=excess, replace=False)
        h0_bin[inds_to_zero] = 0
    if current_weight < target_weight:
        # Add more ones
        deficit = target_weight - current_weight
        inds_to_one = np.random.choice(zeros_indices, size=deficit, replace=False)
        h0_bin[inds_to_one] = 1
    h0_bin = h0_bin.astype(float)

    # Part of initial budget to Ph1
    remaining_budget = 0.55 * budget_s
    h_best_bin, c5_bin = binary_anneal(h0_bin, remaining_budget)

    # Phase 2: Smooth boundaries and project gradient descent
    h_smooth = smooth_boundaries(h_best_bin, ramp_width=2)
    h_smooth = project_to_feasible(h_smooth, n0)
    remaining_budget = 0.25 * budget_s
    h_final, c5_final = proj_grad_refine(h_smooth, remaining_budget)

    # Phase 3: Progressive upsampling
    n1 = 2 * n0
    dx1 = 2.0 / n1
    h_upsampled = np.interp(np.linspace(0, 2, n1), np.linspace(0, 2, n0), h_final)
    h_upsampled = project_to_feasible(h_upsampled, n1)
    # Short refine
    remaining_budget = 0.15 * budget_s
    h_upsampled, c5_upsampled = proj_grad_refine(h_upsampled, remaining_budget)
    n_points = n1
    final_c5 = c5_upsampled

    return (h_upsampled, final_c5, n_points)
