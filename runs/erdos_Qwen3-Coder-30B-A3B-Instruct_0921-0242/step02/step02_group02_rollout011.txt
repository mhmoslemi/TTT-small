import numpy as np

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def compute_autocorr(h, k):
    """Compute the autocorrelation term."""
    n = len(h)
    dx = 2.0 / n
    # Circular convolution of h with h_{i-k}
    return dx * np.sum(h * np.roll(h, k))

def compute_gradient(h, k):
    """Computes gradient of a single correlation term."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    for i in range(n):
        # Left term: -dx * h_{i + k}
        left_idx = (i + k) % n
        grad[i] -= dx * h[left_idx]
        
        # Right term: dx * h_{i - k}
        right_idx = (i - k) % n
        grad[i] += dx * h[right_idx]
        
    return grad

def compute_smooth_gradient(h, beta=20):
    """Computes smooth approximation of the gradient."""
    n = len(h)
    dx = 2.0 / n
    corr_vals = np.array([compute_autocorr(h, k) for k in range(n)])
    weights = np.exp(beta * corr_vals) / np.sum(np.exp(beta * corr_vals))
    
    grad = np.zeros_like(h)
    for k in range(n):
        g = compute_gradient(h, k)
        grad += weights[k] * g
    
    return grad

def solve_sdp_relaxation(n, max_iter=100):

    target_autocorr = np.zeros(n)

    for k in range(1, min(10, n)):
        if k * 2 < n:
            target_autocorr[k] = 0.2
        else:
            target_autocorr[k] = 0.1
            
    # Reweight to sum of 0.5* n
    target_autocorr /= np.sum(target_autocorr)
    target_autocorr *= n/2
    
    # Add a simplified SDP-like constraint to compute gamma for target
    # Create a dominance-based bound directly
    gamma_bound = 0
    for k in range(n):
        if k == 0:
            gamma_bound = max(gamma_bound, 0.5 * n * dx - 0)  # h^T h = n/2
        else:

    # If still conservative, adjust
    # Simple relaxation from typical optimal solutions to guide

        estimated_lb = np.mean(target_autocorr)
        
        return estimated_lb, h_sdp, target_autocorr

def ramp_smooth_step(h, w=2):
    """Apply smooth step function."""
    dx = 2.0 / len(h)
    ramped = np.zeros_like(h)
    for i in range(len(h)):
        idxl = max(0, i - w)
        idxr = min(len(h), i + w + 1)
        # Smooth transition from 0 to 1 within the window
        mean_val = np.mean(h[idxl:idxr])
        if abs(h[i] - mean_val) < 0.1:
            ramped[i] = h[i]
        else:
            ramped[i] = mean_val
    
    # Now rescale
    scaled = ramped * (1.0 / (np.mean(ramped) + 1e-10)) * (len(h) / 2)
    return np.clip(scaled, 0, 1)

def project_l1_l2(h, n):
    """Project to feasible set using binary projection."""
    h = np.clip(h, 0, 1)
    curr_sum = np.sum(h)
    if abs(curr_sum - n/2) < 1e-6:
        return h
    
    # Water-filling to achieve right sum
    h_flat = h.flatten()
    h_sorted = np.sort(h_flat)[::-1]
    cum_sum = np.cumsum(h_sorted)
    
    # Binary search on drop target
    l, u = 0.0, 1.0
    for _ in range(50):
        mid = (l + u) / 2
        trial = np.clip(h_flat - mid, 0, 1)
        if np.sum(trial) <= n / 2:
            l = mid
        else:
            u = mid
    
    h = np.clip(h_flat - l, 0, 1)
    h = h * (n/2) / np.sum(h)
    return h

def refine_continuous(h, max_iter=2000, beta_start=5, beta_max=100):
    """Refine the solution using projected L-BFGS-B approach."""
    n = len(h)
    dx = 2.0 / n
    start_time = time.time()
    
    # Set up the approximation surrogate for non-smooth objective
    def obj_func(h_vec):
        h = h_vec.reshape(-1)
        corr_vals = np.array([compute_autocorr(h, k) for k in range(n)])
        # Use log-sum-exp for smooth approximation
        beta = beta_start
        return (1.0/beta) * np.log(np.sum(np.exp(beta * corr_vals))) * dx

    def grad_func(h_vec):
        h = h_vec.reshape(-1)
        return compute_smooth_gradient(h)
    
    h_flat = h.copy()
    
    # Initialize with projected values
    h_flat = project_to_feasible(h_flat, n)
    
    # Run projected gradient steps manually for the smooth version
    for i in range(max_iter):
        if i > 0 and i % 100 == 0:
            beta_start = min(beta_start * 2.0, beta_max)
        try:
            grad = grad_func(h_flat)
            h_new = h_flat - 0.01 * grad
            h_new = project_to_feasible(h_new, n)
            if np.abs(obj_func(h_new) - obj_func(h_flat)) < 1e-10:
                break
            h_flat = h_new
        except Exception as e:
            break
        if time.time() - start_time > 590:
            break
    
    # Use final exact computation
    final_h = project_to_feasible(h_flat, n)
    final_c5 = evaluate_c5(final_h)
    
    return final_h, final_c5
