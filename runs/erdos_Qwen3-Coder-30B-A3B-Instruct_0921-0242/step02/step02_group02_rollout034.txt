import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar

def project_feasible(v, n):
    """Project v onto the set {x: sum(x)=n/2, 0<=x<=1} using water-filling."""
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - n/2) < 1e-10:
        return v
    
    # Water filling approach
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    # Find where the water level should be
    water_level = (n/2 - cumsum[-1]) / n
    if water_level > 0:
        # Need to raise all
        water_level = (n/2 - np.sum(v_sorted)) / n + np.mean(v_sorted)
    else:
        # Water level somewhere in middle
        for i in range(n):
            if i == 0:
                if n/2 - cumsum[0] < 0:
                    water_level = (n/2 - cumsum[i]) / (i+1)
                    break
            else:
                if n/2 - cumsum[i] < 0:
                    water_level = (n/2 - cumsum[i-1]) / (i+1)
                    break
        else:
            water_level = n/2 / n
            
    if water_level > 1.0:
        water_level = 1.0
    elif water_level < 0.0:
        water_level = 0.0
        
    # Apply the water level
    updated = np.clip(v - water_level, 0, 1)
    return updated

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT."""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation via FFT
    corr = np.real(ifft(fft(h) * np.conj(fft(1-h))).real)
    # max correlation
    max_corr = np.max(corr) * dx
    return max_corr

def gradient_at_worst(h):
    """Compute subgradient w.r.t. h of max correlation at worst shift."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations and find max
    corr = np.real(ifft(fft(h) * np.conj(fft(1-h))).real)
    k_star = np.argmax(corr)
    
    # Compute gradient for that specific k*; slow but simple
    grad = np.zeros(n)
    for i in range(n):
        i_plus = (i + k_star) % n
        i_minus = (i - k_star) % n
        grad[i] = dx * (h[i_plus] - h[i_minus])
    return grad, k_star

def line_search(h, grad, eta0=0.5):
    """Backtracking line search for gradient step."""
    eta = eta0
    h_new = h - eta * grad
    h_new = project_feasible(h_new, len(h))
    for _ in range(10):
        h_new = h - eta * grad
        h_new = project_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h):
            return h_new, c5_new
        eta /= 2.0
    return h, evaluate_c5(h)

def mutate(v, mode, params):
    """Apply mutation to v."""
    n = len(v)
    if mode == "gauss":
        indices = np.random.choice(n, size=np.random.randint(5,10), replace=False)
        sigma = params["sigma"]
        v_new = v.copy()
        v_new[indices] += np.random.normal(0, sigma, len(indices))
        v_new = project_feasible(v_new, n)
        return v_new
    elif mode == "swap":
        eps = 0.01
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        while v[i] <= eps:
            i = np.random.randint(0, n)
        while v[j] >= 1 - eps:
            j = np.random.randint(0, n)
        delta = np.random.uniform(0.001, min(v[i], 1 - v[j], 0.01))
        v_new = v.copy()
        v_new[i] -= delta
        v_new[j] += delta
        v_new = project_feasible(v_new, n)
        return v_new
    elif mode == "dct":
        # DCT mode tweak
        v_new = v.copy()
        # Apply low freq DCT coefficient
        dct_data = np.fft.dct(v_new)
        F = np.random.randint(1, min(10, len(dct_data)//2 + 1))
        if np.random.rand() < 0.5:
            dct_data[F] += np.random.uniform(-0.01, 0.01)
        else:
            dct_data[F] = 0
        v_new = np.fft.idct(dct_data)
        v_new = project_feasible(v_new, n)
        return v_new
    return v

    # Mutation phase
