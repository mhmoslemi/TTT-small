def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    
    # Water-filling approach
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def sdp_lower_bound(n):
    """Compute lower bound via SDP relaxation."""
    # Build circulant matrix C_k for each k
    dx = 2.0 / n
    C = []
    for k in range(n):
        C_k = np.zeros((n, n))
        for i in range(n):
            C_k[i, (i + k) % n] = 1
        C.append(C_k)
    
    # SDP variables
    h = Variable(n)
    gamma = Variable()
    
    # Constraints: Tr(C_k * X) >= n/2 - gamma/dx for all k
    # where X = h * h^T
    constraints = []
    for k in range(n):
        constraints.append(trace(C[k] @ (h @ h.T)) >= n/2 - gamma/dx)
    
    # Box constraints
    constraints.append(0 <= h)
    constraints.append(h <= 1)
    constraints.append(sum(h) == n/2)
    
    # Objective
    objective = Minimize(gamma)
    
    # Solve
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=SCS, max_iters=1000, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return gamma.value, h.value
    except Exception:
        pass
    
    return None, None

def binary_programming_phase(n, target_autocorr, time_limit, num_cuts=50):
    """Solve binary integer program with cut generation."""
    # Initialize binary variables
    b = cp.Variable(n, boolean=True)
    gamma = cp.Variable()
    
    # Constraints for autocorrelations
    constraints = [
        cp.sum(b) == n/2,
    ]
    
    # Add initial constrained at most num_cuts shifts
    # Use a simple approach: first few largest correlation values from SDP output
    # For simplicity, consider a fixed set of shifts to reduce problem size
    shifts_to_consider = list(range(0, min(10, n), 1)) + list(range(max(10, n//4), min(20, n), 1))

    c = []
    for k in shifts_to_consider:
        # Linearization of b_i * b_{i-k} using McCormick envelopes
        z = cp.Variable(n, boolean=True)
        # b_i * b_{i-k} <= z_i
        constraints.append(z <= b)
        constraints.append(z <= b[np.roll(np.arange(n), k)])
        # b_i * b_{i-k} >= b_i + b_{i-k} - 1
        constraints.append(z >= b + b[np.roll(np.arange(n), k)] - 1)
        # z_i >= 0
        constraints.append(z >= 0)
        s_k = cp.sum(z)
        c.append(s_k)
        constraints.append(dx * (n/2 - s_k) <= gamma)
    
    # Objective
    objective = cp.Minimize(gamma)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=CPLEX, time_limit=time_limit, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return b.value, gamma.value
    except Exception:
        pass
    
    return None, None

def ramp_smooth_and_project(b, n):
    """Convert binary pattern b to continuous one with ramps."""
    dx = 2.0 / n
    h = np.zeros(n)
    
    # Use a ramp function to smooth transitions
    w = 4  # width of the ramp
    
    for i in range(n):
        # Create ramps around transitions from 0 to 1
        # Simple example: assume one ramp around each transition
        if b[i] == 1:
            h[i] = 1
        else:
            h[i] = 0
    
    # Smoothing via averaging with neighborhood
    # Simple algorithm: convolve with small window
    window_size = max(3, w//2)
    h_smooth = np.zeros(n)
    
    for i in range(n):
        lo = max(0, i - window_size)
        hi = min(n, i + window_size)
        h_smooth[i] = np.mean(h[lo:hi])
    
    h = np.clip(h_smooth, 0, 1)
    h = project_to_feasible(h, n)
    return h

def l_bfgs_refinement(h0, n, max_iter=2000, time_budget=200):
    """Refine using L-BFGS-B with log-sum-exp approximation."""
    dx = 2.0 / n
    beta = 5.0  # starting temperature

    def lse_objective(h):
        corr_values = np.zeros(n)
        for k in range(n):
            # compute correlation for shift k
            corr = np.sum(h * np.roll(1 - h, k))
            corr_values[k] = corr * dx
        return np.log(np.sum(np.exp(beta * corr_values))) / beta

    def grad_lse_objective(h):
        corr_values = np.zeros(n)
        for k in range(n):
            corr = np.sum(h * np.roll(1 - h, k))
            corr_values[k] = corr * dx
            
        # Softmax weights
        w = np.exp(beta * corr_values) / np.sum(np.exp(beta * corr_values))
        
        # Gradient terms
        grad = np.zeros(n)
        for k in range(n):
            shift = np.roll(1 - h, k)
            grad += w[k] * (shift - np.roll(h, k))
        return grad * dx
        
    def project_h(h):
        # Ensure feasible projection
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - n/2) < 1e-10:
            return h
        delta = (n/2 - current_sum) / n
        h = h + delta
        h = np.clip(h, 0, 1)
        h = h * (n/2) / np.sum(h)
        return h
    
    h = h0.copy()
    start_time = time.time()
    
    iteration = 0
    while iteration < max_iter and (time.time() - start_time) < time_budget - 5:
        # Inner loop: approximate optimization with L-BFGS-B or quasi-second-order method
        if iteration % 100 == 0:  # adjust temperature
            beta *= 2
        # Approximate gradient descent on total C5 via LSE surrogate
        try:
            gradient = grad_lse_objective(h)
            new_h = h - 0.001 * gradient
            h = project_h(new_h)
            iteration += 1
        except:
            break
            
    return h
