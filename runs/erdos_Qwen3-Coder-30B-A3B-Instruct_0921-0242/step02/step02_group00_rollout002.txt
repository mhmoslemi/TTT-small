def project_box_sum(y):
    """Project y onto the feasible set {x: sum(x)=n/2, 0<=x[i]<=1}."""
    n = len(y)
    y = np.array(y)
    lo, hi = 0.0, 1.0
    sum_y = np.sum(y)
    target = n / 2.0

    def S(lam):
        return np.sum(np.clip(y - lam, lo, hi))

    if sum_y <= target:
        return np.clip(y + (target - sum_y) / n, 0, 1)
    else:
        # Binary search for lambda
        a, b = np.min(y), np.max(y) + (target - sum_y)
        for _ in range(40):
            mid = (a + b) / 2.0
            s = S(mid)
            if s >= target:
                a = mid
            else:
                b = mid
        lam = (a + b) / 2.0
        return np.clip(y - lam, 0, 1)

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorrelation(h):
    """Compute autocorrelation vector of h with itself."""
    n = len(h)
    # (We only need specific parts but easier to compute all)
    auto = correlate(h, h, mode='full')
    return auto

def compute_min_autocorr(h):
    """Returns the minimum autocorrelation value in the full correlation."""
    n = len(h)
    auto = correlate(h, h, mode='full')
    min_auto = np.min(auto)
    return min_auto

def compute_worst_shifts(C):
    """Compute set of indices where C is within tolerance of max C."""
    epsilon = 1e-6
    C5_max = np.max(C)
    return np.where(np.abs(C - C5_max) < epsilon)[0]

def compute_C_from_gamma(gamma, n):
    """Compute C vector from gamma using C = 1 - (2/n) * gamma."""
    return 1.0 - (2.0 / n) * gamma

def compute_score_vector(h, S_w):
    """Compute score vector to guide move selection."""
    n = len(h)
    score = np.zeros(n)
    # For each index p
    for i in range(n):
        for s in S_w:
            idx = s + n - 1  # corresponding shift index
            if 0 <= i + s < n:
                score[i] += h[i + s]
            if 0 <= i - s < n and s != 0:
                score[i] += h[i - s]
    return score

def update_gamma_incremental(gamma, h, i, j, delta):
    """Update gamma incrementally for a transfer of delta units from i to j."""
    n = len(h)
    gamma_new = gamma.copy()
    hat_h = h.copy()
    hat_h[i] = 0.0  # Temporarily remove delta from i

    for s in range(-n+1, n):
        idx = s + n - 1
        # Removal from i:
        if 0 <= i + s < n:
            gamma_new[idx] -= delta * hat_h[i + s]
        if 0 <= i - s < n and s != 0:
            gamma_new[idx] -= delta * hat_h[i - s]

        # Addition to j:
        if 0 <= j + s < n:
            gamma_new[idx] += delta * h[j + s]
        if 0 <= j - s < n and s != 0:
            gamma_new[idx] += delta * h[j - s]

    # Update self term:
    gamma_new[n - 1] += -2 * h[i] * delta + delta * delta + 2 * h[j] * delta + delta * delta
    return gamma_new

def generate_move_simple(h, S_w, score, eta=1e-8, max_delta=0.2):
    """Generate a move using simple method."""
    n = len(h)
    # Score-based selection
    high_score_indices = np.argsort(score)[-max(1, n // 10):]
    low_score_indices = np.argsort(score)[:max(1, n // 10)]
    
    if len(high_score_indices) == 0:
        high_score_indices = [0]
    if len(low_score_indices) == 0:
        low_score_indices = [0]
    
    # Pick source
    i = np.random.choice(high_score_indices)
    if h[i] <= eta:
        return None, None, None

    # Pick target
    j = np.random.choice(low_score_indices)
    if h[j] >= 1 - eta:
        return None, None, None

    delta = min(max_delta, h[i], 1 - h[j])
    # Deliberately allow larger steps occasionally
    if np.random.rand() < 0.05:
        delta = np.random.rand() * max_delta
    delta = min(delta, h[i], 1 - h[j])

    return i, j, delta
