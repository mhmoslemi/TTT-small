import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.linalg import toeplitz
import cvxpy as cp

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2 with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Compute exact C5 using linear correlation."""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation using FFT for efficiency
    h_fft = fft(h, 2 * n)
    ih_fft = fft(1 - h, 2 * n)
    corr = ifft(h_fft * np.conj(ih_fft))
    # Take the real part and get max of non-zero shifts
    corr_vals = corr.real[1:n]  # exclude index 0
    max_corr = np.max(corr_vals) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 256  # even grid size
    dx = 2.0 / n
    t_low, t_high = 0.0, 1.0
    tolerance = 2e-4
    max_sdp_iters = 15
    time_for_sdp = 0.35 * budget_s
    last_improvement_time = time.time()
    
    # SDP bisection search
    t_best = t_low
    r_best = None
    
    for _ in range(max_sdp_iters):
        if time.time() - start_time > time_for_sdp:
            break
            
        t_mid = (t_low + t_high) / 2
        
        # Create SDP problem
        r = cp.Variable(n)
        constraints = [
            r[0] == 1,
            r[1:] >= t_mid,
            r[1:] <= 1,
        ]
        # Create Toeplitz matrix
        T = toeplitz(r)
        constraints.append(T >> 0)  # PSD constraint
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.SCS, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            t_best = t_mid
            r_best = r.value.copy()
            t_low = t_mid
            last_improvement_time = time.time()
        else:
            t_high = t_mid
            
        if t_high - t_low < tolerance:
            break
    
    # Convert r_best to step function via spectral factorisation
    if r_best is None:
        r_best = np.ones(n)
        r_best[0] = 1.0
    
    # Compute power spectrum
    S = np.fft.fft(r_best)
    S = np.maximum(S, 0)
    H = np.sqrt(S)
    
    # Inverse transform to get real sequence
    g = np.real(np.fft.ifft(H))
    
    # Project to feasible set
    h = project_to_feasible(g, n)
    
    # Deterministic worst-shift refinement
    max_iter = 250
    alpha_0 = 0.5 * dx
    for t in range(max_iter):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Compute all overlaps
        h_fft = fft(h, 2 * n)
        ih_fft = fft(1 - h, 2 * n)
        corr = ifft(h_fft * np.conj(ih_fft))
        corr_vals = corr.real[1:n]  # exclude index 0
        worst_shift = np.argmax(corr_vals)
        Ck_worst = corr_vals[worst_shift] * dx
        
        # Compute gradient for worst shift k*
        grad = np.zeros(n)
        for i in range(n):
            left_idx = (i + worst_shift) % n
            right_idx = (i - worst_shift) % n
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Update using projected gradient
        alpha = alpha_0 / np.sqrt(t + 1)
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n)
        
        # Evaluate progress
        C_k_new = evaluate_c5_exact(h_new)
        if C_k_new < Ck_worst - 1e-6:
            h = h_new
        else:
            break  # No improvement
    
    # Optional stochastic polishing
    max_polish_iter = 5000
    for i in range(max_polish_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Randomly choose two indices
        idx_i = np.random.randint(0, n)
        idx_j = np.random.randint(0, n)
        
        if h[idx_i] <= 0 or h[idx_j] >= 1:
            continue
            
        delta = min(0.01, h[idx_i], 1 - h[idx_j])
        if delta <= 0:
            continue
            
        h_test = h.copy()
        h_test[idx_i] -= delta
        h_test[idx_j] += delta
        h_test = project_to_feasible(h_test, n)
        
        c5_test = evaluate_c5_exact(h_test)
        c5_old = evaluate_c5_exact(h)
        
        if c5_test < c5_old:
            h = h_test
    
    # Optional upscaling if time remains
    if time.time() - start_time < budget_s * 0.99:
        # Interpolate and re-run short descent
        old_h = h.copy()
        n_up = n * 2
        dx_up = 2.0 / n_up
        h_up = np.interp(np.linspace(0, 2, n_up), np.linspace(0, 2, n), h)
        h_up = h_up * (n_up / 2) / np.sum(h_up)
        h_up = project_to_feasible(h_up, n_up)
        
        # Finer descent
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
                
            corr = ifft(fft(h_up, 2 * n_up) * np.conj(fft(1 - h_up, 2 * n_up)))
            corr_vals = corr.real[1:n_up]
            worst_shift = np.argmax(corr_vals)
            
            grad = np.zeros(n_up)
            for i in range(n_up):
                left_idx = (i + worst_shift) % n_up
                right_idx = (i - worst_shift) % n_up
                grad[i] = dx_up * ((1 - h_up[left_idx]) - h_up[right_idx])
                
            h_up = h_up - 0.5 * dx_up * grad
            h_up = project_to_feasible(h_up, n_up)
        
        h = h_up
        n = n_up
    
    c5_final = evaluate_c5_exact(h)
    
    return (h, c5_final, n)
