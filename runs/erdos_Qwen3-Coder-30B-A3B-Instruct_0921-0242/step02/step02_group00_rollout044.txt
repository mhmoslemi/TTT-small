import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_box_plus_sum(h, n):
    """Projects h onto the set {0 ≤ h_i ≤ 1, Σ h_i = n/2} using water filling."""
    # Sort in descending order for water filling
    h_sorted = np.sort(h)[::-1]
    
    # Find the threshold lambda
    cumsum = np.cumsum(h_sorted)
    n_vals = np.arange(1, n+1)
    lambda_vals = (cumsum - n/2) / n_vals
    
    # Get the largest lambda such that all elements above are 1, and below are 0
    # or at the boundary
    idx = np.argmax(h_sorted >= 1) if h_sorted[-1] >= 1 else n-1
    while idx >= 0 and h_sorted[idx] <= 0:
        idx -= 1
    
    # Find lambda where the projection is valid
    # This requires finding the largest j such that (cumsum[j] - n/2) / (j+1) <= h_sorted[j]
    # But for simplicity, we check a few cases:
    # Use binary search or check boundary conditions:
    lambda_star = -1e10
    for i in range(n):
        if h_sorted[i] > 0:
            l = (cumsum[i] - n/2) / (i+1)
            if l > lambda_star:
                lambda_star = l
    
    # Now construct the projected vector
    proj = np.maximum(np.minimum(h - lambda_star, 1.0), 0.0)
    
    # Adjust slightly to ensure exact sum
    diff = n/2 - np.sum(proj)
    idx = np.argsort(proj)
    for i in range(int(np.abs(diff) * 1000)):
        if diff > 0 and proj[idx[i]] < 1:
            proj[idx[i]] += 1e-10
        elif diff < 0 and proj[idx[i]] > 0:
            proj[idx[i]] -= 1e-10
    # Small adjustment for sum
    if abs(diff) > 1e-10:
        # Add diff in the largest terms
        sort_idx = np.argsort(proj)[::-1]
        for i in range(min(int(np.abs(diff)*n), len(sort_idx))):
            if diff > 0 and proj[sort_idx[i]] < 1:
                proj[sort_idx[i]] += min(1 - proj[sort_idx[i]], diff)
                diff -= min(1 - proj[sort_idx[i]], diff)
            if diff < 0 and proj[sort_idx[i]] > 0:
                proj[sort_idx[i]] -= min(proj[sort_idx[i]], -diff)
                diff += min(proj[sort_idx[i]], -diff)
    
    return proj

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorrelation(h):
    """Compute the aperiodic autocorrelation using FFT."""
    n = len(h)
    n_fft = 2 ** int(np.ceil(np.log2(2 * n)))
    h_padded = np.pad(h, (0, n_fft - n), 'constant')
    h_fft = fft(h_padded)
    gamma_h = ifft(h_fft * np.conj(h_fft)).real[:2*n - 1]
    return gamma_h

def solve_saddle_point_admm(n_points, max_time_s=500.0):
    """Solve the convex-concave saddle point problem using ADMM."""
    start_time = time.time()
    
    # Initial guess projected onto box + sum constraint
    h = initial_h_values
    if len(h) < n_points:
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_orig, h)
    else:
        h_interp = h[:n_points]
    h = h_interp * (n_points / 2) / np.sum(h_interp)
    h = project_to_box_plus_sum(h, n_points)
    
    # Initialize variables
    z = h.copy()
    u = np.zeros_like(h)
    t = evaluate_c5(z)
    
    # Precompute FFT of all ones vector
    ones = np.ones(n_points)
    n_fft = 2 ** int(np.ceil(np.log2(2 * n_points)))
    ones_pad = np.pad(ones, (0, n_fft - n_points), 'constant')
    ones_fft = fft(ones_pad)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 1000
    tol = 1e-7
    convergence_count = 0
    prev_t = t + 1.0
    
    best_h = z.copy()
    best_t = t
    
    for k in range(max_iter):
        if time.time() - start_time > max_time_s:
            break
            
        # h-update (solve: min_h (ρ/2)||h - (z - u)||^2 s.t. 1 - (2/n)(h * h)_s <= t)
        # In frequency domain we have h = (z - u) / (1 + μ * 1s)
        # We want to solve for μ such that max_s C_s <= t
        # Do a bisection on mu in [0, 1e10]
        mu = 0.0
        mu_low = 0.0
        mu_high = 1e10
        mu_best = 0.0
        tries = 0
        while mu_low < mu_high and tries < 30:
            mu = (mu_low + mu_high) / 2.0
            z_u = z - u
            z_u_fft = fft(np.pad(z_u, (0, n_fft - n_points), 'constant'))
            h_fft = z_u_fft / (1.0 + mu * ones_fft)
            h = ifft(h_fft).real[:n_points]
            
            # compute γ = h * h
            gamma = compute_autocorrelation(h)
            # compute C_s = 1 - (2/n)·γ_s
            C = 1.0 - (2.0 / n_points) * gamma
            
            # check if max is now <= t
            max_new = np.max(C)
            if max_new <= t + 1e-12:
                mu_best = mu
                mu_high = mu
            else:
                mu_low = mu
            tries += 1
            
        mu = mu_best
        
        # update h with the found mu
        z_u = z - u
        z_u_fft = fft(np.pad(z_u, (0, n_fft - n_points), 'constant'))
        h_fft = z_u_fft / (1.0 + mu * ones_fft)
        h = ifft(h_fft).real[:n_points]
        
        # update z by projection
        z = project_to_box_plus_sum(h + u, n_points)
        
        # update u
        u = u + (h - z)
        
        # update t (the max of C)
        gamma = compute_autocorrelation(h)
        C = 1.0 - (2.0 / n_points) * gamma
        t = np.max(C)
        
        # update best ever
        if t < best_t:
            best_t = t
            best_h = z.copy()
        
        # check convergence
        if abs(t - prev_t) / (abs(t) + 1e-10) < tol:
            convergence_count += 1
        else:
            convergence_count = 0
        prev_t = t
        
        if convergence_count >= 5:
            break
    
    return best_h, best_t

def nesterov_refinement(h, t_initial, max_iter=200):
    """Use Nesterov accelerated gradient refinement with smooth max surrogate."""
    n = len(h)
    t_current = t_initial
    h_current = h.copy()
    
    # Compute gamma now
    gamma = compute_autocorrelation(h_current)
    C = 1.0 - (2.0 / n) * gamma  # Specific overlap values
    
    # Initialize parameters
    beta = 5.0
    step_size = 1.0
    h_prev = h_current.copy()
    
    for k in range(max_iter):
        # Compute w_s = exp(β * C_s) / Σ_t exp(β * C_t)
        C_centered = C - np.max(C)  # For numerical stability
        exp_beta_C = np.exp(beta * C_centered)
        w = exp_beta_C / np.sum(exp_beta_C)
        
        # form ∇F_β = -(2/n) * (Σ_s w_s · (shift_+s(h) + shift_-s(h)))
        grad = np.zeros_like(h)
        for s in range(n):
            if s == 0:
                grad += w[s] * 2 * h
            else:
                # shift forward s + backward s
                h_shift_f = np.roll(h, -s)
                h_shift_b = np.roll(h, s)
                grad += w[s] * (h_shift_f + h_shift_b)
        
        grad *= - (2.0 / n)
        
        # Nesterov step
        y = h_current + ((k - 1) / (k + 2)) * (h_current - h_prev)
        y = y - step_size * grad
        
        # project onto box + sum constraint
        h_new = project_to_box_plus_sum(y, n)
        
        # Line search for step size (simplified inline)
        while True:
            gamma_new = compute_autocorrelation(h_new)
            C_new = 1.0 - (2.0 / n) * gamma_new
            t_new = np.max(C_new)
            if t_new < t_current + 1e-5:
                h_current = h_new
                t_current = t_new
                break
            step_size *= 0.5
            y = h_current + ((k - 1) / (k + 2)) * (h_current - h_prev)
            y = y - step_size * grad
            h_new = project_to_box_plus_sum(y, n)
        
        h_prev = h_current.copy()
        
        if k % 10 == 0:
            # Early termination if good improvement
            gamma = compute_autocorrelation(h_current)
            C = 1.0 - (2.0 / n) * gamma
            t_current = np.max(C)
            if time.time() > start_time + 590:
                break
        
        # Increase β gradually
        if k > 0 and k % 20 == 0:
            beta = min(beta * 1.3, 200.0)
    
    return h_current, t_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    start_time = time.time()
    
    # Get the problem size
    n_points = 256
    if budget_s > 300:
        n_points = 300
    
    # Solve with ADMM
    h_best, t_best = solve_saddle_point_admm(n_points, max_time_s=500.0)
    
    # Refine via Nesterov
    h_final, t_final = nesterov_refinement(h_best, t_best, max_iter=200)
    
    # potentially extend if needed
    if len(h_final) < n_points:
        x_orig = np.linspace(0, 2, len(h_final))
        x_new = np.linspace(0, 2, n_points)
        h_final = np.interp(x_new, x_orig, h_final)
    
    return (h_final, t_final, n_points)
