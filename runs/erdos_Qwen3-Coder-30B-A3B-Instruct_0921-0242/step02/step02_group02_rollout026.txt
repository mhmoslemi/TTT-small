import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_feasible(h, n):
    """Project h to feasible set: sum(h) = n/2, h in [0,1].  O(n) exact projection."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.abs(current_sum - n/2) < 1e-12:
        return h
    
    # Water-filling approach
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    # Find where to cut the sorted vector
    for i in range(n):
        if i + 1 < n:
            threshold = (cumsum[i] - n/2) / (i + 1)
        else:
            threshold = (cumsum[i] - n/2) / (i + 1)
        
        if threshold < sorted_h[i]:
            break
    else:
        # Shouldn't reach here in normal cases, but handle edge case
        threshold = (cumsum[-1] - n/2) / n

    h = np.clip(h - threshold, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Compute max_k integral h(x)(1 - h(x+k)) dx = max_k correlate(h, 1-h)[k] * dx"""
    n = len(h)
    dx = 2.0 / n
    # Full correlation via FFT
    corr = ifft(fft(h) * fft(1-h).conj()).real
    max_corr = np.max(corr) * dx
    return max_corr

def gradient_at_worst(h):
    """Return gradient of C5 w.r.t. h, w.r.t. its worst correlation."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation
    corr = ifft(fft(h) * fft(1-h).conj()).real
    k_star = np.argmax(corr)
    # Mapping from full correlate index to shifted index
    if k_star >= n:
        k = k_star - n
    else:
        k = k_star - n + 1
    
    grad = np.zeros_like(h)
    for i in range(n):
        i_plus_k = (i + k) % n
        i_minus_k = (i - k) % n
        grad[i] = dx * (h[i_plus_k] - h[i_minus_k])
    
    return grad

def line_search(h, grad, eta0=0.5):
    """Backtracking line search to find step size that improves C5"""
    eta = eta0
    dx = 2.0 / len(h)
    max_iter = 20
    for _ in range(max_iter):
        h_new = h - eta * grad
        h_new = project_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        if c5_new < c5_old:
            return h_new, c5_new
        eta *= 0.5
    return h, evaluate_c5(h)

def mutate_h(h, mode='gaussian', sigma=0.2, delta=0.01):
    """In-place mutation of h."""
    h_new = h.copy()
    n = len(h)
    if mode == 'gaussian':
        # perturb random subset
        indices = np.random.choice(n, size=np.random.randint(5, 10), replace=False)
        noise = np.random.normal(scale=sigma, size=len(indices))
        h_new[indices] += noise
        h_new = project_feasible(h_new, n)
    elif mode == 'swap_mass':
        i = np.random.randint(n)
        j = np.random.randint(n)
        while h_new[i] <= delta or h_new[j] >= 1 - delta:
            i = np.random.randint(n)
            j = np.random.randint(n)
        move = min(delta, h_new[i], 1 - h_new[j])
        h_new[i] -= move
        h_new[j] += move
        h_new = project_feasible(h_new, n)
    elif mode == 'dct_tweak':
        # Discrete cosine transform
        h_dct = np.fft.dct(h_new)
        # Modify small frequency
        k_freq = np.random.choice(min(5, len(h_dct)-1) + 1, 1)[0] + 1
        h_dct[k_freq] += np.random.normal(0, 0.05)
        h_new = np.fft.idct(h_dct)
        h_new = project_feasible(h_new, n)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Progressively refine n
    n_list = [250, 500]
    best_h = None
    best_c5 = float('inf')
    
    for n in n_list:
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # 1. Generate random pool
        M = 2000
        pool_h = []
        pool_c5 = []
        for _ in range(M):
            u = np.random.rand(n)
            h = project_feasible(u, n)
            c5 = evaluate_c5(h)
            pool_h.append(h)
            pool_c5.append(c5)
            
        pool_c5 = np.array(pool_c5)
        indices = np.argsort(pool_c5)[:30]
        pool_h = [pool_h[i] for i in indices]
        pool_c5 = pool_c5[indices]
        
        # 2. Mutation loop for 0.25 * budget
        t_remaining = budget_s * 0.25
        t_start_mutation = time.time()
        sigma = 0.2
        
        while time.time() - t_start_mutation < t_remaining:
            if time.time() - start_time > budget_s - 0.5:
                break
            # Select parent
            parent_idx = np.random.randint(len(pool_h))
            parent_h = pool_h[parent_idx]
            # Mutate
            mode = np.random.choice(['gaussian', 'swap_mass', 'dct_tweak'])
            child_h = mutate_h(parent_h, mode=mode, sigma=sigma)
            child_c5 = evaluate_c5(child_h)
            
            # Update pool if better
            if child_c5 < pool_c5[-1]:
                idx_replace = np.argmin(pool_c5)
                pool_h[idx_replace] = child_h
                pool_c5[idx_replace] = child_c5
            
            # Slow down sigma over time
            sigma *= 0.995
        
        # 3. Local refinement (targeted gradient descent on top 5)
        t_remaining = budget_s * 0.45
        t_start_refine = time.time()
        k_refine = min(5, len(pool_h))
        
        for i in range(k_refine):
            if time.time() - start_time > budget_s - 0.5:
                break
            h0 = pool_h[i].copy()
            h_now = h0.copy()
            c5_now = pool_c5[i]
            
            # Max 2000 iterations or time
            for _ in range(2000):
                if time.time() - t_start_refine > t_remaining:
                    break
                grad = gradient_at_worst(h_now)
                # Line search for best step
                h_new, c5_new = line_search(h_now, grad)
                if c5_new < c5_now:
                    h_now = h_new
                    c5_now = c5_new
                else:
                    break  # No improvement - stop
        
            # Update if This is the best
            if c5_now < best_c5:
                best_h = h_now.copy()
                best_c5 = c5_now

    # 4. Final polishing with pairwise swaps
    if best_h is not None:
        h_refined = best_h.copy()
        c5_refined = evaluate_c5(h_refined)
        
        t_remaining = budget_s * 0.10
        t_start_swap = time.time()
        
        # 10000 swaps
        delta = 0.001
        for _ in range(10000):
            if time.time() - t_start_swap > t_remaining:
                break
            i = np.random.randint(len(h_refined))
            j = np.random.randint(len(h_refined))
            
            # Ensure will be in interior of [0,1]
            if h_refined[i] < delta or h_refined[j] > 1-delta:
                continue
                
            old_c5 = evaluate_c5(h_refined)
            # Attempt small swap
            swap_amount = np.random.uniform(delta/2, delta*2)
            h_swap = h_refined.copy()
            h_swap[i] -= swap_amount
            h_swap[j] += swap_amount
            h_swap = project_feasible(h_swap, len(h_refined))
            new_c5 = evaluate_c5(h_swap)
            
            if new_c5 < old_c5:
                h_refined = h_swap
                c5_refined = new_c5
        
        best_h = h_refined
        best_c5 = c5_refined
    
    return (best_h, best_c5, len(best_h))
