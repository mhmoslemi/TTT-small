import numpy as np
import cvxpy as cp
import time
from scipy.signal import fftconvolve
from scipy.ndimage import gaussian_filter1d

def aperiodic_autocorr(h):
    """Compute aperiodic autocorrelation via FFT."""
    n = len(h)
    # zero-pad to avoid wraparound
    h_pad = np.pad(h, (0, n-1), mode='constant')
    h_rev = h[::-1]
    h_rev_pad = np.pad(h_rev, (0, n-1), mode='constant')
    corr = fftconvolve(h_pad, h_rev_pad, mode='valid')
    return corr[:n]

def softmax(x, beta=1.0):
    """Compute softmax with temperature beta, numerically stable."""
    x = np.asarray(x)
    x_beta = beta * x
    x_max = np.max(x_beta)
    exp_x = np.exp(x_beta - x_max)
    return exp_x / np.sum(exp_x)

def project_box_sum(h, total):
    """Project h onto {h : 0 ≤ h_i ≤ 1, sum h_i = total}."""
    # Clip to [0,1]
    h_clipped = np.clip(h, 0, 1)
    s = np.sum(h_clipped)
    if abs(s - total) < 1e-10:
        return h_clipped
    # Use water-filling to adjust
    delta = total - s
    if abs(delta) < 1e-12:
        return h_clipped

    # Find indices where we can increase/decrease
    h_sorted = np.sort(h_clipped)[::-1]
    # Binary search for the boundary
    low, high = 0.0, 1.0
    for _ in range(100):
        mid = (low + high) / 2.0
        # How much would we exceed the total if adjustments were at level mid?
        # h_adjusted = h_clipped + (mid - h_clipped) * (h_clipped <= mid) * (h_clipped >= 0)
        # But more efficiently:
        # if mid >= h_clipped[i]: add mid - h_clipped[i]
        # if mid <= 0, don't add anything (since clipped to 0)
        total_change = 0.0
        for i in range(len(h_clipped)):
            val = h_clipped[i]
            if val < mid:
                total_change += (mid - val)
        if total_change > delta:
            high = mid
        else:
            low = mid
    adjustment_level = (low + high) / 2.0

    h_proj = np.clip(h_clipped + adjustment_level - h_clipped, 0, 1)
    h_proj += (total - np.sum(h_proj)) * (h_clipped > 0) / np.sum(h_clipped > 0)
    return h_proj

def evaluate_c5(h):
    """Evaluate the C5 constant."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation of h with (1 - h)
    corr = aperiodic_autocorr(h)
    # Scale by dx and find maximum
    c5 = max(1 - 2 * corr[i] / n for i in range(len(corr)))
    return c5

def score_guided_swap(h, ε=1e-5):
    """Perform a score-guided swap."""
    n = len(h)
    δ = 0.01
    # Find all max shifts (within ε)
    corr = aperiodic_autocorr(h)
    c5 = np.array([1 - 2 * corr[i] / n for i in range(len(corr))])
    max_c5 = np.max(c5)
    worst_shifts = np.where(np.abs(c5 - max_c5) <= ε)[0]
    # since we only care about the relative difference
    # and the hypothesis was about shifts 0 to n-1
    # not indexed by full correlation indices
    # compute score per index
    score = np.zeros(n)
    for s in worst_shifts:
        if s >= n:
            continue
        for i in range(n):
            score[i] += h[(i + s) % n]
            score[i] += h[(i - s) % n]
    # pick index with highest score to reduce
    # and with lowest score to increase
    idx_src = np.argmax(score)
    idx_dst = np.argmin(score)
    if h[idx_src] <= 0 or h[idx_dst] >= 1:
        return h
    amount = min(δ, h[idx_src], 1 - h[idx_dst])
    h[idx_src] -= amount
    h[idx_dst] += amount
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 300
    dx = 2.0 / n_points

    # --- Phase I: solve binary MILP
    n1 = 120  # Binary grid size
    half = n1 // 2
    max_shift = n1 - 1

    # Build MILP model
    b = cp.Variable(n1, boolean=True)
    a = cp.Variable((n1, max_shift + 1), boolean=True)
    gamma = cp.Variable(max_shift + 1)
    L = cp.Variable()

    # Mass constraint
    constraints = [cp.sum(b) == half]

    # McCormick constraints for a[i,s] = b[i] * b[i+s]
    for i in range(n1):
        for s in range(max_shift + 1):
            if i + s < n1:
                constraints.extend([
                    a[i,s] <= b[i],
                    a[i,s] <= b[i + s],
                    a[i,s] >= b[i] + b[i + s] - 1
                ])

    # Define gamma[s] = sum_{i=0}^{n1-1-s} a[i,s]
    for s in range(max_shift + 1):
        if s == 0:
            # gamm0 = sum b[i] = half
            constraints.append(gamma[s] == half)
        else:
            # TODO replace with proper indexing
            terms = [a[i,s] for i in range(n1 - s)]
            constraints.append(gamma[s] == cp.sum(terms))

    # Uniform lower bound: all gamma_s >= L
    for s in range(max_shift + 1):
        constraints.append(gamma[s] >= L)

    # Objective: max L
    obj = cp.Maximize(L)

    # Create problem and solve
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CBC, time_limit=0.2 * budget_s, relative_gap=1e-3, verbose=False)

    if prob.status != cp.OPTIMAL and prob.status != cp.SOLVED:
        # fallback to use best known binary pattern if solver failed
        b_star = np.zeros(n1)
        b_star[:half] = 1
        np.random.shuffle(b_star)
    else:
        b_star = np.round(b.value).astype(int)

    # --- Phase II: Upscaling and smoothing
    h0 = np.interp(np.linspace(0, 2, n_points, endpoint=False),
                   np.linspace(0, 2, n1, endpoint=False),
                   b_star)
    h0 = project_box_sum(h0, total=n_points / 2)
    h0 = gaussian_filter1d(h0, sigma=1.0)
    h0 = project_box_sum(h0, total=n_points / 2)

    # --- Phase III: Smoothed projected gradient descent
    h = h0.copy()
    β = 2.0
    η = 1.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    iteration = 0
    t0 = time.time()
    while (time.time() - start_time < 0.85 * budget_s):
        iteration += 1
        # compute all gamma_s
        gamma = aperiodic_autocorr(h)
        C = np.array([1 - 2 * gamma[i] / n_points for i in range(len(gamma))])
        # soft max weights
        w = softmax(β * C)
        # compute gradient:
        # ∇ h = - (2/n_points) * ( w * h  + reverse(w) * h )
        h_fft = np.fft.fft(h)
        w_fft = np.fft.fft(w)
        w_rev_fft = np.fft.fft(np.flip(w))
        grad_fft = - (2.0/n_points) * (w_fft * h_fft + w_rev_fft * h_fft)
        grad = np.real(np.fft.ifft(grad_fft))
        # backtracking line search:
        α = η
        h_trial = project_box_sum(h - α * grad, total=n_points / 2)
        c5_trial = evaluate_c5(h_trial)
        while c5_trial > best_c5 and α > 1e-8:
            α *= 0.5
            h_trial = project_box_sum(h - α * grad, total=n_points / 2)
            c5_trial = evaluate_c5(h_trial)

        h = h_trial
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # schedule
        if iteration % 150 == 0:
            β = min(β * 1.5, 200)
        if iteration % 200 == 0:
            h = score_guided_swap(h)
        if iteration % 1000 == 0:
            if time.time() - start_time > 0.85 * budget_s - 1:
                break

    # --- Phase IV: Final polishing with subgradient
    while (time.time() - start_time < budget_s - 0.1 * budget_s):
        corr = aperiodic_autocorr(h)
        c5_vals = np.array([1 - 2 * corr[i] / n_points for i in range(len(corr))])
        max_c5 = np.max(c5_vals)
        worst_shifts = np.where(np.abs(c5_vals - max_c5) <= 1e-5)[0]
        g = np.zeros(n_points)
        for s in worst_shifts:
            shift_left = np.roll(h, s)
            shift_right = np.roll(h, -s)
            g += (shift_left + shift_right)
        g = - (2 / n_points) * g / len(worst_shifts)
        η = 1.0 / np.sqrt(iteration + 1)
        h_new = project_box_sum(h - η * g, total=n_points / 2)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        iteration += 1

    return (best_h, best_c5, n_points)
