import numpy as np
import time
import math
from scipy.signal import correlate

def project_box_sum(y, target_sum):
    """Project y onto the set {x: 0 <= x <= 1, sum(x) = target_sum} using water filling."""
    y = np.asarray(y)
    lb = 0.0
    ub = 1.0
    n = len(y)
    
    def S(lam):
        return np.sum(np.clip(y - lam, lb, ub))
    
    # Binary search for lambda
    left, right = np.min(y) - 1.0, np.max(y) + 1.0
    eps = 1e-12
    while right - left > eps:
        mid = (left + right) / 2
        s = S(mid)
        if s >= target_sum:
            left = mid
        else:
            right = mid
    
    lam = (left + right) / 2
    x = np.clip(y - lam, lb, ub)
    return x

def evaluate_c5(h):
    """Compute C5 for discrete h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def evaluate_autocorr(h):
    """Compute full autocorrelation of h."""
    return correlate(h, h, mode='full')

def update_autocorr_incremental(γ, h, i, j, Δ):
    """Update γ for a transfer of Δ from index i to j."""
    # γ is shifted version of full correlation
    n = len(h)
    Δγ = np.zeros_like(γ)
    
    # Temporary: zero out source
    # h_prime = h.copy()
    # h_prime[i] -= Δ
    # Handle all affected γ_s entries where s = -(n-1) to (n-1)
    # The index of shift s in γ is s + n - 1
    # i = source, j = target
    # γ_s = Σ_p h[p] * h[p+s]
    
    # Remove contribution from i
    for s in range(-n+1, n):
        idx = s + n - 1  # Index for shift s
        if 0 <= i + s < n and s != 0:
            Δγ[idx] -= Δ * h[i+s]
        if 0 <= i - s < n and s != 0:
            Δγ[idx] -= Δ * h[i-s]
            
    Δγ[n-1] += -2 * h[i] * Δ + Δ * Δ   # self-term (the contribution of h[i]*h[i])
    
    # Add contribution from j
    for s in range(-n+1, n):
        idx = s + n - 1
        if 0 <= j + s < n:
            Δγ[idx] += Δ * h[j+s]
        if 0 <= j - s < n and s != 0:
            Δγ[idx] += Δ * h[j-s]
    
    Δγ[n-1] += 2 * h[j] * Δ + Δ * Δ   # self-term (the contribution of h[j]*h[j])

    return γ + Δγ

def compute_worst_shifts(C, C5, ε=1e-5):
    """Return indices s such that C[s] within ε of C5."""
    return np.where(np.abs(C - C5) < ε)[0]

def compute_shift_scores(h, S_w, n):
    """Compute score for each index based on contribution to worst shifts."""
    scores = np.zeros(n)
    for s in S_w:
        # s corresponds to index in full array as s + n - 1
        # contributions from i to shift s:
        # h[i] * h[i + s] for i such that i+s is in valid range
        # and similarly for i - s
        if s == 0:
            # auto-correlation, no need
            continue
        elif s > 0:
            # s = shift, so h[i] contributes to shift s via h[i+s]
            # h[i+s] contributes to shift (-s) via h[i]
            valid_indices = np.arange(0, n - s)
            scores[valid_indices] += h[valid_indices + s]
            scores[valid_indices + s] += h[valid_indices]
        else:
            # s < 0, so shift s means h[i] comes from h[i - s]
            valid_indices = np.arange(-s, n)
            scores[valid_indices] += h[valid_indices + s]
            scores[valid_indices + s] += h[valid_indices]
    return scores

def generate_move(h, scores, S_w, η=1e-8, max_delta=0.2):
    """Generate a move: transfer Δ from i to j."""
    n = len(h)
    
    # Pick i with probability biased towards high scores (except saturated)
    candidates_i = np.where(h > η)[0]
    if len(candidates_i) == 0:
        return None
    i = np.random.choice(candidates_i)
    
    # Pick j with probability biased towards low scores (except saturated)
    candidates_j = np.where(h < 1 - η)[0]
    if len(candidates_j) == 0:
        return None
    j = np.random.choice(candidates_j)
    
    if i == j:
        return None
        
    # Δ is in [0, min(h[i], 1 - h[j])]
    Δ = min(h[i], 1 - h[j], max_delta)
    
    if Δ <= 0:
        return None
        
    return i, j, Δ
