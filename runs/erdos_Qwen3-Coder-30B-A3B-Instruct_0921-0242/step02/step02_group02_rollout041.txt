import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_to_feasible(h, n):
    """Project h to feasible set: 0 <= h[i] <= 1, sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    if abs(diff) < 1e-10:
        return h
    # Water-filling with binary search on lambda
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Compute maximum overlap for h."""
    n = len(h)
    dx = 2.0 / n
    H = fft(h, n)
    autocorr = np.real_if_close(ifft(H * np.conj(H), n).real)
    # c_k = dx * (n/2 - autocorr_k)
    # max c_k is the C5 bound
    c = dx * (n/2 - autocorr)
    return np.max(c)

def compute_gradient(h, k_star):
    """Compute subgradient of C5 at k_star."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    grad += dx * np.roll(h, k_star)
    grad += dx * np.roll(h, -k_star)
    return grad

def backtracking_line_search(h, grad, max_steps=10):
    """Backtracking line search for step size"""
    alpha = 1.0
    dx = 2.0 / len(h)
    old_c5 = evaluate_c5(h)
    for _ in range(max_steps):
        h_new = h + alpha * grad
        h_new = project_to_feasible(h_new, len(h))
        new_c5 = evaluate_c5(h_new)
        if new_c5 < old_c5:
            return h_new, alpha
        alpha /= 2.0
    return h, 0.0

def smooth_boundaries(h_bin, ramp_width=2):
    """Convert binary pattern to smooth step function with ramps"""
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    # Identify boundaries where 0->1 or 1->0
    boundaries = []
    for i in range(n):
        if h_bin[i] != h_bin[(i+1)%n]:
            boundaries.append(i)
    
    # Insert ramps for each boundary
    new_h = h.copy()
    for i in boundaries:
        # Insert ramp from h[i] to h[i+1]
        start_idx = (i + 1) % n
        for r in range(ramp_width):
            idx = (start_idx + r) % n
            new_h[idx] = (r + 1) / (ramp_width + 1)
            
    # Project to feasible set
    new_h = project_to_feasible(new_h, n)
    return new_h

def binary_anneal(h, T0=5*0.008, alpha=0.9997, time_limit=300):
    """Simulated annealing on binary patterns."""
    n = len(h)
    dx = 2.0 / n
    start_time = time.time()
    
    # Compute initial autocorrelations
    H = fft(h, n)
    autocorr = np.real_if_close(ifft(H * np.conj(H), n).real)
    c = dx * (n/2 - autocorr)
    best_c = np.max(c)
    best_h = h.copy()
    
    T = T0
    swap_count = 0
    stagnant = 0
    
    while time.time() - start_time < time_limit and T > dx * 1e-4:
        # Pick a random 1 and 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute delta autocorrelation for this swap
        delta = np.zeros(n)
        # Shifts affected: positions u where u == i or u == j or u+shift == i or u+shift == j
        # This is easier to do with explicit function
        delta[i] -= 1
        delta[j] += 1
        
        # Update all autocorrelations that depend on positions i and j
        # This can be done via the shift function
        if i != j:
            delta[i] += 1  # adding 1 @ i in its own autocorrelation
            delta[j] -= 1  # removing 1 @ j from its own autocorrelation
            # self shifts
            delta[0] += 2   # both i and j get +1 in autocorr 0, and we subtract 1 from each -- net +1
            delta[(i - j) % n] -= 1  # subtract 1 from a[i-j] (i shifts to j)
            delta[(j - i) % n] -= 1  # subtract 1 from a[j-i] (i shifts to j)
            # But we also need to subtract 1 from a[i] and a[j] when removing 1 from that position in autocorr
            # Actually, let's just directly compute what happens to all elements of autocorr
            # When h[i]=1, h[j]=0, we are changing h[i],h[j] -> 0, 1 respectively
            # For an index u, the contribution to autocorrelation with shift k is h[u] * h[u+k]
            # For k=0: h[i]h[i] + h[j] h[j] = 1*1 + 0*0 = 1 is replaced by 0*0 + 1*1 = 1
            # So there is no change!
            # But when k != 0, there is change. 
            # For shift k, u = i -> contributes h[i] * h[i+k] = 1 * h[i+k]
            #               u = j -> contributes h[j] * h[j+k] = 0 * h[j+k] = 0
            # So the change in autocorr_k is -h[i+k] + h[j+k]
            # We apply this change for all k (the indices in delta are positions, not shifts)
            # so the index k in autocorr_k is at index k in delta, so delta[k] = (h[j+k]-h[i+k])
            # Ah but we defined the sign at the bottom
            delta = np.zeros(n)
            for k in range(n):
                delta[k] = h[(j - k) % n] - h[(i - k) % n]
            
            delta[0] += 2  # follow the formula from above
            delta[(i - j) % n] -= 1
            delta[(j - i) % n] -= 1
            
        # Now update autocorr with delta
        new_autocorr = autocorr + delta
        new_c = dx * (n/2 - new_autocorr)
        new_bound = np.max(new_c)
        
        # Accept or reject
        if new_bound <= c.max() or np.random.rand() < np.exp(-(new_bound - c.max()) / T):
            h[i] = 0
            h[j] = 1
            autocorr = new_autocorr
            c = new_c
            if new_bound < best_c:
                best_c = new_bound
                best_h = h.copy()
                stagnant = 0
        else:
            stagnant += 1
            
        swap_count += 1
        if swap_count % 1000 == 0:
            T = alpha * T
        if stagnant > 200000:
            # Large jump
            n_ones = np.sum(h)
            # Reassign 5% of ones to random zeros
            indices = np.random.choice(n, int(0.05 * n), replace=False)
            for idx in indices:
                if h[idx] == 1:
                    h[idx] = 0
            # Find random zeros
            target_indices = np.where(h == 0)[0]
            for idx in indices:
                if h[idx] == 0 and len(target_indices) > 0:
                    h[idx] = 1
            stagnant = 0

    return best_h, best_c

def proj_grad_refine(h, time_limit=150):
    """Projected gradient refinement of the continuous function."""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    
    while time.time() - start_time < time_limit:
        # Find worst shift
        H = fft(h, n)
        autocorr = np.real_if_close(ifft(H * np.conj(H), n).real)
        c = dx * (n/2 - autocorr)
        k_star = np.argmax(c)
        
        # Gradient w.r.t. this shift
        grad = compute_gradient(h, k_star)
        
        # Line search
        h_new, alpha = backtracking_line_search(h, grad)
        
        if alpha > 0:
            h = h_new
        else:
            # Fall back to some small step
            h_new = h + 0.001 * grad
            h = project_to_feasible(h_new, n)
            
        # Check improvement
        c5 = evaluate_c5(h)
        # Optional: pairwise fractional swaps to escape plateaus
        
    return h, c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Initialize
    n0 = 256
    dx0 = 2.0 / n0
    h0 = np.random.rand(n0)
    h0 = project_to_feasible(h0, n0)
    
    # Phase 2: Binary simulated annealing
    h_bin, c5_bin = binary_anneal(h0, time_limit=0.55 * budget_s)
    
    # Phase 3: Boundary smoothing
    h_smooth = smooth_boundaries(h_bin, ramp_width=2)
    
    # Phase 4: Projected gradient refinement
    h_refined, c5_refined = proj_grad_refine(h_smooth, time_limit=0.25 * budget_s)
    
    # Phase 5: Upsample and polish
    n1 = 2 * n0
    dx1 = 2.0 / n1
    h_upsampled = np.interp(np.linspace(0, 2, n1), np.linspace(0, 2, n0), h_refined)
    h_upsampled = project_to_feasible(h_upsampled, n1)
    # Short gradient polish on the new grid
    old_c5_refined = c5_refined
    for _ in range(500):
        if time.time() - start_time > budget_s - 0.5:
            break
        k_star = np.argmax(np.real_if_close(ifft(fft(h_upsampled, n1) * np.conj(fft(h_upsampled, n1)), n1).real))
        # Avoid k_star out of bounds - will need periodic indexing
        grad = compute_gradient(h_upsampled, k_star)
        h_new, alpha = backtracking_line_search(h_upsampled, grad)
        if alpha > 0:
            h_upsampled = h_new
        else:
            h_upsampled = h_upsampled + 0.001 * grad
            h_upsampled = project_to_feasible(h_upsampled, n1)
        c5_new = evaluate_c5(h_upsampled)
        if c5_new < old_c5_refined:
            old_c5_refined = c5_new
    
    return (h_upsampled, old_c5_refined, n1)
