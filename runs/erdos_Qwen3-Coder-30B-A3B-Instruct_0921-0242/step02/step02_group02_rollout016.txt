import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_feasible(h, n):
    """Project h onto the feasible set: sum(h) = n/2, 0 <= h[i] <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    # Water-filling algorithm
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    # Find the point where we need to fill
    # sum(h_sorted[:i]) + (n-i) * x = n/2
    # So: x = (n/2 - sum(h[:i])) / (n-i)
    target = n/2
    i = 0
    while i < len(h_sorted) and (cumsum[i] + (n-i-1) * h_sorted[i]) < target:
        i += 1
    if i == len(h_sorted):
        x = (target - cumsum[i-1]) / (n-i)
        h = np.clip(h_sorted, 0, x)
    else:
        x = (target - cumsum[i-1]) / (n-i) if i > 0 else target / n
        if x > h_sorted[i]:
            h = np.clip(h_sorted, 0, x)
        else:
            h = np.clip(h_sorted, 0, h_sorted[i])
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Use FFT for faster correlation, gives full correlation in length 2*n-1
    N = len(h)
    v = fft(np.concatenate([h[:N], [0]*(N-1)]))
    w = fft(np.concatenate([1-h[:N], [0]*(N-1)]))
    corr = ifft(v * np.conj(w)).real[:2*N-1]
    max_corr = np.max(corr) * dx
    return max_corr

def gradient_at_worst(h):
    """Find the worst shift k* and compute gradient w.r.t. it."""
    N = len(h)
    dx = 2.0 / N
    # Full cross correlation
    corr = correlate(h, 1-h, mode='full')
    # Add small value to avoid numerical issues if all are equal
    k_star = np.argmax(corr) - (N-1)
    grad = np.zeros(N)
    # Implement the gradient of corr[k*] w.r.t. h[i]
    for i in range(N):
        # corr[k_star] = sum(h[j] * (1 - h[j + k_star]))
        # So for fixed k_star:
        # d corr[k_star] / d h[i] = (1 - h[i + k_star]) - (1 - h[i - k_star]) = h[i - k_star] - h[i + k_star]
        # But indices must be modulo N
        i_plus = (i + k_star) % N
        i_minus = (i - k_star) % N
        grad[i] = dx * (h[i_minus] - h[i_plus])
    return grad

def line_search(h, grad, eta0=0.5):
    """Line search to find the best step size."""
    dx = 2.0 / len(h)
    eta = eta0
    current_c5 = evaluate_c5(h)
    for _ in range(10):
        h_new = h - eta * grad
        h_new = project_feasible(h_new, len(h))
        new_c5 = evaluate_c5(h_new)
        if new_c5 < current_c5:
            return h_new, new_c5
        eta /= 2.0
    return h, current_c5

def mutate(v, mode='gauss', param=None):
    """Apply mutation to a vector."""
    n = len(v)
    param = param or {}
    if mode == 'gauss':
        sigma = param.get('sigma', 0.1)
        nchg = param.get('nchg', np.random.randint(5, 10))
        indices = np.random.choice(n, size=nchg, replace=False)
        v_new = v.copy()
        v_new[indices] += np.random.normal(0, sigma, nchg)
        v_new = project_feasible(v_new, n)
    elif mode == 'swap':
        delta = param.get('delta', 0.01)
        # Find valid indices (h[i] > delta, h[j] < 1-delta)
        valid_i = np.where(v > delta)[0]
        valid_j = np.where(v < 1 - delta)[0]
        if len(valid_i) > 0 and len(valid_j) > 0:
            i = np.random.choice(valid_i)
            j = np.random.choice(valid_j)
            transfer = min(delta, v[i], 1 - v[j])
            v_new = v.copy()
            v_new[i] -= transfer
            v_new[j] += transfer
            v_new = project_feasible(v_new, n)
        else:
            v_new = v.copy()
    elif mode == 'dct':
        # DCT mode tweak: modify some low frequency component
        v = project_feasible(v, n)
        dt = np.fft.rfft(v)
        dt = dt.copy()
        # Modify a few low frequency coefficients
        n_coeffs = min(6, len(dt))
        indices = np.random.choice(n_coeffs, size=2, replace=False)
        for i in indices:
            dt[i] += np.random.normal(0, 0.01)
        v_new = np.fft.irfft(dt)
        v_new = v_new[:n]
        v_new = project_feasible(v_new, n)
    else:
        v_new = v.copy()
    return v_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Guide discretization - start with small, refined later
    n_points = 250
    dx = 2.0 / n_points

    # Phase 1: Generate initial pool
    M = 2000
    pool = []
    for _ in range(M):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        pool.append(h)
    
    # Evaluate
    evals = []
    for i, h in enumerate(pool):
        c5 = evaluate_c5(h)
        evals.append((c5, i, h))
    evals.sort()
    top_k = 30
    pool = [evals[i][2] for i in range(top_k)]

    # Mutation loop
    t0 = time.time()
    max_mutation_iter = 100000
    mutation_count = 0

    while time.time() - t0 < 0.25 * budget_s and mutation_count < max_mutation_iter:
        if time.time() - start_time > 0.9 * budget_s:
            break
        mutation_count += 1
        # Select parent
        parent_idx = np.random.randint(0, len(pool))
        parent = pool[parent_idx]
        mode = np.random.choice(['gauss', 'swap', 'dct'])
        if mode == 'gauss':
            params = {'sigma': 0.2 * (1.0 - mutation_count / max_mutation_iter),
                      'nchg': np.random.randint(5, 10)}
        elif mode == 'dct':
            params = {}
        else:
            params = {'delta': 0.01}
        child = mutate(parent, mode=mode, param=params)
        child_c5 = evaluate_c5(child)

        # Check if to replace worst in pool
        worst_idx = np.argmax([evaluate_c5(h) for h in pool])
        if child_c5 < evaluate_c5(pool[worst_idx]):
            pool[worst_idx] = child

    # Phase 2: Local refinement of top candidates
    top_candidates = []
    for i in range(min(5, len(pool))):
        h = pool[i].copy()
        current_c5 = evaluate_c5(h)
        prev_c5 = current_c5 + 1e-6
        iter_count = 0
        while abs(current_c5 - prev_c5) > 1e-6 and iter_count < 2000 and time.time() - start_time < 0.8 * budget_s:
            prev_c5 = current_c5
            grad = gradient_at_worst(h)
            h_new, new_c5 = line_search(h, grad)
            h = h_new
            current_c5 = new_c5
            iter_count += 1
        top_candidates.append((current_c5, h))

    top_candidates.sort()
    final_h = top_candidates[0][1]
    final_c5 = top_candidates[0][0]

    # Phase 3: Final polishing
    for _ in range(10000):
        if time.time() - start_time > budget_s - 0.5:
            break
        i, j = np.random.choice(n_points, size=2, replace=False)
        if final_h[i] > 1e-4 and final_h[j] < 1 - 1e-4:
            delta = min(0.01, final_h[i], 1 - final_h[j])
            h_trial = final_h.copy()
            h_trial[i] -= delta
            h_trial[j] += delta
            h_trial = project_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            if trial_c5 < final_c5:
                final_h = h_trial
                final_c5 = trial_c5

    # Optionally upscale
    if time.time() - start_time < 0.9 * budget_s:
        n_points = 500
        dx = 2.0 / n_points
        # Interpolate
        h_interp = np.interp(np.linspace(0, 2, n_points),
                             np.linspace(0, 2, len(final_h)), final_h)
        final_h = project_feasible(h_interp, n_points)
        # Local refine
        current_c5 = evaluate_c5(final_h)
        prev_c5 = current_c5 + 1e-6
        iter_count = 0
        while abs(current_c5 - prev_c5) > 1e-6 and iter_count < 1000 and time.time() - start_time < budget_s - 0.5:
            prev_c5 = current_c5
            grad = gradient_at_worst(final_h)
            final_h, current_c5 = line_search(final_h, grad)
            iter_count += 1
        final_c5 = current_c5

    return (final_h.tolist(), final_c5, n_points)
