def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def solve_sdp_relaxation(n):
    """Solve SDP relaxation to get a lower bound and target autocorrelation."""
    dx = 2.0 / n
    # Assemble circulant matrices C_k
    C = []
    for k in range(n):
        C_k = np.zeros((n, n))
        for i in range(n):
            C_k[i, (i - k) % n] = 1
        C.append(C_k)

    # Variables
    h = Variable(n, nonneg=True)
    
    # SDP variables
    one = np.ones((n, 1))
    X = h @ one.T
    
    # Objective
    gamma = Variable()
    
    # Constraints
    constraints = [
        trace(C_k @ X) >= n/2 - gamma/dx for C_k in C
    ]
    constraints.append(sum_entries(h) == n/2)
    constraints.append(X >> 0)  # positive semidefinite
    
    # Solve
    prob = Problem(Minimize(gamma), constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=2000, eps=1e-4)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            return h_opt, gamma.value
    except Exception:
        pass
    return None, None

def solve_binary_mip(h_target, n, gamma_upper_bound):
    """Solve binary integer program with lazy cuts."""
    dx = 2.0 / n
    
    # Prepare for MX
    b = cp.Variable(n, boolean=True)
    gamma = cp.Variable()
    
    # Accumulator - sum over products
    total = 0
    product_constraints = []
    
    for k in range(n):
        # For shift k
        z = cp.Variable(n, boolean=True)
        scalings = b @ b[(np.arange(n) + k) % n]
        # z = b * b_shifted.  Use McCormick envelope
        # z_{i} <= b_i,  z_{i} <= b_{i-k},  z_i >= b_i + b_{i-k} - 1, z_i >= 0
        product_constraints.extend([
            z <= b,
            z <= b[(np.arange(n) + k) % n],
            z >= b + b[(np.arange(n) + k) % n] - 1,
            z >= 0
        ])
        # s_k = sum(z)
        s_k = cp.sum(z)
        # Linear constraint on output
        # c_k = dx * (n/2 - s_k) <= gamma 
        product_constraints.append(s_k <= (n/2 - gamma/dx))
    
    # Set weight constraint
    product_constraints.append(cp.sum(b) == n/2)
    
    # Solve
    prob = Problem(Minimize(gamma), product_constraints)
    
    try:
        # This requires cooling and enforce.
        prob.solve(solver=cp.CBC, verbose=False, timeLimit=15)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return b.value
    except Exception:
        pass
    return None

def ramp_smooth(h_bin, w=3):
    """
    Convert binary to smooth ramped step function.
    """
    n = len(h_bin)
    h_smooth = np.zeros(n)
    # Find transitions
    idx_01 = np.where(np.diff(h_bin) == 1)[0]
    idx_10 = np.where(np.diff(h_bin) == -1)[0]
    # Associate each
    for idx in idx_01:
        start = max(0, idx - w + 1)
        stop = min(n, idx + w + 1)
        x = np.linspace(0, 1, stop - start)
        h_smooth[start:stop] = x

    for idx in idx_10:
        start = max(0, idx - w + 1)
        stop = min(n, idx + w + 1)
        x = np.linspace(1, 0, stop - start)
        h_smooth[start:stop] = x

    # Normalize
    h_smooth = h_smooth / np.sum(h_smooth) * (n/2)
    return h_smooth

def build_autocorr_matrix(h):
    """
    Build row-wise circulant autocorrelation matrix of h
    """
    n = len(h)
    A = np.zeros((n, n))
    for k in range(n):
        A[k] = np.roll(h, k)
    return A

def refine_continuous(h0, target_gamma, max_iter=2000, beta_schedule=None):
    """
    Refine h using projected L-BFGS-B with log-sum-exp surrogate
    """
    if beta_schedule is None:
        beta_schedule = [5 * (2 ** (i // 200)) for i in range(max_iter)]
    
    dx = 2.0 / len(h0)
    n = len(h0)
    h = h0.astype(float)
    
    def compute_obj(h, beta=5.0):
        corr = correlate(h, 1-h, mode='full')
        c_vals = corr * dx
        loss = (1.0 / beta) * np.log(np.sum(np.exp(beta * c_vals)))  # lse
        return loss
    
    def compute_grad(h, beta=5.0):
        corr = correlate(h, 1-h, mode='full')
        c_vals = corr * dx
        wts = np.exp(beta * c_vals)
        wts /= np.sum(wts)
        grad_h = np.zeros_like(h)
        
        for k in range(n):
            if k <= n:
                grad_h += wts[k] * (h[np.roll(np.arange(n), k)] - h)
            else:
                grad_h += wts[k] * (h[np.roll(np.arange(n), k - n)] - h)
            grad_h += wts[k] * (-h[np.roll(np.arange(n), -k)]) # -j-k index
        return grad_h
 

    # Custom L-BFGS for scipy-like 'f' and 'g'
    from scipy.optimize import minimize
    outer_evals = 0

    def closure(h_in):
        return compute_obj(h_in, beta_schedule[outer_evals]), compute_grad(h_in, beta_schedule[outer_evals])
    
    # Make an outer loop for beta scaling
    h_buf = h.copy()
    for i in range(max_iter):
        if i < len(beta_schedule):
            current_beta = beta_schedule[i]
        else:
            current_beta = beta_schedule[-1]
        
        # Perform one optimization step
        h_new = h_buf.copy()
        
        # Smoothed problem projection + gradient step with ARMJO's condition
        # Manually do projected BFGS step on smaller time budget
        try:
            # Optimization step on actual multi-shift outputs
            step_size = 0.01
            g = compute_grad(h_buf, current_beta)
            h_new = h_buf - step_size * g
            
            # Project to feasible
            h_new = project_to_feasible(h_new, n)
            h_buf = h_new.copy()
            # Increment outer eval
            outer_evals += 1

        except:
            # Fallback
            break
            
        # Check termination condition
        if evaluate_c5(h_buf) < target_gamma:
            break
            
    return h_buf
