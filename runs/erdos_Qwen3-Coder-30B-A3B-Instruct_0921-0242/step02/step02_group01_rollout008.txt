```python
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp
from scipy.linalg import toeplitz

def project_to_feasible(h, n):
    """Project h to feasible set: [0,1] and sum = n/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def compute_c5(h):
    """Compute C5 using FFT correlation"""
    dx = 2.0 / len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    max_corr = np.max(corr.real) * dx
    return max_corr

def spectral_factorization(r):
    """Convert autocorrelation r to step function h via spectral factorization"""
    n = len(r)
    # Compute power spectrum
    S = np.fft.fft(r)
    # Clip negative values due to numerical error
    S = np.maximum(S, 0)
    # Spectral factorization
    H = np.sqrt(S)
    # Inverse FFT to get signal
    g = np.real(np.fft.ifft(H))
    return g

def initialize_h(n):
    """Initialize h from a reasonable starting point or uniform"""
    # Use the initial construction if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = initial_h_values
        if len(h) > n:
            h = h[:n]
        else:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n) * 0.5
    return project_to_feasible(h, n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: SDP bisection for maximum t
    n = 256
    t_low, t_high = 0.0, 1.0
    epsilon = 2e-4
    max_sdp_iters = 15
    sdp_time_budget = 0.35 * budget_s
    last_feasible_r = None
    last_feasible_t = 0.0
    
    for _ in range(max_sdp_iters):
        if time.time() - start_time > sdp_time_budget:
            break
        t_trial = (t_low + t_high) / 2
        
        # Build SDP problem
        r = cp.Variable(n)
        constraints = [
            r[0] == 1,
            r[1:] >= t_trial,
            r <= 1,
        ]
        # Toeplitz PSD constraint
        T_r = toeplitz(r)
        constraints.append(cp.PSD(T_r))
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            prob.solve(solver=cp.SCS, verbose=False)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            last_feasible_r = r.value.copy()
            last_feasible_t = t_trial
            t_low = t_trial
        else:
            t_high = t_trial
    
    if last_feasible_r is None:
        h = initialize_h(n)
        return (h, compute_c5(h), n)
    
    # Phase 2: Spectral factorization
    r = last_feasible_r
    g = spectral_factorization(r)
    h0 = project_to_feasible(g, n)
    
    # Phase 3: Deterministic worst-shift refinement
    dx = 2.0 / n
    max_iter = 250
    alpha0 = 0.5 * dx
    
    for i in range(max_iter):
        if time.time() - start_time > 0.7 * budget_s:
            break
        # Compute all correlations
        h_fft = fft(h0, 2 * n)
        ih_fft = fft(1 - h0, 2 * n)
        corr = ifft(h_fft * np.conj(ih_fft))
        Ck = np.real(corr) * dx
        
        # Find worst shift (excluding k=0)
        k_star = np.argmax(Ck[1:]) + 1
        if Ck[k_star] <= 1e-12:  # Already minimal
            break
            
        # Gradient w.r.t. h for worst shift
        grad = np.zeros(n)
        for i in range(n):
            left_idx = (i + k_star) % n
            right_idx = (i - k_star) % n
            grad[i] = dx * ((1 - h0[left_idx]) - h0[right_idx])
            
        # Sub-gradient descent step
        alpha = alpha0 / np.sqrt(i + 1)
        h_new = h0 - alpha * grad
        h0 = project_to_feasible(h_new, n)
    
    # Phase 4: Stochastic polishing with pair transfers
    h_best = h0.copy()
    c5_best = compute_c5(h_best)
    
    max_it = 5000
    for i in range(max_it):
        if time.time() - start_time > 0.95 * budget_s:
            break
        # Pick indices
        i_idx = np.random.randint(0, n)
        j_idx = np.random.randint(0, n)
        
        if h_best[i_idx] <= 1e-10 or h_best[j_idx] >= 1 - 1e-10:
            continue
            
        # Transfer amount
        delta = min(0.01, h_best[i_idx], 1 - h_best[j_idx])
        if delta < 1e-6:
            continue
            
        h_trial = h_best.copy()
        h_trial[i_idx] -= delta
        h_trial[j_idx] += delta
        h_trial = project_to_feasible(h_trial, n)
        
        c5_trial = compute_c5(h_trial)
        if c5_trial < c5_best:
            c5_best = c5_trial
            h_best = h_trial
            
    # Final upsampling and polishing if sufficient time
    if time.time() - start_time < 0.99 * budget_s:
        n_up = n * 2
        h_up = np.interp(np.linspace(0, 2, n_up), np.linspace(0, 2, n), h_best)
        h_up = project_to_feasible(h_up, n_up)
        dx_up = 2.0 / n_up
        
        # Short gradient descent on up-sampled grid
        max_iter_up = 50
        alpha_up = 0.25 * dx_up
        for _ in range(max_iter_up):
            if time.time() - start_time > 0.99 * budget_s:
                break
            h_fft = fft(h_up, 2 * n_up)
            ih_fft = fft(1 - h_up, 2 * n_up)
            corr = ifft(h_fft * np.conj(ih_fft))
            Ck = np.real(corr) * dx_up
            
            k_star = np.argmax(Ck[1:]) + 1
            grad = np.zeros(n_up)
            for i in range(n_up):
                left_idx = (i + k_star) % n_up
                right_idx = (i - k_star) % n_up
                grad[i] = dx_up * ((1 - h_up[left_idx]) - h_up[right_idx])
                
            h_up -= alpha_up * grad
            h_up = project_to_feasible(h_up, n_up)
            
        h_best = h_up
        c5_best = compute_c5(h_best)
    
    return (h_best, c5_best, len(h_best))
```
