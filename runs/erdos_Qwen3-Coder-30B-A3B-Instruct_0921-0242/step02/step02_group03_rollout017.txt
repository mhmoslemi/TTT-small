import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cma

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def memetic_descent(h, n_points, max_steps=25):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        # The maximum is at shift 0, so we need to find the min among non-zero shifts
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        # Gradient wrt h[i] is dx * (1 - h[i + worst_shift_idx] - h[i - worst_shift_idx])
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def build_symmetric_h(p, q, n_points):
    """Build symmetric piecewise constant h from parameters p, q."""
    M = len(p)  # number of segments
    # Transform p to widths via softmax
    w = np.exp(p)
    w = w / np.sum(w)
    # Transform q to heights via logistic
    a = 1.0 / (1.0 + np.exp(-q))
    
    # Build left half of h with n_points/2 elements
    half_length = n_points // 2
    h_half = np.zeros(half_length)
    cum_width = 0
    for i in range(M):
        segment_width = int(np.round(w[i] * half_length))
        if i == M-1:  # Make sure we fill the entire segment
            segment_width = half_length - int(cum_width)
        start_idx = int(cum_width)
        h_half[start_idx:start_idx + segment_width] = a[i]
        cum_width += segment_width
    
    h_full = np.concatenate([h_half, h_half[::-1]])
    return h_full

def cmaes_erdos_optimize(start_time, time_limit, n_points=400, M=7):
    """Optimize h using CMA-ES."""
    # Initialize CMA-ES
    # Mean for p (widths) and q (heights)
    p0 = np.zeros(M)
    q0 = np.zeros(M)
    # Set up middle segments with higher height
    mid_indices = [M//2 - 1, M//2, M//2 + 1]
    for i in range(M):
        if i in mid_indices:
            q0[i] = 5.0  # High value = ~0.993 height
        else:
            q0[i] = -5.0  # Low value = ~0.007 height
    
    # widths give middle 3 segments 0.3 total width
    p0[mid_indices] = 0.3 / len(mid_indices)  # initial equal widths
    # Distribute remainder among others
    if M > 3:
        remaining_weights = 1.0 - 0.3
        num_other = M - 3
        p0[[i for i in range(M) if i not in mid_indices]] = remaining_weights / num_other
    
    # Objective function for CMA-ES
    def obj_func(params):
        # params is p1, q1, ..., pM, qM (length 2M)
        p = params[:M]
        q = params[M:]
        
        h = build_symmetric_h(p, q, n_points)
        h = project_to_feasible(h, n_points)
        c5 = evaluate_c5(h)
        return c5
    
    # Setup arguments for CMA-ES
    # For 2*M-dimensional search, 4 + floor(3*log(2*M)) = 4 + floor(3*log(14)) = 10
    popsize = 4 + int(3 * np.log(2 * M))
    # Initial std dev
    sigma = 0.3
    # CMA-ES options
    options = cma.CMAOptions()
    options.set('maxiter', 500)
    options.set('verbose', -9)  # Silent
    
    try:
        es = cma.CMAEvolutionStrategy(np.concatenate([p0, q0]), 
                                      sigma, 
                                      options)
        best_c5 = float('inf')
        best_h = None
        best_gen = 0
        
        while not es.stop() and time.time() < time_limit - 30:
            # Get candidate solutions
            X = es.ask()
            F = []
            for x in X:
                try:
                    f = obj_func(x)
                    F.append(f)
                except:
                    F.append(float('inf'))
            # Tell the optimization
            es.tell(X, F)
            
            # Track best here
            current_best = min(F)
            if current_best < best_c5:
                best_c5 = current_best
                # Get the corresponding individual
                best_x = X[F.index(current_best)]
                p_best = best_x[:M]
                q_best = best_x[M:]
                h_candidate = build_symmetric_h(p_best, q_best, n_points)
                h_candidate = project_to_feasible(h_candidate, n_points)
                best_h = h_candidate
                best_gen = es.generations
        
        if best_h is not None:
            return best_h, best_c5
        else:
            # If no improvement, return last best
            return None, float('inf')
            
    except Exception as e:
        return None, float('inf')
