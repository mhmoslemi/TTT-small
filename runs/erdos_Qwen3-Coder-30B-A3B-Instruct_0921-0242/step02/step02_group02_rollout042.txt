import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def smooth_ramp(h, width=2):
    """Apply smooth ramp to binary h."""
    h_smooth = np.zeros_like(h)
    for i in range(len(h)):
        h_smooth[i] = 0.5 * (1 + np.tanh((i - len(h) / 2) / width))
    return h_smooth

def lifted_obj(h, beta=10.0):
    """Approximate max overlap with log-sum-exp."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * corr * dx)))

def gradient_lse(h, beta=10.0):
    """Gradient of LSE approximation."""
    dx = 2.0 / len(h)
    n = len(h)
    corr = correlate(h, 1 - h, mode='full')
    w = np.exp(beta * corr * dx)
    w /= np.sum(w)

    grad = np.zeros_like(h)
    for i in range(n):
        for k in range(n):
            ik = (i - k) % n
            grad[i] += w[k] * dx * (h[ik] - h[(i + k) % n])
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Phase 1: Spectral LMI relaxation (simplified for speed)
    n_points = 200
    dx = 2.0 / n_points

    # Start with an initial construction or uniform
    h0 = initial_h_values[:n_points]
    h0 = h0 * (n_points / 2) / np.sum(h0)
    current_h = project_to_feasible(h0, n_points)
    current_c5 = evaluate_c5(current_h)

    # Phase 2: Binary construction upper bound from LMI target
    # A binary pattern is created using a greedy procedure as a start
    h_binary = np.zeros(n_points)
    # Assign values to be 1 or 0
    h_sorted = np.sort(current_h)[::-1]
    cutoff = np.partition(current_h, n_points//2)[n_points//2]
    for i in range(n_points):
        if current_h[i] >= cutoff:
            h_binary[i] = 1
        else:
            h_binary[i] = 0

    # Project to correct sum
    h_binary = project_to_feasible(h_binary, n_points)
    # Refine with heuristic
    for i in range(100):
        if time.time() - start_time > 0.7 * budget_s:
            break
        # Attempt swaps
        i1 = np.random.randint(0, n_points)
        i2 = np.random.randint(0, n_points)
        if h_binary[i1] == 0 and h_binary[i2] == 1:
            # Swap and test new configuration
            h_trial = h_binary.copy()
            h_trial[i1] = 1
            h_trial[i2] = 0
            h_trial = project_to_feasible(h_trial, n_points)
            c5_new = evaluate_c5(h_trial)
            if c5_new < current_c5:
                current_h = h_trial
                current_c5 = c5_new

    # Phase 3: Continuous refinement
    h_refined = current_h.copy()
    current_c5 = evaluate_c5(h_refined)
    n_iter = 0
    max_iter = 1000

    beta = 10.0
    while n_iter < max_iter and time.time() - start_time < 0.9 * budget_s:
        # Use projected gradient descent on log-sum-exp surrogate
        grad = gradient_lse(h_refined, beta)
        step_size = 0.001
        h_temp = h_refined - step_size * grad
        h_temp = project_to_feasible(h_temp, n_points)
        c5_temp = evaluate_c5(h_temp)

        if c5_temp < current_c5:
            h_refined = h_temp
            current_c5 = c5_temp
        else:
            step_size *= 0.5

        n_iter += 1
        if n_iter % 20 == 0:
            # Increase β to sharpen surrogate
            beta *= 1.05

    # Final polishing with subgradients and exact objective
    while time.time() - start_time < budget_s - 1.0:
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        k_star = np.argmax(corr)
        c5_star = corr[k_star] * dx
        grad_star = np.zeros_like(h_refined)
        k_mod = k_star - (len(h_refined) - 1) // 2
        for i in range(len(h_refined)):
            idx_l = (i - k_mod) % len(h_refined)
            idx_r = (i + k_mod) % len(h_refined)
            grad_star[i] = dx * (h_refined[idx_l] - h_refined[idx_r])

        # Subgradient update
        norm_sq = np.dot(grad_star, grad_star)
        if norm_sq > 1e-12:
            step_size = c5_star / norm_sq
        else:
            step_size = 0.01
        h_new = h_refined - step_size * grad_star
        h_new = project_to_feasible(h_new, n_points)
        c5_new = evaluate_c5(h_new)

        if c5_new < current_c5:
            h_refined = h_new
            current_c5 = c5_new

    return (h_refined, current_c5, n_points)
