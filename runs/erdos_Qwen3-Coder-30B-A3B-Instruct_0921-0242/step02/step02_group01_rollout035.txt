import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum and h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    
    return h

def compute_correlation(h, dx):
    """Compute correlation values for all shifts using FFT."""
    n = len(h)
    # Full correlation computation
    h_fft = rfft(h, 2 * n)
    ih_fft = rfft(1 - h, 2 * n)
    corr = irfft(h_fft * np.conj(ih_fft), 2 * n)
    # Take the first n values (positive shifts), skipping shift 0
    return corr[:n] * dx

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = compute_correlation(h, dx)
    # Ignore shift 0 (which is -∫h(1-h)dx)
    max_corr = np.max(corr[1:])
    return max_corr

def gradient_worst_shift(h):
    """Compute gradient with respect to worst shift."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h, dx)
    # Find worst shift k (excluding 0)
    k_star = np.argmax(corr[1:]) + 1
    grad = np.zeros_like(h)
    
    for i in range(n):
        left_idx = (i + k_star) % n
        right_idx = (i - k_star) % n
        grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
    
    return grad

def gradient_descent_worst_shift(h, max_iter=200, alpha0=0.1):
    """Perform gradient descent on the worst shift."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for t in range(max_iter):
        alpha = alpha0 / np.sqrt(t + 1)
        grad = gradient_worst_shift(h)
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n / 2)
        if evaluate_c5(h_new) < evaluate_c5(h):
            h = h_new
        else:
            # Check if no significant improvement
            if abs(evaluate_c5(h_new) - evaluate_c5(h)) < 1e-7:
                break
    
    return h

def differential_evolution_step(population, fitnesses, F=0.6, CR=0.9):
    """Perform one step of DE/rand/1/bin."""
    new_population = []
    n = len(population[0])
    
    for i in range(len(population)):
        # Select three distinct individuals
        a, b, c = np.random.choice(len(population), 3, replace=False)
        x_a, x_b, x_c = population[a], population[b], population[c]
        
        # Create donor vector
        donor = x_a + F * (x_b - x_c)
        donor = np.clip(donor, 0, 1)
        
        # Apply crossover
        trial = np.zeros_like(donor)
        j_rand = np.random.randint(n)
        for j in range(n):
            if np.random.rand() < CR or j == j_rand:
                trial[j] = donor[j]
            else:
                trial[j] = population[i][j]
        
        # Ensure sum constraint
        trial = project_to_feasible(trial, n / 2)
        new_population.append(trial)
    
    return new_population

    n_points = 400
    target_sum = n_points / 2
    F = 0.6
    CR = 0.9
    max_generations = 4000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():

        # Interpolate to n_points

        h_interp = np.interp(x_new, x_orig, h_init)
        h_interp = project_to_feasible(h_interp, target_sum)
        population.append(h_interp)
    else:
        h_interp = np.ones(n_points) * 0.5
        h_interp = project_to_feasible(h_interp, target_sum)
        population.append(h_interp)
    
    # Fill rest of population
    for _ in range(39):  # 40 in total
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = project_to_feasible(h, target_sum)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # DE main loop
    generation_count = 0
    last_improvement = time.time()
    
    while time.time() - start_time < 0.75 * budget_s and generation_count < max_generations:
        generation_count += 1
        
        # Perform DE step
        population = differential_evolution_step(population, fitnesses, F, CR)
        
        # Evaluate new population
        fitnesses = [evaluate_c5(h) for h in population]
        
        # Update best
        new_best_idx = np.argmin(fitnesses)
        if fitnesses[new_best_idx] < best_c5:
            best_c5 = fitnesses[new_best_idx]
            best_h = population[new_best_idx].copy()
            last_improvement = time.time()
        
        # Occasionally perform local refinement
        if generation_count % 200 == 0:
            refined_h = gradient_descent_worst_shift(best_h, max_iter=50)
            refined_c5 = evaluate_c5(refined_h)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
            # Possibly re- seed a few individuals if stagnation
            if time.time() - last_improvement > 5.0:
                for _ in range(4):
                    h = np.random.rand(n_points)
                    h = h * (n_points / 2) / np.sum(h)
                    h = project_to_feasible(h, target_sum)
                    population[np.random.randint(len(population))] = h
                last_improvement = time.time()
    
    # Further refinement on the best before final polishing
    if time.time() - start_time < 0.9 * budget_s:
        refined_h = gradient_descent_worst_shift(best_h, max_iter=200)
        refined_c5 = evaluate_c5(refined_h)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h
    
    # Final late-stage polishing via stochastic swaps
    if time.time() - start_time < 0.995 * budget_s:
        h_final = best_h.copy()
        c5_final = best_c5
        temp = 0.01  # Start with high temperature
        decay = 0.999
        swap_count = 0
        
        while time.time() - start_time < 0.995 * budget_s and swap_count < 5000:
            # Pick two indices to swap
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            if h_final[i] <= 0 or h_final[j] >= 1:
                swap_count += 1
                continue
            
            # Transfer small amount
            delta = np.random.uniform(0.0001, 0.01)
            if h_final[i] < delta:
                swap_count += 1
                continue
            
            h_test = h_final.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, target_sum)
            c5_test = evaluate_c5(h_test)
            
            if c5_test < c5_final or np.random.rand() < np.exp(-(c5_test - c5_final) / temp):
                c5_final = c5_test
                h_final = h_test
            
            temp *= decay
            swap_count += 1
    
    # If using higher resolution to improve bounds:
    if time.time() - start_time < 0.995 * budget_s:
        n_high_res = 800
        # Interpolate best_h to higher resolution
        x_low = np.linspace(0, 2, n_points)
        x_high = np.linspace(0, 2, n_high_res)
        h_high = np.interp(x_high, x_low, h_final)
        h_high = project_to_feasible(h_high, n_high_res / 2)
        
        # Short gradient descent at higher resolution
        h_high_refined = gradient_descent_worst_shift(h_high, max_iter=50)
        c5_high = evaluate_c5(h_high_refined)
        
        if c5_high < c5_final:
            h_final = h_high_refined
            c5_final = c5_high
    
    return (h_final, c5_final, n_points)
