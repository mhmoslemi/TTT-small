import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_feasible(v, n):
    """Project v onto feasible set: 0 <= v[i] <= 1, sum(v) = n/2."""
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - n/2) < 1e-10:
        return v
    
    # Water-filling to adjust to target sum
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    target = n/2
    idx = np.searchsorted(cumsum, target, side="right")
    
    if idx < len(v):
        pivot = (target - cumsum[idx-1]) / (len(v) - idx)
        if pivot < 0:
            pivot = 0
        return np.clip(np.where(np.arange(len(v)) < idx, v_sorted[idx-1], pivot), 0, 1)
    else:
        return np.clip(v_sorted, 0, 1)

def evaluate_c5(h):
    """Compute C5 = max_k integrate h(x)(1-h(x+k))dx."""
    n = len(h)
    dx = 2.0 / n
    # Use FFT for fast correlation computation
    # correlate(h, 1-h, mode='full') produces len(h)*2 - 1 values
    # But we want to align arrays properly.
    # The correct way is corr = ifft(fft(h) * conj(fft(1-h))) 
    # In frequency domain we have:
    # F_{1-h} = F_1 - F_h = 1 - F_h
    # So we compute fft(h) * (1 - fft(h)).conj() and take real part.
    corr_fft = fft(h) * (1 - fft(h)).conj()
    corr = np.real(ifft(corr_fft))
    # corr is complex, may have small imaginary part due to numerical errors.
    corr_max = np.max(corr) * dx
    return corr_max

def gradient_at_worst(h):
    """Compute gradient w.r.t. h of the worst case correlation."""
    n = len(h)
    dx = 2.0 / n
    C = fft(h) * (1 - fft(h)).conj()
    c = np.real(ifft(C))
    
    k_star = np.argmax(c)
    
    # The gradient we want to compute is:
    # d/dh_i (h * (1 - h_{i+k_star})) = 1 - h_{i+k_star} - h_{i-k_star} 
    # but since i-k_star and i+k_star are mod n, we use the full correlation logic.
    # Actually, the vector gradient of c[k*] w.r.t. h[i] is 
    # grad[i] =  dx * ( h[(i+k_star)%n] - h[(i-k_star)%n] )
    
    grad = np.zeros(n)
    for i in range(n):
        idx1 = (i + k_star) % n
        idx2 = (i - k_star) % n
        grad[i] = dx * (h[idx1] - h[idx2])
    
    return grad, k_star

def line_search(v, grad, eta0=0.5):
    """Backtracking line search to ensure improvement."""
    eta = eta0
    current_c5 = evaluate_c5(v)
    
    for _ in range(20):
        v_new = v - eta * grad
        v_new = project_feasible(v_new, len(v))
        c5_new = evaluate_c5(v_new)
        if c5_new < current_c5:
            return v_new, c5_new
        eta /= 2
    return v, current_c5

def mutate(v, mode="gaussian", sigma=0.1, delta=0.01):
    """Generate a mutated version of vector v."""
    n = len(v)
    v_new = v.copy()
    if mode == "gaussian":
        # Perturb idx of 5-10 random indices
        idxs = np.random.choice(n, np.random.randint(5, 11), replace=False)
        for i in idxs:
            v_new[i] += np.random.normal(0, sigma)
        v_new = project_feasible(v_new, n)
    elif mode == "swap":
        # Pick i with h[i] > delta, j with h[j] < 1-delta
        mask_low = v_new < 1 - delta
        mask_high = v_new > delta
        if np.any(mask_low) and np.any(mask_high):
            i = np.random.choice(np.where(mask_high)[0])
            j = np.random.choice(np.where(mask_low)[0])
            d = np.random.uniform(0.001, min(v_new[i], 1-v_new[j], delta))
            v_new[i] -= d
            v_new[j] += d
            v_new = project_feasible(v_new, n)
    elif mode == "dct":
        # Modify a low frequency DCT component
        v_dct = np.fft.fftshift(np.fft.fft(v))
        # Choose random low frequency mode
        k = np.random.randint(1, min(n//4 + 1, 20))
        # Modify one coefficient with some noise
        v_dct[0:2*k] += np.random.normal(0, sigma, 2*k)
        v_new = np.real(np.fft.ifft(np.fft.ifftshift(v_dct)))
        v_new = project_feasible(v_new, n)
    return v_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    safety_margin = 2.0  # seconds

    # Phase 1: Generate random initial pool
    M = 2000  # pool size
    n_points = 250  # start with moderately fine grid
    dx = 2.0 / n_points
    pool = []
    pool_c5 = []
    
    for _ in range(M):
        if time.time() - t_start > 0.15 * budget_s - safety_margin:
            break
        u = np.random.rand(n_points)
        v = project_feasible(u, n_points)
        c5 = evaluate_c5(v)
        pool.append(v)
        pool_c5.append(c5)

    # Sort pool by c5
    indices = np.argsort(pool_c5)
    pool = [pool[i] for i in indices[:30]]
    pool_c5 = [pool_c5[i] for i in indices[:30]]

    # Phase 2: Mutation loop
    t_total = time.time() - t_start
    idx_mutation = 0
    sigma = 0.2
    mutations_per_budget = 500
    while t_total < 0.25 * budget_s - safety_margin and idx_mutation < mutations_per_budget:
        # Pick parent
        parent = pool[np.random.randint(len(pool))]
        # Mutate
        mode = np.random.choice(["gaussian", "swap", "dct"])
        child = mutate(parent, mode=mode, sigma=sigma)
        child_c5 = evaluate_c5(child)
        if child_c5 < pool_c5[-1]:  # replace worst
            pool[-1] = child
            pool_c5[-1] = child_c5
            # re-sort
            indices = np.argsort(pool_c5)
            pool = [pool[i] for i in indices]
            pool_c5 = [pool_c5[i] for i in indices]
        sigma *= 0.99
        idx_mutation += 1
        t_total = time.time() - t_start

    # Phase 3: Local refinement of top candidates
    t_total = time.time() - t_start
    # Refine top 5 only to save time
    pool_refined = []
    pool_c5_refined = []
    for i in range(min(5, len(pool))):
        if t_total > 0.45 * budget_s - safety_margin:
            break
        v = pool[i]
        current_c5 = pool_c5[i]
        max_iter = 2000
        iter_counter = 0
        while iter_counter < max_iter:
            if time.time() - t_start > 0.45 * budget_s - safety_margin:
                break
            # Get gradient
            grad, k_star = gradient_at_worst(v)
            # Line search
            new_v, new_c5 = line_search(v, grad, eta0=0.5)
            if new_c5 < current_c5:
                v = new_v
                current_c5 = new_c5
            else:
                # Compute the k* based on a new v (in case of change)
                c = correlate(v, 1 - v, mode='full')
                k_star = np.argmax(c)
                grad = np.zeros(len(v))
                for j in range(len(v)):
                    idx1 = (j + k_star) % len(v)
                    idx2 = (j - k_star) % len(v)
                    grad[j] = dx * (v[idx1] - v[idx2])
                new_v, new_c5 = line_search(v, grad, eta0=0.5)
                if new_c5 < current_c5:
                    v = new_v
                    current_c5 = new_c5
                else:
                    break
            iter_counter += 1
            if abs(current_c5 - evaluate_c5(v)) < 1e-6:
                break
        pool_refined.append(v)
        pool_c5_refined.append(current_c5)
        t_total = time.time() - t_start

    # Refine with swap heuristic
    t_total = time.time() - t_start
    best_idx = np.argmin(pool_c5_refined)
    best_v = pool_refined[best_idx]
    best_c5 = pool_c5_refined[best_idx]
    
    # Apply swap heuristic
    for _ in range(10000):
        if time.time() - t_start > budget_s - safety_margin:
            break
        # Pick two indices
        i = np.random.randint(len(best_v))
        j = np.random.randint(len(best_v))
        delta = 1e-3 * np.random.uniform(0.1, 1.0)
        # Ensure i, j are valid
        if best_v[i] < 1 - delta or best_v[j] < 1 - delta:
            if best_v[i] > delta and best_v[j] < 1 - delta and best_v[i] > best_v[j]:
                h_trial = best_v.copy()
                h_trial[i] -= delta
                h_trial[j] += delta
                h_trial = project_feasible(h_trial, len(h_trial))
                trial_c5 = evaluate_c5(h_trial)
                if trial_c5 < best_c5:
                    best_v = h_trial
                    best_c5 = trial_c5
                    
    # Final postprocessing
    # Try upsampling to 500 points, if time left
    if time.time() - t_start < 0.9 * budget_s:
        # Interpolate
        n_points_new = 500
        x_old = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_new)
        h_new = np.interp(x_new, x_old, best_v)
        # Project
        h_new = project_feasible(h_new, n_points_new)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_v = h_new
            best_c5 = c5_new

        # Local refine once more on the finer grid
        max_iter = 500
        iter_counter = 0
        v = best_v.copy()
        current_c5 = best_c5
        while iter_counter < max_iter:
            if time.time() - t_start > budget_s - safety_margin:
                break
            # Get gradient
            grad, k_star = gradient_at_worst(v)
            # Backtracking line search
            new_v, new_c5 = line_search(v, grad, eta0=0.2)
            if new_c5 < current_c5:
                v = new_v
                current_c5 = new_c5
            else:
                break
            iter_counter += 1
        best_v = v
        best_c5 = current_c5

    # Final result
    return (best_v.tolist(), best_c5, len(best_v))
