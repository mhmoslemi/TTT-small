import numpy as np
import time
from scipy.optimize import minimize_scalar
from scipy.fft import fft, ifft
import cvxpy as cp

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def build_cyclic_shift_matrix(k, n):
    """Build a sparse cyclic shift matrix P_k."""
    P = np.zeros((n, n))
    for i in range(n):
        j = (i - k) % n
        P[i, j] = 1
    return P

def subgradient_refine(h, dx, time_budget):
    """Use sub-gradient descent to refine the solution."""
    start_time = time.time()
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    iterations = 0
    while time.time() - start_time < time_budget and iterations < 2000:
        # Compute full correlation
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        # Find k that maximizes the correlation (worst overlap)
        k_star = np.argmax(corr)
        if k_star > len(current_h):
            k_shift = k_star - len(current_h)
        else:
            k_shift = k_star - len(current_h) + 1
            
        # Compute gradient w.r.t. h for this shift
        grad = np.zeros_like(current_h)
        for i in range(len(current_h)):
            left_idx = (i + k_shift) % len(current_h)
            right_idx = (i - k_shift) % len(current_h)
            grad[i] = dx * (current_h[left_idx] - current_h[right_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_iter = 10
        updated = False
        for _ in range(max_iter):
            trial_h = current_h - alpha * grad
            trial_h = project_to_feasible(trial_h, len(current_h))
            trial_c5 = evaluate_c5(trial_h)
            if trial_c5 < current_c5:
                current_h = trial_h
                current_c5 = trial_c5
                updated = True
                break
            alpha /= 2.0
            
        # If no improvement, break
        if not updated:
            break
            
        iterations += 1

    return current_h, current_c5

def upsample_and_project(h, n2):
    """Upsample h to n2 points and project to feasible set."""
    # Linear interpolation
    x_orig = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n2)
    h_up = np.interp(x_new, x_orig, h, period=2)
    
    # Re-project to correct sum
    h_up = h_up * (n2 / 2) / np.sum(h_up)
    h_up = np.clip(h_up, 0, 1)
    
    # Water filling
    h_up = project_to_feasible(h_up, n2)
    
    return h_up

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # SDP setting
    n = 180  # moderate resolution
    dx = 2.0 / n
    
    # Build shift matrices (sparse versions for efficiency)
    P = [build_cyclic_shift_matrix(k, n) for k in range(n)]
    
    # Create optimization variables
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # Positive semi-definite
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n / 2,
    ] + [cp.trace(P[k] @ X) >= t for k in range(n)]
    
    # Objective
    objective = cp.Maximize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve the SDP
    prob.solve(solver='SCS', max_iters=25000, eps=1e-5)
    
    # Extract the solution
    X_opt = X.value
    
    # Spectral factorization and extract h0
    eigvals, eigvecs = np.linalg.eigh(X_opt)
    leading_eig = eigvecs[:, -1] * np.sqrt(eigvals[-1])
    h0 = project_to_feasible(leading_eig, n)
    
    # Run with several random draws to increase chance of getting a better one
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    for _ in range(15):
        # Random draw from the solution
        g = np.random.randn(n) @ np.linalg.cholesky(X_opt)
        h = project_to_feasible(np.clip(g, 0, 1), n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Sub-gradient refinement
    while time.time() - start_time < budget_s * 0.5:
        best_h, best_c5 = subgradient_refine(best_h, dx, budget_s - (time.time() - start_time))
        
    # Try upsampling refinement if time permits
    if time.time() - start_time < budget_s * 0.85:
        n2 = 2 * n
        h_up = upsample_and_project(best_h, n2)
        best_h, best_c5 = subgradient_refine(h_up, 2.0 / n2, budget_s - (time.time() - start_time))

    return best_h, best_c5, len(best_h)
