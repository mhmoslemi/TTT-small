import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def fast_cross_correlation(h):
    """Fast computation of correlation using FFT."""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
    
    # Using FFT for efficient cross-correlation
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    corr_vals = corr_vals[:n]  # Take only the non-cyclic part
    return corr_vals

def compute_correlation_changes(h, i, j):
    """Compute change in correlation if we swap h[i] and h[j]."""
    n = len(h)
    delta = np.zeros(n)
    for s in range(n):
        # i contributes to c_s if i + s < n
        if i + s < n:
            delta[s] -= (1 - h[i + s]) - h[i + s]
        if j + s < n:
            delta[s] += (1 - h[j + s]) - h[j + s]
        # j contributes to c_s if j + s < n
        # Already considered above
    return delta

def anneal_binary(h0, n_points, time_limit):
    """Perform simulated annealing on binary vector."""
    np.random.seed()
    h = h0.astype(np.int8)
    c = fast_cross_correlation(h)
    current_energy = np.max(c)
    best_energy = current_energy
    best_h = h.copy()
    
    T0 = 0.02 * n_points  # Initial temperature
    alpha = 0.997  # Cooling factor
    T = T0
    
    max_moves = 100000
    moves = 0
    stagnation = 0
    max_stagnation = 5000
    
    while T > 1e-5 and moves < max_moves and time.time() < time_limit:
        # Pick two random positions
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if h[i] == h[j]:
            continue
        
        # Compute energy change
        delta_c = compute_correlation_changes(h, i, j)
        new_energy = current_energy + np.max(delta_c)
        
        # Acceptance probability
        if new_energy <= current_energy:
            accept = True
        else:
            accept = np.random.rand() < np.exp(-(new_energy - current_energy) / T)
        
        if accept:
            # Perform swap
            h[i], h[j] = h[j], h[i]
            # Update correlations incrementally
            c += delta_c
            current_energy = new_energy
            stagnation = 0
            if current_energy < best_energy:
                best_energy = current_energy
                best_h = h.copy()
        else:
            stagnation += 1
        
        if stagnation >= max_stagnation:
            break
            
        T *= alpha
        moves += 1
    
    return best_h, best_energy

def polish_continuous(h0, n_points):
    """Polish the solution with continuous gradient steps."""
    dx = 2.0 / n_points
    h = h0.astype(float)
    h = project_to_feasible(h, n_points)
    
    # Smooth max surrogate
    def smooth_max(values, lam):
        return (1.0 / lam) * np.log(np.sum(np.exp(lam * values)))
    
    # Gradient components
    def compute_gradient_h(h, s):
        # Build the gradient for shift s of the correlation sum
        n = len(h)
        grad = np.zeros(n)
        for i in range(n-s):
            grad[i] += (1 - h[i + s]) - h[i]
        return grad
    
    # Directional derivatives approximation
    num_steps = 30
    eta = 0.1 * dx
    lambda_start = 5.0
    lambda_end = 30.0
    
    for step in range(num_steps):
        # Compute all the gradients w.r.t. shifts
        shift_grads = []
        for s in range(1, n_points):
            grad = compute_gradient_h(h, s)
            shift_grads.append(grad)
        
        # Calculate soft maximum with growing lambda
        corr = fast_cross_correlation(h)
        lam = lambda_start + (lambda_end - lambda_start) * step / num_steps
        soft_max = smooth_max(corr, lam)
        
        # Compute full gradient of soft_max
        grad_total = np.zeros(n_points)
        for s in range(1, n_points):
            grad_s = shift_grads[s-1]
            # Gradient of exp(lambda * corr[s]) = lambda * exp(lambda * corr[s]) * grad_s
            scale = np.exp(lam * corr[s])
            grad_total += lam * scale * grad_s
        
        # Normalize by the sum of exponentials
        exp_corr = np.exp(lam * corr[1:])
        norm_factor = np.sum(exp_corr)
        if norm_factor > 0:
            grad_total /= norm_factor
        
        h_new = h - eta * grad_total
        h_new = project_to_feasible(h_new, n_points)
        h = h_new
        
        eta *= 0.95
    
    return h, evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    # Use odd n for better symmetry
    n_points = 221
    dx = 2.0 / n_points
    h_best = None
    c5_best = float('inf')
    
    # Run several independent simulated annealing trials
    num_trials = 10
    for trial in range(num_trials):
        if time.time() >= time_limit - 5:
            break
        
        # Choose initialization
        if np.random.rand() < 0.6:
            # Block initialization
            h_initial = np.zeros(n_points, dtype=np.int8)
            mid = n_points // 2
            start = mid - n_points // 4
            end = start + n_points // 2
            h_initial[start:end] = 1
        else:
            # Random initialization
            h_initial = np.zeros(n_points, dtype=np.int8)
            ones = np.random.choice(n_points, n_points // 2, replace=False)
            h_initial[ones] = 1
        
        # Anneal
        h_annealed, energy = anneal_binary(h_initial, n_points, time_limit)
        
        # Convert to float for polishing
        h_float = h_annealed.astype(float)
        h_processed = project_to_feasible(h_float, n_points)
        
        # Polish with continuous refinement
        try:
            h_final, c5 = polish_continuous(h_processed, n_points)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_final.copy()
        except Exception:
            pass
    
    # Final fallback if nothing was found
    if h_best is None:
        # Initialize with a known good approximation
        h_initial = np.zeros(n_points)
        h_initial[n_points//3:2*n_points//3] = 1.0
        h_initial = project_to_feasible(h_initial, n_points)
        h_best, c5_best = polish_continuous(h_initial, n_points)
        
    return (h_best, c5_best, n_points)
