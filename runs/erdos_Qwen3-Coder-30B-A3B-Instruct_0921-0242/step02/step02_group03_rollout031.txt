import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def build_trapezoid(t, H, n):
    """Construct a trapezoidal function h(x) with parameters t (ramp width) and H (height)."""
    dx = 2.0 / n
    h = np.zeros(n)
    
    L = 1.0 / H - t
    a = (2.0 - 2.0 * t - L) / 2.0
    
    if L < 0 or a < 0:
        return h  # Infeasible, return all zeros
    
    # Indices for different regions
    left_ramp_start = int(np.round(a / dx))
    left_ramp_end = int(np.round((a + t) / dx))
    top_start = left_ramp_end
    top_end = int(np.round((a + t + L) / dx))
    right_ramp_start = top_end
    right_ramp_end = int(np.round((a + 2 * t + L) / dx))
    
    # Fill left ramp
    if left_ramp_start < left_ramp_end:
        xvals = np.arange(left_ramp_start, left_ramp_end) * dx + dx / 2.0
        h[left_ramp_start:left_ramp_end] = H * (xvals - a) / t
    
    # Fill top
    if top_start < top_end:
        h[top_start:top_end] = H
    
    # Fill right ramp
    if right_ramp_start < right_ramp_end:
        xvals = np.arange(right_ramp_start, right_ramp_end) * dx + dx / 2.0
        h[right_ramp_start:right_ramp_end] = H * (a + 2 * t + L - xvals) / t
        
    return h

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT correlation."""
    dx = 2.0 / len(h)
    N = 2 * len(h)
    Hc = fft(h, N)
    Gc = fft(1 - h, N)
    corr = ifft(Hc * np.conj(Gc)).real[:2*len(h)-1]
    max_corr = np.max(corr) * dx
    return max_corr

def memetic_descent(h, n_points, max_steps=30):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only non-cyclic part
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5_fft(h_trial)
            current_c5 = evaluate_c5_fft(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def random_feasible(n):
    """Generate a random feasible function."""
    h = np.random.rand(n)
    return project_to_feasible(h, n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    n = 800
    dx = 2.0 / n
    best_val = float('inf')
    best_h = None
    
    # ---- 2-D parametric search ----
    H_vals = np.linspace(0.5, 1.0, 41)
    for H in H_vals:
        t_max = min(1.0/H, 2.0 - 1.0/H)
        t_vals = np.linspace(0.0, t_max, 81)
        for t in t_vals:
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
    
    # ---- refinement around the best pair ----
    if best_h is not None and (time.time() - start_time) < budget_s - 10:
        # Find the parameters for the best h
        H_best = 1.0 / (np.sum(best_h) * dx)  # H from area constraint
        # This assumes that the shape is simple enough for a cleaned lookup
        # Try to determine t and L from the actual structure
        # We'll do it via a very coarse numerical estimation
        h = best_h
        t = 0.1  # We expect a ramp length
        H = 0.8   # Some middle value
        # Since we don't compute exact inverse, we just do minimal correction
        # For now reuse already found h as the golden one
        pass  # The current best is already good from grid search
        
    # ---- polishing with memetic descent ----
    if best_h is not None and (time.time() - start_time) < budget_s - 5:
        best_h = memetic_descent(best_h, n, max_steps=20)
        best_val = evaluate_c5_fft(best_h)

    while (time.time() < time_limit - 5) and best_h is not None:
        h0 = random_feasible(n)
        h_opt = memetic_descent(h0, n, max_steps=30)
        val = evaluate_c5_fft(h_opt)
        if val < best_val:
            best_val, best_h = val, h_opt.copy()
