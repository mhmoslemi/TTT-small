import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Computes the exact C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def smooth_max(h, p):
    """Computes a smooth approximation of max using p-norm."""
    dx = 2.0 / len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    corr_vals = corr.real[:len(h)] * dx
    C_vec = 1 - corr_vals
    C_vec = np.maximum(C_vec, 0)  # Ensure non-negative (should be already)
    if p <= 0:
        return np.sum(C_vec)
    return np.power(np.sum(np.power(C_vec, p)), 1/p)

def compute_smooth_grad(h, p):
    """Compute gradient of smooth max with respect to h."""
    dx = 2.0 / len(h)
    n = len(h)
    
    h_fft = fft(h, 2 * n)
    ih_fft = fft(1 - h, 2 * n)
    corr = ifft(h_fft * np.conj(ih_fft))
    C_vals = 1 - corr.real[:n] * dx
    
    if p <= 0:
        # For p near 0, special case
        weights = np.ones_like(C_vals)
    else:
        # Using the smooth max p-norm gradient formula
        # C_i = 1 - dx * sum(h_j * h_{j+i})
        # We need to differentiate C_i w.r.t. h_k
        
        # Vectorized approach:
        # dC_i / dh_k = dx * ( (1 - h_{k+i}) - h_{k-i} ) 
        # if i != 0 and similar for the other h_{k-i} component
        
        # We compute weighted gradient
        # Let w_k = C(k)^{p-1}
        # grad_k = (1 / S_p^{p-1}) * sum_k w_k * dC_k / dh_k
        weights = np.power(np.maximum(C_vals, 1e-15), p-1)
        
        # Normalize (avoid division by zero)
        if np.sum(weights) == 0:
            weights = np.ones_like(weights)
        else:
            weights /= np.sum(weights)
    
    # Use finite differences in the direction of embedded gradients
    # Since we have C(k) ≈ 1 - dx * sum_{j} h[j] * h[(j+k)%n]
    # ∂C(k)/∂h_i = -dx * h_{i+k} for forward shift (note: we flipped the sign in C)
    
    gradients = np.zeros_like(h)
    for i in range(n):
        grad_sum = 0
        for k in range(n):
            right_idx = (i+k) % n
            left_idx = (i-k) % n
            grad_sum += weights[k] * dx * ( (1-h[right_idx]) - h[left_idx] )
        gradients[i] = grad_sum
    
    return gradients

def generate_seeds(n_points, n_seeds=30):

    seeds = []
    
    # Uniform fill
    seeds.append(np.full(n_points, 0.5))

    dirichlet_sample = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
    seeds.append(dirichlet_sample)
    
    # Interpolated from initial

        h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
        seeds.append(h_interp)
    
    # Binary distribution (determined by n)
    binary_h = np.zeros(n_points)
    mid = n_points // 2
    binary_h[:mid] = 0.0
    binary_h[mid:] = 1.0
    seeds.append(binary_h)
    
    # Add a few random ones
    for i in range(n_seeds - len(seeds)):
        # Basic random sample from Unif(0,1), then scale to sum = n/2
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        seeds.append(h)
    
    # Project all to feasible
    for i in range(len(seeds)):
        seeds[i] = project_to_feasible(seeds[i], n_points)
        
    return seeds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 350
    p_vals = [8, 16, 32, 64, 128, 256]
    
    # Generate seeds
    all_seeds = generate_seeds(n_points, 30)
    
    best_h = None
    best_c5 = float('inf')
    
    # Optimization loop
    for p in p_vals:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        best_seed_h = None
        best_seed_c5 = float('inf')
        
        per_seed_time = (budget_s * 0.95 - (time.time() - start_time)) / len(all_seeds)
        
        for i, seed_h in enumerate(all_seeds):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            def func(h):
                return smooth_max(h, p)
                
            def grad(h):
                return compute_smooth_grad(h, p)
                
            # Optimize with L-BFGS-B
            # Use penalty for sum constraint (normal penalty method)
            lamb = 1e4
            
            def penalized_func(h):
                # Ensure sum==n/2 via penalty
                penalty = lamb * (np.sum(h) - n_points/2)**2
                return func(h) + penalty
            
            def penalized_grad(h):
                # Gradient comes from smooth part, plus penalty
                g = grad(h)
                g += 2 * lamb * (np.sum(h) - n_points/2)
                return g
                
            try:
                res = minimize(
                    penalized_func,
                    seed_h,
                    method='L-BFGS-B',
                    bounds=[(0, 1) for _ in range(n_points)],
                    jac=penalized_grad,
                    options={'maxiter': 150}
                )
                if res.success:
                    h_opt = res.x
                    h_opt = project_to_feasible(h_opt, n_points)
                    c5 = evaluate_c5_exact(h_opt)
                    if c5 < best_seed_c5:
                        best_seed_c5 = c5
                        best_seed_h = h_opt.copy()
            except:
                continue
                
        if best_seed_c5 < best_c5:
            best_c5 = best_seed_c5
            best_h = best_seed_h.copy()
            
        # If we did not converge overall or if it's very close to final
        if p == p_vals[-1]:
            # Apply worst-shift refinement to the final best solution
            if best_h is not None:
                h_final = best_h.copy()
                h_last = best_h.copy()
                
                for _ in range(100):
                    if time.time() - start_time > budget_s * 0.995:
                        break
                        
                    # find worst shift (argmax)
                    h_fft = fft(h_final, 2 * len(h_final))
                    ih_fft = fft(1 - h_final, 2 * len(h_final))
                    corr = ifft(h_fft * np.conj(ih_fft))
                    corr_vals = corr.real[:len(h_final)]  # exclude shift 0
                    worst_shift = np.argmax(corr_vals)
                    
                    # compute gradient for this shift
                    grad = np.zeros_like(h_final)
                    dx = 2.0 / len(h_final)
                    for i in range(len(h_final)):
                        left_idx = (i + worst_shift) % len(h_final)
                        right_idx = (i - worst_shift) % len(h_final)
                        grad[i] = dx * ((1 - h_final[left_idx]) - h_final[right_idx])
                    
                    # Backtracking line search
                    alpha = 0.1 * dx
                    max_iter = 20
                    for _ in range(max_iter):
                        h_new = h_final - alpha * grad
                        h_new = project_to_feasible(h_new, len(h_new))
                        c5_new = evaluate_c5_exact(h_new)
                        
                        if c5_new < evaluate_c5_exact(h_final):
                            h_final = h_new
                            break
                        else:
                            alpha *= 0.5
                    
                    # Stop early if less than 1e-6 improvement
                    if abs(evaluate_c5_exact(h_final) - evaluate_c5_exact(h_last)) < 1e-6:
                        break
                    h_last = h_final.copy()
                
                if evaluate_c5_exact(h_final) < best_c5:
                    best_c5 = evaluate_c5_exact(h_final)
                    best_h = h_final.copy()
    
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        best_c5 = float('inf')
    
    return (best_h, best_c5, n_points)
