import numpy as np
import cvxpy as cp
import time

def full_to_sym(h_full):
    """Convert full h vector to symmetric version (first half + middle)."""
    n = len(h_full)
    if n % 2 == 0:
        half = n // 2
        return np.concatenate([h_full[:half], h_full[half-1::-1]])
    else:
        half = n // 2
        return np.concatenate([h_full[:half+1], h_full[half-1::-1]])

def sym_to_full(h_sym):
    """Convert symmetric h vector back to full version."""
    n = len(h_sym)
    if n % 2 == 0:
        half = n // 2
        return np.concatenate([h_sym[:half], h_sym[half-1::-1]])
    else:
        half = n // 2
        return np.concatenate([h_sym[:half+1], h_sym[half-1::-1]])

def build_constraints(x, t, shift_set, n):
    """Build linear constraints for the LP."""
    constraints = []
    dx = 2.0 / n
    # Ensure 0 <= x <= 1
    constraints.extend([x >= 0, x <= 1])
    
    # Integral constraint (sum of full vector = n/2)
    x_full = sym_to_full(x)
    constraints.append(cp.sum(x_full) == n/2)
    
    # Each shift constraint: sum_i x_i * (1 - x_{i+s}) <= t
    for s in shift_set:
        if s == 0:
            # Direct correlation for shift 0
            expr = 0
            for i in range(n):
                expr += x_full[i] * (1 - x_full[i])
            expr *= dx
        else:
            # Correlation for shift s
            expr = 0
            for i in range(n - s):
                expr += x_full[i] * (1 - x_full[i + s])
            expr *= dx
        constraints.append(expr <= t)
    
    return constraints

def solve_lp(shift_set, x0=None, n=400):
    """Solve the LP with given shift set."""
    # Symmetric version of h
    m = n // 2 + 1  # Number of variables
    x = cp.Variable(m)
    t = cp.Variable()
    
    constraints = build_constraints(x, t, shift_set, n)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        if x0 is not None:
            x.value = x0
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            return sym_to_full(x.value), t.value
        else:
            return None, float('inf')
    except Exception:
        return None, float('inf')

def correlation_vals(h):
    """Compute full correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr[n-1:] * dx
