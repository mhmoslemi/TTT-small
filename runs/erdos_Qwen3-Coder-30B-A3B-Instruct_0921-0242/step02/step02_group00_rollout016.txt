import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from math import isclose

def project_box_sum(y, total):
    """Project vector y onto the set {x : 0 ≤ x_i ≤ 1, sum x_i = total}."""
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if isclose(s, total):
        return y
    # Binary search for lambda
    tol = 1e-12
    low, high = -1.0, 1.0
    while high - low > tol:
        mid = (low + high) / 2
        z = np.clip(y - mid, 0, 1)
        if np.sum(z) > total:
            low = mid
        else:
            high = mid
    return np.clip(y - low, 0, 1)

def compute_autocorrelation(h):
    """Compute the cyclic autocorrelation γ_k of h."""
    n = len(h)
    H = fft(h)
    P = np.real(H * np.conj(H))
    gamma = np.real(ifft(P))
    return gamma

def compute_C5(h):
    """Compute the C5 constant for h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def generate_legendre_seed(n):
    """Generate a Legendre sequence of given length."""
    # Use prime p=257, then take Legendre symbols for 1...p-1, drop last element
    p = 257
    if n != 256:
        raise ValueError("Legendre seed is fixed for n=256")
    ℓ = np.array([int(((k % p) ** ((p - 1) // 2)) % p == 1) if k != 0 else 0 for k in range(1, p)])
    h = 2 * ℓ - 1  # map {0,1} → {-1,1}
    h = (h + 1) / 2  # map {-1,1} → {0,1}
    h = h[:n]
    return h

def generate_mseq_seed(n):
    """Generate a maximal-length LFSR sequence."""
    if n != 256:
        raise ValueError("m-sequence seed is fixed for n=256")
    # Generate a degree 8 primitive polynomial
    # Low-weight primitive polynomial x^8 + x^4 + x^3 + x^2 + 1
    # Initial state
    state = np.array([1, 0, 0, 0, 0, 0, 0, 0], dtype=int)
    sequence = []
    for i in range(n):
        output = state[0]
        # Shift left
        state = np.roll(state, -1)
        # XOR feedback
        feedback = state[7] ^ state[4] ^ state[3] ^ state[2]
        state[0] = feedback
        sequence.append(output)
    sequence = np.array(sequence)
    return sequence.astype(float)

def generate_balanced_random(n):
    """Generate a balanced random binary sequence."""
    h = np.zeros(n, dtype=int)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1
    return h.astype(float)

def binary_hill_climbing(h, maxSwaps=3*n):
    """Optimize a binary h vector using swaps to minimize C5."""
    n = len(h)
    # Compute initial autocorrelation
    # Using fast correlation via FFT: C = FFT(FFT(h).*FFT(1-h))
    H = fft(h)
    H_conj = np.conj(H)
    One_minus_h = 1 - h
    One_minus_h_fft = fft(One_minus_h)
    C = ifft(H * One_minus_h_fft)  # Correlation
    gamma = np.real(C)
    
    # Store the original values
    h_current = h.copy()
    c5_current = compute_C5(h_current)
    
    # Build fast update formula:
    # For each site i, a 1 is swapped with a 0 at site j
    # This affects γ_k = Σ h[i] h[i+k] and similarly for the changes
    
    swaps_done = 0
    no_improvement_count = 0
    while swaps_done < maxSwaps and no_improvement_count < 3 * n:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h_current[i] != 1 or h_current[j] != 0:
            continue
        # Compute Δγ_k for all k
        # h[i] = 1 turns to 0
        # h[j] = 0 turns to 1
        # Δγ_k = -h[i] h[i+k] - h[i] h[i-k] + h[j] h[j+k] + h[j] h[j-k]
        delta_gamma = np.zeros(n)
        for k in range(-n+1, n):
            # shifted indexes modulo n
            k_plus = (i + k) % n
            k_minus = (i - k) % n
            delta_gamma[k + n - 1] -= (h_current[i]*h_current[k_plus] + h_current[i]*h_current[k_minus])
            k_plus2 = (j + k) % n
            k_minus2 = (j - k) % n
            delta_gamma[k + n - 1] += (h_current[j]*h_current[k_plus2] + h_current[j]*h_current[k_minus2])
        
        # New autocorrelation prediction
        gamma_new = gamma + delta_gamma
        # Adjusts back to original labels
        c5_new = 1 - 2/n * np.max(gamma_new[1:])  # This simplifies assuming γ[0] immaterial
        # It's more direct to sum up the correlation terms
        corr_new = correlate(h_current, 1 - h_current, mode='full')
        c5_new_corr = np.max(corr_new) * (2.0 / n) * (1.0 / 2.0)  # Fix as before
        # A bit cleaner continuing with corr function
        corr_new = correlate(h_current, 1 - h_current, mode='full')
        c5_new = np.max(corr_new) * (2.0 / n)  # This is correct
        # Actually best to be achieved via direct C5 evaluation method
        if c5_new < c5_current:
            h_current[i] = 0
            h_current[j] = 1
            # Update gamma directly using previous change
            gamma = gamma_new
            c5_current = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        swaps_done += 1
    return h_current, c5_current

def apply_spectral_alternating_projection(h_start, max_iter=400, α_start=0.7):
    """Apply Spectral Alternating Projection to flatten the autocorrelation."""
    h = h_start.astype(float)
    n = len(h)
    best_c5 = compute_C5(h)
    best_h = h.copy()
    α = α_start
    
    for i in range(max_iter):
        if i > 0 and i % 20 == 0:
            # check improvement
            if compute_C5(h) > best_c5 - 1e-6:
                # Maybe reduce α to allow more stabilizing apply
                α = max(0.1, α - 0.02)
        
        # Compute autocorrelation
        gamma = compute_autocorrelation(h)
        
        # Compute average side lobe level
        avg = (np.sum(gamma) - gamma[0]) / (n - 1)

        # Target γ
        gamma_target = gamma.copy()
        gamma_target[1:] = (1 - α) * gamma[1:] + α * avg

        # Apply FFT to target autocorrelation to get power spectrum
        P_target = fft(gamma_target)
        # Force to be real and non-negative
        P_target = np.maximum(np.real(P_target), 0)
        A_target = np.sqrt(P_target)

        # Preserve phase
        H = fft(h)
        Phi = np.angle(H)
        H_new = A_target * np.exp(1j * Phi)

        # Return to time domain
        h_new = np.real(ifft(H_new))
        # Project back
        h_new = project_box_sum(h_new, n / 2)
        
        h = h_new
        c5 = compute_C5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if i > 0 and i % 200 == 0:
            α = min(α + 0.02, 0.99)

    return best_h, best_c5

def projected_gradient_polish(h_start, max_iter=300):
    """Refine solution using projected gradient descent."""
    h = h_start.copy()
    n = len(h)
    β = 5.0  # Temperature for soft-max approximation
    η = 1.0   # Initial step size
    prev_c5 = compute_C5(h)

    for i in range(max_iter):
        # Compute overlaps
        corr = correlate(h, 1 - h, mode='full')
        C = 1 - (2 / n) * corr  # Corresponds to C_k values
        C = np.maximum(C, 0)  # clip to avoid issues in small negative values
        
        # Compute weights (soft max)
        w = np.exp(β * C)
        w_sum = np.sum(w)
        w = w / w_sum if w_sum > 0 else w

        # Gradient computation using convolutions
        grad1 = correlate(w, h, mode='full')
        grad2 = correlate(w[::-1], h, mode='full')
        grad = (2 / n) * (grad1 + grad2)

        # Line search
        h_trial = h - η * grad
        h_trial = project_box_sum(h_trial, n / 2)
        c5_trial = compute_C5(h_trial)
        if c5_trial < prev_c5:
            h = h_trial
            prev_c5 = c5_trial
            η *= 0.99  # Decaying step size
        else:
            η *= 0.5   # Backtracking
            if η < 1e-8:
                η = 1e-8
        β *= 1.07  # Temperature decay to sharpen max

    return h, prev_c5
