import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cma

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def build_symmetric_h(p, q, n, M):
    """Build symmetric piecewise constant h from p (widths) and q (heights)."""
    # Transform p to actual widths via softmax
    w = np.exp(p)
    w = w / np.sum(w)
    
    # Transform q to actual heights via logistic
    a = 1.0 / (1.0 + np.exp(-q))
    
    # Build left half h_half from segments
    n_half = n // 2
    h_half = np.zeros(n_half)
    
    # Use cumulative widths for segment boundaries
    cum_w = np.concatenate([[0.0], np.cumsum(w)])
    seg_bounds = (cum_w * n_half).astype(int)
    
    # Fill each segment
    for i in range(M):
        start_idx = seg_bounds[i]
        end_idx = seg_bounds[i + 1]
        if end_idx > start_idx:
            h_half[start_idx:end_idx] = a[i]
    
    # Construct full h (symmetric about center)
    h_full = np.concatenate([h_half, h_half[-2::-1]])
    
    return h_full

def eval_objective(params, n, M):
    """Evaluate the objective function for given parameters."""
    p = params[:M]
    q = params[M:]
    
    # Build the function h
    h = build_symmetric_h(p, q, n, M)
    
    # Project to feasible set
    h_projected = project_to_feasible(h, n)
    
    # Compute C5 bound
    C5 = evaluate_c5(h_projected)
    
    return C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 400
    M = 7
    D = 2 * M  # total dimension
    
    # Initial mean
    p0 = np.zeros(M)
    p0[2:5] = 0.3  # middle 3 segments wider
    p0[0] = p0[1] = p0[5] = p0[6] = 0.1
    q0 = np.zeros(M)
    q0[2:5] = 5.0  # middle 3 segments high
    q0[0:2] = q0[5:7] = -5.0  # outer segments low
    
    mean = np.concatenate([p0, q0])
    sigma = 0.3
    
    # Set up CMA-ES
    es = cma.CMAEvolutionStrategy(mean, sigma, {'popsize': 4 + int(3 * np.log(D))})

    fallback_solution = None
    fallback_c5 = float('inf')
    
    # Generational loop
    G_noimp = 80
    num_restarts = 0
    max_restarts = 10
    
    best_C5 = float('inf')
    best_h = None
    
    while es.should_stop() is False:
        if time.time() - start_time > budget_s - 30:
            break
            
        # Get new candidate solutions
        X = es.ask()
        fitvals = []
        
        for x in X:
            fit = eval_objective(x, n, M)
            fitvals.append(fit)
            
        # Tell the optimizer
        es.tell(X, fitvals)
        
        # Update the best solution
        current_best_idx = np.argmin(fitvals)
        current_best_val = fitvals[current_best_idx]
        
        if current_best_val < best_C5:
            best_C5 = current_best_val
            # Store the best h from the top solution
            p_best = X[current_best_idx][:M]
            q_best = X[current_best_idx][M:]
            h = build_symmetric_h(p_best, q_best, n, M)
            best_h = project_to_feasible(h, n)
        
        # Check for restart
        if es.countiter % G_noimp == 0:
            if num_restarts < max_restarts:
                # Check if improvement was negligible
                if es.countiter > 100:
                    recent_improvements = np.diff(es.fbest_history[-10:]) if len(es.fbest_history) >= 10 else [1e10]
                    if np.all(np.abs(recent_improvements) < 1e-5):
                        # Restart
                        es.restart()
                        num_restarts += 1
                        if num_restarts == 1:
                            fallback_solution = best_h.copy()
                            fallback_c5 = best_C5

    # If we didn't restart, store the best solution found
    if num_restarts == 0:
        # Get last one from the final population if needed
        fitvals = [eval_objective(x, n, M) for x in es.ask()]
        if fitvals and min(fitvals) < best_C5:
            best_C5 = min(fitvals)
            best_idx = np.argmin(fitvals)
            p_best = es.ask()[best_idx][:M]
            q_best = es.ask()[best_idx][M:]
            h = build_symmetric_h(p_best, q_best, n, M)
            best_h = project_to_feasible(h, n)

    # Local polishing
    if best_h is not None:
        # Memetic descent for local polishing
        # Import outside scope to ensure it's available
        from scipy.signal import correlate
        from scipy.fft import fft, ifft
        
        def compute_worst_shift_grad(h, n_points):
            h_padded = np.pad(h, (0, n_points-1), mode='constant', constant_values=0)
            one_minus_h_padded = np.pad(1-h, (0, n_points-1), mode='constant', constant_values=0)
            
            corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
            corr_vals = ifft(corr_freq).real
            corr_vals = corr_vals[:n_points]
            dx = 2.0 / n_points
            corr_vals = corr_vals * dx
            
            non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
            worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
            
            grad = np.zeros_like(h)
            for i in range(n_points):
                right_idx = (i + worst_shift_idx) % n_points
                left_idx = (i - worst_shift_idx) % n_points
                grad[i] = dx * (1 - h[right_idx] - h[left_idx])
            
            return grad, worst_shift_idx

        def memetic_descent(h, n_points, max_steps=25):
            dx = 2.0 / n_points
            h_current = h.copy()
            
            for _ in range(max_steps):
                grad, worst_shift = compute_worst_shift_grad(h_current, n_points)
                
                alpha = 1.0
                while alpha > 1e-9:
                    h_trial = h_current - alpha * grad
                    h_trial = project_to_feasible(h_trial, n_points)
                    trial_c5 = evaluate_c5(h_trial)
                    current_c5 = evaluate_c5(h_current)
                    
                    if trial_c5 < current_c5:
                        h_current = h_trial
                        break
                    else:
                        alpha *= 0.5
                        if alpha < 1e-9:
                            break
            
            return h_current
        
        # Only polish if we're within the time limit
        if time.time() - start_time < budget_s - 10:
            best_h_refined = memetic_descent(best_h, n, max_steps=25)
            refined_c5 = evaluate_c5(best_h_refined)
            if refined_c5 < best_C5:
                best_C5 = refined_c5
                best_h = best_h_refined

    # Final fallback check
    if fallback_solution is not None and fallback_c5 < best_C5:
        best_h = fallback_solution
        best_C5 = fallback_c5
        
    if best_h is None:
        # Fallback to constant 0.5
        best_h = np.ones(n) * 0.5
        best_C5 = evaluate_c5(best_h)
    
    return (best_h, best_C5, n)
