import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def fast_correlation_update(h, old_corr, i, j):
    """Compute new correlation values after swapping positions i and j."""
    n = len(h)
    new_corr = old_corr.copy()
    for s in range(n):
        if i + s < n:
            new_corr[s] += (1 - h[i+s]) - h[i+s]  # contribution removed
        if j + s < n:
            new_corr[s] += (1 - h[j+s]) - h[j+s]  # contribution added
    return new_corr

def anneal_binary(h0, n, time_limit):
    """Perform simulated annealing on binary vector to maximise minimum correlation."""
    T0 = 0.02 * n
    alpha = 0.997
    max_moves = 50000
    
    h = h0.copy()
    h = h.astype(np.int8)
    
    # Compute initial correlation
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    corr_vals = corr_vals[:n]
    dx = 2.0 / n
    corr_vals = corr_vals * dx

    current_energy = np.max(corr_vals)
    best_h = h.copy()
    best_energy = current_energy

    T = T0
    moves = 0
    no_improvement_count = 0
    while moves < max_moves and T > 1e-5 and time.time() < time_limit:
        # Swap two random positions
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h[i] == h[j]:
            continue
            
        # Compute new correlation values efficiently
        new_corr = fast_correlation_update(h, corr_vals, i, j)
        new_energy = np.max(new_corr)
        
        # Accept or reject
        if new_energy <= current_energy or np.random.rand() < np.exp(-(new_energy - current_energy) / T):
            h[i], h[j] = h[j], h[i]
            corr_vals = new_corr
            current_energy = new_energy
            if new_energy < best_energy:
                best_energy = new_energy
                best_h = h.copy()
                no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 5000:
            break
            
        T *= alpha
        moves += 1
        
    return best_h, best_energy

def polish_continuous(h_binary, n_points):
    """Polish the binary solution using projected gradient descent."""
    h = h_binary.astype(float)
    dx = 2.0 / n_points
    
    # Smooth max surrogate
    def smooth_max(corr, lam):
        return 1.0/lam * np.log(np.sum(np.exp(lam * corr)))
        
    # Gradient computation
    def compute_gradient(h, corr):
        # Reconstruct to make it work with full correlation
        h_padded = np.pad(h, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h, (0, n_points-1), mode='constant', constant_values=0)
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points] * dx
        
        grad = np.zeros_like(h)
        # For each position, compute gradient of the smooth max
        for i in range(n_points):
            grad[i] = 0
            for s in range(n_points):
                if i + s < n_points:
                    # Gradient of h_i * (1 - h_{i+s})
                    grad[i] += (1 - h[i+s]) * dx
                if i - s >= 0 and s > 0:
                    grad[i] -= h[i-s] * dx
        return grad
        
    # Gradient descent
    max_iter = 30
    lambda_schedule = [5, 10, 15, 20, 25, 30]
    stepsize = 0.1 * dx
    current_h = h.copy()
    
    for iter_num in range(max_iter):
        # Compute current correlation and gradient
        h_padded = np.pad(current_h, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-current_h, (0, n_points-1), mode='constant', constant_values=0)
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points] * dx
        
        # Find worst correlation position
        worst_idx = np.argmax(corr_vals)
        
        # Compute gradient - simplified version 
        grad = np.zeros_like(current_h)
        # This is a more symbolic gradient for smooth function approximating max
        # Visualize it as being approximated with a smooth max
        # For computation purposes, numerically approximate gradient 
        # Gradients can be derived from correlation formula but are expensive to compute
        # simpler: sample movement in direction to reduce the max 
        
        # Approximate gradient for just one term. Based on structure:
        # Incidentally, since the derivative of smooth max with respect to h_i
        # involves terms of contributions from the positions where this h_i enters,
        # its algebraic form from full convolution shows it's related to a 
        # sliding exclusive correlation of h with 1-h.
        
        # But for precision, let's optimize with robust noise reduction
        # Instead of numericals, keep a known good structure:
        # The current approach with full correlation FFt and estimation is already robust
        
        # Making a conservative gradient step to move down:
        # Step one component in direction that reduces worst correlation
        # This is an approximation but reasonably good for small steps
        freeze_positions = [np.argmax(corr_vals)]
        relative_grad = np.zeros_like(current_h)
        
        # Get approximate gradient step from directional analysis
        for i in range(n_points):
            relative_grad[i] = 0.0
            # Casually approximate derivative on key positions using discrete finite diff approach
            shift_grad = np.zeros_like(current_h)
            for sh in range(1, n_points - 1):
                if sh + i < n_points:
                    shift_grad[i] += (1 - current_h[sh + i]) * dx
                if i - sh >= 0:
                    shift_grad[i] -= current_h[i - sh] * dx
            
            relative_grad[i] = shift_grad[i]
    
        # Clip and project as necessary for feasibility
        h_trial = current_h - stepsize * relative_grad
        h_trial = np.clip(h_trial, 0, 1)
        
        # Adjust if not exactly correct integral
        h_trial = project_to_feasible(h_trial, n_points)
        
        current_h = h_trial
        
        # Rescale and update make small pieces for gradient step
        stepsize *= 0.97  # gradual decay 
   
    return current_h, evaluate_c5(current_h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    # Parameters
    n = 221  # odd number, ~200-300 points to resolve well but stay fast
    dx = 2.0 / n
    h_best = None
    c5_best = float('inf')
    
    np.random.seed(int(seed))
    
    restarts = 12  # Number of independent restarts
    for i in range(restarts):
        if time.time() > time_limit - 10:
            break
            
        # Initializations: choose between block or random binary
        if np.random.rand() < 0.5:
            # Block initialization  
            h = np.zeros(n, dtype=np.int8)
            mid = n // 2
            h[mid - n//4:mid + n//4] = 1
            np.random.shuffle(h)
        else:
            # Random initialization  
            indices = np.random.permutation(n)
            h = np.zeros(n, dtype=np.int8)
            h[indices[:n//2]] = 1
            
        h = project_to_feasible(h.astype(float), n).astype(np.int8)
        
        # Anneal to get binary solution
        try:
            h_annealed, _ = anneal_binary(h, n, time_limit)
        except:
            continue
            
        # Polish the binary solution
        try:
            h_final, c5_polished = polish_continuous(h_annealed, n)
        except:
            continue
            
        if c5_polished < c5_best:
            c5_best = c5_polished
            h_best = h_final.copy()
            
    # Final evaluation
    if h_best is not None:
        c5_best = evaluate_c5(h_best)
    else:
        h_best = np.ones(n) * 0.5
        c5_best = evaluate_c5(h_best)
    
    return (h_best, c5_best, n)
