import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT correlation."""
    n = len(h)
    dx = 2.0 / n
    # Full linear correlation using FFT
    N = 2 * n
    Hc = fft(h, N)
    Gc = fft(1 - h, N)
    corr = ifft(Hc * np.conj(Gc)).real[:2*n-1]
    max_corr = np.max(corr) * dx
    return max_corr

def build_trapezoid(t, H, n):
    """Build a trapezoidal function with given parameters."""
    dx = 2.0 / n
    h = np.zeros(n)
    
    L = 1.0 / H - t
    a = (2.0 - 2.0 * t - L) / 2.0
    
    if L < 0 or a < 0:
        return h  # Infeasible
    
    # Indices for the three parts
    # Left ramp: [a, a+t)
    # Flat top: [a+t, a+t+L)
    # Right ramp: [a+t+L, a+2*t+L)
    
    a_idx = int(a / dx)
    t_idx = int(t / dx)
    L_idx = int(L / dx)
    
    if a_idx >= n:
        return h
    if a_idx + t_idx >= n:
        return h
    if a_idx + t_idx + L_idx >= n:
        return h
    if a_idx + 2 * t_idx + L_idx > n:
        return h
    
    # Left ramp
    if t > 0:
        ramp_indices = np.arange(a_idx, min(a_idx + t_idx, n))
        h[ramp_indices] = H * (ramp_indices - a_idx) * dx / t
    
    # Flat top
    if L > 0:
        top_indices = np.arange(a_idx + t_idx, min(a_idx + t_idx + L_idx, n))
        h[top_indices] = H
    
    # Right ramp
    if t > 0:
        ramp_indices = np.arange(a_idx + t_idx + L_idx, min(a_idx + 2 * t_idx + L_idx, n))
        h[ramp_indices] = H * (a_idx + 2 * t_idx + L_idx - ramp_indices) * dx / t
    
    return h

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def memetic_descent(h, n_points, max_steps=30):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        # The maximum is at shift 0, so we need to find the min among non-zero shifts
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        # Gradient wrt h[i] is dx * (1 - h[i + worst_shift_idx] - h[i - worst_shift_idx])
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5_fft(h_trial)
            current_c5 = evaluate_c5_fft(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    n = 800
    dx = 2.0 / n
    
    best_val = float('inf')
    best_h = None
    
    # ---- 2-D parametric search ----
    H_vals = np.linspace(0.5, 1.0, 41)
    for H in H_vals:
        t_max = min(1.0/H, 2.0 - 1.0/H)
        if t_max <= 0:
            continue
        t_vals = np.linspace(0.0, t_max, 81)
        for t in t_vals:
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
    
    if best_h is None:
        # Fallback
        h = np.ones(n) * 0.5
        best_val = evaluate_c5_fft(h)
        best_h = h.copy()
    
    # ---- refinement around the best pair ----
    # Reverse engineer: find t and H from best_h, using the integral = 1 and shape properties
    # Here we do a coarse grid around the current found optimum, as a safeguard
    # A richer grid near t=0.2723, H=0.8897 based on empirical finding
    t_best = 0.2723
    H_best = 0.8897
    delta_t = 0.01
    delta_H = 0.025
    t_range = np.linspace(t_best - 2*delta_t, t_best + 2*delta_t, 41)
    H_range = np.linspace(H_best - 2*delta_H, H_best + 2*delta_H, 41)
    
    for H in H_range:
        t_max = min(1.0/H, 2.0 - 1.0/H)
        for t in t_range:
            if t < 0 or t > t_max or H < 0.5 or H > 1:
                continue
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
    
    # ---- local Nelder-Mead optimisation (if we want to)
    # This would be for small changes to parameters, ignored for consistency
    
    # ---- polishing with memetic descent ----
    best_h = memetic_descent(best_h, n, max_steps=20)
    best_val = evaluate_c5_fft(best_h)
    
    # ---- fallback random memetic runs if time permits ----
    remaining_time = time_limit - time.time()
    if remaining_time > 5.0:
        n_small = 300
        for _ in range(10):
            if time.time() >= time_limit - 5:
                break
            h0 = np.random.rand(n_small)
            h0 = project_to_feasible(h0, n_small)
            h_opt = memetic_descent(h0, n_small, max_steps=30)
            val = evaluate_c5_fft(h_opt)
            if val < best_val:
                best_val, best_h = val, h_opt.copy()
    
    return (best_h, best_val, n)
