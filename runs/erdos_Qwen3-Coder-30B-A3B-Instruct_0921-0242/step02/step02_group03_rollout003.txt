import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_correlations(h):
    """Efficiently compute all correlations using FFT."""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
    # Using FFT for efficient cross-correlation
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    corr_vals = corr_vals[:n]  # Take only the non-cyclic part
    dx = 2.0 / n
    corr_vals = corr_vals * dx
    return corr_vals

def incrementally_update_correlations(corr_vals, h, i, j):
    """Update correlations in O(n) for swapping h[i] and h[j]."""
    # Compute changes in all correlations due to swap
    n = len(h)
    delta = np.zeros(n)
    
    # contribution removed at shift s from position i
    for s in range(n):
        if i+s < n:
            delta[s] -= (1 - h[i+s]) + h[i+s]
        if j+s < n:
            delta[s] += (1 - h[j+s]) + h[j+s]

    # Apply update
    corr_vals = corr_vals + delta
    return corr_vals

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def polish_continuous(h_binary, n_points, max_iter=30):
    """Post-process binary solution with projected gradient on smooth max."""
    h = h_binary.astype(float)
    
    # Set up smooth max surrogate
    lambda_param = 5.0
    dx = 2.0 / n_points
    
    def smooth_max(corr_vals, lam):
        return (1.0 / lam) * np.log(np.sum(np.exp(lam * corr_vals)))
        
    def compute_grad_smooth(corr_vals, h, lam):
        """Compute gradient of smooth max with respect to h."""
        n = len(h)
        exp_vals = np.exp(lam * corr_vals)
        sum_exp = np.sum(exp_vals)
        
        grad = np.zeros(n)
        for i in range(n):
            # Derivative w.r.t. h[i]
            dci_dh_i = 0
            for s in range(n):
                if i + s < n:
                    dci_dh_i += (1 - h[i+s]) - h[i - s]  # This is approximate
                if i - s >= 0:
                    dci_dh_i -= (1 - h[i-s]) - h[i + s]
            grad[i] = (lam / sum_exp) * exp_vals @ (dci_dh_i) 
        return grad

    eta = 0.1 * dx  # initial step size
    for iteration in range(max_iter):
        corr_vals = compute_correlations(h)
        c5_smooth = smooth_max(corr_vals, lambda_param)
        grad = compute_grad_smooth(corr_vals, h, lambda_param)
        
        h_new = h - eta * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Update parameters
        lambda_param += 5.0
        eta *= 0.9  # decay learning rate
        
        h = h_new

    return h, evaluate_c5(h_new)

def anneal_binary(h_initial, n, time_limit):
    """Simulated annealing on binary sequences."""
    np.random.seed()
    
    h = h_initial.copy()
    corr_vals = compute_correlations(h)
    energy = np.max(corr_vals)
    
    T = 0.02 * n  # initial temperature
    alpha = 0.997  # cooling factor
    
    best_h = h.copy()
    best_energy = energy
    
    iter_count = 0
    early_stop_count = 0
    max_early_stop = 5000
    
    while T > 1e-5 and time.time() < time_limit:
        # Pick two random indices to swap
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
            
        # Update correlations incrementally
        old_corr_vals = corr_vals.copy()
        corr_vals = incrementally_update_correlations(corr_vals, h, i, j)
        new_energy = np.max(corr_vals)
        
        # Accept or reject
        delta = new_energy - energy
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            # Accept
            h[i], h[j] = h[j], h[i]
            energy = new_energy
            # Update best
            if energy < best_energy:
                best_energy = energy
                best_h = h.copy()
                early_stop_count = 0
            else:
                early_stop_count += 1
        else:
            # Reject, restore correlations
            corr_vals = old_corr_vals
        
        T *= alpha
        iter_count += 1
        if early_stop_count > max_early_stop:
            break
            
    return best_h, best_energy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 10  # leave buffering time
    
    # Use 221 for nice balance between resolution and computation
    n = 221
    dx = 2.0 / n
    
    h_best = None
    c5_best = float('inf')
    
    # Run 8 simulated annealing trials in parallel (sequential)
    for trial in range(8):
        # Create an initial balanced binary sequence
        h = np.zeros(n, dtype=int)
        h[:n//2] = 1
        np.random.shuffle(h)
        
        # Occasionally use a block pattern initialization
        if np.random.rand() < 0.3:
            h = np.zeros(n, dtype=int)
            middle = n // 2
            start = np.random.randint(0, n//2)
            h[start:start+middle] = 1
            
        # Run simulated annealing
        h_final, energy = anneal_binary(h, n, time_limit)
        
        # Polish the solution
        h_polished, c5 = polish_continuous(h_final, n)
        
        # Keep track of best
        if c5 < c5_best:
            c5_best = c5
            h_best = h_polished.copy()
    
    return (h_best, c5_best, n)
