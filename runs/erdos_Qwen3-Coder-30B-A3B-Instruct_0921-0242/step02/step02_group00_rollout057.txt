import numpy as np

from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import bisect

def project_box_sum(y, total):
    """Project y onto the set {x : 0 <= x_i <= 1, sum(x) = total} using water filling."""
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if abs(s - total) < 1e-10:
        return y
    # Use bisection to find lambda
    def sum_with_lambda(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    a, b = -1.0, 2.0
    try:
        lam = bisect(lambda x: sum_with_lambda(x) - total, a, b, xtol=1e-12)
    except:
        lam = 0.0
    return np.clip(y - lam, 0, 1)

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorrelation(h):
    """Compute aperiodic autocorrelation using FFT."""
    H = fft(h)
    P = np.abs(H)**2
    gamma = ifft(P).real
    return gamma

def build_target_autocorrelation(gamma, alpha=0.7):
    """Build target autocorrelation by convex combination."""
    avg = (np.sum(gamma) - gamma[0]) / (len(gamma) - 1)
    gamma_target = gamma.copy()
    gamma_target[1:] = alpha * gamma[1:] + (1 - alpha) * avg
    return gamma_target

def swap_autocorr_update(gamma, h, i, j):
    """Efficiently compute new autocorrelation when swapping h[i] and h[j]."""
    n = len(h)
    gamma_new = gamma.copy()
    # Assume the values to swap are 1 and 0
    for k in range(-n+1, n):
        # Use periodic indexing
        idx1 = (i + k) % n
        idx2 = (j + k) % n
        # Remove contributions from h[i] and h[j]
        gamma_new[k + n - 1] -= h[i] * h[idx1] + h[i] * h[idx2]
        # Add contributions from h[j] and h[i]
        gamma_new[k + n - 1] += h[j] * h[idx1] + h[j] * h[idx2]
    return gamma_new

def hill_climb_binary_seed(h, max_iter=3000):
    """Perform efficient binary hill climbing on a seed."""
    n = len(h)
    h = h.astype(int)
    gamma = compute_autocorrelation(h.astype(float))
    c5 = 1.0 - (2.0 / n) * np.max(gamma[1:])
    
    # Initialize swaps to avoid infinite loops
    iter_count = 0
    improved = True
    while improved and iter_count < max_iter:
        improved = False
        iter_count += 1
        
        # Randomly select i1 from 1s, i2 from 0s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute new gamma after swap
        gamma_new = swap_autocorr_update(gamma, h, i, j)
        c5_new = 1.0 - (2.0 / n) * np.max(gamma_new[1:])
        
        if c5_new < c5:
            h[i] = 0
            h[j] = 1
            gamma = gamma_new
            c5 = c5_new
            improved = True
    
    return h, c5, gamma

def generate_legendre_seed(n):
    """Generate Legendre symbol sequence (prime p=257)."""
    p = 257
    if n + 1 >= p:
        raise ValueError("Need n + 1 < p for Legendre sequence")
    h = [0] * n
    for i in range(1, p):
        if i < n + 1:
            # Use Legendre symbol (i|p)
            if pow(i, (p - 1) // 2, p) == 1:
                h[i - 1] = 1
            else:
                h[i - 1] = 0
    return np.array(h)

def generate_random_balanced_seed(n):
    """Generate a random balanced binary string."""
    h = np.zeros(n)
    ones = np.random.choice(n, size=n//2, replace=False)
    h[ones] = 1
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    n_points = 256
    dx = 2.0 / n_points
    h_best = None
    c5_best = float('inf')
    
    # Part 1: Generate high-quality seeds
    seeds = []
    # Legendre (prime = 257)
    try:
        legendre_seed = generate_legendre_seed(n_points)
        legendre_seed = project_box_sum(legendre_seed, n_points / 2)
        seeds.append(('legendre', legendre_seed))
    except:
        # fallback: use random
        rand_seed = generate_random_balanced_seed(n_points)
        rand_seed = project_box_sum(rand_seed, n_points / 2)
        seeds.append(('legendre', rand_seed))
    
    # m-sequence
    try:
        mseq_seed = generate_msequence_seed(n_points)
        mseq_seed = project_box_sum(mseq_seed, n_points / 2)
        seeds.append(('mseq', mseq_seed))
    except:
        pass
    # Random balanced strings, 5 times
    for i in range(5):
        rand_seed = generate_random_balanced_seed(n_points)
        rand_seed = project_box_sum(rand_seed, n_points / 2)
        seeds.append(('random_{}'.format(i), rand_seed))
    
    # Perform binary hill climbing on each seed
    for seed_name, h_seed in seeds:
        if time.time() - start_time > 0.7 * budget_s:
            break
        h_bin, c5_bin, gamma_bin = hill_climb_binary_seed(h_seed, 3000)
        if c5_bin < c5_best:
            c5_best = c5_bin
            h_best = h_bin.copy()

    # Part 2: Spectral Alternating Projection (SAP)
    if h_best is not None:
        h = h_best.astype(float)
        alpha = 0.7
        max_iter_sap = 400
        best_h_sap = h.copy()
        best_c5_sap = c5_best
        
        for i in range(max_iter_sap):
            if time.time() - start_time > 0.9 * budget_s:
                break
            
            H = fft(h)
            gamma = ifft(np.abs(H)**2).real
            gamma_target = build_target_autocorrelation(gamma, alpha)
            P_target = fft(gamma_target)
            P_target = np.maximum(P_target, 0)  # Non-negativity
            A_target = np.sqrt(P_target)
            H_new = A_target * np.exp(1j * np.angle(H))
            h_new = ifft(H_new).real
            
            h_new = project_box_sum(h_new, n_points / 2)
            
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5_sap:
                best_c5_sap = c5_new
                best_h_sap = h_new.copy()
                
            h = h_new
            if alpha < 0.99:
                alpha = min(alpha + 0.02, 0.99)
                
        h_best = best_h_sap
        c5_best = best_c5_sap
    
    # Part 3: Final gradient polishing
    if time.time() - start_time < budget_s - 0.1:
        beta = 5.0
        eta = 1.0
        max_iter_grad = 300
        best_h_grad = h_best.copy()
        best_c5_grad = c5_best
        
        for i in range(max_iter_grad):
            if time.time() - start_time > budget_s - 0.01:
                break
        
            # Compute weights
            gamma = compute_autocorrelation(best_h_grad)
            C = 1 - (2 / n_points) * gamma
            w = np.exp(beta * C)
            w = w / np.sum(w)
            
            # Compute gradient (approximate)
            # ∇ = -(2/n)* [ conv(w, h) + conv(w[::-1], h) ]
            W_fft = fft(w)
            H_fft = fft(best_h_grad)
            conv1_fft = W_fft * H_fft
            conv2_fft = fft(w[::-1]) * H_fft
            g1 = ifft(conv1_fft).real
            g2 = ifft(conv2_fft).real
            grad = -(2 / n_points) * (g1 + g2)
            
            # Backtracking line search
            h_trial = project_box_sum(best_h_grad - eta * grad, n_points / 2)
            c5_trial = evaluate_c5(h_trial)
            
            if c5_trial < best_c5_grad:
                best_h_grad = h_trial
                best_c5_grad = c5_trial
                
                # Update for next iteration
                beta *= 1.2  # Sharpens max approximation
                if eta > 0.01:
                    eta = 0.5 * eta  # Gradually decrease step size
            else:
                eta = 0.5 * eta  # Reduce step size if no improvement
                
        h_best = best_h_grad
        c5_best = best_c5_grad
    
    return (h_best, c5_best, n_points)
