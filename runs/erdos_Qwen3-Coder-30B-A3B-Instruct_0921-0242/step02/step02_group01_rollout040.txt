import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for a given step function h."""
    dx = 2.0 / len(h_values)
    # Full correlation using FFT for efficiency
    h_fft = fft(h_values, 2 * len(h_values))
    ih_fft = fft(1 - h_values, 2 * len(h_values))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def compute_autocorrelation(x):
    """Compute the circular autocorrelation of binary vector x."""
    n = len(x)
    # Use FFT convolve
    x_fft = fft(x, 2 * n)
    conj_x_fft = np.conj(x_fft)
    corr = ifft(x_fft * conj_x_fft)
    return corr.real[:n]

def round_and_evaluate(X, n, t_star):
    """Rounds SDP solution X to binary vector and evaluates."""
    # Compute leading eigenvector using SVD for numerical stability
    try:
        U, s, Vt = np.linalg.svd(X, full_matrices=False)
        v = U[:, 0] * np.sqrt(s[0])
    except:
        # Fallback to direct eigenvalue decomposition
        try:
            eigvals, eigvecs = np.linalg.eigh(X)
            v = eigvecs[:, -1] * np.sqrt(eigvals[-1])
        except:
            # Return deterministic thresholding if anything goes wrong
            return np.array(v > 0.5, dtype=int)
    
    # Random hyperplane rounding (try multiple)
    best_min_corr = -1
    best_x = None
    
    R = 30
    for _ in range(R):
        g = np.random.randn(n)
        z = Vt.T @ (g * np.sqrt(s))
        # Sort indices by z values in descending order and select top n/2
        indices = np.argsort(z)[::-1][:n//2]
        x_candidate = np.zeros(n)
        x_candidate[indices] = 1
        
        # Evaluate min autocorrelation
        ac = compute_autocorrelation(x_candidate)
        min_ac = np.min(ac[1:])  # exclude k=0
        
        if min_ac > best_min_corr:
            best_min_corr = min_ac
            best_x = x_candidate
    
    # Also consider deterministic thresholding (backup)  
    if best_x is None:
        indices = np.argsort(v)[::-1][:n//2]
        best_x = np.zeros(n)
        best_x[indices] = 1
    else:
        # Ensure we got a solution
        best_x = np.array(best_x, dtype=int)
    
    # Re-evaluate with final exact value
    final_ac = compute_autocorrelation(best_x)
    final_min_ac = np.min(final_ac[1:])
    if final_min_ac > t_star:
        t_star = final_min_ac
    
    return best_x

def greedy_swap_refinement(h_binary, max_iter=5000):
    """Perform swap-based refinement of binary vector for maximum minimum autocorrelation."""
    n = len(h_binary)
    dx = 2.0 / n
    
    # Compute initial autocorrelation vector
    def compute_autocorr(h):
        h_fft = fft(h, 2 * n)
        ih_fft = fft(1 - h, 2 * n)
        corr = ifft(h_fft * np.conj(ih_fft))
        return corr.real[:n] * dx
    
    # Initial autocorrelations (only for non-zero shifts)
    def get_autocorrelogram(h):
        ac = compute_autocorr(h)
        return ac
    
    R = get_autocorrelogram(h_binary)
    
    # Track worst shift and corresponding values
    def find_worst_shift_and_contribs(h, R):
        k_star = np.argmin(R[1:]) + 1  # Exclude shift 0
        # Compute contribution of ones to that shift
        c = np.zeros(n)
        for i in range(n):
            c[i] = h[(i + k_star) % n] + h[(i - k_star) % n]
        # Compute contribution of zeros to that shift
        d = np.zeros(n)
        for i in range(n):
            d[i] = h[(i + k_star) % n] + h[(i - k_star) % n]
        return k_star, c, d
    
    def adjust_autocorrelations(R_old, h_old, h_new, k_star):
        # Recalculate all correlations
        # Optimized: handle only changes
        R_new = np.copy(R_old)
        # Find positions that changed
        changed_ones = np.where((h_old != h_new) & (h_old == 1))[0]
        changed_zeros = np.where((h_old != h_new) & (h_old == 0))[0]
        
        # Efficiency: just recalculate manually for affected shifts
        # Here we simply recompute for all shifts.
        ac = ifft(fft(h_new, 2 * n) * np.conj(fft(1 - h_new, 2 * n)))
        R_new = ac.real[:n] * dx
        return R_new
    
    # Small efficiency optimization: work with binary only
    h = h_binary.copy()
    if np.sum(h) != n // 2:
        # Normalize reasonably
        h = project_to_feasible(h, n)
        h = h_binary.copy()  # Or compute true support of h
        h = np.array(h > 0.5, dtype=int)
    
    iter_count = 0
    while iter_count < max_iter:
        iter_count += 1
        # Recompute full autocorrelation 
        old_R = get_autocorrelogram(h)
        k_star, c, d = find_worst_shift_and_contribs(h, old_R)
        
        # Find best swap: 1 with max c, 0 with min d
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        best_1 = ones[np.argmax(c[ones])]
        best_0 = zeros[np.argmin(d[zeros])]
        
        if d[best_0] > c[best_1]:
            # Perform the swap
            h[best_1] = 0
            h[best_0] = 1
            
            new_R = get_autocorrelogram(h)
            # Check if improvement in worst
            old_worst = np.min(old_R[1:])
            new_worst = np.min(new_R[1:])
            
        else:
            break  # No improvement, stop
    
    return h

def run(seed=42, budget_s=600, **kwargs):

    budget_s_left = budget_s
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    
    # Set initial h from previous solution (if available)
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) != n_points:

        else:
            h_start = orig_h * (n_points / 2) / np.sum(orig_h)
    else:
        h_start = np.ones(n_points) * 0.5
    
    h_start = project_to_feasible(h_start, n_points)
    
    # Handle memory/time budget: reduce n if needed
    while n_points > 100:
        # Estimate memory is acceptable, but if we just started, maybe reduce it
        break  # keep going with 200
    
    # Phase 1: SDP relaxation
    print("Starting SDP relaxation...", time.time() - start_time)
    
    # Binary variables as 0/1 in {0,1}^n
    # x_i = 1 means h_i = 1, etc.
    try:
        X = cp.Variable((n_points, n_points), symmetric=True)
        t = cp.Variable()
        
        # Linear constraints
        constraints = [
            cp.diag(X) == 0.5 * np.ones(n_points),  # Diagonal entries should be 0.5 for x = 1 (safe assumption)
            X >> 0,  # PSD constraint
            t <= 0.5  # bounded initially. Higher t means more effective correlation
        ]
        
        # Add additional bounding constraints for X_{ij}
        for i in range(n_points):
            for j in range(n_points):
                if i != j:
                    constraints.append(X[i, j] >= 0)
                    constraints.append(X[i, j] <= X[i,i])
                    constraints.append(X[i, j] <= X[j,j])
        
        # Autocorrelation constraints
        # For k = 1 to n-1, trace(A_k * X) >= t
        for k in range(1, n_points):
            A_k = np.zeros((n_points, n_points))
            for i in range(n_points):
                A_k[i, (i + k) % n_points] = 1
            # Trace is sum over diagonal elements
            constraints.append(cp.trace(A_k @ X) >= t)
        
        # Objective
        prob = cp.Problem(cp.Maximize(t), constraints)
        
        # Solve with ECOS or SCS, shorter time budget allocation for SDP
        budget_s_left = budget_s * 0.3
        start_solve = time.time()
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=500)
        end_solve = time.time()
        
        t_star = t.value if t.value is not None else 0
        X_star = X.value if X.value is not None else np.eye(n_points)
        
        print("SDP solved in {:.3f}s. t_star = {:.6f}".format(end_solve - start_solve, t_star))
        
        # Round SDP to binary vector
        print("Rounding solution...")
        start_round = time.time()
        
        h_rounded = round_and_evaluate(X_star, n_points, t_star)
        h_rounded = np.array(h_rounded, dtype=float)
        h_rounded = project_to_feasible(h_rounded, n_points)
        
        print("Rounding done in {:.3f}s".format(time.time() - start_round))
        
        # Phase 2: Greedy swap refinement
        print("Greedy swap refinement...")
        start_refine = time.time()
        
        h_final = greedy_swap_refinement(h_rounded, 1000)
        
        print("Refinement took {:.3f}s".format(time.time() - start_refine))
        
        # Evaluate final result
        c5_bound = evaluate_erdos_solution(h_final)
        h_values = h_final
        
        print("Final c5:", c5_bound)
        
    except Exception as e:
        print("Error in SDP phase:", str(e))
        # Fallback to simple random search with swap refinement
        print("Using fallback to random search")
        best_c5 = float('inf')
        for trial in range(100):
            h_trial = np.random.rand(n_points)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            h_trial = project_to_feasible(h_trial, n_points)
            h_refined = greedy_swap_refinement(h_trial, 1000)
            c5_trial = evaluate_erdos_solution(h_refined)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                h_values = h_refined
        
        c5_bound = best_c5
        
    # Final polishing step using continuous gradient descent
    try:
        def continuous_polish(h):
            h = project_to_feasible(h, n_points)
            max_iter = 30  # few iterations only
            dx = 2.0 / n_points
            for i in range(max_iter):
                # Find worst shift (i*k) for k=k* where we wanted largest overlap
                corr = ifft(fft(h, 2 * n_points) * np.conj(fft(1 - h, 2 * n_points)))
                corr_vals = corr.real[1:n_points//2]
                k_star = np.argmax(corr_vals) + 1
                
                # Compute gradient w.r.t. k_star
                grad = np.zeros_like(h)
                for j in range(n_points):
                    l_shift = (j + k_star) % n_points
                    r_shift = (j - k_star) % n_points
                    grad[j] = dx * (1 - h[l_shift]) - h[r_shift]
                
                # Step
                alpha = 0.5 * dx / np.sqrt(i + 1)
                h = h - alpha * grad
                
                # Project back
                h = project_to_feasible(h, n_points)
                
                # Update if better
                new_c5 = evaluate_erdos_solution(h)
                if new_c5 < c5_bound:
                    c5_bound = new_c5
                    h_values = h.copy()
            return h_values, c5_bound
        
        h_values, c5_bound = continuous_polish(h_values)
    except:
        pass
    
    return (h_values, c5_bound, n_points)
