def full_to_sym(h_full):
    """Convert full vector to symmetric (half + center) representation."""
    n = len(h_full)
    if n % 2 == 0:
        return np.concatenate([h_full[:n//2], h_full[n//2-1::-1]])
    else:
        return np.concatenate([h_full[:(n+1)//2], h_full[(n-1)//2::-1]])

def sym_to_full(h_sym):
    """Convert symmetric vector back to full."""
    n = len(h_sym)
    if n % 2 == 0:
        return np.concatenate([h_sym[:n//2], h_sym[n//2-1::-1]])
    else:
        return np.concatenate([h_sym[:(n+1)//2], h_sym[(n-1)//2::-1]])

def compute_correlation_vals(h_full):
    """Compute the full linear correlation as a function of shift."""
    n = len(h_full)
    dx = 2.0 / n
    # compute correlations manually for culprit point now (should be okay with the full loop)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    corr = corr[n-1:]  # take non-negative shift part
    corr = corr * dx
    return corr

def solve_lp(shift_set, h_initial, n_full, x0=None):
    dx = 2.0 / n_full
    n_sym = n_full // 2 + (1 if n_full % 2 else 0)
    # Variable
    x = cp.Variable(n_sym)
    t = cp.Variable()
    # Objective
    objective = cp.Minimize(t)
    # Constraints
    constraints = []
    # Bounds on x
    constraints += [0 <= x, x <= 1]
    # Integral constraint:
    # We need sum(h) * dx == 1, i.e., sum(h) = n/2
    # sym_to_full maps x to full h; 
    # so sum_full = sum(sym_to_full(x)) == n/2   => ~ 2*sum(x[:n//2]) + x[n//2] == n/2 for odd length
    # But simplifying: Let’s build integral from x variable directly:
    h_full = sym_to_full(x)
    integral_constr = cp.sum(h_full) * dx == 1.
    constraints.append(integral_constr)
    
    # For shifts:
    # For every shift s in shift_set, we have expression sum_i x[i] * (1 - x[i+s]) <= t
    # We will try to only work with full form (for now), delaying the exact buildup
    # This creates a very hard-to-compose method so calling simulate underlining below
    # This will build proper index constraints manually for performance.
    
    # we need to simplify shifting in the symmetric space means.
    
    # Build constraints iteratively:
    for s in shift_set:
        expr = 0.0
        for i in range(len(x)):
            j = i + s
            if j < len(x):
                # actually focus on safe indexing for indexing: i,j are in x index range.
                term = x[i] * (1 - x[j])
                expr += term
            # skip if out of bound (not a problem in this case)

        constraints.append(expr <= t)
    
    # Solve the problem.
    prob = cp.Problem(objective, constraints)  
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=200)
        x_sol = x.value
        t_sol = t.value
        if x_sol is not None:
            # return both full form and t
            h_sol = sym_to_full(x_sol)
            return h_sol, t_sol, True
    except Exception:
        # fallback to a random initial starting point evaluation
        pass
    # default: return initial estimate
    h_ft = sym_to_full(h_initial)
    return h_ft, np.max(compute_correlation_vals(h_ft)), False

def project_to_feasible(h, n):
    """Ensure sum(h) == n/2 and h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n / 2
    diff = target_sum - current_sum

    if abs(diff) < 1e-10:
        return h

    if diff > 0:
        # Need to increase sum
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    return np.clip(h_copy, 0, 1)
