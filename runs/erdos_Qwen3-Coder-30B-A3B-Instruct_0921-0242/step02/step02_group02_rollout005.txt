import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.interpolate import interp1d

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Use FFT for efficient computation of correlations
    H = rfft(h)
    auto = irfft(np.abs(H)**2, len(h))
    c = dx * (len(h)/2 - auto)
    return c.max()

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def build_cyclic_shift_matrix(k, n):
    """Build the cyclic shift matrix P_k."""
    P = np.zeros((n, n))
    for i in range(n):
        j = (i - k) % n
        P[i, j] = 1
    return P

def subgradient_refine(h, dx, budget_s):
    """Perform subgradient descent refinement."""
    start = time.time()
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    iterations = 0
    max_iter = 10000  # Maximum iterations to prevent infinite loop
    
    while (time.time() - start < budget_s and 
           iterations < max_iter):
        # Compute overlaps
        H = rfft(current_h)
        auto = irfft(np.abs(H)**2, len(current_h))
        c = dx * (len(current_h)/2 - auto)
        k_star = np.argmax(c)
        
        # Compute gradient
        h_k = np.roll(current_h, k_star)
        h_negk = np.roll(current_h, -k_star)
        grad = dx * (h_k - h_negk)
        
        # Line search
        alpha = 1.0
        step_size = 0.01
        for _ in range(20):
            trial_h = current_h - alpha * grad
            trial_h = project_to_feasible(trial_h, len(trial_h))
            trial_c5 = evaluate_c5(trial_h)
            if trial_c5 < current_c5:
                current_h = trial_h
                current_c5 = trial_c5
                break
            alpha *= 0.5
            
        iterations += 1
        
    return current_h, current_c5

def upsample_and_project(h_coarse, n_new):
    """Upsample by interpolation and project."""
    n_old = len(h_coarse)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h_coarse, kind='linear', fill_value='extrapolate')
    h_up = f(x_new)
    h_up = project_to_feasible(h_up, n_new)
    return h_up

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: SDP (n=180)
    n = 180
    dx = 2.0 / n
    P = [build_cyclic_shift_matrix(k, n) for k in range(n)]
    
    # SDP formulation
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()
    
    constraints = [
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n/2,
    ] + [cp.trace(P[k] @ X) >= t for k in range(n)]
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Solve SDP with reasonable time limit
    prob.solve(solver='SCS', max_iters=25000, eps=1e-5)
    
    # Extract candidate
    X_opt = X.value
    eigvals, eigvecs = np.linalg.eigh(X_opt)
    lead = eigvecs[:, -1] * np.sqrt(eigvals[-1])
    h0 = project_to_feasible(lead, n)
    
    # Keep best of random draws
    best_h = h0
    best_c5 = evaluate_c5(h0)
    
    for _ in range(15):
        if time.time() - start_time > 0.9 * budget_s:
            break
        g = np.random.randn(n) @ np.linalg.cholesky(X_opt)
        h = project_to_feasible(np.clip(g, 0, 1), n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 2: Sub-gradient refinement
    remaining_time = 0.5 * budget_s
    best_h, best_c5 = subgradient_refine(best_h, dx, remaining_time)
    
    # Phase 3: Upsampling if time allows
    if time.time() - start_time < 0.85 * budget_s:
        n2 = 2 * n
        h_up = upsample_and_project(best_h, n2)
        best_h, best_c5 = subgradient_refine(h_up, 2.0/n2, remaining_time)
    
    return (best_h, best_c5, len(best_h))
