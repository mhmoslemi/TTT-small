import numpy as np
import time
import math

def project_box_sum(y, total):
    """Project y onto box [0,1] intersected with sum = total."""
    y = np.asarray(y)
    low, high = 0.0, 1.0
    # bisection to find lambda
    l, u = -100.0, 100.0
    for _ in range(40):
        lam = (l + u) / 2.0
        x = np.clip(y - lam, low, high)
        s = np.sum(x)
        if s < total:
            u = lam
        else:
            l = lam
    lam = (l + u) / 2.0
    x = np.clip(y - lam, low, high)
    return x

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def softmax_weights(C, beta):
    """Compute softmax weights for a vector C with temperature beta."""
    C = np.asarray(C)
    C_shifted = C - np.max(C)
    exp_C = np.exp(beta * C_shifted)
    return exp_C / np.sum(exp_C)

def gradient_from_weights(h, w):
    """Compute gradient from softmax weights w."""
    n = len(h)
    # Compute convolutions: w * h and w[::-1] * h
    # w vector is length 2n - 1
    w_full = np.zeros(2 * n - 1)
    w_full[n-1:n-1+len(w)] = w
    # Forward convolution: w * h
    conv1 = np.convolve(w_full, h, mode='full')
    # Backward convolution: w[::-1] * h
    conv2 = np.convolve(w_full[::-1], h, mode='full')
    # Extract central slice
    grad = - (2.0 / n) * (conv1[n-1:n-1+n] + conv2[n-1:n-1+n])
    return grad
