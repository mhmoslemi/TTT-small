import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Project h to feasible set: 0 <= h_i <= 1, sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Binary search for lambda to shift
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n / 2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def evaluate_autocorr(h, k):
    """Evaluate the autocorrelation at shift k."""
    n = len(h)
    return np.sum(h * np.roll(h, k))

def smooth_boundaries(h_bin, ramp_width=2):
    """Create a continuous step function from binary with fractional ramps."""
    n = len(h_bin)
    h_cont = h_bin.astype(float)
    
    # Find transition indices
    transitions = []
    for i in range(n):
        if h_bin[i] != h_bin[(i+1) % n]:
            transitions.append(i)
    
    # Insert ramps at transitions
    for i in transitions:
        # If 0->1 transition
        if h_bin[i] == 0 and h_bin[(i+1) % n] == 1:
            start = i + 1
            for r in range(ramp_width):
                j = (start + r) % n
                h_cont[j] = (r + 1) / (ramp_width + 1)
        elif h_bin[i] == 1 and h_bin[(i+1) % n] == 0:
            start = i - ramp_width + 1
            for r in range(ramp_width):
                j = (start + r) % n
                h_cont[j] = (ramp_width - r) / (ramp_width + 1)
    
    return project_to_feasible(h_cont, n)

def compute_subgradient(h, k_star):
    """Compute subgradient for shift k_star."""
    n = len(h)
    dx = 2.0 / n
    g = -dx * (np.roll(h, k_star) + np.roll(h, -k_star))
    return g

def backtracking_line_search(h, g, max_iter=10):
    """Backtracking line search for step size."""
    alpha = 1.0
    dx = 2.0 / len(h)
    current_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        h_new = h + alpha * g
        h_new = project_to_feasible(h_new, len(h))
        new_c5 = evaluate_c5(h_new)
        if new_c5 < current_c5:
            return h_new, alpha
        alpha /= 2.0
    return h, alpha

def proj_grad_refine(h_start, max_time_s):
    """Project gradient refinement of step function."""
    start_time = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h_start)
    
    iteration = 0
    while time.time() - start_time < max_time_s:
        if iteration % 100 == 0:
            # Recheck if still valid
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Compute k_star and gradient
        corr = correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        if k_star >= len(h):
            k_star = k_star - len(h)
        else:
            k_star = k_star - len(h) + 1
            
        g = compute_subgradient(h, k_star)
        h_new, step_size = backtracking_line_search(h, g)
        h = h_new
        
        iteration += 1

    c5 = evaluate_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    return best_h, best_c5

def binary_anneal(n, max_time_s):
    """Binary simulated annealing for the autocorrelation maximization."""
    dx = 2.0 / n
    start_time = time.time()

    # Initialize binary array
    h = np.zeros(n)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    h[ones_indices] = 1.0
    
    # Initial autocorrelations
    H = fft(h, n)
    ac = np.fft.irfft(H * np.conj(H), n)
    ac = np.real_if_close(ac)
    
    c = dx * (n/2 - ac)
    current_c5 = np.max(c)
    
    best_h = h.copy()
    best_c5 = current_c5
    
    T = 5.0 * dx  # initial temperature
    T_min = dx * 1e-4
    alpha = 0.9997
    swap_count = 0
    
    while time.time() - start_time < max_time_s:
        # Sample two different indices
        i = np.random.randint(0, n)
        while h[i] == 0:
            i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        while h[j] == 1:
            j = np.random.randint(0, n)
        
        # Propose swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1

        # Compute changes in autocorrelation at all shifts
        delta = np.zeros(n)
        delta = delta - h_new[i] + h[i]
        delta = delta - h_new[j] + h[j]
        delta[i] += 2  # self term
        delta[j] += 2
        if i != j:
            delta[(i - j) % n] += -1
            delta[(j - i) % n] += -1
        delta = np.roll(delta, -j)  # compensate for circular shift
        delta = np.roll(delta, j)
        
        # Compute new autocorrelations
        ac_new = ac + delta
        c_new = dx * (n/2 - ac_new)
        new_c5 = np.max(c_new)
        
        # Accept or reject
        if new_c5 <= current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new
            ac = ac_new
            c = c_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        swap_count += 1
        if swap_count % 1000 == 0:
            T = max(T * alpha, T_min)
        
        if swap_count % 200000 == 0:
            # Restart: reassign 5% of ones to random locations
            n_ones = int(n / 2)
            old_ones = np.where(h == 1)[0]
            new_ones = np.random.choice(n, size=int(0.05 * n_ones), replace=False)
            h = np.zeros(n)
            keep_ones = np.setdiff1d(old_ones, new_ones, assume_unique=True)
            h[keep_ones] = 1
            h[new_ones] = 1

    return best_h, best_c5

def upsample_and_refine(h_start, n_new, max_time_s):
    """Upsample step function."""
    start_time = time.time()
    h = h_start.copy()
    dx = 2.0 / len(h)
    dx_new = 2.0 / n_new
    h_upsampled = np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)
    h_upsampled = project_to_feasible(h_upsampled, n_new)
    
    # Projected gradient refinement
    h_refined, c5_refined = proj_grad_refine(h_upsampled, max_time_s)
    
    return h_refined, c5_refined

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n0 = 256
    dx0 = 2.0 / n0

    # Stage 1: Binary simulated annealing
    h_bin, c5_bin = binary_anneal(n0, 0.55 * budget_s)
    
    # Stage 2: Boundary smoothing and gradient refinement
    h_cont = smooth_boundaries(h_bin, ramp_width=2)
    h_refined, c5_refined = proj_grad_refine(h_cont, 0.25 * budget_s)
    
    # Stage 3: Progressive up-sampling
    n1 = 2 * n0
    h_upsampled, c5_upsampled = upsample_and_refine(c5_refined, n1, 0.15 * budget_s)

    return (h_upsampled, c5_upsampled, n1)
