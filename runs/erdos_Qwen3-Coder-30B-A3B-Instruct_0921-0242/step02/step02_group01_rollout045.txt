import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region: sum(h) = target_sum, 0 <= h[i] <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def gradient_worst_shift(h, worst_shift):
    """Compute gradient with respect to worst shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    
    # Since correlation uses circular shift, we take h_{i+k} and h_{i-k}
    # for k = worst_shift
    for i in range(n):
        left_idx = (i + worst_shift) % n
        right_idx = (i - worst_shift) % n
        grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
    
    return grad

def gradient_worst_shift_descent(h, max_iter=200, alpha0=0.1):
    """Perform gradient descent on the worst shift."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for t in range(max_iter):
        # Compute all correlations
        h_fft = fft(h, 2 * n)
        ih_fft = fft(1 - h, 2 * n)
        corr = ifft(h_fft * np.conj(ih_fft))
        corr_vals = corr.real[1:n]  # Ignore shift 0
        if len(corr_vals) == 0:
            break
        worst_shift = np.argmax(corr_vals) + 1
        
        # Compute gradient w.r.t. worst shift
        grad = gradient_worst_shift(h, worst_shift)
        
        # Elbow method: reducing step size based on iteration
        alpha = alpha0 / np.sqrt(t + 1)
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n/2)
        
        # Stopping condition
        if t > 10 and np.linalg.norm(h_new - h) < 1e-7:
            break
            
        h = h_new
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 400
    target_sum = n_points / 2
    pop_size = min(40, 2 * n_points // 10)
    max_generations = 4000
    F = 0.6
    CR = 0.9
    
    # Initial warmup
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
    else:
        orig_h = np.ones(300) * 0.5
    
    # Use a moderate size
    if len(orig_h) > n_points:
        h_start = orig_h[:n_points]
    else:
        x_orig = np.linspace(0, 2, len(orig_h))
        x_new = np.linspace(0, 2, n_points)
        h_start = np.interp(x_new, x_orig, orig_h)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = project_to_feasible(h_start, n_points)
    
    # Population initialization
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, target_sum)
        population.append(h)
        
    population[0] = h_start  # Place the warmup at the first position
    
    # Track best solution
    global_best = None
    global_best_c5 = float('inf')
    
    # DE main loop
    gen = 0
    last_improvement = 0
    
    while gen < max_generations and time.time() - start_time < budget_s * 0.75:
        for i in range(pop_size):
            # DE mutation
            idx = [j for j in range(pop_size) if j != i]
            a, b, c = np.random.choice(idx, 3, replace=False)
            donor = population[a] + F * (population[b] - population[c])
            donor = np.clip(donor, 0, 1)
            donor = project_to_feasible(donor, target_sum)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, donor, population[i])
            
            # Evaluate new solution
            c5_new = evaluate_c5(trial)
            c5_old = evaluate_c5(population[i])
            
            if c5_new < c5_old:
                population[i] = trial
                if c5_new < global_best_c5:
                    global_best_c5 = c5_new
                    global_best = trial.copy()
                    last_improvement = time.time()
        
        # Every 200 generations, run local refinement on the best
        if gen % 200 == 0 and global_best is not None:
            if time.time() - start_time > budget_s * 0.75:
                break
            ref = gradient_worst_shift_descent(global_best, max_iter=100)
            c5_ref = evaluate_c5(ref)
            if c5_ref < global_best_c5:
                global_best = ref.copy()
                global_best_c5 = c5_ref
        
        gen += 1
    
    # If nothing found, use warmup
    if global_best is None:
        global_best = h_start
        global_best_c5 = evaluate_c5(global_best)
    
    # Memory intensive polishing after DE
    final_h = global_best.copy()
    final_c5 = global_best_c5
    
    # Local refinement (deterministic)
    if time.time() - start_time < budget_s * 0.9:
        ref_h = gradient_worst_shift_descent(final_h, max_iter=200)
        ref_c5 = evaluate_c5(ref_h)
        if ref_c5 < final_c5:
            final_h = ref_h
            final_c5 = ref_c5
    
    # Stochastic fine tuning (simulated annealing style)
    if time.time() - start_time < budget_s * 0.995:
        current_h = final_h.copy()
        current_c5 = final_c5
        temp = 0.5
        decay = 0.999
        max_steps = 5000
        for _ in range(max_steps):
            if time.time() - start_time > budget_s * 0.995:
                break
            # Pick two random indices
            i, j = np.random.randint(0, n_points, 2)
            if i == j:
                continue
            # Try a small transfer of mass
            delta = np.random.uniform(0.005, 0.05)
            if current_h[i] < delta or current_h[j] > 1 - delta:
                continue
            # Try to transfer
            h_test = current_h.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, target_sum)
            c5_test = evaluate_c5(h_test)
            
            # Accept based on Metropolis criterion
            if c5_test < current_c5 or np.random.rand() < np.exp(-(c5_test - current_c5) / temp):
                current_h = h_test
                current_c5 = c5_test
            
            temp *= decay
            
        final_h = current_h
        final_c5 = current_c5

    # Final check with 800 points to ensure discrete effect doesn't affect value
    # Don't spend too much time here
    if time.time() - start_time < budget_s * 0.999:
        h_fine = final_h.copy()
        n_fine = 800
        dx_fine = 2.0 / n_fine
        # Interpolate to finer grid
        x_orig = np.linspace(0, 2, n_points)
        x_fine = np.linspace(0, 2, n_fine)
        h_fine = np.interp(x_fine, x_orig, h_fine)
        h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
        h_fine = project_to_feasible(h_fine, n_fine / 2)
        
        # Local descent on finer grid
        h_refined = gradient_worst_shift_descent(h_fine, max_iter=50)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < final_c5:
            final_h = h_refined
            final_c5 = c5_refined

    return (final_h, final_c5, n_points)
