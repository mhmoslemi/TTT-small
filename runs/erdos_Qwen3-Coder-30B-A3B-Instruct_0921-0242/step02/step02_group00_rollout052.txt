import numpy as np
import time
import math

def project_box_sum(y, total):
    """Project y onto the set {x : 0 <= x[i] <= 1, sum(x) = total}."""
    # bisection to find lambda
    low, high = -1.0, 1.0
    eps = 1e-12
    for _ in range(100):  # At most 100 iterations for double precision
        mid = (low + high) / 2.0
        x = np.clip(y - mid, 0, 1)
        s = np.sum(x)
        if s > total:
            low = mid
        else:
            high = mid
    lambda_opt = (low + high) / 2.0
    x = np.clip(y - lambda_opt, 0, 1)
    return x

def softmax_weights(C, beta):
    """Compute softmax weights with temperature beta."""
    C_max = np.max(C)
    exp_C = np.exp(beta * (C - C_max))
    return exp_C / np.sum(exp_C)

def gradient_from_weights(h, w):
    """Compute gradient given weights w."""
    n = len(h)
    # w has 2n-1 elements (full convolution mode)
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    # Extract the valid part
    grad = -(2.0 / n) * (conv1[n-1:n-1+n] + conv2[n-1:n-1+n])
    return grad

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

        # Inner loop for fixed beta

            gamma = np.correlate(h, h, mode='full')

            alpha = step_init
