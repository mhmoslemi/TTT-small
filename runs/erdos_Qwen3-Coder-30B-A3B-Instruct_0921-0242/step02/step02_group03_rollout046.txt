import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def full_to_sym(h_full):
    """Convert full array to symmetric representation (first half + center)."""
    n = len(h_full)
    if n % 2 == 0:
        half_n = n // 2
        return np.concatenate([h_full[:half_n], h_full[half_n-1::-1]])
    else:
        half_n = n // 2
        return np.concatenate([h_full[:half_n+1], h_full[half_n-1::-1]])

def sym_to_full(h_sym):
    """Convert symmetric representation back to full array."""
    n = len(h_sym)
    if n % 2 == 0:
        half_n = n // 2
        return np.concatenate([h_sym[:half_n], h_sym[half_n-1::-1]])
    else:
        half_n = n // 2
        return np.concatenate([h_sym[:half_n+1], h_sym[half_n-1::-1]])

def build_constraints(x, t, shift_set, n):
    """Build constraints for the LP involving correlations."""
    constraints = []
    
    # Integral constraint
    dx = 2.0 / n
    constraints.append(cp.sum(x) == n/2)

    # Non-negativity
    constraints.append(x >= 0)
    constraints.append(x <= 1)

    # For each shift, construct the constraint
    for s in shift_set:
        if s == 0:
            # This is equivalent to sum(h_i * (1 - h_i)) = sum(h_i) - sum(h_i^2)
            expr = cp.sum(x) - cp.sum(cp.multiply(x, x))
            constraints.append(expr * dx <= t)
        else:
            # We need to loop over i and form the sum over valid indices
            # x is the symmetric representation
            # Need h[i] * (1 - h[i+s]) = x[i] * (1 - x[i+s]) where i+s is valid
            # But we have to map indices appropriately
            # e.g., x[i] corresponds to h[i] in the full array
            # So the shifted values are x[i+s] where valid
            
            # Let's define the full vector h from x, then compute correlation
            h = sym_to_full(x)
            h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
            one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
            corr_i = np.zeros_like(h)
            corr_i = one_minus_h_padded[s:s+n] * h_padded[:n]

            expr = cp.sum(corr_i) * dx
            constraints.append(expr <= t)

    return constraints

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    # Take only the non-cyclic correlations (correlations at shifts 0..n-1)
    non_cyclic = corr[len(h)-1:]
    max_corr = np.max(non_cyclic) * dx
    return max_corr

def solve_lp(shift_set, h0, n):
    """Solve the LP for given shifts."""
    x = cp.Variable(n//2 + 1 if n % 2 == 1 else n//2, name='x')
    t = cp.Variable(1, name='t')

    constraints = build_constraints(x, t, shift_set, n)

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_val = x.value
            t_val = t.value
            h_full = sym_to_full(x_val)
            return h_full, t_val
        else:
            return None, None
    except Exception:
        return None, None
