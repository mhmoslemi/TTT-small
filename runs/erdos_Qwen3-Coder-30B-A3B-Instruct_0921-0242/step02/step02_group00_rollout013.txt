import numpy as np
import time
import math

def project_box_sum(y, total):
    """Project y onto the set {x : 0 <= x[i] <= 1, sum(x) = total}"""
    y = np.asarray(y)
    lo, hi = 0.0, 1.0
    a, b = lo, hi
    
    def S(lam):
        return np.sum(np.clip(y - lam, lo, hi))
    
    # Binary search for lambda
    for _ in range(100):
        lam = (a + b) / 2
        s = S(lam)
        if s > total:
            a = lam
        elif s < total:
            b = lam
        else:
            break
    return np.clip(y - lam, lo, hi)

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def softmax_weights(C, beta):
    """Compute softmax weights corresponding to C with temperature beta"""
    C = np.asarray(C)
    C_shifted = C - np.max(C)
    exp_C = np.exp(beta * C_shifted)
    return exp_C / np.sum(exp_C)

def gradient_from_weights(h, w):
    """Compute gradient from weights w corresponding to h."""
    n = len(h)
    # two convolutions: w * h and w[::-1] * h
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    # slice to get the middle portion for the gradient
    gradient = -(2.0 / n) * (conv1[n-1:n-1+n] + conv2[n-1:n-1+n])
    return gradient

                step_size = min(step_size * 1.1, 2.0)
            else:
                step_size *= 0.9

        if it - last_improvement > 500 and (time.time() - start_time) < time_remaining:
