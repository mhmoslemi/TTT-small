import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, target_sum):
    """Ensure that sum(h) = target_sum, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def generate_seeds(n_points, target_sum):
    """Generate various starting seeds."""
    seeds = []
    
    # 1. Uniform
    seeds.append(np.ones(n_points) * 0.5)
    
    # 2. Random Dirichlet (Dirichlet(1,...,1) is uniform on simplex)
    seeds.append(np.random.dirichlet(np.ones(n_points)) * target_sum)
    
    # 3. Binary + smoothing
    binary = np.zeros(n_points)
    binary[:n_points // 2] = 1
    np.random.shuffle(binary)
    # Add small noise and re-project
    binary = binary + np.random.normal(0, 0.01, n_points)
    seeds.append(project_to_feasible(binary, target_sum))
    
    # 4. Interpolated initial_h_values (if available)
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) != n_points:
            x_orig = np.linspace(0, 2, len(orig_h))
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_orig, orig_h)
        else:
            h_interp = orig_h.copy()
        seeds.append(project_to_feasible(h_interp, target_sum))
    
    # 5. Sobol-like quasi-random (simplified version)
    sobol_like = np.random.rand(n_points)
    seeds.append(project_to_feasible(sobol_like, target_sum))
    
    return seeds

def build_lp_data(h0):
    """Build LP data from h0: A*h + 1*t <= b, where h and t are variables."""
    n = len(h0)
    dx = 2.0 / n
    # Compute auto-correlations (non-circular)
    # raw correlogram with all lags
    c0 = np.correlate(h0, h0, mode='full')
    c0 = c0[len(c0)//2:]  # only non-negative lags (0 to n-1)
    
    # Build the matrix W for LP constraints
    W = np.zeros((n, n))
    for k in range(n):
        W[k, :] = h0[(np.arange(n) + k) % n] + h0[(np.arange(n) - k) % n]
    A = -W
    b = -n/2 + c0
    
    return (A, b, c0)

def solve_lp(A, b, n):
    """Solve the LP via scipy -- use highs method as it's efficient"""
    # We solve:
    # minimize t
    # subject to Ax + t*1 <= b
    #             0 <= x <= 1
    #             sum x = n/2
    #             t >= 0
    
    # Variables are [h[0], ..., h[n-1], t]
    # Constraints:
    # -[W][h] + [1] * t <= b
    # 0 <= h <= 1
    # sum(h) = n/2
    # 0 <= t
    
    # Rearrange as:
    # [W, -1] * [h; t] <= b
    # [1, 0] * [h; t] = n/2
    # [I, 0] * [h; t] >= 0
    # [-I, 0] * [h; t] <= 1
    
    # As per 'highs' format excised, build explicit bounds directly:
    A_ub = np.column_stack([A, -np.ones(A.shape[0])])
    b_ub = b.copy()
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = np.ones(n)
    b_eq = np.array([n/2])
    
    bounds = [(0, 1)] * n + [(0, None)];  # h in [0,1], t >= 0
    
    res = linprog(c=[0]*n + [1], 
                  A_ub=A_ub, 
                  b_ub=b_ub, 
                  A_eq=A_eq, 
                  b_eq=b_eq,
                  bounds=bounds, 
                  method='highs')
    
    if res.success:
        h_sol = res.x[:n]
        t_sol = res.x[n]
        return h_sol, t_sol
    else:
        return None, None

def ccp_loop(h0, target_sum, max_iter=30):
    """Run the CCP (Convex Concave Procedure) relaxation loop."""
    h_current = h0.copy()
    best_c5 = float('inf')
    best_h = h_current.copy()
    
    for iter in range(max_iter):
        A, b, _ = build_lp_data(h_current)
        h_new, t_new = solve_lp(A, b, len(h_current))
        if h_new is None:
            break
        
        h_projected = project_to_feasible(h_new, target_sum)
        true_c5 = evaluate_c5(h_projected)
        
        if true_c5 < best_c5:
            best_c5 = true_c5
            best_h = h_projected.copy()
        
        # Convergence criterion
        if np.max(np.abs(h_new - h_current)) < 1e-8:
            break
        h_current = h_new.copy()
    
    return best_h, best_c5

def project_gradient_descent(h_start, c0, max_iter=200):
    """Projected gradient descent with non-linear objective."""
    h = h_start.copy()
    dx = 2.0 / len(h)
    step_size = 0.1
    
    for t in range(max_iter):
        # Compute worst shift
        corr = ifft(fft(h, 2 * len(h)) * np.conj(fft(1 - h, 2 * len(h))))
        corr_vals = corr.real[1: len(h)]  # exclude index 0
        worst_shift = np.argmin(corr_vals) + 1
        
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            left_idx = (i + worst_shift) % len(h)
            right_idx = (i - worst_shift) % len(h)
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Update with a diminishing step size
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, len(h) / 2)
        
        # Check if we improved
        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h):
            h = h_new
            step_size *= 0.95  # diminishing
        else:
            step_size *= 0.98  # still slide down
            
        if t % 50 == 0:
            if eval_c5(h) < 0.381:
                break  # early exit
    
    return h

def simulated_annealing(h_start, max_time_s, start_T=1.0, decay=0.95):
    """Simulated Annealing pair-exchange polishing."""
    h = h_start.copy()
    c5 = evaluate_c5(h)
    T = start_T
    np.random.seed(42)
    
    end_time = time.time() + max_time_s
    
    while time.time() < end_time:
        i, j = np.random.randint(0, len(h), size=2)
        
        if h[i] <= 0 or h[j] >= 1:
            continue
            
        delta = min(0.01, h[i], 1 - h[j])
        
        h_test = h.copy()
        h_test[i] -= delta
        h_test[j] += delta
        h_test = project_to_feasible(h_test, len(h) / 2)
        
        c5_test = evaluate_c5(h_test)
        
        if c5_test < c5:
            h = h_test
            c5 = c5_test
        else:
            # Accept with probability exp(-Δ/T)
            if np.random.rand() < np.exp(-(c5_test - c5) / T):
                h = h_test
                c5 = c5_test
        
        # Cool temperature
        T *= decay
    
    return h, c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start = time.time()
    
    # Parameters
    n_points = 400
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Seed generation:
    seeds = generate_seeds(n_points, target_sum)
    best_c5 = float('inf')
    best_h = None
    
    # Multi-start CCP
    for i, h0 in enumerate(seeds):
        if time.time() - start > 0.7 * budget_s:
            break
        h_final, c5 = ccp_loop(h0, target_sum, max_iter=30)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()
        
        # If we're getting very close to target, consider closer inspection
        if best_c5 < 0.3805:
            break
    
    if best_h is None:
        return np.ones(n_points) * 0.5, float('inf'), n_points
    
    # Refinement: projected gradient descent
    if time.time() - start < 0.9 * budget_s:
        best_h = project_gradient_descent(best_h, None, max_iter=200)
        best_c5 = evaluate_c5(best_h)
    
    # Polish with simulated annealing
    remaining_time = budget_s - (time.time() - start)
    if remaining_time > 5:
        best_h, best_c5 = simulated_annealing(best_h, remaining_time, start_T=1.0, decay=0.98)
    
    # Optional upscale if time allows
    if time.time() - start < budget_s - 10:
        # Upsample
        n_up = 800
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_up)
        h_up = np.interp(x_new, x_orig, best_h)
        h_up = project_to_feasible(h_up, n_up / 2)
        
        # Run short CCP loop on this
        if n_up == 800:
            A, b, c0 = build_lp_data(h_up)
            h_refined, t = solve_lp(A, b, n_up)
            if h_refined is not None:
                h_up = project_to_feasible(h_refined, n_up / 2)
                best_c5 = evaluate_c5(h_up)  # recalculate C5 with new resolution
                best_h = h_up
    
    return (best_h, best_c5, n_points)
