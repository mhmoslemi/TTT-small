import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2 with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling to adjust the sum
    # Find optimal constant lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Evaluate the C5 constant of a function."""
    dx = 2.0 / len(h)
    # FFT-based autocorrelation
    H = rfft(h, len(h))
    autocorr = irfft(H * np.conj(H), len(h))
    autocorr = np.real_if_close(autocorr)
    # Compute overlap using formula from strategy
    c = dx * (len(h) / 2 - autocorr)
    return np.max(c)

def binary_anneal(h_binary, time_limit):
    """Perform binary simulated annealing."""
    start_time = time.time()
    n = len(h_binary)
    dx = 2.0 / n
    # Autocorrelation vector
    H = rfft(h_binary, n)
    autocorr = irfft(H * np.conj(H), n)
    autocorr = np.real_if_close(autocorr)
    c = dx * (n / 2 - autocorr)  # overlap per shift k
    best_c = np.max(c)
    best_h = h_binary.copy()

    T0 = 5 * dx
    alpha = 0.9997
    T = T0
    swaps = 0
    stagnation = 0

    while time.time() - start_time < time_limit:
        # Perform sweep
        # Determine ones and zeros
        ones = np.where(h_binary == 1)[0]
        zeros = np.where(h_binary == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # Create new candidate h
        h_new = h_binary.copy()
        h_new[i] = 0
        h_new[j] = 1

        # Update autocorrelation exactly
        # This is the difference in autocorrelation due to one swap
        # Row of autocorrelation matrix
        delta = np.zeros(n)
        for k in range(n):
            # Shift by k
            idx_i = (i - k) % n
            idx_j = (j - k) % n
            delta[k] = h_binary[idx_i] - h_new[idx_i]
            delta[k] += h_binary[idx_j] - h_new[idx_j]
        delta = delta * (h_binary[i] - h_new[i])
        delta[0] += 2  # autocorr shift 0
        # Update
        autocorr_new = autocorr + delta
        c_new = dx * (n / 2 - autocorr_new)
        c_new_max = np.max(c_new)

        # Accept or reject
        if c_new_max <= best_c or np.random.rand() < np.exp(-(c_new_max - best_c) / T):
            h_binary = h_new.copy()
            autocorr = autocorr_new.copy()
            c = c_new.copy()
            if c_new_max < best_c:
                best_c = c_new_max
                best_h = h_binary.copy()
                stagnation = 0
            else:
                stagnation += 1

        # Update temperature
        swaps += 1
        if swaps % 1000 == 0:
            T = max(T0 * (alpha ** (swaps / 1000)), dx * 1e-4)

        # Restart large jump if stagnation
        if stagnation > 200000:
            h_binary = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
            # Normalize
            ones = np.sum(h_binary)
            if ones == n / 2:
                pass  # good
            elif ones < n / 2:
                # Fill by setting zeros to ones
                zero_idx = np.where(h_binary == 0)[0]
                zero_idx = np.random.choice(zero_idx, size=int(n/2 - ones), replace=False)
                h_binary[zero_idx] = 1
            else:
                # Make some ones zeros
                one_idx = np.where(h_binary == 1)[0]
                one_idx = np.random.choice(one_idx, size=int(ones - n/2), replace=False)
                h_binary[one_idx] = 0
            # reset
            H = rfft(h_binary, n)
            autocorr = irfft(H * np.conj(H), n)
            autocorr = np.real_if_close(autocorr)
            c = dx * (n / 2 - autocorr)
            best_c = np.max(c)
            best_h = h_binary.copy()
            stagnation = 0
            T = T0

    return best_h, best_c

def smooth_boundaries(h_bin, ramp_width=2):
    """Convert binary pattern to a continuous function by smoothing."""
    ramp_width = max(1, ramp_width)
    n = len(h_bin)
    h = np.zeros(n)
    h[:] = h_bin  # Start with binary
    # Apply smoothing to boundaries
    # Index of 1s and 0s
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]

    # For 0→1 transitions, add ramps
    for i in range(len(ones) - 1):
        if ones[i+1] - ones[i] == 1:
            j = ones[i]  # end of 0
            for r in range(ramp_width):
                idx = j + 1 + r
                if idx >= n:
                    break
                h[idx] = float(r + 1) / (ramp_width + 1)
    
    # For 1→0 transitions
    for i in range(len(zeros) - 1):
        if zeros[i+1] - zeros[i] == 1:
            j = zeros[i]  # end of 1
            for r in range(ramp_width):
                idx = j + 1 + r
                if idx >= n:
                    break
                h[idx] = 1.0 - float(r + 1) / (ramp_width + 1)

    # Ensure the sum is correct
    h = project_to_feasible(h, n)
    return h

def proj_grad_refine(h, time_limit, use_smooth=False):
    """Projected gradient descent to refine continuous patterns."""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = evaluate_c5(h)
    
    if use_smooth:
        gamma = 10.0  # smooth surrogate parameter
    
    while time.time() - start_time < time_limit:
        # Compute overlaps for every shift
        H = rfft(h, n)
        autocorr = irfft(H * np.conj(H), n)
        autocorr = np.real_if_close(autocorr)
        c_values = dx * (n / 2 - autocorr)
        k_star = np.argmax(c_values)
        c_star = c_values[k_star]

        # Gradient with respect to max
        g = np.zeros(n)
        g += np.roll(h, k_star)
        g += np.roll(h, -k_star)
        g = -dx * g  # gradient points where to go to increase the autocorrelation at this shift

        # Backtracking line search
        eta = 0.5  # initial stepsize
        direction = g
        
        # To avoid too many evaluations, we compute with a few steps of gradient
        for _ in range(100):
            h_new = h - eta * direction
            h_new = project_to_feasible(h_new, n)
            c_new = evaluate_c5(h_new)
            
            if c_new < best_c:
                h = h_new.copy()
                best_c = c_new
                best_h = h.copy()
                
                # Re-evaluate k_star and gradient
                H = rfft(h, n)
                autocorr = irfft(H * np.conj(H), n)
                autocorr = np.real_if_close(autocorr)
                c_values = dx * (n / 2 - autocorr)
                k_star = np.argmax(c_values)
                g = np.zeros(n)
                g += np.roll(h, k_star)
                g += np.roll(h, -k_star)
                g = -dx * g
                
                # Update direction for next iteration
                direction = g
            else:
                eta *= 0.5
                if eta < 1e-10:
                    break
        
    return best_h, best_c

def interpolate(h, new_n):
    """Interpolate to new length."""
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n)
    new_x = np.linspace(0, 2, new_n)
    
    resampled = np.interp(new_x, old_x, h)
    return resampled

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Stage 1: Binary optimisation - use n=256 for FFT efficiency
    n0 = 256
    dx0 = 2.0 / n0
    h0 = np.random.choice([0, 1], size=n0, p=[0.5, 0.5])
    # Ensure correct weight
    ones = np.sum(h0)
    if ones == n0/2:
        pass  # good
    elif ones < n0/2:
        # Fill zeros to ones
        zero_idx = np.where(h0 == 0)[0]
        zero_idx = np.random.choice(zero_idx, size=int(n0/2 - ones), replace=False)
        h0[zero_idx] = 1
    else:
        # Clear ones to zeros
        one_idx = np.where(h0 == 1)[0]
        one_idx = np.random.choice(one_idx, size=int(ones - n0/2), replace=False)
        h0[one_idx] = 0
    
    time_limit_1 = 0.55 * budget_s
    h_bin, c5_bin = binary_anneal(h0, time_limit_1)
    print(f"After binary annealing: C5 = {c5_bin}")
    
    # Stage 2: Smoothing & gradient refinement
    time_limit_2 = 0.25 * budget_s
    h_smooth = smooth_boundaries(h_bin, ramp_width=2)
    h_final, c5_final = proj_grad_refine(h_smooth, time_limit_2)
    print(f"After smoothing and gradient refinement: C5 = {c5_final}")

    # Stage 3: Upsample
    time_limit_3 = 0.15 * budget_s
    if time_limit_3 > 1.0:
        n1 = 2 * n0
        dx1 = 2.0 / n1

        # Interpolate
        h_up = interpolate(h_final, n1)
        h_up = project_to_feasible(h_up, n1)  # rebound to feasible
        
        # Polish in fine grid
        h_up_final, c5_up = proj_grad_refine(h_up, time_limit_3)
        h_return = h_up_final
        c5_return = c5_up
        n_return = n1
    else:
        h_return = h_final
        c5_return = c5_final
        n_return = n0

    return (h_return, c5_return, n_return)
