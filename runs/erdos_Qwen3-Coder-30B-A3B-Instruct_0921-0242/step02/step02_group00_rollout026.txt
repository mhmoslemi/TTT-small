import numpy as np
import time
from scipy.signal import correlate, fftconvolve
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Project h to feasible set: 0 <= h[i] <= 1 and sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Adjust by adding diff/n to each component
    h = h + diff / n
    
    # Re-clip
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def water_fill_projection(x, target_sum):
    """Project x to the set {y : 0 <= y <= 1, sum(y) = target_sum} using waterfilling."""
    # Sort in descending order to implement waterfilling
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Find lambda such that sum(clamp(x_i - lambda, 0, 1)) = target_sum
    # This uses a binary search over lambda
    
    lower, upper = np.min(x) - 1.0, np.max(x) + 1.0
    tolerance = 1e-12
    
    while upper - lower > tolerance:
        mid = (lower + upper) / 2
        projected = np.clip(sorted_x - mid, 0, 1)
        total = np.sum(projected)
        if total >= target_sum:
            lower = mid
        else:
            upper = mid
            
    lambda_opt = lower
    result = np.clip(x - lambda_opt, 0, 1)
    
    # Correct small numerical errors
    # If we deviate from target_sum, adjust some of the components
    diff = np.sum(result) - target_sum
    if abs(diff) > 1e-10:
        # Find largest values to decrease
        indices = np.argsort(result)[::-1]
        for i in indices:
            if diff == 0:
                break
            delta = min(result[i], -diff)
            result[i] -= delta
            diff += delta
    
    return result

def compute_h_update(z, u, t, n, rho=1.0):

    # Prepare for FFT
    n_fft = 2**int(np.ceil(np.log2(2*n)))
    z_u = z - u  # z^k - u^k
    
    # Precompute FFT of all-ones vector
    ones_vec = np.ones(n)
    ones_fft = fft(ones_vec, n_fft)
    
    # Add zero padding
    z_u_pad = np.pad(z_u, (0, n_fft - n), 'constant')
    
    # Use bisection on mu to find the smallest mu such that max_s 1−(2/n)(h∘h)_s <= t
    # i.e. max_s (h∘h)_s >= (n/2)(1 - t)
    lower_mu = 0.0
    upper_mu = 1e10  # starts large
    tolerance = 1e-12
    max_iterations = 30
    
    for _ in range(max_iterations):
        mu = (lower_mu + upper_mu) / 2
        # h = (z^k - u^k) / (1 + mu * ones_fft)
        # But we also multiply by (n_fft/n) due to scaling from frequency domain
        denom = 1 + mu * ones_fft
        h_fft = fft(z_u_pad) / denom
        h = np.real(ifft(h_fft))[:n]
        # Compute h∘h using convolution
        h_conv = correlate(h, h, mode='full')[:, 0]  # [0, 2n-1]
        h_conv = h_conv[:n]  # Take just the autocorrelations we need
        rhs_check = (n/2) * (1 - t)
        
        # If x is not a vector we might have mismatched dimensions.
        # Body: h = z - u - (2 / (rho * n)) * (h.conv.h)
        # But in frequency we do: h = (z - u) * (1/(1 + mu * ones_fft))
        # We want: max_s ( h∘h_s ) >= rhs_check  
        # So this is a performance check with the constraint rewritten:
        # max_s 1 - (2/n)\sum_{i} h_i h_{i+s} <= t
        # i.e. min_s (2/n) \sum h_i h_{i+s} >= (n/2)(1 - t)
        
        # Since we are using full correlations we still have a matrix.
        # We must evaluate the autocorrelation values...
        # Make a dedicated method for correct evaluation.
        
        # Instead of due to circularity or shifting errors:
        # 1. Use i in [0, n), for the shift indices:
        # 2. Compute auto-corr correctly
        auto_corr = correlate(h, h, mode='full')[n-1:n-1+2*n-1] # key auto cors
        auto_corr = auto_corr[:n]
        min_corr = np.min(auto_corr) # numpy auto_corr with 0s offset gives valid left shift.
        value_constrained = (2/n)*min_corr
        val_check = 1 - value_constrained
        
        if val_check > t:
            # Infeasible, too large
            lower_mu = mu
        else:
            # Feasible, try smaller mu
            upper_mu = mu
            
    mu_opt = upper_mu
    denom = 1 + mu_opt * ones_fft
    h_fft = fft(z_u_pad) / denom
    h = np.real(ifft(h_fft))[:n]
    return h    
    
def admm_step(z, u, t, n, rho=1.0):
    """Perform one ADMM step."""
    
    # Get h from current z and u
    h = compute_h_update(z, u, t, n, rho)
    
    # Project h to T: Z = {0 <= y <= 1, sum(y) = n/2}
    z_new = water_fill_projection(h + u, n/2)
    
    # Update dual variable u
    u_new = u + (h - z_new)
    
    # Compute new t using h
    t_new = evaluate_c5(h)
    
    return h, z_new, u_new, t_new
