def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def create_symmetric_binary_pattern(h_initial, n_points):
    """Create a symmetric, monotone binary pattern from continuous h."""
    # Threshold to binary
    b = (h_initial > 0.5).astype(int)
    # Ensure correct sum
    current_sum = np.sum(b)
    target_sum = n_points // 2
    
    if current_sum > target_sum:
        # Reduce some 1s to 0s
        indices = np.where(b == 1)[0]
        indices = indices[np.argsort(h_initial[indices])[::-1]]  # Sort by original value descending
        for idx in indices:
            if current_sum <= target_sum:
                break
            b[idx] = 0
            current_sum -= 1
    elif current_sum < target_sum:
        # Increase some 0s to 1s
        indices = np.where(b == 0)[0]
        indices = indices[np.argsort(h_initial[indices])[::-1]]  # Sort by original value descending
        for idx in indices:
            if current_sum >= target_sum:
                break
            b[idx] = 1
            current_sum += 1
    
    # Symmetrize
    half_len = n_points // 2
    b_left = b[:half_len]
    b_right = b_left[::-1]
    if n_points % 2 == 1:
        b_middle = b[half_len:half_len+1]
        b = np.concatenate([b_left, b_middle, b_right])
    else:
        b = np.concatenate([b_left, b_right])
    
    # Ensure monotonicity (non-increasing on left half)
    for i in range(half_len-1):
        if b[i] < b[i+1]:
            b[i+1] = b[i]
    
    return b

def compute_shift_correlations(b, n_points):
    """Compute the full set of correlations."""
    dx = 2.0 / n_points
    # Convert to float
    h = b.astype(float)
    # Use FFT for fast convolution
    h_padded = np.pad(h, (0, n_points-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, n_points-1), mode='constant', constant_values=0)
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
    # Normalize by dx
    corr_vals = corr_vals * dx
    
    # Also compute maximal correlation
    max_corr = np.max(corr_vals)
    
    return corr_vals, max_corr
