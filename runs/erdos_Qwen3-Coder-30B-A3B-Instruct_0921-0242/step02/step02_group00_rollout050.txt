import numpy as np

def aperiodic_autocorr(h):
    """Compute aperiodic autocorrelation using FFT."""
    n = len(h)
    # Pad to 2*n-1 for full correlation
    h_pad = np.pad(h, (0, n-1), mode='constant')
    h_rev = h[::-1]
    h_rev_pad = np.pad(h_rev, (0, n-1), mode='constant')
    # FFT-based convolution
    H = fft(h_pad)
    H_rev = fft(h_rev_pad)
    corr_fft = ifft(H * np.conj(H_rev)).real
    # Return the first n terms (positive shifts)
    return corr_fft[:n]

def softmax(beta, x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(beta * x_shifted)
    return exp_x / np.sum(exp_x)

def conv(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    a_pad = np.pad(a, (0, n-1), mode='constant')
    b_pad = np.pad(b, (0, n-1), mode='constant')
    return ifft(fft(a_pad) * fft(b_pad)).real[:n]

def project_box_sum(h, total):
    """Project h onto [0,1]^n with sum = total using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = total - current_sum
    
    if abs(diff) < 1e-12:
        return h
    
    # Simple approach: just shift values
    if diff > 0:
        # Need to increase values
        # Simple method: uniformly add diff/n to each
        h = h + diff / len(h)
        h = np.clip(h, 0, 1)
    else:
        # Need to decrease values
        h = h + diff / len(h)
        h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Evaluate C5 = max_s (1 - 2*gamma_s/n)."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def score_guided_swap(h, eps=1e-5):
    """Swap small and large values to escape flat regions."""
    n = len(h)
    c5_val = evaluate_c5(h)
    # Find worst shifts
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    worst_shifts = np.where(np.abs(corr - max_corr) < eps)[0]
    
    # Ensure shifts are in valid range: [0, n-1] first
    worst_shifts = worst_shifts - (n-1) # convert to shifts in [-n+1, n-1]
    worst_shifts = worst_shifts[np.abs(worst_shifts) < n//2]
    
    # Compute scores for each index
    scores = np.zeros(n)
    for s in worst_shifts:
        if s > 0:
            scores[:-s] += h[s:]
        elif s < 0:
            scores[-s:] += h[:s]
        else:
            scores += h  # s == 0 contributes to all indices
    
    # Find indices to swap
    source_idx = np.argmax(scores)
    target_idx = np.argmin(scores)
    
    # Pick move amount
    delta = min(0.01, h[source_idx], 1 - h[target_idx])
    
    if delta < 1e-6:
        return h
    
    # Perform swap
    h_new = h.copy()
    h_new[source_idx] -= delta
    h_new[target_idx] += delta
    
    return project_box_sum(h_new, n/2)
