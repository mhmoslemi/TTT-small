import numpy as np
import time
import random
from scipy.fft import fft, ifft

def evaluate_c5(h):
    """Evaluate C5 for a given step function h."""
    dx = 2.0 / len(h)
    # Compute full cross-correlation using FFT
    h_padded = np.pad(h, (0, len(h)-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, len(h)-1), mode='constant', constant_values=0)
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    corr_vals = corr_vals[:len(h)]  # Take only non-cyclic part
    corr_vals = corr_vals * dx
    return np.max(corr_vals)

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    if diff > 0:
        # Need to increase sum
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    return np.clip(h_copy, 0, 1)

def compute_correlation_changes(h, i, j, corr_vals):
    """Compute the change in correlation when swapping positions i and j."""
    # Compute the updated cross-correlation values
    delta = np.zeros_like(corr_vals)
    n = len(h)
    for s in range(n):
        # Handle indices in the correlation computation considering wrapping
        left_i = (i - s) % n
        right_i = (i + s) % n
        left_j = (j - s) % n
        right_j = (j + s) % n
        
        # Contribution from i and j in C_s
        delta[s] += (1 - h[left_i]) + (1 - h[right_i]) - h[left_j] - h[right_j]
        delta[s] -= (1 - h[left_j]) + (1 - h[right_j]) - h[left_i] - h[right_i]
    return delta

def simulated_annealing_binary(h0, n, max_moves=60000, time_limit=None):
    """Run simulated annealing on binary vectors to minimize max correlation."""
    h = h0.copy()
    corr_vals = np.zeros(n)
    # Compute initial correlation values (we'll update incrementally)
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    corr_vals = corr_vals[:n]
    
    current_energy = np.max(corr_vals)
    best_h = h.copy()
    best_energy = current_energy
    
    T = 0.02 * n  # Initial temperature
    alpha = 0.997  # Cooling factor
    
    for move in range(max_moves):
        if time_limit and time.time() >= time_limit:
            break
            
        # Select two random positions to swap: one 1 and one 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute change in correlation values
        delta_corr = compute_correlation_changes(h, i, j, corr_vals)
        new_corr_vals = corr_vals + delta_corr
        
        # Compute new energy (max correlation)
        new_energy = np.max(new_corr_vals)
        
        # Accept or reject the move
        if new_energy <= current_energy:
            accept = True
        else:
            prob = np.exp(-(new_energy - current_energy) / T)
            accept = np.random.rand() < prob
            
        # If accepted, update everything
        if accept:
            h[i], h[j] = h[j], h[i]  # Swap in h
            corr_vals = new_corr_vals  # Update correlation vector
            current_energy = new_energy
            
            if current_energy < best_energy:
                best_energy = current_energy
                best_h = h.copy()
                
        # Update temperature
        T *= alpha
        if T < 1e-5:
            break
            
    return best_h, best_energy

def polish_continuous(h, n, max_iter=30):
    """Perform continuous polishing using smooth max surrogate."""
    dx = 2.0 / n
    h = h.astype(float)
    h = project_to_feasible(h, n)
    
    # Use smooth max with increasing λ (from 5 to 30)
    for i in range(max_iter):
        # Use smooth max with λ
        lambda_val = 5 + 25 * (i / max_iter)
        
        # Compute true correlations
        h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n] * dx
        
        # Compute smooth max
        smooth_corr = np.log(np.sum(np.exp(lambda_val * corr_vals))) / lambda_val
        
        # Compute gradient of the smooth max w.r.t h
        grad = np.zeros(n)
        for s in range(n):
            for t in range(n-s):
                if s == 0:
                    # Derivative of h[t] * (1 - h[t]) in first term
                    grad[t] += 1 - 2*h[t]  # This is only for current shifts?
                else:
                    # Derivative of h[t] * (1 - h[t+s]) 
                    deriv = 1 - h[t+s]
                    grad[t] += deriv
                    # Also the derivative of h[t+s] *(1 - h[t]) 
                    if t+s < n:
                        grad[t+s] += -(1 - h[t])
        
        grad = grad * dx  # Scale gradient
        step_size = 0.1 * dx * (0.5 ** (i // 10))
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n)
        h = h_new
        
    # Final evaluation with true max
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    corr_vals = corr_vals[:n] * dx
    
    return h, np.max(corr_vals)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 10  # Leave 10 s for cleanup
    
    n = 221  # Good balance between resolution and computation time
    dx = 2.0 / n
    h_best = None
    c5_best = float('inf')
    n_points = n

    # Launch parallel trials
    num_trials = 12
    for trial in range(num_trials):
        if time.time() >= time_limit:
            break
            
        # Initialize a good starting point
        if np.random.rand() < 0.5:
            # Start with a block pattern
            h = np.zeros(n, dtype=np.int8)
            mid = n // 2
            h[mid - n//4:mid + n//4] = 1
            # Perturb a bit
            for _ in range(5):
                i = np.random.randint(0, n)
                j = np.random.randint(0, n)
                if h[i] != h[j]:
                    h[i], h[j] = h[j], h[i]
        else:
            # Random permutation of same number of 1s and 0s
            h = np.zeros(n, dtype=np.int8)
            h[:n//2] = 1
            np.random.shuffle(h)
        
        # Run simulated annealing
        h_anneal, c5_anneal = simulated_annealing_binary(h, n, time_limit=time_limit)
        h_polish, c5_polish = polish_continuous(h_anneal, n)
        
        if c5_polish < c5_best:
            c5_best = c5_polish
            h_best = h_polish.copy()
            
        if time.time() >= time_limit:
            break
            
    if h_best is not None:
        return (h_best, c5_best, n_points)
    else:
        # Fallback
        h_fallback = np.ones(n_points) * 0.5
        c5_fallback = evaluate_c5(h_fallback)
        return (h_fallback, c5_fallback, n_points)
