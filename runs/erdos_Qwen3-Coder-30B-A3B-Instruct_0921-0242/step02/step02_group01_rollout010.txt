import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region: 0 <= h <= 1 and sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum

    if abs(diff) < 1e-10:
        return h

    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)

    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = rfft(h)
    ih_fft = rfft(1 - h)
    corr = irfft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    # Excluding shift 0
    max_corr = np.max(corr[1:]) * dx
    return max_corr

def gradient_worst_shift(h):
    """Compute gradient with respect to worst shift."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    h_fft = rfft(h)
    ih_fft = rfft(1 - h)
    corr = irfft(h_fft * np.conj(ih_fft))
    # exclude shift 0
    corr_vals = corr[1:]
    worst_shift = np.argmax(corr_vals)

    # Compute gradient
    grad = np.zeros_like(h)
    for i in range(n):
        left_idx = (i + worst_shift) % n
        right_idx = (i - worst_shift) % n
        grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])

    return grad

def gradient_worst_shift_descent(h, max_iter=200, alpha0=0.1):
    """Gradient descent on the worst shift."""
    n = len(h)
    dx = 2.0 / n
    h_current = h.copy()
    alpha = alpha0 * dx

    for t in range(max_iter):
        grad = gradient_worst_shift(h_current)
        h_new = h_current - alpha * grad
        h_new = project_to_feasible(h_new, n / 2.0)
        if evaluate_c5(h_new) < evaluate_c5(h_current):
            h_current = h_new
            if t % 10 == 0:
                alpha *= 0.95

    return h_current

def differential_evolution(seed, n_points, pop_size, max_iter, budget_s):
    """Differential evolution with FFT evaluation"""
    np.random.seed(seed)
    start_time = time.time()

    # Initialize population
    target_sum = n_points / 2.0
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = h * (target_sum / np.sum(h))
        h = project_to_feasible(h, target_sum)
        population.append(h)

    # Include initial guess if available
    if 'initial_h_values' in globals():
        h_init = initial_h_values
        if len(h_init) > n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_start = np.interp(x_new, x_orig, h_init)
        else:
            h_start = h_init[:n_points]
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        h_start = project_to_feasible(h_start, n_points)
        population[0] = h_start

    # DE parameters
    F = 0.6
    CR = 0.9

    # Store best
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h)

    iteration = 0
    while iteration < max_iter and (time.time() - start_time < budget_s * 0.75):
        iteration += 1

        # Shuffle population order for randomness
        np.random.shuffle(population)

        # Update each individual
        for i in range(pop_size):
            # Randomly sample three other individuals
            a, b, c = np.random.choice(pop_size, size=3, replace=False)
            x_a, x_b, x_c = population[a], population[b], population[c]

            # Mutation step
            donor = np.clip(x_a + F * (x_b - x_c), 0, 1)
            donor = project_to_feasible(donor, target_sum)

            # Crossover
            trial = np.copy(population[i])
            j_rand = np.random.randint(n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = donor[j]

            # Evaluate
            c5_trial = evaluate_c5(trial)
            c5_current = evaluate_c5(population[i])

            if c5_trial <= c5_current:
                population[i] = trial

        # Track best
        for h in population:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Occasionally refine best
        if iteration % 200 == 0 and (time.time() - start_time < budget_s * 0.75):
            refined = gradient_worst_shift_descent(best_h, max_iter=100)
            new_c5 = evaluate_c5(refined)
            if new_c5 < best_c5:
                best_h = refined
                best_c5 = new_c5

    return best_h, best_c5

def stochastic_swap_refinement(h, budget_s):
    """Stochastic swap refinement"""
    start_time = time.time()
    h_current = h.copy()
    c5_current = evaluate_c5(h_current)
    T = 0.1
    T_min = 1e-6
    decay = 0.999
    max_iter = 50000  # limit number of steps

    for _ in range(max_iter):
        if time.time() - start_time > budget_s * 0.995:
            break

        i = np.random.randint(0, len(h_current))
        j = np.random.randint(0, len(h_current))
        if i == j:
            continue

        # Small transfer
        delta = min(0.01, h_current[i], 1 - h_current[j])
        if delta == 0:
            continue

        h_test = h_current.copy()
        h_test[i] -= delta
        h_test[j] += delta
        h_test = project_to_feasible(h_test, len(h_current) / 2)

        c5_test = evaluate_c5(h_test)
        delta_c5 = c5_test - c5_current

        if delta_c5 < 0:
            h_current = h_test
            c5_current = c5_test
        elif np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_test
            c5_current = c5_test

        # Cool down
        T = max(T_min, T * decay)

    return h_current, c5_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 400
    pop_size = min(40, 2 * n_points // 10)
    max_iter = int(budget_s * 1000 / (pop_size * n_points)) if n_points > 0 else 1000
    
    # Run DE
    best_h, best_c5 = differential_evolution(seed, n_points, pop_size, max_iter, budget_s)

    # Refine with gradient descent
    if time.time() - start_time < budget_s * 0.9:
        refined = gradient_worst_shift_descent(best_h, max_iter=200)
        refined_c5 = evaluate_c5(refined)
        if refined_c5 < best_c5:
            best_h = refined
            best_c5 = refined_c5

    # Stochastic polishing
    if time.time() - start_time < budget_s * 0.995:
        h_final, c5_final = stochastic_swap_refinement(best_h, budget_s)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # Optionally double resolution and refine
    if time.time() - start_time < budget_s * 0.999:
        # Interpolate to higher resolution
        n_high = 800
        x_low = np.linspace(0, 2, n_points)
        x_high = np.linspace(0, 2, n_high)
        h_high_interp = np.interp(x_high, x_low, best_h)
        h_high = project_to_feasible(h_high_interp, n_high / 2)
        c5_high = evaluate_c5(h_high)
        # Short refine
        h_refined_high = gradient_worst_shift_descent(h_high, max_iter=100)
        c5_refined_high = evaluate_c5(h_refined_high)
        if c5_refined_high < best_c5:
            best_h = h_refined_high
            best_c5 = c5_refined_high

    return (best_h, best_c5, n_points)
