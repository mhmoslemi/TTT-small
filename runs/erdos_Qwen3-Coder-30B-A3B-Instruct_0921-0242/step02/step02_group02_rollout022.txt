import numpy as np
import time
from scipy.optimize import differential_evolution, linprog
from scipy.fft import rfft, irfft
from scipy.interpolate import interp1d

def project_to_feasible(h, target_sum):
    """Ensure sum(h) = target_sum with h in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple uniform adjustment
    scale = target_sum / current_sum
    h = h * scale
    return h

def evaluate_c5(h):
    """Evaluate C5 for a given h vector using fft;
    The equivalent of max_k integral h(x)(1-h(x+k)) dx
    """
    dx = 2.0 / len(h)
    # (1-h) auto-correlation
    h1 = 1 - h
    # FFT-based convolution with flipped h1
    H1 = rfft(h1)
    H = rfft(h)
    # cross-correlation
    corr = irfft(H * np.conj(H1))
    corr = np.concatenate([corr, np.zeros(len(h))]) # full mode
    # max over shifts
    max_corr = np.max(corr[:len(h)]) * dx
    return max_corr

def basis_evaluation(x, M, n):
    """Generate coefficient basis vectors for cos/sin"""
    basis = np.zeros((2*M, n))
    for m in range(1, M+1):
        basis[2*(m-1), :] = np.cos(m * np.pi * x)
        basis[2*(m-1) + 1, :] = np.sin(m * np.pi * x)
    return basis

def compute_h_from_coeffs(c, basis, n_points):
    """Given coefficients c and basis, compute h on grid points"""
    M = len(c) // 2
    x = np.linspace(0, 2, n_points, endpoint=False)
    h_raw = 0.5 + basis @ c  # a0=0.5
    h = np.clip(h_raw, 0, 1)
    h = project_to_feasible(h, n_points / 2)
    return h

def fit_c5_function(c, basis, n_points):
    """Objective function for DE - evaluates C5 given coefficients"""
    h = compute_h_from_coeffs(c, basis, n_points)
    return evaluate_c5(h)

def full_fit_c5_function(c, basis, n_points):
    """Exact fit for better evaluation"""
    h = compute_h_from_coeffs(c, basis, n_points)
    return evaluate_c5(h)

def solve_linearised_lp(h, M, max_constraints=10):
    # compute current A(k) = h^T P_k h
    n = len(h)
    A_vals = np.zeros(n)
    k_star = 0
    min_A = np.inf
    
    for k in range(n):
        # This is A(k) = sum h(i) h((i+k)%n)
        A_i = np.sum(h * np.roll(h, k))
        A_vals[k] = A_i
        if A_i < min_A:
            min_A = A_i
            k_star = k
    
    # Active constraints (Worst shifts)
    # Get ~5 worst shifts
    worst_indices = np.argsort(A_vals)[:max_constraints]
    # Build constraints for all active ones
    A = np.zeros((len(worst_indices) + 2, n + 1))
    
    for i, k in enumerate(worst_indices):
        # constraint: h^T P_k h >= t
        # Gradient: 2*roll(h, k) so constraint is: g^T h >= t - g^T h_old
        g = np.roll(h, k)
        A[i, :n] = g
        A[i, n] = -1  # -t <= g^T h - t
    
    # bounds constraint (0 <= h <= 1)
    A[len(worst_indices), :n] = np.ones(n)
    A[len(worst_indices), n] = 0  # 0 <= sum h
    A[len(worst_indices)+1, :n] = -np.ones(n)
    A[len(worst_indices)+1, n] = 0  # sum h <= n/2
    
    b = np.zeros(len(worst_indices) + 2)
    b[len(worst_indices)] = 0
    b[len(worst_indices)+1] = -n/2

    # Solve LP to maximize t
    c_lp = np.zeros(n+1)
    c_lp[n] = -1  # maximize -t (minimize t)
    try:
        res = linprog(c_lp, A_ub=A, b_ub=b, method='highs', bounds=[(0,1)]*n + [None, None])
        if res.success:
            if -res.fun < min_A:
                # Use the optimal value from LP
                new_h = res.x[:n]
                t_opt = -res.fun
                return new_h, t_opt
    except:
        pass
    
    return h, min_A

def subgradient_step(h, target_sum):
    """Perform one step of subgradient descent"""
    # evaluate correlations to find k_star
    n = len(h)
    dx = 2.0 / n
    h1 = 1 - h
    H1 = rfft(h1)
    H = rfft(h)
    corr = irfft(H * np.conj(H1))
    corr = np.concatenate([corr, np.zeros(n)])
    k_star = np.argmax(corr[:n])
    
    # gradient + signs find shift to get to max overlap score
    grad = np.roll(h, k_star)  # 2*...
    # now move mass from large to small values in direction of grad
    # find idx where h is large and where grad is small / k_star is small
    
    # Alternating greedy - touch 2 random indices that improve the score
    evaluator = lambda x: evaluate_c5(x)
    
    # try swapping 2 pairs
    i = np.random.randint(n)
    i = np.argmin(h)  # use small value
    j = np.argmax(h)  # use large value
    
    if i == j: 
        return h
        
    # small delta
    delta = max(0.01, min(0.5, h[i]))
    if delta > h[j]:
        delta = h[j] / 2
    
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    
    h_new = project_to_feasible(h_new, target_sum)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Create bases
    n0 = 250
    M = 12  # Number of freq components
    x = np.linspace(0, 2, n0, endpoint=False)
    basis = basis_evaluation(x, M, n0)
    
    # Bounds for coefficients [-0.5, 0.5] for both a and b
    lb = [-0.5] * (2 * M)
    ub = [0.5] * (2 * M)
    bounds = list(zip(lb, ub))
    
    # Phase 2: DE search
    max_eval = 3000
    pop_size = 15 * (2*M)  # default value
    
    def fit_c5_local(c):
        return fit_c5_function(c, basis, n0)
    
    result = differential_evolution(
        fit_c5_local,
        bounds,
        seed=seed,
        maxiter=1000,
        popsize=pop_size,
        atol=0.0001,
        x0=None,
        worker=None,
        callback=None,
        strategy='best1bin',
        recombination=0.7,
        mutation=(0.5, 1),
        errtol=1e-08,
        polish=True,
        disp=False,
        catol=5e-03,
        timeout=budget_s * 0.45  # Allow some overhead for others
    )
    
    # Phase 3: Get top 5 candidates by distance
    candidates = []
    candidates_c = []
    c_best = result.x
    
    best_h = compute_h_from_coeffs(c_best, basis, n0)
    best_c5 = evaluate_c5(best_h)
    candidates.append((c_best, best_h, best_c5))
    candidates_c.append(c_best)
    
    # Get additional candidates from center of DE solution
    # artificial exploration by perturbation in good direction
    for _ in range(4):
        if time.time() - start_time > 0.7 * budget_s:
            break
        pert = np.random.normal(0, 0.01, 2*M)
        new_c = np.clip(c_best + pert, -0.5, 0.5)
        new_h = compute_h_from_coeffs(new_c, basis, n0)
        new_c5 = evaluate_c5(new_h)
        candidates.append((new_c, new_h, new_c5))
        candidates_c.append(new_c)
    
    # Phase 4: Select candidates & refine with SLP
    candidate_scores = [c[2] for c in candidates]
    # Select top 5 with distinct coeffs
    top_indices = sorted(range(len(candidate_scores)), key=lambda i: candidate_scores[i])[:5]
    top_candidates = [candidates[i] for i in top_indices]
    
    refinements = []
    
    for i, (c, h, score) in enumerate(top_candidates):
        if time.time() - start_time > 0.9 * budget_s:
            break
        
        h_refined = h.copy()
        t_prev = score
        iter_count = 0
        max_iter = 300
        
        while iter_count < max_iter:
            if time.time() - start_time > 0.9 * budget_s:
                break
            h_prev = h_refined.copy()
            t_prev = evaluate_c5(h_refined)
            
            # Linearized model
            new_h, t_new = solve_linearised_lp(h_refined, M, max_constraints=8)
            
            if t_new > t_prev + 1e-6:
                h_refined = new_h
            else:
                # Backtrack to ensure descent
                alpha = 0.5
                h_trial = (1-alpha) * h_prev + alpha * new_h
                h_trial = project_to_feasible(h_trial, n0/2)
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < score:
                    h_refined = h_trial
                else:
                    break  # past break or no improvement
            
            if abs(t_new - t_prev) < 1e-6:
                break
            iter_count += 1
        
        refinements.append((c, h_refined, evaluate_c5(h_refined)))
    
    # Phase 5: upsampling and final polishing
    candidate_results = [(h, c5) for _, h, c5 in refinements]
    candidate_results.sort(key=lambda x: x[1])
    best_refined_h = candidate_results[0][0]
    
    # Now upsample to higher resolution
    n1 = 500  # or try more
    if time.time() - start_time > budget_s * 0.8:
        n1 = 300  # or even less
    
    dx = 2.0 / n0
    x0 = np.arange(n0) * dx
    x1 = np.linspace(0, 2, n1, endpoint=False)
    
    # Interpolate using continuous function
    h_interp = interp1d(x0, best_refined_h, kind='linear', periodic=True)
    h_upsample = h_interp(x1)
    # Force scaling to sum correctly
    h_upsample = project_to_feasible(h_upsample, n1 / 2)
    c5_final = evaluate_c5(h_upsample)
    
    # Final refinement
    final_h = h_upsample.copy()
    for _ in range(2000):
        if time.time() - start_time > budget_s - 0.5:
            break
        final_h = subgradient_step(final_h, n1 / 2)
        c5_final = evaluate_c5(final_h)
    
    return (final_h, c5_final, n1)
