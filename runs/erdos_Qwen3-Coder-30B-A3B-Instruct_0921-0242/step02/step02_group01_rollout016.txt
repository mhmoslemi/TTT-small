import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, target_sum):
    """Project h to feasible region: 0 <= h <= 1 and sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h

    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h using FFT."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # Take the real part of the correlation (excluding shift 0)
    real_corr = np.real_if_close(corr)
    max_corr = np.max(real_corr[1:len(h)]) * dx  # ignore shift 0
    return max_corr

def gradient_worst_shift_descent(h, max_iter=200, alpha0=0.1):
    """Gradient descent on the worst shift."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for t in range(max_iter):
        # Compute all correlations
        h_fft = fft(h, 2 * n)
        ih_fft = fft(1 - h, 2 * n)
        corr = ifft(h_fft * np.conj(ih_fft))
        real_corr = np.real_if_close(corr)
        
        # Ignore shift 0 (self-correlation)
        max_corr_idx = np.argmax(real_corr[1:n]) + 1
        # Compute gradient for worst shift
        grad = np.zeros_like(h)
        for i in range(n):
            left_idx = (i + max_corr_idx) % n
            right_idx = (i - max_corr_idx) % n
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Update with step size
        alpha = alpha0 / np.sqrt(t + 1) if t > 0 else alpha0
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n/2)
        # Check improvement
        if evaluate_c5(h_new) < evaluate_c5(h):
            h = h_new
        else:
            break
    return h

def differential_evolution_step(pop, F=0.6, CR=0.9):
    """Perform a single DE step on population."""
    new_pop = []
    for i in range(len(pop)):
        # Select three distinct individuals
        idxs = [j for j in range(len(pop)) if j != i]
        a, b, c = [pop[j] for j in np.random.choice(idxs, 3, replace=False)]
        # Mutation
        donor = a + F * (b - c)
        donor = np.clip(donor, 0, 1)
        # Rescale to target sum
        donor = donor * (len(donor)/2) / np.sum(donor)
        donor = project_to_feasible(donor, len(donor)/2)
        # Crossover (binomial)
        trial = donor.copy()
        for j in range(len(donor)):
            if np.random.rand() > CR:
                trial[j] = pop[i][j]
        # Selection (should the trial even be better? we compute pop[i] again to save time)
        corr_old = evaluate_c5(pop[i])
        corr_new = evaluate_c5(trial)
        if corr_new < corr_old:
            new_pop.append(trial)
        else:
            new_pop.append(pop[i])
    return new_pop

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 400
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Build initial population
    if 'initial_h_values' in globals():
        h_start = initial_h_values
        if len(h_start) != n_points:
            x_orig = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, n_points)
            h_start = np.interp(x_new, x_orig, h_start)
    else:
        h_start = np.ones(n_points) * 0.5
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = project_to_feasible(h_start, n_points/2)

    POP_SIZE = min(40, max(10, 2 * n_points // 10))
    population = []
    for _ in range(POP_SIZE):
        if len(population) == 0:
            h = h_start.copy()
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = project_to_feasible(h, n_points/2)
        population.append(h)

    global_best = None
    global_best_c5 = float('inf')

    # DE iterations
    max_gens = int(budget_s / (0.001 * POP_SIZE))
    if max_gens < 10:
        max_gens = 10
    
    for gen in range(max_gens):
        if time.time() - start_time > 0.8 * budget_s:
            break
        # Perform DE step
        population = differential_evolution_step(population)
        # Track best
        for i, h in enumerate(population):
            c5 = evaluate_c5(h)
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best = h.copy()

        # Periodically refine best
        if gen % 200 == 0 and global_best is not None:
            if time.time() - start_time > 0.95 * budget_s:
                break
            ref_h = gradient_worst_shift_descent(global_best)
            ref_c5 = evaluate_c5(ref_h)
            if ref_c5 < global_best_c5:
                global_best_c5 = ref_c5
                global_best = ref_h.copy()
    
    # Final gradient refinement on the best found
    if global_best is not None and time.time() - start_time < 0.99 * budget_s:
        ts = time.time()
        global_best = gradient_worst_shift_descent(global_best)
        global_best_c5 = evaluate_c5(global_best)
        ts = time.time() - ts
        if ts > 0.05:  # Avoid noise from timing
            pass

    # Final stochastic polishing
    if global_best is not None and time.time() - start_time < 0.995 * budget_s:
        h_final = global_best.copy()
        c5_final = global_best_c5
        # Cooling schedule for SA
        T = 0.01
        T_min = 1e-6
        alpha = 0.99
        iter_per_temp = 1000
        temp_count = 0
        while T > T_min and time.time() - start_time < 0.995 * budget_s:
            for _ in range(iter_per_temp):
                i, j = np.random.randint(0, n_points, 2)
                if h_final[i] <= 0 or h_final[j] >= 1:
                    continue
                delta = np.random.uniform(0, min(h_final[i], 1 - h_final[j], 0.01))
                h_test = h_final.copy()
                h_test[i] -= delta
                h_test[j] += delta
                h_test = project_to_feasible(h_test, n_points/2)
                c5_test = evaluate_c5(h_test)
                if c5_test < c5_final:
                    c5_final = c5_test
                    h_final = h_test
                else:
                    if np.random.rand() < np.exp(-(c5_test - c5_final) / T):
                        c5_final = c5_test
                        h_final = h_test
            T *= alpha
            temp_count += 1
            
    if global_best is not None:
        return (h_final, c5_final, n_points)
    else:
        # fallback to initial point
        h_fallback = project_to_feasible(h_start, n_points / 2)
        c5_fallback = evaluate_c5(h_fallback)
        return (h_fallback, c5_fallback, n_points)
