import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_box_sum(y, total):
    """Project y onto {x : 0 ≤ x_i ≤ 1, sum x_i = total} using water-filling."""
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if abs(s - total) < 1e-10:
        return y
    # bisection on lambda
    lo, hi = -10.0, 10.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        z = np.clip(y - lam, 0, 1)
        sz = np.sum(z)
        if sz > total:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0, 1)

def compute_autocorrelation(h):
    """Compute cyclic autocorrelation via FFT."""
    H = fft(h)
    P = np.abs(H)**2
    gamma = ifft(P).real
    return gamma

def compute_c5_from_autocorr(gamma, n):
    """Compute C5 from autocorrelation."""
    # gamma is indexed from -(n-1) to (n-1)
    # the zero-lag term is gamma[n-1]
    gamma0 = gamma[n-1]
    avg = (np.sum(gamma) - gamma0) / (n-1)
    gamma_target = np.where(np.arange(2*n-1) != n-1, avg, gamma0)
    return 1.0 - (2.0/n) * np.max(gamma_target)

def generate_legendre_seed(n):
    # Use a large prime
    p = 257  # 257 is prime
    if n > p - 1:
        raise ValueError("n too large for this prime")
    # Generate Legendre symbols (1 to p-1)
    k_vals = np.arange(1, n+1)
    legendre = np.array([np.sign(np.power(k, (p-1)//2, p) - 1) for k in k_vals])
    # Map -1 -> 0, 1 -> 1
    seed = (legendre + 1) // 2
    # Ensure sum = n/2
    current_sum = np.sum(seed)
    if current_sum < n/2:
        # find zero positions and flip some
        zero_idx = np.where(seed == 0)[0]
        if len(zero_idx) >= int(n/2 - current_sum):
            for i in range(int(n/2 - current_sum)):
                seed[zero_idx[i]] = 1
    elif current_sum > n/2:
        # find one positions and flip some
        one_idx = np.where(seed == 1)[0]
        if len(one_idx) >= int(current_sum - n/2):
            for i in range(int(current_sum - n/2)):
                seed[one_idx[i]] = 0
    return seed.astype(float)

def generate_msequence_seed(n):
    # Generate max length sequence (LFSR of order 8)
    # Not a true implementation but a sample for illustrative purpose
    # For production, use a real LFSR or a table of such
    shift_reg = 0x12345678  # some arbitrary starting value
    poly = 0x100000001  # primitive polynomial for degree 8
    seeds = []
    for i in range(255):  # max period for degree 8
        lsb = shift_reg & 1
        shift_reg = (shift_reg >> 1) ^ (-lsb & poly)
        seeds.append(lsb)
    # Take first n bits
    seq = np.array(seeds[:n], dtype=float)
    seq = seq / np.max(seq) if np.max(seq) > 0 else seq  # Normalize to 0 or 1
    current_sum = np.sum(seq)
    if current_sum < n/2:
        # flip some zeros to ones
        flip = int(n/2 - current_sum)
        zeros = np.where(seq == 0)[0]
        for i in range(min(flip, len(zeros))):
            seq[zeros[i]] = 1
    return seq

def generate_random_balanced_seed(n):
    seed = np.zeros(n)
    # Choose approx. n/2 positions to set to 1
    ones = np.random.choice(n, size=n//2, replace=False)
    seed[ones] = 1.0
    return seed

def binary_hill_climbing(h, max_iter=3*n, time_budget=None):
    start = time.time()
    n = len(h)
    γ = compute_autocorrelation(h)
    h = h.copy()
    current_c5 = compute_c5_from_autocorr(γ, n)
    best_h = h.copy()
    best_c5 = current_c5
    iter_no_improvement = 0
    max_no_improvement = 3*n
    moves = 0

    while moves < max_iter and iter_no_improvement < max_no_improvement:
        if time_budget and (time.time() - start) > time_budget*0.95:
            break

        # Find indices with 1 and 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # pick a random swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Compute change in autocorrelation from swapping i and j
        Δγ = np.zeros(2*n-1)
        # Entries involving i
        for k in range(2*n-1):
            if (i+k) % n < n:
                Δγ[k] -= h[i] * h[(i + k) % n]  # since h[i] becomes 0
            if (i-k) % n < n:
                Δγ[k] -= h[i] * h[(i - k) % n]  # since h[i] becomes 0
            if (j+k) % n < n:
                Δγ[k] += h[j] * h[(j + k) % n]  # since h[j] becomes 1
            if (j-k) % n < n:
                Δγ[k] += h[j] * h[(j - k) % n]  # since h[j] becomes 1

        # Apply the change to autocorrelation to get new γ
        γ_new = γ + Δγ

        # Get new C5 value
        new_c5 = compute_c5_from_autocorr(γ_new, n)

        if new_c5 < current_c5:
            h[i] = 0.0
            h[j] = 1.0
            γ = γ_new
            current_c5 = new_c5
            iter_no_improvement = 0
            moves += 1
        else:
            iter_no_improvement += 1
            moves += 1
    return h, current_c5

def spectral_alternating_projection(h, max_iter=400, time_budget=None):
    start = time.time()
    n = len(h)
    h = h.astype(float)
    h = project_box_sum(h, n/2)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    α = 0.7

    for iter in range(max_iter):
        if time_budget and (time.time() - start) > time_budget*0.98:
            break

        # 1. Compute current autocorrelation
        γ = compute_autocorrelation(h)
        γ0 = γ[n-1]
        avg_lag0 = (np.sum(γ) - γ0) / (n - 1)

        # 2. Build target autocorrelation
        γ_target = γ.copy()
        γ_target[range(n-1)] = (1 - α) * γ[range(n-1)] + α * avg_lag0
        γ_target[n-1] = γ0

        # 3. Compute needed magnitude
        P_target = fft(γ_target)
        # Clamp to non-negative
        P_target = np.maximum(P_target.real, 0)
        A_target = np.sqrt(P_target)

        # 4. Reconstruct with new magnitudes
        H = fft(h)
        H_new = A_target * np.exp(1j * np.angle(H))
        h_new = ifft(H_new).real

        # 5. Project onto feasible set
        h_new = project_box_sum(h_new, n/2)
        h = h_new

        # 6. Update best
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # 7. Increase α slightly
        α = min(α + 0.02, 0.99)

    return best_h, best_c5

def evaluate_c5(h):
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n = 256
    dx = 2.0 / n

    # Phase 1: Generate good binary seeds
    best_c5 = float('inf')
    best_h = None

    # Generate three kinds of seeds
    seeds = []
    seeds.append(("legendre", generate_legendre_seed(n)))
    seeds.append(("msequence", generate_msequence_seed(n)))
    seeds.append(("random", generate_random_balanced_seed(n)))

    print(f"Starting binary hill-climbing on {len(seeds)} seeds...")
    for seed_name, seed_h in seeds:
        if time.time() - start_time > 0.8 * budget_s:
            break
        seed_h = seed_h.copy()
        # Project to feasibility for binary seed
        seed_h = project_box_sum(seed_h, n/2)
        # Binary hill-climbing
        refined_h, refined_c5 = binary_hill_climbing(seed_h, time_budget=budget_s*0.8)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h.copy()
        print(f"{seed_name}: {refined_c5:.6f}")

    print(f"After binary hill-climbing, best C5 = {best_c5:.6f}")

    # Phase 2: SAP
    if time.time() - start_time < 0.95 * budget_s:
        best_h, best_c5 = spectral_alternating_projection(best_h, time_budget=budget_s*0.95)

    print(f"After SAP, best C5 = {best_c5:.6f}")

    # Phase 3: Final polishing
    if time.time() - start_time < 0.99 * budget_s:
        # Perform projected gradient descent
        β = 5.0
        η = 1.0
        max_steps = 300
        h = best_h.copy()
        current_c5 = best_c5

        for i in range(max_steps):
            if time.time() - start_time > 0.995 * budget_s:
                break
            # Evaluate current γ
            γ = compute_autocorrelation(h)
            C = 1.0 - (2.0/n) * γ
            # Softmax weights
            w = np.exp(β * C)
            w = w / np.sum(w)

            # compute gradient
            w_fft = np.fft.fft(w)
            h_fft = np.fft.fft(h)
            grad_fft = -(2.0/n) * (w_fft * h_fft + np.conj(w_fft[::-1]) * h_fft[::-1])
            # in time domain
            grad = np.fft.ifft(grad_fft).real

            # Line search
            while True:
                h_new = h - η * grad
                h_new = project_box_sum(h_new, n/2)
                new_c5 = evaluate_c5(h_new)
                if new_c5 < current_c5:
                    h = h_new
                    current_c5 = new_c5
                    break
                else:
                    η /= 2.0
                    if η < 1e-8:
                        break
            β *= 1.2
            if (i % 20) == 0:
                print(f"Gradient polishing iter {i}: C5 = {current_c5:.6f}")

        best_h = h.copy()
        best_c5 = current_c5

    return (best_h, best_c5, n)
