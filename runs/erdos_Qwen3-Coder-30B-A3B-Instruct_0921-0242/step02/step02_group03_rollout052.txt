import numpy as np
import cvxpy as cp
import time

def full_to_sym(h_full):
    """Convert full vector to symmetric (half + center) form."""
    n = len(h_full)
    # Use the first half + center (for odd n, center is n//2)
    if n % 2 == 0:
        return h_full[:n//2]
    else:
        return h_full[:n//2 + 1]

def sym_to_full(h_sym):
    """Convert symmetric vector back to full vector."""
    n = 2 * len(h_sym) - 1 if len(h_sym) > 1 else 1
    # For symmetry h[i] = h[n-1-i]
    h_full = np.zeros(n)
    for i in range(len(h_sym)):
        h_full[i] = h_sym[i]
        h_full[n-1-i] = h_sym[i]
    if n % 2 == 1:
        # correct the center element which was added twice in the loop above
        h_full[n//2] = h_sym[n//2]
    return h_full

def solve_lp(shift_set, h_initial, n):
    """Solve the LP with given shift set using symmetry."""
    # Only consider non-negative shifts (we'll use symmetry)
    # We need to enforce: sum(h) = n/2, and each h[i] in [0,1]
    m = n // 2 + 1  # size of symmetric vector
    x = cp.Variable(m)
    t = cp.Variable()
    
    # Integral constraint scaled by dx = 2/n
    integral_constraint = cp.sum(x) * (2.0 / n) == 1.0
    
    # Bounds
    bounds = [0 <= x, x <= 1]
    
    # Add shift constraints
    constraints = bounds + [integral_constraint]
    dx = 2.0 / n
    
    # Build constraints efficiently with pre-computed shift matrices
    # Build a sparse matrix for each shift s that computes sum_i x_i * (1 - x_{i+s})
    # For shift s, compute indices i, i+s (with zero padding beyond boundaries)
    shift_exprs = []
    for s in shift_set:
        if s == 0:
            expr = dx * cp.sum(x)
        else:
            # Construct sparse matrix for mapping purposes
            # We only do explicit computation
            expr = 0.0
            for i in range(m):
                right_i = i + s
                if right_i < m:
                    expr += x[i] * (1 - x[right_i])
                else:
                    # For out-of-range indices, the corresponding h value is zero
                    expr += x[i] * 1.0
            expr = expr * dx  # Scale by dx
        shift_exprs.append(expr)
    
    c_constraints = [expr <= t for expr in shift_exprs]
    constraints += c_constraints
    
    # Solve using ECOS
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=2000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_val = x.value
            t_val = t.value
        else:
            # fallback to a good feasible point if optimization failed
            return h_initial.copy(), float('inf')
    except Exception:
        return h_initial.copy(), float('inf')
        
    h_full = sym_to_full(x_val)
    return h_full, t_val

def correlation_vals(h):
    """Compute correlation values of h with (1-h)."""
    n = len(h)
    # full linear correlation (no periodic wrap)
    corr = np.correlate(h, 1 - h, mode='full')
    # We return the non-negative shifts (1-indexed in the middle)
    # corr index 0 corresponds to shift -(n-1)
    return corr[n - 1:] * (2.0 / n)

            c5_best = c5_final
            h_best = h_current.copy()
    
    return (h_best, c5_best, n)
