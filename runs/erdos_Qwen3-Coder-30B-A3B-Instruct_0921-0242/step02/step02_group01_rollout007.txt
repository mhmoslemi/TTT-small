import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def compute_autocorrelations_binary(x):
    """Compute autocorrelations for binary vector x."""
    n = len(x)
    # FFT-based convolution for fast computation
    x_fft = fft(x, 2 * n)
    conj_fft = fft(1 - x, 2 * n)
    corr = ifft(x_fft * conj_fft).real
    # Return only the non-trivial autocorrelations (k=1 to n-1)
    return corr[1:n]

def compute_autocorrelations_binary_fast(x):
    """Fast computation of autocorrelations for binary vector x."""
    n = len(x)
    ac = np.zeros(n)
    # Use optimized loop
    for k in range(1, n):
        # Compute R_k = sum x_i * x_{i+k}
        for i in range(n):
            j = (i + k) % n
            ac[k] += x[i] * x[j]
    return ac

def round_sdp_solution(X, v, n, R=30):
    """Round SDP solution to binary vector via random hyperplane and deterministic."""
    # Eigen-decomposition
    try:
        eigenvals, eigenvecs = np.linalg.eigh(X)
        idx = np.argsort(eigenvals)[::-1]
        eigenvals = eigenvals[idx]
        eigenvecs = eigenvecs[:, idx]
    except Exception:
        # fallback to deterministic rounding
        if np.allclose(v, 0):
            return np.zeros(n)
        sorted_idx = np.argsort(-v)
        x_rounded = np.zeros(n)
        x_rounded[sorted_idx[:n//2]] = 1
        return x_rounded
    
    best_x = None
    best_min_ac = -1
    
    # Deterministic rounding using the principal component
    sorted_idx = np.argsort(-v)
    x_det = np.zeros(n)
    x_det[sorted_idx[:n//2]] = 1
    ac = compute_autocorrelations_binary_fast(x_det)
    min_ac = np.min(ac)
    if min_ac > best_min_ac:
        best_min_ac = min_ac
        best_x = x_det.copy()
    
    # Random hyperplane rounding
    for _ in range(R):
        g = np.random.randn(n)
        z = eigenvecs @ (np.sqrt(np.maximum(eigenvals, 0)) * g)
        sorted_idx = np.argsort(-z)
        x_test = np.zeros(n)
        x_test[sorted_idx[:n//2]] = 1
        ac = compute_autocorrelations_binary_fast(x_test)
        min_ac = np.min(ac)
        if min_ac > best_min_ac:
            best_min_ac = min_ac
            best_x = x_test.copy()
    
    return best_x if best_x is not None else np.zeros(n)

def swap_improvement(x, ac, n, max_iter=5000):
    """Apply greedy swap refinement to improve the worst autocorrelation."""
    # Copy the input to avoid modifying the original
    x = x.copy()
    ac = ac.copy()
    iter_count = 0
    
    # Precompute shift contributions for efficiency
    contrib_1 = {}
    contrib_0 = {}
    
    while iter_count < max_iter:
        # Find the worst shift
        k_star = np.argmin(ac) + 1  # k=1 to n-1
        iter_count += 1
        
        # Update contributions for this k_star
        if k_star not in contrib_1:
            contrib_1[k_star] = np.zeros(n)
            contrib_0[k_star] = np.zeros(n)
            for i in range(n):
                j = (i + k_star) % n
                k = (i - k_star) % n
                contrib_1[k_star][i] = x[j] + x[k]
        if k_star not in contrib_0:
            contrib_0[k_star] = np.zeros(n)
            for j in range(n):
                i = (j + k_star) % n
                l = (j - k_star) % n
                contrib_0[k_star][j] = x[i] + x[l]
        
        # Get contribution vectors
        c_1 = contrib_1[k_star]
        d_0 = contrib_0[k_star]
        
        # Find indices with x=1 and x=0
        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # n_select = min(len(ones), 10)
        # max_c1 = np.sort(c_1[ones])[-n_select:], indices = np.argsort(c_1[ones])[-n_select:]
        # n_select = min(len(zeros), 10)
        # min_d0 = np.sort(d_0[zeros])[:n_select], indices = np.argsort(d_0[zeros])[:n_select]
        
        max_c1 = np.max(c_1[ones])
        min_d0 = np.min(d_0[zeros])
        
        if min_d0 <= max_c1:
            break
        
        # Find the specific indices to swap
        i_bad = np.argmax(c_1[ones])
        j_bad = np.argmin(d_0[zeros])
        
        i = ones[i_bad]
        j = zeros[j_bad]
        
        # Perform the swap
        x[i] = 0
        x[j] = 1
        
        # Update ac values efficiently
        # Remove old contributions of x[i] and x[j]
        old_k = k_star
        for k in range(1, n):
            if k == old_k:
                # For the specific shift, actually update
                if k != 0:
                    # Remove old contributions
                    # Contribution of x[i], x_{i+k}, x_{i-k}
                    if (i + k) % n < n:
                        ac[k] -= x[(i + k) % n]
                    if (i - k) % n < n:
                        ac[k] -= x[(i - k) % n]
                    # Add new contributions
                    if (j + k) % n < n:
                        ac[k] += x[(j + k) % n]
                    if (j - k) % n < n:
                        ac[k] += x[(j - k) % n]
            else:
                # For all other k, update near self
                pass  # For simplicity, ignore other updates in this step by step method
        
        # Recompute ac for all shifts for accuracy
        # But skip the O(n²) recomputation here to save time; try to update precisely
        
        # Actually, we use a more accurate recomputation at the end for each iteration
        # update just the current shift, the main interference
        
        # In hope of simple update, recompute ac from scratch for every few iterations
        # or account only for disturbing terms, not full recalculation
        
        # For large accuracy, we just recompute (trade-precise vs time)  
        # Do not do precise update here due to computational complexity; just return good result
        
        # Alternative: recalculate for more precise handling
        # But for 200 length, just recompute ac every 50 or so steps at most
        # Given the high importance of keeping it correct for the algorithm's behavior
        
        ac = compute_autocorrelations_binary_fast(x)
        if np.min(ac) >= np.min(ac):
            pass  # accept the step
    return x

def solve_sdp_round_and_refine(seed, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose n to be even and small enough to solve in time
    n = 200
    dx = 2.0 / n
    
    # Variable for SDP relaxation
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [X >> 0]  # X is psd
    constraints += [X[i, i] >= 0 for i in range(n)]  # 0 <= X_{ii} <= 1
    constraints += [X[i, i] <= 1 for i in range(n)]  # 0 <= X_{ii} <= 1
    constraints += [cp.sum(X.diagonal()) == n/2]  # Trace = n/2
    
    # Matrix A_k for each k (shift)
    # Compute autocorrelation lower bound for all k ≠ 0
    # A_k[i, j] = 1 iff x_i = 1 and x_{i+k} = 1 (mod n)
    
    # Shift matrix for shift k
    for k in range(1, n):
        if time.time() - start_time > 0.7 * budget_s:
            break
        A_k = np.zeros((n, n))
        for i in range(n):
            j = (i + k) % n
            A_k[i, j] = 1
        constraints += [cp.trace(A_k @ X) >= t]
    
    # Pairwise bounds (tighten relaxation)
    for i in range(n):
        for j in range(i+1, n):
            if time.time() - start_time > 0.7 * budget_s:
                break
            constraints += [X[i, j] >= 0]
            constraints += [X[i, j] <= X[i, i]]
            constraints += [X[i, j] <= X[j, j]]
            constraints += [X[i, j] <= 1]
    
    # Objective
    objective = cp.Maximize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Choose solver
    solver = cp.CBC  # can be changed if available
    try:
        prob.solve(solver=solver, verbose=False, max_seconds=0.6 * budget_s)
    except Exception:
        # fallback to simpler solver or smaller size
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_seconds=0.5 * budget_s)
        except Exception:
            # reduce to a much smaller size
            n = 100
            dx = 2.0 / n
            X = cp.Variable((n, n), symmetric=True)
            t = cp.Variable()
            
            constraints = [X >> 0]
            constraints += [X[i, i] >= 0 for i in range(n)]
            constraints += [X[i, i] <= 1 for i in range(n)]
            constraints += [cp.sum(X.diagonal()) == n/2]
            
            for k in range(1, n):
                A_k = np.zeros((n, n))
                for i in range(n):
                    j = (i + k) % n
                    A_k[i, j] = 1
                constraints += [cp.trace(A_k @ X) >= t]
                
            # Pairwise bounds for small size are removed to speed up
    
            constraints += [X[i, j] >= 0 for i in range(n) for j in range(i+1, n)]
            constraints += [X[i, j] <= X[i, i] for i in range(n) for j in range(i+1, n)]
            constraints += [X[i, j] <= X[j, j] for i in range(n) for j in range(i+1, n)]
            
            objective = cp.Maximize(t)
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.ECOS, verbose=False, max_seconds=0.5 * budget_s)
            except Exception:
                return None, float('inf'), n
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        X_opt = X.value
        try:
            v = np.sqrt(np.linalg.eigh(X_opt)[0][-1]) * np.linalg.eigh(X_opt)[1][:, -1]
        except Exception:
            # fallback round
            v = np.zeros(n)
            v[0] = 1
    else:
        # fallback to random binary vector search
        return None, float('inf'), n
    
    # Rounding
    x_candidate = round_sdp_solution(X_opt, v, n)
    
    # Ensure cardinality is exactly n/2
    ones_count = np.sum(x_candidate)
    if abs(ones_count - n/2) > 0.2:
        # simple rounding to balance number of ones
        x_candidate = np.zeros(n)
        sorted_idx = np.argsort(-v)
        x_candidate[sorted_idx[:n//2]] = 1
    
    # Improved binary refinement using swap logic
    ac = compute_autocorrelations_binary_fast(x_candidate)
    
    # Refine using swap-based approach
    h_refined = swap_improvement(x_candidate, ac, n, max_iter=int(0.2 * budget_s))
    
    # Optional polishing
    h_polished = h_refined.copy()
    c5_initial = evaluate_erdos_solution(h_polished)
    
    if time.time() - start_time < 0.95 * budget_s:
        # substitute back the real computation
        current_c5 = evaluate_erdos_solution(h_polished)
        
        for k in range(2000):
            if time.time() - start_time > 0.95 * budget_s:
                break
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if h_polished[i] <= 0 or h_polished[j] >= 1:
                continue
            delta = np.random.uniform(0, min(h_polished[i], 1 - h_polished[j], 0.01))
            h_test = h_polished.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, n)
            c5_test = evaluate_erdos_solution(h_test)
            if c5_test < current_c5:
                current_c5 = c5_test
                h_polished = h_test
    
    # Final evaluation with fast FFT-based computation
    final_c5 = evaluate_erdos_solution(h_polished)
    
    return h_polished, final_c5, n

def run(seed=42, budget_s=600, **kwargs):
    h_values, c5_bound, n_points = solve_sdp_round_and_refine(seed, budget_s)
    
    if h_values is None:
        # fallback to simpler method
        np.random.seed(seed)
        n = 200
        h_start = np.random.rand(n)
        h_start = h_start * (n / 2) / np.sum(h_start)
        h_start = project_to_feasible(h_start, n)
        
        # simple local improvement with edges swapping from high density to low
        h_refined = h_start.copy()
        c5_old = evaluate_erdos_solution(h_refined)
        for _ in range(20000):
            if time.time() - start_time > budget_s * 0.95:
                break
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if h_refined[i] <= 0 or h_refined[j] >= 1:
                continue
            delta = np.random.uniform(0, min(h_refined[i], 1 - h_refined[j], 0.01))
            h_test = h_refined.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, n)
            c5_test = evaluate_erdos_solution(h_test)
            if c5_test < c5_old:
                c5_old = c5_test
                h_refined = h_test
        h_values, c5_bound, n_points = h_refined, c5_old, n
        
    return h_values, c5_bound, n_points
