import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp
from scipy.linalg import toeplitz

def project_to_feasible(h, target_sum):
    """Ensure sum(h) = target_sum with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Evaluates the C5 constant given step function h."""
    dx = 2.0 / len(h)
    # Full correlation with zero-padding
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # Take real part and maximum of non-zero shifts
    max_corr = np.max(corr.real[1:len(h)]) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 256  # Even number for stability in FFT
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Phase 1: SDP Bisection for largest feasible t
    t_low, t_high = 0.0, 1.0
    best_feasible_t = t_low
    best_feasible_r = None
    remaining_time = budget_s * 0.35

    while (t_high - t_low) > 2e-4 and (time.time() - start_time < remaining_time):
        t_mid = (t_low + t_high) / 2.0

        # Construct SDP problem
        r = cp.Variable(n_points)
        constraints = [r[0] == 1.0,
                       r[1:] >= t_mid,
                       r[1:] <= 1.0]
        
        # Construct Toeplitz matrix from r
        T = toeplitz(r)
        constraints.append(T >> 0)  # PSD constraint
        
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                best_feasible_t = t_mid
                best_feasible_r = r.value
                t_low = t_mid
            else:
                t_high = t_mid
        except Exception:
            t_high = t_mid

    if best_feasible_r is None:
        # Fallback in case no SDP was solved
        r = np.ones(n_points)
        r[0] = 1
        r[1:] = 0.5
        best_feasible_r = r

    # Phase 2: Spectral Factorisation and Projection
    r = best_feasible_r
    s = np.fft.fft(r)  # Power spectrum
    s = np.maximum(s, 0.0)  # Ensure non-negative
    h_fft = np.sqrt(s)
    h = np.real(np.fft.ifft(h_fft))  # good estimate but probably not feasible
    h = project_to_feasible(h, target_sum)

    # Phase 3: Worst-shift Gradient Descent
    num_refine_iter = 250
    step_size = 0.5 * dx
    for i in range(num_refine_iter):
        if time.time() - start_time > budget_s * 0.7:
            break
        # Compute full correlation
        corr = np.real(np.fft.ifft(fft(h, 2 * n_points) * np.conj(fft(1 - h, 2 * n_points))))
        max_shift = np.argmax(corr[1:])
        if max_shift == 0:
            max_shift = 1
        # Compute gradient
        g = np.zeros(n_points)
        for k in range(n_points):
            g[k] = dx * ((1 - h[(k + max_shift) % n_points]) - h[(k - max_shift) % n_points])
        
        h_new = h - step_size * g
        h = project_to_feasible(h_new, target_sum)
        step_size = 0.5 * dx / np.sqrt(1 + i)

    # Phase 4: Stochastic Pair-Transfer Polishing
    num_mc_steps = 5000
    for i in range(num_mc_steps):
        if time.time() - start_time > budget_s * 0.95:
            break
        i_idx = np.random.choice(np.where(h > 0)[0])
        j_idx = np.random.choice(np.where(h < 1)[0])
        if i_idx == j_idx:
            continue
        delta = min(0.01, h[i_idx], 1 - h[j_idx])
        h_test = h.copy()
        h_test[i_idx] -= delta
        h_test[j_idx] += delta
        h_test = project_to_feasible(h_test, target_sum)
        c5_old = evaluate_c5(h)
        c5_new = evaluate_c5(h_test)
        if c5_new < c5_old:
            h = h_test

    # Final evaluation
    final_c5 = evaluate_c5(h)

    return (h, final_c5, n_points)
