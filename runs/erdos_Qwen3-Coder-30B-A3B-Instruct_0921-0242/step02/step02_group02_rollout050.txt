import numpy as np
import time
from scipy.optimize import differential_evolution, linprog
from scipy.fft import fft, ifft
from scipy.interpolate import interp1d

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def basis_evaluation(x, M):
    """Evaluate cosine and sine basis functions at points x."""
    basis = np.zeros((len(x), 2 * M))
    for m in range(1, M + 1):
        basis[:, 2*(m-1)] = np.cos(m * np.pi * x)
        basis[:, 2*(m-1)+1] = np.sin(m * np.pi * x)
    return basis

def sample_from_coefficients(c, n=250, M=12):
    """Sample a function from coefficients."""
    x = np.linspace(0, 2, n, endpoint=False)
    basis = basis_evaluation(x, M)
    h_raw = 0.5 + basis @ c
    h_raw = np.clip(h_raw, 0, 1)
    h_scaled = h_raw * (n / 2) / np.sum(h_raw)
    return h_scaled

def de_fitness(c, M, n):
    """Fitness function for differential evolution."""
    h = sample_from_coefficients(c, n, M)
    return evaluate_c5(h)

def linearise_and_solve_lp(h, M, n, max_constraints=10):
    """Solve the linear programming problem to refine h."""
    # Compute current autocorrelations
    dx = 2.0 / n
    h_fft = fft(h)
    A_fft = np.abs(h_fft)**2
    A = np.real(ifft(A_fft))
    A = A[:n]  # First n values only
    
    # Maximize t such that A(k) >= t for all k
    # This is equivalent to minimizing -t, subject to:
    # A(k) - t >= 0, i.e. t <= A(k)
    # We also have bounds on h and the integral constraint.
    # For each constraint, we linearize it:
    # A(k) ≈ A(k_0) + ∇A(k)^T (h - h_0)
    # where h_0 is the current point
    
    # We want to activate the worst constraints i.e. those closest to the minimum autocorrelation
    k_star = np.argmin(A)
    min_corr = A[k_star]
    
    # Select top worst shifts
    worst_indices = np.argsort(A)[-max_constraints:]
    # Remove duplicate indices and ensure k_star is included if not there
    if k_star not in worst_indices:
        worst_indices = np.append(worst_indices, k_star)
        worst_indices = np.unique(worst_indices)
    
    # Build LP problem
    # Variables are h[0], ..., h[n-1], t
    num_vars = n + 1
    
    # Objective: maximize t
    # We solve min -t by adding a unit variable
    c_lp = np.zeros(num_vars)
    c_lp[-1] = -1  # coefficient for t
    
    # Inequality constraints: A(k) >= t
    # For k in worst_indices, we approximate A(k) ≈ A[k] + gradient^T * (h - h0)
    # Let's compute A(k) in matrix form
    A_ineq = np.zeros((len(worst_indices), num_vars))
    b_ineq = np.zeros(len(worst_indices))
    
    # Gradient computation: dA(k)/dh = 2 * roll(h, k)
    for i, k in enumerate(worst_indices):
        # Compute gradient vector d/dh A(k) at current h
        # This is 2 * roll(h, k) but we work with indexed variables
        gradient = np.zeros(n)
        # Rotate h to apply shift k
        shifted = np.roll(h, k)
        gradient = 2 * shifted
        
        # Construct constraint matrix row
        A_ineq[i, :n] = gradient
        A_ineq[i, n] = -1  # coefficient of t
        # since dA(k) >= t, we get dA(k) - t >= 0 
        b_ineq[i] = A[k]  # This is the linearization using A[k]
    
    A_eq = np.zeros((2, num_vars))
    A_eq[0, :n] = 1  # integral constraint: sum h = n/2
    A_eq[1, :n] = 1  # weight of h in p
    b_eq = np.array([n/2, 0])  # Fixed sum, normalize correction later
    
    bounds = [(0, 1)]*n + [(-np.inf, np.inf)]  # bounds on h and t
    
    # We don't have the ii'th variable fixed to its average value, so the second
    # equality doesn't make sense for an LP. We must rethink this.
    # We correct: integrate = n/2. We solve it correctly:
    # SUM h[i] = n/2
    # and 0 <= h[i] <= 1
    # So A_eq should be only:
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1  # sum h = n/2
    b_eq = np.array([n/2])
    
    # Solve LP
    lp_result = linprog(c=c_lp, A_ub=-A_ineq, b_ub=-b_ineq, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs', options={"presolve": False})
    
    if lp_result.success:
        # Extract h solution from the result
        h_new = lp_result.x[:-1]
        t_new = lp_result.x[-1]
        
        # Compute exact C5 for the new h
        c5_new = evaluate_c5(h_new)
        return h_new, c5_new
    else:
        # Failed to solve, return original h
        return h, evaluate_c5(h)

def subgradient_step(h, n):
    """Perform a subgradient-style update on h. """
    # Find worst shift index
    corr = np.correlate(h, 1 - h, mode='full')
    k_star = np.argmax(corr)
    
    # Compute gradient (approximate)
    # Approximate gradient of A_k(h) at h: 2 * roll(h, k_star)
    grad = np.roll(h, k_star)
    grad = 2 * grad
    
    # Try to move mass from high values to low values, in the direction of the gradient
    # Also add small random perturbation to avoid convergence issues
    update = grad * 1e-4 + np.random.normal(0, 1e-5, len(h))
    
    h_new = h - update
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_feasible(h_new, n)
    return h_new
