import numpy as np
import time
from scipy.optimize import differential_evolution, linprog
from scipy.fft import rfft, irfft
from scipy.interpolate import interp1d

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    # Water-filling approach
    diff = n/2 - current_sum
    if diff > 0:
        # Add uniform offset
        h += diff / n
    else:
        # Subtract uniform offset
        h += diff / n
    h = np.clip(h, 0, 1)
    scale = (n/2) / np.sum(h)
    h *= scale
    return h

def evaluate_c5(h):
    """Evaluates C5 for given step function h discretised over [0,2] with dx = 2/n."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def basis_evaluation(x_vals, coeffs, M):
    """Evaluate a truncated Fourier series with cosine/sine coefficients."""
    # First term is a0 = 0.5
    result = 0.5 * np.ones_like(x_vals)
    for m in range(1, M + 1):
        a_m = coeffs[2*(m-1)]
        b_m = coeffs[2*(m-1)+1]
        result += a_m * np.cos(m * np.pi * x_vals) + b_m * np.sin(m * np.pi * x_vals)
    return result

def compute_autocorr(h):
    """Compute autocorrelation from a step function using FFT."""
    n = len(h)
    H = rfft(h)
    A = irfft(H * np.conj(H), n)
    return A

def evaluate_h_on_basis(coeffs, M, n):
    """Construct h(x) from basis coefficients and return sampled vector."""
    x_vals = np.linspace(0, 2, n, endpoint=False)
    h_raw = basis_evaluation(x_vals, coeffs, M)
    h_raw = np.clip(h_raw, 0, 1)
    h_projected = project_to_feasible(h_raw, n)
    return h_projected

def fitness_function(coeffs, M, n0):
    """Evaluate the fitness of a coefficient vector (i.e., -C5)."""
    h = evaluate_h_on_basis(coeffs, M, n0)
    c5 = evaluate_c5(h)
    return c5

def linearise_and_solve_lp(h0, A0, k_star, n0):
    """
    Solve a linearised LP subproblem to improve h.
    
    Args:
        h0: Current h vector
        A0: Autocorrelation values A(k) for all k
        k_star: Index of worst shift (where min A(k) is achieved)
        n0: size of h
    Returns:
        (h_new, t_new) where t_new is the improved lower bound on A(k).
    """
    # Construct LP problem
    # max t
    # s.t. A0[k] + (P_k + P_k.T) @ (h - h0) >= t for some k
    #      sum(h) = n0/2
    #      0 <= h <= 1
    
    # Select a few key constraints (active or near-active)
    active_k = []
    for i in range(n0):
        if abs(A0[i] - A0[k_star]) < 1e-8:
            active_k.append(i)
    # Also include some nearby ones
    all_candidates = set(active_k)
    for i in range(10):
        for k in [k_star + i, k_star - i]:
            if 0 <= k < n0:
                all_candidates.add(k)
    
    active_k = list(all_candidates)
    active_k.sort()
    
    # Fix M = 1: Remove duplicates and limit number
    if len(active_k) > 10:
        active_k = active_k[:10]
    
    # Set up the A matrix (each row is the linearisation of A(k))
    # A(k) = h^T P_k h, where P_k is a permutation matrix
    # For small shift, P_k h with k shifts x elements of h cyclically
    # Linear term: grad A(k) = P_k h + P_k.T h = 2 * roll(h, k)
    # The constraint is (P_k h + P_k.T h)^T * (h - h0) >= t - A0[k]
    # i.e., (2 * roll(h, k))^T * (h - h0) >= t - A0[k]
    # i.e., (2 * roll(h, k))^T h - (2 * roll(h, k))^T h0 >= t - A0[k]
    # i.e., -t >= (2 * roll(h, k))^T h0 - A0[k] - (2 * roll(h, k))^T h
    
    # Simple encoding for the LP
    n_vars = n0 + 1  # h + t
    c = np.concatenate([np.zeros(n0), [-1]])  # maximize t
    A_ub = []
    b_ub = []
    
    for k in active_k:
        # grad A(k) = 2 * roll(h0, k) 
        # Constraint: (grad A(k))^T h - (grad A(k))^T h0 >= t - A0[k]  
        # i.e., -t + (grad A(k))^T h >= -A0[k] + (grad A(k))^T h0
        grad = 2 * np.roll(h0, k)
        row = np.concatenate([grad, [-1]])
        A_ub.append(row)
        b_ub.append(-A0[k] + np.dot(grad, h0))
        
    # Bounds
    bounds = [(0, 1)] * n0 + [None, None]  # h in [0,1], t unbounded
    A_eq = np.zeros((1, n0 + 1))
    A_eq[0, :n0] = 1
    b_eq = [n0 / 2]
    
    # Solve using linprog
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if result.success:
            h_new = np.clip(result.x[:n0], 0, 1)
            t_new = result.x[-1]
            return h_new, t_new
        else:
            # Fallback: return original
            return h0.copy(), np.max(A0)  # Would be precise, but no improvement
    except:
        return h0.copy(), np.max(A0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    total_time = budget_s

    # Phase A: Setup basis functions (small reference grid)
    n0 = 250  # Medium size grid
    dx0 = 2.0 / n0
    M = 12  # Number of cosine/sine terms - relatively small for low dim
    d = 2 * M  # dimensionality of coefficient space

    # Phase B: Global DE search
    def fit_func(coeffs):
        return fitness_function(coeffs, M, n0)
    
    # DE bounds [-0.5, 0.5] for each coefficient
    bounds = [(-0.5, 0.5)] * d

    # DE settings
    popsize = max(15, 3 * d)  # 15 * d but not less than 15 
    maxiter = int(3000 / popsize)  # number of iterations to do 
    if maxiter < 1:
        maxiter = 1
    # Allocate time for DE
    de_time = 0.5 * total_time
    de_start = time.time()
    
    # Run DE
    result = differential_evolution(fit_func, bounds, maxiter=maxiter, popsize=popsize, seed=seed, disp=False)
    best_coeffs = result.x

    best_h = evaluate_h_on_basis(best_coeffs, M, n0)
    best_c5 = evaluate_c5(best_h)
    
    # Phase C: Select top 5 candidates from DE
    # Just one for now
    candidates = [best_coeffs]
    
    # Phase D: SLP refinement per candidate
    refinement_start = time.time()
    slp_time = 0.3 * total_time
    
    refined_h = best_h.copy()
    refined_c5 = best_c5
    
    # Work through the candidates
    for i, c in enumerate(candidates):
        if time.time() - refinement_start > slp_time or (time.time() - start_time > total_time - 0.5):
            break
            
        h_current = evaluate_h_on_basis(c, M, n0)
        c5_current = evaluate_c5(h_current)
        
        # Run SLP refinement iteratively until convergence or timeout
        max_iters = 400
        for it in range(max_iters):
            if time.time() - refinement_start > slp_time or (time.time() - start_time > total_time - 0.5):
                break
                
            A_vals = compute_autocorr(h_current)
            k_star = np.argmin(A_vals)
            # Compute A_vals = h_current^T P_k h_current for each k
            # But we already computed A_vals from the FFT
            
            # Linearise and solve the LP (sequential linear programming)
            h_new, t_new = linearise_and_solve_lp(h_current, A_vals, k_star, n0)
            
            # Compute exact C5 for h_new - re-evaluate
            c5_new = evaluate_c5(h_new)
            
            # Accept if there's improvement, else try a small damped step
            if c5_new < c5_current - 1e-8:
                h_current = h_new
                c5_current = c5_new
            else:
                # Small hybrid step - interpolate slightly towards h_new
                h_current = 0.9 * h_current + 0.1 * h_new
                h_current = project_to_feasible(h_current, n0)
                c5_current = evaluate_c5(h_current)  # Re-evaluate

        # If refined version is better, keep it
        if c5_current < refined_c5:
            refined_h = h_current
            refined_c5 = c5_current

    # Phase E: Upsampling and fine-tuning
    # Target grid size
    n_final = min(800, 500)
    if n_final > n0:
        # Interpolate using cubic with periodic boundary
        x_orig = np.linspace(0, 2, n0, endpoint=False)
        x_upsampled = np.linspace(0, 2, n_final, endpoint=False)
        
        # Reconstruct high-res version using basis, then interpolate
        # We have an analytic representation so sample it accurately
        
        h_upsampled = evaluate_h_on_basis(best_coeffs, M, n_final)
        
        # Final polishing with subgradients
        h_polished = h_upsampled.copy()
        c5_polished = evaluate_c5(h_polished)
        
        max_polish_steps = 2000
        polish_start = time.time()
        for step in range(max_polish_steps):
            if time.time() - polish_start > 0.1 * total_time or (time.time() - start_time > total_time - 0.5):
                break
                
            # Re-evaluate correlation to find worst k
            corr = np.correlate(h_polished, 1 - h_polished, mode='full')
            k_worst = np.argmax(corr)  # high overlap is bad
            
            # Negative gradient (to reduce overlap)
            # Amount to move in implicit direction of gradient of overlap integral
            # This step just approximates a one-step sub-gradient update
            
            # Simulation of sub-gradient steps
            h_old = h_polished.copy()
            # Compute shift k_worst
            # Use linear interpolation effect: look at the shift
            # and transfer "some" mass from idx_i where h[i] is higher to where h[j] is lower
            # Simple: gradually move from high to low points
            
            # klargest = len(corr) - k_worst - 1 if k_worst >= len(corr)//2 else k_worst
            # So for k_worst, we want to shift that many positions forward  
            start_idx = 10  # arbitrary sample start
            end_idx = n_final - 10
            
            # Sub-gradient approximation: we move mass along shift k_worst
            # e.g., subtract from index i, add to index (i + k_worst) % n
            
            # Find where to move mass: move from high to low, possibly using gradient 
            # Here just do pair moving
            for i in range(start_idx, end_idx):
                j = (i + k_worst) % n_final
                if h_old[i] < 0.001 or h_old[j] > 0.999:
                    continue
                delta = min(0.001, h_old[i], 1 - h_old[j])
                h_polished[i] -= delta
                h_polished[j] += delta
                
            # Project to feasibility  
            h_polished = project_to_feasible(h_polished, n_final)
            new_c5 = evaluate_c5(h_polished)
            if new_c5 < c5_polished - 1e-10:
                c5_polished = new_c5
            if abs(new_c5 - c5_polished) < 1e-8:
                # Premature convergence
                break
            
        # Final output 
        final_h = h_polished
        final_c5 = c5_polished
    else:
        final_h = refined_h
        final_c5 = refined_c5
    
    n_points = len(final_h)
    return (final_h, final_c5, n_points)
