import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cma

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def memetic_descent(h, n_points, max_steps=25):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        # The maximum is at shift 0, so we need to find the min among non-zero shifts
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        # Gradient wrt h[i] is dx * (1 - h[i + worst_shift_idx] - h[i - worst_shift_idx])
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def setup_symmetric_piecewise_params(M):
    """Set up the initial parameter vector for a symmetric piecewise constant function with M segments."""
    # Initial widths (soft-maxed)
    widths = np.ones(M)
    # Initial heights (logistic)
    heights = np.ones(M) * -5.0  # Most negative values gives small values
    
    # Force middle 3 blocks to be larger
    widths[M//2-1:M//2+2] = 5.0  
    
    # Middle heights should be large (close to 1)
    heights[M//2-1:M//2+2] = 5.0
    
    # Flatten to get parameter vector of 2*M values
    param_vector = np.concatenate([widths, heights])
    return param_vector.copy()

def params_to_h(params, n_points, M):
    """Convert parameters to a symmetric piecewise constant function h."""
    # Extract widths and heights
    widths_raw = params[:M]
    heights_raw = params[M:]
    
    # Apply transformations
    widths = np.exp(widths_raw) / np.sum(np.exp(widths_raw))
    heights = 1.0 / (1.0 + np.exp(-heights_raw))
    
    # Build left half
    h_half = np.zeros(n_points // 2)
    cumulative = 0.0
    for i in range(M):
        start_idx = int(np.round(cumulative * (n_points // 2)))
        end_idx = int(np.round((cumulative + widths[i]) * (n_points // 2)))
        if i == M - 1:  # Ensure last segment hits the end
            end_idx = n_points // 2
        h_half[start_idx:end_idx] = heights[i]
        cumulative += widths[i]
    
    # Mirror to get full h
    h_full = np.concatenate([h_half, h_half[::-1]])
    
    return h_full

def get_constraints(n_points, M):
    """Return constraints for the factory."""
    def project(params):
        h = params_to_h(params, n_points, M)
        h = project_to_feasible(h, n_points)
        return h
    
    return project

def cmaes_optimization(seed, budget_s):
    """Run CMA-ES optimization on the symmetric piecewise function space."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 400
    M = 7
    k = 2 * M  # dimension
    CMA_OPTS = {'popsize': 4 + int(3 * np.log(k)), 'verbose': -9}
    σ = 0.3
    max_evals = 20000  # Limit for safety
    num_restarts = 10
    improved_threshold = 1e-5
    no_improvement_generations = 80
    
    # Initial parameters
    x0 = setup_symmetric_piecewise_params(M)
    
    # Setup constraints
    project = get_constraints(n_points, M)

    # CMA-ES control variables
    best_c5 = float('inf')
    best_h = None
    restart_count = 0
    g_noimp = 0
    λ = CMA_OPTS['popsize']
    μ = λ // 2
    
    # CMA-ES loop
    try:
        # Initial run
        es = cma.CMAEvolutionStrategy(x0, σ, CMA_OPTS)
        
        while not es.stop() and restart_count <= num_restarts and time.time() < start_time + budget_s - 30:
            # Generate offspring
            X = es.ask()
            fitness = []
            
            # Evaluate each individual
            for x in X:
                try:
                    h = params_to_h(x, n_points, M)
                    h = project(h)
                    c5 = evaluate_c5(h)
                    fitness.append(c5)
                except Exception:
                    fitness.append(float('inf'))
            
            # Tell ES new fitness values
            es.tell(X, fitness)
            
            # Update best solution
            best_idx = np.argmin(fitness)
            if fitness[best_idx] < best_c5:
                best_c5 = fitness[best_idx]
                best_x = X[best_idx]
                h = params_to_h(best_x, n_points, M)
                best_h = project(h)
                g_noimp = 0
            else:
                g_noimp += 1

            # Check for restart
            if g_noimp >= no_improvement_generations and restart_count < num_restarts:
                restart_count += 1
                λ = min(λ * 2, 100)  # At most 100
                σ = 0.3
                print(f"Restart {restart_count} with λ={λ}")
                # Keep best as new mean
                x0 = best_x
                es = cma.CMAEvolutionStrategy(x0, σ, {'popsize': λ, 'verbose': -9})

    except Exception:
        pass
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main function to run the optimization."""
    start = time.time()
    
    # Run CMA-ES
    h_best, c5_best = cmaes_optimization(seed, budget_s)
    
    # Polishing
    if h_best is not None:
        h_best = memetic_descent(h_best, 400, max_steps=25)
        c5_best = evaluate_c5(h_best)
    
    if h_best is None:
        # Fallback back to simple initialization if needed
        h_best = np.ones(400) * 0.5
        c5_best = evaluate_c5(h_best)
    
    return (h_best, c5_best, 400)
