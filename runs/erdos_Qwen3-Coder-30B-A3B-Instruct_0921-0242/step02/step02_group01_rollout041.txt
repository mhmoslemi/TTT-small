import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Computes C5 for a given h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def compute_autocorrelations(x):
    """Compute all autocorrelations of a binary vector using FFT."""
    n = len(x)
    x_fft = fft(x, 2 * n)
    xx_fft = fft(1 - x, 2 * n)
    corr = ifft(x_fft * np.conj(xx_fft))
    return corr.real

def compute_min_autocorr(h):
    """Compute minimum non-trivial autocorrelation of h."""
    n = len(h)
    if n <= 1:
        return 0
    # Compute correlation
    corr = compute_autocorrelations(h)
    # Minimize over non-zero shifts
    min_val = np.min(corr[1:n])
    return min_val

def solve_sdp_binary(n, max_time_s=180):
    """Solve SDP relaxation of binary max-min problem."""
    # Create SDP variables
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints for SDP relaxation
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n/2,
        t <= 1  # to ensure it's feasible to start
    ]
    
    # Add pairwise bounds
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(X[i,j] >= 0)
            constraints.append(X[i,j] <= X[i,i])
            constraints.append(X[i,j] <= X[j,j])
    
    # Add autocorrelation constraints for each shift k
    for k in range(1, n):
        # Compute A_k - matrix with 1s at (i, (i+k)%n)
        A_k = np.zeros((n, n))
        for i in range(n):
            A_k[i, (i+k) % n] = 1
        # Trace constraint
        constraints.append(cp.trace(A_k @ X) >= t)
    
    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    start_t = time.time()
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
        end_t = time.time()
        if end_t - start_t > max_time_s:
            return None, float("-inf")
        return X.value, t.value
    except Exception as e:
        return None, float("-inf")

def random_round_sdp(X_sol, n, iterations=30):
    """Random hyperplane rounding of SDP solution."""
    # Eigen-decomposition: X = V * Lambda * V^T
    eigvals, eigvecs = np.linalg.eigh(X_sol)
    v = np.sqrt(eigvals[-1]) * eigvecs[:, -1]
    
    best_x = None
    best_mincorr = float("-inf")
    
    for _ in range(iterations):
        # Generate random Gaussian vector
        g = np.random.randn(n)
        z = eigvecs @ (eigvals ** 0.5) * g
        
        # Round to binary
        x = np.zeros(n)
        idx = np.argsort(z)[::-1][:n//2]
        x[idx] = 1
        
        # Compute minimum correlation
        mincorr = compute_min_autocorr(x)
        if mincorr > best_mincorr:
            best_mincorr = mincorr
            best_x = x.copy()
            
    return best_x

def deterministic_round_sdp(X_sol, n):
    """Eigenvector threshold rounding."""
    eigvals, eigvecs = np.linalg.eigh(X_sol)
    v = np.sqrt(eigvals[-1]) * eigvecs[:, -1]
    x = np.zeros(n)
    idx = np.argsort(v)[::-1][:n//2]
    x[idx] = 1
    return x

def apply_swap_refinement(x, n, max_iterations=50000):
    """Refine binary solution using swaps that improve min autocorrelation."""
    # Precompute all shifts
    x_padded = np.concatenate([x, x, x])
    
    # Compute initial autocorrelations
    corr = compute_autocorrelations(x)
    min_corr_idx = np.argmin(corr[1:n]) + 1  # Exclude shift 0
    min_corr_value = corr[min_corr_idx]
    violations = []
    for i in range(n):
        violations.append(corr[i] - min_corr_value)
    
    # Compute contribution vectors: c_i for 1s, d_j for 0s
    def compute_contributions(shift_idx):
        c_i = np.zeros(n)
        d_j = np.zeros(n)
        x_shifted = np.roll(x, shift_idx)
        for i in range(n):
            if x[i] == 1:
                c_i[i] = x_shifted[(i + shift_idx) % n] + x_shifted[(i - shift_idx) % n]
            if x[i] == 0:
                d_j[i] = x_shifted[(i + shift_idx) % n] + x_shifted[(i - shift_idx) % n]
        return c_i, d_j

    start_t = time.time()
    num_swaps = 0
    improved = True

    while improved and time.time() - start_t < 150:  # 30% of total time
        improved = False
        # Current worst shift
        shift_k = min_corr_idx
        c_i, d_j = compute_contributions(shift_k)
        
        # Find the best swap pair (i with highest c_i, j with lowest d_j)
        ones = [i for i in range(n) if x[i] == 1]
        zeros = [i for i in range(n) if x[i] == 0]
        
        if not ones or not zeros:
            break
            
        # Find best candidate indices
        best_i = ones[np.argmax([c_i[i] for i in ones])]
        best_j = zeros[np.argmin([d_j[j] for j in zeros])]
        
        # Compute impact
        delta = d_j[best_j] - c_i[best_i]
        if delta > 0:
            num_swaps += 1
            improved = True
            # Perform the swap
            x[best_i] = 0
            x[best_j] = 1
            
            # Update correlation vector more efficiently by direct update
            # Not modeling this精确 because of caching overhead more than computational cost
            
            # Just recompute the full thing since is needed after several swaps anyway
            corr = compute_autocorrelations(x)
            min_corr_idx = np.argmin(corr[1:n]) + 1
            min_corr_value = corr[min_corr_idx]
            
    return x

def continuous_polish(h, target_min_autocorr, max_iter=200):
    """Fine-tune solution using gradient-based method."""
    dx = 2.0 / len(h)
    for i in range(max_iter):
        # Find worst shift
        corr = compute_autocorrelations(h)
        worst_shift = np.argmin(corr[1:]) + 1
        
        # Compute gradient for current shift
        g = np.zeros_like(h)
        for j in range(len(h)):
            left = (j + worst_shift) % len(h)
            right = (j - worst_shift) % len(h)
            g[j] = dx * (1 - h[left] - h[right])
            
        # Update with step size
        alpha = 0.5 * dx / np.sqrt(i+1)
        h_new = h - alpha * g
        
        # Project back to feasible
        h_new = project_to_feasible(h_new, len(h))
        
        # Check if this helps
        new_min_corr = compute_min_autocorr(h_new)
        if new_min_corr > target_min_autocorr:
            h = h_new
            target_min_autocorr = new_min_corr
        else:
            # Reduce step size
            alpha *= 0.9
            if alpha < 1e-10:
                break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Start with small n to save time
    n = 200
    
    # Phase 1: SDP relaxation
    print("Solving SDP relaxation...")
    X_sol, t_star = solve_sdp_binary(n, max_time_s=180)  # 30% of budget
    
    if X_sol is None:

        print("Falling back to random generation...")
        best_x = None
        best_mincorr = float("-inf")
        for _ in range(1000):
            x = np.random.rand(n)
            x = (x > 0.5).astype(float)  # binary
            x = project_to_feasible(x, n)
            mincorr = compute_min_autocorr(x)
            if mincorr > best_mincorr:
                best_mincorr = mincorr
                best_x = x.copy()
        x = best_x
    else:
        # Rounding phase: random hyperplane and deterministic
        x = deterministic_round_sdp(X_sol, n)
        # Optionally random round to improve
        x_random = random_round_sdp(X_sol, n, iterations=30)
        if compute_min_autocorr(x_random) > compute_min_autocorr(x):
            x = x_random
            
    # Phase 2: Greedy swap refinement
    print("Applying swap refinement...")
    x_final = apply_swap_refinement(x, n, max_iterations=50000)  # 30% of budget
    
    # Phase 3: optional continuous polishing
    print("Polishing with continuous method...")
    target_min_autocorr = compute_min_autocorr(x_final)
    h_final = continuous_polish(x_final, target_min_autocorr, max_iter=200)  # 10% of budget
    
    # Final evaluation
    c5_bound = evaluate_erdos_solution(h_final)
    
    return (h_final, c5_bound, n)
