import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.linalg import toeplitz
import cvxpy as cp

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def compute_c5(h):
    """Computes C5 constant using correlation."""
    dx = 2.0 / len(h)
    corr = np.real(ifft(fft(h) * np.conj(fft(1 - h))))
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Settings
    n = 256
    dx = 2.0 / n
    eps = 2e-4
    t_low, t_high = 0.0, 1.0
    
    # SDP bisection on t
    t = None
    r_best = None
    while t_high - t_low > eps and time.time() - start_time < 0.35 * budget_s:
        t = (t_low + t_high) / 2
        # Create PSD feasibility SDP
        r = cp.Variable(n)
        constraints = [
            r[0] == 1,
            r[1:] >= t,
            r[1:] <= 1
        ]
        # Toeplitz PSD constraint
        T = toeplitz(r)
        constraints.append(T >> 0)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                r_best = r.value
                t_low = t
            else:
                t_high = t
        except:
            t_high = t
        if time.time() - start_time > 0.35 * budget_s:
            break
    
    # If we have a feasible r, do spectral factorization
    if r_best is not None:
        r = r_best
    else:
        r = np.ones(n)
        r[0] = 1
        # fallback to a simple binary search
        for _ in range(20):
            t = (t_low + t_high) / 2
            r = np.ones(n)
            r[0] = 1
            r[1:] = np.clip(r[1:], t, 1)
            T = toeplitz(r)
            if np.all(np.linalg.eigvals(T) >= -1e-6):  # numerical tolerance
                t_low = t
            else:
                t_high = t
        r[0] = 1
        t = t_low
        r[1:] = t
    
    # Compute spectral factorization
    S = np.fft.fft(r)
    S = np.maximum(S, 0)
    H = np.sqrt(S)
    g = np.real(np.fft.ifft(H))
    
    # Project to feasible set
    h = project_to_feasible(g, n)
    
    # Deterministic worst-shift gradient descent
    max_iter = min(250, int(0.35 * budget_s / 0.002) if n == 256 else 250)
    alpha0 = 0.5 * dx
    for iter_num in range(max_iter):
        if time.time() - start_time > 0.7 * budget_s:
            break
        # Compute full correlation using FFT
        corr = np.real(ifft(fft(h) * np.conj(fft(1 - h))))
        worst_shift = np.argmax(corr[1:]) + 1  # Exclude k=0
        # Compute gradient
        grad = np.zeros(n)
        for i in range(n):
            left = (i + worst_shift) % n
            right = (i - worst_shift) % n
            grad[i] = dx * ((1 - h[left]) - h[right])
        # Projected GD step
        alpha = alpha0 / np.sqrt(iter_num + 1)
        h_trial = h - alpha * grad
        h = project_to_feasible(h_trial, n)
    
    # Optional fine-grained stochastic polishing if time allows
    if time.time() - start_time < 0.95 * budget_s:
        for _ in range(1000):
            if time.time() - start_time > 0.95 * budget_s:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if h[i] <= 0 or h[j] >= 1:
                continue
            delta = min(0.01, h[i], 1 - h[j])
            h_test = h.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, n)
            if compute_c5(h_test) < compute_c5(h):
                h = h_test
    
    # Final computation
    c5_bound = compute_c5(h)
    
    return (h, c5_bound, n)
