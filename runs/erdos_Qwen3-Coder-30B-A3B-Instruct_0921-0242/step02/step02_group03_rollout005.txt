import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cma

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def build_symmetric_h(p, q, n_points):
    """Build symmetric piecewise constant h from piecewise parameters."""
    M = len(p)  # number of segments
    # Softmax for widths
    w = np.exp(p)
    w = w / np.sum(w)
    
    # Logistic for heights
    a = 1.0 / (1.0 + np.exp(-q))
    
    # Compute cumulative widths for segment boundaries
    cumw = np.concatenate([[0.0], np.cumsum(w)])
    
    # Build half function
    h_half = np.zeros(n_points // 2)
    dx = 1.0 / (n_points // 2)
    for i in range(n_points // 2):
        x = i * dx
        # Find which segment x belongs to
        segment = np.searchsorted(cumw, x, side='right') - 1
        h_half[i] = a[segment]
    
    # Mirror it to get full
    h_full = np.concatenate([h_half, h_half[::-1]])
    return h_full

def eval_c5_symmetric(params, n_points=400):
    """Evaluate C5 for symmetric piecewise constant parametrization."""
    p = params[:len(params)//2]
    q = params[len(params)//2:]
    
    h = build_symmetric_h(p, q, n_points)
    h = project_to_feasible(h, n_points)
    return evaluate_c5(h)

def memetic_descent(h, n_points, max_steps=25):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)

        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx

        # Find the worst shift index (non-zero)
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1

        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        # Gradient wrt h[i] is dx * (1 - h[i + worst_shift_idx] - h[i - worst_shift_idx])
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 30  # Leave time for polishing
    
    # Parameters
    n_points = 400
    M = 7  # Number of segments
    D = 2 * M  # Dimension
    
    # Initialize mean parameters for CMA-ES
    # Initial widths: give middle three segments more weight
    p_mean = np.array([0.1]*M)
    p_mean[3:6] = 1.0  # Middle thirds
    p_mean /= np.sum(p_mean)
    # Make logit of heights, centered to handle middle more peaked than edges
    q_mean = np.array([1.0] * M)
    q_mean[3:6] = 5.0  # Those in middle should be high
    q_mean[:3] = -5.0  # Edges low
    q_mean[6:] = -5.0
    
    # Initial CMA-ES run setup
    opts = cma.CMAOptions()
    opts['bounds'] = [None]*D  # Unbounded in the CMA parameter space
    opts['CMA_damps'] = 1.0
    opts['popsize'] = 4 + int(3 * np.log(D))  # Effective population size
    opts['verbose'] = -9
    
    # Run CMA-ES optimization
    es = cma.CMAES(cma.fmin2(eval_c5_symmetric, 
                              np.concatenate([p_mean, q_mean]), 
                              0.3, 
                              opts, 
                              maxeval=20000,
                              maxtime=2000)
                        )
    
    best_c5 = float('inf')
    best_h = None
    
    # Use the best CMA-ES result
    final_result = es.result
    final_params = final_result[0]  # The best parameters found by CMA-ES
    p_best = final_params[:len(final_params)//2]
    q_best = final_params[len(final_params)//2:]
    
    # Compute h from best parameters
    h_cma = build_symmetric_h(p_best, q_best, n_points)
    h_cma = project_to_feasible(h_cma, n_points)
    c5_cma = evaluate_c5(h_cma)
    
    # Polishing with memetic descent
    h_polished = memetic_descent(h_cma, n_points, max_steps=25)
    c5_polished = evaluate_c5(h_polished)
    
    if c5_polished < c5_cma:
        best_h = h_polished
        best_c5 = c5_polished
    else:
        best_h = h_cma
        best_c5 = c5_cma
        
    return (best_h, best_c5, n_points)
