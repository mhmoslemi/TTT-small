import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar
import math

def project_box_sum(y, target_sum):
    """Project vector y onto the feasible set with sum = target_sum and box constraints [0,1]."""
    def S(lam):
        return np.sum(np.clip(y - lam, 0, 1))
            
    low, high = np.min(y), np.max(y) + 1.0
    while high - low > 1e-12:
        mid = (low + high) / 2.0
        s = S(mid)
        if s > target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    return np.clip(y - lam, 0, 1)

def compute_autocorrelation(h):
    """Compute auto-correlation vector."""
    return correlate(h, h, mode='full')

def fast_autocorrelation_update(γ, h, old_h, i, j, Δ):
    """Update γ incrementally after moving Δ from i to j."""
    n = len(h)
    Δγ = np.zeros_like(γ)
    # Update contribution for each shift s
    for s in range(-(n-1), n):
        idx = s + (n-1)
        # Remove from i:
        if 0 <= i + s < n:
            Δγ[idx] -= Δ * old_h[i+s]
        if 0 <= i - s < n and s != 0:
            Δγ[idx] -= Δ * old_h[i-s]
        # Add to j:
        if 0 <= j + s < n:
            Δγ[idx] += Δ * old_h[j+s]
        if 0 <= j - s < n and s != 0:
            Δγ[idx] += Δ * old_h[j-s]

    # Self-term correction
    Δγ[n-1] += -2*h[i]*Δ + Δ*Δ + 2*h[j]*Δ + Δ*Δ

    return γ + Δγ

def get_worst_shifts(C, ε=1e-5):
    """Get indices of worst shifts in C."""
    C5 = np.max(C)
    return np.where(np.abs(C - C5) <= ε)[0]

def score_indices(h, S_w, n):
    """Compute scores for indices based on contribution to worst shifts."""
    score = np.zeros(n)
    for s in S_w:
        # Add contribution from left side (shift s)
        if s >= 0:
            score[s:] += h[:-s]
        else:
            score[:s] += h[-s:]
        # Add contribution from right side (shift -s)
        if -s >= 0:
            score[-s:] += h[:-(-s)]
        else:
            score[:(-s)] += h[-(-s):]
    return score

def evaluate_c5(h):
    """Evaluate C5 for given h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 300
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = initial_h_values
        if len(h_start) < n_points:
            x_orig = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, n_points)
            h_interpolated = np.interp(x_new, x_orig, h_start)
        else:
            h_interpolated = h_start[:n_points]
        h = h_interpolated * (n_points / 2) / np.sum(h_interpolated)
    else:
        h = np.ones(n_points) * 0.5  # fallback
        h = h * (n_points / 2) / np.sum(h)
        
    h = np.clip(h, 0, 1)
    h = project_box_sum(h, n_points / 2)
    
    # Initialization
    γ = compute_autocorrelation(h)
    C = 1 - (2 / n_points) * γ
    c5_best = np.max(C)
    h_best = h.copy()
    γ_best = γ.copy()
    
    # Phase A: Repetitive seeding for diversity
    best_overall_h = h.copy()
    best_overall_c5 = c5_best
    n_seeds = 5
    for _ in range(n_seeds):
        h_temp = np.random.rand(n_points)
        h_temp = project_box_sum(h_temp, n_points / 2)
        c5_temp = evaluate_c5(h_temp)
        if c5_temp < best_overall_c5:
            best_overall_h = h_temp.copy()
            best_overall_c5 = c5_temp
    h = best_overall_h.copy()
    c5_best = best_overall_c5
    
    # Phase B: Coarse Simulated Annealing
    Δ_max = 0.2
    T0 = 0.3 * (np.max(C) - np.min(C))  
    α = 0.9995  # decay factor for temperature
    T = T0
    n_iter = int(0.3 * 1e6)  # 30% of total for coarse
    step = 0
    accept_count = 0
    T_boost_count = 0
    last_thermal_check = 0
    
    for _ in range(n_iter):
        if time.time() - start_time > 0.7 * budget_s:
            break
        step += 1
        # Binary swap or fraction transfer
        if np.random.rand() < 0.7:  # Binary swap
            i, j = np.random.choice(n_points, 2, replace=False)
            if (h[i] > 0.99 and h[j] < 0.01) or (h[i] < 0.01 and h[j] > 0.99):
                Δ = 1
                h_new = h.copy()
                h_new[i] = 1 - h_new[i]
                h_new[j] = 1 - h_new[j]
                old_h = h.copy()
                old_γ = γ.copy()
                h_new = project_box_sum(h_new, n_points / 2)
                γ_new = fast_autocorrelation_update(γ, h_new, old_h, i, j, Δ)
                C_new = 1 - (2 / n_points) * γ_new
                c5_new = np.max(C_new)
                if c5_new < c5_best:
                    h = h_new.copy()
                    γ = γ_new.copy()
                    c5_best = c5_new
                    if c5_new < best_overall_c5:
                        best_overall_c5 = c5_new
                        best_overall_h = h.copy()
                elif np.random.rand() < np.exp(- (c5_new - c5_best) / T):
                    h = h_new.copy()
                    γ = γ_new.copy()
                    c5_best = c5_new
                else:
                    h = old_h
                    γ = old_γ
                accept_count += 1
            else:
                continue  # Skip to try again
        else:  # Fractional transfer
            i = np.random.choice(n_points)
            if h[i] < 1e-8:
                continue
            j = np.random.choice(n_points)
            if h[j] > 1 - 1e-8:
                continue
            Δ = np.random.rand() * Δ_max
            if Δ > min(h[i], 1 - h[j]):
                continue
            old_h = h.copy()
            old_γ = γ.copy()
            h[i] -= Δ
            h[j] += Δ
            h = project_box_sum(h, n_points / 2)
            γ = fast_autocorrelation_update(γ, h, old_h, i, j, Δ)
            C_new = 1 - (2 / n_points) * γ
            c5_new = np.max(C_new)
            if c5_new < c5_best:
                h = h.copy()
                γ = γ.copy()
                c5_best = c5_new
                if c5_new < best_overall_c5:
                    best_overall_c5 = c5_new
                    best_overall_h = h.copy()
            elif np.random.rand() < np.exp(- (c5_new - c5_best) / T):
                h = h.copy()
                γ = γ.copy()
                c5_best = c5_new
            else:
                h = old_h
                γ = old_γ
            accept_count += 1

        T = T * α  # Temperature decay

        # Adaptive temperature adjustment
        if step % 5000 == 0:
            if accept_count < 0.25 * 5000:
                T = T * 1.2  # Re-heat
                T_boost_count += 1
            accept_count = 0

    # Phase C: Fine SA with memory and scoring
    h = best_overall_h.copy()
    c5_best = best_overall_c5
    γ = compute_autocorrelation(h)
    C = 1 - (2 / n_points) * γ
    c5_best = np.max(C)
    Δ_max = 0.1  # Decrease for fine tuning
    T = 0.05
    
    for _ in range(int(0.45 * 1e6)):
        if time.time() - start_time > 0.95 * budget_s:
            break
        # Compute worst shifts and scores
        S_w = get_worst_shifts(C)
        scores = score_indices(h, S_w, n_points)
        
        # Select i (highest score) and j (lowest score)
        idxs_i = np.where(scores > np.percentile(scores, 90))[0]
        idxs_j = np.where(scores < np.percentile(scores, 10))[0]
        
        if len(idxs_i) == 0 or len(idxs_j) == 0:
            # fallback to random
            i = np.random.choice(n_points)
            j = np.random.choice(n_points)
        else:
            i = np.random.choice(idxs_i)
            j = np.random.choice(idxs_j)
        
        # Transfer fraction Δ with bounded chance
        Δ = np.random.rand() * Δ_max
        if Δ > min(h[i], 1 - h[j]):
            continue
            
        old_h = h.copy()
        old_γ = γ.copy()
        h[i] -= Δ
        h[j] += Δ
        h = project_box_sum(h, n_points / 2)
        γ = fast_autocorrelation_update(γ, h, old_h, i, j, Δ)
        C_new = 1 - (2 / n_points) * γ
        c5_new = np.max(C_new)
        
        if c5_new < c5_best:
            h = h.copy()
            γ = γ.copy()
            c5_best = c5_new
            if c5_new < best_overall_c5:
                best_overall_c5 = c5_new
                best_overall_h = h.copy()

        elif np.random.rand() < np.exp(- (c5_new - c5_best) / T):
            h = h.copy()
            γ = γ.copy()
            c5_best = c5_new

        else:
            h = old_h.copy()
            γ = old_γ.copy()
            
        # Randomly update T
        T = T * α

    # Phase E: Subgradient Polish (non-smoothed max)
    h = best_overall_h.copy()
    c5_best = best_overall_c5
    γ = compute_autocorrelation(h)
    C = 1 - (2 / n_points) * γ
    S_w = get_worst_shifts(C)
    
    max_iter = 100
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
        # Subgradient: just average of all worst conjs
        g = np.zeros_like(h)
        for s in S_w:
            # shiftfunc +s
            left = np.roll(h, s, axis=0)
            if s > 0:
                left[:s] = 0
            else:
                left[s:] = 0
            # shiftfunc -s
            right = np.roll(h, -s, axis=0)
            if s > 0:
                right[s:] = 0
            else:
                right[:s] = 0
            g -= (2 / n_points) * (left + right)
        # Line search for step size
        def objective(alpha):
            h_trial = h - alpha * g
            h_trial = project_box_sum(h_trial, n_points / 2)
            return evaluate_c5(h_trial)
        # Binary search for best step size
        l, u = 0, 1.0
        for _ in range(20):
            mid = (l + u) / 2
            test_val = objective(mid)
            if test_val <= c5_best:
                l = mid
            else:
                u = mid
        dx = (l + u) / 2
        h = h - dx * g
        h = project_box_sum(h, n_points / 2)
        c5_best = evaluate_c5(h)
        γ = compute_autocorrelation(h)
        C = 1 - (2 / n_points) * γ
        S_w = get_worst_shifts(C)
        
        # early break:
        if c5_best < 0.3790:
            break

    # Final check and exact evaluation
    final_h = best_overall_h
    final_c5 = evaluate_c5(final_h)
    
    # Return best final solution
    return (final_h, final_c5, n_points)
