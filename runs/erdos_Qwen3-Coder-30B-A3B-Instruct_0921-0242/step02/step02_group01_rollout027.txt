```python
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.linalg import toeplitz
import cvxpy as cp

def project_to_feasible(h, target_sum):
    """Ensure that sum(h) = target_sum, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def gradient_worst_shift(h, worst_shift):
    """Compute gradient with respect to worst shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    
    # Since correlation uses circular shift, we take h_{i+k} and h_{i-k}
    # for k = worst_shift
    for i in range(n):
        left_idx = (i + worst_shift) % n
        right_idx = (i - worst_shift) % n
        grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Bisection setup
    n = 256  # Even, for numerical stability
    t_low, t_high = 0.0, 1.0
    eps = 2e-4
    
    # Phase 1: SDP bisection to find largest feasible t
    r_best = None
    while (time.time() - start_time < budget_s * 0.35) and (t_high - t_low > eps):
        t_mid = (t_low + t_high) / 2.0
        
        # Build variables and constraints for this t
        r = cp.Variable(n)
        constraints = [
            r[0] == 1,
            r[1:] >= t_mid,
            r <= 1
        ]
        
        # Build Toeplitz matrix
        T = toeplitz(r)
        constraints.append(T >> 0)  # PSD
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            r_best = r.value
            t_low = t_mid
        else:
            t_high = t_mid
    
    if r_best is None:
        # Fall back to uniform if SDP fails
        r_best = np.ones(n)
        r_best[0] = 1.0
        r_best[1:] = 0.0
  
    # Phase 2: Spectral factorisation and projection
    if time.time() - start_time > budget_s * 0.4:
        # Not enough time for next steps
        h_final = np.ones(n) * 0.5
        c5_final = float('inf')
        return (h_final, c5_final, n)
    
    r = r_best
    # Compute power spectrum
    S = np.fft.fft(r)
    # Clamp negative values
    S = np.maximum(S, 0)
    H = np.sqrt(S)
    
    # Spectral factorization
    g = np.real(np.fft.ifft(H))
    
    # Project to feasible set
    h_start = project_to_feasible(g, n / 2)
    
    # Phase 3: Worst-shift gradient descent
    h_refined = h_start.copy()
    c5_prev = evaluate_c5(h_refined)
    
    max_iter = 250
    step_size = 0.5 * (2.0 / n)
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.75:
            break
        # Compute correlation
        corr = ifft(fft(h_refined, 2 * len(h_refined)) * np.conj(fft(1 - h_refined, 2 * len(h_refined))))
        # Ignore shift 0 (self-correlation)
        corr_vals = corr.real[1: len(h_refined)]  # exclude index 0
        worst_shift = np.argmax(corr_vals)
        
        # Compute gradient
        grad = gradient_worst_shift(h_refined, worst_shift)
        
        # Update with step size
        step_size = 0.5 * (2.0 / n) / np.sqrt(i + 1)
        h_new = h_refined - step_size * grad
        h_new = project_to_feasible(h_new, n / 2)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5_prev:
            h_refined = h_new
            c5_prev = c5_new
        else:
            # Small tolerance to avoid stall
            if c5_prev - c5_new < 1e-6:
                break
    
    # Phase 4: Polishing with stochastic transfer
    h_final = h_refined.copy()
    c5_final = evaluate_c5(h_final)
    
    iter_count = 0
    while (time.time() - start_time < budget_s * 0.95) and iter_count < 5000:
        # Pick i with h_i > 0 and j with h_j < 1
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        
        if h_final[i] <= 0 or h_final[j] >= 1:
            continue

        delta = min(0.01, h_final[i], 1 - h_final[j])
        if delta <= 0:
            continue
        
        # Test the move
        h_test = h_final.copy()
        h_test[i] -= delta
        h_test[j] += delta
        h_test = project_to_feasible(h_test, n / 2)
 
        c5_test = evaluate_c5(h_test)
        if c5_test < c5_final:
            c5_final = c5_test
            h_final = h_test
            
        iter_count += 1
    
    # Phase 5: Upscale if time allows
    if time.time() - start_time < budget_s * 0.995:
        # Linear interpolate to 4x the resolution (like 1024 points)
        n_upscale = 1024
        dx_up = 2.0 / n_upscale
        # Interpolate h using linear interpolation
        x_orig = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_upscale)
        h_up = np.interp(x_new, x_orig, h_final)
        # Renormalize
        h_up = h_up * (n_upscale / 2) / np.sum(h_up)
        # Small gradient pass on upsampled
        h_final_up = h_up.copy()
        c5_final_up = evaluate_c5(h_up)
        step_size_up = 0.5 * dx_up
        for _ in range(50):
            if time.time() - start_time > budget_s * 0.995:
                break
            corr = ifft(fft(h_final_up, 2 * len(h_final_up)) * np.conj(fft(1 - h_final_up, 2 * len(h_final_up))))
            corr_vals = corr.real[1: len(h_final_up)]
            worst_shift = np.argmax(corr_vals)
            grad = gradient_worst_shift(h_final_up, worst_shift)
            step_size_up = 0.5 * dx_up / np.sqrt(1 + _)
            h_new = h_final_up - step_size_up * grad
            h_new = project_to_feasible(h_new, n_upscale / 2)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5_final_up:
                h_final_up = h_new
                c5_final_up = c5_new
            else:
                break
        h_final = h_final_up
        n = n_upscale
        c5_final = c5_final_up
    
    return (h_final, c5_final, n)
```
