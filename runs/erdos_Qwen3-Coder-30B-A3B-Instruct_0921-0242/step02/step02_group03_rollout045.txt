import numpy as np
import cvxpy as cp
import time

def full_to_sym(x_full):
    """Convert full vector to symmetric (middle element included)"""
    n = len(x_full)
    return x_full[:n//2 + 1]

def sym_to_full(x_sym):
    """Convert symmetric vector back to full"""
    n = 2 * (len(x_sym) - 1) + 1
    x_full = np.empty(n)
    x_full[:len(x_sym)] = x_sym
    x_full[len(x_sym):] = x_sym[-2::-1]
    return x_full

def build_constraints(x, t, shift_set, n_full):
    """Build linear constraints for given shift set."""
    constraints = []
    dx = 2.0 / n_full
    # For each shift, compute constraint sum_i x_i * (1 - x_{i+s}) <= t
    for s in shift_set:
        if s == 0:
            # Direct correlation: sum x_i * (1 - x_i) <= t
            expr = 0
            for i in range(n_full):
                expr += x[i] * (1 - x[i])
            constraints.append(expr * dx <= t)
        else:
            # Off-diagonal
            expr = 0
            for i in range(n_full):
                j = (i + s) % n_full
                expr += x[i] * (1 - x[j])
            constraints.append(expr * dx <= t)
    return constraints

def solve_lp(shift_set, x0=None, n_full=400):
    """Solve the LP with given shift set."""
    n_sym = n_full // 2 + 1
    x = cp.Variable(n_sym)
    t = cp.Variable()
    
    # Constraints
    constraints = [0 <= x, x <= 1]
    # Integral constraint: sum(h) = 1
    x_full = sym_to_full(x)
    integral_constraint = cp.sum(x_full) * (2.0 / n_full) == 1.0
    constraints.append(integral_constraint)
    
    # Add correlation constraints
    corr_constraints = build_constraints(x, t, shift_set, n_full)
    constraints.extend(corr_constraints)
    
    # Objective
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INaccurate]:
            x_opt = x.value
            t_opt = t.value
            return x_opt, t_opt
        else:
            return None, float('inf')
    except Exception:
        return None, float('inf')

def correlation_vals(h_full):
    """Compute correlation values for full vector."""
    n = len(h_full)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    # The reference point (shift 0) is at index n-1
    # Return from 0 to n-1 shifts (non-negative)
    return corr[n-1:] * (2.0 / n)
