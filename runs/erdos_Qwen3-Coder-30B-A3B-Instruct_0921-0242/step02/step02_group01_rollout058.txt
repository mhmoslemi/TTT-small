```python
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp
from scipy.linalg import toeplitz

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2 with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    if abs(diff) < 1e-10:
        return h
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    max_corr = np.max(corr.real) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial settings
    n_points = 256
    dx = 2.0 / n_points
    t_low, t_high = 0.0, 1.0
    epsilon = 2e-4
    max_sdp_iters = 15
    
    # SDP bisection
    r_best = None
    t_best = None
    
    sdp_time_budget = 0.35 * budget_s
    sdp_start = time.time()
    
    for iter_count in range(max_sdp_iters):
        if time.time() - sdp_start > sdp_time_budget:
            break
            
        t_mid = (t_low + t_high) / 2.0
        
        # Create SDP problem for feasibility
        r = cp.Variable(n_points)
        constraints = [
            r[0] == 1,
            r[1:] >= t_mid,
            r <= 1,
        ]
        
        # Toeplitz PSD constraint using the fact that T(r) is PSD
        # We will construct the matrix from the r vector
        T = toeplitz(r)
        
        # Add psd constraint via cvxpy
        constraints.append(cp.PSD(T))
        
        obj = cp.Minimize(0)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                r_best = r.value
                t_best = t_mid
                t_low = t_mid
            else:
                t_high = t_mid
        except:
            t_high = t_mid
    
    # If we have a feasible r, use it to construct h
    if r_best is None:
        # fallback to simple construction
        h = np.ones(n_points) * 0.5
    else:
        # Spectral factorization
        r = r_best
        S = np.fft.fft(r)
        # Ensure non-negative spectrum
        S = np.maximum(S, 0)
        H = np.sqrt(S)
        g = np.real(ifft(H))
        
        # Project to feasible polytope
        h = project_to_feasible(g, n_points)
    
    # Deterministic worst-shift gradient descent
    max_grad_iter = 250
    alpha0 = 0.5 * dx
    grad_start = time.time()
    
    for i in range(max_grad_iter):
        if time.time() - grad_start > 0.35 * budget_s:
            break
            
        # Compute correlation with FFT
        h_fft = fft(h, 2 * n_points)
        ih_fft = fft(1 - h, 2 * n_points)
        corr = ifft(h_fft * np.conj(ih_fft))
        corr_real = np.real(corr)
        # Ignore shift 0 in max computation
        worst_shift = np.argmax(corr_real[1:]) + 1
        
        # Compute gradient for worst shift
        g = np.zeros_like(h)
        for j in range(n_points):
            left_idx = (j + worst_shift) % n_points
            right_idx = (j - worst_shift) % n_points
            g[j] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Update with projected step
        alpha = alpha0 / np.sqrt(i + 1)
        h_new = h - alpha * g
        h = project_to_feasible(h_new, n_points)
    
    # Optional stochastic polishing
    if time.time() - start_time < 0.95 * budget_s:
        polish_start = time.time()
        while time.time() - polish_start < 0.05 * budget_s:
            i = np.random.randint(0, n_points)
            if h[i] <= 0:
                continue
            j = np.random.randint(0, n_points)
            if h[j] >= 1:
                continue
            delta = min(0.01, h[i], 1 - h[j])
            if delta <= 0:
                continue
            h_test = h.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, n_points)
            c5_test = evaluate_c5(h_test)
            c5_current = evaluate_c5(h)
            if c5_test < c5_current:
                h = h_test
                
    # Final check
    c5_bound = evaluate_c5(h)
    return (h, c5_bound, n_points)
```
