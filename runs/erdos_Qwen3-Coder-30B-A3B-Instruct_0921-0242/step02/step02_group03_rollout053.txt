import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum

    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def evaluate_correlations(h, n_points):
    """Evaluate all correlations of h with (1-h) efficiently using FFT."""
    h_padded = np.pad(h, (0, n_points-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, n_points-1), mode='constant', constant_values=0)
    
    # Using FFT for efficient cross-correlation
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
    # Normalize by dx
    dx = 2.0 / n_points
    corr_vals = corr_vals * dx
    
    # Return full array of correlations
    return corr_vals

def update_correlations_diff(corr_vals, h, i, j, n_points):
    """Efficiently update all correlations after a swap of h[i] and h[j]."""
    if i == j:
        return corr_vals
    
    # Precompute the differences for all shifts where this makes change
    new_corr = corr_vals.copy()
    dx = 2.0 / n_points
    
    for s in range(n_points):
        # swap affects correlation at shift s
        # Only the terms where i or j are involved change
        if i+s < n_points:
            delta_i = -(1 - h[i+s]) + h[i+s]  # contribution from h[i] to shift s
            delta_j = (1 - h[j+s]) - h[j+s]   # contribution from h[j] to shift s
            new_corr[s] += dx * (delta_i + delta_j)
        
        if j+s < n_points:
            delta_j = -(1 - h[j+s]) + h[j+s]
            delta_i = (1 - h[i+s]) - h[i+s]
            new_corr[s] += dx * (delta_i + delta_j)
    
    return new_corr

def anneal_binary(h_initial, n_points, time_limit, T0=0.02 * 221, alpha=0.997):
    """Perform simulated annealing on binary vectors."""
    h = h_initial.copy()
    corr_vals = evaluate_correlations(h, n_points)
    current_max = np.max(corr_vals)
    
    T = T0
    iter_count = 0
    max_iter = 50000
    best_h = h.copy()
    best_max = current_max
    
    # Track last improving iteration (for early stopping)
    last_improvement = 0
    
    while T > 1e-5 and iter_count < max_iter and time.time() < time_limit:
        # Accept the swap with probability 1 if it improves or Metropolis if not
        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        
        # If same index, we want a different one
        while j == i:
            j = np.random.randint(n_points)
        
        # Update correlation values efficiently
        old_corr_vals = corr_vals.copy()
        corr_vals = update_correlations_diff(corr_vals, h, i, j, n_points)
        new_max = np.max(corr_vals)
        
        if new_max <= current_max:
            # Accept improvement
            h[i], h[j] = h[j], h[i]
            current_max = new_max
            if new_max < best_max:
                best_h = h.copy()
                best_max = new_max
            last_improvement = iter_count
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_max - current_max) / T):
                h[i], h[j] = h[j], h[i]
                current_max = new_max
                if new_max < best_max:
                    best_h = h.copy()
                    best_max = new_max
                last_improvement = iter_count
            else:
                # Revert
                corr_vals = old_corr_vals
                
        iter_count += 1
        if iter_count - last_improvement > 5000:
            break
        if iter_count % 100 == 0:
            T *= alpha
            
    return best_h, best_max

def polish_continuous(h_binary, n_points, max_iter=30):
    """Perform gradient-based polishing from a binary solution."""
    h = h_binary.astype(float)
    dx = 2.0 / n_points
    lambda_param = 5.0  # Start with λ=5
    eta = 0.1 * dx  # Initial step size
    
    for i in range(max_iter):
        # Evaluate correlation efficiently with FFT
        corr_vals = evaluate_correlations(h, n_points)
        current_max = np.max(corr_vals)
        
        # Smooth max surrogate
        exp_vals = np.exp(lambda_param * corr_vals)
        Phi = np.log(np.sum(exp_vals)) / lambda_param
        
        # Gradient of the smooth max (approximate)
        # ∂Φ/∂h_k = 1/λ Σ_s exp(λ c_s) * ∂c_s/∂h_k / Σ_s exp(λ c_s)
        # ∂c_s/∂h_k = (1 - h_{k+s}) - (1 - h_{k-s}) = h_{k-s} - h_{k+s} 
        # when k+s, k-s are within bounds
        grad = np.zeros_like(h)
        
        # Compute gradients carefully for all shifts
        for k in range(n_points):
            for s in range(n_points - k):
                if k + s < n_points:
                    grad[k] += (1.0 - h[k + s]) * exp_vals[s] / np.sum(exp_vals)
                if k - s >= 0:
                    grad[k] -= h[k - s] * exp_vals[s] / np.sum(exp_vals)
        
        # Update h
        h_new = h - eta * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_feasible(h_new, n_points)
        
        # Check if improvement
        corr_new = evaluate_correlations(h_new, n_points)
        new_max = np.max(corr_new)
        
        if new_max < current_max:
            h = h_new.copy()
        else:
            # Break early if not improving anymore
            break
            
        # Decrease step size
        if i % 10 == 0 and i > 0:
            eta *= 0.5
            
        # Increase λ for sharper gradient
        lambda_param += 5.0
                
    return h, np.max(evaluate_correlations(h, n_points))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 10  # Leave some buffer
    
    n_points = 221  # Odd number for better symmetry
    dx = 2.0 / n_points
    
    h_best = None
    c5_best = float('inf')
    
    # Run multiple independent SA trials
    num_trials = 10
    for trial in range(num_trials):
        if time.time() >= time_limit:
            break
            
        # Initialize with block structure
        h = np.zeros(n_points, dtype=np.int8)
        mid = n_points // 2
        h[:mid] = 1  # First half ones
        np.random.shuffle(h)
        
        # Ensure equal sum
        ones_count = np.sum(h)
        if ones_count != n_points // 2:
            # Swap some ones with zeros to balance it
            ones_indices = np.where(h == 1)[0]
            zeros_indices = np.where(h == 0)[0]
            swaps = n_points // 2 - ones_count
            if swaps > 0:
                for i in range(swaps):
                    if i < len(ones_indices) and i < len(zeros_indices):
                        h[ones_indices[i]], h[zeros_indices[i]] = h[zeros_indices[i]], h[ones_indices[i]]
            else:
                for i in range(-swaps):
                    if i < len(ones_indices) and i < len(zeros_indices):
                        h[ones_indices[i]], h[zeros_indices[i]] = h[zeros_indices[i]], h[ones_indices[i]]
        
        # Run SA
        h_local, val_local = anneal_binary(h, n_points, time_limit)
        
        # Polish
        h_polished, val_polished = polish_continuous(h_local, n_points)
        
        # Try to improve with converging initializations
        if val_polished < c5_best:
            c5_best = val_polished
            h_best = h_polished.copy()
            
    if h_best is None:
        h_best = np.ones(n_points) * 0.5
        c5_best = evaluate_c5(h_best)
        
    return (h_best, c5_best, n_points)
