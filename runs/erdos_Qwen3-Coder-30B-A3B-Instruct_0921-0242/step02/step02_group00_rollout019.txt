import numpy as np
import time
from scipy.signal import fftconvolve
import cvxpy as cp
from scipy.interpolate import interp1d
from scipy.ndimage import gaussian_filter1d

def aperiodic_autocorr(h):
    """Compute aperiodic autocorrelation using FFT."""
    n = len(h)
    # Zero-pad to avoid circular correlation
    h_padded = np.pad(h, (0, n-1), 'constant')
    corr = fftconvolve(h_padded, h[::-1], 'valid')
    # Return only positive shifts
    return corr[n-1:]

def softmax(beta, x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(beta * x_shifted)
    return exp_x / np.sum(exp_x)

def project_box_sum(h, total):
    """Project h to [0,1] and ensure sum(h) = total using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = total - current_sum
    
    if abs(diff) < 1e-12:
        return h
    
    # Water-filling approach
    # Sort values in ascending order
    h_sorted = np.sort(h)
    n = len(h)
    
    # Find threshold lambda
    lambda_min, lambda_max = -1.0, 1.0
    tol = 1e-12
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(h - lam, 0, 1))
    
    # Binary search
    for _ in range(100):
        mid = (lambda_min + lambda_max) / 2
        s = sum_at_lambda(mid)
        if s > total:
            lambda_min = mid
        else:
            lambda_max = mid
        if abs(s - total) < tol:
            break
    
    lambda_opt = (lambda_min + lambda_max) / 2
    h_proj = np.clip(h - lambda_opt, 0, 1)
    return h_proj

def score_guided_swap(h, eps=1e-5):
    """Perform score-guided swaps in h."""
    n = len(h)
    # Find worst shifts
    gamma = aperiodic_autocorr(h)
    C = 1 - (2 / n) * gamma
    max_C = np.max(C)
    worst_indices = np.where(np.abs(C - max_C) < eps)[0]
    
    if len(worst_indices) == 0:
        return h

    # Score for each index: sum of values at shift positions
    scores = np.zeros(n)
    for s in worst_indices:
        if s > 0:
            scores[:-s] += h[s:]  # shift +s
            scores[s:] += h[:-s]  # shift -s (reverse)
        else:
            scores += h  # s == 0 case
    
    # Select indices with high score (source) and low score (target)
    high_score_idx = np.argsort(scores)[-10:]  # top 10 indices
    low_score_idx = np.argsort(scores)[:10]   # bottom 10 indices
    
    # Pick one pair
    if len(high_score_idx) > 0 and len(low_score_idx) > 0:
        i = high_score_idx[np.random.randint(len(high_score_idx))]
        j = low_score_idx[np.random.randint(len(low_score_idx))]
        
        # Move small amount
        delta = min(0.01, h[i], 1 - h[j])
        if delta > 0:
            h[i] -= delta
            h[j] += delta
            h = project_box_sum(h, total=n/2)
    
    return h

def evaluate_c5(h):
    """Evaluate the maximum overlap."""
    n = len(h)
    dx = 2.0 / n
    gamma = aperiodic_autocorr(h)
    C = 1 - (2 / n) * gamma
    return np.max(C)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase I: Binary MILP optimization
    n1 = 120  # Size for binary optimization
    half_n1 = n1 // 2
    
    # Define problem variables
    b = cp.Variable(n1, boolean=True)
    a = cp.Variable((n1, n1), boolean=True)  # a_i,s = b_i * b_{i+s}
    gamma = cp.Variable(n1)  # gamma_s
    L = cp.Variable()  # lower bound for all gamma_s
    
    # Define the MILP
    # Mass constraint: sum b_i = n1/2
    mass_constraint = cp.sum(b) == half_n1
    
    # McCormick constraints for a_i,s = b_i * b_{i+s} (i+s < n1)
    constraints = [mass_constraint]
    
    for s in range(n1):
        for i in range(n1 - s):
            # a[i,s] <= b[i]
            constraints.append(a[i, s] <= b[i])
            # a[i,s] <= b[i+s]
            constraints.append(a[i, s] <= b[i + s])
            # a[i,s] >= b[i] + b[i+s] - 1
            constraints.append(a[i, s] >= b[i] + b[i + s] - 1)
    
    # Gamma definition: gamma_s = sum_{i=0}^{n1-1-s} a[i,s]
    for s in range(n1):
        constraints.append(gamma[s] == cp.sum(a[:n1 - s, s]))
    
    # All gamma_s >= L
    for s in range(n1):
        constraints.append(gamma[s] >= L)
    
    # Objective: maximize L
    objective = cp.Maximize(L)
    
    prob = cp.Problem(objective, constraints)
    
    # Set parameters for solver
    rel_gap = 1e-3
    solver_time = min(120.0, 0.2 * budget_s)
    
    # Solve using CBC solver
    try:
        prob.solve(solver=cp.CBC, time_limit=solver_time, relative_gap=rel_gap)
    except Exception:
        pass  # Proceed without optimal solution if solver fails
    
    # Extract binary solution
    b_val = b.value
    if not b_val is None:
        b_val = np.round(b_val).astype(int)
    else:
        # Fall back to random solution if solver failed
        b_val = np.random.choice([0, 1], size=n1, p=[0.5, 0.5])
        # Make sure sum is exactly n1//2
        ones = np.where(b_val == 1)[0]
        if len(ones) > half_n1:
            to_drop = np.random.choice(ones, size=len(ones) - half_n1, replace=False)
            b_val[to_drop] = 0
        elif len(ones) < half_n1:
            zeros = np.where(b_val == 0)[0]
            to_set = np.random.choice(zeros, size=half_n1 - len(ones), replace=False)
            b_val[to_set] = 1

    # Phase II: Upsampling to n2 = 300
    n2 = 300
    x1 = np.linspace(0, 2, n1, endpoint=False)
    x2 = np.linspace(0, 2, n2, endpoint=False)
    h = np.interp(x2, x1, b_val)
    h = project_box_sum(h, total=n2 / 2)
    
    # Optional smoothing
    h = gaussian_filter1d(h, sigma=1)
    h = project_box_sum(h, total=n2 / 2)
    
    # Phase III: Smooth-max projected gradient descent
    beta = 2.0
    eta = 1.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    iteration = 0
    while time.time() - start_time < 0.85 * budget_s:
        gamma = aperiodic_autocorr(h)
        C = 1 - (2 / n2) * gamma
        w = softmax(beta, C)
        w_f = np.fft.fft(w)
        h_f = np.fft.fft(h)
        
        # Compute convolutions using FFT
        conv_w_h = np.fft.ifft(w_f * h_f).real
        conv_rev_w_h = np.fft.ifft(np.conj(w_f) * h_f).real
        
        grad = -(2 / n2) * (conv_w_h + conv_rev_w_h)
        
        # Line search with backtracking
        alpha = eta
        h_trial = h - alpha * grad
        h_trial = project_box_sum(h_trial, total=n2 / 2)
        c5_trial = evaluate_c5(h_trial)
        while c5_trial > best_c5 and alpha > 1e-8:
            alpha *= 0.5
            h_trial = h - alpha * grad
            h_trial = project_box_sum(h_trial, total=n2 / 2)
            c5_trial = evaluate_c5(h_trial)
        
        h = h_trial
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        iteration += 1
        if iteration % 150 == 0:
            beta = min(beta * 1.5, 200)
        if iteration % 200 == 0:
            h = score_guided_swap(h)
    
    # Phase IV: Final polishing
    while time.time() - start_time < budget_s - 0.1 * budget_s:
        gamma = aperiodic_autocorr(h)
        C = 1 - (2 / n2) * gamma
        max_C = np.max(C)
        worst_indices = np.where(np.abs(C - max_C) < 1e-5)[0]
        
        # Sub-gradient
        subgrad = np.zeros(n2)
        for s in worst_indices:
            if s == 0:
                subgrad += h  # shift zero
            else:
                h_shifted = np.roll(h, s)
                subgrad += h_shifted
                h_shifted = np.roll(h, -s)
                subgrad += h_shifted
        subgrad = -(2 / n2) * subgrad / len(worst_indices)
        
        eta_new = 1 / np.sqrt(iteration + 1)
        h_trial = h - eta_new * subgrad
        h_trial = project_box_sum(h_trial, total=n2 / 2)
        
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        h = h_trial
        iteration += 1
    
    return (best_h, best_c5, n2)
