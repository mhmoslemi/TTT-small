import numpy as np
import time

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    if abs(diff) < 1e-10:
        return h
    h = h + diff / n
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorrelation(h):
    """Compute full autocorrelation of h with itself and translate to shift domain."""
    n = len(h)
    gamma = np.correlate(h, h, mode='full')
    return gamma

def update_autocorrelation_delta(gamma, h, i, j, delta):
    """Exact O(n) update to gamma when transferring `delta` from h[i] to h[j]."""
    n = len(h)
    gamma_new = gamma.copy()
    h_copy = h.copy()
    h_copy[i] -= delta
    h_copy[j] += delta

    for s in range(-n+1, n):
        idx = s + n - 1
        if 0 <= i + s < n:
            gamma_new[idx] -= delta * h_copy[i + s]
        if 0 <= i - s < n and s != 0:
            gamma_new[idx] -= delta * h_copy[i - s]
        if 0 <= j + s < n:
            gamma_new[idx] += delta * h_copy[j + s]
        if 0 <= j - s < n and s != 0:
            gamma_new[idx] += delta * h_copy[j - s]
    # Specific update for s=0
    gamma_new[n - 1] += -2 * h[i] * delta + delta * delta + 2 * h[j] * delta + delta * delta

    return gamma_new

def get_worst_shifts(C, C5, epsilon=1e-5):
    """Return shift indices where overlap is close to maximum."""
    mask = np.abs(C - C5) < epsilon
    worst_s = np.where(mask)[0]
    return worst_s

def compute_score(h, worst_shifts, n):
    """Compute the score of each position."""
    score = np.zeros(n)
    for s in worst_shifts:
        if s > 0:
            idxs_left = np.arange(s, n)
            idxs_right = np.arange(0, n - s)
            # For s > 0, we look at h[i+s] for i in range(n-s)
            score[idxs_left] += h[idxs_right]
            score[idxs_right] += h[idxs_left]
        elif s < 0:
            idxs_left = np.arange(0, n + s)
            idxs_right = np.arange(-s, n)
            score[idxs_left] += h[idxs_right]
            score[idxs_right] += h[idxs_left]
        else:
            # For s=0 (self-correlation), handle specially
            score += 2 * h
    return score

def random_source_target(h, worst_shifts, score, eta=1e-8):
    """Select source and target position using scores."""
    n = len(h)
    indices_high = np.where((h > eta) & (score > np.quantile(score, 0.9)))[0]
    indices_low = np.where((h < 1 - eta) & (score < np.quantile(score, 0.1)))[0]
    if len(indices_high) == 0:
        i = np.random.randint(0, n)
    else:
        i = np.random.choice(indices_high)
    if len(indices_low) == 0:
        j = np.random.randint(0, n)
    else:
        j = np.random.choice(indices_low)
    return i, j

def project_box_sum(y, target_sum, lower=0, upper=1):
    """Handle projection onto box-constrained sum with line search."""
    def S(lam):
        return np.sum(np.clip(y - lam, lower, upper)) - target_sum

    l = np.min(y) - (np.sum(y) - target_sum)
    r = np.max(y) - (np.sum(y) - target_sum)
    for _ in range(50):
        mid = (l + r) / 2
        s = S(mid)
        if s > 0:
            l = mid
        else:
            r = mid
    lam = (l + r) / 2
    return np.clip(y - lam, lower, upper)
