import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft
from scipy.spatial.distance import cdist

def project_to_feasible(h, n_points):
    """Ensure that sum(h) = n_points/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n_points / 2 - current_sum

    if abs(diff) < 1e-10:
        return h

    # Water-filling approach
    h = h + diff / n_points
    h = np.clip(h, 0, 1)
    # Adjust for any remaining difference
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def build_lp_data(h0):
    """Build the linear program data for the current h0"""
    n = len(h0)
    # Compute auto-correlation
    c = ifft(fft(h0) * np.conj(fft(h0))).real
    C0 = c
    
    # Build matrix W
    W = np.zeros((n, n))
    for k in range(n):
        W[k, :] = h0[(np.arange(n) + k) % n] + h0[(np.arange(n) - k) % n]
    
    A = -W
    b = -n/2 + C0
    
    return A, b

def solve_lp(A, b, n_points):
    """Solve the linear program."""
    t = np.ones(1)
    # Objective: minimize t
    c = np.append(np.zeros(n_points), 1.0)
    
    # Constraint matrix: A h + t * 1 <= b
    # The constraint is -[W] h + 1 * t <= -n/2 + C0
    
    # Note: constraint matrix is (n_rows, n_cols)
    # We have n_rows = n, n_cols = n+1  
    n_rows = A.shape[0]
    # Zero pad and extend row by 1
    A_aug = np.column_stack([A, np.ones(n_rows)])
    
    # Add bounds: 0 <= h <= 1, sum(h) = n/2, t >= 0
    bounds = [(0, 1)] * n_points + [(0, None)]

    # Add sum constraint manually 
    A_eq = np.zeros((1, n_points + 1))
    A_eq[0, :n_points] = 1
    b_eq = np.array([n_points / 2])

    # All constraints should be in the form A * x <= b
    # Extending A_aug to account for equality constraint built from
    # sum h_i + 0 * t = n/2 which is equivalent to -sum h_i + 0 * t = -n/2

    # Add equality constraint
    A_eq_full = np.vstack([A_aug, A_eq])
    b_full = np.append(b, b_eq)

    # Return solution: first n entries are h, last is t
    result = linprog(c, A_ub=A_eq_full, b_ub=b_full, bounds=bounds, method='highs')
    if result.success:
        h_new = result.x[:n_points]
        t_new = result.x[-1]
        
        return h_new, t_new
    else:
        # Revert to soft projection
        h_new = np.clip(h0, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new, evaluate_c5(h_new)

def generate_seeds(n_points):
    """Generate different initial seeds."""
    seeds = []
    # 1. Uniform
    h_uniform = np.ones(n_points) * 0.5
    seeds.append(("uniform", h_uniform))

    # 2. Random Dirichlet
    h_dirichlet = np.random.dirichlet([1.0] * n_points)
    h_dirichlet = h_dirichlet * (n_points / 2) / np.sum(h_dirichlet)
    seeds.append(("dirichlet", h_dirichlet))

    # 3. Binary random
    h_binary = np.zeros(n_points)
    indices = np.random.permutation(n_points)[:n_points // 2]
    h_binary[indices] = 1.0
    h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
    # Add small noise for smoothness
    h_binary = np.clip(h_binary + np.random.normal(0, 0.01, n_points), 0, 1)
    h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
    seeds.append(("binary", h_binary))

    # 4. Interpolated from initial_h_values if available
    try:
        if 'initial_h_values' in globals():
            x_orig = np.linspace(0, 2, len(initial_h_values))
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_orig, initial_h_values)
            h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
            seeds.append(("interpolated", h_interp))
    except:
        pass

    # 5. Sobol-like sample (a sketch)
    # Random but avoid extreme clustering by using space-filling
    # Here we generate some low-discrepancy-like sample
    sobol_points = []
    for _ in range(n_points):
        x = np.random.rand(n_points)
        x = np.sort(x)
        x = x * (n_points / 2) / np.sum(x)
        x = np.clip(x, 0, 1)
        sobol_points.append(x)
    
    if sobol_points:
        # Pick one representative from Sobol points
        representative = np.mean(sobol_points, axis=0)
        seeds.append(("sobol", representative))
    
    return seeds

def projected_gradient_descent(h_start, max_iter=200):
    """Perform simple projected gradient descent on true objective"""
    h = h_start.copy()
    dx = 2.0 / len(h)
    for _ in range(max_iter):
        # Compute correlation on full data to find worst shift
        h_fft = fft(h, 2 * len(h))
        ih_fft = fft(1 - h, 2 * len(h))
        corr = ifft(h_fft * np.conj(ih_fft))
        corr_vals = corr.real[1:len(h)]  # ignore shift 0
        worst_shift = np.argmin(corr_vals) + 1
        
        # Gradient computation
        grad = np.zeros_like(h)
        for i in range(len(h)):
            left_idx = (i + worst_shift) % len(h)
            right_idx = (i - worst_shift) % len(h)
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Update
        h = h - 0.1 * grad
        h = project_to_feasible(h, len(h))
        
    return h

def simulated_annealing(h_start, max_time_s=120, temp0=1.0):
    """Use simulated annealing to polish the solution."""
    h = h_start.copy()
    dx = 2.0 / len(h)
    t_start = time.time()
    temp = temp0
    
    while time.time() - t_start < max_time_s:
        # Pick two indices i and j
        i = np.random.randint(0, len(h))
        j = np.random.randint(0, len(h))
        if i == j:
            continue
            
        # Check bounds
        if h[i] <= 0 or h[j] >= 1:
            continue
        
        # Move amount
        delta = min(0.01, h[i], 1 - h[j])
        if delta < 1e-5:
            continue
            
        # Test move
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = project_to_feasible(h_new, len(h))  # Ensure feasibility
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        
        if c5_new < c5_old:
            h = h_new
        else:
            # Accept with probability
            dE = c5_new - c5_old
            if np.random.rand() < np.exp(-dE / temp):
                h = h_new
                
        # Cool temperature
        temp *= 0.9999
        if temp < 1e-8:
            temp = 1e-8
    
    return h

def upscale_and_refine(h_coarse, n_fine=800, n_ccp=5):
    """Upscale coarse solution and do a few CCP passes."""
    dx_coarse = 2.0 / len(h_coarse)
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    
    # Run some CCP passes
    for _ in range(n_ccp):
        A, b = build_lp_data(h_fine)
        h_fine, _ = solve_lp(A, b, n_fine)
        h_fine = project_to_feasible(h_fine, n_fine)
        
    return h_fine

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Duration allocation
    ccp_time = 0.55 * budget_s
    gradient_time = 0.10 * budget_s
    sa_time = 0.20 * budget_s
    upscale_time = 0.10 * budget_s
    
    # Integers
    n_points = 400
    max_iter = 30
    
    best_c5 = float('inf')
    best_h = None
    
    # Generate seeds
    seeds = generate_seeds(n_points)
    
    # Run CCP loop for each seed
    for name, h_seed in seeds:
        if time.time() - start_time > ccp_time * 0.9:
            break
            
        h_current = h_seed.copy()
        
        for it in range(max_iter):
            if time.time() - start_time > ccp_time:
                break
                
            A, b = build_lp_data(h_current)
            h_new, t_new = solve_lp(A, b, n_points)
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
            if np.max(np.abs(h_new - h_current)) < 1e-8:
                break
                
            h_current = h_new

    # Check if we found anything
    if best_h is None:
        h_final = np.ones(n_points) * 0.5
        c5_final = evaluate_c5(h_final)
    else:
        h_final = best_h.copy()
        c5_final = best_c5
        
        # Do refined gradient descent
        if time.time() - start_time < gradient_time + 0.1:
            h_final = projected_gradient_descent(h_final, max_iter=200)
            c5_final = evaluate_c5(h_final)
        
        # Polishing with SA
        if time.time() - start_time < sa_time + 0.1:
            h_final = simulated_annealing(h_final, max_time_s=sa_time)
            c5_final = evaluate_c5(h_final)
        
        # Optional upscale if time is still available
        if time.time() - start_time < upscale_time + 0.1:
            h_final = upscale_and_refine(h_final.copy(), n_fine=800, n_ccp=5)
            c5_final = evaluate_c5(h_final)
    
    return (h_final, c5_final, n_points)
