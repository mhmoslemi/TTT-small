import numpy as np
import cvxpy as cp
import time

def full_to_sym(h_full):
    """Convert full vector to symmetric (half + center) representation."""
    n = len(h_full)
    if n % 2 == 0:
        return np.concatenate([h_full[:n//2], h_full[n//2-1::-1]])
    else:
        return np.concatenate([h_full[:(n+1)//2], h_full[:(n-1)//2][::-1]])

def sym_to_full(h_sym):
    """Convert symmetric vector to full representation."""
    n = len(h_sym) * 2 - 1
    if n % 2 == 1:
        # odd length
        return np.concatenate([h_sym[:n//2], h_sym[n//2::-1]])
    else:
        # even length
        return np.concatenate([h_sym[:n//2], h_sym[n//2-1::-1]])

def compute_shifted_sum(h, shift):
    """Compute sum_i h[i] * (1 - h[i+shift]) with proper boundary handling."""
    n = len(h)
    if shift >= n:
        return 0.0
    # We compute: sum_i h[i] * (1 - h[i+shift]) = sum_i h[i] - sum_i h[i]*h[i+shift]
    # But we need to avoid out-of-bounds access.
    # We can pad h at both ends with zeros.
    h_padded = np.pad(h, (0, n), mode='constant', constant_values=0)
    shifted = h_padded[shift:shift + n]
    return np.sum(h * (1 - shifted))

def solve_lp(shift_set, h0_sym=None, n=400):
    """Solve the LP with given shift set."""
    # Use symmetric representation (n//2 + 1 variables)
    m = n // 2 + 1
    x = cp.Variable(m)
    t = cp.Variable()
    
    # Symmetry constraints
    constraints = [0 <= x, x <= 1]
    
    # Integral constraint
    h_full = sym_to_full(x)
    integral_con = cp.sum(h_full) * (2.0 / n) == 1.0
    constraints.append(integral_con)
    
    # Shift constraints
    for s in shift_set:
        # Compute sum_i x_i * (1 - x_{i+s}) where x is symmetric
        # We'll manually handle the shift
        # For each i, we compute x_i * (1 - x_{i+s}) for valid indices
        shifted_indices = np.arange(m) + s
        if s >= m:
            # No contribution
            constraint_val = 0.0
        else:
            # Padding at the end
            # First, extract x_i (valid) for i=0..m-1
            # Then x_{i+s} but want to respect symmetric structure
            # Let's build the full h and compute
            h_full = sym_to_full(x)
            shifted = np.zeros_like(h_full)
            if s < len(h_full):
                shifted[s:] = h_full[:-s] if s > 0 else h_full
            shift_sum = np.sum(h_full * (1 - shifted))
            constraint_val = shift_sum * (2.0 / n)
        
        constraints.append(constraint_val <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_opt = x.value
            t_opt = t.value
            return x_opt, t_opt
        else:
            # Revert to h0 if not solvable
            return h0_sym, None
    except Exception:
        return h0_sym, None
