import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def build_trapezoid(t, H, n):
    """Build a piecewise linear trapezoid function based on parameters t and H."""
    dx = 2.0 / n
    h = np.zeros(n)
    
    # Compute L and a
    L = 1.0 / H - t
    a = (2.0 - 2.0 * t - L) / 2.0
    
    # Check feasibility
    if L < 0 or a < 0:
        return np.zeros(n)
    
    # Fill in the trapezoid
    x = np.arange(n) * dx + dx / 2.0  # midpoints of subintervals
    
    # Left ramp
    mask = (a <= x) & (x < a + t)
    if t > 0:
        h[mask] = H * (x[mask] - a) / t
    
    # Flat top
    mask = (a + t <= x) & (x < a + t + L)
    h[mask] = H
    
    # Right ramp
    mask = (a + t + L <= x) & (x < a + 2 * t + L)
    if t > 0:
        h[mask] = H * (a + 2 * t + L - x[mask]) / t
    
    return h

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT correlation."""
    dx = 2.0 / len(h)
    N = 2 * len(h)
    Hc = fft(h, N)
    Gc = fft(1 - h, N)
    corr = ifft(Hc * np.conj(Gc)).real[:2*len(h)-1]
    max_corr = np.max(corr) * dx
    return max_corr

def memetic_descent(h, n_points, max_steps=20):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        # The maximum is at shift 0, so we need to find the min among non-zero shifts
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        # Gradient wrt h[i] is dx * (1 - h[i + worst_shift_idx] - h[i - worst_shift_idx])
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5_fft(h_trial)
            current_c5 = evaluate_c5_fft(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def random_feasible(n):
    """Generate a random feasible step function."""
    h = np.random.rand(n)
    return project_to_feasible(h, n)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    # Initialize parameters
    n_points = 800
    dx = 2.0 / n_points
    best_val = float('inf')
    best_h = None
    
    # ---- 2-D parametric search ----
    H_vals = np.linspace(0.5, 1.0, 41)
    
    for H in H_vals:
        t_max = min(1.0 / H, 2.0 - 1.0 / H)
        if t_max <= 0:
            continue
        t_vals = np.linspace(0.0, t_max, 81)
        for t in t_vals:
            if time.time() >= time_limit:
                break
            h = build_trapezoid(t, H, n_points)
            h = project_to_feasible(h, n_points)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
    
    if best_val == float('inf'):
        # Fallback to a simple construction
        h = np.zeros(n_points)
        h[n_points//4:3*n_points//4] = 1.0
        h = project_to_feasible(h, n_points)
        best_val = evaluate_c5_fft(h)
        best_h = h.copy()
    
    # ---- refinement around the best pair ----
    # Estimate t, H from best_h
    # For simplicity, get projected center value
    # Use very coarse grid around it
    if (time.time() - start_time) < budget_s - 10:
        # Find approximate t, H from best_h
        # Assume a reasonably good estimate by examining smooth transition points
        last_zero_idx = np.where(best_h > 0)[0][-1]
        first_zero_idx = np.where(best_h > 0)[0][0]
        estimate_t = (last_zero_idx - first_zero_idx) * dx / 2.0
        estimate_H = np.mean(best_h[first_zero_idx:last_zero_idx])
        
        # Assume not very sharp peaks for safety
        delta_H = 0.025
        delta_t = 0.01
        H_vals_fine = np.arange(max(0.5, estimate_H - 2*delta_H), min(1.0, estimate_H + 2*delta_H), delta_H)
        t_vals_fine = np.arange(max(0.0, estimate_t - 2*delta_t), min(2.0, estimate_t + 2*delta_t), delta_t)
        
        for H in H_vals_fine:
            t_max = min(1.0 / H, 2.0 - 1.0 / H)
            if t_max <= 0:
                continue
            for t in t_vals_fine:
                if t > t_max:
                    continue
                if time.time() >= time_limit:
                    break
                h = build_trapezoid(t, H, n_points)
                h = project_to_feasible(h, n_points)
                val = evaluate_c5_fft(h)
                if val < best_val:
                    best_val, best_h = val, h.copy()
    
    # ---- polishing with memetic descent ----
    if (time.time() - start_time) < budget_s - 5:
        best_h = memetic_descent(best_h, n_points, max_steps=20)
        best_val = evaluate_c5_fft(best_h)
    
    # ---- fallback random memetic runs if time permits ----
    while (time.time() - start_time) < time_limit - 5:
        try:
            h0 = random_feasible(n_points)
            h_opt = memetic_descent(h0, n_points, max_steps=30)
            val = evaluate_c5_fft(h_opt)
            if val < best_val:
                best_val, best_h = val, h_opt
        except Exception:
            continue
    
    return (best_h, best_val, n_points)
