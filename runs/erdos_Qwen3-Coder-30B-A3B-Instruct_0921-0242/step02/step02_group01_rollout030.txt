import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Computes the exact C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def smooth_max(h, p=100):
    """Computes a smooth approximation of the maximum of the overlap values."""
    dx = 2.0 / len(h)
    # Auto-correlation
    h_fft = fft(h, 2 * len(h))
    corr = ifft(h_fft * np.conj(h_fft))
    corr_vals = corr.real[:len(h)] * dx  # Only first half
    overlap = corr_vals
    C_vals = 1 - overlap
    # Smooth max
    S_p = np.sum(C_vals ** p) ** (1/p)
    return S_p

def compute_gradient(h, p=100):
    """Computes the gradient of the smooth max with respect to h."""
    dx = 2.0 / len(h)
    n = len(h)
    # Compute auto-correlation
    h_fft = fft(h, 2*n)
    h_conj_fft = np.conj(h_fft)
    auto_corr = ifft(h_fft * h_conj_fft).real[:n] * dx
    C_vals = 1 - auto_corr
    
    # Compute weights for p-norm gradient
    eps = 1e-12  # Prevent division by zero
    weights = (C_vals + eps) ** (p - 1)
    
    # Gradient of C(k) = 1 - dx * sum(h_i * h_{i+k})
    # ∂C(k)/∂h_i = -dx * (h_{i+k} + h_{i-k})
    # We compute ∂S_p / ∂h_i = (1/S_p^(p-1)) * sum_k w_k * ∂C(k)/∂h_i
    
    grad = np.zeros(n)
    for k in range(n):
        # Forward shift h_{i+k}
        h_forward = np.roll(h, -k)
        h_backward = np.roll(h, k)
        partial_grad = dx * (h_forward - h_backward)
        grad += weights * partial_grad
    
    grad /= (np.sum(C_vals ** p) ** ((p-1)/p) + eps)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 350
    dx = 2.0 / n_points
    p_vals = [8, 16, 32, 64, 128, 256]
    
    # Seeding strategies
    seed_pool = []
    
    # 1. Uniform half-filled
    seed_pool.append(np.ones(n_points) * 0.5)
    
    # 2. Random Dirichlet sample
    random_dirichlet = np.random.dirichlet(np.ones(n_points)) * (n_points/2)
    seed_pool.append(random_dirichlet)
    
    # 3. Interpolated initial_h_values if available
    if 'initial_h_values' in globals():
        init_h = initial_h_values
        x_orig = np.linspace(0, 2, len(init_h))
        x_new = np.linspace(0, 2, n_points)
        interp_h = np.interp(x_new, x_orig, init_h)
        seed_pool.append(interp_h)
    
    # 4. Binary pattern with soft transition
    binary_h = np.zeros(n_points)
    binary_h[:n_points//2] = 1
    # Apply Gaussian blur with std=5
    from scipy import signal
    binary_h = signal.convolve(binary_h, np.exp(-np.linspace(-5, 5, 21)**2/20), mode='same')
    binary_h = binary_h * (n_points/2) / np.sum(binary_h)
    seed_pool.append(binary_h)
    
    # Refine on seeds: for each seed, run homotopy optimization
    best_h = None
    best_c5 = float('inf')
    
    for seed_idx, h_seed in enumerate(seed_pool):
        h_seed = project_to_feasible(h_seed, n_points)
        current_h = h_seed.copy()
        
        for p in p_vals:
            # Time check before expensive step
            if time.time() - start_time > budget_s * 0.9:
                break
                
            # Minimize smooth surrogate with L-BFGS-B
            def obj(h):
                return smooth_max(h, p)
            
            def grad(h):
                return compute_gradient(h, p)
            
            # Optimization with penalty for sum constraint
            lambda_sum = 1e4  # Penalty weight
            
            def obj_with_penalty(h):
                penalty = lambda_sum * ((np.sum(h) - n_points/2) ** 2)
                return obj(h) + penalty
                
            def grad_with_penalty(h):
                penalty_grad = 2 * lambda_sum * (np.sum(h) - n_points/2)
                return grad(h) + penalty_grad
            
            # Angle-soft bounds
            bounds = [(0, 1) for _ in range(n_points)]
            options = {"maxiter": 150}
            
            try:
                res = minimize(
                    obj_with_penalty,
                    current_h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    jac=grad_with_penalty,
                    options=options,
                    tol=1e-6
                )
                
                if res.success or res.fun < float('inf'):
                    current_h = project_to_feasible(res.x, n_points)
                else:
                    current_h = project_to_feasible(current_h, n_points)
                    
            except Exception:
                current_h = project_to_feasible(current_h, n_points)
                
            # Update best if better
            if time.time() - start_time < budget_s * 0.9:
                current_c5 = evaluate_c5_exact(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
    
    # Refinement via worst-shift subgradient ascent
    if best_h is not None and time.time() - start_time < budget_s * 0.95:
        h_refined = project_to_feasible(best_h, n_points)
        for _ in range(100):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Find worst shift
            corr = ifft(fft(h_refined, 2 * len(h_refined)) * np.conj(fft(1 - h_refined, 2 * len(h_refined))))
            corr_vals = corr.real[1: len(h_refined)]  # exclude index 0
            worst_shift = np.argmax(corr_vals) + 1
            
            # Compute gradient for this shift
            grad = np.zeros(len(h_refined))
            h_forward = np.roll(h_refined, -worst_shift)
            h_backward = np.roll(h_refined, worst_shift)
            grad = dx * (1 - h_forward - h_backward)
            
            # Backtracking line search
            alpha = 0.1 * dx
            max_iter = 10
            for _ in range(max_iter):
                h_new = h_refined - alpha * grad
                h_new = project_to_feasible(h_new, n_points)
                c5_new = evaluate_c5_exact(h_new)
                c5_old = evaluate_c5_exact(h_refined)
                if c5_new < c5_old:
                    h_refined = h_new
                    break
                alpha *= 0.5
                
        # Final evaluation
        final_c5 = evaluate_c5_exact(h_refined)
        best_h = h_refined
        best_c5 = final_c5
    
    # Return best solution
    if best_h is not None:
        return (best_h, best_c5, n_points)
    else:
        h_default = np.ones(n_points) * 0.5
        return (h_default, float('inf'), n_points)
