import numpy as np
import time
import math

def project_box_sum(y, total):
    """Project y onto the set {x : 0 <= x <= 1, sum(x) = total}."""
    a = 0.0
    b = 1.0
    y = np.array(y)
    while b - a > 1e-12:
        mid = (a + b) / 2.0
        x = np.clip(y - mid, 0, 1)
        s = np.sum(x)
        if s >= total:
            a = mid
        else:
            b = mid
    return np.clip(y - a, 0, 1)

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def softmax_weights(C, beta):
    """Compute softmax weights from C values with temperature beta."""
    C = np.array(C)
    C_max = np.max(C)
    exp_C = np.exp(beta * (C - C_max))
    return exp_C / np.sum(exp_C)

def gradient_from_weights(h, w):
    """Compute gradient using softmax weights w."""
    n = len(h)
    # Convolve w with h and w[::-1] with h
    conv1 = np.convolve(w, h, mode='full')
    conv2 = np.convolve(w[::-1], h, mode='full')
    # Take the middle part: [n-1:n-1+n]
    grad = -(2.0/n) * (conv1[n-1:n-1+n] + conv2[n-1:n-1+n])
    return grad

def line_search(h, grad, total_mass, step_init=1.0):
    """Perform backtracking line search."""
    alpha = step_init
    h_current = h.copy()
    c5_current = evaluate_c5(h_current)
    
    for _ in range(20):  # Max 20 backtracks
        h_trial = project_box_sum(h_current - alpha * grad, total_mass)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial <= c5_current:
            return h_trial, c5_trial, alpha
        alpha *= 0.5
        if alpha < 1e-10:
            break
    return h_current, c5_current, alpha
