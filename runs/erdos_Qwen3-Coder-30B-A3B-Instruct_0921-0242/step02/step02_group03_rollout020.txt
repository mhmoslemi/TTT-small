import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def full_to_sym(h_full):
    """Convert full symmetric vector to half + center."""
    n = len(h_full)
    if n % 2 == 0:
        return (h_full[:n//2] + h_full[n//2:][::-1]) / 2.0
    else:
        return (h_full[:n//2] + h_full[n//2+1:][::-1]) / 2.0

def sym_to_full(h_sym):
    """Convert symmetric half vector back to full."""
    n = 2 * len(h_sym) - 1
    h_full = np.zeros(n)
    h_full[:len(h_sym)] = h_sym
    h_full[len(h_sym):] = h_sym[-2::-1]
    return h_full

def correlation_vals(h):
    """Compute full correlation of h with (1-h) and return non-negative shifts."""
    n = len(h)
    corr = correlate(h, 1-h, mode='full')
    return corr[n-1:] * (2.0/n)   # dx = 2/n

def solve_lp(shift_set, x0=None, n_points=400):
    """Solve the LP to minimize the maximum correlation for given shifts."""
    # Half size for symmetric case
    m = n_points // 2 + 1
    x = cp.Variable(m)
    t = cp.Variable()

    # Constraints for x
    constraints = [0 <= x, x <= 1, cp.sum(x) == 0.5]  # For integral constraint in symmetric case

    # Add correlation constraints
    dx = 2.0 / n_points
    for s in shift_set:
        if s == 0:
            expr = dx * cp.sum(x)
        else:
            # Create sparse matrix for shift s (only valid indices)
            expr = 0.0
            for i in range(m):
                j = i + s
                if 0 <= j < m:
                    expr += x[i] * (1.0 - x[j])
                else:
                    # Handle out of range - fill with 0
                    pass
        constraints.append(expr <= t)

    # Solve the LP
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)

    x_opt = x.value
    t_opt = t.value
    return x_opt, t_opt

    max_iter = 100
