import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cvxpy as cp

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def compute_autocorrelation(h):
    """Compute autocorrelation vector for a binary vector h."""
    dx = 2.0 / len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # Return real part, ignoring shift 0 (self-correlation)
    return corr.real[1:len(h)]

def solve_sdp_binary(n, max_time=180):
    """Solves the SDP relaxation for the binary case."""
    start_time = time.time()
    
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Box constraints
    for i in range(n):
        constraints.append(0 <= X[i,i])
        constraints.append(X[i,i] <= 1)
    
    # Cardinality constraint
    constraints.append(cp.sum(X.diagonal()) == n/2)
    
    # Autocorrelation constraints (k = 1 to n-1)
    for k in range(1, n):
        A = np.zeros((n, n))
        for i in range(n):
            A[i, (i+k) % n] = 1
        # Trace(A * X) >= t
        constraints.append(cp.trace(A @ X) >= t)
    
    # Pairwise bounds (improve relaxation)
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(X[i,j] >= 0)
            constraints.append(X[i,j] <= X[i,i])
            constraints.append(X[i,j] <= X[j,j])
    
    # PSD constraint
    constraints.append(X >> 0)
    
    # Objective
    objective = cp.Maximize(t)
    
    # Problem setup
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        if prob.status != cp.OPTIMAL:
            return None, None
        X_opt = X.value
        t_opt = t.value
        return X_opt, t_opt
    except Exception:
        return None, None

def round_binary_sdp(X_opt, n, R=30):
    """Randomized hyperplane rounding for SDP solution to binary vector."""
    if X_opt is None:
        return None
        
    # Eigen-decomposition
    try:
        evals, evecs = np.linalg.eigh(X_opt)
        evals = np.maximum(evals, 0)  # ensure non-negative
        sqrt_evals = np.sqrt(evals)
        v = evecs[:, -1] * sqrt_evals[-1]  # principal component
    except Exception:
        # Fallback to deterministic rounding
        return None
    
    best_min_corr = -1
    best_x = None
    
    # Try multiple randomizations
    for i in range(R):
        g = np.random.normal(0, 1, n)
        z = evecs @ (sqrt_evals * g)
        
        # Sort indices by z and take top n/2
        tmp_indices = np.argsort(z)[::-1]
        x = np.zeros(n)
        x[tmp_indices[:n//2]] = 1
        
        # Compute the minimum correlation
        corr = compute_autocorrelation(x)
        min_corr = np.min(corr)
        
        if min_corr > best_min_corr:
            best_min_corr = min_corr
            best_x = x
    
    # Also try deterministic thresholding
    tmp_indices = np.argsort(v)[::-1]
    x_det = np.zeros(n)
    x_det[tmp_indices[:n//2]] = 1
    
    corr = compute_autocorrelation(x_det)
    min_corr = np.min(corr)
    if min_corr > best_min_corr:
        best_min_corr = min_corr
        best_x = x_det
    
    return best_x

def optimize_binary_greedy(x, n, max_time=180):
    """Greedy swap-based optimization on binary vector."""
    start_time = time.time()
    
    # Precompute autocorrelations using FFT
    corr = compute_autocorrelation(x)
    
    count = 0
    was_improved = True
    while was_improved and time.time() - start_time < max_time:
        was_improved = False
        min_corr_idx = np.argmin(corr)
        if min_corr_idx == 0:
            break
        k = min_corr_idx
        
        # Evaluate contribution of ones and zeros for this shift
        contrib_ones = np.zeros(n)
        contrib_zeros = np.zeros(n)
        
        # For i with x_i=1 compute sum_{j=1}^{n-1} x_{i+j}+x_{i-j} 
        # Effectively this is look at the contribution of the ones to shift k.
        for i in range(n):
            if x[i] != 1:
                continue
            contrib_ones[i] = x[(i+k)%n] + x[(i-k)%n]
            
        for j in range(n):
            if x[j] != 0:
                continue
            contrib_zeros[j] = x[(j+k)%n] + x[(j-k)%n]
        
        # Find i that maximizes contrib_ones[i] (i.e. where removing 1 helps in k)
        max_i = np.argmax(contrib_ones)
        min_j = np.argmin(contrib_zeros)  # where adding 1 helps in k
        
        if contrib_ones[max_i] < contrib_zeros[min_j]:
            continue  # No benefit
        # Perform the swap:
        # x[max_i] = 0
        # x[min_j] = 1
        x[max_i] = 0
        x[min_j] = 1
        was_improved = True
        count += 1
        
        # Update correlation vector for this k directly 
        # Remove old contributions and add new contributions
        for i in range(n):
            for k_shift in range(1, n):
                old_val = x[(i-k_shift)%n] * x[i]
                new_val = x[(i-k_shift)%n] * x[i]  # unchanged unless i changed
                corr[k_shift-1] -= old_val
                new_val = x[(i-k_shift)%n] * x[i] 
                corr[k_shift-1] += new_val
        
        # Repeat update to efficiently recalculate all correlations 
        corr = compute_autocorrelation(x)
        continue
    return x

def continuous_refinement(h, max_time=60):
    """Refine fractional solution using gradient-based method."""
    start_time = time.time()
    dx = 2.0 / len(h)
    alpha = 0.5 * dx
    iter_count = 0
    old_c5 = evaluate_c5(h)
    
    while time.time() - start_time < max_time:
        # Find worst shift
        corr = ifft(fft(h, 2*len(h)) * np.conj(fft(1-h, 2*len(h))))
        corr_vals = corr.real[1:len(h)]
        worst_shift = np.argmin(corr_vals) + 1
        
        # Gradient: g_i = dx * [(1 - h_{i+k}) - h_{i-k}]
        grad = np.zeros_like(h)
        for i in range(len(h)):
            left_idx = (i + worst_shift) % len(h)
            right_idx = (i - worst_shift) % len(h)
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Take step
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h))
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < old_c5:
            h = h_new
            old_c5 = new_c5
        else:
            alpha *= 0.5  # backtracking
        
        iter_count += 1
        if alpha < 1e-6:
            break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 200
    h = np.ones(n) * 0.5  # fallback
    
    # Phase 1: Solve SDP relaxations
    if time.time() - start_time + 200 > budget_s:
        n = 150
    
    X_opt, t_opt = solve_sdp_binary(n, max_time=min(200, budget_s * 0.3))
    
    # Phase 2: Rounding step
    if X_opt is not None:
        x = round_binary_sdp(X_opt, n, R=30)
        if x is not None:
            h = x
    
    # Phase 3: Greedy swap refinement
    h = optimize_binary_greedy(h, n, max_time=min(180, budget_s * 0.3))
    
    # Phase 4: Optional continuous polishing
    if time.time() - start_time < budget_s * 0.95:
        h = continuous_refinement(h, max_time=min(60, budget_s * 0.1))
    
    # Final evaluation
    c5_bound = evaluate_c5(h)
    
    return (h, c5_bound, n)
