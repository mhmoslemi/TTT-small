import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    if diff > 0:
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    h_copy = np.clip(h_copy, 0, 1)
    return h_copy

def compute_correlations(h):
    """Fast computation of correlations using FFT."""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    return corr_vals[:n]

def update_correlations_delta(corr, h, i, j):
    """Compute correlation changes after swapping h[i] and h[j]."""
    n = len(h)
    delta = np.zeros(n)
    for s in range(n):
        if i+s < n and j+s < n:
            delta[s] += -(1-h[i+s]) + h[i+s]  # change at position i
            delta[s] += +(1-h[j+s]) - h[j+s]  # change at position j
    return delta

def anneal_binary(h0, n, time_limit):
    """Simulated annealing for binary sequences."""
    T0 = 0.02 * n
    alpha = 0.997
    T = T0
    
    h = h0.copy()
    corr = compute_correlations(h)
    current_max = np.max(corr)
    
    best_h = h.copy()
    best_max = current_max
    
    iteration = 0
    last_improvement = 0
    
    while T > 1e-5 and last_improvement < 5000 and iteration < 60000:
        if time.time() > time_limit:
            break
        
        # Find all 1s and 0s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute delta correlation
        delta = update_correlations_delta(corr, h, i, j)
        new_corr = corr + delta
        new_max = np.max(new_corr)
        
        if new_max <= current_max or np.random.rand() < np.exp(-(new_max - current_max)/T):
            h[i], h[j] = h[j], h[i]
            corr = new_corr
            current_max = new_max
            
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                last_improvement = 0
        else:
            last_improvement += 1
            
        T *= alpha
        iteration += 1
        
    return best_h, best_max

def polish_continuous(h0, n, max_steps=30):
    """Perform continuous polishing using smooth max surrogate."""
    h = h0.astype(float)
    dx = 2.0 / n
    
    # Smooth max parameter
    lam = 5.0
    eta = 0.1 * dx
    
    # Project to feasible set
    h = project_to_feasible(h, n)
    
    for step in range(max_steps):
        if step % 10 == 0 and step > 0:
            eta /= 2.0
        if eta < 1e-8:
            break
            
        # Compute correlations
        corr = compute_correlations(h)
        max_corr = np.max(corr)
        
        # Compute gradient of smooth max
        max_corr_exp = np.exp(lam * corr)
        total = np.sum(max_corr_exp)
        prob = max_corr_exp / total
        
        # Gradient of log-sum-exp with respect to h
        grad = np.zeros(n)
        for s in range(n):
            for i in range(n - s):
                grad[i] += prob[s] * (1 - h[i+s]) 
                grad[i+s] -= prob[s] * h[i]
                
        # Update h
        h_new = h - eta * grad
        h_new = project_to_feasible(h_new, n)
        
        # Check improvement
        h = h_new
        
    return h, evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 10  # Leave 10s for final computation
    
    # Use odd number of points for better balance
    n = 221
    dx = 2.0 / n
    best_h = None
    best_c5 = float('inf')
    
    # Run multiple parallel trials
    num_trials = 12
    
    for trial in range(num_trials):
        if time.time() > time_limit:
            break
            
        # Try different initial configurations
        if np.random.rand() < 0.5:
            # Block initialization
            h = np.zeros(n, dtype=int)
            start = np.random.randint(0, n//2)
            h[start:start+n//2] = 1
        else:
            # Random balanced binary construction
            ones = np.random.permutation(n)[:n//2]
            h = np.zeros(n, dtype=int)
            h[ones] = 1
            
        # Simulated annealing
        h_annealed, c5_annealed = anneal_binary(h, n, time_limit)
        
        # Polishing
        h_polished, c5_polished = polish_continuous(h_annealed, n)
        
        # Keep the best
        if c5_polished < best_c5:
            best_c5 = c5_polished
            best_h = h_polished.copy()
            
    if best_h is None:
        # Fallback initialization
        h_fallback = np.zeros(n)
        h_fallback[:n//2] = 1
        h_fallback = project_to_feasible(h_fallback, n)
        best_h, best_c5 = polish_continuous(h_fallback, n)
        
    return (best_h, best_c5, n)
