import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def build_trapezoid(t, H, n):
    """Build a trapezoidal h function with given t and H."""
    dx = 2.0 / n
    h = np.zeros(n)
    
    # Compute derived parameters
    L = 1.0 / H - t
    a = (2.0 - 2.0 * t - L) / 2.0
    
    if L < 0 or a < 0:
        return h  # invalid parameters
    
    # Indices for the ramps and flat top
    left_ramp_start = int(a / dx)
    left_ramp_end = int((a + t) / dx)
    flat_start = left_ramp_end
    flat_end = int((a + t + L) / dx)
    right_ramp_start = flat_end
    right_ramp_end = int((a + 2.0 * t + L) / dx)
    
    # Fill left ramp
    for i in range(left_ramp_start, min(left_ramp_end, n)):
        x = (i + 0.5) * dx
        h[i] = H * (x - a) / t
    
    # Fill flat top
    for i in range(flat_start, min(flat_end, n)):
        h[i] = H
    
    # Fill right ramp
    for i in range(right_ramp_start, min(right_ramp_end, n)):
        x = (i + 0.5) * dx
        h[i] = H * (a + 2.0 * t + L - x) / t
    
    return h

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def evaluate_c5_fft(h):
    """Compute the C5 constant using FFT for correlation."""
    dx = 2.0 / len(h)
    N = 2 * len(h)
    Hc = fft(h, N)
    Gc = fft(1 - h, N)
    corr = ifft(Hc * np.conj(Gc)).real[:2 * len(h) - 1]
    max_corr = np.max(corr) * dx
    return max_corr

def memetic_descent(h, n_points, max_steps=30):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        # The maximum is at shift 0, so we need to find the min among non-zero shifts
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        # Gradient wrt h[i] is dx * (1 - h[i + worst_shift_idx] - h[i - worst_shift_idx])
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5_fft(h_trial)
            current_c5 = evaluate_c5_fft(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def random_feasible(n_points):
    """Generate a random feasible step function."""
    h = np.random.rand(n_points)
    h = project_to_feasible(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    n = 800
    dx = 2.0 / n
    best_val = float('inf')
    best_h = None
    
    # Coarse 2D grid search
    H_values = np.linspace(0.5, 1.0, 41)
    
    for H in H_values:
        t_max = min(1.0/H, 2.0 - 1.0/H)
        if t_max <= 0:
            continue
        t_values = np.linspace(0, t_max, 81)
        for t in t_values:
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
    
    # Refinement around the best
    # Get best parameters from the resulting h
    t_best, H_best = 0., 0.5
    max_ratio = 0.0
    
    # Try to estimate parameters from best_h
    for H in H_values[1:-1]:  # avoid boundaries
        t_max = min(1.0/H, 2.0 - 1.0/H)
        if t_max <= 0:
            continue
        t_values = np.linspace(0, t_max, 81)
        for t in t_values:
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
                t_best, H_best = t, H
    
    # Do refinement with finer grid
    delta_H = 0.025
    delta_t = 0.01
    H_range = np.linspace(max(0.5, H_best - 2*delta_H), min(1.0, H_best + 2*delta_H), 41)
    t_range = np.linspace(max(0.0, t_best - 2*delta_t), min(t_max, t_best + 2*delta_t), 41)
    
    for H in H_range:
        t_max_ref = min(1.0/H, 2.0 - 1.0/H)
        if t_max_ref <= 0:
            continue
        for t in t_range:
            if t > t_max_ref:
                continue
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
    
    # Optional local refinement with Nelder-Mead if time allows
    if time.time() < time_limit - 10:
        try:
            def objective(params):
                t, H = params
                h = build_trapezoid(t, H, n)
                h = project_to_feasible(h, n)
                return evaluate_c5_fft(h)
            
            res = minimize(objective, [t_best, H_best], method='Nelder-Mead', options={'maxiter': 20})
            if res.success:
                h_opt = build_trapezoid(res.x[0], res.x[1], n)
                h_opt = project_to_feasible(h_opt, n)
                val_opt = evaluate_c5_fft(h_opt)
                if val_opt < best_val:
                    best_val, best_h = val_opt, h_opt
        except:
            pass
    
    # Polishing with memetic descent
    if time.time() < time_limit - 5:
        best_h = memetic_descent(best_h, n, max_steps=20)
        best_val = evaluate_c5_fft(best_h)
        
        # Fallback random memetic searches
        remaining_time = time_limit - time.time()
        num_fallbacks = min(10, max(1, int(remaining_time / 10)))  # Up to 10 trials
        for _ in range(num_fallbacks):
            if time.time() >= time_limit:
                break
            h0 = random_feasible(n)
            h_opt = memetic_descent(h0, n, max_steps=30)
            val = evaluate_c5_fft(h_opt)
            if val < best_val:
                best_val, best_h = val, h_opt
    
    if best_h is None:
        best_h = np.ones(n) * 0.5
        best_val = evaluate_c5_fft(best_h)
    
    return (best_h, best_val, n)
