import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cma

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)
    return h_copy

def memetic_descent(h, n_points, max_steps=25):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute the entire correlation to identify the worst shift
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points] 
        corr_vals = corr_vals * dx
        
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1

        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def create_symmetric_piecewise_h(p, q, n_points, M):
    """Create symmetric h from piecewise constant parameters."""
    # Apply transformations to get widths and heights
    w = np.exp(p)
    w = w / np.sum(w)
    a = 1. / (1. + np.exp(-q))
    
    # Build h_half based on the widths and heights
    h_half = np.zeros(n_points // 2)
    cumulative_w = 0
    for i in range(M):
        w_i = w[i]
        a_i = a[i]
        start_idx = int(np.round(cumulative_w * (n_points // 2)))
        end_idx = int(np.round((cumulative_w + w_i) * (n_points // 2)))
        if end_idx > start_idx:
            h_half[start_idx:end_idx] = a_i
        cumulative_w += w_i
    
    # Construct full h
    h_full = np.concatenate([h_half, h_half[::-1]])
    return h_full

def objective(pq, n_points, M):
    """Objective to minimize (C5) for given parameters."""
    p = pq[:M]
    q = pq[M:]
    h = create_symmetric_piecewise_h(p, q, n_points, M)
    h = project_to_feasible(h, n_points)
    c5 = evaluate_c5(h)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 30  # Leave 30s for polishing
    
    n_points = 400
    M = 7
    D = 2 * M
    
    # CMA-ES parameters
    sigma = 0.3
    popsize = 4 + int(3 * np.log(D))
    max_generations = 500
    no_improve_generations = 80
    max_restarts = 10
    
    # Initial mean for CMA-ES
    # Initialize widths to make middle blocks wider
    p_init = np.zeros(M)
    p_init[M//3:2*M//3] = 0.3 / (M//3)  # Middle blocks wider
    p_init[:M//3] = (1 - 0.3) / (M//3)
    p_init[2*M//3:] = (1 - 0.3) / (M//3)

    # Initialize heights to give high values in middle and low in outer regions
    q_init = np.zeros(M)
    q_init[M//3:2*M//3] = 5  # High values
    q_init[:M//3] = -5  # Low values
    q_init[2*M//3:] = -5

    pq_mean = np.concatenate([p_init, q_init])
    
    # Initial best solution
    best_c5 = float('inf')
    best_h = None

    # Run CMA-ES with restarts
    restart_count = 0
    generation = 0
    no_improve = 0
    
    while restart_count <= max_restarts and time.time() < time_limit:
        if no_improve >= no_improve_generations:
            # Trigger restart
            popsize *= 2
            sigma = 0.3
            restart_count += 1
            no_improve = 0
        else:
            if time.time() >= time_limit:
                break
                
        es = cma.CMAEvolutionStrategy(pq_mean, sigma, {'popsize': popsize, 'seed': seed + restart_count})
        generation = 0

        while (es.countiter <= max_generations) and (time.time() < time_limit):
            X = es.ask()
            F = [objective(x, n_points, M) for x in X]
            es.tell(X, F)
            
            if np.min(F) < best_c5:
                best_c5 = np.min(F)
                best_x = X[np.argmin(F)]
                best_h = create_symmetric_piecewise_h(best_x[:M], best_x[M:], n_points, M)
                best_h = project_to_feasible(best_h, n_points)
                
                no_improve = 0
            else:
                no_improve += 1
                
            generation += 1
            if time.time() >= time_limit:
                break
                
        pq_mean = es.result.xbest
    
    # Final polishing step
    if best_h is not None:
        h_polished = memetic_descent(best_h, n_points, max_steps=25)
        c5_polished = evaluate_c5(h_polished)
        if c5_polished < best_c5:
            best_c5 = c5_polished
            best_h = h_polished.copy()
    
    if best_h is None:
        # Fallback to uniform distribution
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, n_points)
