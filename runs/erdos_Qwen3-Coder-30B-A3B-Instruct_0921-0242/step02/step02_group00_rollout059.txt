import numpy as np
import time
from scipy.signal import fftconvolve
import cvxpy as cp
from scipy.ndimage import gaussian_filter1d

def aperiodic_autocorr(h):
    """Compute aperiodic autocorrelation using FFT."""
    n = len(h)
    # Pad to 2*n-1 to avoid circular convolution effects
    h_padded = np.pad(h, (0, n-1), mode='constant')
    corr = fftconvolve(h_padded, h[::-1], mode='valid')
    return corr[:n]

def project_box_sum(h, total):
    """Project h onto the feasible set {h | 0 ≤ h ≤ 1, sum(h) = total}."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = total - current_sum
    if abs(diff) < 1e-12:
        return h
    # Use water-filling
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    n = len(h)
    # Find threshold index
    for i in range(n):
        if sorted_h[i] * (i + 1) - cumsum[i] >= diff:
            threshold = (cumsum[i] + diff) / (i + 1)
            # Apply threshold
            h = np.clip(h, 0, 1)
            h = np.where(h >= threshold, h, threshold)
            # `h` now has sum not quite `total` due to quantization
            # Adjust the difference using optimized approach
            err = np.sum(h) - total
            for j in range(n):
                if h[j] < 1 and h[j] > 0:
                    delta = min(-err, 1 - h[j])
                    if delta > 1e-10:
                        h[j] += delta
                        err += delta
                        if abs(err) < 1e-10:
                            break
            return h
    # Fall back to uniform adjustment (if needed)
    h += diff / n
    # Clip and re-adjust if necessary
    h = np.clip(h, 0, 1)
    err = np.sum(h) - total
    if abs(err) > 1e-10:
        # Adjust with water-filling logic
        h = np.clip(h, 0, 1)
        h = np.where(h > 0, h, 0)
        ratio = total / max(np.sum(h), 1e-12)
        h *= ratio
    return h

def softmax(beta, x):
    """Numerically stable softmax with temperature beta."""
    x = np.asarray(x)
    x_shifted = x - np.max(x)
    exp_x = np.exp(beta * x_shifted)
    return exp_x / np.sum(exp_x)

def conv(a, b):
    """Fast convolution of a and b with zero-padding."""
    return fftconvolve(a, b, mode='full')[:len(a)]

def score_guided_swap(h, ε=1e-5):
    """Perform a score-guided swap."""
    n = len(h)
    # Get current errors
    gamma = aperiodic_autocorr(h)
    C = 1 - (2/n) * gamma
    max_val = np.max(C)
    W = np.where(np.abs(C - max_val) < ε)[0]
    if len(W) == 0:
        return h
    # Compute scores as sum of neighbors for each index
    scores = np.zeros(n)
    for s in W:
        # Forward
        if 0 <= s <= n-1:
            for i in range(n-s):
                scores[i] += h[i+s]
        # Backward
        if 0 <= s <= n-1:
            for i in range(s, n):
                scores[i] += h[i-s]
    # Find top scoring indices (for swapping out)
    # and bottom (for swapping in)
    top_indices = np.argsort(scores)[-10:]
    bot_indices = np.argsort(scores)[:10]
    if len(top_indices) == 0 or len(bot_indices) == 0:
        return h
    # Pick one to swap
    i = top_indices[np.random.choice(len(top_indices))]
    j = bot_indices[np.random.choice(len(bot_indices))]
    if h[i] > 0 and h[j] < 1:
        delta = min(0.01, h[i], 1 - h[j])
        h[i] -= delta
        h[j] += delta
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n2 = 300
    dx = 2.0 / n2
    n1 = 120  # smaller binary grid
    half1 = n1 // 2
    # === Phase I: MILP binary optimization ===
    print("Phase I: Solving Binary MILP...")
    b = cp.Variable(n1, boolean=True)
    a = cp.Variable((n1, n1), boolean=True)  # additional variables for linearization
    gamma = cp.Variable(n1)  # stay symmetric
    L = cp.Variable()
    
    # Constrain total number of ones
    constraints = [cp.sum(b) == half1]
    
    # McCormick constraints for a[i,s] = b[i]*b[i+s]
    # Only for valid i,s combinations (i+s < n1)
    for i in range(n1):
        for s in range(i+1, n1):
            # Lagrange method: a[i,s] <= b[i], a[i,s] <= b[i+s], a[i,s] >= b[i] + b[i+s] - 1
            if i + s < n1:
                constraints.extend([
                    a[i, s] <= b[i],
                    a[i, s] <= b[i+s],
                    a[i, s] >= b[i] + b[i+s] - 1
                ])
    
    # Define gamma based on a
    for s in range(n1):
        if s == 0:
            gamma[s] = cp.sum(b)  # special case
        else:
            terms = [a[i, s] for i in range(n1) if i+s < n1]
            constraints.append(gamma[s] == cp.sum(terms))
            
    # Constraint that all gamma are lower bounded
    for s in range(n1):
        constraints.append(gamma[s] >= L)
    
    # Objective
    objective = cp.Maximize(L)
    
    prob = cp.Problem(objective, constraints)
    solver_time = min(0.2 * budget_s, 120)
    prob.solve(solver_time=solver_time, verbose=False, solver=cp.CBC, rel_gap=5e-3)
    
    # Extract binary solution
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        b_star = np.round(b.value).astype(int)
    else:
        # In case solver didn't converge, fallback to a heuristic solution
        b_star = np.zeros(n1, dtype=int)
        ones = np.random.choice(n1, half1, replace=False)
        b_star[ones] = 1
    print(f"Binary solution status: {prob.status}")
    
    # === Phase II: Upsample to target resolution ===
    print("Phase II: Upsampling...")
    x1 = np.linspace(0, 2, n1, endpoint=False)
    x2 = np.linspace(0, 2, n2, endpoint=False)
    # Interpolate
    h0 = np.interp(x2, x1, b_star)
    # Project to ensure the mass is exactly 1
    h0 = project_box_sum(h0, total=n2/2)
    h = h0.copy()
    
    # Optional smoothing: remove sharp peaks
    h = gaussian_filter1d(h, sigma=1.0)
    h = project_box_sum(h, total=n2/2)
    
    # === Phase III: Smooth max projected gradient descent ===
    print("Phase III: Smooth max gradient descent...")
    β = 2.0
    η = 1.0
    best_h = h.copy()
    best_c5 = 1 - (2 / n2) * np.min(aperiodic_autocorr(h))
    iteration = 0
    
    # Initialize momentum
    m = np.zeros(n2)
    beta_power = 1
    
    while time.time() - start_time < 0.85 * budget_s:
        if time.time() - start_time > 0.85 * budget_s:
            break
            
        # Compute autocorrelations
        gamma = aperiodic_autocorr(h)
        C = 1 - (2 / n2) * gamma
        w = softmax(β, C)
        grad = -(2 / n2) * (conv(w, h) + conv(w[::-1], h))
        
        # Back-tracking line search
        α = η
        for _ in range(20):  # Max steps before fail
            h_trial = project_box_sum(h - α * grad, total=n2/2)
            c5_trial = 1 - (2 / n2) * np.min(aperiodic_autocorr(h_trial))
            if c5_trial <= 1 - (2 / n2) * np.min(aperiodic_autocorr(h)):
                break
            α *= 0.5
            if α < 1e-8:
                break
        h = h_trial
        c5 = 1 - (2 / n2) * np.min(aperiodic_autocorr(h))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Update schedule for β
        iteration += 1
        if iteration % 150 == 0:
            β = min(β * 1.5, 200.0)
            beta_power += 1
            
        # Nesterov momentum update
        m = 0.9 * m - α * grad
        h = project_box_sum(h + 0.9 * m, total=n2/2)
        
        # Occasionally swap (score guided)
        if iteration % 200 == 0:
            h = score_guided_swap(h)
    
    # === Phase IV: Final sub-gradient polishing ===
    print("Phase IV: Final polishing...")
    while time.time() - start_time < budget_s - 0.1 * budget_s:
        # Compute C values
        gamma = aperiodic_autocorr(best_h)
        C = 1 - (2 / n2) * gamma
        worst_indices = np.where(np.abs(C - best_c5) < 1e-5)[0]
        # Compute sub-gradient
        g = np.zeros(n2)
        for s in worst_indices:
            if s == 0:
                g += best_h.copy()
            else:
                g[s:] += best_h[:-s].copy()
                g[:-s] += best_h[s:].copy()
        g = -(2 / n2) * g / len(worst_indices)
        # Apply step with diminishing learning rate
        η_step = 0.01 / np.sqrt(iteration + 1)
        h_new = project_box_sum(best_h - η_step * g, total=n2/2)
        c5_new = 1 - (2 / n2) * np.min(aperiodic_autocorr(h_new))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        best_h = h_new

    # Final evaluation
    final_c5 = 1 - (2 / n2) * np.min(aperiodic_autocorr(best_h))
    return (best_h, final_c5, n2)
