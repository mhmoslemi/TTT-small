def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def soft_ramp(h, w=2):
    """Smoothly ramp the binary step function h to real values"""
    # Input: h is a binary vector
    h = h.astype(float)
    n = len(h)
    h_smooth = np.zeros_like(h)
    for i in range(n):
        for j in range(int(max(0, i - w)), int(min(n, i + w + 1))):
            h_smooth[i] += h[j] * max(0, 1 - abs(i - j) / w)
    return h_smooth
    
def compute_corr_matrix(n):
    """Compute the matrix of circulant correlation operators."""
    C = np.zeros((n, n))
    for k in range(n):
        for i in range(n):
            j = (i - k) % n
            C[i, j] = 1
    return C

def alternative_initialization(n):
    """Heuristically construct a better starting solution."""
    h = np.zeros(n)
    # Place a central peak with wings
    mid = n // 2
    half_w = n // 6
    for i in range(max(0, mid - half_w), min(n, mid + half_w)):
        h[i] = 1.0
    return project_to_feasible(h, n)
