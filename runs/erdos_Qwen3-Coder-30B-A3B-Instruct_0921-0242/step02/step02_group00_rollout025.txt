import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import bisect


def project_box_sum(y, total):
    """Project y onto the set {x : 0 <= x_i <= 1, sum(x) = total}."""
    # Clip to [0,1]
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if abs(s - total) < 1e-10:
        return y

    # Use bisection on lambda to find correct shift
    def sum_no_lambda(lam):
        return np.sum(np.clip(y - lam, 0, 1))

    # Find bounds for lambda
    lb, ub = -1.0, 2.0
    while sum_no_lambda(lb) > total:
        lb -= 1.0
    while sum_no_lambda(ub) < total:
        ub += 1.0

    # Bisection
    def func(lam):
        return sum_no_lambda(lam) - total

    try:
        lam = bisect(func, lb, ub, xtol=1e-12, maxiter=100)
    except:
        # fallback if bisection fails
        lam = (total - s) / len(y)

    y = np.clip(y - lam, 0, 1)
    return y


def compute_autocorrelation(h):
    """Compute aperiodic autocorrelation of h."""
    n = len(h)
    # equivalent to correlate(h, h, mode='full')
    H = fft(h)
    P = np.abs(H)**2
    gamma = np.real(ifft(P))
    return gamma


def binary_swap_update(h, gamma, i, j):
    """Given h with indices i=1, j=0, return Δγ for all lags."""
    n = len(h)
    delta_gamma = np.zeros(n * 2 - 1)
    for k in range(-(n - 1), n):  # lag k from -(n-1) to (n-1)
        # cumulative terms from reading h
        if 0 <= i + k < n:
            delta_gamma[k + n - 1] -= h[i]
        if 0 <= i - k < n:
            delta_gamma[k + n - 1] -= h[i]
        if 0 <= j + k < n:
            delta_gamma[k + n - 1] += h[j]
        if 0 <= j - k < n:
            delta_gamma[k + n - 1] += h[j]
    return delta_gamma


def compute_c5_from_gamma(gamma, n):
    """Compute C5 from (cyclic) autocorrelation gamma."""
    gamma0 = gamma[n - 1]
    avg_side = (np.sum(gamma) - gamma0) / (n - 1) if n > 1 else 0.0
    c5 = 1.0 - (2.0 / n) * avg_side
    return c5


def binary_hill_climb(h_init, max_swaps=3000):
    """Performs hill-climbing with binary swaps."""
    h = h_init.copy()
    n = len(h)
    gamma = compute_autocorrelation(h)
    gamma0 = gamma[n - 1]
    avg_side = (np.sum(gamma) - gamma0) / (n - 1) if n > 1 else 0.0
    current_c5 = 1.0 - (2.0 / n) * avg_side

    best_h = h.copy()
    best_c5 = current_c5

    swaps_made = 0
    no_improvement_count = 0
    while swaps_made < max_swaps:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h[i] != 1 or h[j] != 0:
            continue

        delta_gamma = binary_swap_update(h, gamma, i, j)
        new_gamma = gamma + delta_gamma

        new_gamma0 = new_gamma[n - 1]
        new_avg_side = (np.sum(new_gamma) - new_gamma0) / (n - 1) if n > 1 else 0.0
        new_c5 = 1.0 - (2.0 / n) * new_avg_side

        if new_c5 < current_c5:
            h[i] = 0
            h[j] = 1
            gamma = new_gamma
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        swaps_made += 1
        if no_improvement_count >= max_swaps:
            break

    return best_h, best_c5


def generate_legendre_sequence(n):
    """Generate a Legendre sequence of length n, if possible."""
    # Find largest prime is 257.
    # Legendre(k) = (k | 257)
    def legendre_symbol(k, p):
        return pow(k, (p - 1) // 2, p) - 1 if k % p != 0 else 0

    p = 257
    if n > p - 1:
        n = p - 1
    s = np.array([1 if legendre_symbol(k, p) == 1 else 0 for k in range(1, p)])
    return s[:n]


def generate_m_sequence(n):
    """Generate a maximal-length m-sequence of length n."""
    # 8-bit LFSR: feedback polynomial x^8 + x^6 + x^5 + x^4 + 1
    lfsr = np.zeros(8, dtype=int)
    lfsr[0] = 1
    taps = [7, 5, 4, 3]
    sequence = []
    for _ in range(n - 1):
        out = lfsr[0]
        newbit = 0
        for t in taps:
            newbit ^= lfsr[t]
        lfsr = np.roll(lfsr, -1)
        lfsr[-1] = newbit
        sequence.append(out)
    # append a 0
    sequence.append(0)
    return np.array(sequence)


def generate_random_balanced(n, num_samples=5):
    """Generate random balanced binary strings."""
    samples = []
    total = n // 2
    for _ in range(num_samples):
        indices = np.random.choice(n, total, replace=False)
        s = np.zeros(n, dtype=int)
        s[indices] = 1
        samples.append(s)
    return samples


def eval_c5(h):
    """Evaluate C5 with proper scaling."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx


def spectral_alternating_projection(h, max_iter=400, alpha_start=0.7, alpha_inc=0.02):
    """Apply spectral alternating projection to flatten autocorrelation."""
    h = h.astype(float)
    n = len(h)
    alpha = alpha_start
    best_h = h.copy()
    best_c5 = eval_c5(h)
    prev_c5 = best_c5
    no_improvement_count = 0

    for i in range(max_iter):
        H = fft(h)
        P = np.abs(H)**2
        gamma = np.real(ifft(P))
        gamma0 = gamma[n - 1]
        avg_side = (np.sum(gamma) - gamma0) / (n - 1) if n > 1 else 0.0

        # construct γ_target
        gamma_target = gamma.copy()
        gamma_target[1:] = alpha * avg_side + (1 - alpha) * gamma_target[1:]

        # rebuild from target spectrum
        P_target = fft(gamma_target)
        A_target = np.sqrt(np.maximum(P_target.real, 0))
        H_new = A_target * np.exp(1j * np.angle(H))
        h_new = np.real(ifft(H_new))

        # project back onto T
        h_new = project_box_sum(h_new, n / 2)
        c5 = eval_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

        if abs(c5 - prev_c5) < 1e-6:
            no_improvement_count += 1
        else:
            no_improvement_count = 0

        if no_improvement_count >= 20:
            break

        h = h_new.copy()
        prev_c5 = c5
        alpha = min(alpha + alpha_inc, 0.99)

    return best_h, best_c5


def projected_gradient_polish(h_start, max_iter=300, beta_start=5.0):
    """Perform projected gradient descent on smooth surrogate."""
    h = h_start.copy()
    n = len(h)
    beta = beta_start
    prev_c5 = eval_c5(h)
    best_c5 = prev_c5
    best_h = h.copy()

    for i in range(max_iter):
        # compute C_k for all shifts k
        corr = correlate(h, 1 - h, mode='full')
        C = corr * (2.0 / n)  # scaled C_k
        C = 1.0 - C  # now C_k = 1 - (2/n)*γ_k

        w = np.exp(beta * C)
        w = w / np.sum(w)

        # compute gradient (approximate using FFT)
        h_fft = fft(h)
        w_fft = fft(w)
        g_fft = (w_fft.conj() * h_fft) + (np.flip(w_fft, axis=0).conj() * h_fft)
        g = -np.real(ifft(g_fft)) * (2.0 / n)

        # Backtracking line search
        eta = 1.0
        max_step = 10
        for _ in range(max_step):
            h_trial = h - eta * g
            h_trial = project_box_sum(h_trial, n / 2)
            c5_trial = eval_c5(h_trial)
            if c5_trial < prev_c5:
                h = h_trial
                prev_c5 = c5_trial
                if prev_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = prev_c5
                break
            else:
                eta *= 0.5

        beta *= 1.2  # sharpen max approximation

    return best_h, best_c5
