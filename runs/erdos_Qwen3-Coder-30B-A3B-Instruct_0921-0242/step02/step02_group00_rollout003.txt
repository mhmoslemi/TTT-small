import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import bisect
import math

def project_box_sum(y, total):
    """Project y onto the set {x: 0 ≤ x_i ≤ 1, sum x_i = total}"""
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if abs(s - total) < 1e-10:
        return y
    
    def sum_func(lam):
        return np.sum(np.clip(y - lam, 0, 1)) - total
    
    if s < total:
        # Need to increase values
        a, b = 0, 1 - np.min(y)  # Upper bound for lam
        lam = bisect(sum_func, a, b)
        return np.clip(y + lam, 0, 1)
    else:
        # Need to decrease values
        a, b = -np.min(1 - y), 0  # Lower bound for lam
        lam = bisect(sum_func, a, b)
        return np.clip(y - lam, 0, 1)

def compute_autocorrelation(h):
    """Compute the cyclic autocorrelation of h"""
    n = len(h)
    H = fft(h)
    P = np.abs(H)**2
    gamma = np.real(ifft(P))
    return gamma

def compute_c5(h):
    """Evaluate C5 for given h"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def build_binary_seed(n, seed_type):
    """Generate binary seeds of length n"""
    if seed_type == "legendre":
        # Use Legendre symbol for prime 257 to get 256 values
        p = 257
        h = np.zeros(n)
        for i in range(1, p):
            if i < n + 1:
                symbol = np.sign(np.power(i, (p-1)//2, p) - 1)
                if symbol == 0:
                    symbol = 1
                h[i-1] = (symbol + 1) // 2
        h = h[:n]
    elif seed_type == "mseq":
        # Maximal length sequence over GF(2) with degree 8 (255 elements + zeros)
        # We generate one m-sequence of length 255 and pad with 0
        lfsr = np.ones(8, dtype=int)
        sequence = np.zeros(255, dtype=int)
        for i in range(255):
            feedback = np.sum(lfsr[:2] * lfsr[6:8]) % 2
            sequence[i] = lfsr[7]
            lfsr = np.roll(lfsr, 1)
            lfsr[0] = feedback
        h = np.pad(sequence, (0, n - len(sequence)), 'constant')
    elif seed_type == "random_balanced":
        # Random binary vector with roughly n/2 ones
        h = np.zeros(n)
        ones_idx = np.random.choice(n, size=n//2, replace=False)
        h[ones_idx] = 1
    else:
        raise ValueError("Unsupported seed type")
    return h

def binary_hill_climb(h, max_iter=3000, timeout_s=50):
    """Perform local search on binary vectors to reduce C5"""
    n = len(h)
    gamma = compute_autocorrelation(h)
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5
    iter_count = 0
    
    # Precompute gamma from input
    gamma = compute_autocorrelation(h)
    
    while iter_count < max_iter:
        # Try to swap 1->0 and 0->1
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h[i] == 1 and h[j] == 0:
            # Compute change in autocorrelation values
            new_gamma = gamma.copy()  # Protect original
            for k in range(2*n-1):
                offset = k - (n-1)  # shift
                # Handle wraparound
                i_plus = (i + offset) % n
                i_minus = (i - offset) % n
                j_plus = (j + offset) % n
                j_minus = (j - offset) % n
                # Updated via convolution property
                delta_gamma = (
                    - h[i] * h[i_plus] - h[i] * h[i_minus] +
                    h[j] * h[j_plus] + h[j] * h[j_minus]
                )
                new_gamma[k] += delta_gamma
            
            # New c5 is one minus the worst shift in terms of new_gamma
            # Note: instead of computing the full correlation correctly,
            #   we directly update it:
            new_corr = 1 - (2/n) * new_gamma  # This is what C_k looks like
            c5_new = 1 - np.max(new_corr)
            
            if c5_new < c5:
                # Accept
                gamma = new_gamma
                c5 = c5_new
                h[j] = 1
                h[i] = 0
                if c5 < c5_best:
                    h_best = h.copy()
                    c5_best = c5
        iter_count += 1
        if timeout_s and time.time() - t0 > timeout_s:
            break
    return h_best, c5_best

def spectral_alternating_projection(h_bin, max_iter=400):
    """Perform SAP using target power spectrum approach"""
    n = len(h_bin)
    h = h_bin.astype(float)
    alpha = 0.7
    best_c5 = compute_c5(h)
    best_h = h.copy()
    last_improvement = 0
    
    for iter_i in range(max_iter):
        # Compute current power spectrum and autocorrelation
        H = fft(h)
        P = np.abs(H)**2
        gamma = np.real(ifft(P))
        
        # Compute target autocorrelation
        gamma0 = gamma[0]
        avg_side = (np.sum(gamma) - gamma0) / (n-1) if n > 1 else 0
        gamma_target = (1 - alpha) * gamma + alpha * avg_side
        gamma_target[0] = gamma0  # Keep lag zero
        
        # Convert to frequency domain
        P_target = fft(gamma_target)
        P_target = np.maximum(P_target, 0)
        A_target = np.sqrt(P_target)
        
        # Preserve phase
        H_target = A_target * np.exp(1j * np.angle(H))
        
        # Back to time domain
        h_new = np.real(ifft(H_target))
        h_new = project_box_sum(h_new, n/2)
        
        h = h_new
        
        # Compute new C5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iter_i
        
        if iter_i - last_improvement > 20:
            break
            
        # Slowly increase alpha to reduce peakiness
        if alpha < 0.95:
            alpha += 0.01
            
    return best_h, best_c5

def projected_gradient_polish(h, max_iter=300, beta_start=5):
    """Apply short gradient descent with soft maximum"""
    n = len(h)
    h = h.copy()
    beta = beta_start
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for i in range(max_iter):
        # Compute side-lobe values
        corr = correlate(h, 1 - h, mode='full')
        c_values = 1 - (2/n) * corr  # Not tied with actual definition yet, we’ll fix it.
        # Reconstruct proper avg_corr:
        dx = 2./n
        avg_lobes = (np.sum(c_values) - c_values[n-1]) / (2*n - 2)
        c_values = (1 - alpha) * c_values + alpha * avg_lobes
        
        # We'll approximate using soft max via FFT, 
        # though fast version uses a linear approximation to gradient of a smoothed max over all shifts
        # Soft max weights:
        w = np.exp(beta * c_values)
        w = w / np.sum(w)
        
        # Now compute ∇ = -(2/n)*(conv(w,h) + conv(w[::-1], h)
        # Requires FFT:
        w_fft = fft(w)
        h_fft = fft(h)
        conv_w_h = ifft(w_fft * h_fft)
        conv_w_rev_h = ifft(w_fft[::-1] * h_fft)
        grad = -(2/n) * (np.real(conv_w_h) + np.real(conv_w_rev_h))
        
        # Back-tracking line search
        eta = 1.0
        converged = False
        while eta > 1e-8 and not converged:
            h_trial = h - eta * grad
            h_trial = project_box_sum(h_trial, n/2)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                h = h_trial
                best_c5 = c5_trial
                best_h = h.copy()
                converged = True
            else:
                eta *= 0.5
                
        # Update beta to sharpen soft max
        beta = beta * 1.2
        
        if np.abs(best_c5 - compute_c5(h)) < 1e-7 and i > 5:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    global t0
    t0 = time.time()
    
    n = 256
    dx = 2.0 / n
    
    # Build initial seeds
    seeds = []
    for _ in range(5):
        # Create a few static seeds
        h = build_binary_seed(n, "legendre")
        h = project_box_sum(h, n/2)
        seeds.append(h)
        
        h = build_binary_seed(n, "mseq")
        h = project_box_sum(h, n/2)
        seeds.append(h)
        
        h = build_binary_seed(n, "random_balanced")
        h = project_box_sum(h, n/2)
        seeds.append(h)
    
    # If there's an initial construction from external source, interpolate that
    if 'initial_h_values' in globals():
        init_h = initial_h_values[:n]
        if len(init_h) < n:
            x_orig = np.linspace(0, 2, len(init_h))
            x_new = np.linspace(0, 2, n)
            init_h_interp = np.interp(x_new, x_orig, init_h)
        else:
            init_h_interp = init_h
    
        init_h_proj = project_box_sum(init_h_interp, n/2)
        seeds.append(init_h_proj)

    # Binary hill-climb for all seeds
    best_h_bin = None
    best_c5_bin = float('inf')
    for i_seed, seed_h in enumerate(seeds):
        h_final, c5_final = binary_hill_climb(seed_h, max_iter=3000, timeout_s=budget_s * 0.05)
        if c5_final < best_c5_bin:
            best_c5_bin = c5_final
            best_h_bin = h_final.copy()
    
    # SAP
    h_sap, c5_sap = spectral_alternating_projection(best_h_bin, max_iter=400)
    
    # Final polish
    h_final, c5_final = projected_gradient_polish(h_sap, max_iter=300)
    
    # Final evaluation of all three
    final_eval = compute_c5(h_final)
    
    return (h_final, final_eval, n)
