import numpy as np
import time
from scipy.optimize import differential_evolution, linprog
from scipy.fft import rfft, irfft
from scipy.interpolate import interp1d

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    # Water-filling approach
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def build_basis(n0):
    """Build cosine and sine basis functions for a grid of size n0."""
    x = np.linspace(0, 2, n0, endpoint=False)
    basis_cos = [np.cos(m * np.pi * x) for m in range(1, 13)]
    basis_sin = [np.sin(m * np.pi * x) for m in range(1, 13)]
    return np.array(basis_cos + basis_sin)

def basis_evaluation(coefficients, basis):
    """Generate h from given basis coefficients."""
    a0 = 0.5
    h_raw = a0 + basis.T @ coefficients
    h_raw = np.clip(h_raw, 0, 1)
    return h_raw

def de_fitness(coefficients, basis, n0):
    """Fitness for differential evolution."""
    h = basis_evaluation(coefficients, basis)
    h = project_to_feasible(h, n0)  # Ensures ∫ h dx = 1
    c5 = evaluate_c5(h)
    return c5

def linearise_and_solve_lp(h, k_star):
    """Linearise and solve LP for improvement."""
    n = len(h)
    # Active constraints
    A_k = []
    b = []
    t = 0  # placeholder
    # Build linearized constraints for the worst k shifts
    # Since we're doing it heuristically:
    A_k = []
    b = []
    # Start with a default selection of several worst shifts
    # For now, we use a small heuristic and will avoid too dense constraints
    worst_k_indices = [k_star]  # Start with only the worst one
    for k in worst_k_indices:
        # gradient of A(k) at h
        # A(k) = h^T P_k h where P_k shifts by k
        g = np.roll(h, k) + np.roll(h, -k)
        A_k.append(g)
        b.append(0)  # Constraint vector to be adjusted later

    A_k = np.array(A_k)  # Shape (num_constraints, n)
    
    # Construct LP: maximize t subject to
    # A_k @ h >= t * ones
    # Let C be a matrix formed by appending [A_k, -1] (column for t)
    # Add bounds on h, but t can go arbitrarily up (but our goal is increasing t)

    # Use standard formulation:
    # minimize -t (i.e., maximize t)
    # subject to
    # A_k @ h - t >= 0
    # 0 <= h_i <= 1
    # sum(h) = n/2

    # Linearized constraints:
    num_constraints = A_k.shape[0]
    c = np.zeros(n + 1)
    c[-1] = -1.  # maximize t

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.  # sum h = n/2
    b_eq = [n / 2.]

    A_ub = np.zeros((num_constraints, n + 1))
    A_ub[:, :n] = -A_k  # for h^T * g >= t  => -g^T h + t <= 0
    b_ub = [0.] * num_constraints

    bounds = [(0, 1)] * n + [(None, None)]  # h bounded [0,1], t unbounded

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        new_h = result.x[:-1]
        t_opt = result.x[-1]
        return new_h, t_opt
    else:
        # fallback to identity
        return h.copy(), np.inf

def subgradient_step(h, n, dx):
    """Perform one subgradient update, switching mass to decrease C5."""
    # Find the worst shift
    corr = np.correlate(h, 1 - h, mode='full')
    k_star = np.argmax(corr)
    if k_star >= len(h):
        k_shift = k_star - len(h)
    else:
        k_shift = k_star - len(h) + 1
    
    # Compute gradient
    # gradient of h^T P_k (1-h) = (P_k h) - (P_k^T h) = 2 * roll(h, k)
    g = (np.roll(h, k_shift) + np.roll(h, -k_shift))  # actually -2 * sign shift
    # Let's recompute corr again with correct mode for base case
    corr_full = np.correlate(h, 1 - h, mode='full')
    max_corr_value = corr_full[k_star]
    # Corr = h^T P_k h
    # We're minimizing max_k of corr, so setting the k* constraint:
    # h^T P_k* h >= max_corr_value
    # Gradient (翕)?    h^T (A_k* h) => both terms are === (P_k h + P_{-k} h) = 2*(roll(h,-k) + roll(h,k)) 
    # ulteriorly, should getignment of contribution of each component, 
    # Alternative: Compute subgradient direction from roll(h,-k) and roll(h,k)
    
    # Corresponds to sum over i: (h[i] * 1 + h[i] * 1) = 2 * h[i], but we do it with circles
    # ∇_i A(k) = (1_{i-k} + 1_{i+k}) = 2*roll(h,k) ?

    estimated_grad = 2 * np.roll(h, -k_shift) * (1 - h)
    estimated_grad = np.roll(estimated_grad, k_shift) - 2 * h * (1 - h)
    # better way, from sheer standpoint of change
    # h is being modified by swapping mass from indices j to i
    # i moves towards lower; j towards higher (k shift)
    # Elect its slope components.
    # We already now the direction that goes down from particular point
    # for each element, we can compute direct change
    # Track largest % change in specific point  
    # For binary moves "subgradient computation"]]
    # Apply simple swap trick on worsening side only
    if np.sum(h) < n/2:
        delta = n/2 - np.sum(h)  # excess
        # Tend to increase h at low extract its mass at high:
        # both ends point in the direction of decreasing overlap
        usei = np.argmax(h[:n//2])   # Start high energy and high
        used = np.argmin(h[n//2:])      # End should be low/high  
        used += n//2                   # Adjust to full array indexing
        low = h[usei] - 0.001  # Low boundary
        high = h[used]  # High boundary
        if low > 0 and high < 1:
            if low + high > 1:
                h[used] = 1 - low
                h[usei] = 0.001
            else:
                h[used] -= 0.001
                h[usei] += 0.001
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # Phase A: Setup
    n0 = 250  # low resolution for global search
    basis = build_basis(n0)
    d = basis.shape[0]  # number of coefficients

    # Phase B: Global DE Search
    bounds = [(-0.5, 0.5)] * d  # coefficient bounds
    result = differential_evolution(de_fitness, bounds, args=(basis, n0), maxiter=200, popsize=20, seed=seed)
    best_c = result.x
    current_h = basis_evaluation(best_c, basis)
    # Refine with projection
    current_h = project_to_feasible(current_h, n0)
    current_c5 = evaluate_c5(current_h)

    # keep 5 best candidates
    candidates = [(best_c, current_h, current_c5)]
    
    # Phase D: Sequential Linear Programming (SLP) refinement
    best_refined_c5 = current_c5
    best_refined_h = current_h.copy()

    for round in range(300):  # limited loop in time
        if time_left() < 0.1:
            break
        # get best approx from each param
        c = candidates[0][0]
        h = candidates[0][1]
        c5 = candidates[0][2]
        
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        
        # linearize and solve LP
        h_new, t_new = linearise_and_solve_lp(h, k_star)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5:
            candidates[0] = (c, h_new, c5_new)
            if c5_new < best_refined_c5:
                best_refined_c5 = c5_new
                best_refined_h = h_new.copy()
        else:
            # damped search
            alpha = 0.5
            h_trial = (1 - alpha) * h + alpha * h_new
            h_trial = project_to_feasible(h_trial, n0)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < c5:
                candidates[0] = (c, h_trial, c5_trial)
                if c5_trial < best_refined_c5:
                    best_refined_c5 = c5_trial
                    best_refined_h = h_trial.copy()
            else:
                break  # No improvement, move on
    h = best_refined_h.copy()
    c5 = best_refined_c5

    # Phase E: Upsampling and polish
    # interpolate up to higher resolution
    n1 = 500
    x_original = np.linspace(0, 2, n0, endpoint=False)
    x_upsampled = np.linspace(0, 2, n1, endpoint=False)

    # rebuild the function from coefficients, then interpolate
    # no need to re-run optim, just rebuild
    h_upsampled = basis_evaluation(best_c, build_basis(n1))
    h_upsampled = project_to_feasible(h_upsampled, n1)

    # Final polish with sub-gradient
    for i in range(500):
        if time_left() < 0.1:
            break
        h_upsampled = subgradient_step(h_upsampled, n1, 2.0/n1)
        c5_upsampled = evaluate_c5(h_upsampled)
        if c5_upsampled < c5:
            c5 = c5_upsampled
            h = h_upsampled.copy()
        else:
            break

    return (h, c5, n1)
