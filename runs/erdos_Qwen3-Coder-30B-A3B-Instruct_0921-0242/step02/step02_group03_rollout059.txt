import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

            # Find zeros and flip to ones
            zero_indices = np.where(b_initial == 0)[0]

                b_initial[zero_indices[:n_flip]] = 1
            else:
                b_initial[zero_indices] = 1
        else:

            one_indices = np.where(b_initial == 1)[0]

            else:
                b_initial[one_indices] = 0

        dx = 2.0 / n_points

        corr = correlate(h, 1 - h, mode='full')
        # Ignore shift 0 which is trivial
        non_zero_corr = np.concatenate([corr[1:], corr[:n_points-1]])

        # Compute gradient for that shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            right_idx = (i + worst_shift) % n_points
            left_idx = (i - worst_shift) % n_points
            grad[i] = (1 - h[right_idx] - h[left_idx]) * dx

        step = 0.01

            h_trial = project_to_feasible(h_trial, n_points)
            if evaluate_c5(h_trial) < evaluate_c5(h):
                return h_trial
            step *= 0.5
        return h

    # Run several steps of memetic refinement to sharpen
    for _ in range(10):
        h_continuous = memetic_descent_step(h_continuous, n)
        if time.time() > time_limit - 2:
            break

    c5_opt = evaluate_c5(h_continuous)
    
    return (h_continuous, c5_opt, n)
