import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import bisect

def project_box_sum(y, target_sum):
    """Project y onto the set {x: 0 ≤ x ≤ 1, sum(x) = target_sum}."""
    def S(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    
    # bisection to find lambda such that S(lambda) = target_sum
    lb = np.min(y) - 1.0
    ub = np.max(y) + 1.0
    eps = 1e-12
    while ub - lb > eps:
        mid = (lb + ub) / 2.0
        if S(mid) > target_sum:
            lb = mid
        else:
            ub = mid
    lam = (lb + ub) / 2.0
    x = np.clip(y - lam, 0, 1)
    return x

def evaluate_c5(h):
    """Computes C5 = max_k ∫ h(x)(1-h(x+k)) dx."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_worst_shifts(C, eps=1e-5):
    """Return the indices s such that C[s] is within eps of max(C)."""
    max_C = np.max(C)
    return np.where(np.abs(C - max_C) < eps)[0]

def compute_autocorr(h):
    """Compute autocorrelation: γ_s = Σ_i h[i] h[i+s] for all s."""
    return correlate(h, h, mode='full')

def compute_autocorr_update(γ, h, i, j, Δ):
    """Incrementally update autocorrelation γ for transfer from i to j."""
    n = len(h)
    new_γ = γ.copy()
    # Create temporary vector with i not contributing (h[i]=0)
    new_h = h.copy()
    new_h[i] = 0

    for s in range(-n+1, n):
        idx = s + n - 1
        # removal of Δ from i
        if 0 <= i + s < n:
            new_γ[idx] -= Δ * new_h[i + s]
        if 0 <= i - s < n and s != 0:
            new_γ[idx] -= Δ * new_h[i - s]

        # addition of Δ to j
        if 0 <= j + s < n:
            new_γ[idx] += Δ * h[j + s]
        if 0 <= j - s < n and s != 0:
            new_γ[idx] += Δ * h[j - s]

    # self-term (s=0): exact correction for quadratic change
    new_γ[n-1] += -2 * h[i] * Δ + Δ * Δ + 2 * h[j] * Δ + Δ * Δ
    return new_γ

def compute_scores(h, worst_shifts):
    """Compute the score for each index based on worst shifts."""
    n = len(h)
    scores = np.zeros(n)
    for s in worst_shifts:
        # shift +s
        if s >= 0:
            scores[:n-s] += h[s:]
        else:
            scores[-s:] += h[:n+s]
        # shift -s (but skip s=0 since reshape is_same)
        if s != 0:
            if s > 0:
                scores[s:] += h[:-s]
            else:
                scores[:n+s] += h[-s:]
    return scores

def get_best_move_source(h, worst_shifts, eta=1e-8):
    """Select a source index with max score."""
    score = compute_scores(h, worst_shifts)
    candidates = np.where(h > eta)[0]
    if len(candidates) == 0:
        return None
    idx = np.argmax(score[candidates])
    return candidates[idx]

def get_best_move_target(h, worst_shifts, eta=1e-8):
    """Select a target index with min score."""
    score = compute_scores(h, worst_shifts)
    candidates = np.where(h < 1 - eta)[0]
    if len(candidates) == 0:
        return None
    idx = np.argmin(score[candidates])
    return candidates[idx]

def get_move_delta(h, i, j, Δ_max):
    """Get a delta ensuring h remains in valid box."""
    Δ = np.random.uniform(0, min(Δ_max, h[i], 1 - h[j]))
    return Δ
