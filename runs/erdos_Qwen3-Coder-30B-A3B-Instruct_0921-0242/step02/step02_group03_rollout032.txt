import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def build_trapezoid(t, H, n):
    """Build a trapezoidal function h(x) with parameters t (ramp length) and H (height)."""
    dx = 2.0 / n
    h = np.zeros(n)
    
    L = 1.0 / H - t
    a = (2.0 - 2.0 * t - L) / 2.0
    
    if L < 0 or a < 0:
        return h
    
    # Indices for ramps and top
    i_a = int(a / dx)
    i_t1 = int((a + t) / dx)
    i_t2 = int((a + t + L) / dx)
    i_t3 = int((a + 2 * t + L) / dx)
    
    # Left ramp
    if t > 0:
        for i in range(i_a, i_t1):
            x = (i + 0.5) * dx
            h[i] = H * (x - a) / t
    
    # Top
    for i in range(i_t1, i_t2):
        h[i] = H
    
    # Right ramp
    if t > 0:
        for i in range(i_t2, i_t3):
            x = (i + 0.5) * dx
            h[i] = H * (a + 2.0 * t + L - x) / t
    
    return h

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    if diff > 0:
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    h_copy = np.clip(h_copy, 0, 1)
    return h_copy

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT correlation."""
    dx = 2.0 / len(h)
    N = 2 * len(h)
    
    Hc = fft(h, N)
    Gc = fft(1 - h, N)
    corr = ifft(Hc * np.conj(Gc)).real
    corr = corr[:2 * len(h) - 1]
    
    max_corr = np.max(corr) * dx
    return max_corr

def memetic_descent(h, n_points, max_steps=30):
    """Refine h using projected gradient descent."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Full correlation of h with (1 - h)
        corr = correlate(h_current, 1 - h_current, mode='full')
        # Scale by dx and find maximum
        max_corr = np.max(corr) * dx
        
        # Find the worst shift index
        non_zero_shifts = np.concatenate([corr[1:], corr[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5_fft(h_trial)
            current_c5 = evaluate_c5_fft(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s
    
    n = 800
    dx = 2.0 / n
    best_val = float('inf')
    best_h = None
    
    # ---- 2-D parametric search ----
    H_vals = np.linspace(0.5, 1.0, 41)
    for H in H_vals:
        t_max = min(1.0/H, 2.0 - 1.0/H)
        t_vals = np.linspace(0.0, t_max, 81)
        for t in t_vals:
            h = build_trapezoid(t, H, n)
            h = project_to_feasible(h, n)
            val = evaluate_c5_fft(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
    
    # ---- refine around the best pair ----
    if time.time() < time_limit - 1.0:
        # Extract the best (t,H) from the best_h
        # since we don't have a direct inverse, we approximate
        # (this is a simplification; in practice, would need to derive t, H from best_h)
        # Just recompute over fine grid around current best
        H_refined = np.linspace(max(0.5, H - 0.025), min(1.0, H + 0.025), 41)
        t_refined = np.linspace(max(0.0, t - 0.01), min(t_max, t + 0.01), 41)
        
        for H in H_refined:
            t_max = min(1.0/H, 2.0 - 1.0/H)
            for t in t_refined:
                if t > t_max:
                    continue
                h = build_trapezoid(t, H, n)
                h = project_to_feasible(h, n)
                val = evaluate_c5_fft(h)
                if val < best_val:
                    best_val, best_h = val, h.copy()
    
    # ---- polishing with memetic descent ----
    if time.time() < time_limit - 1.0:
        best_h = memetic_descent(best_h, n, max_steps=20)
        best_val = evaluate_c5_fft(best_h)
    
    # ---- fallback random memetic runs if time permits ----
    while time.time() < time_limit - 5.0:
        h0 = np.random.rand(n)
        h0 = project_to_feasible(h0, n)
        h_opt = memetic_descent(h0, n, max_steps=30)
        val = evaluate_c5_fft(h_opt)
        if val < best_val:
            best_val, best_h = val, h_opt.copy()
    
    return (best_h, best_val, n)
