import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from math import sqrt

def project_box_sum(y, total):
    """Project y onto {x : 0 ≤ x_i ≤ 1, sum x_i = total}."""
    y = np.array(y, dtype=float)
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if abs(s - total) < 1e-10:
        return y
    
    # Find lambda such that sum(clamp(y - lambda, 0, 1)) = total
    # We know that S(lambda) = sum(clamp(y - lambda, 0, 1)) is decreasing in lambda
    # Start with a reasonable guess and use bisection
    low, high = -1.0, 1.0
    for _ in range(40):  # Bisection iterations for double precision
        mid = (low + high) / 2.0
        z = np.clip(y - mid, 0, 1)
        s_z = np.sum(z)
        if s_z > total:
            low = mid
        else:
            high = mid
    lambda_opt = (low + high) / 2.0
    y = np.clip(y - lambda_opt, 0, 1)
    return y

def compute_autocorr(h):
    """Compute the cyclic autocorrelation of h."""
    H = fft(h)
    P = np.abs(H)**2
    gamma = ifft(P).real
    return gamma

def compute_C5_from_autocorr(gamma, n):
    """Compute C5 from cyclic autocorrelation gamma."""
    gamma0 = gamma[0]
    avg = (np.sum(gamma) - gamma0) / (n - 1)
    # Target γ to flatten side-lobe energy
    gamma_target = gamma.copy()
    gamma_target[1:] = 0.7 * gamma[1:] + 0.3 * avg
    return 1.0 - (2.0 / n) * np.max(gamma_target)

def compute_C5(h):
    """Compute C5 for a given h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def generate_legendre_seed(n):
    """Generate a Legendre symbol sequence of length n."""
    # Use prime p=257 -> sequence length 256
    p = 257
    if n > p - 1:
        n = p - 1
    h = np.array([1 if ((k * k) % p) < p // 2 else 0 for k in range(1, p)]).astype(int)
    return h[:n]

def generate_msequence_seed(n):
    """Generate a maximal-length LFSR sequence of length n."""
    # Primitive polynomial for degree 8: x^8 + x^6 + x^5 + x^4 + 1
    # or equivalently, for the sequence of length 255, add one zero to make it n=256.
    # We generate an m-sequence by shifting and taking parity of feedback.
    # Let's instead attach a zero to the truncated output of a real LFSR.
    lfsr = np.ones(8, dtype=int)
    seq = np.zeros(255, dtype=int)
    for i in range(255):
        new_bit = lfsr[0] ^ lfsr[4] ^ lfsr[5] ^ lfsr[6]
        lfsr = np.roll(lfsr, -1)
        lfsr[-1] = new_bit
        seq[i] = new_bit
    seq = np.append(seq, 0)  # length 256
    return seq[:n]

def generate_random_balanced_seed(n, seed=42):
    """Generate a random balanced binary seed."""
    np.random.seed(seed)
    h = np.zeros(n, dtype=int)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1
    return h

def binary_hill_climb(h, max_swaps=None):
    """Perform binary hill-climbing using exact autocorrelation updates."""
    n = len(h)
    gamma = compute_autocorr(h)
    if max_swaps is None:
        max_swaps = 3 * n
    
    iter_count = 0
    best_h = h.copy()
    best_gamma = gamma.copy()
    best_c5 = compute_C5_from_autocorr(best_gamma, n)
    
    # Precompute all non-zero shifts
    shifts = list(range(1, n))
    
    while iter_count < max_swaps:
        # Pick a random pair (i, j) with h[i] = 1, h[j] = 0
        i_candidates = np.where(h == 1)[0]
        j_candidates = np.where(h == 0)[0]
        if len(i_candidates) == 0 or len(j_candidates) == 0:
            break
        i = np.random.choice(i_candidates)
        j = np.random.choice(j_candidates)
        
        # Compute delta gamma
        dg = np.zeros(n)
        for k in shifts:
            dg[k] = -h[i] * h[(i + k) % n] - h[i] * h[(i - k) % n] \
                     + h[j] * h[(j + k) % n] + h[j] * h[(j - k) % n]
            if k != 0:
                dg[-k] = dg[k]  # symmetry of autocorrelation for real sequence
        
        gamma_new = gamma + dg
        gamma_max = np.max(gamma_new)
        
        # Evaluate change in C5
        new_c5 = 1.0 - (2.0 / n) * gamma_max
        if new_c5 < best_c5:
            h[i] = 0
            h[j] = 1
            gamma = gamma_new
            best_h = h.copy()
            best_gamma = gamma.copy()
            best_c5 = new_c5
            
        iter_count += 1
    return best_h, best_c5

def sap_phase(h_bin, max_iter=400):
    """Perform Spectral Alternating Projection."""
    n = len(h_bin)
    h = h_bin.astype(float)
    alpha = 0.7
    best_h = h.copy()
    best_c5 = compute_C5(h)
    
    for iter_num in range(max_iter):
        # Compute current autocorrelation
        gamma = compute_autocorr(h)
        gamma0 = gamma[0]
        avg = (np.sum(gamma) - gamma0) / (n - 1)
        
        # Target autocorrelation to flatten side lobes
        gamma_target = gamma.copy()
        gamma_target[1:] = alpha * gamma[1:] + (1 - alpha) * avg
        gamma_target[0] = gamma0  # preserve energy
        
        # Compute target magnitude in frequency domain
        P_target = fft(gamma_target)
        A_target = np.sqrt(np.maximum(P_target.real, 0))
        
        # Replace magnitude preserving phases
        H = fft(h)
        H_new = A_target * np.exp(1j * np.angle(H))
        h_new = ifft(H_new).real
        
        # Project back to feasible set
        h_new = project_box_sum(h_new, n / 2)
        h = h_new
        c5 = compute_C5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Adjust alpha
        if iter_num % 10 == 0:
            alpha = min(alpha + 0.02, 0.99)
    return best_h, best_c5

def gradient_polish(h, max_iter=300, beta_start=5.0):
    """Polish the solution via projected gradient descent with soft-max surrogate."""
    n = len(h)
    beta = beta_start
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_C5(h)
    
    for i in range(max_iter):
        # Compute weights
        gamma = compute_autocorr(h)
        C = 1.0 - (2.0 / n) * gamma
        w = np.exp(beta * C)
        w = w / np.sum(w)
        
        # Compute gradient
        h_conv = np.convolve(w, h, mode='full')
        h_conv = h_conv[:n]
        h_conv_reverse = np.convolve(w[::-1], h, mode='full')
        h_conv_reverse = h_conv_reverse[:n]
        grad = -(2.0 / n) * (h_conv + h_conv_reverse)
        
        # Backtracking line search
        eta = 1.0
        while True:
            h_new = project_box_sum(h - eta * grad, n / 2)
            c5_new = compute_C5(h_new)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                break
            else:
                eta *= 0.5
                if eta < 1e-10:
                    break
        beta *= 1.2
        if beta > 20:
            beta = 20
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization run."""
    np.random.seed(seed)
    start_time = time.time()
    
    n = 256  # Use power of two for FFT efficiency
    dx = 2.0 / n
    
    print("Starting SAP-based optimization for Erdős constant...")
    
    # Phase 1: Generate seeds and hill-climb
    print("Phase 1: Generating seeds and binary hill-climbing...")
    seeds = []
    seeds.append(("Legendre", generate_legendre_seed(n)))
    seeds.append(("m-sequence", generate_msequence_seed(n)))
    seeds.append(("Random-balanced", generate_random_balanced_seed(n, seed=41)))
    seeds.append(("Random-balanced2", generate_random_balanced_seed(n, seed=43)))
    
    # Add initial_h_values if available
    try:
        # Interpolate from initial_h_values, which may be different length
        x_orig = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_orig, initial_h_values)
        h_interp = project_box_sum(h_interp, n/2)
        seeds.append(("Initial", h_interp))
    except:
        pass
    
    # Evaluate and hill-climb all seeds
    best_h_before_sap = None
    best_c5_before_sap = float('inf')
    
    for name, h_seed in seeds:
        # Project to feasible domain
        h_projected = project_box_sum(h_seed, n/2)
        h_bin, c5_bin = binary_hill_climb(h_projected)
        if c5_bin < best_c5_before_sap:
            best_c5_before_sap = c5_bin
            best_h_before_sap = h_bin.copy()
        print(f"{name} seeds: C5 = {c5_bin:.6f}")
    
    if time.time() - start_time > 0.9 * budget_s:
        return (best_h_before_sap, best_c5_before_sap, n)
        
    print(f"Best binary solution: C5 = {best_c5_before_sap:.6f}")

    # Phase 2: SAP
    print("Phase 2: Spectral alternating projection...")
    h_sap, c5_sap = sap_phase(best_h_before_sap)
    print(f"After SAP: C5 = {c5_sap:.6f}")
    best_h = h_sap.copy()
    best_c5 = c5_sap
    
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining < 20:
        return (best_h, best_c5, n)

    # Phase 3: Projected gradient polishing
    print("Phase 3: Final gradient polishing...")
    h_polished, c5_polished = gradient_polish(best_h)
    print(f"After polishing: C5 = {c5_polished:.6f}")
    
    if c5_polished < best_c5:
        best_h = h_polished
        best_c5 = c5_polished

    return (best_h, best_c5, n)
