import numpy as np
import time
from scipy.signal import fftconvolve
import cvxpy as cp
from scipy.ndimage import gaussian_filter

def aperiodic_autocorr(h):
    """Compute aperiodic autocorrelation via FFT."""
    n = len(h)
    # Pad to 2*n - 1 for cross-correlation
    H = np.fft.fft(h, 2*n - 1)
    ac = np.fft.ifft(H * np.conj(H)).real
    return ac[:n]

def softmax(beta, x):
    """Numerically stable softmax with temperature beta."""
    x_stable = x - np.max(x)
    exp_x = np.exp(beta * x_stable)
    return exp_x / np.sum(exp_x)

def convolve_fft(a, b):
    """Convolve a and b with zero-padding, FFT-based."""
    n = len(a)
    m = len(b)
    pad_len = n + m - 1
    A = np.fft.fft(a, pad_len)
    B = np.fft.fft(b, pad_len)
    result = np.fft.ifft(A * B).real
    return result[:n]

def project_box_sum(h, total):
    """Project h to [0,1] and normalize to sum = total."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - total) < 1e-10:
        return h
    diff = total - current_sum
    # Simple additive shift
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def compute_c5(h):
    """Compute C5 = max_s (1 - 2*gamma_s/n)."""
    n = len(h)
    dx = 2.0 / n
    gamma = aperiodic_autocorr(h)
    C_s = 1 - (2 / n) * gamma
    return np.max(C_s)

def score_guided_swap(h, num_swaps=5):
    """Swap small amounts between indices with extremes in scoring."""
    n = len(h)
    # Identify worst shifts
    gamma = aperiodic_autocorr(h)
    C_s = 1 - (2 / n) * gamma
    worst_thresh = np.max(C_s) - 1e-5
    worst_indices = np.where(C_s >= worst_thresh)[0]
    if len(worst_indices) == 0:
        return h
    
    # Compute scores for each index
    scores = np.zeros(n)
    for s in worst_indices:
        # Forward and backward shifts
        if s > 0:
            scores += h[s:] + np.pad(h[:n-s], (s, 0), 'constant')
        else:
            scores += h + h  # This is for s = 0, self-correlation
    scores[1:] += scores[:-1]  # Simplified for efficiency
    scores = scores[:n]  # Ensure right size
    
    # Find indices with low/high scores
    low_idx = np.argmin(scores)
    high_idx = np.argmax(scores)
    if low_idx == high_idx:
        return h
    # Move small amount from high_idx to low_idx
    move_amount = min(0.01, h[high_idx], 1 - h[low_idx])
    if move_amount < 1e-6:
        return h
    h[high_idx] -= move_amount
    h[low_idx] += move_amount
    # Project back
    h = project_box_sum(h, n / 2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # === Phase I: Binary MILP to find good seed pattern ===
    n1 = 120  # smaller size for MILP
    half = n1 // 2  # number of 1s needed
    
    # Build the model
    b = cp.Variable(n1, boolean=True)
    a = cp.Variable((n1, n1), boolean=True)  # a[i,s] = b[i]*b[i+s]
    gamma = cp.Variable(n1)  # autocorrelations
    L = cp.Variable()  # lower bound on all gamma_s
    
    # Define constraints
    constraints = [
        cp.sum(b) == half,
    ]
    
    # McCormick constraints for a[i,s] = b[i]*b[i+s]
    for i in range(n1):
        for s in range(n1 - i):
            constraints += [
                a[i, s] <= b[i],
                a[i, s] <= b[i + s],
                a[i, s] >= b[i] + b[i + s] - 1,
            ]
    
    # Define gamma terms
    for s in range(n1):
        constraints += [
            gamma[s] == cp.sum(a[:, s])
        ]
    
    # Enforce lower bound
    for s in range(n1):
        constraints += [gamma[s] >= L]
    
    # Objective
    objective = cp.Maximize(L)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC, time_limit=0.2 * budget_s, verbose=False, ccp=True)
    
    # Get the result
    b_star = np.round(b.value).astype(int)
    
    # === Phase II: Upscale and smooth ===
    n2 = 300
    x1 = np.linspace(0, 2, n1, endpoint=False)
    x2 = np.linspace(0, 2, n2, endpoint=False)
    h0 = np.interp(x2, x1, b_star)
    h0 = project_box_sum(h0, n2 / 2)
    
    # Apply mild Gaussian blur
    h0 = gaussian_filter(h0, sigma=1)
    h0 = project_box_sum(h0, n2 / 2)
    
    # === Phase III: Smooth max gradient descent ===
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    beta = 2.0
    eta = 1.0
    max_iter = 0
    
    while time.time() - start_time < 0.85 * budget_s:
        max_iter += 1
        # Compute C values
        gamma = aperiodic_autocorr(h)
        C = 1 - (2 / n2) * gamma
        # Softmax for weights
        w = softmax(beta, C)
        # Compute gradient using FFT
        conv1 = convolve_fft(w, h)
        # Reverse the weight for the second convolution
        w_rev = w[::-1]
        conv2 = convolve_fft(w_rev, h)
        grad = -(2 / n2) * (conv1 + conv2)
        
        # Backtracking line search
        alpha = eta
        while True:
            h_trial = project_box_sum(h - alpha * grad, n2 / 2)
            c5_trial = compute_c5(h_trial)
            if c5_trial <= best_c5 or alpha < 1e-8:
                break
            alpha *= 0.5
        
        h = h_trial
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Anneal beta
        if max_iter % 150 == 0:
            beta = min(beta * 1.5, 200.0)
        
        # Occasional score-guided swaps
        if max_iter % 200 == 0:
            h = score_guided_swap(h)
    
    # === Phase IV: Final sub-gradient polishing ===
    while time.time() - start_time < budget_s - 0.1 * budget_s:
        # Identify worst shifts
        gamma = aperiodic_autocorr(best_h)
        C_s = 1 - (2 / n2) * gamma
        worst_thresh = best_c5 + 1e-5
        worst_indices = np.where(C_s >= worst_thresh)[0]
        if len(worst_indices) == 0:
            break
        # Sub-gradient
        g = np.zeros(n2)
        for s in worst_indices:
            # shift right
            if s > 0:
                g += np.pad(best_h[s:], (0, s), 'constant')
            else:
                g += best_h
            # shift left
            if s > 0:
                g += np.pad(best_h[:-s], (s, 0), 'constant')
            else:
                g += best_h
        g = -(2 / n2) * g / len(worst_indices)
        # Step size
        step = max_iter
        eta = 0.01 / np.sqrt(step + 1)
        h_new = project_box_sum(best_h - eta * g, n2 / 2)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        best_h = h_new
        max_iter += 1
    
    return (best_h, best_c5, n2)
