import numpy as np
import time
from scipy.signal import correlate
import math

def project_box_sum(y):
    """Project vector y onto box [0,1] with sum = n/2 using water-filling."""
    n = len(y)
    target_sum = n / 2
    
    def S(lam):
        return np.sum(np.clip(y - lam, 0, 1))
    
    # Binary search for lambda
    low, high = np.min(y) - 1.0, np.max(y) + 1.0
    for _ in range(40):
        mid = (low + high) / 2
        s = S(mid)
        if s > target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    
    x = np.clip(y - lam, 0, 1)
    return x

def evaluate_c5(h):
    """Computes C5 constant for given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorr(h):
    """Compute full autocorrelation of h."""
    return correlate(h, h, mode='full')

def compute_worst_shifts(C, epsilon=1e-5):
    """Return indices of worst shifts."""
    C5 = np.max(C)
    return np.where(np.abs(C - C5) < epsilon)[0]

def update_autocorr_delta(γ, h, i, j, Δ):
    """Update autocorrelation by moving Δ from h[i] to h[j]."""
    n = len(h)
    Δγ = np.zeros_like(γ)
    # Temporarily zero out h[i]
    ĥ = h.copy()
    ĥ[i] = 0
    
    for s in range(-n+1, n):
        idx = s + n - 1
        # Removal of Δ from i
        if 0 <= i + s < n:
            Δγ[idx] -= Δ * ĥ[i + s]
        if 0 <= i - s < n and s != 0:
            Δγ[idx] -= Δ * ĥ[i - s]
        # Addition of Δ to j
        if 0 <= j + s < n:
            Δγ[idx] += Δ * ĥ[j + s]
        if 0 <= j - s < n and s != 0:
            Δγ[idx] += Δ * ĥ[j - s]
    
    # Self-term correction
    Δγ[n-1] += -2 * h[i] * Δ + Δ * Δ + 2 * h[j] * Δ + Δ * Δ
    return Δγ

def compute_scores(h, S_w):
    """Compute score for each position based on worst shifts."""
    n = len(h)
    score = np.zeros(n)
    for s in S_w:
        # s denotes the shift index in full correlation array (stored at s+n-1)
        shift = s - (n - 1)
        for i in range(n):
            if 0 <= i + shift < n:
                score[i] += h[i + shift]
            if 0 <= i - shift < n and shift != 0:
                score[i] += h[i - shift]
    return score

def generate_move(h, S_w, eta=1e-8):
    """Generate a move: (src, tgt, Δ)"""
    n = len(h)
    
    scores = compute_scores(h, S_w)
    
    # Source: top 10% of high scoring indices with h[i] > eta
    idxs = np.nonzero(h > eta)[0]
    if len(idxs) == 0:
        return None
    
    # Sort top 10%
    sorted_indices = idxs[np.argsort(scores[idxs])[::-1]]
    src_set = sorted_indices[:max(1, len(sorted_indices)//10)]
    
    # Target: bottom 10% of low scoring indices with h[j] < 1 - eta
    idxs = np.nonzero(h < 1 - eta)[0]
    if len(idxs) == 0:
        return None
    
    sorted_indices = idxs[np.argsort(scores[idxs])]
    tgt_set = sorted_indices[:max(1, len(sorted_indices)//10)]
    
    if len(src_set) == 0 or len(tgt_set) == 0:
        return None
    
    i = np.random.choice(src_set)
    j = np.random.choice(tgt_set)
    
    # Δ
    Δ_max = 0.2 
    Δ = np.random.uniform(0, Δ_max)
    Δ = min(Δ, h[i], 1 - h[j])
    
    return (i, j, Δ)

def acceptance_probability(C5_new, C5_old, T):
    """Return acceptance probability for simulated annealing."""
    if C5_new < C5_old:
        return 1.0
    else:
        return math.exp((C5_old - C5_new) / T)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 300
    dx = 2.0 / n_points
    
    # Initialize using pre-computed construction
    if 'initial_h_values' in globals() and len(initial_h_values) < n_points:
        x_orig = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, initial_h_values)
    else:
        h = initial_h_values[:n_points] if 'initial_h_values' in globals() else np.random.rand(n_points)
    
    h = project_box_sum(h)
    
    # Run random seeds first
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    num_seeds = 5
    for _ in range(num_seeds):
        h_seed = np.random.rand(n_points)
        h_seed = project_box_sum(h_seed)
        c5_seed = evaluate_c5(h_seed)
        if c5_seed < best_c5:
            best_h = h_seed.copy()
            best_c5 = c5_seed
    
    h = best_h.copy()
    gamma = compute_autocorr(h)
    C = 1 - (2 / n_points) * gamma
    S_w = compute_worst_shifts(C)
    
    # Parameters
    T0 = 0.3 * (np.max(C) - np.min(C))
    α = 0.9995  # decay factor
    max_iter = int(100000 / 0.9995)
    accepted_count = 0
    last_accepted = 0
    accept_ratio = 0.4  # target acceptance rate

    # Phase B - Coarse SA
    Δ_max = 0.2
    T = T0

    p_iter = 0
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        p_iter += 1
        if p_iter % 10000 == 0:
            T = T0 * (α ** it)
            
        move = generate_move(h, S_w)
        if move is None:
            continue
            
        i, j, Δ = move
        
        gamma_new = gamma + update_autocorr_delta(gamma, h, i, j, Δ)
        C_new = 1 - (2 / n_points) * gamma_new
        C5_new = np.max(C_new)
        
        if np.random.rand() < acceptance_probability(C5_new, np.max(C), T):
            h[i] -= Δ
            h[j] += Δ
            h = project_box_sum(h)
            C = C_new
            gamma = gamma_new
            
            accepted_count += 1
            last_accepted = it
            
            S_w = compute_worst_shifts(C)
            
            if C5_new < best_c5:
                best_c5 = C5_new
                best_h = h.copy()
        
        # Reset Δ_max after 5000 would-be acceptances
        if accepted_count - last_accepted > 5000:
            Δ_max /= 2.0
            
        # Check convergence
        if accepted_count > 1000 and (accepted_count - last_accepted) > 20000:
            break
            
    # Phase C
    Δ_max = 0.05
    for it in range(200000):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        move = generate_move(h, S_w)
        if move is None:
            continue
            
        i, j, Δ = move
        γ_upd = update_autocorr_delta(gamma, h, i, j, Δ)
        C_new = 1 - (2 / n_points) * (gamma + γ_upd)
        C5_new = np.max(C_new)
        
        if np.random.rand() < acceptance_probability(C5_new, np.max(C), T):
            h[i] -= Δ
            h[j] += Δ
            h = project_box_sum(h)
            gamma = gamma + γ_upd
            C = C_new
            S_w = compute_worst_shifts(C)
            accepted_count += 1
            last_accepted = it
            if C5_new < best_c5:
                best_c5 = C5_new
                best_h = h.copy()
        
        # Periodic recomputation of gamma
        if accepted_count % 5000 == 0:
            gamma = compute_autocorr(h)
            C = 1 - (2 / n_points) * gamma
            S_w = compute_worst_shifts(C)

    # Phase E - Subgradient polishing
    def subgradient_step(h, S_w, stepsize=0.001):
        """Compute one subgradient descent step."""
        n = len(h)
        g = np.zeros(n)
        
        for s in S_w:
            shift = s - (n - 1)
            # Add contributions from positive and negative shifts
            if shift < 0:
                g[:-shift] -= h[-shift:]  # negative shift
                g[-shift:] -= h[:-shift]  # positive shift
            elif shift > 0:
                g[shift:] -= h[:-shift]  # negative shift
                g[:-shift] -= h[shift:]  # positive shift
            else:
                g -= 2 * h  # self term (s=0)
                
        # Normalize
        g = -g / n
        h_new = h - stepsize * g
        return project_box_sum(h_new)
    
    # Iteratively perform subgradient steps
    for _ in range(300):
        if time.time() - start_time > budget_s - 0.1:
            break
        C = 1 - (2 / n_points) * gamma
        S_w = compute_worst_shifts(C)
        h_new = subgradient_step(h, S_w)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        gamma = compute_autocorr(h)

    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
