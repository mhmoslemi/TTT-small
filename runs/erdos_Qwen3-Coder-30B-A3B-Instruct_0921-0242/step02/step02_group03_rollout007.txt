import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def full_to_sym(h_full):
    """Convert full vector to symmetric (half + center) representation."""
    n = len(h_full)
    half_n = n // 2
    if n % 2 == 0:
        return (h_full[:half_n] + h_full[:half_n][::-1]) / 2
    else:
        return np.concatenate([h_full[:half_n], [h_full[half_n]], h_full[:half_n][::-1]])

def sym_to_full(h_sym):
    """Convert symmetric representation to full vector."""
    n = len(h_sym) * 2 - 1
    h_full = np.zeros(n)
    h_full[:len(h_sym)] = h_sym
    h_full[len(h_sym):] = h_sym[-2::-1]
    return h_full

def correlation_vals(h):
    """Compute correlation values from a step function h."""
    n = len(h)
    corr = correlate(h, 1-h, mode='full')
    # index 0 corresponds to shift -(n-1)  
    return corr[n-1:] * (2.0/n)   # dx = 2/n

def build_constraints(x, t, shift_set, n_sym):
    """Build linear constraints for the LP from a set of shifts."""
    constraints = []
    
    # Integral constraint: sum(h) = 1
    constraints.append(cp.sum(x) == 0.5)  # because h is half-size and sum should be 1
    
    # Bounds
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # Additional constraints from each shift
    for s in shift_set:
        # This computes sum_i x_i * (1 - x_{i+s}) for valid i
        expr = 0
        for i in range(n_sym - s):
            expr += x[i] * (1 - x[i + s])
        constraints.append(expr <= t)
        
    return constraints

def solve_lp(shift_set, x0=None, n_sym=201):
    """Solve the LP with specified shifts."""
    # Variables: x (symmetric), t (scalar)
    x = cp.Variable(n_sym)
    t = cp.Variable()
    
    # Constraints
    constraints = build_constraints(x, t, shift_set, n_sym)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = x.value.copy()
            t_sol = t.value
        else:
            return None, None
    except:
        return None, None
    
    return x_sol, t_sol

    S = [0, 1, 2, n // 4, n // 2]
    
    h_best = None
    c5_best = float('inf')
    t_best = float('inf')
    
    # Cached correlation computation and maximum shift tracking
    def get_max_shift_and_corr(h_full):
        c = correlation_vals(h_full)
        max_shift = np.argmax(c)
        max_val = c[max_shift]
        return max_shift, max_val, c

    # Column generation / cutting plane loop
    time_limit = start_time + budget_s - 2  # reserved for final polishing

        x_sol, t_sol = solve_lp(S, h_sym, n_sym=len(h_sym))

        h_full = sym_to_full(x_sol)
