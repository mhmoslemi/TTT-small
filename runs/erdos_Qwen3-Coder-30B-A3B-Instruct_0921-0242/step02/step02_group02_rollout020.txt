import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def compute_autocorrelation(h, k):
    """Compute the k-th autocorrelation of h."""
    n = len(h)
    dx = 2.0 / n
    # Compute h_i * h_{i-k}
    corr = 0.0
    for i in range(n):
        j = (i - k) % n
        corr += h[i] * h[j]
    return corr

def sdp_relaxation(n):
    """Solve SDP relaxation to get a lower bound and target autocorrelation."""
    dx = 2.0 / n
    # Generate the circulant matrices C_k for k = 0, ..., n-1
    C_list = []
    for k in range(n):
        C = np.zeros((n, n))
        for i in range(n):
            C[i, (i - k) % n] = 1.0
        C_list.append(C)
    
    # Variables
    h = cp.Variable(n)
    gamma = cp.Variable()
    
    # Constraints  
    constraints = [cp.sum(h) == n/2, h >= 0, h <= 1]
    for k in range(n):
        constraints.append(cp.trace(C_list[k] @ cp.outer(h, h)) >= n/2 - gamma/dx)
    
    # Objective
    objective = cp.Minimize(gamma)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
    except:
        return None, None

    if prob.status != cp.OPTIMAL:
        return None, None
        
    # Extract solution
    h_sol = h.value
    gamma_sol = gamma.value
    
    # Create target autocorrelation
    if h_sol is not None:
        # Compute the target autocorrelation for each shift
        target = []
        for k in range(n):
            corr = 0.0
            for i in range(n):
                j = (i - k) % n
                corr += h_sol[i] * h_sol[j]
            target.append(corr)
        target = np.array(target)
        target = target / (n/2)  # Normalize by half the number of elements
        return h_sol, target
    else:
        return None, None

def solve_binary_mip(n, target_correlation, max_time=15):
    """Solve binary MIP to approximately match target autocorrelation."""
    # Binary vector b
    b = cp.Variable(n, boolean=True)
    gamma = cp.Variable()
    
    # Constraints: sum(b) = n/2
    constraints = [cp.sum(b) == n/2]
    
    # Constraint: c_k <= gamma for all k, linearized
    # s_k = sum_i b_i b_{i-k}
    for k in range(n):
        s_k = cp.Variable()
        # Add McCormick envelopes for z_{i,k} = b_i * b_{i-k}
        z_vars = cp.Variable(n)
        constraints.extend([
            z_vars >= b - (1 - b[(k+1)%n:n] + b[:k]),
            z_vars >= 0,
            z_vars <= b,
            z_vars <= b[(k+1)%n:n] + b[:k] - 1,
            s_k == cp.sum(z_vars)
        ])
        # c_k = dx * (n/2 - s_k) <= gamma
        constraints.append(dx*(n/2 - s_k) <= gamma)
    
    # Warm start with a heuristic (use initial h if given)
    # Start simple heuristic for now
    initial_h = np.random.rand(n)
    initial_h = initial_h * (n / 2) / np.sum(initial_h)
    initial_b = (initial_h > 0.5).astype(float)
    b.value = initial_b
    
    # Objective
    objective = cp.Minimize(gamma)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, time_limit=max_time, verbose=False)
    except:
        # fallback to heuristics
        pass

    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        b_sol = b.value
        if b_sol is not None:
            return b_sol, evaluate_c5(b_sol)
    else:
        # Heuristic: simple thresholding of target autocorrelation
        b_sol = np.zeros(n)
        h_target = np.ones(n)
        h_target = h_target * (n / 2) / np.sum(h_target)
        # Use simple rounding
        b_sol = (h_target > 0.5).astype(float)
        return b_sol, evaluate_c5(b_sol)
        
    return None, None

def ramp_smoothing(b, width=3):
    """Convert binary to continuous with smooth transitions."""
    n = len(b)
    h = np.zeros(n)
    for i in range(n):
        h[i] = b[i]
        # Smooth transitions
        for k in range(1, width+1):
            j1 = (i + k) % n
            j2 = (i - k) % n
            if b[i] == 0 and b[j1] == 1:
                h[i] += 0.5 * (k / (width + 1))
            if b[i] == 1 and b[j2] == 0:
                h[i] += 0.5 * (k / (width + 1))
    return h

def lbfgsb_refinement(h0, max_iter=2000, beta=5.0):
    """Refine h using smooth surrogate with L-BFGS-B."""
    n = len(h0)
    dx = 2.0 / n
    # Initial structure
    h = h0.copy()
    def softmax(x, beta):
        """Softmax with binned version."""
        x = np.array(x)
        exps = np.exp(beta * x)
        return exps / np.sum(exps)
    
    def lse_objective(h):
        """Objective with log-sum-exp approximation."""
        corr = correlate(h, 1-h, mode="full")
        corr = corr[:n] * dx
        return np.log(np.sum(np.exp(beta * corr))) / beta
    
    def grad_lse(h):
        """Gradient of log-sum-exp objective."""
        corr = correlate(h, 1-h, mode="full")[:n] * dx
        w = softmax(corr, beta)
        # Compute gradient from each correlation
        grad = np.zeros_like(h)
        for k in range(n):
            if k == 0:
                shifted = h
            else:
                shifted = np.roll(h, -k)
            grad += w[k] * (1 - 2*h) * shifted
        return grad * dx
    
    # Other auxiliary functions
    def project_h(x):
        return project_to_feasible(x, n)
    
    # BFGS with projection
    for i in range(max_iter):
        if i % 200 == 0 and beta < 80:
            beta *= 2
        grad = grad_lse(h)
        h -= 0.01 * grad
        h = project_h(h)
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use slightly larger n for better resolution
    n_points = 200
    dx = 2.0 / n_points

    # Phase 1: Solve SDP relaxation to get target autocorrelation
    h_target, target_corr = sdp_relaxation(n_points)
    
    # fallback to simple target
    if h_target is None:
        target_corr = np.ones(n_points) * 0.5  # Uniform target
        h_target = np.ones(n_points) * 0.5
    else:
        # Ensure h_target reflects the target_corr
        h_target = project_to_feasible(h_target, n_points)

    # Phase 2: Binary MIP construction
    binary_pattern, c5_binary = solve_binary_mip(n_points, target_corr, 15)

    if binary_pattern is None:
        binary_pattern = (h_target > 0.5).astype(float)
    
    # Phase 3: Refine binary to continuous
    h_continuous = ramp_smoothing(binary_pattern, width=3)
    h_continuous = project_to_feasible(h_continuous, n_points)
    
    # Phase 4: Continuous refinement with L-BFGS-B
    h_refined = lbfgsb_refinement(h_continuous)

    # Final polishing
    while time.time() - start_time < budget_s - 5:
        corr_full = correlate(h_refined, 1-h_refined, mode="full")
        k_max = np.argmax(corr_full[:n_points])  # Shift causing the max overlap
        k_max = k_max if k_max < n_points else k_max - n_points
        
        # Gradient of c_k_max
        grad_corr = np.zeros(n_points)
        shifted = np.roll(h_refined, k_max)
        grad_corr = (1 - 2 * h_refined) * shifted
        
        # Update with gradient step
        h_new = h_refined - 0.001 * grad_corr
        h_refined = project_to_feasible(h_new, n_points)
        c5_current = evaluate_c5(h_refined)
        if c5_current >= 0.3807:  # Threshold for stopping early if beyond goal
            break

    # Final evaluation
    final_c5 = evaluate_c5(h_refined)
    return (h_refined, final_c5, n_points)
