def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Computes the exact C5 constant using numpy correlate."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def smooth_max(h, p=256):
    """Smooth approximation to max using p-norm."""
    dx = 2.0 / len(h)
    # Compute overlap by auto-correlation
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    corr_vals = corr.real[:len(h)] * dx
    
    # Compute smooth max
    C_vec = 1.0 - corr_vals
    S_p = np.sum(C_vec ** p) ** (1.0 / p)
    return S_p, C_vec

def run(seed=42, budget_s=600, **kwargs):

    for p_idx, p in enumerate(p_vals):

        for i, h_seed in enumerate(seeds):
            # Make this to be testable
            h_current = h_seed.copy()
            
            # Ensure sum constraint and clipping
            h_current = project_to_feasible(h_current, n_points)

            opt_start = time.time()
            def objective(h_vec):
                S_p, C_vec = smooth_max(h_vec, p)

                return -S_p

                return -compute_gradient(h_vec, p)

            def sum_penalty(h_vec):
                return 1e4 * (np.sum(h_vec) - n_points / 2) ** 2
            
            # simplify to always apply sum penalty
            penalty = lambda h_vec: sum_penalty(h_vec)
            def objective_with_penalty(h_vec):
                return -smooth_max(h_vec, p)[0] + penalty(h_vec)
            
            res = minimize(objective_with_penalty, h_current, method='L-BFGS-B',
