import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.stats import dirichlet

def project_to_feasible(h, target_sum):
    """Ensure that sum(h) = target_sum, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def generate_seeds(n_points, target_sum):
    """Generate multiple seeds for the optimization."""
    seeds = []
    
    # 1. Uniform seed
    seeds.append(np.ones(n_points) * 0.5)
    
    # 2. Random Dirichlet seed
    dirichlet_sample = dirichlet.rvs(np.ones(n_points), size=1).flatten()
    dirichlet_sample = dirichlet_sample * target_sum / np.sum(dirichlet_sample)
    seeds.append(dirichlet_sample)
    
    # 3. Binary random seed
    binary_h = np.zeros(n_points)
    binary_h[:n_points//2] = 1
    np.random.shuffle(binary_h)
    # Add small noise and re-project
    binary_h += np.random.uniform(0, 0.05, n_points)
    binary_h = project_to_feasible(binary_h, target_sum)
    seeds.append(binary_h)
    
    # 4. Interpolated initial_h_values seed
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        x_orig = np.linspace(0, 2, len(orig_h))
        x_new = np.linspace(0, 2, n_points)
        interpolated_h = np.interp(x_new, x_orig, orig_h)
        interpolated_h = project_to_feasible(interpolated_h, target_sum)
        seeds.append(interpolated_h)
    
    # 5. Sobol-like low-discrepancy seed
    # Using a pseudo-random sample but injecting some structure
    sobol_like = np.sin(np.arange(n_points) * 1.618033988749895)
    sobol_like = (sobol_like + 1) / 2  # Scale to [0,1]
    sobol_like = sobol_like * target_sum / np.sum(sobol_like)
    sobol_like = project_to_feasible(sobol_like, target_sum)
    seeds.append(sobol_like)
    
    return seeds

def build_lp_data(h0):
    """Build the linear program data for given h0."""
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2
    
    # Compute correlation C_k^0 = sum_i h_i^0 * h_{i+k}^0 for all k
    c0 = ifft(fft(h0, 2 * n) * np.conj(fft(h0, 2 * n))).real
    
    # Build matrix W where row k is (h0_{i+k} + h0_{i-k})
    W = np.zeros((n, n))
    for k in range(n):
        left_indices = (np.arange(n) + k) % n
        right_indices = (np.arange(n) - k) % n
        W[k, :] = h0[left_indices] + h0[right_indices]
    
    A = -W
    b = -target_sum + c0  # From equation (2)
    
    return A, b, c0

def run_ccp_loop(h0, target_sum, max_iter=30, time_left_func=lambda: 1e9):
    """Run one CCP iteration."""
    n = len(h0)
    dx = 2.0 / n
    
    prev_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    best_h = h0.copy()
    
    for _ in range(max_iter):
        if time_left_func() < 1e-3:
            break
        A, b, c0 = build_lp_data(prev_h)
        
        # Set up the linear program
        c = np.append(np.zeros(n), 1.0)  # optimize t, not h
        A_ub = np.hstack([A, np.ones((A.shape[0], 1))])
        b_ub = b
        
        # Add bounds: 0 <= h <= 1 and sum h = target_sum, t >= 0
        bounds = [(0, 1) for _ in range(n)] + [(0, None)]
        
        # Solve the LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',  
                      options={'presolve': True})
        
        if not res.success:
            break
        
        # Extract the solution
        h_new = res.x[:-1]
        t_new = res.x[-1]  # This is the approximate value of the max
        h_new = project_to_feasible(h_new, target_sum)
        
        # Check true C5
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Check convergence
        if np.max(np.abs(h_new - prev_h)) < 1e-8:
            break
            
        prev_h = h_new.copy()
    
    return best_h, best_c5

def gradient_descent_step(h, worst_shift, dx, max_iter=200):
    """Perform a few gradient descent steps on the true objective."""
    n = len(h)
    h = h.copy()
    for t in range(max_iter):
        # Compute worst shift
        corr_vals = []
        for k in range(1, n):
            corr = np.sum(h * np.roll(1 - h, k))
            corr_vals.append(corr)
        worst_shift = np.argmin(corr_vals) + 1
        
        # Compute gradient
        grad = np.zeros(n)
        for i in range(n):
            left_idx = (i + worst_shift) % n
            right_idx = (i - worst_shift) % n
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Step size
        alpha = 1.0 / np.sqrt(1 + t)
        # Update
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n / 2)
        h = h_new
    return h

def simulated_annealing(h, dx, max_iter=5000, time_left_func=lambda: 1e9):
    """Perform simulated annealing pair exchange."""
    n = len(h)
    current_h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    T_max = 0.1
    T_min = 1e-6
    T = T_max
    alpha = 0.95
    
    for _ in range(max_iter):
        if time_left_func() < 1e-3:
            break
            
        # Find worst shift
        corr_vals = []
        for k in range(1, n):
            corr = np.sum(current_h * np.roll(1 - current_h, k))
            corr_vals.append(corr)
        worst_shift = np.argmin(corr_vals) + 1
        
        # Make move
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if current_h[i] <= 0 or current_h[j] >= 1:
            continue
        
        delta = np.random.uniform(0, min(current_h[i], 1 - current_h[j], 0.01))
        h_test = current_h.copy()
        h_test[i] -= delta
        h_test[j] += delta
        h_test = project_to_feasible(h_test, n / 2)
        c5_test = evaluate_c5(h_test)
        
        # Accept with probability
        delta_E = c5_test - best_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):  
            current_h = h_test
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
        
        # Update temperature
        T = alpha * T if T > T_min else T_min
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 400  # Should be feasible within the time constraints
    target_sum = n_points / 2
    
    # Generate multiple starting seeds
    seeds = generate_seeds(n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    # Multi-start approach
    for seed_idx, h0 in enumerate(seeds):
        if time.time() - start_time > budget_s * 0.7:
            break
            
        time_left = lambda: budget_s * 0.7 - (time.time() - start_time)
        
        # Run CCP loop
        h_final, c5_final = run_ccp_loop(h0, target_sum, max_iter=30, time_left_func=time_left)
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
        
    if best_h is None:
        # Fallback to uniform
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_c5(best_h)
    
    # Attempt polishing - gradient descent
    if time.time() - start_time < budget_s * 0.95:
        dx = 2.0 / n_points
        time_left = lambda: budget_s * 0.95 - (time.time() - start_time)
        h_polished = gradient_descent_step(best_h, 1, dx, max_iter=100)
        c5_polished = evaluate_c5(h_polished)
        if c5_polished < best_c5:
            best_c5 = c5_polished
            best_h = h_polished
    
    # Final simulated annealing polishing
    if time.time() - start_time < budget_s * 0.99:
        time_left = lambda: budget_s * 0.99 - (time.time() - start_time)
        h_final, c5_final = simulated_annealing(best_h, 2.0/n_points, max_iter=1000,  
                                               time_left_func=time_left)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Optional upsampling refinement if time permits
    if time.time() - start_time < budget_s * 0.995:
        # Resample to 800 points
        new_n = 800
        dx_new = 2.0 / new_n
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, new_n)
        h_upsampled = np.interp(x_new, x_orig, best_h)
        h_upsampled = project_to_feasible(h_upsampled, new_n / 2)
        
        # Run one short CCP pass on upsampled h
        h_fine, c5_fine = run_ccp_loop(h_upsampled, new_n/2, max_iter=5,
                                      time_left_func=lambda: 1e9)
        
        # Final gradient step on fine grid
        dx = 2.0 / new_n
        h_final = gradient_descent_step(h_fine, 1, dx, max_iter=50)
        c5_final = evaluate_c5(h_final)
        
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            n_points = new_n
    
    # Ensure feasible final result
    best_h = project_to_feasible(best_h, n_points / 2)
    best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, n_points)
