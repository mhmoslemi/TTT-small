import numpy as np
import time
import math

def project_box_sum(y, total):
    """Project y onto the set {x : 0 <= x[i] <= 1, sum x = total}"""
    # Use bisection to find lambda
    a, b = -1.0, 2.0  # bounds for lambda
    for _ in range(50):  # sufficient for double precision
        mid = (a + b) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > total:
            a = mid
        else:
            b = mid
    lambda_ = (a + b) / 2.0
    return np.clip(y - lambda_, 0.0, 1.0)

def softmax_weights(beta_C):
    """Compute softmax weights from beta*C, with numerical stability."""
    max_C = np.max(beta_C)
    exp_C = np.exp(beta_C - max_C)
    return exp_C / np.sum(exp_C)

def gradient_from_weights(w, h):
    """Compute gradient from weight vector w and h."""
    n = len(h)
    w_full = np.zeros(2 * n - 1)
    w_full[n-1:n-1+len(w)] = w
    w_rev = w_full[::-1]
    
    # Two convolutions: w * h and w_rev * h
    conv1 = np.convolve(w_full, h, mode='full')
    conv2 = np.convolve(w_rev, h, mode='full')
    
    # Extract central part (n elements)
    grad = -(2.0 / n) * (conv1[n-1:n-1+n] + conv2[n-1:n-1+n])
    return grad

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
