import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import bisect

def project_box_sum(y, total):
    """Project y onto {x : 0 <= x_i <= 1, sum x_i = total}."""
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if abs(s - total) < 1e-10:
        return y
    
    # Binary search for lambda
    def sum_at_lambda(lam):
        return np.sum(np.clip(y - lam, 0, 1))
        
    def f(lam):
        return sum_at_lambda(lam) - total
    
    a, b = np.min(y), np.max(y) - 1e-10
    if f(a) > 0:
        a, b = b, a
    try:
        lam = bisect(f, a, b, xtol=1e-12)
    except:
        lam = (a + b) / 2.0
    return np.clip(y - lam, 0, 1)

def compute_autocorr(h):
    """Compute aperiodic autocorrelation of h."""
    n = len(h)
    # Use FFT method for speed
    H = fft(h)
    P = np.abs(H)**2
    gamma = ifft(P).real
    return gamma

def max_c5(h):
    """Evaluate C5 for discrete h."""
    n = len(h)
    dx = 2.0 / n
    # This computes the same integral as in the task
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def swap_autocorr_update(gamma, h, i, j):
    """Compute the change in autocorrelation when swapping h[i] and h[j]."""
    # h[i] is set to 0 (was 1), h[j] is set to 1 (was 0)
    # Update gamma via linear increments
    # γ modifies as:
    # shifts where both i,j are involved are adjusted
    n = len(h)
    delta_gamma = np.zeros_like(gamma)
    
    # For each k, we compute change in γ_k
    # γ_k = sum_{i} h[i] * h[i+k]  (with wraparound)
    # Removing h[i], adding h[j]
    
    # Change for k = 0 (lag 0)
    # only h[i] and h[j] appear here
    delta_gamma[0] = h[j] * h[j] - h[i] * h[i]
    
    # For k = 1 to n-1 (and k = -(n-1) to -1): 
    for k in range(1, n):
        # j+k is wrapped, j-k is wrapped
        jk = (j + k) % n
        jmk = (j - k) % n
        ik = (i + k) % n
        imk = (i - k) % n
        
        # Changed:
        # h[i] is being removed (now 0)
        # h[j] is being added (now 1)
        delta_gamma[k] = - h[i] * h[ik] + h[j] * h[jk]
        delta_gamma[-k] = - h[i] * h[imk] + h[j] * h[jmk]
    
    return delta_gamma

def binary_hill_climb(seed_h, max_iter=None):
    """Improve a binary seed by swapping 1s and 0s."""
    if max_iter is None:
        max_iter = 3 * len(seed_h)
    
    h = seed_h.copy().astype(float)
    gamma = compute_autocorr(h)
    c5 = max_c5(h)
    
    improved = 0
    for _ in range(max_iter):
        # For pick the indices to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        try:
            delta_gamma = swap_autocorr_update(gamma, h, i, j)
            new_gamma = gamma + delta_gamma
            # max is at index n-1
            c5_new = 1 - 2 * new_gamma[n//2] / len(h)
            if c5_new < c5:
                h[i] = 0.0
                h[j] = 1.0
                gamma = new_gamma
                c5 = c5_new
                improved += 1
        except Exception:
            continue
            
    # Make sure we are still feasible
    h = project_box_sum(h, len(h)/2)
    return h, max_c5(h)

def legendre_sequence(n):
    """Generate a Legendre sequence of length n, that sums to about n/2."""
    # We want n prime so we can generate a full Legendre sequence
    # For small n, build from the first prime larger than n
    p = n + 1
    while True:
        if is_prime(p):
            break
        p += 1
    
    assert p <= 257  # Safe assumption for our context
    legendre = np.array([jacobi_symbol(k, p) for k in range(1, p)])
    
    # Convert +1, -1 to 1, 0
    legendre = (legendre + 1) // 2
    
    # Truncate to exactly n
    return legendre[:n]

def is_prime(n):
    """Check if a number is prime."""
    if n < 2:
        return False
    for i in range(2, int(np.sqrt(n)) + 1):
        if n % i == 0:
            return False
    return True

def jacobi_symbol(a, n):
    """Compute the Jacobi symbol of a/n."""
    if n <= 0 or n % 2 == 0:
        return 0
    a = a % n
    result = 1
    while a != 0:
        while a % 2 == 0:
            a //= 2
            if n % 8 in (3, 5):
                result = -result
        a, n = n, a
        if a % 4 == 3 and n % 4 == 3:
            result = -result
        a = a % n
    return result if n == 1 else 0

def m_sequence(length):
    """Generate a maximal-length m-sequence of length length."""
    # For Modulo 2^8 - 1, it's 255 
    # This is an implementation of the LFSR generator
    # Primitive polynomial: x^8 + x^4 + x^3 + x^2 + 1
    # Initial values (we'll use 100000000)
    state = 1
    sequence = []
    for _ in range(length):
        # Output bit
        output = state & 1
        sequence.append(output)
        # Feedback bit: bit 7 ^ bit 3 ^ bit 2 ^ bit 1
        feedback = ((state >> 7) ^ (state >> 3) ^ (state >> 2) ^ (state >> 1)) & 1
        # Shift left and insert feedback bit
        state = ((state << 1) | feedback) & 0xFF
    return np.array(sequence)

def random_balanced(n):
    """Random balanced binary string of length n."""
    h = np.zeros(n)
    ones_indices = np.random.choice(n, n//2, replace=False)
    h[ones_indices] = 1
    return h

def spectral_alternating_projection(h_start, max_iter=400, alpha_start=0.7):
    """SAP procedure to flatten the autocorrelation by reshaping in frequency domain."""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    
    alpha = alpha_start
    best_h = h.copy()
    best_c5 = max_c5(h)
    
    for iter in range(max_iter):
        # Forward FFT to get power spectrum
        H = fft(h)
        P = np.abs(H)**2
        gamma = ifft(P).real
        nonzero_avg = np.sum(gamma[1:]) / (n-1)  # Average of non-zero lags
        
        # Define target autocorrelation (soft combination)
        gamma_target = (1-alpha) * gamma + alpha * nonzero_avg
        gamma_target[0] = gamma[0]
        
        # Convert to magnitude in frequency domain
        P_target = fft(gamma_target)
        P_target = np.maximum(P_target, 0)
        A_target = np.sqrt(P_target)
        
        # Reconstruct and reconstruct with original phase
        H_new = A_target * np.exp(1j * np.angle(H))
        h_new = ifft(H_new).real
        
        # Project to box and sum constraint
        h_new = project_box_sum(h_new, n/2)
        
        # Check improvement
        c5 = max_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            
        h = h_new.copy()
        
        # Gradually increase alpha (smoothing)
        if iter > 0 and iter % 20 == 0:
            alpha = min(alpha + 0.01, 0.99)
        
    return best_h, best_c5
