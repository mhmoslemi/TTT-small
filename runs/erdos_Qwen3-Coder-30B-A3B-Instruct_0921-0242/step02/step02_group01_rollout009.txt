import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import fftconvolve

def project_to_sum_constraint(h, target_sum):
    """Project h to have sum=target_sum, keeping values in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Add diff/n to each element, clip, repeat if needed
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5_fast(h):
    """Evaluate C5 using FFT for speed."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation with 1-h using convolution
    corr = fftconvolve(h, 1 - h, mode='full')
    # Extract positive shifts (excluding shift 0)
    corr_vals = corr[n-1:2*n-1]
    max_corr = np.max(corr_vals) * dx
    return max_corr

def gradient_worst_shift_descent(h, max_iter=200, alpha0=0.1):
    """Perform gradient descent with respect to the worst shift."""
    n = len(h)
    dx = 2.0 / n
    alpha = alpha0 * dx
    h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    for t in range(max_iter):
        # Compute all correlations
        corr = fftconvolve(h, 1 - h, mode='full')
        corr_vals = corr[n-1:2*n-1]
        # Find worst shift (largest value -> max correlation)
        worst_shift = np.argmax(corr_vals)
        worst_corr = corr_vals[worst_shift]
        
        # Compute gradient w.r.t. h
        grad = np.zeros_like(h)
        for i in range(n):
            left_idx = (i + worst_shift) % n
            right_idx = (i - worst_shift) % n
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_to_sum_constraint(h_new, n / 2)
        c5_new = evaluate_c5_fast(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
        else:
            # diminishing step size
            alpha *= 0.995
            if alpha < 1e-10:
                break
    
    return h, best_c5

def differential_evolution(hs, target_sum, generations=200, F=0.6, CR=0.9):
    """Perform DE with FFT-based evaluation."""
    n = len(hs[0])
    dx = 2.0 / n
    pop_size = len(hs)
    best_h = None
    best_c5 = float('inf')
    
    def evaluate_individual(h):
        return evaluate_c5_fast(h)
    
    for _ in range(generations):
        for i in range(pop_size):
            # Select three different vectors
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            v1, v2, v3 = hs[a], hs[b], hs[c]
            # Create donor vector
            trial = v1 + F * (v2 - v3)
            # Binomial crossover
            mask = np.random.random(n) < CR
            u = np.where(mask, trial, hs[i])
            # Project to feasible set
            u = project_to_sum_constraint(u, target_sum)
            
            # Evaluate
            c5_u = evaluate_individual(u)
            c5_i = evaluate_individual(hs[i])
            if c5_u < c5_i:
                hs[i] = u.copy()
                if c5_u < best_c5:
                    best_c5 = c5_u
                    best_h = hs[i].copy()
    
    return hs, best_h, best_c5

def stochastic_swap_polishing(h, max_iter=5000):
    """Stochastic polishing using pair swaps."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    current_c5 = evaluate_c5_fast(h)
    
    for _ in range(max_iter):
        # Sample i, j and small delta
        i, j = np.random.randint(0, n, 2)
        if h[i] <= 0 or h[j] >= 1:
            continue
        delta = np.random.uniform(0, min(h[i], 1 - h[j], 0.01))
        
        h_trial = h.copy()
        h_trial[i] -= delta
        h_trial[j] += delta
        h_trial = project_to_sum_constraint(h_trial, n / 2)
        trial_c5 = evaluate_c5_fast(h_trial)
        
        if trial_c5 < current_c5:
            h = h_trial
            current_c5 = trial_c5
        # else: accepted with some probability (not strictly required for this)
    
    return h, current_c5

def interpolate_to_higher_resolution(h, new_n):
    """Interpolate h to a higher resolution."""
    old_x = np.linspace(0, 2, len(h))
    new_x = np.linspace(0, 2, new_n)
    h_new = np.interp(new_x, old_x, h)
    return h_new

    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 400
    target_sum = n_points / 2
    
    # Use initial construction if available

        h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Initialize population
    pop_size = min(40, 2 * n_points // 10)
    hs = []
    hs.append(project_to_sum_constraint(h_init, target_sum))
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        h = project_to_sum_constraint(h, target_sum)
        hs.append(h)
    
    best_h = hs[0]
    best_c5 = evaluate_c5_fast(best_h)
    
    # Differential Evolution
    for gen in range(500):
        if time.time() - start_time > 0.75 * budget_s:
            break
        hs, gen_best_h, gen_best_c5 = differential_evolution(hs, target_sum, generations=1)
        if gen_best_c5 < best_c5:
            best_c5 = gen_best_c5
            best_h = gen_best_h.copy()
    
    # Gradient descent on best
    if time.time() - start_time < 0.9 * budget_s:
        best_h, best_c5 = gradient_worst_shift_descent(best_h, max_iter=200)
    
    # Polishing (swap-based)
    if time.time() - start_time < 0.99 * budget_s:
        best_h, best_c5 = stochastic_swap_polishing(best_h, max_iter=5000)
    
    # Optional: interpolate and do fine tuning
    if time.time() - start_time < 0.995 * budget_s:
        fine_h = interpolate_to_higher_resolution(best_h, 800)
        fine_h = project_to_sum_constraint(fine_h, 400)
        fine_h, fine_c5 = gradient_worst_shift_descent(fine_h, max_iter=100)
        if fine_c5 < best_c5:
            best_c5 = fine_c5
            best_h = fine_h
    
    # Final safety check
    if time.time() - start_time > 0.999 * budget_s:
        # Fallback if needed
        h_fallback = project_to_sum_constraint(h_init, target_sum)
        c5_fallback = evaluate_c5_fast(h_fallback)
        if c5_fallback < best_c5:
            best_c5 = c5_fallback
            best_h = h_fallback
    
    return (best_h, best_c5, n_points)
