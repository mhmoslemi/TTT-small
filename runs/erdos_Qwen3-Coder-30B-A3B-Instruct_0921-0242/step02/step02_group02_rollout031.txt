import numpy as np
import time
from scipy.optimize import differential_evolution, linprog
from scipy.fft import rfft, irfft
from scipy.interpolate import interp1d

def project_to_feasible(h, n):
    """Project h to [0,1] and ensure sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    # Add uniform offset to fix sum
    delta = (n/2 - current_sum) / n
    h = h + delta
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Compute C5 = max_k int h(x)(1 - h(x+k)) dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def basis_evaluation(x, M):
    """Generate cosine and sine basis functions up to order M."""
    basis = np.zeros((len(x), 2*M))
    for m in range(1, M+1):
        basis[:, 2*(m-1)] = np.cos(m * np.pi * x)
        basis[:, 2*(m-1)+1] = np.sin(m * np.pi * x)
    return basis

def construct_h_from_coeffs(c, n0, M):
    """Construct h(x) from Fourier coefficients c = [a1,b1,...,aM,bM]."""
    x = np.linspace(0, 2, n0, endpoint=False)
    basis = basis_evaluation(x, M)
    h_raw = 0.5 + np.dot(basis, c)  # a0 = 0.5
    h_raw = np.clip(h_raw, 0, 1)
    scale = 1.0 / (np.sum(h_raw) * 2 / n0)
    h = h_raw * scale
    return h

def fitness_function(c, n0, M):
    """Fitness evaluation for a given coefficient vector."""
    h = construct_h_from_coeffs(c, n0, M)
    return evaluate_c5(h)

def solve_linearized_lp(h_current, k_star, n0):
    """Solve linearized LP to improve h."""
    # Build the constraint matrix for active shifts
    # Only consider the worst 10 shifts, as they are likely to dominate
    active_indices = []
    corr = np.correlate(h_current, h_current, mode='full')
    k_indices = np.arange(len(corr))
    k_indices = np.concatenate([k_indices[len(corr)//2:], k_indices[:len(corr)//2]])
    k_indices = k_indices[np.argsort(-corr)]
    for i in range(min(10, len(corr))):
        idx = k_indices[i]
        if idx < len(corr) // 2:
            k = idx
        else:
            k = idx - len(corr) + 1
        if k == k_star:
            continue
        active_indices.append(k)
    if k_star not in active_indices:
        active_indices.append(k_star)
    active_indices = np.array(active_indices)

    # Reshape constraint matrix
    A_eq = np.zeros((n0 + 1, n0 + 1))  # last column is for t
    b_eq = np.zeros(n0 + 1)
    A_eq[:n0, :n0] = np.eye(n0)
    A_eq[:n0, n0] = -1  # t is in the last column
    b_eq[:n0] = 0
    A_eq[n0, :n0] = 1  # sum constraint
    b_eq[n0] = n0 / 2
    # Add constraints
    A_ub = []
    b_ub = []
    for k in active_indices:
        if k > 0:
            grad = np.roll(h_current, k)
        elif k < 0:
            grad = np.roll(h_current, -k)
        else:
            grad = h_current
        grad_row = np.append(-grad, 1)
        A_ub.append(grad_row)
        corr_val = np.dot(h_current, np.roll(h_current, k))
        b_ub.append(corr_val)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c_obj = np.zeros(n0 + 1)
    c_obj[n0] = -1  # maximize t
    # Solve LP
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
        if res.success:
            h_new = res.x[:n0]
            t_new = res.x[n0]
        else:
            h_new = h_current
            t_new = np.min([np.dot(h_current, np.roll(h_current, k)) for k in active_indices])
    except:
        h_new = h_current
        t_new = np.min([np.dot(h_current, np.roll(h_current, k)) for k in active_indices])
    return h_new, t_new

def subgradient_step(h, n):
    """Perform a simple subgradient sub-step."""
    dx = 2.0 / n
    # Find worst shift
    corr = np.correlate(h, 1 - h, mode='full')
    k_worst = np.argmax(corr)
    if k_worst > len(h):
        k = k_worst - len(h)
    else:
        k = k_worst - len(h) + 1
    # Compute gradient
    if k > 0:
        grad = -np.roll(h, k)
    elif k < 0:
        grad = np.roll(h, -k)
    else:
        grad = -h
    # Small step in negative gradient direction
    h_new = h + dx * grad * 0.1
    h_new = project_to_feasible(h_new, n)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phases allocation
    phase_time_budgets = {
        'basis': 0.02 * budget_s,
        'de': 0.50 * budget_s,
        'select': 0.03 * budget_s,
        'slp': 0.30 * budget_s,
        'upsample': 0.15 * budget_s
    }
    
    # 1. Basis preparation
    n0 = 250
    M = 12
    dx = 2.0 / n0
    x0 = np.linspace(0, 2, n0, endpoint=False)
    basis = basis_evaluation(x0, M)
    
    # 2. Global search with DE
    t_start = time.time()
    bounds = [(-0.5, 0.5)] * (2 * M)
    de_result = differential_evolution(
        fitness_function,
        bounds,
        args=(n0, M),
        popsize=15 * (2*M),
        maxiter=3000,
        seed=seed,
        disp=False
    )
    best_c = de_result.x
    t_elapsed = time.time() - t_start
    
    # Select good candidates
    candidates = [best_c]
    subsample = 5
    for i in range(1, subsample):
        np.random.seed(i + seed)
        fit_func = lambda c: fitness_function(c, n0, M)
        c = np.random.uniform(-0.5, 0.5, 2 * M)
        result = differential_evolution(fit_func, bounds, maxiter=100, seed=seed+i)
        candidates.append(result.x)
    
    best_candidates = sorted(candidates, key=lambda c: fitness_function(c, n0, M))[:5]
    
    best_h = None
    best_c5 = np.inf
    
    # 3. SLP refinement
    for c in best_candidates:
        if time.time() - start_time > phase_time_budgets['slp'] + 0.01:
            break
        h = construct_h_from_coeffs(c, n0, M)
        c5 = evaluate_c5(h)
        t_start = time.time()
        max_iter = 200
        for i in range(max_iter):
            if time.time() - start_time > phase_time_budgets['slp'] + 0.01:
                break
            corr = np.correlate(h, 1-h, mode='full')
            k_star = np.argmax(corr)
            if k_star > len(h):
                k = k_star - len(h)
            else:
                k = k_star - len(h) + 1
            h_new, t_new = solve_linearized_lp(h, k, n0)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5 - 1e-6:
                h = h_new
                c5 = c5_new
            else:
                h = (1 - 0.1) * h + 0.1 * h_new
                c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    if best_h is None:
        # Fallback to best from DE
        best_h = construct_h_from_coeffs(best_c, n0, M)
        best_c5 = evaluate_c5(best_h)
    
    # 4. Upsampling
    n_up = int(200 + (budget_s - 0.01) / 2)
    if n_up > 800:
        n_up = 800
    if n_up < 500:
        n_up = 500
        
    # Interpolation
    x_up = np.linspace(0, 2, n_up, endpoint=False)
    # Use Fourier approximations as a smooth function
    basis_up = basis_evaluation(x_up, M)
    h_interp = 0.5 + np.dot(basis_up, best_c)
    h_interp = np.clip(h_interp, 0, 1)
    
    # Scaling
    scale = (n_up / 2) / np.sum(h_interp)
    h_up = h_interp * scale
    h_up = project_to_feasible(h_up, n_up)
    
    # Final polishing
    for i in range(1000):  # Limit steps
        if time.time() - start_time > budget_s - 0.5:
            break
        h_up = subgradient_step(h_up, n_up)
        
    final_c5 = evaluate_c5(h_up)
    
    return (h_up, final_c5, len(h_up))
