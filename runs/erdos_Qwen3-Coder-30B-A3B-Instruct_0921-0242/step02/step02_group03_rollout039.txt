import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def compute_correlations(h):
    """Compute all correlations c_s for the given h."""
    n = len(h)
    dx = 2.0 / n
    # Pad and use FFT for efficient computation
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    corr_vals = corr_vals[:n]  # Take only the non-cyclic part
    # Normalize by dx
    corr_vals = corr_vals * dx
    return corr_vals

def compute_correlation_changes(h, i, j):
    """Compute correlation changes when swapping h[i] and h[j]."""
    n = len(h)
    delta = np.zeros(n)
    # For each shift s, compute the change in correlation
    for s in range(n):
        if i + s < n:
            delta[s] += (1 - h[i + s]) - h[i + s]
        if j + s < n:
            delta[s] += h[j + s] - (1 - h[j + s])
        if i - s >= 0:
            delta[s] -= (1 - h[i - s]) - h[i - s]
        if j - s >= 0:
            delta[s] -= h[j - s] - (1 - h[j - s])
    return delta

def anneal_binary(h, n, time_limit):
    """Perform simulated annealing on a binary vector."""
    T0 = 0.02 * n
    alpha = 0.997
    max_moves = 50000
    T = T0
    current_h = h.copy()
    current_corr = compute_correlations(current_h)
    current_max_corr = np.max(current_corr)
    best_h = current_h.copy()
    best_max_corr = current_max_corr
    
    moves_without_improvement = 0
    move = 0
    while T > 1e-5 and move < max_moves and time.time() < time_limit:
        # Pick two random positions to swap
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue

        # Compute the change in correlations
        delta = compute_correlation_changes(current_h, i, j)
        new_corr = current_corr + delta
        new_max_corr = np.max(new_corr)

        # Metropolis criterion
        if new_max_corr <= current_max_corr or np.random.rand() < np.exp(-(new_max_corr - current_max_corr)/T):
            current_h[i], current_h[j] = current_h[j], current_h[i]
            current_corr = new_corr
            current_max_corr = new_max_corr
            moves_without_improvement = 0
            if current_max_corr < best_max_corr:
                best_h = current_h.copy()
                best_max_corr = current_max_corr
        else:
            moves_without_improvement += 1
            if moves_without_improvement > 5000:
                break

        move += 1
        T *= alpha

    return best_h, best_max_corr

def polish_continuous(h, n):
    """Perform smooth gradient ascent to refine continuous solution."""
    dx = 2.0 / n
    h_float = h.astype(float)
    max_iter = 30
    initial_eta = 0.1 * dx
    lambda_param = 5.0
    
    for it in range(max_iter):
        # Compute correlations
        corr = compute_correlations(h_float)
        # Smooth max approximation
        phi = (1.0 / lambda_param) * np.log(np.sum(np.exp(lambda_param * corr)))
        # Compute gradient of the smooth max
        grad = np.zeros(n)
        for i in range(n):
            for s in range(n):
                if i + s < n:
                    grad[i] += (1 - h_float[i + s]) * np.exp(lambda_param * corr[s])
                if i - s >= 0:
                    grad[i] -= h_float[i - s] * np.exp(lambda_param * corr[s])
        grad /= np.sum(np.exp(lambda_param * corr))
        
        # Gradient step with diminishing step size
        eta = initial_eta * (0.5 ** (it // 10))
        h_float = h_float - eta * grad
        
        # Project back onto constraints
        h_float = np.clip(h_float, 0, 1)
        h_float = project_to_feasible(h_float, n)
        
        # Update lambda for sharper approximation
        lambda_param = min(lambda_param + 5.0, 30.0)
    
    return h_float, evaluate_c5(h_float)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 10  # Keep some buffer for final steps
    
    # Choose odd number for better symmetry properties
    n = 221  # This is a good balance for our resolution needs
    dx = 2.0 / n
    
    best_h = None
    best_val = float('inf')
    
    num_trials = 10
    for trial in range(num_trials):
        if time.time() > time_limit:
            break
            
        # Construct initial binary state
        h = np.zeros(n, dtype=np.int8)
        h[:n//2] = 1
        np.random.shuffle(h)
        
        # Local search with simulated annealing
        h_local, val_local = anneal_binary(h, n, time_limit)
        
        # Polish continuous solution
        h_polished, val_polished = polish_continuous(h_local, n)
        
        if val_polished < best_val:
            best_val = val_polished
            best_h = h_polished.copy()
            
    if best_h is None:
        # Fallback to a simple construction
        h_fallback = np.zeros(n)
        h_fallback[n//4:3*n//4] = 1.0
        h_fallback = project_to_feasible(h_fallback, n)
        best_h = h_fallback.copy()
        best_val = evaluate_c5(best_h)
    
    return (best_h, best_val, n)
