import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5_exact(h):
    """Computes the exact C5 constant using full correlation."""
    dx = 2.0 / len(h)
    # With mode='full', correlate computes auto-correlations for all lags
    corr = correlate(h, 1 - h, mode='full')
    # Take only positive lags, cut off the first element (lag 0)
    corr_positive = corr[len(h)-1:]  # This slice gives us lags k=1 to n-1
    max_corr = np.max(corr_positive) * dx
    return max_corr

def smooth_max(h, p=128):
    """Smooth approximation of max."""
    dx = 2.0 / len(h)
    # auto-correlation of h with itself using FFT (linear correlation)
    h_fft = fft(h, 2 * len(h))
    hh_fft = fft(h * h, 2 * len(h))
    corr = ifft(h_fft * np.conj(h_fft))
    corrhh = ifft(h_fft * np.conj(hh_fft))
    # Compute overlap values for lags 1 to n-1
    overlap = np.array([corrhh[k].real * dx for k in range(1, len(h))])
    # C(k, h) = 1 - dx * h_k * h_{k+i}
    C_values = 1.0 - overlap  # Values for k=1 to n-1
    # Smooth max
    S_p = np.sum(C_values ** p) ** (1.0 / p)
    return S_p, C_values

def compute_gradient(h, p=128):
    """Compute gradient of smooth max with respect to h."""
    dx = 2.0 / len(h)
    n = len(h)
    grad = np.zeros(n)
    # Compute overlap as in smooth_max
    h_fft = fft(h, 2 * n)
    hh_fft = fft(h * h, 2 * n)
    corr = ifft(h_fft * np.conj(h_fft))
    corrhh = ifft(h_fft * np.conj(hh_fft))
    # Compute overlap values
    overlap = np.array([corrhh[k].real * dx for k in range(1, n)])
    C_values = 1.0 - overlap

    # Compute gradient for each index i
    # ∂S_p / ∂h_i = (1 / S_p^(p-1)) * Σ_k w_k * ∂C(k)/∂h_i
    # ∂C(k) / ∂h_i = dx * [ (1 - h_{i+k}) - h_{i-k} ]  (indices modulo n)

    # w_k = C(k)^{p-1}
    w = C_values ** (p - 1)
    # Total gradient
    for i in range(n):
        # For k=1 to n-1:
        tmp_grad = 0.0
        for k in range(1, n):
            left_idx = (i + k) % n
            right_idx = (i - k) % n
            tmp_grad += w[k - 1] * dx * ((1.0 - h[left_idx]) - h[right_idx])
        grad[i] = tmp_grad / (np.sum(C_values ** p) ** ((p - 1.0) / p))

    # Add Lagrange multiplier term for sum constraint (penalty method)
    lambda_sum = 1e4
    grad += 2 * lambda_sum * (np.sum(h) - n / 2)

    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Sample count
    n = 350
    dx = 2.0 / n
    
    # Generate seeds
    seeds = []
    # 1. Half-filled
    h = np.ones(n) * 0.5
    seeds.append(project_to_feasible(h, n))
    
    # 2. Random Dirichlet
    h = np.random.dirichlet(np.ones(n)) * (n / 2)
    seeds.append(project_to_feasible(h, n))
    
    # 3. Interpolated initial
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) > n:
            x_orig = np.linspace(0, 2, len(orig_h))
            x_new = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_orig, orig_h)
        else:
            h_interp = orig_h[:n]
        h = h_interp * (n / 2) / np.sum(h_interp)
        h = project_to_feasible(h, n)
        seeds.append(h)
    
    # 4. Low-high binary
    h = np.zeros(n)
    h[:n//2] = 1.0
    h = project_to_feasible(h, n)
    seeds.append(h)
    
    # 5. Normal random with clamping
    h = np.random.normal(0.5, 0.2, n)
    h = project_to_feasible(h, n)
    seeds.append(h)
    
    # Additional random seeds
    for _ in range(25):
        h = np.random.rand(n)
        h = h * (n / 2) / np.sum(h)
        h = project_to_feasible(h, n)
        seeds.append(h)

    # Homotopy parameters
    p_values = [8, 16, 32, 64, 128, 256]
    best_h = None
    best_c5 = np.inf
    best_p = 0

    for p in p_values:
        if time.time() - start_time > budget_s * 0.8:
            break
        print(f"Running with p={p}")
        best_here_h = None
        best_here_c5 = np.inf
        for h_seed in seeds:
            if time.time() - start_time > budget_s * 0.8:
                break
            try:
                def objective(h):
                    S_p, _ = smooth_max(h, p)
                    return S_p
                def grad(h):
                    return compute_gradient(h, p)
                # Minimize
                res = minimize(objective, h_seed, method='L-BFGSB', jac=grad,
                               bounds=[(0, 1) for _ in range(n)],
                               options={'maxiter': 100})
                if res.success:
                    h_opt = res.x
                    h_opt = project_to_feasible(h_opt, n)
                    c5 = evaluate_c5_exact(h_opt)
                    if c5 < best_here_c5:
                        best_here_c5 = c5
                        best_here_h = h_opt.copy()
                else:
                    # fallback to original if not successful
                    h_opt = h_seed.copy()
                    c5 = evaluate_c5_exact(h_opt)
                    if c5 < best_here_c5:
                        best_here_c5 = c5
                        best_here_h = h_opt.copy()
            except Exception as e:
                continue
            if best_here_c5 < best_c5:
                best_c5 = best_here_c5
                best_h = best_here_h.copy()
                best_p = p
        
        # Use the best from this stage as starting point for next p
        if best_here_h is not None:
            seeds = [best_here_h] + seeds[:10]  # Keep 10 best ones
        if time.time() - start_time > budget_s * 0.8:
            break

    # Final deterministic refinement with worst-shift update
    if best_h is not None:
        h_refined = best_h.copy()
        if time.time() - start_time < budget_s * 0.98:
            for _ in range(50):
                if time.time() - start_time > budget_s * 0.98:
                    break
                # Compute actual C values (to find worst shift)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                # Take max from 1 to n-1 (excluding 0 which is self-correlation)
                c_vals = corr[len(h_refined):]
                k_worst = np.argmax(c_vals) + 1  # k=1,2,...,n-1
                # Compute gradient for that shift only
                grad_worst = np.zeros_like(h_refined)
                for i in range(len(h_refined)):
                    left_idx = (i + k_worst) % len(h_refined)
                    right_idx = (i - k_worst) % len(h_refined)
                    grad_worst[i] = dx * ((1 - h_refined[left_idx]) - h_refined[right_idx])
                
                # Backtracking line search
                alpha = 0.1
                max_iter = 20
                for _ in range(max_iter):
                    h_candidate = h_refined - alpha * grad_worst
                    h_candidate = project_to_feasible(h_candidate, n)
                    c5_candidate = evaluate_c5_exact(h_candidate)
                    if c5_candidate < evaluate_c5_exact(h_refined):
                        h_refined = h_candidate
                        break
                    alpha *= 0.5
                if abs(evaluate_c5_exact(h_refined) - evaluate_c5_exact(best_h)) < 1e-6:
                    break
    
    # Final polishing with simulated annealing
    if time.time() - start_time < budget_s * 0.99:
        h_final = h_refined.copy()
        c5_final = evaluate_c5_exact(h_final)
        temp = 0.1
        decay = 0.95
        for _ in range(1000):
            if time.time() - start_time > budget_s * 0.99:
                break
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if h_final[i] <= 0 or h_final[j] >= 1:
                continue
            delta = np.random.uniform(0, min(h_final[i], 1 - h_final[j], 0.01))
            h_test = h_final.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, n)
            c5_test = evaluate_c5_exact(h_test)
            if c5_test < c5_final:
                h_final = h_test
                c5_final = c5_test
            elif np.random.rand() < np.exp(-(c5_test - c5_final) / temp):
                h_final = h_test
                c5_final = c5_test
            temp *= decay
    else:
        h_final = h_refined
        c5_final = evaluate_c5_exact(h_refined)

    return (h_final, c5_final, n)
