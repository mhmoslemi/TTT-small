import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def full_to_sym(h_full):
    """Convert full symmetric vector to symmetric half + center."""
    n = len(h_full)
    mid = n // 2
    return h_full[:mid+1]

def sym_to_full(x_sym):
    """Convert symmetric half + center to full symmetric vector."""
    n = 2 * (len(x_sym) - 1)
    x_full = np.zeros(n)
    x_full[:len(x_sym)] = x_sym
    x_full[len(x_sym):] = x_sym[-2::-1]
    return x_full

def build_constraints(x, t, shift_set, n):
    """Build linear constraints for the LP based on shift_set."""
    constraints = []
    dx = 2.0 / n
    
    # Create a sparse matrix for each shift
    for s in shift_set:
        # Linear constraint: sum_i x_i * (1 - x_{i+s}) <= t
        # Precompute indices for efficient constraint formation
        expr = 0
        for i in range(n):
            j = i + s
            if j >= n:
                # Don't include out-of-bounds terms
                continue
            expr += x[i] * (1 - x[j])  # This will be *dx later
        # Scale by dx to match original scaling
        expr = expr * dx
        constraints.append(expr <= t)
    return constraints

def solve_lp(shift_set, x0=None, n=400):
    """Solve the LP with given shifts."""
    # Use symmetry: reduce to length n//2 + 1
    m = n // 2 + 1
    x = cp.Variable(m)
    t = cp.Variable()
    
    # Add constraints for non-negativity and upper bound
    constraints = [0 <= x, x <= 1, cp.sum(x) == 0.5]  # Integral constraint: sum * dx = 1
    
    # Add correlation constraints
    constraints.extend(build_constraints(x, t, shift_set, n))
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            x_val = x.value
            t_val = t.value
            return x_val, t_val
        else:
            return None, None
    except Exception:
        return None, None

def compute_full_correlation(h):
    """Compute full correlation including all shifts."""
    dx = 2.0 / len(h)
    corr_full = correlate(h, 1 - h, mode='full')
    # The middle index corresponds to shift 0
    return corr_full * dx
