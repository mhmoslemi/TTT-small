import numpy as np
import time

def project_box_sum(y, total):
    """Project y onto {x : 0 <= x[i] <= 1, sum x = total}."""
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if abs(s - total) < 1e-10:
        return y
    # Use bisection on lambda to find the right shift
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        test = np.sum(np.clip(y - mid, 0, 1))
        if test > total:
            lo = mid
        else:
            hi = mid
    lambda_opt = (lo + hi) / 2.0
    return np.clip(y - lambda_opt, 0, 1)

def compute_autocorrelation(h):
    """Compute aperiodic autocorrelation of h."""
    n = len(h)
    H = np.fft.fft(h)
    P = np.abs(H) ** 2
    gamma = np.fft.ifft(P).real
    return gamma

def compute_c5_from_autocorr(gamma, n):
    """Compute C5 from autocorrelation gamma."""
    avg = (np.sum(gamma) - gamma[0]) / (n - 1)
    gamma_target = gamma.copy()
    gamma_target[1:] = 0.7 * gamma_target[1:] + 0.3 * avg
    return 1.0 - 2.0 * np.max(gamma_target) / n

def compute_c5(h):
    """Compute C5 directly from h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def compute_delta_autocorr(h, i, j):
    """Compute Δγ for swapping h[i]=1, h[j]=0."""
    n = len(h)
    dgamma = np.zeros(2 * n - 1)
    # For k from -(n-1) to n-1
    for k in range(-(n-1), n):
        # At lag k
        left_i = (i + k) % n
        left_j = (j + k) % n
        right_i = (i - k) % n
        right_j = (j - k) % n
        dgamma[k + (n - 1)] = -h[i] * h[left_i] - h[i] * h[right_i] + h[j] * h[left_j] + h[j] * h[right_j]
    return dgamma

def binary_hill_climb(h, max_swaps=3 * 256):
    """Improve a binary h by swaps."""
    n = len(h)
    gamma = compute_autocorrelation(h)
    c5_current = compute_c5_from_autocorr(gamma, n)
    swapped = 0
    while swapped < max_swaps:
        candidates = np.where((h == 1) & (np.roll(h, 1) == 0))[0]
        if len(candidates) == 0:
            break
        i = np.random.choice(candidates)
        j = np.random.choice(np.where(h == 0)[0])
        if h[i] == 1 and h[j] == 0:
            delta_gamma = compute_delta_autocorr(h, i, j)
            new_gamma = gamma + delta_gamma
            c5_after = compute_c5_from_autocorr(new_gamma, n)
            if c5_after < c5_current:
                h[i] = 0
                h[j] = 1
                gamma = new_gamma
                c5_current = c5_after
                swapped = 0
            else:
                swapped += 1
        else:
            swapped += 1
    return h, c5_current

def generate_seeds(n):
    """Generate binary seeds."""
    # 1. Legendre
    p = 257  # prime
    legendre = np.array([1 if pow(k, (p - 1) // 2, p) == 1 else 0 for k in range(1, p)])
    h_legendre = legendre[:n]
    
    # 2. m-sequence
    lfsr = np.zeros(256, dtype=int)
    lfsr[0] = 1
    for i in range(1, 255):
        lfsr[i] = (lfsr[i-1] + lfsr[i-13] + lfsr[i-14] + lfsr[i-16]) & 1
    h_mseq = np.roll(lfsr, -1)  # rm first, add 0 to make length 256
    h_mseq[-1] = 0
    
    # 3. Random balanced
    h_random = np.zeros(n, dtype=int)
    ones_idx = np.random.choice(n, n // 2, replace=False)
    h_random[ones_idx] = 1
    
    seeds = [h_legendre, h_mseq, h_random]
    return seeds

def sap_refinement(h, max_iter=400):
    """Spectral Alternating Projection."""
    n = len(h)
    h = h.astype(float)
    beta = 5.0
    c5_best = compute_c5(h)
    h_best = h.copy()
    alpha = 0.7
    
    for it in range(max_iter):
        H = np.fft.fft(h)
        gamma = np.fft.ifft(np.abs(H)**2).real
        avg = (np.sum(gamma) - gamma[0]) / (n - 1)
        gamma_targ = gamma.copy()
        gamma_targ[1:] = alpha * gamma_targ[1:] + (1 - alpha) * avg
        P_target = np.fft.fft(gamma_targ)
        P_target = np.maximum(P_target, 0)
        A_target = np.sqrt(P_target)
        H_new = A_target * np.exp(1j * np.angle(H))
        h_new = np.fft.ifft(H_new).real
        h_new = project_box_sum(h_new, n / 2)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
        h = h_new
        
        if it > 20:
            # Stop if no improvement in last 20 iterations
            improvement = max(1e-10, c5_best - c5_new)
            if improvement < 1e-6:
                break
        if alpha < 0.99:
            alpha = min(alpha + 0.02, 0.99)
    return h_best, c5_best

def polish_with_gradient(h, max_iter=100):
    """Projected gradient descent with soft-max weight."""
    n = len(h)
    h = h.copy()
    beta = 5.0
    c5_current = compute_c5(h)
    eta = 1.0
    for it in range(max_iter):
        # Compute weights
        gamma = compute_autocorrelation(h)
        C = 1.0 - 2.0 * gamma / n
        w = np.exp(beta * C)
        w = w / np.sum(w)
        
        # Compute gradient
        h_fft = np.fft.fft(h)
        w_fft = np.fft.fft(w)
        grad_real = np.fft.ifft(w_fft.conj() * h_fft).real
        grad_imag = np.fft.ifft(w_fft * h_fft).real
        
        grad = 2.0 / n * (grad_real + grad_imag)
        
        # Backtracking line search
        h_trial = h - eta * grad
        h_trial = project_box_sum(h_trial, n / 2)
        c5_trial = compute_c5(h_trial)
        if c5_trial < c5_current:
            h = h_trial
            c5_current = c5_trial
            eta = min(eta * 1.2, 5.0)
        else:
            eta = eta * 0.5
            if eta < 1e-6:
                break
        if it % 20 == 0:
            beta = beta * 1.2
    return h, c5_current

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 256
    dx = 2.0 / n
    
    # Generate and improve seeds
    seeds = generate_seeds(n)
    h_best = None
    c5_best = float('inf')
    
    # Add initial_h_values
    if 'initial_h_values' in globals():
        h_init = initial_h_values
        if len(h_init) < n:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n)
            h_interpolated = np.interp(x_new, x_orig, h_init)
        else:
            h_interpolated = h_init[:n]
        h_init_norm = h_interpolated * (n / 2) / np.sum(h_interpolated)
        h_s = [h_init_norm]
        h_s.extend(seeds)
    else:
        h_s = seeds
    
    # Binary hill-climbing
    for h_seed in h_s:
        h_bin = h_seed.copy()
        h_bin = project_box_sum(h_bin, n / 2)
        h_bin = h_bin.astype(int)
        if np.sum(h_bin) != n // 2:
            # Force the sum to be exactly n/2
            ones_pos = np.where(h_bin == 1)[0]
            zero_pos = np.where(h_bin == 0)[0]
            if len(ones_pos) > n // 2:
                # Remove excess 1s
                indices_to_remove = np.random.choice(ones_pos, len(ones_pos) - n // 2, replace=False)
                h_bin[indices_to_remove] = 0
            elif len(ones_pos) < n // 2:
                # Add extra 1s
                indices_to_add = np.random.choice(zero_pos, n // 2 - len(ones_pos), replace=False)
                h_bin[indices_to_add] = 1
        # Ensure binary
        h_bin = h_bin.astype(int)
        h_improved, c5_improved = binary_hill_climb(h_bin)
        if c5_improved < c5_best:
            c5_best = c5_improved
            h_best = h_improved.copy()
    
    # SAP
    if h_best is not None:
        h_sap, c5_sap = sap_refinement(h_best)
        if c5_sap < c5_best:
            c5_best = c5_sap
            h_best = h_sap.copy()
    
    # Final polishing
    if time.time() - start_time < budget_s - 0.5:
        h_final, c5_final = polish_with_gradient(h_best)
        if c5_final < c5_best:
            c5_best = c5_final
            h_best = h_final.copy()
    
    return (h_best, c5_best, n)
