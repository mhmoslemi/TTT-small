import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_to_sum_constraint(h, target_sum):
    """Project h to satisfy sum(h) = target_sum with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum

    if abs(diff) < 1e-10:
        return h

    # Water-filling approach: add uniform adjustment
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)

    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def compute_correlations(h):
    """Compute full correlations for all shifts."""
    dx = 2.0 / len(h)
    # Full correlation using FFT
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # Take real part
    corr_vals = np.real(corr)
    return corr_vals * dx

def gradient_worst_shift(h):
    """Compute the gradient with respect to the worst shift."""
    # Compute all correlations
    corr_vals = compute_correlations(h)
    # Exclude shift 0
    corr_excl_zero = corr_vals[1:len(h)]
    # Find worst shift (minimum correlation) for gradient computation
    worst_shift = np.argmin(corr_excl_zero) + 1
    # Compute gradient for shift_worst
    n = len(h)
    grad = np.zeros_like(h)
    for i in range(n):
        left_idx = (i + worst_shift) % n
        right_idx = (i - worst_shift) % n
        grad[i] = (1 - h[left_idx]) - h[right_idx]
    return grad * (2.0 / n)

def gradient_descent_step(h, step_size):
    """Perform one gradient descent step."""
    grad = gradient_worst_shift(h)
    h_new = h - step_size * grad
    # Project to feasible domain
    h_new = project_to_sum_constraint(h_new, len(h) / 2)
    return h_new

def gradient_worst_shift_descent(h, max_iter=200, step_size=0.1, tol=1e-7):
    """Run gradient descent on the worst shift."""
    current_h = h.copy()
    for i in range(max_iter):
        h_new = gradient_descent_step(current_h, step_size)
        if np.linalg.norm(h_new - current_h) < tol:
            break
        current_h = h_new
    return current_h

def generate_population(n_points, pop_size, initial_h=None):
    """Generate a population of step functions."""
    population = []
    # Always include the initial solution
    if initial_h is not None:
        h_init = initial_h.copy()
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_init = project_to_sum_constraint(h_init, n_points / 2)
        population.append(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = project_to_sum_constraint(h_init, n_points / 2)
        population.append(h_init)
    
    # Fill remaining with random samples
    for i in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = project_to_sum_constraint(h, n_points / 2)
        population.append(h)
    
    return population

def differential_evolution_step(population, fitnesses, F=0.6, CR=0.9):
    """One step of differential evolution."""
    new_pop = []
    for i in range(len(population)):
        # Select three distinct vectors
        idxs = np.random.choice(len(population), 3, replace=False)
        a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
        
        # Generate donor vector
        donor = a + F * (b - c)
        # Clip to [0,1]
        donor = np.clip(donor, 0, 1)
        # Enforce sum constraint
        donor = project_to_sum_constraint(donor, len(donor) / 2)
        
        # Binomial crossover
        trial = np.zeros_like(donor)
        j_rand = np.random.randint(0, len(donor))
        for j in range(len(donor)):
            if np.random.rand() < CR or j == j_rand:
                trial[j] = donor[j]
            else:
                trial[j] = population[i][j]
        
        new_pop.append(trial)
    
    return new_pop

    # Use a moderate size
    n_points = min(400, len(orig_h))
    if len(orig_h) > n_points:
        h_start = orig_h[:n_points]
    else:
        x_orig = np.linspace(0, 2, len(orig_h))
        x_new = np.linspace(0, 2, n_points)
        h_start = np.interp(x_new, x_orig, orig_h)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = project_to_sum_constraint(h_start, n_points / 2)
    
    # Parameters for DE
    pop_size = min(40, 2 * n_points // 10)
    F = 0.6
    CR = 0.9
    max_generations = 5000
    freq_refinement = 200
    
    # Initialize population
    population = generate_population(n_points, pop_size, h_start)
    
    # Track best
    global_best = None
    global_best_c5 = float('inf')
    last_improvement = 0
    
    # Run DE
    gen = 0
    while gen < max_generations and (time.time() - start_time < budget_s * 0.75):
        # Evaluate current population
        fitnesses = []
        for h in population:
            c5 = evaluate_c5(h)
            fitnesses.append(c5)
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best = h.copy()
                last_improvement = time.time()
                
        # Let DE evolve
        new_population = differential_evolution_step(population, fitnesses, F, CR)
        population = new_population
        gen += 1
        
        # Occasionally refine
        if gen % freq_refinement == 0 and global_best is not None:
            ref_h = gradient_worst_shift_descent(global_best.copy(), max_iter=100)
            ref_c5 = evaluate_c5(ref_h)
            if ref_c5 < global_best_c5:
                global_best_c5 = ref_c5

        c5_refined = evaluate_c5(h_refined)
        if c5_refined < global_best_c5:
            global_best_c5 = c5_refined
            global_best = h_refined.copy()

    if global_best is not None and time.time() - start_time < budget_s * 0.995:

        c5_final = global_best_c5
        temp = 1.0
        cooling_rate = 0.9999

            h_test = project_to_sum_constraint(h_test, n_points / 2)
            
            c5_test = evaluate_c5(h_test)
            total += 1
            
            if c5_test < c5_final:
                c5_final = c5_test
                h_final = h_test
                accepted += 1
            else:
                # Accept with probability
                prob = np.exp(-(c5_test - c5_final) / temp)
                if np.random.rand() < prob:
                    c5_final = c5_test
                    h_final = h_test
                    accepted += 1
            
            temp *= cooling_rate
        
        if accepted > 0:
            global_best_c5 = c5_final
            global_best = h_final.copy()
    
    # Optionally refine at higher resolution
    if global_best is not None and time.time() - start_time < budget_s * 0.999:
        # Interpolate to higher resolution
        n_high = 2 * n_points
        x_orig = np.linspace(0, 2, n_points)

        h_high = h_high * (n_high / 2) / np.sum(h_high)
        h_high = project_to_sum_constraint(h_high, n_high / 2)
        
        # Short gradient descent at higher resolution
        h_high_refined = gradient_worst_shift_descent(h_high.copy(), max_iter=50)
        c5_high = evaluate_c5(h_high_refined)

            global_best_c5 = c5_high
            global_best = h_high_refined.copy()
            n_points = n_high
    
    # Return final result
    if global_best is not None:
        return (global_best, global_best_c5, n_points)
    else:
        h_fallback = np.ones(n_points) * 0.5
        h_fallback = project_to_sum_constraint(h_fallback, n_points / 2)
        c5_fallback = evaluate_c5(h_fallback)
        return (h_fallback, c5_fallback, n_points)
