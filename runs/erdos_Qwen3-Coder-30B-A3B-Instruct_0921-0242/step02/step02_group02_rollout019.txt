import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def project_to_feasible(h, n):
    """Project h onto [0,1] and ensure sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
        
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n / 2:
            ub = mid
        else:
            lb = mid

    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def evaluate_c5(h):
    """Evaluate C5 using FFT-based correlation."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation via FFT
    H = rfft(h, n)
    autocorr = irfft(H * np.conj(H), n)
    autocorr = np.real_if_close(autocorr)
    c = dx * (n/2.0 - autocorr)
    return np.max(c)

def fix_h_to_feasible(h, n):
    """Ensure h sums to n/2 with bounds in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    if abs(diff) < 1e-10:
        return h
    # Normalize
    h = h * (n / 2) / current_sum
    return h

def generate_random_binary(n, weight):
    """Generate a random binary vector with given weight."""
    h = np.zeros(n)
    indices = np.random.permutation(n)[:weight]
    h[indices] = 1.0
    return h

def binary_anneal(h_bin, time_limit):
    """Perform binary simulated annealing with exact autocorrelation updates."""
    n = len(h_bin)
    dx = 2.0 / n
    
    # Compute initial autocorrelation
    H = rfft(h_bin, n)
    autocorr = irfft(H * np.conj(H), n)
    autocorr = np.real_if_close(autocorr)
    c = dx * (n/2.0 - autocorr)
    best_c = np.max(c)
    best_h = h_bin.copy()
    
    # Temperature schedule
    T = 5 * dx
    T_min = dx * 1e-4
    alpha = 0.9997
    counter = 0
    start_time = time.time()
    
    # Metrics
    swaps = 0
    stagnation = 0
    
    while time.time() - start_time < time_limit:
        if stagnation > 200000:
            # Large jump: reassign 5% of ones
            ones = np.where(h_bin == 1)[0]
            zero_indices = np.where(h_bin == 0)[0]
            num_to_change = max(1, len(ones) // 20)
            to_remove = np.random.choice(ones, size=num_to_change, replace=False)
            to_add = np.random.choice(zero_indices, size=num_to_change, replace=False)
            h_bin[to_remove] = 0
            h_bin[to_add] = 1
            # Recompute autocorrelation
            H = rfft(h_bin, n)
            autocorr = irfft(H * np.conj(H), n)
            autocorr = np.real_if_close(autocorr)
            c = dx * (n/2.0 - autocorr)
            best_c = np.max(c)
            best_h = h_bin.copy()
            stagnation = 0
            T = max(T * 0.9, T_min)
            continue
            
        # Pick random 0 and 1
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute update to autocorrelation
        h_shift_i = np.roll(h_bin, i)
        h_shift_j = np.roll(h_bin, j)
        delta = -h_shift_i + h_shift_j
        # Self terms for shift (i-j) and (j-i)
        delta[i - j] += 1.0
        delta[j - i] += 1.0
        autocorr_new = autocorr + delta
        c_new = dx * (n/2.0 - autocorr_new)
        c_new_max = np.max(c_new)
        
        # Accept or reject
        if c_new_max <= c.max() or np.random.rand() < np.exp(-(c_new_max - c.max()) / T):
            h_bin[i] = 0
            h_bin[j] = 1
            autocorr = autocorr_new
            c = c_new
            if c_new_max < best_c:
                best_c = c_new_max
                best_h = h_bin.copy()
            stagnation = 0
        else:
            stagnation += 1
            
        swaps += 1
        if swaps % 1000 == 0:
            T *= alpha
            T = max(T, T_min)
    
    return best_h, best_c

def smooth_boundaries(h_bin, ramp_width=2):
    """Insert linear ramps at binary transitions."""
    n = len(h_bin)
    h_cont = h_bin.astype(float)
    
    # Scan for zero-to-one transitions
    for i in range(n):
        if h_bin[i] == 0 and h_bin[(i+1)%n] == 1:
            # Insert ramp from 0 to 1 over ramp_width samples
            for r in range(ramp_width):
                j = (i+1+r) % n
                h_cont[j] = float(r+1) / (ramp_width+1)
        elif h_bin[i] == 1 and h_bin[(i+1)%n] == 0:
            # Insert ramp from 1 to 0 over ramp_width samples
            for r in range(ramp_width):
                j = (i+1+r) % n
                h_cont[j] = 1.0 - float(r+1) / (ramp_width+1)
    
    return h_cont

def proj_grad_refine(h_cont, time_limit, use_smooth=False):
    """Refine continuous function using projected gradient descent."""
    n = len(h_cont)
    dx = 2.0 / n
    start_time = time.time()
    best_c = evaluate_c5(h_cont)
    best_h = h_cont.copy()
    
    gamma = 10.0  # For smoothing surrogate
    
    def get_grad(h, k_star):
        """Compute subgradient of c_k_star."""
        h_shift = np.roll(h, k_star)
        h_shift_neg = np.roll(h, -k_star)
        return -dx * (h_shift + h_shift_neg)
    
    def backtrack_line_search(h, g, step_size=0.5):
        """Backtracking line search."""
        alpha = step_size
        c_current = evaluate_c5(h)
        for _ in range(10):
            h_new = h - alpha * g
            h_new = project_to_feasible(h_new, n)
            c_new = evaluate_c5(h_new)
            if c_new < c_current:
                return h_new, c_new
            alpha *= 0.5
        return h, c_current
    
    iter_counter = 0
    while time.time() - start_time < time_limit:
        # Find worst correlation index
        c_vec = np.zeros(n)
        H = rfft(h_cont, n)
        autocorr = irfft(H * np.conj(H), n)
        autocorr = np.real_if_close(autocorr)
        c_vec = dx * (n/2.0 - autocorr)
        k_star = np.argmax(c_vec)
        g = get_grad(h_cont, k_star)
        
        # Line search
        h_new, c_new = backtrack_line_search(h_cont, g)
        h_cont = h_new
        c_current = c_new
        
        if c_current < best_c:
            best_c = c_current
            best_h = h_cont.copy()
        
        iter_counter += 1
        # End early if stagnating
        if iter_counter % 1000 == 0 and (iter_counter % 10000 == 0 or abs(c_current - best_c) < 1e-5):
            # Check for convergence or saturation
            pass
    
    return best_h, best_c

def upsample_and_refine(h, n_new, time_limit):
    """Upsample and refine the function."""
    n = len(h)
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    
    h_new = np.interp(x_new, x_old, h)
    
    # Ensure the new vector satisfies sum = n_new / 2
    h_new = fix_h_to_feasible(h_new, n_new)
    h_new, c_new = proj_grad_refine(h_new, time_limit)
    return h_new, c_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_initial = 256
    dx_initial = 2.0 / n_initial

    # Step 1: Binary optimization
    h_bin = generate_random_binary(n_initial, n_initial // 2)
    h_bin, c5_bin = binary_anneal(h_bin, 0.55 * budget_s)
    
    # Step 2: Smooth boundary
    h_smooth = smooth_boundaries(h_bin)
    h_smooth = project_to_feasible(h_smooth, n_initial)

    # Step 3: Refine smooth curve
    h_refined, c5_refined = proj_grad_refine(h_smooth, 0.25 * budget_s)
    
    # Step 4: Upsample and refine
    n_upsample = 2 * n_initial
    h_upsampled, c5_upsampled = upsample_and_refine(h_refined, n_upsample, 0.15 * budget_s)
    
    # Final evaluation
    final_h = h_upsampled
    final_c5 = c5_upsampled
    final_n = n_upsample
    
    return (final_h, final_c5, final_n)
