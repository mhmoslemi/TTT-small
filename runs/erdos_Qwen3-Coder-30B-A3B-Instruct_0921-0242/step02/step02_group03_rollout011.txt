def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def memetic_descent(h, n_points, max_steps=20):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero)
        # The maximum is at shift 0, so we need to find the min among non-zero shifts
        if n_points == 1:
            non_zero_shifts = np.array([])
        else:
            non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        if len(non_zero_shifts) > 0:
            worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        else:
            worst_shift_idx = 1

        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        # Gradient wrt h[i] is dx * (1 - h[i + worst_shift_idx] - h[i - worst_shift_idx]) 
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def build_symmetric_binary_pattern(init_h, n):
    """
    Build a symmetric binary pattern from initial h values.
    Ensure that it is symmetric and has sum = n/2.
    """
    # Take the left half (including middle if odd)
    mid = n // 2
    left_half = init_h[:mid+1] if n % 2 == 1 else init_h[:mid]
    
    # Threshold at 0.5 to make binary
    b_left = (left_half >= 0.5).astype(int)
    
    # Adjust sum to exactly n/2
    current_sum = np.sum(b_left)
    target_sum = n / 2
    
    if current_sum == target_sum:
        b = np.concatenate([b_left, b_left[::-1][1:]]) if n % 2 == 1 else np.concatenate([b_left, b_left[::-1]])
        return b
    elif current_sum > target_sum:
        # Need to decrease some 1s to 0s
        diff = int(current_sum - target_sum)
        indices = np.argsort(b_left)  # Sorted by ascending
        # Turn diff ones just below the threshold to zeros 
        for i in range(diff):
            i_idx = indices[i] 
            if b_left[i_idx] == 1:
                b_left[i_idx] = 0
    else:
        # Need to increase some 0s to 1s
        diff = int(target_sum - current_sum)
        indices = np.argsort(b_left)[::-1]  # Sorted descending
        for i in range(diff):
            i_idx = indices[i]
            if b_left[i_idx] == 0:
                b_left[i_idx] = 1
    
    # Recompute
    current_sum = np.sum(b_left)
    # Ensure sum matches n/2
    if abs(current_sum - target_sum) > 1e-6:
        # Use a little bit more sophisticated adjustment to hit the target
        err = int(current_sum - target_sum)
        if err > 0:
            # Remove more 1s
            sorted_indices = np.argsort(b_left)
            for i in range(-err):
                if sorted_indices[i] < len(b_left):
                    b_left[sorted_indices[i]] = 0
        elif err < 0:
            # Add more 1s
            sorted_indices = np.argsort(b_left)[::-1]
            for i in range(+err):
                if sorted_indices[i] < len(b_left):
                    b_left[sorted_indices[i]] = 1
    
    # Construct full symmetric pattern
    if n % 2 == 1:
        b = np.concatenate([b_left, b_left[::-1][1:]])
    else:
        b = np.concatenate([b_left, b_left[::-1]])
    
    return b

def morse_function(x):
    """Morse function to handle oscillatory potential."""
    return np.arctan(x) + 0.5 * np.sin(4 * x)
