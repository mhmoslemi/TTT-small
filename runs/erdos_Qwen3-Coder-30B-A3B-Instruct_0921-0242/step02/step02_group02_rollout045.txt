import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar
import cvxpy as cp

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def build_cyclic_shift_matrix(k, n):
    """Build the cyclic shift matrix P_k such that (P_k * h)_i = h_{i-k mod n}"""
    P = np.zeros((n, n))
    for i in range(n):
        j = (i - k) % n
        P[i, j] = 1.0
    return P

def subgradient_refine(h, dx, remaining_time):
    """Perform subgradient descent refinement with time control."""
    start_t = time.time()
    while time.time() - start_t < remaining_time - 1e-3:
        # Compute full correlation
        corr = correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        n = len(h)
        if k_star > n:
            k = k_star - n
        else:
            k = k_star - n + 1
        
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n):
            left_idx = (i + k) % n
            right_idx = (i - k) % n
            grad[i] = dx * (h[left_idx] - h[right_idx])
        
        # Line search with backtracking
        alpha = 1.0
        while alpha > 1e-8:
            h_new = h - alpha * grad
            h_new = project_to_feasible(h_new, len(h))
            c5_new = evaluate_c5(h_new)
            if c5_new < evaluate_c5(h):
                h = h_new
                break
            alpha /= 2.0
        else:
            break  # no acceptable step found
        
        if time.time() - start_t > remaining_time - 1e-3:
            break
    
    return h, evaluate_c5(h)

def upsample_and_project(h_old, n_new):
    """Linearly interpolate and project to new resolution."""
    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / n_new
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    
    # Interpolate using linear interpolation
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    h_new = project_to_feasible(h_new, n_new)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Part 1: SDP relaxation for global approximation
    n = 180  # Increased from older version
    dx = 2.0 / n
    P = [build_cyclic_shift_matrix(k, n) for k in range(n)]

    # Construct SDP problem
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()

    constraints = [
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n/2,
    ] + [cp.trace(P[k] @ X) >= t for k in range(n)]

    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver='SCS', max_iters=25000, eps=1e-5)
    except Exception:
        # Fallback solver if SCS fails
        try:
            prob.solve(solver='ECOS', max_iters=25000, eps=1e-5)
        except Exception:
            # Fallback to a simple projection on a sample
            h_base = np.ones(n) * 0.5
            h_base = project_to_feasible(h_base, n)
            return h_base, evaluate_c5(h_base), n

    if prob.status != 'optimal':
        # Non-optimal solution
        print(f"Normal SDP solver failure {prob.status} with SCS and ECOS.")
        h_base = np.ones(n) * 0.5
        h_base = project_to_feasible(h_base, n)
        return h_base, evaluate_c5(h_base), n

    # Extract feasible vector via spectral factorization
    X_opt = X.value
    try:
        eigvals, eigvecs = np.linalg.eigh(X_opt)
        lead = eigvecs[:, -1] * np.sqrt(eigvals[-1])
        h0 = project_to_feasible(lead, n)
    except (np.linalg.LinAlgError, ValueError):
        # If eigen-decomposition fails, fallback to a simple projection
        h0 = np.ones(n) * 0.5
        h0 = project_to_feasible(h0, n)

    # Refine using random samples to get better candidates
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Sample from a few random vectors
    for _ in range(15):
        if time.time() - start_time > 0.7 * budget_s:
            break
        # Random sample from the distribution
        g = np.random.randn(n) @ np.linalg.cholesky(X_opt)
        h = project_to_feasible(np.clip(g, 0, 1), n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Part 2: Fortify with local optimization using gradient descent
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1.0:
        best_h, best_c5 = subgradient_refine(best_h, dx, remaining_time)

    # Part 3: Upsample to higher resolution if time permits
    if time.time() - start_time < 0.85 * budget_s:
        n2 = 2 * n
        try:
            h_up = upsample_and_project(best_h, n2)
            remaining_time = budget_s - (time.time() - start_time)
            if remaining_time > 1.0:
                best_h, best_c5 = subgradient_refine(h_up, 2.0 / n2, remaining_time)
        except Exception:
            # If upsampling fails, return earlier result
            pass

    return (best_h, best_c5, len(best_h))
