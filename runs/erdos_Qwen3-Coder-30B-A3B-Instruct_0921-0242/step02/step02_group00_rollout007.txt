import numpy as np
import time

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2 and h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Adjust by adding diff/n to each component
    h = h + diff / n
    
    # Re-clip
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def correlation_update_autocorr(h, γ, i, j, Δ):
    """
    Update the autocorrelation vector γ for a move:
    transfer Δ from i to j, in a circular array.
    This function computes the Δγ entrywise.
    """
    n = len(h)
    Δγ = np.zeros_like(γ)
    # Temporary copy of h
    ĥ = h.copy()
    ĥ[i] -= Δ
    
    for s in range(-(n - 1), n):
        idx = s + n - 1
        # Removal from i
        if 0 <= i + s < n:
            Δγ[idx] -= Δ * ĥ[i + s]
        if 0 <= i - s < n and s != 0:
            Δγ[idx] -= Δ * ĥ[i - s]
        # Addition to j
        if 0 <= j + s < n:
            Δγ[idx] += Δ * ĥ[j + s]
        if 0 <= j - s < n and s != 0:
            Δγ[idx] += Δ * ĥ[j - s]
    # Self-term (s=0): exactly compute the change to h[i]*h[i] and h[j]*h[j]
    # h[i] changes by -Δ and h[j] changes by +Δ
    # So it's (h[i]-Δ)^2 - h[i]^2 + (h[j]+Δ)^2 - h[j]^2 = -2*h[i]*Δ + Δ^2 + 2*h[j]*Δ + Δ^2
    # i.e., previous Δγ[n-1] = h[i]*h[i] + h[j]*h[j]
    # final Δγ[n-1] = (h[i]-Δ)^2 + (h[j]+Δ)^2 = h[i]^2 - 2*h[i]*Δ + Δ^2 + h[j]^2 + 2*h[j]*Δ + Δ^2
    # Therefore the moved Δγ[n-1] = -2*h[i]*Δ + Δ^2 + 2*h[j]*Δ + Δ^2 = -2*h[i]*Δ + 2*h[j]*Δ + 2*Δ^2
    Δγ[n - 1] += -2 * h[i] * Δ + 2 * h[j] * Δ + 2 * Δ * Δ
    return Δγ

def get_worst_shifts(C, C5, ε=1e-5):
    """Find indices of roughly worst overlaps."""
    return np.where(np.abs(C - C5) < ε)[0]

def get_scores(h, S_w, n):
    """Score for each index based on its contribution to worst shifts."""
    score = np.zeros(n)
    for s in S_w:
        # Check which indices are valid for contribution from both sides
        if s >= 0:  # forward shift
            score += np.concatenate([h[s:], np.zeros(s)])
        else:  # backward shift, shift=-s
            score += np.concatenate([np.zeros(-s), h[:-s]])
    return score

def apply_move(h, i, j, Δ):
    """Apply a move: transfer Δ from i to j."""
    h[i] -= Δ
    h[j] += Δ
    return h

def project_box_sum(y, n):
    """Project y onto the set {x : 0 <= x <= 1, sum(x) = n/2}."""
    # The function S(λ) = sum_i clip(y_i - λ, 0, 1)
    # This is monotonically decreasing, use binary search
    
    def S(λ):
        clipped = np.clip(y - λ, 0, 1)
        return np.sum(clipped)
    
    # We want S(λ) = n/2
    lower, upper = np.min(y) - 1.0, np.max(y) + 1.0
    # Iterate until calls are needed
    for _ in range(40):
        mid = (lower + upper) / 2.0
        s = S(mid)
        if abs(s - n/2) < 1e-10:
            break
        elif s > n/2:
            lower = mid
        else:
            upper = mid
    
    λ = (lower + upper) / 2.0
    x = np.clip(y - λ, 0, 1)
    return x

def thin_random_initial(n, n_samples=5):
    """Generate n_samples random half-sum feasible vectors."""
    samples = []
    for _ in range(n_samples):
        h = np.random.rand(n)
        h = h * (n / 2) / np.sum(h)
        h = project_to_feasible(h, n)
        samples.append(h)
    return samples

            # Backtracking line search
