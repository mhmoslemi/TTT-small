import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def project_to_sum_constraint(h, target_sum):
    """Project h to the feasible set with sum = target_sum and h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Add diff/n to each, clip, and repeat up to 2 times to ensure sum is exact
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5_fft(h):
    """Evaluate C5 using FFT correlation for speed."""
    n = len(h)
    dx = 2.0 / n
    # Compute full convolution using FFT: h * (1 - h)
    h_fft = rfft(h)
    ih_fft = rfft(1 - h)
    corr_fft = h_fft * np.conj(ih_fft)
    # Inverse FFT to get correlations for each shift
    corr = irfft(corr_fft, n=2*n)
    # Take real part and maximum over the positive shifts (shift 1 to n-1)
    max_c = np.max(corr[1:n]) * dx
    return max_c

def compute_gradient_worst_shift(h):
    """Compute the gradient of the worst shift correlation."""
    n = len(h)
    dx = 2.0 / n
    # Full FFT correlation to find worst shift
    h_fft = rfft(h)
    ih_fft = rfft(1 - h)
    corr_fft = h_fft * np.conj(ih_fft)
    corr = irfft(corr_fft, n=2*n)
    
    # Ignore shift 0
    max_shift = np.argmax(corr[1:n]) + 1
    # Compute gradient w.r.t. h for this max shift
    grad = np.zeros_like(h)
    for i in range(n):
        left_idx = (i + max_shift) % n
        right_idx = (i - max_shift) % n
        grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
    return grad, max_shift

def gradient_worst_shift_descent(h, max_iter=200, alpha0=0.1):
    """Perform gradient descent on worst shift."""
    dx = 2.0 / len(h)
    h = h.copy()
    for t in range(max_iter):
        grad, _ = compute_gradient_worst_shift(h)
        alpha = alpha0 / np.sqrt(t + 1)  # diminishing step size
        h_new = h - alpha * grad
        h_new = project_to_sum_constraint(h_new, len(h) / 2)
        h = h_new
    return h

def differential_evolution(hs, n, target_sum, F=0.6, CR=0.9, max_iter=4000):
    """Differential evolution with FFT-based evaluation."""
    n_pop = len(hs)
    best_h = None
    best_c5 = float('inf')
    
    for i in range(max_iter):
        if i % 200 == 0:
            # Periodic refinement of best individual
            pop_c5 = [evaluate_c5_fft(h) for h in hs]
            current_best_idx = np.argmin(pop_c5)
            h_best = hs[current_best_idx]
            h_refined = gradient_worst_shift_descent(h_best, max_iter=100)
            c5_refined = evaluate_c5_fft(h_refined)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
        
        # DE mutation and crossover
        new_hs = []
        for j in range(n_pop):
            idxs = np.random.choice(n_pop, 3, replace=False)
            a, b, c = hs[idxs[0]], hs[idxs[1]], hs[idxs[2]]
            donor = a + F * (b - c)
            donor = np.clip(donor, 0, 1)
            donor = project_to_sum_constraint(donor, target_sum)
            
            trial = np.zeros_like(donor)
            k = np.random.randint(0, n)
            for l in range(n):
                if np.random.rand() < CR or l == k:
                    trial[l] = donor[l]
                else:
                    trial[l] = hs[j][l]
            
            new_hs.append(trial)
        hs = new_hs
        
        # Update best
        pop_c5 = [evaluate_c5_fft(h) for h in hs]
        current_best_idx = np.argmin(pop_c5)
        if pop_c5[current_best_idx] < best_c5:
            best_c5 = pop_c5[current_best_idx]
            best_h = hs[current_best_idx].copy()
    
    return best_h, best_c5

def thermal_swap_policing(h, max_iter=5000, T0=1.0, decay=0.999):
    """Stochastic swap polishing using simulated annealing logic."""
    dx = 2.0 / len(h)
    h = h.copy()
    c5 = evaluate_c5_fft(h)
    T = T0
    for i in range(max_iter):
        if T < 1e-10:
            break
        i1, i2 = np.random.randint(0, len(h)), np.random.randint(0, len(h))
        if i1 == i2:
            continue
        delta = np.random.uniform(0, min(h[i1], 1 - h[i2], 0.01))
        
        h_new = h.copy()
        h_new[i1] -= delta
        h_new[i2] += delta
        h_new = project_to_sum_constraint(h_new, len(h) / 2)
        
        c5_new = evaluate_c5_fft(h_new)
        if c5_new < c5 or np.random.rand() < np.exp(-(c5_new - c5) / T):
            h = h_new
            c5 = c5_new
        
        T *= decay
    return h, c5

    np.random.seed(seed)

    # Generate population of random individuals
    pop_size = min(40, 2 * n_points // 10)
    hs = [h_start.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = project_to_sum_constraint(h, target_sum)
        hs.append(h)
    
    dec_time = time.time()
    h_best, c5_best = differential_evolution(hs, n_points, target_sum, max_iter=4000)
    
    # Local refinement
    if dec_time - start_time < 0.9 * budget_s:
        h_refined = gradient_worst_shift_descent(h_best, max_iter=200)
        c5_refined = evaluate_c5_fft(h_refined)
        if c5_refined < c5_best:
            h_best = h_refined
            c5_best = c5_refined
    
    # Final polishing
    if time.time() - start_time < 0.92 * budget_s:
        h_final, c5_final = thermal_swap_policing(h_best, max_iter=5000)
        if c5_final < c5_best:
            h_best = h_final
            c5_best = c5_final
    
    # Offer final resolution check if time allows
    if time.time() - start_time < 0.98 * budget_s and n_points < 800:
