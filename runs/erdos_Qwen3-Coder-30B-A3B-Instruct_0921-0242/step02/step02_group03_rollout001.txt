import numpy as np
import cvxpy as cp
import time

def full_to_sym(h_full):
    """Convert full symmetric vector to half + center."""
    n = len(h_full)
    if n % 2 == 0:
        return np.concatenate([h_full[:n//2], [h_full[n//2]]])
    else:
        return h_full[:(n+1)//2]

def sym_to_full(h_sym):
    """Reconstruct full symmetric vector from half + center."""
    n = 2 * (len(h_sym) - 1)
    h_full = np.zeros(n)
    h_full[:len(h_sym)-1] = h_sym[:-1]
    h_full[len(h_sym)-1:] = h_sym[:-1][::-1]
    h_full[len(h_sym)-1] = h_sym[-1]
    return h_full

def build_constraints(x, t, shift_set, n):
    """Construct all correlation constraints for the given shifts."""
    constraints = []
    dx = 2.0 / n
    for s in shift_set:
        # Compute correlation term for shift s
        # We use the full correlation at shift s
        # This corresponds to sum over i of x[i] * (1 - x[i+s])
        # In full vector: overlapping elements of x and 1 - x[s:]
        if s == 0:
            expr = dx * cp.sum(x * (1 - x))
        else:
            # Since x is symmetric, consider how to map the correlation
            # It's easier to build the matrix in full space efficiently
            # But for now we'll handle case-by-case
            
            # Build the product x_i * (1 - x_{i+s})
            # Indexes are from 0 to len(x) - 1
            expr = 0
            for k in range(len(x)):
                if k + s < len(x):
                    expr += dx * x[k] * (1 - x[k + s])
                else:
                    # Pad with 0 for out of bounds
                    pass  # take care below
            expr = cp.sum(expr)
            expr *= dx
        constraints.append(expr <= t)
    return constraints

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def solve_lp(shift_set, x0=None, n=400, use_symmetry=True):
    """Solves the LP for the given shift set."""
    if use_symmetry:
        num_vars = n // 2 + 1
        x = cp.Variable(num_vars)
        t = cp.Variable()
        
        # Symmetry constraint
        x_low = cp.Variable(num_vars)
        x_up = cp.Variable(num_vars)
        constraints = [x_low <= x, x <= x_up,
                       x_low >= 0, x_up <= 1]
        
        # Objective
        objective = cp.Minimize(t)
        
        # Compute constraint matrix
        # For each shift s, we build a constraint sum_{i} x[i] * (1 - x[i+s])
        # For symmetry, we have to be careful with indices
        # Create the full vector from x via symmetry
        # x_full is mapping x_sym to x_full via mirroring
        
        constraints = [t >= 0]
        dx = 2.0 / n
        
        for s in shift_set:
            # We add all correlations but to save time and avoid distribution issues,
            # we compute the full correlation for the half vector x
            # and reconstruct full x later for computation.
            
            # Just add direct constraint: construct correlation manually, use x
            if s == 0:
                px = cp.sum(x * (1 - x)) * dx
            else:
                expr = 0
                for i in range(len(x)):
                    j = i + s
                    if j >= len(x):
                        continue
                    expr += dx * x[i] * (1 - x[j])
                px = expr
            constraints.append(px <= t)
        
        # Integral constraint
        # sum_i x[i] = 1/2 since dx = 2/n, so sum_i x[i] * dx = 1/2 * 2/n * n = 1
        # But the integral is in terms of full vector
        if use_symmetry:
            # Full vector of length n = 2*(length x - 1) + (1 if odd)
            # sum x (full) = 2 * sum_{k=0}^{l-1} x[k] + x[l] = 1  (l = len(x) - 1)
            # x_full
            full_sum = cp.sum(x[:-1]) * 2 + x[-1]
            full_sum = full_sum * dx * 2
            constraints.append(full_sum == 1)
        else:
            # For non-symmetric case
            constraints.append(cp.sum(x) == n / 2)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                x_val = x.value
                t_val = t.value
                return x_val, t_val
            else:
                return None, None
        except Exception:
            return None, None
    else:
        num_vars = n
        x = cp.Variable(num_vars)
        t = cp.Variable()
        # Objective
        objective = cp.Minimize(t)
        # Constraints are created in a loop
        constraints = []
        dx = 2.0 / n
        for s in shift_set:
            # For simplicity, we make this more precise:
            expr = 0
            for i in range(n):
                j = i + s
                if j >= n:
                    # zero padding
                    break
                expr += dx * x[i] * (1 - x[j])
            constraints.append(expr <= t)
        
        # Integral constraint
        constraints.append(cp.sum(x) == n / 2)
        constraints.append(x >= 0)
        constraints.append(x <= 1)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                x_val = x.value
                t_val = t.value
                return x_val, t_val
            else:
                return None, None
        except Exception:
            return None, None
