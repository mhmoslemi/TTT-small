import numpy as np
import time
from scipy.fft import fft, ifft

def evaluate_erdos_solution(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def compute_autocorrelations_binary(x):
    """Compute all periodic autocorrelations for a binary sequence."""
    n = len(x)
    # For efficiency, we compute as circular correlations,
    # without using fft since we're primarily targeting small n
    R = np.zeros(n)
    x = np.array(x)
    for k in range(n):
        R[k] = np.sum(x * np.roll(x, k))
    return R

def greedy_swap_refinement(x, max_iter=5000):
    """Apply local swap-based improvement."""
    n = len(x)
    # Precompute all autocorrelations (same procedure as in original optimize)
    # but resampling many terms swapped for efficiency
    
    # Use the full correlation, with only integer values.
    # Inplace improve minimum correlation
    R = np.zeros(n)
    # We avoid recomputing full correlations each time,
    # and rather update it incrementally on swaps.
    # For practicality we'll recompute at intervals
    
    iters = 0
    while iters < max_iter:
        # compute a fast FFT corr only if needed
        R = compute_autocorrelations_binary(x)
        R = R[1:]  # exclude index 0 (full match)
        k_star = np.argmin(R) + 1
        
        if R[k_star - 1] == 0:
            break  # nothing to do
            
        # find best i to reduce autok and best j to increase it
        # model: gamma_i = x_{i+k_star} + x_{i-k_star}
        # for i where x_i = 1
        # and, it's easy to compute in O(n)
        xarr = np.array(x)
        contrib = np.zeros(n)
        contrib2 = np.zeros(n)
        # we're effectively computing:
        # gamma[i] = x[i+k_star] + x[i-k_star]
        for i in range(n):
            contrib[i] = xarr[(i+k_star)%n] + xarr[(i-k_star)%n]
        # for j where x_j = 0
        for j in range(n):
            contrib2[j] = xarr[(j+k_star)%n] + xarr[(j-k_star)%n]
        
        improve = np.argmax(contrib)
        degrade = np.argmin(contrib2)
        
        # Apply swap improvement if beneficial
        if contrib[improve] < contrib2[degrade]:
            # swapping means: send x[improve] to 0, x[degrade] to 1
            x[improve] = 0
            x[degrade] = 1
            iters += 1
        else:
            break  # no more improvements in current cycle
            
    return x

def rounding_and_evaluation(X, t_value, n, max_rounds=30):
    """Round the symmetric matrix to binary vectors and evaluate."""
    V, Lambda = np.linalg.eigh(X)
    # take first eigenvector
    v = np.sqrt(Lambda[-1]) * V[:, -1]

    best_x = None
    best_t = -1.0
    x_threshold = (np.sort(v)[::-1])[:n//2]

    # deterministic truncate (thresholding)
    x = np.zeros(n)
    x[v >= np.sort(v)[::-1][n//2]] = 1
    # Flip a bit if not exactly middle
    if np.sum(x) != n/2:
        idx = np.argsort(-v)[:n//2]
        # ensure we have exactly n/2
        x = np.zeros(n)
        x[idx] = 1
                
    # evaluate
    R = compute_autocorrelations_binary(x)[1:]
    t_real = np.min(R)
    if t_real > best_t:
        best_t = t_real
        best_x = x.copy()

    # randomialize with hyperplane rounding
    for r in range(max_rounds):
        g = np.random.randn(n)
        z = V @ (np.sqrt(Lambda) * g)
        p = np.argsort(-z)
        x = np.zeros(n)
        x[p[:n//2]] = 1
        # evaluate
        R = compute_autocorrelations_binary(x)[1:]
        t_real = np.min(R)
        if t_real > best_t:
            best_t = t_real
            best_x = x.copy()
    
    return best_x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: SDP relaxation
    n = 200
    dx = 2.0 / n
    t_max = 0.0
    h_final = None
    c5_final = 1.0
    
    # Convert to the computational model of binary vectors to approximate
    # We treat this as a max-min binary quadratic program.
    
    # SDP (maximum t such that all R_k >= t)
    # Variables
    X = Semidef(n)
    t = Variable()
    
    # Constraints
    constraints = []
    # diagonal bounds
    for i in range(n):
        constraints.append(X[i,i] >= 0)
        constraints.append(X[i,i] <= 1)
    # cardinality
    constraints.append(sum([X[i,i] for i in range(n)]) == n/2)
    
    # Now we add the constraints of the form:
    # sum_{i=0}^{n-1} X[i,i+k] >= t
    # where indices are considered mod n, and k from 1 to n-1.
    
    # A_k matrix (for shift k)
    for k in range(1, n):
        # Compute the sum of diagonal elements of X along the kth diagonal
        constraint = 0
        for i in range(n):
            j = (i + k) % n
            constraint += X[i,j]
        constraints.append(constraint >= t)
    
    # Enforce pairwise bounds (connected to strong relaxation), for stability
    for i in range(n):
        for j in range(i+1, n):  # j > i
            constraints.append(X[i,j] >= 0)
            constraints.append(X[i,j] <= X[i,i])
            constraints.append(X[i,j] <= X[j,j])
    
    # Problem
    prob = Problem(Maximize(t), constraints)
    
    # Time allocation for SDP: 0.3 of budget
    remaining_budget = budget_s * 0.3
    try:
        prob.solve(solver=ECOS, verbose=False, eps=1e-6, max_iters=2000)
        # Check that the solver succeeded
        if prob.status == 'optimal':
            t_opt = t.value
            X_opt = X.value
        else:
            raise ValueError("SDP did not solve to optimality")
    except Exception as e:
        # fallback to pure random search
        print("SDP failed with error:", str(e))
        c5_final = 0.5
        h_final = np.random.rand(n)
        h_final = project_to_feasible(h_final, n)
        return (h_final, c5_final, n)

    
    # Stage 2: Rounding and evaluation
    if X_opt is not None:
        x = rounding_and_evaluation(X_opt, t_opt, n)
        x = np.array(x).astype(int)
        R = compute_autocorrelations_binary(x)
        min_t = np.min(R[1:])
        c5_current = 1 - 2*min_t / n
        
        # refine using greedy swaps
        x_swap = x.copy()
        try:
            x_swap = greedy_swap_refinement(x_swap, max_iter=1000)
        except Exception:
            pass
        
        # Final evaluation
        R_final = compute_autocorrelations_binary(x_swap)[1:]
        min_t_final = np.min(R_final)
        c5_final = 1 - 2*min_t_final / n
        h_final = x_swap.astype(float)
    
    # Stage 3: If within time limits, do further polishing
    # clarify: Hoe to use evaluate_erdos_solution here in floating representation?
    if h_final is not None and time.time() - start_time < budget_s * 0.99:
        # Perform small continuous fine-tuning if time allows
        h = h_final.copy()
        dx = 2.0 / n
        
        # estimate k_star as before from h
        R = compute_autocorrelations_binary(h.astype(int))
        k_star = np.argmin(R[1:]) + 1
        
        # Grad decent / local improvement
        for i in range(50):  # small number of iterations
            if time.time() - start_time > budget_s * 0.99:
                break
                
            # compute gradient for direction of k_star
            g = np.zeros(n)
            for i in range(n):
                # gradient direction of current k_star
                k = k_star
                g[i] = dx * ( (1 - h[(i+k) % n]) - h[(i-k) % n] )
            
            # Update via small step
            alpha = 0.05 * dx
            h_new = h - alpha * g
            
            # Project to feasible
            h_new = project_to_feasible(h_new, n)
            
            # evaluate new C5 (in case it's better)
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < c5_final:
                c5_final = c5_new
                h = h_new.copy()
                # recompute k_star
                R = compute_autocorrelations_binary(h.astype(int))
                k_star = np.argmin(R[1:]) + 1
        
        h_final = h
    
    if h_final is None:
        # fallback
        h_final = np.ones(n) * 0.5
        c5_final = 0.5
    
    return (h_final, c5_final, n)
