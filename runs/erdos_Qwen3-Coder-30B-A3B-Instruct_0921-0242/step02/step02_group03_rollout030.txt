import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def update_correlations(h, i, j, c_old):
    """Update correlations when swapping h[i] and h[j] in a binary vector."""
    n = len(h)
    c_new = c_old.copy()
    for s in range(n):
        if i + s < n:
            if h[i] == 1 and h[i+s] == 0:
                c_new[s] -= 1
            elif h[i] == 0 and h[i+s] == 1:
                c_new[s] += 1
        if j + s < n:
            if h[j] == 1 and h[j+s] == 0:
                c_new[s] -= 1
            elif h[j] == 0 and h[j+s] == 1:
                c_new[s] += 1
    return c_new

def anneal_binary(h0, n, time_limit):
    """Simulated Annealing for binary balanced sequences."""
    # Convert to binary (int8) and initialize correlation array
    h = h0.astype(np.int8).copy()
    # Precompute initial correlations for full range
    dx = 2.0 / n
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    c_old = ifft(corr_freq).real[:n]
    c_old = c_old * dx  # Scale by dx

    max_old = np.max(c_old)
    best_h = h.copy()
    best_c = max_old
    max_steps = 60000
    T0 = 0.02 * n
    alpha = 0.997
    T = T0
    step = 0

    while T > 1e-5 and step < max_steps:
        if time.time() > time_limit:
            break
        step += 1
        # Pick two random positions to swap
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        while i == j:
            j = np.random.randint(0, n)
        # Skip if indices are same
        if i == j:
            continue

        # Update correlation
        c_new = update_correlations(h, i, j, c_old)
        max_new = np.max(c_new)

        # Metropolis acceptance criterion
        delta = max_new - max_old
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            # Accept the move
            h[i], h[j] = h[j], h[i]
            c_old = c_new
            max_old = max_new
            if max_new < best_c:
                best_c = max_new
                best_h = h.copy()

        T = T * alpha

    return best_h, best_c

def polish_continuous(h_binary, n):
    """Polish the binary solution with continuous refinement."""
    dx = 2.0 / n
    h = h_binary.astype(float)
    # Smooth max surrogate
    lambda_val = 5.0

    def smooth_max(h_array, lambda_param):
        # Compute correlations
        corr = correlate(h_array, 1 - h_array, mode='full')
        corr = corr[:n]  # Just the non-cyclic part
        corr_scaled = corr * dx  # Scale by dx
        # Smooth max: 1/λ * log(sum(exp(λ * corr)))
        exp_corr = np.exp(lambda_param * corr_scaled)
        return np.log(np.sum(exp_corr)) / lambda_param

    def gradient(h_array, lambda_param):
        # For each position, compute gradient of the smooth max
        corr = correlate(h_array, 1 - h_array, mode='full')
        corr = corr[:n]  # Non-cyclic
        corr_scaled = corr * dx
        exp_corr = np.exp(lambda_param * corr_scaled)
        sum_exp = np.sum(exp_corr)
        inv_sum = 1.0 / sum_exp
        # The gradient for contribution to each correlation
        d = np.zeros_like(h_array)
        for i in range(n):
            grad_term = 0.0
            for s in range(n):
                if i+s < n:
                    # Contribution from h[i]
                    if s == 0:
                        grad_term += (1 - h_array[i+s]) * dx
                    else:
                        grad_term += (1 - h_array[i+s]) * dx
                        # And from h[i-s] if exists and i-s >= 0
                        if i >= s:
                            grad_term -= h_array[i-s] * dx
            d[i] = grad_term
        return d * inv_sum * lambda_param

    eta = 0.1 * dx
    max_polish_steps = 30
    for i_step in range(max_polish_steps):
        if lambda_val < 30:
            lambda_val += 5.0
        # Compute smooth max and gradient
        val = smooth_max(h, lambda_val)
        grad = gradient(h, lambda_val)
        h_new = h - eta * grad
        # Project h_new back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_feasible(h_new, n)
        h = h_new
        eta *= 0.95  # diminish stepsize

    return h, evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 10  # Leave some margin for cleanup

    # Choose n such that np.ceil(n/2) is not too large, and n is odd
    n = 221
    dx = 2.0 / n
    h_best = None
    c5_best = float('inf')

    restarts = 10
    for trial in range(restarts):
        if time.time() > time_limit:
            break
        # Either block of 1s or random binary construction
        if np.random.rand() < 0.5:
            # Block construction
            h = np.zeros(n, dtype=np.int8)
            start = np.random.randint(0, n)
            length = n // 2
            end = min(start + length, n)
            if start + length > n:
                # Ensure we don't overflow
                h[start:] = 1
                h[:length - (n - start)] = 1
            else:
                h[start:end] = 1
        else:
            # Random construction
            h = np.zeros(n, dtype=np.int8)
            ones_indices = np.random.choice(n, n//2, replace=False)
            h[ones_indices] = 1

        # Anneal
        h_final, c5 = anneal_binary(h, n, time_limit)
        # Polish
        h_polished, c5_polished = polish_continuous(h_final, n)
        if c5_polished < c5_best:
            c5_best = c5_polished
            h_best = h_polished.copy()

    if h_best is None:
        h_best = np.ones(n) * 0.5
        c5_best = evaluate_c5(h_best)

    return (h_best, c5_best, n)
