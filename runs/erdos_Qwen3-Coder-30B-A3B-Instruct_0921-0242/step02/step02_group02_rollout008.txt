import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2 with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n / 2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n / 2:
            ub = mid
        else:
            lb = mid

    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Evaluates C5 for the given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def evaluate_autocorr(h):
    """Compute the autocorrelation vector."""
    n = len(h)
    H = fft(h, n)
    autocorr = np.real_if_close(ifft(H * np.conj(H), n))
    return autocorr

def compute_gradient(h, k):
    """Compute subgradient of C5 at shift k."""
    dx = 2.0 / len(h)
    n = len(h)
    grad = np.zeros_like(h)
    # +h_{i+k} term
    grad += np.roll(h, k)
    # -h_{i-k} term
    grad += np.roll(h, -k)
    return -dx * grad

def evaluate_c5_smooth(h, gamma=10.0):
    """Smooth approximation of C5 using log-sum-exp."""
    dx = 2.0 / len(h)
    n = len(h)
    H = fft(h, n)
    autocorr = np.real_if_close(ifft(H * np.conj(H), n))
    c = dx * (n/2 - autocorr)
    return (1.0 / gamma) * np.log(np.sum(np.exp(gamma * c)))

def binary_anneal(h_bin, time_limit):
    """Perform binary simulated annealing to maximize minimal autocorrelation."""
    start_time = time.time()
    n = len(h_bin)
    dx = 2.0 / n
    T0 = 5 * dx
    alpha = 0.9997
    T = T0
    best_h = h_bin.copy()
    ac = evaluate_autocorr(h_bin)
    c = dx * (n/2 - ac)
    best_c5 = np.max(c)
    num_swaps = 0
    stagnation_counter = 0
    best_ac = ac.copy()
    best_c = c.copy()

    # Precompute all affected shifts
    self_terms = np.zeros(n)
    self_terms[0] = 2.0
    self_terms[1] = -1.0
    self_terms[-1] = -1.0

    while time.time() - start_time < time_limit:
        # Propose swap: pick a 1 and 0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_new = h_bin.copy()
        h_new[i] = 0
        h_new[j] = 1

        # Compute delta to autocorrelation
        delta = np.zeros(n)
        delta = -h_new + h_bin
        ac_new = ac + delta + self_terms

        c_new = dx * (n/2 - ac_new)
        c5_new = np.max(c_new)
        delta_c5 = c5_new - best_c5

        if (c5_new < best_c5 or np.random.rand() < np.exp(-delta_c5 / T)):
            h_bin = h_new
            ac = ac_new
            c = c_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_bin.copy()
                best_ac = ac.copy()
                best_c = c.copy()
                stagnation_counter = 0

        num_swaps += 1
        stagnation_counter += 1

        if stagnation_counter > 200000:
            # Large jump: reinitialise 5% of ones
            ones = np.where(h_bin == 1)[0]
            num_to_change = int(0.05 * len(ones))
            to_change = np.random.choice(ones, size=num_to_change, replace=False)
            for i in to_change:
                h_bin[i] = 0
            # Pick new zeros to turn to ones
            zeros = np.where(h_bin == 0)[0]
            to_fill = np.random.choice(zeros, size=num_to_change, replace=False)
            for i in to_fill:
                h_bin[i] = 1
            # Recompute
            ac = evaluate_autocorr(h_bin)
            c = dx * (n/2 - ac)
            c5_new = np.max(c)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_bin.copy()
                best_ac = ac.copy()
                best_c = c.copy()
            stagnation_counter = 0

        if num_swaps % 1000 == 0:
            T *= alpha

    return best_h, best_c5

def smooth_boundaries(h_bin, ramp_width=2):
    """Convert binary h to continuous step function with ramps."""
    n = len(h_bin)
    h = h_bin.astype(float)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    # Find transitions
    transitions = []
    for i in range(n):
        # Check if 0->1
        if h_bin[i] == 0 and h_bin[(i+1)%n] == 1:
            transitions.append((i, '01'))
        # Check if 1->0
        if h_bin[i] == 1 and h_bin[(i+1)%n] == 0:
            transitions.append((i, '10'))

    # Place ramps
    for bound, typ in transitions:
        if typ == '01':
            if ramp_width == 1:
                h[(bound+1)%n] = 0.5
            elif ramp_width == 2:
                h[(bound+1)%n] = 1.0/3
                h[(bound+2)%n] = 2.0/3
        else:  # typ == '10'
            if ramp_width == 1:
                h[(bound+1)%n] = 0.5
            elif ramp_width == 2:
                h[(bound+1)%n] = 2.0/3
                h[(bound+2)%n] = 1.0/3

    # Re-project to integral constraint
    h = project_to_feasible(h, n)
    return h

def proj_grad_refine(h, time_limit):
    """Project gradient descent (refine continuous solution)."""
    n = len(h)
    dx = 2.0 / n
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Gradient descent
    max_steps = 5000
    step_count = 0
    for i in range(max_steps):
        if time.time() - start_time > time_limit:
            break
            
        # Determine k* - worst shift
        corr = correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        if k_star > len(h):
            k_star = k_star - len(h)
        else:
            k_star = k_star - len(h) + 1

        # Compute sub-gradient
        grad = compute_gradient(h, k_star)

        # Backtracking line search
        eta = 0.5
        for _ in range(10):  # up to 10 tries
            h_new = h - eta * grad
            h_new = project_to_feasible(h_new, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5 or np.random.rand() < 0.01:  # accept with some randomness
                h = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            eta *= 0.5

        step_count += 1

    return best_h, best_c5

def upsample_and_refine(h, time_limit):
    """Upsample to double resolution and refine."""
    start_time = time.time()
    n = len(h)
    n_up = 2 * n
    dx_up = 2.0 / n_up

    # Interpolation
    h_up = np.interp(np.linspace(0, 2, n_up), np.linspace(0, 2, n), h)
    # Re-project
    h_up = project_to_feasible(h_up, n_up)
    
    # Short gradient polish
    h_up, c5 = proj_grad_refine(h_up, time_limit)
    return h_up, c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n0 = 256
    dx0 = 2.0 / n0

    # Stage 1: Binary optimization
    h_bin = np.zeros(n0)
    half_n = n0 // 2
    h_bin[:half_n] = 1
    np.random.shuffle(h_bin)
    h_bin, c5_bin = binary_anneal(h_bin, 0.55 * budget_s)
    
    # Stage 2: Smoothing & gradient refinement
    h_smooth = smooth_boundaries(h_bin, ramp_width=2)
    h_refined, c5_refined = proj_grad_refine(h_smooth, 0.25 * budget_s)

    # Stage 3: Upsampling
    h_up, c5_up = upsample_and_refine(h_refined, 0.15 * budget_s)

    # Return best of all
    if c5_up < c5_refined:
        final_h = h_up
        final_c5 = c5_up
    else:
        final_h = h_refined
        final_c5 = c5_refined

    return (final_h, final_c5, len(final_h))
