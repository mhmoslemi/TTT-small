def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum

    if abs(diff) < 1e-10:
        return h

    # Water-filling approach for adjusting sum
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)

    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid

    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)

    return h

def compute_autocorrelation_profile(h):
    """Compute the (circulant) auto-correlation profile of h."""
    n = len(h)
    dx = 2.0 / n
    acorr = correlate(h, h, mode='full')
    acorr = acorr[n-1:2*n-1]
    return acorr * dx

def log_sum_exp_surrogate(h, beta=5.0):
    """Smooth approximation to the maximum overlap."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    corr = corr[n-1:2*n-1]
    corr = corr * dx
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * corr)))

def compute_gradient_of_surrogate(h, beta=5.0):
    """Compute the gradient of the log-sum-exp surrogate."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    corr = corr[n-1:2*n-1] 
    corr = corr * dx
    exp_corr = np.exp(beta * corr)
    Z = np.sum(exp_corr)
    weights = exp_corr / Z
    # Derivative is weighted combination of individual gradients
    grad = np.zeros_like(h)
    for k in range(n):
        # Backward shift of h(k) becomes h(k - k)
        lag_k = np.roll(h, k)
        # Forward shift h(k - k) becomes h(k) 
        lead_k = np.roll(h, -k)
        grad_k = dx * (lead_k - lag_k)
        grad += weights[k] * grad_k
    return grad

def projected_lbfgsb_refinement(h0, max_iter=2000, beta_start=5.0):
    """L-BFGS-B with projection, using the LSE surrogate."""
    n = len(h0)
    h = h0.copy()
    dx = 2.0 / n

    # Initial beta
    beta = beta_start

    # L-BFGS-B optimization
    for i in range(max_iter):
        if i > 0 and i % 200 == 0:
            beta *= 2  # Sharpen approximation

        # Exponential log-sum-exp surrogate
        phi_current = log_sum_exp_surrogate(h, beta)
        grad = compute_gradient_of_surrogate(h, beta)

        # Simple projected BFGS step (manually coded here)
        h_new = h - 0.005 * grad
        # Project back
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n / 2) / np.sum(h_new)
        h_new = project_to_feasible(h_new, n)

        # Accept or reject based on objective improvement
        phi_new = log_sum_exp_surrogate(h_new, beta)
        if phi_new < phi_current:
            h = h_new.copy()
        else:
            # No improvement; decrease step size
            pass

        if np.linalg.norm(h - h_new) < 1e-7:
            break

    # Final polishing with exact gradient under worst-case shift
    t0 = time.time()
    while time.time() - t0 < 15 and i < max_iter:
        # Locate worst shift k* 
        corr = correlate(h, 1 - h, mode='full') * dx
        k_star = np.argmax(corr)
        # Extract gradient for shift k* (indexed from 0..2n-2)
        k_shift = k_star - (n - 1) # Conv encoding shifts
        if k_shift > 0:
            grad = dx * (np.roll(h, -k_shift) - np.roll(h, k_shift))
        else:
            grad = dx * (np.roll(h, k_shift) - np.roll(h, -k_shift))
        # Perform projected subgradient step
        eta = corr[k_star] / np.sum(grad**2) if np.sum(grad**2) > 1e-15 else 1e-5
        h_new = h - eta * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n / 2) / np.sum(h_new)
        h_new = project_to_feasible(h_new, n)
        h = h_new.copy()
        i += 1

    return h

def solve_binary_mip_with_cuts(h_init, target_autocorr=None, n=None):
    """Use cvxpy to solve a binary MIP; builds initial binary pattern."""
    if n is None:
        n = len(h_init)
    dx = 2.0 / n

    # Create a binary vector
    b = cp.Variable(n, boolean=True)
    gamma = cp.Variable()

    # Constraint: sum is n/2
    constraints = [cp.sum(b) == n/2]

    # Add autocorrelation constraints (simplified here)
    # We'll add a subset of critical constraints based on SDP duals
    for k in range(n):
        s_k = cp.sum(cp.multiply(b, cp.roll(b, k)))
        constraint = dx * (n/2 - s_k) <= gamma
        constraints.append(constraint)

    # Objective: minimize gamma
    objective = cp.Minimize(gamma)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=CP.SCS, verbose=False, max_iters=3000)
        b_sol = b.value
        gamma_sol = gamma.value
        if b_sol is not None:
            b_sol = np.round(b_sol).astype(int)
            return b_sol / (n/2), gamma_sol
    except Exception as e:
        pass

    return None, None

def apply_ramp_smoothing(b, w=4): 
    """Convert binary pattern to continuous with ramps."""
    n = len(b)
    h = np.zeros_like(b, dtype=float)
    h[0] = b[0]
    
    # Smooth step transitions
    for i in range(n):
        if b[i] != b[(i+1) % n]:
            h1 = i
            h2 = (i+1) % n
            # Ramp length: min(w, (h2 - h1) % n)
            length = min(w, (h2 - h1) % n, (h1 - h2) % n) 
            for j in range(length):
                ratio = float(j+1) / (length + 1)
                idx1 = (i + j + 1) % n
                idx2 = (i - j - 1) % n
                if b[i] < b[(i+1) % n]: # 0 -> 1 transition
                    h[idx1] += ratio * (b[(i+1) % n] - b[i])
                if b[i] > b[(i+1) % n]:  # 1 -> 0 transition
                    h[idx2] += ratio * (b[i] - b[(i+1) % n])
    return h
