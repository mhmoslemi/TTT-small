import numpy as np
import time
import cvxpy as cp
from scipy.signal import fftconvolve
from scipy.ndimage import gaussian_filter1d

def aperiodic_autocorr(h):
    """Compute aperiodic autocorrelation of h using FFT."""
    n = len(h)
    H = np.fft.fft(h, 2*n-1)
    acorr = np.fft.ifft(H * np.conj(H)).real[:n]
    return acorr

def softmax(beta, x):
    """Numerically stable softmax with temperature beta."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(beta * x_shifted)
    return exp_x / np.sum(exp_x)

def conv(a, b):
    """Linear convolution using FFT."""
    n = len(a)
    m = len(b)
    result_len = n + m - 1
    A = np.fft.fft(a, result_len)
    B = np.fft.fft(b, result_len)
    return np.fft.ifft(A * B).real[:n]

def project_box_sum(h, total):
    """Project h onto box and sum constraints using water-filling."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    
    if abs(current_sum - total) < 1e-10:
        return h
    
    # Water-filling to adjust to exact sum
    h_sorted = np.sort(h)[::-1]
    n = len(h)
    
    # Find the threshold
    cumsum = np.cumsum(h_sorted)
    # We want sum of clipped values to be total
    # This is like finding lambda s.t. sum(max(h[i] - lambda, 0)) = total
    # We compute sums = sum(max(h_sorted[i] - lambda, 0)) for different lambdas
    
    if current_sum <= total:
        # Need to add more
        diff = total - current_sum
        h = h + diff / n
        h = np.clip(h, 0, 1)
        return h
    
    # Need to reduce; find lambda
    lambda_min = 0.0
    lambda_max = h_sorted[0]
    eps = 1e-12
    
    # Binary search for lambda
    while lambda_max - lambda_min > eps:
        lambda_mid = (lambda_min + lambda_max) / 2.0
        # Sum of max(h_sorted[i] - lambda_mid, 0)
        pos_sum = np.sum(np.maximum(h_sorted - lambda_mid, 0))
        if pos_sum > total:
            lambda_min = lambda_mid
        else:
            lambda_max = lambda_mid
    
    lambda_opt = (lambda_min + lambda_max) / 2.0
    
    # Apply lambda
    h_final = np.maximum(h - lambda_opt, 0)
    # Rescale to have proper mass
    current_final_sum = np.sum(h_final)
    h_final = h_final * total / current_final_sum
    return h_final  # Should be good now

def evaluate_c5(h):
    """Simple C5 evaluation."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def score_guided_swap(h, k=20):
    """Do a batch of swaps guided by scores."""
    # For an increasingly high-gain operation.
    n = len(h)
    # Focus on worse shifts
    gamma = aperiodic_autocorr(h)
    C = 1 - (2/n) * gamma
    worst_idx = np.argmax(C)
    worst_C = C[worst_idx]
    
    # Score targets
    scores = np.zeros(n)
    # Consider shifts near worst, but not too much (too wide will cause thinning)
    W = np.abs(C - worst_C) < 1e-5
    worse_shifts = np.where(W)[0]
    
    for i in range(n):
        # Compute weighted sum over relevant parts
        #这个地方是精细相关的
        # 尽量不尝试太远的移位, 直接在搜索空间缩减为 [0, n//2] 
        total_S = 0
        for s in worse_shifts:
            if s == 0: continue  # skip the zero shift
            if i + s < n:
                total_S += h[i+s]
            if i - s >= 0:
                total_S += h[i-s]
        scores[i] = total_S
    
    # Pick best to worst
    sorted_indices = np.argsort(-scores)
    
    # Try 20 swaps
    h_new = h.copy()
    for _ in range(k):
        # pick a random source with high score
        source_idx = sorted_indices[np.random.randint(0, min(20, len(sorted_indices)))]
        # pick a random target with low score
        target_idx = np.random.choice(np.arange(n), p=np.exp(-scores)/np.sum(np.exp(-scores)))
        if abs(h_new[source_idx] - h_new[target_idx]) < 1e-7: continue
        
        # Small delta
        delta = min(0.01, h_new[source_idx], 1 - h_new[target_idx])
        if delta < 1e-6: continue  # too small to move
        
        # Move
        h_new[source_idx] -= delta
        h_new[target_idx] += delta
        # Project
        h_new = project_box_sum(h_new, total=len(h)/2)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Phase I: Solve MILP for binary solution
    n1 = 120
    half = n1 // 2

    # Define variables
    b = cp.Variable(n1, boolean=True)
    a = cp.Variable((n1, n1), boolean=True)
    L = cp.Variable()
    
    # 1. Mass constraint
    constraints = [cp.sum(b) == half]
    
    # 2. Linearization: a[i,s] = b[i] * b[i+s] (i+s < n1)
    # Only add relevant terms not out of bounds
    for i in range(n1):
        for s in range(n1 - i):
            # McCormick envelopes
            constraints.append(a[i, s] <= b[i])
            constraints.append(a[i, s] <= b[i+s])
            constraints.append(a[i, s] >= b[i] + b[i+s] - 1)
    
    # 3. Define γ_s from a_{i,s}
    gamma = cp.Variable(n1)
    for s in range(n1):
        constraints.append(gamma[s] == cp.sum(a[:, s]))
    
    # 4. Uniform lower bound
    for s in range(n1):
        # gamma_s >= L is equivalent to gamma_s - L >= 0, i.e. L <= gamma_s
        constraints.append(L <= gamma[s])
    
    # 5. Objective: maximize L
    objective = cp.Maximize(L)
    
    prob = cp.Problem(objective, constraints)
    
    # Solve using CBC with timeout
    prob.solve(solver=cp.CBC, verbose=False, max_seconds=0.2 * budget_s, 
               rel_gap=5e-3)
    
    # Get the best feasible binary pattern
    b_star = np.round(b.value).astype(int)
    
    # Phase II: Upscaling
    n2 = 300
    x1 = np.linspace(0, 2, n1, endpoint=False)
    x2 = np.linspace(0, 2, n2, endpoint=False)
    h = np.interp(x2, x1, b_star)
    
    # Correct mass
    h = project_box_sum(h, total=n2 / 2)
    
    # Optional smoothing step
    h = gaussian_filter1d(h, sigma=1.0)
    h = project_box_sum(h, total=n2 / 2)
    
    # Phase III: Smooth-max projected gradient descent
    beta = 2.0
    eta = 1.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    iteration = 0
    step = 1
    
    def evaluate_c5_smooth(h, beta=100):
        gamma = aperiodic_autocorr(h)
        C = 1 - (2/n2) * gamma
        return 1.0 / beta * np.log(np.sum(np.exp(beta * C)))
    
    nesterov_grad = np.zeros_like(h)
    momentum_factor = 0.0
    while time.time() - start_time < 0.85 * budget_s:
        gamma = aperiodic_autocorr(h)
        C = 1 - (2/n2) * gamma
        w = softmax(beta, C)
        
        # Gradient of smooth surrogate F_beta(h)
        w_rev = w[::-1]
        conv_w_h = conv(w, h)
        conv_w_rev_h = conv(w_rev, h)
        grad = -(2.0/n2) * (conv_w_h + conv_w_rev_h)
        
        # Backtracking line search
        alpha = eta
        h_trial = project_box_sum(h - alpha * grad, total=n2/2)
        c5_trial = evaluate_c5(h_trial)
        
        while c5_trial >= evaluate_c5(h) and alpha > 1e-8:
            alpha *= 0.5
            h_trial = project_box_sum(h - alpha * grad, total=n2/2)
            c5_trial = evaluate_c5(h_trial)
        
        h = h_trial
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        
        # Schedule
        if iteration % 150 == 0:
            beta = min(beta * 1.5, 200.0)
        
        # Occasionally swap
        if iteration % 200 == 0:
            h = score_guided_swap(h)
        
        iteration += 1
        
        # Nesterov momentum step (adaptive)
        new_grad = grad
        momentum_factor = (iteration - 1) / (iteration + 2)
        nesterov_grad = momentum_factor * (new_grad - nesterov_grad) + new_grad
        h = project_box_sum(h - eta * nesterov_grad, total=n2/2)

    # Phase IV: Final sub-gradient polishing
    while time.time() - start_time < budget_s - 0.1 * budget_s:
        h_copy = h.copy()
        gamma = aperiodic_autocorr(h)
        C = 1 - (2/n2) * gamma
        worst_shifts = np.where(np.abs(C - np.max(C)) < 1e-5)[0]
        
        # Subgradient
        g = np.zeros_like(h)
        for s in worst_shifts:
            if s != 0:
                g += np.roll(h, s)  # h[i+s]
                g += np.roll(h, -s) # h[i-s]
        
        g = -(2.0 / n2) * g / len(worst_shifts)
        
        # Step size
        eta = 0.1 / np.sqrt(step)
        h_new = project_box_sum(h - eta * g, total=n2/2)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new.copy()
        h = h_new
        step += 1
    
    return (best_h, best_c5, n2)
