def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def build_cyclic_shift_matrix(k, n):
    """Build a sparse cyclic shift matrix P_k."""
    P_k = np.zeros((n, n))
    for i in range(n):
        j = (i - k) % n
        P_k[i, j] = 1
    return P_k

def upsample_and_project(h, n_new):
    """Upsample using linear interpolation and project."""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    
    # Periodic linear interpolation
    f = interp1d(x_old, h, kind='linear', period=2)
    h_up = f(x_new)
    
    # Project to feasibility
    h_up = project_to_feasible(h_up, n_new)
    
    return h_up
