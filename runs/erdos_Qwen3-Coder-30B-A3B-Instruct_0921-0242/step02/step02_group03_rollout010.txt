import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cma

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        # Sort h in ascending order to fill smaller values first
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        # Sort h in descending order to reduce larger values first
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def memetic_descent(h, n_points, max_steps=30):
    """Refine h using projected gradient descent with the worst shift."""
    dx = 2.0 / n_points
    h_current = h.copy()
    
    for _ in range(max_steps):
        # Compute cross-correlation between h and 1-h using FFT for efficiency
        h_padded = np.pad(h_current, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h_padded = np.pad(1-h_current, (0, n_points-1), mode='constant', constant_values=0)
        
        # Using FFT for efficient cross-correlation
        corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
        corr_vals = ifft(corr_freq).real
        corr_vals = corr_vals[:n_points]  # Take only the non-cyclic part
        # Normalize by dx
        corr_vals = corr_vals * dx
        
        # Find the worst shift index (non-zero) by finding minimum among non-zero shifts
        non_zero_shifts = np.concatenate([corr_vals[1:], corr_vals[:n_points-1]])
        worst_shift_idx = np.argmin(non_zero_shifts) % (n_points - 1) + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h_current)
        # Gradient wrt h[i] is dx * (1 - h[i + worst_shift_idx] - h[i - worst_shift_idx])
        for i in range(n_points):
            right_idx = (i + worst_shift_idx) % n_points
            left_idx = (i - worst_shift_idx) % n_points
            grad[i] = dx * (1 - h_current[right_idx] - h_current[left_idx])
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 1.0
        converged = False
        while alpha > 1e-9:
            h_trial = h_current - alpha * grad
            h_trial = project_to_feasible(h_trial, n_points)
            trial_c5 = evaluate_c5(h_trial)
            current_c5 = evaluate_c5(h_current)
            
            if trial_c5 < current_c5:
                h_current = h_trial
                converged = True
                break
            else:
                alpha *= 0.5
                if alpha < 1e-9:
                    break
        
        if not converged:
            break
    
    return h_current

def symmetric_piecewise_constant(p, q, n, M):
    """Build a symmetric piecewise constant function from parameters."""
    # Transform widths to sum to 1
    w = np.exp(p)
    w = w / np.sum(w)
    
    # Transform heights to [0, 1] using logistic
    a = 1 / (1 + np.exp(-q))
    
    # Create left half
    h_half = np.zeros(n//2)
    cumsum_w = np.concatenate([[0], np.cumsum(w)])
    
    # Map indices to segments
    for i in range(len(cumsum_w)-1):
        left_idx = int((cumsum_w[i] * (n//2)))
        right_idx = int((cumsum_w[i+1] * (n//2)))
        h_half[left_idx:right_idx] = a[i]
    
    # Create full h by mirroring
    h_full = np.concatenate([h_half, h_half[::-1]])
    
    return h_full

def objective_function(params, n, M):
    p = params[:M]  # First M parameters for widths
    q = params[M:]  # Next M parameters for heights
    
    # Build function
    h = symmetric_piecewise_constant(p, q, n, M)
    
    # Project to ensure sum equals n/2
    h = project_to_feasible(h, n)
    
    # Evaluate C5
    c5 = evaluate_c5(h)
    
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 30  # Reserve 30s for polishing
    
    n = 400
    dx = 2.0 / n
    M = 7  # Number of segments in symmetric piecewise constant function
    
    # Initial parameter vector for CMA-ES
    p0 = np.zeros(M)
    q0 = np.zeros(M)
    
    # Set up middle block with 0.3 width and large sigmoid value for height
    mid = M // 2
    p0[mid-1:mid+2] = 0.3
    p0[:mid-1] = (1 - 0.9) / (mid - 1)
    p0[mid+2:] = (1 - 0.9) / (M - mid - 2)
    
    # Heights in middle region large positive, others negative
    q0[mid-1:mid+2] = 5.0
    q0[:mid-1] = -5.0
    q0[mid+2:] = -5.0
    
    initial_params = np.concatenate([p0, q0])
    
    # Run CMA-ES optimization
    es = cma.CMAEvolutionStrategy(initial_params, 0.3, {'popsize': 4 + int(3 * np.log(2 * M))})
    best_params = initial_params.copy()
    best_c5 = np.inf
    
    num_restarts = 0
    max_restarts = 10
    gen_interval = 50
    
    while time.time() < time_limit and num_restarts <= max_restarts:
        # Sample population
        population = es.ask()
        fitnesses = []
        
        for indiv in population:
            fit = objective_function(indiv, n, M)
            fitnesses.append(fit)
            
            if fit < best_c5:
                best_c5 = fit
                best_params = indiv.copy()
        
        # Tell CMA-ES
        es.tell(population, fitnesses)
        
        # Check for restart condition
        if es.countiter % gen_interval == 0:
            if es.countiter > 0 and es.fbest < best_c5:
                continue  # Better solution found recently
            
            # Reset with increased population size
            if num_restarts < max_restarts:
                es = cma.CMAEvolutionStrategy(
                    best_params, 
                    0.3, 
                    {'popsize': 2 * es.popsize, 'seed': seed}
                )
                num_restarts += 1
            else:
                break
    
    # Perform final polishing with memetic descent
    p_opt = best_params[:M]
    q_opt = best_params[M:]
    h_best = symmetric_piecewise_constant(p_opt, q_opt, n, M)
    h_best = project_to_feasible(h_best, n)
    c5_best = evaluate_c5(h_best)
    
    # Polishing
    h_polished = memetic_descent(h_best, n, max_steps=25)
    c5_polished = evaluate_c5(h_polished)
    
    if c5_polished < c5_best:
        h_best = h_polished
        c5_best = c5_polished
        
    return (h_best, c5_best, n)
