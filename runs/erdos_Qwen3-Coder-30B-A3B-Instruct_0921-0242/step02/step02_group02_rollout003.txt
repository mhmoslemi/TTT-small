import numpy as np
import time
from scipy.optimize import differential_evolution, linprog
from scipy.fft import rfft, irfft
from scipy.interpolate import interp1d

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2 with h in [0,1] using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum

    if abs(diff) < 1e-10:
        return h

    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid

    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using FFT
    H = rfft(h)
    A = irfft(H * np.conj(H), n)
    A[0] = n/2
    max_corr = np.max(A) * dx
    return 1 - max_corr

def basis_evaluation(coefs, n_points):
    """Evaluate h_c(x) = 0.5 + sum(am*cos(m*pi*x) + bm*sin(m*pi*x))"""
    x_vals = np.linspace(0, 2, n_points, endpoint=False)
    h_raw = 0.5 * np.ones_like(x_vals)
    M = len(coefs) // 2
    for m in range(1, M+1):
        a_m, b_m = coefs[2*(m-1)], coefs[2*(m-1)+1]
        h_raw += a_m * np.cos(m * np.pi * x_vals) + b_m * np.sin(m * np.pi * x_vals)
    return h_raw

def fitness(c, n_points):
    """Fitness function for DE, returns C5."""
    h_raw = basis_evaluation(c, n_points)
    h = np.clip(h_raw, 0, 1)
    scale = 1.0 / np.sum(h) * (n_points / 2)
    h = h * scale
    return evaluate_c5(h)

def linearise_and_solve_lp(h, k_star, n_points, max_constraints=10):
    """Solve the linearized LP subproblem."""
    # Compute gradients (cyclic shifts)
    h_grads = [np.roll(h, k) for k in range(-n_points+1, n_points)]
    # Get active constraints (i.e. shifts near k_star)
    active_k_indices = np.argsort([np.abs(k - k_star) for k in range(n_points)])[:max_constraints]
    # Compute A(k) for each shift
    A_values = [np.dot(h, h_grads[k]) for k in active_k_indices]
    # Assemble LP
    c = np.zeros(n_points + 1)  # h + t variable
    c[-1] = -1.0  # maximize t
    A_ub = []
    b_ub = []
    # Put into canonical form Ax <= b
    for i, k in enumerate(active_k_indices):
        # Gradient of A(k) w.r.t. h
        grad_A = h_grads[k]
        row = np.concatenate([grad_A, [-1.0]])  # h_grads + t
        A_ub.append(row)
        b_ub.append(A_values[i])  # >= t => -grad_A * h + t <= A_values[i]

    # Box constraints
    bounds = [(0, 1) for _ in range(n_points)] + [(-np.inf, np.inf)]
    # Sum constraint
    A_eq = np.zeros(n_points + 1)
    A_eq[:n_points] = 1.0
    b_eq = n_points / 2.0
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=[A_eq], b_eq=[b_eq], bounds=bounds, method='highs')
        if res.success:
            h_new = res.x[:-1]
            t_new = res.x[-1]
            return h_new, t_new
    except:
        pass
    return h, -np.inf

def subgradient_step(h, n_points, max_iter=100):
    """Perform a few iterations of sub-gradient descent."""
    dx = 2.0 / n_points
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        k_star = np.argmax(corr)
        if k_star > n_points:
            k_star = k_star - n_points
        else:
            k_star = k_star - n_points + 1
        # Compute gradient (circular shifts)
        grad_i = np.roll(h, k_star)
        # Move mass from high to low values along neg.grad
        i_high = np.argmax(h)
        i_low = np.argmin(h)
        delta = min(0.1, 0.5 * (h[i_high] - h[i_low]))
        if delta < 1e-4:
            break
        h[i_high] -= delta
        h[i_low] += delta
        h = project_to_feasible(h, n_points)

    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n0 = 250  # low resolution for global search
    M = 12  # number of Fourier terms per side
    d = 2*M  # dimensionality of coeff vector
    pop = 15 * d
    maxiter = 3000 // pop
    time_left = budget_s - 0.5

    # Build basis functions once
    x_ref = np.linspace(0, 2, n0, endpoint=False)
    cos_basis = np.array([[np.cos(m * np.pi * x_ref) for m in range(1, M+1)]])
    sin_basis = np.array([[np.sin(m * np.pi * x_ref) for m in range(1, M+1)]])

    # Global DE search on coefficient space
    bounds = [(-0.5, 0.5) for _ in range(d)]
    res = differential_evolution(fitness, bounds, args=(n0,), popsize=pop, maxiter=maxiter, seed=seed, recombination=0.7, atol=1e-6, disp=False)
    best_c = res.x

    # Get top candidates (keep unique ones)
    candidates = [best_c]
    all_cands = []
    all_cands.append(best_c)
    for _ in range(20):  # try 20 additional random starts
        c0 = np.random.uniform(-0.5, 0.5, d)
        res0 = differential_evolution(fitness, bounds, args=(n0,), x0=c0, popsize=pop//2, maxiter=50, seed=seed//3, recombination=0.7, atol=1e-6, disp=False)
        all_cands.append(res0.x)

    # Keep distinct candidates by Euclidean distance
    unique_cands = [all_cands[0]]
    for c in all_cands[1:]:
        is_distinct = True
        for uc in unique_cands:
            if np.linalg.norm(c - uc) < 1e-4:
                is_distinct = False
                break
        if is_distinct:
            unique_cands.append(c)
        if len(unique_cands) >= 5:
            break
    candidates = unique_cands[:5]

    # SLP refinement for top candidates
    best_h = None
    best_c5 = float('inf')
    for i, c in enumerate(candidates):
        if time.time() - start_time > time_left:
            break
        h_raw = basis_evaluation(c, n0)
        h_clip = np.clip(h_raw, 0, 1)
        h = h_clip * (n0 / 2) / np.sum(h_clip)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # refine with SLP
        remaining_time = time_left - (time.time() - start_time)
        if remaining_time < 1.0:
            continue
        max_iter = min(400, int(remaining_time * 10))
        for it in range(max_iter):
            if time.time() - start_time > time_left:
                break
            corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n0)
            k_star = np.argmax(corr)
            if k_star > n0:
                k_star = k_star - n0
            else:
                k_star = k_star - n0 + 1

            # solve LP
            h_new, t_new = linearise_and_solve_lp(h, k_star, n0)
            if t_new > np.max(corr) - 1e-6:  # Keep improvement
                h = h_new.copy()
                c5_new = evaluate_c5(h)
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
            else:
                break  # no meaningful improvement

    # Upsample and polish
    n1 = 500
    x_up = np.linspace(0, 2, n1, endpoint=False)

    h_up = basis_evaluation(best_c, n1)
    h_up = np.clip(h_up, 0, 1)
    h_final = h_up * (n1 / 2) / np.sum(h_up)

    # Additional polish with gradient-like steps
    for _ in range(2000):
        if time.time() - start_time > budget_s - 0.5:
            break
        corr = np.correlate(h_final, 1 - h_final, mode='full') * (2.0 / n1)
        k_star = np.argmax(corr)
        if k_star > n1:
            k_star = k_star - n1
        else:
            k_star = k_star - n1 + 1
        # Simple subgradient-like step
        grad_i = np.roll(h_final, k_star)
        delta = 0.01 * np.min(h_final)  # small move
        if delta < 1e-5:
            break
        h_final = np.clip(h_final - delta * grad_i, 0, 1)
        h_final = h_final * (n1 / 2) / np.sum(h_final)

    final_c5 = evaluate_c5(h_final)
    final_h = h_final

    # Final refinement if needed
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h.copy()

    return (best_h, best_c5, len(best_h))
