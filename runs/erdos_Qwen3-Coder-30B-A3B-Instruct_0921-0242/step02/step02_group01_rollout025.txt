import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cvxpy as cp

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def compute_autocorrelations_binary(x):
    """Compute autocorrelations for a binary vector"""
    n = len(x)
    # Compute the autocorrelations with FFT for efficiency
    x_fft = fft(x, 2 * n)
    x_conj_fft = np.conj(x_fft)
    corr = ifft(x_fft * x_conj_fft)
    # Take only the first n components (the relevant ones)
    return corr.real[:n]

def update_autocorrelations(R, x, i, j):
    """Update autocorrelations after swapping positions i and j"""
    n = len(x)
    if x[i] == x[j]:
        return R.copy()
    # Swap the values in x
    x_new = x.copy()
    x_new[i], x_new[j] = x_new[j], x_new[i]
    
    # Recompute the full autocorrelation vector
    # Optimization: only update the values that may change
    R_new = np.zeros(n)
    # For all k, compute the autokorrelation
    for k in range(n):
        R_new[k] = sum(x_new[(i+k)%n] * x_new[i] for i in range(n)) if k != 0 else 0
    return R_new

def find_worst_shift(R):
    """Find the shift index that minimizes autocorrelation"""
    return np.argmin(R[1:]) + 1

def compute_autocorrelation_shift(x, k):
    """Compute auto-correlation for a given shift"""
    n = len(x)
    corr = 0
    for i in range(n):
        corr += x[i] * x[(i + k) % n]
    return corr

def eval_c5_binary(x):
    """Evaluate C5 for binary vector x"""
    n = len(x)
    min_corr = float('inf')
    for k in range(1, n):
        corr = compute_autocorrelation_shift(x, k)
        min_corr = min(min_corr, corr)
    dx = 2.0 / n
    return 1 - 2 * min_corr / n

def swap_improvement(R, x, k):
    """Evaluate if a swap improves the worst shift k"""
    n = len(x)
    # Check all swaps that might change the k-th autocorrelation
    best_swap = None
    min_c_i = float('inf')
    max_d_j = float('-inf')
    
    c_i_vals = []
    d_j_vals = []
    
    # For all i with x_i=1, compute c_i
    for i in range(n):
        if x[i] == 1:
            c_i = x[(i + k) % n] + x[(i - k) % n]
            c_i_vals.append((c_i, i))
    
    # For all j with x_j=0, compute d_j  
    for j in range(n):
        if x[j] == 0:
            d_j = x[(j + k) % n] + x[(j - k) % n]
            d_j_vals.append((d_j, j))

    if not c_i_vals or not d_j_vals:
        return None
        
    # Find best i and worst j
    c_i_vals.sort(reverse=True)
    d_j_vals.sort()
    
    i_best = c_i_vals[0][1]
    j_best = d_j_vals[0][1] 
    
    # Compute improvement
    c_i = x[(i_best + k) % n] + x[(i_best - k) % n]
    d_j = x[(j_best + k) % n] + x[(j_best - k) % n]
    improvement = d_j - c_i
    
    if improvement > 0:
        return (i_best, j_best)
    return None

def implement_swap(x, i, j):
    """Perform a swap in binary vector"""
    x_new = x.copy()
    x_new[i], x_new[j] = x_new[j], x_new[i]
    return x_new

def greedy_swap_refine_binary(x, max_time):
    start_time = time.time()
    n = len(x)
    
    # Compute initial autocorrelations
    R = np.array([compute_autocorrelation_shift(x, k) for k in range(n)])
    worst_k = find_worst_shift(R) # k in [1, n-1]
    min_R = R[worst_k]
    
    cycle_count = 0
    
    while time.time() - start_time < max_time:
        cycle_count += 1
        # Try to improve the worst shift k
        improvement = swap_improvement(R, x, worst_k)
        
        if improvement is None:
            # No improvement possible for current k
            # Find new worst shift
            worst_k = find_worst_shift(R)
            min_R = R[worst_k]
            # If no improvement, break
            if min_R == 0:
                break
            else:
                continue
                
        i, j = improvement
        # Perform the swap
        x = implement_swap(x, i, j)
        # Update the autocorrelations affected by the swap
        R = np.array([compute_autocorrelation_shift(x, k) for k in range(n)])
        
        # Update the current worst shift and minimum
        worst_k = find_worst_shift(R)
        min_R = R[worst_k]
        
        # Early termination if we reach close to target
        if min_R >= 40 and cycle_count > 2000:  # heuristic
            break
            
    return x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s * 0.7
    
    # Use 200 points gives good balance of accuracy and speed
    n = 200
    dx = 2.0 / n
    
    # Step 1: Solve SDP relaxation
    # Maximize t subject to:
    # X is PSD
    # 0 <= X[i,i] <= 1
    # sum X[i,i] = n/2  
    # trace(A_k X) >= t for all k=1..n-1
    # X[i,j] >= 0, X[i,j] <= X[i,i], X[i,j] <= X[j,j]
    
    # Construct A_k matrices efficiently
    A = []
    for k in range(1, n):
        # A_k[i,j] = 1 if (i, (i+k)%n) == (j, (j+k)%n) 
        # But that's wrong. A_k has ones at positions (i, (i+k)%n) 
        # Let's construct A_k properly
        temp = np.zeros((n,n))
        for i in range(n):
            temp[i, (i+k) % n] = 1
        A.append(temp)
    
    # Variables for the SDP
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints: PSD
    SDP_constraints = [X >> 0]  # X is psd
    
    # Constraints: diagonal bounds
    for i in range(n):
        SDP_constraints.append(0 <= X[i,i])
        SDP_constraints.append(X[i,i] <= 1)
    
    # Constraints: sum constraint  
    SDP_constraints.append(cp.sum(X.diagonal()) == n/2)
    
    # Autocorrelation constraints
    for A_k in A:
        SDP_constraints.append(cp.trace(A_k @ X) >= t)
        
    # Pairwise bounds (tighten relaxation)
    for i in range(n):
        for j in range(i+1, n):
            SDP_constraints.append(X[i,j] >= 0)
            SDP_constraints.append(X[i,j] <= X[i,i])
            SDP_constraints.append(X[i,j] <= X[j,j])
        
    # Objective
    objective = cp.Maximize(t)
    
    # Formulation 
    prob = cp.Problem(objective, SDP_constraints)
    try:
        # Solve SDP (may take time)
        prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
        if prob.status != 'optimal':
            pass # Fallback to randomization may be needed
    except:
        pass
    
    # Step 2: Rounding strategy
    
    # Create candidates
    best_x = None
    best_c5 = float('inf')
    
    # Use eigenvector rounding approach: take leading eigenvector of X*
    if "X.value" in locals() and X.value is not None:
        # Get eigenvalues and vectors of the optimal matrix
        evals, evecs = np.linalg.eigh(X.value)
        v = evecs[:, -1]  # Assumed to be large eigenvalue
        
        # Random rounding (R = 30)
        for _ in range(30):
            g = np.random.randn(n)
            z = evecs @ np.diag(np.sqrt(np.maximum(evals, 0))) @ g
            x_candidate = np.zeros(n)
            idxs = np.argsort(z)[::-1][:n//2] # Top n//2 indices
            x_candidate[idxs] = 1
            x_candidate = project_to_feasible(x_candidate, n)
            c5 = eval_c5_binary(x_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_x = x_candidate.copy()
    else:
        best_x = np.ones(n // 2)
        best_x = np.pad(best_x, (0, n - len(best_x)), 'constant')
        best_x = project_to_feasible(best_x, n)
        best_c5 = eval_c5_binary(best_x)
    
    # Step 3: Greedy local Improvement
    # Run until time is almost used or no improvement possible 
    max_refinement_time = budget_s * 0.3
    final_x = greedy_swap_refine_binary(best_x, max_refinement_time)
    c5 = eval_c5_binary(final_x)
    
    # Step 4: Continuous polishing (small)
    # Convert to continuous domain:
    # Since the result is binary we can do a gentle optimization
    final_h = final_x
    # This isn't very sensitive, so a simple run
    for i in range(100):
        if time.time() - start_time > budget_s * 0.99:
            break
        corr = ifft(fft(final_h, 2 * len(final_h)) * np.conj(fft(1 - final_h, 2 * len(final_h))))
        # Maximize over shift 1 to n-1
        corr_vals = corr.real[1: len(final_h)]  
        worst_shift = np.argmin(corr_vals) + 1
        # Compute gradient
        grad = np.zeros_like(final_h)
        for j in range(len(final_h)):
            left_idx = (j + worst_shift) % len(final_h)
            right_idx = (j - worst_shift) % len(final_h)
            grad[j] = dx * ((1 - final_h[left_idx]) - final_h[right_idx])
        # Take small step 
        final_h = final_h - 0.005 * grad
        final_h = project_to_feasible(final_h, n)
        c5 = eval_c5_binary(final_h)
    
    # Final evaluation
    return (final_h, c5, n)
