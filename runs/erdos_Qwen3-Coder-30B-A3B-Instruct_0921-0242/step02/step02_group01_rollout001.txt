import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.signal import correlate

def project_to_sum_constraint(h, target_sum):
    """Project h to have sum = target_sum while keeping in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Water-filling adjustment
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5_fft(h):
    """Fast evaluation of C5 constant using FFT."""
    n = len(h)
    dx = 2.0 / n
    # Construct 1 - h
    ih = 1.0 - h
    # Pad to 2n for full correlation (avoid wraparound)
    pad_len = 2 * n
    h_pad = np.pad(h, (0, pad_len - n), mode='constant')
    ih_pad = np.pad(ih, (0, pad_len - n), mode='constant')
    # FFT correlation
    H_fft = rfft(h_pad)
    Ih_fft = rfft(ih_pad)
    corr_fft = H_fft * np.conj(Ih_fft)
    corr = irfft(corr_fft)
    # Take full correlation, ignore index == 0 and keep only first n values
    corr_vals = corr[:n]
    c5 = np.max(corr_vals[1:]) * dx  # exclude shift 0
    return c5

def gradient_worst_shift(h):
    """Compute gradient w.r.t. h for the worst shift."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all shifts
    ih = 1.0 - h
    H_fft = rfft(h)
    Ih_fft = rfft(ih)
    corr_fft = H_fft * np.conj(Ih_fft)
    corr = irfft(corr_fft)
    corr_vals = corr[:n]
    # Find worst shift (excluding k=0)
    worst_k = np.argmax(corr_vals[1:]) + 1
    # Compute gradient for the worst shift
    grad = np.zeros_like(h)
    for i in range(n):
        left_idx = (i + worst_k) % n
        right_idx = (i - worst_k) % n
        grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
    return grad

def gradient_descent_on_worst_shift(h, max_iter=200, step_size=0.1):
    """Projected gradient descent on current worst shift."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for t in range(max_iter):
        grad = gradient_worst_shift(h)
        h_new = h - step_size * grad
        h_new = project_to_sum_constraint(h_new, n / 2)
        if evaluate_c5_fft(h_new) < evaluate_c5_fft(h):
            h = h_new
        else:
            step_size *= 0.95  # Decrease step size
    return h

def de_step(pop, F=0.6, CR=0.9):
    """Single DE step for z_n+1 = x_n + F·(x_a - x_b) + CR·(x_c - x_b)"""
    new_pop = []
    for i, x in enumerate(pop):
        a, b, c = np.random.choice(len(pop), 3, replace=False)
        donor = pop[a] + F * (pop[b] - pop[c])
        donor = np.clip(donor, 0, 1)
        donor = project_to_sum_constraint(donor, len(donor) / 2)
        # Binomial crossover
        mask = np.random.rand(len(donor)) < CR
        trial = np.where(mask, donor, x)
        new_pop.append(trial)
    return new_pop

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialization
    initial_h_values = np.array([0.5] * 300)  # fallback
    if 'initial_h_values' in globals():
        # Interpolate to 400 points
        x_orig = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, 400)
        initial_h = np.interp(x_new, x_orig, initial_h_values)
    else:
        initial_h = np.ones(400) * 0.5
    n_points = 400
    target_sum = n_points / 2

    # Initialize population
    pop_size = min(40, 2 * n_points // 10)
    population = [project_to_sum_constraint(np.random.rand(n_points), target_sum)]
    population[0] = initial_h.copy()
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_to_sum_constraint(h, target_sum)
        population.append(h)

    # DE main loop
    best_h = None
    best_c5 = float('inf')
    gen_count = 0
    max_gens = int(4000)  # budget estimate
    freq_refine = 200  # Refine every so often

    while gen_count < max_gens and time.time() - start_time < 0.75 * budget_s:
        gen_count += 1
        new_pop = de_step(population, F=0.6, CR=0.9)

        for i, h in enumerate(new_pop):
            c5 = evaluate_c5_fft(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            population[i] = h  # Overwrite at this point

        if gen_count % freq_refine == 0:
            refined = gradient_descent_on_worst_shift(best_h, max_iter=50)
            new_c5 = evaluate_c5_fft(refined)
            if new_c5 < best_c5:
                best_h = refined
                best_c5 = new_c5

        if time.time() - start_time > 0.75 * budget_s:
            break

    # Final gradient refinement
    if best_h is not None:
        refined_best = gradient_descent_on_worst_shift(best_h, max_iter=200, step_size=0.05)

        # Final polishing with stochastic swaps
        h_best = refined_best.copy()
        current_c5 = evaluate_c5_fft(h_best)
        temp = 1.0
        decay_rate = 0.995
        eval_count = 0
        max_evals = 1000

        while eval_count < max_evals and time.time() - start_time < 0.995 * budget_s:
            eval_count += 1
            i, j = np.random.randint(0, n_points, 2)
            if i == j:
                continue
            delta = np.random.uniform(0, min(h_best[i], 1 - h_best[j], 0.01))
            h_test = h_best.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_sum_constraint(h_test, n_points / 2)

            test_c5 = evaluate_c5_fft(h_test)
            if test_c5 < current_c5:
                h_best = h_test
                current_c5 = test_c5
            else:
                prob = np.exp(-(test_c5 - current_c5) / temp)
                if np.random.random() < prob:
                    h_best = h_test
                    current_c5 = test_c5
            temp *= decay_rate

        # Optional: interpolate to higher resolution and refine
        if n_points == 400:
            x_new = np.linspace(0, 2, 800)
            x_old = np.linspace(0, 2, n_points)
            h_high_res = np.interp(x_new, x_old, h_best)
            h_high_res = project_to_sum_constraint(h_high_res, 800 / 2)
            h_refined = gradient_descent_on_worst_shift(h_high_res, max_iter=100)
            c5_TEST = evaluate_c5_fft(h_refined)
            if c5_TEST < current_c5:
                h_best = h_refined
                current_c5 = c5_TEST

        return (h_best, current_c5, len(h_best))
    else:
        h_fallback = np.ones(n_points) * 0.5
        h_fallback = project_to_sum_constraint(h_fallback, n_points / 2)
        c5_fallback = evaluate_c5_fft(h_fallback)
        return (h_fallback, c5_fallback, n_points)
