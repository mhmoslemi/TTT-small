import numpy as np
import time
from scipy import signal
from scipy.fft import fft, ifft
from scipy.optimize import bisect

def project_box_sum(y, total):
    """Project vector y onto box [0,1] with sum constraint."""
    y = np.clip(y, 0, 1)
    s = np.sum(y)
    if abs(s - total) < 1e-10:
        return y
    # Find lambda such that sum(clamp(y - lambda, 0, 1)) = total
    def sum_func(lam):
        return np.sum(np.clip(y - lam, 0, 1)) - total
    if s < total:
        # We need to increase values, lambda must be < 0
        a, b = -1.0, 0.0 
        lam = bisect(sum_func, a, b, xtol=1e-12)
    else:
        # We need to decrease values, lambda must be > 0
        a, b = 0.0, 1.0
        lam = bisect(sum_func, a, b, xtol=1e-12)
    y = np.clip(y - lam, 0, 1)
    return y

def evaluate_c5(h):
    """Evaluate C5 for given h."""
    dx = 2.0 / len(h)
    corr = signal.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorr(h):
    """Compute aperiodic autocorrelation of h."""
    n = len(h)
    # Use FFT for efficiency
    H = fft(h)
    P = np.abs(H)**2
    gamma = ifft(P).real
    return gamma

def update_autocorr_delta(gamma, h, i, j):
    """Update autocorrelation with a 1 at i and 0 at j."""
    # originally h[i] = 1, h[j] = 0
    # now h[i] = 0, h[j] = 1
    # so this is the difference from changing i from 1 to 0 and j from 0 to 1
    n = len(h)
    delta = np.zeros_like(gamma)
    for k in range(-n+1, n):
        delta[k + n - 1] = (
            -h[i] * h[(i+k) % n] 
            - h[i] * h[(i-k) % n]
            + h[j] * h[(j+k) % n]
            + h[j] * h[(j-k) % n]
        )
    return delta

def binary_hill_climb(h_start, n_iter=3000):
    """Perform hill-climbing on binary vectors."""
    h = h_start.copy().astype(int)
    gamma = compute_autocorr(h.astype(float))
    best_h = h.copy()
    best_C5 = evaluate_c5(h.astype(float))
    
    n = len(h)
    
    for _ in range(n_iter):
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h[i] == 0 or h[j] == 1:
            continue
        # Compute the change in autocorrelation
        delta_gamma = update_autocorr_delta(gamma, h, i, j)
        new_gamma = gamma - delta_gamma
        # Compute the new C5
        # C5 = max_k (1 - (2/n) * gamma_k)
        max_C5_new = 1.0 - (2.0 / n) * np.max(new_gamma)
        
        if max_C5_new < best_C5:
            h[i] = 0
            h[j] = 1
            gamma = new_gamma
            best_C5 = max_C5_new
            best_h = h.copy()
            
    return best_h.astype(float), best_C5

def spectral_alternating_projection(h_start, max_iter=400):
    """SAP with soft target autocorrelation."""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    alpha = 0.7
    best_C5 = evaluate_c5(h)
    best_h = h.copy()
    
    for iter in range(max_iter):
        # compute current autocorrelation
        gamma = compute_autocorr(h)
        gamma_0 = gamma[0]
        avg = np.sum(gamma[1:]) / (n - 1)
        
        # Soft target
        gamma_target = np.full_like(gamma, avg)
        gamma_target[0] = gamma_0
        gamma_target = alpha * gamma + (1 - alpha) * gamma_target
        
        # Update magnitudes in Fourier domain
        ifft_result = ifft(gamma_target)
        P_target = np.abs(ifft_result)**2
        A_target = np.sqrt(np.maximum(P_target, 0))
        H = fft(h)
        H_new = A_target * np.exp(1j * np.angle(H))
        
        h_new = ifft(H_new).real
        
        h_new = project_box_sum(h_new, n/2)
        c5 = evaluate_c5(h_new)
        
        if c5 < best_C5:
            best_C5 = c5
            best_h = h_new.copy()
            
        h = h_new
        
        # Update alpha slowly
        if iter < 100:
            alpha = min(0.99, alpha + 0.02)
    
    return best_h, best_C5

def projected_gradient_polishing(h_start, max_iter=300, beta_init=5.0):
    """Final polishing step."""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    beta = beta_init
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for i in range(max_iter):
        # Compute weight vector
        gamma = compute_autocorr(h)
        C = 1 - (2.0 / n) * gamma
        w = np.exp(beta * C)
        w = w / np.sum(w)
        
        # compute gradient
        conv = np.fft.ifft(np.fft.fft(w) * np.fft.fft(h)).real
        conv_flip = np.fft.ifft(np.fft.fft(w[::-1]) * np.fft.fft(h)).real
        grad = -(2.0 / n) * (conv + conv_flip)
        
        # backtracking line search
        eta = 1.0
        while eta > 1e-12:
            h_new = h - eta * grad
            h_new = project_box_sum(h_new, n/2)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                break
            eta *= 0.5
            
        # Reduce beta
        beta *= 1.2
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n = 256
    dx = 2.0 / n
    
    # Phase 1: construct seeds and binary hill-climb
    seeds = build_seeds(n)
    best_h = None
    best_C5 = float('inf')
    
    for seed_name, h_seed in seeds:
        h_iter, c5_iter = binary_hill_climb(h_seed, n_iter=3000)
        if c5_iter < best_C5:
            best_C5 = c5_iter
            best_h = h_iter.copy()
            
        if time.time() - start_time > 0.1 * budget_s:
            break
            
    # Phase 2: SAP
    if time.time() - start_time > 0.8 * budget_s:
        return (best_h, best_C5, n)
    
    final_h, final_c5 = spectral_alternating_projection(best_h, max_iter=400)
    if final_c5 < best_C5:
        best_C5 = final_c5
        best_h = final_h.copy()
        
    # Phase 3: final polishing
    if time.time() - start_time > 0.95 * budget_s:
        return (best_h, best_C5, n)
    
    polshed_h, polshed_c5 = projected_gradient_polishing(final_h, max_iter=300)
    if polshed_c5 < best_C5:
        best_C5 = polshed_c5
        best_h = polshed_h.copy()
        
    return (best_h, best_C5, n)
