import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_box_plus_sum(h, n):
    """Project h onto the set {0 <= h <= 1, sum(h) = n/2} using water-filling."""
    # Subtract a scalar lambda from h, clip to [0,1], adjust lambda to meet sum
    y = h.copy()
    target_sum = n / 2.0

    def sum_clipped(lam):
        return np.sum(np.clip(y - lam, 0, 1))

    # Binary search on lambda
    low, high = np.min(y - 1), np.max(y)
    for _ in range(40):
        lam = (low + high) / 2.0
        s = sum_clipped(lam)
        if s > target_sum:
            low = lam
        else:
            high = lam
    lam = (low + high) / 2.0
    h_proj = np.clip(y - lam, 0, 1)
    return h_proj

def compute_autocorrelation(h, n_fft):
    """Compute aperiodic autocorrelation using FFT."""
    h_pad = np.pad(h, (0, n_fft - len(h)), 'constant', constant_values=0)
    H = fft(h_pad)
    gamma = ifft(H * np.conj(H)).real
    return gamma[:2*n-1]

def compute_c5_from_h(h):
    """Evaluate C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

        iter_count += 1

        # minimize (rho/2) * ||h - (z - u)||^2 s.t. 1 - (2/n)(h * h)_s <= t

        z_minus_u = z - u
        z_minus_u_fft = fft(z_minus_u)

        # Bisect to find the smallest mu s.t. max_s (1 - (2/n)(h * h)_s) <= t
        mu_old = 0.0
        mu_high = 1e10  # large upper bound

        mu = 1e-6

        for _ in range(30):

            h_fft = (z_minus_u_fft) / (1 + mu * ones_fft)
            h = ifft(h_fft).real[:n]
            h = np.clip(h, 0, 1)  # enforce bounds

            gamma = compute_autocorrelation(h, 2*n - 1)

            if max_c <= t:

        if abs(mu-high) < 1e-12:

def nesterov_polish(z, n):
    """Apply Nesterov accelerated gradient with smooth surrogate."""
    # Parameters
    beta = 5.0
    max_iter = 200
    eta = 1.0
    eps = 1e-10
    
    h = z.copy()
    h_prev = z.copy()
    y = z.copy()
    t = 1.0

    # Helper to compute gradient of smooth max
    def compute_smooth_gradient(h, beta, n):
        gamma = compute_autocorrelation(h, 2*n - 1)
        C_vals = 1.0 - (2.0/n) * gamma
        C_max = np.max(C_vals)
        exp_C = np.exp(beta * (C_vals - C_max))
        w = exp_C / np.sum(exp_C)
        
        # gradient of smooth max
        # ∇F_β = -(2/n) * Σ_s w_s * shift_{+s}(h) + shift_{-s}(h)
        grad = np.zeros_like(h)
        # Compute shifted sums
        for s in range(1, n):
            right_sum = h[(s + np.arange(n)) % n]
            left_sum = h[(np.arange(n) - s) % n]
            grad -= (2.0/n) * w[s] * (left_sum + right_sum)
        # Special case for s = 0: flat contribution
        grad -= (2.0/n) * w[0] * 2.0 * h  # h + h shift = 0, so just 2h
        return grad
    
    for k in range(max_iter):
        y = h + ((k-1) / (k + 2)) * (h - h_prev)
        grad = compute_smooth_gradient(y, beta, n)
        
        # Line search for eta
        while True:
            h_new = project_to_box_plus_sum(y - eta * grad, n)
            c5_new = compute_c5_from_h(h_new)  # Use exact eval
            if c5_new < compute_c5_from_h(h):
                break
            eta *= 0.5
            if eta < 1e-10:
                break

        if np.abs(compute_c5_from_h(h_new) - compute_c5_from_h(h)) < 1e-10:
            break
        
        h_prev = h.copy()
        h = h_new.copy()
        
        # Update beta
        if k % 20 == 0 and beta < 200:
            beta *= 1.3

        if k > 10 and abs(compute_c5_from_h(h) - compute_c5_from_h(h_prev)) < 1e-10:
            break
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()

    n = 256  # Using 256 points for faster convergence
    z_best, t_best = admm_solve(n, budget_s * 0.7)

    c5_bound = compute_c5_from_h(h)
