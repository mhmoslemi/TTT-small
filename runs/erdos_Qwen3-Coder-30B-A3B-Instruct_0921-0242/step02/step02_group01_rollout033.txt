import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp
from scipy.linalg import toeplitz

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def gradient_worst_shift(h, worst_shift):
    """Compute gradient with respect to worst shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    
    # Since correlation uses circular shift, we take h_{i+k} and h_{i-k}
    # for k = worst_shift
    for i in range(n):
        left_idx = (i + worst_shift) % n
        right_idx = (i - worst_shift) % n
        grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
    
    return grad

def backtrack_line_search(h, grad, max_iter=10):
    """Backtracking line search for gradient update."""
    alpha = 1.0
    dx = 2.0 / len(h)
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        if c5_new < c5_old:
            return h_new
        alpha *= 0.5
    return h  # fallback

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Early return for very short budget
    if budget_s < 10:
        n_points = 256
        h = np.ones(n_points) * 0.5
        return (h, 0.5, n_points)
    
    # Phase 1: SDP-based relaxation to find good autocorrelation
    n_points = 256  # Even number for simplicity and FFT
    t_low = 0.0
    t_high = 1.0
    epsilon = 2e-4
    dx = 2.0 / n_points

    # Start with the initial solution or balanced
    if 'initial_h_values' in globals():
        h_start = initial_h_values
        h_start = h_start[:n_points] if len(h_start) >= n_points else np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_start)), h_start)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        h_start = project_to_feasible(h_start, n_points)
    else:
        h_start = np.ones(n_points) * 0.5

    last_feasible_r = None
    best_t = 0.0

    # Binary bisection on t for autocorrelation
    while t_high - t_low > epsilon and time.time() - start_time < budget_s * 0.35:
        t_mid = (t_low + t_high) / 2
        
        # Create PSD feasibility problem
        r = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            r[0] == 1,
            r[1:] >= t_mid,
            r <= 1,
        ]
        
        # Toeplitz PSD matrix
        T = toeplitz(r)
        constraints.append(cp.PSD(T))
        
        # Feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                last_feasible_r = r.value
                best_t = t_mid
                t_low = t_mid
            else:
                t_high = t_mid
        except Exception:
            t_high = t_mid

    if last_feasible_r is None:
        last_feasible_r = np.ones(n_points)
        last_feasible_r[0] = 1.0

    # Phase 2: Spectral factorization
    r = last_feasible_r.copy()
    S = np.fft.fft(r)
    S = np.maximum(S, 0.0)  # ensure non-negative
    H = np.sqrt(S)
    g = np.real(np.fft.ifft(H))

    # Phase 3: Project to feasible set
    h = project_to_feasible(g, n_points)

    # Phase 4: Refine using worst-shift descent
    max_iter = min(250, int(500 * (budget_s * 0.35) / 600))
    alpha0 = 0.5 * dx
    for iter_count in range(max_iter):
        if time.time() - start_time > 0.9 * budget_s:
            break
        corr = np.real(ifft(fft(h) * np.conj(fft(1 - h))))
        corr = corr * dx
        worst_shift = np.argmax(corr[1:]) + 1
        grad = gradient_worst_shift(h, worst_shift)
        alpha = alpha0 / np.sqrt(iter_count + 1) if iter_count > 0 else alpha0
        h_new = h - alpha * grad
        h = project_to_feasible(h_new, n_points)

    # Phase 5: Stochastic polishing
    max_iter_stochastic = min(1000, int(1000 * (budget_s * 0.15) / 600))
    for _ in range(max_iter_stochastic):
        if time.time() - start_time > 0.99 * budget_s:
            break
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if h[i] <= 0 or h[j] >= 1:
            continue
        delta = min(0.01, h[i], 1 - h[j])
        h_test = h.copy()
        h_test[i] -= delta
        h_test[j] += delta
        h_test = project_to_feasible(h_test, n_points)
        c5_test = evaluate_c5(h_test)
        c5_current = evaluate_c5(h)
        if c5_test < c5_current:
            h = h_test

    # Final computation of C5
    c5_bound = evaluate_c5(h)

    # Phase 6: Optional upscaling for verification
    final_h = h
    if time.time() - start_time < 0.995 * budget_s:
        n_high = 2 * n_points
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_high)
        final_h = np.interp(x_new, x_orig, h)
        final_h = final_h * (n_high / 2) / np.sum(final_h)
        final_h = project_to_feasible(final_h, n_high)
        if time.time() - start_time < 0.995 * budget_s:
            # Short descent on fine grid
            for _ in range(min(50, int(50 * (budget_s * 0.02) / 600))):
                if time.time() - start_time > 0.995 * budget_s:
                    break
                corr = np.real(ifft(fft(final_h) * np.conj(fft(1 - final_h))))
                corr = corr * (2.0 / n_high)
                worst_shift = np.argmax(corr[1:]) + 1
                grad = gradient_worst_shift(final_h, worst_shift)
                alpha = 0.5 * (2.0 / n_high) / np.sqrt(1 + _)
                h_new = final_h - alpha * grad
                final_h = project_to_feasible(h_new, n_high)

    return (final_h, c5_bound, len(final_h))
