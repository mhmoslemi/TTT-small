import numpy as np
import time
from scipy.signal import fftconvolve
import cvxpy as cp
from scipy.ndimage import gaussian_filter1d

def aperiodic_autocorr(h):
    """Compute aperiodic autocorrelation via FFT."""
    n = len(h)
    # pad to avoid wraparound
    padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    # compute auto correlation via convolution
    corr = fftconvolve(padded, h[::-1], mode='valid')
    return corr[:n]

def softmax(beta, x):
    """Stable softmax computation."""
    x_centered = x - np.max(x)
    exp_x = np.exp(beta * x_centered)
    return exp_x / np.sum(exp_x)

def project_box_sum(h, total):
    """Project h onto {h: 0≤h_i≤1, sum h_i = total}."""
    # clip to [0,1]
    h = np.clip(h, 0, 1)
    # adjust to have sum = total
    current_sum = np.sum(h)
    diff = total - current_sum
    if abs(diff) < 1e-10:
        return h
    # add diff/n to each entry
    h = h + diff / len(h)
    # re-clip
    h = np.clip(h, 0, 1)
    return h

def conv(a, b):
    """Fast convolution using FFT."""
    n = len(a)
    # pad to avoid wraparound
    a_p = np.pad(a, (0, n-1), mode='constant', constant_values=0)
    b_p = np.pad(b, (0, n-1), mode='constant', constant_values=0)
    result = fftconvolve(a_p, b_p[::-1], mode='valid')
    return result[:n]

def score_guided_swap(h, n_swaps=5):
    """Perform swap moves to escape plateaus."""
    n = len(h)
    # Get the worst shifts
    gamma = aperiodic_autocorr(h)
    C = 1 - 2 * gamma / n
    eps = 1e-5
    worst_idx = np.where(np.abs(C - np.max(C)) < eps)[0]
    if len(worst_idx) == 0:
        worst_idx = [np.argmax(C)]
    
    # Compute scores
    scores = np.zeros(n)
    for s in worst_idx:
        if s == 0:
            scores += h
        else:
            scores[:-s] += h[s:]
            scores[s:] += h[:-s]

    # Find candidates
    max_score_idx = np.argmax(scores)
    min_score_idx = np.argmin(scores)

    # Perform small swap
    delta = 0.001
    if h[max_score_idx] > delta and h[min_score_idx] < 1 - delta:
        h[max_score_idx] -= delta
        h[min_score_idx] += delta
        h = project_box_sum(h, n/2)
    
    return h

def evaluate_c5(h):
    """Evaluate C5 for a given function h."""
    n = len(h)
    gamma = aperiodic_autocorr(h)
    C = 1 - 2 * gamma / n
    return np.max(C)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 300
    dx = 2.0 / n_points

    print("Starting Phase I - MILP Optimisation...")
    n1 = 120
    half = n1 // 2

    # Variables for the MILP
    b = cp.Variable(n1, boolean=True)
    a = cp.Variable((n1, n1), boolean=True)
    gamma = cp.Variable(n1)
    L = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(b) == half,
        L <= gamma[0],
        L <= gamma[1],
        L <= gamma[2],
        L <= gamma[3],
        L <= gamma[4],
        L <= gamma[5],
        L <= gamma[6],
        L <= gamma[7],
        L <= gamma[8],
        L <= gamma[9],
        L <= gamma[10],
    ]

    # Product constraints (McCormick)
    for s in range(1, n1):
        for i in range(n1-s):
            constraints.extend([
                a[i,s] <= b[i],
                a[i,s] <= b[i+s],
                a[i,s] >= b[i] + b[i+s] - 1
            ])
    
    # Autocorrelation constraints
    for s in range(n1):
        constraints.append(gamma[s] == cp.sum(a[:,s]))

    # Objective
    objective = cp.Maximize(L)
    
    # Formulate and solve
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC, time_limit=0.2 * budget_s, 
               relative_gap=1e-3, verbose=False)

    # Extract binary solution
    b_star = np.round(b.value).astype(int)
    print(f"MILP Status: {prob.status}")
    print(f"MILP objective value: {L.value}")

    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        print("Binary pattern found via MILP.")
    else:
        print(f"Warning: MILP status: {prob.status}. Using incumbent.")
        # Still try to get something useful
        b_star = np.round(b.value).astype(int)
    
    # Phase II: Upsample
    x1 = np.linspace(0, 2, n1, endpoint=False)
    x2 = np.linspace(0, 2, n_points, endpoint=False)
    h = np.interp(x2, x1, b_star)
    h = project_box_sum(h, n_points / 2)
    h = gaussian_filter1d(h, sigma=1.0)
    h = project_box_sum(h, n_points / 2)

    # Phase III: SmoothMax projected gradient descent
    print("Starting SmoothMax GD Phase...")
    beta = 2.0
    eta = 1.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    max_iter = 10000
    iteration = 0
    last_improvement = 0
    last_c5 = best_c5

    while time.time() - start_time < 0.85 * budget_s and iteration < max_iter:
        gamma = aperiodic_autocorr(h)
        C = 1 - 2 * gamma / n_points
        w = softmax(beta, C)
        # compute gradient
        grad = -(2 / n_points) * (conv(w, h) + conv(w[::-1], h))

        # backtracking line search
        alpha = eta
        h_trial = h - alpha * grad
        h_trial = project_box_sum(h_trial, n_points / 2)
        c5_trial = evaluate_c5(h_trial)

        while c5_trial > best_c5 and alpha > 1e-8:
            alpha *= 0.5
            h_trial = h - alpha * grad
            h_trial = project_box_sum(h_trial, n_points / 2)
            c5_trial = evaluate_c5(h_trial)

        h = h_trial
        c5 = evaluate_c5(h)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            last_c5 = c5

        iteration += 1

        if iteration % 150 == 0:
            beta = min(beta * 1.5, 200)
        
        if iteration % 200 == 0:
            h = score_guided_swap(h)

        # Early stopping if no improvement for long
        if iteration - last_improvement > 500 and iteration > 2000:
            break

    # Phase IV: Final polishing
    print("Starting Final Polishing Phase...")
    step = 1
    eta0 = 0.01
    while time.time() - start_time < budget_s - 0.1 * budget_s:
        gamma = aperiodic_autocorr(best_h)
        C = 1 - 2 * gamma / n_points
        worst_shifts = np.where(np.abs(C - np.max(C)) < 1e-5)[0]
        if len(worst_shifts) == 0:
            worst_shifts = [np.argmax(C)]
        
        g = np.zeros(n_points)
        for s in worst_shifts:
            if s == 0:
                g += 2 * best_h
            else:
                # shift forward
                g[:-s] += best_h[s:]
                # shift backward
                g[s:] += best_h[:-s]

        g = -(2 / n_points) * g
        eta = eta0 / np.sqrt(step)
        h_new = best_h - eta * g
        h_new = project_box_sum(h_new, n_points / 2)
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        best_h = h_new
        step += 1

    return (best_h, best_c5, n_points)
