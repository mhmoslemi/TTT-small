import numpy as np
import time
from scipy.fft import fft, ifft
import math

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorr(h):
    """Compute periodic autocorrelation using FFT."""
    H = fft(h, len(h))
    ac = np.real_if_close(ifft(H * np.conj(H), len(h)))
    return ac

def compute_correlation_vector(h):
    """Compute full correlation vector based on autocorrelation."""
    n = len(h)
    dx = 2.0 / n
    ac = compute_autocorr(h)
    corr = dx * (n/2 - ac)
    return corr

def binary_anneal(h0, time_limit):
    """Binary simulated annealing for maximizing minimal autocorrelation."""
    n = len(h0)
    dx = 2.0 / n
    T = 5.0 * dx  # Initial temperature
    alpha = 0.9997  # Cooling rate
    n_swap = 0
    max_swap = int(1e6 * (time_limit/200)) if time_limit > 200 else 100000
    
    h = h0.copy()
    ac = compute_autocorr(h)
    c = dx * (n/2 - ac)
    best_c = np.max(c)
    best_h = h.copy()
    stagnation_counter = 0
    last_improvement = 0
    
    start_time = time.time()
    while time.time() - start_time < time_limit:
        # Propose a swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute change in autocorrelation with no need for full recomputation
        delta = np.zeros(n)
        delta[i] = -1
        delta[j] = 1
        delta[(i - np.arange(n)) % n] = -1  # shift by one
        delta[(j - np.arange(n)) % n] = 1  # shift by one
        if i != j:
            delta[(i - j) % n] = 1  # delta for shift from i to j
            delta[(j - i) % n] = 1  # delta for shift from j to i
        
        # Handle self-terms
        delta[0] += 2
        
        ac_new = ac + delta
        c_new = dx * (n/2 - ac_new)
        max_c_new = np.max(c_new)
        
        if max_c_new <= best_c:
            h[i], h[j] = 0, 1
            ac = ac_new
            c = c_new
            best_c = max_c_new
            best_h = h.copy()
            last_improvement = n_swap
        elif np.random.rand() < np.exp(-(max_c_new - best_c) / T):
            h[i], h[j] = 0, 1
            ac = ac_new
            c = c_new
        
        n_swap += 1
        if n_swap % 1000 == 0:
            T *= alpha
        if n_swap - last_improvement > 200000:
            # Large jump
            ones_now = np.where(h == 1)[0]
            to_flip = np.random.choice(ones_now, int(0.05*len(ones_now)), replace=False)
            for idx in to_flip:
                h[idx] = 0
            zeros_now = np.where(h == 0)[0]
            to_flip = np.random.choice(zeros_now, len(to_flip), replace=False)
            for idx in to_flip:
                h[idx] = 1
            ac = compute_autocorr(h)
            c = dx * (n/2 - ac)
            max_c_new = np.max(c)
            if max_c_new < best_c:
                best_c = max_c_new
                best_h = h.copy()
            last_improvement = n_swap
        
        if n_swap >= max_swap:
            break
    
    return best_h, best_c

def smooth_boundaries(h_bin, ramp_width=2):
    """
    Insert fractional ramps at transitions to smoothen steps
    """
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    
    # Find all boundaries (0->1 or 1->0)
    boundary_indices = []
    for i in range(n):
        j = (i+1) % n
        if h_bin[i] != h_bin[j]:
            boundary_indices.append((i, j))
    
    # Insert ramps
    for i, j in boundary_indices:
        if h_bin[i] == 0 and h_bin[j] == 1:
            # 0->1 transition; add ramp at j
            for r in range(1, ramp_width+1):
                k = (j + r) % n
                h[k] = float(r+1) / (ramp_width+1)
        elif h_bin[i] == 1 and h_bin[j] == 0:
            # 1->0 transition; add ramp before j
            for r in range(1, ramp_width+1):
                k = (j - r) % n
                h[k] = float(ramp_width+1-r) / (ramp_width+1)
    
    return project_to_feasible(h, n)

def proj_grad_refine(h0, time_limit):
    """
    Refine by projected gradient descent on the max correlation
    """
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c = evaluate_c5(h)
    
    start_time = time.time()
    step_count = 0
    
    while time.time() - start_time < time_limit:
        # Compute all correlations
        c = compute_correlation_vector(h)
        k_star = np.argmax(c)
        
        # Compute sub-gradient of c_{k*}
        # df/dh_i = -dx * (h_{i-k*} + h_{i+k*})
        g = -dx * (np.roll(h, k_star) + np.roll(h, -k_star))
        
        # Backtracking line search
        alpha = 0.5
        max_iter = 10
        for _ in range(max_iter):
            h_new = h - alpha * g
            h_new = project_to_feasible(h_new, n)
            c_new = evaluate_c5(h_new)
            if c_new < best_c:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                break
            alpha *= 0.5
        
        # Occasional pairwise fractional swaps to escape plateaus
        if step_count % 100 == 0:
            for _ in range(10):
                i, j = np.random.choice(n, size=2, replace=False)
                if h[i] >= 1e-6 and h[j] <= 1 - 1e-6:
                    delta = min(0.01, min(h[i], 1 - h[j]))
                    h[i] -= delta
                    h[j] += delta
                    h = project_to_feasible(h, n)
                    c_new = evaluate_c5(h)
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
        
        step_count += 1
    
    return best_h, best_c

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Budget allocation
    # Stage 0: Initialization = 30s
    n0 = 256
    dx0 = 2.0 / n0
    # Start with a random binary configuration
    h0 = np.random.rand(n0) > 0.5
    h0 = h0.astype(float)
    h0 = project_to_feasible(h0, n0)
    
    # Simple binary heuristic as starting point
    h_bin = h0.copy()
    current_i = 0
    for i in range(n0):
        # Ensure exactly half are ones
        if i < n0 // 2:
            h_bin[i] = 1
        else:
            h_bin[i] = 0
    h_bin = project_to_feasible(h_bin, n0)
    
    t1 = time.time() - start_time
    if t1 > 0.1 * budget_s:
        return (h_bin, evaluate_c5(h_bin), n0)
    
    t_limit1 = 0.55 * budget_s
    h_bin, c5_bin = binary_anneal(h_bin, t_limit1)
    
    if time.time() - start_time > 0.6 * budget_s:
        h_smoother = h_bin.copy()
        c5_smoother = c5_bin
    else:
        # Smoothing and boundary adjustment
        h_smoother = smooth_boundaries(h_bin, ramp_width=2)
        c5_smoother = evaluate_c5(h_smoother)
        
        t_limit2 = 0.25 * budget_s
        h_smoother, c5_smoother = proj_grad_refine(h_smoother, t_limit2)
    
    if time.time() - start_time > 0.8 * budget_s:
        return (h_smoother, c5_smoother, n0)
    
    # Upsampling
    n1 = 2 * n0  # Double resolution
    dx1 = 2.0 / n1
    h_interp = np.interp(np.linspace(0, 2, n1), np.linspace(0, 2, n0), h_smoother)
    h_upsampled = project_to_feasible(h_interp, n1)
    
    # Quick gradient polish
    t_limit3 = 0.15 * budget_s
    h_final, c5_final = proj_grad_refine(h_upsampled, t_limit3)
    
    return (h_final, c5_final, len(h_final))
