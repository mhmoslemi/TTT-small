def project_box_sum(y, target_sum):
    """Project y onto {x : 0 ≤ x ≤ 1, sum x = target_sum} using water filling."""
    y = np.asarray(y)
    low, high = 0.0, 1.0
    while high - low > 1e-12:
        mid = (low + high) / 2.0
        x = np.clip(y - mid, 0, 1)
        if np.sum(x) > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(y - low, 0, 1)

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorr(h):
    """Returns full autocorrelation array of h with itself."""
    return correlate(h, h, mode='full')

def compute_gamma_min_from_autocorr(gamma_full):
    """Return min of autocorrelation values."""
    n = (len(gamma_full) + 1) // 2
    # Take the center slice (shifts from -n+1 to n-1)
    gamma = gamma_full[n-1:n*2-1]
    return np.min(gamma)

def full_gamma_update(h, gamma, i, j, delta):
    """Update gamma in O(n) after transfer from i to j."""
    n = len(h)
    # Create temporary vector with h[i] reduced by delta
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta

    # The update is linear for all shifts except self term
    gamma_new = gamma.copy()
    for s in range(-n+1, n):
        idx = s + n - 1
        # redundancy is handled in the loop logic
        if 0 <= i + s < n:
            gamma_new[idx] -= delta * h[i+s]
        if 0 <= i - s < n and s != 0:
            gamma_new[idx] -= delta * h[i-s]

        if 0 <= j + s < n:
            gamma_new[idx] += delta * h[j+s]
        if 0 <= j - s < n and s != 0:
            gamma_new[idx] += delta * h[j-s]

    # Self-term correction (s == 0)
    gamma_new[n-1] += -2 * delta * h[i] + delta * delta + 2 * delta * h[j] + delta * delta

    return gamma_new

def score_sources(h, gamma, eps=1e-5):
    """Compute scores for source indices based on worst shifts."""
    n = len(h)
    gamma_full = gamma
    gamma = gamma_full[n-1:n*2-1]
    C = 1 - (2.0/n) * gamma
    worst_indices = np.where(np.abs(C - np.max(C)) < eps)[0]
    scores = np.zeros(n)
    for s in worst_indices:
        s_shift = s - n + 1  # actual shift in range -(n-1) to (n-1)
        # Score contribution = sum of neighbors across all worst shifts
        for i in range(n):
            if 0 <= i + s_shift < n:
                scores[i] += h[i + s_shift]
            if 0 <= i - s_shift < n and s_shift != 0:
                scores[i] += h[i - s_shift]
    return scores

def get_best_source(h, scores, threshold=0.1, eta=1e-8):
    """Select source with highest score."""
    valid = np.where(h > eta)[0]
    if len(valid) == 0:
        return -1
    # Get top N
    top_indices = valid[np.argsort(-scores[valid])[:max(1, len(valid)//10)]]
    return np.random.choice(top_indices)

def get_best_target(h, scores, threshold=0.1, eta=1e-8):
    """Select target with lowest score."""
    valid = np.where(h < 1 - eta)[0]
    if len(valid) == 0:
        return -1
    # Get bottom N
    bottom_indices = valid[np.argsort(scores[valid])[:max(1, len(valid)//10)]]
    return np.random.choice(bottom_indices)
