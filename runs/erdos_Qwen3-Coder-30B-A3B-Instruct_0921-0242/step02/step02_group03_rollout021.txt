import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    # Re-clip after adjustment
    h_copy = np.clip(h_copy, 0, 1)

    return h_copy

def evaluate_correlations(h):
    """Evaluate correlations efficiently using FFT and return full correlation array."""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant', constant_values=0)
    
    # Using FFT for efficient cross-correlation
    corr_freq = fft(h_padded) * np.conj(fft(one_minus_h_padded))
    corr_vals = ifft(corr_freq).real
    corr_vals = corr_vals[:n]  # Take only the non-cyclic part
    # Normalize by dx
    dx = 2.0 / n
    corr_vals = corr_vals * dx
    return corr_vals

def update_correlations_fast(corr_vals, h, i, j):
    """Efficiently update correlations when swapping h[i] and h[j]."""
    n = len(h)
    # Compute change in contributions at each shift
    delta = np.zeros(n)
    for s in range(n):
        if i + s < n:
            delta[s] += -(1 - h[i + s]) + h[i + s]  # contribution removed from i
        if j + s < n:
            delta[s] += (1 - h[j + s]) - h[j + s]   # contribution added from j
        if i - s >= 0:
            delta[s] += -(1 - h[i - s]) + h[i - s]  # contribution removed from i (reverse shift)
        if j - s >= 0:
            delta[s] += (1 - h[j - s]) - h[j - s]   # contribution added from j (reverse shift)
        # For forward shift s = 0, we don't double-count
        if s == 0:
            pass
        
    # Update full correlation array
    corr_vals += delta
    return corr_vals

def anneal_binary(h, n, max_moves=60000, T0=0.02 * 221, alpha=0.997, time_limit=None):
    """Simulated annealing for binary balanced sequences."""
    dx = 2.0 / n
    corr_vals = evaluate_correlations(h)
    current_energy = np.max(corr_vals)
    
    # Let's preserve binary nature and the exact value
    h = h.astype(np.int8)
    best_h = h.copy()
    best_energy = current_energy
    
    # Track the last L swaps to avoid tabu
    tabu_list = []
    tabu_size = 100
    
    T = T0
    moves = 0
    last_improvement = 0
    while moves < max_moves and T > 1e-5:
        if time_limit and time.time() > time_limit:
            break
        
        # Pick two positions to swap: one 1 and one 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Check tabu list
        if (i, j) in tabu_list:
            continue
            
        # Create new state with swapped positions
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        # Update correlations efficiently
        corr_vals_new = corr_vals.copy()
        update_correlations_fast(corr_vals_new, h_new, i, j)
        new_energy = np.max(corr_vals_new)
        
        # Accept or not
        if new_energy <= current_energy:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_energy - current_energy) / T)
                accept = np.random.rand() < prob
            else:
                accept = False
                
        if accept:
            h = h_new
            corr_vals = corr_vals_new
            current_energy = new_energy
            moves += 1
            
            # Update tabu list
            tabu_list.append((i, j))
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
                
            # Check for improvement
            if current_energy < best_energy:
                best_energy = current_energy
                best_h = h.copy()
                last_improvement = moves
                
        else:
            moves += 1
            if moves - last_improvement > 5000:
                break

    return best_h, best_energy

def run(seed=42, budget_s=600, **kwargs):
    assert budget_s == 600, "Budget must be 600s"
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s - 10  # Keep 10s safety margin
    
    # Use an odd number of points for better balance
    n = 221
    dx = 2.0 / n
    best_h = None
    best_c5 = float('inf')
    
    trials = 12
    
    for trial in range(trials):
        if time.time() >= time_limit:
            break
            
        # Initialize with block or Wiener
        # Use boundary sweep for better construction
        h = np.zeros(n, dtype=np.int8)
        mid = n // 2
        # Different initializations:
        if np.random.rand() > 0.5:
            # Block pattern is common
            h[mid - n//4:mid + n//4] = 1  # A contiguous block
        else:
            # Random shuffle of ones and zeros
            h[:n//2] = 1
            np.random.shuffle(h)
            
        # Annealing
        h_local, energy = anneal_binary(h, n, time_limit=time_limit)
        
        # Polish
        h_polished, c5 = polish_continuous(h_local, n)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_polished.copy()
    
    if best_h is None:
        # Fallback solution if no good one found
        best_h = np.ones(n) * 0.5
        best_c5 = evaluate_c5(best_h)
        
    return (best_h, best_c5, n)
