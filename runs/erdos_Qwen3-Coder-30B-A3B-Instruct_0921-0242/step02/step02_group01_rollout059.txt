```python
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.linalg import toeplitz
import cvxpy as cp

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 256  # Even number for spectral factorization
    dx = 2.0 / n
    eps = 2e-4  # Tolerance for bisection
    
    # SDP bisection on t in [t_low, t_high]
    t_low, t_high = 0.0, 1.0
    best_r = None
    
    # Time limit for SDP phase
    sdp_time_limit = 0.35 * budget_s
    
    while (t_high - t_low > eps) and (time.time() - start_time < sdp_time_limit):
        t_mid = (t_low + t_high) / 2
        
        # Build SDP
        r = cp.Variable(n)
        constraints = [
            r[0] == 1,
            r[1:] >= t_mid,
            r <= 1
        ]
        # Construct Toeplitz matrix
        T = toeplitz(r)
        constraints.append(T >> 0)  # PSD constraint
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            best_r = r.value
            t_low = t_mid
        else:
            t_high = t_mid
    
    if best_r is None:
        # Fallback if bisection failed
        best_r = np.ones(n)
        best_r[0] = 1
        best_r[1:] = t_low
    
    # Convert r to h using spectral factorization
    S = np.fft.fft(best_r)
    S = np.maximum(S, 0)
    H = np.sqrt(S)
    
    # Inverse FFT (real part only)
    g = np.real(np.fft.ifft(H))
    
    # Project to feasible region
    h = project_to_feasible(g, n)
    
    # Deterministic worst-shift gradient descent
    max_iter = 250
    alpha0 = 0.5 * dx
    
    # Compute initial overlaps
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    Ck = np.real(corr) * dx  # Full correlation
    
    for iter_num in range(max_iter):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Find worst shift
        Ck_no_self = Ck[1:n]  # Exclude shift 0
        worst_shift = np.argmax(Ck_no_self) + 1
        
        # Compute gradient
        grad = np.zeros(n)
        for i in range(n):
            left_idx = (i + worst_shift) % n
            right_idx = (i - worst_shift) % n
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Update with step size
        alpha = alpha0 / np.sqrt(iter_num + 1) if iter_num > 0 else alpha0
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n)
        
        # Evaluate new solution
        h = h_new
        h_fft = fft(h, 2 * len(h))
        ih_fft = fft(1 - h, 2 * len(h))
        corr = ifft(h_fft * np.conj(ih_fft))
        Ck = np.real(corr) * dx  # Recomputed overlaps
        
        # Stop if improvement negligible
        # Note: we're not tracking previous, but we do it for the last few steps.
    
    # Stochastic polishing
    max_polish_iter = 3000
    last_improvement_time = time.time()
    
    for _ in range(max_polish_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
        
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        
        # Only allow transfer between valid indices
        if h[i] <= 0 or h[j] >= 1:
            continue
            
        delta = min(0.01, h[i], 1 - h[j])
        if delta <= 0:
            continue
            
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        h_new = project_to_feasible(h_new, n)
        
        c5_old = evaluate_c5(h)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5_old:
            h = h_new
            last_improvement_time = time.time()
    
    # Final C5 evaluation
    final_c5 = evaluate_c5(h)
    
    # Upsample if time permits
    if time.time() - start_time < 0.99 * budget_s:
        upsample_factor = 2
        n_up = n * upsample_factor
        dx_up = 2.0 / n_up
        x_upsample = np.linspace(0, 2, n_up, endpoint=False)
        h_upsampled = np.interp(x_upsample, np.linspace(0, 2, n, endpoint=False), h)
        h_upsampled = project_to_feasible(h_upsampled, n_up)
        
        # Short gradient descent on finer grid
        fine_max_iter = 50
        fine_alpha0 = 0.5 * dx_up
        for iter_num in range(fine_max_iter):
            if time.time() - start_time > budget_s:
                break
                
            h_fft = fft(h_upsampled, 2 * len(h_upsampled))
            ih_fft = fft(1 - h_upsampled, 2 * len(h_upsampled))
            corr = ifft(h_fft * np.conj(ih_fft))
            Ck = np.real(corr) * dx_up

            # Find worst shift
            Ck_no_self = Ck[1:n_up]
            worst_shift = np.argmax(Ck_no_self) + 1

            grad = np.zeros(n_up)
            for i in range(n_up):
                left_idx = (i + worst_shift) % n_up
                right_idx = (i - worst_shift) % n_up
                grad[i] = dx_up * ((1 - h_upsampled[left_idx]) - h_upsampled[right_idx])

            alpha = fine_alpha0 / np.sqrt(iter_num + 1) if iter_num > 0 else fine_alpha0
            h_upsampled_new = h_upsampled - alpha * grad
            h_upsampled_new = project_to_feasible(h_upsampled_new, n_up)
            h_upsampled = h_upsampled_new

        h = h_upsampled
        final_c5 = evaluate_c5(h)
        
    return (h, final_c5, len(h))
```
