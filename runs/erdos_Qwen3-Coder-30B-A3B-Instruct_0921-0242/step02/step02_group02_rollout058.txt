import numpy as np
import time
from scipy.optimize import differential_evolution, linprog
from scipy.fft import rfft, irfft
from scipy.interpolate import interp1d

def project_to_feasible(h, n):
    """Ensure sum(h) = n/2 with h in [0,1] using water-filling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) < 1e-10:
        return h
    # Find lambda such that sum(h + lambda) = n/2 with clipping
    lb, ub = -1.0, 1.0
    for _ in range(50):
        mid = (lb + ub) / 2
        h_test = np.clip(h + mid, 0, 1)
        if np.sum(h_test) < n/2:
            ub = mid
        else:
            lb = mid
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    """Computes C5 value from sampled h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def basis_evaluation(n, M):
    """Generate cos/sin basis vectors for a grid of size n."""
    x = np.linspace(0, 2, n, endpoint=False)  # [0, 2)
    cos_basis = np.array([np.cos(m * np.pi * x) for m in range(1, M + 1)])
    sin_basis = np.array([np.sin(m * np.pi * x) for m in range(1, M + 1)])
    return cos_basis, sin_basis

def coefficients_to_h(coefficients, cos_basis, sin_basis, n):
    """Construct h from coefficients using basis functions."""
    M = len(cos_basis)
    a0 = 0.5  # because integral must be 1 and all basis functions have 0 mean
    # Reshape coefficients into am, bm pairs
    am = coefficients[:M]
    bm = coefficients[M:]
    
    # Evaluate h(x)
    h_raw = a0 + np.dot(am, cos_basis) + np.dot(bm, sin_basis)
    
    # Clip and renormalize
    h = np.clip(h_raw, 0, 1)
    h = project_to_feasible(h, n)
    return h

def fitness(coefficients, cos_basis, sin_basis, n, M):
    """Evaluate fitness of coefficient vector."""
    h = coefficients_to_h(coefficients, cos_basis, sin_basis, n)
    return evaluate_c5(h)

def linearise_and_solve_lp(h, n, k_star, A_vals, max_constraints=10):
    """Solve linear program to improve h by linearizing worst shift constraint."""
    # Get indices of the worst constraints
    # Also include a few constraints around k_star to weaken the model
    indices = np.argsort(A_vals)[-max_constraints:]
    
    A_eq = np.array([1]*n)
    b_eq = n/2
    
    # Constraints: A(k) >= t for selected k
    # A(k) = h^T P_k h ≈ A(k_0) + grad(k_0)^T (h - h0)
    # = A(k_0) + 2 * (h_0 >> k) + (h_0 << k) ! if shifted in circular fashion
    rows = []
    b_ineq = []
    
    for k in indices:
        # Estimate gradient of A(k) at h
        h_shifted = np.roll(h, k)
        h_reversed_shifted = np.roll(h, -k)
        grad = 2 * (h_shifted + h_reversed_shifted)
        
        rows.append(grad)
        b_ineq.append(A_vals[k])  # Need A(k) >= t, so b = A(k)

    A_ub = -np.array(rows)
    b_ub = -np.array(b_ineq)
    
    # Variable bounds (h in [0, 1], t in [-inf, inf])
    bounds = [(0, 1) for _ in range(n)] + [(-np.inf, np.inf)]  # t is last var

    c = np.zeros(n + 1)
    c[-1] = -1  # maximize t
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success:
            solution = res.x[:-1]  # h values
            return solution
    except:
        pass
    return h  # fallback

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Dedicate phases of the budget
    n0 = 250  # low resolution grid
    M = 12    # basis functions (12 pairs)
    d = 2 * M # dimension
    pop = 15 * d  # population for DE
    
    cos_basis, sin_basis = basis_evaluation(n0, M)
    
    global_best_h = None
    global_best_c5 = float('inf')

    # Phase A: Basis preparation (minimal effort)
    pass  # already done above

    # Phase B: DE
    euclidean_tol = 1e-3
    def de_fitness(c):
        return fitness(c, cos_basis, sin_basis, n0, M)

    # Use a wide bound for coefficient search, but not too extreme
    bounds = [(-0.5, 0.5)] * d
    de_result = differential_evolution(de_fitness, bounds, seed=seed, maxiter=3000,
                                       popsize=pop, disp=False, workers=1)
    
    # Phase C: Top-k distinct solutions
    top_candidates = [de_result.x]
    top_c5 = [de_result.fun]
    
    # Select 5 (or fewer) distinct candidates
    candidates = [de_result.x]
    for i in range(30):
        if len(candidates) >= 5:
            break
        c = np.random.uniform(-0.5, 0.5, d)
        min_dist = min(np.linalg.norm(c - prev, ord=2) for prev in candidates)
        if min_dist > euclidean_tol:
            candidates.append(c)
    
    # Phase D: Sequential linear programming refinement
    max_iter = 300
    time_left = budget_s * 0.3  # allocate time to phase D
    for i, candidate in enumerate(candidates):
        if time.time() - start_time > budget_s - 1:
            break
        h0 = coefficients_to_h(candidate, cos_basis, sin_basis, n0)
        h_best = h0.copy()
        c5_best = evaluate_c5(h_best)

        # Loop of SLP Steps
        t0 = time.time()
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            # Evaluate A_vals
            fft_h = rfft(h_best)
            A_vals = irfft(fft_h * np.conj(fft_h), n0)
            A_vals[0] = n0 / 2  # Ensure the energy constraint (0-shift)
            
            # Find k* corresponding to min A(k)
            k_star = np.argmin(A_vals)
            
            # Linearize and solve LP
            h_new = linearise_and_solve_lp(h_best, n0, k_star, A_vals)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
            else:
                # Backtrack step or quit
                break
            
            if time.time() - t0 > time_left:  # Limit refinement time
                break

        if c5_best < global_best_c5:
            global_best_c5 = c5_best
            global_best_h = h_best

    # Phase E: Upsampling and fine-tuning
    if global_best_h is not None:
        n = 800  # higher resolution grid
        x_ref = np.linspace(0, 2, n0, endpoint=False)
        x_up = np.linspace(0, 2, n, endpoint=False)
        
        # Periodic cubic interpolation
        f_interp = interp1d(x_ref, global_best_h, kind='cubic', bound_error=False, period=2)
        h_up = f_interp(x_up)
        
        # Renormalize and clip
        h_up = project_to_feasible(h_up, n)
        
        # Fine-tune with subgradients
        t0 = time.time()
        for _ in range(2000):
            if time.time() - start_time > budget_s - 0.5:
                break
            corr_vals = np.correlate(h_up, 1 - h_up, mode='full')
            k_star = np.argmax(corr_vals)
            # subgradient step
            h_up[k_star] = max(0, h_up[k_star] - 0.01 * (np.prod(h_up[k_star])))
            h_up = project_to_feasible(h_up, n)
            
        global_best_h = h_up
        global_best_c5 = evaluate_c5(global_best_h)
    
    return (global_best_h, global_best_c5, len(global_best_h))
