import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def project_to_feasible(h, target_sum):
    """Ensure that sum(h) = target_sum, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def generate_seeds(n, target_sum):
    """Generate multiple starting points for optimization."""
    seeds = []
    
    # 1. Uniform
    h_uniform = np.ones(n) * 0.5
    seeds.append(project_to_feasible(h_uniform, target_sum))
    
    # 2. Random Dirichlet (sample from uniform distribution and normalize)
    h_dirichlet = np.random.rand(n)
    h_dirichlet = h_dirichlet * target_sum / np.sum(h_dirichlet)
    seeds.append(project_to_feasible(h_dirichlet, target_sum))
    
    # 3. Binary with some smoothing
    h_binary = np.zeros(n)
    h_binary[:int(target_sum)] = 1.0
    np.random.shuffle(h_binary)
    h_binary = h_binary * target_sum / np.sum(h_binary)
    seeds.append(project_to_feasible(h_binary, target_sum))
    
    # 4. Interpolate initial_h_values if available

        seeds.append(project_to_feasible(h_interp, target_sum))
    
    # 5. Sobol-like sampling (simple random in simplex)
    h_sobol = np.random.rand(n)
    h_sobol = h_sobol * target_sum / np.sum(h_sobol)
    seeds.append(project_to_feasible(h_sobol, target_sum))
    
    return seeds

def build_lp_data(h0):
    """Build the linear program data for the CCP approach."""
    n = len(h0)
    # Compute auto-correlation to get C0_k
    h_fft = fft(h0, 2 * n)
    corr_auto = ifft(h_fft * np.conj(h_fft))
    c0 = corr_auto.real[:n]
    
    # Construct W matrix
    W = np.zeros((n, n))
    for k in range(n):
        W[k, :] = h0[(np.arange(n) + k) % n] + h0[(np.arange(n) - k) % n]
    
    # LP data
    A = -W
    b = -n/2 + c0
    
    return A, b, c0

def solve_ccp_step(h0, target_sum, max_iter=30, eps=1e-8):
    """Solve one CCP iteration step."""
    n = len(h0)
    A, b, c0 = build_lp_data(h0)
    
    # LP problem:
    # min  t
    # s.t. A*x + t*1 <= b  (one row per shift)
    #      0 <= x <= 1
    #      sum(x) = target_sum
    #      t >= 0
    
    # Add constraints as a list
    c = np.zeros(n + 1)
    c[-1] = 1  # minimize t

    # Equality constraint for sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :-1] = 1
    b_eq = np.array([target_sum])
    
    # Inequality constraints for shifting
    A_ub = A
    b_ub = b
    
    # Bound constraints
    bounds = [(0, 1)] * n + [(0, None)]  # (min, max) for each variable

    # Solve using high quality solver
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs', options={'presolve': True})
    
    if result.success:
        h_new = result.x[:-1]
        t_new = result.x[-1]
        return h_new, t_new
    else:
        return None, None

def projected_gradient_descent(h, target_sum, max_iter):
    """Projected gradient descent for fine tuning."""
    dx = 2.0 / len(h)
    for iter in range(max_iter):
        # Compute full correlation
        h_fft = fft(h, 2 * len(h))
        ih_fft = fft(1 - h, 2 * len(h))
        corr = ifft(h_fft * np.conj(ih_fft))
        corr_vals = corr.real[1:len(h)]  # exclude self correlation
        worst_k = np.argmin(corr_vals) + 1
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            left_idx = (i + worst_k) % len(h)
            right_idx = (i - worst_k) % len(h)
            grad[i] = dx * ((1 - h[left_idx]) - h[right_idx])
        # Step
        alpha = 0.01 / np.sqrt(iter + 1)
        h_new = h - alpha * grad
        # Project to feasible region
        h_new = project_to_feasible(h_new, target_sum)
        h = h_new
        
        if iter % 10 == 0:
            c5_new = evaluate_c5(h)
            if c5_new > 0.001:  # Early stopping if improvement stops
                pass
            else:
                break
    return h

def simulated_annealing(h, target_sum, max_iter, time_limit):
    """Polish solution with simulated annealing.""" 
    start_time = time.time()
    dx = 2.0 / len(h)
    current_c5 = evaluate_c5(h)
    temperature = 0.05
    
    for i in range(max_iter):
        if time.time() - start_time >= time_limit:
            break
        # Pick two indices
        i1 = np.random.randint(0, len(h))
        i2 = np.random.randint(0, len(h))
        if h[i1] <= 0 or h[i2] >= 1:
            continue
        # Move small amount
        delta = np.random.uniform(0, min(h[i1], 1 - h[i2], 0.01))
        h_test = h.copy()
        h_test[i1] -= delta
        h_test[i2] += delta
        h_test = project_to_feasible(h_test, target_sum)
        c5_test = evaluate_c5(h_test)
        if c5_test < current_c5:
            h = h_test
            current_c5 = c5_test
        else:
            # Accept with probability
            prob = np.exp(-(c5_test - current_c5) / temperature)
            if np.random.rand() < prob:
                h = h_test
                current_c5 = c5_test
        temperature *= 0.995
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup
    n_points = 400
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Generate seeds
    seeds = generate_seeds(n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    # Run CCP loops for all seeds
    for i, h0 in enumerate(seeds):
        if time.time() - start_time > 0.7 * budget_s:
            break
        
        h_current = h0.copy()
        c5_prev = evaluate_c5(h_current)
        
        for iter in range(30):  # max iterations
            if time.time() - start_time > 0.7 * budget_s:
                break
            
            h_new, t_new = solve_ccp_step(h_current, target_sum)
            if h_new is None:
                break
                
            h_new = project_to_feasible(h_new, target_sum)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            if abs(c5_new - c5_prev) < 1e-8:
                break
                
            h_current = h_new
            c5_prev = c5_new
    
    # If no good seed is found, start with uniform
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        best_h = project_to_feasible(best_h, target_sum)
        best_c5 = evaluate_c5(best_h)
    
    # Refine with projected gradient descent
    if time.time() - start_time < 0.9 * budget_s:
        best_h = projected_gradient_descent(best_h, target_sum, max_iter=200)
        best_c5 = evaluate_c5(best_h)
    
    # Polish with simulated annealing
    if time.time() - start_time < 0.99 * budget_s:
        best_h = simulated_annealing(best_h, target_sum, max_iter=5000,
                                    time_limit=0.09 * budget_s)
        best_c5 = evaluate_c5(best_h)
    
    # Optional upsampling and refinement
    if time.time() - start_time < 0.995 * budget_s:
        # Upsample to higher resolution
        n_up = 800
        target_up = n_up / 2
        x_orig = np.linspace(0, 2, n_points)
        x_up = np.linspace(0, 2, n_up)
        h_up = np.interp(x_up, x_orig, best_h)
        h_up = project_to_feasible(h_up, target_up)
        # CCP refinement for small number of iterations
        for _ in range(5):
            h_new, t_new = solve_ccp_step(h_up, target_up)
            if h_new is None:
                break
            h_up = project_to_feasible(h_new, target_up)
            best_c5 = evaluate_c5(h_up)
            
        best_h = h_up
        best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, n_points)
