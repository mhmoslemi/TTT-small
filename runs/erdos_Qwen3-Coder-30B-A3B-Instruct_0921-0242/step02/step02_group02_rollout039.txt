import numpy as np
import time
import cvxpy as cp
from scipy.linalg import circulant
from scipy.signal import correlate
from scipy.fft import rfft, irfft

def build_cyclic_shift_matrix(k, n):
    """Build a cyclic shift matrix P_k for shift k."""
    p = np.zeros(n)
    p[(np.arange(n) - k) % n] = 1
    return p

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def subgradient_refine(h, dx, budget_s):
    """Refine h using projected subgradient descent."""
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Fast FFT-based correlation calculation
    n = len(h)
    H = rfft(h)
    auto_ref = irfft(np.abs(H)**2, n)
    c_ref = dx * ( n/2 - auto_ref )
    k_star = np.argmax(c_ref)
    
    while time.time() - start < budget_s - 0.1:
        # Compute gradient for this shift
        grad = np.zeros_like(h)
        grad = np.roll(h, k_star) - np.roll(h, -k_star)
        grad *= -dx  # descent direction
        
        # Line search
        alpha = 1.0
        max_iter = 10
        for _ in range(max_iter):
            h_new = h + alpha * grad
            h_new = project_to_feasible(h_new, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
                break
            alpha /= 2.0
        else:
            # Line search failed, break
            break
            
        # Recompute to find new k_star
        H = rfft(h)
        auto_ref = irfft(np.abs(H)**2, n)
        c_ref = dx * ( n/2 - auto_ref )
        k_star = np.argmax(c_ref)
        
        # Early stopping if not improving
        if alpha < 1e-6 or alpha * np.linalg.norm(grad) < 1e-6:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # 1. Choose n, build shift matrices
    n = 180
    dx = 2.0 / n
    P = [build_cyclic_shift_matrix(k, n) for k in range(n)]
    
    # 2. SDP
    X = cp.Variable((n,n), PSD=True)
    t = cp.Variable()
    constraints = [
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n/2,
    ] + [cp.trace(P[k] @ X) >= t for k in range(n)]
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver='SCS', max_iters=25000, eps=1e-5)
    except Exception:
        # fallback to a smaller size if needed
        n = 150
        dx = 2.0 / n
        P = [build_cyclic_shift_matrix(k, n) for k in range(n)]
        X = cp.Variable((n,n), PSD=True)
        t = cp.Variable()
        constraints = [
            cp.diag(X) >= 0,
            cp.diag(X) <= 1,
            cp.sum(cp.diag(X)) == n/2,
        ] + [cp.trace(P[k] @ X) >= t for k in range(n)]
        prob = cp.Problem(cp.Maximize(t), constraints)
        prob.solve(solver='SCS', max_iters=10000, eps=1e-4)
    
    # 3. Extract candidates
    X_opt = X.value
    if X_opt is None: 
        # fallback if SCS failed
        h0 = np.ones(n) * 0.5 
        h0 = project_to_feasible(h0, n)
        return h0, evaluate_c5(h0), n
    
    eigvals, eigvecs = np.linalg.eigh(X_opt)
    lead = eigvecs[:, -1] * np.sqrt(eigvals[-1])
    h0 = project_to_feasible(lead, n)
    
    # 4. Evaluate and keep best of a few random draws
    best_h, best_c5 = h0, evaluate_c5(h0)
    
    for _ in range(10):
        if time.time() - start_time >= 0.8 * budget_s:
            break
        g = np.random.randn(n) @ np.linalg.cholesky(X_opt)
        h = project_to_feasible(np.clip(g,0,1), n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # 5. Sub-gradient refinement (time-controlled)
    remaining_budget = budget_s - (time.time() - start_time) - 0.5
    if remaining_budget > 0.5:
        best_h, best_c5 = subgradient_refine(best_h, dx, remaining_budget)
    
    # 6. Upsample if time remains
    # Only upsample if still have time
    if time.time() - start_time < 0.8 * budget_s:
        n2 = 2 * n
        h_up = np.interp(np.linspace(0, 2, n2), np.linspace(0, 2, n), best_h)
        h_up = project_to_feasible(h_up, n2)
        remaining_budget = budget_s - (time.time() - start_time) - 0.5
        if remaining_budget > 0.5:
            best_h, best_c5 = subgradient_refine(h_up, 2.0/n2, remaining_budget)
    
    return best_h, best_c5, len(best_h)
