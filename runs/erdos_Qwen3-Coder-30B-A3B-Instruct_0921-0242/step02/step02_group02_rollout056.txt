import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def evaluate_c5_exact(h):
    """Exact fft-based evaluation"""
    n = len(h)
    dx = 2.0 / n
    H = fft(h, n)
    autocorr = ifft(H * np.conj(H)).real
    c = dx * (n/2 - autocorr)
    return np.max(c)

def binary_anneal(h, time_limit):
    """Perform simulated annealing for binary patterns."""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    
    def update_autocorr(ac, i, j, h):
        """Update autocorrelation vector after a single bit-swap."""
        # delta vector for h_i=1, h_j=0 turning into h_i=0, h_j=1
        delta = np.zeros(n)
        delta[i] = -1
        delta[j] = 1
        # Add self terms for shift 0, up to 2
        # For shift k: ac_new[k] = ac[k] + h[i-k] + h[j+k] - h[i+k] - h[j-k]
        # For shift 0: -1 +1 = 0, so no self term
        # But we need to handle the coefficients of 1s at positions where 
        # h_i appears or h_j appears. Easiest is to update all shifts directly.
        delta_at_k = np.zeros(n)
        # Simplified approach: compute the full update vector manually
        for k in range(n):
            h_i_k = h[(i - k) % n]
            h_j_k = h[(j - k) % n]
            h_i_k_plus = h[(i + k) % n]
            h_j_k_plus = h[(j + k) % n]
            delta_at_k[k] = (h_j_k + h_i_k_plus) - (h_i_k + h_j_k_plus)
        
        # The formula for autocorr change in single swap is as follows:
        # ac' = ac + delta + self_terms
        # self terms including shift 0
        self_terms = np.zeros(n)
        self_terms[0] = 2  # h_i h_{i} + h_j h_{j} = 1*1 + 0*0 = 1 -> new 0*0 + 1*1 = 1, net change=0
        # Actually: figure this more carefully as:
        # The change in ac[k] is due to h[i] h[i-k] and h[j] h[j-k] changing
        # h[i] h[i-k] goes from 1 to 0, so subtract 1 h[i-k]
        # h[j] h[j-k] goes from 0 to 1, so add 1 h[j-k] 
        # But these are not both flipped at the same time in general.
        # Should be: 
        # For entry k:
        #   if i-k is j, then -h[j]
        #   if j-k is i, then +h[i] 
        #   if i+k is j, then -h[j]
        #   if j+k is i, then +h[i] 
        # If that's not the pattern we need a more direct accounting
        # Need to recompute the effect of change of h[i] from 1 to 0 and h[j] from 0 to 1
        
        # Construct delta_at_k based on autocorrelation updates
        delta_at_k = np.zeros(n)
        # Flip h[i] from 1 to 0, h[j] from 0 to 1
        # This means we flip sign of h[i] at shift 0 (self)
        # and then for shifts k, the term h[i]*h[i-k] flips into 0 from 1 (subtract h[i-k])
        # and h[j]*h[j-k] flips from 0 to 1 (add h[j-k])
        # Similarly for terms involving horizontal swaps...
        # Use a clean form:
        for k in range(n):
            if i == (j + k) % n:
                delta_at_k[k] -= h[i]  # unsufficient
            if i == (j - k) % n:
                delta_at_k[k] = h[j]
            # and similar for h[i] h[i-k] flipping to 0
            delta_at_k[k] -= h[(i - k) % n]
            delta_at_k[k] += h[(j - k) % n]
            # Special case for k=0: still a point-here
            if k == 0:
                delta_at_k[k] += h[i] - h[j]  # no self flip on stack
            # Better to get individual values:
        # Cleaner derivation:
        # ac_new[k] = ac[k] - h[i] h[i-k] - h[j] h[j-k] + h[i] h[i-k] (XOR) + h[j] h[j-k]
        # * happens first at position i (or at i+k or i-k) the difference depends on k etc...
        # Best: Compute actual changes as differences from the direct update formula:
        
        # Use Thuemmler-style update:
        # ac_new[k] = ac[k] + (h'[i] h'[i-k] - h[i] h[i-k]) + (h'[j] h'[j-k] - h[j] h[j-k])
        # = ac[k] - h[i] h[i-k] + h[j] h[j-k] + h'[i] h'[i-k] - h'[j] h'[j-k]
        # For a bit swap i->0 j->1: 
        # entirely new: h'[i] =0, h'[j] =1
        # So:  
        #   h'[i] h'[i-k] = 0 if i!=k or = h'[i-k] = 0 if i==k  
        #   h'[j] h'[j-k] = h[j-k] if j==k else 0
        # so change in autocorr:  
        #  cases of i=0 for h[i] h[i-k] -> 0
        # and j=1 for h[j] h[j-k] -> h[j-k]
        
        # So the net change per k:
        # - h[i] h[i-k] (0) + h[j] h[j-k] (1) = 0 + h[j] h[j-k] - h[i] h[i-k].  
        #     (This is just the correct delta for i or j being fixed)
        # But if j-k = i  then h[j] h[j-k] = h[j] h[i] = 0*1 = 0, so net improvement  
        # manually form update
        delta_at_k = np.zeros(n)
        # [i+k] in i, and [j-k] in j
        # at a concrete k
        # current h[i]h[i-k] and h[j]h[j-k] 
        # after swap h[i]=0, h[j]=1 
        # new h[i]h[i-k] = 0  
        # new h[j]h[j-k] = h[j-k] (i.e. h[j_skip])
        # Change = - h[i] h[i-k] + h[j] h[j-k] - 0 + h[j-k] = -h[i] h[i-k] + h[j] h[j-k] + h[j-k]
        # = - h[i] h[i-k] + h[j-k] * (h[j] + 1) = - h[i] h[i-k] + h[j-k] * 2  (if h[j]==1, i.e. original present)
        # So:
        # delta[k] = hole1[k] + hole2[k]  
        # where hole1[k] is -h[i] h[i-k] = -h[i] h[i-k] if bit = 1 anyway
        #      hole2[k] is +h[j] h[j-k] + h[j-k] = h[j-k] (1 + h[j]) = h[j-k] (1 + 1) = 2 * h[j-k]
        # this gives stale if h[i] or h[j] changed. Let's just iterate manually...
        # Simplify formula, especially easy to do what's below exactly:
        # rewrite for correctness:
        for k in range(n):
            new_h = h.copy()
            new_h[i] = 0
            new_h[j] = 1
            ac_new = ifft(fft(new_h) * np.conj(fft(new_h))).real
            # So change is ac_new - ac
            # delta_at_k[k] = ac_new[k] - ac[k]
            delta_at_k[k] = ac_new[k] - ac[k]

        return delta_at_k

    # Start with original binary
    ac = ifft(fft(h) * np.conj(fft(h))).real
    current_c5 = np.max(dx * (n / 2 - ac))
    best_c5 = current_c5
    best_h = h.copy()
    
    # Annealing schedule
    T = 5.0 * dx
    alpha = 0.9997
    max_swaps = 200000
    temp_cooldown = 1000
    swaps = 0
    
    # Shuffling good candidates
    indices_1 = np.where(h == 1)[0]
    indices_0 = np.where(h == 0)[0]
    rand_indices_1 = np.random.permutation(indices_1)
    rand_indices_0 = np.random.permutation(indices_0)
    
    # reset all permutation structures when we're too stuck
    stagnation_count = 0
    last_improvement = 0
    stagnation_idx = 0
    
    while time.time() - start_time < time_limit:
        t_elapsed = time.time() - start_time
        if t_elapsed > time_limit - 0.1:
            break
        
        # Swap a 1 with a 0
        if len(indices_1) > 0 and len(indices_0) > 0:
            i = indices_1[np.random.randint(len(indices_1))]
            j = indices_0[np.random.randint(len(indices_0))]
            
            original = h.copy()
            h[i] = 0
            h[j] = 1
            
            # Compute change in autocorrelation
            new_ac = ifft(fft(h) * np.conj(fft(h))).real
            new_c5 = np.max(dx * (n / 2 - new_ac))
            
            # Accept or not based on simulated annealing
            if new_c5 <= current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
                # Accept
                ac = new_ac
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    last_improvement = swaps
            else:
                # Reject
                h[:] = original[:]  # restore
                
            # Adjust T and swap counter
            swaps += 1
            if swaps % temp_cooldown == 0:
                T *= alpha
            
            # Reset stagnation fields
            if swaps - last_improvement > 100000:
                stagnation_count += 1
                if stagnation_count > 2:
                    # Force a large jump: randomly reassign 5%  
                    num_changes = int(0.05 * n)
                    ones_to_change = np.random.choice(n, size=num_changes, replace=False)
                    for idx in ones_to_change:
                        if h[idx] == 1:
                            h[idx] = 0
                        else:
                            h[idx] = 1
                    # Recalculate ac
                    ac = ifft(fft(h) * np.conj(fft(h))).real
                    current_c5 = np.max(dx * (n / 2 - ac))
                    last_improvement = swaps
                    stagnation_count = 0  # reset
            
            # rewrite indices after swapping to maintain good tracking
            indices_1 = np.where(h == 1)[0]
            indices_0 = np.where(h == 0)[0]
            
        else:
            # Small adjustments just in case
            # recompute and try to make it work
            break
            
    return best_h, best_c5

def smooth_boundaries(h_bin, ramp_width=2):
    """Convert binary h to a step function with ramped boundaries."""
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    
    # Scan the binary pattern and insert ramps for 0->1 transitions
    # roughly: for every boundary, create ramp of width ramp_width
    
    # First identify transitions 0 -> 1 and 1 -> 0
    transitions = []
    for i in range(n):
        i_prev = (i - 1) % n
        prev_val = h_bin[i_prev]
        curr_val = h_bin[i]
        if prev_val == 0 and curr_val == 1:
            # 0 -> 1 transition found at index i
            transitions.append(('0to1', i))
        elif prev_val == 1 and curr_val == 0:
            # 1 -> 0 transition found at index i
            transitions.append(('1to0', i))
            
    # create ramps
    for typename, start_idx in transitions:
        # For 0 -> 1 at start_idx, go from 0 to 1 in regions
        if typename == '0to1':
            # smooth from start_idx-ramp_width to start_idx
            for r in range(1, ramp_width + 1):
                pos = (start_idx - r) % n
                h[pos] = r / (ramp_width + 1)
        elif typename == '1to0':
            # smooth from start_idx to start_idx+ramp_width
            for r in range(1, ramp_width + 1):
                pos = (start_idx + r) % n
                h[pos] = 1 - r / (ramp_width + 1)
    
    # Project to ensure constraint sum(h) = n/2
    h = project_to_feasible(h, n)
    return h

def proj_grad_refine(h, time_limit):
    """Perform projected sub-gradient descent using the 'true max' method."""
    start_time = time.time()
    n = len(h)
    dx = 2.0 / n
    
    def subgradient(h, k):
        """Compute sub-gradient of C5 at shift k."""
        g = -dx * (np.roll(h, k) + np.roll(h, -k))
        return g
    
    def line_search(h, g, max_iter=10):
        alpha = 1.0
        start_c5 = evaluate_c5(h)
        for _ in range(max_iter):
            h_trial = h - alpha * g
            h_trial = project_to_feasible(h_trial, n)
            trial_c5 = evaluate_c5(h_trial)
            if trial_c5 < start_c5:
                return h_trial, trial_c5
            else:
                alpha *= 0.5
        return h, start_c5
    
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    max_steps = 10000
    iter_count = 0
    
    while time.time() - start_time < time_limit and iter_count < max_steps:
        # Step 1: Find the worst shift
        corr = correlate(h, 1 - h, mode="full")
        k_star = np.argmax(corr)
        # Get actual index in [0, n)
        if k_star >= n:
            k_star = k_star - n
        else:
            k_star = k_star - (n - 1)  # nc = n start at -n+1 to n -1
        
        # Fix k_star to be in correct range [-n+1, n-1]
        if k_star >= n // 2:
            k_star -= n
        elif k_star < -n // 2:
            k_star += n
        
        # Compute subgradient
        g = subgradient(h, k_star)
        
        # Perform line search
        h_new, c5_new = line_search(h, g)
        
        if c5_new < current_c5:
            h = h_new
            current_c5 = c5_new
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        iter_count += 1
        
        t = time.time() - start_time
        if t > time_limit - 0.1:
            break
            
    return best_h, best_c5
