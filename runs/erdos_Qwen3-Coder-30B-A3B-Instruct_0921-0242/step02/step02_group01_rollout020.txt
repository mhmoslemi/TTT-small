import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate C5 using the provided FFT approach to compute overlaps."""
    dx = 2.0 / len(h)
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    max_corr = np.max(corr.real) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def autocorr_k_binary(x, k, n):
    """Computes R_k for a binary vector x, using periodic indexing"""
    return np.sum(x * np.roll(x, k))

def core_autocorrelations(x):
    """Computes all R_k for k from 1 to n-1 using FFT (faster for vectorized computation)"""
    n = len(x)
    x_fft = fft(x, 2 * n)
    x_conj_fft = np.conj(x_fft)
    corr = ifft(x_fft * x_conj_fft)
    # Extract all R_k (excluding k=0)
    autocs = corr.real[1:n]
    return autocs

def compute_autocorrelations(x, n):
    """Compute autocorrelations for all shifts k from 1 to n-1 efficiently."""
    return core_autocorrelations(x)

def find_worst_shift(autocorrelations):
    """Given the autocorrelations of all shifts, return the index of the smallest."""
    return np.argmin(autocorrelations)

def compute_swap_impact(autocorr_worst, x, i, j, n):
    """Compute change in autocorr_worst if we swap x[i]=1 with x[j]=0"""
    # x_i changes from 1 to 0, x_j changes from 0 to 1
    delta = 0
    # All k such that autocorr_k is affected
    for k in range(1, n):
        # Old contribution of i to R_k (before swap)
        old_i_k = x[(i+k) % n]
        # New contribution of i to R_k (after swap)
        new_i_k = 0
        # Old contribution of j to R_k (before swap)
        old_j_k = x[(j+k) % n]
        # New contribution of j to R_k (after swap)
        new_j_k = 1
        # Change in R_k
        delta += (new_i_k + new_j_k) - (old_i_k + old_j_k)
    return delta

def greedy_swap_refinement(x, n, max_iter=10000):
    """Improve a binary solution using swap operations on the worst shift."""
    # Copy to avoid modifying input
    x = x.copy()
    
    # Early termination based on time budget
    start_time = time.time()
    
    limits = 1000000  # Prevent infinite loops even with negligible safety

    iterations = 0
    improvement_count = 0
    
    while iterations < limits and time.time() - start_time < 180:
        iterations += 1

        # Compute all R_k values
        autocs = core_autocorrelations(x)
        # Skip k=0 (no self-correlation)
        worst_k_idx = find_worst_shift(autocs)

        if worst_k_idx is None:
            break

        # Compute contribution of each 1 to worst_k
        contribution_of_ones = np.zeros(n)
        for i in range(n):
            if x[i] == 1:
                contribution_of_ones[i] = x[(i + worst_k_idx) % n] + x[(i - worst_k_idx) % n]

        # Compute contribution of each 0 to worst_k
        contribution_of_zeros = np.full(n, np.inf)
        for j in range(n):
            if x[j] == 0:
                contribution_of_zeros[j] = x[(j + worst_k_idx) % n] + x[(j - worst_k_idx) % n]

        # Find best 1 (MAX contribution to its R_worst) and worst 0 (MIN contribution)
        good_ones = np.where(x == 1)[0]
        good_ones_contributions = [contribution_of_ones[i] for i in good_ones]
        best_one_idx = good_ones[np.argmax(good_ones_contributions)]
        
        bad_zeros = np.where(x == 0)[0]
        bad_zeros_contributions = [contribution_of_zeros[j] for j in bad_zeros]
        best_zero_idx = bad_zeros[np.argmin(bad_zeros_contributions)]

        # If swapping helps, apply the swap
        if contribution_of_zeros[best_zero_idx] < contribution_of_ones[best_one_idx]:
            x[best_one_idx] = 0
            x[best_zero_idx] = 1
            
            improvement_count += 1
        else:
            break  # No more improvement possible

    return x

def random_round(X, n, R=30):
    """Random hyperplane rounding of SDP matrix X."""
    # Eigen-decomposition
    w, v = np.linalg.eigh(X)
    if w[-1] < 0:
        # Make sure all eigenvalues are non-negative
        v = v[:, w > 0]
        w = w[w > 0]
    v = np.column_stack([v[:, -1]])  # principal component
    e = np.sqrt(w[-1])
    v = v * e
    
    best_x = None
    best_t = -np.inf
    
    for _ in range(R):
        g = np.random.randn(n)
        z = v.T @ g  # Use the largest eigenvector direction
        x = np.zeros(n)
        x[np.argpartition(-z, n//2)[:n//2]] = 1
        
        # Check if it's better
        autocs = core_autocorrelations(x)
        min_autocorr = np.min(autocs)
        if min_autocorr > best_t:
            best_t = min_autocorr
            best_x = x.astype(int)
            
    if best_x is not None:
        # Apply swap refinement to the best rounded solution
        refined = greedy_swap_refinement(best_x, n, max_iter=2000)
        return refined
    else:
        # Fallback if no good rounding found
        return np.random.choice([0, 1], size=n, p=[0.5, 0.5])

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Determine n to work with
    n = 200
    if n % 2 != 0:
        n -= 1  # Ensure it's even
    
    dx = 2.0 / n
    
    # Define SDP relaxation problem
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints:
    # 1. X is positive semidefinite
    constraints = [X >> 0]
    
    # 2. Diagonal bounds (0 <= X_ii <= 1)
    for i in range(n):
        constraints.append(X[i, i] >= 0)
        constraints.append(X[i, i] <= 1)
    
    # 3. Cardinality constraint: sum of diagonal = n/2
    constraints.append(cp.sum(X.diagonal()) == n / 2)
    
    # 4. Autocorrelation bounds (R_k >= t, for all k ≠ 0)
    A = np.zeros((n, n))
    shift_index = []
    for k in range(1, n):
        A = np.zeros((n, n))
        for i in range(n):
            idx_right = (i + k) % n
            A[i, idx_right] = 1
        shift_index.append((k, A))
    
    for k, A in shift_index:
        trace = cp.trace(A @ X)
        constraints.append(trace >= t)
    
    # Optional pairwise bounds
    for i in range(n):
        for j in range(i):
            constraints.append(X[i, j] >= 0)
            constraints.append(X[i, j] <= X[i, i])
            constraints.append(X[i, j] <= X[j, j])
    
    # Objective: Maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve SDP
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, eps=1e-6)
    except:
        # fallback in case of problem
        print("SDP solve failed; falling back to random.")
        best_x = np.random.choice([0,1], size=n, p=[0.5, 0.5])
        best_x = greedy_swap_refinement(best_x, n)
        c5 = evaluate_erdos_solution(best_x)
        return (best_x, c5, n)
    
    # Get the solution
    X_val = X.value
    t_val = t.value
    
    # Perform rounding
    x = random_round(X_val, n, R=30)
    
    # L1 local search refinement using Greedy Swap
    x = greedy_swap_refinement(x, n, max_iter=10000)
    
    # Final evaluation
    c5 = evaluate_erdos_solution(x)
    
    # Output as required
    h_values = x.astype(float)
    return (h_values, c5, n)
