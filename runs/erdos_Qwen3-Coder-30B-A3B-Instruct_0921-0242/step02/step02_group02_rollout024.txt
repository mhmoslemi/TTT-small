import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft
from scipy.interpolate import interp1d
import cvxpy as cp

def build_cyclic_shift_matrix(k, n):
    """Build the cyclic shift matrix P_k where (P_k)_{ij} = 1 iff j = i-k (mod n)"""
    P = np.zeros((n, n))
    for i in range(n):
        j = (i - k) % n
        P[i, j] = 1
    return P

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    # Find optimal constant shift lambda such that projection sums to n/2
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Simple bisection to find lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def subgradient_refine(h, dx, budget_s):
    """Use projected subgradients to refine h."""
    start = time.time()
    h = h.copy()
    current_c5 = evaluate_c5(h)

    max_iter = 5000
    # Just a quick check to get off the ground
    final_iter = min(5000, max_iter)
    for _ in range(final_iter):
        if time.time() - start > budget_s - 0.5:
            break
            
        # Compute full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Compute c = dx * ( n/2 - auto )
        c = dx * ( len(h)/2 - np.correlate(h, h, mode='full') )
        k_star = np.argmax(c)
        
        if k_star > len(h):
            k_star = k_star - len(h)
        else:
            k_star = k_star - len(h) + 1
            
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            left_idx = (i + k_star) % len(h)
            right_idx = (i - k_star) % len(h)
            grad[i] = dx * (h[left_idx] - h[right_idx])
            
        # Perform backtrack line search
        alpha = 1.0
        alpha_min = 1e-10
        
        while alpha >= alpha_min:
            h_new = h - alpha * grad
            h_new = project_to_feasible(h_new, len(h))
            c5_new = evaluate_c5(h_new)
            
            if c5_new < current_c5 - 1e-12:
                h = h_new
                current_c5 = c5_new
                break
            alpha /= 2

    return h, current_c5

def upsample_and_project(h, n_new):
    """Upsample h from length n to n_new"""
    n = len(h)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(np.linspace(0, 2, n, endpoint=False), h, kind='linear', 
                 fill_value='extrapolate')
    h_up = f(x_new)
    h_up = h_up * (n_new / 2) / np.sum(h_up)
    return h_up

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    n = 180
    dx = 2.0 / n
    P = [build_cyclic_shift_matrix(k, n) for k in range(n)]

    # SDP formulation
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()
    constraints = [
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n/2,
    ] + [cp.trace(P[k] @ X) >= t for k in range(n)]
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    # Solve the SDP
    try:
        prob.solve(solver='SCS', max_iters=25000, eps=1e-5)
    except Exception:
        # Fallback to faster solver
        try:
            prob.solve(solver='ECOS', max_iters=5000, eps=1e-4)
        except Exception:
            # Worst case: fallback to small size SDP
            n = 150
            dx = 2.0 / n
            P = [build_cyclic_shift_matrix(k, n) for k in range(n)]
            X = cp.Variable((n, n), PSD=True)
            t = cp.Variable()
            constraints = [
                cp.diag(X) >= 0,
                cp.diag(X) <= 1,
                cp.sum(cp.diag(X)) == n/2,
            ] + [cp.trace(P[k] @ X) >= t for k in range(n)]
            prob = cp.Problem(cp.Maximize(t), constraints)
            prob.solve(solver='SCS', max_iters=10000, eps=1e-4)
    
    X_opt = X.value
    if X_opt is None:
        # Default construct with uniform
        h = np.ones(n) * 0.5
        h = project_to_feasible(h, n)
        c5 = evaluate_c5(h)
        return (h, c5, n)
    
    # Compute eigenvalues/vectors
    try:
        eigvals, eigvecs = np.linalg.eigh(X_opt)
        # Take the leading eigenvector
        lead = eigvecs[:, -1] * np.sqrt(eigvals[-1])
        h0 = lead
    except:
        # Fallback if eig fails
        h0 = np.mean(X_opt, axis=0)

    # Project onto feasible set
    h0 = project_to_feasible(h0, n)
    h0 = np.clip(h0, 0, 1)

    # Evaluate candidate
    c5_current = evaluate_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_current

    # Try several randomizations
    for i in range(20):
        if time.time() - start_time > budget_s - 1.0:
            break
        # Add a random sample from the covariance
        g = np.random.randn(n) @ np.linalg.cholesky(X_opt)
        h = project_to_feasible(np.clip(g, 0, 1), n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Local refinement with subgradient descent
    time_left = budget_s - (time.time() - start_time)
    if time_left > 1.0:
        best_h, best_c5 = subgradient_refine(best_h, dx, time_left)

    # Optionally upsample if time allows
    if time.time() - start_time < 0.85 * budget_s:
        n_high = min(2 * n, 400)
        h_up = upsample_and_project(best_h, n_high)
        if n_high > 200:
            h_up = project_to_feasible(h_up, n_high)
        time_left = budget_s - (time.time() - start_time)
        if time_left > 1.0:
            best_h, best_c5 = subgradient_refine(h_up, 2.0 / n_high, time_left)

    return (best_h, best_c5, len(best_h))
