import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_to_sum_constraint(h, target_sum):
    """Project h to have sum = target_sum with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    
    # Simple water-filling approach
    h = h + diff / len(h)
    h = np.clip(h, 0, 1)
    return h

def evaluate_c5(h):
    """Evaluate C5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Use FFT for efficient cross-correlation
    h_fft = fft(h, 2 * n)
    ih_fft = fft(1 - h, 2 * n)
    corr = ifft(h_fft * np.conj(ih_fft))
    # Extract positive shifts only, ignoring shift 0
    corr_vals = np.real(corr)[1:n]
    return np.max(corr_vals) * dx

def gradient_worst_shift_descent(h, max_iter=200, alpha0=0.1):
    """Refine h using gradient descent on the worst shift."""
    n = len(h)
    dx = 2.0 / n
    alpha = alpha0 * dx
    
    for i in range(max_iter):
        # Compute all correlations
        h_fft = fft(h, 2 * n)
        ih_fft = fft(1 - h, 2 * n)
        corr = ifft(h_fft * np.conj(ih_fft))
        corr_vals = np.real(corr)[1:n]
        worst_shift = np.argmax(corr_vals)
        
        # Compute gradient for worst shift
        grad = np.zeros(n)
        for j in range(n):
            # h_i contributes to correlation with h_{i+k}
            left_idx = (j + worst_shift) % n
            # h_{i-k} contributes
            right_idx = (j - worst_shift) % n
            grad[j] = dx * ((1 - h[left_idx]) - h[right_idx])
        
        # Gradient step
        h_new = h - alpha * grad
        h_new = project_to_sum_constraint(h_new, n/2)
        
        # Check for convergence
        if np.linalg.norm(h_new - h) < 1e-8:
            break
        h = h_new
        alpha *= 0.995  # Decaying step size
    
    return h

def differential_evolution(h_pop, target_sum, pop_size=40, F=0.6, CR=0.9, generations=1000):
    """Differential Evolution with proper projection."""
    n = len(h_pop[0])
    for gen in range(generations):
        for i in range(pop_size):
            # Select three distinct individuals
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            # Generate donor vector
            v = h_pop[a] + F * (h_pop[b] - h_pop[c])
            # Clip to [0,1]
            v = np.clip(v, 0, 1)
            # Enforce sum constraint
            v = project_to_sum_constraint(v, target_sum)
            
            # Binomial crossover
            j_rand = np.random.randint(n)
            u = np.copy(h_pop[i])
            for j in range(n):
                if np.random.rand() < CR or j == j_rand:
                    u[j] = v[j]
            
            # Evaluate and accept if better
            c5_old = evaluate_c5(h_pop[i])
            c5_new = evaluate_c5(u)
            if c5_new < c5_old:
                h_pop[i] = u
    return h_pop

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 400
    target_sum = n_points / 2
    pop_size = min(40, 2 * n_points // 10)
    F = 0.6
    CR = 0.9
    max_de_generations = int(budget_s * 0.75 / (0.001 * pop_size)) if pop_size > 0 else 1000
    
    # Initialize population
    if 'initial_h_values' in globals() and len(initial_h_values) >= n_points:
        orig_h = initial_h_values[:n_points]
    else:
        orig_h = np.ones(n_points) * 0.5
    
    h_pop = []
    # Add real initial point
    h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(orig_h)), orig_h)
    h_interp = project_to_sum_constraint(h_interp, target_sum)
    h_pop.append(h_interp)
    
    # Fill with random samples
    for _ in range(pop_size - len(h_pop)):
        h = np.random.rand(n_points)
        h = project_to_sum_constraint(h, target_sum)
        h_pop.append(h)
    
    h_pop = np.array(h_pop)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    # Differential Evolution main loop
    last_improvement = time.time()
    for gen in range(max_de_generations):
        if time.time() - start_time > 0.75 * budget_s:
            break
        h_pop = differential_evolution(h_pop, target_sum, pop_size, F, CR, 1)
        
        # Update best solution
        for i in range(pop_size):
            c5 = evaluate_c5(h_pop[i])
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_pop[i].copy()
                last_improvement = time.time()
        
        # Occasionally refine the best solution with gradient descent
        if gen % 200 == 0 and best_h is not None:
            if time.time() - start_time > 0.75 * budget_s:
                break
            h_refined = gradient_worst_shift_descent(best_h.copy(), max_iter=50)
            c5_refined = evaluate_c5(h_refined)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined
    
    # Early termination if no improvement yet
    if best_h is None:
        best_h = h_pop[0].copy()
        best_c5 = evaluate_c5(best_h)
    
    # Post-process with local refinement if time permits
    if time.time() - start_time < 0.9 * budget_s and best_h is not None:
        h_refined = gradient_worst_shift_descent(best_h.copy(), max_iter=200)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined
    
    # Final stochastic swap polishing
    if time.time() - start_time < 0.99 * budget_s and best_h is not None:
        h_final = best_h.copy()
        c5_final = evaluate_c5(h_final)
        # Annealing parameters
        temp = 0.01
        decay = 0.995
        min_temp = 1e-6
        
        while temp > min_temp and time.time() - start_time < 0.995 * budget_s:
            for _ in range(100):
                i, j = np.random.randint(0, n_points, 2)
                if h_final[i] <= 0 or h_final[j] >= 1:
                    continue
                delta = np.random.uniform(0, min(h_final[i], 1 - h_final[j], 0.01))
                h_test = h_final.copy()
                h_test[i] -= delta
                h_test[j] += delta
                h_test = project_to_sum_constraint(h_test, target_sum)
                c5_test = evaluate_c5(h_test)
                if c5_test < c5_final or np.random.rand() < np.exp(-(c5_test - c5_final) / temp):
                    c5_final = c5_test
                    h_final = h_test
            temp *= decay
    
    if 'h_final' in locals():
        return (h_final, c5_final, n_points)
    else:
        return (best_h, best_c5, n_points)
