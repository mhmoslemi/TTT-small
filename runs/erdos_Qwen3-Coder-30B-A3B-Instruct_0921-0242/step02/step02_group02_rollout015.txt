```python
import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.fft import rfft, irfft

def build_cyclic_shift_matrix(k, n):
    """Builds the cyclic shift matrix P_k for k-th shift."""
    P = np.zeros((n, n))
    for i in range(n):
        j = (i - k) % n
        P[i, j] = 1
    return P

def project_to_feasible(h, n):
    """Project h to feasible set: 0 <= h[i] <= 1, sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-12:
        return h
    
    # Water-filling approach
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Find the index where adding diff would cause overflow
    idx = np.searchsorted(cumsum + np.arange(1, n+1) * diff, n/2, side="right")
    
    # Adjust all elements above idx
    if idx > 0:
        lambda_ = (n/2 - cumsum[idx-1]) / (n - idx)
        h = np.clip(h + lambda_, 0, 1)
    else:
        lambda_ = diff / n
        h = np.clip(h + lambda_, 0, 1)
    
    # Normalize to ensure exact sum
    h = h * (n / 2) / np.sum(h)
    
    return h

def evaluate_c5(h):
    """Evaluates C5 = max_k int h(x)(1 - h(x+k)) dx."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def subgradient_refine(h, dx, budget_s):
    """Performs projected sub-gradient descent to refine h."""
    start_time = time.time()
    max_iter = 5000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1e-3:
            break
        # Compute exact correlation
        H = rfft(h)
        auto = irfft(np.abs(H)**2, len(h))
        c = dx * (len(h)/2 - auto)
        k_star = np.argmax(c)
        
        # Compute gradient direction
        grad = dx * (np.roll(h, k_star) - np.roll(h, -k_star))
        
        # Backtracking linesearch
        alpha = 1.0
        while alpha > 1e-10:
            h_new = h - alpha * grad
            h_new = project_to_feasible(h_new, len(h))
            if evaluate_c5(h_new) < evaluate_c5(h):
                h = h_new
                break
            alpha /= 2.0
    
    return h, evaluate_c5(h)

def upsample_and_project(h_old, n_new):
    """Upsample h_old to n_new points and re-project."""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    
    # Linear interpolation
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    
    # Re-project to sum = n_new/2
    h_new = project_to_feasible(h_new, n_new)
    
    return h_new

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # SDP stage
    n = 180
    dx = 2.0 / n
    P = [build_cyclic_shift_matrix(k, n) for k in range(n)]

    # Define SDP variables
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.diag(X) >= 0,
        cp.diag(X) <= 1,
        cp.sum(cp.diag(X)) == n/2,
    ] + [cp.trace(P[k] @ X) >= t for k in range(n)]

    # Objective
    objective = cp.Maximize(t)
    problem = cp.Problem(objective, constraints)

    # Solve
    problem.solve(solver='SCS', max_iters=25000, eps=1e-5)
    X_opt = X.value
    t_opt = t.value

    # Extract candidates via spectral factorization
    eigvals, eigvecs = np.linalg.eigh(X_opt)
    lead_vec = eigvecs[:, -1] * np.sqrt(eigvals[-1])
    h0 = project_to_feasible(lead_vec, n)

    # Select best of several random samples
    best_h = h0
    best_c5 = evaluate_c5(h0)

    # Multiple randomizations with different starting points
    for _ in range(15):
        if time.time() - start_time > budget_s - 2.0:
            break
        g = np.random.randn(n) @ np.linalg.cholesky(X_opt)
        h = project_to_feasible(np.clip(g,0,1), n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Local refinement
    remaining_time = budget_s - (time.time() - start_time) - 1.0
    if remaining_time > 0.1:
        best_h, best_c5 = subgradient_refine(best_h, dx, remaining_time)

    # Upsampling and final polish
    remaining_time = budget_s - (time.time() - start_time) - 1.0
    if remaining_time > 0.1:
        h_up = upsample_and_project(best_h, 2 * n)
        best_h, best_c5 = subgradient_refine(h_up, 2.0 / (2*n), remaining_time)

    return (best_h, best_c5, len(best_h))
```
