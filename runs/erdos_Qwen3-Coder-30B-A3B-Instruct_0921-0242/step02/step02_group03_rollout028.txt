import numpy as np
import cvxpy as cp
import time

def full_to_sym(h_full):
    """Convert full vector to symmetric representation."""
    n = len(h_full)
    return h_full[:n//2 + 1]

def sym_to_full(h_sym):
    """Convert symmetric representation to full vector."""
    n = 2 * (len(h_sym) - 1)
    h_full = np.empty(n)
    h_full[:len(h_sym)] = h_sym
    h_full[len(h_sym):] = h_sym[-2::-1]
    return h_full

def build_constraints(x, t, shift_set, n):
    """Build constraints for the LP."""
    constraints = []
    dx = 2.0 / n
    
    for s in shift_set:
        # Compute linear constraint for shift s
        # x_i * (1 - x_{i+s}) for i = 0,...,n-1-s
        # We compute the sum explicitly
        expr = 0
        for i in range(n - s):
            expr += x[i] * (1 - x[i+s])
        expr *= dx
        constraints.append(expr <= t)
    
    return constraints

def solve_lp(shift_set, x0=None, n=400):
    """Solve the LP for given shift set."""
    # Create symmetric variables
    m = n // 2 + 1
    x = cp.Variable(m, nonneg=True)
    t = cp.Variable()
    
    # Constraints: 0 <= x <= 1
    constraints = [x <= 1, x >= 0]
    
    # Integral constraint
    dx = 2.0 / n
    # full h = [x[0], ..., x[m-1], x[m-2], ..., x[1]]
    full_coeffs = np.concatenate([x, x[-2:0:-1]])
    integral_constraint = cp.sum(full_coeffs) * dx == 1.0
    constraints.append(integral_constraint)
    
    # Add correlation constraints
    constraints.extend(build_constraints(x, t, shift_set, n))
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        # Solve with ECOS
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_sol = x.value
            if x0 is not None and not np.isnan(x_sol).any():
                x.value = x0
                prob.solve(solver=cp.ECOS, verbose=False, max_iters=500)
                x_sol = x.value
            return True, x_sol, t.value
        else:
            return False, None, None
    except Exception:
        return False, None, None

def evaluate_c5_full(h):
    """Compute full correlation for h."""
    n = len(h)
    dx = 2.0 / n
    # Full linear correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # We only care about non-negative shifts (as in task instruction)
    corr_vals = corr[n-1:] * dx  # Only positive shifts
    return np.max(corr_vals)

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h

    # Water-filling method to adjust sum
    if diff > 0:
        # Need to increase sum
        sorted_indices = np.argsort(h)
        h_copy = h.copy()
        for index in sorted_indices:
            if diff <= 0:
                break
            max_add = 1.0 - h_copy[index]
            add = min(max_add, diff)
            h_copy[index] += add
            diff -= add
    else:
        # Need to decrease sum
        sorted_indices = np.argsort(h)[::-1]
        h_copy = h.copy()
        for index in sorted_indices:
            if diff >= 0:
                break
            max_subtract = h_copy[index]
            subtract = max(-diff, max_subtract)
            h_copy[index] -= subtract
            diff += subtract

    h_copy = np.clip(h_copy, 0, 1)
    return h_copy
