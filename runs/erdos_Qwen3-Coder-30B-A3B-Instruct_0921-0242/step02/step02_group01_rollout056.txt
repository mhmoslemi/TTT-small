import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluates the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def compute_autocorrelations(x):
    """Compute autocorrelations using FFT."""
    n = len(x)
    # Use real FFT for efficiency
    fft_x = fft(x, 2 * n)
    corr = ifft(fft_x * np.conj(fft_x))
    return corr.real[:n]

def swap_improvement_update(R, x, k_star, i, j):
    """Efficiently update autocorrelations after swapping x[i] and x[j]."""
    # Get the effective change in autocorrelations
    # x_{i+k}, x_{i-k}, x_{j+k}, x_{j-k}
    n = len(x)
    # Update contributions
    # Remove old contributions
    for k in range(1, n):
        # indices
        i_k = (i + k) % n
        i_m = (i - k) % n
        j_k = (j + k) % n
        j_m = (j - k) % n
        # Update R_k
        R[k] -= x[i] * x[i_k] + x[i_m] * x[i]
        R[k] += x[j] * x[j_k] + x[j_m] * x[j]
    return R

def greedy_swap_improvement(x, R, n_points, budget_s, start_time):
    """Perform greedy swaps to improve worst autocorrelation."""
    # Ensure we work on a copy to avoid side effects
    x = x.copy()
    R = R.copy()
    # Stores shift k* that corresponds to min R_k
    k_star = np.argmin(R[1:]) + 1  # excluding index 0
    iter_count = 0
    iter_limit = 20000  # Conservative limit for iteration count
    while iter_count < iter_limit and time.time() - start_time < budget_s * 0.7:
        iter_count += 1
        # Find worst shift k_star
        k_star = np.argmin(R[1:]) + 1  # excluding index 0

        # Get contributions c_i for all positions i with x_i = 1
        c_i = np.zeros(n_points)
        d_j = np.zeros(n_points)
        for i in range(n_points):
            if x[i] == 1:
                # Indices involved
                i_k = (i + k_star) % n_points
                i_m = (i - k_star) % n_points
                c_i[i] = x[i_k] + x[i_m]
        
        for j in range(n_points):
            if x[j] == 0:
                # Indices involved
                j_k = (j + k_star) % n_points
                j_m = (j - k_star) % n_points
                d_j[j] = x[j_k] + x[j_m]
        
        # Find best i and worst j for swap
        # We want i where c_i contributes most to worsening
        i_star = np.argmax(c_i)
        if x[i_star] == 0:
            continue  # Safety check

        j_star = np.argmin(d_j)
        if x[j_star] == 1:
            continue
        # Check if it's beneficial
        if d_j[j_star] > c_i[i_star]:
            # Swap
            x[i_star] = 0
            x[j_star] = 1
            # Update correlations
            R = swap_improvement_update(R, x, k_star, i_star, j_star)
        else:
            break
    return x, R

def random_perturbation_swap(x, budget_s, start_time):
    """Perform a random swap to escape plateaus."""
    if time.time() - start_time > budget_s * 0.95:
        return x

    n = len(x)
    # Random 1, 0 pair
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return x

    i = np.random.choice(ones)
    j = np.random.choice(zeros)

    x = x.copy()
    x[i] = 0
    x[j] = 1
    return x

def continuous_polishing(h, budget_s, start_time):
    """Perform projected gradient descent to improve fractional solution."""
    if time.time() - start_time > budget_s * 0.95:
        return h
    dx = 2.0 / len(h)
    # Initial iteration parameters
    alpha = 0.5 * dx
    max_iter = 200
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Find worst shift k*
        h_fft = fft(h, 2 * len(h))
        ih_fft = fft(1 - h, 2 * len(h))
        corr = ifft(h_fft * np.conj(ih_fft))
        corr_vals = corr.real[1:]  # exclude index 0
        worst_shift = np.argmin(corr_vals) + 1

        # Compute gradient
        grad = np.zeros_like(h)
        for k in range(len(h)):
            left_idx = (k + worst_shift) % len(h)
            right_idx = (k - worst_shift) % len(h)
            grad[k] = dx * ((1 - h[left_idx]) - h[right_idx])

        # Take step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)

        # Project to constraint
        current_sum = np.sum(h_new)
        diff = len(h) / 2 - current_sum
        h_new = h_new + diff / len(h)
        h_new = np.clip(h_new, 0, 1)

        # If accepted, update
        c5_old = evaluate_erdos_solution(h)
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < c5_old:
            h = h_new.copy()
        else:
            alpha *= 0.95  # Reduce step size
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Step 1: Prepare
    n_points = 200  # Lower for faster runtime and to fit budget
    dx = 2.0 / n_points

    # Step 2: SDP Relaxation
    # Formulate the SDP:
    # Maximize t subject to:
    # X is psd
    # X_{ii} in [0,1] for all i
    # sum X_{ii} = n/2
    # trace(A_k * X) >= t for all k=1..n-1 where A_k has ones in positions (i, i+k)
    # X_ij >= 0, X_ij <= X_ii, X_ij <= X_jj

    # Construct the problem
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [X >> 0]  # PSD
    for i in range(n_points):
        constraints.append(X[i,i] >= 0)
        constraints.append(X[i,i] <= 1)
    constraints.append(cp.sum(X.diagonal()) == n_points / 2)
    
    # For each shift k=1 to n/2 (exclude 0)
    for k in range(1, n_points):
        # Construction of matrix A_k
        A_k = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = (i + k) % n_points
            A_k[i,j] = 1
        # trace(A_k * X) >= t
        constraints.append(cp.trace(A_k @ X) >= t)

    # Pairwise bounds
    for i in range(n_points):
        for j in range(i+1, n_points):
            constraints.append(X[i,j] >= 0)
            constraints.append(X[i,j] <= X[i,i])
            constraints.append(X[i,j] <= X[j,j])

    # Objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    # Try to solve SDP
    solve_start = time.time()
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
    except Exception as e:
        pass  # fallback to binary at end
    
    if prob.status != 'optimal' or time.time() - solve_start > 0.3 * budget_s:
        # If SDP did not solve or takes too long, fallback
        # Use random binary construction
        x = np.random.rand(n_points)
        x = (x > 0.5).astype(int)  # Almost binary
        # Ensure sum = n_points/2
        current_sum = np.sum(x)
        diff = n_points / 2 - current_sum
        # Adjust by swapping as needed
        while abs(diff) > 0.1:
            if diff > 0:
                # We need to make some 0s into 1s
                zero_indices = np.where(x == 0)[0]
                if len(zero_indices) > 0:
                    idx = np.random.choice(zero_indices)
                    x[idx] = 1
                diff = n_points / 2 - np.sum(x)
            else:
                # Make some 1s into 0s
                one_indices = np.where(x == 1)[0]
                if len(one_indices) > 0:
                    idx = np.random.choice(one_indices)
                    x[idx] = 0
                diff = n_points / 2 - np.sum(x)
        # Now convert to h
        h = x.astype(float)
        h = project_to_feasible(h, n_points)
    else:
        # Extract the solution
        X_opt = X.value
        t_opt = t.value
        if X_opt is None or np.any(np.isnan(X_opt)):
            # Fallback to random construction anyway
            x = np.random.rand(n_points)
            x = (x > 0.5).astype(int)
            # Ensure sum = n_points/2
            current_sum = np.sum(x)
            diff = n_points / 2 - current_sum
            # Adjust by swapping as needed
            while abs(diff) > 0.1:
                if diff > 0:
                    zero_indices = np.where(x == 0)[0]
                    if len(zero_indices) > 0:
                        idx = np.random.choice(zero_indices)
                        x[idx] = 1
                    diff = n_points / 2 - np.sum(x)
                else:
                    one_indices = np.where(x == 1)[0]
                    if len(one_indices) > 0:
                        idx = np.random.choice(one_indices)
                        x[idx] = 0
                    diff = n_points / 2 - np.sum(x)
            h = x.astype(float)
            h = project_to_feasible(h, n_points)
        else:
            # Round to binary vector using principal component
            # Eigen decomposition
            try:
                _, V = np.linalg.eigh(X_opt)
                if len(V) > 0:
                    v = V[:, -1]  # Principal component (largest eigenvalue)
                    # Use random hyperplane rounding with 30 samples
                    best_min = -1
                    best_x = None
                    for _ in range(30):
                        g = np.random.randn(n_points)
                        z = V @ (np.sqrt(np.linalg.eigvals(X_opt)) * g)
                        x = np.zeros_like(z)
                        idx = np.argpartition(z, -int(n_points/2))[-int(n_points/2):]
                        x[idx] = 1.0
                        # Ensure sum constraint
                        current_sum = np.sum(x)
                        diff = n_points / 2 - current_sum
                        if diff > 0:
                            # Needs to increase to 1s
                            diff_indices = np.where(x == 0)[0]
                            if len(diff_indices) >= int(diff):
                                select_idx = np.random.choice(diff_indices, int(diff), replace=False)
                                x[select_idx] = 1.0
                        elif diff < 0:
                            # Needs to decrease from 1s
                            diff_indices = np.where(x == 1)[0]
                            if len(diff_indices) >= int(-diff):
                                select_idx = np.random.choice(diff_indices, int(-diff), replace=False)
                                x[select_idx] = 0.0
                        # Ensure sum constraint exactly
                        x = project_to_feasible(x, n_points)
                        # Check if x is valid
                        if np.sum(x) > 0:
                            corr = compute_autocorrelations(x)
                            min_correlation = np.min(corr[1:])
                            if min_correlation > best_min or best_x is None:
                                best_min = min_correlation
                                best_x = x.copy()
                    h = best_x
                else:
                    # Fallback to matrix
                    v = np.ones(n_points)
                    x = np.zeros(n_points)
                    idx = np.argpartition(v, -int(n_points/2))[-int(n_points/2):]
                    x[idx] = 1.0
                    h = project_to_feasible(x, n_points)
            except Exception:
                x = np.zeros(n_points)
                idx = np.argpartition(np.random.rand(n_points), -int(n_points/2))[-int(n_points/2):]
                x[idx] = 1.0
                h = project_to_feasible(x, n_points)

    # Step 3: Swap refinement
    if time.time() - start_time < budget_s * 0.8:
        corr = compute_autocorrelations(h)
        h, _ = greedy_swap_improvement(h, corr, n_points, budget_s, start_time)

        # Include random perturbations to escape stuck local minima
        for _ in range(10):
            if time.time() - start_time > budget_s * 0.9:
                break
            h = random_perturbation_swap(h, budget_s, start_time)
            corr = compute_autocorrelations(h)
            h, _ = greedy_swap_improvement(h, corr, n_points, budget_s, start_time)

    # Step 4: Continuous polishing
    if time.time() - start_time < budget_s * 0.95:
        h = continuous_polishing(h, budget_s, start_time)

    # Final step
    c5_bound = evaluate_erdos_solution(h)
    return (h, c5_bound, n_points)
