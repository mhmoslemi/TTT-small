import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar

def project_feasible(v, n):
    """Project v onto the set {x: 0 <= x <= 1, sum(x) = n/2} using water filling."""
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    diff = n/2 - current_sum
    if abs(diff) < 1e-12:
        return v
    # Find the water level
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    # Find where to cut
    idx = np.searchsorted(cumsum, n/2)
    if idx == 0:
        level = n/2 / n
        return np.clip(v + level, 0, 1)
    else:
        # We have to fill up to level where cumulative sum of v_sorted[0:idx] adds up to n/2, 
        # then distribute remaining.
        total_pre = cumsum[idx - 1]
        remaining = n/2 - total_pre
        level = remaining / (n - idx)
        # Adjust the bottom elements to level
        v_new = v.copy()
        v_new[v_new > level] = level
        v_new[v_new <= level] = level
        # But we also need to correctly clip to water level in the top elements.
        # More precise approach:
        # Let water level = L
        # We want sum_{i: v[i] >= L} (v[i] - L) + sum_{i: v[i] < L} (L) = n/2
        # This is convex in L. Find L by bisection.

        # Let's simply do a binary search
        lower = 0
        upper = 1
        for _ in range(100):
            mid = (lower + upper) / 2
            new_sum = np.sum(np.clip(v - mid, 0, 1))
            if new_sum < n/2:
                upper = mid
            else:
                lower = mid
        level = lower
        v_new = np.clip(v - level, 0, 1)
        v_new = v_new * (n/2) / np.sum(v_new)
        return v_new

def evaluate_c5(h):
    """Evaluate C5 using FFT for speed."""
    n = len(h)
    dx = 2.0 / n
    # Use ifft to compute correlation efficiently
    H1 = fft(h)
    H2_conj = fft(1 - h).conj()
    corr = np.real(ifft(H1 * H2_conj))
    # We need only the full correlation, we can also compute directly:
    # But for now let's just be safe and use scipy.correlate for comparison
    # Or use fft if we trust its result
    max_corr = np.max(corr) * dx
    return max_corr

def gradient_at_worst(h):
    """Compute gradient of the maximum correlation term, which is the one we are 
    optimizing over."""
    n = len(h)
    dx = 2.0 / n
    # Critical: compute full correlation first, find worst index k*
    corr = correlate(h, 1-h, mode='full')
    k_star = np.argmax(corr)
    k_shift = k_star - (n-1) 
    grad = np.zeros(n)
    for i in range(n):
        i_plus = (i + k_shift) % n
        i_minus = (i - k_shift) % n
        grad[i] = dx * (h[i_plus] - h[i_minus])
    return grad

def line_search(h, grad, eta0=0.5):
    """Backtracking line search."""
    eta = eta0
    score_old = evaluate_c5(h)
    while eta > 1e-8:
        h_new = h - eta * grad
        h_new = project_feasible(h_new, len(h))
        score_new = evaluate_c5(h_new)
        if score_new < score_old:
            return h_new, score_new, eta
        eta /= 2
    return h, score_old, eta

def mutate_gaussian(v, overlap_size=5, sigma=0.2):
    """Apply Gaussian perturbation to subset of indices."""
    v = v.copy()
    idxs = np.random.choice(len(v), overlap_size, replace=False)
    delta = np.random.normal(0, sigma, overlap_size)
    for i, d in zip(idxs, delta):
        v[i] += d
    return project_feasible(v, len(v))

def mutate_swap(v, eps=0.01, delta=0.01):
    """Swap a small amount of mass between two components."""
    v = v.copy()
    i = np.random.choice(np.where(v > eps)[0])
    j = np.random.choice(np.where(v < 1-eps)[0])
    transfer = min(delta, v[i], 1 - v[j])
    v[i] -= transfer
    v[j] += transfer
    return project_feasible(v, len(v))

def mutate_dct(v, freq_coeff=0):
    """Modify a DCT coefficient to change function shape."""
    v = v.copy()
    # DCT of a real vector
    dct_coeffs = np.fft.dct(v)
    # Modify low-frequency coefficient (say first one)
    dct_coeffs[freq_coeff] += np.random.normal(0, 0.01)
    # Inverse DCT
    v_new = np.fft.idct(dct_coeffs)
    return project_feasible(v_new, len(v))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 250
    dx = 2.0 / n_points
    
    # Phase 1: Generate random initial samples
    n_pool = 1000
    pool = []
    pool_scores = []
    for i in range(n_pool):
        if time.time() - start_time > 0.1 * budget_s:
            break
        u = np.random.rand(n_points)
        h = project_feasible(u, n_points)
        score = evaluate_c5(h)
        pool.append(h)
        pool_scores.append(score)
        
    pool_scores = np.array(pool_scores)
    # Sort best
    indices = np.argsort(pool_scores)
    pool = [pool[i] for i in indices]
    pool_scores = pool_scores[indices]
    top_pool = list(zip(pool, pool_scores))[:30]
    
    # Phase 2: Mutation loop
    time_left = 0.25 * budget_s
    iter_count = 0
    while time.time() - start_time < 0.3 * budget_s - 0.1:
        iter_count += 1
        if iter_count > 50000:
            break
        parent_idx = np.random.randint(0, len(top_pool))
        parent = top_pool[parent_idx][0]
        # Select mutation type:
        choice = np.random.choice(3, p=[0.4, 0.4, 0.2])
        
        if choice == 0:  # Gaussian
            child = mutate_gaussian(parent, np.random.randint(5, 15), 0.2)
        elif choice == 1:  # Swap
            child = mutate_swap(parent)
        else:  # DCT
            child = mutate_dct(parent, np.random.randint(0, 3))
        
        score = evaluate_c5(child)
        # Replace worst in pool
        if score < top_pool[-1][1]:
            top_pool[-1] = (child, score)
            # Re-sort
            top_pool.sort(key=lambda x: x[1])
    
    # Phase 3: Local refinement on best solutions
    time_left = 0.45 * budget_s
    
    best_h = top_pool[0][0]
    best_c5 = top_pool[0][1]
    
    # Refine best few
    for i in range(min(5, len(top_pool))):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        h = top_pool[i][0]
        current_h = h.copy()
        current_c5 = top_pool[i][1]
        
        for _ in range(2000):
            if time.time() - start_time > 0.7 * budget_s - 0.1:
                break
            # Compute gradient at current best k
            grad = gradient_at_worst(current_h)
            # Backtrack to get a good step size
            h_new, score_new, _ = line_search(current_h, grad)
            if score_new < current_c5:
                current_c5 = score_new
                current_h = h_new
            else:
                # Sometimes our gradient wasn't good enough so let's also shoot random
                # small changes and try to climb up.
                break
                
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    # Phase 4: Progressive refinement
    if n_points == 250:
        # Upscale to 500
        h_up = np.interp(np.linspace(0, 2, 500), np.linspace(0, 2, 250), best_h)
        h_up = project_feasible(h_up, 500)
        c5_up = evaluate_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up
            n_points = 500
    
    # Phase 5: Swap refinement
    time_left = 0.1 * budget_s
    final_h = best_h.copy()
    current_c5 = best_c5
    for _ in range(10000):
        if time.time() - start_time > 0.9 * budget_s:
            break
        # Pick two points that are not at boundary
        i = np.random.randint(0, len(final_h))
        j = np.random.randint(0, len(final_h))
        if final_h[i] < 1e-5 or final_h[i] > 1 - 1e-5:
            continue
        if final_h[j] < 1e-5 or final_h[j] > 1 - 1e-5:
            continue
        delta = np.random.uniform(1e-4, 0.01)
        h_trial = final_h.copy()
        if h_trial[i] < delta:
            continue
        h_trial[i] -= delta
        h_trial[j] += delta
        h_trial = project_feasible(h_trial, len(h_trial))
        trial_c5 = evaluate_c5(h_trial)
        if trial_c5 < current_c5:
            final_h = h_trial
            current_c5 = trial_c5

    return (final_h, current_c5, n_points)
