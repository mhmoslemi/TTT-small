import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import fft, ifft

def project_to_box_plus_sum(v, n):
    """Project vector v onto the set {x : 0 ≤ x_i ≤ 1, sum x_i = n/2}."""
    # Water-filling approach
    x = v.copy()
    s = np.sum(x) - n/2
    if abs(s) < 1e-12:
        return np.clip(x, 0, 1)

    # Sort in descending order to find threshold level
    sorted_x = np.sort(x)[::-1]
    # Compute cumulative sums from the back
    cumsum = np.cumsum(sorted_x[::-1])[::-1]
    # Compute thresholds
    diff = sorted_x - (cumsum - n/2) / np.arange(1, len(sorted_x) + 1)
    # Find the threshold index
    threshold_idx = np.argmax(diff >= 0)
    if diff[threshold_idx] < 0:
        threshold_idx = len(sorted_x)
    # If threshold_idx == 0, no clipping needed on the upper bound
    if threshold_idx == 0:
        lam = sorted_x[0] + (n/2 - np.sum(sorted_x)) / len(sorted_x)
        # Ensure non-negativity
        lam = max(lam, 0.0)
        clipped = np.clip(x - lam, 0, 1)
    else:
        # The threshold cut-off
        if threshold_idx < len(sorted_x):
            # Actually, use the 'later' threshold 
            lam = sorted_x[threshold_idx - 1] - (n/2 - cumsum[threshold_idx - 1]) / threshold_idx
        else:
            # All were clipped
            lam = sorted_x[-1] - (n/2 - cumsum[-1]) / len(sorted_x)
        
        # And we clip all of them
        lam = max(lam, 0.0)
        clipped = np.clip(x - lam, 0, 1)
    return clipped

def evaluate_c5(h):
    """Evaluates the C5 value for h according to the objective."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorrelation(h):
    """Compute aperiodic autocorrelation of h."""
    n = len(h)
    # zero pad for linear convolution
    n_fft = 2 ** int(np.ceil(np.log2(2 * n)))
    H = fft(np.pad(h, (0, n_fft - n)), n_fft)
    auto_corr_fft = H * np.conj(H)
    result = ifft(auto_corr_fft).real
    # take only the correct length as full convolution
    return result[:2 * n - 1]

def construct_all_ones(n):
    """Construct vector of ones for convolution."""
    n_fft = 2 ** int(np.ceil(np.log2(2 * n)))
    ones = np.pad(np.ones(n), (0, n_fft - n))
    return fft(ones, n_fft)
