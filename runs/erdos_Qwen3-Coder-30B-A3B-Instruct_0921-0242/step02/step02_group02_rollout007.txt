import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_feasible(v, n):
    """Project v to the feasible set: sum(v) = n/2 and 0 <= v[i] <= 1."""
    v = np.clip(v, 0, 1)
    current_sum = np.sum(v)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return v
    
    # Water-filling approach
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    n_excess = 0
    for i, cum in enumerate(cumsum):
        if i == 0:
            avg = cum
        else:
            avg = cum / (i + 1)
        if avg > 1:
            n_excess = i
            break
    else:
        n_excess = len(v)
    
    # Handle excess mass
    if n_excess <= len(v) and n_excess > 0:
        excess = cumsum[n_excess - 1] - n_excess * 1
        v = np.clip(v - excess / n_excess, 0, 1)
    
    # Now redistribute difference
    v = v + diff / len(v)
    v = np.clip(v, 0, 1)
    v = v * (n / 2) / np.sum(v)
    return v

def evaluate_c5(v):
    """Evaluate the C5 constant."""
    n = len(v)
    dx = 2.0 / n
    # Compute cross-correlation via FFT
    # correlate(h, 1-h) is equivalent to ifft(fft(h) * conj(fft(1-h)))
    c = ifft(fft(v) * np.conj(fft(1 - v))).real
    max_corr = np.max(c) * dx
    return max_corr

def gradient_at_worst(v):
    """Compute gradient w.r.t. h for the maximum correlation."""
    n = len(v)
    dx = 2.0 / n
    # Compute full correlation
    c = ifft(fft(v) * np.conj(fft(1 - v))).real
    k_star = np.argmax(c)
    # Convert k* to shift (in range [-n+1, n-1])
    if k_star >= n:
        k_shift = k_star - n + 1
    else:
        k_shift = k_star - n + 1
    
    grad = np.zeros(n)
    # Compute gradient entries for k* shift
    for i in range(n):
        i_plus = (i + k_shift) % n
        i_minus = (i - k_shift) % n
        # grad[i] = dx * (v[i_minus] - v[i_plus])
        grad[i] = dx * (v[i_plus] - v[i_minus])
    return grad

def line_search(v, grad, eta0=0.5):
    """Backtracking line search."""
    eta = eta0
    dx = 2.0 / len(v)
    while eta > 1e-10:
        v_new = v - eta * grad
        v_new = project_feasible(v_new, len(v))
        c5_new = evaluate_c5(v_new)
        c5_orig = evaluate_c5(v)
        if c5_new < c5_orig:
            return v_new, c5_new
        eta *= 0.5
    return v, evaluate_c5(v)

def mutate(v, mode='gauss', params=None):
    """Mutate v using one of three strategies."""
    n = len(v)
    v_new = v.copy()
    if mode == 'gauss':
        # perturb subset of indices
        indices_to_perturb = np.random.choice(n, size=np.random.randint(5, 10))
        perturb_magnitude = 0.2 if params is None else params.get('sigma', 0.2)
        for i in indices_to_perturb:
            new_val = v_new[i] + np.random.normal(0, perturb_magnitude)
            v_new[i] = new_val
    elif mode == 'swap':
        min_val = 0.01
        max_val = 0.99
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        while v_new[i] <= min_val:
            i = np.random.randint(0, n)
        while v_new[j] >= max_val:
            j = np.random.randint(0, n)
        delta = np.random.uniform(0.001, 0.05)
        if v_new[i] >= delta and (1 - v_new[j]) >= delta:
            v_new[i] -= delta
            v_new[j] += delta
    elif mode == 'dct':
        if n < 4:
            return v
        # DCT this
        v_dct = np.fft.dct(v)
        # Modify a low-frequency coefficient
        k = np.random.randint(0, min(5, n//2 + 1))
        change = np.random.uniform(-0.05, 0.05) * v_dct[k]
        v_dct[k] += change
        v_new = np.fft.idct(v_dct)
        # Project to [0,1] and renormalize
    # Clamp values
    v_new = np.clip(v_new, 0, 1)
    return project_feasible(v_new, n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize points
    initial_n = 250
    dx = 2.0 / initial_n
    # Use initial_h_values
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values[:initial_n])
    else:
        h0 = np.random.rand(initial_n)
    h0 = h0 * (initial_n / 2) / np.sum(h0)
    h0 = project_feasible(h0, initial_n)

    # Generate initial population
    M = 2000  # Population size
    pool = []
    for _ in range(M):
        h = np.random.rand(initial_n)
        h = project_feasible(h, initial_n)
        pool.append(h)
    
    # Evaluate pool and keep top K
    K = 30
    pool_evals = [(h, evaluate_c5(h)) for h in pool]
    pool_evals.sort(key=lambda x: x[1])
    pool = [h for h, _ in pool_evals][:K]
    best_c5 = pool_evals[0][1]
    best_h = pool[0].copy()

    # Run mutation loop
    t_elapsed = time.time() - start_time
    budget_mutate = 0.25 * budget_s
    i_iter = 0
    while t_elapsed < budget_mutate and (time.time() - start_time) < budget_s - 0.5:
        # Select parent
        parent = pool[np.random.randint(0, len(pool))]
        # Try multiple mutation modes
        modes = ['gauss', 'swap', 'dct']
        mode = np.random.choice(modes)
        child = mutate(parent, mode)
        child_c5 = evaluate_c5(child)
        # Evict worst if better
        if child_c5 < pool[-1][1]:
            pool.insert(0, child)
            pool = pool[:-1]  # Remove last
            if child_c5 < best_c5:
                best_c5 = child_c5
                best_h = child.copy()

        i_iter += 1
        t_elapsed = time.time() - start_time

    # Local refinement
    budget_refine = 0.45 * budget_s
    refine_start = time.time()
    refine_h = []
    for h in pool[:5]:  # Top 5 for refinement
        # Local improvement via gradient descent
        h_local = h.copy()
        current_c5 = evaluate_c5(h_local)
        for iteration in range(2000):
            if time.time() - refine_start > budget_refine or \
               time.time() - start_time > budget_s - 0.5:
                break
            grad = gradient_at_worst(h_local)
            if np.linalg.norm(grad) < 1e-10 and current_c5 < 0.5:
                break  # Stop if not improving significantly
            h_local, new_c5 = line_search(h_local, grad)
            if new_c5 < current_c5:
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_local.copy()
            else:
                # Try different step size directions
                h_local = project_feasible(h_local, len(h_local))
                new_c5 = evaluate_c5(h_local)
                if new_c5 < current_c5:
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h_local.copy()
    # Try upsampling
    final_n = initial_n  # Minimal size to pass test
    # Maybe upscale if time permits
    if (time.time() - start_time) < 0.9 * budget_s:
        # Interpolate from 250 to 500
        n_upscale = 500
        # Interpolate values
        x_original = np.linspace(0, 2, initial_n)
        x_new = np.linspace(0, 2, n_upscale)
        h_up = np.interp(x_new, x_original, best_h)
        h_up = project_feasible(h_up, n_upscale)
        current_c5 = evaluate_c5(h_up)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_up
            final_n = n_upscale
    
    # Final polishing
    final_start = time.time()
    for _ in range(10000):
        if time.time() - start_time > budget_s - 0.5:
            break
        i = np.random.randint(0, final_n)
        j = np.random.randint(0, final_n)
        # Both should be within (0.01, 0.99)
        if best_h[i] <= 0.01 or best_h[j] >= 0.99:
            continue
        # Move small amount
        delta = np.random.uniform(1e-4, 1e-2)
        h_trial = best_h.copy()
        h_trial[i] -= delta
        h_trial[j] += delta
        h_trial = project_feasible(h_trial, final_n)
        trial_c5 = evaluate_c5(h_trial)
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = h_trial.copy()

    return (best_h.tolist(), best_c5, final_n)
