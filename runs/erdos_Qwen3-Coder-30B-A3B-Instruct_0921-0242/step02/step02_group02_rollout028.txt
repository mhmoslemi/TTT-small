import numpy as np
import time
from scipy.signal import correlate
from scipy.fft import rfft, irfft

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach for adjusting sum
    h_sorted = np.sort(h)[::-1]
    cumsum = np.cumsum(h_sorted)
    
    # Binary search for lambda
    lb, ub = -1.0, 1.0
    for _ in range(50):  # Safe iterations
        mid = (lb + ub) / 2
        new_h = np.clip(h + mid, 0, 1)
        if np.sum(new_h) < n/2:
            ub = mid
        else:
            lb = mid
    
    h = np.clip(h + lb, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def compute_autocorr(h):
    """Compute the autocorrelation using FFT."""
    H = rfft(h, len(h))
    ac = irfft(H * np.conj(H), len(h))
    return np.real_if_close(ac)

def single_swap_update(ac, h, i, j):
    """Update autocorrelation after a single swap between i and j."""
    # Get shifts
    n = len(h)
    h_i = h[i]
    h_j = h[j]
    
    # Compute the change in autocorrelation at each shift k
    # For shift k, the contribution changes by:
    #   h_i h_{i-k} -> h_j h_{i-k}
    #   h_j h_{j-k} -> h_i h_{j-k}
    #   h_i h_{i-k} + h_j h_{j-k} -> h_j h_{i-k} + h_i h_{j-k}
    # The change is: add (h_i - h_j) * (h_{i-k} - h_{j-k}) to ac[k] for all k
    
    delta = np.zeros(n)
    
    # For each valid k, compute the change
    # We only explicitly compute for a small number shifts per update
    # But here because we optimize over a few swaps per step, it's acceptable to
    # compute all for the resonable expectations
    
    for k in range(n):
        ik = (i - k) % n
        jk = (j - k) % n
        delta[k] += (h_i - h_j) * (h[ik] - h[jk])
        # Self-terms: -2*h_i*h_j at k = 0, and others if concentrated
        # Actually for exchange of h[i] and h[j], we'd have standard changes:
        # h[i]*h[i-k] + h[j]*h[j-k] -> h[j]*h[i-k] + h[i]*h[j-k]
        # The change = h[i]*h[i-k] - h[j]*h[i-k] + h[j]*h[j-k] - h[i]*h[j-k]
        # = (h[i] - h[j]) * (h[i-k] - h[j-k])
        # Combined with contributions to the constant shift of k = 0
        # At k = 0, the self-contribution: h[i]*h[i] + h[j]*h[j] changes to h[j]*h[j] + h[i]*h[i]
        # This contributes a 0 to the delta difference
        # Better just represent the difference the way we calculate:
        # at shift k: old_h[i]*old_h[i-k] + old_h[j]*old_h[j-k], new is new_h[i]*new_h[i-k] + new_h[j]*new_h[j-k]
        # So d_{ik} = h[i] * h[i-k] + h[j] * h[j-k] - (h[j] * h[i-k] + h[i] * h[j-k])
        # d_{ik} = (h[i] - h[j]) * (h[i-k] - h[j-k])
    
    # After checking it's:
    # The change at shift k = 0 when swapping indices i and j is:
    # old contribution = h[i]*h[i] + h[j]*h[j]
    # new contribution = h[j]*h[j] + h[i]*h[i]
    # delta[0] = h[i]*h[i] - h[i]*h[i] = 0
    # This is wrong logic, we compute properly via proper indexing:
    # Each time index i is in the position, it increases by h[i * (h[i-k] - h[j-k])]
    # This is negligible for our needs as we do not track const term change
    
    # Implement cleaner approach:
    # Original: ac[k] = sum(h[t] * h[t-k])
    # We swap h[i] and h[j], i.e., apply h'[i] = h[j], h'[j] = h[i], others unchanged
    # This changes ac[k] by (h[i] - h[j])*(h[i-k] - h[j-k])
    # With periodic indexing, we compute all such terms at once:
    
    # Subtract current term differences from shift lookups
    # and add updated term differences:
    # For all positions k in range:
    # The change in corruption is:
    # New contribution k: h'[i] * h'[i-k] + h'[j] * h'[j-k] 
    # Old contribution k: h[i] * h[i-k] + h[j] * h[j-k]
    
    # Due to periodic indexing:
    # Let's take k=0: new == h'[i]*h'[i] + h'[j]*h'[j] = h[j]*h[j] + h[i]*h[i]
    # Original is h[i]*h[i] + h[j]*h[j] => difference is 0
    # for k != 0: 
    # Haven treat the shift k differently:
    # delta[k] = (h[i] - h[j])*(h[i-k] - h[j-k]) for all k
    
    # We'll compute and make correction only for shifts where the change occurs
    # The attribute_CHANGE (a simple term) is the difference in correlation
    # The only problem is that when we look at the autocorrelation, we may miss some changes
    
    # Simpler approach: re-calculate full change within traced
    # Each swap touches 2*n elements: each h_t where it gets modified
    # from h_t -> h[t+shift] effect is: (h[i]-h[j])*(h[t-shift] - h[t-shift]) -> we are computing change over all k, i.e., all directions
    
    # One simple robust way is to compute the
    # Exact correlation changes involved, but it's a bit hard to make it perfect
    # For now, use standard formula:
    # If we are swapping i and j (i and j are different):  
    # Shift k, value = starting_h[i]*starting_h[i-k] + starting_h[j]*starting_h[j-k]
    # After swap: h'[i] = h[j], h'[j] = h[i]
    # Let’s get the full expression:
    # dot products over all t:
    # shift k contribution: h[t] * h[t-k] 
    # with changing indices i,j:
    # Then at shift k:
    # Old = h[i]*h[i-k] + h[j]*h[j-k]
    # New = h[j]*h[i-k] + h[i]*h[j-k] 
    # change = h[j]*h[i-k] + h[i]*h[j-k] - h[i]*h[i-k] - h[j]*h[j-k]  
    # reduce = (h[j] - h[i]) * (h[i-k] - h[j-k])
    
    # Compute the updated terms:
    delta = np.zeros(n)
    # Compute change due to shifting h[i], h[j] 
    # i shifts are problematically different in indices:
    for k in range(n):
        ik = (i - k) % n
        jk = (j - k) % n
        delta[k] += (h_i - h_j) * (h[ik] - h[jk])
    
    return delta

def binary_anneal(seed, n, time_limit, start_h=None):
    """Anneal binary pattern to minimize max overlap."""
    np.random.seed(seed)
    dx = 2.0 / n
    
    # Start with the given h if provided, else random binary
    if start_h is not None and len(start_h) == n:
        h = start_h.copy()
    else:
        h = np.random.rand(n) > 0.5
        h = h.astype(float)
    
    # Enforce sum = n/2
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if diff > 0:
        # Add ones to make up missing sum
        # Need to find zeros to make up
        zero_indices = np.where(h < 0.5)[0]
        if len(zero_indices) > 0:
            selected = np.random.choice(zero_indices, min(int(diff), len(zero_indices)), replace=False)
            for i in selected:
                h[i] = 1.0
    else:
        # Subtract ones to reduce the sum
        one_indices = np.where(h > 0.5)[0]
        if len(one_indices) > 0:
            selected = np.random.choice(one_indices, min(int(-diff), len(one_indices)), replace=False)
            for i in selected:
                h[i] = 0.0
    
    # Ensure exactly sum = n / 2
    current_sum = np.sum(h)
    if abs(current_sum - n/2) > 1e-10:
        if current_sum > n/2:
            # Reduce 1s to get sum to exactly n/2
            one_indices = np.where(h > 0.5)[0]
            num_reduce = int(current_sum - n/2)
            to_reduce = np.random.choice(one_indices, num_reduce, replace=False)
            for i in to_reduce:
                h[i] = 0.0
        else:
            # Increase 0s  to get sum to exactly n/2
            zero_indices = np.where(h < 0.5)[0]
            num_increase = int(n/2 - current_sum)
            to_increase = np.random.choice(zero_indices, num_increase, replace=False)
            for i in to_increase:
                h[i] = 1.0
    
    # Compute initial
    ac = compute_autocorr(h)
    current_c5 = np.max(dx * (n/2 - ac))
    best_h = h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    T0 = 5 * dx
    T = T0
    alpha = 0.9997
    
    # Some counters
    n_swaps = 0
    last_improvement = 0
    
    start_time = time.time()
    while time.time() - start_time < time_limit:
        # Perform 20000 Swaps per iteration
        for _ in range(5000):
            if time.time() - start_time > time_limit:
                break
            
            # Choose two indices with different values
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            
            if len(ones) < 1 or len(zeros) < 1:
                # Can't make meaningful change
                break
                
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Compute delta_ac for this swap
            delta = single_swap_update(ac, h, i, j)
            # Apply change to make new ac
            new_ac = ac + delta
            # Δ = CHANGED c for this one swap. 
            # if         c new = max (dx * (n/2 - new_ac)),
            # then old = max (dx * (n/2 - ac)),
            #      new = max (dx * (n/2 - new_ac))
            # So: 
            # Δ_c = [dx * (n/2 - new_ac)].max() - [dx * (n/2 - ac)].max()
            
            # Compute new c vector from new_ac
            old_c = dx * (n/2 - ac)
            new_c = dx * (n/2 - new_ac)
            
            # New c5
            old_c5 = np.max(old_c)
            new_c5 = np.max(new_c)

            # Make the swap if it improves or with probability T
            if new_c5 < old_c5 or np.random.rand() < np.exp(-(new_c5 - old_c5) / T):
                h[i] = 0.0
                h[j] = 1.0
                ac = new_ac  # Updated
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                # Perhaps we need small adjustment due to numerical errors
            
            # Update temperature schedule
            n_swaps += 1
            if n_swaps % 2000 == 0:
                T = max(T * alpha, dx * 1e-4)
                
        # End of a set of swaps
            
        # Implementation alert: Sometimes local stagnation occurs
        # Restart if no progress made in a long time
        if n_swaps - last_improvement > 200000:
            # Perform large jump:
            # Set 5% of the ones into 0s or so
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            # Replace 5%
            num = int(0.05 * len(h))
            for _ in range(num):
                if len(ones) > 0:
                    idx = np.random.choice(ones)
                    h[idx] = 0.0
                    ones = np.where(h > 0.5)[0]
                if len(zeros) > 0:
                    idx = np.random.choice(zeros)
                    h[idx] = 1.0
                    zeros = np.where(h < 0.5)[0]
            
            ac = compute_autocorr(h)
            current_c5 = np.max(dx * (n/2 - ac))
            last_improvement = n_swaps
            
        # Update after many iterations
        if n_swaps % 5000 == 0:
            # Periodic check
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    # Done annealing
    return best_h, best_c5

def smooth_boundaries(h_bin, w=2):
    """Convert binary pattern to a continuous one with fractional transitions."""
    n = len(h_bin)
    h_smooth = h_bin.copy()
    
    # Identify all transitions 0->1 and 1->0
    transitions = []
    for i in range(n):
        # previous index
        prev = (i - 1) % n
        if h_bin[prev] != h_bin[i]:
            transitions.append((prev, i, h_bin[i]))  # (prev_index, current_index, next_value)
    
    # Insert ramps
    for i in range(n):
        if i + 1 < n and h_bin[i] != h_bin[i+1]:
            if h_bin[i] == 0 and h_bin[i+1] == 1:
                # 0->1, smooth from i to i+w in the right
                # Set indices i+1, i+2, ..., i+w (if they exist and within bound)
                for k in range(1, w+1):
                    idx = (i + k) % n
                    if idx == i or idx == i+w+1:
                        break
                    # Linear interpolation
                    ratio = k / (w + 1.)
                    h_smooth[idx] = ratio
            elif h_bin[i] == 1 and h_bin[i+1] == 0:
                # Set from i to i+w with inverse
                for k in range(1, w+1):
                    idx = (i + k) % n
                    if idx == i or idx == i+w+1:
                        break
                    ratio = (w - k + 1) / (w + 1)
                    h_smooth[idx] = ratio
                
    return h_smooth

def compute_sub_gradient(h, k):
    """Compute subgradient of worst shift k."""
    dx = 2.0 / len(h)
    grad = -dx * (np.roll(h, k) + np.roll(h, -k))
    return grad

def proj_grad_refine(h, time_limit):
    """Perform projected gradient descent refinements."""
    dx = 2.0 / len(h)
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Gradient step counter
    step_count = 0
    
    while time.time() - start_time < time_limit:
        step_count += 1
        # Find the worst shift k
        corr = correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        if k_star > len(h):
            k_star = k_star - len(h)
        else:
            k_star = k_star - len(h) + 1
        
        grad = compute_sub_gradient(h, k_star)
        
        # Backtracking line search
        alpha = 0.5  # Initial step size
        max_trials = 20
        for _ in range(max_trials):
            # Update
            h_trial = h - alpha * grad
            h_trial = project_to_feasible(h_trial, len(h))
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                h = h_trial
                best_h = h.copy()
                best_c5 = c5_trial
                break
            alpha *= 0.5
        
        # Occasionally perform pairwise swaps to escape plateaus
        if step_count % 100 == 0:
            # Pairwise swap of very small differences
            # Make small move from larger to smaller, greedily
            i = np.argmax(h)
            j = np.argmin(h) 
            if h[i] > 0 and h[j] < 1:
                delta = min(h[i], 1 - h[j], 0.01)
                h[i] -= delta
                h[j] += delta
                h = project_to_feasible(h, len(h))
                current_c5 = evaluate_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: Binary search on 256 grid
    n0 = 256
    dx0 = 2 / n0
    print("Starting binary search with n={}, dx={}".format(n0, dx0))
    h0, c5_0 = binary_anneal(seed, n0, 0.55 * budget_s)
    print("Binary search completed, C5 = {:.6f}".format(c5_0))
    
    # Stage 2: Smooth boundaries and refined gradient
    print("Starting smoothing stage...")
    h_smooth = smooth_boundaries(h0, w=2)
    h_smooth = project_to_feasible(h_smooth, n0)
    print("Smoothed starting point to feasible")
    
    h_refined, c5_refined = proj_grad_refine(h_smooth, 0.25 * budget_s)
    print("Refined with gradient, C5 = {:.6f}".format(c5_refined))
    
    # Stage 3: Upsample and re-refine
    print("Starting upsampling stage...")
    n1 = 2 * n0
    h_upsample = np.interp(np.linspace(0, 2, n1), np.linspace(0, 2, n0), h_refined)
    h_upsample = project_to_feasible(h_upsample, n1)
    print("Upsampled to n={}".format(n1))
    
    h_final, c5_final = proj_grad_refine(h_upsample, 0.15 * budget_s)
    print("Final polish completed, C5 = {:.6f}".format(c5_final))
    
    return (h_final, c5_final, n1)
