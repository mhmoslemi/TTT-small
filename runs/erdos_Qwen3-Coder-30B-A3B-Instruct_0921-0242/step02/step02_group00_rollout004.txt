import numpy as np
import time
from scipy.signal import correlate, fftconvolve
from scipy.fft import fft, ifft

def project_to_feasible(h, n):
    """Project h to feasible set {0 <= h <= 1, sum(h) = n/2} using waterfilling."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum

    if abs(diff) < 1e-10:
        return h

    # Waterfilling to adjust the sum
    if diff > 0:
        # We need to increase values
        # Sort in ascending order to increase smaller values first
        idx = np.argsort(h)
        accum = 0.0
        for i in range(n):
            j = idx[i]
            if accum + (1 - h[j]) <= diff:
                accum += (1 - h[j])
                h[j] = 1.0
            else:
                h[j] += diff - accum
                break
    elif diff < 0:
        # We need to decrease values
        idx = np.argsort(h, kind='stable')[::-1]
        accum = 0.0
        for i in range(n):
            j = idx[i]
            if accum + h[j] <= abs(diff):
                accum += h[j]
                h[j] = 0.0
            else:
                h[j] -= abs(diff) - accum
                break
    return h

def evaluate_c5(h):
    """Evaluates the C5 constant for a given h."""
    dx = 2.0 / len(h)
    # Full correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def build_fft_structure(n):
    """Build FFT structure for convolution."""
    n_fft = 2 ** int(np.ceil(np.log2(2*n)))
    ones_vec = np.ones(n)
    ones_fft = fft(ones_vec, n_fft)
    return ones_fft, n_fft

def update_h_with_mu(z, u, t, mu, ones_fft, n_fft, n):
    """Update h for given mu using FFT."""
    z_minus_u = z - u
    z_fft = fft(z_minus_u, n_fft)
    denominator = 1 + mu * ones_fft
    h_fft = z_fft / denominator
    h = ifft(h_fft).real[:n]
    return h

def find_optimal_mu(z, u, t, ones_fft, n_fft, n):
    """Find optimal mu using bisection."""
    eps = 1e-12
    mu_low, mu_high = 0.0, 1e10
    tol = 1e-12

    # Ensure the current h is feasible
    h = update_h_with_mu(z, u, t, 0.0, ones_fft, n_fft, n)

    # Check if h is already feasible:
    hcorr = correlate(h, h, mode='full')
    max_corr = 1 - (2.0 / n) * np.max(hcorr)
    if max_corr <= t + tol:
        return 0.0, h

    for _ in range(50):
        mu = (mu_low + mu_high) / 2.0
        h = update_h_with_mu(z, u, t, mu, ones_fft, n_fft, n)

        # Check if h satisfies constraint for t
        hcorr = correlate(h, h, mode='full')
        max_corr = 1 - (2.0 / n) * np.max(hcorr)

        if max_corr <= t + tol:
            mu_high = mu
        else:
            mu_low = mu

        if abs(mu_high - mu_low) < tol:
            break

    h = update_h_with_mu(z, u, t, mu_high, ones_fft, n_fft, n)
    return mu_high, h

def admm_solve(h0, z0, t0, ones_fft, n_fft, n, max_iter=500, tol=1e-7):
    """Solve the ADMM subproblems."""

    h = h0.copy()
    z = z0.copy()
    u = np.zeros_like(h)
    t = t0

    last_t = t
    no_improve_count = 0
    best_t = t
    best_z = z.copy()

    for it in range(max_iter):
        # Recompute h with optimal mu
        mu, h_new = find_optimal_mu(z, u, t, ones_fft, n_fft, n)
        h = h_new

        # Update z via projection
        z_new = project_to_feasible(h + u, n)
        z = z_new

        # Update dual variable
        u = u + (h - z)

        # Update t
        corr = correlate(h, h, mode='full')
        t = max(1 - (2.0 / n) * np.max(corr), 0.0)

        # Update best solution
        if t < best_t:
            best_t = t
            best_z = z.copy()

        # Check convergence
        if abs(t - last_t) / (1 + abs(t)) < tol:
            no_improve_count += 1
        else:
            no_improve_count = 0
        last_t = t

        if no_improve_count >= 5:
            break

    return best_z, best_t

def nesterov_refine(z, n, eps=1e-6, max_iter=200):
    """Nesterov accelerated gradient refinement with smooth max."""
    # Initial setup
    h = z.copy()
    h_prev = z.copy()
    y = z.copy()
    beta = 5.0
    eta = 1.0
    best_h = z.copy()
    best_c5 = evaluate_c5(z)

    for k in range(1, max_iter + 1):
        # Smooth max surrogate
        corr = correlate(h, h, mode='full')
        C = 1 - (2.0 / n) * corr
        C_max = np.max(C)
        C_stable = C - C_max
        exp_C = np.exp(beta * C_stable)
        w = exp_C / np.sum(exp_C)

        # Compute gradient of surrogate
        shift_plus = np.roll(h, 1)
        shift_minus = np.roll(h, -1)
        grad = -(2.0 / n) * (shift_plus @ w + shift_minus @ w)

        # Nesterov step
        y = h + ((k - 1) / (k + 2)) * (h - h_prev)
        h_new = project_to_feasible(y - eta * grad, n)

        # Backtracking line search for eta
        c5_trial = evaluate_c5(h_new)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_new.copy()

        h_prev = h.copy()
        h = h_new.copy()

        # Adaptive beta
        if k % 20 == 0:
            beta *= 1.3

    return best_h, best_c5
