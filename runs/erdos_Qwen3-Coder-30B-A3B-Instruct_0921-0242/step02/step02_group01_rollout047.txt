import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def project_to_feasible(h, n):
    """Ensure that sum(h) = n/2, with h in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = n/2 - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    h = h + diff / n
    h = np.clip(h, 0, 1)
    
    return h

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Full correlation using FFT for efficiency
    h_fft = fft(h, 2 * len(h))
    ih_fft = fft(1 - h, 2 * len(h))
    corr = ifft(h_fft * np.conj(ih_fft))
    # We take the real part and maximum value of the full correlation
    max_corr = np.max(corr.real) * dx
    return max_corr

def build_lp_data(h0):
    """Build linear programming data for the current iterate."""
    n = len(h0)
    
    # Compute auto-correlation for shift k=0 (needed later)
    auto_corr = ifft(fft(h0) * np.conj(fft(h0))).real
    C0 = auto_corr
    
    # Build weight matrix W for all shifts
    # Row k of W is (h0[(i+k)%n] + h0[(i-k)%n]) for i in [0,n-1]
    W = np.zeros((n, n))
    for k in range(n):
        W[k, :] = h0[(np.arange(n) + k) % n] + h0[(np.arange(n) - k) % n]
    
    A = -W
    b = -n/2 + C0
    
    return A, b

def solve_ccp_step(A, b, n):
    """Solve the CCP subproblem as an LP."""
    # Construct the LP
    c = np.zeros(n + 1)
    c[-1] = 1  # minimize t
    
    # The constraints A * h + 1 * t <= b
    # Form augmented constraint matrix, add sum constraint
    # A_aug = [A, ones] so that A_aug * [h; t] <= b
    A_aug = np.column_stack([A, np.ones(A.shape[0])])
    
    # Add bounds: 0 <= h <= 1, sum h = n/2, t >= 0
    bounds = [(0, 1) for _ in range(n)] + [(0, None)]
    # Add sum constraint
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1
    b_eq = [n/2]
    
    # Convert to standard form: minimize c^T x subject to A_aug x <= b, A_eq x = b_eq
    result = linprog(c, A_ub=A_aug, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        h_new = result.x[:-1].copy()
        t_new = result.x[-1]
        return h_new, t_new
    else:
        # Fall back to simple projection
        print("Warning: LP did not converge.")
        return np.zeros(n), float('inf')

def generate_seeds(n):
    """Generate multiple feasible starting points."""
    seeds = []
    
    # 1. Uniform
    seeds.append(np.ones(n) * 0.5)
    
    # 2. Random Dirichlet (sample from uniform distribution + normalize)
    h = np.random.rand(n)
    h = h * (n/2) / np.sum(h)
    seeds.append(h)
    
    # 3. Binary random with exactly n/2 ones, then smooth
    h = np.zeros(n)
    h[:n//2] = 1
    np.random.shuffle(h)
    # Add small noise to soften
    noise = np.random.normal(0, 0.01, n)
    h = np.clip(h + noise, 0, 1)
    h = h * (n/2) / np.sum(h)
    seeds.append(h)
    
    # 4. Interpolate initial_h_values if available
    if 'initial_h_values' in globals():
        orig_h = initial_h_values
        if len(orig_h) != n:
            x_orig = np.linspace(0, 2, len(orig_h))
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_orig, orig_h)
        else:
            h = orig_h.copy()
        h = h * (n/2) / np.sum(h)
        seeds.append(h)
        
    # 5. Sobol-style low-discrepancy (empirically good direction)
    # seed the random number generator to ensure reproducibility
    np.random.seed(42)
    base = np.random.rand(n)
    # perturb to make distribution more uniform-like
    h = np.clip(base + np.random.rand(n) * 0.1, 0, 1)
    h = h * (n/2) / np.sum(h)
    seeds.append(h)
    
    return seeds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Length of discretization - smaller to keep fast
    n_points = 400
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = np.ones(n_points) * 0.5
    last_improvement = time.time()
    
    # Multi-start CCP
    seeds = generate_seeds(n_points)
    max_iter = 25
    
    for i, h0 in enumerate(seeds):
        if time.time() - start_time > budget_s * 0.7:
            break
        h = h0.copy()
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s * 0.7:
                break
            A, b = build_lp_data(h)
            h_new, t_new = solve_ccp_step(A, b, n_points)
            
            # Project to feasible
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                last_improvement = time.time()
            if c5_new > t_new + 1e-6:
                raise RuntimeError("LP gave a tighter bound than actual objective.")
                
            # Check for convergence
            if np.max(np.abs(h_new - h)) < 1e-8:
                break
            h = h_new
    
    # Refinement via projected gradient descent (few steps)
    if time.time() - start_time < budget_s * 0.9:
        h_refined = best_h.copy()
        for i in range(200):
            if time.time() - start_time > budget_s * 0.9:
                break
            # Find worst shift
            corr = ifft(fft(h_refined, 2 * len(h_refined)) * np.conj(fft(1 - h_refined, 2 * len(h_refined))))
            corr_vals = corr.real[1: len(h_refined)]  # exclude index 0
            worst_shift = np.argmin(corr_vals) + 1
            # Compute gradient
            grad = np.zeros_like(h_refined)
            for j in range(len(h_refined)):
                left_idx = (j + worst_shift) % len(h_refined)
                right_idx = (j - worst_shift) % len(h_refined)
                grad[j] = dx * ((1 - h_refined[left_idx]) - h_refined[right_idx])
            # Update with fixed step size
            alpha = 0.1
            h_new = h_refined - alpha * grad
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_refined = h_new
    
    # Final polishing with simulated annealing
    if time.time() - start_time < budget_s * 0.995:
        h_final = best_h.copy()
        c5_final = evaluate_c5(h_final)
        temp_start = 1.0
        temp_min = 1e-6
        cooling_rate = 0.999
        max_attempts = 50000
        
        t = 0
        for _ in range(max_attempts):
            if time.time() - start_time > budget_s * 0.995:
                break
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            # Pick a small quantity to move
            delta = min(0.01, h_final[i], 1 - h_final[j])
            if delta < 1e-8:
                continue
            h_test = h_final.copy()
            h_test[i] -= delta
            h_test[j] += delta
            h_test = project_to_feasible(h_test, n_points)
            c5_test = evaluate_c5(h_test)
            if c5_test < c5_final:
                c5_final = c5_test
                h_final = h_test
            else:
                # Accept with simulated annealing probability
                prob = np.exp(-(c5_test - c5_final) / (temp_start * (cooling_rate ** t)))
                if np.random.rand() < prob:
                    h_final = h_test
                    c5_final = c5_test
            t += 1
    
    return (h_final, c5_final, n_points)
