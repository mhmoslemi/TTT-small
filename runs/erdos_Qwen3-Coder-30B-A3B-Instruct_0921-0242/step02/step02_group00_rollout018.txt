import numpy as np

def project_to_box_plus_sum(h, n_points):
    """Project h onto the set {0 ≤ h ≤ 1, sum(h) = n/2}."""
    # Water-filling algorithm
    y = h.copy()
    total = np.sum(y)
    target = n_points / 2.0
    
    if abs(total - target) < 1e-12:
        return np.clip(y, 0, 1)
    
    # Sort in descending order
    sorted_y = np.sort(y)[::-1]
    cumsum = np.cumsum(sorted_y)
    
    # Find the threshold
    threshold = 0.0
    n_above = 0
    for i in range(len(sorted_y)):
        if sorted_y[i] > (target - cumsum[i]) / (i + 1):
            n_above = i + 1
        else:
            break
    
    # Calculate threshold
    threshold = (target - cumsum[n_above - 1]) / n_above if n_above > 0 else 0.0
    
    # Apply projection
    result = np.clip(y - threshold, 0, 1)
    
    return result

def evaluate_c5(h):
    """Computes the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def fft_autocorr(h):
    """Compute the autocorrelation of h using FFT."""
    n = len(h)
    nfft = 2**(int(np.ceil(np.log2(2*n))))
    H = fft(h, nfft)
    autocorr = ifft(H * np.conj(H))
    return np.real(autocorr[:n])

def compute_shifted_sums(h, n):
    """Compute shifted sums (used for gradient computation)."""
    h_fft = fft(h, 2*n)
    ones = np.ones(2*n)
    ones_fft = fft(ones, 2*n)
    shifted_sum_fft = h_fft * ones_fft
    shifted_sum = ifft(shifted_sum_fft)
    return np.real(shifted_sum[:n])
