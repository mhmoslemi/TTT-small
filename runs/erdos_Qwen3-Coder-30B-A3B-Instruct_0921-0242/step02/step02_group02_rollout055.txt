import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def project_feasible(v, n):
    """Project v onto the set {x : sum(x) = n/2, 0 <= x <= 1}."""
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if abs(s - n/2) < 1e-10:
        return v
    # Water-filling to adjust sum to n/2
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    # Find the water level
    target = n/2
    idx = np.searchsorted(cumsum, target, side='right')
    if idx == 0:
        level = target / n
        v = np.full_like(v, level)
    else:
        level = (target - cumsum[idx-1]) / (n - idx)
        # Clip levels above 1 or below 0
        level = np.clip(level, 0, 1)
        v = np.where(np.arange(n) < idx, v_sorted[:idx], level)
        # Recompute and reproject to be safe
        v = np.clip(v, 0, 1)
        v = v * (n/2) / np.sum(v)
    return v

def evaluate_c5(h):
    """Evaluates the C5 constant for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using FFT
    f_h = fft(h)
    f_inv = fft(1 - h)
    corr = ifft(f_h * f_inv.conj()).real
    # The correlation is a circular correlation, mode='full'
    # Take only the valid part: the maximum over all shifts
    max_corr = np.max(corr) * dx
    return max_corr

def compute_gradient(h, k_star):
    """Compute gradient of C5 at h for the worst shift k_star."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        left_idx = (i - k_star) % n
        right_idx = (i + k_star) % n
        grad[i] = dx * (h[right_idx] - h[left_idx])
    return grad

def line_search(h, grad, max_steps=10):
    """Backtracking line search for gradient descent step."""
    eta = 0.5
    dx = 2.0 / len(h)
    for _ in range(max_steps):
        h_new = h - eta * grad
        h_new = project_feasible(h_new, len(h))
        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h):
            return h_new, c5_new
        eta /= 2.0
    return h, evaluate_c5(h)

def mutate(v, mode='gaussian', params=None):
    """Create a mutated version of v."""
    n = len(v)
    v = v.copy()
    if mode == 'gaussian':
        # Perturb a random subset of indices
        k = np.random.randint(5, 11)
        indices = np.random.choice(n, k, replace=False)
        scale = params.get('sigma', 0.2) if params else 0.2
        v[indices] += np.random.normal(0, scale, k)
        return project_feasible(v, n)
    elif mode == 'swap':
        delta = params.get('delta', 0.01) if params else 0.01
        # Pick indices where we can transfer mass
        mask_lo = v < 1 - delta
        mask_hi = v > delta
        if np.sum(mask_lo) < 1 or np.sum(mask_hi) < 1:
            return v  # Don't swap if impossible
        i = np.random.choice(np.where(mask_hi)[0])
        j = np.random.choice(np.where(mask_lo)[0])
        amount = np.random.uniform(0.001, min(v[i], 1 - v[j], delta))
        v[i] -= amount
        v[j] += amount
        return project_feasible(v, n)
    elif mode == 'dct_tweak':
        # Modify a low frequency DCT coefficient
        try:
            from scipy.fft import dct, idct
            if params is None:
                params = {}
            sig = params.get('sigma', 0.02)
            num_coeffs = params.get('num', 2)
            # DCT transform
            v_dct = dct(v, type=2, norm='ortho')
            # Modify a few low-frequency coefficients
            indices = np.random.choice(len(v_dct), num_coeffs, replace=False)
            changes = np.random.normal(0, sig, num_coeffs)
            v_dct[indices] += changes
            # Inverse
            v_new = idct(v_dct, type=2, norm='ortho')
            # Clip and reproject
            v_new = np.clip(v_new, 0, 1)
            return project_feasible(v_new, n)
        except:
            return v  # fallback to no change
    else:
        return v

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 250  # Start coarse
    dx = 2.0 / n_points
    M = 2000  # Initial pool size
    K = 30    # Keep top K candidates
    
    # Phase 1: Generate initial pool
    pool = []
    for _ in range(M):
        u = np.random.rand(n_points)
        h = project_feasible(u, n_points)
        c5 = evaluate_c5(h)
        pool.append((h, c5))
    pool.sort(key=lambda x: x[1])
    pool = pool[:K]
    
    # Phase 2: Mutation loop, 0.25 budget
    t_end = start_time + 0.25 * budget_s
    iteration = 0
    while time.time() < t_end:
        iteration += 1
        parent = np.random.choice(len(pool))
        h_parent, _ = pool[parent]
        modes = ['gaussian', 'swap', 'dct_tweak']
        mode = np.random.choice(modes)
        params = {}
        if mode == 'gaussian':
            params['sigma'] = max(0.02, 0.2 * 0.999 ** iteration)
        elif mode == 'dct_tweak':
            params['sigma'] = 0.02
            params['num'] = np.random.randint(1, 4)
        
        h_child = mutate(h_parent, mode, params)
        c5_child = evaluate_c5(h_child)
        
        if len(pool) > 0 and c5_child < pool[-1][1]:
            # Replace worst element
            pool = pool[:-1] + [(h_child, c5_child)]
            pool.sort(key=lambda x: x[1])
    
    # Phase 3: Local refinement for top pool elements, 0.45 budget
    t_end = start_time + 0.7 * budget_s
    for i in range(min(5, len(pool))):
        h_ref = pool[i][0].copy()
        c5_ref = pool[i][1]
        t_inner = time.time() + 0.45 * budget_s / 5
        for _ in range(2000):
            if time.time() > t_inner or time.time() > t_end:
                break
            # Find worst shift
            corr = correlate(h_ref, 1 - h_ref, mode='full')
            k_star = np.argmax(corr)
            if k_star > len(h_ref):
                k_star = k_star - len(h_ref)
            else:
                k_star = k_star - len(h_ref) + 1
                
            grad = compute_gradient(h_ref, k_star)
            h_new, c5_new = line_search(h_ref, grad)
            if c5_new < c5_ref:
                h_ref = h_new
                c5_ref = c5_new
            else:
                break  # no improvement, stop
        pool[i] = (h_ref, c5_ref)

    # Final refinement: swap heuristic
    h_best, c5_best = pool[0]
    t_final = start_time + 0.9 * budget_s
    for _ in range(10000):
        if time.time() > t_final:
            break
        # Swap small amounts between pair of indices
        i, j = np.random.choice(n_points, 2, replace=False)
        if not (0.01 < h_best[i] < 0.99 and 0.01 < h_best[j] < 0.99):
            continue
        delta = np.random.uniform(1e-4, 1e-2)
        # Try transfer of mass from i to j
        if h_best[i] > delta and h_best[j] < 1 - delta:
            h_trial = h_best.copy()
            h_trial[i] -= delta
            h_trial[j] += delta
            h_trial = project_feasible(h_trial, n_points)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < c5_best:
                h_best = h_trial
                c5_best = c5_trial

    return (h_best.tolist(), c5_best, n_points)
