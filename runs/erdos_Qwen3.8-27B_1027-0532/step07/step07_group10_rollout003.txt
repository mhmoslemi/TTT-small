def frank_wolfe(h_init, n, target, time_limit=60, max_iter=10000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = 1.0  # Will use 2/(t+2) schedule or backtracking
    no_improve = 0
    
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active set
        c_max = np.max(corr)
        # Use a small tolerance for active set
        active = np.where(corr >= c_max - 1e-10 * c_max)[0]
        
        if len(active) > 20:
            # Only keep top 20
            active = np.argsort(corr)[-20:]
        
        # Compute subgradient
        grad = np.zeros(n)
        for idx in active:
            k = int(idx) - (n - 1)
            # ∂corr[idx]/∂h[j] = (1-h[j+k])*1[0≤j+k<n] - h[j-k]*1[0≤j-k<n]
            jp = idxs + k
            jm = idxs - k
            mask_p = (jp >= 0) & (jp < n)
            mask_m = (jm >= 0) & (jm < n)
            grad[mask_p] += (1.0 - h[jp[mask_p]])
            grad[mask_m] -= h[jm[mask_m]]
        
        grad /= len(active)
        
        # Frank-Wolfe step: min <grad, h'> s.t. 0≤h'≤1, sum(h')=target
        # Optimal: h'=1 where grad is smallest (target elements), h'=0 elsewhere
        sorted_idx = np.argsort(grad)
        h_fw = np.zeros(n)
        h_fw[sorted_idx[:int(target)]] = 1.0
        
        # Step size: use 2/(t+2) or backtracking
        # h_new = (1-tau)*h + tau*h_fw
        tau = 2.0 / (t + 2.0)
        h_new = (1.0 - tau) * h + tau * h_fw
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < c5_val - 1e-15:
            h = h_new
        else:
            # Try smaller step
            tau = 0.5 * tau
            h_new = (1.0 - tau) * h + tau * h_fw
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < c5_val - 1e-15:
                h = h_new
            else:
                # Skip this step
                pass
    
    return best_h, best_score

                wp = (idxs + k < n)
                wm = (idxs - k >= 0)
                grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
                grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
            return grad
        def cons_fun(x):
            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)
        try:
            res = minimize(objective, h_cur, jac=objective_grad,
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def gradient_descent_active(h_init, n, target, time_limit=60, max_steps=50000,
                            base_lr=0.03, n_active=8):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = base_lr
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.9
                if lr < 1e-7:
                    break
        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[0]]
        weights = np.exp(-50.0 * (c_max - corr[top_shifts]))
        weights /= np.sum(weights)
        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val - 1e-15:
            h = h_try
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    if tleft() > 100:
        h_fw, sc_fw = frank_wolfe(best_h, best_n, best_target,
                                  time_limit=min(120, tleft() * 0.45),
                                  max_iter=80000)
        if sc_fw < best_score:
            best_score = sc_fw
            best_h = h_fw

    if tleft() > 80:
        h_cd, sc_cd = pair_cd_fast(best_h, best_n, best_target,
                                   time_limit=min(100, tleft() * 0.4),
                                   max_sweeps=15000,
                                   n_cand_start=20, n_cand_end=55,
                                   max_no_improve=350)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 60:
        h_gd, sc_gd = gradient_descent_active(best_h, best_n, best_target,
                                              time_limit=min(50, tleft() * 0.35),
                                              max_steps=40000, base_lr=0.03,
                                              n_active=10)
        if sc_gd < best_score:
            best_score = sc_gd
            best_h = h_gd

    if tleft() > 80:
        h_pol, sc_pol = slsqp_round(best_h, best_n, best_target,
                                    [200, 1000, 5000], 800, 1e-14)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    if tleft() > 60:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.08, 0.15]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            h_p, sc_p = frank_wolfe(h_pert, best_n, best_target,
                                    time_limit=min(40, tleft() * 0.4),
                                    max_iter=30000)
            if sc_p < best_score:
                best_score = sc_p
                best_h = h_p

    n1 = 600
    if tleft() > 250:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 200:
            h1_fw, sc1_fw = frank_wolfe(h1, n1, target1,
                                        time_limit=min(100, tleft() * 0.4),
                                        max_iter=50000)
            if sc1_fw < sc1:
                h1 = h1_fw
                sc1 = sc1_fw
        if tleft() > 120:
            h1_cd, sc1_cd = pair_cd_fast(h1, n1, target1,
                                         time_limit=min(100, tleft() * 0.4),
                                         max_sweeps=15000,
                                         n_cand_start=20, n_cand_end=60,
                                         max_no_improve=300)
            if sc1_cd < best_score:
                best_score = sc1_cd
                best_h = h1_cd
                best_n = n1
                best_target = target1
            elif sc1_cd < sc1:
                h1 = h1_cd
                sc1 = sc1_cd
        if tleft() > 60:
            h1_pol, sc1_pol = slsqp_round(h1, n1, target1,
                                          [500, 3000], 600, 1e-14)
            if sc1_pol < best_score:
                best_score = sc1_pol
                best_h = h1_pol
                best_n = n1
                best_target = target1

    n2 = 800
    if tleft() > 350:
        target2 = n2 / 2.0
        if best_n == n1:
            h2 = interpolate_to(best_h, n2, target2)
        else:
            h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 280:
            h2_fw, sc2_fw = frank_wolfe(h2, n2, target2,
                                        time_limit=min(120, tleft() * 0.4),
                                        max_iter=40000)
            if sc2_fw < sc2:
                h2 = h2_fw
                sc2 = sc2_fw
        if tleft() > 180:
            h2_cd, sc2_cd = pair_cd_fast(h2, n2, target2,
                                         time_limit=min(120, tleft() * 0.4),
                                         max_sweeps=12000,
                                         n_cand_start=25, n_cand_end=70,
                                         max_no_improve=300)
            if sc2_cd < best_score:
                best_score = sc2_cd
                best_h = h2_cd
                best_n = n2
                best_target = target2
            elif sc2_cd < sc2:
                h2 = h2_cd
                sc2 = sc2_cd
        if tleft() > 80:
            h2_pol, sc2_pol = slsqp_round(h2, n2, target2,
                                          [1000, 5000], 800, 1e-14)
            if sc2_pol < best_score:
                best_score = sc2_pol
                best_h = h2_pol
                best_n = n2
                best_target = target2

    if tleft() > 80:
        h_pol2, sc2 = slsqp_round(best_h, best_n, best_target,
                                  [5000, 20000], 1200, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 60:
        h_cd_final, c5_final = pair_cd_fast(best_h, best_n, best_target,
                                            time_limit=min(80, tleft() * 0.6),
                                            max_sweeps=10000,
                                            n_cand_start=30, n_cand_end=70,
                                            max_no_improve=250)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 40:
        h_gd_final, c5_gf = gradient_descent_active(best_h, best_n, best_target,
                                                    time_limit=min(30, tleft() * 0.5),
                                                    max_steps=20000, base_lr=0.01,
                                                    n_active=15)
        if c5_gf < best_score:
            best_score = c5_gf
            best_h = h_gd_final

    if tleft() > 30:
        h_pol3, sc3 = slsqp_round(best_h, best_n, best_target,
                                  [50000, 100000], 800, 1e-14)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    if tleft() > 20:
        rng2 = np.random.RandomState(seed + 7)
        for ns in [0.02, 0.05]:
            if tleft() < 15:
                break
            h_prt = best_h + rng2.randn(best_n) * ns
            h_prt = project_feasible(h_prt, best_target)
            h_ref, sc_ref = slsqp_round(h_prt, best_n, best_target,
                                        [2000, 20000], 500, 1e-14)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, best_n
