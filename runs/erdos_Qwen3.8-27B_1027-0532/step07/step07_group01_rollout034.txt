```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, alpha):
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(corr, alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
```

∂LSE/∂h[i] = (1/α) * sum_m exp(α*(corr[m] - max)) / sum_m' exp(α*(corr[m'] - max)) * ∂corr[m]/∂h[i]

The weights w[m] = exp(α*(corr[m] - max)) / sum(exp(α*(corr - max))) are the "softmax" weights.

So the gradient is a weighted average of the individual correlation gradients, with weights given by softmax.

This is much smoother than the subgradient of the max, and should allow for more effective optimization.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def proj_box_sum(v, target):
    lo = -(np.max(v) + 2.0)
    hi = -(np.min(v) - 2.0)
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(v + lam, 0.0, 1.0)

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def lse_gradient(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    weight_sum = np.sum(weights)
    if weight_sum < 1e-300:
        weight_sum = 1.0
    
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    
    # Only consider significant weights
    threshold = 1e-10
    sig_mask = weights / weight_sum > threshold
    if not sig_mask.any():
        sig_mask = np.zeros(len(weights), dtype=bool)
        sig_mask[np.argmax(weights)] = True
    
    for m in range(len(weights)):
        w = weights[m] / weight_sum
        if w < threshold:
            continue
        k = m - mid
        if k >= n or k < -(n-1):
            continue
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if k == 0:
            # corr[mid] = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
            # d/dh[i] = 1 - 2*h[i]
            grad[v1] += w * (1.0 - 2.0 * h[v1])
        else:
            if v1.any():
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
            if v2.any():
                grad[v2] -= w * h[idxs[v2] + k]
    
    return grad
```

Let me implement block moves:

A block move of type "shift": select a block [i, i+m-1] and shift it left or right by 1 position.
This means: h_new[i] = h[i+1], h_new[i+1] = h[i+2], ..., h_new[i+m-1] = h[i+m]
(or the reverse for left shift)

Actually, this changes the sum, so we'd need to re-project. Let me think of a different block move.

**Block move type "ramp"**: Add a linear ramp to a block:
h_new[j] = h[j] + delta * (j - i + 0.5) / m for j in [i, i+m-1]
This has zero mean over the block (if the ramp is symmetric), so the sum is preserved (approximately).

Actually, for the sum to be exactly preserved, I need: sum_{j=i}^{i+m-1} (j - i + 0.5)/m = sum_{k=0}^{m-1} (k+0.5)/m = (1/2 * m*(m-1) + m/2) / m = (m-1)/2 + 1/2 = m/2
So the sum of the ramp is delta * m/2, which is not zero unless delta = 0.

Let me use a zero-mean ramp instead:
ramp[j] = (j - i) / (m-1) - 0.5 for j in [i, i+m-1]
Sum of ramp = 0 (by symmetry)
So h_new = h + delta * ramp preserves the sum.

This is a "tilt" of the block. It's useful for adjusting the slope of features.

But implementing this in the CD framework (with line search) is more complex. Let me skip block moves for now and focus on other improvements.

```python
def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=100, max_no_improve=400,
            restart_every=150, restart_scale=0.05, n_shifts=5,
            seed_rng=777):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(seed_rng)
    
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        if consecutive_no_improve > restart_every and consecutive_no_improve % 30 == 0:
            scale = restart_scale * (1.0 + 0.5 * np.sin(consecutive_no_improve * 0.1))
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0
        
        # Multi-shift gradient
        top_shifts = np.argsort(corr)[-n_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0
        
        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if v1.any():
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
            if v2.any():
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def proj_box_sum(v, target):
    lo = -np.max(v) - 2.0
    hi = -np.min(v) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(v + lam, 0.0, 1.0)

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if v1.any():
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
            if v2.any():
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def momentum_grad_descend(h_init, n, target, time_limit,
                          alpha_seq=None, beta=0.9, eta_init=0.03,
                          max_iters=30000):
    start = time.time()
    idxs = np.arange(n)
    mid = n - 1

    if alpha_seq is None:
        alpha_seq = [100, 500, 2000, 10000, 50000, 200000]

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    eta

    for alpha in alpha_seq:
        if time.time() - start > time_limit - 10:
            break
        no_improve = 0
        for it in range(max_iters):
            if time.time() - start > time_limit:
                break
            if no_improve > 4000:
                break

            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = np.max(corr)
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -500.0, 0.0)
            weights = np.exp(shifted)
            weight_sum = np.sum(weights)
            if weight_sum < 1e-300:
                weight_sum = 1.0

            grad = np.zeros(n)
            threshold = 1e-8
            for m in range(len(weights)):
                w = weights[m] / weight_sum
                if w < threshold:
                    continue
                k = m - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if v1.any():
                    grad[v1] += w * (1.0 - h[idxs[v1] - k])
                if v2.any():
                    grad[v2] -= w * h[idxs[v2] + k]

            grad -= np.mean(grad)
            v = beta * v - eta * grad
            h_new = h + v
            h_new = proj_box_sum(h_new, target)

            score = compute_c5(h_new)
            if score < best_score - 1e-14:
                best_score = score
                best_h = h_new.copy()
                h = h_new
                no_improve = 0
                if it % 200 == 0:
                    eta = min(eta * 1.01, 0.3)
            else:
                no_improve += 1
                if no_improve % 150 == 0:
                    eta = max(eta * 0.92, 0.001)
                    v *= 0.5

        v = np.zeros(n)

    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)
```
