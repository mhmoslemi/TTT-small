def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_cubic(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    try:
        cs = CubicSpline(x_old, h_src, bc_type='not-a-knot')
        h_out = cs(x_new)
    except Exception:
        h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import CubicSpline
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_cubic(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    try:
        cs = CubicSpline(x_old, h_src, bc_type='not-a-knot')
        h_out = cs(x_new)
    except Exception:
        h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def _slsqp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _slsqp_cons(x):
    return np.sum(x) - _g_target


def _slsqp_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change


def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]


def get_top_weights(corr, n_active):
    top_idx = np.argsort(corr)[-n_active:]
    w_sum = np.sum(corr[top_idx])
    if w_sum <= 0:
        return top_idx, np.ones(n_active) / n_active
    return top_idx, corr[top_idx] / w_sum


def compute_grad_from_lags(h, top_idx, weights, n):
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = int(top_idx[idx]) - mid
        w = weights[idx]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad -= np.mean(grad)
    return grad


def compute_grad_full(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-300:
        return np.zeros(n), corr
    w = w / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for lag in range(-n + 1, n):
        wi = w[lag + n - 1]
        if wi < 1e-14:
            continue
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 + lag:hi2 + lag]
    grad -= np.mean(grad)
    return grad, corr

                 n_cand_start=30, n_cand_end=80, max_no_improve=500,
                 bisection_iters=14, alpha_smooth=500, use_full_grad=False,
                 n_active_top=5, max_pairs=300):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    rng = np.random.RandomState(12345)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        if use_full_grad and sweep % 5 == 0:
            grad, corr = compute_grad_full(h, n, alpha_smooth)
        else:
            n_act = n_active_top
            top_idx, weights = get_top_weights(corr, n_act)
            grad = compute_grad_from_lags(h, top_idx, weights, n)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_h = len(high_set)
        n_l = len(low_set)
        pair_list = []
        if n_h * n_l <= max_pairs:
            for ii in range(n_h):
                for jj in range(n_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:

                lo = int(frac * (n_l - 1))
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pair_list.append((iv, jv))
            remaining = max_pairs - len(pair_list)
            for _ in range(remaining):

            delta_hi = min(h[i], 1.0 - h[j])

                continue

            g = change[j] - change[i]

            a_d = 0.0
            b_d = delta_hi

                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)

                continue

            approx_max = np.max(corr + delta_star * g) * dx

                best_i = i
                best_j = j
                best_delta = delta_star

                              use_full_grad=True, max_pairs=350)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    if tleft() > 60:
        h6 = slsqp_polish(best_h, n1, target1,
                          alphas=[50000, 200000],
                          maxiter_per_alpha=300)
        s6 = compute_c5(h6)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    if tleft() > 120:
        n2 = 1200
        target2 = n2 / 2.0
        h_hi = interpolate_cubic(best_h, n2, target2)
        s_hi = compute_c5(h_hi)
        if s_hi < best_score:
            best_score = s_hi
            best_h = h_hi
        if tleft() > 80:
            h7, s7 = ecd_optimize(best_h, n2, target2,
                                  time_limit=min(90, tleft() * 0.5),
                                  max_sweeps=25000,
                                  n_cand_start=40, n_cand_end=80,
                                  max_no_improve=300, bisection_iters=14,
                                  alpha_smooth=500, n_active_top=5, max_pairs=300)
            if s7 < best_score:
                best_score = s7
                best_h = h7
            if tleft() > 40:
                h7b = slsqp_polish(best_h, n2, target2,
                                   alphas=[5000, 20000],
                                   maxiter_per_alpha=300)
                s7b = compute_c5(h7b)
                if s7b < best_score:
                    best_score = s7b
                    best_h = h7b

    if tleft() > 50:
        nf = len(best_h)
        h8, s8 = ecd_optimize(best_h, nf, nf / 2.0,
                              time_limit=min(40, tleft() * 0.4),
                              max_sweeps=20000,
                              n_cand_start=60, n_cand_end=100,
                              max_no_improve=200, bisection_iters=16,
                              alpha_smooth=5000, n_active_top=8,
                              use_full_grad=True, max_pairs=350)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    if tleft() > 30:
        nf = len(best_h)
        h9 = slsqp_polish(best_h, nf, nf / 2.0,
                          alphas=[100000, 500000],
                          maxiter_per_alpha=200)
        s9 = compute_c5(h9)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
