            break

                break

            h_rest[bs:bs+block_size] = 0  # placeholder

                h_try = h_rest.copy()
                h_try[bs:bs+block_size] = x

                c_max = np.max(corr)

                corr = np.correlate(h_try, 1.0 - h_try, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

def compute_block_grad(w, h, bs, block_size, n):
    """Compute gradient of smooth max at block positions bs:bs+block_size."""
    # w has length 2n-1, h has length n
    # For position i, grad = sum_k w[k+n-1] * [(1-h[i+k])*valid - h[i-k]*valid]
    grad = np.zeros(block_size)
    for bi in range(block_size):
        i = bs + bi
        # First term: sum_k w_k * (1-h[i+k]) for 0<=i+k<n
        k_pos = np.arange(0, n - i)  # k values where i+k < n, i.e., k < n-i
        # But k can also be negative: i+k >= 0 means k >= -i
        k_valid = np.arange(max(-(n-1), -i), min(n-1, n-1-i) + 1)
        # Actually k ranges from -(n-1) to n-1, and we need 0<=i+k<n
        # So -i <= k <= n-1-i, intersected with [-(n-1), n-1]
        k_lo = max(-(n-1), -i)
        k_hi = min(n-1, n-1-i)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            wks = w[ks + (n-1)]
            grad[bi] += np.sum(wks * (1.0 - h[i + ks]))
        
        # Second term: -sum_k w_k * h[i-k] for 0<=i-k<n
        # i-k >= 0 means k <= i, i-k < n means k > i-n
        k_lo2 = max(-(n-1), i - n + 1)
        k_hi2 = min(n-1, i)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            wks2 = w[ks2 + (n-1)]
            grad[bi] -= np.sum(wks2 * h[i - ks2])
    return grad

            wks = w[ks + mid]
            grad[bi] += np.dot(wks, 1.0 - h[i + ks])
        k_lo2 = max(-(n - 1), i - n + 1)
        k_hi2 = min(n - 1, i)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            wks2 = w[ks2 + mid]
            grad[bi] -= np.dot(wks2, h[i - ks2])
    return grad


def slsqp_base(h, n, target, alpha, maxiter, ftol=1e-14):
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_val(corr, alpha) * dx

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        w = smooth_max_weights(corr, alpha)
        g = np.zeros(n)
        for k_idx in range(2 * n - 1):
            wk = w[k_idx]
            if wk < 1e-10:
                continue
            k = k_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v1] += wk * (1.0 - x[idxs[v1] - k])
            g[v2] -= wk * x[idxs[v2] + k]
        return g * dx

    def cons_fun(x):
        return np.sum(x) - target

    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()


def block_cd(h_init, n, target, time_limit, block_size=40, alpha=500,
             maxiter=150, n_passes=200, rng_seed=42):
    start = time.time()
    dx = 2.0 / n
    rng = np.random.RandomState(rng_seed)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    block_starts = list(range(0, n - block_size + 1, block_size))
    if not block_starts:
        return best_h, best_c5

    no_improve_passes = 0

    for outer in range(n_passes):
        if time.time() - start > time_limit:
            break

        rng.shuffle(block_starts)
        improved_this_pass = False

        for bs in block_starts:
            if time.time() - start > time_limit:
                break

            h_rest = h.copy()
            old_block = h[bs:bs + block_size].copy()
            h_rest[bs:bs + block_size] = 0.0
            block_sum_target = target - np.sum(h_rest)

            if block_sum_target < -1e-10 or block_sum_target > block_size + 1e-10:
                continue

            def obj_block(x):
                h_try = h_rest.copy()
                h_try[bs:bs + block_size] = x
                corr = np.correlate(h_try, 1.0 - h_try, mode='full')
                return smooth_max_val(corr, alpha) * dx

            def obj_grad_block(x):
                h_try = h_rest.copy()
                h_try[bs:bs + block_size] = x
                corr = np.correlate(h_try, 1.0 - h_try, mode='full')
                w = smooth_max_weights(corr, alpha)
                g = block_grad(w, h_try, bs, block_size, n)
                return g * dx

            def cons_block(x):
                return np.sum(x) - block_sum_target

            bounds = [(0.0, 1.0)] * block_size
            cons = [{'type': 'eq', 'fun': cons_block}]

            try:
                res = minimize(obj_block, old_block, jac=obj_grad_block,
                               method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-12})

                if hasattr(res, 'x') and len(res.x) == block_size:
                    new_block = np.clip(res.x, 0.0, 1.0)
                    h_new = h.copy()
                    h_new[bs:bs + block_size] = new_block
                    h_new = project_feasible(h_new, target)
                    c5_new = compute_c5(h_new)

                    if c5_new < cur_c5 - 1e-14:
                        h = h_new
                        cur_c5 = c5_new
                        if c5_new < best_c5 - 1e-15:
                            best_c5 = c5_new
                            best_h = h.copy()
                        improved_this_pass = True
            except Exception:
                pass

        if improved_this_pass:
            no_improve_passes = 0
        else:
            no_improve_passes += 1
            if no_improve_passes > 8:
                break

    return best_h, best_c5


def pair_cd_quick(h_init, n, target, time_limit, max_sweeps=30000,
                  n_cand=40, top_k_eval=30, n_shifts=2, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_shifts = np.argsort(corr)[-n_shifts:]
        k_star_idx = int(top_shifts[-1])

        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(10):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fourier_perturb(h, n, target, rng, n_modes, amplitude):
    x = np.linspace(0.0, 2.0, n)
    perturbation = np.zeros(n)
    for k in range(1, n_modes + 1):
        a = rng.randn() * amplitude
        b = rng.randn() * amplitude
        perturbation += a * np.sin(np.pi * k * x) + b * np.cos(np.pi * k * x)
    h_pert = h + perturbation
    return project_feasible(h_pert, target)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    tl = _time_left(start_time, budget_s)
    if tl > 50:
        h_pol = slsqp_base(best_h, n0, target0, alpha=2000, maxiter=300)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    tl = _time_left(start_time, budget_s)
    if tl > 200:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    tl = _time_left(start_time, budget_s)
    if tl > 150:
        h_bcd, c5_bcd = block_cd(best_h, 800, 400.0,
                                 time_limit=min(350, tl * 0.75),
                                 block_size=40, alpha=500,
                                 maxiter=150, n_passes=300, rng_seed=seed)
        if c5_bcd < best_score:
            best_score = c5_bcd
            best_h = h_bcd

    tl = _time_left(start_time, budget_s)
    if tl > 100:
        h_pc, c5_pc = pair_cd_quick(best_h, 800, 400.0,
                                    time_limit=min(150, tl * 0.7),
                                    max_sweeps=30000, n_cand=45,
                                    top_k_eval=40, n_shifts=3,
                                    max_no_improve=500)
        if c5_pc < best_score:
            best_score = c5_pc
            best_h = h_pc

    tl = _time_left(start_time, budget_s)
    if tl > 80:
        h_pol = slsqp_base(best_h, 800, 400.0, alpha=5000, maxiter=400)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    tl = _time_left(start_time, budget_s)
    if tl > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

        tl = _time_left(start_time, budget_s)
        if tl > 200:
            h_bcd2, c5_bcd2 = block_cd(best_h, n2, target2,
                                       time_limit=min(300, tl * 0.75),
                                       block_size=40, alpha=500,
                                       maxiter=150, n_passes=250, rng_seed=seed + 1)
            if c5_bcd2 < best_score:
                best_score = c5_bcd2
                best_h = h_bcd2

    tl = _time_left(start_time, budget_s)
    if tl > 100:
        h_pc2, c5_pc2 = pair_cd_quick(best_h, len(best_h), len(best_h) / 2.0,
                                      time_limit=min(120, tl * 0.7),
                                      max_sweeps=25000, n_cand=50,
                                      top_k_eval=45, n_shifts=3,
                                      max_no_improve=500)
        if c5_pc2 < best_score:
            best_score = c5_pc2
            best_h = h_pc2

    tl = _time_left(start_time, budget_s)
    if tl > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        rng = np.random.RandomState(seed + 100)
        for amp in [0.03, 0.08, 0.15]:
            tl = _time_left(start_time, budget_s)
            if tl < 60:
                break
            h_pert = fourier_perturb(best_h, n_cur, target_cur, rng, n_modes=5, amplitude=amp)
            h_ref, c5_ref = block_cd(h_pert, n_cur, target_cur,
                                     time_limit=min(40, tl * 0.4),
                                     block_size=40, alpha=500,
                                     maxiter=100, n_passes=80, rng_seed=seed + 200)
            if c5_ref < best_score:
                best_score = c5_ref
                best_h = h_ref

    tl = _time_left(start_time, budget_s)
    if tl > 80:
        h_bcd_final, c5_bcd_final = block_cd(best_h, len(best_h), len(best_h) / 2.0,
                                             time_limit=min(100, tl * 0.7),
                                             block_size=25, alpha=1000,
                                             maxiter=200, n_passes=200, rng_seed=seed + 300)
        if c5_bcd_final < best_score:
            best_score = c5_bcd_final
            best_h = h_bcd_final

    tl = _time_left(start_time, budget_s)
    if tl > 50:
        n_cur = len(best_h)
        h_final_pol = slsqp_base(best_h, n_cur, n_cur / 2.0,
                                 alpha=10000, maxiter=300)
        sc = compute_c5(h_final_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
