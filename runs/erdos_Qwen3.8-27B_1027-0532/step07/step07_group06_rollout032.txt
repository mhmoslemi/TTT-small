import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _topk_obj(x, k):
    corr = np.correlate(x, 1.0 - x, mode='full')
    topk = np.sort(corr)[-k:]
    return float(np.mean(topk) ** 0.5 * np.max(corr) ** 0.5)

def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

def _cons_fun(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()

        try:

                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):

def admm_solve(h_init, n, target, time_limit, start_time, rho=10.0, max_iter=5000):

    u = np.zeros(n_corr)

            break

        # where A(x) = corr(x), subject to sum(x) = target, 0 <= x <= 1

        for k_idx in range(n_corr):

            if abs(r) < 1e-15:
                continue
            k = shifts[k_idx]
            if k >= 0:
                for i in range(n - k):
                    grad[i] += rho * r * (1.0 - h[i+k])
                    # d/dh[i+k] of h[i]*(1-h[i+k]) = -h[i]
                    if i + k < n:

            else:

                for i in range(n - kk):
                    grad[i+kk] += rho * r * (1.0 - h[i])
                    grad[i] += rho * r * (-h[i+kk])
        
        # Projected gradient step
        lr = 1.0 / (rho * n)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        # t-update (projection onto t >= max(corr))
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        t_new = max(t_val, np.max(corr_new))
        
        # u-update
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        u = u + rho * (corr_new - t_new)
        
        h = h_new
        t_val = t_new
        
        sc = compute_c5(h)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h.copy()
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def stochastic_refine_v2(h_init, n, target, time_limit, start_time, 
                          max_steps=50000, base_scale=0.02, n_pert_range=(1,4)):
    rng = np.random.RandomState(42)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    temperature = 0.01
    
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 5000:
            break
        
        scale = base_scale * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.randint(n_pert_range[0], n_pert_range[1] + 1))
        n_pert = min(n_pert, n)
        perm = rng.permutation(n)
        idxs = perm[:n_pert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        
        # Simulated annealing acceptance
        if sc < cur_c5:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            # Accept with probability exp(-(sc - cur_c5) / temperature)
            delta = sc - cur_c5
            if delta < temperature:
                accept_prob = np.exp(-delta / temperature)
                if rng.random() < accept_prob:
                    h = h_try
                    cur_c5 = sc
            no_improve += 1
            if no_improve > 200:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0
        
        # Cool down
        temperature *= 0.9999
    
    return best_h, best_c5
