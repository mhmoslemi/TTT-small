def subgradient_with_backtrack(h_init, n, target, time_limit, max_iter=20000, lr_start=0.005):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    lr = lr_start

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.05, lr_start)
        else:
            no_improve += 1
            lr *= 0.95

        max_corr = np.max(corr)
        tol = max(1e-10, 1e-6 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)

        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val - 1e-15:
            h = h_new

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h_start, n, target, alpha_list, maxiter_list, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for i in range(len(alpha_list)):
        alpha = alpha_list[i]
        maxiter = maxiter_list[i]
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score - 1e-15:
                    cur_score = sc
                    h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    n = len(h)

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
