_g_target = 0.0

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad

def corr_gradient_multi(h, k_stars, weights):

        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]

def local_search_gradient(h_init, n, target, time_limit, max_sweeps=10000):
    """Gradient-based local search: move mass in the direction of steepest descent."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    step_size = 0.01
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            step_size = min(step_size * 1.05, 0.05)
        else:
            no_improve += 1
            if no_improve > 50:
                step_size *= 0.8
            if no_improve > 300:
                break
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        # Compute gradient of active constraint
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        # Also consider nearly-active constraints
        active_mask = corr >= corr[k_star_idx] - 5e-5
        if active_mask.sum() > 1:
            extra_grad = np.zeros(n)
            for k_idx in np.where(active_mask)[0]:
                k = k_idx - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                extra_grad[v1] += (1.0 - h[idxs[v1] - k])
                extra_grad[v2] -= h[idxs[v2] + k]
            extra_grad -= np.mean(extra_grad)
            grad = 0.5 * (grad + extra_grad)
        
        # Move in negative gradient direction
        step = step_size * grad / (np.max(np.abs(grad)) + 1e-15)
        h_new = h - step
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        
        if sc < c5_val:
            # Update change matrix for changed entries
            changed = np.where(np.abs(h_new - h) > 1e-10)[0]
            h = h_new
            for p in changed:
                change[p, :] = 0.0
                pos_m = (p + shifts >= 0) & (p + shifts < n)
                neg_m = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_m] += (1.0 - h[p + shifts[pos_m]])
                change[p, neg_m] -= h[p - shifts[neg_m]]
        else:
            step_size *= 0.7
    
    return best_h, best_c5

Actually, let me reconsider the bisection. We're looking for the minimum of f(δ) = max_m (corr[m] + δ * g[m]) over δ ∈ [δ_lo, δ_hi]. 

A faster approach: Instead of bisection, we can note that f(δ) is the upper envelope of lines {corr[m] + δ*g[m]}. The minimum of this upper envelope occurs where the "kink" is, i.e., where two lines cross and the maximum switches from one to the other.

An even faster approach: We can compute the subgradient condition directly. At the optimal δ*, 0 must be in the subgradient of f, meaning:
- max_{m: corr[m] + δ*g[m] = f(δ*)} g[m] ≤ 0
- min_{m: corr[m] + δ*g[m] = f(δ*)} g[m] ≥ 0

This is equivalent to: for the active set A(δ) = {m: corr[m] + δ*g[m] = f(δ)}, we need max(A(δ)) ≤ 0 and min(A(δ)) ≥ 0 (where max/min are of g values).

A simpler sufficient condition: max_m g[m] ≤ 0 (then f is non-increasing, so minimum is at δ_hi) or min_m g[m] ≥ 0 (then f is non-decreasing, so minimum is at δ_lo).

import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def corr_gradient_single(h, k_idx):
    n = len(h)
    mid = n - 1
    k = k_idx - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad


def cutting_plane_multi(h_init, n, target, time_limit, max_iters=8000, tol=1e-9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_f = []
    cuts_h = []
    max_cuts = 250
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1500:
            if len(cuts_g) > 30:
                keep = max(10, len(cuts_g) // 3)
                cuts_g = cuts_g[-keep:]
                cuts_f = cuts_f[-keep:]
                cuts_h = cuts_h[-keep:]
            no_improve = 0

        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])

        active_tol = 1e-6
        active_mask = corr >= corr[k_star] - active_tol
        active_indices = np.where(active_mask)[0]

        n_add = min(len(active_indices), 5)
        for k_idx in active_indices[:n_add]:
            g = corr_gradient_single(h, int(k_idx))
            cuts_g.append(g)
            cuts_f.append(float(corr[k_idx]))
            cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = cuts_g[-keep:]
            cuts_f = cuts_f[-keep:]
            cuts_h = cuts_h[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')

                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)

                    best_h = h.copy()
                break

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break

    return best_h, best_score


def pairwise_local_search(h_init, n, target, time_limit, max_sweeps=10000,
                          n_cand_start=50, n_cand_end=180, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:

                change[p, pos_m] += (1.0 - h[p + shifts[pos_m]])
                change[p, neg_m] -= h[p - shifts[neg_m]]

    return best_h, best_c5


def random_perturb_search(h_init, n, target, time_limit, seed=42,
                          n_sweeps_ls=2000, sigma_list=(0.03, 0.01, 0.005)):
    start = time.time()
    rng = np.random.RandomState(seed)
    h_best = h_init.copy().astype(float)
    best_score = compute_c5(h_best)

    for sigma in sigma_list:

            break
        h_pert = h_best + rng.randn(n) * sigma

        remaining = time_limit - (time.time() - start)

        h_ls, sc_ls = pairwise_local_search(h_pert, n, target,

    return h_best, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def gradient_descent_smooth(h_init, n, target, time_limit, max_iters=5000,
                            alpha_start=2000.0, alpha_end=50000.0):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    h_curr = h_init.copy().astype(float)
    h_prev = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)
    idxs = np.arange(n)
    momentum = 0.9

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum)) / 2.0
        y = h_curr + ((t_k - 1.0) / t_k) * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)

        frac = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        corr_y = compute_corr(y)
        c_max = np.max(corr_y)
        shifted = np.clip(alpha * (corr_y - c_max), -500.0, 0.0)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-30:
            p = np.ones(len(corr_y)) / len(corr_y)
        else:
            p = p / p_sum

        grad = np.zeros(n)
        for m in range(len(corr_y)):
            if p[m] < 1e-15:
                continue
            k = m - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += p[m] * (1.0 - y[idxs[v1] + k])
            grad[v2] -= p[m] * y[idxs[v2] - k]
        grad /= alpha

        step = 0.01
        for _ in range(8):
            h_new = y - step * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc <= compute_c5(h_curr) + 1e-12:
                break
            step *= 0.5

        h_prev = h_curr.copy()
        h_curr = h_new
        sc = compute_c5(h_curr)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_curr.copy()

    return best_h, best_score

                                             max_iters=6000, tol=1e-10)
        if sc_cp3 < best_score:
            best_score = sc_cp3
            best_h = h_cp3

    if tleft() > 100:
        h_gs, sc_gs = gradient_descent_smooth(best_h, n0, target0,
                                               time_limit=min(80, tleft() * 0.4),
                                               max_iters=3000,
                                               alpha_start=3000.0, alpha_end=80000.0)
        if sc_gs < best_score:
            best_score = sc_gs
            best_h = h_gs

    if tleft() > 80:
        h_ls4, sc_ls4 = pairwise_local_search(best_h, n0, target0,
                                               time_limit=min(60, tleft() * 0.5),
                                               max_sweeps=5000,
                                               n_cand_start=120, n_cand_end=200,
                                               max_no_improve=150)
        if sc_ls4 < best_score:
            best_score = sc_ls4
            best_h = h_ls4

    if tleft() > 50:
        h_final_sls, sc_final = slsqp_round(best_h, n0, target0,
                                             [10000, 50000], 400, 1e-14)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_sls

    if tleft() > 30:
        h_ls5, sc_ls5 = pairwise_local_search(best_h, n0, target0,
                                               time_limit=min(25, tleft() * 0.6),
                                               max_sweeps=3000,
                                               n_cand_start=150, n_cand_end=200,
                                               max_no_improve=100)
        if sc_ls5 < best_score:
            best_score = sc_ls5
            best_h = h_ls5

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
