This is a 1D piecewise linear minimax problem. The optimal delta is where the maximum is minimized, which occurs at a point where two (or more) of the lines corr[k] + delta*g_i[k] intersect at the maximum.

We can solve this efficiently:
1. Sort the corr[k] values
2. The maximum as a function of delta changes at breakpoints where two lines cross
3. Use a parametric search or binary search on the value t: "is there a delta such that corr[k] + delta*g_i[k] <= t for all k?"

- If g_i[k] > 0: delta <= (t - corr[k]) / g_i[k]
- If g_i[k] < 0: delta >= (t - corr[k]) / g_i[k]
- If g_i[k] = 0: need corr[k] <= t (else infeasible)

    increasing h[i] on corr[k_idx]."""

    mid = n - 1

def compute_change_matrix(h, n):

    mid = n - 1
    shifts = np.arange(-(n-1), n)  # shape (n_corr,)

    k_vals = shifts[None, :]  # (1, n_corr)
    ik = i_vals + k_vals  # (n, n_corr)
    ik_valid = (ik >= 0) & (ik < n)
    ikm = i_vals - k_vals  # (n, n_corr)
    ikm_valid = (ikm >= 0) & (ikm < n)
    
    # (1 - h[i+k]) where valid, 0 otherwise
    pos_term = np.where(ik_valid, 1.0 - h[ik[ik_valid].ravel()], 0.0)
    # Hmm, this is getting complicated with the indexing. Let me use a different approach.
    
    # Actually, let me just use the loop version but with numpy for the inner loop
    change = np.zeros((n, n_corr))

    #   change[i, k+mid] += (1-h[i+k])
    #   neg: k such that 0 <= i-k < n, i.e., i-n < k <= i

            targets = i + np.arange(k_min_pos, k_max_pos + 1)

    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for i in range(n):
        k_min_pos = -i
        k_max_pos = n - 1 - i
        if k_min_pos <= k_max_pos:
            ks = np.arange(k_min_pos, k_max_pos + 1)
            idxs_pos = ks + mid
            targets = i + ks
            change[i, idxs_pos] += (1.0 - h[targets])
        k_min_neg = i - (n - 1)
        k_max_neg = i
        if k_min_neg <= k_max_neg:
            ks = np.arange(k_min_neg, k_max_neg + 1)
            idxs_neg = ks + mid
            targets = i - ks
            change[i, idxs_neg] -= h[targets]
    return change


def pair_best_response_cd(h_init, n, target, time_limit, n_cand=40, n_bisect=30, max_no_improve=150):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(np.max(corr))
        c5_val = current_max * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_n = min(6, n_corr)
        top_shifts = np.argsort(corr)[-top_n:]
        c_top = corr[top_shifts[0]]
        wts = np.exp(-30.0 * (c_top - corr[top_shifts]))
        wts /= np.sum(wts)

        grad = np.zeros(n)
        for wi, si in zip(wts, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        nc = min(n_cand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:nc]
        high_set = sorted_idx[-nc:]

        best_improvement = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue

                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-13:
                    continue

                g = change[i] - change[j]

                pos_mask = g > 1e-14
                neg_mask = g < -1e-14
                zero_mask = np.abs(g) <= 1e-14

                lo = 0.0
                hi = current_max

                for _ in range(n_bisect):
                    mid_t = 0.5 * (lo + hi)
                    lb = d_lo
                    ub = d_hi
                    ok = True
                    if np.any(zero_mask):
                        if np.max(corr[zero_mask]) > mid_t:
                            ok = False
                    if ok and np.any(pos_mask):
                        cand_ub = np.min((mid_t - corr[pos_mask]) / g[pos_mask])
                        ub = min(ub, cand_ub)
                    if ok and np.any(neg_mask):
                        cand_lb = np.max((mid_t - corr[neg_mask]) / g[neg_mask])
                        lb = max(lb, cand_lb)
                    if ok and lb <= ub:
                        hi = mid_t
                    else:
                        lo = mid_t

                opt_t = hi
                if opt_t < current_max - 1e-12:
                    imp = current_max - opt_t
                    if imp > best_improvement:
                        lb = d_lo
                        ub = d_hi
                        if np.any(pos_mask):
                            ub = min(ub, np.min((opt_t - corr[pos_mask]) / g[pos_mask]))
                        if np.any(neg_mask):
                            lb = max(lb, np.max((opt_t - corr[neg_mask]) / g[neg_mask]))
                        if lb > ub:
                            best_i = i
                            best_j = j
                            best_delta = 0.5 * (d_lo + d_hi)
                            best_improvement = imp
                        else:
                            best_i = i
                            best_j = j
                            best_delta = 0.5 * (lb + ub)
                            best_improvement = imp

        if best_i < 0:
            break

        h[best_i] = np.clip(h[best_i] + best_delta, 0.0, 1.0)
        h[best_j] = np.clip(h[best_j] - best_delta, 0.0, 1.0)
        h = project_feasible(h, target)

        change = compute_change_matrix(h, n)

    return best_h, best_c5


def nesterov_subgrad(h_init, n, target, time_limit, max_steps=50000, base_lr=0.01):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = base_lr
    no_improve = 0
    beta = 0.9

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_n = min(5, len(corr))
        top_shifts = np.argsort(corr)[-top_n:]
        c_top = corr[top_shifts[0]]
        wts = np.exp(-40.0 * (c_top - corr[top_shifts]))
        wts /= np.sum(wts)

        grad = np.zeros(n)
        for wi, si in zip(wts, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        v = beta * v + (1.0 - beta) * grad
        h_try = h - lr * v
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < c5_val - 1e-15:
            h = h_try
        else:
            v = 0.0
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < c5_val - 1e-15:
                h = h_try
                v = grad
            else:
                lr *= 0.95
                if lr < 1e-7:
                    break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def block_cd(h_init, n, target, time_limit, block_size=3, n_cand=25, max_no_improve=200):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    rng = np.random.RandomState(123)

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(np.max(corr))
        c5_val = current_max * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_n = min(5, n_corr)
        top_shifts = np.argsort(corr)[-top_n:]
        c_top = corr[top_shifts[0]]
        wts = np.exp(-30.0 * (c_top - corr[top_shifts]))
        wts /= np.sum(wts)

        grad = np.zeros(n)
        for wi, si in zip(wts, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        nc = min(n_cand, n // 4)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:nc]
        high_set = sorted_idx[-nc:]

        change = compute_change_matrix(h, n)

        # Try random blocks of size 2 or 3
        best_improvement = 0.0
        best_update = None

        n_trials = 30
        for _ in range(n_trials):
            bs = rng.randint(2, block_size + 1)
            idxs_lo = rng.choice(low_set, size=bs // 2, replace=False)
            idxs_hi = rng.choice(high_set, size=bs - bs // 2, replace=False)

            # For a block, we solve a 1D problem: distribute delta among the "hi" indices
            # and -delta among the "lo" indices proportionally to gradient
            # Simplified: equal split
            n_hi = len(idxs_hi)
            n_lo = len(idxs_lo)

            g_block = np.zeros(n_corr)
            for idx in idxs_hi:
                g_block += change[int(idx)] / n_hi
            for idx in idxs_lo:
                g_block -= change[int(idx)] / n_lo

            # Delta bounds
            d_lo = max(max(-h[int(idx)] / n_hi for idx in idxs_hi),
                       max(h[int(idx)] - 1.0) / n_lo for idx in idxs_lo) if False else 0.0
            # More careful:
            d_lo_bounds = []
            d_hi_bounds = []
            for idx in idxs_hi:
                idx = int(idx)
                d_hi_bounds.append((1.0 - h[idx]) * n_hi)
                d_lo_bounds.append(-h[idx] * n_hi)
            for idx in idxs_lo:
                idx = int(idx)
                d_hi_bounds.append(h[idx] * n_lo)
                d_lo_bounds.append((h[idx] - 1.0) * n_lo)

            d_lo = max(d_lo_bounds) if d_lo_bounds else 0.0
            d_hi = min(d_hi_bounds) if d_hi_bounds else 1.0
            if d_hi - d_lo < 1e-12:
                continue

            pos_mask = g_block > 1e-14
            neg_mask = g_block < -1e-14
            zero_mask = np.abs(g_block) <= 1e-14

            lo_t = 0.0
            hi_t = current_max
            for _ in range(25):
                mid_t = 0.5 * (lo_t + hi_t)
                lb = d_lo
                ub = d_hi
                ok = True
                if np.any(zero_mask) and np.max(corr[zero_mask]) > mid_t:
                    ok = False
                if ok and np.any(pos_mask):
                    ub = min(ub, np.min((mid_t - corr[pos_mask]) / g_block[pos_mask]))
                if ok and np.any(neg_mask):
                    lb = max(lb, np.max((mid_t - corr[neg_mask]) / g_block[neg_mask]))
                if ok and lb <= ub:
                    hi_t = mid_t
                else:
                    lo_t = mid_t

            opt_t = hi_t
            if opt_t < current_max - 1e-12:
                imp = current_max - opt_t
                if imp > best_improvement:
                    best_improvement = imp
                    lb = d_lo
                    ub = d_hi
                    if np.any(pos_mask):
                        ub = min(ub, np.min((opt_t - corr[pos_mask]) / g_block[pos_mask]))
                    if np.any(neg_mask):
                        lb = max(lb, np.max((opt_t - corr[neg_mask]) / g_block[neg_mask]))
                    if lb > ub:
                        delta_opt = 0.5 * (d_lo + d_hi)
                    else:
                        delta_opt = 0.5 * (lb + ub)
                    best_update = (idxs_hi, idxs_lo, delta_opt)

        if best_update is not None:
            idxs_hi, idxs_lo, delta_opt = best_update
            for idx in idxs_hi:
                h[int(idx)] += delta_opt / len(idxs_hi)
            for idx in idxs_lo:
                h[int(idx)] -= delta_opt / len(idxs_lo)
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    # Phase 1: Quick SLSQP warmup
    if tleft() > 120:
        h1, sc1 = slsqp_round(best_h, best_n, best_target,
                              [50, 200, 1000, 5000], 800, 1e-15)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: Main CD (best-response)
    if tleft() > 250:
        t_alloc = min(tleft() * 0.4, 400.0)
        h2, sc2 = pair_best_response_cd(best_h, best_n, best_target,
                                        time_limit=t_alloc, n_cand=40,
                                        n_bisect=30, max_no_improve=200)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Nesterov subgradient
    if tleft() > 150:
        t_alloc = min(tleft() * 0.2, 150.0)
        h3, sc3 = nesterov_subgrad(best_h, best_n, best_target,
                                   time_limit=t_alloc, max_steps=80000, base_lr=0.008)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: SLSQP polish
    if tleft() > 100:
        h4, sc4 = slsqp_round(best_h, best_n, best_target,
                              [2000, 10000, 50000], 1200, 1e-15)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Another CD round
    if tleft() > 150:
        t_alloc = min(tleft() * 0.35, 250.0)
        h5, sc5 = pair_best_response_cd(best_h, best_n, best_target,
                                        time_limit=t_alloc, n_cand=50,
                                        n_bisect=30, max_no_improve=150)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: Block CD for diversity
    if tleft() > 100:
        t_alloc = min(tleft() * 0.3, 120.0)
        h6, sc6 = block_cd(best_h, best_n, best_target,
                           time_limit=t_alloc, block_size=3, n_cand=30,
                           max_no_improve=150)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: Try higher resolution
    n2 = 600
    if tleft() > 300:
        target2 = n2 / 2.0
        h_hi = interpolate_to(best_h, n2, target2)
        sc_hi = compute_c5(h_hi)

        if tleft() > 250:
            h_hi, sc_hi = slsqp_round(h_hi, n2, target2,
                                      [100, 500, 2000, 10000], 600, 1e-15)
        if tleft() > 150:
            t_alloc = min(tleft() * 0.4, 200.0)
            h_hi, sc_hi = pair_best_response_cd(h_hi, n2, target2,
                                                time_limit=t_alloc, n_cand=40,
                                                n_bisect=28, max_no_improve=150)
        if tleft() > 60:
            h_hi, sc_hi = slsqp_round(h_hi, n2, target2,
                                      [5000, 20000], 800, 1e-15)
        if sc_hi < best_score:
            best_score = sc_hi
            best_h = h_hi
            best_n = n2
            best_target = target2

    # Phase 8: Final SLSQP + CD
    if tleft() > 80:
        hf, scf = slsqp_round(best_h, best_n, best_target,
                              [10000, 50000, 100000], 1000, 1e-15)
        if scf < best_score:
            best_score = scf
            best_h = hf

    if tleft() > 60:
        t_alloc = min(tleft() * 0.5, 80.0)
        hc, sc_c = pair_best_response_cd(best_h, best_n, best_target,
                                         time_limit=t_alloc, n_cand=50,
                                         n_bisect=30, max_no_improve=100)
        if sc_c < best_score:
            best_score = sc_c
            best_h = hc

    # Phase 9: Perturbation restarts
    if tleft() > 60:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.08, 0.15]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            hp, scp = slsqp_round(h_pert, best_n, best_target,
                                  [500, 5000, 20000], 400, 1e-14)
            if scp < best_score:
                best_score = scp
                best_h = hp
            if tleft() > 40:
                t_alloc = min(30.0, tleft() * 0.3)
                hpc, scp2 = pair_best_response_cd(hp, best_n, best_target,
                                                  time_limit=t_alloc, n_cand=40,
                                                  n_bisect=25, max_no_improve=80)
                if scp2 < best_score:
                    best_score = scp2
                    best_h = hpc

    # Final validation
    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
