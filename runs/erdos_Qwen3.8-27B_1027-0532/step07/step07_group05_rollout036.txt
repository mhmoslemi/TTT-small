```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_logsumexp(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_eq(x):
    return np.sum(x) - _g_target

def cons_eq_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_logsumexp, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_logsumexp(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_eq_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_logsumexp, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change


def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]


def enhanced_cd(h_init, n, target, time_limit,
                max_sweeps=100000,
                n_active_start=10, n_active_end=30,
                n_cand_start=25, n_cand_end=55,
                max_no_improve=400,
                bisection_iters=12,
                max_pairs=40,
                weight_mode='proportional'):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        n_active = int(n_active_start + (n_active_end - n_active_start) * progress)
        n_active = min(n_active, n_corr)

        top_idx = np.argsort(corr)[-n_active:]
        if weight_mode == 'proportional':
            w_sum = np.sum(corr[top_idx])
            if w_sum <= 0:
                w_sum = 1.0
            weights = corr[top_idx] / w_sum
        else:
            weights = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_h = len(high_set)
        n_l = len(low_set)

        pair_list = []
        if n_h * n_l <= max_pairs:
            for ii in range(n_h):
                for jj in range(n_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pair_list.append((iv, jv))
        else:
            n_strata = min(8, n_h, n_l)
            for s in range(n_strata):
                ii = int(s * (n_h - 1) / max(1, n_strata - 1)) if n_strata > 1 else n_h // 2
                jj = int(s * (n_l - 1) / max(1, n_strata - 1)) if n_strata > 1 else n_l // 2
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))
            while len(pair_list) < max_pairs:
                ii = np.random.randint(n_h)
                jj = np.random.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i, j in pair_list:
            if i == j:
                continue

            delta_lo = -h[i]
            delta_hi = 1.0 - h[j]

            if delta_hi - delta_lo < 1e-12:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi

            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-12
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = float(np.max(corr + delta_star * g)) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5
