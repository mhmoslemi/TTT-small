def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_grad_matrix(h):
    n = len(h)
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    grad = np.zeros((n, 2 * n - 1))
    for p in range(2 * n - 1):
        k = shifts[p]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1, p] += (1.0 - h[idxs[v1] - k])
        grad[v2, p] -= h[idxs[v2] + k]
    return grad


def coordinate_descent(h_init, n, target, time_limit, n_cand=40, grid_size=40,
                       max_no_improve=300):

    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    grad_mat = compute_grad_matrix(h)

    no_improve = 0
    rng = np.random.RandomState(12345)

    while time.time() - start < time_limit:
        if no_improve > max_no_improve:
            break

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        active_mask = corr >= corr.max() - 1e-9
        grad = (grad_mat[active_mask].mean(axis=0) if active_mask.any()
                else grad_mat.argmax(axis=1).sum())
        if not isinstance(grad, np.ndarray):
            grad = np.zeros(n)
        grad_centered = grad - np.mean(grad)

        sorted_idx = np.argsort(grad_centered)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_h = len(high_set)
        n_l = len(low_set)

        j_l = perm[:try_count] % n_l
        high_arr = high_set[i_h]
        low_arr = low_set[j_l]

        d_eff = grad_mat[low_arr] - grad_mat[high_arr]

        h_i = h[high_arr]
        h_j = h[low_arr]
        d_min = np.maximum(h_i - 1.0, -h_j)
        d_max = np.minimum(h_i, 1.0 - h_j)

        valid = (d_max - d_min) > 1e-12
        if not valid.any():
            continue

        h_i_v = h_i[valid]
        h_j_v = h_j[valid]

        d_max_v = d_max[valid]
        d_eff_v = d_eff[valid]

        n_valid = len(h_i_v)
        if n_valid == 0:
            continue

        grid = np.linspace(d_min_v, d_max_v, grid_size).T
        corr_col = corr[None, :]
        vals = corr_col + grid[:, :, None] * d_eff_v[None, :]
        F_vals = vals.max(axis=2)

        best_local = F_vals.min(axis=0)
        best_local_idx = F_vals.argmin(axis=0)
        best_delta = grid[np.arange(n_valid), best_local_idx]

        baseline = c5_val
        improve_mask = best_local * dx < baseline - 1e-12
        if not improve_mask.any():
            continue

        improve_idx = np.where(improve_mask)[0]
        imp_best = best_local[improve_idx]
        best_ii = improve_idx[np.argmin(imp_best)]

        hi_pos = high_arr[valid[high_arr.searchsorted(high_arr) - 1] if False else np.where(valid)[0][best_ii]]
        lj_pos = low_arr[valid[low_arr.searchsorted(low_arr) - 1] if False else np.where(valid)[0][best_ii]]

        # Recompute properly
        valid_indices = np.where(valid)[0]
        best_py = valid_indices[best_ii]
        hi = int(high_arr[best_py])
        lj = int(low_arr[best_py])
        delta = float(best_delta[best_ii])

        if abs(delta) < 1e-12:
            continue

        h[hi] -= delta
        h[lj] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)

        corr = compute_corr(h)
        grad_mat = compute_grad_matrix(h)

    return best_h, best_c5


def binary_local_search(h_init, n, target, time_limit, max_no_improve=2000):

    shifts = np.arange(-(n - 1), n)
    rng = np.random.RandomState(999)

    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]

    target_ones = n // 2
    if n_ones > target_ones:
        h[ones[target_ones:]] = 0.0
    elif n_ones < target_ones:
        h[zeros[:target_ones - n_ones]] = 1.0
    h = project_feasible(h, target)

    corr = compute_corr(h)
    grad_mat = compute_grad_matrix(h)

            best_h = h.copy()

        else:
            no_improve += 1

        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:

        grad_centered = grad - np.mean(grad)

        i_picks = ones_idx[rng.randint(len(ones_idx), size=n_try)]
        j_picks = zeros_idx[rng.randint(len(zeros_idx), size=n_try)]

        d_eff = grad_mat[j_picks] - grad_mat[i_picks]

        h_i = h[i_picks]
        h_j = h[j_picks]
        d_min = np.maximum(h_i - 1.0, -h_j)
        d_max = np.minimum(h_i, 1.0 - h_j)
        valid = (d_max - d_min) > 1e-12
        if not valid.any():
            continue

        d_eff_v = d_eff[valid]

            continue

        # or we swap: delta = -1 if h[i]=1,h[j]=0 (move 1 from i to j)
        # Since binary, valid moves are delta = 1 (if h[i]=0,h[j]=1) or delta = -1 (if h[i]=1,h[j]=0)

        # For binary with h[i]=1, h[j]=0: delta=1 gives h[i]=0, h[j]=1
        # For binary with h[i]=0, h[j]=1: delta=-1 gives h[i]=1, h[j]=0

        best_c5_try = c5_val
        best_delta_try = 0.0
        best_pi = -1

                best_delta_try = trial_delta

        if best_pi >= 0:
            valid_indices = np.where(valid)[0]
            hi = int(i_picks[valid_indices[best_pi]])
            lj = int(j_picks[valid_indices[best_pi]])
            h[hi] = 0.0 if best_delta_try > 0 else 1.0
            h[lj] = 1.0 if best_delta_try > 0 else 0.0
            h = project_feasible(h, target)
            corr = compute_corr(h)
            grad_mat = compute_grad_matrix(h)

def multiplicative_weights(h_init, n, target, time_limit, eta=5.0,
                           n_inner=50, step_decay=0.999):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.05

    corr = compute_corr(h)
    n_corr = 2 * n - 1
    p = np.ones(n_corr) / n_corr

    while time.time() - start < time_limit:
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        # Update weights (softmax)
        log_p = -eta * corr
        log_p -= log_p.max()
        p = np.exp(log_p)
        p /= p.sum()

        # Compute weighted gradient
        grad = np.zeros(n)
        shifts = np.arange(-(n - 1), n)
        for m in range(n_corr):
            if p[m] < 1e-15:
                continue
            k = shifts[m]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += p[m] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= p[m] * h[idxs[v2] + k]

        grad_centered = grad - np.mean(grad)
        h_new = h - step * grad_centered
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < c5_val + 1e-10:
            h = h_new
        else:
            h_new = h - 0.5 * step * grad_centered
            h_new = project_feasible(h_new, target)
            h = h_new

        step *= step_decay
        step = max(step, 0.001)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
