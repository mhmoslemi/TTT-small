def gradient_line_search(h, n, target, time_limit, n_active=5, max_iters=5000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h_cur = h.copy().astype(float)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h_cur.copy()
        
        # Get active shifts
        top_k = np.argsort(corr)[-n_active:]
        weights = corr[top_k] / np.sum(corr[top_k])
        
        # Compute combined gradient
        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
            grad[v2] -= w * h_cur[idxs[v2] + k]
        
        # Remove mean to stay in the affine space of the constraint
        grad -= np.mean(grad)
        
        # Line search: find best step size
        # Move in direction -grad
        # Need to maintain 0 <= h <= 1
        max_step_up = min((1.0 - h_cur)[grad < 0] / (-grad[grad < 0])) if (grad < 0).any() else np.inf
        max_step_down = min(h_cur[grad > 0] / grad[grad > 0]) if (grad > 0).any() else np.inf
        max_step = min(max_step_up, max_step_down)
        
        if max_step < 1e-10:
            break
        
        # Try several step sizes
        best_step = 0
        best_f = c5_val
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = frac * max_step
            if step < 1e-12:
                continue
            h_trial = h_cur - step * grad
            h_trial = project_feasible(h_trial, target)
            f_trial = compute_c5(h_trial)
            if f_trial < best_f:
                best_f = f_trial
                best_step = step
        
        if best_step == 0:
            break
        
        h_cur = h_cur - best_step * grad
        h_cur = project_feasible(h_cur, target)
    
    return best_h, best_score

        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def fast_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=80, max_no_improve=500,
            n_active_shifts=4, reheat_interval=100, reheat_scale=0.03):

    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    best_c5 = compute_c5(h)
    best_h = h.copy()

    sweeps_since_improve = 0

            break

            best_c5 = c5_val
            best_h = h.copy()

            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

            break

            n_pert = max(5, n // 30)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 0.5 * reheat_scale:
                h = h_pert
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                continue

        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            weights = corr[top_k] / corr_top_sum
        
        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue

                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                # Skip if this move would increase the current max
                if g[k_star_idx] > 0.2 * np.max(np.abs(g)) + 1e-10:
                    continue

                b_d = delta_hi

                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-12
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    
    return best_h, best_c5

    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    alpha = 0.1
    stale = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h_cur.copy()
            stale = 0
            alpha = min(alpha * 1.05, 0.5)
        else:
            stale += 1
            alpha *= 0.97
            if stale > 300:
                break
        
        c_max = float(np.max(corr))
        threshold = c_max - 0.1 * max(c_max - corr.min(), 0.01)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        elif len(active) > n_active:
            active = np.argsort(corr)[-n_active:]
        
        grad = np.zeros(n)
        n_act = len(active)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = 1.0 / n_act
            grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
            grad[v2] -= w * h_cur[idxs[v2] + k]
        
        grad -= np.mean(grad)
        h_new = h_cur - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h_cur = h_new
    
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break

            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break

        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -100.0, 0.0)
    s = np.sum(np.exp(shifted))
    return float(c_max + np.log(s) / alpha)

            rng = np.random.RandomState(sweep * 13 + 7)
            n_pert = max(5, n // 30)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 0.5 * reheat_scale:
                h = h_pert
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                continue

        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            weights = corr[top_k] / corr_top_sum

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star_idx = int(np.argmax(corr))
        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                if g[k_star_idx] > 0.1 * np.max(np.abs(g)) + 1e-10:
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-12
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def gradient_method(h_init, n, target, time_limit, n_active=6, max_iters=20000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h_cur = h_init.copy().astype(float)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    alpha = 0.15
    stale = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h_cur.copy()
            stale = 0
            alpha = min(alpha * 1.05, 0.5)
        else:
            stale += 1
            alpha *= 0.97
            if stale > 400:
                break

        c_max = float(np.max(corr))
        spread = c_max - float(np.min(corr))
        threshold = c_max - 0.1 * max(spread, 0.01)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        elif len(active) > n_active:
            active = np.argsort(corr)[-n_active:]

        grad = np.zeros(n)
        n_act = len(active)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = 1.0 / n_act
            grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
            grad[v2] -= w * h_cur[idxs[v2] + k]

        grad -= np.mean(grad)
        h_new = h_cur - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h_cur = h_new

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    def elapsed():
        return time.time() - start_time

    def time_left():
        return max(0.0, budget_s - 25.0 - elapsed())

    # Phase 1: Aggressive SLSQP at n=400
    if time_left() > 120:
        deadline = time.time() + min(90, time_left() * 0.5)
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [30, 100, 400, 1500, 6000, 25000, 100000, 400000],
                                   1000, 1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD at n=400
    if time_left() > 100:
        h_cd, sc_cd = fast_cd(best_h, n0, target0,
                              time_limit=min(60, time_left() * 0.4),
                              max_sweeps=50000,
                              n_cand_start=25, n_cand_end=80,
                              max_no_improve=400,
                              n_active_shifts=5,
                              reheat_interval=80, reheat_scale=0.04)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 3: SLSQP again at n=400 (high alpha)
    if time_left() > 80:
        deadline = time.time() + min(50, time_left() * 0.35)
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [10000, 50000, 200000, 800000],
                                   1200, 1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Interpolate to n=800
    if time_left() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)

        if time_left() > 250:
            deadline = time.time() + min(80, time_left() * 0.3)
            h_pol, sc = slsqp_optimize(h1, n1, target1,
                                       [50, 200, 1000, 5000, 20000, 100000, 400000],
                                       1200, 1e-15, deadline=deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_pol
                current_n = n1
                current_target = target1
            else:
                best_h = h1
                current_n = n1
                current_target = target1

        if time_left() > 150:
            h_cd, sc_cd = fast_cd(best_h, current_n, current_target,
                                  time_limit=min(100, time_left() * 0.45),
                                  max_sweeps=40000,
                                  n_cand_start=30, n_cand_end=100,
                                  max_no_improve=500,
                                  n_active_shifts=5,
                                  reheat_interval=100, reheat_scale=0.03)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd

        if time_left() > 80:
            deadline = time.time() + min(55, time_left() * 0.35)
            h_pol, sc = slsqp_optimize(best_h, current_n, current_target,
                                       [5000, 20000, 100000, 500000],
                                       1500, 1e-15, deadline=deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 5: Interpolate to n=1600
    if time_left() > 350:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        if time_left() > 250:
            deadline = time.time() + min(100, time_left() * 0.28)
            h_pol, sc = slsqp_optimize(h2, n2, target2,
                                       [100, 500, 2000, 10000, 50000, 200000],
                                       1500, 1e-15, deadline=deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_pol
                current_n = n2
                current_target = target2
            else:
                current_n = n2
                current_target = target2
                best_h = h2

        if time_left() > 150:
            h_cd, sc_cd = fast_cd(best_h, current_n, current_target,
                                  time_limit=min(120, time_left() * 0.45),
                                  max_sweeps=25000,
                                  n_cand_start=25, n_cand_end=90,
                                  max_no_improve=400,
                                  n_active_shifts=5,
                                  reheat_interval=80, reheat_scale=0.03)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd

        if time_left() > 80:
            deadline = time.time() + min(60, time_left() * 0.35)
            h_pol, sc = slsqp_optimize(best_h, current_n, current_target,
                                       [10000, 50000, 200000, 800000],
                                       1500, 1e-15, deadline=deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 6: Multi-start perturbation
    if time_left() > 120:
        for attempt in range(5):
            if time_left() < 45:
                break
            noise_scales = [0.02, 0.04, 0.07, 0.1, 0.15]
            noise_scale = noise_scales[attempt % 5]
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(20, time_left() * 0.2)
            h_ref, sc_ref = slsqp_optimize(h_pert, current_n, current_target,
                                           [200, 1000, 5000, 25000, 100000],
                                           800, 1e-15, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 7: Gradient method refinement
    if time_left() > 50:
        h_sg, c5_sg = gradient_method(best_h, current_n, current_target,
                                      time_limit=min(35, time_left() * 0.5),
                                      n_active=8, max_iters=15000)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: Final SLSQP polish
    if time_left() > 25:
        deadline = time.time() + min(15, time_left() * 0.5)
        h_final, sc_final = slsqp_optimize(best_h, current_n, current_target,
                                           [50000, 200000, 800000, 3000000],
                                           1500, 1e-15, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
