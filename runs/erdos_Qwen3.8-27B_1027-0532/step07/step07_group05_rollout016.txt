import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothed_logsumexp(corr, alpha):
    c_max = np.max(corr)
    exp_vals = np.exp(alpha * (corr - c_max))
    return c_max + np.log(np.sum(exp_vals)) / alpha, exp_vals / np.sum(exp_vals)

def compute_grad_vectorized(h, w):
    n = len(h)
    mid = n - 1
    grad = np.zeros(n)
    for c in range(2 * n - 1):
        wc = w[c]
        if abs(wc) < 1e-15:
            continue
        s = c - mid
        j_start = max(0, -s)
        j_end = min(n, n - s)
        if j_start < j_end:
            grad[j_start:j_end] += wc * (1.0 - h[j_start + s:j_end + s])
        j_start2 = max(0, s)
        j_end2 = min(n, s + n)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= wc * h[j_start2 - s:j_end2 - s]
    return grad

def projected_gradient(h_init, n, target, time_limit, alpha_schedule, step_schedule, momentum=0.9, max_iter=5000):
    start = time.time()
    h = h_init.copy().astype(float)
    velocity = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        alpha = min(alpha_schedule, (it + 1) * 0.5 + 5.0)
        lse_val, weights = smoothed_logsumexp(corr, alpha)
        obj = lse_val * dx

        if obj < best_c5 - 1e-15:
            best_c5 = obj
            best_h = h.copy()

        grad = compute_grad_vectorized(h, weights) * dx
        grad -= np.mean(grad)

        step = min(step_schedule, 2.0 / (alpha + 1.0))
        velocity = momentum * velocity - step * grad
        h_new = h + velocity
        h_new = project_feasible(h_new, target)

        corr_new = compute_corr(h_new)
        lse_new, _ = smoothed_logsumexp(corr_new, alpha)
        obj_new = lse_new * dx

        if obj_new < obj + 0.01 * step * np.max(np.abs(grad)):
            h = h_new
        else:
            velocity = 0.5 * velocity
            h_new = h - 0.5 * step * grad
            h_new = project_feasible(h_new, target)
            h = h_new

    final_c5 = compute_c5(best_h)
    return best_h, min(best_c5, final_c5)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):

    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if n > 1000:
            break
        try:
            def obj(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                return smoothed_logsumexp(corr, alpha)[0]

            def obj_grad(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                _, w = smoothed_logsumexp(corr, alpha)
                return compute_grad_vectorized(x, w)

            res = minimize(obj, h_cur, jac=obj_grad,
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur
