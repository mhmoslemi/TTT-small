```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    """
    change[p, k_index] = how corr[k] changes when h[p] increases by 1
    """
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change
```

```python
                n_cand_start=30, n_cand_end=100, 
                max_no_improve=400, n_active=3,
                safety_margin=0.0):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(12345)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0

            break

        # Restart logic
        if consecutive_no_improve > 200 and consecutive_no_improve % 40 == 0:
            scale = 0.01 + 0.03 * (consecutive_no_improve // 200)
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0

        top_max = corr[top_idx[-1]]

        for s_idx in top_idx:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        stuck_factor = min(1.0, consecutive_no_improve / 100.0)
        base_ncand = n_cand_start + (n_cand_end - n_cand_start) * progress

        n_h = len(high_set)
        n_l = len(low_set)

            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:

                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:

                g = change[j] - change[i]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):

                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                
                if abs(delta_star) < 1e-13:

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:

                    best_triplet = (i, j, delta_star)

            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def improved_cd(h_init, n, target, time_limit, max_sweeps=60000,
                n_cand_start=30, n_cand_end=100,
                max_no_improve=500, n_active=5,
                restart_every=150, restart_scales=None):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    if restart_scales is None:
        restart_scales = [0.02, 0.03, 0.05, 0.08, 0.12]

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(42)
    restart_count = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_every and consecutive_no_improve % 30 == 0:
            scale = restart_scales[restart_count % len(restart_scales)]
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0
            restart_count += 1

        # Find active shifts
        n_act = min(n_active, n_corr)
        top_idx = np.argsort(corr)[-n_act:]
        top_sum = np.sum(corr[top_idx])
        if top_sum < 1e-15:
            top_sum = 1.0

        # Combined gradient
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Adaptive candidate set

        stuck_boost = 1.0 + 0.5 * min(1.0, consecutive_no_improve / 100.0)
        ncand = int((n_cand_start + (n_cand_end - n_cand_start) * progress) * stuck_boost)
        ncand = min(ncand, n // 3)
        ncand = max(ncand, 15)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)
```

```python
def fast_screen_pairs(h, change, corr, active_idx, idxs, n, n_screen=200, n_exact=30):
    """
    Fast screening: rank pairs by potential, then exact evaluation for top ones.
    active_idx: indices of active shifts in corr
    Returns best (i, j, delta, approx_c5)
    """
    mid = n - 1
    # For each position, compute potential at active shifts
    # change[p, k_idx] is how corr[k] changes when h[p] increases
    # We want to decrease high positions and increase low positions
    # For decreasing h[i]: corr changes by -change[i, :]
    # For increasing h[j]: corr changes by +change[j, :]
    # Net change: change[j, :] - change[i, :]
    # Approx quality: min over active k of (change[j,k] - change[i,k])
    # (we want this positive - it increases the min of active set)
    # Actually we want to DECREASE the max, so we want the change at the max to be negative
    # Let me reconsider...
    
    # We want to minimize max(corr + delta * g) where g = change[j] - change[i]
    # The max is at some index m. We want corr[m] + delta * g[m] < corr[m]
    # i.e., g[m] < 0
    # For the top active shifts, we want g[active] < 0
    # Approx: min over active of g[active] should be as negative as possible
    # = min over active of (change[j, active] - change[i, active])
    # = min(change[j, active]) - max(change[i, active])
    
    # For fast screening, compute for each p:
    #   worst_potential[p] = min over active k of change[p, k]
    #   best_potential[p] = max over active k of change[p, k]
    
    active_changes = change[:, active_idx]  # shape (n, n_active)
    min_pot = np.min(active_changes, axis=1)  # for each p, min over active
    max_pot = np.max(active_changes, axis=1)  # for each p, max over active
    
    # For pair (i,j): approximate improvement = min_pot[j] - max_pot[i]
    # We want this to be very negative (decrease the worst active shift a lot)
    
    # Sort by potential
    # "high" positions: high max_pot (decreasing them helps the most)
    # "low" positions: low min_pot (increasing them helps the most)
    
    high_order = np.argsort(-max_pot)  # most negative impact if decreased
    low_order = np.argsort(min_pot)  # most positive impact if increased
    
    n_h = min(n_screen, n // 3)
    n_l = min(n_screen, n // 3)
    
    high_candidates = high_order[:n_h]
    low_candidates = low_order[:n_l]
    
    # Score all pairs approximately
    # For pair (i,j): score = min_pot[j] - max_pot[i] (want very negative)
    # = min_pot[j] + (-max_pot[i])
    
    best_score = np.inf
    best_pair = None
    
    # Vectorized: for each i in high_candidates, compute score for all j
    for i in high_candidates:
        scores = min_pot[low_candidates] - max_pot[i]
        # Only consider valid pairs (i != j)
        valid = low_candidates != i
        if not valid.any():
            continue
        scores_valid = scores[valid]
        low_valid = low_candidates[valid]
        best_local = np.argmin(scores_valid)
        if scores_valid[best_local] < best_score:
            best_score = scores_valid[best_local]
            best_pair = (int(i), int(low_valid[best_local]))
    
    return best_pair
```

4. Swap perturbation (swap two random blocks)

This should help escape local minima more effectively.

**Alternative: Direct optimization of the min-autocorrelation**

Recall: C5 = n/2 * dx - min_k A_k where A_k = sum_i h[i]h[i+k].
So minimizing C5 = maximizing min_k A_k.

This is a max-min problem. I can use a **bisection** approach:
- Guess a target value T for min_k A_k
- Check if there exists h with sum(h) = n/2, 0 <= h <= 1, and A_k >= T for all k
- If yes, increase T; if no, decrease T

The feasibility check is a convex feasibility problem? No, because A_k is quadratic. But if I fix h and linearize A_k around h, I get a linear constraint:
A_k ≈ A_k(h) + sum_p (h_p+k + h_p-k) * (delta_p - mean(delta))

This is a linear constraint! So I can use a linearized approach:
1. At current h, compute A_k(h)
2. For the active set (where A_k is closest to T), add linear constraint: sum_p (h_p+k + h_p-k) * delta_p >= T - A_k(h)
3. Solve: minimize ||delta||^2 subject to linear constraints and box constraints
4. Update h, repeat

This is essentially a QCQP (quadratically constrained QP) or a sequence of QPs. Each QP is convex!

Solving this with scipy or cvxpy would be slow. But I can implement a simple projected gradient method for the QP:
- Minimize 0.5 * ||delta||^2 + T * max(0, T - A_k_linearized) subject to sum(delta) = 0, 0 <= h+delta <= 1

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_cd(h_init, n, target, time_limit, max_sweeps=100000,
            n_screen=150, n_exact=40, n_active=5,
            max_no_improve=400, restart_every=120):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(99)
    restart_count = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_every and consecutive_no_improve % 25 == 0:
            rtype = restart_count % 4
            if rtype == 0:
                scale = 0.01 + 0.02 * (restart_count // 4)
                h = best_h + rng.randn(n) * scale
            elif rtype == 1:
                lo = rng.randint(0, n - 40)
                hi = lo + rng.randint(20, 40)
                h = best_h.copy()
                h[lo:hi] += rng.randn(hi - lo) * 0.05
            elif rtype == 2:
                freq = rng.choice([2, 3, 5, 7, 11])
                phase = rng.uniform(0, 2 * np.pi)
                x = np.linspace(0, 2 * np.pi, n)
                pert = 0.03 * np.sin(freq * x + phase)
                h = best_h + pert
            else:
                scale = 0.05 + 0.03 * (restart_count % 3)
                h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0
            restart_count += 1

        n_act = min(n_active, n_corr)
        top_idx = np.argsort(corr)[-n_act:]

        grad = np.zeros(n)
        top_sum = np.sum(corr[top_idx])
        if top_sum < 1e-15:
            top_sum = 1.0
        for s_idx in top_idx:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        active_changes = change[:, top_idx]
        min_pot = np.min(active_changes, axis=1)
        max_pot = np.max(active_changes, axis=1)

        stuck_boost = 1.0 + 0.3 * min(1.0, float(consecutive_no_improve) / 80.0)
        ns = int(n_screen * stuck_boost)
        ns = min(ns, n // 2)
        ns = max(ns, 30)

        high_order = np.argsort(-max_pot)[:ns]
        low_order = np.argsort(min_pot)[:ns]

        scores = min_pot[low_order[None, :]] - max_pot[high_order[:, None]]
        for ii in range(len(high_order)):
            scores[ii, high_order[ii] == low_order] = np.inf

        exact_candidates = []
        for _ in range(min(n_exact, len(high_order) * len(low_order))):
            flat_idx = int(np.argmin(scores))
            ii = flat_idx // len(low_order)
            jj = flat_idx % len(low_order)
            i = int(high_order[ii])
            j = int(low_order[jj])
            if i == j:
                scores[ii, jj] = np.inf
                continue
            exact_candidates.append((i, j))
            scores[ii, jj] = np.inf

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for (i, j) in exact_candidates:
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(14):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)
```
