def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):

            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def pair_cd_v2(h_init, n, target, time_limit, max_sweeps=50000,

def subgradient_descent(h_init, n, target, time_limit, n_iter=10000, 
                        step_init=0.01, top_k=10):
    """Subgradient method focused on active set."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        # Find active shifts (top-k)
        active_idx = np.argsort(corr)[-top_k:]
        
        # Compute subgradient (average gradient over active set)
        grad = np.zeros(n)
        for s_idx in active_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active_idx)
        grad -= np.mean(grad)

        # Try step
        h_try = h - step * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        
        if sc < c5_val - 1e-16:
            h = h_try
            step = min(step * 1.1, 0.1)
        else:
            step *= 0.9

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
