        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_trial = c5_val
        k_star = int(np.argmax(corr))

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-13:
                    continue
                di = np.zeros(n)
                di[i] = -1.0
                di[j] = 1.0
                g = corr + di * 0.0
                for _ in range(10):
                    mid_d = 0.5 * (d_lo + d_hi)
                    vals = corr + mid_d * (g - g + np.zeros_like(corr))
                    vals2 = corr.copy()
                    vals2 += mid_d * (np.zeros_like(corr))
                    actual = corr + mid_d * (di[k_star] if False else 0)
                    pass
                a_d = d_lo
                b_d = d_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    new_corr = np.zeros(2 * n - 1)
                    new_h = h.copy()
                    new_h[i] -= mid_d
                    new_h[j] += mid_d
                    new_corr = np.correlate(new_h, 1.0 - new_h, mode='full')
                    if float(np.max(new_corr)) > corr[k_star] + 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                new_h = h.copy()
                new_h[i] -= delta_star
                new_h[j] += delta_star
                trial_c5 = compute_c5(new_h)
                if trial_c5 < best_trial - 1e-15:
                    best_trial = trial_c5
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def smooth_obj(x, alpha, target, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)

def _g_n_global():
    return _g_state['n']

_g_state = {'n': 0, 'target': 0.0}

def _cons_fun(x):
    return np.sum(x) - _g_state['target']

def _cons_jac(x):
    return np.ones(_g_state['n'])

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    _g_state['n'] = n
    _g_state['target'] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, target, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
