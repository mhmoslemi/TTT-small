def smooth_max_gradient(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    corr_shifted = alpha * (corr - np.max(corr))
    w = np.exp(corr_shifted)
    w /= np.sum(w)
    
    # Gradient: sum_k w_k * grad f_k
    # f_k(h) = sum_i h[i](1-h[i+k])*dx (for appropriate i range)
    # df_k/dh[i] = (1-h[i+k])*dx - h[i-k]*dx (with boundary handling)
    
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    
    for k_idx in range(2*n - 1):
        w_k = w[k_idx]
        if w_k < 1e-10:
            continue
        k = k_idx - mid  # actual shift
        if k >= 0:
            # f_k = sum_{i=0}^{n-1-k} h[i](1-h[i+k])*dx
            # df/dh[i] = (1-h[i+k])*dx for i in [0, n-1-k]
            # df/dh[i] = -h[i-k]*dx for i in [k, n-1]
            if k == 0:
                grad[:n] += w_k * (1 - h) * dx - h * dx  # (1-h)-h = 1-2h
            else:

            if kk == 0:

            else:
                grad[kk:] += w_k * (1 - h[:n-kk]) * dx
                grad[:n-kk] -= w_k * h[kk:] * dx
    
    return grad, w
