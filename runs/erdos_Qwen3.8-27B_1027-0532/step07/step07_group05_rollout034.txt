```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        adj = diff / mask.sum()
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    for alpha in alphas:
        try:
            def obj(x):
                return smooth_obj(x, alpha, n)
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
```

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def cd_optimize(h_init, n, target, time_limit, max_sweeps=20000,
                active_frac=0.9, n_cand_start=10, n_cand_end=40,
                max_no_improve=300, bisection_iters=12, check_top=40):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        # Adaptive active set
        max_c = float(np.max(corr))
        thr = max_c * active_frac
        active_count = int(np.sum(corr >= thr))
        n_active = max(3, min(active_count, 15))
        top_idx = np.argsort(corr)[-n_active:]

        # Compute gradient from active correlations
        grad = np.zeros(n)
        w_sum = float(np.sum(corr[top_idx]))
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            jp = idxs + k
            vp = (jp >= 0) & (jp < n)
            grad[vp] += w * (1.0 - h[jp[vp]])
            jm = idxs - k
            vm = (jm >= 0) & (jm < n)
            grad[vm] -= w * h[jm[vm]]
        grad -= np.mean(grad)

        # Candidate selection
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        # Top correlation indices for fast evaluation
        top_corr_idx = np.argsort(corr)[-check_top:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        max_pairs = 60

        # Generate pairs
        if n_h * n_l <= max_pairs:
            pairs = []
            for ii in range(n_h):
                for jj in range(n_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pairs.append((iv, jv))
        else:
            pairs = []
            n_strat = min(max_pairs - 10, max_pairs)
            for frac in np.linspace(0.05, 0.95, n_strat):
                hi = int(frac * (n_h - 1))
                lo = int(frac * (n_l - 1))
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pairs.append((iv, jv))
            # Add random pairs
            for _ in range(min(10, max_pairs - len(pairs))):
                ii = int(np.random.randint(n_h))
                jj = int(np.random.randint(n_l))
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pairs.append((iv, jv))

        for i, j in pairs:
            if i == j:
                continue
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max < 1e-12:
                continue

            # Compute gradient change for top correlations
            g_top = np.empty(check_top)
            for ci in range(check_top):
                k_idx = int(top_corr_idx[ci])
                k_s = k_idx - mid
                dc_j = 0.0
                if 0 <= j + k_s < n:
                    dc_j += 1.0 - h[j + k_s]
                if 0 <= j - k_s < n:
                    dc_j -= h[j - k_s]
                dc_i = 0.0
                if 0 <= i + k_s < n:
                    dc_i += 1.0 - h[i + k_s]
                if 0 <= i - k_s < n:
                    dc_i -= h[i - k_s]
                g_top[ci] = dc_j - dc_i

            # Bisection
            a_d = 0.0
            b_d = delta_max
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr[top_corr_idx] + mid_d * g_top
                f_mid = np.max(vals)
                active_b = vals >= f_mid - 1e-14
                f_prime = np.max(g_top[active_b]) if active_b.any() else 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = float(np.max(corr[top_corr_idx] + delta_star * g_top)) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def subgradient_explore(h_init, n, target, time_limit, max_iters=100000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    eta0 = 0.5
    for t in range(1, max_iters + 1):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        max_c = float(np.max(corr))
        thr = max_c * 0.8
        active = np.where(corr >= thr)[0]
        n_act = len(active)
        
        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            jp = idxs + k
            vp = (jp >= 0) & (jp < n)
            grad[vp] += (1.0 - h[jp[vp]])
            jm = idxs - k
            vm = (jm >= 0) & (jm < n)
            grad[vm] -= h[jm[vm]]
        
        grad *= dx / n_act
        grad -= np.mean(grad)
        
        eta = eta0 / np.sqrt(float(t))
        h = project_feasible(h - eta * grad, target)

    return best_h, best_c5
