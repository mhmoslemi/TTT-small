    n = len(h)
    mid = n - 1
    log_w = alpha * (corr - np.max(corr))
    log_w = np.clip(log_w, -600.0, 0.0)
    max_lw = np.max(log_w)
    weights = np.exp(log_w - max_lw)
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        weights = np.zeros(2 * n - 1)
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= w_sum
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-14:
            continue
        k = s - mid
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            ka = -k
            grad[ka:] += w * (1.0 - h[:n - ka])
            grad[:n - ka] -= w * h[ka:]
    val = np.max(corr) + (max_lw + np.log(max(w_sum, 1e-300))) / alpha
    return val, grad

def momentum_sgd(h_start, n, target, time_limit, alpha_init=200.0,
                 lr_init=0.004, momentum=0.93, plateau_every=3000):
    start = time.time()
    h = h_start.copy().astype(float)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    alpha = alpha_init
    no_improve = 0
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        _, grad = fast_smooth_max_grad(h, alpha, corr)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = lr / (1.0 + 0.05 * gnorm)
        v = momentum * v - step * grad
        h_new = project_feasible(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve == 200:
            lr *= 0.7
            momentum = max(0.5, momentum - 0.03)
            v *= 0.5
        elif no_improve == 500:
            lr *= 0.5
            v = np.zeros(n)
        elif no_improve == 2000:
            break
        if it > 0 and it % plateau_every == 0:
            h_bin = (h > 0.5).astype(float)
            h_bin = project_feasible(h_bin, target)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                h = h_bin.copy()
                v = np.zeros(n)
                no_improve = 0
                alpha = max(50.0, alpha * 0.5)
    return best_h, best_c5

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=50, n_cand_end=200, max_no_improve=500,
            n_binsearch=22, stochastic_frac=0.08, seed=123):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        n_pairs = n_h * n_l
        n_random = max(1, int(stochastic_frac * n_pairs))
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if n_random > 0:
            for _ in range(n_random):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        elif consecutive_no_improve > 100:
            if consecutive_no_improve % 200 == 0:
                n_pert = max(5, n // 20)
                pert_idxs = rng.choice(n, size=n_pert, replace=False)
                h[pert_idxs] += rng.randn(n_pert) * 0.01
                h = project_feasible(h, target)
                change = compute_change_matrix(h, n)
    return best_h, best_c5

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        def obj(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
        def cons_fun(x):
            return np.sum(x) - _g_target
        def cons_jac(x):
            return np.ones(_g_n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def binary_swap_annealing(h_start, n, target, time_limit, T_init=0.003,
                          T_final=0.00005, max_sweeps=200, seed=42):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = (h_start > 0.5).astype(float)
    h = project_feasible(h, target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    change = compute_change_matrix(h, n)
    shifts = np.arange(-(n - 1), n)
    ones_list = np.where(h > 0.5)[0]
    zeros_list = np.where(h <= 0.5)[0]
    T = T_init
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        progress = float(sweep) / max_sweeps
        T = T_init * (T_final / T_init) ** progress
        n_one = len(ones_list)
        n_zero = len(zeros_list)
        order = rng.permutation(n_one)
        for oi in order:
            i = int(ones_list[oi])
            oz = int(rng.randint(n_zero))
            j = int(zeros_list[oz])
            g = change[j] - change[i]
            new_corr = corr + g
            new_max = float(np.max(new_corr)) * dx
            delta_e = new_max - best_c5
            if delta_e < -1e-14:
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                ones_list = np.delete(ones_list, oi)
                zeros_list = np.delete(zeros_list, oz)
                ones_list = np.append(ones_list, j)
                zeros_list = np.append(zeros_list, i)
                best_c5 = new_max
                best_h = h.copy()
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
            elif T > 1e-8 and rng.random() < np.exp(-delta_e / max(T, 1e-8)):
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                ones_list = np.delete(ones_list, oi)
                zeros_list = np.delete(zeros_list, oz)
                ones_list = np.append(ones_list, j)
                zeros_list = np.append(zeros_list, i)
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    if tleft() > 100:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                        [100, 500, 2000, 10000], 1000, 1e-15)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol
    if tleft() > 250:
        h_sg, c5_sg = momentum_sgd(best_h, n0, target0,
                                    time_limit=min(200, tleft() * 0.5),
                                    alpha_init=300.0, lr_init=0.003,
                                    momentum=0.92, plateau_every=5000)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg
    if tleft() > 150:
        h_bin, c5_bin = binary_swap_annealing(best_h, n0, target0,
                                              time_limit=min(120, tleft() * 0.5),
                                              T_init=0.004, T_final=0.00003,
                                              max_sweeps=150, seed=seed)
        if c5_bin < best_score:
            best_score = c5_bin
            best_h = project_feasible(h_bin, target0)
    if tleft() > 250:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                               time_limit=min(250, tleft() * 0.6),
                               max_sweeps=60000,
                               n_cand_start=50, n_cand_end=200,
                               max_no_improve=600,
                               n_binsearch=22,
                               stochastic_frac=0.08, seed=42)
        if c5_cd < best_score:
            best_score = c5_cd

    if tleft() > 100:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                        [5000, 20000, 100000], 2000, 1e-15)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol
    if tleft() > 200:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                               time_limit=min(200, tleft() * 0.6),
                               max_sweeps=50000,
                               n_cand_start=80, n_cand_end=200,
                               max_no_improve=500,
                               n_binsearch=22,
                               stochastic_frac=0.12, seed=137)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
    if tleft() > 120:
        h_sg, c5_sg = momentum_sgd(best_h, n0, target0,
                                    time_limit=min(100, tleft() * 0.5),
                                    alpha_init=500.0, lr_init=0.002,
                                    momentum=0.9, plateau_every=5000)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg
    if tleft() > 80:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                        [100000, 500000], 3000, 1e-15)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol
    if tleft() > 200:
        n1 = 600
        target1 = n1 / 2.0
        x_old = np.linspace(0.0, 1.0, n0)
        x_new = np.linspace(0.0, 1.0, n1)
        h1 = np.interp(x_new, x_old, best_h)
        h1 = project_feasible(h1, target1)
        if tleft() > 150:
            h_cd1, c5_cd1 = pair_cd(h1, n1, target1,
                                     time_limit=min(150, tleft() * 0.5),
                                     max_sweeps=30000,
                                     n_cand_start=40, n_cand_end=130,
                                     max_no_improve=400,
                                     n_binsearch=20,
                                     stochastic_frac=0.05, seed=251)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                n0 = n1
                target0 = target1
        if tleft() > 80:
            h_pol1, sc_pol1 = slsqp_optimize(best_h, n0, target0,
                                              [5000, 20000, 100000], 2000, 1e-15)

                best_score = sc_pol1

        h_cd_final, c5_final = pair_cd(best_h, n0, target0,

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
