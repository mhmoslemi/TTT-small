The change in correlation at shift k is:
Δcorr_k = delta * [∂corr_k/∂h[j] - ∂corr_k/∂h[i]]

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        ws = window_sizes[step % len(window_sizes)]
        n_pos = n - ws + 1
        if n_pos <= 1:
            continue

        src_pos = rng.randint(0, n_pos)
        dst_pos = rng.randint(0, n_pos)
        if src_pos == dst_pos:
            continue

        src_block = h[src_pos:src_pos + ws].copy()
        dst_block = h[dst_pos:dst_pos + ws].copy()

        delta_max = min(src_block.max(), 1.0 - dst_block.min())
        delta_min = max(-src_block.min(), dst_block.max() - 1.0)
        if delta_max - delta_min < 1e-12:
            continue

        g = np.zeros(n_corr)
        for t in range(ws):
            g += change[dst_pos + t, :] - change[src_pos + t, :]

        a_d = delta_min
        b_d = delta_max
        for _ in range(12):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr + mid_d * g
            f_mid = np.max(vals)
            active_mask = vals >= f_mid - 1e-14
            if active_mask.any():
                f_prime = np.max(g[active_mask])
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                b_d = mid_d
            else:
                a_d = mid_d

        delta_star = 0.5 * (a_d + b_d)
        if abs(delta_star) < 1e-12:
            continue

        approx_max = np.max(corr + delta_star * g)
        if approx_max < c5_val - 1e-14:
            h[src_pos:src_pos + ws] -= delta_star
            h[dst_pos:dst_pos + ws] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for t in range(ws):
                change[src_pos + t, :] = 0.0
                pos_m = (src_pos + t + shifts >= 0) & (src_pos + t + shifts < n)
                neg_m = (src_pos + t - shifts >= 0) & (src_pos + t - shifts < n)
                change[src_pos + t, pos_m] += (1.0 - h[src_pos + t + shifts[pos_m]])
                change[src_pos + t, neg_m] -= h[src_pos + t - shifts[neg_m]]
                change[dst_pos + t, :] = 0.0
                pos_m = (dst_pos + t + shifts >= 0) & (dst_pos + t + shifts < n)
                neg_m = (dst_pos + t - shifts >= 0) & (dst_pos + t - shifts < n)
                change[dst_pos + t, pos_m] += (1.0 - h[dst_pos + t + shifts[pos_m]])
                change[dst_pos + t, neg_m] -= h[dst_pos + t - shifts[neg_m]]

    return best_h, best_c5


def projected_gradient(h_init, n, target, time_limit, max_iter=50000,
                       lr_start=0.005, decay=0.999, smooth_alpha=30.0,
                       n_active=12):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    lr = lr_start

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        grad = smooth_gradient(h, smooth_alpha, n_active=n_active)
        grad -= np.mean(grad)

        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val:
            h = h_new
        else:
            lr *= 0.5

        lr = max(lr, lr_start * (decay ** it))

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [50, 300, 1500, 8000],
                                400, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n1 = 800
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)

    if tleft() > 100:
        h1, sc1 = slsqp_round(h1, n1, target1,
                              [200, 1000, 5000],
                              500, 1e-15)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    if tleft() > 300:
        h_cd, c5_cd = fast_pair_cd(h1, n1, target1,
                                   time_limit=min(200, tleft() * 0.6),
                                   max_sweeps=40000,
                                   n_cand_start=15, n_cand_end=50,
                                   max_no_improve=400,
                                   smooth_alpha=20.0, n_active=10)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_pg, c5_pg = projected_gradient(best_h, n1, target1,
                                         time_limit=min(60, tleft() * 0.7),
                                         max_iter=30000, lr_start=0.004,
                                         decay=0.9995, smooth_alpha=30.0,
                                         n_active=12)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    if tleft() > 100:
        h1b, sc1b = slsqp_round(best_h, n1, target1,
                                [1000, 5000, 20000],
                                600, 1e-15)
        if sc1b < best_score:
            best_score = sc1b
            best_h = h1b

    n2 = 1200
    target2 = n2 / 2.0
    h2 = interpolate_to(best_h, n2, target2)

    if tleft() > 200:
        h2, sc2 = slsqp_round(h2, n2, target2,
                              [500, 2000, 10000],
                              400, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    if tleft() > 300:
        h_cd2, c5_cd2 = fast_pair_cd(h2, n2, target2,
                                     time_limit=min(250, tleft() * 0.6),
                                     max_sweeps=30000,
                                     n_cand_start=12, n_cand_end=40,
                                     max_no_improve=350,
                                     smooth_alpha=25.0, n_active=10)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 120:
        h_w, c5_w = window_cd(best_h, n2, target2,
                              time_limit=min(80, tleft() * 0.6),
                              max_steps=15000, window_sizes=(2, 3, 4, 5))
        if c5_w < best_score:
            best_score = c5_w
            best_h = h_w

    if tleft() > 100:
        h2b, sc2b = slsqp_round(best_h, n2, target2,
                                [2000, 10000, 50000],
                                500, 1e-15)
        if sc2b < best_score:
            best_score = sc2b
            best_h = h2b

    if tleft() > 150:
        h_cd3, c5_cd3 = fast_pair_cd(best_h, n2, target2,
                                     time_limit=min(100, tleft() * 0.6),
                                     max_sweeps=20000,
                                     n_cand_start=20, n_cand_end=55,
                                     max_no_improve=300,
                                     smooth_alpha=40.0, n_active=8)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 100:
        h_pg2, c5_pg2 = projected_gradient(best_h, n2, target2,
                                           time_limit=min(60, tleft() * 0.7),
                                           max_iter=20000, lr_start=0.002,
                                           decay=0.9998, smooth_alpha=50.0,
                                           n_active=8)
        if c5_pg2 < best_score:
            best_score = c5_pg2
            best_h = h_pg2

    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, n2 / 2.0)
            h_ref, sc_ref = slsqp_round(h_pert, n2, n2 / 2.0,
                                        [500, 3000, 15000],
                                        300, 1e-15)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 60:
        h_final_cd, c5_fc = fast_pair_cd(best_h, n2, n2 / 2.0,
                                         time_limit=min(40, tleft() * 0.7),
                                         max_sweeps=10000,
                                         n_cand_start=25, n_cand_end=60,
                                         max_no_improve=200,
                                         smooth_alpha=60.0, n_active=6)
        if c5_fc < best_score:
            best_score = c5_fc
            best_h = h_final_cd

    if tleft() > 30:
        h_fin, sc_fin = slsqp_round(best_h, n2, n2 / 2.0,
                                    [10000, 50000, 100000],
                                    400, 1e-15)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
