import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
            else:
                h_cur = project_feasible(h_cur, target)
        except Exception:
            pass
    return h_cur, cur_score


def compute_subgrad(h, n, k_star):
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    grad -= np.mean(grad)
    return grad


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break
                elif approx_max - c5_val < T * 5:
                    if rng.random() < np.exp(-(approx_max - c5_val) / T):
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
                        break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def polyak_subgradient(h_init, n, target, time_limit, n_iter=5000,
                       step_scale=0.002, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    step = step_scale
    no_improve = 0

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            step = min(step * 1.1, 0.02)
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                step *= 0.9

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        idxs = np.arange(n)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        velocity = momentum * velocity - step * grad
        h_try = project_feasible(h + velocity, target)
        h = h_try

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_scale_optimize(h, n, target, time_limit, coarse_factor=4):
    start = time.time()
    n_coarse = max(100, n // coarse_factor)
    target_coarse = n_coarse / 2.0

    h_coarse = interpolate_to(h, n_coarse, target_coarse)

    t1 = min(60, (time_limit * 0.3))
    if time.time() - start < time_limit - t1:
        h_coarse, sc_coarse = slsqp_round(h_coarse, n_coarse, target_coarse,
                                          [100, 500, 2000], 400, 1e-14)

    t2 = min(60, (time_limit * 0.3))
    if time.time() - start < time_limit - t2:
        h_coarse, sc_coarse2 = focused_pair_cd(h_coarse, n_coarse, target_coarse,
                                               time_limit=t2, max_sweeps=10000,
                                               n_low=20, n_high=20,
                                               max_no_improve=200)

    h_up = interpolate_to(h_coarse, n, target)

    t3 = time_limit - (time.time() - start)
    if t3 > 30:
        h_up, sc_up = slsqp_round(h_up, n, target, [500, 2000], 300, 1e-14)

    return h_up, compute_c5(h_up)


def structural_perturb(h, n, target, rng):
    h_new = h.copy()
    n_perturb = rng.randint(3, 10)
    indices = rng.choice(n, n_perturb, replace=False)
    for idx in indices:
        width = rng.randint(5, 20)
        lo = max(0, idx - width)
        hi = min(n, idx + width + 1)
        scale = rng.uniform(0.3, 0.7)
        h_new[lo:hi] = 1.0 - h_new[lo:hi] * scale
    return project_feasible(h_new, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    n_cur = n0
    target_cur = target0

    # Phase 1: SLSQP with moderate alphas
    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, n_cur, target_cur,
                                [200, 1000, 5000, 20000],
                                500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Focused pair CD (single worst shift)
    if tleft() > 200:
        t_cd = min(200, tleft() * 0.5)
        h_cd, c5_cd = focused_pair_cd(best_h, n_cur, target_cur,
                                      time_limit=t_cd, max_sweeps=40000,
                                      n_low=50, n_high=50,
                                      max_no_improve=600)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP polish after CD
    if tleft() > 80:
        h_pol2, sc2 = slsqp_round(best_h, n_cur, target_cur,
                                  [10000, 50000, 100000],
                                  500, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 4: Multi-scale to escape local minima
    if tleft() > 250:
        h_ms, sc_ms = multi_scale_optimize(best_h, n_cur, target_cur,
                                           time_limit=min(220, tleft() * 0.5))
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms
            # Polish after multi-scale
            if tleft() > 60:
                h_pol3, sc3 = slsqp_round(best_h, n_cur, target_cur,
                                          [5000, 20000], 400, 1e-15)
                if sc3 < best_score:
                    best_score = sc3
                    best_h = h_pol3

    # Phase 5: Annealed pair CD
    if tleft() > 200:
        t_an = min(180, tleft() * 0.5)
        h_an, c5_an = annealed_pair_cd(best_h, n_cur, target_cur,
                                       time_limit=t_an, max_sweeps=20000,
                                       n_cand=45, max_no_improve=500,
                                       weight_beta=4.0, T0=0.002, T_min=1e-6)
        if c5_an < best_score:
            best_score = c5_an
            best_h = h_an

    # Phase 6: Polyak subgradient (focus on worst shift)
    if tleft() > 100:
        h_sg, c5_sg = polyak_subgradient(best_h, n_cur, target_cur,
                                         time_limit=min(80, tleft() * 0.5),
                                         n_iter=8000, step_scale=0.003,
                                         momentum=0.85)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 7: Focused pair CD again
    if tleft() > 120:
        t_cd2 = min(100, tleft() * 0.5)
        h_cd2, c5_cd2 = focused_pair_cd(best_h, n_cur, target_cur,
                                        time_limit=t_cd2, max_sweeps=25000,
                                        n_low=60, n_high=60,
                                        max_no_improve=400)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 8: Structural perturbation + re-optimize
    if tleft() > 200:
        for n_struct in range(3):
            if tleft() < 80:
                break
            h_pert = structural_perturb(best_h, n_cur, target_cur, rng)
            if tleft() > 60:
                h_sp, sc_sp = slsqp_round(h_pert, n_cur, target_cur,
                                          [500, 2000, 10000], 400, 1e-15)
            else:
                h_sp, sc_sp = h_pert, compute_c5(h_pert)
            if sc_sp < best_score:
                best_score = sc_sp
                best_h = h_sp
            if tleft() > 50:
                h_spcd, c5_spcd = focused_pair_cd(best_h, n_cur, target_cur,
                                                  time_limit=min(40, tleft() * 0.4),
                                                  max_sweeps=10000,
                                                  n_low=40, n_high=40,
                                                  max_no_improve=200)
                if c5_spcd < best_score:
                    best_score = c5_spcd
                    best_h = h_spcd

    # Phase 9: Another multi-scale cycle
    if tleft() > 200:
        h_ms2, sc_ms2 = multi_scale_optimize(best_h, n_cur, target_cur,
                                             time_limit=min(180, tleft() * 0.5),
                                             coarse_factor=3)
        if sc_ms2 < best_score:
            best_score = sc_ms2
            best_h = h_ms2
            if tleft() > 50:
                h_pol4, sc4 = slsqp_round(best_h, n_cur, target_cur,
                                          [5000, 20000, 50000], 400, 1e-15)
                if sc4 < best_score:
                    best_score = sc4
                    best_h = h_pol4

    # Phase 10: Final focused CD
    if tleft() > 80:
        h_cd3, c5_cd3 = focused_pair_cd(best_h, n_cur, target_cur,
                                        time_limit=min(60, tleft() * 0.6),
                                        max_sweeps=20000,
                                        n_low=70, n_high=70,
                                        max_no_improve=300)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 11: Final SLSQP with very high alpha
    if tleft() > 40:
        h_fin, sc_fin = slsqp_round(best_h, n_cur, target_cur,
                                    [50000, 100000, 200000],
                                    500, 1e-15)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
