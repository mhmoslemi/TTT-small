    """Find exact minimum of max_k (corr[k] + delta * g[k]) for delta in [lo, hi]."""
    # Get the top |g| correlations (most likely to be active)
    # Actually, we want correlations that could be at the max
    # At delta=0, the top correlations are most likely active
    # At the delta we're looking for, the active set might be different
    # But for a small step, the top correlations at delta=0 are a good approximation
    
    top_idx = np.argsort(corr)[-top_m:]
    c_top = corr[top_idx]
    g_top = g[top_idx]
    
    # Candidate breakpoints
    candidates = [lo, hi]
    for i in range(len(top_idx)):
        for j in range(len(top_idx)):
            if i != j and abs(g_top[j] - g_top[i]) > 1e-14:
                delta = (c_top[i] - c_top[j]) / (g_top[j] - g_top[i])
                if lo <= delta <= hi:
                    candidates.append(delta)
    
    # Evaluate at each candidate
    best_delta = None
    best_val = np.inf
    for d in candidates:
        vals = corr + d * g  # Full evaluation
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_delta = d
    
    return best_delta, best_val

                if g[k_star_idx] > 0.2 * np.max(np.abs(g)):
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-13
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def active_set_gradient_cd(h_init, n, target, time_limit, tol_frac=0.02, step_scale=0.01,
                           max_steps=200000, stagnation_limit=300):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step_size = step_scale
    stagnant = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        c5_val = c_max * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stagnant = 0
            step_size = min(step_size * 1.05, step_scale)
        else:
            stagnant += 1
            step_size *= 0.97
            if stagnant > stagnation_limit:
                break

        threshold = c_max - tol_frac * max(c_max, 0.001)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        n_act = len(active)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = 1.0 / n_act
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        grad_normalized = grad / grad_norm
        h_new = h - step_size * grad_normalized
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: Quick SLSQP polish at n=400
    if tleft() > 60:
        deadline = time.time() + min(35, tleft() * 0.3)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [200, 1000, 5000, 20000],
                                400, 1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Interpolate to n=800, long CD
    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            current_n = n1
            current_target = target1

        if tleft() > 200:
            h_cd, c5_cd = pair_cd_improved(h1, n1, target1,
                                           time_limit=min(220, tleft() * 0.55),
                                           max_sweeps=40000,
                                           n_cand_start=20, n_cand_end=75,
                                           max_no_improve=500,
                                           n_active_shifts=3,
                                           reheat_interval=100, reheat_scale=0.035,
                                           use_rotating_active=True, phase_offset=0)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                current_n = n1
                current_target = target1

    # Phase 3: SLSQP polish at n=800
    if tleft() > 80:
        deadline = time.time() + min(55, tleft() * 0.4)
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [2000, 10000, 50000],
                                1500, 1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Second CD with different parameters
    if tleft() > 200:
        h_cd2, c5_cd2 = pair_cd_improved(best_h, current_n, current_target,
                                         time_limit=min(180, tleft() * 0.5),
                                         max_sweeps=35000,
                                         n_cand_start=30, n_cand_end=90,
                                         max_no_improve=600,
                                         n_active_shifts=4,
                                         reheat_interval=80, reheat_scale=0.025,
                                         use_rotating_active=True, phase_offset=50)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 5: Active set gradient CD
    if tleft() > 80:
        h_asg, c5_asg = active_set_gradient_cd(best_h, current_n, current_target,
                                              time_limit=min(60, tleft() * 0.5),
                                              tol_frac=0.015, step_scale=0.008,
                                              max_steps=50000, stagnation_limit=200)
        if c5_asg < best_score:
            best_score = c5_asg
            best_h = h_asg

    # Phase 6: SLSQP with gradient
    if tleft() > 60:
        deadline = time.time() + min(45, tleft() * 0.45)
        h_pol2, sc2 = slsqp_round(best_h, current_n, current_target,
                                  [5000, 20000, 80000],
                                  1000, 1e-15, deadline=deadline, use_grad=True)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 7: Multi-start perturbation
    if tleft() > 150:
        noise_scales = [0.02, 0.05, 0.08, 0.12, 0.18]
        for attempt, ns in enumerate(noise_scales):
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(current_n) * ns
            h_pert = project_feasible(h_pert, current_target)
            deadline_ps = time.time() + min(25, tleft() * 0.25)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [500, 2000, 8000, 30000],
                                        600, 1e-15, deadline=deadline_ps)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 8: Final CD
    if tleft() > 70:
        h_cd_f, c5_f = pair_cd_improved(best_h, current_n, current_target,
                                        time_limit=min(55, tleft() * 0.55),
                                        max_sweeps=20000,
                                        n_cand_start=40, n_cand_end=100,
                                        max_no_improve=400,
                                        n_active_shifts=5,
                                        reheat_interval=60, reheat_scale=0.02,
                                        use_rotating_active=True, phase_offset=100)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    # Phase 9: Final SLSQP (highest alpha)
    if tleft() > 40:
        deadline_final = time.time() + min(30, tleft() * 0.6)
        h_final, sc_final = slsqp_round(best_h, current_n, current_target,
                                        [20000, 80000, 200000],
                                        1200, 1e-15, deadline=deadline_final)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 10: Try brief optimization at n=1000 if time allows
    if tleft() > 120:
        n3 = 1000
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        sc3 = compute_c5(h3)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3
            current_n = n3
            current_target = target3
        if tleft() > 60:
            deadline_h3 = time.time() + min(50, tleft() * 0.5)
            h_pol3, sc_pol3 = slsqp_round(best_h, current_n, current_target,
                                          [5000, 20000, 60000],
                                          800, 1e-15, deadline=deadline_h3)
            if sc_pol3 < best_score:
                best_score = sc_pol3
                best_h = h_pol3
            if tleft() > 30:
                h_cd3, c5_cd3 = pair_cd_improved(best_h, current_n, current_target,
                                                 time_limit=min(30, tleft() * 0.4),
                                                 max_sweeps=10000,
                                                 n_cand_start=30, n_cand_end=70,
                                                 max_no_improve=300,
                                                 n_active_shifts=4,
                                                 reheat_interval=50, reheat_scale=0.02,
                                                 use_rotating_active=True, phase_offset=200)
                if c5_cd3 < best_score:
                    best_score = c5_cd3
                    best_h = h_cd3

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
