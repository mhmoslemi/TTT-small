def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

    _g_target = target

            sc = compute_c5(h_new)

                h_cur = h_new

            pass

import time

_g_target = 0.0

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):

    n = len(h)
    for _ in range(100):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target


def slsqp_con_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h0, n, target, alphas, max_iters):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

    h_cur = h0.copy()

            res = minimize(slsqp_obj_func, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,

def build_phi(h, n):
    n_corr = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    phi = np.zeros((n, n_corr))
    for p in range(n):
        pos = (lags + p >= 0) & (lags + p < n)
        neg = (p - lags >= 0) & (p - lags < n)
        phi[p, pos] += (1.0 - h[lags[pos] + p])
        phi[p, neg] -= h[p - lags[neg]]
    return phi


def find_optimal_delta(corr, g, delta_max):
    if delta_max < 1e-14:
        return 0.0
    max_val = corr.max()
    active = np.where(corr >= max_val - 1e-12)[0]
    slope0 = g[active].max()
    if slope0 > 1e-12:
        return 0.0
    if slope0 > -1e-12:
        vals_end = corr + delta_max * g
        if vals_end.max() < max_val - 1e-12:
            return delta_max
        return 0.0
    lo = 0.0
    hi = delta_max
    for _ in range(25):
        mid = 0.5 * (lo + hi)
        vals = corr + mid * g
        max_v = vals.max()
        act = np.where(vals >= max_v - 1e-12)[0]
        slope = g[act].max()
        if slope > 1e-12:
            hi = mid
        else:
            lo = mid
    best_d = 0.0
    best_f = max_val
    for d in [lo, hi, 0.5 * (lo + hi), 0.0, delta_max]:
        if 0.0 <= d <= delta_max:
            f_d = (corr + d * g).max()
            if f_d < best_f:
                best_f = f_d
                best_d = d
    return best_d


def pair_cd_v2(h0, n, target, time_limit, start_time,
               max_sweeps=50000, n_cand=25, n_shifts=3,
               max_no_improve=500, temp_init=0.005, temp_decay=0.999,
               swap_every=200, swap_prob=0.3):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h0.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    temp = temp_init

    corr = np.correlate(h, 1.0 - h, mode='full')
    phi = build_phi(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_no_improve:
            break

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idx = np.argsort(corr)[-n_shifts:]
        sw = corr[top_idx].copy()
        sw_sum = sw.sum()
        if sw_sum < 1e-15:
            sw = np.ones(n_shifts) / n_shifts
        else:
            sw = sw / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            k = int(top_idx[si]) - mid
            w = sw[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            hi_val = h[i]
            if hi_val < 1e-12:
                continue
            phi_i = phi[i]
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_max = min(hi_val, 1.0 - h[j])
                if delta_max < 1e-12:
                    continue
                g = phi[j] - phi_i
                delta_star = find_optimal_delta(corr, g, delta_max)
                if delta_star < 1e-12:
                    continue
                new_max = (corr + delta_star * g).max()
                new_c5 = float(new_max) * dx
                if new_c5 < best_trial - 1e-14:
                    best_trial = new_c5
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                elif temp > 1e-10 and new_c5 < c5_val + temp * 50.0:
                    if np.random.random() < np.exp(-(new_c5 - c5_val) * 2000.0 / temp):
                        best_trial = new_c5
                        best_i = i
                        best_j = j
                        best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            phi = build_phi(h, n)
            temp *= temp_decay
        else:
            temp *= 0.98
            if swap_every > 0 and sweep % swap_every == 0 and np.random.random() < swap_prob:
                p1 = int(np.random.randint(n))
                p2 = int(np.random.randint(n))
                if p1 != p2:

                    corr = np.correlate(h, 1.0 - h, mode='full')
                    phi = build_phi(h, n)

    return best_h, best_c5
