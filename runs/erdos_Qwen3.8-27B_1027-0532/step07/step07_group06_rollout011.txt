```python
    # np.correlate(a, b, 'full')[i] = sum_n a[n] * b[n + i - (len(b)-1)]

    # np.correlate(h, v, 'full')[i] = sum_m h[m] * v[m + i - (n-1)]

    # sum_j v[j] * h[j-k]: let m = j-k, j = m+k
    # = sum_m v[m+k] * h[m]
    # = sum_m h[m] * v[m+k]
    
    # np.correlate(h, v, 'full')[i] = sum_m h[m] * v[m + i - (len(v)-1)]
    # len(v) = n, so = sum_m h[m] * v[m + i - n + 1]
    # Want: sum_m h[m] * v[m+k]
    # So m + i - n + 1 = m + k → i = k + n - 1
    
    # Hmm, both give the same index? That seems wrong. Let me recheck.
    
    # Actually wait. Let me re-examine.
    # np.correlate(a, b, 'full') has length len(a) + len(b) - 1
    # The i-th element (0-indexed) is: sum_n a[n] * b[n + i - (len(b)-1)]

    # Matching: m + i - n + 1 = m - k → i = k + n - 1

    # g_k = correlate(1-h, v, 'full')[k+n-1] - correlate(h, v, 'full')[k+n-1]
    #     = correlate((1-h)-h, v, 'full')[k+n-1]
    #     = correlate(1-2h, v, 'full')[k+n-1]

    # = correlate(1-h, v, 'full')[k+n-1]  (based on above)

    # = correlate(h, v, 'full')[k+n-1]  (based on above)

    #        = correlate(1-2h, v)[k+n-1]

    # n=3, h=[0.5, 0.5, 0.5], v=[1, -1, 0]

    # For k=0: g_0 = v[0]*(1-h[0]) - v[0]*h[0] + v[1]*(1-h[1]) - v[1]*h[1] + v[2]*(1-h[2]) - v[2]*h[2]

    # 1-2h = [-1, 1, 0]

    # = correlate([-1,1,0], [1,-1,0], 'full')

    # I'm getting confused with the numpy definition. Let me just use the formula directly in code.
    
    # Actually, let me just verify numerically in the code and use the formula if it checks out.
    # For now, let me use a direct O(n^2) computation but vectorized over j.
    
    shifts = np.arange(-(n-1), n)  # all shifts k
    # For each j, the contribution to g_k is v[j] * [(1-h[j+k]) if 0<=j+k<n else 0 - h[j-k] if 0<=j-k<n else 0]
    
    # Vectorized: create a matrix M[j, k_idx] = d(c_k)/d(h_j)
    # Then g = v @ M  (v is 1D of length n, M is n × (2n-1))
    
    # M[j, k_idx] where k = k_idx - (n-1):
    # = (1-h[j+k]) if 0<=j+k<n else 0, minus h[j-k] if 0<=j-k<n else 0
    
    # This can be computed as:
    # term1[j, k_idx] = (1-h[j+k]) if 0<=j+k<n
    # term2[j, k_idx] = h[j-k] if 0<=j-k<n
    
    # For term1: j+k = j + k_idx - (n-1), need 0 <= j+k_idx-(n-1) < n
    #   → n-1-j <= k_idx <= n-1+j
    # For term2: j-k = j - (k_idx-(n-1)) = j-n+1-k_idx, need 0 <= j-n+1-k_idx < n
    #   → j-n+1-n < k_idx <= j-n+1  →  -n+1+j < k_idx <= j+1-n... 
    
    # This is getting messy. Let me just do it with a loop over j but vectorized over k.
    
    g = np.zeros(2*n - 1)
    for j in range(n):
        if v[j] == 0:
            continue
        # Contribution of h_j to all c_k:
        # (1-h[j+k]) for k where 0<=j+k<n → k in [-j, n-1-j]
        # -h[j-k] for k where 0<=j-k<n → k in [j-n+1, j]
        
        k_vals = np.arange(-(n-1), n)
        contrib = np.zeros(2*n-1)
        
        mask1 = (j + k_vals >= 0) & (j + k_vals < n)
        contrib[mask1] = (1.0 - h[j + k_vals[mask1]])
        
        mask2 = (j - k_vals >= 0) & (j - k_vals < n)
        contrib[mask2] -= h[j - k_vals[mask2]]
        
        g += v[j] * contrib
    
    return g
```

This is O(n^2) but with inner loop vectorized. For n=400, the outer loop is 400 iterations, each doing O(n) vectorized work. Total: 400*800 = 320,000 operations, which is fast.

But wait, for the block CD, at each sweep we need:
1. np.correlate(h, 1-h, 'full'): O(n log n) via FFT or O(n^2) direct
2. compute_g: O(n^2) as above
3. Bisection: 20 iterations, each O(n) to compute max

Total per sweep: O(n^2). For n=400, that's 160,000 operations. At ~1 ns per op, that's 0.16 ms per sweep. We can do ~6000 sweeps per second. In 100 seconds, that's 600,000 sweeps. That's plenty.

Actually, np.correlate for n=400 might be slower than O(n^2) in practice due to overhead. Let me check: np.correlate with mode='full' for arrays of length 400 does about 400*800 = 320,000 multiply-adds. In C, that's very fast, maybe 10-20 μs.

OK so the block CD should be fast enough. Let me now design the full algorithm.

**Algorithm:**

1. **Initialization**: Use `initial_h_values` (n=400).

2. **Phase 1 - SLSQP warm-start** (~60s):
   - Run SLSQP with increasing alpha on the smooth max objective.
   - This gives a good local optimum.

3. **Phase 2 - Block coordinate descent** (~400s):
   - Generalized pair CD with block size m (adaptively increasing from 1 to 5).
   - At each sweep: compute correlation, gradient, select block, bisect.
   - Use top-k shifts for gradient computation.

4. **Phase 3 - Resolution upgrade** (~200s):
   - Interpolate to n=600 or n=800.
   - Run block CD at higher resolution.

5. **Phase 4 - Subgradient with momentum** (~100s):
   - Use Nesterov-accelerated subgradient method for structural changes.
   - This can escape local optima that CD gets stuck in.

6. **Phase 5 - Final polish** (~100s):
   - SLSQP with very high alpha.
   - Final block CD with small blocks.

7. **Phase 6 - Random restarts** (remaining time):
   - Perturb the best solution and try to improve.

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_max_obj, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_grad_at_shift(h, k, n):
    """Compute gradient of c_k with respect to h."""
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    grad[v1] += (1.0 - h[idxs[v1] - k])
    grad[v2] -= h[idxs[v2] + k]
    return grad

def compute_g_vector(v, h, n):
    """Compute g_k = v · grad_k for all shifts k, vectorized."""
    shifts = np.arange(-(n-1), n)
    n_corr = 2*n - 1
    g = np.zeros(n_corr)
    for j in range(n):
        vj = v[j]
        if abs(vj) < 1e-15:
            continue
        mask1 = (j + shifts >= 0) & (j + shifts < n)
        mask2 = (j - shifts >= 0) & (j - shifts < n)
        contrib = np.zeros(n_corr)
        if mask1.any():
            contrib[mask1] = (1.0 - h[j + shifts[mask1]])
        if mask2.any():
            contrib[mask2] -= h[j - shifts[mask2]]
        g += vj * contrib
    return g

def block_cd(h_init, n, target, time_limit, start_time,
             max_sweeps=50000, m_start=2, m_end=8,
             max_no_improve=500, n_top_shifts=3):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Compute combined gradient from top shifts
        top_k = n_top_shifts
        top_idxs = np.argsort(corr)[-top_k:]
        weights = corr[top_idxs].astype(float)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            weights = weights / wsum
        
        grad = np.zeros(n)
        for s_idx, w in zip(top_idxs, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Adaptive block size
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        m = int(m_start + (m_end - m_start) * progress)
        m = max(m, 2)
        m = min(m, n // 4)
        
        # Select block: top m positive and bottom m negative gradient
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:m]
        high_set = sorted_idx[-m:]
        
        # Form direction vector
        v = np.zeros(n)
        v[high_set] = 1.0
        v[low_set] = -1.0
        
        # Compute bounds on t
        # h + t*v must stay in [0,1]
        pos_mask = v > 0
        neg_mask = v < 0
        t_max = np.inf
        if pos_mask.any():
            t_max = min(t_max, np.min((1.0 - h[pos_mask]) / v[pos_mask]))
        if neg_mask.any():
            t_max = min(t_max, np.min((0.0 - h[neg_mask]) / v[neg_mask]))
        t_lo = 0.0
        t_hi = max(t_max, 0.0)
        
        if t_hi - t_lo < 1e-14:
            continue
        
        # Compute g = sensitivity of all c_k to the direction v
        g = compute_g_vector(v, h, n)
        
        # Bisect on t to minimize max(corr + t * g)
        for _ in range(25):
            t_mid = 0.5 * (t_lo + t_hi)
            new_corr = corr + t_mid * g
            f_mid = np.max(new_corr)
            active = new_corr >= f_mid - 1e-14
            if active.any():
                f_prime = np.max(g[active])
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                t_hi = t_mid
            else:
                t_lo = t_mid
        
        t_opt = 0.5 * (t_lo + t_hi)
        if abs(t_opt) < 1e-14:
            continue
        
        h_new = h + t_opt * v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        sc_new = compute_c5(h_new)
        if sc_new < c5_val - 1e-14:
            h = h_new
        else:
            # Try a smaller step
            h_new2 = h + 0.5 * t_opt * v
            h_new2 = np.clip(h_new2, 0.0, 1.0)
            h_new2 = project_feasible(h_new2, target)
            sc_new2 = compute_c5(h_new2)
            if sc_new2 < c5_val - 1e-14:
                h = h_new2
    
    return best_h, best_c5

def subgradient_momentum(h_init, n, target, time_limit, start_time,
                         lr_init=0.01, momentum=0.9, max_steps=5000):
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx if (dx := 2.0/n) else 0
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Adaptive learning rate
        lr = lr_init * (1.0 + step) ** (-0.5)
        
        # Subgradient at argmax
        k_star = int(np.argmax(corr)) - mid
        g = compute_grad_at_shift(h, k_star, n)
        g -= np.mean(g)
        
        # Momentum update
        v = momentum * v + lr * g
        h_new = h - v
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < c5_val:
            h = h_new
        else:
            # Rejected step - reduce momentum
            v *= 0.5
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
```
