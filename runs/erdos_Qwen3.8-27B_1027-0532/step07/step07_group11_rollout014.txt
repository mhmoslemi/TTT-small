            temp = max(temp * 0.95, 1e-10)

    return best_h, best_c5


def subgradient_multi(h_init, n, target, time_limit, n_steps=30000, n_weights=5, step_scale=0.004):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        p = np.exp(corr * 80.0)
        p /= p.sum()

        grad = np.zeros(n)
        top_k = np.argsort(p)[-n_weights:]
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.9

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def local_search_slsqp(h_init, n, target, time_limit, n_restarts=8, noise_scales=None):
    start = time.time()
    if noise_scales is None:
        noise_scales = [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3]
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    rng = np.random.RandomState(12345)

    for r in range(n_restarts):
        if time.time() - start > time_limit - 30:
            break
        scale = noise_scales[r % len(noise_scales)]
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pert.copy()
        if time.time() - start > time_limit - 40:
            break
        h_ref, sc2 = slsqp_round(h_pert, n, target, [200, 1000, 5000], 400, 1e-14)
        if sc2 < best_c5:
            best_c5 = sc2
            best_h = h_ref

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [100, 500, 2000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 400:
        h_cd, c5_cd = pair_cd_advanced(best_h, cur_n, cur_target,
                                       time_limit=min(350, tleft() * 0.6),
                                       max_sweeps=80000,
                                       n_cand_start=40, n_cand_end=200,
                                       n_shifts=3, max_no_improve=1000,
                                       anneal_temp=0.03, anneal_decay=0.9995)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 150:
        h_sg, c5_sg = subgradient_multi(best_h, cur_n, cur_target,
                                        time_limit=min(80, tleft() * 0.4),
                                        n_steps=30000, n_weights=5,
                                        step_scale=0.005)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 120:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [3000, 10000], 800, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 350:
        h_cd2, c5_cd2 = pair_cd_advanced(best_h, cur_n, cur_target,
                                         time_limit=min(300, tleft() * 0.6),
                                         max_sweeps=70000,
                                         n_cand_start=60, n_cand_end=250,
                                         n_shifts=4, max_no_improve=1200,
                                         anneal_temp=0.01, anneal_decay=0.9997)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [5000, 15000, 30000], 1200, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            cur_n = n1
            cur_target = target1
        if tleft() > 200:
            h_cd1, c5_cd1 = pair_cd_advanced(h1, n1, target1,
                                             time_limit=min(180, tleft() * 0.55),
                                             max_sweeps=30000,
                                             n_cand_start=30, n_cand_end=100,
                                             n_shifts=3, max_no_improve=500,
                                             anneal_temp=0.02, anneal_decay=0.999)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                cur_n = n1
                cur_target = target1

    if tleft() > 80:
        h_ls, c5_ls = local_search_slsqp(best_h, cur_n, cur_target,
                                         time_limit=min(70, tleft() * 0.7),
                                         n_restarts=5,
                                         noise_scales=[0.02, 0.05, 0.1, 0.2, 0.4])
        if c5_ls < best_score:
            best_score = c5_ls
            best_h = h_ls

    if tleft() > 60:
        h_cd3, c5_cd3 = pair_cd_advanced(best_h, cur_n, cur_target,
                                         time_limit=min(50, tleft() * 0.7),
                                         max_sweeps=25000,
                                         n_cand_start=80, n_cand_end=250,
                                         n_shifts=5, max_no_improve=300,
                                         anneal_temp=0.0, anneal_decay=1.0)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 40:
        h_final, sc_final = slsqp_round(best_h, cur_n, cur_target,
                                        [10000, 30000, 50000], 1000, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
