- But we can relax to y[i,k] ≥ 0 and y[i,k] ≤ min(h[i], h[i+k]) (for 0≤h≤1)
- Or use an SDP relaxation

This might be too expensive for the time budget.

**Approach: Frank-Wolfe (Conditional Gradient) method**

The feasible set {h: 0≤h≤1, sum(h)=n/2} is a polytope (slice of a hypercube). The Frank-Wolfe method works well for such sets.

The conditional gradient at point h is: min_{v: 0≤v≤1, sum(v)=n/2} <∇f(h), v-h>

Since the gradient of f (smooth max of correlations) is a linear combination of the gradients of individual correlations, and each correlation gradient has a specific structure, the linear minimization oracle can be computed efficiently.

The gradient of corr_k with respect to h[p] is:
∂corr_k/∂h[p] = (1-h[p-k])*I(0≤p-k<n) - h[p+k]*I(0≤p+k<n)

For the smooth max objective f = (1/α) * log(Σ exp(α*corr_k)), the gradient is:
∇f = (1/Σ exp(α*corr_k)) * Σ exp(α*corr_k) * ∇corr_k

This is a weighted sum of the correlation gradients. The softmax weights concentrate on the top correlations.

The linear minimization oracle: min_v <g, v> subject to 0≤v≤1, sum(v)=n/2
= (n/2) * min(g) + Σ_{i: g[i] > median} max(0, ...) 

Actually for a box constraint with a sum constraint, the optimal v is:
- v[i] = 1 if g[i] < threshold
- v[i] = 0 if g[i] > threshold  
- v[i] = fraction if g[i] = threshold
where threshold is chosen so that sum(v) = n/2.

This is O(n log n) to compute.

The Frank-Wolfe step then is: h_new = (1-t)*h + t*v where t is the step size (full or line search).

This is a clean approach! Let me implement it.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    func = pnorm_obj if use_pnorm else smooth_lse_obj
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def frank_wolfe(h_init, n, target, time_limit, start_time, alpha=1000):
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for t in range(10000):
        if time.time() - start_time > time_limit:
            break
        
        # Compute gradient of smooth max
        corr = np.correlate(h, 1.0 - h, mode='full')
        weights = np.exp(alpha * (corr - np.max(corr)))
        weights /= np.sum(weights)
        
        # Gradient = sum_k weights[k] * grad_corr_k
        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            if weights[k_idx] < 1e-10:
                continue
            k = k_idx - (n - 1)
            v1 = (np.arange(n) - k >= 0) & (np.arange(n) - k < n)
            v2 = (np.arange(n) + k >= 0) & (np.arange(n) + k < n)
            grad[v1] += weights[k_idx] * (1.0 - h[np.arange(n)[v1] - k])
            grad[v2] -= weights[k_idx] * h[np.arange(n)[v2] + k]
        
        # Linear minimization oracle
        v = fw_oracle(grad, n)
        
        # Step size: full step or half step
        # Try full step first
        step = 2.0 / (t + 2)  # standard FW step size
        h_new = h + step * (v - h)
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_new.copy()
            no_improve = 0
            h = h_new
        else:
            # Try half step
            h_half = h + 0.5 * step * (v - h)
            h_half = project_feasible(h_half, target)
            sc_half = compute_c5(h_half)
            if sc_half < compute_c5(h) - 1e-14:
                h = h_half
            no_improve += 1
            if no_improve > 200:
                break
    
    return best_h, best_c5

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h
```

This is a simple iterative projection. It's O(n * iterations) where iterations is typically 1-3. Fine.

But wait, this doesn't compute the true Euclidean projection. It just adjusts the free variables to meet the sum constraint. The true projection would be:
h'[i] = min(max(h[i] - lambda, 0), 1) for some lambda.

Let me implement the true projection for better quality:

```python
def project_feasible_v2(h, target_sum):
    n = len(h)
    h = h.astype(float).copy()
    
    # Binary search for lambda
    lo, hi = -1.0, 1.0  # lambda range
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target_sum:
            lo = mid  # need to decrease more
        else:
            hi = mid
    h_final = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    # Fix any numerical issues
    diff = target_sum - np.sum(h_final)
    if abs(diff) > 1e-12:
        mask = (h_final > 0) & (h_final < 1)
        if mask.any():
            h_final[mask] += diff / mask.sum()
            h_final = np.clip(h_final, 0.0, 1.0)
    return h_final
```

This is O(n * 60) = O(n) which is fine.

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    # Proper Euclidean projection onto {0<=h<=1, sum(h)=target}
    n = len(h)
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        ht = np.clip(h - mid, 0.0, 1.0)
        s = float(np.sum(ht))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    diff = target_sum - float(np.sum(h_out))
    if abs(diff) > 1e-12:
        mask = (h_out > 1e-15) & (h_out < 1.0 - 1e-15)
        if mask.any():
            h_out[mask] += diff / int(mask.sum())
            h_out = np.clip(h_out, 0.0, 1.0)
        else:
            h_out += diff / n
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def smooth_lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cm = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cm), -500.0, 0.0)
    return cm + float(np.log(np.sum(np.exp(shifted)))) / alpha

def pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cm = float(np.max(corr))
    if cm <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / cm
    return cm * float(np.sum(scaled ** p)) ** (1.0/p)

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    func = pnorm_obj if use_pnorm else smooth_lse_obj
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def fast_cd(h_init, n, target, time_limit, start_time,
            max_sweeps=200000, n_cand=40, n_active=8,
            bisection_iters=14, max_no_improve=1000):
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max_val = float(np.max(corr))
        c5_val = cur_max_val * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Active shifts
        top_idx = np.argsort(corr)[-n_active:]
        top_k = (top_idx - mid).astype(int)
        active_corr = corr[top_idx].astype(float)
        
        # Gradient
        grad = np.zeros(n)
        for a in range(n_active):
            k = int(top_k[a])
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Candidates
        nc = min(n_cand, n // 4)
        nc = max(nc, 5)
        sidx = np.argsort(grad)
        low_set = sidx[:nc]
        high_set = sidx[-nc:]
        
        best_impr = 0.0
        best_i = best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            hi_val = h[i]
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                hj_val = h[j]
                d_lo = max(hi_val - 1.0, -hj_val)
                d_hi = min(hi_val, 1.0 - hj_val)
                if d_hi - d_lo < 1e-12:
                    continue
                
                # Active g
                g_act = np.zeros(n_active)
                for a in range(n_active):
                    k = int(top_k[a])
                    bjk = 0.0
                    bei = 0.0
                    jp = j - k
                    if 0 <= jp < n:
                        bjk += 1.0 - h[jp]
                    jp2 = j + k
                    if 0 <= jp2 < n:
                        bjk -= h[jp2]
                    ip = i - k
                    if 0 <= ip < n:
                        bei += 1.0 - h[ip]
                    ip2 = i + k
                    if 0 <= ip2 < n:
                        bei -= h[ip2]
                    g_act[a] = bjk - bei
                
                # Bisection
                a_d, b_d = d_lo, d_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = active_corr + mid_d * g_act
                    fm = float(np.max(vals))
                    am = vals >= fm - 1e-14
                    if float(np.max(g_act[am])) > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                ds = 0.5 * (a_d + b_d)
                if abs(ds) < 1e-12:
                    continue
                amx = float(np.max(active_corr + ds * g_act))
                if amx < cur_max_val - 1e-14:
                    impr = cur_max_val - amx
                    if impr > best_impr:
                        best_impr = impr
                        best_i = i
                        best_j = j
                        best_delta = ds
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def frank_wolfe(h_init, n, target, time_limit, start_time, alpha=500, max_iters=50000):
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    for t in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = float(np.max(corr)) * (2.0/n)
        
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Gradient of smooth max
        w = np.exp(alpha * (corr - np.max(corr)))
        w /= np.sum(w)
        
        grad = np.zeros(n)
        for ki in range(len(corr)):
            wk = w[ki]
            if wk < 1e-12:
                continue
            k = ki - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wk * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wk * h[idxs[v2] + k]
        
        # FW oracle: v = argmin <grad, v> s.t. 0<=v<=1, sum(v)=n/2
        v = np.zeros(n)
        target_count = n // 2
        bottom = np.argpartition(grad, target_count)[:target_count]
        v[bottom] = 1.0
        
        # Step size
        step = 2.0 / (t + 2)
        h_new = project_feasible(h + step * (v - h), target)
        
        sc = compute_c5(h_new)
        if sc < compute_c5(h) - 1e-14:
            h = h_new
        else:
            # Try half step
            h_half = project_feasible(h + 0.5 * step * (v - h), target)
            if compute_c5(h_half) < compute_c5(h) - 1e-14:
                h = h_half
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    if len(h_src) == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, len(h_src))
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def stochastic_kick(h, n, target, scale, seed):
    rng = np.random.RandomState(seed)
    h_new = h + rng.randn(n) * scale
    return project_feasible(h_new, target)
