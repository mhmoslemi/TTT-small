    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

import time

_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _compute_change(x, shifts):
    n = len(x)
    n_corr = 2 * n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - x[p + shifts[pos_mask]])
        change[p, neg_mask] -= x[p - shifts[neg_mask]]
    return change

def _smooth_obj_only(x):
    n = len(x)
    alpha = _g_alpha
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_grad(x):
    n = len(x)
    alpha = _g_alpha
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    obj = float(c_max + np.log(np.sum(exp_vals)) / alpha)
    
    shifts = np.arange(-(n - 1), n)
    change = _compute_change(x, shifts)
    weights = exp_vals / np.sum(exp_vals)
    grad = change @ weights
    return obj, grad

    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15, use_grad=True):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            if use_grad:
                res = minimize(_smooth_obj_grad, h_cur,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            else:
                res = minimize(_smooth_obj_only, h_cur,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

                 n_cand_start=25, n_cand_end=100, max_no_improve=400,
                 restart_every=200, restart_scale=0.015):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(99999)

    change = _compute_change(h, shifts)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

            best_h = h.copy()

def _compute_change(x, shifts):
    n = len(x)
    n_corr = 2 * n - 1
    # shifts has shape (n_corr,)
    # For each p, change[p, k] = (1-x[p+shifts[k]]) if valid - x[p-shifts[k]] if valid
    # Let's vectorize using broadcasting
    
    # pos: p + shifts[k] in [0, n) for each p, k
    # This is a 2D condition: (p + shifts) in [0, n)
    # p ranges over [0, n), shifts ranges over [-(n-1), n-1]
    
    p_idx = np.arange(n)[:, None]  # (n, 1)
    s_idx = shifts[None, :]  # (1, n_corr)
    
    pos_sum = p_idx + s_idx  # (n, n_corr)
    neg_sum = p_idx - s_idx  # (n, n_corr)
    
    pos_valid = (pos_sum >= 0) & (pos_sum < n)  # (n, n_corr)
    neg_valid = (neg_sum >= 0) & (neg_sum < n)  # (n, n_corr)
    
    # change[p, k] = (1 - x[p+shifts[k]]) * pos_valid[p,k] - x[p-shifts[k]] * neg_valid[p,k]
    pos_vals = np.zeros((n, n_corr))
    pos_vals[pos_valid] = 1.0 - x[pos_sum[pos_valid]]
    
    neg_vals = np.zeros((n, n_corr))
    neg_vals[neg_valid] = -x[neg_sum[neg_valid]]
    
    return pos_vals + neg_vals

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _compute_change_vec(x):
    n = len(x)
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    p_idx = np.arange(n)[:, None]

    pos_sum = p_idx + s_idx
    neg_sum = p_idx - s_idx
    pos_valid = (pos_sum >= 0) & (pos_sum < n)
    neg_valid = (neg_sum >= 0) & (neg_sum < n)
    result = np.zeros((n, n_corr))
    result[pos_valid] = 1.0 - x[pos_sum[pos_valid]]
    result[neg_valid] -= x[neg_sum[neg_valid]]

def interpolate_to(h_src, n_dst, target_dst):

    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0

def pair_cd(h_init, n, target, time_limit, max_sweeps=60000,
            n_cand_start=20, n_cand_end=90, max_no_improve=500,
            restart_every=250, restart_scale=0.015, n_refine=25):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(42)

    change = _compute_change_vec(h)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_every and consecutive_no_improve % 40 == 0:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            change = _compute_change_vec(h)
            consecutive_no_improve = 0

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        screened = []
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g_k = (change[j] - change[i])[k_star_idx]
                if g_k >= -1e-15:
                    continue
                est_d = min(delta_hi, max(delta_lo, (corr[k_star_idx] * 0.01) / (-g_k)))
                est_val = (corr[k_star_idx] + est_d * g_k) * dx
                if est_val < best_trial_c5:
                    screened.append((est_val, i, j, delta_lo, delta_hi))

        screened.sort(key=lambda t: t[0])
        n_to_refine = min(n_refine, len(screened))
        for c in screened[:n_to_refine]:
            _, i, j, delta_lo, delta_hi = c
            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(16):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            _update_change_row(change, best_i, h, n, shifts)
            _update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

            best_h = h_pol

    if tleft() > 300:
        h_cd, c5_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(250, tleft() * 0.6),
                              max_sweeps=50000,
                              n_cand_start=20, n_cand_end=80,
                              max_no_improve=500, n_refine=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol, sc = slsqp_grad_round(best_h, current_n, current_target,
                                     [2000, 10000, 50000], 800, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 400:
        n1 = 500
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        current_n = n1
        current_target = target1
        h_cd1, c5_cd1 = pair_cd(h1, n1, target1,
                                time_limit=min(300, tleft() * 0.6),
                                max_sweeps=40000,
                                n_cand_start=20, n_cand_end=85,
                                max_no_improve=500, n_refine=30)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1

        if tleft() > 80:
            h_pol1, sc1 = slsqp_grad_round(best_h, current_n, current_target,
                                           [5000, 20000, 100000], 1000, 1e-16)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1

    if tleft() > 250:
        h_ms, c5_ms = multi_shift_cd(best_h, current_n, current_target,
                                     time_limit=min(200, tleft() * 0.6),
                                     max_sweeps=25000, n_cand=40, n_shifts=4,
                                     n_refine=25)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if tleft() > 150:
        h_cd2, c5_cd2 = pair_cd(best_h, current_n, current_target,
                                time_limit=min(120, tleft() * 0.6),
                                max_sweeps=30000,
                                n_cand_start=30, n_cand_end=95,
                                max_no_improve=400, n_refine=35)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 80:
        h_pol2, sc2 = slsqp_grad_round(best_h, current_n, current_target,
                                       [10000, 50000, 200000], 1200, 1e-16)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 150:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.06]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = slsqp_grad_round(h_pert, current_n, current_target,
                                             [200, 1000, 5000], 400, 1e-16)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 60:
                h_cd_p, c5_p = pair_cd(h_ref, current_n, current_target,
                                       time_limit=min(50, tleft() * 0.4),
                                       max_sweeps=10000,
                                       n_cand_start=25, n_cand_end=70,
                                       max_no_improve=300, n_refine=25)
                if c5_p < best_score:
                    best_score = c5_p
                    best_h = h_cd_p

    if tleft() > 80:
        h_final_cd, c5_final = pair_cd(best_h, current_n, current_target,
                                       time_limit=min(60, tleft() * 0.7),
                                       max_sweeps=20000,
                                       n_cand_start=35, n_cand_end=100,
                                       max_no_improve=350, n_refine=40)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if tleft() > 30:
        h_final_pol, sc_final = slsqp_grad_round(best_h, current_n, current_target,
                                                  [50000, 200000, 500000], 800, 1e-16)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
