def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_shift(h, k, n):
    """Compute gradient of correlation at lag k w.r.t. all h."""
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[:n-k] += (1.0 - h[k:])
            if k > 0:
                grad[k:] -= h[:n-k]
    else:
        kp = -k
        if kp < n:
            grad[kp:] += (1.0 - h[:n-kp])
            if kp > 0:
                grad[:n-kp] -= h[kp:]
    return grad

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                a_d = delta_lo
                b_d = delta_hi

                        f_prime = 0.0

                    else:
                        a_d = mid_d

                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    
    return best_h, best_c5

def smooth_max_slsqp(h_start, n, target, alphas, maxiter_per_alpha):
    """SLSQP with smooth max approximation."""
    dx = 2.0 / n
    
    def obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def con_fun(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        try:
            res = minimize(obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, cur_score

    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_shift(h, k, n):
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[:n - k] += (1.0 - h[k:])
            if k > 0:
                grad[k:] -= h[:n - k]
    else:
        kp = -k
        if kp < n:
            grad[kp:] += (1.0 - h[:n - kp])
            if kp > 0:
                grad[:n - kp] -= h[kp:]
    return grad

def compute_change_matrix_vec(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for k in range(n_corr):
        lag = shifts[k]
        if lag >= 0:
            if lag < n:
                change[:n - lag, k] += (1.0 - h[lag:])
                if lag > 0:
                    change[lag:, k] -= h[:n - lag]
        else:
            kp = -lag
            if kp < n:
                change[kp:, k] += (1.0 - h[:n - kp])
                if kp > 0:
                    change[:n - kp, k] -= h[kp:]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

        return np.sum(x) - target

            res = minimize(_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_perturb(h, n, target, rng, n_blocks=3, max_block_size=60):
    h_pert = h.copy()
    for _ in range(n_blocks):
        bs = rng.randint(10, max_block_size)
        if bs >= n:
            bs = n // 2
        start = rng.randint(0, n - bs)
        shift_amt = rng.randint(-8, 9)
        block = h_pert[start:start + bs].copy()
        h_pert[start:start + bs] = np.roll(block, shift_amt)
        h_pert[start] = block[0]
        h_pert[start + bs - 1] = block[-1]
    return project_feasible(h_pert, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Fast Nesterov at initial resolution
    if tleft() > 120:
        h1, sc1 = nesterov_opt(best_h, n, target,
                               time_limit=min(130, tleft() * 0.45),
                               top_k=15, step_init=0.012, beta_init=0.85,
                               temp_factor=0.04)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: Pair CD for fine structure
    if tleft() > 150:
        h2, sc2 = pair_cd(best_h, n, target,
                          time_limit=min(160, tleft() * 0.45),
                          max_sweeps=35000,
                          n_cand_start=12, n_cand_end=45,
                          max_no_improve=400,
                          top_k_shifts=20, temp_factor=0.015)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Nesterov again after CD
    if tleft() > 80:
        h3, sc3 = nesterov_opt(best_h, n, target,
                               time_limit=min(80, tleft() * 0.4),
                               top_k=20, step_init=0.006, beta_init=0.92,
                               temp_factor=0.03)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: SLSQP polish
    if tleft() > 80:
        h4, sc4 = smooth_max_slsqp(best_h, n, target,
                                   [500, 3000], 350)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Multi-start with diverse perturbations
    if tleft() > 200:
        rng = np.random.RandomState(seed)
        strategies = [
            (0.005, 'noise'), (0.015, 'noise'), (0.04, 'noise'),
            (None, 'block'), (None, 'block')
        ]
        for scale, stype in strategies:
            if tleft() < 80:
                break
            if stype == 'noise':
                h_pert = best_h + rng.randn(n) * scale
                h_pert = project_feasible(h_pert, target)
            else:
                h_pert = block_perturb(best_h, n, target, rng, n_blocks=rng.randint(2, 5))

            hp, scp = nesterov_opt(h_pert, n, target,
                                   time_limit=min(40, tleft() * 0.3),
                                   top_k=15, step_init=0.01, beta_init=0.85,
                                   temp_factor=0.04)
            if scp < best_score:
                best_score = scp
                best_h = hp
            if tleft() > 40:
                hcp, scp2 = pair_cd(best_h, n, target,
                                    time_limit=min(35, tleft() * 0.3),
                                    max_sweeps=12000,
                                    n_cand_start=12, n_cand_end=40,
                                    max_no_improve=200,
                                    top_k_shifts=20, temp_factor=0.02)
                if scp2 < best_score:
                    best_score = scp2
                    best_h = hcp

    # Phase 6: Interpolate to higher resolution
    n_high = 1000
    target_high = n_high / 2.0
    if tleft() > 300:
        h_hi = interpolate_to(best_h, n_high, target_high)
        sc_hi = compute_c5(h_hi)

        if tleft() > 250:
            h_hi1, sc_hi1 = nesterov_opt(h_hi, n_high, target_high,
                                         time_limit=min(120, tleft() * 0.4),
                                         top_k=18, step_init=0.01, beta_init=0.87,
                                         temp_factor=0.04)
            if sc_hi1 < best_score:
                best_score = sc_hi1
                best_h = h_hi1
            else:
                best_h = h_hi1

        if tleft() > 120:
            h_hi2, sc_hi2 = pair_cd(best_h, n_high, target_high,
                                    time_limit=min(130, tleft() * 0.45),
                                    max_sweeps=25000,
                                    n_cand_start=15, n_cand_end=50,
                                    max_no_improve=350,
                                    top_k_shifts=25, temp_factor=0.015)
            if sc_hi2 < best_score:
                best_score = sc_hi2
                best_h = h_hi2

        if tleft() > 60:
            h_hi3, sc_hi3 = nesterov_opt(best_h, n_high, target_high,
                                         time_limit=min(60, tleft() * 0.4),
                                         top_k=20, step_init=0.005, beta_init=0.93,
                                         temp_factor=0.03)
            if sc_hi3 < best_score:
                best_score = sc_hi3
                best_h = h_hi3

    # Phase 7: Final pair CD polish
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if tleft() > 60:
        h7, sc7 = pair_cd(best_h, n_cur, target_cur,
                          time_limit=min(55, tleft() * 0.55),
                          max_sweeps=15000,
                          n_cand_start=20, n_cand_end=55,
                          max_no_improve=250,
                          top_k_shifts=30, temp_factor=0.01)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 8: Final SLSQP
    if tleft() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h8, sc8 = smooth_max_slsqp(best_h, n_cur, target_cur,
                                   [5000, 20000], 350)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: One more CD pass if time allows
    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h9, sc9 = pair_cd(best_h, n_cur, target_cur,
                          time_limit=min(25, tleft() * 0.6),
                          max_sweeps=8000,
                          n_cand_start=25, n_cand_end=60,
                          max_no_improve=150,
                          top_k_shifts=30, temp_factor=0.008)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
