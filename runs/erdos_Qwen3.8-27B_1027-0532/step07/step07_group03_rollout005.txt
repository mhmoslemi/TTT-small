def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    rng = np.random.RandomState(seed)
    if tleft() > 180:
        h_cp, sc_cp = cp_method(best_h, n0, target0,
                                time_limit=min(170, tleft() * 0.55),
                                max_iters=20000, max_cuts=400)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp
    if tleft() > 150:
        h_cd, sc_cd = interdictive_cd(best_h, n0, target0,
                                      time_limit=min(140, tleft() * 0.5),
                                      max_sweeps=25000,
                                      n_cand_start=15, n_cand_end=60,
                                      max_no_improve=350)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd
    if tleft() > 120:
        h_cp2, sc_cp2 = cp_method(best_h, n0, target0,
                                  time_limit=min(110, tleft() * 0.5),
                                  max_iters=15000, max_cuts=350)
        if sc_cp2 < best_score:
            best_score = sc_cp2
            best_h = h_cp2
    if tleft() > 100:
        h_sls, sc_sls = slsqp_round(best_h, n0, target0,
                                    [500, 2000, 8000], 400)
        if sc_sls < best_score:
            best_score = sc_sls
            best_h = h_sls
    if tleft() > 100:
        h_cd2, sc_cd2 = interdictive_cd(best_h, n0, target0,
                                        time_limit=min(90, tleft() * 0.5),
                                        max_sweeps=20000,
                                        n_cand_start=30, n_cand_end=100,
                                        max_no_improve=300)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2
    if tleft() > 80:
        h_cp3, sc_cp3 = cp_method(best_h, n0, target0,
                                  time_limit=min(70, tleft() * 0.5),
                                  max_iters=12000, max_cuts=300)
        if sc_cp3 < best_score:
            best_score = sc_cp3
            best_h = h_cp3
    if tleft() > 120:
        n1 = 800
        target1 = n1 / 2.0
        h800 = interpolate_to(best_h, n1, target1)
        sc800 = compute_c5(h800)
        if tleft() > 80:
            h800_cp, sc800_cp = cp_method(h800, n1, target1,
                                           time_limit=min(70, tleft() * 0.45),
                                           max_iters=8000, max_cuts=250)
            if sc800_cp < best_score:
                best_score = sc800_cp
                best_h = h800_cp
                n0 = n1
                target0 = target1
    if tleft() > 80:
        h_sg, sc_sg = subgradient_avg(best_h, n0, target0,
                                       time_limit=min(70, tleft() * 0.45),
                                       max_iters=3000)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
    if tleft() > 60:
        for pert_scale in [0.03, 0.07]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(n0) * pert_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc_ref = cp_method(h_pert, n0, target0,
                                       time_limit=min(25, tleft() * 0.3),
                                       max_iters=5000, max_cuts=200)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
    if tleft() > 40:
        h_cd3, sc_cd3 = interdictive_cd(best_h, n0, target0,
                                        time_limit=min(35, tleft() * 0.5),
                                        max_sweeps=15000,
                                        n_cand_start=40, n_cand_end=120,
                                        max_no_improve=250)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3
    if tleft() > 25:
        h_sls2, sc_sls2 = slsqp_round(best_h, n0, target0,
                                       [10000, 30000], 400)
        if sc_sls2 < best_score:
            best_score = sc_sls2
            best_h = h_sls2
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
