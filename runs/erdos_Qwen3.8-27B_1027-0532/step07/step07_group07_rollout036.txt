def aug_lag_lbfgs(h_init, n, target, time_limit, alpha_start=100, alpha_end=50000):
    start = time.time()
    h = h_init.copy()
    mu = 1.0
    lam = 0.0
    best_h = h.copy()
    best_score = compute_c5(h)
    alpha = alpha_start
    
    N = 1
    while N < 4*n:
        N *= 2
    
    for outer in range(50):
        if time.time() - start > time_limit - 30:
            break
        
        # Inner L-BFGS-B optimization
        def obj_grad(x):
            corr = np.correlate(x, 1-x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            exp_vals = np.exp(shifted)
            s = np.sum(exp_vals)
            w = exp_vals / s
            
            # FFT-based gradient
            fft_1x = np.fft.rfft(1-x, n=N)
            fft_w = np.fft.rfft(w, n=N)
            fft_x = np.fft.rfft(x, n=N)
            
            cross = np.fft.irfft(fft_1x * np.conj(fft_w), n=N)
            conv = np.fft.irfft(fft_x * fft_w, n=N)
            
            grad = cross[:n] - conv[:n]
            
            # Add augmented Lagrangian
            viol = np.sum(x) - target
            grad += (lam + mu * viol)
            obj = c_max + np.log(s) / alpha + lam * viol + 0.5 * mu * viol**2
            
            return obj, grad
        
        res = minimize(obj_grad, h, jac=True, method='L-BFGS-B', 
                       bounds=[(0,1)]*n, options={'maxiter': 200, 'ftol': 1e-16})
        h_new = res.x
        viol = np.sum(h_new) - target
        
        # Update Lagrange multiplier
        lam += mu * viol
        h = h_new
        
        # Increase penalty if constraint violation is large
        if abs(viol) > 1e-6:
            mu = min(mu * 2.0, 1e8)
        
        # Increase alpha (sharpen the max approximation)
        alpha = min(alpha * 1.5, alpha_end)
        
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
        
        # If constraint is well satisfied and alpha is large, try to converge
        if abs(viol) < 1e-10 and alpha >= alpha_end:
            break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def next_pow2(x):
    p = 1
    while p < x:
        p *= 2
    return p


def smooth_max_grad_fft(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        return c_max, np.zeros(n)
    w = exp_vals / s
    N = next_pow2(4 * n)
    fft_1h = np.fft.rfft(1.0 - h, n=N)
    fft_h = np.fft.rfft(h, n=N)
    fft_w = np.fft.rfft(w, n=N)
    C = np.fft.irfft(fft_1h * np.conj(fft_w), n=N)
    D = np.fft.irfft(fft_h * fft_w, n=N)
    T1 = np.concatenate([C[N - n + 1:], C[:1]])
    T2 = D[n - 1:2 * n - 1]
    obj = c_max + float(np.log(s)) / alpha
    grad = T1 - T2
    return obj, grad

def aug_lag_lbfgs(h_init, n, target, time_limit, alpha_start=200.0, alpha_end=50000.0, max_outer=40):

    h = h_init.copy().astype(float)
    mu = 10.0
    lam = 0.0
    best_h = h.copy()
    best_score = compute_c5(h)
    alpha = alpha_start
    bounds = [(0.0, 1.0)] * n

    for outer in range(max_outer):

            grad = grad_sm + (lam + mu * viol)
            aug_obj = obj + lam * viol + 0.5 * mu * viol ** 2
            return aug_obj, grad

        res = minimize(obj_grad, h, jac=True, method='L-BFGS-B',

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):

def pair_cd_v2(h_init, n, target, time_limit, max_sweeps=50000,
               n_cand_start=30, n_cand_end=80, top_k_eval=50,
               max_no_improve=500, n_shifts_grad=3):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        n_cand = max(n_cand, 10)

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx]
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def slsqp_optimize(h_start, n, target, alphas, maxiter=600, ftol=1e-14):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    tx = np.float64(target)

    for alpha in alphas:
        def obj_fn(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

        def obj_grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            exp_s = np.exp(shifted)
            s = np.sum(exp_s)
            if s < 1e-300:
                return np.zeros(n)
            weights = exp_s / s
            idxs = np.arange(n)
            grad = np.zeros(n)
            top_k = min(10, 2 * n - 1)
            top_idxs = np.argsort(weights)[-top_k:]
            for k_idx in top_idxs:
                w = weights[k_idx]
                if w < 1e-10:
                    continue
                k = k_idx - (n - 1)
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
                grad[v2] -= w * x[idxs[v2] + k]
            return grad

        def cons_fun(x):
            return np.sum(x) - tx

        try:
            res = minimize(obj_fn, h_cur, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=[{'type': 'eq', 'fun': cons_fun}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multiplicative_update(h, n, target, n_iter=50, lr=0.5):
    h = h.copy().astype(float)
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        top_shifts = np.argsort(corr)[-5:]
        grad = np.zeros(n)
        idxs = np.arange(n)
        for s_idx in top_shifts:
            k = int(s_idx) - (n - 1)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        h_new = h * np.exp(-lr * grad)
        h_new = np.clip(h_new, 1e-8, 1.0 - 1e-8)
        h_new = project_feasible(h_new, target)
        if compute_c5(h_new) <= compute_c5(h) + 1e-12:
            h = h_new
    return h

        h_cd2, c5_cd2 = pair_cd_v2(best_h, n0, target0,
                                    time_limit=min(250, tl * 0.7),
                                    max_sweeps=80000,
                                    n_cand_start=40, n_cand_end=100,
                                    top_k_eval=70, max_no_improve=800,
                                    n_shifts_grad=4)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    tl = time_left()
    if tl > 150:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
        else:
            best_h = h2
        n0 = n2
        target0 = target2

        tl = time_left()
        if tl > 100:
            h_aug3, sc_aug3 = aug_lag_lbfgs(best_h, n0, target0,
                                             time_limit=min(120, tl * 0.6),
                                             alpha_start=200.0, alpha_end=50000.0,
                                             max_outer=25)
            if sc_aug3 < best_score:
                best_score = sc_aug3
                best_h = h_aug3

    tl = time_left()
    if tl > 100:
        h_cd3, c5_cd3 = pair_cd_v2(best_h, n0, target0,
                                    time_limit=min(120, tl * 0.65),
                                    max_sweeps=40000,
                                    n_cand_start=40, n_cand_end=90,
                                    top_k_eval=60, max_no_improve=500,
                                    n_shifts_grad=3)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    tl = time_left()
    if tl > 60:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.08, 0.15]:
            tl = time_left()
            if tl < 50:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc_ref = aug_lag_lbfgs(h_pert, n0, target0,
                                           time_limit=min(40, tl * 0.4),
                                           alpha_start=500.0, alpha_end=30000.0,
                                           max_outer=15)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    tl = time_left()
    if tl > 50:
        h_cd_final, c5_final = pair_cd_v2(best_h, n0, target0,
                                           time_limit=min(60, tl * 0.7),
                                           max_sweeps=30000,
                                           n_cand_start=50, n_cand_end=100,
                                           top_k_eval=80, max_no_improve=300,
                                           n_shifts_grad=3)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    tl = time_left()
    if tl > 30:
        h_final = slsqp_optimize(best_h, n0, target0,
                                  [
                                      10000, 50000, 200000], 400, 1e-15)
        sc_final = compute_c5(h_final)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
