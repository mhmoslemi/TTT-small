def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

    return np.sum(x) - _g_target

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target

    cur_score = compute_c5(h_cur)

        func = _pnorm_obj

        func = _smooth_lse

            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})

                h_cur = h_new

def de_global(h_center, n, target, time_limit, start_time, pop_size=40, max_gen=300):

        fit = compute_c5(pop[i])

            best_fit = fit
            best = pop[i].copy()
    F = 0.8
    CR = 0.9
    for gen in range(max_gen):
        if time.time() - start_time > time_limit:
            break
        if gen % 50 == 0 and gen > 0:
            F = max(0.3, F * 0.9)
        for i in range(pop_size):
            if time.time() - start_time > time_limit:
                break
            a = (i - 1) % pop_size
            b = (i - 2) % pop_size
            c = (i - 3) % pop_size
            while a == b or a == c or b == c:
                c = rng.randint(pop_size)
            trial = pop[a].copy()
            mask = rng.rand(n) < CR

def simulated_annealing(h_init, n, target, time_limit, start_time, max_steps=10000, t_start=0.05, t_end=0.001):
    rng = np.random.RandomState(999)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        progress = step / float(max_steps)
        T = t_start * (t_end / t_start) ** progress
        n_pert = rng.choice([1, 2, 3, 4, 6])
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = 0.02 * (1.0 - progress) + 0.005
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        delta = sc - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_try
            cur_c5 = sc
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            best_h = h7
            n0 = n2
            target0 = target2

    # Phase 8: SLSQP at 800
    if remaining_time() > 100:
        h8, sc8 = slsqp_round(best_h, n0, target0,
                              [2000, 8000, 30000],
                              1200, start_time + min(100, remaining_time() - 15), start_time)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: Pair CD at high resolution
    if remaining_time() > 120:
        h9, sc9 = pair_cd(best_h, n0, target0,
                          start_time + min(120, remaining_time() - 15), start_time,
                          max_sweeps=25000, n_cand_start=25, n_cand_end=60,
                          max_no_improve=300, n_top_shifts=1, bisection_iters=20)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    # Phase 10: Simulated annealing
    if remaining_time() > 80:
        h10, sc10 = simulated_annealing(best_h, n0, target0,
                                        start_time + min(60, remaining_time() - 10), start_time,
                                        max_steps=15000, t_start=0.03, t_end=0.001)
        if sc10 < best_score:
            best_score = sc10
            best_h = h10

    # Phase 11: Noise perturbation + SLSQP (multiple trials)
    rng = np.random.RandomState(seed)
    for noise_scale in [0.01, 0.02, 0.04]:
        if remaining_time() < 50:
            break
        h_pert = best_h + rng.randn(n0) * noise_scale
        h_pert = project_feasible(h_pert, target0)
        h11, sc11 = slsqp_round(h_pert, n0, target0,
                                [500, 3000, 10000],
                                500, start_time + min(40, remaining_time() - 10), start_time)
        if sc11 < best_score:
            best_score = sc11
            best_h = h11

    # Phase 12: Final pair CD
    if remaining_time() > 60:
        h12, sc12 = pair_cd(best_h, n0, target0,
                            start_time + min(60, remaining_time() - 10), start_time,
                            max_sweeps=20000, n_cand_start=30, n_cand_end=70,
                            max_no_improve=200, n_top_shifts=1, bisection_iters=20)
        if sc12 < best_score:
            best_score = sc12
            best_h = h12

    # Phase 13: Final SLSQP polish
    if remaining_time() > 40:
        h13, sc13 = slsqp_round(best_h, n0, target0,
                                [10000, 50000],
                                500, start_time + min(40, remaining_time() - 5), start_time)
        if sc13 < best_score:
            best_score = sc13
            best_h = h13

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    if final_score < best_score:
        best_score = final_score
    return final_h, best_score, len(final_h)
