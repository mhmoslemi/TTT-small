from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_h_ref = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Shift to avoid overflow
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_grad(x, alpha):
    n = len(x)
    mid = n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    Z = np.sum(weights)
    
    # Gradient of C_smooth = (1/Z) * sum_k weights[k] * grad_k
    # grad_k[h[i]] = (1 - h[i+k]) if i+k valid, and -(h[i-k]) if i-k valid
    # Actually: f_k = sum_x h[x](1-h[x+k])
    # df_k/dh[i] = (1-h[i+k]) if i+k<n, plus -h[i-k] if i-k>=0
    grad = np.zeros(n)
    idxs = np.arange(n)
    # Only compute for shifts with non-negligible weight
    threshold = -50.0  # exp(-50) ≈ 2e-22, negligible
    active = np.where(shifted > threshold)[0]
    for s in active:
        k = int(s) - mid
        w = weights[s] / Z
        if k >= 0:
            v1 = idxs[:n-k]  # x where x+k < n
            grad[v1] += w * (1.0 - x[v1 + k])
        if k <= 0:
            v2 = idxs[-k:]  # x where x-k >= 0
            grad[v2] -= w * x[v2 - k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_list, ftol=1e-14, deadline=None, use_grad=False):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for i, alpha in enumerate(alphas):
        if deadline is not None and time.time() > deadline:
            break
        maxiter = maxiter_list[i] if i < len(maxiter_list) else maxiter_list[-1]
        try:
            if use_grad:
                _g_alpha = alpha
                res = minimize(_smooth_obj, h_cur, args=(alpha,),
                             method='SLSQP', bounds=bounds, constraints=cons,
                             jac=lambda x: _smooth_obj_grad(x, alpha),
                             options={'maxiter': maxiter, 'ftol': ftol})
            else:
                res = minimize(_smooth_obj, h_cur, args=(alpha,),
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    row = np.zeros(n_corr)
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    row[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    row[neg_mask] -= h[p - shifts[neg_mask]]
    change[p, :] = row
    return change

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _smooth_grad(x, alpha):
    n = len(x)
    mid = n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    Z = np.sum(weights)
    if Z < 1e-300:
        return np.zeros(n)
    idxs = np.arange(n)
    grad = np.zeros(n)
    active = np.where(shifted > -30.0)[0]
    for s in active:
        k = int(s) - mid
        w = weights[s] / Z
        if k >= 0:
            v1 = idxs[k:]
            if len(v1) > 0:
                grad[v1] += w * (1.0 - x[v1 - k])
            v2 = idxs[:n - k]
            if len(v2) > 0:
                grad[v2] -= w * x[v2 + k]
        else:
            v1 = idxs[:n + k]
            if len(v1) > 0:
                grad[v1] += w * (1.0 - x[v1 - k])
            v2 = idxs[-k:]
            if len(v2) > 0:
                grad[v2] -= w * x[v2 + k]
    return grad


def _smooth_grad_wrap(x):
    return _smooth_grad(x, _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_list, ftol=1e-14, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for i in range(len(alphas)):
        if deadline is not None and time.time() > deadline:
            break
        alpha = alphas[i]
        _g_alpha = alpha
        maxiter = maxiter_list[i] if i < len(maxiter_list) else maxiter_list[-1]
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           jac=_smooth_grad_wrap,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    change = np.zeros((n, n_corr))
    for s in range(n_corr):
        k = s - mid
        if k >= 0:
            v1 = idxs[k:]
            if len(v1) > 0:
                change[v1, s] += (1.0 - h[v1 - k])
            v2 = idxs[:n - k]
            if len(v2) > 0:
                change[v2, s] -= h[v2 + k]
        else:
            v1 = idxs[:n + k]
            if len(v1) > 0:
                change[v1, s] += (1.0 - h[v1 - k])
            v2 = idxs[-k:]
            if len(v2) > 0:
                change[v2, s] -= h[v2 + k]
    return change


def update_change_row(change, p, h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    row = np.zeros(n_corr)
    for s in range(n_corr):
        k = s - mid
        idx = p - k
        if 0 <= idx < n:
            row[s] += (1.0 - h[idx])
        idx2 = p + k
        if 0 <= idx2 < n:
            row[s] -= h[idx2]
    change[p, :] = row


def cd_optimize(h_init, n, target, time_limit, n_cand=30, n_active=5,
                max_sweeps=50000, reheat_interval=150, reheat_scale=0.05):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    corr = np.correlate(h, 1.0 - h, mode='full')
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
            if no_improve % reheat_interval == 0:
                rng = np.random.RandomState(sweep * 17 + 11)
                n_pert = max(8, n // 40)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                h_pert = h.copy()
                h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
                h_pert = project_feasible(h_pert, target)
                sc_pert = compute_c5(h_pert)
                if sc_pert < c5_val + 1e-4:
                    h = h_pert
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    change = compute_change_matrix(h, n)
                    c5_val = float(np.max(corr)) * dx
                    no_improve = 0
            if no_improve > 200:
                continue

        c_max = float(np.max(corr))
        threshold = c_max - 0.15 * max(c_max, 0.01)
        active_shifts = np.where(corr >= threshold)[0]
        if len(active_shifts) > n_active:
            active_shifts = np.argsort(corr)[-n_active:]
        if len(active_shifts) == 0:
            active_shifts = np.array([int(np.argmax(corr))])

        n_act = len(active_shifts)
        weights = np.ones(n_act) / n_act
        grad = np.zeros(n)
        for s_idx, w in zip(active_shifts, weights):
            k = int(s_idx) - mid
            if k >= 0:
                v1 = idxs[k:]
                if len(v1) > 0:
                    grad[v1] += w * (1.0 - h[v1 - k])
                v2 = idxs[:n - k]
                if len(v2) > 0:
                    grad[v2] -= w * h[v2 + k]
            else:
                v1 = idxs[:n + k]
                if len(v1) > 0:
                    grad[v1] += w * (1.0 - h[v1 - k])
                v2 = idxs[-k:]
                if len(v2) > 0:
                    grad[v2] -= w * h[v2 + k]

        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        k_star = int(np.argmax(corr))
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_trial = c5_val

        candidates = []
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                d_ks = change[j, k_star] - change[i, k_star]
                if d_ks < -0.005:
                    candidates.append((d_ks, i, j, delta_lo, delta_hi))

        candidates.sort(key=lambda x: x[0])
        candidates = candidates[:25]

        for d_ks, i, j, delta_lo, delta_hi in candidates:
            d = change[j] - change[i]
            a = delta_lo
            b = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a + b)
                vals = corr + mid_d * d
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                if active_mask.any():
                    f_prime = np.max(d[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-10:
                    b = mid_d
                else:
                    a = mid_d
            delta_star = 0.5 * (a + b)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = float(np.max(corr + delta_star * d)) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            if trial_count < 5:
                block_avg, bi = block_starts[trial_count]
            else:
                block_avg, bi = block_starts[len(block_starts) - 1 - (trial_count - 5)]

            block_high = list(range(bi, bi + block_size))

            best_j = -1
            best_j_delta = 0.0
            for jj in range(n_cand):
                j = int(np.argsort(grad)[-n_cand:][jj])
                if j in block_high:
                    continue
                d = np.zeros(2 * n - 1)
                for p in block_high:
                    d += change[j] - change[p]
                d_ks = d[k_star]
                if d_ks >= 0:
                    continue
                delta_lo = max(h[j] - 1.0, -min([h[p] for p in block_high]))
                delta_hi = min(1.0 - h[j], min([1.0 - h[p] for p in block_high]))
                if delta_hi - delta_lo < 1e-12:
                    continue
                a = delta_lo
                b = delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a + b)
                    vals = corr + mid_d * d
                    f_mid = np.max(vals)
                    am = vals >= f_mid - 1e-12
                    fp = np.max(d[am]) if am.any() else 0.0
                    if fp > 1e-10:
                        b = mid_d
                    else:
                        a = mid_d
                delta_star = 0.5 * (a + b)
                if abs(delta_star) < 1e-12:
                    continue
                approx = float(np.max(corr + delta_star * d)) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_delta = delta_star
                    best_block = bi
                    best_j = j

        if best_block >= 0 and best_j >= 0:
            bi = best_block
            for p in range(bi, bi + block_size):
                h[p] += best_delta
            h[best_j] -= best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in range(bi, bi + block_size):
                update_change_row(change, p, h, n)
            update_change_row(change, best_j, h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 120:
        deadline = time.time() + min(100, tleft() * 0.5)
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [50, 200, 1000, 5000, 30000],
                                   [300, 500, 800, 1200, 2000],
                                   ftol=1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 300:
        h_cd, c5_cd = cd_optimize(best_h, current_n, current_target,
                                  time_limit=min(250, tleft() * 0.55),
                                  n_cand=25, n_active=4,
                                  max_sweeps=50000,
                                  reheat_interval=120, reheat_scale=0.05)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        deadline = time.time() + min(60, tleft() * 0.5)
        h_pol, sc = slsqp_optimize(best_h, current_n, current_target,
                                   [2000, 10000, 50000, 200000],
                                   [500, 1000, 1500, 2000],
                                   ftol=1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_bc, c5_bc = block_cd(best_h, current_n, current_target,
                               time_limit=min(120, tleft() * 0.5),
                               block_size=3, n_cand=20,
                               max_sweeps=20000,
                               reheat_interval=150, reheat_scale=0.04)
        if c5_bc < best_score:
            best_score = c5_bc
            best_h = h_bc
        deadline = time.time() + min(40, tleft() * 0.3)
        h_pol, sc = slsqp_optimize(best_h, current_n, current_target,
                                   [10000, 50000, 200000],
                                   [800, 1200, 1500],
                                   ftol=1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score + 0.001:
            current_n = n1
            current_target = target1
            best_h = h1
            if tleft() > 250:
                deadline = time.time() + min(120, tleft() * 0.4)
                h_pol1, sc1b = slsqp_optimize(h1, n1, target1,
                                              [200, 1000, 5000, 20000],
                                              [400, 600, 1000, 1500],
                                              ftol=1e-14, deadline=deadline)
                if sc1b < best_score:
                    best_score = sc1b
                    best_h = h_pol1
            if tleft() > 150:
                h_cd1, c5_cd1 = cd_optimize(best_h, current_n, current_target,
                                            time_limit=min(100, tleft() * 0.45),
                                            n_cand=20, n_active=4,
                                            max_sweeps=30000,
                                            reheat_interval=100, reheat_scale=0.04)
                if c5_cd1 < best_score:
                    best_score = c5_cd1
                    best_h = h_cd1
            if tleft() > 60:
                deadline = time.time() + min(40, tleft() * 0.5)
                h_pol1b, sc1c = slsqp_optimize(best_h, current_n, current_target,
                                               [10000, 50000, 200000],
                                               [800, 1200, 1500],
                                               ftol=1e-15, deadline=deadline)
                if sc1c < best_score:
                    best_score = sc1c
                    best_h = h_pol1b

    if tleft() > 150:
        for attempt in range(3):
            if tleft() < 50:
                break
            noise_scales = [0.02, 0.05, 0.08]
            h_pert = best_h + rng.randn(current_n) * noise_scales[attempt]
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(40, tleft() * 0.3)
            h_ref, sc_ref = slsqp_optimize(h_pert, current_n, current_target,
                                           [500, 2000, 10000, 50000],
                                           [300, 500, 800, 1000],
                                           ftol=1e-14, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 50:
        deadline = time.time() + min(35, tleft() * 0.6)
        h_final, sc_final = slsqp_optimize(best_h, current_n, current_target,
                                           [50000, 200000, 500000],
                                           [1000, 1500, 2000],
                                           ftol=1e-15, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
