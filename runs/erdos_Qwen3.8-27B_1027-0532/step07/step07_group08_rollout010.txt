import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, alpha):
    N = len(h)
    dx = 2.0 / N
    mid = N - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        return np.zeros(N)
    weights = exp_vals / total
    
    grad = np.zeros(N)
    threshold = 1e-10
    active = np.where(weights > threshold)[0]
    
    for m in active:
        k = int(m) - mid
        w = weights[m] * dx
        if k >= 0:
            j1 = np.arange(0, N - k)
            grad[j1] += w * (1.0 - h[j1 + k])
            j2 = np.arange(k, N)
            grad[j2] -= w * h[j2 - k]
        else:
            j1 = np.arange(-k, N)
            grad[j1] += w * (1.0 - h[j1 + k])
            j2 = np.arange(0, N + k)
            grad[j2] -= w * h[j2 - k]
    
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def _slsqp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def grad_descent_momentum(h_init, n, target, time_limit, alpha=200.0, 
                          lr=0.001, momentum=0.9, max_steps=100000):
    """Gradient descent with momentum and adaptive step size."""
    start = time.time()
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    stale = 0
    current_lr = lr
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        grad = smooth_grad(h, alpha)
        
        # Momentum update
        v = momentum * v - current_lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        score = compute_c5(h_new)
        if score < best_score - 1e-14:
            best_score = score
            best_h = h_new.copy()
            stale = 0
            # Optionally increase lr slightly
            current_lr = min(current_lr * 1.05, lr * 2)
        else:
            stale += 1
            current_lr *= 0.8
            if stale > 200:
                break
        
        h = h_new
    
    return best_h, best_score

def compute_change_matrix(h, N):
    n_corr = 2 * N - 1
    M = np.zeros((N, n_corr))
    mid = N - 1
    
    for m in range(n_corr):
        k = m - mid
        if k >= 0:
            if k < N:
                j1 = np.arange(0, N - k)
                M[j1, m] = (1.0 - h[j1 + k])
            if k > 0:
                j2 = np.arange(k, N)
                M[j2, m] -= h[j2 - k]
        else:
            kk = -k
            if kk < N:
                j1 = np.arange(kk, N)
                M[j1, m] = (1.0 - h[j1 + k])
            if kk > 0:
                j2 = np.arange(0, N - kk)
                M[j2, m] -= h[j2 - k]
    
    return M

def update_change_row(M, p, h, N, mid):
    n_corr = 2 * N - 1
    M[p, :] = 0.0
    for m in range(n_corr):

        if k >= 0:
            if 0 <= p + k < N:
                M[p, m] += (1.0 - h[p + k])
            if 0 <= p - k < N:

        else:
            kk = -k
            if 0 <= p + k < N:
                M[p, m] += (1.0 - h[p + k])
            if 0 <= p - k < N:
                M[p, m] -= h[p - k]
    return M

def pair_cd(h_init, n, target, time_limit, max_sweeps=10000,

        c5_val = float(np.max(corr)) * dx

            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

            grad[v1] += w * (1.0 - h[np.arange(n)[v1] - k])

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

                if g[k_star] > 0.3 * np.max(np.abs(g)):
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(10):

                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, alpha):
    N = len(h)
    dx = 2.0 / N
    mid = N - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        return np.zeros(N)
    weights = exp_vals / total
    
    grad = np.zeros(N)
    active = np.where(weights > 1e-10)[0]
    
    for m in active:
        k = int(m) - mid
        w = weights[m] * dx
        if k >= 0:
            j1 = np.arange(0, N - k)
            grad[j1] += w * (1.0 - h[j1 + k])
            if k > 0:
                j2 = np.arange(k, N)
                grad[j2] -= w * h[j2 - k]
        else:
            kk = -k
            j1 = np.arange(kk, N)
            grad[j1] += w * (1.0 - h[j1 + k])
            if kk > 0:
                j2 = np.arange(0, N - kk)
                grad[j2] -= w * h[j2 - k]
    
    return grad

    return np.sum(x) - _g_target

    return np.ones(_g_n)

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter, deadline=None):

    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

            break
