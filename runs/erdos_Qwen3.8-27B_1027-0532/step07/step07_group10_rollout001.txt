                     max_iters=50000, base_lr=None):

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    significant = np.where(weights > 1e-12)[0]
    for k_idx in significant:
        k = int(k_idx) - mid
        wp = (idxs + k >= 0) & (idxs + k < n)
        wm = (idxs - k >= 0) & (idxs - k < n)
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def nesterov_gd(h_init, n, target, time_limit, max_iters=100000,
                alpha_start=50.0, alpha_end=2000.0,
                lr0=None, momentum=0.9, n_sig=80):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    if lr0 is None:
        lr0 = 0.5 / alpha_start

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 20000:
            break

        progress = min(float(it) / max_iters, 1.0)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        lr = lr0 * (1.0 / (1.0 + 0.1 * progress))
        lr = min(lr, 2.0 / (alpha))

        y = h - momentum * lr * v
        y = project_feasible(y, target)

        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            weights = np.zeros(n_corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= wsum

        sig_mask = weights > 1e-10
        n_sig_actual = np.minimum(int(np.sum(sig_mask)), n_sig)
        if n_sig_actual == 0:
            n_sig_actual = 1
        sig_idx = np.argsort(weights)[-n_sig_actual:]
        w_masked = np.zeros(n_corr)
        w_masked[sig_idx] = weights[sig_idx]
        w_masked /= np.sum(w_masked)

        grad = np.zeros(n)
        for k_idx in sig_idx:
            if w_masked[k_idx] < 1e-12:
                continue
            k = int(k_idx) - mid
            w = w_masked[k_idx]
            wp = (idxs + k >= 0) & (idxs + k < n)
            wm = (idxs - k >= 0) & (idxs - k < n)
            grad[wp] += w * (1.0 - y[idxs[wp] + k])
            grad[wm] -= w * y[idxs[wm] - k]
        grad -= np.mean(grad)

        v = momentum * v + grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target)

        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_new.copy()
            no_improve = 0
            h = h_new
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                h = best_h.copy()
                v = np.zeros(n)
                h = h_new
            h = h_new

    return best_h, best_c5


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=30, n_cand_end=100, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-15:
                    best_trial_c5 = float(approx_max) * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def multi_active_cd(h_init, n, target, time_limit, n_cand=60, n_active=5,
                    max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(30000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-n_active:]
        weights = np.zeros(n_corr)
        c_max = corr[top_shifts[0]]
        for si in top_shifts:
            weights[si] = np.exp(-30.0 * (c_max - corr[si]))
        sw = np.sum(weights)
        if sw > 0:
            weights /= sw
        else:
            weights[top_shifts[0]] = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            w = weights[s_idx]
            if w < 1e-10:
                continue
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fista_refine(h_init, n, target, time_limit, max_iters=50000,
                 alpha=500.0, lr0=0.01):
    start = time.time()
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    x = h_init.copy().astype(float)
    y = x.copy()
    t = 1.0
    best_h = x.copy()
    best_c5 = compute_c5(x)
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 10000:
            break

        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            weights = np.zeros(n_corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= wsum

        sig_mask = weights > 1e-10
        sig_idx = np.where(sig_mask)[0]
        if len(sig_idx) == 0:
            sig_idx = np.array([np.argmax(corr)])

        grad = np.zeros(n)
        for k_idx in sig_idx:
            k = int(k_idx) - mid
            w = weights[k_idx]
            wp = (idxs + k >= 0) & (idxs + k < n)
            wm = (idxs - k >= 0) & (idxs - k < n)
            grad[wp] += w * (1.0 - y[idxs[wp] + k])
            grad[wm] -= w * y[idxs[wm] - k]
        grad -= np.mean(grad)

        lr = lr0 / (1.0 + 0.01 * it)
        x_new = y - lr * grad
        x_new = project_feasible(x_new, target)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = project_feasible(y, target)
        x = x_new
        t = t_new

        c5_val = compute_c5(x)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = x.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    # Phase 1: Quick Nesterov at original resolution
    if tleft() > 60:
        h_ng, sc_ng = nesterov_gd(best_h, best_n, best_target,
                                  time_limit=min(40, tleft() * 0.4),
                                  max_iters=20000, alpha_start=20.0,
                                  alpha_end=500.0, lr0=0.05, momentum=0.85,
                                  n_sig=40)
        if sc_ng < best_score:
            best_score = sc_ng
            best_h = h_ng

    # Phase 2: Interpolate to n=800
    n1 = 800
    target1 = n1 / 2.0
    if tleft() > 300:
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            best_n = n1
            best_target = target1

        # Nesterov at n=800
        if tleft() > 250:
            h1_ng, sc1_ng = nesterov_gd(best_h, best_n, best_target,
                                        time_limit=min(180, tleft() * 0.55),
                                        max_iters=60000, alpha_start=30.0,
                                        alpha_end=1500.0, lr0=0.03,
                                        momentum=0.9, n_sig=60)
            if sc1_ng < best_score:
                best_score = sc1_ng
                best_h = h1_ng

        # FISTA at n=800
        if tleft() > 120:
            h1_fi, sc1_fi = fista_refine(best_h, best_n, best_target,
                                         time_limit=min(80, tleft() * 0.4),
                                         max_iters=30000, alpha=800.0,
                                         lr0=0.008)
            if sc1_fi < best_score:
                best_score = sc1_fi
                best_h = h1_fi

        # Pair CD at n=800
        if tleft() > 100:
            h1_cd, sc1_cd = fast_pair_cd(best_h, best_n, best_target,
                                         time_limit=min(80, tleft() * 0.5),
                                         max_sweeps=30000,
                                         n_cand_start=30, n_cand_end=100,
                                         max_no_improve=300)
            if sc1_cd < best_score:
                best_score = sc1_cd
                best_h = h1_cd

    # Phase 3: Multi-active CD at n=800
    if tleft() > 100:
        h_ma, sc_ma = multi_active_cd(best_h, best_n, best_target,
                                      time_limit=min(80, tleft() * 0.5),
                                      n_cand=60, n_active=6,
                                      max_no_improve=250)
        if sc_ma < best_score:
            best_score = sc_ma
            best_h = h_ma

    # Phase 4: Try n=1000
    n2 = 1000
    target2 = n2 / 2.0
    if tleft() > 350:
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        work_at_n2 = sc2 < best_score + 0.005

        if work_at_n2:
            if tleft() > 250:
                h2_ng, sc2_ng = nesterov_gd(best_h if not work_at_n2 else h2,
                                            best_n if not work_at_n2 else n2,
                                            best_target if not work_at_n2 else target2,
                                            time_limit=min(180, tleft() * 0.5),
                                            max_iters=50000, alpha_start=30.0,
                                            alpha_end=2000.0, lr0=0.025,
                                            momentum=0.9, n_sig=50)
                if sc2_ng < best_score:
                    best_score = sc2_ng
                    best_h = h2_ng
                    best_n = n2
                    best_target = target2

            if tleft() > 150:
                h2_fi, sc2_fi = fista_refine(best_h, best_n, best_target,
                                             time_limit=min(80, tleft() * 0.4),
                                             max_iters=25000, alpha=1000.0,
                                             lr0=0.006)
                if sc2_fi < best_score:
                    best_score = sc2_fi
                    best_h = h2_fi

            if tleft() > 100:
                h2_cd, sc2_cd = fast_pair_cd(best_h, best_n, best_target,
                                             time_limit=min(70, tleft() * 0.5),
                                             max_sweeps=25000,
                                             n_cand_start=35, n_cand_end=110,
                                             max_no_improve=250)
                if sc2_cd < best_score:
                    best_score = sc2_cd
                    best_h = h2_cd

    # Phase 5: SLSQP polish (only if time remains)
    if tleft() > 60:
        h_sp, sc_sp = slsqp_round(best_h, best_n, best_target,
                                  [2000, 10000], 800, 1e-15)
        if sc_sp < best_score:
            best_score = sc_sp
            best_h = h_sp

    # Phase 6: Final Nesterov polish
    if tleft() > 80:
        h_fin, sc_fin = nesterov_gd(best_h, best_n, best_target,
                                    time_limit=min(60, tleft() * 0.6),
                                    max_iters=30000, alpha_start=200.0,
                                    alpha_end=5000.0, lr0=0.01,
                                    momentum=0.92, n_sig=40)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 7: Final CD
    if tleft() > 50:
        h_cd_f, sc_cd_f = fast_pair_cd(best_h, best_n, best_target,
                                       time_limit=min(35, tleft() * 0.5),
                                       max_sweeps=15000,
                                       n_cand_start=50, n_cand_end=120,
                                       max_no_improve=200)
        if sc_cd_f < best_score:
            best_score = sc_cd_f
            best_h = h_cd_f

    # Phase 8: Random restarts
    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.05, 0.08]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            h_ref, sc_ref = nesterov_gd(h_pert, best_n, best_target,
                                        time_limit=min(25, tleft() * 0.3),
                                        max_iters=10000, alpha_start=100.0,
                                        alpha_end=3000.0, lr0=0.015,
                                        momentum=0.9, n_sig=40)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Final small-scale CD
    if tleft() > 30:
        h_last, sc_last = fast_pair_cd(best_h, best_n, best_target,
                                       time_limit=min(20, tleft() * 0.5),
                                       max_sweeps=8000,
                                       n_cand_start=60, n_cand_end=120,
                                       max_no_improve=100)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, best_n
