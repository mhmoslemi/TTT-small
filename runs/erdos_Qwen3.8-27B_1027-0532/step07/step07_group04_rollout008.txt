        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_highs = None
        best_lows = None
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        hi_idx = n_h // 2
        lo_idx = n_l // 2

        for ii in range(max(0, hi_idx - block_size), min(n_h, hi_idx + block_size)):
            highs = list(high_set[max(0, ii):min(n_h, ii + block_size)])
            for jj in range(max(0, lo_idx - block_size), min(n_l, lo_idx + block_size)):
                lows = list(low_set[max(0, jj):min(n_l, jj + block_size)])
                for offset in range(max(0, len(highs) - block_size + 1)):
                    hi_block = highs[offset:offset + min(block_size, len(highs) - offset)]
                    for lo_off in range(max(0, len(lows) - block_size + 1)):
                        lo_block = lows[lo_off:lo_off + min(block_size, len(lows) - lo_off)]
                        if len(hi_block) != len(lo_block) or len(hi_block) == 0:
                            continue
                        overlap = set(hi_block) & set(lo_block)
                        if overlap:
                            continue

                        delta_lo = max(min(h[i] - 1.0 for i in hi_block),
                                      -min(h[j] for j in lo_block) / len(lo_block))
                        delta_hi = min(min(h[i] for i in hi_block),
                                      (1.0 - max(h[j] for j in lo_block)) / len(lo_block))
                        if delta_hi - delta_lo < 1e-12:
                            continue

                        g_block = np.zeros(n_corr)
                        for i in hi_block:
                            g_block -= change[i]
                        for j in lo_block:
                            g_block += change[j] / len(lo_block) * len(hi_block)

                        a_d = delta_lo
                        b_d = delta_hi
                        for _ in range(10):
                            mid_d = 0.5 * (a_d + b_d)
                            vals = corr + mid_d * g_block
                            f_mid = np.max(vals)
                            active_mask = vals >= f_mid - 1e-14
                            if active_mask.any():
                                f_prime = np.max(g_block[active_mask])
                            else:
                                f_prime = 0.0
                            if f_prime > 1e-12:
                                b_d = mid_d
                            else:
                                a_d = mid_d

                        delta_star = 0.5 * (a_d + b_d)
                        if abs(delta_star) < 1e-12:
                            continue

                        approx_max = np.max(corr + delta_star * g_block) * dx
                        if approx_max < best_trial_c5 - 1e-15:
                            best_trial_c5 = approx_max
                            best_highs = hi_block
                            best_lows = lo_block
                            best_delta = delta_star

        if best_highs is not None:
            for i in best_highs:
                h[i] -= best_delta
            for j in best_lows:
                h[j] += best_delta * len(best_highs) / len(best_lows)
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for i in best_highs:
                update_change_row(change, i, h, n, shifts)
            for j in best_lows:
                update_change_row(change, j, h, n, shifts)

    return best_h, best_c5

def diversify_restart(h_best, n, target, rng, n_variants=5):
    results = []
    for r in range(n_variants):
        scale = 0.03 + 0.04 * r
        h_pert = h_best + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        results.append((sc, h_pert.copy()))
    results.sort(key=lambda x: x[0])
    return results

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    # Phase 1: Quick SLSQP polish at initial resolution
    if tleft() > 60:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [500, 2000], 500, 1e-14,
                                   time_limit=min(40, tleft() * 0.5),
                                   start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Interpolate to n=800 and run adaptive CD
    if tleft() > 200:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_cd1, c5_cd1 = adaptive_cd(h1, n1, target1,
                                    time_limit=min(120, tleft() * 0.5),
                                    max_sweeps=40000,
                                    n_cand_base=25, n_cand_max=60,
                                    max_no_improve=400, top_k=4,
                                    weight_temp=6.0)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1
            best_n = n1
            best_target = target1

    # Phase 3: Interpolate to n=1600 and run adaptive CD
    if tleft() > 250:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_cd2, c5_cd2 = adaptive_cd(h2, n2, target2,
                                    time_limit=min(180, tleft() * 0.5),
                                    max_sweeps=30000,
                                    n_cand_base=25, n_cand_max=50,
                                    max_no_improve=400, top_k=4,
                                    weight_temp=8.0)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
            best_n = n2
            best_target = target2

    # Phase 4: SLSQP at higher resolution
    if tleft() > 100:
        h_p4, sc4 = slsqp_optimize(best_h, best_n, best_target,
                                   [2000, 8000], 800, 1e-14,
                                   time_limit=min(80, tleft() * 0.6),
                                   start_time=start_time)
        if sc4 < best_score:
            best_score = sc4
            best_h = h_p4

    # Phase 5: Block CD for structured moves
    if tleft() > 100:
        h_b5, sc5 = block_cd(best_h, best_n, best_target,
                             time_limit=min(60, tleft() * 0.5),
                             max_sweeps=10000, block_size=2,
                             n_cand=30, max_no_improve=200)
        if sc5 < best_score:
            best_score = sc5
            best_h = h_b5

    # Phase 6: Try n=2000 with adaptive CD
    if tleft() > 250:
        n3 = 2000
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        h_cd3, c5_cd3 = adaptive_cd(h3, n3, target3,
                                    time_limit=min(180, tleft() * 0.5),
                                    max_sweeps=25000,
                                    n_cand_base=20, n_cand_max=40,
                                    max_no_improve=350, top_k=5,
                                    weight_temp=10.0)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
            best_n = n3
            best_target = target3

    # Phase 7: Diversified restarts + refinement
    if tleft() > 150:
        variants = diversify_restart(best_h, best_n, best_target, rng, n_variants=4)
        for sc_v, h_v in variants:
            if tleft() < 60:
                break
            if sc_v < best_score + 0.005:
                h_r, sc_r = adaptive_cd(h_v, best_n, best_target,
                                        time_limit=min(40, tleft() * 0.4),
                                        max_sweeps=10000,
                                        n_cand_base=20, n_cand_max=40,
                                        max_no_improve=200, top_k=3,
                                        weight_temp=6.0)
                if sc_r < best_score:
                    best_score = sc_r
                    best_h = h_r

    # Phase 8: Final SLSQP with very high alpha
    if tleft() > 80:
        h_p8, sc8 = slsqp_optimize(best_h, best_n, best_target,
                                   [5000, 20000, 50000], 1000, 1e-14,
                                   time_limit=min(60, tleft() * 0.6),
                                   start_time=start_time)
        if sc8 < best_score:
            best_score = sc8
            best_h = h_p8

    # Phase 9: Final adaptive CD burst
    if tleft() > 40:
        h_cd9, c5_cd9 = adaptive_cd(best_h, best_n, best_target,
                                    time_limit=min(30, tleft() * 0.7),
                                    max_sweeps=10000,
                                    n_cand_base=30, n_cand_max=50,
                                    max_no_improve=200, top_k=3,
                                    weight_temp=5.0)
        if c5_cd9 < best_score:
            best_score = c5_cd9
            best_h = h_cd9

    # Phase 10: Final block CD
    if tleft() > 30:
        h_b10, sc10 = block_cd(best_h, best_n, best_target,
                               time_limit=min(20, tleft() * 0.7),
                               max_sweeps=5000, block_size=2,
                               n_cand=25, max_no_improve=100)
        if sc10 < best_score:
            best_score = sc10
            best_h = h_b10

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
