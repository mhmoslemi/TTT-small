Let me flesh out the Frank-Wolfe idea:

The feasible set is: S = {h : 0 ≤ h ≤ 1, sum(h) = n/2}

This is a polytope (a slice of the hypercube). The extreme points are vectors with exactly n/2 entries equal to 1 and the rest 0 (assuming n/2 is an integer).

For the Frank-Wolfe step:
- Compute gradient g (subgradient of the objective)
- Find v = argmin_{v ∈ S} <g, v>
  - This means: sort g, set the n/2 smallest to 1, rest to 0
  - If n/2 is not integer, one coordinate is fractional
- Update: h ← h + γ(v - h), where γ is the step size

This is O(n log n) per step (for sorting), which is much faster than SLSQP!

For the subgradient of C₅(h) = dx * max_k corr_k(h):
- If only one shift k* is active: g = ∂/∂h [correlation at shift k*]
- If multiple shifts are active: use the average (or convex combination) of their gradients

Let me also think about a **conditional gradient with away steps** (AGD method):
- Regular FW: h ← (1-γ)h + γv where v is the minimizer
- Away step: h ← (1-γ)h + γv_away where v_away is the MAXIMIZER in the support of h
- This avoids the slow convergence of vanilla FW

def frank_wolfe(h_init, n, target, time_limit, n_steps=50000):
    # Frank-Wolfe / conditional gradient with away steps
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Track support for away steps
    support = set(np.where(h > 1e-10)[0])
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        if (step + 1) % 100 == 0 and time.time() - start > time_limit * 0.9:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Compute subgradient using top shifts
        top_k = np.argsort(corr)[-3:]
        p = np.exp(corr[top_k] * 100.0)
        p /= p.sum()
        
        grad = np.zeros(n)
        for idx, s_idx in enumerate(top_k):
            k = int(s_idx) - mid
            w = p[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)  # project out constraint normal
        
        # Line search or fixed step
        gamma = 2.0 / (step + 2)
        
        # FW step: find v that minimizes <grad, v> over feasible set
        # Sort grad, set smallest target to 1, rest to 0
        sorted_idx = np.argsort(grad)
        v = np.zeros(n)
        v[sorted_idx[:int(target)]] = 1.0
        # Handle fractional part
        if target != int(target):
            v[sorted_idx[int(target)]] = target - int(target)
        
        # Away step: find max grad in support
        if len(support) > int(target):
            support_arr = np.array(list(support))
            away_idx = support_arr[np.argmax(grad[support_arr])]
            v_away = np.zeros(n)
            v_away[away_idx] = -1.0  # direction to remove
            
            # Try both FW and away step
            h_fw = h + gamma * (v - h)
            h_fw = project_feasible(h_fw, target)
            sc_fw = compute_c5(h_fw)
            
            h_away = h + gamma * (v_away[away_idx] * (h[away_idx] / h[away_idx]) * (v - h_away))
            # Actually away step: h_away = h - gamma * h[away_idx] * e_away_idx
            # Then renormalize
            h_away = h.copy()
            h_away[away_idx] -= gamma * h_away[away_idx]
            h_away = np.clip(h_away, 0, 1)
            h_away = project_feasible(h_away, target)
            sc_away = compute_c5(h_away)
            
            if sc_fw <= sc_away:
                h = h_fw
            else:
                h = h_away
                support.discard(int(away_idx))
                support.update(int(i) for i in np.where(h > 1e-10)[0])
        
        # Update support
        support = set(int(i) for i in np.where(h > 1e-10)[0])
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha, n_corr):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def frank_wolfe(h_init, n, target, time_limit, n_shifts=3, use_away=True, alpha_w=50.0):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    no_improve = 0
    for step in range(500000):
        if time.time() - start > time_limit:
            break
        if no_improve > 10000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        top_k = np.argsort(corr)[-n_shifts:]
        p = np.exp(corr[top_k] * alpha_w)
        p /= p.sum()
        
        grad = np.zeros(n)
        for idx, s_idx in enumerate(top_k):
            k = int(s_idx) - mid
            w = p[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        gamma = 2.0 / (step + 2.0)
        
        sorted_idx = np.argsort(grad)
        v = np.zeros(n)
        n_full = int(target)
        v[sorted_idx[:n_full]] = 1.0
        if target != n_full:
            v[sorted_idx[n_full]] = target - n_full
        
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)
        
        if use_away:
            support = np.where(h > 1e-10)[0]
            if len(support) > n_full:
                j = int(support[np.argmax(grad[support])])
                h_away = h.copy()
                h_away[j] = max(0.0, h_away[j] - gamma * h_away[j])
                h_away = project_feasible(h_away, target)
                sc_away = compute_c5(h_away)
                sc_fw = compute_c5(h_new)
                if sc_away < sc_fw:
                    h_new = h_away
        
        h = h_new
    
    return best_h, best_c5


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=100, n_shifts=5,
            max_no_improve=500, anneal_temp=0.0, anneal_decay=0.999):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = np.sum(shift_weights)
        if sw_sum < 1e-15:
            sw_sum = 1.0
        # Use exponential weighting for concentration
        pw = np.exp((shift_weights - shift_weights.max()) * 10.0)
        shift_weights = pw / pw.sum()

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 1000.0
                    if np.random.random() < np.exp(-delta_obj / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 2:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.95, 1e-10)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_descent(h_init, n, target, time_limit, n_steps=20000, step_scale=0.005, n_shifts=5):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.corralate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        top_k = np.argsort(corr)[-n_shifts:]
        p = np.exp(corr[top_k] * 30.0)
        p /= p.sum()

        grad = np.zeros(n)
        for idx, s_idx in enumerate(top_k):
            k = int(s_idx) - mid
            w = p[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
            eta = min(eta * 1.05, step_scale)
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.9

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

# Global variables for SLSQP constraint
_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def frank_wolfe(h_init, n, target, time_limit, n_shifts=3, use_away=True, alpha_w=30.0):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    step = 0
    
    while time.time() - start < time_limit:
        if no_improve > 15000:
            break
        step += 1
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Subgradient
        top_k = np.argsort(corr)[-n_shifts:]
        p = np.exp(corr[top_k] * alpha_w)
        p /= p.sum()
        
        grad = np.zeros(n)
        for idx_k in range(n_shifts):
            k = int(top_k[idx_k]) - mid
            w = p[idx_k]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # FW step size
        gamma = 2.0 / (step + 2.0)
        
        # Linear oracle: argmin <grad, v> over {0<=v<=1, sum(v)=target}
        sorted_idx = np.argsort(grad)
        v = np.zeros(n)
        n_full = int(target)
        v[sorted_idx[:n_full]] = 1.0
        if target != n_full:
            v[sorted_idx[n_full]] = target - n_full
        
        # FW update
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, target)
        
        # Away step
        if use_away:
            support = np.where(h > 1e-10)[0]
            if len(support) > n_full + 5:
                j = int(support[np.argmax(grad[support])])
                h_away = h.copy()
                h_away[j] *= (1.0 - gamma)
                h_away = project_feasible(h_away, target)
                sc_fw = compute_c5(h_new)
                sc_away = compute_c5(h_away)
                if sc_away < sc_fw:
                    h_new = h_away
        
        h = h_new
    
    return best_h, best_c5

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=100, n_shifts=5,

    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        # Gradient from top shifts
        top_shifts = np.argsort(corr)[-n_shifts:]
        pw = np.exp((corr[top_shifts] - corr[top_shifts].max()) * 10.0)
        shift_weights = pw / pw.sum()

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 1000.0
                    if np.random.random() < np.exp(-delta_obj / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 2:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.95, 1e-10)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_multi(h_init, n, target, time_limit, n_steps=30000, n_weights=5, step_scale=0.004):
