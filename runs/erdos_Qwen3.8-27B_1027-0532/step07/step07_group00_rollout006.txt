```python
def subgradient_step(h, n, tol_frac=0.005, nw_iter=50):
    """Compute steepest descent direction for max of correlations using MWU"""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    active_mask = corr >= c_max - tol_frac * c_max
    active_shifts = np.where(active_mask)[0]
    n_active = len(active_shifts)
    
    if n_active == 0:
        return np.zeros(n), c_max * dx
    
    # Precompute gradients for active shifts
    grads = np.zeros((n_active, n))
    for ai, s_idx in enumerate(active_shifts):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grads[ai, v1] += (1.0 - h[idxs[v1] - k])
        grads[ai, v2] -= h[idxs[v2] + k]
    
    # Project gradients onto sum=0 subspace
    for ai in range(n_active):
        grads[ai] -= np.mean(grads[ai])
    
    # Multiplicative weights update
    lambdas = np.ones(n_active) / n_active
    for _ in range(nw_iter):
        d = -np.sum(lambdas[:, None] * grads, axis=0)
        d -= np.mean(d)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm
        
        # Compute g_k^T d for all active k
        gd = grads @ d
        eta = 1.0 / np.sqrt(np.log(n_active))
        lambdas *= np.exp(eta * gd)
        lambdas /= lambdas.sum()
        # Clip to avoid underflow
        lambdas = np.clip(lambdas, 1e-15, None)
        lambdas /= lambdas.sum()
    
    return d, c_max * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def weighted_pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
                     n_cand_start=20, n_cand_end=60, max_no_improve=300,
                     weight_beta=5.0, use_anneal=False, T0=0.0003,
                     seed_val=12345):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed_val)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        T = T0 * 0.01 ** progress if use_anneal else 0.0

        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        grad = np.zeros(n)
        top_k_shifts = min(40, n_corr)
        top_idx = np.argsort(weights)[-top_k_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_i = -1
        best_j = -1
        best_delta = 0.0

                delta_hi = min(h[i], 1.0 - h[j])

                    continue

                b_d = delta_hi

                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break
                elif T > 1e-12 and approx_max - c5_val < T * 3:
                    if rng.random() < np.exp(-(approx_max - c5_val) / T):
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
                        break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def mwu_subgradient_direction(h, n, tol_frac=0.005, nw_iter=80):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    active_mask = corr >= c_max - tol_frac * c_max
    active_shifts = np.where(active_mask)[0]
    n_active = len(active_shifts)

    if n_active == 0:
        return np.zeros(n), c_max * dx

    grads = np.zeros((n_active, n))
    for ai, s_idx in enumerate(active_shifts):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grads[ai, v1] += (1.0 - h[idxs[v1] - k])
        grads[ai, v2] -= h[idxs[v2] + k]

    for ai in range(n_active):
        grads[ai] -= np.mean(grads[ai])

    lambdas = np.ones(n_active) / n_active
    d = np.zeros(n)
    for _ in range(nw_iter):
        d = -np.sum(lambdas[:, None] * grads, axis=0)
        d -= np.mean(d)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        gd = grads @ d
        eta = 1.0 / np.sqrt(np.log(max(n_active, 2)))
        lambdas *= np.exp(eta * gd)
        lam_sum = lambdas.sum()
        if lam_sum < 1e-300:
            lambdas = np.ones(n_active) / n_active
        else:
            lambdas /= lam_sum
        lambdas = np.clip(lambdas, 1e-15, None)
        lambdas /= lambdas.sum()

    return d, c_max * dx


def subgradient_descent(h_init, n, target, time_limit, tol_frac=0.005,
                        nw_iter=60, max_iter=8000, step_init=0.008):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 300:
            break

        d, c5_val = mwu_subgradient_direction(h, n, tol_frac, nw_iter)
        if np.linalg.norm(d) < 1e-12:
            break

        best_score = c5_val
        best_delta = 0.0
        for delta in [step, step * 0.5, step * 2.0, step * 0.25, step * 4.0]:
            h_try = project_feasible(h + delta * d, target)
            sc = compute_c5(h_try)
            if sc < best_score - 1e-15:
                best_score = sc
                best_delta = delta

        if best_delta > 0:
            h = project_feasible(h + best_delta * d, target)
            if best_score < best_c5 - 1e-14:
                best_c5 = best_score
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            step = min(step * 1.03, 0.05)
        else:
            step *= 0.7
            no_improve += 1

    return best_h, best_c5


def structured_perturb(h_base, n, seed_val, noise_scale=0.03):
    rng = np.random.RandomState(seed_val)
    noise = np.zeros(n)
    n_modes = rng.randint(3, 8)
    for _ in range(n_modes):
        freq = rng.choice(range(1, min(100, n // 4)), size=1)[0]
        phase = rng.uniform(0, 2 * np.pi)
        amp = noise_scale * rng.uniform(0.3, 1.0)
        noise += amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    h_pert = h_base + noise
    return project_feasible(h_pert, n / 2.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def plateau_walker(h_init, n, target, time_limit, n_steps=200):

        overlap = src_mask & dst_mask
        src_mask[overlap] = False

        h_fin, sc_fin = slsqp_round(best_h, n_cur, target_cur,
                                     [50000, 100000], 300, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
```
