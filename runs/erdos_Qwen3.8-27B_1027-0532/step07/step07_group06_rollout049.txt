from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_max(corr, alpha):
    """Smooth approximation of max(corr) using log-sum-exp"""
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def gradient_c5_smooth(h, alpha):
    """Gradient of the smooth max of correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted) / alpha  # proportional to softmax(corr * alpha)
    w_sum = np.sum(weights)
    weights = weights / w_sum  # normalized weights
    
    # Now compute weighted sum of gradients of each corr[k]
    grad = np.zeros(n)
    mid = n - 1
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-15:
            continue
        k = k_idx - mid
        if k >= 0:
            # corr[k_idx] = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
            # d/dh[j] = (1-h[j+k]) for j+k <= n-1, i.e., j <= n-1-k
            #           -h[j-k] for j-k >= 0, i.e., j >= k
            # Combined: for j in [0, n-1-k]: (1-h[j+k])
            #            for j in [k, n-1]: -h[j-k]
            v1 = (np.arange(n) <= n - 1 - k)
            v2 = (np.arange(n) >= k)
            grad[v1] += w * (1.0 - h[v1 + k])
            grad[v2] -= w * h[v2 - k]
        else:
            kk = -k
            # corr[k_idx] = sum_{i=0}^{n-1-kk} h[i+kk] * (1-h[i])
            # d/dh[j] = (1-h[j-kk]) for j >= kk, j-kk <= n-1-kk, i.e., j in [kk, n-1]
            #           -h[j+kk] for j+kk <= n-1-kk, i.e., j <= n-1-kk
            v1 = (np.arange(n) >= kk)
            v2 = (np.arange(n) <= n - 1 - kk)
            grad[v1] += w * (1.0 - h[v1 - kk])
            grad[v2] -= w * h[v2 + kk]
    
    return grad * dx

def fw_step(grad, n, target):
    """Frank-Wolfe linear minimization oracle
    Minimize grad · z subject to 0 <= z <= 1, sum(z) = target
    Solution: set z[i]=1 for the floor(target) indices with smallest grad,
    z[i]=0 for the rest, and one fractional value if target is not integer.
    """
    sorted_idx = np.argsort(grad)
    z = np.zeros(n)
    # target should be n/2 which is integer for even n
    k = int(round(target))
    z[sorted_idx[:k]] = 1.0
    return z

def frank_wolfe(h_init, n, target, alphas, max_iters, time_limit, start_time):
    """Frank-Wolfe with Nesterov acceleration and adaptive alpha"""
    h = h_init.copy().astype(float)
    h_ym = h.copy()  # for Nesterov
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 0
    
    for alpha in alphas:
        for it in range(max_iters):
            if time.time() - start_time > time_limit:
                    break
            t += 1
            
            # Nesterov momentum
            beta = 2.0 / (t + 2.0)
            h_ym = h + (beta - 1.0) / beta * (h - h_ym) if t > 1 else h.copy()
            h_ym = np.clip(h_ym, 0.0, 1.0)
            
            # Compute gradient at y_m
            corr = np.correlate(h_ym, 1.0 - h_ym, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            weights = np.exp(shifted)
            w_sum = np.sum(weights)
            weights = weights / w_sum
            
            dx = 2.0 / n
            grad = np.zeros(n)
            mid = n - 1
            idxs = np.arange(n)
            for k_idx in range(len(corr)):
                w = weights[k_idx]
                if w < 1e-15:

                k = k_idx - mid
                if k >= 0:
                    v1 = idxs <= n - 1 - k
                    v2 = idxs >= k
                    grad[v1] += w * (1.0 - h_ym[v1 + k])
                    grad[v2] -= w * h_ym[v2 - k]
                else:

                    v1 = idxs >= kk
                    v2 = idxs <= n - 1 - kk
                    grad[v1] += w * (1.0 - h_ym[v1 - kk])
                    grad[v2] -= w * h_ym[v2 + kk]
            grad *= dx
            
            # FW step
            z = fw_step(grad, n, target)
            
            # Update with step size
            gamma = 2.0 / (t + 2.0)
            h_new = (1.0 - gamma) * h + gamma * z
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            
            # Check improvement

            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            h = h_new
        
        # Interpolation step (for better convergence)
        # Skip for now, basic FW
    
    return best_h, best_c5

import time

    n = len(h)

        else:
            mask = h > 1e-14

        h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    weights = weights / w_sum
    val = c_max + np.log(w_sum) / alpha

    grad = np.zeros(n)

            v2 = idxs <= n - 1 - kk
            grad[v1] += w * (1.0 - h[v1 - kk])
            grad[v2] -= w * h[v2 + kk]

    z = np.zeros(n)
    k = int(round(target))
    z[sorted_idx[:k]] = 1.0

def frank_wolfe_nesterov(h_init, n, target, alphas, iters_per_alpha, time_limit, start_time):

    h = h_init.copy().astype(float)
    y = h.copy()  # Nesterov y
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_global = 0
    
    for alpha in alphas:
        t = 0
        for it in range(iters_per_alpha):
            if time.time() - start_time > time_limit:
                break
            t += 1
            t_global += 1
            
            # Nesterov momentum
            if t == 1:
                y = h.copy()
            else:
                beta_t = 2.0 / (t + 2.0)
                beta_prev = 2.0 / (t + 1.0)
                y = h + (beta_prev - 1.0) / beta_t * (h - y)
                y = np.clip(y, 0.0, 1.0)

            grad = gradient_smooth_corrs(y, weights)

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def _slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target, _g_alpha

    _g_target = target

    cur_score = compute_c5(h_cur)

        if time.time() - start_time > time_limit:
            break

        if use_pnorm:
            func = _pnorm_obj
        else:

            sc = compute_c5(h_new)

    grad = np.zeros(n)
    for k_idx in range(len(weights)):
        w = weights[k_idx]
        if w < 1e-15:
            continue
        k = k_idx - mid
        if k >= 0:
            v1 = idxs <= n - 1 - k
            v2 = idxs >= k
            grad[v1] += w * (1.0 - h[v1 + k])
            grad[v2] -= w * h[v2 - k]
        else:
            kk = -k
            v1 = idxs >= kk
            v2 = idxs <= n - 1 - kk
            grad[v1] += w * (1.0 - h[v1 - kk])
            grad[v2] -= w * h[v2 + kk]
    return grad * dx


def fw_solve(grad, n, target):
    sorted_idx = np.argsort(grad)
    z = np.zeros(n)
    k = int(round(target))
    z[sorted_idx[:k]] = 1.0
    return z


def frank_wolfe_nesterov(h_init, n, target, alphas, iters_per_alpha, time_limit, start_time):
    h = h_init.copy().astype(float)
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for alpha in alphas:
        t = 0
        for it in range(iters_per_alpha):
            if time.time() - start_time > time_limit:
                break
            t += 1
            if t == 1:
                y = h.copy()
            else:
                beta_t = 2.0 / (t + 2.0)
                beta_prev = 2.0 / (t + 1.0)
                y = h + (beta_prev - 1.0) / beta_t * (h - y)
                y = np.clip(y, 0.0, 1.0)

            corr = np.correlate(y, 1.0 - y, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            weights = np.exp(shifted)
            w_sum = np.sum(weights)
            weights = weights / w_sum

            grad = gradient_smooth_corrs(y, weights)
            z = fw_solve(grad, n, target)

            gamma = 2.0 / (t + 2.0)
            h_new = (1.0 - gamma) * h + gamma * z
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, target)

            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new

        c5_check = compute_c5(h)
        if c5_check < best_c5 - 1e-14:
            best_c5 = c5_check
            best_h = h.copy()

    return best_h, best_c5


def accelerated_grad(h_init, n, target, alphas, iters_per_alpha, time_limit, start_time, lr=0.02, momentum=0.85):
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for alpha in alphas:
        for it in range(iters_per_alpha):
            if time.time() - start_time > time_limit:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            weights = np.exp(shifted)
            w_sum = np.sum(weights)
            weights = weights / w_sum

            grad = gradient_smooth_corrs(h, weights)
            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project_feasible(h_new, target)

            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new

    return best_h, best_c5


def pair_cd_sa(h_init, n, target, time_limit, start_time,
               max_sweeps=30000, n_cand_start=15, n_cand_end=60,
               max_no_improve=400, n_top_shifts=1, bisection_iters=16,
               T_initial=0.002, T_min=0.0):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(99999)

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        temperature = T_initial * max(T_min / max(T_initial, 1e-30), (1.0 - float(sweep) / max_sweeps) ** 2)

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_shifts].astype(float)
            wsum = np.sum(weights)
            if wsum < 1e-15:
                weights = np.ones(n_top_shifts) / n_top_shifts
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            if best_trial_c5 > c5_val and temperature > 1e-8:
                delta_obj = best_trial_c5 - c5_val
                accept_prob = np.exp(-delta_obj / temperature)
                if rng.random() < accept_prob:
                    h[best_i] -= best_delta
                    h[best_j] += best_delta
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    for p in [best_i, best_j]:
                        change[p, :] = 0.0
                        pos_mask = (p + shifts >= 0) & (p + shifts < n)
                        neg_mask = (p - shifts >= 0) & (p - shifts < n)
                        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                        change[p, neg_mask] -= h[p - shifts[neg_mask]]
            else:
                h[best_i] -= best_delta
                h[best_j] += best_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                for p in [best_i, best_j]:
                    change[p, :] = 0.0
                    pos_mask = (p + shifts >= 0) & (p + shifts < n)
                    neg_mask = (p - shifts >= 0) & (p - shifts < n)
                    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                    change[p, neg_mask] -= h[p - shifts[neg_mask]]
        elif temperature > 1e-8:
            nrand = rng.randint(2, min(5, n))
            perm = rng.permutation(n)
            idxs_r = perm[:nrand]
            h_try = h.copy()
            h_try[idxs_r] += rng.randn(nrand) * 0.02
            h_try = project_feasible(h_try, target)
            sc_try = compute_c5(h_try)
            delta_obj = sc_try - c5_val
            if delta_obj < 0 or rng.random() < np.exp(-delta_obj / temperature):
                h = h_try
                for p in idxs_r:
                    change[p, :] = 0.0
                    pos_mask = (p + shifts >= 0) & (p + shifts < n)
                    neg_mask = (p - shifts >= 0) & (p - shifts < n)
                    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                    change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5


def block_move_cd(h_init, n, target, time_limit, start_time, max_sweeps=5000, block_size=4):
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_sweeps // 4:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        window = block_size
        n_starts = max(1, (n - window) // 200)
        starts = np.linspace(0, n - window, n_starts, dtype=int)

        best_block_score = np.inf
        best_block_start = 0
        for start in starts:
            block_grad = grad[start:start + window]
            score = np.sum(block_grad ** 2)
            if score < best_block_score:
                best_block_score = score
                best_block_start = int(start)

        block_idx = np.arange(best_block_start, best_block_start + window)
        h_fixed = h.copy()
        h_block = h[block_idx].copy()
        rest_sum = np.sum(h) - np.sum(h_block)

        def block_obj(xb):
            h_trial = h_fixed.copy()
            h_trial[block_idx] = np.clip(xb, 0.0, 1.0)
            needed = target - rest_sum
            cur = np.sum(np.clip(xb, 0.0, 1.0))
            if abs(needed - cur) > 1e-10:
                h_trial = project_feasible(h_trial, target)
            c = np.correlate(h_trial, 1.0 - h_trial, mode='full')
            cm = np.max(c)
            if cm <= 0:
                return 0.0
            shifted = 800.0 * (c - cm)
            return float(cm + np.log(np.sum(np.exp(np.clip(shifted, -500, 0)))) / 800.0)

        bounds_block = [(0.0, 1.0)] * window
        try:
            res = minimize(block_obj, h_block, method='Nelder-Mead',
                           bounds=bounds_block,
                           options={'maxiter': 1500, 'xatol': 1e-12, 'fatol': 1e-14})
            h_new_block = np.clip(res.x, 0.0, 1.0)
            h_trial = h.copy()
            h_trial[block_idx] = h_new_block
            h_trial = project_feasible(h_trial, target)
            sc = compute_c5(h_trial)
            if sc < c5_val - 1e-14:
                h = h_trial
        except Exception:
            pass

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_refine(h_init, n, target, time_limit, start_time, max_steps=20000, base_scale=0.03):
    rng = np.random.RandomState(777)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 3000:
            break

        scale = base_scale * max(0.05, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 5]))
        n_pert = min(n_pert, n)
        perm = rng.permutation(n)
        idxs = perm[:n_pert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 100:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Frank-Wolfe global exploration on coarse grid
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 80:
        n_coarse = 160
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        t1 = time.time() + min(50, remaining - 20)
        h_fw, sc_fw = frank_wolfe_nesterov(h_coarse, n_coarse, target_coarse,
                                           [100, 500, 2000], 400, t1, start_time)
        h_back = interpolate_to(h_fw, n0, target0)
        sc_back = compute_c5(h_back)
        if sc_back < best_score:
            best_score = sc_back
            best_h = h_back

        # Also try accelerated gradient from perturbed start
        rng = np.random.RandomState(seed)
        h_pert = h_coarse + rng.randn(n_coarse) * 0.08
        h_pert = project_feasible(h_pert, target_coarse)
        t1b = time.time() + min(30, remaining - 40)
        h_ag, sc_ag = accelerated_grad(h_pert, n_coarse, target_coarse,
                                       [200, 1000], 400, t1b, start_time, lr=0.03, momentum=0.85)
        h_back2 = interpolate_to(h_ag, n0, target0)
        sc_back2 = compute_c5(h_back2)
        if sc_back2 < best_score:
            best_score = sc_back2
            best_h = h_back2

    # Phase 2: Pair CD with simulated annealing at n0
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 150:
        t2 = time.time() + min(130, remaining - 20)
        h_cd, sc_cd = pair_cd_sa(best_h, n0, target0, t2, start_time,
                                  max_sweeps=50000, n_cand_start=12, n_cand_end=55,
                                  max_no_improve=350, n_top_shifts=3, bisection_iters=16,
                                  T_initial=0.001, T_min=0.0)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 3: SLSQP refinement
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 60:
        t3 = time.time() + min(50, remaining - 15)
        h_sl, sc_sl = _slsqp_round(best_h, n0, target0,
                                    [500, 2000, 8000], 1200, t3, start_time, use_pnorm=False)
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    # Phase 4: Higher resolution n=550
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 200:
        n1 = 550
        target1 = n1 / 2.0
        h_int = interpolate_to(best_h, n1, target1)
        t4 = time.time() + min(150, remaining - 20)
        h_hr, sc_hr = pair_cd_sa(h_int, n1, target1, t4, start_time,
                                  max_sweeps=35000, n_cand_start=10, n_cand_end=45,
                                  max_no_improve=300, n_top_shifts=3, bisection_iters=16,
                                  T_initial=0.0008, T_min=0.0)
        if sc_hr < best_score:
            best_score = sc_hr
            best_h = h_hr
            n0 = n1
            target0 = target1

    # Phase 5: SLSQP at higher resolution
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 60:
        t5 = time.time() + min(50, remaining - 15)
        h_sl2, sc_sl2 = _slsqp_round(best_h, n0, target0,
                                      [2000, 10000], 1500, t5, start_time, use_pnorm=False)
        if sc_sl2 < best_score:
            best_score = sc_sl2
            best_h = h_sl2

    # Phase 6: Block moves for local improvement
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 80:
        t6 = time.time() + min(60, remaining - 15)
        h_bl, sc_bl = block_move_cd(best_h, n0, target0, t6, start_time,
                                     max_sweeps=3000, block_size=4)
        if sc_bl < best_score:
            best_score = sc_bl
            best_h = h_bl

    # Phase 7: More pair CD at current resolution
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 120:
        t7 = time.time() + min(100, remaining - 20)
        h_cd2, sc_cd2 = pair_cd_sa(best_h, n0, target0, t7, start_time,
                                    max_sweeps=30000, n_cand_start=20, n_cand_end=60,
                                    max_no_improve=250, n_top_shifts=5, bisection_iters=18,
                                    T_initial=0.0005, T_min=0.0)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 8: Try even higher resolution
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 150:
        n2 = 700
        target2 = n2 / 2.0
        h_int2 = interpolate_to(best_h, n2, target2)
        t8 = time.time() + min(120, remaining - 20)
        h_hr2, sc_hr2 = pair_cd_sa(h_int2, n2, target2, t8, start_time,
                                    max_sweeps=25000, n_cand_start=10, n_cand_end=35,
                                    max_no_improve=250, n_top_shifts=3, bisection_iters=16,
                                    T_initial=0.0005, T_min=0.0)
        if sc_hr2 < best_score:
            best_score = sc_hr2
            best_h = h_hr2
            n0 = n2
            target0 = target2

    # Phase 9: Final SLSQP
    remaining = budget_s - 25.0 - (time.time() - start_time)
    if remaining > 50:
        t9 = time.time() + min(40, remaining - 10)
        h_fin, sc_fin = _slsqp_round(best_h, n0, target0,
                                      [5000, 20000, 50000], 1000, t9, start_time, use_pnorm=False)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 10: Final pair CD polish
    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 40:
        t10 = time.time() + min(30, remaining - 10)
        h_fin2, sc_fin2 = pair_cd_sa(best_h, n0, target0, t10, start_time,
                                      max_sweeps=15000, n_cand_start=30, n_cand_end=70,
                                      max_no_improve=150, n_top_shifts=1, bisection_iters=20,
                                      T_initial=0.0, T_min=0.0)
        if sc_fin2 < best_score:
            best_score = sc_fin2
            best_h = h_fin2

    # Multi-start perturbation if time remains
    remaining = budget_s - 15.0 - (time.time() - start_time)
    if remaining > 60:
        rng2 = np.random.RandomState(seed + 1)
        for noise_scale in [0.01, 0.03]:
            remaining = budget_s - 15.0 - (time.time() - start_time)
            if remaining < 40:
                break
            h_pert2 = best_h + rng2.randn(n0) * noise_scale
            h_pert2 = project_feasible(h_pert2, target0)
            t_p = time.time() + min(30, remaining - 10)
            h_r, sc_r = pair_cd_sa(h_pert2, n0, target0, t_p, start_time,
                                    max_sweeps=10000, n_cand_start=15, n_cand_end=50,
                                    max_no_improve=200, n_top_shifts=3, bisection_iters=16,
                                    T_initial=0.0005, T_min=0.0)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
