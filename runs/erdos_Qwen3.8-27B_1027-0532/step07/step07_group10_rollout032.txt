**New idea: Use the KKT conditions to identify the optimal structure**

At the optimum, we have:

- For any feasible direction d with ∑d = 0, we need ∑_{k in K*} dC(k)/dh * d ≥ 0

Wait, that's not quite right. The KKT condition is: there exist multipliers λ_k ≥ 0 for k in K* such that:
∑_{k in K*} λ_k * g_k + μ * 1 = 0 (where 1 is the all-ones vector, from the equality constraint)

And complementary slackness: λ_k * (C* - C(k)) = 0 for all k.

This means: ∑_{k in K*} λ_k * g_k = -μ * 1

At the optimum, the weighted average of the active gradients is a constant vector (which can be made zero by the equality constraint).

This gives us a way to check if we're at a local optimum: compute the active set, then check if there's a non-negative combination of the active gradients that equals a constant vector (plus the bound constraints).

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = (idxs + k >= 0) & (idxs + k < n)
        wm = (idxs - k >= 0) & (idxs - k < n)
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def nesterov_optimize(h_init, n, target, time_limit, beta=200.0,
                      base_lr=0.05, momentum_init=0.5, max_steps=200000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = momentum_init
    no_improve = 0
    lr = base_lr

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 20000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.05, base_lr)
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.95
            if no_improve > 3000:
                lr *= 0.9
            if lr < 1e-7:
                break

        c_max = np.max(corr)
        shifted = np.clip(beta * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            weights = np.zeros(n_corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= wsum

        grad = np.zeros(n)
        for k_idx in range(n_corr):
            wi = weights[k_idx]
            if wi < 1e-10:
                continue
            k = k_idx - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] + k])
            grad[v2] -= wi * h[idxs[v2] - k]
        grad *= dx
        grad -= np.mean(grad)

        v = momentum * v - lr * grad
        h_try = h + v
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < c5_val + 1e-15:
            h = h_try
            momentum = min(0.999, momentum * 1.001)
        else:
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            sc2 = compute_c5(h_try)
            if sc2 < c5_val + 1e-15:
                h = h_try
            v *= 0.5
            lr *= 0.7
            momentum = min(momentum, 0.999)

    return best_h, best_c5


def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=20, n_cand_end=80, max_no_improve=500,
                 batch_size=1):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        for batch in range(batch_size):
            best_trial_c5 = c5_val
            best_i = -1
            best_j = -1
            best_delta = 0.0

            n_h = len(high_set)
            n_l = len(low_set)
            for ii in range(n_h):
                i = int(high_set[ii])
                for jj in range(n_l):
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo < 1e-13:
                        continue

                    g = change[j] - change[i]

                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(18):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-14
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d

                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-13:
                        continue

                    approx_max = np.max(corr + delta_star * g)
                    if approx_max < best_trial_c5 / dx - 1e-14:
                        best_trial_c5 = float(approx_max) * dx
                        best_i = i
                        best_j = j
                        best_delta = delta_star

            if best_i >= 0:
                h[best_i] -= best_delta
                h[best_j] += best_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                corr += best_delta * (change[best_j] - change[best_i])
                update_change_row(change, best_i, h, n, shifts)
                update_change_row(change, best_j, h, n, shifts)
                c5_val = float(np.max(corr)) * dx
            else:
                break

    return best_h, best_c5


def multi_active_cd(h_init, n, target, time_limit, n_cand=50, n_active=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(20000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 300:
            break

        top_shifts = np.argsort(corr)[-n_active:]
        weights = np.zeros(n_corr)
        c_max = corr[top_shifts[0]]
        for si in top_shifts:
            weights[si] = np.exp(-30.0 * (c_max - corr[si]))
        sw = np.sum(weights)
        weights /= sw

        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            w = weights[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr += best_delta * (change[best_j] - change[best_i])
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def thue_morse_init(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = 1.0 - t.astype(float)
    target = n / 2.0
    return project_feasible(h, target)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    if tleft() > 200:
        h_nm, sc_nm = nesterov_optimize(best_h, n0, target0,
                                        time_limit=min(120, tleft() * 0.6),
                                        beta=200.0, base_lr=0.05,
                                        momentum_init=0.5)
        if sc_nm < best_score:
            best_score = sc_nm
            best_h = h_nm

    if tleft() > 150:
        h_cd, sc_cd = fast_pair_cd(best_h, n0, target0,
                                   time_limit=min(150, tleft() * 0.8),
                                   max_sweeps=30000,
                                   n_cand_start=25, n_cand_end=80,
                                   max_no_improve=400)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 80:
        h_s, sc_s = slsqp_round(best_h, n0, best_target,
                                [200, 1000, 5000], 1500, 1e-15)
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    if tleft() > 100:
        h_cd2, sc_cd2 = fast_pair_cd(best_h, n0, target0,
                                     time_limit=min(100, tleft() * 0.7),
                                     max_sweeps=20000,
                                     n_cand_start=40, n_cand_end=100,
                                     max_no_improve=300)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    n1 = 600
    if tleft() > 350:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 250:
            h1_nm, sc1_nm = nesterov_optimize(h1, n1, target1,
                                              time_limit=min(100, tleft() * 0.4),
                                              beta=300.0, base_lr=0.04,
                                              momentum_init=0.5)
            if sc1_nm < best_score:
                best_score = sc1_nm
                best_h = h1_nm
                best_n = n1
                best_target = target1
            else:
                h1 = h1_nm if sc1_nm < compute_c5(h1) else h1

        if tleft() > 200:
            h1_cd, sc1_cd = fast_pair_cd(best_h, best_n, best_target,
                                         time_limit=min(200, tleft() * 0.6),
                                         max_sweeps=25000,
                                         n_cand_start=30, n_cand_end=90,
                                         max_no_improve=400)
            if sc1_cd < best_score:
                best_score = sc1_cd
                best_h = h1_cd
                best_n = best_n
                best_target = best_target

        if tleft() > 100:
            h1_s, sc1_s = slsqp_round(best_h, best_n, best_target,
                                      [500, 3000, 10000], 1500, 1e-15)
            if sc1_s < best_score:
                best_score = sc1_s
                best_h = h1_s

        if tleft() > 80:
            h1_mc, sc1_mc = multi_active_cd(best_h, best_n, best_target,
                                            time_limit=min(60, tleft() * 0.5),
                                            n_cand=60, n_active=6)
            if sc1_mc < best_score:
                best_score = sc1_mc
                best_h = h1_mc

    if tleft() > 120:
        h_cd3, sc_cd3 = fast_pair_cd(best_h, best_n, best_target,
                                     time_limit=min(120, tleft() * 0.7),
                                     max_sweeps=20000,
                                     n_cand_start=50, n_cand_end=120,
                                     max_no_improve=250)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    if tleft() > 80:
        h_s2, sc_s2 = slsqp_round(best_h, best_n, best_target,
                                  [10000, 50000], 2000, 1e-15)
        if sc_s2 < best_score:
            best_score = sc_s2
            best_h = h_s2

    if tleft() > 60:
        h_nm2, sc_nm2 = nesterov_optimize(best_h, best_n, best_target,
                                          time_limit=min(50, tleft() * 0.6),
                                          beta=500.0, base_lr=0.02,
                                          momentum_init=0.7)
        if sc_nm2 < best_score:
            best_score = sc_nm2
            best_h = h_nm2

    if tleft() > 100:
        h_cd4, sc_cd4 = fast_pair_cd(best_h, best_n, best_target,
                                     time_limit=min(80, tleft() * 0.6),
                                     max_sweeps=15000,
                                     n_cand_start=60, n_cand_end=130,
                                     max_no_improve=200)
        if sc_cd4 < best_score:
            best_score = sc_cd4
            best_h = h_cd4

    if tleft() > 200:
        for noise_scale in [0.01, 0.03, 0.05, 0.08, 0.12]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            h_ref, sc = nesterov_optimize(h_pert, best_n, best_target,
                                          time_limit=min(30, tleft() * 0.3),
                                          beta=300.0, base_lr=0.03,
                                          momentum_init=0.5)
            if tleft() > 30:
                h_ref2, sc2 = fast_pair_cd(h_ref, best_n, best_target,
                                           time_limit=min(20, tleft() * 0.3),
                                           max_sweeps=5000,
                                           n_cand_start=50, n_cand_end=100,
                                           max_no_improve=200)
                if sc2 < sc:
                    h_ref = h_ref2
                    sc = sc2
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 30:
        h_s3, sc_s3 = slsqp_round(best_h, best_n, best_target,
                                  [50000, 100000], 1000, 1e-15)
        if sc_s3 < best_score:
            best_score = sc_s3
            best_h = h_s3

    if tleft() > 20:
        h_cd_final, sc_final = fast_pair_cd(best_h, best_n, best_target,
                                            time_limit=min(15, tleft() * 0.6),
                                            max_sweeps=5000,
                                            n_cand_start=80, n_cand_end=150,
                                            max_no_improve=100)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_cd_final

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
