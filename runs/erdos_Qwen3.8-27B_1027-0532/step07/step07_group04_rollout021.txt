    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -50.0, 0.0)

            break

            gamma *= 0.95  # decrease step size
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find active shifts (within epsilon of max)
        c_max = np.max(corr)
        active = np.where(corr > c_max - 1e-10)[0]
        
        # Compute subgradient (average of active gradients)
        grad = np.zeros(n)
        for m in active:
            k = int(m) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)

            grad[v2] -= h[idxs[v2] + k]

        grad -= np.mean(grad)  # project onto sum constraint
        
        # Take step
        h = h - gamma * grad
        h = project_feasible(h, target)
    
    return best_h, best_c5

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    _g_target = target

        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def multi_shift_cd(h_init, n, target, time_limit, max_sweeps=15000,
                   n_cand=60, n_top=4):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 150:
            break
        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-n_top:]
        tens = np.exp(-4.0 * (c_max - corr[top_idx]) / max(c_max * dx, 1e-10))
        tens = tens / tens.sum()
        grad = np.zeros(n)
        for wi, s_idx in enumerate(top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += tens[wi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= tens[wi] * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def sa_refine(h_init, n, target, time_limit, max_steps=20000,
              t_start=0.03, t_end=0.0003, n_cand=50):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(123)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 400:
            break
        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue
        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue
        g = change[j] - change[i]
        approx_new_c5 = float(np.max(corr + delta * g) * dx)
        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-delta_c / max(temperature * dx, 1e-15)):
                accept = True
        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            cur_c5 = approx_new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP on initial grid
    if tleft() > 50:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [300, 2000, 8000], 600, 1e-16,
                                   time_limit=min(40, tleft()*0.7),
                                   start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Interpolate to n=800, subgradient + SLSQP
    if tleft() > 250:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 200:
            h_sg1, sc_sg1 = subgradient_method(h1, n1, target1,
                                               time_limit=min(40, tleft()*0.3),
                                               n_steps=30000, gamma0=0.04)
            if sc_sg1 < best_score:
                best_score = sc_sg1
                best_h = h_sg1
        if tleft() > 120:
            h_pol1, sc1 = slsqp_optimize(best_h, n1, target1,
                                         [2000, 10000, 40000],
                                         1000, 1e-16,
                                         time_limit=min(90, tleft()*0.6),
                                         start_time=start_time)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1
        if tleft() > 60:
            h_cd1, c5_cd1 = fast_cd(best_h, n1, target1,
                                    time_limit=min(40, tleft()*0.5),
                                    max_sweeps=15000, n_cand=70,
                                    max_no_improve=150)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

    # Phase 3: Interpolate to n=1200
    if tleft() > 400:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 300:
            h_sg2, sc_sg2 = subgradient_method(h2, n2, target2,
                                               time_limit=min(50, tleft()*0.25),
                                               n_steps=40000, gamma0=0.03)
            if sc_sg2 < best_score:
                best_score = sc_sg2
                best_h = h_sg2
        if tleft() > 200:
            h_pol2, sc2 = slsqp_optimize(best_h, n2, target2,
                                         [5000, 20000, 80000],
                                         1200, 1e-16,
                                         time_limit=min(150, tleft()*0.5),
                                         start_time=start_time)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2
        if tleft() > 60:
            h_ms2, c5_ms2 = multi_shift_cd(best_h, n2, target2,
                                           time_limit=min(40, tleft()*0.4),
                                           max_sweeps=10000, n_cand=60, n_top=5)
            if c5_ms2 < best_score:
                best_score = c5_ms2
                best_h = h_ms2

    # Phase 4: Try n=1600 if time allows
    if tleft() > 500:
        n3 = 1600
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        if tleft() > 400:
            h_sg3, sc_sg3 = subgradient_method(h3, n3, target3,
                                               time_limit=min(60, tleft()*0.2),
                                               n_steps=30000, gamma0=0.02)
            if sc_sg3 < best_score:
                best_score = sc_sg3
                best_h = h_sg3
        if tleft() > 250:
            h_pol3, sc3 = slsqp_optimize(best_h, n3, target3,
                                         [10000, 50000],
                                         1000, 1e-16,
                                         time_limit=min(200, tleft()*0.45),
                                         start_time=start_time)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3
        if tleft() > 60:
            h_cd3, c5_cd3 = fast_cd(best_h, n3, target3,
                                    time_limit=min(40, tleft()*0.3),
                                    max_sweeps=8000, n_cand=60,
                                    max_no_improve=100)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

    # Phase 5: SA escape on current best
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sa, c5_sa = sa_refine(best_h, n_cur, target_cur,
                                time_limit=min(50, tleft()*0.5),
                                max_steps=15000, t_start=0.02,
                                t_end=0.0002, n_cand=50)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 6: Final SLSQP polish
    if tleft() > 50:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fp, sc_fp = slsqp_optimize(best_h, n_cur, target_cur,
                                     [20000, 100000, 400000],
                                     800, 1e-16,
                                     time_limit=min(40, tleft()*0.6),
                                     start_time=start_time)
        if sc_fp < best_score:
            best_score = sc_fp
            best_h = h_fp

    # Phase 7: Final CD
    if tleft() > 25:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cf, c5_cf = fast_cd(best_h, n_cur, target_cur,
                              time_limit=min(15, tleft()*0.7),
                              max_sweeps=8000, n_cand=80,
                              max_no_improve=80)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
