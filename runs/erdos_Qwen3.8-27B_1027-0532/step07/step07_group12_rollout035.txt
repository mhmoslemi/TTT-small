def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def hessian_free_step(h, corr, weights, idxs, mid, n, step_size):
    """Take a step in the direction of the negative gradient from weighted active constraints."""
    grad = np.zeros(n)
    active_mask = weights > 1e-10

        if np.any(v2):
            grad[v2] -= w * h[idxs[v2] + k]

def lbfgs_optimize(h_start, n, target, alpha, maxiter, time_limit, penalty=100.0):

def topk_smoothed_max(corr, k, alpha):
    """Compute a smoothed approximation of the max using top-k values."""
    top_k = np.sort(corr)[-k:]
    shifted = np.clip(alpha * (top_k - corr.max()), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return corr.max() + log_sum / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def force_binary(h, target, threshold):
    h = h.copy()
    n = len(h)
    h[h < threshold] = 0.0
    h[h > 1.0 - threshold] = 1.0
    diff = target - np.sum(h)
    free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
    if len(free) == 0:
        if diff > 1e-10:
            zeros = np.where(h == 0)[0]
            if len(zeros) > 0:
                n_flip = min(int(np.ceil(diff)), len(zeros))
                h[zeros[:n_flip]] = 1.0
        elif diff < -1e-10:
            ones = np.where(h == 1)[0]
            if len(ones) > 0:
                n_flip = min(int(np.ceil(-diff)), len(ones))
                h[ones[:n_flip]] = 0.0
        residual = target - np.sum(h)
        if abs(residual) > 1e-10:
            h[:] += residual / n
            h = np.clip(h, 0.0, 1.0)
    else:
        h[free] += diff / len(free)
        h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        free2 = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if len(free2) > 0:
            h[free2] += residual / len(free2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_optimize(h_start, n, target, alpha, maxiter, penalty=100.0):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_objective, h_start, args=(alpha, target, penalty),
                      jac=lse_gradient, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12,
                               'maxls': 40, 'maxfun': 8000})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start

def pair_cd_vectorized(h_init, n, target, time_limit, n_cand=80, n_shifts=1,
                       force_thresh=0.0, force_interval=200, fractions=(0.3, 0.5, 0.7, 0.9, 1.0)):

    h = h_init.copy()

        sweep += 1

            if np.any(v2):

        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            aw = 50.0
            w = np.exp(aw * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid

                if np.any(v1):

        sorted_idx = np.argsort(-grad)

            if np.any(valid1):

            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        nh = len(high_set)
        nl = len(low_set)
        high_profiles = np.array([profiles[cand_pos[int(i)]] for i in high_set])
        low_profiles = np.array([profiles[cand_pos[int(j)]] for j in low_set])

        diffs = high_profiles[:, None, :] - low_profiles[None, :, :]

        num = cur_max - corr
        safe_diffs = np.where(diffs > 1e-10, diffs, 1e10)
        ratios = num[None, None, :] / safe_diffs
        delta_limits = np.min(ratios, axis=2)

        hi_vals = np.array([float(h[i]) for i in high_set])
        lo_vals = np.array([1.0 - float(h[j]) for j in low_set])
        box_limits = np.minimum(hi_vals[:, None], lo_vals[None, :])
        delta_limits = np.minimum(delta_limits, box_limits)
        delta_limits = np.maximum(delta_limits, 0.0)

        best_improve = 0.0
        best_fi = -1
        best_fj = -1
        best_delta = 0.0

        for frac in fractions:
            deltas = frac * delta_limits
            new_maxes = np.max(corr[None, None, :] + deltas[:, :, None] * diffs, axis=2)
            improvements = cur_max - new_maxes
            flat_idx = np.argmax(improvements)
            fi, fj = np.unravel_index(flat_idx, improvements.shape)
            imp = improvements[fi, fj]
            if imp > best_improve:
                best_improve = imp
                best_fi = fi
                best_fj = fj
                best_delta = deltas[fi, fj]

        if best_improve > 1e-14 and best_fi >= 0:
            i = int(high_set[best_fi])
            j = int(low_set[best_fj])
            h[i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0.0, 1.0)
