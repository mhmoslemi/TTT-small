Actually no. np.correlate(h, 1-h) gives the sum without dx. Then we multiply by dx to get the integral.

So let C_k = np.correlate(h, 1-h)[k] (the raw correlation sum)
And the objective is max_k C_k * dx.

For the smoothing, we optimize max_k C_k (then multiply by dx at the end).

C_k = Σ_i h[i] * (1-h[i+k]) for valid indices.

∂C_k/∂h[j] = (1-h[j+k]) if j+k is valid, plus -h[j-k] if j-k is valid.

So ∂C_k/∂h = vector where component j is:
(1-h[j+k]) * [j+k valid] - h[j-k] * [j-k valid]

Now for the smoothed objective:
S = c_max + (1/α) * log(Σ_k exp(α*(C_k - c_max)))

∂S/∂h[j] = (1/α) * Σ_k [α * ∂C_k/∂h[j] * exp(α*(C_k - c_max))] / Σ_k exp(α*(C_k - c_max))
           = Σ_k [∂C_k/∂h[j] * exp(α*(C_k - c_max))] / Σ_k exp(α*(C_k - c_max))
           = Σ_k w_k * ∂C_k/∂h[j]

where w_k = exp(α*(C_k - c_max)) / Σ_m exp(α*(C_m - c_max))

This is a weighted average of the individual gradients, with weights proportional to how "active" each shift is. This is computable!

def smoothed_obj_and_grad(x, alpha, n, target):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    mid = n - 1
    idxs = np.arange(n)
    
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    
    # Log-sum-exp for numerical stability
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    
    val = c_max + np.log(sum_exp) / alpha
    pen = 500.0 * (np.sum(x) - target) ** 2
    val += pen
    
    # Gradient
    weights = exp_shifted / sum_exp  # w_k
    grad = np.zeros(n)
    
    for s_idx in range(n_corr):
        w = weights[s_idx]
        if w < 1e-10:
            continue
        k = s_idx - mid
        # ∂C_k/∂h[j] = (1-h[j+k]) if j+k valid, minus h[j-k] if j-k valid
        v1 = idxs - k
        v2 = idxs + k
        m1 = (v1 >= 0) & (v1 < n)
        m2 = (v2 >= 0) & (v2 < n)
        grad[m1] += w * (1.0 - x[v1[m1]])
        grad[m2] -= w * x[v2[m2]]
    
    # Add penalty gradient
    grad += 1000.0 * (np.sum(x) - target)
    
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def build_change_matrix(h, n):
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    return change

def update_change_row(change, p, h, shifts):
    n = len(h)
    change[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos] += (1.0 - h[p + shifts[pos]])
    change[p, neg] -= h[p - shifts[neg]]

def _smoothed_obj(x, alpha, n, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    pen = 1000.0 * (np.sum(x) - target) ** 2
    return val + pen

def _smoothed_grad(x, alpha, n, target):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp

    n_active = min(80, n_corr)
    top_w_idx = np.argsort(weights)[-n_active:]
    grad = np.zeros(n)

    for s_idx in top_w_idx:
        w = weights[s_idx]
        if w < 1e-10:
            continue
        k = int(s_idx) - mid
        v1 = idxs - k
        v2 = idxs + k
        m1 = (v1 >= 0) & (v1 < n)
        m2 = (v2 >= 0) & (v2 < n)
        grad[m1] += w * (1.0 - x[v1[m1]])
        grad[m2] -= w * x[v2[m2]]

    grad += 2000.0 * (np.sum(x) - target)
    return grad

def lbfgs_round(h_init, n, target, alphas, time_limit, maxiter=500):
    start = time.time()
    h_cur = h_init.copy()
    best_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit - 5.0:
            break
        try:
            res = minimize(_smoothed_obj, h_cur,
                           args=(alpha, n, target),
                           jac=lambda x, a, nn, tt: _smoothed_grad(x, a, nn, tt),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': maxiter * 5, 'maxls': 30})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

def focused_cd(h_init, n, target, time_limit, max_sweeps=30000,
               n_cand=30, n_active_focus=5, max_no_improve=400):
    start_t = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        c_max = float(np.max(corr))
        top_idx = np.argsort(corr)[-n_active_focus:]

        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            v1 = idxs - k
            v2 = idxs + k
            m1 = (v1 >= 0) & (v1 < n)
            m2 = (v2 >= 0) & (v2 < n)
            grad[m1] += (1.0 - h[v1[m1]])
            grad[m2] -= h[v2[m2]]
        grad -= np.mean(grad)

        n_c = min(n_cand, n // 4)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_c]
        high_set = sorted_idx[-n_c:]

        best_trial = c5_val
        best_move = None

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d, b_d = d_lo, d_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid < c_max - 1e-15:
                        a_d = mid_d
                    else:
                        active = vals >= f_mid - 1e-14
                        f_prime = np.max(g[active]) if active.any() else 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_move = (i, j, delta_star)
                    break
            if best_move is not None:
                break

        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, shifts)
            update_change_row(change, j, h, shifts)

    return best_h, best_c5
