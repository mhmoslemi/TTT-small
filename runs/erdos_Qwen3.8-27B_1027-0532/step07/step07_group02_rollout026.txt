import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def smooth_obj_and_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum > 1e-30:
        weights /= wsum
    else:
        weights = np.zeros(len(corr))
        weights[int(np.argmax(corr))] = 1.0
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return obj, grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(lambda x: smooth_obj_and_grad(x, alpha)[0],
                           h_cur, jac=lambda x: smooth_obj_and_grad(x, alpha)[1],
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def nesterov_smooth(h0, n, target, alpha, maxiter=5000, time_limit=100, eta_init=0.01):
    start = time.time()
    h = h0.copy()
    y = h0.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    eta = eta_init
    t = 0
    for it in range(maxiter):
        if time.time() - start > time_limit:
            break
        obj, grad_y = smooth_obj_and_grad(y, alpha)
        grad_centered = grad_y - np.mean(grad_y)
        momentum = (it) / (it + 2.0)
        x_new = y - eta * grad_centered
        x_new = project_feasible(x_new, target)
        y_new = x_new + momentum * (x_new - h)
        y_new = np.clip(y_new, 0.0, 1.0)
        f_new = compute_c5(x_new)
        if f_new > obj * 1.5:
            eta = eta * 0.5
        elif f_new < best_f:
            best_f = f_new
            best_h = x_new.copy()
            eta = min(eta * 1.05, 0.1)
        else:
            eta = max(eta * 0.95, 1e-6)
        h = x_new
        y = y_new
    return best_h, best_f

        if sc_sub2 < best_score:

    if tleft() > 150:
        h_cd2, c5_cd2 = fast_pair_cd(best_h, n0, target0,

                                      n_cand_start=40, n_cand_end=130,

        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,

            best_h = h_pol
    
    if tleft() > 200:
        rng = np.random.RandomState(seed)
        scales = [0.02, 0.05, 0.1, 0.15]

            h_pert = best_h + rng.randn(n0) * sc

            h_cd, c5_cd = fast_pair_cd(h_pert, n0, target0,

        h_sub3, sc_sub3 = subset_optimize(best_h, n0, target0, d=50, alpha=3000,

        h_cd_f, c5_cd_f = fast_pair_cd(best_h, n0, target0,

    final_h = project_feasible(best_h, len(best_h) / 2.0)
