import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

def enhanced_cd(h_init, n, target, time_limit, max_sweeps=20000,
                n_active=3, n_cand_start=15, n_cand_end=50,
                max_no_improve=400, bisection_iters=12):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        max_pairs = 250
        if n_h * n_l <= max_pairs:
            pair_list = []
            for ii in range(n_h):
                for jj in range(n_l):
                    i_val = int(high_set[ii])
                    j_val = int(low_set[jj])
                    if i_val != j_val:
                        pair_list.append((i_val, j_val))
        else:
            pair_list = []
            for frac in np.linspace(0.0, 1.0, max_pairs):
                hi = int(frac * (n_h - 1))
                lo = int(frac * (n_l - 1))
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pair_list.append((iv, jv))
        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue
            g = change[j] - change[i]
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
    return best_h, best_c5

def block_cd(h_init, n, target, time_limit, block_size=8, max_passes=200, alpha=800):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_blocks = n // block_size
    rng = np.random.RandomState(123)
    corr = np.correlate(h, 1.0 - h, mode='full')
    sum_h = np.sum(h)
    for p in range(max_passes):
        if time.time() - start > time_limit * 0.95:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        sum_h = np.sum(h)
        block_order = rng.permutation(n_blocks)
        for b in block_order:
            if time.time() - start > time_limit * 0.98:
                break
            lo = b * block_size
            hi = lo + block_size
            h_block = h[lo:hi].copy()
            contrib = np.zeros((block_size, n_corr))
            for bi in range(block_size):
                m = lo + bi
                lo_l = np.arange(-(m), n - m)
                valid1 = lo_l >= 0
                if valid1.any():
                    contrib[bi, m + lo_l[valid1]] = (1.0 - h[m + lo_l[valid1]])
                lo_l2 = np.arange(-(n - 1 - m), m + 1)
                valid2 = lo_l2 <= 0
                if valid2.any():
                    neg_idx = m - lo_l2[valid2]
                    contrib[bi, neg_idx] -= h[m - lo_l2[valid2]]
            c_max = np.max(corr)
            w = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
            w_sum = w.sum()
            if w_sum > 0:
                w = w / w_sum
            else:
                w = np.ones(n_corr) / n_corr
            grad_block = contrib.T.dot(w)
            penalty = 2.0 * (sum_h - target)
            grad_block += penalty
            lr = 0.05
            improved = False
            for ls in range(15):
                h_trial_block = np.clip(h_block - lr * grad_block, 0.0, 1.0)
                delta = h_trial_block - h_block
                corr_trial = corr + contrib.dot(delta)
                c5_trial = float(np.max(corr_trial)) * dx
                if c5_trial < best_c5 - 1e-16:
                    h[lo:hi] = h_trial_block
                    corr = corr_trial
                    sum_h += np.sum(delta)
                    improved = True
                    break
                lr *= 0.5
            if not improved and lr < 0.001:
                pass
        if (p + 1) % 5 == 0:
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
    h = project_feasible(h, target)
    cur_c5 = compute_c5(h)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    return best_h, best_c5

def subgradient_momentum(h_init, n, target, time_limit, n_active=5, max_iters=5000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t = 1.0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        lr = 1.0 / (t_new * np.sqrt(np.sum(grad * grad) + 1e-10))
        lr = min(lr, 0.1)
        h_new = y - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h_prev = h
        h = h_new
        t = t_new
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 40:
        h_pol = slsqp_round(best_h, n0, target0, [100, 500], 400, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        h_cd, c5_cd = enhanced_cd(best_h, n0, target0,
                                   time_limit=min(60, tleft() * 0.3),
                                   max_sweeps=15000, n_active=3,
                                   n_cand_start=15, n_cand_end=40,
                                   max_no_improve=300, bisection_iters=10)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 200:
        n1 = 1200
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        tl1 = min(120, tleft() * 0.35)
        h_cd1, c5_cd1 = enhanced_cd(h1, n1, target1,
                                     time_limit=tl1, max_sweeps=15000,
                                     n_active=3, n_cand_start=20, n_cand_end=45,
                                     max_no_improve=300, bisection_iters=10)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1
            n0 = n1
            target0 = target1

    if tleft() > 300:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        tl2 = min(180, tleft() * 0.35)
        h_bc, c5_bc = block_cd(h2, n2, target2,
                                time_limit=tl2, block_size=8,
                                max_passes=300, alpha=800)
        if c5_bc < best_score:
            best_score = c5_bc
            best_h = h_bc
            n0 = n2
            target0 = target2

    if tleft() > 200:
        tl3 = min(120, tleft() * 0.35)
        h_cd3, c5_cd3 = enhanced_cd(best_h, n0, target0,
                                     time_limit=tl3, max_sweeps=12000,
                                     n_active=4, n_cand_start=20, n_cand_end=50,
                                     max_no_improve=300, bisection_iters=10)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 150:
        tl4 = min(80, tleft() * 0.3)
        h_bc2, c5_bc2 = block_cd(best_h, n0, target0,
                                  time_limit=tl4, block_size=6,
                                  max_passes=200, alpha=1500)
        if c5_bc2 < best_score:
            best_score = c5_bc2
            best_h = h_bc2

    if tleft() > 100:
        h_final = slsqp_round(best_h, n0, target0,
                               [2000, 10000], 600, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    rng = np.random.RandomState(seed)
    for pert_scale in [0.02, 0.05, 0.10]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n0) * pert_scale
        h_pert = project_feasible(h_pert, target0)
        tl_pert = min(50, tleft() * 0.3)
        h_re, sc_re = block_cd(h_pert, n0, target0,
                                time_limit=tl_pert, block_size=8,
                                max_passes=150, alpha=1000)
        if sc_re < best_score:
            best_score = sc_re
            best_h = h_re

    if tleft() > 80:
        tl5 = min(50, tleft() * 0.4)
        h_cd5, c5_cd5 = enhanced_cd(best_h, n0, target0,
                                     time_limit=tl5, max_sweeps=10000,
                                     n_active=5, n_cand_start=30, n_cand_end=60,
                                     max_no_improve=200, bisection_iters=10)
        if c5_cd5 < best_score:
            best_score = c5_cd5
            best_h = h_cd5

    if tleft() > 60:
        h_final = slsqp_round(best_h, n0, target0,
                               [10000, 50000], 400, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
