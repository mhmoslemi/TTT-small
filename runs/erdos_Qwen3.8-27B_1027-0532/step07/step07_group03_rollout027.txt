    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    p = np.exp(shifted)
    p = p / np.sum(p)
    
    n = len(x)

The correlation is: corr[m] = ∑_i x[i] * (1-x[i+k]) where k = m - (n-1)

The gradient of corr[m] with respect to x[j] is:

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_and_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    p = np.exp(shifted)
    p = p / np.sum(p)
    
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for m in range(len(corr)):
        if p[m] < 1e-15:
            continue
        k = m - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += p[m] * (1.0 - x[idxs[v1] + k])
        grad[v2] -= p[m] * x[idxs[v2] - k]
    grad /= alpha
    return obj, grad

def smooth_optimize(h_start, n, target, time_limit, alphas, maxiter=300):
    start = time.time()
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    bounds = [(0.0, 1.0)] * n
    
    # Equality constraint
    def cons_eq(x):
        return np.sum(x) - target
    def cons_eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        
        def obj(x):
            val, _ = _smooth_obj_and_grad(x, alpha)
            return val
        
        def grad(x):
            _, g = _smooth_obj_and_grad(x, alpha)
            return g
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_score

    x: length n
    p: length 2n-1 (probability distribution over lags)

    """

```python
def fast_gradient_fft(x, p):
    """Compute gradient using FFT. O(n log n)."""
    n = len(x)
    N = 4 * n  # FFT size (power of 2 for efficiency)
    
    # q[k] = p[k] for k in [0, 2n-2], 0 otherwise
    # We want A[j] = sum_k q[k] * (1-x[j+k-n+1]) for valid j+k-n+1
    # Let's define t = k - (n-1), so A[j] = sum_t p[t+n-1] * (1-x[j+t])
    # q[t] = p[t+n-1] for t in [-(n-1), n-1]
    
    # For FFT cross-correlation:
    # If a is length n and b is length 2n-1 (centered at n-1),
    # then cross-correlation is ifft(fft(a_padded) * conj(fft(b_padded)))
    
    # Actually, let me use a simpler approach:
    # A[j] = sum_{k: 0<=j+k-(n-1)<n} p[j+k-(n-1)] * (1-x[j+k-(n-1)])
    # Hmm, this is getting complicated with the indexing.
    
    # Let me just use the O(n^2) vectorized approach for now.
    # It's fast enough for n up to ~2000.
    idxs = np.arange(n)
    k_vals = np.arange(2*n-1) - (n-1)
    
    # Build offset matrix: shape (n, 2n-1)
    # offsets[j, m] = j + k_vals[m]
    offsets = idxs[:, None] + k_vals[None, :]
    valid = (offsets >= 0) & (offsets < n)
    offsets_safe = np.clip(offsets, 0, n-1)
    
    # A[j] = sum_m p[m] * (1-x[offsets[j,m]]) * valid[j,m]
    ones_minus_x = 1.0 - x
    weighted = p[None, :] * ones_minus_x[offsets_safe]
    A = np.where(valid, weighted, 0.0).sum(axis=1)
    
    # B[j] = sum_m p[m] * x[j - k_vals[m]] * 1_{valid}
    offsets2 = idxs[:, None] - k_vals[None, :]
    valid2 = (offsets2 >= 0) & (offsets2 < n)
    offsets2_safe = np.clip(offsets2, 0, n-1)
    weighted2 = p[None, :] * x[offsets2_safe]
    B = np.where(valid2, weighted2, 0.0).sum(axis=1)
    
    return A - B
```

                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, n)
            update_change_row(change, h, best_j, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def coordinate_descent(h0, n, target, time_limit, max_iters=50000):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    rng = np.random.RandomState(12345)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if (it % 200) == 0:
            corr = compute_corr(h)
            c5_val = float(np.max(corr)) * dx
            if c5_val < best_c5 - 1e-13:
                best_c5 = c5_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 100:
                break

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue

        corr = compute_corr(h)
        k_i = i - mid
        k_j = j - mid
        idxs = np.arange(n)

        di = np.zeros(2 * n - 1)
        v1i = (i + np.arange(2 * n - 1) - mid >= 0) & (i + np.arange(2 * n - 1) - mid < n)
        v2i = (i - np.arange(2 * n - 1) + mid >= 0) & (i - np.arange(2 * n - 1) + mid < n)
        di[v1i] = (1.0 - h[i + (np.arange(2 * n - 1))[v1i] - mid])
        di[v2i] -= h[i - (np.arange(2 * n - 1))[v2i] + mid]

        dj = np.zeros(2 * n - 1)
        v1j = (j + np.arange(2 * n - 1) - mid >= 0) & (j + np.arange(2 * n - 1) - mid < n)
        v2j = (j - np.arange(2 * n - 1) + mid >= 0) & (j - np.arange(2 * n - 1) + mid < n)
        dj[v1j] = (1.0 - h[j + (np.arange(2 * n - 1))[v1j] - mid])
        dj[v2j] -= h[j - (np.arange(2 * n - 1))[v2j] + mid]

        g_vec = dj - di
        a_d = delta_lo
        b_d = delta_hi
        for _ in range(10):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr + mid_d * g_vec
            f_mid = np.max(vals)
            active_m = vals >= f_mid - 1e-14
            if active_m.any():
                f_prime = np.max(g_vec[active_m])
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                b_d = mid_d
            else:
                a_d = mid_d

        delta_star = 0.5 * (a_d + b_d)
        if abs(delta_star) > 1e-12:
            new_c5 = np.max(corr + delta_star * g_vec) * dx
            if new_c5 < compute_c5(h) + 1e-14:
                h[i] -= delta_star
                h[j] += delta_star
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init.copy(), target0)
    best_score = compute_c5(best_h)

    if tleft() > 100:
        h_s, sc = do_smooth_opt(best_h, n0, target0,
                                time_limit=min(60, tleft() * 0.5),
                                alpha_seq=[80, 250, 800, 2500])
        if sc < best_score:
            best_score = sc
            best_h = h_s

    if tleft() > 80:
        h_cd, sc_cd = interceptive_cd(best_h, n0, target0,
                                      time_limit=min(50, tleft() * 0.5),
                                      max_sweeps=15000,
                                      n_cand_start=10, n_cand_end=40,
                                      max_no_improve=300)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    n_high = 800
    target_high = n_high / 2.0
    h_high = interpolate_to(best_h, n_high, target_high)
    sc_high = compute_c5(h_high)
    n_work = n_high
    target_work = target_high
    h_work = h_high.copy()

    if sc_high < best_score:
        best_score = sc_high
        best_h = h_high.copy()

    if tleft() > 400:
        h_s2, sc2 = do_smooth_opt(h_work, n_work, target_work,
                                  time_limit=min(300, tleft() * 0.45),
                                  alpha_seq=[50, 150, 500, 1500, 5000, 15000, 40000])
        if sc2 < best_score:
            best_score = sc2
            best_h = h_s2.copy()
            h_work = h_s2.copy()

    if tleft() > 300:
        h_cd2, sc_cd2 = interceptive_cd(h_work, n_work, target_work,
                                        time_limit=min(250, tleft() * 0.5),
                                        max_sweeps=25000,
                                        n_cand_start=15, n_cand_end=50,
                                        max_no_improve=400)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2.copy()
            h_work = h_cd2.copy()

    if tleft() > 200:
        h_s3, sc3 = do_smooth_opt(h_work, n_work, target_work,
                                  time_limit=min(150, tleft() * 0.5),
                                  alpha_seq=[10000, 30000, 100000])
        if sc3 < best_score:
            best_score = sc3
            best_h = h_s3.copy()
            h_work = h_s3.copy()

    if tleft() > 180:
        h_cd3, sc_cd3 = interceptive_cd(h_work, n_work, target_work,
                                        time_limit=min(150, tleft() * 0.55),
                                        max_sweeps=20000,
                                        n_cand_start=20, n_cand_end=65,
                                        max_no_improve=350)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3.copy()
            h_work = h_cd3.copy()

    if tleft() > 120:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.05]:
            if tleft() < 50:
                break
            h_pert = h_work + rng.randn(n_work) * noise_scale
            h_pert = project_feasible(h_pert, target_work)
            h_rp, sc_rp = interceptive_cd(h_pert, n_work, target_work,
                                          time_limit=min(40, tleft() * 0.35),
                                          max_sweeps=8000,
                                          n_cand_start=10, n_cand_end=35,
                                          max_no_improve=200)
            if sc_rp < best_score:
                best_score = sc_rp
                best_h = h_rp.copy()
                h_work = h_rp.copy()

    if tleft() > 80:
        h_s4, sc4 = do_smooth_opt(h_work, n_work, target_work,
                                  time_limit=min(60, tleft() * 0.6),
                                  alpha_seq=[30000, 80000, 200000])
        if sc4 < best_score:
            best_score = sc4
            best_h = h_s4.copy()
            h_work = h_s4.copy()

    if tleft() > 60:
        h_cd4, sc_cd4 = interceptive_cd(h_work, n_work, target_work,
                                        time_limit=min(50, tleft() * 0.7),
                                        max_sweeps=15000,
                                        n_cand_start=30, n_cand_end=80,
                                        max_no_improve=250)
        if sc_cd4 < best_score:
            best_score = sc_cd4
            best_h = h_cd4.copy()
            h_work = h_cd4.copy()

    if tleft() > 40:
        h_cd5, sc_cd5 = interceptive_cd(best_h, n_work, target_work,
                                        time_limit=min(30, tleft() * 0.7),
                                        max_sweeps=10000,
                                        n_cand_start=40, n_cand_end=100,
                                        max_no_improve=150)
        if sc_cd5 < best_score:
            best_score = sc_cd5
            best_h = h_cd5.copy()

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
