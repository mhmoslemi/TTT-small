import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_fn(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj_fn, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def cd_optimize(h_init, n, target, time_limit, max_sweeps=30000,

            consecutive_no_improve = 0

            break

            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)

def spectral_kick(h, n, target, rng, max_freq=8, max_amp=0.15):

        phase = rng.uniform(0, 2 * np.pi)
        if rng.random() < 0.5:

        else:
            pert = amp * np.cos(2 * np.pi * m * np.arange(n) / n + phase)
        h_trial = h + pert
        h_trial = project_feasible(h_trial, target)
        sc = compute_c5(h_trial)
        if sc < compute_c5(h_new):
            h_new = h_trial
    return h_new


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)

                                             n_cand_start=40, n_cand_end=100,
                                             n_top_shifts=3, max_no_improve=300)
                if c5_cd2 < best_score:
                    best_score = c5_cd2
                    best_h = h_cd2

    # Phase 4: Upsample to n=700
    if tleft() > 350:
        n1 = 700
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            current_n = n1
            current_target = target1

        if tleft() > 250:
            h_cd1, c5_cd1 = cd_optimize(h1, current_n, current_target,
                                         time_limit=min(220, tleft() * 0.55),
                                         max_sweeps=30000,
                                         n_cand_start=25, n_cand_end=100,
                                         n_top_shifts=4, max_no_improve=600,
                                         kick_every=90, kick_scale=0.03)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        if tleft() > 80:
            h_pol1, sc1b = slsqp_round(best_h, current_n, current_target,
                                        [1000, 5000, 20000],
                                        800, 1e-15)
            if sc1b < best_score:
                best_score = sc1b
                best_h = h_pol1

    # Phase 5: Upsample to n=1000
    if tleft() > 300:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            current_n = n2
            current_target = target2

        if tleft() > 200:
            h_cd2, c5_cd2 = cd_optimize(h2, current_n, current_target,
                                         time_limit=min(180, tleft() * 0.5),
                                         max_sweeps=20000,
                                         n_cand_start=20, n_cand_end=80,
                                         n_top_shifts=3, max_no_improve=500,
                                         kick_every=70, kick_scale=0.025)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

        if tleft() > 80:
            h_pol2, sc2b = slsqp_round(best_h, current_n, current_target,
                                        [2000, 10000, 50000],
                                        600, 1e-15)
            if sc2b < best_score:
                best_score = sc2b
                best_h = h_pol2

    # Phase 6: Final CD refinement at best resolution
    if tleft() > 100:
        h_final_cd, c5_final = cd_optimize(best_h, current_n, current_target,
                                            time_limit=min(90, tleft() * 0.7),
                                            max_sweeps=20000,
                                            n_cand_start=35, n_cand_end=110,
                                            n_top_shifts=4, max_no_improve=500,
                                            kick_every=60, kick_scale=0.03)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Phase 7: Final SLSQP polish
    if tleft() > 40:
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                             [5000, 20000, 100000],
                                             800, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
