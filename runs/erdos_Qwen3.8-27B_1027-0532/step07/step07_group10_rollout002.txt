    # c = [0, 0, ..., 0, 1] (minimize t)

    # b_ub: corresponding RHS

    # g_s[p] = (1 - h0[p+k]) - h0[p-k] where k = s - mid

        g[v2] -= h0[idxs[v2] - k]

        f_k0 = 0.0
        if k >= 0:
            f_k0 = np.sum(h0[:n-k] * (1.0 - h0[k:]))
        else:
            f_k0 = np.sum(h0[-k:] * (1.0 - h0[:n+k]))

        A_rows.append(np.append(g, -1.0))
        b_vals.append(const)

    b_ub = np.array(b_vals)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_eq = np.zeros((1, n + 1))

    b_eq = np.array([n / 2.0])

        if res.success:
            return res.x[:n]
    except Exception:
        pass

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def get_active_shifts(h, n, frac=0.05, n_top=40):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    threshold = c_max * (1.0 - frac)
    active = np.where(corr >= threshold)[0]
    top = np.argsort(corr)[-n_top:]
    active = np.unique(np.concatenate([active, top]))
    return active, corr, c_max


def solve_lp_direction(h0, n, active_shifts):
    mid = n - 1
    idxs = np.arange(n)
    A_rows = []
    b_vals = []
    for s in active_shifts:
        k = int(s) - mid
        g = np.zeros(n)
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        g[v1] += (1.0 - h0[idxs[v1] + k])
        g[v2] -= h0[idxs[v2] - k]
        if k >= 0:
            f_k0 = np.sum(h0[:n - k] * (1.0 - h0[k:]))
        else:
            f_k0 = np.sum(h0[-k:] * (1.0 - h0[:n + k]))
        rhs = np.sum(g * h0) - f_k0
        A_rows.append(np.append(g, -1.0))
        b_vals.append(rhs)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return None, None


def lp_optimize(h_init, n, target, time_limit):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    max_no_improve = 80
    while time.time() - start < time_limit:
        if no_improve > max_no_improve:
            break
        active_shifts, corr, c_max = get_active_shifts(h, n, frac=0.03, n_top=50)
        h_lp, t_lp = solve_lp_direction(h, n, active_shifts)
        if h_lp is None:
            active_shifts2, _, _ = get_active_shifts(h, n, frac=0.15, n_top=120)
            h_lp, t_lp = solve_lp_direction(h, n, active_shifts2)
            if h_lp is None:
                no_improve += 5
                continue
        improved = False
        for alpha in [1.0, 0.75, 0.55, 0.4, 0.3, 0.2, 0.12, 0.06, 0.03]:
            h_try = h + alpha * (h_lp - h)
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                no_improve = 0
                improved = True
                break
        if not improved:
            no_improve += 1
    return best_h, best_c5


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def pair_cd(h_init, n, target, time_limit, n_cand_start=30, n_cand_end=90,
            max_sweeps=50000, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-14:
                    best_trial_c5 = float(approx_max) * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5


def gradient_descent_active(h_init, n, target, time_limit, max_steps=100000,

    start = time.time()

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = base_lr
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 6000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                lr *= 0.85
                if lr < 1e-7:
                    break
        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[-1]]
        weights = np.exp(-40.0 * (c_max - corr[top_shifts]))
        weights /= np.sum(weights)
        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
        else:
            lr *= 0.5
            if lr < 1e-9:
                break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
