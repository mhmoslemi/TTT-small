def compute_corr(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

def _smooth_obj(x):
    alpha = _g_alpha
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    for alpha in alphas:

def compute_corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad


def focused_cd(h_init, n, target, time_limit, m_active=3, n_cand_start=15,
               n_cand_end=80, max_sweeps=50000, max_no_improve=800):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_m = np.argsort(corr)[-m_active:]
        corr_top = corr[top_m]
        cmax_top = float(np.max(corr_top))
        cmin_top = float(np.min(corr_top))
        spread = max(cmax_top - cmin_top, 1e-10)

        w_k = np.exp((corr_top - cmax_top) / (0.3 * spread))
        w_k /= w_k.sum()

        change_top = change[:, top_m]
        grad = change_top.dot(w_k)
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)

        sorted_idx = np.argsort(grad)

        high_set = sorted_idx[-ncand:]

                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)

                    best_j = j

def cutting_plane_method(h_init, n, target, time_limit, max_iters=10000, tol=1e-10,

    start = time.time()
    mid = n - 1
    dx = 2.0 / n

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_b = []

    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            if cuts_g:
                cuts_g = cuts_g[-min(20, len(cuts_g)):]
                cuts_b = cuts_b[-min(20, len(cuts_b)):]
            no_improve = 0

        active_mask = corr >= corr[int(np.argmax(corr))] - 1e-6
        active_lags = np.where(active_mask)[0]
        n_act = len(active_lags)

        if n_act >= m_active_cuts and n_act > 1:
            w = np.exp((corr[active_lags] - corr[active_lags].max()) / (0.5 * max(corr[active_lags].max() - corr[active_lags].min(), 1e-10)))
            w /= w.sum()
        else:
            w = np.ones(len(active_lags)) / len(active_lags)

        g = np.zeros(n)
        idxs = np.arange(n)
        for k_idx, wv in zip(active_lags, w):
            k = int(k_idx) - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            g[v1] += wv * (1.0 - h[idxs[v1] + k])
            g[v2] -= wv * h[idxs[v2] - k]

        f_val = float(w @ corr[active_lags])
        cuts_g.append(g)
        cuts_b.append(float(g @ h) - f_val)

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 3
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_b = [cuts_b[0]] + cuts_b[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = cuts_b[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                    m_active=5, n_cand_start=30, n_cand_end=100,
                                    max_sweeps=30000, max_no_improve=1000)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    if tleft() > 100:
        h_nest, sc_nest = nesterov_accurate(best_h, n0, target0,
                                             time_limit=min(100, tleft() * 0.6),
                                             alpha_start=500.0, alpha_end=80000.0,
                                             max_iters=10000, momentum=0.8)
        if sc_nest < best_score:
            best_score = sc_nest
            best_h = h_nest

    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.08]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc_ref = focused_cd(h_pert, n0, target0,
                                        time_limit=min(50, tleft() * 0.4),
                                        m_active=3, n_cand_start=20, n_cand_end=60,
                                        max_sweeps=15000, max_no_improve=400)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 80:
        h_cp_final, sc_cp_final = cutting_plane_method(best_h, n0, target0,
                                                        time_limit=min(80, tleft() * 0.5),
                                                        max_iters=5000, tol=1e-11,
                                                        max_cuts=100, m_active_cuts=8)
        if sc_cp_final < best_score:
            best_score = sc_cp_final
            best_h = h_cp_final

    if tleft() > 60:
        h_cd_final, sc_cd_final = focused_cd(best_h, n0, target0,
                                              time_limit=min(60, tleft() * 0.5),
                                              m_active=3, n_cand_start=40, n_cand_end=120,
                                              max_sweeps=20000, max_no_improve=300)
        if sc_cd_final < best_score:
            best_score = sc_cd_final
            best_h = h_cd_final

    if tleft() > 40:
        h_sls_final, sc_sls_final = slsqp_round(best_h, n0, target0,
                                                 [10000, 50000], 400)
        if sc_sls_final < best_score:
            best_score = sc_sls_final
            best_h = h_sls_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
