        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

    _g_n = n
    _g_target = target

                h_cur = h_new

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break
        
        # Find active correlations (within tolerance of max)
        c_max = np.max(corr)
        active = corr >= c_max - 1e-7
        active_idx = np.where(active)[0]
        
        # Compute gradient of the worst correlation for candidate selection
        k_star = int(np.argmax(corr)) - (n-1)
        grad = np.zeros(n)
        v1 = (idxs + k_star >= 0) & (idxs + k_star < n)
        v2 = (idxs - k_star >= 0) & (idxs - k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] + k_star])
        grad[v2] -= h[idxs[v2] - k_star]
        grad -= np.mean(grad)
        
        # Adaptive k
        progress = min(float(sweep) / max_sweeps, 1.0)
        cur_k = int(k_start + (k_end - k_start) * progress)
        cur_k = min(cur_k, n // 4)
        
        # Select candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:cur_k]
        high_set = sorted_idx[-cur_k:]
        
        # Quick filter: for each (i, j) pair, check if the direction is improving
        n_h = len(high_set)
        n_l = len(low_set)
        
        best_trial = c5_val
        best_swap = None
        
        # Vectorized filter
        # For pair (i, j): g = change[j] - change[i]
        # Derivative at 0: max(g[active_idx])
        # We want this to be negative
        candidates = []
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi < 1e-10:
                    continue
                g = change[j, active_idx] - change[i, active_idx]
                deriv = np.max(g)
                if deriv < -1e-10:
                    candidates.append((deriv, i, j, delta_hi))
        
        # Sort by derivative (most negative first) and take top few
        candidates.sort(key=lambda x: x[0])
        top_cands = candidates[:15]
        
        for _, i, j, delta_hi in top_cands:
            g_vec = change[j] - change[i]
            
            # Binary search for optimal delta
            a_d = 0.0
            b_d = delta_hi
            for _ in range(30):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g_vec
                f_mid = np.max(vals)
                act = vals >= f_mid - 1e-14
                f_prime = np.max(g_vec[act]) if act.any() else 0.0
                if f_prime > 1e-10:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-12:
                continue
            
            approx = np.max(corr + delta_star * g_vec) * dx
            if approx < best_trial:
                best_trial = approx
                best_swap = (i, j, delta_star)
        
        if best_swap is not None:
            i, j, delta_star = best_swap
            h[i] -= delta_star
            h[j] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            
            # Update change matrix
            change[i, :] = 0.0
            change[j, :] = 0.0
            pos_i = (i + shifts >= 0) & (i + shifts < n)
            neg_i = (i - shifts >= 0) & (i - shifts < n)
            change[i, pos_i] += (1.0 - h[i + shifts[pos_i]])
            change[i, neg_i] -= h[i - shifts[neg_i]]
            pos_j = (j + shifts >= 0) & (j + shifts < n)
            neg_j = (j - shifts >= 0) & (j - shifts < n)
            change[j, pos_j] += (1.0 - h[j + shifts[pos_j]])
            change[j, neg_j] -= h[j - shifts[neg_j]]
    
    return best_h, best_c5

One more thought: the SLSQP with log-sum-exp smoothing. The key parameter is alpha. Small alpha (e.g., 100) gives a smooth approximation that's easy to optimize but doesn't closely match the true objective. Large alpha (e.g., 100000) gives a close approximation but is harder to optimize (more "peaky").

The strategy is to start with small alpha and gradually increase. The previous code does this, which is good.

However, there's a subtlety: SLSQP uses numerical gradients by default. For large alpha, the function is very "spiky" and numerical gradients might be inaccurate. We should provide an analytical gradient.

Let me add the analytical gradient for the smooth objective:

f(x) = max(corr) + (1/alpha) * log(sum(exp(alpha * (corr - max(corr)))))

∂f/∂x[j] = (1/alpha) * sum_k exp(alpha*(corr[k] - max)) * ∂corr[k]/∂x[j] / sum_k exp(alpha*(corr[k] - max))

= sum_k p_k * ∂corr[k]/∂x[j]

where p_k = exp(alpha*(corr[k] - max)) / sum_m exp(alpha*(corr[k] - max))

And ∂corr[k]/∂x[j] = (1 - x[j+k]) if j+k valid, else 0) - (x[j-k] if j-k valid, else 0)

So the gradient is a weighted sum of the individual correlation gradients, with weights p_k.

This is the same as the "soft" subgradient. Let me implement this.

```python
def _smooth_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if p[k] < 1e-10:
            continue
        shift = k - mid
        v1 = (idxs + shift >= 0) & (idxs + shift < n)
        v2 = (idxs - shift >= 0) & (idxs - shift < n)
        grad[v1] += p[k] * (1.0 - x[idxs[v1] + shift])
        grad[v2] -= p[k] * x[idxs[v2] - shift]
    return grad
```

        h = np.clip(h, 0.0, 1.0)

**Idea: Exploit the structure of the problem**

**Idea: Use the Fourier transform**

The autocorrelation of h is the Fourier transform of |H|² (where H is the Fourier transform of h). Wait, that's not quite right. The autocorrelation is:

We want to minimize C₅, which means maximizing min_k ∫h(x)h(x+k)dx.

The autocorrelation R[k] = ∫h(x)h(x+k)dx is related to the Fourier transform: R = F^{-1}(|H|²) where H = F(h).

So we want to make R[k] as large as possible at its minimum, which means making |H|² as "spread out" as possible (inverse of concentrated).
