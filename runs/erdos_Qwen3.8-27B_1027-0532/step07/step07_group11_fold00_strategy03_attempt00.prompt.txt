def project_feasible(h, target_sum):

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):

def subgradient_descent(h_init, n, target, time_limit, n_steps=20000, step_scale=0.005):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.95

        if step % 500 == 0 and step > 0:
            eta = max(step_scale * max(0.1, 1.0 - step / n_steps), 1e-6)

    return best_h, best_c5

def block_cd(h_init, n, target, time_limit, block_size=5, max_sweeps=5000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 1000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        n_blocks = n // block_size
        block_grads = np.zeros(n_blocks)
        for b in range(n_blocks):
            block_grads[b] = np.mean(grad[b * block_size:(b + 1) * block_size])

        sorted_blocks = np.argsort(block_grads)
        low_blocks = sorted_blocks[:5]
        high_blocks = sorted_blocks[-5:]

        best_trial = c5_val
        best_move = None

        for hb in high_blocks:
            for lb in low_blocks:
                if hb == lb:
                    continue
                hi_start = hb * block_size
                hi_end = (hb + 1) * block_size
                lo_start = lb * block_size
                lo_end = (lb + 1) * block_size

                max_move = min(np.sum(h[hi_start:hi_end]),
                               np.sum(1.0 - h[lo_start:lo_end]))
                if max_move < 1e-10:
                    continue

                for frac in [0.25, 0.5, 0.75, 1.0]:
                    delta = frac * max_move / block_size
                    h_try = h.copy()
                    h_try[hi_start:hi_end] -= delta
                    h_try[lo_start:lo_end] += delta
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_trial - 1e-14:
                        best_trial = sc
                        best_move = (h_try.copy(), sc)

        if best_move is not None:
            h = best_move[0]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_perturbation(h_init, n, target, time_limit, n_restarts=5, base_scale=0.05):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    rng = np.random.RandomState(777)

    for r in range(n_restarts):
        if time.time() - start > time_limit:
            break
        scale = base_scale * (0.5 + 0.5 * (r / max(n_restarts - 1, 1)))
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pert.copy()

    return best_h, best_c5
