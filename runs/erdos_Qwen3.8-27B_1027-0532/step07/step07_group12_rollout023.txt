def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)

def soft_max_obj(x, alpha, target, penalty):

    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def soft_max_grad(x, alpha, target, penalty):

            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]

    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(soft_max_obj, h_cur, args=(alpha, target, pen),
                              jac=soft_max_grad, method='L-BFGS-B',

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def soft_max_grad(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(soft_max_obj, h_cur, args=(alpha, target, pen),
                              jac=soft_max_grad, method='L-BFGS-B',

                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:

                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)

        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        active_at_0 = np.where(corr >= cur_max - 1e-12)[0]

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = high_set[ii]
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            prof_i = profiles[ci]
            for jj in range(len(low_set)):
                j = low_set[jj]
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = prof_i - profiles[cj]

                g_active = g[active_at_0]
                if np.all(g_active >= -1e-14):
                    continue

                lo = 0.0
                hi_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def subgradient_phase(h_init, n, target, time_limit, max_steps=100000, n_shifts=5, base_lr=0.05):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    momentum = np.zeros(n)
    beta = 0.9

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 25.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= (2.0 / n) * np.sum(grad)

        momentum = beta * momentum + (1 - beta) * grad

        improved = False
        decay = 1.0 / (1.0 + 0.003 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * momentum
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved and step > 5000:
            break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick CD to establish baseline
    if tleft() > 80:
        hp, sc = pair_cd(best_h, n, target, time_limit=min(60, tleft() * 0.1),
                         n_cand=30, n_shifts=1, no_improve_limit=200)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: L-BFGS smoothing
    if tleft() > 100:
        hp, sc = lbfgs_optimize(best_h, n, target,
                                alphas=[100, 500, 2000],
                                maxiter=300, time_limit=min(60, tleft() * 0.12),
                                penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 3: Long CD with moderate candidates
    if tleft() > 200:
        hp, sc = pair_cd(best_h, n, target, time_limit=min(200, tleft() * 0.2),
                         n_cand=50, n_shifts=3, no_improve_limit=400)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 4: L-BFGS with higher alpha
    if tleft() > 80:
        hp, sc = lbfgs_optimize(best_h, n, target,
                                alphas=[5000, 20000],
                                maxiter=200, time_limit=min(50, tleft() * 0.1),
                                penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 5: Long CD with larger candidates and multi-shift
    if tleft() > 250:
        hp, sc = pair_cd(best_h, n, target, time_limit=min(250, tleft() * 0.22),
                         n_cand=70, n_shifts=5, no_improve_limit=500)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 6: Subgradient with momentum to explore
    if tleft() > 80:
        hp, sc = subgradient_phase(best_h, n, target, time_limit=min(60, tleft() * 0.12),
                                   max_steps=100000, n_shifts=7, base_lr=0.06)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 7: CD after subgradient (recover)
    if tleft() > 100:
        hp, sc = pair_cd(best_h, n, target, time_limit=min(80, tleft() * 0.15),
                         n_cand=50, n_shifts=3, no_improve_limit=300)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 8: Resolution bump to higher n
    if tleft() > 400:
        n_new = 1200
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        c5_interp = compute_c5(h_interp)
        hp, sc = lbfgs_optimize(h_interp, n_new, target_new,
                                alphas=[50, 500, 5000],
                                maxiter=200, time_limit=min(80, tleft() * 0.12),
                                penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new
        elif tleft() > 200:
            # Even if L-BFGS didn't improve, try CD at higher res
            pass

        if tleft() > 150:
            hp, sc = pair_cd(best_h, n, target, time_limit=min(150, tleft() * 0.2),
                             n_cand=40, n_shifts=3, no_improve_limit=400)
            if sc < best_score:
                best_score = sc
                best_h = hp

    # Phase 9: L-BFGS final smoothing at current resolution
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_optimize(best_h, n_use, target_use,
                                alphas=[50000, 200000],
                                maxiter=200, time_limit=min(60, tleft() * 0.15),
                                penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 10: Random perturbation + re-optimize
    if tleft() > 150:
        rng = np.random.RandomState(seed + 1)
        n_use = len(best_h)
        target_use = n_use / 2.0
        noise_scale = 0.03
        h_pert = best_h + rng.randn(n_use) * noise_scale
        h_pert = project_feasible(h_pert, target_use)
        hp, sc = lbfgs_optimize(h_pert, n_use, target_use,
                                alphas=[2000, 20000],
                                maxiter=200, time_limit=min(40, tleft() * 0.15),
                                penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
            if tleft() > 60:
                hp2, sc2 = pair_cd(best_h, n_use, target_use, time_limit=min(50, tleft() * 0.2),
                                   n_cand=40, n_shifts=3, no_improve_limit=200)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = hp2

    # Phase 11: Final long CD
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = pair_cd(best_h, n_use, target_use, time_limit=min(50, tleft() * 0.6),
                         n_cand=60, n_shifts=5, no_improve_limit=300)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 12: Final L-BFGS
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_optimize(best_h, n_use, target_use,
                                alphas=[500000, 2000000],
                                maxiter=150, time_limit=min(30, tleft() * 0.7),
                                penalty=50000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
