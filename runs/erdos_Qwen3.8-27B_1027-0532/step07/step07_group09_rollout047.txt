```python
_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:

            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new
        except Exception:
```

The idea is:
1. Maintain a momentum vector m
2. At each iteration:

   b. Compute gradient at y
   c. Update: m = beta * m + (x - y) (approximately)
   d. x_new = project(x - alpha * grad(y) + m)

```python
def smooth_gradient(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        
        # First term: (1-h[p+s]) for p in [max(0,-s), min(n-1, n-1-s)]
        p_low1 = max(0, -s)
        p_high1 = min(n - 1, n - 1 - s)
        mask1 = (idxs >= p_low1) & (idxs <= p_high1)
        if mask1.any():
            grad[mask1] += w * (1.0 - x[idxs[mask1] + s])
        
        # Second term: -h[p-s] for p in [max(s,0), min(n-1+s, n-1)]
        p_low2 = max(s, 0)
        p_high2 = min(n - 1 + s, n - 1)
        mask2 = (idxs >= p_low2) & (idxs <= p_high2)
        if mask2.any():
            grad[mask2] -= w * x[idxs[mask2] - s]
    
    return grad
```

```python
            break

            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        # Decay learning rate
        # lr *= 0.99999
```

```python
    no_improve = 0

        if no_improve > 3000:
            break

            grad = smooth_gradient(h, alpha)

            active = np.where(corr >= max_val - tol)[0]

                mask2 = (idxs >= p_low2) & (idxs <= p_high2)

            best_h = h_new.copy()

            # Backtrack if no improvement
            if no_improve % 100 == 0:
                lr *= 0.9
        
        h = h_new
        lr *= lr_decay
    
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            n_free = np.sum((h > 1e-15) & (h < 1.0 - 1e-15))
            if n_free == 0:
                break
            h[(h > 1e-15) & (h < 1.0 - 1e-15)] += diff / n_free
        else:
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_and_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    val = float(c_max + log_sum / alpha)
    
    n = len(x)
    total = np.sum(exp_vals)
    if total < 1e-300:
        weights = np.ones(len(corr)) / len(corr)
    else:
        weights = exp_vals / total
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        p_low1 = max(0, -s)
        p_high1 = min(n - 1, n - 1 - s)
        if p_low1 <= p_high1:
            m1 = (idxs >= p_low1) & (idxs <= p_high1)
            if m1.any():
                grad[m1] += w * (1.0 - x[idxs[m1] + s])
        p_low2 = max(s, 0)
        p_high2 = min(n - 1 + s, n - 1)
        if p_low2 <= p_high2:
            m2 = (idxs >= p_low2) & (idxs <= p_high2)
            if m2.any():
                grad[m2] -= w * x[idxs[m2] - s]
    
    return val, grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha=300):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_and_grad, h_cur, args=(alpha,), jac=True,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000, n_cand=30, max_no_improve=500):

    n_corr = 2 * n - 1

    idxs = np.arange(n)

    best_h = h.copy()

    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)

        change[p, neg_mask] -= h[p - shifts[neg_mask]]

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Weighted gradient from top correlations
        n_active = min(10, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        # Exponential weighting

        w_norm = np.sum(w)
        if w_norm > 0:

        else:

            grad[v2] -= wi * h[idxs[v2] + k]

            G[r, :] = change[int(all_cands[r]), :]

        n_l = len(low_set)

        best_i = -1
        best_j = -1
        best_delta = 0.0

                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                    continue

                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

            else:

            pos_mask = (best_i + shifts >= 0) & (best_i + shifts < n)
            neg_mask = (best_i - shifts >= 0) & (best_i - shifts < n)

            change[best_i, neg_mask] -= h[best_i - shifts[neg_mask]]
```
