def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

                    best_i = i
                    best_j = j
                    best_delta = delta_star

            h[best_j] += best_delta

            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

def lp_linearize(h_init, n, target, max_iters=20, time_limit=60):
    """Solve successive LPs by linearizing the quadratic constraints."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = float(np.max(corr)) * dx
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()

        # Build linearized LP
        # Variables: h_0,...,h_{n-1}, t → n+1 variables
        # Constraints:
        # For each shift k (n_corr constraints):
        #   corr_k_linearized ≤ t/dx
        #   where corr_k_lin = corr_k(h) + sum_j g_kj (h_j - h_j^0)
        #   = (corr_k(h) - sum_j g_kj * h_j^0) + sum_j g_kj * h_j
        #   So: sum_j g_kj * h_j - t/dx ≤ -(corr_k(h) - sum_j g_kj * h_j^0)
        #   i.e.: sum_j g_kj * h_j - (1/dx) * t ≤ -corr_k(h) + sum_j g_kj * h_j^0
        #
        #   But wait, sum_j g_kj * h_j^0 = gradient dot h = 2*corr_k(h) (by Euler's theorem for quadratic forms)
        #   Actually no, corr_k is not homogeneous. Let me just compute it directly.
        #
        # Simpler: corr_k_lin(h_new) ≈ corr_k(h) + sum_j g_kj * (h_j_new - h_j)
        #         = corr_k(h) - sum_j g_kj * h_j + sum_j g_kj * h_j_new
        # Let const_k = corr_k(h) - sum_j g_kj * h_j
        # Then: const_k + sum_j g_kj * h_j_new - t/dx ≤ 0
        # i.e.: sum_j g_kj * h_j_new - (1/dx) * t ≤ -const_k
        #
        # Also: 0 ≤ h_j ≤ 1, sum h_j = target

        # Compute gradients for all shifts
        # g_kj = ∂corr_k/∂h_j = (1 - h_{j+k}) - h_{j-k} (with appropriate bounds)
        # This is the same as the "change" matrix!
        # change[p, k_idx] gives ∂corr_{k_idx}/∂h_p
        
        # Wait, actually change[p, m] where m is the correlation index:
        # change[p, m] = ∂(corr at index m)/∂h_p
        # For m corresponding to shift k = m - mid:
        # change[p, m] = (1-h[p+k]) if p+k is valid, minus h[p-k] if p-k is valid
        # This is exactly the gradient!

        # So I need the full gradient matrix: grad_matrix[j, m] = ∂corr_m/∂h_j = change[j, m]
        # But change is already n × n_corr, which is what I need.

        # For the LP:
        # Minimize: t
        # Subject to:
        #   change @ h_new - (1/dx) * t ≤ -const, where const[m] = corr[m] - change[:,m] @ h
        #   0 ≤ h_new ≤ 1
        #   sum(h_new) = target

        # Let me compute const:
        change_mat = compute_change_matrix(h, n)  # n × n_corr
        # corr[m] - change_mat.T @ h  ... this is n_corr dimensional
        # Actually: const[m] = corr[m] - sum_j change_mat[j,m] * h[j]
        
        # Hmm, this is expensive to compute directly. Let me use the identity:
        # sum_j change_mat[j,m] * h[j] = sum_j ∂corr_m/∂h_j * h[j]
        # For corr_m = sum_i h_i (1-h_{i+k}), this is:
        # = sum_i [(1-h_{i+k}) - h_{i-k}] * h_i (with appropriate bounds)
        # = sum_i h_i(1-h_{i+k}) - sum_i h_i * h_{i-k}
        # = corr_m - (sum_i h_i * h_{i-k})
        
        # So const[m] = corr[m] - corr[m] + sum_i h_i * h_{i-k} = sum_i h_i * h_{i-k}
        # Which is the "autocorrelation-like" term!
        
        # Actually, let me just compute it numerically:
        gh = change_mat.T @ h  # n_corr dimensional
        const = corr - gh  # should be the "residual"
        
        # Now build the LP:
        # c = [0, 0, ..., 0, 1] (minimize t, last variable)
        # A_ub: for each m, [change_mat[:,m], -1/dx] @ [h_new; t] ≤ -const[m]
        # A_ub shape: (n_corr, n+1)
        # b_ub: -const
        # A_eq: [1, 1, ..., 1, 0] @ [h_new; t] = target
        # b_eq: [target]
        # bounds: [(0,1)]*n + [(0, None)]
        
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        
        A_ub = np.zeros((n_corr, n + 1))
        A_ub[:, :n] = change_mat
        A_ub[:, -1] = -1.0 / dx
        b_ub = -const
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
        
        try:
            from scipy.optimize import linprog
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
            if res.status == 0:
                h_new = res.x[:n]
                t_new = res.x[-1]
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < best_c5 - 1e-15:
                    h = h_new
                    best_c5 = sc
                    best_h = h.copy()
                elif sc < cur_c5 + 1e-10:
                    # Even if not improving, accept to continue linearization
                    h = h_new
            else:
                break
        except Exception:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def thue_morse_init(n):
    """Create a Thue-Morse-based initial solution."""
    tm = np.zeros(n)
    for i in range(n):
        # Count bits
        tm[i] = bin(i).count('1') % 2
    # Scale to satisfy integral constraint
    # We want sum(h) = n/2, so if tm has n1 ones, scale by (n/2)/n1
    n1 = tm.sum()
    if n1 == 0:
        return np.ones(n) * 0.5
    h = tm * (n / 2.0) / n1
    return project_feasible(h, n / 2.0)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        change[p, :] = 0.0
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def lp_linearize(h_init, n, target, max_iters=20, time_limit=60):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = float(np.max(corr)) * dx
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()

        change_mat = compute_change_matrix(h, n)
        gh = change_mat.T @ h
        const = corr - gh

        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0

        A_ub = np.zeros((n_corr, n + 1))
        A_ub[:, :n] = change_mat
        A_ub[:, -1] = -1.0 / dx
        b_ub = -const

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
            if res.status == 0:
                h_new = res.x[:n]
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < best_c5 - 1e-15:
                    h = h_new
                    best_c5 = sc
                    best_h = h.copy()
                elif sc < cur_c5 + 1e-12:
                    h = h_new
            else:
                break
        except Exception:
            break

    return best_h, best_c5


def fast_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=25, n_cand_end=60, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = min(6, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_max = np.max(w)
        w = np.exp(w - w_max)
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)])

        # Use active set for fast inner loop
        active_corr = corr[top_idx].copy()
        active_G = G[:, top_idx]  # n_cands x n_active

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            aG_hr = active_G[hr, :]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                if delta_max < 1e-12:
                    continue

                ag = active_G[lr, :] - aG_hr

                a_d = 0.0
                b_d = delta_max
                for _ in range(8):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = active_corr + mid_d * ag
                    f_mid = np.max(vals)
                    if f_mid < best_trial - 1e-14:
                        active_mask = vals >= f_mid - 1e-14
                        if active_mask.any():
                            f_prime = np.max(ag[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            a_d = mid_d
                        else:
                            b_d = mid_d
                    else:
                        b_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue

                g_full = G[lr, :] - G[hr, :]
                approx_max = np.max(corr + delta_star * g_full)
                if approx_max < best_trial - 1e-14:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def thue_morse_init(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    n1 = tm.sum()
    if n1 == 0:
        return np.ones(n) * 0.5
    h = tm * (n / 2.0) / n1
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        h_lp, c5_lp = lp_linearize(best_h, n0, target0,
                                   max_iters=20, time_limit=min(45, tleft()))
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 50:
        h_pol = slsqp_round(best_h, n0, target0,
                            [200, 1000, 5000], 400, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n1 = 800
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)
    best_h = h1
    n0 = n1
    target0 = target1
    best_score = compute_c5(best_h)

    if tleft() > 250:
        h_cd, c5_cd = fast_cd(h1, n1, target1,
                              time_limit=min(200, tleft() * 0.5),
                              max_sweeps=40000,
                              n_cand_start=25, n_cand_end=65,
                              max_no_improve=500)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_lp2, c5_lp2 = lp_linearize(best_h, n0, target0,
                                     max_iters=12, time_limit=min(50, tleft()))
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    if tleft() > 60:
        h_pol2 = slsqp_round(best_h, n0, target0,
                             [500, 2000, 10000], 600, 1e-14)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 300:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h = h2
        n0 = n2
        target0 = target2
        best_score = compute_c5(best_h)

        if tleft() > 200:
            h_cd3, c5_cd3 = fast_cd(h2, n2, target2,
                                    time_limit=min(150, tleft() * 0.55),
                                    max_sweeps=30000,
                                    n_cand_start=20, n_cand_end=55,
                                    max_no_improve=400)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

        if tleft() > 60:
            h_lp3, c5_lp3 = lp_linearize(best_h, n0, target0,
                                         max_iters=10, time_limit=min(40, tleft()))
            if c5_lp3 < best_score:
                best_score = c5_lp3
                best_h = h_lp3

        if tleft() > 50:
            h_pol3 = slsqp_round(best_h, n0, target0,
                                 [1000, 5000, 20000], 500, 1e-14)
            sc3 = compute_c5(h_pol3)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    if tleft() > 120:
        rng = np.random.RandomState(seed)
        h_tm = thue_morse_init(n0)
        h_tm = h_tm * 0.5 + best_h * 0.5
        h_tm = project_feasible(h_tm, target0)
        h_tm_pol = slsqp_round(h_tm, n0, target0,
                               [200, 1000, 5000], 400, 1e-14)
        sc_tm = compute_c5(h_tm_pol)
        if sc_tm < best_score:
            best_score = sc_tm
            best_h = h_tm_pol

    if tleft() > 100:
        rng = np.random.RandomState(seed + 1)
        for noise_scale in [0.015, 0.03]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref = slsqp_round(h_pert, n0, target0,
                                [200, 1000, 5000], 300, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 60:
        h_cd_final, c5_final = fast_cd(best_h, n0, target0,
                                       time_limit=min(45, tleft() * 0.55),
                                       max_sweeps=20000,
                                       n_cand_start=35, n_cand_end=75,
                                       max_no_improve=300)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 35:
        h_final = slsqp_round(best_h, n0, target0,
                              [3000, 10000, 30000], 400, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
