```python
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
        if np.any(neg_mask):
            change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change, shifts

def update_change_row(change, p, h, n, shifts):
    n_corr = 2 * n - 1
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    if np.any(pos_mask):
        change[p, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
    if np.any(neg_mask):
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

def get_gradient_multi(h, corr, n, top_k, temperature=1.0):
    """Compute weighted gradient from top-k shifts"""
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    
    # Get top-k shifts
    top_indices = np.argsort(corr)[-top_k:]
    c_max = corr[top_indices[0]] if corr[top_indices[0]] == np.max(corr) else np.max(corr)
    c_max = np.max(corr)
    
    # Softmax weights over top-k
    weights = np.exp(-temperature * (c_max - corr[top_indices]))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for wi, s_idx in enumerate(top_indices):
        k = int(s_idx) - mid
        if k == 0:
            v1 = idxs
            v2 = idxs
        elif k > 0:
            v1 = idxs[idxs >= k]  # i such that i-k >= 0 -> i >= k
            v2 = idxs[idxs < n - k]  # i such that i+k < n -> i < n-k
        else:
            k_abs = -k
            v1 = idxs[idxs >= k_abs]  # i such that i+k_abs >= 0
            v2 = idxs[idxs < n - k_abs]  # i such that i-k_abs < n
        
        # grad[i] += w * (1 - h[i-k]) for valid i-k
        # grad[i] -= w * h[i+k] for valid i+k
        if len(v1) > 0:
            grad[v1] += weights[wi] * (1.0 - h[v1 - k])
        if len(v2) > 0:
            grad[v2] -= weights[wi] * h[v2 + k]
    
    grad -= np.mean(grad)
    return grad

def diversity_restart(h_best, n, target, n_restarts=20, rng=None):
    """Generate diverse perturbations and return the best one after projection"""
    if rng is None:
        rng = np.random.RandomState(0)
    
    best_h = h_best.copy()
    best_score = compute_c5(h_best)
    
    for r in range(n_restarts):
        # Type 1: Gaussian perturbation
        if r % 3 == 0:
            scale = 0.1 + 0.05 * (r // 3)
            h_pert = h_best + rng.randn(n) * scale
        
        # Type 2: Binary flip
        elif r % 3 == 1:
            h_pert = h_best.copy()
            # Flip a random subset
            flip_mask = rng.rand(n) < 0.15
            h_pert[flip_mask] = 1.0 - h_pert[flip_mask]
        
        # Type 3: Block swap
        else:
            h_pert = h_best.copy()
            block_size = max(3, n // 20)
            start_pos = rng.randint(0, n - 2 * block_size)
            # Swap two blocks
            b1_start = start_pos
            b1_end = b1_start + block_size
            b2_start = start_pos + block_size + rng.randint(0, n - 2*block_size - block_size)
            b2_end = b2_start + block_size
            if b2_end < n:
                h_pert[b1_start:b1_end], h_pert[b2_start:b2_end] = \

def slsqp_smooth(h_start, n, target, alphas, maxiter, time_limit, start_time):
    """SLSQP with smooth max approximation"""
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alphas[0] * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alphas[0]) * dx
    
    def obj_grad(x):
        # Gradient of smooth max
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alphas[0] * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        
        # grad of corr[k] w.r.t. x[i]
        grad = np.zeros(n)
        mid = n - 1
        for k_idx in range(len(corr)):
            if weights[k_idx] < 1e-10:
                continue
            k = k_idx - mid
            if k >= 0:
                valid = np.arange(k, n)
                grad[valid] += weights[k_idx] * (1.0 - x[valid - k])
                # derivative of x[i]*(1-x[i+k]) w.r.t. x[i] is (1-x[i+k])
                # derivative w.r.t. x[i+k] is -x[i]
                grad2 = np.zeros(n)
                grad2[valid] += (1.0 - x[valid - k])  # d/dx[i]
                if k > 0:
                    grad2[valid + k] -= x[valid]  # d/dx[i+k]
                grad += weights[k_idx] * grad2
            else:
                k_abs = -k
                valid = np.arange(n - k_abs)
                # x[i] * (1-x[i+k]) where k is negative, so i+k = i-k_abs
                # valid when i-k_abs >= 0, i.e., i >= k_abs
                valid2 = np.arange(k_abs, n)
                # corr at shift k: sum over i where i>=0, i+k in [0,n)
                # i+k = i-k_abs, need i-k_abs >= 0 -> i >= k_abs
                # and i < n
                vals = valid2
                grad_partial = np.zeros(n)
                grad_partial[vals] += (1.0 - x[vals - k_abs])  # d/dx[i] of x[i]*(1-x[i-k_abs])
                grad_partial[vals - k_abs] -= x[vals]  # d/dx[i-k_abs]
                grad += weights[k_idx] * grad_partial
        
        return grad * dx
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > start_time + time_limit:
            break
        
        # Redefine with current alpha
        def obj_alpha(x, a=alpha):

            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
        
        try:

                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, cur_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        if np.any(pos_mask):
            change[p, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
        if np.any(neg_mask):
            change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change, shifts

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    if np.any(pos_mask):
        change[p, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
    if np.any(neg_mask):
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

def get_gradient_multi(h, corr, n, top_k, temperature=2.0):
    idxs = np.arange(n)
    mid = n - 1
    top_indices = np.argsort(corr)[-top_k:]
    c_max = np.max(corr)
    weights = np.exp(-temperature * (c_max - corr[top_indices]))
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for wi, s_idx in enumerate(top_indices):
        k = int(s_idx) - mid
        j = idxs
        v1 = (j + k >= 0) & (j + k < n)
        v2 = (j - k >= 0) & (j - k < n)
        if np.any(v1):
            grad[v1] += weights[wi] * (1.0 - h[j[v1] + k])
        if np.any(v2):
            grad[v2] -= weights[wi] * h[j[v2] - k]
    grad -= np.mean(grad)
    return grad

def fast_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand=50, top_k=5, temperature=2.5,
            max_no_improve=400, n_binsearch=12, top_M=30):
    dx = 2.0 / n
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change, shifts = compute_change_matrix(h, n)
    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max_no_improve:
            break

        grad = get_gradient_multi(h, corr, n, top_k, temperature)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        top_M_idx = np.argsort(corr)[-top_M:]
        corr_top = corr[top_M_idx]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                g_top = g[top_M_idx]
                a_d, b_d = d_lo, d_hi
                for _ in range(n_binsearch):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr_top + m_d * g_top
                    f_m = np.max(vals)
                    act = vals >= f_m - 1e-13
                    if not np.any(act):
                        break
                    g_act = g_top[act]
                    if np.all(g_act > 1e-12):
                        b_d = m_d
                    elif np.all(g_act < -1e-12):
                        a_d = m_d
                    else:
                        break
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) < 1e-13:
                    continue
                est = np.max(corr_top + d_star * g_top) * dx
                if est < best_trial - 1e-15:
                    best_trial = est
                    best_i, best_j, best_delta = i, j, d_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def binary_push(h, target):
    n = len(h)
    h_bp = h.copy()
    thresh_lo = 0.05
    thresh_hi = 0.95
    for i in range(n):
        if h_bp[i] < thresh_lo:
            h_bp[i] = 0.0
        elif h_bp[i] > thresh_hi:
            h_bp[i] = 1.0
    h_bp = project_feasible(h_bp, target)
    return h_bp

def diversity_restart(h_best, n, target, n_restarts=20, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h_best.copy()
    best_score = compute_c5(h_best)
    for r in range(n_restarts):
        if r % 4 == 0:
            scale = 0.08 + 0.03 * (r // 4)
            h_pert = h_best + rng.randn(n) * scale
        elif r % 4 == 1:
            h_pert = h_best.copy()
            flip_mask = rng.rand(n) < 0.2
            h_pert[flip_mask] = 1.0 - h_pert[flip_mask]
        elif r % 4 == 2:
            h_pert = h_best.copy()
            block_sz = max(3, n // 15)
            n_blocks = max(1, rng.randint(1, 4))
            for _ in range(n_blocks):
                s1 = rng.randint(0, n - 2 * block_sz)
                s2 = rng.randint(0, n - 2 * block_sz)
                if abs(s1 - s2) < block_sz:
                    continue
                h_pert[s1:s1+block_sz], h_pert[s2:s2+block_sz] = \
                    h_pert[s2:s2+block_sz].copy(), h_pert[s1:s1+block_sz].copy()
        else:
            h_pert = h_best.copy()
            n_flips = rng.randint(n // 10, n // 4)
            flip_idx = rng.choice(n, size=n_flips, replace=False)
            h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        if sc < best_score:
            best_score = sc
            best_h = h_pert.copy()
    return best_h, best_score

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    for alpha in alphas:
        if time.time() - start_time > start_time + time_limit:
            break
        def obj_a(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
        try:
            res = minimize(obj_a, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
