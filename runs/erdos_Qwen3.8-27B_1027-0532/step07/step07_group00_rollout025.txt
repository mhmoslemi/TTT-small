from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

            grad[v2] -= w * h[idxs[v2] + k]

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0

def _smooth_obj_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_pwr(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Power sum approximation: (sum |corr|^alpha)^(1/alpha)
    # Normalize to avoid overflow
    c_max = np.max(np.abs(corr))
    if c_max < 1e-20:
        return 0.0
    normed = corr / c_max
    val = np.sum(np.abs(normed) ** alpha) ** (1.0 / alpha) * c_max
    return float(val)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, use_pwr=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    obj = _smooth_obj_pwr if use_pwr else _smooth_obj_lse
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break
                elif T > 1e-12 and approx_max - c5_val < T * 3:
                    if rng.random() < np.exp(-(approx_max - c5_val) / T):
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
                        break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def exhaustive_cd(h_init, n, target, time_limit, max_sweeps=10000,
                  n_cand=80, max_no_improve=300, ls_iters=26):
    start_t = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        c_max = float(np.max(corr))
        threshold = c_max - 3.0 * dx
        active = corr >= threshold
        if active.sum() < 5:
            active_idx = np.argsort(corr)[-15:]
        else:
            active_idx = np.where(active)[0]

        grad = np.zeros(n)
        for s_idx in active_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(ls_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def subgradient_nudge(h_init, n, target, n_steps=200, step_scale=0.002):
    start_t = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    step = step_scale
    for it in range(n_steps):
        if time.time() - start_t > 30:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        k_star = int(np.argmax(corr))
        k = k_star - mid
        grad = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - step * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val:
            h = h_try
            step = min(step * 1.03, 0.01)
        else:
            step *= 0.8

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP polish at initial resolution
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000],
                                400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Single-shift SLSQP (cycle through active shifts)
    if tleft() > 120:
        h_ss, sc_ss = slsqp_single_shift(best_h, n0, target0,
                                         max_iter=400, max_rounds=8)
        if sc_ss < best_score:
            best_score = sc_ss
            best_h = h_ss

    # Phase 3: Weighted pair CD - aggressive
    if tleft() > 200:
        h_cd, c5_cd = weighted_pair_cd(best_h, n0, target0,
                                       time_limit=min(180, tleft() * 0.6),
                                       max_sweeps=25000,
                                       n_cand_start=25, n_cand_end=75,
                                       max_no_improve=400,
                                       weight_beta=5.0,
                                       ls_iters=22, top_k_shifts=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP with high alpha
    if tleft() > 120:
        h_pol2, sc2 = slsqp_round(best_h, n0, target0,
                                  [5000, 20000],
                                  500, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 5: Weighted pair CD - focused (higher beta, fewer candidates)
    if tleft() > 150:
        h_cd2, c5_cd2 = weighted_pair_cd(best_h, n0, target0,
                                         time_limit=min(130, tleft() * 0.6),
                                         max_sweeps=20000,
                                         n_cand_start=15, n_cand_end=50,
                                         max_no_improve=300,
                                         weight_beta=10.0,
                                         ls_iters=24, top_k_shifts=50)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 6: Subgradient nudge to escape local minima
    if tleft() > 100:
        h_sg, sc_sg = subgradient_nudge(best_h, n0, target0,
                                        n_steps=150, step_scale=0.001)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        if tleft() > 60:
            h_pol3, sc3 = slsqp_round(best_h, n0, target0,
                                      [2000, 10000],
                                      300, 1e-14)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    # Phase 7: Multi-start with structured perturbations
    if tleft() > 200:
        rng = np.random.RandomState(seed)
        n_blocks = 8
        block_len = n0 // n_blocks
        for noise_scale in [0.02, 0.05, 0.10]:
            if tleft() < 80:
                break
            h_pert = best_h.copy()
            for _ in range(3):
                b = rng.randint(n_blocks)
                lo = b * block_len
                hi = lo + block_len
                h_pert[lo:hi] += rng.uniform(-noise_scale, noise_scale)
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc_ref = slsqp_round(h_pert, n0, target0,
                                        [500, 3000],
                                        300, 1e-14)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 50:
                h_cd_ref, c5_cd_ref = weighted_pair_cd(best_h, n0, target0,
                                                       time_limit=min(25, tleft() * 0.3),
                                                       max_sweeps=8000,
                                                       n_cand_start=20, n_cand_end=50,
                                                       max_no_improve=150,
                                                       weight_beta=6.0,
                                                       ls_iters=20, top_k_shifts=30)
                if c5_cd_ref < best_score:
                    best_score = c5_cd_ref
                    best_h = h_cd_ref

    # Phase 8: Exhaustive CD (large candidate set, accurate line search)
    if tleft() > 180:
        h_ex, c5_ex = exhaustive_cd(best_h, n0, target0,
                                    time_limit=min(160, tleft() * 0.7),
                                    max_sweeps=10000,
                                    n_cand=90, max_no_improve=250,
                                    ls_iters=26)
        if c5_ex < best_score:
            best_score = c5_ex
            best_h = h_ex

    # Phase 9: Single-shift cycling
    if tleft() > 100:
        h_ss2, sc_ss2 = slsqp_single_shift(best_h, n0, target0,
                                           max_iter=500, max_rounds=10)
        if sc_ss2 < best_score:
            best_score = sc_ss2
            best_h = h_ss2

    # Phase 10: Final SLSQP with very high alpha
    if tleft() > 80:
        h_pol4, sc4 = slsqp_round(best_h, n0, target0,
                                  [50000, 100000],
                                  500, 1e-14)
        if sc4 < best_score:
            best_score = sc4
            best_h = h_pol4

    # Phase 11: Final exhaustive CD polish
    if tleft() > 60:
        h_final, c5_final = exhaustive_cd(best_h, n0, target0,
                                          time_limit=min(50, tleft() * 0.7),
                                          max_sweeps=5000,
                                          n_cand=70, max_no_improve=100,
                                          ls_iters=28)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final

    # Phase 12: Final SLSQP
    if tleft() > 35:
        h_fin, sc_fin = slsqp_round(best_h, n0, target0,
                                    [100000],
                                    400, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
