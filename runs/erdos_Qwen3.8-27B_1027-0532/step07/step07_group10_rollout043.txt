def project_to_simplex_box(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h) = target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)

            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

def slsqp_smoothed(h_init, n, target, alphas, maxiter_per_alpha):
    """SLSQP with log-sum-exp smoothing of the max."""
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    
    def obj_func(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -600.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def obj_grad(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -600.0, 0.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= w_sum
        
        grad = np.zeros(n)
        for k_idx in np.where(weights > 1e-10)[0]:
            k = int(k_idx) - mid
            w = weights[k_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - x[idxs[v1] - k])
            grad[v2] -= w * x[idxs[v2] + k]
        return grad
    
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, a=target: np.sum(x) - a,
             'jac': lambda x, a=target: np.ones(n)}]
    
    for alpha in alphas:
        def _f(x):

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    target_dst = n_dst / 2.0
    if n_src == n_dst:
        return project_to_simplex_box(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_to_simplex_box(h_out, target_dst)

def thue_morse_h(n):
    """Generate Thue-Morse based initial function."""
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = 1.0 - t.astype(float)
    return project_to_simplex_box(h, n / 2.0)

def periodic_h(n, period, duty=0.5):
    """Generate periodic initial function."""
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        h[i] = 1.0 if phase < duty else 0.0
    return project_to_simplex_box(h, n / 2.0)

def random_binary_h(n, seed=0):
    rng = np.random.RandomState(seed)
    h = rng.randint(0, 2, size=n).astype(float)
    return project_to_simplex_box(h, n / 2.0)

def project_to_box_sum(h, target_sum):

            tl = min(60.0, tleft() * 0.25)
            h1_pol, sc1 = slsqp_round(h1, n1, target1, [100, 500, 2000], 400)
            if sc1 < base1:
                h1 = h1_pol
                base1 = sc1
        if tleft() > 150:
            tl = min(100.0, tleft() * 0.45)
            h1_cd, sc1 = pairwise_cd_precomputed(h1, n1, target1,
                                                  time_limit=tl, n_cand_start=30, n_cand_end=80)
            if sc1 < best_score:
                best_score = sc1
                best_h = h1_cd
                best_n = n1
                best_target = target1
        if tleft() > 80:
            tl = min(50.0, tleft() * 0.4)
            h1_nes, sc1 = nesterov_accel(best_h, best_n, best_target,
                                         time_limit=tl, alpha=300.0, base_lr=0.04)
            if sc1 < best_score:
                best_score = sc1
                best_h = h1_nes
        if tleft() > 60:
            tl = min(40.0, tleft() * 0.5)
            h1_ds, sc1 = direction_search(best_h, best_n, best_target,
                                          time_limit=tl, n_active=5)
            if sc1 < best_score:
                best_score = sc1
                best_h = h1_ds

    # Phase 8: Try n=800
    n2 = 800
    if tleft() > 400:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2)
        base2 = compute_c5(h2)
        if tleft() > 320:
            tl = min(80.0, tleft() * 0.3)
            h2_pol, sc2 = slsqp_round(h2, n2, target2, [200, 1000, 5000], 500)
            if sc2 < base2:
                h2 = h2_pol
                base2 = sc2
        if tleft() > 200:
            tl = min(130.0, tleft() * 0.5)
            h2_cd, sc2 = pairwise_cd_precomputed(h2, n2, target2,
                                                  time_limit=tl, n_cand_start=35, n_cand_end=90)
            if sc2 < best_score:
                best_score = sc2
                best_h = h2_cd
                best_n = n2
                best_target = target2
        if tleft() > 100:
            tl = min(60.0, tleft() * 0.4)
            h2_nes, sc2 = nesterov_accel(best_h, best_n, best_target,
                                         time_limit=tl, alpha=400.0, base_lr=0.03)
            if sc2 < best_score:
                best_score = sc2
                best_h = h2_nes
        if tleft() > 60:
            tl = min(40.0, tleft() * 0.5)
            h2_bal, sc2 = active_balance(best_h, best_n, best_target,
                                         time_limit=tl, n_active=6)
            if sc2 < best_score:
                best_score = sc2
                best_h = h2_bal

    # Phase 9: Final SLSQP
    if tleft() > 80:
        tl = min(60.0, tleft() * 0.65)
        h_fin, sc_fin = slsqp_round(best_h, best_n, best_target,
                                    [10000, 50000, 200000], int(1000 * tl / 60))
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 10: Final CD
    if tleft() > 70:
        tl = min(50.0, tleft() * 0.6)
        h_cd_f, sc_f = pairwise_cd_precomputed(best_h, best_n, best_target,
                                               time_limit=tl, n_cand_start=50, n_cand_end=100)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_cd_f

    # Phase 11: Perturbation restarts
    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for scale in [0.02, 0.05, 0.10, 0.20]:
            if tleft() < 35:
                break
            h_pert = best_h + rng.randn(best_n) * scale
            h_pert = project_to_box_sum(h_pert, best_target)
            h_ref, sc_ref = slsqp_round(h_pert, best_n, best_target,
                                        [500, 5000, 20000], 300)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 25:
                h_ref2, sc_ref2 = pairwise_cd_precomputed(h_ref, best_n, best_target,
                                                          time_limit=min(15, tleft() * 0.4),
                                                          n_cand_start=40, n_cand_end=80)
                if sc_ref2 < best_score:
                    best_score = sc_ref2
                    best_h = h_ref2

    # Phase 12: Final direction search
    if tleft() > 30:
        tl = min(20.0, tleft() * 0.6)
        h_ds_f, sc_ds = direction_search(best_h, best_n, best_target,
                                         time_limit=tl, n_active=6)
        if sc_ds < best_score:
            best_score = sc_ds
            best_h = h_ds_f

    # Phase 13: Final SLSQP
    if tleft() > 30:
        tl = min(20.0, tleft() * 0.7)
        h_last, sc_last = slsqp_round(best_h, best_n, best_target,
                                      [50000, 200000], int(500 * tl / 20))
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    final_h = project_to_box_sum(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, best_n
