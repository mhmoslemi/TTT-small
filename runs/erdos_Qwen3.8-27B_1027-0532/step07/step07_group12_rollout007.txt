                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(18):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def subgradient_phase(h_init, n, target, time_limit, max_steps=100000, n_shifts=5, base_lr=0.05):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 20.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= (2.0 / n) * np.sum(grad)
        improved = False
        decay = 1.0 / (1.0 + 0.003 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved and step > 5000:
            break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = (np.arange(n_src) + 0.5) / n_src
    x_new = (np.arange(n_dst) + 0.5) / n_dst
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_restart_pair_cd(h_init, n, target, time_limit, n_restarts=3, n_cand=40):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    rng = np.random.RandomState(123)
    per_restart_time = time_limit / (n_restarts + 1)
    time_per = 0
    h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=per_restart_time, n_cand=n_cand, max_sweeps=100000, n_shifts=1)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd
    time_per = time.time() - start - per_restart_time
    for r in range(n_restarts):
        if time.time() - start > time_limit - 30:
            break
        remaining = time_limit - (time.time() - start)
        if remaining < 30:
            break
        noise_scale = 0.03 / (r + 1)
        h_perturbed = best_h + rng.randn(n) * noise_scale
        h_perturbed = project_feasible(h_perturbed, target)
        t_this = min(remaining * 0.7, 120.0)
        h_try, sc = pair_cd(h_perturbed, n, target, time_limit=t_this, n_cand=n_cand, max_sweeps=100000, n_shifts=1)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_try
    return best_h, best_c5
