        if not mask.any():

        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    """Smooth max of correlation values times dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Smooth max: (1/alpha) * log(sum(exp(alpha * corr)))
    # Scale by dx at the end
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    log_sum = np.log(np.sum(np.exp(np.clip(shifted, -500, 0)))) / alpha
    return (max_corr + log_sum) * dx

def smooth_max_grad(h, alpha):
    """Gradient of smooth max of correlation values times dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = np.clip(alpha * (corr - max_corr), -500, 0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        weights = np.zeros(len(corr))
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= wsum
    
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(len(corr)):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad * dx

    h_cur = h_init.copy().astype(float)

        momentum = 0.0

            g_norm = np.linalg.norm(g)

            y_new = x_new + beta * (x_new - y)

            if f_new < f_old + 1e-15:
                h_cur = x_new
                y = y_new
                momentum = t_plus
                n_iter += 1
                sc = compute_c5(h_cur)
                if sc < best_score:
                    best_score = sc
                    best_h = h_cur.copy()
            else:
                # Reduce step size
                step *= 0.5
                if step < 1e-15:
                    break
    
    return best_h, best_score

def smooth_max_val(corr, alpha):
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    return max_c + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_weights(corr, alpha):
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-30:
        w = np.zeros(len(corr))
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= s
    return w

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = smooth_max_weights(corr, alpha)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(len(corr)):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad * dx

def slsqp_pass(h_start, n, target, alphas, maxiter, time_budget):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    
    def obj(x, a):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_val(corr, a) * (2.0 / n)
    
    def obj_grad(x, a):
        return smooth_max_grad(x, a)
    
    def con_fun(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n)
    
    for alpha in alphas:
        if time.time() - start > time_budget:
            break
        try:
            res = minimize(obj, h_cur, args=(alpha,), jac=lambda x, a=alpha: obj_grad(x, a),
                          method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, alpha):
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    return max_c + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_weights(corr, alpha):
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-30:
        w = np.zeros(len(corr))
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= s
    return w

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = smooth_max_weights(corr, alpha)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(len(corr)):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad * dx

def _slsqp_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_val(corr, alpha) * (2.0 / n)

def _slsqp_grad(x, alpha, n):
    return smooth_max_grad(x, alpha)

def _cons_fun(x, target):
    return np.sum(x) - target

def _cons_jac(x, target):
    return np.ones(len(x))

def slsqp_pass(h_start, n, target, alphas, maxiter, time_budget):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_budget:
            break
        try:
            obj = lambda x: _slsqp_obj(x, alpha, n)
            jac = lambda x: _slsqp_grad(x, alpha, n)
            cf = lambda x: _cons_fun(x, target)
            cj = lambda x: _cons_jac(x, target)
            res = minimize(obj, h_cur, jac=jac, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

def _update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def _init_change_matrix(h, n):
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        _update_change_row(change, p, h, n, shifts)
    return change

def pair_cd(h_init, n, target, time_limit, max_sweeps, n_cand_start, n_cand_end,
            max_no_improve, n_binsearch, stochastic_frac, temp_for_softmax, seed):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = _init_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]

        if temp_for_softmax is not None and temp_for_softmax > 0:
            sw = smooth_max_weights(corr, 1.0 / temp_for_softmax)
            sw_grad = np.zeros(n)
            for s in range(n_corr):
                w = sw[s]
                if w < 1e-10:
                    continue
                k = s - mid
                v1s = (idxs - k >= 0) & (idxs - k < n)
                v2s = (idxs + k >= 0) & (idxs + k < n)
                sw_grad[v1s] += w * (1.0 - h[idxs[v1s] - k])
                sw_grad[v2s] -= w * h[idxs[v2s] + k]
            grad = 0.5 * grad + 0.5 * sw_grad

        grad -= np.mean(grad)
        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-14:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_min_val(vals, f_mid)
                    f_prime = np.max(g[active_mask]) if active_mask.any() else 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-14:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        n_random = max(1, int(stochastic_frac * n_h * n_l))
        for _ in range(n_random):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-14:
                continue
            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(n_binsearch):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                f_prime = np.max(g[active_mask]) if active_mask.any() else 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-14:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            _update_change_row(change, best_i, h, n, shifts)
            _update_change_row(change, best_j, h, n, shifts)
        elif consecutive_no_improve > 80 and consecutive_no_improve % 100 == 0:
            n_pert = max(5, n // 20)
            idxs_pert = rng.choice(n, size=n_pert, replace=False)
            h[idxs_pert] += rng.randn(n_pert) * 0.04
            h = project_feasible(h, target)
            for p in idxs_pert:
                _update_change_row(change, int(p), h, n, shifts)

    return best_h, best_c5

def f_min_val(vals, f_mid):
    return f_mid - 1e-14

def multi_start_cd(h_best, n, target, time_limit, seed):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    scales = [0.01, 0.03, 0.06, 0.1, 0.15, 0.25]
    n_attempts = 0
    while time.time() - start < time_limit:
        if time.time() - start > time_limit * 0.9:
            break
        n_attempts += 1
        scale = scales[rng.randint(len(scales))]
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        remaining = time_limit - (time.time() - start)
        if remaining < 40:
            break
        alloc = min(remaining * 0.35, 120)
        h_cd, c5_cd = pair_cd(h_pert, n, target,
                               time_limit=alloc,
                               max_sweeps=8000,
                               n_cand_start=35, n_cand_end=100,
                               max_no_improve=200,
                               n_binsearch=16,
                               stochastic_frac=0.06,
                               temp_for_softmax=6.0,
                               seed=int(rng.randint(100000)))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
