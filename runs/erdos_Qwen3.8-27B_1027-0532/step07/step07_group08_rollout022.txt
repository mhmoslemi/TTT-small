def compute_change_matrix(h, n):

            best_h = h.copy()
            stale = 0
            alpha = min(alpha * 1.05, alpha0)
        else:
            stale += 1
            alpha *= 0.97
            if stale > 400:
                break

        c_max = float(np.max(corr))
        threshold = c_max - 0.03 * max(c_max, 0.01)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        grad = np.zeros(n)
        n_act = len(active)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = 1.0 / n_act
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def wiggle_perturb(h, n, target, rng, scale=0.05, freq=None):
    if freq is None:
        freq = rng.randint(2, max(3, n // 100))
    phases = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    pert = scale * np.sin(2 * np.pi * freq * x / n + phases)
    h_pert = h + pert
    return project_feasible(h_pert, target)


def flip_block_perturb(h, n, target, rng):
    block_len = rng.randint(max(5, n // 20), max(10, n // 8))
    pos = rng.randint(0, n - block_len)
    h_pert = h.copy()
    h_pert[pos:pos+block_len] = 1.0 - h_pert[pos:pos+block_len]
    return project_feasible(h_pert, target)

    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)

    best_score = compute_c5(best_h)
    current_n = n0

    if tleft() > 70:
        deadline = time.time() + min(50, tleft() * 0.5)
        h_pol, sc = slsqp_polish(best_h, n0, target0,
                                  [80, 300, 1500, 8000],

            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        deadline = time.time() + min(50, tleft() * 0.4)
        h_trf, sc_trf = trf_polish(best_h, current_n, current_target,
                                    [300, 1500, 8000],
                                    1000, deadline=deadline)
        if sc_trf < best_score:
            best_score = sc_trf
            best_h = h_trf

    if tleft() > 300:
        h_cd, c5_cd = pair_cd_improved(best_h, current_n, current_target,

                                        max_no_improve=800,

                                        reheat_interval=120, reheat_scale=0.04,

                                    1000, 1e-16, deadline=deadline)
        if sc2 < best_score:

    if tleft() > 100:
        deadline = time.time() + min(60, tleft() * 0.4)
        h_trf2, sc_trf2 = trf_polish(best_h, current_n, current_target,
                                      [5000, 20000, 80000],
                                      1500, deadline=deadline)
        if sc_trf2 < best_score:
            best_score = sc_trf2
            best_h = h_trf2

    if tleft() > 250:
        h_cd2, c5_cd2 = pair_cd_improved(best_h, current_n, current_target,
                                          time_limit=min(180, tleft() * 0.55),
                                          max_sweeps=35000,
                                          n_cand_start=50, n_cand_end=140,
                                          max_no_improve=1000,
                                          n_active_shifts=5,
                                          reheat_interval=80, reheat_scale=0.03,
                                          use_triple=True)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 200:
        for attempt in range(5):
            if tleft() < 50:
                break
            if attempt % 3 == 0:
                h_pert = wiggle_perturb(best_h, current_n, current_target, rng,
                                         scale=0.04)
            elif attempt % 3 == 1:
                h_pert = flip_block_perturb(best_h, current_n, current_target, rng)
            else:
                noise_scale = 0.05
                h_pert = best_h + rng.randn(current_n) * noise_scale
                h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(30, tleft() * 0.3)
            h_ref, sc_ref = slsqp_polish(h_pert, current_n, current_target,
                                          [500, 3000, 15000],
                                          800, 1e-15, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 100:
        h_sg, c5_sg = subgradient_cd(best_h, current_n, current_target,
                                      time_limit=min(50, tleft() * 0.5),
                                      alpha0=0.15, alpha_min=0.003)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

        if c5_bc < best_score:
            best_score = c5_bc

            if tleft() > 20:

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_h = np.clip(final_h, 0.0, 1.0)

    return final_h, final_score, len(final_h)
