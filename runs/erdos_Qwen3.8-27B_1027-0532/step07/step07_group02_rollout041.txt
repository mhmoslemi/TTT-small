        if abs(diff) < 1e-14:
            break
        if diff > 0:

        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_and_grad(h, k):
    """Compute corr[k] and its gradient w.r.t. h."""
    n = len(h)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    
    # corr[k] = sum_i h[i] * (1 - h[i+k]) for valid i
    # = sum_{i: 0<=i<n, 0<=i+k<n} h[i] * (1 - h[i+k])
    c = np.sum(h[idxs[v1]] * (1.0 - h[idxs[v1] + k]))
    
    # gradient
    grad = np.zeros(n)
    grad[idxs[v1]] += (1.0 - h[idxs[v1] + k])  # d/dh[i] of h[i]*(1-h[i+k])
    grad[idxs[v2]] -= h[idxs[v2] - k]  # d/dh[i] of h[i-k]*(1-h[i])
    
    return c, grad

maximize t
subject to: tr(X * A_k) ≥ t for all k (where A_k is the shift matrix)
           X = h*h^T (rank constraint - makes it non-convex)
           0 ≤ h[i] ≤ 1
           sum h = n/2
           X ⪰ 0

If we drop the rank constraint, we get an SDP relaxation! This would give a lower bound on the optimal value (upper bound on C₅).

But solving an SDP with n=400 variables might be too slow. However, the structure of the shift matrices A_k is very special (they're circulant-like), so the SDP has a special structure that might be exploitable.

Actually, for a circulant matrix, the eigendecomposition is via the FFT. So the constraint tr(X * A_k) ≥ t for all k becomes a set of constraints on the FFT of x = vec(X). This might be solvable.

def bottleneck_optimize(h_init, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    
    idxs = np.arange(n)
    mid = n - 1
    
    while time.time() - start < time_limit:
        # Compute autocorrelation
        acorr = np.correlate(h, h, mode='full')
        # autocorr[k] for shift k corresponds to acorr[mid + k]
        # corr[k] = n/2 - autocorr[shift k]
        # So C5 = max_k corr[k] * dx = (n/2 - min_k autocorr[k]) * dx
        
        # Find the bottleneck (minimum autocorrelation)
        # Note: we should exclude lag 0 (which is sum h[i]^2)
        # Actually, for the overlap problem, all shifts matter including 0
        # corr[0] = sum h[i]*(1-h[i]) which is also part of C5
        
        min_acorr = np.min(acorr)
        bottleneck_idx = int(np.argmin(acorr))
        k_bottleneck = bottleneck_idx - mid
        
        # Try to increase autocorr at the bottleneck
        # autocorr[k] = sum_i h[i]*h[i+k]
        # d autocorr[k] / d h[i] = h[i+k] + h[i-k] (for valid indices)
        
        c_k, g_k = corr_and_grad(h, k_bottleneck)
        # g_k is the gradient of corr[k] = n/2 - autocorr[k]
        # So gradient of autocorr[k] = -g_k
        
        acorr_grad = -g_k
        acorr_grad -= np.mean(acorr_grad)
        
        # Take a step in the direction that increases the bottleneck autocorrelation
        # But check that it doesn't decrease other autocorrelations too much
        gnorm = np.linalg.norm(acorr_grad)
        if gnorm < 1e-12:
            break
        
        best_step = 0
        best_c5 = compute_c5(h)
        for lr in [0.1/gnorm, 0.05/gnorm, 0.01/gnorm, 0.5/gnorm]:
            h_trial = h + lr * acorr_grad
            h_trial = project_feasible(h_trial, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_step = lr
        
        if best_step > 0:
            h = h + best_step * acorr_grad
            h = project_feasible(h, target)
            new_c5 = compute_c5(h)
            if new_c5 < best_score - 1e-12:
                best_score = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Try random pair moves targeting the bottleneck
            v1 = (idxs + k_bottleneck >= 0) & (idxs + k_bottleneck < n)
            if np.sum(v1) > 0:
                # Find pairs where h[i] is high and h[i+k] is low (they contribute to low autocorr)
                products = h[idxs[v1]] * h[idxs[v1] + k_bottleneck]
                # We want to increase these products
                # Increase h[i+k] where h[i] is high but h[i+k] is low
                scores = h[idxs[v1]] * (1.0 - h[idxs[v1] + k_bottleneck])
                top_idx = np.argsort(scores)[-5:]
                for li in top_idx:
                    i = int(idxs[v1][li])
                    j = i + k_bottleneck
                    delta = 0.01 * h[i] * (1.0 - h[j])
                    if delta > 1e-8:
                        h[i] -= delta
                        h[j] += delta
                        h = project_feasible(h, target)
        
        if no_improve > 200:
            # Restart with perturbation
            h = best_h + rng.randn(n) * 0.02
            h = project_feasible(h, target)
            no_improve = 0
    
    return best_h, best_score

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def autocorr_and_grad_acorr(h, k):
    """Compute autocorrelation at shift k and its gradient."""
    n = len(h)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    
    # autocorr[k] = sum_i h[i] * h[i+k] for valid i
    ac = np.sum(h[idxs[v1]] * h[idxs[v1] + k])
    
    # gradient of autocorr[k] w.r.t. h
    grad = np.zeros(n)
    grad[idxs[v1]] += h[idxs[v1] + k]
    grad[idxs[v2]] += h[idxs[v2] - k]
    
    return ac, grad

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(h)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x, alpha):
        return smooth_max_obj(x, alpha)
    
    def _grad(x, alpha):
        return smooth_max_grad(x, alpha)
    
    def _cons_fun(x):
        return np.sum(x) - target
    
    def _cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_obj, h_cur, args=(alpha,), jac=lambda x, a: _grad(x, a),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def bottleneck_gradient_optimize(h_init, n, target, time_limit, seed=42):
    """Maximize min autocorrelation using targeted gradient steps."""
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr)) * (2.0/n)
        
        if c5 < best_score - 1e-12:
            best_score = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            h = best_h + rng.randn(n) * 0.015
            h = project_feasible(h, target)
            no_improve = 0
            continue
        
        # Find the shift that gives the worst (maximum) overlap
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        # Compute gradient to reduce corr at k_star
        # corr[k] = sum h[i]*(1-h[i+k]), gradient w.r.t. h[i]:
        # (1-h[i+k]) - h[i-k] (for valid indices)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Also consider the second-worst shift
        corr_sorted = np.sort(corr)[::-1]
        if len(corr_sorted) >= 2 and corr_sorted[1] > corr_sorted[0] * 0.95:
            # Multiple shifts are near the max, use smooth max gradient
            alpha = 50.0
            sm_grad = smooth_max_grad(h, alpha)
            sm_grad -= np.mean(sm_grad)
            grad = 0.5 * (grad / gnorm + sm_grad / max(np.linalg.norm(sm_grad), 1e-12)) * max(gnorm, np.linalg.norm(sm_grad))
            gnorm = np.linalg.norm(grad)
        
        # Line search
        best_step = 0
        best_c5_trial = c5
        for lr in [0.05/gnorm, 0.02/gnorm, 0.01/gnorm, 0.1/gnorm]:
            h_trial = h - lr * grad
            h_trial = project_feasible(h_trial, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5_trial:
                best_c5_trial = c5_trial
                best_step = lr
        
        if best_step > 0:
            h = h - best_step * grad
            h = project_feasible(h, target)
        else:
            # Try a targeted pair move
            v1_pairs = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.sum(v1_pairs) > 0:
                # Find the pair with the largest h[i]*(1-h[i+k])
                contribs = h[idxs[v1_pairs]] * (1.0 - h[idxs[v1_pairs] + k_star])
                top_n = min(10, len(contribs))
                top_idxs = np.argsort(contribs)[-top_n:]
                for ti in top_idxs:
                    i = int(idxs[v1_pairs][ti])
                    j = i + k_star
                    if 0 <= j < n:
                        # Try to increase h[j] (and decrease h[i])
                        delta = 0.02 * h[i] * (1.0 - h[j])
                        if delta > 1e-6:
                            h_trial = h.copy()
                            h_trial[i] -= delta
                            h_trial[j] += delta
                            h_trial = project_feasible(h_trial, target)
                            c5_trial = compute_c5(h_trial)
                            if c5_trial < c5:
                                h = h_trial
                                c5 = c5_trial
                                break
    
    return best_h, best_score

def screw_cd_optimize(h_init, n, target, time_limit, seed=42):
    """Use 3-cycle (screw) moves in addition to pair moves."""
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr)) * (2.0/n)
        
        if c5 < best_score - 1e-12:
            best_score = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            h = best_h + rng.randn(n) * 0.01
            h = project_feasible(h, target)
            no_improve = 0
            continue
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        # Gradient targeting worst shift
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        # Identify good candidates for decrease (high grad) and increase (low grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:30]
        high_set = sorted_idx[-30:]
        
        best_trial = c5
        best_move = None
        
        # Try pair moves
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                # Try a small transfer from i to j
                delta = min(h[i], 1.0 - h[j]) * 0.3
                if delta < 1e-8:
                    continue
                h_trial = h.copy()
                h_trial[i] -= delta
                h_trial[j] += delta
                h_trial = project_feasible(h_trial, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_trial:
                    best_trial = c5_trial
                    best_move = ('pair', i, j, delta)
        
        # Try screw moves: decrease h[i], increase h[j], and adjust h[k]
        # A screw move: h[i] -= d, h[j] += d (preserves sum)
        # But also try: h[i] -= d1, h[j] += d1, h[k] -= d2, h[l] += d2
        if no_improve > 50:
            # Try a 4-point move
            for _ in range(5):
                i = int(high_set[rng.randint(len(high_set))])
                j = int(low_set[rng.randint(len(low_set))])
                k2 = int(high_set[rng.randint(len(high_set))])
                l2 = int(low_set[rng.randint(len(low_set))])
                if len(set([i,j,k2,l2])) < 4:
                    continue
                d1 = min(h[i], h[k2]) * 0.2
                d2 = min(1.0-h[j], 1.0-h[l2]) * 0.2
                d = min(d1, d2)
                if d < 1e-8:
                    continue
                h_trial = h.copy()
                h_trial[i] -= d
                h_trial[k2] -= d
                h_trial[j] += d
                h_trial[l2] += d
                h_trial = project_feasible(h_trial, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_trial:
                    best_trial = c5_trial
                    best_move = ('quad', i, j, k2, l2, d)
        
        if best_move is not None:
            if best_move[0] == 'pair':
                _, i, j, d = best_move
                h[i] -= d
                h[j] += d
            else:
                _, i, j, k2, l2, d = best_move
                h[i] -= d
                h[k2] -= d
                h[j] += d
                h[l2] += d
            h = project_feasible(h, target)
    
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
