import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def frank_wolfe_step(h, n, target):
    """Frank-Wolfe style step: compute gradient at active constraint,
    solve LP subproblem, return the FW direction and target point."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star_idx = int(np.argmax(corr))
    mid = n - 1
    k_star = k_star_idx - mid
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    # Solve: min grad · h' s.t. 0 <= h' <= 1, sum(h') = target
    # Solution: sort grad, set h'=1 for smallest target values
    sorted_idx = np.argsort(grad)
    h_fw = np.zeros(n)
    h_fw[sorted_idx[:int(target)]] = 1.0
    # Handle fractional part
    if target != int(target):
        h_fw[sorted_idx[int(target)]] = target - int(target)
    return h_fw, grad

def fw_optimize(h_init, n, target, time_limit, max_iters=5000, gamma_schedule='adaptive'):

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    gamma = 1.0

            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx if (dx := 2.0/n) else 0
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        h_fw, grad = frank_wolfe_step(h, n, target)
        # Gap evaluation
        gap = np.dot(grad, h - h_fw)
        if gap < 1e-12:
            break
        # Adaptive step size

            gamma = 2.0 / (it + 2.0)
        # Try step

        h_trial = project_feasible(h_trial, target)

def block_cd(h_init, n, target, time_limit, block_size=10, max_sweeps=2000, seed=777):
    """Block coordinate descent: optimize a block of coordinates at a time."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 200:
            break
        
        # Select a random block of consecutive indices
        block_start = int(rng.randint(0, n - block_size + 1))
        block_idx = np.arange(block_start, block_start + block_size)
        h_block = h[block_idx].copy()
        block_sum = np.sum(h_block)
        
        # Optimize the block using SLSQP on the smooth objective
        # (restricted to the block)
        def block_obj(x):
            h_temp = h.copy()
            h_temp[block_idx] = x
            c = np.correlate(h_temp, 1.0 - h_temp, mode='full')
            c_max = np.max(c)
            alpha = 100.0
            shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
            return c_max + np.log(np.sum(np.exp(shifted))) / alpha
        
        # ... this is too slow per iteration. Let me skip this approach.
        # Instead, do a simple gradient step on the block
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        idxs = np.arange(n)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        
        # Projected gradient step on the block
        lr = 0.01
        h_block_new = h_block - lr * grad[block_idx]
        h_block_new = np.clip(h_block_new, 0.0, 1.0)
        # Fix sum
        diff = block_sum - np.sum(h_block_new)
        mask = (h_block_new < 1.0 - 1e-12) & (h_block_new > 1e-12)
        if mask.any():
            h_block_new[mask] += diff / mask.sum()
            h_block_new = np.clip(h_block_new, 0.0, 1.0)
        
        h[block_idx] = h_block_new
        h = project_feasible(h, target)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):

    if n_src == n_dst:

    h_out = np.interp(x_new, x_old, h_src)

            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
