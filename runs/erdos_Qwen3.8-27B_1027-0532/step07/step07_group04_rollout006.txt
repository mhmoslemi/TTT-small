        for _ in range(extra):
            i = int(high_set[rng_local.randint(len(high_set))])
            j = int(low_set[rng_local.randint(len(low_set))])
            if i != j:
                candidate_pairs.append((i, j))

        best_trial_c5 = c5_val
        best_move = None
        for (i, j) in candidate_pairs:
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-12:
                continue
            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(14):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, delta_star = best_move
            h[i] -= delta_star
            h[j] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [i, j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5


def sa_v2(h_init, n, target, time_limit, max_steps=30000,
          t_start=0.03, t_end=0.0002, n_cand=50):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(999)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    consecutive_no_improve = 0

    def update_change_row(change_mat, p):
        change_mat[p, :] = 0.0
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change_mat[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change_mat[p, neg_mask] -= h[p - shifts[neg_mask]]

    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        update_change_row(change, p)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 400:
            break
        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue
        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue
        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx
        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-12:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature, 1e-15)):
                accept = True
        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i)
            update_change_row(change, j)
            cur_c5 = approx_new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - 15.0 - (time.time() - start_time)

    n = 600
    target = n / 2.0
    h_init = interpolate_to(initial_h_values, n, target)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    if tl() > 80:
        h1, c1 = gradient_method(h_init, n, target, min(70, tl() * 0.9),
                                 alphas=(20, 50, 100, 200, 500, 1000, 2000))
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    if tl() > 100:
        h2, c2 = fast_cd_v3(best_h, n, target, min(90, tl() * 0.8),
                            max_sweeps=20000, n_cand=40, max_no_improve=200)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    if tl() > 80:
        h3, c3 = sa_v2(best_h, n, target, min(70, tl() * 0.8),
                       max_steps=25000, t_start=0.03, t_end=0.0002, n_cand=50)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    if tl() > 60:
        h4, c4 = fast_cd_v3(best_h, n, target, min(50, tl() * 0.8),
                            max_sweeps=15000, n_cand=30, max_no_improve=150)
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    n2 = 800
    target2 = n2 / 2.0
    if tl() > 150:
        h5 = interpolate_to(best_h, n2, target2)
        c5_init = compute_c5(h5)
        if c5_init < best_c5:
            best_c5, best_h = c5_init, h5
        h6, c6 = gradient_method(h5, n2, target2, min(60, tl() * 0.5),
                                 alphas=(50, 100, 300, 1000, 3000, 10000))
        if c6 < best_c5:
            best_c5, best_h = c6, h6
        if tl() > 60:
            h7, c7 = fast_cd_v3(best_h, n2, target2, min(55, tl() * 0.7),
                                max_sweeps=10000, n_cand=35, max_no_improve=150)
            if c7 < best_c5:
                best_c5, best_h = c7, h7
        if tl() > 40:
            h8, c8 = sa_v2(best_h, n2, target2, min(35, tl() * 0.7),
                           max_steps=15000, t_start=0.02, t_end=0.0001, n_cand=40)
            if c8 < best_c5:
                best_c5, best_h = c8, h8
        n = n2
        target = target2

    if tl() > 200:
        n3 = 1000
        target3 = n3 / 2.0
        h9 = interpolate_to(best_h, n3, target3)
        c9_init = compute_c5(h9)
        if c9_init < best_c5:
            best_c5, best_h = c9_init, h9
        h10, c10 = gradient_method(h9, n3, target3, min(80, tl() * 0.5),
                                   alphas=(50, 200, 500, 2000, 5000))
        if c10 < best_c5:
            best_c5, best_h = c10, h10
        if tl() > 60:
            h11, c11 = fast_cd_v3(best_h, n3, target3, min(55, tl() * 0.7),
                                  max_sweeps=8000, n_cand=30, max_no_improve=120)
            if c11 < best_c5:
                best_c5, best_h = c11, h11
        n = n3
        target = target3

    if tl() > 30:
        h12, c12 = fast_cd_v3(best_h, n, target, min(25, tl() * 0.8),
                              max_sweeps=5000, n_cand=40, max_no_improve=100)
        if c12 < best_c5:
            best_c5, best_h = c12, h12

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
