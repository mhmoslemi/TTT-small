One more thought: instead of the bisection for step size, I can use a cheaper approximation. The function f(δ) = max(corr + δ*g) is piecewise linear and convex in δ. The minimum is at the point where the subgradient crosses 0, i.e., where the maximum switches from one element to another.

Cheaper approach: 
- Compute the current max and the index where it's achieved
- Compute g at that index
- If g[active] < 0, the max will decrease as δ increases
- The max decrease continues until another element catches up
- The crossover δ* = (corr[other] - corr[active]) / (g[active] - g[other])

This gives an O(n) step size computation (find the second max and the crossover).

Actually, the simplest good approach:
1. Compute d_max = max(corr) and the active set A = {i : corr[i] > d_max - eps}
2. Compute the maximum of g over A: g_max_A = max(g[i] for i in A)
3. If g_max_A >= 0, try the other direction (or skip)
4. The step is limited by: for each i not in A, corr[i] + δ*g[i] <= d_max
   → δ <= (d_max - corr[i]) / g[i] for g[i] > 0
5. Also bounded by [δ_lo, δ_hi]
6. δ* = min(δ_hi, min over i not in A with g[i] > 0 of (d_max - corr[i]) / g[i])

But this is only valid if g is constant on A (which it's not in general). For a tighter bound, use:
δ* = min(δ_hi, min over all i of (d_max - corr[i]) / g[i] for g[i] > 0)

This is O(n) and gives an upper bound on the optimal step. The actual optimal step might be smaller (if g varies within A), but this is a good approximation.

def compute_change_diff(h, i, j, n):
    """Compute change[j, m] - change[i, m] for all m, vectorized."""
    n_corr = 2 * n - 1
    k_vals = np.arange(n_corr) - (n - 1)  # all possible shifts
    
    # change[p, m] = (1-h[p+k]) * [0<=p+k<n] - h[p-k] * [0<=p-k<n]
    # where k = k_vals[m]
    
    result = np.zeros(n_corr)
    
    # For index j:
    jk = j + k_vals  # p+k for all k
    mask1 = (jk >= 0) & (jk < n)
    result[mask1] += (1.0 - h[jk[mask1]])
    
    jkm = j - k_vals  # p-k for all k
    mask2 = (jkm >= 0) & (jkm < n)
    result[mask2] -= h[jkm[mask2]]
    
    # For index i (subtract):
    ik = i + k_vals
    mask3 = (ik >= 0) & (ik < n)
    result[mask3] -= (1.0 - h[ik[mask3]])
    
    ikm = i - k_vals
    mask4 = (ikm >= 0) & (ikm < n)
    result[mask4] += h[ikm[mask4]]
    
    return result

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

        grad /= float(n_shifts_grad)
        grad -= np.mean(grad)

        progress = min(float(sweep) / 5000.0, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_l = np.array([change_at_shift(h, int(p), k_star, n) for p in low_set])
        ch_h = np.array([change_at_shift(h, int(p), k_star, n) for p in high_set])
        score_mat = ch_l[None, :] - ch_h[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = compute_change_diff(h, i, j, n)

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def multi_cd_optimize(h_init, n, target, time_limit,
                      n_cand=50, top_k_eval=30, n_shifts=4,
                      n_bisect=5, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    while True:
        if time.time() - start > time_limit:
            break
        corr = fft_correlate(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_shifts = np.argsort(corr)[-n_shifts:]
        k_star_idx = int(top_shifts[-1])
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[int(s_idx)]
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_l = np.array([change_at_shift(h, int(p), k_star, n) for p in low_set])
        ch_h = np.array([change_at_shift(h, int(p), k_star, n) for p in high_set])
        score_mat = ch_l[None, :] - ch_h[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            g = compute_change_diff(h, i, j, n)
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def projected_subgrad(h_init, n, target, time_limit, n_shifts=5):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    step = 0.02
    momentum = 0.0
    v = np.zeros(n)

    for step_i in range(200000):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break

        corr = fft_correlate(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            step = min(step * 1.05, 0.05)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                step *= 0.8
                v *= 0.5

        top_shifts = np.argsort(corr)[-n_shifts:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(n_shifts)

        v = 0.3 * v - step * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < compute_c5(h) + 1e-12:
            h = h_new
        else:
            v *= 0.5

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha, _g_target
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    tl = _time_left(start_time, budget_s)
    if tl > 50:
        h_pol = slsqp_optimize(best_h, target0, [200, 1000, 5000], 300, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    tl = _time_left(start_time, budget_s)
    if tl > 500:
        n1 = 4000
        target1 = float(n1) / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
        n_work = n1
        target_work = target1

        tl = _time_left(start_time, budget_s)
        cd_time1 = min(500, tl * 0.55)
        h_cd, c5_cd = cd_optimize(h1, n_work, target_work, cd_time1,
                                  n_cand_start=20, n_cand_end=80,
                                  top_k_eval=25, n_shifts_grad=5,
                                  n_bisect=5, max_no_improve=250)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

        tl = _time_left(start_time, budget_s)
        if tl > 100:
            h_pol2 = slsqp_optimize(best_h, target_work, [1000, 5000], 250, 1e-14)
            sc2 = compute_c5(h_pol2)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

        tl = _time_left(start_time, budget_s)
        if tl > 200:
            cd_time2 = min(180, tl * 0.6)
            h_cd2, c5_cd2 = cd_optimize(best_h, n_work, target_work, cd_time2,
                                        n_cand_start=30, n_cand_end=100,
                                        top_k_eval=30, n_shifts_grad=7,
                                        n_bisect=6, max_no_improve=200)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

        tl = _time_left(start_time, budget_s)
        if tl > 80:
            h_ms, c5_ms = multi_cd_optimize(best_h, n_work, target_work,
                                            min(70, tl * 0.7),
                                            n_cand=60, top_k_eval=30,
                                            n_shifts=4, n_bisect=5,
                                            max_no_improve=200)
            if c5_ms < best_score:
                best_score = c5_ms
                best_h = h_ms

    tl = _time_left(start_time, budget_s)
    if tl > 60:
        h_pol3 = slsqp_optimize(best_h, len(best_h) / 2.0, [5000, 20000], 200, 1e-15)
        sc3 = compute_c5(h_pol3)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    tl = _time_left(start_time, budget_s)
    if tl > 60 and len(best_h) < 2000:
        n2 = 4000
        target2 = float(n2) / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2b = compute_c5(h2)
        if sc2b < best_score:
            best_score = sc2b
            best_h = h2
        tl = _time_left(start_time, budget_s)
        if tl > 40:
            h_cd3, c5_cd3 = cd_optimize(best_h, n2, target2,
                                        min(35, tl * 0.7),
                                        n_cand_start=30, n_cand_end=70,
                                        top_k_eval=25, n_shifts_grad=5,
                                        n_bisect=5, max_no_improve=150)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

    tl = _time_left(start_time, budget_s)
    if tl > 30:
        rng = np.random.RandomState(seed)
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for noise_scale in [0.03, 0.08]:
            tl = _time_left(start_time, budget_s)
            if tl < 25:
                break
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, target_cur)
            h_ref = slsqp_optimize(h_pert, target_cur, [1000, 5000], 200, 1e-14)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
