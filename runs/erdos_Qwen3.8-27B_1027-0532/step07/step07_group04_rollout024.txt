            if time.time() - start_time > time_limit:
                return best_h, best_score
            obj, grad = smooth_max_and_grad(y, alpha, n_active)
            x_new = project_feasible(y - step * grad, target)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - h)
            y = y_new
            h = x_new
            t_mom = t_new
            step *= momentum_decay
            score = compute_c5(h)
            if score < best_score - 1e-14:
                best_score = score
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 3000:
                break
        h = best_h.copy()
    return best_h, best_score

def momentum_gd(h_init, n, target, time_limit, start_time,
                alphas=None, step_base=0.015, max_iter=8000,
                n_active=25, mom=0.85):
    if alphas is None:
        alphas = [50, 100, 300]
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    for alpha in alphas:
        v = np.zeros(n)
        step = step_base
        no_improve = 0
        for it in range(max_iter):
            if time.time() - start_time > time_limit:
                return best_h, best_score
            _, grad = smooth_max_and_grad(h, alpha, n_active)
            v = mom * v - step * grad
            h_new = project_feasible(h + v, target)
            h = h_new
            step *= 0.9998
            score = compute_c5(h)
            if score < best_score - 1e-14:
                best_score = score
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 4000:
                break
        h = best_h.copy()
    return best_h, best_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def cd_refine(h_init, n, target, time_limit, max_sweeps=30000,
              n_cand_start=30, n_cand_end=80, max_no_improve=400,
              n_shifts_weight=3):
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        top_shifts = np.argsort(corr)[-n_shifts_weight:]
        c_max = corr[np.argmax(corr)]
        if c_max > 1e-10:
            weights = np.exp(-5.0 * (c_max - corr[top_shifts]) / c_max)
        else:
            weights = np.ones(len(top_shifts))
        weights = weights / np.sum(weights)
        grad = np.zeros(n)
        for wi, s_idx in enumerate(top_shifts):
            k = int(s_idx) - mid
            lo1 = max(0, k)
            hi1 = min(n, n + k)
            if lo1 < hi1:
                grad[lo1:hi1] += weights[wi] * (1.0 - h[lo1 - k:hi1 - k])
            lo2 = max(0, -k)
            hi2 = min(n, n - k)
            if lo2 < hi2:
                grad[lo2:hi2] -= weights[wi] * h[lo2 + k:hi2 + k]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def sa_escape(h_init, n, target, time_limit, max_steps=20000,
              t_start=0.03, t_end=0.0002, n_cand=50):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    rng = np.random.RandomState(77)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 500:
            break
        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        lo1 = max(0, k_star)
        hi1 = min(n, n + k_star)
        lo2 = max(0, -k_star)
        hi2 = min(n, n - k_star)
        grad = np.zeros(n)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1 - k_star:hi1 - k_star])
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2 + k_star:hi2 + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue
        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue
        g = change[j] - change[i]
        approx_new_c5 = float(np.max(corr + delta * g)) * dx
        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature, 1e-15)):
                accept = True
        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            cur_c5 = approx_new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_quick(h_init, n, target, time_limit, start_time, alphas=None, maxiter=500):
    if alphas is None:
        alphas = [200, 1000]
    dx = 2.0 / n
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def obj_fn(x):
            obj, _ = smooth_max_and_grad(x, alpha, 15)
            return obj * dx
        grad_cache = {}
        def jac_fn(x):
            _, g = smooth_max_and_grad(x, alpha, 15)
            return g * dx
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj_fn, h_cur, jac=jac_fn, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(budget_s - 25.0 - (time.time() - start_time), 0.0)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    # Phase 1: FISTA on initial grid
    if tleft() > 120:
        h_f, sc = fista_phase(best_h, cur_n, cur_target,
                              time_limit=min(100, tleft() * 0.7),
                              start_time=start_time,
                              alphas=[30, 80, 200, 500],
                              step_base=0.015, max_iter=6000, n_active=20)
        if sc < best_score:
            best_score = sc
            best_h = h_f

    # Phase 2: Momentum GD on initial grid
    if tleft() > 100:
        h_g, sc = momentum_gd(best_h, cur_n, cur_target,
                              time_limit=min(80, tleft() * 0.7),
                              start_time=start_time,
                              alphas=[100, 300, 800],
                              step_base=0.012, max_iter=5000, n_active=20, mom=0.8)
        if sc < best_score:
            best_score = sc
            best_h = h_g

    # Phase 3: CD refinement on initial grid
    if tleft() > 80:
        h_cd, sc = cd_refine(best_h, cur_n, cur_target,
                             time_limit=min(60, tleft() * 0.7),
                             max_sweeps=20000, n_cand_start=25,
                             n_cand_end=60, max_no_improve=300)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 4: Upscale to n=800
    if tleft() > 250:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_f1, sc1 = fista_phase(h1, n1, target1,
                                 time_limit=min(120, tleft() * 0.5),
                                 start_time=start_time,
                                 alphas=[50, 150, 400, 1000],
                                 step_base=0.010, max_iter=6000, n_active=25)
        if sc1 < best_score:
            best_score = sc1
            best_h = h_f1
            cur_n = n1
            cur_target = target1
        else:
            best_h = h1
            cur_n = n1
            cur_target = target1

    # Phase 5: CD on n=800
    if tleft() > 120:
        h_cd1, sc1 = cd_refine(best_h, cur_n, cur_target,
                               time_limit=min(100, tleft() * 0.6),
                               max_sweeps=25000, n_cand_start=30,
                               n_cand_end=70, max_no_improve=400)
        if sc1 < best_score:
            best_score = sc1
            best_h = h_cd1

    # Phase 6: SA escape on n=800
    if tleft() > 80:
        h_sa, sc_sa = sa_escape(best_h, cur_n, cur_target,
                                time_limit=min(60, tleft() * 0.6),
                                max_steps=15000, t_start=0.025,
                                t_end=0.0001, n_cand=45)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    # Phase 7: FISTA again after SA
    if tleft() > 80:
        h_f2, sc2 = fista_phase(best_h, cur_n, cur_target,
                                 time_limit=min(60, tleft() * 0.7),
                                 start_time=start_time,
                                 alphas=[200, 600, 1500],
                                 step_base=0.008, max_iter=4000, n_active=25)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_f2

    # Phase 8: Try n=1000
    if tleft() > 300:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_f3, sc3 = fista_phase(h2, n2, target2,
                                 time_limit=min(130, tleft() * 0.4),
                                 start_time=start_time,
                                 alphas=[50, 150, 400],
                                 step_base=0.008, max_iter=5000, n_active=30)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_f3
            cur_n = n2
            cur_target = target2
        else:
            cur_n = n2
            cur_target = target2
            best_h = h2

        if tleft() > 80:
            h_cd2, sc_cd2 = cd_refine(best_h, cur_n, cur_target,
                                       time_limit=min(70, tleft() * 0.6),
                                       max_sweeps=15000, n_cand_start=30,
                                       n_cand_end=65, max_no_improve=350)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_h = h_cd2

    # Phase 9: SLSQP polish
    if tleft() > 60:
        h_sp, sc_sp = slsqp_quick(best_h, cur_n, cur_target,
                                   time_limit=min(40, tleft() * 0.7),
                                   start_time=start_time,
                                   alphas=[500, 2000], maxiter=300)
        if sc_sp < best_score:
            best_score = sc_sp
            best_h = h_sp

    # Phase 10: Final FISTA + CD
    if tleft() > 50:
        h_ff, sc_ff = fista_phase(best_h, cur_n, cur_target,
                                   time_limit=min(30, tleft() * 0.7),
                                   start_time=start_time,
                                   alphas=[1000, 3000],
                                   step_base=0.005, max_iter=3000, n_active=30)
        if sc_ff < best_score:
            best_score = sc_ff
            best_h = h_ff

    if tleft() > 30:
        h_cf, sc_cf = cd_refine(best_h, cur_n, cur_target,
                                 time_limit=min(20, tleft() * 0.8),
                                 max_sweeps=8000, n_cand_start=40,
                                 n_cand_end=60, max_no_improve=200)
        if sc_cf < best_score:
            best_score = sc_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
