from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha=1000):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

    h_out = np.interp(x_new_mid, x_old_mid, h_src)

def compute_grad_matrix(h, n):
    """Compute G[p, l] = ∂corr_l/∂h[p] for all p, l."""
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    G = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        G[p, pos] += (1.0 - h[p + shifts[pos]])
        G[p, neg] -= h[p - shifts[neg]]
    return G

def update_grad_row(G, p, h, n, shifts):
    G[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    G[p, pos] += (1.0 - h[p + shifts[pos]])
    G[p, neg] -= h[p - shifts[neg]]
    return G

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_grad_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    G = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        G[p, pos] += (1.0 - h[p + shifts[pos]])
        G[p, neg] -= h[p - shifts[neg]]
    return G

def _update_G_row(G, p, h, n, shifts):
    G[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    G[p, pos] += (1.0 - h[p + shifts[pos]])
    G[p, neg] -= h[p - shifts[neg])
    return G

def pair_cd(h_init, n, target, time_limit, n_cand=30, max_sweeps=30000,
            max_no_improve=800, restart_every=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    shifts = np.arange(-(n - 1), n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    rng = np.random.RandomState(99)
    
    G = compute_grad_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        if no_improve > restart_every and no_improve % 100 == 0:
            h = best_h + rng.randn(n) * 0.015
            h = project_feasible(h, target)
            G = compute_grad_matrix(h, n)
            no_improve = 0
            continue
        
        k_star = int(np.argmax(corr)) - mid
        grad = G[:, k_star].copy()
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = best_j = -1
        best_delta = 0.0
        
        n_h = min(len(high_set), 25)
        n_l = min(len(low_set), 25)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            Gi = G[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = G[j] - Gi
                a_d, b_d = delta_lo, delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= f_mid - 1e-14
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i, best_j, best_delta = i, j, delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            _update_G_row(G, best_i, h, n, shifts)
            _update_G_row(G, best_j, h, n, shifts)
    
    return best_h, best_c5


def _triple_subproblem(corr, u, v, h_i, h_j, h_k):
    """Solve the 2D subproblem for a triple move.
    d_i = -(a+b), d_j = a, d_k = b
    Minimize max_l (corr_l + a*u_l + b*v_l)
    s.t. box constraints.
    """
    # Feasible region for (a, b):
    # -h[i] ≤ -(a+b) ≤ 1-h[i]  →  h[i]-1 ≤ a+b ≤ h[i]
    # -h[j] ≤ a ≤ 1-h[j]
    # -h[k] ≤ b ≤ 1-h[k]
    
    a_lo = max(-h_j, h_i - 1 - (1 - h_k))  # from a+b ≥ h_i-1 and b ≤ 1-h_k
    a_hi = min(1 - h_j, h_i - (-h_k))  # from a+b ≤ h_i and b ≥ -h_k
    
    # Simplify:
    # a ≥ -h_j
    # a ≤ 1-h_j
    # a ≥ h_i - 1 - b  → for each b, a ≥ h_i - 1 - b
    # a ≤ h_i - b  → for each b, a ≤ h_i - b
    # b ≥ -h_k
    # b ≤ 1-h_k
    # a+b ≥ h_i - 1
    # a+b ≤ h_i
    
    # The feasible region is a polygon. Let me just use scipy with bounds + constraint.
    
    def _obj_2d(x):
        a, b = x
        di = -(a + b)
        # Check feasibility
        if di < -h_i - 1e-10 or di > 1 - h_i + 1e-10:
            return 1e10
        if b < -h_k - 1e-10 or b > 1 - h_k + 1e-10:
            return 1e10
        vals = corr + a * u + b * v
        return float(np.max(vals))
    
    # Use bounds
    a_min = max(-h_j, h_i - 1 - (1 - h_k))
    a_max = min(1 - h_j, h_i - (-h_k))
    b_min = -h_k
    b_max = 1 - h_k
    
    if a_min >= a_max or b_min >= b_max:
        return None
    
    res = minimize(_obj_2d, [0.0, 0.0], method='Nelder-Mead',
                   options={'xatol': 1e-10, 'fatol': 1e-12, 'maxiter': 500})
    
    a_star, b_star = res.x
    # Project to feasible
    di = -(a_star + b_star)
    if di < -h_i: a_star = h_i - 1 - b_star  # clamp
    if di > 1 - h_i: a_star = h_i - b_star
    
    return a_star, b_star, di


def triple_cd(h_init, n, target, time_limit, n_cand=20, max_sweeps=10000,
