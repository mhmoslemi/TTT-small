    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12

            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    _g_target = target
    bounds = [(0.0, 1.0)] * n

            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

        M = float(np.max(corr))

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = shift_weights.sum()
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            if w < 1e-15:
                continue
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        moves_made = 0
        for _ in range(multi_move):
            if time.time() - start > time_limit:
                break

            best_trial = M
            best_i = -1
            best_j = -1
            best_delta = 0.0

            n_h = len(high_set)
            n_l = len(low_set)

            for ii in range(n_h):
                i = int(high_set[ii])
                hi = h[i]
                if hi < 1e-12:
                    continue
                for jj in range(n_l):
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    lj = h[j]
                    if lj > 1.0 - 1e-12:
                        continue

                    delta_hi = min(hi, 1.0 - lj)
                    if delta_hi < 1e-12:
                        continue

                    g = change[j] - change[i]
                    pos_g = g > 1e-14
                    if pos_g.any():
                        ratios = (M - corr[pos_g]) / g[pos_g]
                        valid = ratios >= 0.0
                        if valid.any():
                            delta_max_corr = float(np.min(ratios[valid]))
                        else:
                            delta_max_corr = 0.0
                    else:
                        delta_max_corr = delta_hi

                    delta_opt = min(delta_hi, delta_max_corr) * 0.9999
                    if delta_opt < 1e-12:
                        continue

                    new_max = float(np.max(corr + delta_opt * g))

                    accept = False
                    if new_max < best_trial - 1e-12:
                        accept = True
                    elif temp > 1e-10 and new_max < M + 1e-10:
                        delta_obj = (new_max - M) * 2000.0
                        if np.random.random() < np.exp(-delta_obj / temp):
                            accept = True

                    if accept and new_max < best_trial + 1e-10:
                        best_trial = new_max
                        best_i = i
                        best_j = j
                        best_delta = delta_opt

            if best_i >= 0:
                h[best_i] -= best_delta
                h[best_j] += best_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, best_i, h, n, shifts)
                update_change_row(change, best_j, h, n, shifts)
                moves_made += 1
                temp *= anneal_decay
            else:
                break

        if moves_made == 0:
            temp = max(temp * 0.96, 1e-10)

    return best_h, best_c5


def subgradient_momentum(h_init, n, target, time_limit, n_steps=50000,
                         step_scale=0.008, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        top_k = 5
        top_shifts = np.argsort(corr)[-top_k:]
        top_corr = corr[top_shifts]
        sw = top_corr / top_corr.sum()

        grad = np.zeros(n)
        for si in range(top_k):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = sw[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        v = momentum * v - eta * grad
        h_try = h + v
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
        else:
            v = 0.5 * v
            eta *= 0.9
            h_try2 = project_feasible(h - eta * grad, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2

        if step % 1000 == 0 and step > 0:
            eta = max(step_scale * 0.5, step_scale * (1.0 - step / (2.0 * n_steps)))

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, n_src - 1, n_dst)
    h_out = np.interp(x_new, np.arange(n_src), h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    if tleft() > 60:
        h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                   [100, 500, 2000], 400)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_cd, c5_cd = cd_analytical(best_h, cur_n, cur_target,
                                    time_limit=min(150, tleft() * 0.6),
                                    max_sweeps=80000,
                                    n_cand_start=50, n_cand_end=130,
                                    n_shifts=5, max_no_improve=400,
                                    anneal_temp=0.015, anneal_decay=0.9997,
                                    multi_move=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                   [500, 2000, 8000], 600)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_cd2, c5_cd2 = cd_analytical(best_h, cur_n, cur_target,
                                      time_limit=min(100, tleft() * 0.6),
                                      max_sweeps=60000,
                                      n_cand_start=60, n_cand_end=150,
                                      n_shifts=7, max_no_improve=300,
                                      anneal_temp=0.005, anneal_decay=0.9998,
                                      multi_move=2)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 100:
        h_sg, c5_sg = subgradient_momentum(best_h, cur_n, cur_target,
                                           time_limit=min(60, tleft() * 0.5),
                                           n_steps=30000, step_scale=0.006,
                                           momentum=0.85)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 250:
            h_cd1, c5_cd1 = cd_analytical(h1, n1, target1,
                                          time_limit=min(200, tleft() * 0.65),
                                          max_sweeps=50000,
                                          n_cand_start=30, n_cand_end=100,
                                          n_shifts=4, max_no_improve=350,
                                          anneal_temp=0.01, anneal_decay=0.9997,
                                          multi_move=2)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                cur_n = n1
                cur_target = target1
            else:
                sc1 = compute_c5(h1)
                if sc1 < best_score:
                    best_score = sc1
                    best_h = h1
                    cur_n = n1
                    cur_target = target1

    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                   [2000, 10000], 800)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_cd3, c5_cd3 = cd_analytical(best_h, cur_n, cur_target,
                                      time_limit=min(100, tleft() * 0.6),
                                      max_sweeps=50000,
                                      n_cand_start=50, n_cand_end=140,
                                      n_shifts=6, max_no_improve=300,
                                      anneal_temp=0.003, anneal_decay=0.9999,
                                      multi_move=3)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.06, 0.1, 0.15]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(cur_n) * noise_scale
            h_pert = project_feasible(h_pert, cur_target)
            h_ref, sc = slsqp_optimize(h_pert, cur_n, cur_target,
                                       [500, 3000], 400)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 60:
        h_cd4, c5_cd4 = cd_analytical(best_h, cur_n, cur_target,
                                      time_limit=min(40, tleft() * 0.6),
                                      max_sweeps=30000,
                                      n_cand_start=70, n_cand_end=160,
                                      n_shifts=6, max_no_improve=200,
                                      anneal_temp=0.0, anneal_decay=1.0,
                                      multi_move=4)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    if tleft() > 40:
        h_final, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                     [5000, 20000], 1000)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
