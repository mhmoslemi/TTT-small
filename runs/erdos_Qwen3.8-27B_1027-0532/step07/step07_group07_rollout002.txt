def enhanced_pair_cd(h_init, n, target, time_limit,

        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Use top n_shifts to compute gradient
        top_shifts = np.argsort(corr)[-n_shifts:]
        k_star_idx = int(top_shifts[-1])
        
        # Weighted gradient using correlation values as weights
        grad = np.zeros(n)
        total_w = 0.0
        for s_idx in top_shifts:
            w = corr[s_idx]
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
            total_w += w
        if total_w > 0:
            grad /= total_w
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Score based on active shift
        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]
        
        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval - 1)[:n_to_eval]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            
            g = change[j] - change[i]
            
            # Binary search for optimal delta
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g

                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)

                continue
            
            approx_max = np.max(corr + delta_star * g) * dx

                best_j = j

            h = np.clip(h, 0.0, 1.0)

Let me also think about **4-point moves**: instead of moving mass from i to j, move mass from i to j AND from k to l simultaneously. This is a 2D move in the space of (delta1, delta2). The feasible region is a rectangle (or box), and we can optimize over both deltas.

For a 4-point move (i→j, k→l):
- delta1 moves from i to j
- delta2 moves from k to l
- The effect on correlation at shift s is: delta1 * (change[j,s] - change[i,s]) + delta2 * (change[l,s] - change[k,s])
- We want to minimize max_s (corr[s] + delta1 * g1[s] + delta2 * g2[s])

This is a 2D convex optimization problem (minimax of affine functions), which can be solved efficiently.
