import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    n = len(h)
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(hp)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(hp)
    if abs(diff) > 1e-13:
        active = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        if active.any():
            hp[active] += diff / active.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def get_grad_direct(h, weights, n):
    idxs = np.arange(n)
    mid = n - 1
    grad = np.full(n, 1.0)
    active_k = np.where(weights > 1e-14)[0]
    for k_idx in active_k:
        w = weights[k_idx]
        k = int(k_idx) - mid
        pos = idxs + k
        vp = (pos >= 0) & (pos < n)
        neg = idxs - k
        vn = (neg >= 0) & (neg < n)
        grad[vp] -= w * h[pos[vp]]
        grad[vn] -= w * h[neg[vn]]
    return grad

def fista_optimize(h_init, n, target, time_limit, alpha_start=20, alpha_end=200000,
                   n_alpha_steps=12, max_iter_per_alpha=100000, restart_every=80):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t_mom = 1.0

    alphas = np.geomspace(alpha_start, alpha_end, n_alpha_steps)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        t_mom = 1.0
        h_prev = h.copy()
        no_improve = 0
        step = 1.0 / (2.0 * alpha + 1.0)

        for it in range(max_iter_per_alpha):
            if time.time() - start > time_limit:
                break
            if it % 100 == 0 and time.time() - start > time_limit:
                break

            corr = get_corr(h)
            c_max = np.max(corr)
            c5_val = c_max * dx

            if c5_val < best_score - 1e-15:
                best_score = c5_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 3000:
                    if it > 50:
                        break

            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            w = np.exp(shifted)
            ws = np.sum(w)
            if ws < 1e-200:
                w = np.zeros_like(corr)
                w[int(np.argmax(corr))] = 1.0
            else:
                w /= ws

            grad = get_grad_direct(h, w, n)

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            mu = (t_mom - 1.0) / t_new
            y = h + mu * (h - h_prev)

            h_new = project_box_sum(y - step * grad, target)
            h_prev = h
            h = h_new
            t_mom = t_new

            if it % restart_every == restart_every - 1:
                t_mom = 1.0
                h_prev = h.copy()

    return best_h, best_score

def pair_cd(h_init, n, target, time_limit, max_sweeps=20000, n_cand=50, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    def compute_chg(p, h):
        chg = np.zeros(n_corr)
        pos = p + shifts
        vp = (pos >= 0) & (pos < n)
        chg[vp] += (1.0 - h[pos[vp]])
        neg = p - shifts
        vn = (neg >= 0) & (neg < n)
        chg[vn] -= h[neg[vn]]
        return chg

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if sweep % 200 == 0 and time.time() - start > time_limit:
            break

        corr = get_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star = int(np.argmax(corr)) - mid

        grad = np.zeros(n)
        pos = idxs + k_star
        vp = (pos >= 0) & (pos < n)
        neg = idxs - k_star
        vn = (neg >= 0) & (neg < n)
        grad[vp] += (1.0 - h[pos[vp]])
        grad[vn] -= h[neg[vn]]
        grad -= np.mean(grad)

        nc = min(n_cand, n // 4)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:nc]
        high_set = sorted_idx[-nc:]

        best_trial = c5_val
        best_i, best_j, best_delta = -1, -1, 0.0

        for ii in range(nc):
            i = int(high_set[ii])
            chg_i = compute_chg(i, h)
            for jj in range(nc):
                j = int(low_set[jj])
                if i == j:
                    continue
                dlo = max(h[i] - 1.0, -h[j])
                dhi = min(h[i], 1.0 - h[j])
                if dhi - dlo < 1e-13:
                    continue
                chg_j = compute_chg(j, h)
                g = chg_j - chg_i
                a, b = dlo, dhi
                for _ in range(14):
                    m = 0.5 * (a + b)
                    vals = corr + m * g
                    fm = np.max(vals)
                    am = vals >= fm - 1e-15
                    if am.any():
                        fp = np.max(g[am])
                    else:
                        fp = 0.0
                    if fp > 1e-12:
                        b = m
                    else:
                        a = m
                ds = 0.5 * (a + b)
                if abs(ds) < 1e-13:
                    continue
                amax = np.max(corr + ds * g) * dx
                if amax < best_trial - 1e-15:
                    best_trial = amax
                    best_i, best_j, best_delta = i, j, ds

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, target)

    return best_h, best_c5
