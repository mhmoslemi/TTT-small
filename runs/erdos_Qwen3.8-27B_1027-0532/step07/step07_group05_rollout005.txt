    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    # Project h to {0 ≤ x ≤ 1, sum(x) = target_sum}
    h = np.clip(h.astype(float), 0.0, 1.0)

        else:

        if not mask.any():

        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_eq_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, n - p)
        change[p, mid + lags] += (1.0 - h[p + lags])
        if p > 0:
            lags2 = np.arange(-p, 0)
            change[p, mid + lags2] -= h[p + lags2]
    return change


def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, n - p)
    change[p, mid + lags] += (1.0 - h[p + lags])
    if p > 0:
        lags2 = np.arange(-p, 0)
        change[p, mid + lags2] -= h[p + lags2]

def swap_perturb(h, n, target, seed, n_swaps=50):
    rng = np.random.RandomState(seed)

        j = rng.randint(n)

            h_new[i], h_new[j] = h_new[j], h_new[i]
    return project_feasible(h_new, target)
