_g_target = 0.0
_g_alpha = 0.0

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

            break

        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)

def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = idxs + k < n
        wm = idxs - k >= 0
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

        if not mask.any():

            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = idxs + k < n
        wm = idxs - k >= 0
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
                 n_cand_start=20, n_cand_end=80, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if sweep % 50 == 0 and sweep > 0:
            if time.time() - start > time_limit:
                break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = 0.0
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = float(approx_max)
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def nesterov_gd(h_init, n, target, time_limit, max_steps=50000, base_lr=0.05, n_active=8):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    mu = 0.9
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        y = h + mu * v
        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[0]]
        weights = np.exp(-30.0 * (c_max - corr[top_shifts]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - y[idxs[v1] - k])
            grad[v2] -= wi * y[idxs[v2] + k]
        grad -= np.mean(grad)

        new_v = mu * v - lr * grad
        h_try = h + new_v
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < c5_val * dx:
            mu_next = min(mu + 0.01, 0.95)
            v = new_v
            h = h_try
        else:
            lr *= 0.8
            v = mu * v
            if lr < 1e-7:
                break

    return best_h, best_c5


def multi_active_cd(h_init, n, target, time_limit, n_cand=50, n_active=6):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(20000):
        if time.time() - start > time_limit:
            break
        if sweep % 50 == 0 and consecutive_no_improve > 500:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 500:
            break

        top_shifts = np.argsort(corr)[-n_active:]
        weights = np.zeros(n_corr)
        for si in top_shifts:
            weights[si] = 1.0
        sw = np.sum(weights)
        weights /= sw

        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            w = weights[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = 0.0
                b_d = delta_hi
                for _ in range(13):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturb(h, n, target, rng, scale=0.05):
    h_pert = h.copy()
    blocks = n // 8
    for _ in range(8):
        start = rng.randint(0, n - blocks)
        end = start + blocks
        delta = rng.uniform(-scale, scale)
        h_pert[start:end] += delta
    return project_feasible(h_pert, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 10.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    # Phase 1: SLSQP continuation with aggressive alpha schedule
    if tleft() > 60:
        alphas = [50, 200, 1000, 5000, 20000]
        h1, sc1 = slsqp_round(best_h, best_n, best_target, alphas, 800, 1e-15)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    if tleft() > 60:
        alphas = [50000, 200000, 500000]
        h2, sc2 = slsqp_round(best_h, best_n, best_target, alphas, 1500, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 2: Pair coordinate descent
    if tleft() > 100:
        h_cd, c5_cd = fast_pair_cd(best_h, best_n, best_target,
                                   time_limit=min(120, tleft() * 0.4),
                                   max_sweeps=30000,
                                   n_cand_start=20, n_cand_end=80,
                                   max_no_improve=500)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Nesterov gradient descent
    if tleft() > 60:
        h_ng, c5_ng = nesterov_gd(best_h, best_n, best_target,
                                  time_limit=min(50, tleft() * 0.3),
                                  max_steps=50000, base_lr=0.04, n_active=10)
        if c5_ng < best_score:
            best_score = c5_ng
            best_h = h_ng

    # Phase 4: Multi-active CD
    if tleft() > 80:
        h_macd, c5_macd = multi_active_cd(best_h, best_n, best_target,
                                          time_limit=min(80, tleft() * 0.4),
                                          n_cand=60, n_active=6)
        if c5_macd < best_score:
            best_score = c5_macd
            best_h = h_macd

    # Phase 5: SLSQP polish at current resolution
    if tleft() > 60:
        h_pol, sc_pol = slsqp_round(best_h, best_n, best_target,
                                    [100000, 500000, 1000000], 2000, 1e-15)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Phase 6: Try higher resolution
    for n_try in [500, 600]:
        if tleft() < 200:
            break
        target_try = n_try / 2.0
        h_try = interpolate_to(best_h, n_try, target_try)

        if tleft() > 150:
            h_t1, sc_t1 = slsqp_round(h_try, n_try, target_try,
                                      [100, 500, 2000, 10000], 600, 1e-15)
            if sc_t1 < compute_c5(h_try):
                h_try = h_t1

        if tleft() > 100:
            h_t2, sc_t2 = slsqp_round(h_try, n_try, target_try,
                                      [50000, 200000], 1500, 1e-15)
            if sc_t2 < compute_c5(h_try):
                h_try = h_t2

        sc_try = compute_c5(h_try)
        if tleft() > 80 and sc_try < best_score + 0.001:
            h_tc, c5_tc = fast_pair_cd(h_try, n_try, target_try,
                                       time_limit=min(80, tleft() * 0.4),
                                       max_sweeps=20000,
                                       n_cand_start=25, n_cand_end=80,
                                       max_no_improve=400)
            if c5_tc < sc_try:
                h_try = h_tc
                sc_try = c5_tc

        if tleft() > 40 and sc_try < best_score + 0.001:
            h_tp, sc_tp = slsqp_round(h_try, n_try, target_try,
                                      [200000, 500000], 1500, 1e-15)
            if sc_tp < sc_try:
                h_try = h_tp
                sc_try = sc_tp

        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try
            best_n = n_try
            best_target = target_try

    # Phase 7: Multi-start perturbation at final resolution
    if tleft() > 80:
        for noise_scale in [0.01, 0.03, 0.08]:
            if tleft() < 30:
                break
            h_pert = structured_perturb(best_h, best_n, best_target, rng, scale=noise_scale)
            h_ref, sc = slsqp_round(h_pert, best_n, best_target,
                                    [500, 5000, 50000, 200000], 800, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
            if tleft() > 30:
                h_ref2, sc2 = fast_pair_cd(h_ref, best_n, best_target,
                                           time_limit=min(30, tleft() * 0.4),
                                           max_sweeps=10000,
                                           n_cand_start=30, n_cand_end=70,
                                           max_no_improve=200)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_ref2

    # Phase 8: Final SLSQP polish
    if tleft() > 40:
        h_fin, sc_fin = slsqp_round(best_h, best_n, best_target,
                                    [500000, 2000000, 5000000], 2000, 1e-15)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 9: Final CD polish
    if tleft() > 30:
        h_fc, c5_fc = fast_pair_cd(best_h, best_n, best_target,
                                   time_limit=min(30, tleft() * 0.5),
                                   max_sweeps=10000,
                                   n_cand_start=40, n_cand_end=90,
                                   max_no_improve=200)
        if c5_fc < best_score:
            best_score = c5_fc
            best_h = h_fc

    # Phase 10: Final SLSQP
    if tleft() > 20:
        h_fl, sc_fl = slsqp_round(best_h, best_n, best_target,
                                  [2000000, 5000000], 1000, 1e-15)
        if sc_fl < best_score:
            best_score = sc_fl
            best_h = h_fl

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
