    lr_end = 0.001

        if no_improve > 500:

Let me also add one more idea: **alternating projection onto "shift-alignment" constraints**.

The idea:
1. Start with current h
2. For the worst shift k*, find the optimal "adjustment" that reduces corr[k*] the most while keeping sum(h) = target
3. This adjustment is: move mass from positions where h is high and h(·+k*) is low to positions where h is low and h(·+k*) is high
4. This is a transportation problem!

The transportation problem formulation:
- Source: positions p where h[p] is high (and h[p+k*] is low)
- Sink: positions q where h[q] is low (and h[q+k*] is high)
- Cost: the reduction in corr[k*] per unit moved

This is exactly what the CD does, but the CD only moves between ONE source and ONE sink at a time. A "global" transportation approach would move mass optimally across many source-sink pairs simultaneously.

However, implementing a full transportation algorithm is complex. A simpler approximation:
1. Compute the "supply" at each position: s[p] = h[p] * (1 - h[p+k*])
2. Compute the "demand" at each position: d[p] = (1 - h[p]) * h[p-k*]
3. Move mass from high-supply to high-demand positions

But this is approximate because moving mass changes both the supply and demand.

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

            if sc < compute_c5(h_start):
                h_start = h_new
        except Exception:
            pass
        if time.time() - start_time > time_limit:
            break

    h_final = project_feasible(h_start, target)
    return h_final, compute_c5(h_final)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def sa_escape(h_init, n, target, time_limit, max_steps=20000,
              t_start=0.03, t_end=0.0002, n_cand=30):
    dx = 2.0 / n
    start_t = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(123)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        if no_improve > 400:
            break

        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_raw = float(np.max(corr))
        c5_val = c5_raw * dx

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue

        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, shifts)
            update_change_row(change, j, h, shifts)
            if approx_new_c5 < best_c5 - 1e-14:
                best_c5 = approx_new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0

    best_h = project_feasible(h0, target0)
    best_c5 = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    # Phase 1: Fast multi-shift CD at initial resolution
    if tleft() > 80:
        h1, c51 = fast_cd(best_h, cur_n, cur_target,
                          time_limit=min(50, tleft() * 0.7),
                          n_cand_max=25, max_no_improve=500,
                          use_multi_shift=True, n_top_shifts=3)
        if c51 < best_c5:
            best_c5 = c51
            best_h = h1

    # Phase 2: Wider CD at initial resolution
    if tleft() > 120:
        h2, c52 = fast_cd(best_h, cur_n, cur_target,
                          time_limit=min(90, tleft() * 0.7),
                          n_cand_max=50, max_no_improve=350,
                          use_multi_shift=True, n_top_shifts=4)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2

    # Phase 3: L-BFGS-B refinement
    if tleft() > 60:
        h3, c53 = slsqp_refine(best_h, cur_n, cur_target,
                                alphas=[500, 2000, 8000],
                                maxiter_per_alpha=400,
                                time_limit=min(50, tleft() * 0.7),
                                start_time=start_time)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3

    # Phase 4: Move to n=600
    if tleft() > 250:
        n1 = 600
        target1 = n1 / 2.0
        h4 = interpolate_to(best_h, n1, target1)
        h4b, c54 = fast_cd(h4, n1, target1,
                           time_limit=min(100, tleft() * 0.55),
                           n_cand_max=30, max_no_improve=400,
                           use_multi_shift=True, n_top_shifts=3)
        if c54 < best_c5:
            best_c5 = c54
            best_h = h4b
            cur_n = n1
            cur_target = target1

    # Phase 5: Wide CD at n=600
    if tleft() > 150:
        h5, c55 = fast_cd(best_h, cur_n, cur_target,
                          time_limit=min(120, tleft() * 0.7),
                          n_cand_max=55, max_no_improve=300,
                          use_multi_shift=True, n_top_shifts=4)
        if c55 < best_c5:
            best_c5 = c55
            best_h = h5

    # Phase 6: SA escape at n=600
    if tleft() > 80:
        h6, c56 = sa_escape(best_h, cur_n, cur_target,
                             time_limit=min(60, tleft() * 0.7),
                             max_steps=15000, t_start=0.025,
                             t_end=0.0002, n_cand=35)
        if c56 < best_c5:
            best_c5 = c56
            best_h = h6

    # Phase 7: L-BFGS-B at n=600
    if tleft() > 60:
        h7, c57 = slsqp_refine(best_h, cur_n, cur_target,
                                alphas=[3000, 10000],
                                maxiter_per_alpha=500,
                                time_limit=min(50, tleft() * 0.7),
                                start_time=start_time)
        if c57 < best_c5:
            best_c5 = c57
            best_h = h7

    # Phase 8: Move to n=800
    if tleft() > 280:
        n2 = 800
        target2 = n2 / 2.0
        h8 = interpolate_to(best_h, n2, target2)
        h8b, c58 = fast_cd(h8, n2, target2,
                           time_limit=min(120, tleft() * 0.5),
                           n_cand_max=30, max_no_improve=350,
                           use_multi_shift=True, n_top_shifts=3)
        if c58 < best_c5:
            best_c5 = c58
            best_h = h8b
            cur_n = n2
            cur_target = target2

        if tleft() > 120:
            h9, c59 = fast_cd(best_h, cur_n, cur_target,
                              time_limit=min(100, tleft() * 0.7),
                              n_cand_max=50, max_no_improve=250,
                              use_multi_shift=True, n_top_shifts=4)
            if c59 < best_c5:
                best_c5 = c59
                best_h = h9

    # Phase 9: Final CD polish at highest resolution
    if tleft() > 60:
        h10, c510 = fast_cd(best_h, cur_n, cur_target,
                            time_limit=min(45, tleft() * 0.7),
                            n_cand_max=60, max_no_improve=200,
                            use_multi_shift=True, n_top_shifts=5)
        if c510 < best_c5:
            best_c5 = c510
            best_h = h10

    # Phase 10: Final L-BFGS-B
    if tleft() > 40:
        h11, c511 = slsqp_refine(best_h, cur_n, cur_target,
                                  alphas=[10000, 30000],
                                  maxiter_per_alpha=300,
                                  time_limit=min(30, tleft() * 0.7),
                                  start_time=start_time)
        if c511 < best_c5:
            best_c5 = c511
            best_h = h11

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
