        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-14:
                    best_trial_c5 = float(approx_max) * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def nesterov_pg(h_init, n, target, time_limit, max_steps=200000,
                 base_lr=0.05, n_active=10, momentum_decay=0.9):
    """Nesterov-accelerated projected gradient with active set."""
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    h_prev = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    lr = base_lr

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        # Nesterov step
        beta = (1.0 + np.sqrt(1.0 + 4.0 * step * step)) / 2.0
        beta = (step - 1) / (step + 2.0) if step > 1 else 0.0
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)

        # Compute active set gradient
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[-n_active:]
        c_max_val = corr[top_idx[-1]]
        weights = np.exp(-30.0 * (c_max_val - corr[top_idx]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_idx):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - y[idxs[v1] - k])
            grad[v2] -= wi * y[idxs[v2] + k]
        grad -= np.mean(grad)

        # Momentum
        v = momentum_decay * v - lr * grad
        h_new = y + v
        h_new = project_feasible(h_new, target)

        h_prev = h
        h = h_new

        sc = compute_c5(h)
        if sc < best_c5 - 1e-15:
            best_c5 = sc
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.9
                if lr < 1e-7:
                    break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = idxs + k < n
        wm = idxs - k >= 0
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,

            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-n_active:]
        weights = np.zeros(n_corr)
        c_top = corr[top_shifts[-1]]
        for si in top_shifts:
            weights[si] = np.exp(-20.0 * (c_top - corr[si]))
        sw = np.sum(weights)
        if sw > 0:
            weights /= sw

        grad = np.zeros(n)
        for si in top_shifts:
            k = int(si) - mid
            w = weights[si]
            if w < 1e-10:
                continue
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-13
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-11:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def nesterov_pg(h_init, n, target, time_limit, max_steps=200000,
                base_lr=0.04, n_active=10, momentum_decay=0.85):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    h_prev = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    lr = base_lr

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 8000:
            break

        if step > 1:
            beta = (step - 1.0) / (step + 2.0)
        else:
            beta = 0.0
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)

        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[-n_active:]
        c_max_val = corr[top_idx[-1]]
        weights = np.exp(-25.0 * (c_max_val - corr[top_idx]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_idx):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - y[idxs[v1] - k])
            grad[v2] -= wi * y[idxs[v2] + k]
        grad -= np.mean(grad)

        v = momentum_decay * v - lr * grad
        h_new = y + v
        h_new = project_feasible(h_new, target)

        h_prev = h.copy()
        h = h_new

        sc = compute_c5(h)
        if sc < best_c5 - 1e-15:
            best_c5 = sc
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                lr *= 0.85
                if lr < 1e-7:
                    break

    return best_h, best_c5


def projected_subgradient(h_init, n, target, time_limit, max_steps=100000,
                          n_active=8, base_lr=0.02):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    lr = base_lr
    avg_lr_legendre = base_lr

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 6000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 300 == 0:
                lr *= 0.8
                if lr < 1e-7:
                    break

        top_idx = np.argsort(corr)[-n_active:]
        c_max_val = corr[top_idx[-1]]
        weights = np.exp(-20.0 * (c_max_val - corr[top_idx]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_idx):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
