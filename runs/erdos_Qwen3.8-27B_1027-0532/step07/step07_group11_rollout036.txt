    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        
        # Adaptive alpha
        progress = it / n_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Smoothed gradient using softmax weights
        weights = np.exp(alpha * (corr - np.max(corr)))
        weights /= weights.sum()

            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

            best_h = h_new.copy()
            h = h_new
        
        # Backtracking if no improvement
        if c5_new >= compute_c5(h) * (1 - 1e-15):
            lr *= 0.95
    
    return best_h, best_c5

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective(x, alpha, n, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha=500, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_objective, h_cur, args=(alpha, n, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except:
            pass
    return h_cur, cur_score

    best_c5 = compute_c5(h)
    
    for alpha_idx, alpha in enumerate(alpha_list):
        if time.time() - start > time_limit:
            break
        lr = lr_init * (0.6 ** alpha_idx)
        no_improve = 0
        
        while time.time() - start < time_limit and no_improve < patience:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = np.max(corr)
            c5_cur = float(c_max) * dx
            
            if c5_cur < best_c5 - 1e-13:
                best_c5 = c5_cur
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            top_idx = np.argsort(corr)[-top_k:]
            w = np.exp(alpha * (corr[top_idx] - c_max))
            w /= w.sum()
            
            grad = np.zeros(n)
            for si in range(len(top_idx)):
                s = int(top_idx[si])
                k = s - mid
                wi = w[si]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                grad[v2] -= wi * h[idxs[v2] + k]
            grad -= np.mean(grad)
            
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            
            if sc < c5_cur - 1e-15:
                h = h_try
            else:
                h_try2 = h - 0.5 * lr * grad
                h_try2 = project_feasible(h_try2, target)
                sc2 = compute_c5(h_try2)
                if sc2 < c5_cur - 1e-15:
                    h = h_try2
                else:
                    lr *= 0.7
                    if lr < 1e-10:
                        break
    
    return best_h, best_c5

def pair_cd_fast(h_init, n, target, time_limit, max_sweeps=10000,
                 n_cand=40, n_shifts=3, max_no_improve=200):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    rng = np.random.RandomState(123)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        top_shifts = np.argsort(corr)[-n_shifts:]
        sw = corr[top_shifts].astype(float)
        sw_sum = sw.sum()
        if sw_sum < 1e-15:
            sw_sum = 1.0
        sw = sw / sw_sum
        
        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = sw[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_move = None
        
        # Try a subset of pairs (random sampling if n_cand is large)
        n_try = min(n_cand * n_cand, 500)
        pairs = []
        for ii in range(n_cand):
            for jj in range(n_cand):
                pairs.append((int(high_set[ii]), int(low_set[jj])))
        if len(pairs) > n_try:
            pairs = [pairs[i] for i in rng.choice(len(pairs), n_try, replace=False)]
        
        for i, j in pairs:
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-12:
                continue
            
            # Binary search for optimal delta
            a_d, b_d = delta_lo, delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                # Compute effect on correlations
                # Changing h[i] by -mid_d and h[j] by +mid_d
                # Effect on corr[s] for each s:
                # For shift k = s - mid:
                #   h[i] appears as h[i] in term h[i]*(1-h[i+k]) and h[i-k]*(1-h[i])
                #   h[j] appears similarly
                # This is complex, so just evaluate directly
                h_trial = h.copy()
                h_trial[i] -= mid_d
                h_trial[j] += mid_d
                h_trial = np.clip(h_trial, 0.0, 1.0)
                corr_trial = np.correlate(h_trial, 1.0 - h_trial, mode='full')
                f_mid = np.max(corr_trial)
                
                # Check if we can increase mid_d
                # If the max is at a unique shift, we can check the derivative
                active = np.where(corr_trial >= f_mid - 1e-14)[0]
                if len(active) == 1:
                    s_act = int(active[0])
                    k_act = s_act - mid
                    # Derivative of corr[s_act] w.r.t. delta
                    # = effect of decreasing h[i] and increasing h[j]
                    # h[i] change: -(1-h[i+k]) + h[i-k] (if valid)
                    # h[j] change: +(1-h[j+k]) - h[j-k] (if valid)
                    deriv = 0.0
                    if 0 <= i + k_act < n:
                        deriv -= (1.0 - h[i + k_act])
                    if 0 <= i - k_act < n:
                        deriv += h[i - k_act]
                    if 0 <= j + k_act < n:
                        deriv += (1.0 - h[j + k_act])
                    if 0 <= j - k_act < n:
                        deriv -= h[j - k_act]
                    if deriv > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                else:
                    # Multiple active shifts, be conservative
                    b_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            
            h_trial = h.copy()
            h_trial[i] -= delta_star
            h_trial[j] += delta_star
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = project_feasible(h_trial, target)
            sc = compute_c5(h_trial)
            
            if sc < best_trial - 1e-14:
                best_trial = sc
                best_move = h_trial
        
        if best_move is not None:
            h = best_move
        else:
            # Try a small random perturbation
            if rng.random() < 0.1:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i != j:
                    delta = 0.01 * rng.random()
                    if h[i] >= delta and h[j] <= 1.0 - delta:
                        h_try = h.copy()
                        h_try[i] -= delta
                        h_try[j] += delta
                        h_try = project_feasible(h_try, target)
                        if compute_c5(h_try) < c5_val:
                            h = h_try
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
