def frank_wolfe(h_init, n, target, time_limit, max_iters=100000, 
                n_active_min=3, n_active_max=20):

    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        c_max = np.max(corr)
        # Adaptive number of active shifts
        diff_from_max = c_max - corr
        # Use shifts within some tolerance
        tol = 0.05 * c_max * dx  # relative tolerance
        active = np.where(diff_from_max < tol)[0]
        if len(active) < n_active_min:
            active = np.argsort(corr)[-n_active_min:]
        if len(active) > n_active_max:
            active = np.argsort(corr)[-n_active_max:]

        grad = np.zeros(n)
        for si in active:

        sorted_idx = np.argsort(grad)

        best_t = 0.0
        best_f = c5_val / dx
        for t in t_candidates:

            h_try = np.clip(h_try, 0.0, 1.0)

            h = project_feasible(h, target)
        else:
            # Stuck, try smaller step
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / len(h), 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k_idx in active:
        k = int(k_idx) - mid
        wp = (idxs + k >= 0) & (idxs + k < n)
        wm = (idxs - k >= 0) & (idxs - k < n)
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=25, n_cand_end=100, max_no_improve=600,
                 bsearch_iters=14):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-5:]
        weights = np.zeros(n_corr)
        for si in top_shifts:
            weights[si] = 1.0
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bsearch_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-15:
                    best_trial_c5 = float(approx_max) * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def subgradient_method(h_init, n, target, time_limit, max_iters=2000000,
                       n_active_min=3, n_active_max=15, init_lr=0.01,
                       lr_decay=0.99999, patience=5000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    lr = init_lr

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > patience:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            if lr < init_lr:
                lr = min(lr * 1.5, init_lr)
        else:
            no_improve += 1

        c_max = np.max(corr)
        diff_from_max = c_max - corr
        tol = 0.02 * (c_max + 1e-10)
        active = np.where(diff_from_max < tol)[0]
        if len(active) < n_active_min:
            active = np.argsort(corr)[-n_active_min:]
        if len(active) > n_active_max:
            active_idx = np.argsort(corr)[-n_active_max:]
            active = active_idx

        grad = np.zeros(n)
        for si in active:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(active))
        grad -= np.mean(grad)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < c5_val - 1e-16:
            h = h_try
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
