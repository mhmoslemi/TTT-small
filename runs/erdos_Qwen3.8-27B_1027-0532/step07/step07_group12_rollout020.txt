def incremental_corr_update(corr, h, i, j, delta, n):
    """Update corr after h[i] += delta, h[j] -= delta. Vectorized."""
    # For lag k (shift), the contribution of position p is:
    # h[p] * (1 - h[p+k]) if p+k in [0,n)
    # h[p] * 1 if p+k not in [0,n)
    # plus h[p-k] * (1 - h[p]) if p-k in [0,n)
    
    # For position i, the lags where it contributes are:
    # As "left" factor: lags k where i+k in [0,n) -> k in [-i, n-1-i]
    # As "right" factor: lags k where i-k in [0,n) -> k in [i-(n-1), i]
    
    # The change at lag k due to h[i] += delta:
    # delta * (1 - h[i+k]) if i+k in [0,n), else delta * 1
    # -delta * h[i-k] if i-k in [0,n), else 0
    
    # Similarly for h[j] -= delta (change is -delta):
    # -delta * (1 - h[j+k]) if j+k in [0,n), else -delta * 1
    # +delta * h[j-k] if j-k in [0,n), else 0
    
    lags = np.arange(-(n-1), n)  # all possible shifts
    # corr index = lag + (n-1)
    
    # Change from h[i] += delta:
    # Term 1: delta * (1 - h[i+lag]) for i+lag in [0,n)
    term1_mask = (i + lags >= 0) & (i + lags < n)
    term1 = np.zeros(2*n-1)
    term1[term1_mask] = delta * (1.0 - h[i + lags[term1_mask]])
    term1[~term1_mask] = delta  # h[i+lag] = 0, so 1-0 = 1
    
    # Term 2: -delta * h[i-lag] for i-lag in [0,n)
    term2_mask = (i - lags >= 0) & (i - lags < n)
    term2 = np.zeros(2*n-1)
    term2[term2_mask] = -delta * h[i - lags[term2_mask]]
    
    # Change from h[j] -= delta:
    # Term 3: -delta * (1 - h[j+lag]) for j+lag in [0,n)
    term3_mask = (j + lags >= 0) & (j + lags < n)
    term3 = np.zeros(2*n-1)
    term3[term3_mask] = -delta * (1.0 - h[j + lags[term3_mask]])
    term3[~term3_mask] = -delta
    
    # Term 4: +delta * h[j-lag] for j-lag in [0,n)
    term4_mask = (j - lags >= 0) & (j - lags < n)
    term4 = np.zeros(2*n-1)
    term4[term4_mask] = delta * h[j - lags[term4_mask]]
    
    return corr + term1 + term2 + term3 + term4
