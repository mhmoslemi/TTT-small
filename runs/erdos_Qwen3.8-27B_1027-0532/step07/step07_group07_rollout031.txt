def compute_optimal_weights(corr, h, n, top_m=20):
    # Find top m shifts
    n_corr = len(corr)
    mid = n - 1
    top_indices = np.argsort(corr)[-top_m:]
    
    # Build gradient matrix for these shifts
    idxs = np.arange(n)
    grads = np.zeros((len(top_indices), n))
    for t, s_idx in enumerate(top_indices):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grads[t, v1] = (1.0 - h[idxs[v1] - k])
        grads[t, v2] -= h[idxs[v2] + k]
    
    # Solve: min w' G'G w s.t. w >= 0, sum w = 1
    # where G is the gradient matrix
    # This is min ||G'w||^2 / ||G'w||^2 ... no wait
    # We want min ||sum_k w_k g_k||^2 s.t. w >= 0, sum w = 1
    # = min w' (G G') w s.t. w >= 0, 1'w = 1
    # where G G' is top_m x top_m
    
    GG = grads @ grads.T  # top_m x top_m
    # Solve QP: min w' GG w s.t. w >= 0, sum w = 1
    # Use simple projected gradient or scipy
    from scipy.optimize import minimize
    w0 = np.ones(len(top_indices)) / len(top_indices)
    res = minimize(lambda w: w @ GG @ w, w0, method='SLSQP',
                   bounds=[(0,1)]*len(top_indices),
                   constraints=[{'type':'eq', 'fun': lambda w: w.sum()-1}])
    return res.x, top_indices

        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

                best_j = j
                best_delta = delta_star

        applied = False
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)
            applied = True

        # Periodic exploration: small random perturbation
        if not applied and sweep > 0 and sweep % explore_interval == 0:
            n_pert = rng.randint(2, 6)
            pert_idxs = rng.choice(n, size=n_pert, replace=False)
            h[pert_idxs] += rng.randn(n_pert) * explore_scale
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for pi in pert_idxs:
                update_change_row(change, h, pi, shifts, n)
            no_improve = 0

    return best_h, best_c5


def pair_cd_multishift(h_init, n, target, time_limit, max_sweeps=30000,
                       n_cand=60, top_k_eval=60, n_shifts=4,
                       max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_shifts = np.argsort(corr)[-n_shifts:]
        k_star_idx = int(top_shifts[-1])

        w_corr = corr[top_shifts]
        w_sum = w_corr.sum()
        if w_sum < 1e-15:
            w_shifts = np.ones(len(top_shifts)) / len(top_shifts)
        else:
            w_shifts = w_corr / w_sum

        grad = np.zeros(n)
        for si, s_idx in enumerate(top_shifts):
            wi = w_shifts[si]
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(11):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def block_cd(h_init, n, target, time_limit, block_size=8, max_iters=5000):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(999)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        top_shifts = np.argsort(corr)[-4:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))

        # Find block with most negative mean gradient (best candidate to increase)
        if n - block_size <= 0:
            b_start = 0
        else:
            block_means = np.convolve(grad, np.ones(block_size)/block_size, mode='valid')
            b_start = int(np.argmin(block_means))
        b_end = b_start + block_size
        b_idx = np.arange(b_start, b_end)

        # Optimize the block using a small SLSQP
        h_block = h[b_idx].copy()
        rest_sum = target - np.sum(h_block)

        def block_obj(xb):
            h_full = h.copy()
            h_full[b_idx] = xb
            c = np.correlate(h_full, 1.0 - h_full, mode='full')
            return float(np.max(c))

        def block_grad(xb):
            h_full = h.copy()
            h_full[b_idx] = xb
            c = np.correlate(h_full, 1.0 - h_full, mode='full')
            k_star = int(np.argmax(c)) - mid
            gb = np.zeros(len(xb))
            for ii, gi in enumerate(b_idx):
                v1 = (gi - k_star >= 0) & (gi - k_star < n)
                v2 = (gi + k_star >= 0) & (gi + k_star < n)
                # dc/dh[gi] contribution to block var ii
                # Actually we need dc_k*/dx_ii where x_ii = h[b_idx[ii]]
                # dc_k/dh[j] = (1-h[j+k])*[valid] - h[j-k]*[valid]
                pos = 0.0
                if 0 <= gi + k_star < n:
                    pos = 1.0 - h_full[gi + k_star]
                neg = 0.0
                if 0 <= gi - k_star < n:
                    neg = h_full[gi - k_star]
                gb[ii] = pos - neg
            return gb

        bounds_b = [(0.0, 1.0)] * block_size
        cons_b = [{'type': 'eq', 'fun': lambda xb: np.sum(xb) - np.sum(h_block)}]
        try:
            res = minimize(block_obj, h_block, jac=block_grad, method='SLSQP',
                           bounds=bounds_b, constraints=cons_b,
                           options={'maxiter': 50, 'ftol': 1e-15})
            h_new_block = project_feasible(res.x, np.sum(h_block))
            h[b_idx] = h_new_block
            h = project_feasible(h, target)
        except Exception:
            pass

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP polish at initial resolution
    tl = _time_left(start_time, budget_s)
    if tl > 60:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [200, 1000, 5000], 400, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

            best_h = h1

            h_pol1 = slsqp_optimize(h_work, n_work, target_work,

            if sc < best_score:

            h_work = h_cd

        # Phase 5: Fine-tune CD (small candidate pool for tiny adjustments)

                                       n_shifts_grad=8)
            if c5_ft < best_score:
                best_score = c5_ft

            h_work = h_ft

            if sc < best_score:

                               block_size=6, max_iters=3000)

            best_h = h_bc

    # Phase 13: Stochastic perturbation + reoptimize
    tl = _time_left(start_time, budget_s)
    if tl > 100:
        rng = np.random.RandomState(seed)
        n_work = len(best_h)
        target_work = n_work / 2.0
        for noise_scale in [0.01, 0.03, 0.06, 0.10]:
            tl = _time_left(start_time, budget_s)
            if tl < 40:
                break
            h_pert = best_h + rng.randn(n_work) * noise_scale
            h_pert = project_feasible(h_pert, target_work)
            h_ref = slsqp_optimize(h_pert, n_work, target_work,
                                   [200, 1000, 5000, 20000], 400, 1e-15)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 14: Final fine-tune CD
    tl = _time_left(start_time, budget_s)
    if tl > 40:
        h_work = best_h
        n_work = len(h_work)
        target_work = n_work / 2.0
        h_fd, c5_fd = fine_tune_cd(h_work, n_work, target_work,
                                   time_limit=min(35, tl * 0.7),
                                   max_sweeps=30000,
                                   n_cand=10, top_k_eval=20,
                                   n_shifts_grad=8)
        if c5_fd < best_score:
            best_score = c5_fd
            best_h = h_fd

    # Phase 15: Final SLSQP
    tl = _time_left(start_time, budget_s)
    if tl > 25:
        h_work = best_h
        n_work = len(h_work)
        target_work = n_work / 2.0
        h_final = slsqp_optimize(h_work, n_work, target_work,
                                 [20000, 100000], 400, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
