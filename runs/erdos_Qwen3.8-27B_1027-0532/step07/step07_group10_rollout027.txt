import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)

def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = idxs + k < n
        wm = idxs - k >= 0
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def interpolate_to(h_src, n_dst, target_dst):

    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def gradient_descent_active(h_init, n, target, time_limit, max_steps=100000,
                            base_lr=0.02, n_active=8, lr_decay=0.9999):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = base_lr
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.9
                if lr < 1e-7:
                    break
        
        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[0]]
        weights = np.exp(-50.0 * (c_max - corr[top_shifts]))
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        
        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
        else:
            lr *= 0.5
            if lr < 1e-9:
                break
    
    return best_h, best_c5

import numpy as np

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    weights /= wsum
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)

        wm = (idxs - k >= 0)
        grad[wp] += weights[ki] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[ki] * x[idxs[wm] - k]
    return grad

    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):

    _g_n = n
    _g_target = target

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pm = (p + shifts >= 0) & (p + shifts < n)
    nm = (p - shifts >= 0) & (p - shifts < n)
    change[p, pm] += (1.0 - h[p + shifts[pm]])
    change[p, nm] -= h[p - shifts[nm]]
    return change

            delta_lo = np.maximum(h[i] - 1.0, -h[low_set].astype(float))
            delta_hi = np.minimum(h[i], 1.0 - h[low_set].astype(float))

            valid = delta_hi - delta_lo > 1e-13
            if not valid.any():
                continue

            a = delta_lo[valid].copy()
            b = delta_hi[valid].copy()
            g_v = g_all[valid]

            for _ in range(n_bs_iter):
                mid_d = 0.5 * (a + b)
                vals = corr[None, :] + mid_d[:, None] * g_v
                f_mid = np.max(vals, axis=1)
                active = vals >= (f_mid[:, None] - 1e-14)
                f_prime = np.max(g_v * active, axis=1)
                go_up = f_prime > 1e-12
                idx_up = np.where(go_up)[0]
                idx_dn = np.where(~go_up)[0]
                if idx_up.size > 0:
                    a[idx_up] = mid_d[idx_up]
                if idx_dn.size > 0:
                    b[idx_dn] = mid_d[idx_dn]

            delta_star = 0.5 * (a + b)
            approx_max = np.max(corr[None, :] + delta_star[:, None] * g_v, axis=1)

            valid_ids = low_set[valid]
            best_local = int(np.argmin(approx_max))
            if approx_max[best_local] * dx < best_trial_c5 - 1e-15:
                best_trial_c5 = float(approx_max[best_local]) * dx
                best_i = i
                best_j = int(valid_ids[best_local])
                best_delta = float(delta_star[best_local])

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def gradient_descent_nesterov(h_init, n, target, time_limit, max_steps=50000,
                              base_lr=0.03, n_active=8):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    h_mom = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = base_lr
    momentum = 0.9
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 4000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.95
                if lr < 1e-7:
                    break

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = float(corr[top_shifts[0]])
        weights = np.exp(-60.0 * (c_max - corr[top_shifts]))
        wsum = np.sum(weights)
        weights /= wsum

        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_mom = project_feasible(h_mom - lr * grad, target)
        h_candidate = h_mom
        h_try = project_feasible(0.5 * (h + h_mom), target)

        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
        else:
            h_try2 = project_feasible(h_mom, target)
            sc2 = compute_c5(h_try2)
            if sc2 < cur_c5 - 1e-15:
                h = h_try2
                cur_c5 = sc2
            else:
                lr *= 0.5
                if lr < 1e-9:
                    break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
