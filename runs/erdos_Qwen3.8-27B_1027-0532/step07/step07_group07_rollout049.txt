import time

_g_alpha = 0.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _time_left(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def smooth_max_obj(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

    return smooth_max_grad(x, _g_alpha)

# Change matrix for pair CD
def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

# Pair coordinate descent
def pair_cd(h_init, n, target, time_limit, max_sweeps, n_cand_start, n_cand_end,
            top_k_eval, max_no_improve, n_shifts_grad):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5

# Interpolation
def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

# Gradient flow with projected gradient
def gradient_flow(h_init, n, target, time_limit, lr_init=0.01, decay=0.999):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    alpha = 500.0  # smooth max parameter
    
    step = 0
    no_improve = 0
    while time.time() - start < time_limit:
        if no_improve > 2000:
            break
        
        # Compute gradient of smooth max
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        current_c5 = float(c_max) * dx
        
        if current_c5 < best_c5 - 1e-15:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gradient
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_shifted = np.exp(shifted)
        s = np.sum(exp_shifted)
        if s < 1e-300:
            break
        weights = exp_shifted / s
        
        grad = np.zeros(n)
        for k_idx in range(n_corr):
            w = weights[k_idx]
            if w < 1e-10:
                continue
            k = k_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        # Only use top shifts for gradient (faster and more focused)
        top_n = 20
        top_shifts = np.argsort(corr)[-top_n:]
        grad = np.zeros(n)
        for k_idx in top_shifts:
            w = corr[k_idx] / np.sum(corr[top_shifts])
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        grad -= np.mean(grad)  # maintain sum constraint
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Line search
        direction = -grad / gnorm
        step_c5 = current_c5
        best_step = 0.0
        for t in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:
            h_try = h + t * direction
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < step_c5:
                step_c5 = sc
                best_step = t
        
        if best_step > 0:
            h = h + best_step * direction
            h = project_feasible(h, target)
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_c5

# Stochastic restart
def stochastic_perturbation(h_best, n, target, time_limit, rng):
    start = time.time()
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    
    scales = [0.01, 0.03, 0.06, 0.1, 0.15]
    for scale in scales:
        if time.time() - start > time_limit:
            break
        n_trials = 20
        for _ in range(n_trials):
            if time.time() - start > time_limit:
                break
            h_pert = h_best + rng.randn(n) * scale
            h_pert = project_feasible(h_pert, target)
            h_pol = slsqp_optimize(h_pert, n, target, [500, 2000, 10000], 300, 1e-14)
            sc = compute_c5(h_pol)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_pol.copy()
    
    return best_h, best_c5

        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _time_left(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)


def smooth_max_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_max_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _slsqp_obj(x):
    return smooth_max_obj(x, _g_alpha)


def _slsqp_grad(x):
    return smooth_max_grad(x, _g_alpha)


def _slsqp_cons(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_alpha, _g_target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, jac=_slsqp_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def pair_cd(h_init, n, target, time_limit, max_sweeps,
            n_cand_start, n_cand_end, top_k_eval,
            max_no_improve, n_shifts_grad):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(13):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def triple_cd(h_init, n, target, time_limit, max_sweeps,
              n_cand, top_k_eval, n_shifts_grad, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_l_star = change[low_set, k_star_idx]
        ch_h_star = change[high_set, k_star_idx]

        n_tl = len(low_set)
        n_th = len(high_set)

        best_trial_c5 = c5_val
        best_it = None

        score_ml = ch_l_star[None, :]
        score_mh = ch_h_star[:, None]

        n_to_eval_l = min(top_k_eval, n_tl)
        n_to_eval_h = min(top_k_eval, n_th)

        candidates = []
        for ii in range(n_to_eval_h):
            i = int(high_set[ii])
            for jj in range(n_to_eval_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                score = ch_l_star[jj] - ch_h_star[ii]
                candidates.append((score, i, j))

        candidates.sort(key=lambda x: x[0], reverse=True)
        n_eval = min(30, len(candidates))

        for score, i, j in candidates[:n_eval]:
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(13):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_it = (i, j, delta_star)

        if best_it is not None:
            i, j, delta = best_it
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, i, shifts, n)
            update_change_row(change, h, h and j, shifts, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def gradient_flow(h_init, n, target, time_limit, lr_init=0.003, n_top=15):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    no_improve = 0

    while time.time() - start < time_limit:
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_shifts = np.argsort(corr)[-n_top:]
        top_weights = corr[top_shifts]
        top_weights = top_weights / top_weights.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(top_weights, top_shifts):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]

        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        direction = -grad / gnorm
        best_step = 0.0
        best_step_c5 = c5_val
        for t in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05, lr * 0.02]:
            h_try = h + t * direction
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_step_c5:
                best_step_c5 = sc
                best_step = t

        if best_step > 0:
            h = h + best_step * direction
            h = project_feasible(h, target)
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-7:
                break

    return best_h, best_c5


def targeted_repair(h_init, n, target, time_limit, n_top=5):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        top_shifts = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        free_mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if not free_mask.any():
            break

        n_free = int(np.sum(free_mask))
        step_size = 0.01 * n_free / max(1, int(np.sum(np.abs(grad[free_mask]))))
        step_size = min(step_size, 0.05)

        h_try = h.copy()
        h_try[free_mask] -= step_size * grad[free_mask]
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val - 1e-15:
            h = h_try
        else:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP at original resolution
    tl = _time_left(start_time, budget_s)
    if tl > 100:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [200, 1000, 5000, 20000], 600, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Interpolate to n=1600 and optimize
    tl = _time_left(start_time, budget_s)
    if tl > 500:
        n1 = 1600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        # Pair CD round 1 at n=1600
        tl = _time_left(start_time, budget_s)
        if tl > 350:
            h_cd, c5_cd = pair_cd(h1, n1, target1,
                                   time_limit=min(300, tl * 0.55),
                                   max_sweeps=60000,
                                   n_cand_start=30, n_cand
  =100,
                                   top_k_eval=80, max_no_improve=1200,
                                   n_shifts_grad=3)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
        else:
            best_h = h1
            best_score = compute_c5(h1)

        n0 = n1
        target0 = target1

        # SLSQP at n=1600
        tl = _time_left(start_time, budget_s)
        if tl > 150:
            h_pol = slsqp_optimize(best_h, n0, target0,
                                   [2000, 8000, 30000], 1200, 1e-15)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 3: Gradient flow
    tl = _time_left(start_time, budget_s)
    if tl > 150:
        h_gf, c5_gf = gradient_flow(best_h, n0, target0,
                                     time_limit=min(120, tl * 0.6),
                                     lr_init=0.004, n_top=12)
        if c5_gf < best_score:
            best_score = c5_gf
            best_h = h_gf

    # Phase 4: Targeted repair
    tl = _time_left(start_time, budget_s)
    if tl > 80:
        h_tr, c5_tr = targeted_repair(best_h, n0, target0,
                                       time_limit=min(60, tl * 0.5),
                                       n_top=5)
        if c5_tr < best_score:
            best_score = c5_tr
            best_h = h_tr

    # Phase 5: Second pair CD round
    tl = _time_left(start_time, budget_s)
    if tl > 180:
        h_cd2, c5_cd2 = pair_cd(best_h, n0, target0,
                                 time_limit=min(160, tl * 0.7),
                                 max_sweeps=40000,
                                 n_cand_start=50, n_cand_end=120,
                                 top_k_eval=100, max_no_improve=1000,
                                 n_shifts_grad=4)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 6: SLSQP polish
    tl = _time_left(start_time, budget_s)
    if tl > 100:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [5000, 20000, 80000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: Stochastic perturbation + re-optimize
    tl = _time_left(start_time, budget_s)
    if tl > 120:
        scales = [0.02, 0.05, 0.1, 0.2]
        for scale in scales:
            tl = _time_left(start_time, budget_s)
            if tl < 40:
                break
            n_trials = 5
            for t in range(n_trials):
                tl = _time_left(start_time, budget_s)
                if tl < 25:
                    break
                h_pert = best_h + rng.randn(n0) * scale
                h_pert = project_feasible(h_pert, target0)
                h_pol_p = slsqp_optimize(h_pert, n0, target0,
                                         [500, 3000, 10000], 400, 1e-14)
                sc = compute_c5(h_pol_p)
                if sc < best_score:
                    best_score = sc
                    best_h = h_pol_p

    # Phase 8: Final pair CD
    tl = _time_left(start_time, budget_s)
    if tl > 80:
        h_cd_final, c5_final = pair_cd(best_h, n0, target0,
                                        time_limit=min(60, tl * 0.7),
                                        max_sweeps=25000,
                                        n_cand_start=60, n_cand_end=150,
                                        top_k_eval=120, max_no_improve=600,
                                        n_shifts_grad=5)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    # Phase 9: Final SLSQP
    tl = _time_left(start_time, budget_s)
    if tl > 40:
        h_final_pol = slsqp_optimize(best_h, n0, target0,
                                     [10000, 50000], 600, 1e-15)
        sc = compute_c5(h_final_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    # Phase 10: Final targeted repair
    tl = _time_left(start_time, budget_s)
    if tl > 30:
        h_final_tr, c5_final_tr = targeted_repair(best_h, n0, target0,
                                                   time_limit=min(20, tl * 0.6),
                                                   n_top=8)
        if c5_final_tr < best_score:
            best_score = c5_final_tr
            best_h = h_final_tr

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
