def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _sum_constraint(x):
    return np.sum(x) - _g_target


def _sum_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def subgradient_momentum(h_init, n, target, time_limit, n_steps=50000,
                         beta=0.85, eta_start=0.008, eta_decay=0.9995):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = eta_start
    momentum = np.zeros(n)

    p = np.ones(n_corr) / n_corr

    for step in range(n_steps):
        if time.time() - start > time_limit - 2.0:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        p = np.exp(corr * 40.0)
        p /= p.sum()

        grad = np.zeros(n)
        top_k = np.argsort(p)[-4:]
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        momentum = beta * momentum - eta * grad
        h_try = h + momentum
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc <= c5_val + 1e-14:
            h = h_try
        else:
            momentum *= 0.5
            h_try2 = h + momentum
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 <= c5_val + 1e-14:
                h = h_try2
            else:
                eta *= eta_decay

        if step % 1000 == 0 and step > 0:
            eta = max(eta_start * (0.3 + 0.7 * (1.0 - step / n_steps)), 1e-5)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):

        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_restart_cd(h_init, n, target, time_limit, n_restarts=8,
                     base_scale=0.04, cd_time_frac=0.7):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    rng = np.random.RandomState(12345)
    time_per_restart = (time_limit - 5.0) / n_restarts

    for r in range(n_restarts):
        if time.time() - start > time_limit - 3.0:
            break
        scale = base_scale * (0.5 + 0.7 * (r / max(n_restarts - 1, 1)))
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        sc_pert = compute_c5(h_pert)

        cd_budget = time_per_restart * cd_time_frac
        h_cd, c5_cd = pair_cd_improved(h_pert, n, target,
                                       time_limit=cd_budget,
                                       max_sweeps=15000,
                                       n_cand_start=30, n_cand_end=100,
                                       n_shifts_start=3, n_shifts_end=5,
                                       max_no_improve=300,
                                       anneal_temp=0.02, anneal_decay=0.9995)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    return best_h, best_c5
