def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

Wait no. c_k involves the product h[l]*h[l+k] (in the autocorrelation interpretation). Position i appears in c_k when:
- i is in [0,n) AND i+k is in [0,n): i.e., -i < k < n-i
- OR i-k is in [0,n) AND i is in [0,n): i.e., i-n < k < i

So position i affects c_k for k in the range (max(-n+1, i-n+1), min(n-1, n-i-1))... this is getting complicated.

Actually, for the full correlation, position i affects c_k for all k such that either (i, i+k) or (i-k, i) is a valid pair. This means k ranges over a window of size approximately n around i.

So a swap at (i, j) affects c_k for k in the union of the windows around i and j, which is essentially all k. So we can't do incremental updates easily.

However, we CAN compute the effect of a specific swap on a specific c_k in O(1) time using the formula above:
Δc_k = δ * (h[i-k] + h[i+k] - h[j-k] - h[j+k]) (with 0 for out-of-bounds)

So the algorithm is:
1. For the current max shift k*, find all pairs (i,j) with h[i]=1, h[j]=0
2. For each pair, compute Δc_k* = h[i-k*] + h[i+k*] - h[j-k*] - h[j+k*]
3. Pick the pair with the most negative Δc_k*
4. Apply the swap
5. Recompute c (or update incrementally)

Step 2 is O(n^2) which is too slow for n=600 per iteration. But we can be smarter:

For a fixed k*, Δc_k* = (h[i-k*] + h[i+k*]) - (h[j-k*] + h[j+k*])

Let A[i] = h[i-k*] + h[i+k*] (with 0 for out of bounds)
Let B[j] = h[j-k*] + h[j+k*]

Then Δc_k* = A[i] - B[j]

We want to maximize B[j] - A[i] over i with h[i]=1 and j with h[j]=0.

This is: max_{j: h[j]=0} B[j] - min_{i: h[i]=1} A[i]

Which can be computed in O(n)!

So the algorithm per iteration is:
1. Find k* = argmax c_k
2. Compute A[i] and B[j] for all i, j: O(n)
3. Find best swap: O(n)
4. Apply swap: O(1)
5. Update c: O(n) (recompute affected terms)

Total per iteration: O(n). For n=600, we can do millions of iterations in a few seconds!

This is MUCH faster than the pair CD which does O(n_cand^2 * n_corr) per sweep.

But we need to be careful: after the swap, other c_k values change too, and the argmax might shift. So we need to track the top few shifts, not just the argmax.

1. Maintain a set of "active" shifts (top M shifts by c_k value)
2. For each active shift, compute the best swap
3. Among all candidate swaps, pick the one that most reduces the maximum
4. Apply it
5. Update c_k for all affected k

Let me think about the update more carefully. After swapping h[i] and h[j]:
- c_k changes for all k where position i or j appears

So the overall algorithm is O(n) per iteration, and we can do many iterations.

def binary_local_search(h_init, n, target, time_limit, n_active=5, max_iter=1000000, ann_temp=0.0, seed=42):
    """
    Efficient binary local search.
    At each step, finds the best swap (i,j) that reduces the max overlap.
    """
    start = time.time()
    rng = np.random.RandomState(seed)
    
    # Binarize
    h = (h_init > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target_ones = n // 2  # Should be exact when n is even
    if n_ones > target_ones:
        # Remove excess ones: remove ones at positions with lowest "importance"
        one_pos = np.where(h == 1)[0]
        # Remove randomly for simplicity
        remove = rng.choice(one_pos, size=n_ones - target_ones, replace=False)
        h[remove] = 0
    elif n_ones < target_ones:
        zero_pos = np.where(h == 0)[0]
        add = rng.choice(zero_pos, size=target_ones - n_ones, replace=False)
        h[add] = 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5_val = float(np.max(corr))
    best_h = h.copy()
    
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    
    for iteration in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        cur_max = float(np.max(corr))
        if cur_max < best_c5_val:
            best_c5_val = cur_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break
        
        # Find top n_active shifts
        top_idx = np.argsort(corr)[-n_active:]
        
        best_swap = None
        best_score = -np.inf
        
        for si in top_idx:
            k = int(si) - mid
            
            # Compute A[p] = h[p+k] + h[p-k] for all p
            A = np.zeros(n)
            valid_plus = (idxs + k >= 0) & (idxs + k < n)
            valid_minus = (idxs - k >= 0) & (idxs - k < n)
            if np.any(valid_plus):
                A[valid_plus] += h[idxs[valid_plus] + k]
            if np.any(valid_minus):
                A[valid_minus] += h[idxs[valid_minus] - k]
            
            # Find i with h[i]=1 that minimizes A[i]
            # Find j with h[j]=0 that maximizes A[j]
            min_A = np.min(A[h == 1]) if np.any(h == 1) else np.inf
            max_A = np.max(A[h == 0]) if np.any(h == 0) else -np.inf
            
            if min_A >= np.inf or max_A <= -np.inf:
                continue
            
            improvement = max_A - min_A
            if improvement <= 0:
                continue
            
            # Find the actual positions
            i_candidates = np.where((h == 1) & (A <= min_A + 1e-10))[0]
            j_candidates = np.where((h == 0) & (A >= max_A - 1e-10))[0]
            
            # Pick the first valid pair (i != j)
            for i in i_candidates[:5]:
                for j in j_candidates[:5]:
                    if i != j:
                        score = improvement
                        # Check that this actually helps the max
                        if score > best_score:
                            best_score = score
                            best_swap = (int(i), int(j))
                        break
                else:
                    continue
                break
        
        if best_swap is None:
            # Annealing: try a random move
            if ann_temp > 0 and no_improve < 2000:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = int(rng.choice(ones))
                    j = int(rng.choice(zeros))
                    h[i], h[j] = 0.0, 1.0
                    # Update corr
                    # Δc_k = h[i+k] + h[i-k] - h[j+k] - h[j-k] for each k
                    # But h has changed, so we need to recompute
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    ann_temp *= 0.999
            continue
        
        i, j = best_swap
        
        # Apply swap
        h[i], h[j] = 0.0, 1.0
        
        # Update corr efficiently
        # For each shift k, Δc_k = old_h[i+k] + old_h[i-k] - old_h[j+k] - old_h[j-k]
        # where old_h is h BEFORE the swap
        # But we already applied the swap, so old_h[i] = 1, old_h[j] = 0
        # new_h[i] = 0, new_h[j] = 1
        
        # Actually, let's just recompute. For n=600, it's fast enough.
        corr = np.correlate(h, 1.0 - h, mode='full')
        # This is O(n^2) but with numpy it's fast (FFT-based)
    
    return best_h, best_c5_val * (2.0 / n)

For the second term: h_new[j]*(1-h_new[j+k]) - h_old[j]*(1-h_old[j+k]) = 1*(1-h[j+k]) - 0 = 1-h[j+k]

For the third term (i appears as l+k, so l = i-k): h[i-k]*(1-h_new[i]) - h[i-k]*(1-h_old[i]) = h[i-k]*1 - h[i-k]*0 = h[i-k]

For the fourth term (j appears as l+k, so l = j-k): h[j-k]*(1-h_new[j]) - h[j-k]*(1-h_old[j]) = h[j-k]*0 - h[j-k]*1 = -h[j-k]

(This uses h_old values, which is the state BEFORE the swap.)

So I can update corr in O(n) time:
For each k (0 to 2n-2):
  actual_k = k - (n-1)  # the actual shift
  pos1 = i + actual_k (valid if 0 <= pos1 < n)
  pos2 = i - actual_k (valid if 0 <= pos2 < n)
  pos3 = j + actual_k (valid if 0 <= pos3 < n)
  pos4 = j - actual_k (valid if 0 <= pos4 < n)
  delta = (h[pos1] if valid else 0) - (h[pos3] if valid else 0) + (h[pos2] if valid else 0) - (h[pos4] if valid else 0)
  corr[k] += delta

This is O(n) per update (since there are 2n-1 values of k, and each takes O(1)).
