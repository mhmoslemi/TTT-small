import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value_and_grad(h, alpha):
    """Compute LSE smoothed max and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    val = max_c + np.log(np.sum(exp_s)) / alpha
    
    # Gradient: convolve(1-h, w, 'full')[1:n+1] - correlate(h, w, 'full')[1:n+1]
    term1 = np.convolve(1.0 - h, w, mode='full')[1:n+1]
    term2 = np.correlate(h, w, mode='full')[1:n+1]
    grad = term1 - term2
    return val, grad

def lseb_step(h, n, target, alpha, maxiter=200):
    """One step of L-BFGS-B with augmented Lagrangian."""
    rho = 100.0
    lam = 0.0
    
    def obj(x):
        v, _ = lse_value_and_grad(x, alpha)
        return v + 0.5 * rho * (np.sum(x) - target)**2 + lam * (np.sum(x) - target)
    
    def obj_grad(x):
        _, g = lse_value_and_grad(x, alpha)
        g = g + rho * (np.sum(x) - target) + lam
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except:
        return h

import numpy as np
from scipy.optimize import minimize
import time

_start_time = 0.0
_budget_s = 1300.0


def tleft():
    return _budget_s - 30.0 - (time.time() - _start_time)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    w = exp_s / s
    val = max_c + np.log(s) / alpha
    term1 = np.convolve(1.0 - h, w, mode='full')[1:n + 1]
    term2 = np.correlate(h, w, mode='full')[1:n + 1]
    grad = term1 - term2
    return val, grad


def lseb_subproblem(x, alpha, target, rho, lam):
    v, g = lse_value_and_grad(x, alpha)
    s = np.sum(x) - target
    obj = v + 0.5 * rho * s * s + lam * s
    grad = g + rho * s + lam
    return obj, grad


def lseb_step(h, n, target, alpha, maxiter=300):
    rho = 500.0
    lam = 0.0
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        return lseb_subproblem(x, alpha, target, rho, lam)[0]

    def grad_func(x):
        return lseb_subproblem(x, alpha, target, rho, lam)[1]

    try:
        res = minimize(obj_func, h, jac=grad_func, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

                n_active=3, n_cand_start=15, n_cand_end=50,
                max_no_improve=400, bisection_iters=10, max_pairs=200):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    rng_local = np.random.RandomState(12345)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        if n_h * n_l <= max_pairs:
            pair_list = []
            for ii in range(n_h):

                    jv = int(low_set[jj])

        else:

                jj = rng_local.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))

        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g = change[j] - change[i]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i, :] = 0.0
            lags = np.arange(0, best_i + 1)
            change[best_i, mid + lags] += (1.0 - h[best_i - lags])
            lags2 = np.arange(-best_i, n - best_i)
            change[best_i, mid + lags2] -= h[best_i + lags2]
            change[best_j, :] = 0.0
            lags = np.arange(0, best_j + 1)
            change[best_j, mid + lags] += (1.0 - h[best_j - lags])
            lags2 = np.arange(-best_j, n - best_j)
            change[best_j, mid + lags2] -= h[best_j + lags2]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
