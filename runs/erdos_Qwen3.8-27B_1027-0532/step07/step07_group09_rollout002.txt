import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

# Smooth objective using power approximation
def smooth_obj_power(h, p):
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_pos = np.maximum(corr, 0)
    return float(np.sum(corr_pos ** p) ** (1.0 / p))

def smooth_obj_logsumexp(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

# Gradient of smooth objective
def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    weights = np.exp(np.clip(alpha * (corr - c_max), -500, 0))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-15:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[k_idx] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= weights[k_idx] * h[idxs[v2] + k]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def _smooth_obj_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_pow(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    corr_clipped = np.maximum(corr, 0.0)
    s = np.sum(corr_clipped ** p)
    if s <= 0:
        return 0.0
    return float(s ** (1.0 / p))

def slsqp_lse(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_lse, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def slsqp_pow(h_start, n, target, powers, maxiter_per_power, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

            res = minimize(_smooth_obj_pow, h_cur, args=(p,),

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def pair_cd(h_init, n, target, time_limit, max_sweeps=30000,

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = int(n_active_start + (n_active_end - n_active_start) * progress)

        top_idx = np.argsort(corr)[-n_active:]

        w_norm = np.sum(w)

            w = np.ones(n_active) / n_active

            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)

        high_set = sorted_idx[-ncand:]

        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue

                g = G[lr] - G[hr]

                a_d = delta_min
                b_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:

                    best_trial = approx_max

                    best_j = j
                    best_delta = delta_star

            h[best_j] += best_delta

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        tol = max(1e-10, 1e-7 * max_corr)

            v1 = (idxs - k >= 0) & (idxs - k < n)

            grad[v2] -= h[idxs[v2] + k]

        new_c5 = compute_c5(h_new)

    return best_h, best_c5

def multi_pair_cd(h_init, n, target, time_limit, max_sweeps=15000,
                  n_cand=30, batch_size=3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 300:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = min(6, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        ncand = min(n_cand, n // 4)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)

        best_trial = c5_val
        best_moves = []
        best_total_delta = 0.0

        for ii in range(min(n_h, batch_size)):
            i = int(high_set[ii])
            hr = cand_map[i]
            h_i = h[i]
            for jj in range(min(n_l, batch_size)):
                j = int(low_set[jj])
                lr = cand_map[j]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = 0.0
                if delta_max < 1e-12:
                    continue

                g = G[lr] - G[hr]
                a_d = delta_min
                b_d = delta_max
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_moves = [(i, j, delta_star)]
                    best_total_delta = delta_star

        if best_moves:
            i, j, delta = best_moves[0]
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP at initial resolution
    if tleft() > 50:
        h_pol, sc = slsqp_lse(best_h, n0, target0,
                               [200, 1000, 5000], 500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Go to n=800
    n1 = 800
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)
    sc1 = compute_c5(h1)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    # Phase 3: Pair CD at n=800 (main improvement driver)
    if tleft() > 250:
        h_cd, c5_cd = pair_cd(h1, n1, target1,
                              time_limit=min(180, tleft() * 0.5),
                              max_sweeps=30000,
                              n_cand_start=20, n_cand_end=60,
                              max_no_improve=500)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP polish at n=800
    if tleft() > 120:
        h_pol, sc = slsqp_lse(best_h, n1, target1,
                               [1000, 5000, 20000, 80000], 600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Power-approximation SLSQP at n=800
    if tleft() > 100:
        h_pow, sc = slsqp_pow(best_h, n1, target1,
                               [50, 100, 200, 500], 400, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pow

    # Phase 6: Second CD round at n=800
    if tleft() > 200:
        h_cd2, c5_cd2 = pair_cd(best_h, n1, target1,
                                time_limit=min(120, tleft() * 0.5),
                                max_sweeps=20000,
                                n_cand_start=25, n_cand_end=65,
                                max_no_improve=400)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 7: Subgradient polish
    if tleft() > 80:
        h_sg, c5_sg = subgradient_polish(best_h, n1, target1,
                                         time_limit=min(50, tleft() * 0.5),
                                         max_iter=30000, lr_start=0.0015)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: SLSQP polish again
    if tleft() > 100:
        h_pol2, sc2 = slsqp_lse(best_h, n1, target1,
                                 [5000, 20000, 80000, 200000], 800, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 9: Go to n=1600 for finer resolution
    n2 = 1600
    target2 = n2 / 2.0
    h2 = interpolate_to(best_h, n2, target2)
    sc2 = compute_c5(h2)
    work_n = n2
    work_target = target2
    if sc2 < best_score:
        best_score = sc2
        best_h = h2

    # Phase 10: SLSQP at n=1600
    if tleft() > 150:
        h_pol3, sc3 = slsqp_lse(best_h, n2, target2,
                                 [500, 2000, 10000], 400, 1e-15)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    # Phase 11: Power SLSQP at n=1600
    if tleft() > 80:
        h_pow2, sc_p = slsqp_pow(best_h, n2, target2,
                                  [50, 200, 500], 300, 1e-15)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_pow2

    # Phase 12: Random restarts at n=800
    if tleft() > 200:
        rng = np.random.RandomState(seed)
        for attempt in range(8):
            if tleft() < 50:
                break
            noise_scale = 0.01 + 0.02 * (attempt % 3)
            h_pert = best_h[:n1] + rng.randn(n1) * noise_scale
            h_pert = project_feasible(h_pert, target1)
            h_ref, sc_ref = slsqp_lse(h_pert, n1, target1,
                                       [500, 2000, 10000], 300, 1e-15)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
                work_n = n1
                work_target = target1

    # Phase 13: Final CD at working resolution
    if tleft() > 120:
        h_cd3, c5_cd3 = pair_cd(best_h, work_n, work_target,
                                 time_limit=min(80, tleft() * 0.5),
                                 max_sweeps=15000,
                                 n_cand_start=30, n_cand_end=70,
                                 max_no_improve=300)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 14: Interpolate to n=2000 if time allows
    if tleft() > 200:
        n3 = 2000
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        sc3 = compute_c5(h3)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3
            work_n = n3
            work_target = target3

        if tleft() > 100:
            h_pol4, sc4 = slsqp_lse(best_h, n3, target3,
                                     [500, 2000, 8000], 300, 1e-15)
            if sc4 < best_score:
                best_score = sc4
                best_h = h_pol4

    # Phase 15: Final polish
    if tleft() > 50:
        h_final_pol, sc_f = slsqp_lse(best_h, work_n, work_target,
                                       [10000, 50000, 100000], 400, 1e-16)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_final_pol

    # Final feasibility check
    final_n = len(best_h)
    final_h = project_feasible(best_h, final_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, final_n
