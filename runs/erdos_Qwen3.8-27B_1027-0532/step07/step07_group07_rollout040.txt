import time

_g_alpha = 0.0
_g_target = 0.0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):

        if abs(diff) < 1e-16 * max(1.0, target_sum):
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:

            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    best_h = h.copy()
    no_improve = 0

        if time.time() - start > time_limit:

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:

            best_h = h.copy()
            no_improve = 0

        n_cand = min(n_cand, n // 4)

            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        best_j = -1

            g = change[j] - change[i]

                if f_prime > 1e-12:

                else:

            delta_star = 0.5 * (a_d + b_d)

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:

                best_i = i
                best_j = j

            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_n, _g_target, _g_alpha
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def pair_cd_multishift_score(h_init, n, target, time_limit, max_sweeps=50000,
                             n_cand_start=30, n_cand_end=80, top_k_eval=60,
                             n_shifts_grad=3, n_shifts_score=3,
                             max_no_improve=800):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        top_score_shifts = np.argsort(corr)[-n_shifts_score:]
        top_grad_shifts = np.argsort(corr)[-n_shifts_grad:]

        grad = np.zeros(n)
        for s_idx in top_grad_shifts:
            w = corr[s_idx]
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad /= float(len(top_grad_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h = change[high_set, top_score_shifts]
        ch_l = change[low_set, top_score_shifts]

        score_weights = corr[top_score_shifts]
        score_weights = score_weights / score_weights.sum()
        score_mat = np.zeros((len(high_set), n_cand))
        for s in range(len(top_score_shifts)):
            score_mat += score_weights[s] * (ch_l[:, s][None, :] - ch_h[s, :][:, None])

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_local_search(h_init, n, target, time_limit, max_steps=20000):
    start = time.time()
    rng = np.random.RandomState(12345)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        scale = 0.03 * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.choice([2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 500:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def fourier_perturb(h, n, seed, n_modes):
    idx = np.arange(n)
    h_new = h.copy()
    rng = np.random.RandomState(seed)
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.005, 0.03)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += amp * np.cos(np.pi * k * idx / n + phase)
    return project_feasible(h_new, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_target, _g_alpha
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    def _tl():
        return budget_s - 40.0 - (time.time() - start_time)

    tl = _tl()
    if tl > 60:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [200, 1000, 5000], 400, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    tl = _tl()
    if tl > 200:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

        tl = _tl()
        if tl > 150:
            h_cd, c5_cd = pair_cd_multishift_score(
                h1, n1, target1, time_limit=min(160, tl * 0.7),
                max_sweeps=40000, n_cand_start=30, n_cand_end=70,
                top_k_eval=50, n_shifts_grad=3, n_shifts_score=3,
                max_no_improve=600)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                n0 = n1
                target0 = target1

    tl = _tl()
    if tl > 60:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [2000, 10000], 800, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    tl = _tl()
    if tl > 350:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

        tl = _tl()
        if tl > 250:
            h_cd2, c5_cd2 = pair_cd_multishift_score(
                h2, n2, target2, time_limit=min(280, tl * 0.8),
                max_sweeps=30000, n_cand_start=30, n_cand_end=70,
                top_k_eval=50, n_shifts_grad=3, n_shifts_score=3,
                max_no_improve=800)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                n0 = n2
                target0 = target2

        tl = _tl()
        if tl > 60:
            h_pol2 = slsqp_optimize(best_h, n0, target0,
                                    [5000, 20000], 600, 1e-14)
            sc2b = compute_c5(h_pol2)
            if sc2b < best_score:
                best_score = sc2b
                best_h = h_pol2

    tl = _tl()
    if tl > 150:
        h_cd3, c5_cd3 = pair_cd_multishift_score(
            best_h, n0, target0, time_limit=min(180, tl * 0.8),
            max_sweeps=25000, n_cand_start=40, n_cand_end=90,
            top_k_eval=70, n_shifts_grad=4, n_shifts_score=4,
            max_no_improve=600)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    tl = _tl()
    if tl > 40:
        h_ss, c5_ss = stochastic_local_search(
            best_h, n0, target0, time_limit=min(50, tl * 0.7),
            max_steps=15000)
        if c5_ss < best_score:
            best_score = c5_ss
            best_h = h_ss

    tl = _tl()
    if tl > 60:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.08]:
            tl = _tl()
            if tl < 40:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref = slsqp_optimize(h_pert, n0, target0,
                                   [1000, 5000, 20000], 400, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    tl = _tl()
    if tl > 50:
        h_harm = fourier_perturb(best_h, n0, seed + 1, 5)
        h_ref = slsqp_optimize(h_harm, n0, target0,
                               [500, 5000, 20000], 400, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    tl = _tl()
    if tl > 40:
        h_cd_final, c5_final = pair_cd_multishift_score(
            best_h, n0, target0, time_limit=min(45, tl * 0.8),
            max_sweeps=15000, n_cand_start=50, n_cand_end=100,
            top_k_eval=80, n_shifts_grad=4, n_shifts_score=4,
            max_no_improve=300)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    tl = _tl()
    if tl > 30:
        h_final_pol = slsqp_optimize(best_h, n0, target0,
                                     [20000, 80000], 400, 1e-15)
        sc = compute_c5(h_final_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
