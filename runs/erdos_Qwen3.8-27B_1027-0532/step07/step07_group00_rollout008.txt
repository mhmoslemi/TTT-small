        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(x, alpha, n):
    """Compute smooth max objective and its gradient."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Softmax weights
    max_corr = np.max(corr)
    exp_corr = np.exp(alpha * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    obj = (max_corr + np.log(np.sum(exp_corr)) / alpha)
    
    # Gradient: weighted sum of individual gradients
    n_half = n - 1
    grad = np.zeros(n)
    # Only compute for shifts with significant weight
    significant = np.where(weights > 1e-6)[0]
    for s_idx in significant:
        w = weights[s_idx]
        k = int(s_idx) - n_half
        # d/dh[j] corr_k
        for j in range(n):
            # forward: j and j+k valid
            if 0 <= j + k < n:
                grad[j] += w * (1.0 - x[j + k])
            # backward: j-k and j valid
            if 0 <= j - k < n:
                grad[j] += w * (-x[j - k])
    return obj, grad, corr

def coarsen(h, factor):
    """Downsample by averaging blocks."""
    n = len(h)
    n_new = n // factor
    h_trimmed = h[:n_new * factor]
    h_coarse = h_trimmed.reshape(n_new, factor).mean(axis=1)
    return h_coarse

def upsample(h_coarse, n_dst):
    """Upsample using linear interpolation."""
    n_src = len(h_coarse)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst)
    return np.interp(x_dst, x_src, h_coarse)

def pair_cd(h_init, n, target, time_limit, max_sweeps=20000,
            n_cand=40, max_no_improve=300, weight_beta=5.0):
    """Weighted pair coordinate descent with large candidate sets."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))
        
        # Gradient
        grad = np.zeros(n)
        top_k_shifts = min(20, n_corr)
        top_idx = np.argsort(weights)[-top_k_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            found = False
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    found = True
                    break
            
            if found:

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows

                pos_mask = (p + shifts >= 0) & (p + shifts < n)

    return best_h, best_c5

        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(120, tleft() * 0.4),
                              max_sweeps=20000, n_cand=50,
                              max_no_improve=300, weight_beta=5.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
    
    # Phase 3: Multi-scale exploration
    if tleft() > 300:
        h_ms, sc_ms = multiscale_optimize(best_h, n0, target0,
                                          time_limit=min(200, tleft() * 0.5),
                                          coarse_n=100, n_starts=20)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms
        else:
            # Even if multi-scale didn't improve at coarse level,
            # the upsampled version might be better after local optimization
            h_up = project_feasible(h_ms, target0)
            h_pol_up, sc_up = slsqp_round(h_up, n0, target0,
                                          [200, 1000, 5000], 400, 1e-14)
            if sc_up < best_score:
                best_score = sc_up
                best_h = h_pol_up
    
    # Phase 4: Pair CD again (possibly starting from better point)
    if tleft() > 150:
        h_cd2, c5_cd2 = pair_cd(best_h, n0, target0,
                                time_limit=min(100, tleft() * 0.4),
                                max_sweeps=15000, n_cand=60,
                                max_no_improve=250, weight_beta=6.0)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
    
    # Phase 5: Second multi-scale round (with updated best)
    if tleft() > 250:
        h_ms2, sc_ms2 = multiscale_optimize(best_h, n0, target0,
                                            time_limit=min(180, tleft() * 0.5),
                                            coarse_n=80, n_starts=15)
        # Upsample and polish
        if sc_ms2 < best_score:
            best_score = sc_ms2
            best_h = h_ms2
        else:
            h_up2 = project_feasible(h_ms2, target0)
            h_pol2, sc_pol2 = slsqp_round(h_up2, n0, target0,
                                          [200, 1000, 5000, 20000], 400, 1e-14)
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_h = h_pol2
    
    # Phase 6: Final pair CD
    if tleft() > 80:
        h_cd3, c5_cd3 = pair_cd(best_h, n0, target0,
                                time_limit=min(60, tleft() * 0.5),
                                max_sweeps=10000, n_cand=70,
                                max_no_improve=200, weight_beta=8.0)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
    
    # Phase 7: Final SLSQP with high alpha
    if tleft() > 50:
        h_fin, sc_fin = slsqp_round(best_h, n0, target0,
                                    [10000, 50000], 300, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin
    
    # Phase 8: Try n=1200 if time allows
    if tleft() > 200:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, n0), best_h)
        h2 = project_feasible(h2, target2)
        
        h2_pol, sc2 = slsqp_round(h2, n2, target2, [200, 1000, 5000], 300, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2_pol
        
        if tleft() > 80:
            h2_cd, c5_2 = pair_cd(best_h, n2, target2,
                                  time_limit=min(60, tleft() * 0.5),
                                  max_sweeps=8000, n_cand=50,
                                  max_no_improve=200, weight_beta=5.0)
            if c5_2 < best_score:
                best_score = c5_2
                best_h = h2_cd
        
        if tleft() > 40:
            h2_fin, sc2_fin = slsqp_round(best_h, n2, target2,
                                          [10000, 50000], 300, 1e-14)
            if sc2_fin < best_score:
                best_score = sc2_fin
                best_h = h2_fin
    
    # Final check
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    
    return final_h, final_score, len(final_h)
