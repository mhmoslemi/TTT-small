def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad

def alternating_method(h_init, n, target, time_limit, beta_init=10.0, beta_end=200.0, max_iters=100000):
    """Alternating minimization for the dual-primal game."""
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    shifts = np.arange(-(n-1), n)

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_i = -1
        best_j = -1
        best_delta = 0.0

                if i == j:
                    continue

                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:

                a_d = delta_lo

                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:

                    else:
                        a_d = mid_d

                if abs(delta_star) < 1e-12:
                    continue

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i, :] = 0.0
            change[best_j, :] = 0.0
            pi = best_i
            pj = best_j
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj - shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

The alternating method does:
- Given h, compute p ∝ exp(β * corr)
- Compute u[i] = sum_k p[k] * (1-h[i+k])
- Set h_new[i] = 1 if u[i] is in the smallest n/2 values

This is essentially a "best response" iteration. The question is whether it makes progress toward the global optimum.

For the max-norm problem (minimize max_k corr[k]), the optimal p should be concentrated on the maximizing k. So as β→∞, p concentrates on argmax corr, and the update becomes:
- Compute u[i] = 1 - h[i+k*] (for k* = argmax)
- Set h_new[i] = 1 for the n/2 smallest u[i] = largest h[i+k*]

This is like: shift the "mass" of h to align with h shifted by k*. This makes sense intuitively - we want h to overlap with its shifted version as much as possible at the worst shift.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad


def compute_corr_gradient_multi(h, k_stars, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(k_stars, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def cutting_plane_method(h_init, n, target, time_limit, max_iters=10000, tol=1e-9, max_cuts=400):

def interceptive_cd(h_init, n, target, time_limit, max_sweeps=30000,
                    n_cand_start=20, n_cand_end=80, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pi = best_i
            pj = best_j
            change[pi, :] = 0.0
            change[pj, :] = 0.0
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    return best_h, best_c5


def binary_sa(h_init, n, target, time_limit, rng, t_start=1.0, t_end=0.001,
              max_iters=200000):
    start = time.time()
    sorted_h = np.sort(h_init)
    threshold = sorted_h[n // 2 - 1]
    h = (h_init >= threshold).astype(float)
    ones_count = int(np.sum(h))
    if ones_count != n // 2:
        idxs_all = np.argsort(h)
        if ones_count > n // 2:
            for i in range(ones_count - n // 2):
                h[idxs_all[-(i + 1)]] = 0.0
        else:
            for i in range(n // 2 - ones_count):
                h[idxs_all[i]] = 1.0
    h = project_feasible(h, target)

    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        progress = min(float(it) / max_iters, 1.0)
        t_curr = t_start * (t_end / t_start) ** progress

        i = int(rng.randint(len(ones_idx)))
        j = int(rng.randint(len(zeros_idx)))
        p = int(ones_idx[i])
        q = int(zeros_idx[j])

        h[p] = 0.0
        h[q] = 1.0
        new_score = compute_c5(h)

        delta = new_score - curr_score
        if delta < 0 or rng.random() < np.exp(-delta / max(t_curr, 1e-10)):
            curr_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_h = h.copy()
            ones_idx = np.where(h == 1.0)[0]
            zeros_idx = np.where(h == 0.0)[0]
        else:
            h[p] = 1.0
            h[q] = 0.0

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def nesterov_pg(h_init, n, target, time_limit, alpha_init=200.0, alpha_end=50000.0,
                max_iters=20000, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    h_curr = h_init.copy().astype(float)
    h_prev = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)
    idxs = np.arange(n)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum)) / 2.0
        y = h_curr + ((t_k - 1.0) / t_k) * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)

        alpha = alpha_init * (alpha_end / alpha_init) ** (it / max_iters)
        corr_y = compute_corr(y)
        c_max = np.max(corr_y)
        shifted = np.clip(alpha * (corr_y - c_max), -500.0, 0.0)
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(len(corr_y))
            p[int(np.argmax(corr_y))] = 1.0
        else:
            p /= p_sum

        grad = np.zeros(n)
        for m in range(len(corr_y)):
            if p[m] < 1e-14:
                continue
            k = m - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += p[m] * (1.0 - y[idxs[v1] + k])
            grad[v2] -= p[m] * y[idxs[v2] - k]
        grad /= alpha

        h_new = y - 0.1 * grad
        h_new = project_feasible(h_new, target)
        h_prev = h_curr.copy()
        h_curr = h_new
        sc = compute_c5(h_curr)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_curr.copy()

    return best_h, best_score

                                 max_iters=150000)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    if tleft() > 60:
        h_cd2, sc_cd2 = interceptive_cd(best_h, n0, target0,
                                         time_limit=min(50, tleft() * 0.5),
                                         max_sweeps=15000,
                                         n_cand_start=30, n_cand_end=80,
                                         max_no_improve=300)
        if sc_cd2 < best_score:

            best_h = h_cd2

    if tleft() > 250:
        n1 = 800
        target1 = n1 / 2.0
        h800 = interpolate_to(best_h, n1, target1)

        if tleft() > 200:
            h800_alt, sc800a = alternating_binary(h800, n1, target1,
                                                   time_limit=min(100, tleft() * 0.4),
                                                   beta=25.0, max_iters=200000)
            if sc800a < best_score:
                best_score = sc800a
                best_h = h800_alt
                best_n = n1

        if tleft() > 150:
            n_cur = len(best_h)
            tgt_cur = n_cur / 2.0
            h800_cd, sc800c = interceptive_cd(best_h, n_cur, tgt_cur,
                                               time_limit=min(120, tleft() * 0.5),
                                               max_sweeps=20000,
                                               n_cand_start=15, n_cand_end=50,
                                               max_no_improve=300)
            if sc800c < best_score:
                best_score = sc800c
                best_h = h800_cd
                best_n = n_cur

        if tleft() > 100:
            n_cur = len(best_h)
            tgt_cur = n_cur / 2.0
            h800_cp, sc800p = cutting_plane_method(best_h, n_cur, tgt_cur,
                                                    time_limit=min(80, tleft() * 0.4),
                                                    max_iters=10000, tol=1e-10, max_cuts=300)
            if sc800p < best_score:
                best_score = sc800p
                best_h = h800_cp
                best_n = n_cur

        if tleft() > 60:
            n_cur = len(best_h)
            tgt_cur = n_cur / 2.0
            h800_sa, sc800s = binary_sa(best_h, n_cur, tgt_cur,
                                         time_limit=min(50, tleft() * 0.4),
                                         rng=rng, t_start=0.3, t_end=0.001,
                                         max_iters=100000)
            if sc800s < best_score:

    if tleft() > 150:
        n2 = 1600
        target2 = n2 / 2.0
        h1600 = interpolate_to(best_h, n2, target2)

        if tleft() > 100:

            tgt_cur = n_cur / 2.0
            h1600_cd, sc1600c = interceptive_cd(best_h, n_cur, tgt_cur,

                                                 max_no_improve=200)
            if sc1600c < best_score:

    if tleft() > 80:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_final_cd, sc_final = interceptive_cd(best_h, n_cur, tgt_cur,
                                                time_limit=min(60, tleft() * 0.5),
                                                max_sweeps=10000,
                                                n_cand_start=30, n_cand_end=80,
                                                max_no_improve=200)
        if sc_final < best_score:

    if tleft() > 40:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_final_cp, sc_final2 = cutting_plane_method(best_h, n_cur, tgt_cur,

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
