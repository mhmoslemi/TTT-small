def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):

def softmax_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    p = np.exp(shifted)
    p = p / p.sum()
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    active_thresh = 1e-8
    for k_idx in range(len(corr)):
        if p[k_idx] < active_thresh:
            continue
        k = k_idx - mid
        w = p[k_idx]
        v1 = (idxs + k >= 0) & (idxs + k < n)

        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]

            h = h_new

    return best_h, best_score


def simulated_annealing(h_init, n, target, time_limit, n_changes=4,
                        t_init=0.1, t_end=0.005, max_iters=500000):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    rng = np.random.RandomState(42)
    corr = np.correlate(h, 1.0 - h, mode='full')
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        progress = it / max_iters
        temperature = t_init * (t_end / t_init) ** progress

        i_idx = rng.randint(n, size=n_changes)
        j_idx = rng.randint(n, size=n_changes)

        old_vals = h[i_idx].copy()
        new_vals = h[j_idx].copy()
        h[i_idx] = new_vals
        h[j_idx] = old_vals
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)

        corr_new = np.correlate(h, 1.0 - h, mode='full')
        sc_new = float(np.max(corr_new)) * (2.0 / n)

        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        diff = sc_new - compute_c5(h)
        if diff > 0 and rng.random() < np.exp(-diff / max(temperature, 1e-10)):
            corr = corr_new
        else:
            h[i_idx] = old_vals
            h[j_idx] = new_vals
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')

        if no_improve > 20000:
            break

    return best_h, best_score


def multi_resolution_refine(h_init, n, target, time_limit):
    start = time.time()
    h = h_init.copy()
    best_score = compute_c5(h)
    best_h = h.copy()

    for res_factor in [2, 4]:
        if time.time() - start > time_limit:
            break
        n_low = n // res_factor
        if n_low < 50:
            continue
        target_low = n_low / 2.0

        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, n_low)
        h_low = np.interp(x_new, x_old, h)
        h_low = project_feasible(h_low, target_low)

        sc_low = compute_c5(h_low)
        remaining = time_limit - (time.time() - start)
        t_per = remaining / (res_factor + 1)

        if t_per > 10:
            h_low2, sc_low2 = transfer_local_search(h_low, n_low, target_low,
                                                    time_limit=t_per * 0.5,
                                                    max_iters=100000,
                                                    n_active_shifts=2)
            if sc_low2 < sc_low:
                h_low = h_low2
                sc_low = sc_low2

        x_new2 = np.linspace(0.0, 1.0, n)
        h_up = np.interp(x_new2, x_new, h_low)
        h_up = project_feasible(h_up, target)
        sc_up = compute_c5(h_up)

        if sc_up < best_score - 1e-14:
            best_score = sc_up
            best_h = h_up.copy()

    return best_h, best_score


def slsqp_polish(h_init, n, target, alphas, maxiter_per_alpha, time_limit):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    start = time.time()

    def _smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def _cons_fun(x):
        return np.sum(x) - _g_target

    def _cons_jac(x):
        return np.ones(_g_n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, cur_score


def cutting_plane_refine(h_init, n, target, time_limit, max_iters=5000):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    idxs = np.arange(n)

    cuts_g = []
    cuts_f = []
    cuts_h = []
    max_cuts = 100
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 1000:
            break

        k_star = int(np.argmax(corr))
        k = k_star - mid
        f_val = float(corr[k_star])

        g = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        g[v1] += (1.0 - h[idxs[v1] - k])
        g[v2] -= h[idxs[v2] + k]

        cuts_g.append(g)
        cuts_f.append(f_val)
        cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        h_f, sc_f = fourier_parametric_optimize(n0, target0,
                                                 time_limit=min(40, tleft() * 0.3),
                                                 n_modes=8)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f

    if tleft() > 200:
        h_ng, sc_ng = nesterog_gradient(best_h, n0, target0,
                                        time_limit=min(180, tleft() * 0.4),
                                        alpha=1000.0, n_iters=100000)
        if sc_ng < best_score:
            best_score = sc_ng
            best_h = h_ng

    if tleft() > 150:
        h_tls, sc_tls = transfer_local_search(best_h, n0, target0,
                                              time_limit=min(130, tleft() * 0.4),
                                              max_iters=300000, n_active_shifts=3)
        if sc_tls < best_score:
            best_score = sc_tls
            best_h = h_tls

    if tleft() > 120:
        h_qp, sc_qp = active_shift_qp(best_h, n0, target0,
                                      time_limit=min(100, tleft() * 0.4),
                                      max_iters=300)
        if sc_qp < best_score:
            best_score = sc_qp
            best_h = h_qp

    if tleft() > 100:
        h_cp, sc_cp = cutting_plane_refine(best_h, n0, target0,
                                           time_limit=min(80, tleft() * 0.4),
                                           max_iters=3000)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp

    if tleft() > 150:
        h_ng2, sc_ng2 = nesterog_gradient(best_h, n0, target0,
                                          time_limit=min(130, tleft() * 0.4),
                                          alpha=5000.0, n_iters=100000)
        if sc_ng2 < best_score:
            best_score = sc_ng2
            best_h = h_ng2

    if tleft() > 120:
        h_tls2, sc_tls2 = transfer_local_search(best_h, n0, target0,
                                                time_limit=min(100, tleft() * 0.4),
                                                max_iters=200000, n_active_shifts=5)
        if sc_tls2 < best_score:
            best_score = sc_tls2
            best_h = h_tls2

    if tleft() > 100:
        h_mr, sc_mr = multi_resolution_refine(best_h, n0, target0,
                                              time_limit=min(80, tleft() * 0.4))
        if sc_mr < best_score:
            best_score = sc_mr
            best_h = h_mr

    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for ns in [0.03, 0.08, 0.15]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(n0) * ns
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc_ref = transfer_local_search(h_pert, n0, target0,
                                                  time_limit=min(30, tleft() * 0.3),
                                                  max_iters=100000, n_active_shifts=3)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 80:
        h_sls, sc_sls = slsqp_polish(best_h, n0, target0,
                                     alphas=[1000, 5000, 20000],
                                     maxiter_per_alpha=400,
                                     time_limit=min(70, tleft() * 0.4))
        if sc_sls < best_score:
            best_score = sc_sls
            best_h = h_sls

    if tleft() > 60:
        h_cp2, sc_cp2 = cutting_plane_refine(best_h, n0, target0,
                                             time_limit=min(50, tleft() * 0.4),
                                             max_iters=3000)
        if sc_cp2 < best_score:
            best_score = sc_cp2
            best_h = h_cp2

    if tleft() > 50:
        h_tls3, sc_tls3 = transfer_local_search(best_h, n0, target0,
                                                time_limit=min(40, tleft() * 0.4),
                                                max_iters=200000, n_active_shifts=5)
        if sc_tls3 < best_score:
            best_score = sc_tls3
            best_h = h_tls3

    if tleft() > 40:
        h_sls2, sc_sls2 = slsqp_polish(best_h, n0, target0,
                                       alphas=[20000, 80000],
                                       maxiter_per_alpha=500,
                                       time_limit=min(35, tleft() * 0.4))
        if sc_sls2 < best_score:
            best_score = sc_sls2
            best_h = h_sls2

    if tleft() > 20:
        h_nl, sc_nl = nesterog_gradient(best_h, n0, target0,
                                        time_limit=min(15, tleft() * 0.3),
                                        alpha=50000.0, n_iters=50000)
        if sc_nl < best_score:
            best_score = sc_nl
            best_h = h_nl

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
