import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)

def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def subgradient_method(h_init, n, target, time_limit, max_steps=50000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    lr = 0.05
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_val = float(c_max) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        # Find active shifts (within tolerance of max)
        tol = 1e-6
        active = np.where(corr >= c_max - tol)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for ai in active:
            k = int(ai) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
            
        h_try = h - lr * grad / gnorm
        h_try = project_feasible(h_try, target)
        
        sc = compute_c5(h_try)
        if sc < c5_val - 1e-15:
            h = h_try
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            if lr < 1e-8:
                break
                
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_cd(h_init, n, target, time_limit, max_sweeps=50000,

    best_c5 = compute_c5(h)
    best_h = h.copy()

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)

        grad[v2] -= h[idxs[v2] + k_star]

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def subgradient_method(h_init, n, target, time_limit, max_steps=50000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    lr = 0.04
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_val = float(c_max) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        tol = max(1e-8, 1e-6 * c_max)
        active = np.where(corr >= c_max - tol)[0]

        grad = np.zeros(n)
        for ai in active:
            k = int(ai) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(active))
        grad -= np.mean(grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break

        h_try = h - lr * grad / (gnorm + 1e-15)
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < c5_val - 1e-15:
            h = h_try
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.8
            if lr < 1e-8:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def thue_morse_init(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = 1.0 - t.astype(float)
    target = n / 2.0
    return project_feasible(h, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    if tleft() > 100:
        h_cd, sc_cd = fast_cd(best_h, best_n, best_target,
                              time_limit=min(80, tleft() * 0.7),
                              max_sweeps=30000, n_cand_start=25, n_cand_end=60,
                              max_no_improve=300, bisect_iters=10)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol, sc_pol = slsqp_round(best_h, best_n, best_target,
                                    [200, 1000, 3000], 600, 1e-15)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    n1 = 600
    if tleft() > 320:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 250:
            h1_cd, sc1_cd = fast_cd(h1, n1, target1,
                                    time_limit=min(160, tleft() * 0.65),
                                    max_sweeps=35000, n_cand_start=25, n_cand_end=70,
                                    max_no_improve=400, bisect_iters=10)
            if sc1_cd < best_score:
                best_score = sc1_cd
                best_h = h1_cd
                best_n = n1
                best_target = target1

        if tleft() > 80:
            h1_pol, sc1_pol = slsqp_round(best_h, best_n, best_target,
                                          [1000, 5000, 10000], 800, 1e-15)
            if sc1_pol < best_score:
                best_score = sc1_pol
                best_h = h1_pol

        if tleft() > 60:
            h1_ms, sc1_ms = multi_shift_cd(best_h, best_n, best_target,
                                           time_limit=min(50, tleft() * 0.7),
                                           n_cand=45, n_active=4, max_no_improve=250)
            if sc1_ms < best_score:
                best_score = sc1_ms
                best_h = h1_ms

    n2 = 800
    if tleft() > 350:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        if tleft() > 280:
            h2_cd, sc2_cd = fast_cd(h2, n2, target2,
                                    time_limit=min(180, tleft() * 0.65),
                                    max_sweeps=40000, n_cand_start=30, n_cand_end=80,
                                    max_no_improve=400, bisect_iters=10)
            if sc2_cd < best_score:
                best_score = sc2_cd
                best_h = h2_cd
                best_n = n2
                best_target = target2

        if tleft() > 100:
            h2_pol, sc2_pol = slsqp_round(best_h, best_n, best_target,
                                          [2000, 8000, 15000], 1000, 1e-15)
            if sc2_pol < best_score:
                best_score = sc2_pol
                best_h = h2_pol

        if tleft() > 60:
            h2_ms, sc2_ms = multi_shift_cd(best_h, best_n, best_target,
                                           time_limit=min(50, tleft() * 0.7),
                                           n_cand=55, n_active=5, max_no_improve=200)
            if sc2_ms < best_score:
                best_score = sc2_ms
                best_h = h2_ms

    if tleft() > 80:
        h_sub, sc_sub = subgradient_method(best_h, best_n, best_target,
                                           time_limit=min(60, tleft() * 0.7))
        if sc_sub < best_score:
            best_score = sc_sub
            best_h = h_sub

    if tleft() > 60:
        h_fc, sc_fc = fast_cd(best_h, best_n, best_target,
                              time_limit=min(50, tleft() * 0.7),
                              max_sweeps=20000, n_cand_start=50, n_cand_end=100,
                              max_no_improve=200, bisect_iters=10)
        if sc_fc < best_score:
            best_score = sc_fc
            best_h = h_fc

    if tleft() > 50:
        h_fp, sc_fp = slsqp_round(best_h, best_n, best_target,
                                  [5000, 20000, 50000], 1500, 1e-15)
        if sc_fp < best_score:
            best_score = sc_fp
            best_h = h_fp

    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.06, 0.10]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            h_ref, sc_ref = slsqp_round(h_pert, best_n, best_target,
                                        [2000, 10000], 500, 1e-14)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 30:

                                          time_limit=min(20, tleft() * 0.5),
                                          max_sweeps=5000, n_cand_start=50, n_cand_end=80,
                                          max_no_improve=150, bisect_iters=10)

                    best_h = h_ref2

        h_tr, sc_tr = fast_cd(best_h, best_n, best_target,
                              time_limit=min(20, tleft() * 0.7),
                              max_sweeps=5000, n_cand_start=60, n_cand_end=100,
                              max_no_improve=100, bisect_iters=10)
        if sc_tr < best_score:
            best_score = sc_tr
            best_h = h_tr

    if tleft() > 20:
        h_lp, sc_lp = slsqp_round(best_h, best_n, best_target,
                                  [50000, 100000], 1000, 1e-15)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
