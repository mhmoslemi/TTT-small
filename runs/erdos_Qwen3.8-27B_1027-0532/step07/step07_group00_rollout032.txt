                          momentum_beta=0.9, step_init=0.01,
                          top_k=50, n_iter=None):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    step = step_init
    
    while time.time() - start < time_limit:

        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find top-K shifts
        top_idx = np.argsort(corr)[-top_k:]
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = corr[s_idx]
            k = int(s_idx) - mid
            if k >= 0:
                # ∂f_k/∂h[j] = (1-h[j+k]) for j in [0, n-1-k]
                #              + (-h[j-k]) for j in [k, n-1]

                grad[valid1] += w * (1.0 - h[k:])  # 1 - h[j+k] for j in [0, n-1-k]
                valid2 = slice(k, n)
                grad[valid2] += w * (-h[:n-k])  # -h[j-k] for j in [k, n-1]
            elif k < 0:
                m = -k
                valid1 = slice(m, n)
                grad[valid1] += w * (1.0 - h[:n-m])  # 1 - h[j-m] for j in [m, n-1]
                valid2 = slice(0, n - m)
                grad[valid2] += w * (-h[m:])  # -h[j+m] for j in [0, n-1-m]
            else:
                grad += w * (1.0 - 2.0 * h)
        
        # Remove mean to maintain constraint
        grad -= np.mean(grad)
        
        # Heavy ball update
        momentum = momentum_beta * momentum - step * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, target)
        
        # Check if it improved
        if compute_c5(h_new) < c5_val:
            h = h_new
            step = min(step * 1.02, 0.1)
        else:
            step *= 0.95
            h = h_new  # still update for exploration
    
    return best_h, best_c5

A "perfect" difference set of order n would have the property that every nonzero difference appears exactly λ times. If we could find such a set of size n/2, the overlap would be constant for all shifts, giving the minimum possible max overlap.

For n=800, a perfect difference set of size 400 would require n = 400² - 400 + 1 = 159601 (from the Singer construction), which is way larger than 800. So a perfect difference set doesn't exist for n=800. But we can get close.

The key parameter is: for a set S of size m in Z_n, the "difference multiset" is {s1 - s2 : s1, s2 ∈ S, s1 ≠ s2}. The autocorrelation at shift k is the number of pairs (s1, s2) with s1 - s2 = k.

For our problem (with the [0,2] domain and the full correlation), the structure is slightly different because of the boundary effects, but the principle is the same.

This suggests that a good initial guess could be based on a "nearly perfect" difference set. But constructing one is non-trivial.

The critical shifts (those achieving the max) are the bottleneck. If we can identify them and focus on reducing them, we can make faster progress.

For a 1D binary-like sequence, the critical shifts are typically:

One more thought: the previous code's pair CD selects candidates based on the gradient (direction of steepest descent). But the actual improvement depends on both the gradient AND the box constraints. A position with a large gradient but h[i] already near 0 won't give much improvement (since δ_hi = h[i] is small).

A better selection criterion: for each position i, compute:
- gradient_g[i] (from the weighted correlation)
- potential[i] = |gradient_g[i]| * min(h[i], 1-h[i]) (the "available room" in the direction of the gradient)

Positions with high potential are the best candidates for moves.

Actually, even better: the improvement from moving δ from i to j is approximately:
improvement ≈ δ * (gradient_g[i] - gradient_g[j]) (for small δ)

where gradient_g[i] is the partial derivative of C₅ with respect to h[i].

So the best pair is one where gradient_g[i] - gradient_g[j] is maximized (i.e., i has the most positive gradient and j has the most negative gradient), subject to the constraint that δ is large enough to matter.

import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=20, n_cand_end=80, max_no_improve=500,
                 weight_beta=5.0, top_k_shifts=30):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        a_max = float(np.max(corr))
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        # Compute weights for gradient
        weights = np.exp(weight_beta * (corr - a_max) / max(a_max, 1e-10))
        
        # Gradient from top-k shifts
        grad = np.zeros(n)
        top_idx = np.argsort(weights)[-top_k_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Candidate selection
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                # Closed-form optimal delta
                g = change[j] - change[i]  # direction: decrease h[i], increase h[j]
                
                # Check if move improves: need max_{k active} g[k] <= 0
                # Active set: k where corr[k] >= a_max - eps
                active_mask = corr >= a_max - 1e-10
                if g[active_mask].max() > 1e-10:
                    continue  # move makes active correlations worse
                
                # Find delta_crit = min_{k: g[k] > 0} (a_max - corr[k]) / g[k]
                pos_mask = g > 1e-15
                if pos_mask.any():
                    delta_crit = np.min((a_max - corr[pos_mask]) / g[pos_mask])
                else:
                    delta_crit = np.inf
                
                if delta_crit <= delta_lo + 1e-12:
                    continue
                
                delta_star = min(delta_hi, delta_crit)
                if delta_star <= delta_lo + 1e-12:
                    continue
                
                # Compute actual C5 after the move
                approx_max = np.max(corr + delta_star * g)
                
                if approx_max * dx < best_trial_c5:
                    best_trial_c5 = approx_max * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    
    return best_h, best_c5

The effect on correlation at shift k:

- Actually, let me use the change matrix. change[i][k] is the derivative of correlation[k] with respect to h[i].

So the new correlation is: corr_new[k] = corr[k] + δ * (change[j][k] - change[i][k])

Let g_k = change[j][k] - change[i][k].

We want: max_k (corr[k] + δ * g_k) < a_max

For this to hold:
- For all k with corr[k] = a_max (active): we need δ * g_k ≤ 0 (strictly < for strict improvement)
- For all k with corr[k] < a_max: we need corr[k] + δ * g_k ≤ a_max, i.e., δ ≤ (a_max - corr[k]) / g_k if g_k > 0

So:
1. If max_{k active} g_k > 0: the move immediately makes an active correlation worse. Skip.
2. Otherwise: δ_crit = min_{k: g_k > 0} (a_max - corr[k]) / g_k (or ∞ if no such k)
3. δ* = min(δ_hi, δ_crit)
4. If δ* > δ_lo: the move is valid. The new max is max_k(corr[k] + δ* * g_k).

def slp_optimize(h_init, n, target, time_limit):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for slp_iter in range(50):
        if time.time() - start > time_limit:
            break
        
        # Build LP
        h_lp = solve_slp_lp(h, n, target)
        if h_lp is None:
            break
        
        # Line search
        best_alpha = 0
        best_alpha_c5 = compute_c5(h)
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            h_try = project_feasible(h + alpha * (h_lp - h), target)
            c5_try = compute_c5(h_try)
            if c5_try < best_alpha_c5:
                best_alpha = alpha
                best_alpha_c5 = c5_try
        
        if best_alpha > 0:
            h = project_feasible(h + best_alpha * (h_lp - h), target)
            if compute_c5(h) < best_c5:
                best_c5 = compute_c5(h)
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5

def solve_slp_lp(h, n, target):
    n_corr = 2 * n - 1
    mid = n - 1
    
    A = np.zeros((n_corr, n + 1))
    b = np.zeros(n_corr)
    
    for corr_idx in range(n_corr):
        k = corr_idx - mid
        if k >= 0:
            const = np.sum(h[:n-k])
            A[corr_idx, k:n] = -h[:n-k]
        else:
            m = -k
            const = np.sum(h[m:])
            A[corr_idx, 0:n-m] = -h[m:]
        A[corr_idx, n] = -1.0
        b[corr_idx] = -const
    
    A_eq = np.zeros((1, n + 1))
    A_eq
