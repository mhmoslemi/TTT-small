- For each point i, compute how much the maximum would decrease if we decreased h[i] by epsilon
- Select the top points by this sensitivity

This is essentially what the gradient does, but computed more accurately.

For a small decrease dh[i] = -eps:
The correlation changes by: corr_new[m] = corr[m] - eps * change[i][m]
The new max is max_m (corr[m] - eps * change[i][m])
≈ max(corr) - eps * max_{m in active} change[i][m] (first order)

So the sensitivity of point i is: max_{m in active} change[i][m]
(where "active" means m where corr[m] is near the maximum)

But we need to maintain the sum constraint, so we also need to increase some other point j. The net effect is:
Δ = -eps * change[i] + eps * change[j] = eps * (change[j] - change[i])
The new max is max_m (corr[m] + eps * (change[j][m] - change[i][m]))
≈ max(corr) + eps * max_{m in active} (change[j][m] - change[i][m])

So we want to minimize max_{m in active} (change[j][m] - change[i][m]), which means:
- We want change[i][m] to be large for active m (decreasing h[i] helps)
- We want change[j][m] to be small for active m (increasing h[j] hurts less)

This is exactly what the gradient-based selection does! The gradient for the active shift k is:
grad[i] = (1-h[i-k]) for i in the "positive" part, -h[i+k] for the "negative" part

        decay = np.exp(-0.8 * np.arange(n_active) / max(1.0, n_active / 3.0))

        w_norm = float(np.sum(w))
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)
        if no_improve > max_no_improve // 2:
            ncand = max(10, ncand - 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = cand_map[i]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = cand_map[j]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue

                g = G[lr] - G[hr]

                a_d = delta_min
                b_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = float(np.max(g[active_mask]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = float(np.max(corr + delta_star * g))
                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            if no_improve > 50 and no_improve % 80 == 0:
                rng = np.random.RandomState(sweep * 7 + 13)
                n_pert = max(3, n // 80)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                scale = lr_perturb / (1.0 + 0.1 * (no_improve // 100))
                h[pert_idx] += rng.randn(n_pert) * scale
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                for p in pert_idx:
                    update_change_row(change, int(p), h, n, shifts)
                no_improve = min(no_improve, max_no_improve - 50)

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, max_steps=20000, block_sizes=(2, 3, 4, 5, 6, 8)):
    start = time.time()
    rng = np.random.RandomState(31337)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = 5
        top_idx = np.argsort(corr)[::-1][:n_active]
        w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        bs = block_sizes[step % len(block_sizes)]
        n_pos = n - bs + 1

        block_grads = np.array([grad[sp:sp + bs].sum() for sp in range(n_pos)])
        best_src = int(np.argmin(block_grads))
        best_dst = int(np.argmax(block_grads))

        if best_src == best_dst:
            continue

        if block_grads[best_src] > block_grads[best_dst]:
            src_start, dst_start = best_src, best_dst
        else:
            src_start, dst_start = best_dst, best_src

        src_block = h[src_start:src_start + bs]
        dst_block = h[dst_start:dst_start + bs]

        delta_max = min(float(src_block.max()), float(1.0 - dst_block.min()))
        delta_min = max(float(-src_block.min()), float(dst_block.min() - 1.0))
        if delta_max - delta_min < 1e-12:
            continue

        g = np.zeros(n_corr)
        for t in range(bs):
            g += change[dst_start + t, :] - change[src_start + t, :]

        a_d = delta_min
        b_d = delta_max
        for _ in range(14):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr + mid_d * g
            f_mid = float(np.max(vals))
            active_mask = vals >= f_mid - 1e-14
            if active_mask.any():
                f_prime = float(np.max(g[active_mask]))
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                b_d = mid_d
            else:
                a_d = mid_d

        delta_star = 0.5 * (a_d + b_d)
        if abs(delta_star) < 1e-12:
            continue

        approx_max = float(np.max(corr + delta_star * g))
        if approx_max < c5_val - 1e-14:
            h[src_start:src_start + bs] -= delta_star
            h[dst_start:dst_start + bs] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for t in range(bs):
                update_change_row(change, src_start + t, h, n, shifts)
                update_change_row(change, dst_start + t, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_step(h_init, n, target, time_limit, max_iter=20000, lr_start=0.0015):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_corr = float(np.max(corr))
        tol = max(1e-10, 1e-7 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)

        progress = min(it / max_iter, 1.0)
        lr = lr_start * (1.0 - 0.9 * progress) ** 2

        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val - 1e-15:
            h = h_new

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 50:
        h_pol = slsqp_round(best_h, n0, target0, [100, 500, 2000], 400)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n1 = 1600
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)

    if tleft() > 450:
        h_cd, c5_cd = improved_cd(h1, n1, target1,
                                   time_limit=min(420, tleft() * 0.55),
                                   max_sweeps=60000,
                                   n_cand_start=15, n_cand_end=45,
                                   max_no_improve=600,
                                   n_active_start=5, n_active_end=12)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        h_pol = slsqp_round(best_h, n1, target1, [500, 2000], 500)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        h_cd2, c5_cd2 = improved_cd(best_h, n1, target1,
                                     time_limit=min(200, tleft() * 0.55),
                                     max_sweeps=50000,
                                     n_cand_start=20, n_cand_end=55,
                                     max_no_improve=400,
                                     n_active_start=8, n_active_end=18)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 150:
        h_bm, c5_bm = block_cd(best_h, n1, target1,
                               time_limit=min(120, tleft() * 0.55),
                               max_steps=20000)
        if c5_bm < best_score:
            best_score = c5_bm
            best_h = h_bm

    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.015, 0.04]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n1) * noise_scale
            h_pert = project_feasible(h_pert, target1)
            h_ref = slsqp_round(h_pert, n1, target1, [200, 1000], 300)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 120:
        h_cd3, c5_cd3 = improved_cd(best_h, n1, target1,
                                     time_limit=min(90, tleft() * 0.55),
                                     max_sweeps=40000,
                                     n_cand_start=25, n_cand_end=60,
                                     max_no_improve=300,
                                     n_active_start=10, n_active_end=22)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 80:
        h_sg, c5_sg = subgradient_step(best_h, n1, target1,
                                        time_limit=min(50, tleft() * 0.5),
                                        max_iter=15000, lr_start=0.001)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 50:
        h_final = slsqp_round(best_h, n1, target1, [5000, 20000], 400)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, n1 / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, n1
