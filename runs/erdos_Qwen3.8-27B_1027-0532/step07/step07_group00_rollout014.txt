def lp_subgradient_step(h, n, active_lags, eps=0.05):
    dx = 2.0 / n
    shifts = np.arange(-(n-1), n)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_active = [corr[k + n - 1] for k in active_lags]
    
    # Compute gradients for active lags
    # grad_k[j] = (1-h[j+k]) - h[j-k] (with boundary handling)
    grads = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        for j in range(n):
            jp = j + k
            jm = j - k
            grad = 0.0
            if 0 <= jp < n:
                grad += (1.0 - h[jp])
            if 0 <= jm < n:
                grad -= h[jm]
            grads[idx, j] = grad
    
    # LP: minimize t
    # s.t. t - sum_j grad_k[j]*delta_j >= c_k for all active k
    #      sum(delta) = 0
    #      -h[j] <= delta[j] <= 1-h[j]
    #      |delta[j]| <= eps
    
    # Variables: t, delta[0], ..., delta[n-1] -> total n+1 vars
    # Minimize: t -> c = [1, 0, ..., 0]
    
    # Constraints:
    # For each active k: t - sum_j grad_k[j]*delta_j >= c_k
    #   -> -t + sum_j grad_k[j]*delta_j <= -c_k
    # sum(delta) = 0
    # -h[j] - eps <= delta[j] <= 1-h[j] - eps (tighter bounds)
    
    # Use scipy.optimize.linprog
    from scipy.optimize import linprog
    
    m = len(active_lags)
    # A_ub: m + 1 constraints (m for active lags, 1 for sum=0)
    A_ub = np.zeros((m + 1, n + 1))
    b_ub = np.zeros(m + 1)
    
    for idx in range(m):
        A_ub[idx, 0] = -1.0  # coefficient of t
        A_ub[idx, 1:] = grads[idx]  # coefficients of delta
        b_ub[idx] = -c_active[idx]
    
    A_ub[m, 1:] = 1.0  # sum(delta) = 0 -> <= 0 and >= 0
    b_ub[m] = 0.0
    
    # Actually, we need equality constraint for sum(delta)=0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = [0.0]
    
    # Bounds: t is free (or bounded below by 0), delta bounded
    bounds = [(None, None)] + [(-min(h[j], eps), min(1-h[j], eps)) for j in range(n)]
    
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.status == 0:
        delta = res.x[1:]
        return h + delta
    return h

Given the current h and the current correlations corr[k], the change in correlation at lag m when we change h[i] by -δ and h[j] by +δ is:

Δc_m = -δ * ∂c_m/∂h[i] + δ * ∂c_m/∂h[j]

where ∂c_m/∂h[i] = (1-h[i+m]) - h[i-m] (with boundary handling)

So Δc_m = δ * (∂c_m/∂h[j] - ∂c_m/∂h[i])

Let g_m = ∂c_m/∂h[j] - ∂c_m/∂h[i] (the "change vector" for moving mass from i to j).

The new correlation at lag m is: corr[m] + δ * g_m

We want to minimize max_m (corr[m] + δ * g_m) over δ ∈ [δ_min, δ_max].

This is a 1D optimization problem! We can solve it exactly:
- The function f(δ) = max_m (corr[m] + δ * g_m) is a convex piecewise-linear function of δ.
- Its minimum is at a "kink" point, which occurs where two of the lines corr[m] + δ * g_m are equal.

So the optimal δ is either:
- δ_min or δ_max (boundary), or
- A kink point: δ* = (corr[m1] - corr[m2]) / (g[m2] - g[m1]) for some pair (m1, m2) where g[m1] ≠ g[m2]

We can find the optimal δ by:
1. Computing all kink points
2. Filtering to those in [δ_min, δ_max]
3. Evaluating f at each kink point and at the boundaries
4. Taking the minimum

def vectorized_bisection(corr, g_matrix, delta_lo, delta_hi, n_iters=16):
    """
    corr: shape (n_corr,)
    g_matrix: shape (T, n_corr) - change vectors for T pairs
    delta_lo: shape (T,) - lower bounds for each pair
    delta_hi: shape (T,) - upper bounds for each pair
    Returns: optimal delta for each pair
    """
    T = g_matrix.shape[0]
    a = delta_lo.copy()
    b = delta_hi.copy()
    
    for _ in range(n_iters):
        mid = 0.5 * (a + b)
        # vals: (T, n_corr)
        vals = corr[None, :] + mid[:, None] * g_matrix
        f_mid = vals.max(axis=1)
        # Find if f'(mid) > 0 for each pair
        # f'(mid)_t = max(g_matrix[t, m] for m where vals[t, m] = f_mid[t])
        # Approximation: check if max(g_matrix[t, vals[t,:] >= f_mid[t] - tol]) > 0
        tol = 1e-14
        active = vals >= f_mid[:, None] - tol
        # g_prime[t] = max(g_matrix[t, active[t, :]])
        # To vectorize: set inactive entries to -inf, then take max
        g_masked = np.where(active, g_matrix, -np.inf)
        g_prime = g_masked.max(axis=1)
        
        # If g_prime > 0, f is increasing, so optimal is to the left -> b = mid
        # If g_prime <= 0, f is decreasing or flat, so optimal is to the right -> a = mid
        inc = g_prime > 1e-12
        b[inc] = mid[inc]
        a[~inc] = mid[~inc]
    
    return 0.5 * (a + b)

            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                lo = max(h[i] - 1.0, -h[j])
                hi = min(h[i], 1.0 - h[j])
                if hi - lo < 1e-12:
                    continue
                pairs_i.append(i)
                pairs_j.append(j)
                d_lo.append(lo)
                d_hi.append(hi)
        
        if len(pairs_i) == 0:
            continue
        
        pairs_i = np.array(pairs_i)
        pairs_j = np.array(pairs_j)
        d_lo = np.array(d_lo)
        d_hi = np.array(d_hi)
        
        T_pairs = len(pairs_i)
        # Cap the number of pairs to avoid memory issues
        max_pairs = 5000
        if T_pairs > max_pairs:
            # Randomly subsample
            idx_sel = rng.choice(T_pairs, max_pairs, replace=False)
            pairs_i = pairs_i[idx_sel]
            pairs_j = pairs_j[idx_sel]
            d_lo = d_lo[idx_sel]
            d_hi = d_hi[idx_sel]
            T_pairs = max_pairs
        
        # Compute g vectors
        g_matrix = change[pairs_j] - change[pairs_i]  # shape (T_pairs, n_corr)
        
        # Vectorized bisection
        delta_opt = vectorized_bisection(corr, g_matrix, d_lo, d_hi, n_iters=16)
        
        # Evaluate resulting C5 for each pair
        # new_corr_max = max_m (corr[m] + delta_opt[t] * g_matrix[t, m])
        vals = corr[None, :] + delta_opt[:, None] * g_matrix
        new_c5 = vals.max(axis=1) * dx
        
        # Find the best move
        best_idx = np.argmin(new_c5)
        best_trial_c5 = new_c5[best_idx]
        
        if best_trial_c5 < c5_val - 1e-15:
            i = int(pairs_i[best_idx])
            j = int(pairs_j[best_idx])
            delta = delta_opt[best_idx]
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
        elif T > 1e-12:
            # Accept with probability based on annealing
            best_accept = np.where(new_c5 < c5_val + T * 3)[0]
            if len(best_accept) > 0:
                probs = np.exp(-(new_c5[best_accept] - c5_val) / T)
                probs /= probs.sum()
                accept_idx = best_accept[rng.choice(len(best_accept), p=probs)]
                i = int(pairs_i[accept_idx])
                j = int(pairs_j[accept_idx])
                delta = delta_opt[accept_idx]
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
    
    return best_h, best_c5


def lp_refinement(h_init, n, target, time_limit, max_iters=200, top_k=15, eps_bound=0.05):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 100:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        # Find active lags
        c_max = float(np.max(corr))
        active_thresh = c_max - 0.02 / dx  # within 0.02 of max (in C5 units)
        active_mask = corr >= min(active_thresh, c_max * 0.99)
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = [int(np.argmax(corr))]
        active_idx = active_idx[:top_k]
        
        # Compute gradients for active lags
        # grad_k[j] = (1-h[j+k]) - h[j-k] (with boundary handling)
        grads = np.zeros((len(active_idx), n))
        for a_idx in range(len(active_idx)):
            k = int(active_idx[a_idx]) - mid
            for j in range(n):
                g = 0.0
                jp = j + k
                jm = j - k
                if 0 <= jp < n:
                    g += (1.0 - h[jp])
                if 0 <= jm < n:
                    g -= h[jm]
                grads[a_idx, j] = g
        
        # LP: minimize t
        # s.t. t - sum_j grad_k[j] * delta_j >= corr[k] for each active k
        #      sum(delta) = 0
        #      -min(h[j], eps) <= delta[j] <= min(1-h[j], eps)
        
        m_act = len(active_idx)
        # Variables: [t, delta_0, ..., delta_{n-1}]
        c_obj = np.zeros(n + 1)
        c_obj[0] = 1.0
        
        # A_ub * x <= b_ub
        # For each active k: -t + sum_j grad_k[j]*delta_j <= -corr[k_active]
        A_ub = np.zeros((m_act, n + 1))
        b_ub = np.zeros(m_act)
        for a_idx in range(m_act):
            A_ub[a_idx, 0] = -1.0
            A_ub[a_idx, 1:] = grads[a_idx]
            b_ub[a_idx] = -corr[int(active_idx[a_idx])]
        
        # Equality: sum(delta) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, 1:] = 1.0
        b_eq = np.zeros(1)
        
        # Bounds
        bounds = [(None, None)]  # t is free
        for j in range(n):
            lo = -min(h[j], eps_bound)
            hi = min(1.0 - h[j], eps_bound)
            bounds.append((lo, hi))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            no_improve += 10
            continue
        
        if res.status != 0:
            no_improve += 1
            continue
        
        delta_star = res.x[1:]
        
        # Line search
        best_alpha = 0
        best_sc = c5_val
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            h_try = h + alpha * delta_star
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha
        
        if best_alpha > 0:
            h = h + best_alpha * delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            if best_sc < best_c5 - 1e-14:
                best_c5 = best_sc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_c5
