A more direct approach: **Frank-Wolfe (Conditional Gradient) method**.

The constraint set C = {h : 0 ≤ h ≤ 1, sum(h) = n/2} is a polytope. The Frank-Wolfe algorithm is:
1. Compute s = argmin_{v ∈ C} ∇f(h)^T v
2. Update h ← h + γ * (s - h) for some step size γ

The linear minimization over C is:
min_v ∇f(h)^T v s.t. 0 ≤ v ≤ 1, sum(v) = n/2

The solution: sort ∇f(h) and set v[i] = 1 for the n/2 smallest gradient values, 0 for the rest.

This is O(n log n) per iteration (for the sorting) plus O(n²) for the gradient computation.

The convergence of Frank-Wolfe is O(1/t), which is slow. But with the "strong step" or "away-step" variants, it can be much faster.

Let me implement the **away-step Frank-Wolfe** (also known as the "frank-wolfe with away steps" or "sparse FW"):
1. Maintain a "support" S (set of coordinates with v[i] ≠ 0)
2. At each step, compute the FW direction AND the away direction
3. Take a step in the better direction

        grad[v2] -= h[idxs[v2] + k_star]

        ncand = int(ncand_lo + (ncand_hi - ncand_lo) * progress)
        ncand = min(ncand, n // 3)
        ncand = max(ncand, 10)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            hi = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_hi = min(hi, 1.0 - h[j])
                if d_hi < 1e-13:
                    continue

                g_vec = change[j] - change[i]

                a_d = 0.0
                b_d = d_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    if f_mid > corr[np.argmax(corr)] + 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0 and best_delta > 1e-12:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_entry(change, shifts, h, best_i, n)
            update_change_entry(change, shifts, h, best_j, n)

        if sweep % binary_check_interval == 0 and binary_check_interval > 0:
            h_bin = (h >= 0.5).astype(float)
            n_ones = int(h_bin.sum())
            tgt_int = int(round(target))
            if n_ones != tgt_int:
                if n_ones > tgt_int:
                    one_idx = np.where(h_bin == 1)[0]
                    one_idx = one_idx[np.argsort(h[one_idx])]
                    h_bin[one_idx[:n_ones - tgt_int]] = 0.0
                else:
                    zero_idx = np.where(h_bin == 0)[0]
                    zero_idx = zero_idx[np.argsort(-h[zero_idx])]
                    h_bin[zero_idx[:tgt_int - n_ones]] = 1.0
            sc_bin = compute_c5(h_bin)
            if sc_bin < best_c5 - 1e-5:
                h = h_bin.copy()
                best_c5 = sc_bin
                best_h = h.copy()
                no_improve = 0
                change, shifts = build_change_matrix(h, n)

    return best_h, best_c5


def smooth_max_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqq_polish(h_init, n, target, time_limit, alphas=(500, 2000, 8000),
                 maxiter=400):
    global _g_n, _g_target
    if time.time() > 0:
        pass
    start = time.time()
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def subgradient_accelerated(h_init, n, target, time_limit,
                            n_lags=20, eta_init=0.5, decay=0.98):
    start = time.time()
    mid = n - 1
    idxs = np.arange(n)
    n_corr = 2 * n - 1

    h_curr = h_init.copy().astype(float)
    h_best = h_curr.copy()
    best_score = compute_c5(h_curr)
    h_mom = h_curr.copy()

    eta = eta_init
    t = 1

    while time.time() - start < time_limit:
        t += 1
        corr = compute_corr(h_mom)
        c_max = np.max(corr)

        # Softmax weights over top n_lags
        top_idx = np.argsort(corr)[-n_lags:]
        top_vals = corr[top_idx]
        exp_vals = np.exp((top_vals - top_vals.max()) * 50.0)
        p = exp_vals / np.sum(exp_vals)

        # Weighted gradient
        grad = np.zeros(n)
        for ki, pi in zip(top_idx, p):
            k = int(ki) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += pi * (1.0 - h_mom[idxs[v1] - k])
            grad[v2] -= pi * h_mom[idxs[v2] + k]

        # Nesterov momentum
        mu = (1.0 + np.sqrt(1.0 + 4.0 * t)) / 2.0
        h_mom = h_curr + ((mu - 1.0) / mu) * (h_curr - h_mom)
        h_mom = np.clip(h_mom, 0.0, 1.0)

        # Gradient step
        step = eta / np.sqrt(t)
        h_new = h_mom - step * grad
        h_new = project_feasible(h_new, target)

        sc = compute_c5(h_new)
        if sc < best_score - 1e-14:
            best_score = sc
            h_best = h_new.copy()

        h_curr = h_new
        eta *= decay

    return h_best, best_score


def multi_start_cd(h_init, n, target, time_limit, n_starts=5,
                   noise_levels=(0.03, 0.05, 0.08, 0.12, 0.2),
                   seed=42):
    start = time.time()
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    rng = np.random.RandomState(seed)

    for i in range(n_starts):
        if time.time() - start > time_limit:
            break
        noise = noise_levels[i % len(noise_levels)]
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        sc_pert = compute_c5(h_pert)

        remaining = time_limit - (time.time() - start)
        if remaining < 10:
            break
        t_per = max(5, remaining / (n_starts - i - 1) * 0.8)

        h_local, sc_local = improved_cd(
            h_pert, n, target, time_limit=t_per,
            max_sweeps=20000, ncand_lo=20, ncand_hi=60,
            max_no_improve=300, top_k_lags=2, binary_check_interval=0)

        if sc_local < best_score:
            best_score = sc_local
            best_h = h_local

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 20.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Long interceptive CD at original resolution
    if tleft() > 200:
        h1, sc1 = improved_cd(best_h, n0, target0,
                              time_limit=min(250, tleft() * 0.7),
                              max_sweeps=100000,
                              ncand_lo=30, ncand_hi=120,
                              max_no_improve=800,
                              top_k_lags=3, binary_check_interval=800)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: Multi-start exploration
    if tleft() > 150:
        h2, sc2 = multi_start_cd(best_h, n0, target0,
                                 time_limit=min(180, tleft() * 0.6),
                                 n_starts=4,
                                 noise_levels=(0.04, 0.07, 0.12, 0.25),
                                 seed=seed)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Subgradient accelerated method
    if tleft() > 120:
        h3, sc3 = subgradient_accelerated(best_h, n0, target0,
                                          time_limit=min(120, tleft() * 0.5),
                                          n_lags=15, eta_init=0.8, decay=0.985)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: Further CD refinement
    if tleft() > 150:
        h4, sc4 = improved_cd(best_h, n0, target0,
                              time_limit=min(180, tleft() * 0.6),
                              max_sweeps=100000,
                              ncand_lo=40, ncand_hi=130,
                              max_no_improve=1000,
                              top_k_lags=4, binary_check_interval=1000)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Upscale to higher resolution
    n_up = 600
    if tleft() > 300:
        h_up = interpolate_to(best_h, n_up, n_up / 2.0)
        sc_up = compute_c5(h_up)
        if sc_up <= best_score + 0.001:
            h_up2, sc_up2 = improved_cd(h_up, n_up, n_up / 2.0,
                                        time_limit=min(200, tleft() * 0.5),
                                        max_sweeps=80000,
                                        ncand_lo=25, ncand_hi=100,
                                        max_no_improve=600,
                                        top_k_lags=3, binary_check_interval=600)
            if sc_up2 < best_score:
                best_score = sc_up2
                best_h = h_up2
                n0 = n_up
                target0 = n_up / 2.0

    # Phase 6: Try n=800 if we have time
    n_up2 = 800
    if tleft() > 250 and n0 == n_up:
        h_up3 = interpolate_to(best_h, n_up2, n_up2 / 2.0)
        sc_up3 = compute_c5(h_up3)
        if sc_up3 <= best_score + 0.001:
            h_up4, sc_up4 = improved_cd(h_up3, n_up2, n_up2 / 2.0,
                                        time_limit=min(200, tleft() * 0.5),
                                        max_sweeps=60000,
                                        ncand_lo=20, ncand_hi=80,
                                        max_no_improve=500,
                                        top_k_lags=3, binary_check_interval=0)
            if sc_up4 < best_score:
                best_score = sc_up4
                best_h = h_up4
                n0 = n_up2
                target0 = n_up2 / 2.0

    # Phase 7: Final CD polish at best resolution
    if tleft() > 80:
        h7, sc7 = improved_cd(best_h, n0, target0,
                              time_limit=min(100, tleft() * 0.6),
                              max_sweeps=50000,
                              ncand_lo=50, ncand_hi=140,
                              max_no_improve=500,
                              top_k_lags=5, binary_check_interval=0)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 8: SLSQP polish (short)
    if tleft() > 50:
        h8, sc8 = slsqq_polish(best_h, n0, target0,
                               time_limit=min(50, tleft() * 0.5),
                               alphas=(2000, 8000, 20000), maxiter=300)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: One more CD pass
    if tleft() > 40:
        h9, sc9 = improved_cd(best_h, n0, target0,
                              time_limit=min(40, tleft() * 0.6),
                              max_sweeps=30000,
                              ncand_lo=60, ncand_hi=150,
                              max_no_improve=300,
                              top_k_lags=5, binary_check_interval=0)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
