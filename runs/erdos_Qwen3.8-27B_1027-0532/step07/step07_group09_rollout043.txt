        h = np.clip(h, 0.0, 1.0)

def alternating_lp(h_init, n, target, n_iter=200, damping=1.0):
    """Alternating LP for the biconvex reformulation"""
    a = h_init.copy()
    b = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    idxs = np.arange(n)
    
    for it in range(n_iter):
        # Fix b, optimize a
        if a is not None and b is not None:
            a_new = solve_lp_fix_b(b, n, target)
        else:
            a_new = a
        
        # Fix a_new, optimize b
        if a_new is not None:
            b_new = solve_lp_fix_a(a_new, n, target)
        else:
            b_new = b
        
        if a_new is None or b_new is None:
            break
        
        # Damping
        a = (1 - damping) * a + damping * a_new
        b = (1 - damping) * b + damping * b_new
        
        # Check the midpoint
        mid = 0.5 * (a + b)
        mid = project_feasible(mid, target)
        score = compute_c5(mid)
        if score < best_score:
            best_score = score
            best_h = mid.copy()

        for candidate in [a, b]:
            cand = project_feasible(candidate, target)
            s = compute_c5(cand)
            if s < best_score:
                best_score = s
                best_h = cand.copy()

import numpy as np
from scipy.optimize import linprog, minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def solve_lp_fix_b(b, n, target):
    a_idxs = np.arange(n)
    n_shifts = 2 * n - 1
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((n_shifts, n_vars))
    b_ub = np.zeros(n_shifts)
    for idx in range(n_shifts):
        k = idx - (n - 1)
        valid = (a_idxs + k >= 0) & (a_idxs + k < n)
        if valid.any():
            A_ub[idx, valid] = 1.0 - b[a_idxs[valid] + k]
        A_ub[idx, -1] = -1.0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def solve_lp_fix_a(a, n, target):
    a_idxs = np.arange(n)
    n_shifts = 2 * n - 1
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    A_ub = np.zeros((n_shifts, n_vars))
    b_ub = np.zeros(n_shifts)
    for idx in range(n_shifts):
        k = idx - (n - 1)
        valid = (a_idxs + k >= 0) & (a_idxs + k < n)
        if not valid.any():
            b_ub[idx] = 0.0
            A_ub[idx, -1] = -1.0
            continue
        const_k = a[a_idxs[valid]].sum()
        A_ub[idx, a_idxs[valid] + k] = -a[a_idxs[valid]]
        A_ub[idx, -1] = -1.0
        b_ub[idx] = -const_k
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def alternating_lp(h_init, n, target, time_limit, n_iter=500, start_it=0):
    start = time.time()
    a = h_init.copy()
    b = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    dx = 2.0 / n
    gap_threshold = 0.01
    for it in range(start_it, n_iter):
        if time.time() - start > time_limit:
            break
        b_new = solve_lp_fix_a(a, n, target)
        if b_new is None:
            break
        a_new = solve_lp_fix_b(b_new, n, target)
        if a_new is None:
            a_new = b_new
        a = a_new
        b = b_new
        mid = 0.5 * (a + b)
        mid = project_feasible(mid, target)
        score = compute_c5(mid)
        if score < best_score:
            best_score = score
            best_h = mid.copy()
        gap = np.max(np.abs(a - b))
        if it > 20 and gap < gap_threshold:
            for cand_list in [a, b]:
                cand = project_feasible(cand_list, target)
                s = compute_c5(cand)
                if s < best_score:
                    best_score = s
                    best_h = cand.copy()
            if gap < gap_threshold * 0.3:
                break
    return best_h, best_score


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def vectorized_pair_cd(h_init, n, target, time_limit, max_sweeps=20000,
                       n_cand_start=20, n_cand_end=55, max_no_improve=350):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = min(5, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)])

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]
                h_j = h[j]
                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue
                g = G[lr] - G[hr]
                a_d = delta_min
                b_d = delta_max
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_dispatch(h_init, n, target, time_limit, max_iter=20000, lr_start=0.003):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        max_corr = np.max(corr)
        tol = max(1e-10, 1e-7 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]
        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)
        progress = min(it / max_iter, 1.0)
        lr = lr_start * (1.0 - 0.95 * progress) ** 2
        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val * dx - 1e-15:
            h = h_new
    return best_h, best_c5
