def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)

    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

    c_max = np.max(corr)

    weights = e / total_e
    grad = np.zeros(n)

def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:

            h = np.clip(h, 0.0, 1.0)

    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):

                    best_h = h_new.copy()

    mid = n - 1
    cur_max = float(np.max(corr))

            grad[v1] += wi * (1.0 - h[idxs[v1] - k])

            best_c5 = c5_val

            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800:
                break

        grad = compute_subgradient(h, corr, n, 5, alpha_w=20.0)
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)

        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        hi_list = [i for i in high_set[:10]]
        lo_list = [j for j in low_set[:10]]
        hi_list = [i for i in hi_list if h[i] > 1e-12]
        lo_list = [j for j in lo_list if h[j] < 1.0 - 1e-12]
        if len(hi_list) < 2 or len(lo_list) < 2:
            no_improve += 10
            continue

        best_improve = 0.0
        best_move = None

        for i1 in hi_list:
            for i2 in hi_list:
                if int(i1) == int(i2):
                    continue
                ci1 = cand_pos[int(i1)]
                ci2 = cand_pos[int(i2)]
                D1 = min(float(h[i1]), 1.0)
                D2 = min(float(h[i2]), 1.0)
                for j1 in lo_list:
                    for j2 in lo_list:
                        if int(j1) == int(j2):
                            continue
                        if int(j1) == int(i1) or int(j1) == int(i2):
                            continue
                        if int(j2) == int(i1) or int(j2) == int(i2):
                            continue
                        cj1 = cand_pos[int(j1)]
                        cj2 = cand_pos[int(j2)]
                        D1b = min(D1, 1.0 - float(h[j1]))
                        D2b = min(D2, 1.0 - float(h[j2]))
                        if D1b < 1e-12 or D2b < 1e-12:
                            continue
                        g1 = profiles[cj1] - profiles[ci1]
                        g2 = profiles[cj2] - profiles[ci2]
                        g_sum = g1 + g2

                        lo = 0.0
                        hi_d = min(D1b, D2b)
                        for _ in range(16):
                            mid_d = 0.5 * (lo + hi_d)
                            vals = corr + mid_d * g_sum
                            f_mid = float(np.max(vals))
                            active = vals >= (f_mid - 1e-14)
                            if not np.any(active):
                                break
                            f_prime = float(np.max(g_sum[active]))
                            if f_prime > 1e-14:
                                hi_d = mid_d
                            else:
                                lo = mid_d
                        delta = 0.5 * (lo + hi_d)
                        if delta < 1e-10:
                            continue
                        new_max = float(np.max(corr + delta * g_sum))
                        improve = cur_max - new_max
                        if improve > best_improve:
                            best_improve = improve
                            best_move = (int(i1), int(i2), int(j1), int(j2), delta)

        if best_move is not None and best_improve > 1e-13:
            i1, i2, j1, j2, delta = best_move
            h[i1] -= delta
            h[i2] -= delta
            h[j1] += delta
            h[j2] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def simulated_annealing_cd(h_init, n, target, time_limit, n_cand=30, t_start=0.1, t_end=0.001, max_sweeps=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(12345)

    no_improve = 0
    n_total = max(max_sweeps, 1)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        frac = sweep / n_total
        T = t_start * (t_end / t_start) ** frac

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        grad = compute_subgradient(h, corr, n, 5, alpha_w=20.0)
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)

        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        valid_high = [i for i in high_set if h[i] > 1e-12]
        valid_low = [j for j in low_set if h[j] < 1.0 - 1e-12]
        if len(valid_high) == 0 or len(valid_low) == 0:
            continue

        i = int(valid_high[rng.randint(len(valid_high))])
        j = int(valid_low[rng.randint(len(valid_low))])
        ci = cand_pos[i]
        cj = cand_pos[j]
        delta_max = min(float(h[i]), 1.0 - float(h[j]))
        if delta_max < 1e-12:
            continue

        g = profiles[ci] - profiles[cj]
        lo = 0.0
        hi_d = delta_max
        for _ in range(16):
            mid_d = 0.5 * (lo + hi_d)
            vals = corr + mid_d * g
            f_mid = float(np.max(vals))
            active = vals >= (f_mid - 1e-14)
            if not np.any(active):
                break
            f_prime = float(np.max(g[active]))
            if f_prime > 1e-14:
                hi_d = mid_d
            else:
                lo = mid_d
        delta = 0.5 * (lo + hi_d)
        if delta < 1e-12:
            delta = delta_max * (0.5 + 0.5 * rng.rand())

        new_max = float(np.max(corr + delta * g))
        delta_obj = (new_max - cur_max) * dx

        if delta_obj < 0 or rng.rand() < np.exp(-delta_obj / (T + 1e-12)):
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def structured_multi_start(h_init, n, target, time_limit, n_freqs=8):
    start = time.time()
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    rng = np.random.RandomState(777)
    dx = 2.0 / n

    for trial in range(n_freqs):
        if time.time() - start > time_limit:
            break
        freq = rng.randint(1, n // 4 + 1)
        phase = rng.rand() * 2.0 * np.pi
        amp = rng.uniform(0.01, 0.04)
        x = np.arange(n)
        pert = amp * np.sin(2.0 * np.pi * freq * x / n + phase)
        h_pert = h_init + pert
        h_pert = project_feasible(h_pert, target)

        trial_time_left = time_limit - (time.time() - start) - 30
        if trial_time_left < 20:
            break
        per_trial = trial_time_left / (n_freqs - trial)

        hp, sc = lbfgs_phase(h_pert, n, target,
                            alphas=[200, 2000],
                            maxiter=100, time_limit=min(40, per_trial * 0.5),
                            penalty=200.0)
        if sc < best_c5:
            best_c5 = sc
            best_h = hp
            if time.time() - start < time_limit - 40:
                hcd, c5cd = pair_cd_improved(hp, n, target,
                                             time_limit=min(40, (time_limit - (time.time() - start)) * 0.5),
                                             n_cand=25, max_sweeps=20000, n_shifts=3)
                if c5cd < best_c5:
                    best_c5 = c5cd
                    best_h = hcd

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 80:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[50, 200, 1000],
                            maxiter=300, time_limit=min(60, tleft() * 0.3),
                            penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 200:
        cd_time = min(250, tleft() * 0.25)
        hcd, c5cd = pair_cd_improved(best_h, n, target, time_limit=cd_time,
                                     n_cand=40, max_sweeps=50000, n_shifts=3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 150:
        fp_time = min(150, tleft() * 0.2)
        hfp, c5fp = four_point_phase(best_h, n, target, time_limit=fp_time,
                                     n_cand=20, max_sweeps=20000)
        if c5fp < best_score:
            best_score = c5fp
            best_h = hfp

    if tleft() > 80:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[5000, 20000, 50000],
                            maxiter=300, time_limit=min(60, tleft() * 0.3),
                            penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 150:
        sa_time = min(150, tleft() * 0.2)
        hsa, c5sa = simulated_annealing_cd(best_h, n, target, time_limit=sa_time,
                                           n_cand=30, t_start=0.05, t_end=0.001,
                                           max_sweeps=50000)
        if c5sa < best_score:
            best_score = c5sa
            best_h = hsa

    if tleft() > 200:
        ms_time = min(180, tleft() * 0.2)
        hms, c5ms = structured_multi_start(best_h, n, target, time_limit=ms_time,
                                           n_freqs=6)
        if c5ms < best_score:
            best_score = c5ms
            best_h = hms

    if tleft() > 200:
        cd_time = min(120, tleft() * 0.25)
        hcd, c5cd = pair_cd_improved(best_h, n, target, time_limit=cd_time,
                                     n_cand=50, max_sweeps=50000, n_shifts=5)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 60:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[50000, 200000, 500000],
                            maxiter=200, time_limit=min(50, tleft() * 0.4),
                            penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 50:
        fp_time = min(40, tleft() * 0.5)
        hfp, c5fp = four_point_phase(best_h, n, target, time_limit=fp_time,
                                     n_cand=15, max_sweeps=10000)
        if c5fp < best_score:
            best_score = c5fp
            best_h = hfp

    if tleft() > 40:
        hcd, c5cd = pair_cd_improved(best_h, n, target, time_limit=min(30, tleft() * 0.6),
                                     n_cand=30, max_sweeps=20000, n_shifts=3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 30:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[1000000, 5000000, 10000000],
                            maxiter=100, time_limit=min(25, tleft() * 0.7),
                            penalty=50000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
