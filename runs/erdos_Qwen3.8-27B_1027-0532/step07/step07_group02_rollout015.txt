import time

_g_target = 0.0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-12:

        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha):

            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        c5_val = c_max * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            # Perturb
            n_pert = max(10, n // 10)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            deltas = rng.randn(n_pert) * perturb_scale
            h[pert_idx] += deltas
            h = project_feasible(h, target)
            # Recompute change for perturbed indices
            for p in pert_idx:
                p = int(p)
                change[p, :] = 0.0
                pos = (p + shifts >= 0) & (p + shifts < n)
                neg = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos] += (1.0 - h[p + shifts[pos]])
                change[p, neg] -= h[p - shifts[neg]]
                # Also update other entries affected by p's change
            # Actually, we need to rebuild the entire change matrix since
            # h changed for multiple indices
            change, shifts = build_change_matrix(h, n)
            consecutive_no_improve = 0
            continue
        
        # Gradient for candidate selection
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 2)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        n_h = len(high_set)
        n_l = len(low_set)
        
        if n_h < 2 or n_l < 2:
            continue
        
        # Check memory
        mem = n_h * n_l * n_corr * 8
        while mem > 100e6 and (n_h > 20 or n_l > 20):
            if n_h >= n_l:
                n_h = max(20, n_h // 2)
                high_set = high_set[:n_h]
            else:
                n_l = max(20, n_l // 2)
                low_set = low_set[:n_l]
            mem = n_h * n_l * n_corr * 8
        
        # Vectorized pair gradient
        ch_i = change[high_set]   # (n_h, n_corr)
        ch_j = change[low_set]    # (n_l, n_corr)
        pair_g = ch_j[None, :, :] - ch_i[:, None, :]  # (n_h, n_l, n_corr)
        
        slack = c_max - corr  # (n_corr,)
        
        # delta_safe
        pos_mask = pair_g > 1e-12
        safe_div = np.where(pos_mask, pair_g, 1.0)
        ratios = np.where(pos_mask, slack[None, None, :] / safe_div, np.inf)
        delta_safe = np.min(ratios, axis=2)  # (n_h, n_l)
        
        deltas = fraction * delta_safe
        
        # Bounds
        upper = np.minimum(h[high_set, None], (1.0 - h[low_set])[None, :])
        deltas = np.clip(deltas, 0.0, upper)
        
        valid = deltas > 1e-10
        if not valid.any():
            continue
        
        # Select top candidates for verification
        if valid.sum() > n_verify:
            # Flatten and select top by delta
            vi, vj = np.where(valid)
            dv = deltas[vi, vj]
            top_idx = np.argsort(dv)[-n_verify:]
            vi = vi[top_idx]
            vj = vj[top_idx]
        else:
            vi, vj = np.where(valid)
        
        best_trial = c5_val
        best_pi = -1
        best_pj = -1
        best_d = 0.0
        
        for t in range(len(vi)):
            pi = int(vi[t])
            pj = int(vj[t])
            d = float(deltas[pi, pj])
            if d < 1e-10:
                continue
            i = int(high_set[pi])
            j = int(low_set[pj])
            g = change[j] - change[i]
            new_c5 = float(np.max(corr + d * g)) * dx
            if new_c5 < best_trial:
                best_trial = new_c5
                best_pi = pi
                best_pj = pj
                best_d = d
        
        if best_pi >= 0:
            i = int(high_set[best_pi])
            j = int(low_set[best_pj])
            d = best_d
            h[i] -= d
            h[j] += d
            # Update change matrix
            update_change_single(change, i, -d, shifts, n)
            update_change_single(change, j, d, shifts, n)
    
    return best_h, best_c5
