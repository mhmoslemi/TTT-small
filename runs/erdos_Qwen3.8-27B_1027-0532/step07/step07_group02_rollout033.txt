def project_feasible(h, target_sum):

def linearization_method(h_init, n, target, time_limit, max_iter=100, eps=0.01, seed=42):
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0 / n)
        if c5_val < best_c5 - 1e-10:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find active set
        threshold = np.max(corr) * (1 - eps)
        active = np.where(corr >= threshold)[0]
        
        # If active set is too large, increase eps
        if len(active) > 50:
            eps *= 2
            threshold = np.max(corr) * (1 - eps)
            active = np.where(corr >= threshold)[0]
        
        # Linearize around current h
        # For each active lag m, compute gradient of overlap wrt h
        n_corr = len(corr)
        mid = n - 1
        idxs = np.arange(n)
        
        # Build the LP: min C s.t. grad_m . (h_new - h) + o(m) ≤ C for m in active
        #                              0 ≤ h_new ≤ 1, sum(h_new) = target
        
        # This is: min C
        #           s.t. A @ h_new - C ≤ b
        #                sum(h_new) = target
        #                0 ≤ h_new ≤ 1
        # where A[m, i] = grad_m[i], b[m] = -o(m)
        
        # Actually: o_lin(m) = o(m) + grad_m . (h_new - h) ≤ C
        # => grad_m . h_new ≤ C - o(m) + grad_m . h
        
        # Let's use cvxpy for the LP
        import cvxpy as cp
        
        h_var = cp.Variable(n)
        C_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for m_idx in active:
            m = m_idx - mid
            # Gradient of overlap(m) wrt h[i]
            # o(m) = sum_i h[i] * (1 - h[i-m]) for valid i
            # ∂o(m)/∂h[i] = (1 - h[i-m]) - h[i] (if i-m is valid)
            #              + ... (if i is used as the second index)
            # More carefully:
            # o(m) = sum_{i: i valid, i-m valid} h[i] * (1 - h[i-m])
            # ∂o(m)/∂h[j] = (1 - h[j-m]) [if j-m valid] - sum_{i: i-m=j} h[i] 
            #              = (1 - h[j-m]) [if j-m valid] - h[j+m] [if j+m valid]
            
            grad_m = np.zeros(n)
            # First term: (1 - h[j-m]) for j where j-m is valid
            v1 = (idxs - m >= 0) & (idxs - m < n)
            if v1.any():
                grad_m[v1] += (1.0 - h[idxs[v1] - m])
            # Second term: -h[j+m] for j where j+m is valid
            v2 = (idxs + m >= 0) & (idxs + m < n)
            if v2.any():
                grad_m[v2] -= h[idxs[v2] + m]
            
            # Linearized constraint: o(m) + grad_m . (h_var - h) ≤ C_var
            # => grad_m . h_var - C_var ≤ -o(m) + grad_m . h
            rhs = -corr[m_idx] + np.dot(grad_m, h)
            constraints.append(np.dot(grad_m, h_var) - C_var <= rhs)
        
        objective = cp.Minimize(C_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLOP)  # or ECOS, SCS
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                try:
                    prob.solve(solver=cp.SCS)
                except Exception:
                    break
        
        if h_var.value is not None:
            h_new = project_feasible(h_var.value, target)
            sc = compute_c5(h_new)
            if sc < best_c5:
                # Damping: move partially towards the LP solution
                h_new = 0.5 * h + 0.5 * h_new
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h_new
            h = h_new
    
    return best_h, best_c5
