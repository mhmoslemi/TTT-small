- The linear subproblem is O(n log n) (just a sort)
- No feasibility issues (convex combination of feasible points stays feasible)
- Simple to implement

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def frank_wolfe_step(h, n, target):
    """Compute Frank-Wolfe direction: binary point that minimizes linear approx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Subgradient: use top few shifts
    top_idx = np.argsort(corr)[-5:]
    g = np.zeros(n)
    for s_idx in top_idx:
        k = int(s_idx) - (n - 1)
        # grad of f_k w.r.t. h[i] = dx * (1 - h[i+k]) for i in valid range
        # and dx * (-h[i-k]) for i in valid range
        idxs = np.arange(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        g[v1] += dx * (1.0 - h[idxs[v1] - k])
        g[v2] -= dx * h[idxs[v2] + k]
    g /= len(top_idx)
    # Sort and pick n/2 smallest as 1s
    sorted_idx = np.argsort(g)
    y = np.zeros(n)
    y[sorted_idx[:target]] = 1.0
    return y, g

def frank_wolfe_optimize(h_init, n, target, time_limit, max_iters=5000):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for t in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        y, g = frank_wolfe_step(h, n, target)
        
        # Line search: h_new = (1-gamma)*h + gamma*y
        # Try several step sizes
        best_gamma = 0.0
        best_val = compute_c5(h)
        for gamma in np.linspace(0.01, 1.0, 20):
            h_try = (1 - gamma) * h + gamma * y
            h_try = project_feasible(h_try, target)
            val = compute_c5(h_try)
            if val < best_val:
                best_val = val
                best_gamma = gamma
        
        if best_gamma > 0:
            h = (1 - best_gamma) * h + best_gamma * y
            h = project_feasible(h, target)
            if best_val < best_c5 - 1e-15:
                best_c5 = best_val
                best_h = h.copy()
    
    return best_h, best_c5
```

For Frank-Wolfe on this problem:
- Feasible set P = {h ∈ [0,1]^n : Σh = n/2}
- The "hard" part is the linear minimization oracle: min_{y∈P} g·y
  - Solution: sort g, set y[i]=1 for the n/2 indices with smallest g[i]
  - (This works because P is the intersection of the hypercube and a hyperplane, and the linear function is minimized at extreme points)

Wait, is that right? The extreme points of P = {h ∈ [0,1]^n : Σh = n/2} are the points where exactly n/2 coordinates are 1 and the rest are 0 (assuming n/2 is an integer). Yes! So the linear oracle is:
- Sort g in ascending order
- Set y = [1,1,...,1,0,0,...,0] (n/2 ones, n/2 zeros) at the sorted indices

This is O(n log n) per iteration.

For the step size, I'll use:
- Backtracking line search: try γ = 1, 0.5, 0.25, ... until the objective decreases
- Or the standard γ_t = 2/(t+2)

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _obj_func(x):
    return smooth_obj(x, _g_alpha)

def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_target, _g_alpha
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000, 
            n_cand_start=30, n_cand_end=80, top_k_eval=50,
            max_no_improve=600, n_shifts_grad=3, cand_multiplier=1.0):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    change = build_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        k_star_idx = int(np.argmax(corr))
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int((n_cand_start + (n_cand_end - n_cand_start) * progress) * cand_multiplier)
        n_cand = min(n_cand, n // 3)
        n_cand = max(n_cand, 10)
        
        # Compute gradient using top shifts (weighted by correlation value)
        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx]
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]
        
        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            
            g = change[j] - change[i]
            
            # Binary search for optimal delta
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):

    if n_src == n_dst:

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _obj_func(x):
    return smooth_obj(x, _g_alpha)


def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_target, _g_alpha
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=80, top_k_eval=50,
            max_no_improve=600, n_shifts_grad=3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        n_cand = max(n_cand, 10)

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx]
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def frank_wolfe(h_init, n, target, time_limit, max_iters=8000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for t in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')

        top_idx = np.argsort(corr)[-5:]
        g = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v1] += dx * (1.0 - h[idxs[v1] - k])
            g[v2] -= dx * h[idxs[v2] + k]
        g /= float(len(top_idx))

        sorted_idx = np.argsort(g)
        y = np.zeros(n)
        y[sorted_idx[:target]] = 1.0

        cur_c5 = compute_c5(h)
        best_gamma = 0.0
        best_trial = cur_c5

        gammas = []
        std_gamma = 2.0 / (t + 2.0)
        gammas.append(std_gamma)
        for gv in [1.0, 0.6, 0.4, 0.25, 0.12, 0.06]:
            if gv > 0 and gv <= 1.0 and gv not in gammas:
                gammas.append(gv)

        for gamma in gammas:
            h_try = (1.0 - gamma) * h + gamma * y
            h_try = project_feasible(h_try, target)
            val = compute_c5(h_try)
            if val < best_trial - 1e-15:
                best_trial = val
                best_gamma = gamma

        if best_gamma > 0:
            h = (1.0 - best_gamma) * h + best_gamma * y
            h = project_feasible(h, target)
            no_improve = 0
            if best_trial < best_c5 - 1e-14:
                best_c5 = best_trial
                best_h = h.copy()
        else:
            no_improve += 1

    return best_h, best_c5

    return project_feasible(h_out, target_dst)

                                 n_shifts_grad=4)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 6: SLSQP with high alpha
    tl = _time_left(start_time, budget_s)
    if tl > 70:
        h_pol2 = slsqp_optimize(best_h, n0, target0,
                                 [2000, 8000, 20000], 800, 1e-15)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 7: Random perturbations + SLSQP
    tl = _time_left(start_time, budget_s)
    if tl > 120:
        for noise_scale in [0.03, 0.07, 0.12, 0.2]:
            tl = _time_left(start_time, budget_s)
            if tl < 50:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref = slsqp_optimize(h_pert, n0, target0,
                                   [200, 1000, 5000], 500, 1e-15)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                tl = _time_left(start_time, budget_s)
                if tl > 40:
                    h_cd_r, c5_r = pair_cd(best_h, n0, target0,
                                            time_limit=min(30, tl * 0.7),
                                            max_sweeps=15000,
                                            n_cand_start=30, n_cand_end=60,
                                            top_k_eval=50, max_no_improve=400,
                                            n_shifts_grad=3)
                    if c5_r < best_score:
                        best_score = c5_r
                        best_h = h_cd_r

    # Phase 8: Try a different grid size (n=600) if time allows
    tl = _time_left(start_time, budget_s)
    if tl > 200 and len(best_h) != 600:
        n_alt = 600
        target_alt = n_alt / 2.0
        h_alt = interpolate_to(best_h, n_alt, target_alt)
        sc_alt = compute_c5(h_alt)
        tl = _time左_left(start_time, budget_s) if False else _time_left(start_time, budget_s)
        if tl > 150:
            h_cd_alt, c5_alt = pair_cd(h_alt, n_alt, target_alt,
                                        time_limit=min(120, tl * 0.7),
                                        max_sweeps=30000,
                                        n_cand_start=25, n_cand_end=60,
                                        top_k_eval=50, max_no_improve=500,
                                        n_shifts_grad=3)
            if c5_alt < sc_alt:
                sc_alt = c5_alt
                h_alt = h_cd_alt
            tl = _time_left(start_time, budget_s)
            if tl > 50:
                h_pol_alt = slsqp_optimize(h_alt, n_alt, target_alt,
                                            [1000, 5000], 600, 1e-15)
                sc_alt2 = compute_c5(h_pol_alt)
                if sc_alt2 < sc_alt:
                    sc_alt = sc_alt2
                    h_alt = h_pol_alt
            if sc_alt < best_score:
                best_score = sc_alt
                best_h = h_alt

    # Phase 9: Final CD push
    tl = _time_left(start_time, budget_s)
    if tl > 60:
        h_cd_final, c5_final = pair_cd(best_h, n0, target0,
                                        time_limit=min(50, tl * 0.7),
                                        max_sweeps=20000,
                                        n_cand_start=40, n_cand_end=90,
                                        top_k_eval=70, max_no_improve=400,
                                        n_shifts_grad=4)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    # Phase 10: Final SLSQP
    tl = _time_left(start_time, budget_s)
    if tl > 40:
        h_final = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                  [5000, 20000], 500, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 11: Fine-grained CD
    tl = _time_left(start_time, budget_s)
    if tl > 30:
        nf = len(best_h)
        tf = nf / 2.0
        h_fine, c5_fine = pair_cd(best_h, nf, tf,
                                   time_limit=min(25, tl * 0.7),
                                   max_sweeps=15000,
                                   n_cand_start=15, n_cand_end=40,
                                   top_k_eval=30, max_no_improve=300,
                                   n_shifts_grad=2)
        if c5_fine < best_score:
            best_score = c5_fine
            best_h = h_fine

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
