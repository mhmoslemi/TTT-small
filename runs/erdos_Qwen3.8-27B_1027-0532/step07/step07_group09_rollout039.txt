```python
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_score
```

Wait, there's an issue with the gradient. The smooth objective F_α(h) = (1/α)log(Σ exp(α·corr[k])) has gradient:
∇F_α = Σ_k p_k · ∇corr[k]

where p_k = exp(α·corr[k]) / Σ_j exp(α·corr[j])

And ∇corr[k] at h is:
For each i, ∂corr[mid+k]/∂h[i] = (1-h[i+k])·1_{i+k valid} - h[i-k]·1_{i-k valid}

This is correct. But the issue is that the sum constraint ∑h = n/2 means we should project the gradient onto the subspace orthogonal to (1,1,...,1), which is what `grad -= np.mean(grad)` does.

But there's a subtlety: the projection onto the feasible set {h: 0≤h≤1, ∑h=n/2} is not simply clip+adjust. The FISTA algorithm assumes a proximal operator that is the projection onto the constraint set. Our `project_feasible` does this approximately.

The main issue with FISTA here is that the "proximal" step (projection) is not the Euclidean projection onto the intersection of the box and the hyperplane. It's an approximation. This might slow convergence.

Alternative: use a penalty method where we add (1/2μ)(∑h - n/2)² to the objective and minimize over the box [0,1]^n (which has a simple projection: clip).

```python
    p = exp_s / sum_exp
    obj = m + np.log(sum_exp) / alpha + (mu / 2.0) * (np.sum(h) - len(h)/2.0)**2
    
    mid = len(h) - 1
    n = len(h)
    grad = np.zeros(n)

    for idx in np.where(active)[0]:

        w = p[idx]
        if k >= 0:

                grad[0:n-k] += w * (1.0 - h[k:n])

            if kk > 0:
                grad[kk:n] += w * (1.0 - h[0:n-kk])
            grad[0:n-kk] -= w * h[kk:n]

    return obj, grad
```

```python
            break

        obj, grad = smooth_obj_and_grad(h, alpha)

        # and the most positive (most beneficial to increase)

        i_inc = np.argmax(grad)

            no_improve += 100
            continue

        # We want to move mass from i_dec to i_inc
        delta_max = min(h[i_dec], 1.0 - h[i_inc])

        best_delta = 0
        best_obj = obj

            if d < 1e-12:

            obj_try, _ = smooth_obj_and_grad(h_try, alpha)

            h[i_inc] += best_delta
            h = project_feasible(h, target)
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500.0, 0.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500.0, 0.0)
    e = np.exp(s)
    p = e / np.sum(e)
    mid = n - 1
    grad = np.zeros(n)
    active = p > 1e-10
    for idx in np.where(active)[0]:
        k = int(idx) - mid
        w = p[idx]
        if k >= 0:
            if k < n:
                grad[:n-k] += w * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= w * h[:n-k]
        else:
            kk = -k
            if kk > 0:
                grad[kk:] += w * (1.0 - h[:n-kk])
            grad[:n-kk] -= w * h[kk:]
    return grad

def fista_optimize(h_init, n, target, time_limit, alpha_start=200.0, alpha_end=5000.0, max_iter=100000):
    start = time.time()
    h = h_init.astype(float).copy()
    h_prev = h.copy()
    t_m = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 10000:
            break
        
        progress = min(float(it) / max_iter, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        grad = smooth_grad(h, alpha)
        grad -= np.mean(grad)
        
        # Adaptive step size
        lr = 2.0 / (alpha * 4.0)  # rough Lipschitz estimate
        
        h_new = project_feasible(h - lr * grad, target)
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_m * t_m)) / 2.0
        h_next = h_new + ((t_m - 1.0) / t_new) * (h_new - h_prev)
        h_next = project_feasible(h_next, target)
        t_m = t_new
        h_prev = h_new
        h = h_next
        
        score = compute_c5(h)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_score

def slsqp_optimize(h_init, n, target, alphas, maxiter=500):
    h_cur = h_init.copy()
    best_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        def obj(x):
            return smooth_obj(x, alpha)
        
        def cons_fun(x):
            return np.sum(x) - target
        
        def cons_jac(x):
            return np.ones(n)
        
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, best_score

def interpolate_to(h_src, n_dst, target_dst):
    if len(h_src) == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, len(h_src))
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:

            break

        h = np.clip(h, 0.0, 1.0)

    n = len(h)

    m = np.max(corr)
    s = alpha * (corr - m)

    e = np.exp(s)
    total = np.sum(e)

    grad = np.zeros(n)
```

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500.0, 0.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def smooth_grad_vec(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500.0, 0.0)
    e = np.exp(s)
    total = np.sum(e)
    mid = n - 1
    grad = np.zeros(n)
    active = e > total * 1e-10
    idxs = np.where(active)[0]
    for idx in idxs:
        k = int(idx) - mid
        w = e[idx] / total
        if k >= 0:
            if k < n:
                grad[:n-k] += w * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= w * h[:n-k]
        else:
            kk = -k
            if kk > 0:
                grad[kk:] += w * (1.0 - h[:n-kk])
            grad[:n-kk] -= w * h[kk:]
    return grad


def _slsqp_obj(x):
    return smooth_obj_val(x, _g_alpha)


def _slsqp_cons(x):
    return np.sum(x) - _g_target


def _slsqp_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h_init, n, target, alphas, maxiter=400):
    global _g_alpha, _g_n, _g_target
    _g_n = n
    _g_target = target
    h_cur = h_init.copy()
    best_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        _g_alpha = float(alpha)
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, best_score


def nesterov_opt(h_init, n, target, time_limit, alpha=2000.0, max_iter=100000):
    start = time.time()
    h = h_init.astype(float).copy()
    h_prev = h.copy()
    t_m = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    step = 1.0 / max(alpha * 2.0, 100.0)
    obj_cur = smooth_obj_val(h, alpha)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 12000:
            break
        if it % 5000 == 0 and it > 0:
            progress = float(it) / max_iter
            alpha = alpha * (1.0 + 3.0 * progress)
            step = 1.0 / max(alpha * 2.0, 100.0)

        grad = smooth_grad_vec(h, alpha)
        grad -= np.mean(grad)

        obj_try = smooth_obj_val(project_feasible(h - step * grad, target), alpha)
        n_bt = 0
        while obj_try > obj_cur + 1e-16 and n_bt < 6:
            step *= 0.5
            h_interp = h + 0.5 * (h_prev - h)
            h_interp = project_feasible(h_interp, target)
            obj_try = smooth_obj_val(project_feasible(h - step * grad, target), alpha)
            n_bt += 1

        h_new = project_feasible(h - step * grad, target)
        obj_new = smooth_obj_val(h_new, alpha)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_m * t_m)) / 2.0
        h_next = h_new + ((t_m - 1.0) / t_new) * (h_new - h_prev)
        h_next = project_feasible(h_next, target)
        t_m = t_new
        h_prev = h_new
        h = h_next

        if it % 5 == 0:
            score = compute_c5(h)
            if score < best_score - 1e-15:
                best_score = score
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 5

    return best_h, best_score


def fast_cd(h_init, n, target, time_limit, alpha=5000.0, max_sweeps=200000):
    start = time.time()
    h = h_init.astype(float).copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 4000:
            break

        grad = smooth_grad_vec(h, alpha)
        grad -= np.mean(grad)

        i_dec = int(np.argmin(grad))
        i_inc = int(np.argmax(grad))

        if grad[i_dec] > -1e-12 or grad[i_inc] < 1e-12:
            no_improve += 1
            continue

        delta_max = min(h[i_dec], 1.0 - h[i_inc])
        if delta_max < 1e-14:
            no_improve += 1
            continue

        best_delta = 0.0
        obj_cur = smooth_obj_val(h, alpha)
        for frac in [0.8, 0.4, 0.2, 0.1]:
            d = frac * delta_max
            if d < 1e-14:
                continue
            h_try = h.copy()
            h_try[i_dec] -= d
            h_try[i_inc] += d
            h_try = project_feasible(h_try, target)
            obj_try = smooth_obj_val(h_try, alpha)
            if obj_try < obj_cur - 1e-15:
                best_delta = d
                break

        if best_delta > 0:
            h[i_dec] -= best_delta
            h[i_inc] += best_delta
            h = project_feasible(h, target)
            no_improve = 0
            if sweep % 20 == 0:
                score = compute_c5(h)
                if score < best_score - 1e-15:
                    best_score = score
                    best_h = h.copy()
        else:
            no_improve += 1

    return best_h, best_score


def pair_cd(h_init, n, target, time_limit, n_cand=30, max_sweeps=50000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.astype(float).copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_score - 1e-14:
            best_score = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = min(4, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)])

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                hr_j = lo_rows[jj]
                h_j = h[j]
                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue
                g = G[hr_j] - G[hr]
                a_d = delta_min
                b_d = delta_max
                for _ in range(8):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    if len(h_src) == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, len(h_src))
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
