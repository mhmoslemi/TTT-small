    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def proj_capped_simplex(h, target):
    n = len(h)
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    ks = np.arange(1, n + 1)
    cond = ks * sorted_h + target > cumsum
    if not cond.any():
        k = 1
    else:
        k = int(np.max(ks[cond]))
    nu = (cumsum[k - 1] - target) / k
    x = np.clip(h - nu, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_change_matrix(h, n):
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, :] = 0.0
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros(len(corr))
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= wsum
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-14:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k]])
        grad[v2] -= w * x[idxs[v2] + k]]
    return grad

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_eq_jac(x):
    return np.ones(_g_n)

def slsqp_full(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-16):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def sparse_slsqp(h_start, n, target, free_tol, alphas, maxiter_per_alpha):
    global _g_n, _g_target, _g_free_idx, _g_h_fixed, _g_alpha
    free_mask = (h_start > free_tol) & (h_start < 1.0 - free_tol)
    free_idx = np.where(free_mask)[0]
    n_free = len(free_idx)
    if n_free == 0:
        return h_start, compute_c5(h_start)

    h_fixed = h_start.copy()
    h_fixed[~free_mask] = np.where(h_fixed[~free_mask] < 0.5, 0.0, 1.0)
    fixed_sum = np.sum(h_fixed[~free_mask])
    target_free = target - fixed_sum

    _g_h_fixed = h_fixed
    _g_free_idx = free_idx
    _g_n = n_free
    _g_target = target_free

    def sparse_obj(x):
        h_full = _g_h_fixed.copy()
        h_full[_g_free_idx] = x
        a = _g_alpha
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / a)

    def sparse_obj_grad(x):
        h_full = _g_h_fixed.copy()
        h_full[_g_free_idx] = x
        a = _g_alpha
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            weights = np.zeros(len(corr))
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights /= wsum
        nf = len(h_full)
        mid = nf - 1
        grad_full = np.zeros(nf)
        idxs = np.arange(nf)
        for s in range(2 * nf - 1):
            w = weights[s]
            if w < 1e-14:
                continue
            kk = s - mid
            v1 = (idxs - kk >= 0) & (idxs - kk < nf)
            v2 = (idxs + kk >= 0) & (idxs + kk < nf)
            grad_full[v1] += w * (1.0 - h_full[idxs[v1] - kk])
            grad_full[v2] -= w * h_full[idxs[v2] + kk]
        return grad_full[_g_free_idx]

    def sparse_cons(x):
        return np.sum(x) - target_free

    def sparse_cons_jac(x):
        return np.ones(n_free)

    bounds_f = [(1e-8, 1.0 - 1e-8)] * n_free
    cons = [{'type': 'eq', 'fun': sparse_cons, 'jac': sparse_cons_jac}]

    h_free = h_start[free_idx].copy()
    h_cur = h_free.copy()
    h_full_cur = h_start.copy()
    cur_score = compute_c5(h_full_cur)

    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(sparse_obj, h_cur, method='SLSQP',
                           bounds=bounds_f, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_free_new = np.clip(res.x, 0.0, 1.0)
            h_full_new = h_fixed.copy()
            h_full_new[free_idx] = h_free_new
            h_full_new = project_feasible(h_full_new, target)
            sc = compute_c5(h_full_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_free_new
                h_full_cur = h_full_new
        except Exception:
            pass
    return h_full_cur, cur_score

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=40, n_cand_end=150, max_no_improve=500,
            n_binsearch=20, stochastic_frac=0.05, seed=123):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        n_pairs = n_h * n_l
        n_random = int(stochastic_frac * n_pairs)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if n_random > 0:
            for _ in range(n_random):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def proj_grad_descent(h_init, n, target, alphas, maxiters_per_alpha, time_limit,
                      lr_scale=1.0, momentum=0.9):
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    for alpha in alphas:
        h_cur = best_h.copy()
        v = np.zeros(n)
        best_local = best_c5
        no_improve = 0
        for it in range(maxiters_per_alpha):
            if time.time() - start > time_limit:
                return best_h, best_c5
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            weights = np.exp(shifted)
            wsum = np.sum(weights)
            if wsum < 1e-300:
                weights = np.zeros(len(corr))
                weights[int(np.argmax(corr))] = 1.0
            else:
                weights /= wsum
            mid = n - 1
            idxs = np.arange(n)
            grad = np.zeros(n)
            for s in range(2 * n - 1):
                w = weights[s]
                if w < 1e-14:
                    continue
                k = s - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h_cur[idxs[v1] - k]])
                grad[v2] -= w * h_cur[idxs[v2] + k]]
            grad -= np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-18:
                break
            lr = lr_scale / (alpha * gnorm)
            v = momentum * v - lr * grad
            h_new = h_cur + v
            h_new = proj_capped_simplex(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_local - 1e-15:
                best_local = sc
                h_cur = h_new.copy()
                no_improve = 0
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h_cur.copy()
            else:
                no_improve += 1
                v *= 0.5
                if no_improve > 100:
                    break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_restart_cd(h_best, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    scales = [0.02, 0.05, 0.1, 0.15, 0.25]
    while time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        if remaining < 30:
            break
        scale = scales[rng.randint(len(scales))]
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        alloc = min(remaining / 3.0, 80.0)
        h_cd, c5_cd = pair_cd(h_pert, n, target, time_limit=alloc,
                               max_sweeps=5000, n_cand_start=30, n_cand_end=80,
                               max_no_improve=200, n_binsearch=16,
                               stochastic_frac=0.1, seed=int(rng.randint(100000)))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    return best_h, best_c5
