```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        res = minimize(lse_objective, h_cur, args=(alpha, target, penalty),
                      jac=lse_gradient, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        h_new = project_feasible(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new

    return best_h, best_c5

def interp_profile(cands, h, n):
    """Vectorized computation of gradient profiles for candidate coordinates."""
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)  # shape (n_corr,)
    c = cands.astype(int)  # shape (C,)
    
    # idx1 = c + s, valid if 0 <= c+s < n
    idx1 = c[:, None] + shifts[None, :]  # shape (C, n_corr)
    valid1 = (idx1 >= 0) & (idx1 < n)
    val1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
    
    # idx2 = c - s, valid if 0 <= c-s < n
    idx2 = c[:, None] - shifts[None, :]
    valid2 = (idx2 >= 0) & (idx2 < n)
    val2 = np.where(valid2, -h[np.clip(idx2, 0, n-1)], 0.0)
    
    return val1 + val2  # shape (C, n_corr)

def fast_pair_cd(h_init, n, target, time_limit, n_cand=40, max_sweeps=100000, n_top=3, n_bsearch=10):

    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    mid = n - 1
    no_improve = 0
    idxs = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        k_star = int(np.argmax(corr)) - mid

        # Gradient at k*: grad[i] = (1-h[i+k*]) - h[i-k*] (with bounds)
        v1 = (idxs + k_star >= 0) & (idxs + k_star < n)
        v2 = (idxs - k_star >= 0) & (idxs - k_star < n)

        if np.any(v1):
            grad_k[v1] += (1.0 - h[idxs[v1] + k_star])
        if np.any(v2):
            grad_k[v2] -= h[idxs[v2] - k_star]

        # And increase h[j] where grad_k[j] < 0 (also reduces the max)

        low_score = np.where(grad_k < 0, (1.0 - h) * (-grad_k), -np.inf)

        low_set = np.argsort(low_score)[-n_cand:][::-1]

        high_cand_pos = [cand_pos[int(i)] for i in high_set]
        low_cand_pos = [cand_pos[int(j)] for j in low_set]

        low_at_max = profiles[low_cand_pos, k_star + mid]

            if p not in seen:

                unique_pairs.append(p)

                else:

                best_i = int(i)
                best_j = int(j)

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5
```
