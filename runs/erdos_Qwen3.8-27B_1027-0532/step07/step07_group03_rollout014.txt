    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    diff = target_sum - np.sum(h_out)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_out < 1.0 - 1e-12
        else:
            mask = h_out > 1e-12
        if mask.any():
            h_out[mask] += diff / mask.sum()
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def minimax_opt(h_init, n, target, time_limit, max_iters=100000):
    start = time.time()
    mid = n - 1
    n_corr = 2 * n - 1
    dx = 2.0 / n
    
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    idxs = np.arange(n)
    # Precompute valid index masks for each lag
    # (This is expensive to precompute all, so compute on the fly)
    
    step = 0.02
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            step *= 0.9
            if step < 1e-6:
                break
            no_improve = 0
        
        # Compute softmax distribution
        progress = min(it / max_iters, 1.0)
        alpha = 20.0 + 200.0 * progress
        
        corr_max = np.max(corr)
        p = np.exp(alpha * (corr - corr_max))
        p_sum = np.sum(p)
        if p_sum < 1e-30:
            p = np.ones(n_corr) / n_corr
        else:
            p = p / p_sum
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for k_idx in range(n_corr):
            if p[k_idx] < 1e-8:
                continue
            pk = p[k_idx]
            k = k_idx - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += pk * (1.0 - h[idxs[v1] + k])
            grad[v2] -= pk * h[idxs[v2] - k]
        
        # Step with backtracking
        h_new = project_feasible(h - step * grad, target)
        sc_new = compute_c5(h_new)
        
        if sc_new < c5_val + 1e-14:
            h = h_new
        else:
            # Try smaller step
            h_half = project_feasible(h - 0.5 * step * grad, target)
            sc_half = compute_c5(h_half)
            if sc_half < c5_val:
                h = h_half
                step *= 0.5
            else:
                step *= 0.5
                h = h_half  # Still take the step, it's probably fine
    
    return best_h, best_score

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:
            step *= 0.85
            if step < 1e-7:
                break
            no_improve = 0

        progress = min(float(it) / max_iters, 1.0)
        alpha = 15.0 + 300.0 * progress

        corr_max = float(np.max(corr))
        exp_vals = np.exp(alpha * (corr - corr_max))
        p_sum = np.sum(exp_vals)
        if p_sum < 1e-30:
            p = np.ones(n_corr) / n_corr
        else:
            p = exp_vals / p_sum

        grad = compute_grad_vec(h, p, mid, idxs, n)

        h_new = project_feasible(h - step * grad, target)
        sc_new = compute_c5(h_new)

        if sc_new < c5_val + 1e-12:
            h = h_new
            step = min(step * 1.01, 0.1)
        else:
            h_half = project_feasible(h - 0.5 * step * grad, target)
            sc_half = compute_c5(h_half)
            if sc_half < c5_val:
                h = h_half
            else:
                h = h_half
            step *= 0.7

    return best_h, best_score


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_refine(h_init, n, target, time_limit, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    start = time.time()
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def pairwise_swap(h_init, n, target, time_limit, max_sweeps=5000, n_cand=30):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    n_corr = 2 * n - 1
    idxs = np.arange(n)

    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break

        corr_max = float(np.max(corr))
        alpha = 50.0
        p = np.exp(alpha * (corr - corr_max))
        p = p / np.sum(p)

        grad = compute_grad_vec(h, p, mid, idxs, n)
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        shifts = np.arange(-(n - 1), n)
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                a_d, b_d = d_lo, d_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * (
                        np.zeros(n_corr)  # placeholder
                    )
                    break
                # Use direct evaluation for small changes
                h_trial = h.copy()
                for frac in [0.3, 0.5, 0.7, 1.0]:
                    d = frac * (d_hi - d_lo) + d_lo
                    if abs(d) < 1e-12:
                        continue
                    h_trial[i] = h[i] - d
                    h_trial[j] = h[j] + d
                    h_trial = np.clip(h_trial, 0.0, 1.0)
                    sc = compute_c5(h_trial)
                    if sc < best_trial:
                        best_trial = sc
                        best_i = i
                        best_j = j
                        best_delta = d

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = project_feasible(h, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def momentum_minimax(h_init, n, target, time_limit, max_iters=100000):
    start = time.time()
    mid = n - 1
    n_corr = 2 * n - 1
    dx = 2.0 / n
    idxs = np.arange(n)

    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)

    beta = 0.9
    step = 0.02
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            step *= 0.8
            if step < 1e-7:
                break
            no_improve = 0

        progress = min(float(it) / max_iters, 1.0)
        alpha = 10.0 + 250.0 * progress
        beta_t = beta * (1.0 - 0.3 * progress)

        corr_max = float(np.max(corr))
        exp_vals = np.exp(alpha * (corr - corr_max))
        p_sum = np.sum(exp_vals)
        if p_sum < 1e-30:
            p = np.ones(n_corr) / n_corr
        else:
            p = exp_vals / p_sum

        grad = compute_grad_vec(h, p, mid, idxs, n)
        v = beta_t * v - step * grad
        h_new = project_feasible(h + v, target)
        sc_new = compute_c5(h_new)

        if sc_new < c5_val + 1e-12:
            h = h_new
        else:
            v = 0.5 * v
            h = h_new
            step *= 0.8

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Minimax at n=400
    if tleft() > 400:
        t1 = min(350, tleft() * 0.4)
        h_mm, sc_mm = minimax_opt(best_h, n0, target0, time_limit=t1, max_iters=200000)
        if sc_mm < best_score:
            best_score = sc_mm
            best_h = h_mm

    # Phase 2: Momentum minimax
    if tleft() > 300:
        t2 = min(250, tleft() * 0.3)
        h_mom, sc_mom = momentum_minimax(best_h, n0, target0, time_limit=t2, max_iters=100000)
        if sc_mom < best_score:
            best_score = sc_mom
            best_h = h_mom

    # Phase 3: SLSQP refinement
    if tleft() > 200:
        t3 = min(150, tleft() * 0.25)
        h_sls, sc_sls = slsqp_refine(best_h, n0, target0, time_limit=t3,
                                      alphas=[500, 2000, 8000], maxiter_per_alpha=400)
        if sc_sls < best_score:
            best_score = sc_sls
            best_h = h_sls

    # Phase 4: Pairwise swaps
    if tleft() > 200:
        t4 = min(180, tleft() * 0.25)
        h_sw, sc_sw = pairwise_swap(best_h, n0, target0, time_limit=t4,
                                    max_sweeps=5000, n_cand=25)
        if sc_sw < best_score:
            best_score = sc_sw
            best_h = h_sw

    # Phase 5: Another minimax round with different params
    if tleft() > 250:
        t5 = min(200, tleft() * 0.3)
        h_mm2, sc_mm2 = minimax_opt(best_h, n0, target0, time_limit=t5, max_iters=150000)
        if sc_mm2 < best_score:
            best_score = sc_mm2
            best_h = h_mm2

    # Phase 6: Try n=800
    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h800 = interpolate_to(best_h, n1, target1)
        sc800 = compute_c5(h800)
        if tleft() > 200:
            t6 = min(180, tleft() * 0.3)
            h800_mm, sc800_mm = minimax_opt(h800, n1, target1, time_limit=t6, max_iters=80000)
            if sc800_mm < best_score:
                best_score = sc800_mm
                best_h = h800_mm
                n0 = n1
                target0 = target1
        if tleft() > 100 and n0 == n1:
            t6b = min(80, tleft() * 0.2)
            h800_sls, sc800_s = slsqp_refine(best_h, n0, target0, time_limit=t6b,
                                              alphas=[2000, 8000], maxiter_per_alpha=300)
            if sc800_s < best_score:
                best_score = sc800_s
                best_h = h800_sls

    # Phase 7: Final SLSQP polish
    if tleft() > 100:
        t7 = min(80, tleft() * 0.2)
        h_fin, sc_fin = slsqp_refine(best_h, n0, target0, time_limit=t7,
                                      alphas=[5000, 20000, 80000], maxiter_per_alpha=500)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 8: Random restart with refinement
    if tleft() > 150:
        for ns in [0.01, 0.03, 0.05]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n0) * ns
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc_ref = minimax_opt(h_pert, n0, target0,
                                        time_limit=min(40, tleft() * 0.3), max_iters=50000)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 9: Final pairwise swaps
    if tleft() > 60:
        t9 = min(45, tleft() * 0.3)
        h_sw2, sc_sw2 = pairwise_swap(best_h, n0, target0, time_limit=t9,
                                      max_sweeps=3000, n_cand=40)
        if sc_sw2 < best_score:
            best_score = sc_sw2
            best_h = h_sw2

    # Phase 10: Final polish
    if tleft() > 40:
        t10 = min(30, tleft() * 0.3)
        h_final_mm, sc_final_mm = minimax_opt(best_h, n0, target0,
                                              time_limit=t10, max_iters=30000)
        if sc_final_mm < best_score:
            best_score = sc_final_mm
            best_h = h_final_mm

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
