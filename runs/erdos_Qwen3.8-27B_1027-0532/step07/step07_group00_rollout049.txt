from scipy.optimize import linprog

def lp_step(h, n, target, trust_radius=None):
    """One step of successive linearization using LP."""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    # Compute gradients of each f_k at h
    grads = np.zeros((n_corr, n))
    f_vals = np.zeros(n_corr)
    
    for s_idx in range(n_corr):
        k = s_idx - mid
        if k > 0:
            i_max = n - k
            f_vals[s_idx] = np.sum(h[:i_max] * (1 - h[k:]))
            grads[s_idx, :i_max] += (1 - h[k:i_max+k])  # = (1 - h[k:])[:i_max]
            grads[s_idx, k:] -= h[:n-k]
        elif k < 0:
            k_neg = -k
            f_vals[s_idx] = np.sum(h[k_neg:] * (1 - h[:n-k_neg]))
            grads[s_idx, k_neg:] += (1 - h[:n-k_neg])
            grads[s_idx, :n-k_neg] -= h[k_neg:]
        else:
            f_vals[s_idx] = np.sum(h * (1 - h))
            grads[s_idx, :] = 1 - 2 * h
    
    # Linearization: f_k(h_new) ≈ f_k(h) + grad_k · (h_new - h)
    # = (f_k(h) - grad_k · h) + grad_k · h_new
    intercepts = f_vals - grads @ h
    
    # LP: min t
    # s.t. grad_k · h_new + intercept_k ≤ t  for all k
    #      sum(h_new) = target
    #      0 ≤ h_new[i] ≤ 1
    
    # Variables: x = [h_new (n vars), t (1 var)]
    # Objective: minimize t => c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: grad_k · h_new - t ≤ -intercept_k
    # A_ub * x ≤ b_ub
    A_ub = np.zeros((n_corr, n + 1))
    for s_idx in range(n_corr):
        A_ub[s_idx, :n] = grads[s_idx]
        A_ub[s_idx, n] = -1.0
    b_ub = -intercepts
    
    # Equality: sum(h_new) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 ≤ h_new[i] ≤ 1, t ≥ 0 (actually t can be unbounded)
    bounds = [(0, 1)] * n + [(0, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None, None
    
    if result.status != 0:
        return None, None
    
    h_new = result.x[:n]
    t_val = result.x[n]
    
    return h_new, t_val

import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def weighted_pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
                     n_cand_start=20, n_cand_end=70, max_no_improve=350,
                     weight_beta=5.0, use_anneal=False, T0=0.0003):

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        T = T0 * 0.01 ** progress if use_anneal else 0.0

        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        grad = np.zeros(n)
        top_k_shifts = min(30, n_corr)
        top_idx = np.argsort(weights)[-top_k_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)

            grad[v2] -= w * h[idxs[v2] + k]

                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    found_break = True
                    break
                elif T > 1e-12 and approx_max - c5_val < T * 3:

                        best_j = j
                        best_delta = delta_star
                        found_break = True
                        break

            h = np.clip(h, 0.0, 1.0)

            update_change_row(change, best_i, h, n, shifts)

def lp_successive_linearization(h_init, n, target, time_limit, max_iters=300, trust_radius=0.1):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 50:
            break

        f_vals = np.zeros(n_corr)
        grads = np.zeros((n_corr, n))

        for s_idx in range(n_corr):
            k = s_idx - mid
            if k > 0:
                i_max = n - k
                f_vals[s_idx] = np.sum(h[:i_max] * (1.0 - h[k:]))
                grads[s_idx, :i_max] += (1.0 - h[k:k + i_max])
                grads[s_idx, k:] -= h[:n - k]
            elif k < 0:
                k_neg = -k
                f_vals[s_idx] = np.sum(h[k_neg:] * (1.0 - h[:n - k_neg]))
                grads[s_idx, k_neg:] += (1.0 - h[:n - k_neg])
                grads[s_idx, :n - k_neg] -= h[k_neg:]
            else:
                f_vals[s_idx] = np.sum(h * (1.0 - h))
                grads[s_idx, :] = 1.0 - 2.0 * h

        intercepts = f_vals - grads @ h

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_corr, n + 1))
        A_ub[:, :n] = grads
        A_ub[:, n] = -1.0
        b_ub = -intercepts

        if trust_radius is not None:
            I_mat = np.eye(n)
            A_trust_up = np.hstack([I_mat, np.zeros((n, 1))])
            b_trust_up = h + trust_radius
            A_trust_lo = np.hstack([-I_mat, np.zeros((n, 1))])
            b_trust_lo = -h + trust_radius
            A_ub = np.vstack([A_ub, A_trust_up, A_trust_lo])
            b_ub = np.concatenate([b_ub, b_trust_up, b_trust_lo])

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_list = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds_list, method='highs')
        except Exception:
            no_improve += 1
            continue

        if result.status != 0 or result.x is None:
            no_improve += 1
            continue

        h_lp = np.clip(result.x[:n], 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)

        best_alpha = 0.0
        best_trial = best_c5
        for alpha in [1.0, 0.75, 0.5, 0.35, 0.2, 0.1, 0.05]:
            h_try = (1.0 - alpha) * h + alpha * h_lp
            h_try = project_feasible(h_try, target)
            sc_try = compute_c5(h_try)
            if sc_try < best_trial:
                best_trial = sc_try
                best_alpha = alpha

        if best_alpha > 0:
            h = (1.0 - best_alpha) * h + best_alpha * h_lp
            h = project_feasible(h, target)
            cur_sc = compute_c5(h)
            if cur_sc < best_c5 - 1e-14:
                best_c5 = cur_sc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        c_max = float(np.max(corr))
        weights = np.exp(5.0 * (corr - c_max) / max(c_max, 1e-10))

        grad = np.zeros(n)
        top_k_shifts = min(20, n_corr)
        top_idx = np.argsort(weights)[-top_k_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_move = None

        n_g = group_size
        # Try moving groups of n_g from high to low
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            # Build group around i
            group_i = []
            offset = 0
            while len(group_i) < n_g:
                if 0 <= i + offset < n and (i + offset) != i or len(group_i) == 0:
                    if i + offset == i:
                        group_i.append(i)
                    elif (i + offset) not in group_i and h[i + offset] > 1e-6:
                        group_i.append(i + offset)
                if offset >= 0 and offset == n_g:
                    offset = -(offset + 1)
                elif offset < 0:
                    offset += 1
                else:
                    offset += 1
                if len(group_i) >= n_g:
                    break

            for jj in range(len(low_set)):
                j = int(low_set[jj])
                group_j = []
                offset = 0
                while len(group_j) < n_g:
                    if 0 <= j + offset < n and (j + offset) != j or len(group_j) == 0:
                        if j + offset == j:
                            group_j.append(j)
                        elif (j + offset) not in group_j and h[j + offset] < 1.0 - 1e-6:
                            group_j.append(j + offset)
                    if offset >= 0 and offset == n_g:
                        offset = -(offset + 1)
                    elif offset < 0:
                        offset += 1
                    else:
                        offset += 1
                    if len(group_j) >= n_g:
                        break

                if not group_i or not group_j:
                    continue
                if set(group_i) & set(group_j):
                    continue

                # Combined gradient for moving 1 from group_i to group_j (uniformly)
                g_combined = np.zeros(n_corr)
                for pi in group_i:
                    g_combined -= change[pi]
                for pj in group_j:
                    g_combined += change[pj]

                max_delta = min(min(h[pi] for pi in group_i),
                                min(1.0 - h[pj] for pj in group_j))
                if max_delta < 1e-12:
                    continue

                a_d = 0.0
                b_d = max_delta
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_combined
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g_combined[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_combined) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_move = (group_i, group_j, delta_star)

        if best_move is not None:
            group_i, group_j, delta = best_move
            for pi in group_i:
                h[pi] -= delta
            for pj in group_j:
                h[pj] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for pi in group_i:
                update_change_row(change, pi, h, n, shifts)
            for pj in group_j:
                update_change_row(change, pj, h, n, shifts)

    return best_h, best_c5


def regularized_gradient_descent(h_init, n, target, time_limit, n_iter=5000,
                                  step_init=0.002, reg_power=6):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    no_improve = 0
    momentum = np.zeros(n)
    beta_m = 0.9

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        c_max = float(np.max(corr))
        reg = np.power(np.maximum(corr / max(c_max, 1e-10), 1e-10), reg_power)
        reg_sum = np.sum(reg)
        if reg_sum < 1e-20:
            reg = np.ones(n_corr) / n_corr
        else:
            reg = reg / reg_sum

        grad = np.zeros(n)
        n_active = min(40, n_corr)
        active_idx = np.argsort(corr)[-n_active:]
        for s_idx in active_idx:
            w = reg[s_idx]
            if w < 1e-10:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        momentum = beta_m * momentum - step * grad
        h_try = h + momentum
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val - 1e-15:
            h = h_try
            if sc < best_c5:
                step = min(step * 1.02, 0.02)
            else:
                step = min(step * 1.01, 0.02)
        else:
            step *= 0.9
            momentum *= 0.5
            h_try2 = h - 0.5 * step * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val - 1e-15:
                h = h_try2
                momentum = -0.5 * step * grad

    return best_h, best_c5
