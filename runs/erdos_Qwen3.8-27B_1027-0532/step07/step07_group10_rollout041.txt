def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def multi_shift_cd(h_start, n, target, time_limit, n_active=4, n_cand=40,
                   max_sweeps=30000, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_start.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-n_active:]
        weights = np.zeros(n_corr)
        c_vals = corr[top_shifts]
        c_max = c_vals[0]
        w_raw = np.exp(-30.0 * (c_max - c_vals))
        sw = np.sum(w_raw)
        if sw < 1e-30:
            w_raw = np.ones(n_active)
            sw = float(n_active)
        for wi, si in zip(w_raw / sw, top_shifts):
            weights[si] = wi

        grad = np.zeros(n)
        for si in top_shifts:
            k = int(si) - mid
            w = weights[si]
            if w < 1e-10:
                continue
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def gradient_descent_active(h_init, n, target, time_limit, max_steps=100000,
                            base_lr=0.02, n_active=8, lr_decay=0.9999):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 8000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.85
                if lr < 1e-7:
                    break

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[0]]
        weights = np.exp(-40.0 * (c_max - corr[top_shifts]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-16:
            h = h_try
            cur_c5 = sc
        else:
            lr *= 0.5
            if lr < 1e-8:
                break

    return best_h, best_c5


def interpolated_slsqp(h_start, n, target, alphas, maxiter_per_alpha, time_limit):
    start = time.time()
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            h_new, sc = slsqp_round(h_cur, n, target, [alpha], maxiter_per_alpha, 1e-15)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def thue_morse_init(n):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    h = 1.0 - t.astype(float)
    target = n / 2.0
    return project_feasible(h, target)


def sawtooth_init(n, period):
    h = np.zeros(n)
    duty = int(period / 2)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    return project_feasible(h, float(n) / 2.0)


def structured_perturb(h, n, target, seed):
    rng = np.random.RandomState(seed)
    h_p = h.copy()
    n_flips = rng.randint(max(2, n // 20), n // 5)
    positions = rng.choice(n, size=n_flips, replace=False)
    for p in positions:
        if h_p[p] > 0.5:
            h_p[p] = 0.0
        else:
            h_p[p] = 1.0
    return project_feasible(h_p, target)
