```python
        # dcorr[k]/dx[j] = (1-x[j-lag])*[0<=j-lag<n] - x[j+lag]*[0<=j+lag<n]
        j_range = np.arange(n)
        valid1 = (j_range - lag >= 0) & (j_range - lag < n)
        valid2 = (j_range + lag >= 0) & (j_range + lag < n)
        grad[valid1] += w * (1.0 - x[j_range[valid1] - lag])
        grad[valid2] -= w * x[j_range[valid2] + lag]
    
    return grad
```

Hmm, this gradient computation is O(n²) because we loop over all 2n-1 lags and for each do O(n) work. That's O(n²) total. For n=1600, that's about 5M operations per gradient evaluation, which is fine.

But actually, we can be smarter. The weights `p_k` are concentrated on the top few lags (for large alpha). So we only need to compute the gradient contribution for lags with significant weight. Let me use a threshold.

Also, for the Frank-Wolfe method, I don't need the exact gradient - I can use a subgradient (just the gradient of the max corr[k] for the argmax k, or an average over the top few). This is much cheaper.

Actually, let me use a simpler approach: **FISTA-like proximal gradient** adapted for this problem.

The proximal operator for the constraint set P = {h: 0≤h≤1, Σh=n/2} is just the projection onto P, which we already have.

So: **Proximal Gradient Descent (PGD)**:
```
h_0 = initial
L = Lipschitz constant of gradient (need to estimate)
for t = 1, 2, ...:
    g_t = ∇f(h_{t-1})  (or subgradient)
    y_t = h_{t-1} - (1/L) * g_t
    h_t = proj_P(y_t)
```

With Nesterov acceleration (FISTA):
```
h_0 = y_0 = initial
t = 1
for t = 1, 2, ...:
    g_t = ∇f(y_t)
    h_t = proj_P(y_t - (1/L) * g_t)
    t_{t+1} = (1 + sqrt(1 + 4*t_t²)) / 2
    y_{t+1} = h_t + ((t_t - 1) / t_{t+1}) * (h_t - h_{t-1})
```

This gives O(1/t²) convergence for smooth convex functions. But our function is non-convex, so convergence isn't guaranteed. Still, it might work well in practice.

The Lipschitz constant L: The gradient of corr[k] is bounded, and the max of n_corr such gradients is bounded by... let me think. Each ∂corr[k]/∂h[j] is at most 1 in magnitude (since it's (1-x) - x' ≤ 1). The sum over active constraints (at most n_corr of them) gives |g| ≤ n_corr * 1 = 2n-1. But with the smooth max weighting, it's bounded by 1 (since the weights sum to 1, and each individual gradient is bounded by 1 in infinity norm... wait no, the L-infinity norm of each ∂corr[k]/∂h is at most 1, so the weighted sum also has L-infinity norm ≤ 1).

Actually wait: ∂corr[k]/∂h[j] = (1-h[j-k])*[valid] - h[j+k]*[valid]
The absolute value is at most max(1-h, h) ≤ 1. But for a fixed k, the gradient vector has entries that are each bounded by 1 in absolute value. So the L-infinity norm of each gradient is ≤ 1. The weighted sum (with weights summing to 1) also has L-infinity norm ≤ 1.

But Lipschitz constant is about the L2 norm of the Hessian (or the change in gradient). The Hessian of corr[k] has entries that are -1 (from the -h[j+k]*h[j-k] term when both indices are valid... wait no, the Hessian involves second derivatives).

Actually, corr[k] = Σ_i h[i+k]*(1-h[i]), so the Hessian has:
∂²corr[k]/∂h[j]∂h[l] = -1 if j = l+k and l = j-k (i.e., if the pair (j,l) appears in the sum)

So the Hessian is a (signed) permutation-like matrix with at most n non-zero entries, each being -1. The spectral norm is at most 1.

For the smooth max, the Hessian is a weighted sum of these Hessians plus the Hessian of the log-sum-exp term. The log-sum-exp Hessian is positive semidefinite and its magnitude depends on alpha.

So L ≈ 1 + O(alpha * range(corr)). For alpha = 100 and range ~ 0.5, L ~ 50.

Let me just use L = 2*n for safety (very conservative) or estimate it empirically.

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fista_subgrad(h_init, n, target, time_limit, n_top=5, beta_start=0.01,
                   max_iter=50000):
    """FISTA with subgradient of max corr."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    
    h = h_init.copy().astype(float)
    y = h.copy()
    t_k = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    L = 2.0  # initial Lipschitz estimate
    stale = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Compute subgradient from top n_top lags
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argpartition(corr, -n_top)[-n_top:]
        corr_max = np.max(corr)
        # Weight by how close to max
        w = np.zeros(len(top_idx))
        for i, idx in enumerate(top_idx):
            w[i] = np.exp(-50 * (corr_max - corr[idx]))
        w /= w.sum()
        
        # Compute subgradient
        g = np.zeros(n)
        for i, idx in enumerate(top_idx):
            if w[i] < 1e-10:
                continue
            lag = int(idx) - mid
            js = np.arange(n)
            v1 = (js - lag >= 0) & (js - lag < n)
            v2 = (js + lag >= 0) & (js + lag < n)
            g[v1] += w[i] * (1.0 - y[js[v1] - lag])
            g[v2] -= w[i] * y[js[v2] + lag]
        
        # FISTA update
        h_new = project_feasible(y - g / L, target)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = h_new + ((t_k - 1.0) / t_next) * (h_new - h)
        y_new = project_feasible(y_new, target)  # keep y feasible too
        t_k = t_next
        h = h_new
        y = y_new
        
        # Check improvement
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 5000:
                # Reset momentum
                y = h.copy()
                t_k = 1.0
                stale = 0
        
        # Adaptive L
        if it % 100 == 0 and it > 0:
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-6:
                break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

def enhanced_cd(h_init, n, target, time_limit, max_sweeps=20000,
                n_active=3, n_cand_start=15, n_cand_end=50,
                max_no_improve=400, bisection_iters=12):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break
        
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum
        
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        max_pairs = 200
        if n_h * n_l <= max_pairs:
            pair_list = []
            for ii in range(n_h):
                for jj in range(n_l):
                    i_val = int(high_set[ii])
                    j_val = int(low_set[jj])
                    if i_val != j_val:
                        pair_list.append((i_val, j_val))
        else:
            pair_list = []
            for frac in [0.0, 0.25, 0.5, 0.75, 1.0]:
                hi = int(frac * (n_h - 1))
                lo = int(frac * (n_l - 1))
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pair_list.append((iv, jv))
            rng = np.random.RandomState(sweep)
            for _ in range(max_pairs - len(pair_list)):
                ii = int(rng.randint(n_h))
                jj = int(rng.randint(n_l))
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))
        
        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue
            
            g = change[j] - change[i]
            
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
    
    return best_h, best_c5
```

    n = len(h)

    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)

            break

            break
