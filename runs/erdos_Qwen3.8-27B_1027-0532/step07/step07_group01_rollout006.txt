```python
def smoothed_gradient_descent(h, n, target, time_limit, T_start=1.0, T_end=0.01, 
                                alpha_start=0.1, momentum=0.9):
    start = time.time()
    dx = 2.0/n
    n_corr = 2*n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h.copy().astype(float)
    v = np.zeros(n)  # momentum
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    n_total = estimate_total_iterations(time_limit)
    
    for it in range(n_total):
        if time.time() - start > time_limit:
            break
        
        progress = float(it) / n_total
        T = T_start * (T_end/T_start) ** progress
        alpha = alpha_start * (1 - progress) ** 0.5
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Compute softmax weights
        shifted = corr / T
        shifted -= np.max(shifted)
        w = np.exp(shifted)
        w /= np.sum(w)
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for k_idx in range(n_corr):
            if w[k_idx] < 1e-10:
                continue
            k = k_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[k_idx] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[k_idx] * h[idxs[v2] + k]
        
        grad -= np.mean(grad)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        h = h - alpha * v
        h = project_feasible(h, target)
    
    return best_h, best_c5
```

  Let m = j + n - 1: sum_k W_np[k] * (1-h)[m - k] = convolve(W_np, 1-h)[j + n - 1] ✓

- For the second convolution: we want sum_k W_np[k] * h[j + (k-(n-1))]
  = sum_k W_np[k] * h[j + k - n + 1]
  
  Let me substitute: this is sum_k W_np[k] * h[(j-n+1) + k]
  
  This is the cross-correlation of h and W_np at offset (j-n+1):
  = correlate(h, W_np)[(j-n+1) + len(W_np) - 1] ... 
  
  Actually, np.correlate(a, v)[k] = sum_n a[n+k] * v[n]
  
  We want sum_k W_np[k] * h[j+k-n+1]. Let n = k, then:
  = sum_n W_np[n] * h[j+n-n+1] ... that doesn't simplify nicely.
  
  Let me try: sum_k W_np[k] * h[j + k - (n-1)]
  Let s = j + k - (n-1), so k = s - j + (n-1):
  = sum_s W_np[s - j + (n-1)] * h[s]
  
  This is np.correlate(h, W_np)[?] 
  np.correlate(h, W_np, mode='full')[k] = sum_s h[s] * W_np[s+k]
  
  We want sum_s W_np[s - j + (n-1)] * h[s] = sum_s h[s] * W_np[s + (n-1-j)]
  = np.correlate(h, W_np, mode='full')[n-1-j]

So:
grad[j] = convolve(W_np, 1-h, mode='full')[j + n - 1] - correlate(h, W_np, mode='full')[n - 1 - j]

Actually, I think there might be an even simpler way. Let me use FFT:

first_part = ifft(fft(W_np, N) * fft(1-h, N)).real  # full convolution
second_part = ifft(fft(h, N) * fft(W_np, N)).real  # this gives correlate(h, W_np) with appropriate indexing

Hmm, the relationship between FFT and correlate/convolve is:
- Convolution: ifft(fft(a) * fft(b))
- Correlation: ifft(fft(conj(a)) * fft(b)) or equivalently ifft(fft(b) * fft(conj(a)))

For real signals: correlate(a, b) = convolve(a, flip(b))

So: correlate(h, W_np) = convolve(h, flip(W_np))
= ifft(fft(h, N) * fft(flip(W_np), N)).real

And the full convolution has length len(h) + len(W_np) - 1 = n + (2n-1) - 1 = 3n - 2.

So:
first_full = ifft(fft(W_np, N) * fft(1-h, N)).real[:3*n-2]
  → first_full[m] = convolve(W_np, 1-h)[m] for m = 0 to 3n-3
  → We need first_full[j + n - 1] for j = 0 to n-1

second_full = ifft(fft(h, N) * fft(W_np[::-1], N)).real[:3*n-2]
  → second_full[m] = convolve(h, flip(W_np))[m] = correlate(h, W_np)[m - (len(W_np)-1)]
  
  Wait, let me be more careful.
  
  correlate(h, W_np, mode='full')[k] = sum_n h[n] * W_np[n+k] (for valid n)
  
  convolve(h, flip(W_np), mode='full')[m] = sum_n h[n] * flip(W_np)[m-n]
  = sum_n h[n] * W_np[len(W_np)-1-(m-n)]
  = sum_n h[n] * W_np[(2n-2)-(m-n)]  [since len(W_np) = 2n-1, so len-1 = 2n-2]
  = sum_n h[n] * W_np[2n-2-m+n]
  
  We want correlate(h, W_np)[k] = sum_n h[n] * W_np[n+k]
  
  Setting 2n-2-m+n = n+k:
  2n-2-m+n = n+k
  2n - 2 - m = k
  m = 2n - 2 - k
  
  So correlate(h, W_np)[k] = convolve(h, flip(W_np))[2n-2-k]
  
  We need correlate(h, W_np)[n-1-j] = convolve(h, flip(W_np))[2n-2-(n-1-j)] = convolve(h, flip(W_np))[n-1+j]

So:
grad[j] = first_full[j + n - 1] - second_full[n - 1 + j]

where:
first_full = ifft(fft(W_np, N) * fft(1-h, N)).real
second_full = ifft(fft(h, N) * fft(W_np[::-1], N)).real

This is O(n log n) per iteration! Much better.

Let me code this up:

```python
def compute_smoothed_grad(w, h, n):
    N = 1
    total_len = 3 * n
    while N < total_len:
        N *= 2
    
    W = fft(w, N)
    H = fft(h, N)
    ONE_MH = fft(1.0 - h, N)
    W_FLIP = fft(w[::-1], N)
    
    first_full = ifft(W * ONE_MH).real[:3*n-2]
    second_full = ifft(H * W_FLIP).real[:3*n-2]
    
    grad = first_full[n-1:n-1+n] - second_full[n-1:n-1+n]
    grad -= np.mean(grad)
    return grad
```

This should work and be very fast!

Now, let me think about the overall algorithm:

1. Start with initial_h_values (n=400)
2. Quick SLSQP polish
3. Interpolate to n=600
4. Run smoothed gradient descent with momentum for most of the budget
5. Periodic SLSQP polish
6. Maybe interpolate to n=800
7. Final polish

Key hyperparameters for SGD:
- Temperature T: start high (exploratory), decrease over time (exploitation)
- Step size α: start moderate, decrease
- Momentum β: 0.8-0.95
- Restart frequency: when stuck

Let me also add a "plateau escape" mechanism: when no improvement for X iterations, add noise and reset momentum.

Let me also think about whether to use the full softmax or just the top-K shifts. Using top-K is cheaper and might be more effective (focuses on the most important constraints).

Actually, for the FFT approach, the cost is O(n log n) regardless of how many shifts are "active" in the softmax. So I might as well use the full softmax.

But wait - with very small T, the softmax becomes concentrated on the top shifts. In that case, I could just use the top shift (or top few) and compute the gradient more simply. Let me use a hybrid:
- If T is large: use FFT-based full gradient
- If T is small: just use the top K shifts

For simplicity, let me just always use the FFT approach since it's efficient enough.

Let me now write the complete solution:

```python
import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_weighted_grad_fft(w, h, n):
    """Compute the smoothed gradient using FFT."""
    total_len = 2 * (2*n - 1) + n - 1  # full conv length
    N = 1
    while N < total_len + 1:
        N <<= 1
    
    W = fft(w, N)
    H = fft(h, N)
    ONE_MH = fft(1.0 - h, N)
    W_FLIP = fft(w[::-1], N)
    
    first_full = ifft(W * ONE_MH).real[:total_len]
    second_full = ifft(H * W_FLIP).real[:total_len]
    
    # grad[j] = first_full[j + n - 1] - second_full[n - 1 + j]
    grad = first_full[n-1:n-1+n] - second_full[n-1:n-1+n]
    grad -= np.mean(grad)
    return grad

def smoothed_gd(h_init, n, target, time_limit, T_start=0.5, T_end=0.005,
                alpha_start=0.05, momentum=0.9, n_restarts=5):
    """Smoothed gradient descent with momentum and restarts."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    rng = np.random.RandomState(12345)
    
    # Estimate how many iterations we can do
    # Do a timing test
    t0 = time.time()
    iters_test = 0
    while time.time() - t0 < 0.1:
        corr = np.correlate(h, 1.0 - h, mode='full')
        shifted = corr / T_start
        shifted -= np.max(shifted)
        w = np.exp(shifted)
        w /= np.sum(w)
        grad = compute_weighted_grad_fft(w, h, n)
        iters_test += 1
    time_per_iter = 0.1 / max(iters_test, 1)
    max_iters = int((time_limit - 5) / time_per_iter)
    
    for it in range(max_iters):
        if time.time() - start > time_limit - 2:
            break
        if (it + 1) % 200 == 0 and time.time() - start > time_limit - 2:
            break
        
        progress = min(float(it) / max_iters, 1.0)
        T = T_start * (T_end / T_start) ** progress
        alpha = alpha_start * (1.0 - 0.5 * progress)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Plateau escape
        if no_improve > 300:
            h = best_h + rng.randn(n) * 0.02
            h = project_feasible(h, target)
            v = np.zeros(n)
            no_improve = 0
            if it > max_iters - 500:
                break
        
        # Compute softmax weights
        shifted = corr / T
        shifted -= np.max(shifted)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.zeros(n_corr)
            w[np.argmax(corr)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        grad = compute_weighted_grad_fft(w, h, n)
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        h = h - alpha * v
        h = project_feasible(h, target)
    
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=80, max_no_improve=500,
            restart_every=250, restart_scale=0.02):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(777)
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        if consecutive_no_improve > restart_every and consecutive_no_improve % 50 == 0:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            for p in range(n):
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
            consecutive_no_improve = 0
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change matrix
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)

        return budget_s - 15.0 - (time.time() - start_time)
```

import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_weighted_grad_fft(w, h, n):
    N = 1
    total_len = 3 * n - 2
    while N < total_len + 1:
        N <<= 1
    W = fft(w, N)
    H = fft(h, N)
    OMH = fft(1.0 - h, N)
    WF = fft(w[::-1], N)
    first = ifft(W * OMH).real[:total_len]
    second = ifft(H * WF).real[:total_len]
    offset = n - 1
    grad = first[offset:offset + n] - second[offset:offset + n]
    grad -= np.mean(grad)
    return grad
