Actually, a better version: given h, for each level t ∈ {0.3, 0.4, 0.5, 0.6, 0.7}, compute h_t = (h > t) and then run a few FISTA steps from h_t. Keep the best.

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    n = len(h)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_grad(h, corr, alpha, n):
    c_max = np.max(corr)
    s = alpha * (corr - c_max)
    s = np.clip(s, -500.0, 0.0)
    ew = np.exp(s)
    sw = np.sum(ew)
    w = ew / sw
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    thresh = 1e-12
    for j in range(n_corr):
        if w[j] < thresh:
            continue
        m = j - mid
        if m >= 0:
            p1 = idxs[:n - m]
            grad[p1] += w[j] * (1.0 - h[p1 + m])
            if m > 0:
                p2 = idxs[m:]
                grad[p2] -= w[j] * h[p2 - m]
        else:
            kk = -m
            p1 = idxs[kk:]
            grad[p1] += w[j] * (1.0 - h[p1 + m])
            p2 = idxs[:n - kk]
            grad[p2] -= w[j] * h[p2 - m]
    grad -= np.mean(grad)
    return grad

def fista_alpha(h_init, n, target, alpha, time_limit, max_iter=20000):
    start = time.time()
    h = project_feasible(h_init.astype(np.float64), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    y = h.copy()
    t = 1.0
    lr = 1.0 / (alpha * 2.0)
    no_improve = 0
    corr_y = np.correlate(y, 1.0 - y, mode='full')
    stall_count = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > max_iter // 2:
            break
        grad = _smooth_grad(y, corr_y, alpha, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        f_y = compute_c5(y)
        h_new = None
        for _ in range(20):
            ht = project_feasible(y - lr * grad, target)
            ft = compute_c5(ht)
            if ft <= f_y + 1e-13:
                h_new = ht
                break
            lr *= 0.5
        if h_new is None:
            y = h.copy()
            t = 1.0
            lr = 1.0 / (alpha * 2.0)
            stall_count += 1
            if stall_count > 5:
                break
            continue
        stall_count = 0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        t = t_new
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        h = h_new
        lr = min(lr * 1.02, 4.0 / (alpha * 2.0))
        sc = compute_c5(h_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                y = h.copy()
                t = 1.0
    return best_h, best_score

def fista_schedule(h_init, n, target, time_limit, alphas):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    for alpha in alphas:
        elapsed = time.time() - start
        if elapsed > time_limit - 5:
            break
        remaining = time_limit - elapsed
        h_f, sc_f = fista_alpha(h, n, target, alpha, remaining - 5,
                                max_iter=min(30000, int(remaining * 50)))
        h = h_f
        if sc_f < best_score:
            best_score = sc_f
            best_h = h.copy()
    return best_h, best_score

def pair_cd(h_init, n, target, time_limit, max_sweeps=15000, n_cand=40):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(np.float64)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    def compute_change_row(p):
        row = np.zeros(n_corr)
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        row[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        row[neg_mask] -= h[p - shifts[neg_mask]]
        return row

    change = np.array([compute_change_row(p) for p in range(n)])

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))
        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        n_active = min(5, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            ch_i = change[i]
            h_i = h[i]
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                ch_j = change[j]
                h_j = h[j]
                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue
                g = ch_j - ch_i
                a_d = delta_min
                b_d = delta_max
                for _ in range(8):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i] = compute_change_row(best_i)
            change[best_j] = compute_change_row(best_j)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_threshold_search(h_smooth, n, target, time_limit):
    start = time.time()
    best_h = h_smooth.copy()
    best_score = compute_c5(h_smooth)
    for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
        if time.time() - start > time_limit - 2:
            break
        h_bin = (h_smooth > thresh).astype(np.float64)
        h_bin = project_feasible(h_bin, target)
        sc = compute_c5(h_bin)
        if sc < best_score:
            best_score = sc
            best_h = h_bin.copy()
        h_mixed = 0.5 * h_smooth + 0.5 * h_bin
        h_mixed = project_feasible(h_mixed, target)
        sc2 = compute_c5(h_mixed)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_mixed.copy()
    return best_h, best_score

def slsqp_polish(h_init, n, target, time_limit):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    h = project_feasible(h_init.astype(np.float64), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    alphas = [500, 2000, 10000]
    for alpha in alphas:
        if time.time() - start > time_limit - 5:
            break
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            s = alpha * (corr - c_max)
            s = np.clip(s, -500.0, 0.0)
            return (c_max + np.log(np.sum(np.exp(s))) / alpha)
        def cons_fun(x):
            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, method='SLSQP',
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           bounds=bounds, options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h = h_new
        except Exception:
            pass
    return best_h, best_score
```
