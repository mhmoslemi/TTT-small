```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
```

```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=10, n_cand_end=40, max_no_improve=300,
            weight_beta=5.0, T0=0.0003, T_decay=0.003, rng_seed=777):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(rng_seed)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        T = T0 * np.exp(-T_decay * sweep) if T0 > 0 else 0.0

        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        top_k = min(40, n_corr)
        top_idx = np.argsort(weights)[-top_k:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break
                elif T > 1e-12 and approx_max - c5_val < T * 5:
                    if rng.random() < np.exp(-(approx_max - c5_val) / T):
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
                        break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def momentum_descent(h_init, n, target, time_limit, n_iter=50000,
                     step_init=0.004, mom=0.85, top_k=30, wt=4.0):
    start = time.time()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    no_improve = 0

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 400:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx_tmp if False else float(np.max(corr)) * (2.0 / n)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        c_max = np.max(corr)
        for s_idx in top_idx:
            w = np.exp(wt * (corr[s_idx] - c_max))
            if w < 1e-6:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        v = mom * v - step * grad
        h_new = project_feasible(h + v, target)
        sc = compute_c5(h_new)

        if sc < c5_val - 1e-15:
            h = h_new
            step = min(step * 1.01, 0.02)
        else:
            step *= 0.93
            v = mom * v

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP at initial resolution
    if tleft() > 120:
        h_p, s_p = slsqp_optimize(best_h, n0, target0, [200, 1000, 5000], 600, 1e-14)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: Pair CD at initial resolution
    if tleft() > 250:
        h_cd, s_cd = pair_cd(best_h, n0, target0,
                             time_limit=min(200, tleft() * 0.5),
                             max_sweeps=30000,
                             n_cand_start=10, n_cand_end=35,
                             max_no_improve=300,
                             weight_beta=6.0, T0=0.0004, T_decay=0.004, rng_seed=1)
        if s_cd < best_score:
            best_score = s_cd
            best_h = h_cd

    # Phase 3: Momentum descent at initial resolution
    if tleft() > 180:
        h_mom, s_mom = momentum_descent(best_h, n0, target0,
                                        time_limit=min(120, tleft() * 0.4),
                                        n_iter=40000, step_init=0.003,
                                        mom=0.8, top_k=25, wt=5.0)
        if s_mom < best_score:
            best_score = s_mom
            best_h = h_mom

    # Phase 4: SLSQP polish
    if tleft() > 100:
        h_p2, s_p2 = slsqp_optimize(best_h, n0, target0, [5000, 20000], 500, 1e-14)
        if s_p2 < best_score:
            best_score = s_p2
            best_h = h_p2

    # Phase 5: Interpolate to n=1600 and optimize
    n1 = 1600
    target1 = n1 / 2.0
    if tleft() > 450:
        h1 = interpolate_to(best_h, n1, target1)
        cur_h = h1.copy()
        cur_score = compute_c5(h1)

        # Pair CD at n=1600
        if tleft() > 350:
            h_cd1, s_cd1 = pair_cd(h1, n1, target1,
                                   time_limit=min(300, tleft() * 0.5),
                                   max_sweeps=15000,
                                   n_cand_start=15, n_cand_end=45,
                                   max_no_improve=300,
                                   weight_beta=5.0, T0=0.0003, T_decay=0.003, rng_seed=2)
            if s_cd1 < cur_score:
                cur_score = s_cd1
                cur_h = h_cd1

        # Momentum at n=1600
        if tleft() > 200:
            h_mom1, s_mom1 = momentum_descent(cur_h, n1, target1,
                                              time_limit=min(150, tleft() * 0.4),
                                              n_iter=30000, step_init=0.002,
                                              mom=0.8, top_k=30, wt=4.0)
            if s_mom1 < cur_score:
                cur_score = s_mom1
                cur_h = h_mom1

        # SLSQP at n=1600
        if tleft() > 120:
            h_p3, s_p3 = slsqp_optimize(cur_h, n1, target1, [2000, 10000], 300, 1e-14)
            if s_p3 < cur_score:
                cur_score = s_p3
                cur_h = h_p3

        # Another round of pair CD
        if tleft() > 100:
            h_cd2, s_cd2 = pair_cd(cur_h, n1, target1,
                                   time_limit=min(80, tleft() * 0.5),
                                   max_sweeps=10000,
                                   n_cand_start=20, n_cand_end=50,
                                   max_no_improve=200,
                                   weight_beta=8.0, T0=0.0, T_decay=0.0, rng_seed=3)
            if s_cd2 < cur_score:
                cur_score = s_cd2
                cur_h = h_cd2

        if cur_score < best_score:
            best_score = cur_score
            best_h = cur_h

    # Phase 6: Multi-start from perturbed solutions
    if tleft() > 150:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.08, 0.15]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, t_cur)
            h_ref, s_ref = slsqp_optimize(h_pert, n_cur, t_cur, [500, 5000], 300, 1e-14)
            if s_ref < best_score:
                best_score = s_ref
                best_h = h_ref
            if tleft() > 60:
                h_cd_ref, s_cd_ref = pair_cd(best_h, n_cur, t_cur,
                                             time_limit=min(40, tleft() * 0.4),
                                             max_sweeps=8000,
                                             n_cand_start=15, n_cand_end=40,
                                             max_no_improve=150,
                                             weight_beta=5.0, T0=0.0003, T_decay=0.003, rng_seed=seed + int(noise_scale * 100))
                if s_cd_ref < best_score:
                    best_score = s_cd_ref
                    best_h = h_cd_ref

    # Phase 7: Final pair CD
    if tleft() > 60:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_fin_cd, s_fin_cd = pair_cd(best_h, n_cur, t_cur,
                                     time_limit=min(50, tleft() * 0.6),
                                     max_sweeps=10000,
                                     n_cand_start=20, n_cand_end=55,
                                     max_no_improve=150,
                                     weight_beta=10.0, T0=0.0, T_decay=0.0, rng_seed=99)
        if s_fin_cd < best_score:
            best_score = s_fin_cd
            best_h = h_fin_cd

    # Phase 8: Final SLSQP
    if tleft() > 30:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_fin, s_fin = slsqp_optimize(best_h, n_cur, t_cur, [20000, 50000], 300, 1e-14)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
```
