        return np.clip(h, 0, 1)

        for s_idx in range(2*n-1):
            k = s_idx - (n-1)
            w = weights[s_idx]
            if w < 1e-12:
                continue
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        # Project gradient onto feasible direction (sum = 0)
        grad -= np.mean(grad)
        
        # Line search
        best_f = c5_val
        best_step = 0.0
        step = step_size
        for _ in range(20):
            trial = h - step * grad
            trial = project_box_sum(trial, target)
            trial_c5 = compute_c5(trial)
            if trial_c5 < best_f - 1e-14:
                best_f = trial_c5
                best_step = step
            step *= 0.7
            if step < 1e-8:
                break
        
        if best_step > 0:
            h = project_box_sum(h - best_step * grad, target)
        else:
            # Small random perturbation to escape
            h = project_box_sum(h + 0.01 * np.random.randn(n), target)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target_sum):
    """Project onto {x: 0<=x_i<=1, sum(x)=target_sum} using bisection on lambda."""
    n = len(h)
    target_sum = max(0.0, min(float(n), target_sum))
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    projected = np.clip(h - lam, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = (projected < 1.0) & (projected > 0.0)
            if not mask.any():
                mask = projected < 1.0
        else:
            mask = (projected > 0.0) & (projected < 1.0)
            if not mask.any():
                mask = projected > 0.0
        if mask.any():
            projected[mask] += diff / mask.sum()
            projected = np.clip(projected, 0.0, 1.0)
    return projected

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def adaptive_projected_gradient(h_init, n, target, time_limit,
                                 beta_start=3.0, beta_end=80.0,
                                 max_iters=20000, step_init=0.05,
                                 max_no_improve=2000):
    """Adaptive projected gradient with softmax-smoothed max."""
    dx = 2.0 / n
    start = time.time()
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    step = step_init
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            step = min(step * 1.05, 0.2)
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve % 100 == 0:
                step = max(step * 0.7, 1e-5)
        
        if consecutive_no_improve > max_no_improve:
            break
        
        progress = min(float(it) / max_iters, 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        c_max = np.max(corr)
        exp_vals = np.exp(-beta * (c_max - corr))
        weights = exp_vals / np.sum(exp_vals)
        
        # Compute weighted gradient (only for shifts with significant weight)
        grad = np.zeros(n)
        threshold = 1.0 / (len(corr) * 100)  # ignore very small weights
        for s_idx in range(len(corr)):
            w = weights[s_idx]
            if w < threshold:
                continue
            k = s_idx - (n - 1)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        # Project gradient to feasible direction
        grad -= np.mean(grad)
        
        # Backtracking line search
        best_trial_f = c5_val
        best_trial_h = h.copy()
        s = step
        for ls in range(15):
            trial = project_box_sum(h - s * grad, target)
            trial_c5 = compute_c5(trial)
            if trial_c5 < best_trial_f - 1e-15:
                best_trial_f = trial_c5
                best_trial_h = trial
            s *= 0.7
            if s < 1e-7:
                break
        
        h = best_trial_h
    
    return best_h, best_c5

def fast_pair_cd(h_init, n, target, time_limit,
                 beta_start=5.0, beta_end=100.0,
                 max_sweeps=30000, n_cand_start=30, n_cand_end=70,
                 max_no_improve=600):
    """Fast pair CD with softmax-weighted gradient for candidate selection."""
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        c_max = np.max(corr)
        exp_vals = np.exp(-beta * (c_max - corr))
        weights = exp_vals / np.sum(exp_vals)
        
        # Weighted gradient
        grad = np.zeros(n)
        for s_idx in range(n_corr):
            w = weights[s_idx]
            if w < 1e-10:
                continue
            k = s_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = -min(1.0 - h[i], h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    
    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14,
                time_limit=None, start_time=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

_g_n = 0
_g_target = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src.copy(), target_dst)

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_sum(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue
        delta_lo = -min(1.0 - h[i], h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue
        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx
        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-12:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True
        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            cur_c5 = approx_new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1
    return best_h, best_c5
