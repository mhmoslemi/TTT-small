    best_h = h.copy()

    for sweep in range(max_sweeps):

            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_val = c_max * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val

            if no_improve > 1000:
                break

        # corr[k] = sum_n h[n] * (1-h[n-mid+k])
        # Let m = k - mid (the shift), then:
        # corr[k] = sum_n h[n] * (1-h[n-m])
        # where both n and n-m are in [0, n-1]
        
        # d(corr[k])/d(h[i]) = (1-h[i-m]) if i-m is valid (0 <= i-m < n)
        #                      - h[i+m] if i+m is valid (0 <= i+m < n)
        #                      (if both are valid, both terms contribute)
        
        # So for each variable i and each shift m:
        # J[i, m_idx] = (1-h[i-m]) * [0<=i-m<n] - h[i+m] * [0<=i+m<n]
        
        # Now, for the LP, we pick block_size variables and solve:
        # min t
        # s.t. corr[k] + sum_i delta_i * J[i_idx_i, k] <= t for all active k
        #      sum delta_i = 0
        #      -(h[i_idx_i]) <= delta_i <= (1 - h[i_idx_i])
        
        # Choose which variables: pick those with most extreme "descent potential"
        # For each variable i, the best improvement if we increase it is:
        #   - sum over active k of J[i,k] * corr[k] (direction of improvement)
        # Actually, we want to minimize max(corr + delta * J_row), so we want
        # J_row to be negative where corr is large.
        
        # Score for variable i: -sum_{k in active} corr[k] * J[i, k]
        # (We want this to be positive, meaning decreasing i would help)
        
        # But computing this for all n variables is O(n * n_corr) = O(n^2) per sweep
        # For n=400, that's 160K operations, which is fine.
        
        # Let's compute scores efficiently
        # score[i] = -sum_{k in active} corr[k] * J[i, k]
        # = -sum_{k in active} corr[k] * [(1-h[i-m_k])*valid - h[i+m_k]*valid]
        
        # This is getting complex. Let me use a simpler heuristic:
        # Compute the full gradient of the smoothed objective and pick extreme values.
        
        # Gradient of sum_k corr[k] (all shifts, not just active):
        # g[i] = sum_k J[i, k] = sum_m [(1-h[i-m])*[valid] - h[i+m]*[valid]]
        # = sum_{m: i-m valid} (1-h[i-m]) - sum_{m: i+m valid} h[i+m]
        # = (number of valid i-m) - sum_{m: i-m valid} h[i-m] - sum_{m: i+m valid} h[i+m]
        
        # For the active set only:
        # g[i] = sum_{k in active} J[i, k]
        
        # Let me just compute the gradient numerically for the active set
        # Using the formula:
        # g[i] = sum_{m in active_shifts} (1-h[i-m])*[0<=i-m<n] - h[i+m]*[0<=i+m<n]
        
        active_shifts = active_idx - mid  # actual shift values
        
        g = np.zeros(n)
        for m in active_shifts:
            if m >= 0:
                # (1-h[i-m]) for i >= m: i-m in [0, n-1]
                valid = np.arange(m, n)
                g[valid] += (1.0 - h[valid - m])
                # -h[i+m] for i < n-m: i+m in [0, n-1]
                valid2 = np.arange(0, n - m)
                g[valid2] -= h[valid2 + m]
            elif m < 0:
                mm = -m
                # (1-h[i-mm]) for i >= mm
                valid = np.arange(mm, n)
                g[valid] += (1.0 - h[valid - mm])
                # -h[i+mm] for i < n-mm
                valid2 = np.arange(0, n - mm)
                g[valid2] -= h[valid2 + mm]
        
        # Wait, I think I'm confusing myself. Let me redo.
        # J[i, m_idx] = d(corr[m_idx])/d(h[i])
        # corr at shift m: corr[m+mid] = sum_j h[j] * (1-h[j-m]) for valid j, j-m
        # d/dh[i]: 
        #   term where j = i: (1-h[i-m]) if i-m valid
        #   term where j-m = i: -h[j] = -h[i+m] if i+m valid (j = i+m)
        # So J[i, m_idx] = (1-h[i-m])*[0<=i-m<n] + (-h[i+m])*[0<=i+m<n]
        
        # OK so my formula above is correct.
        
        # Now pick block_size variables with most extreme g values
        g_centered = g - np.mean(g)
        n_half = block_size // 2
        sorted_idx = np.argsort(g_centered)
        low_vars = sorted_idx[:n_half]  # most negative
        high_vars = sorted_idx[-n_half:]  # most positive
        if block_size % 2 == 1:
            # Add one more from the middle extreme
            mid_candidates = np.setdiff1d(sorted_idx, np.concatenate([low_vars, high_vars]))
            high_vars = np.append(high_vars, mid_candidates[np.argmin(np.abs(g_centered[mid_candidates]))])
        
        block_idx = np.concatenate([low_vars, high_vars])
        
        # Build LP
        # Variables: delta_0, ..., delta_{m-1}, t
        # Minimize t
        # Constraints:
        #   For each active k: sum_i J[block_idx[i], k] * delta_i - t <= -corr[k]
        #   sum delta_i = 0
        #   -h[block_idx[i]] <= delta_i <= 1 - h[block_idx[i]]
        
        m_vars = len(block_idx)
        # Total variables: m_vars + 1 (for t)
        
        # A_ub rows: active constraints + sum constraint
        # c = [0, ..., 0, 1] (minimize t)
        
        J_rows = np.zeros((len(active_idx), m_vars))
        for ai, k_idx in enumerate(active_idx):
            m_val = k_idx - mid
            for bi, i_val in enumerate(block_idx):
                j_valid = (0 <= i_val - m_val < n)
                jp_valid = (0 <= i_val + m_val < n)
                J_rows[ai, bi] = 0.0
                if j_valid:
                    J_rows[ai, bi] += (1.0 - h[i_val - m_val])
                if m_val != 0 and jp_valid:
                    J_rows[ai, bi] -= h[i_val + m_val]
        
        # LP: min [0,...,0,1] . [delta, t]
        # s.t. J_rows * delta - t <= -corr[active_idx]
        #      sum(delta) = 0
        #      -h[block_idx] <= delta <= 1-h[block_idx]
        
        # Using linprog:
        # min c @ x
        # s.t. A_ub @ x <= b_ub
        #      A_eq @ x == b_eq
        #      bounds
        
        c_lp = np.zeros(m_vars + 1)
        c_lp[-1] = 1.0  # minimize t
        
        A_ub = np.zeros((len(active_idx) + 1, m_vars + 1))
        b_ub = np.zeros(len(active_idx) + 1)
        
        # Inequality constraints: J_rows @ delta - t <= -corr_active
        A_ub[:len(active_idx), :m_vars] = J_rows
        A_ub[:len(active_idx), -1] = -1.0
        b_ub[:len(active_idx)] = -corr[active_idx]
        
        # Sum constraint as equality
        A_eq = np.zeros((1, m_vars + 1))
        A_eq[0, :m_vars] = 1.0
        b_eq = np.zeros(1)
        
        bounds_lp = [(-h[i], 1.0 - h[i]) for i in block_idx] + [(0, None)]
        
        try:
            res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
            if res.success:
                delta = res.x[:m_vars]
                h_new = h.copy()
                h_new[block_idx] += delta
                h_new = project_feasible(h_new, target)
                new_c5 = compute_c5(h_new)
                if new_c5 < c5_val - 1e-14:
                    h = h_new
        except Exception:
            pass
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_lse, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def block_lp_cd(h_init, n, target, time_limit, start_time,
                max_sweeps=50000, block_size=4, active_frac=0.1,
                max_no_improve=500, step_scale=1.0):
    dx = 2.0 / n
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    n_half = block_size // 2
    n_extra = block_size - 2 * n_half
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        c5_val = c_max * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Find active shifts
        threshold = c_max * (1.0 - active_frac)
        active_idx = np.where(corr >= threshold)[0]
        
        # Compute gradient for active shifts
        g = np.zeros(n)
        for k_idx in active_idx:
            m = k_idx - mid
            if m == 0:
                g += (1.0 - h) - h  # = 1 - 2h
            elif m > 0:
                valid_neg = np.arange(m, n)
                g[valid_neg] += (1.0 - h[valid_neg - m])
                valid_pos = np.arange(0, n - m)
                g[valid_pos] -= h[valid_pos + m]
            else:
                mm = -m
                valid_neg = np.arange(mm, n)
                g[valid_neg] += (1.0 - h[valid_neg - mm])
                valid_pos = np.arange(0, n - mm)
                g[valid_pos] -= h[valid_pos + mm]
        
        g -= np.mean(g)
        
        # Pick block variables
        sorted_idx = np.argsort(g)
        low_vars = sorted_idx[:n_half]
        high_vars = sorted_idx[-n_half:]
        if n_extra > 0:
            # Pick from the remaining extremes
            used = set(low_vars.tolist() + high_vars.tolist())
            remaining = np.array([i for i in sorted_idx if i not in used])
            if len(remaining) > 0:
                extra_from_low = remaining[:n_extra//2 + n_extra%2]
                extra_from_high = remaining[-(n_extra//2):] if n_extra//2 > 0 else np.array([], dtype=int)
                if len(extra_from_high) == 0 and n_extra//2 > 0:
                    extra_from_high = remaining[-1:]
                low_vars = np.concatenate([low_vars, extra_from_low])
                high_vars = np.concatenate([high_vars, extra_from_high])
        
        block_idx = np.concatenate([low_vars, high_vars]).astype(int)
        m_vars = len(block_idx)
        
        # Build and solve LP
        # Variables: delta (m_vars) + t (1) = m_vars + 1
        # Minimize t
        # s.t.: for each active k: sum_i J[i,k]*delta_i <= t - corr[k]
        #       sum(delta_i) = 0
        #       -h[block_idx[i]] <= delta_i <= 1-h[block_idx[i]]
        
        n_active = len(active_idx)
        
        # Build J matrix: J[ai, bi] = d(corr[active_idx[ai]])/d(h[block_idx[bi]])
        J_mat = np.zeros((n_active, m_vars))
        for bi, i_val in enumerate(block_idx):
            for k_idx in active_idx:
                m = k_idx - mid
                val = 0.0
                if m == 0:
                    val += (1.0 - h[i_val])
                    val -= h[i_val]
                    # Actually for m=0: d(h[i]*(1-h[i]))/dh[i] = 1 - 2*h[i]
                    # But in the corr formula, for m=0:
                    # corr[0] = sum_j h[j]*(1-h[j])
                    # d/dh[i] = 1 - 2*h[i]
                    # But we also have the "other" term: -h[i] from the other index
                    # Wait, for m=0: corr = sum_j h[j] - sum_j h[j]^2
                    # d/dh[i] = 1 - 2*h[i]
                    val = 1.0 - 2.0 * h[i_val]
                elif m > 0:
                    # d(corr)/d(h[i]) = (1-h[i-m])*[i-m valid] - h[i+m]*[i+m valid]
                    if 0 <= i_val - m < n:
                        val += (1.0 - h[i_val - m])
                    if 0 <= i_val + m < n:
                        val -= h[i_val + m]
                else:
                    mm = -m
                    if 0 <= i_val - mm < n:
