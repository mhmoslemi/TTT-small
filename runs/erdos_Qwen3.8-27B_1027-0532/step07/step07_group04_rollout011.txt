    """One step of projected subgradient with smooth max."""
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idx = np.arange(n)
    mid = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)

    grad = np.zeros(n)
    k_vals = np.arange(2 * n - 1) - mid
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-15:
            continue
        k = k_vals[j]
        mask1 = (idx - k >= 0) & (idx - k < n)
        mask2 = (idx + k >= 0) & (idx + k < n)
        grad[mask1] += w * (1.0 - h[idx[mask1] - k])
        grad[mask2] -= w * h[idx[mask2] + k]
    grad -= np.mean(grad)

    h_new = h - lr * grad
    h_new = project_feasible(h_new, target)
    return h_new

def smooth_cd_optimize(h_init, n, target, time_limit, max_iter, alpha_start, alpha_end,
                       lr_start, lr_end):
    dx = 2.0 / n
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        progress = min(it / max_iter, 1.0)
        alpha_cur = alpha_start * (alpha_end / alpha_start) ** progress
        lr_cur = lr_start * (lr_end / lr_start) ** progress

        h = smooth_cd_step(h, n, target, alpha_cur, lr_cur, 50)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def restart_and_refine(h_best, n, target, rng, budget, start_time):
    """Try several random perturbations and keep the best."""
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    n_retries = 8
    for r in range(n_retries):
        if time.time() - start_time > budget:
            break
        scale = 0.02 + 0.01 * r
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pert.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with moderate alpha on initial grid
    tl = tleft()
    if tl > 100:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [100, 500, 2000],
                                   500, 1e-13,
                                   time_limit=min(70, tl * 0.5),
                                   start_time=start_time, topk=30)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Long CD at initial resolution
    tl = tleft()
    if tl > 150:
        h_cd, c5_cd = cd_optimize(best_h, n0, target0,
                                  time_limit=min(350, tl * 0.6),
                                  max_sweeps=80000,
                                  n_cand_start=35, n_cand_end=100,
                                  max_no_improve=400,
                                  n_active=5, weight_scale=4.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP with higher alpha
    tl = tleft()
    if tl > 80:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [5000, 20000],
                                   800, 1e-14,
                                   time_limit=min(60, tl * 0.5),
                                   start_time=start_time, topk=20)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Second CD round
    tl = tleft()
    if tl > 100:
        h_cd, c5_cd = cd_optimize(best_h, n0, target0,
                                  time_limit=min(250, tl * 0.6),
                                  max_sweeps=60000,
                                  n_cand_start=50, n_cand_end=110,
                                  max_no_improve=300,
                                  n_active=6, weight_scale=3.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Smooth CD (subgradient method)
    tl = tleft()
    if tl > 60:
        h_sg, c5_sg = smooth_cd_optimize(best_h, n0, target0,
                                         time_limit=min(50, tl * 0.5),
                                         max_iter=5000,
                                         alpha_start=50.0, alpha_end=500.0,
                                         lr_start=0.005, lr_end=0.0005)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: Final SLSQP polish
    tl = tleft()
    if tl > 50:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [10000, 50000, 100000],
                                   500, 1e-14,
                                   time_limit=min(40, tl * 0.5),
                                   start_time=start_time, topk=15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: Try higher resolution n=500
    tl = tleft()
    n_cur = n0
    target_cur = target0
    if tl > 200:
        n1 = 500
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_cd1, c5_cd1 = cd_optimize(h1, n1, target1,
                                    time_limit=min(120, tl * 0.5),
                                    max_sweeps=40000,
                                    n_cand_start=35, n_cand_end=90,
                                    max_no_improve=400,
                                    n_active=5, weight_scale=4.0)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1
            n_cur = n1
            target_cur = target1
        tl = tleft()
        if tl > 60:
            h_pol1, sc1 = slsqp_optimize(best_h, n_cur, target_cur,
                                         [5000, 20000],
                                         600, 1e-14,
                                         time_limit=min(50, tl * 0.5),
                                         start_time=start_time, topk=20)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1

    # Phase 8: Try n=600 if time permits
    tl = tleft()
    if tl > 200 and n_cur < 600:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_cd2, c5_cd2 = cd_optimize(h2, n2, target2,
                                    time_limit=min(120, tl * 0.5),
                                    max_sweeps=30000,
                                    n_cand_start=40, n_cand_end=100,
                                    max_no_improve=350,
                                    n_active=5, weight_scale=4.0)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
            n_cur = n2
            target_cur = target2
        tl = tleft()
        if tl > 50:
            h_pol2, sc2 = slsqp_optimize(best_h, n_cur, target_cur,
                                         [10000, 30000],
                                         400, 1e-14,
                                         time_limit=min(40, tl * 0.5),
                                         start_time=start_time, topk=15)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

    # Phase 9: Final CD sweep
    tl = tleft()
    if tl > 30:
        h_cf, c5_cf = cd_optimize(best_h, n_cur, target_cur,
                                  time_limit=min(25, tl * 0.6),
                                  max_sweeps=10000,
                                  n_cand_start=60, n_cand_end=110,
                                  max_no_improve=150,
                                  n_active=4, weight_scale=5.0)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    # Phase 10: Final SLSQP
    tl = tleft()
    if tl > 25:
        h_fp, sc_fp = slsqp_optimize(best_h, n_cur, target_cur,
                                     [50000, 100000],
                                     300, 1e-14,
                                     time_limit=min(20, tl * 0.6),
                                     start_time=start_time, topk=10)
        if sc_fp < best_score:
            best_score = sc_fp
            best_h = h_fp

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
