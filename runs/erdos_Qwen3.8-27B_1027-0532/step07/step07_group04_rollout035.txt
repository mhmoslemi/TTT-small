    dx = 2.0 / n
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_val = compute_c5(h)
    
    if lr is None:
        lr = dx  # initial learning rate
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        exp_vals = np.exp(alpha * (corr - max_c))
        weights = exp_vals / exp_vals.sum()
        active = np.where(weights > 1e-10)[0]
        
        # Compute gradient
        grad = np.zeros(n)
        for k_idx in active:
            w = weights[k_idx]
            k = k_idx - (n - 1)
            # ∂g_k/∂h[p] = dx * [(1-h[p-k])*I(0≤p-k<n) - h[p+k]*I(0≤p+k<n)]
            # First part: (1-h[p-k])*I(0≤p-k<n)
            # p-k ranges from 0 to n-1, so p ranges from k to k+n-1, intersected with [0,n-1]
            lo1 = max(0, k)
            hi1 = min(n, n + k)
            if lo1 < hi1:
                grad[lo1:hi1] += w * dx * (1.0 - h[lo1-k:hi1-k])
            # Second part: -h[p+k]*I(0≤p+k<n)
            lo2 = max(0, -k)
            hi2 = min(n, n - k)
            if lo2 < hi2:
                grad[lo2:hi2] -= w * dx * h[lo2+k:hi2+k]
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        
        # Check improvement
        new_val = compute_c5(h_new)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        
        # Adaptive learning rate
        if new_val < compute_c5(h):
            lr = min(lr * 1.1, 1.0)  # increase
        else:
            lr = lr * 0.5  # decrease
        
        h = h_new
    
    return best_h, best_val

            h = np.clip(h + diff / n, 0.0, 1.0)
            break

        h = np.clip(h, 0.0, 1.0)

def _smooth_obj_grad(x, alpha):

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14,
                time_limit=None, start_time=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            def obj_wrapped(x):
                return _smooth_obj_grad(x, alpha)[0]
            def jac_wrapped(x):
                return _smooth_obj_grad(x, alpha)[1]
            res = minimize(obj_wrapped, h_cur, jac=jac_wrapped,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    change[p, neg_mask] -= h[p - shifts[neg_mask]]

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible_fast(h_out, target_dst)
