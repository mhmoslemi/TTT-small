    dx = 2.0 / n
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    idxs = np.arange(n)
    return corr, dx, mid, idxs

    """Compute gradient of f_k(h) w.r.t. h."""
    grad = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    grad[v1] += (1.0 - h[idxs[v1] - k])
    grad[v2] -= h[idxs[v2] + k]
    return grad

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, alphas, maxiter):

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_eq_jac(x):
    return np.ones(_g_n)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, alphas, maxiter=500):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def pair_cd_v2(h_init, n, target, time_limit, max_sweeps=100000,
               n_cand_start=40, n_cand_end=150, n_shifts=5,
               max_no_improve=800, anneal_temp=0.0, anneal_decay=0.9998):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        # Select top shifts with exponential decay weights
        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = np.sum(shift_weights)
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        # Compute weighted gradient
        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Adaptive candidate set size
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):  # Reduced from 16
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active_mask = vals >= f_mid - 1e-13
                    if active_mask.any():
                        f_prime = float(np.max(g[active_mask]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-11:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - best_trial_c5) * 2000.0
                    if np.random.random() < np.exp(-max(0, delta_obj) / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 3:
                for p in range(n):
                    change[p, :] = 0.0
                    pos_mask = (p + shifts >= 0) & (p + shifts < n)
                    neg_mask = (p - shifts >= 0) & (p - shifts < n)
                    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                    change[p, neg_mask] -= h[p - shifts[neg_mask]]
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.98, 1e-10)

    return best_h, best_c5

**Proximal Gradient with Adaptive Step Size**

The objective F(h) = max_k f_k(h) is not smooth (it's a max of differentiable functions). We can use the proximal gradient method for the non-smooth part:

1. Compute the "active" set A = {k : f_k(h) >= F(h) - epsilon}
2. Compute the smoothed objective: F_smooth(h) = (1/|A|) * sum_{k in A} f_k(h) (average over active set)
3. This is a quadratic (smooth) function!
4. Take a gradient step: h_new = h - eta * grad(F_smooth)
5. Project onto the feasible set
6. Update the active set

The key innovation: by averaging over the active set, we get a smooth quadratic objective that we can optimize with Newton's method (since it's quadratic, one Newton step gives the exact minimizer of the quadratic approximation).

Wait, that's interesting! If we fix the active set A, then:
F_A(h) = (1/|A|) * sum_{k in A} f_k(h)

This is a quadratic function of h (since each f_k is quadratic). Its Hessian is:
H = (1/|A|) * sum_{k in A} H_k

where H_k is the Hessian of f_k (which we computed earlier as a sparse matrix).

If H is positive definite (or we work with the unconstrained minimizer and then project), we can solve:
min F_A(h) s.t. 0 <= h <= 1, sum(h) = n/2

This is a QP! And since it's quadratic with a known Hessian, we can solve it efficiently.

The algorithm:
1. Start with initial h
2. Find active set A = {k : f_k(h) within epsilon of max}
3. Solve the QP: min (1/2) h^T H h + c^T h s.t. constraints
4. Update h
5. Update active set
6. Repeat until convergence

The QP in step 3 can be solved with:
- scipy.optimize (if we reformulate)
- A custom projected gradient / Newton method
- cvxpy (if the QP is convex)

For the QP to be convex, we need H to be positive definite. Recall:
H_k has eigenvalues -2*cos(2*pi*m*k/n)
H = (1/|A|) * sum_{k in A} H_k has eigenvalues:
lambda_m = (1/|A|) * sum_{k in A} (-2*cos(2*pi*m*k/n))

For H to be PSD, we need sum_{k in A} cos(2*pi*m*k/n) <= 0 for all m.

This is NOT guaranteed. In fact, for m=0: sum_{k in A} cos(0) = |A| > 0, so lambda_0 > 0. Good.
For m=1: sum_{k in A} cos(2*pi*k/n) could be positive or negative depending on A.

So the QP might not be convex. But we can still solve it as a non-convex QP using:

- A "basin hopping" approach (try multiple starting points)

2. Take a step in the negative gradient direction
3. Project onto feasible set
4. If the max increases, halve the step (backtracking)
5. Update h

This is just gradient descent on the averaged active objective, which is what the previous "subgradient_multi" does! The difference is that we use the exact gradient of the average (which is quadratic) rather than just the subgradient of the max.

**New idea: Active set QP with penalty method**

Instead of projecting onto the feasible set, use a penalty method:
min F_A(h) + (mu/2) * (sum(h) - n/2)^2 + penalty_for_bounds(h)

This turns it into an unconstrained problem that we can solve with Newton's method (since F_A is quadratic, the total objective is still quadratic + penalty terms).

For the bound penalties, we can use: sum_i max(0, h_i - 1)^2 + max(0, -h_i)^2

The resulting objective is a piecewise-quadratic function. We can solve it with:
1. Start with current h
2. Determine which bounds are active (h_i at 0 or 1)
3. For the free variables, solve a linear system (Newton step for the quadratic)
4. Check if any variable violates bounds
5. If so, fix it and go to step 2
6. This is the "active set method" for QPs

This is O(n^3) per iteration (for the linear system) but typically converges in O(n) iterations, so total O(n^4). For n=400, that's 400^4 = 2.6 * 10^10 operations, which is too slow.

We need a faster approach. Since the Hessian H has a special structure (it's a sum of circulant-like matrices), we can solve the linear system using the FFT!

The Hessian H = (1/|A|) * sum_{k in A} H_k where H_k[j, j'] = -(delta[j'-j, k] + delta[j'-j, -k])

So H is a "banded" matrix (bandwidth max(|A|) * max_k). Actually, it's a "structured" matrix where H[j, j'] depends only on (j' - j) mod n (approximately, ignoring boundary effects).

This means H is approximately circulant, and we can solve H * x = b using the FFT in O(n log n)!

This is a huge speedup. Let me implement this:

1. Build H as a "convolution kernel" h_kernel where H[j, j'] = h_kernel[(j' - j) mod n] (approximately)
2. Solve H * x = b using: x = IFFT(FFT(b) / FFT(h_kernel))
3. Handle the boundary effects by using the actual (non-circular) matrix for the few rows near the boundary

Actually, the boundary effects make this more complex. The matrix H is NOT exactly circulant because the correlation is non-circular (it uses the "full" mode, not "wrap" mode).

For the "full" correlation, the valid range of shifts is -(n-1) to (n-1), and the autocorrelation at shift k is:
c[k] = sum_{i: 0<=i<n, 0<=i-k<n} h[i] * h[i-k]

This is NOT a circular correlation, so the Hessian is NOT circulant. It's a "banded" Toeplitz-like matrix.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_eq_jac(x):
    return np.ones(_g_n)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, alphas, maxiter=500):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def pair_cd_improved(h_init, n, target, time_limit,
                     max_sweeps=100000, n_cand_start=40, n_cand_end=160,
                     n_shifts=5, max_no_improve=1200,
                     anneal_temp=0.03, anneal_decay=0.9999,
                     reheat_interval=300, reheat_temp=0.05):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            # Reheat
            temp = reheat_temp
            consecutive_no_improve = 0
            continue

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        # Select top shifts
        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = np.sum(shift_weights)
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        # Compute weighted gradient
        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Adaptive candidate set
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])

                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)

                        f_prime = float(np.max(g[active_mask]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-11:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - best_trial_c5) * 3000.0
                    if delta_obj < 50:  # cap to avoid overflow
                        if np.random.random() < np.exp(-max(0.0, delta_obj) / temp):
                            accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 3:
                for p in range(n):
                    change[p, :] = 0.0
                    pos_mask = (p + shifts >= 0) & (p + shifts < n)
                    neg_mask = (p - shifts >= 0) & (p - shifts < n)
                    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                    change[p, neg_mask] -= h[p - shifts[neg_mask]]
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.97, 1e-10)

    return best_h, best_c5
