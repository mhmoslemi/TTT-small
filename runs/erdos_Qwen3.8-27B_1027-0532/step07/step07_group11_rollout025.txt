import cvxpy as cp

def slp_step_cvxpy(h, n, target):
    mid = n - 1
    idxs = np.arange(n)
    
    # Compute g_k and grad_k for all k
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    grads = np.zeros((n_corr, n))
    for k_idx in range(n_corr):
        k = k_idx - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grads[k_idx, v1] += (1.0 - h[idxs[v1] + k])
        grads[k_idx, v2] -= h[idxs[v2] - k]
    
    # LP: min t
    # s.t. t >= corr[k] + grads[k] @ (x - h) for all k
    #      0 <= x <= 1
    #      sum(x) = target
    x = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [x >= 0, x <= 1, cp.sum(x) == target]
    for k_idx in range(n_corr):
        # t >= corr[k] + grads[k] @ (x - h)
        # t >= corr[k] - grads[k] @ h + grads[k] @ x
        # t - grads[k] @ x >= corr[k] - grads[k] @ h
        rhs = corr[k_idx] - grads[k_idx] @ h
        constraints.append(t - grads[k_idx] @ x >= rhs)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS)  # or GLOP, CBC, etc.
    
    if x.value is not None:
        return np.array(x.value).flatten()
    return h

def slp_step_restricted(h, n, target, top_k=20):
    mid = n - 1
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Get top k shifts
    top_idx = np.argsort(corr)[-top_k:]
    
    grads = np.zeros((len(top_idx), n))
    for ii, k_idx in enumerate(top_idx):
        k = int(k_idx) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grads[ii, v1] += (1.0 - h[idxs[v1] + k])
        grads[ii, v2] -= h[idxs[v2] - k]
    
    # LP: min t
    # s.t. t >= corr[k] + grads_k @ (x - h) for top k
    #      0 <= x <= 1
    #      sum(x) = target
    
    # Using scipy linprog (faster for LPs)
    # min c^T y where y = [t, x_0, ..., x_{n-1}]
    # c = [1, 0, ..., 0]
    
    m = len(top_idx)
    # A_ub * y <= b_ub
    # For each constraint: -t + grads_k @ x <= -corr[k] + grads_k @ h
    A_ub = np.zeros((m + 2*n, 1 + n))
    b_ub = np.zeros(m + 2*n)
    
    for ii, k_idx in enumerate(top_idx):
        A_ub[ii, 0] = -1.0
        A_ub[ii, 1:] = grads[ii]
        b_ub[ii] = -corr[int(k_idx)] + grads[ii] @ h
    
    # x_i <= 1: A_ub row has 1 in column i+1
    for i in range(n):
        A_ub[m + i, i + 1] = 1.0
        b_ub[m + i] = 1.0
        # x_i >= 0: -x_i <= 0
        A_ub[m + n + i, i + 1] = -1.0
        b_ub[m + n + i] = 0.0
    
    # A_eq * y = b_eq: sum(x) = target
    A_eq = np.zeros((1, 1 + n))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([target])
    
    c = np.zeros(1 + n)
    c[0] = 1.0
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=[(None, None)] + [(0, 1)]*n, method='highs')
    
    if res.success:
        x_new = res.x[1:]
        return project_feasible(x_new, target)
    return h

def compute_grads_top(h, n, top_k):

def compute_change_matrix(h, n):

        change[p, neg_mask] -= h[p - shifts[neg_mask]]

def _slsqp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted))) / alpha

def _slsqp_cons(x):
