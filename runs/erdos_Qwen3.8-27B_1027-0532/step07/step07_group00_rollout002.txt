   Let me think about the SLSQP formulation more carefully. The issue with SLSQP is that it needs smooth objectives. The max function is not smooth. Options:
   1. Log-sum-exp (previous approach)
   2. Power mean: (sum corr_k^p)^(1/p) for large p → approximates max
   3. Clipped objective: min(max_corr, T) + λ * sum(max(0, corr_k - T)) for some T near the optimum

   Option 3 is interesting because it's piecewise linear and might work well with SLSQP.

   Actually, for SLSQP, I think the best approach is:
   - Use the log-sum-exp with a very large alpha (like 10000+)
   - This gives a very tight approximation to the max
   - The gradient is well-defined and points in the right direction

   The issue with very large alpha is numerical instability. Let me use a shifted log-sum-exp:
   f(h) = (1/α) * log(sum_k exp(α * (corr_k - M))) + M
   where M = max(corr_k). This is numerically stable.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _lse_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-15:
            continue
        j_val = m
        if 0 <= j_val < n:
            grad[j_val] += w * (1.0 - 2.0 * h_helper(x, j_val))
    return grad


def h_helper(x, j):
    return x[j]


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-13):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_lse_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_vectors(h, n, indices):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    changes = {}
    for p in indices:
        g = np.zeros(n_corr)
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        g[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        g[neg_mask] -= h[p - shifts[neg_mask]]
        changes[p] = g
    return changes


def tabu_cd(h_init, n, target, time_limit, max_sweeps=20000,
            n_cand=40, tabu_len=30, explore_prob=0.0, explore_tol=0.0002,
            weight_beta=4.0):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(999)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    changes_cache = {}

    def get_change(p):
        if p not in changes_cache:
            g = np.zeros(n_corr)
            pos_mask = (p + shifts >= 0) & (p + shifts < n)
            neg_mask = (p - shifts >= 0) & (p - shifts < n)
            g[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
            g[neg_mask] -= h[p - shifts[neg_mask]]
            changes_cache[p] = g
        return changes_cache[p]

    def invalidate(p):
        if p in changes_cache:
            del changes_cache[p]

    tabu_list = []

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 500:
            break

        c_max = float(np.max(corr))
        top_k = min(15, n_corr)
        top_positions = np.argsort(corr)[-top_k:]

        grad = np.zeros(n)
        for s_idx in top_positions:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        acceptor_set = sorted_idx[:n_cand]
        donor_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_move = None

        n_a = len(acceptor_set)
        n_d = len(donor_set)

        for ii in range(n_a):
            i = int(acceptor_set[ii])
            if i in tabu_list:
                continue
            g_i = get_change(i)
            for jj in range(n_d):
                j = int(donor_set[jj])
                if i == j or j in tabu_list:
                    continue
                delta_lo = max(-h[i], h[j] - 1.0)
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = g_i - get_change(j)
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx

                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_move = (i, j, delta_star)
                elif explore_prob > 0 and approx_max < c5_val + explore_tol:
                    if rng.random() < explore_prob:
                        best_trial = approx_max
                        best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, delta = best_move
            h[i] += delta
            h[j] -= delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            invalidate(i)
            invalidate(j)
            tabu_list.append((i, j))
            tabu_list.append((j, i))
            if len(tabu_list) > 2 * tabu_len:
                removed = tabu_list[:len(tabu_list) - 2 * tabu_len]
                tabu_list = tabu_list[len(tabu_list) - 2 * tabu_len:]
        else:
            if explore_prob > 0 and rng.random() < 0.01:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i != j and h[i] > 0.05 and h[j] < 0.95:
                    delta = (h[i] - h[j]) * 0.1 * (1 if h[i] > h[j] else -1)
                    if abs(delta) > 1e-6:
                        h[i] += delta
                        h[j] -= delta
                        h = np.clip(h, 0.0, 1.0)
                        h = project_feasible(h, target)
                        invalidate(i)
                        invalidate(j)

    return best_h, best_c5


def power_mean_slsqp(h_start, n, target, time_limit):
    start = time.time()
    p_vals = [20, 50, 100, 200]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    global _g_n, _g_target
    _g_n = n
    _g_target = target

    for p in p_vals:
        if time.time() - start > time_limit - 20:
            break
        try:
            res = minimize(_lse_obj, h_cur, args=(p * 50,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-13})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def structured_perturb(h, n, target, rng, method='shift'):
    if method == 'shift':
        amt = rng.randint(1, 4)
        h_pert = np.roll(h, amt)
    elif method == 'block_swap':
        blen = rng.randint(2, 6)
        i = rng.randint(0, n - 2 * blen)
        j = i + blen + rng.randint(0, n - 2 * blen - blen)
        if j + blen > n:
            j = rng.randint(0, n - blen)
            if abs(i - j) < blen:
                j = (i + blen + rng.randint(0, n - 2 * blen)) % n
        tmp = h[i:i+blen].copy()
        h_pert = h.copy()
        h_pert[i:i+blen] = h[j:j+blen]
        h_pert[j:j+blen] = tmp
    elif method == 'sinusoidal':
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.05
        x = np.linspace(0, 2, n)
        h_pert = h + amp * np.sin(2 * np.pi * freq * x + phase)
    else:
        h_pert = h + rng.randn(n) * 0.02
    return project_feasible(h_pert, target)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
