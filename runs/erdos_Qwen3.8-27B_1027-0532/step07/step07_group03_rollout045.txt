        if abs(diff) < 1e-14:

        else:

        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

def _smooth_obj(x, alpha):
    corr = compute_corr(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _cons_fun, "jac": _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:

                           method="SLSQP", bounds=bounds, constraints=cons,

            no_improve += 1

                    cuts_f = cuts_f[-10:]
                    cuts_h = cuts_h[-10:]
                no_improve = 0

        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])

        n_active = len(active_lags)

        idxs = np.arange(n)
        if n_active == 1:
            k = k_star - mid
            g = np.zeros(n)
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            g[v1] += (1.0 - h[idxs[v1] + k])
            g[v2] -= h[idxs[v2] - k]
        else:
            w = np.ones(n_active) / n_active
            g = np.zeros(n)
            for lag, wi in zip(active_lags, w):
                k = int(lag) - mid
                v1 = (idxs + k >= 0) & (idxs + k < n)
                v2 = (idxs - k >= 0) & (idxs - k < n)
                g[v1] += wi * (1.0 - h[idxs[v1] + k])
                g[v2] -= wi * h[idxs[v2] - k]

        cuts_f.append(f_val)
        cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:

            cuts_h = [cuts_h[0]] + cuts_h[-keep:]

            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += w * (1.0 - y[idxs[v1] + k])
            grad[v2] -= w * y[idxs[v2] - k]

        eta = eta_init * (eta_end / eta_init) ** (it / max_iters)
        h_new = y - eta * grad
        h_new = project_feasible(h_new, target)

        sc = compute_c5(h_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_new.copy()

        h_prev = h_curr.copy()
        h_curr = h_new
        mu = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0

    return best_h, best_score


def structured_perturbation(h_base, n, target, freqs, amplitudes, phase=0.0):
    x = np.linspace(0.0, 2.0, n)
    pert = np.zeros(n)
    for f, a in zip(freqs, amplitudes):
        pert += a * np.sin(2.0 * np.pi * f * x + phase)
    h_new = h_base + pert
    return project_feasible(h_new, target)
