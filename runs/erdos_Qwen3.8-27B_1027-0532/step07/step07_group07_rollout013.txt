        temp = max(top_vals[-1] - top_vals[0], 1e-10)
        weights = np.exp((top_vals - top_vals[-1]) / temp)
        weights /= weights.sum()

            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)

    global _g_target, _g_alpha, _g_top_k
    start_time = time.time()

    def _tl():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP polish at original resolution with top-K smoothing
    if _tl() > 80:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                     alphas=[300, 1000, 5000],
                                     top_ks=[3, 7, 15],
                                     maxiter=800, ftol=1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Higher resolution n=800
    if _tl() > 400:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

        # Leveling descent
        if _tl() > 180:
            h_lv, sc_lv = leveling_descent(h1, n1, target1,
                                           time_limit=min(140, _tl() * 0.45),
                                           top_k=7, lr_start=0.02, lr_end=0.002)
            if sc_lv < best_score:
                best_score = sc_lv
                best_h = h_lv

        # Pair CD with improved parameters
        if _tl() > 180:
            h_cd, sc_cd = pair_cd(best_h, n1, target1,
                                   time_limit=min(160, _tl() * 0.55),
                                   max_sweeps=50000,
                                   n_cand_start=30, n_cand_end=90,
                                   top_k_eval=70, max_no_improve=1200,
                                   n_shifts_grad=5, soft_temp=0.25)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd

        # SLSQP polish
        if _tl() > 70:
            h_pol2, sc2 = slsqp_optimize(best_h, n1, target1,
                                          alphas=[2000, 8000, 30000],
                                          top_ks=[3, 7],
                                          maxiter=1000, ftol=1e-15)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

        n0 = n1
        target0 = target1

    # Phase 3: Even higher resolution n=1200
    if _tl() > 350:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            n0 = n2
            target0 = target2

        if _tl() > 200:
            h_cd2, sc_cd2 = pair_cd(best_h, n0, target0,
                                     time_limit=min(160, _tl() * 0.6),
                                     max_sweeps=30000,
                                     n_cand_start=25, n_cand_end=80,
                                     top_k_eval=55, max_no_improve=800,
                                     n_shifts_grad=4, soft_temp=0.3)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_h = h_cd2

        if _tl() > 60:
            h_pol3, sc3 = slsqp_optimize(best_h, n0, target0,
                                          alphas=[5000, 20000],
                                          top_ks=[3, 7],
                                          maxiter=800, ftol=1e-15)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    # Phase 4: Multi-restart exploration
    if _tl() > 150:
        h_mr, sc_mr = multi_restart_cd(best_h, n0, target0,
                                        time_limit=min(120, _tl() * 0.7),
                                        seed=seed + 1,
                                        noise_scales=(0.02, 0.05),
                                        cd_time_frac=0.45)
        if sc_mr < best_score:
            best_score = sc_mr
            best_h = h_mr

    # Phase 5: Final pair CD
    if _tl() > 80:
        h_cd3, sc_cd3 = pair_cd(best_h, n0, target0,
                                 time_limit=min(60, _tl() * 0.7),
                                 max_sweeps=25000,
                                 n_cand_start=40, n_cand_end=100,
                                 top_k_eval=80, max_no_improve=500,
                                 n_shifts_grad=5, soft_temp=0.2)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    # Phase 6: Final SLSQP polish
    if _tl() > 40:
        h_fin, sc_fin = slsqp_optimize(best_h, n0, target0,
                                        alphas=[10000, 50000],
                                        top_ks=[3, 5],
                                        maxiter=500, ftol=1e-16)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    if final_score <= best_score:
        return final_h, final_score, len(final_h)
    return best_h, best_score, len(best_h)
