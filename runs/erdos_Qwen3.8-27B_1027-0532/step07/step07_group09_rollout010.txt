    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=100, max_no_improve=600):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_active = max(3, int(10 - 7 * progress))
        n_active = min(n_active, n_corr)

        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = w.sum()
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            h_i = h[i]
            ci = change[i]
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue

                g = change[j] - ci

                a_d = delta_min
                b_d = delta_max
                for _ in range(9):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def nesterov_gradient(h_init, n, target, time_limit, alpha=500.0, max_iter=50000,
                      lr_start=0.05):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    h_prev = h.copy()
    t = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = h + (t - 1.0) / t_new * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr)
        weights = np.exp(np.clip(alpha * (corr - c_max), -50.0, 0.0))
        wsum = weights.sum()
        if wsum > 0:
            weights /= wsum

        grad = np.zeros(n)
        active_idx = np.where(weights > 1e-15)[0]
        for si in active_idx:
            k = int(si) - mid
            wi = weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - y[idxs[v1] - k])
            grad[v2] -= wi * y[idxs[v2] + k]
        grad -= np.mean(grad)

        lr = lr_start / (1.0 + 0.01 * it)
        h_new = project_feasible(y - lr * grad, target)

        cur_score = compute_c5(h_new)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        h_prev = h
        h = h_new
        t = t_new

    return best_h, best_score

def block_cd(h_init, n, target, time_limit, max_steps=20000, block_sizes=(2, 3, 4, 5)):
    start = time.time()
    rng = np.random.RandomState(12345)
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * (2.0 / n) < best_c5 - 1e-14:
            best_c5 = c5_val * (2.0 / n)
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        bs = block_sizes[step % len(block_sizes)]
        n_pos = n - bs + 1
        if n_pos <= 1:
            continue

        src_start = rng.randint(0, n_pos)
        block_grad_src = grad[src_start:src_start + bs].sum()

        n_search = min(80, n_pos)
        offsets = rng.permutation(n_pos)[:n_search]
        best_dst = -1
        best_dst_grad = -np.inf
        for sp in offsets:
            if sp == src_start:
                continue
            bg = grad[sp:sp + bs].sum()
            if bg > best_dst_grad:
                best_dst_grad = bg
                best_dst = int(sp)

        if best_dst < 0:
            continue

        if block_grad_src > best_dst_grad:
            src_s, dst_s = src_start, best_dst
        else:
            src_s, dst_s = best_dst, src_start

        src_block = h[src_s:src_s + bs]
        dst_block = h[dst_s:dst_s + bs]

        delta_max = min(src_block.max(), 1.0 - dst_block.min())
        delta_min = max(-src_block.min(), dst_block.min() - 1.0)
        if delta_max - delta_min < 1e-12:
            continue

        g = np.zeros(n_corr)
        for t_idx in range(bs):
            g += change[dst_s + t_idx, :] - change[src_s + t_idx, :]

        a_d = delta_min
        b_d = delta_max
        for _ in range(9):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr + mid_d * g
            f_mid = np.max(vals)
            active_mask = vals >= f_mid - 1e-14
            if active_mask.any():
                f_prime = np.max(g[active_mask])
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                b_d = mid_d
            else:
                a_d = mid_d

        delta_star = 0.5 * (a_d + b_d)
        if abs(delta_star) < 1e-12:
            continue

        approx_max = np.max(corr + delta_star * g)
        if approx_max < c5_val - 1e-14:
            h[src_s:src_s + bs] -= delta_star
            h[dst_s:dst_s + bs] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for t_idx in range(bs):
                update_change_row(change, src_s + t_idx, h, n, shifts)
                update_change_row(change, dst_s + t_idx, h, n, shifts)

    return best_h, best_c5

def multi_start_perturb(h_best, n, target, time_limit, rng, n_tries=8):
    best_h = h_best.copy()
    best_score = compute_c5(h_best)
    per_try = time_limit / n_tries

    for trial in range(n_tries):
        if time.time() - (time.time() - time_limit) > time_limit:
            break
        noise_scale = 0.03 + 0.04 * (trial % 4) * 0.5
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)

        h_cd, c5_cd = pair_cd(h_pert, n, target,
                              time_limit=min(20, per_try * 0.7),
                              max_sweeps=8000,
                              n_cand_start=25, n_cand_end=60,
                              max_no_improve=300)
        sc = compute_c5(h_cd)
        if sc < best_score:
            best_score = sc
            best_h = h_cd.copy()

    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)

    if tleft() > 150:

        if c5_nc < best_score:

                                max_no_improve=400)

    if tleft() > 200:

                                max_sweeps=25000,
                                n_cand_start=25, n_cand_end=70,
                                max_no_improve=500)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 80:
        h_pol = slsqp_optimize(best_h, cur_n, cur_target,

                               600, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:

            best_h = h_pol

    if tleft() > 150:
        h_cd4, c5_cd4 = pair_cd(best_h, cur_n, cur_target,
                                time_limit=min(120, tleft() * 0.55),
                                max_sweeps=20000,
                                n_cand_start=30, n_cand_end=80,

        if c5_cd4 < best_score:
