    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_fun(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _obj_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    idxs = np.arange(n)

        wi = w[ki]
        if wi < 1e-14:
            continue

        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v2] -= wi * x[idxs[v2] - k]

def do_pair_cd(h0, n, target, tlimit, tstart, ncand=30, bisect_n=14, max_sweep=50000):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    shifts = np.arange(-(n - 1), n)

    h = h0.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweep):
        if time.time() - tstart > tlimit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800:
                break

        c_max = np.max(corr)
        alpha_g = 80.0
        shifted_g = np.clip(alpha_g * (corr - c_max), -500.0, 0.0)
        es_g = np.exp(shifted_g)
        w = es_g / np.sum(es_g)

        grad = np.zeros(n)
        for ki in range(n_corr):
            wi = w[ki]
            if wi < 1e-15:
                continue
            k = ki - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] + k])
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v2] -= wi * h[idxs[v2] - k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                lin_est = grad[j] - grad[i]
                if lin_est >= -1e-10:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                pos_i = (i + shifts >= 0) & (i + shifts < n)
                neg_i = (i - shifts >= 0) & (i - shifts < n)
                g[pos_i] -= (1.0 - h[i + shifts[pos_i]])
                g[neg_i] += h[i - shifts[neg_i]]
                pos_j = (j + shifts >= 0) & (j + shifts < n)
                neg_j = (j - shifts >= 0) & (j - shifts < n)
                g[pos_j] += (1.0 - h[j + shifts[pos_j]])
                g[neg_j] -= h[j - shifts[neg_j]]

                a_d = delta_lo
                b_d = delta_hi
                for _b in range(bisect_n):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                improve = c5_val - approx_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def do_block_opt(h0, n, target, tlimit, tstart, bsize=6, step=4, alpha_b=200):
    dx = 2.0 / n
    h = h0.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    shifts = np.arange(-(n - 1), n)

    for sweep in range(200):

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

                break

        alpha_g = 80.0
        shifted_g = np.clip(alpha_g * (corr - c_max), -500.0, 0.0)
        es_g = np.exp(shifted_g)
        w = es_g / np.sum(es_g)

        grad = np.zeros(n)
        for ki in range(n_corr):

                continue
            k = ki - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] + k])
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v2] -= wi * h[idxs[v2] - k]

        for start in range(0, n - bsize + 1, step):
            if time.time() - tstart > tlimit:
                break
            block_idx = np.arange(start, start + bsize)
            h_block = h[block_idx].copy()
            target_block = target - np.sum(h) + np.sum(h_block)

            bnds_b = [(0.0, 1.0)] * bsize
            cons_b = [{'type': 'eq', 'fun': lambda x, tb=target_block: np.sum(x) - tb,
                       'jac': lambda x, tb=target_block: np.ones(bsize)}]

            def block_obj(xb):
                h_t = h.copy()
                h_t[block_idx] = np.clip(xb, 0.0, 1.0)
                h_t = project_feasible(h_t, target)
                return _sf(h_t, alpha_b)

            def block_grad(xb):
                h_t = h.copy()
                h_t[block_idx] = np.clip(xb, 0.0, 1.0)
                return _sg(h_t, alpha_b)[block_idx]

            try:
                res = minimize(block_obj, h_block, method='SLSQP',
                               bounds=bnds_b, constraints=cons_b,
                               jac=block_grad,
                               options={'maxiter': 80, 'ftol': 1e-16, 'iprint': 0})
                h_new = h.copy()
                h_new[block_idx] = np.clip(res.x, 0.0, 1.0)
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < c5_val - 1e-15:
                    h = h_new
            except Exception:
                pass

    return best_h, best_c5

def make_binary_start(n, pattern_seed=1):
    rng = np.random.RandomState(pattern_seed)
    h = np.zeros(n)
    ones_needed = n // 2
    perm = rng.permutation(n)
    h[perm[:ones_needed]] = 1.0
    return project_feasible(h, n / 2.0)

def make_periodic_start(n, period=5):
    h = np.zeros(n)
    pattern = np.zeros(period)
    pattern[:period // 2 + 1] = 1.0
    h = np.tile(pattern, n // period + 1)[:n]
    return project_feasible(h, n / 2.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
