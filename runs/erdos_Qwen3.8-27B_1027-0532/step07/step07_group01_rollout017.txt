def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):

def pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Scale for numerical stability
    c_max = np.max(corr)
    if c_max < 1e-15:
        return 0.0
    scaled = corr / c_max
    return c_max * np.sum(np.abs(scaled) ** p) ** (1.0 / p)

def compute_subgrad(h, n):
    """Compute subgradient of max_k corr[k] with respect to h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    k = k_star - (n - 1)
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    grad[v1] += (1.0 - h[idxs[v1] - k])
    grad[v2] -= h[idxs[v2] + k]
    # Average with mean zero for the constraint
    grad -= np.mean(grad)
    return grad / dx  # This gives subgradient of C5

    """Subgradient method with Barzilai-Borwein step sizes."""
    start = time.time()

    best_h = h.copy()
    best_c5 = compute_c5(h_cur)
    g_prev = None
    h_prev = None
    alpha = 0.01
    count_no_improve = 0
    
    for it in range(max_iter):

            break

        if g_prev is not None and h_prev is not None:
            d = h_cur - h_prev
            s = g - g_prev
            ds = np.dot(d, s)
            ss = np.dot(s, s)

            count_no_improve = 0
        else:

        g_prev = g.copy()
        h_cur = h_new
        
        if count_no_improve > 2000:

import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change, shifts

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=100, max_no_improve=500,
            restart_every=200, restart_scale=0.015, n_shifts=1, seed=42):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    rng = np.random.RandomState(seed)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change, shifts = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_every:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            change, shifts = compute_change_matrix(h, n)
            consecutive_no_improve = 0

        if n_shifts == 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
            grad -= np.mean(grad)
        else:
            top_shifts = np.argsort(corr)[-n_shifts:]
            top_sum = np.sum(corr[top_shifts])
            if top_sum < 1e-15:
                top_sum = 1.0
            grad = np.zeros(n)
            for s_idx in top_shifts:
                w = corr[s_idx] / top_sum
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
            grad -= np.mean(grad)

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        nh = len(high_set)
        nl = len(low_set)
        for ii in range(nh):
            i = int(high_set[ii])
            for jj in range(nl):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def subgrad_bb(h_init, n, target, time_limit, max_iter=20000):
    start = time.time()
    dx = 2.0 / n
    h_cur = h_init.copy().astype(float)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    h_prev = None
    g_prev = None
    count_no_improve = 0
    alpha = 0.005

    idxs = np.arange(n)
    mid = n - 1

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star_idx = int(np.argmax(corr))
        k = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h_cur[idxs[v1] - k])
        grad[v2] -= h_cur[idxs[v2] + k]
        grad -= np.mean(grad)
        grad /= dx  # subgradient of C5

        if h_prev is not None and g_prev is not None:
            d = h_cur - h_prev
            s = grad - g_prev
            ds = np.dot(d, s)
            ss = np.dot(s, s)
            if ss > 1e-20:
                alpha = max(1e-6, min(0.05, ds / ss))

        h_new = h_cur - alpha * grad
        h_new = project_feasible(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            count_no_improve = 0
        else:
            count_no_improve += 1

        h_prev = h_cur.copy()
        g_prev = grad.copy()
        h_cur = h_new

        if count_no_improve > 3000:
            break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)

        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_every:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            change, shifts = compute_change_matrix(h, n)
            consecutive_no_improve = 0

        if n_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
            grad -= np.mean(grad)
        else:
            top_shifts = np.argsort(corr)[-n_shifts:]
            top_sum = np.sum(corr[top_shifts])
            if top_sum < 1e-15:
                top_sum = 1.0
            grad = np.zeros(n)
            for s_idx in top_shifts:
                w = corr[s_idx] / top_sum
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
            grad -= np.mean(grad)

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        nh = len(high_set)
        nl = len(low_set)
        for ii in range(nh):
            i = int(high_set[ii])
            for jj in range(nl):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def subgrad_bb(h_init, n, target, time_limit, max_iter=20000):
    start = time.time()
    dx = 2.0 / n
    h_cur = h_init.copy().astype(float)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    h_prev = None
    g_prev = None
    count_no_improve = 0
    alpha = 0.003

    idxs = np.arange(n)
    mid = n - 1

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star_idx = int(np.argmax(corr))
        k = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h_cur[idxs[v1] - k])
        grad[v2] -= h_cur[idxs[v2] + k]
        grad -= np.mean(grad)
        grad /= dx

        if h_prev is not None and g_prev is not None:
            d = h_cur - h_prev
            s = grad - g_prev
            ds = np.dot(d, s)
            ss = np.dot(s, s)
            if ss > 1e-20:
                alpha = max(1e-7, min(0.05, ds / ss))

        h_new = h_cur - alpha * grad
        h_new = project_feasible(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            count_no_improve = 0
        else:
            count_no_improve += 1

        h_prev = h_cur.copy()
        g_prev = grad.copy()
        h_cur = h_new

        if count_no_improve > 5000:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    # Phase 1: Quick polish at original resolution
    if tleft() > 70:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000, 10000],
                                600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 50:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(45, tleft() * 0.6),
                              max_sweeps=20000,
                              n_cand_start=20, n_cand_end=70,
                              max_no_improve=400,
                              n_shifts=3, seed=111)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: Upsample to n=800 and intensive optimization
    n1 = 800
    target1 = n1 / 2.0
    current_n = n1
    current_target = target1

    if tleft() > 400:
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 120:
            h_pol, sc = slsqp_round(h1, n1, target1,
                                    [100, 500, 2000, 10000],
                                    800, 1e-15)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

        if tleft() > 200:
            h_cd, c5_cd = pair_cd(best_h, n1, target1,
                                  time_limit=min(180, tleft() * 0.5),
                                  max_sweeps=40000,
                                  n_cand_start=25, n_cand_end=90,
                                  max_no_improve=600,
                                  restart_every=250, restart_scale=0.012,
                                  n_shifts=3, seed=222)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        if tleft() > 80:
            h_pol, sc = slsqp_round(best_h, n1, target1,
                                    [5000, 20000, 100000, 500000],
                                    1200, 1e-15)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 3: Subgradient BB + multi-shift CD
    if tleft() > 250:
        h_sg, sc_sg = subgrad_bb(best_h, current_n, current_target,
                                 time_limit=min(60, tleft() * 0.2),
                                 max_iter=15000)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if tleft() > 180:
        h_cd, c5_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(140, tleft() * 0.5),
                              max_sweeps=30000,
                              n_cand_start=35, n_cand_end=110,
                              max_no_improve=500,
                              restart_every=200, restart_scale=0.01,
                              n_shifts=5, seed=333)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Random restarts for diversity
    rng = np.random.RandomState(seed)
    if tleft() > 200:
        for noise_scale in [0.015, 0.04, 0.08, 0.12]:
            if tleft() < 70:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [200, 1000, 5000, 20000],
                                        500, 1e-15)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 50:
                h_cd_p, c5_p = pair_cd(h_ref, current_n, current_target,
                                       time_limit=min(45, tleft() * 0.4),
                                       max_sweeps=12000,
                                       n_cand_start=30, n_cand_end=85,
                                       max_no_improve=300,
                                       n_shifts=3,
                                       seed=int(rng.randint(0, 99999)))
                if c5_p < best_score:
                    best_score = c5_p
                    best_h = h_cd_p

    # Phase 5: Try higher resolution if time permits
    n2 = 1200
    target2 = n2 / 2.0
    if tleft() > 350 and best_score > 0.3805:
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 200:
            h_pol2, sc2 = slsqp_round(h2, n2, target2,
                                      [200, 1000, 5000],
                                      600, 1e-15)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2
                current_n = n2
                current_target = target2
        if current_n == n2 and tleft() > 100:
            h_cd2, c5_cd2 = pair_cd(best_h, n2, target2,
                                    time_limit=min(90, tleft() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_start=25, n_cand_end=70,
                                    max_no_improve=400,
                                    n_shifts=3, seed=444)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 6: Final intensive polishing
    if tleft() > 120:
        h_final_cd, c5_final = pair_cd(best_h, current_n, current_target,
                                       time_limit=min(90, tleft() * 0.6),
                                       max_sweeps=20000,
                                       n_cand_start=40, n_cand_end=120,
                                       max_no_improve=600,
                                       restart_every=300, restart_scale=0.008,
                                       n_shifts=5, seed=555)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if tleft() > 50:
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                            [100000, 500000, 2000000],
                                            2000, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
