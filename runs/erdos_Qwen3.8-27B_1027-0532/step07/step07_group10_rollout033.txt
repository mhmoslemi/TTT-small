import numpy as np
from scipy.optimize import minimize
import time

def fft_correlate(h):
    """Compute np.correlate(h, 1-h, 'full') using FFT."""
    n = len(h)
    size = 2**int(np.ceil(np.log2(2*n)))
    one_minus_h = 1.0 - h
    # correlate(h, one_minus_h) = convolve(h, flip(one_minus_h))
    fh = np.fft.rfft(h, size)
    fm = np.fft.rfft(one_minus_h[::-1], size)
    result = np.fft.irfft(fh * fm, size)[:2*n-1]
    return result

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h)
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    """Project h onto [0,1]^n ∩ {sum = target_sum}."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def fft_corr(h):
    """Fast correlation using FFT: np.correlate(h, 1-h, 'full')"""
    n = len(h)
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()  # next power of 2
    one_m_h = 1.0 - h
    # correlate(h, one_m_h) = convolve(h, flip(one_m_h))
    Fh = np.fft.rfft(h, size)
    Fm = np.fft.rfft(one_m_h[::-1], size)
    return np.fft.irfft(Fh * Fm, size)[:m]

    dx = 2.0 / n
    corr = fft_corr(h)
    return float(np.max(corr)) * dx

                g_norm_sq = np.dot(grad, grad)
                if f_x < f_y - 0.005 * step * g_norm_sq + 1e-16 * max(1.0, abs(f_y)):
                    x_new = x_trial
                    break
                step *= 0.5
            if x_new is None:
                x_new = project_feasible(y - step * grad, target)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y_new = x_new + ((t - 1.0) / t_new) * (x_new - h)

            c5_new = compute_c5(x_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = x_new.copy()
                stale = 0
            else:
                stale += 1
                if stale % 30 == 0:
                    y_new = x_new.copy()
                    t_new = 1.0

            h = x_new
            y = y_new
            t = t_new

    return best_h, best_c5


def nesterov_subgrad(h_init, n, target, time_limit, max_iters=50000):
    start_time = time.time()
    dx = 2.0 / n
    mid = n - 1
    h = h_init.copy()
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step_base = 0.01
    stale = 0

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        corr_y = fft_corr(y)
        c_max = np.max(corr_y)

        active = corr_y >= c_max - 1e-10
        n_active = np.sum(active)
        if n_active == 0:
            active[np.argmax(corr_y)] = True
            n_active = 1
        weights = np.zeros(len(corr_y))
        weights[active] = 1.0 / n_active

        grad = compute_grad(weights, y, n)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break

        step = step_base
        x_new = None
        for _bt in range(8):
            x_trial = y - step * grad
            x_trial = project_feasible(x_trial, target)
            c5_trial = compute_c5(x_trial)
            if c5_trial < c_max * dx + 1e-12:
                x_new = x_trial
                break
            step *= 0.5
        if x_new is None:
            x_new = project_feasible(y - step * grad, target)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - h)

        c5_new = compute_c5(x_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = x_new.copy()
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                y_new = x_new.copy()
                t_new = 1.0
            if stale > 2000:
                break

        h = x_new
        y = y_new
        t = t_new

    return best_h, best_c5


def coordinate_descent(h_init, n, target, time_limit, max_sweeps=30000,
                       n_cand_start=30, n_cand_end=100, max_no_improve=600):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        if k_star == 0:
            grad[:] = (1.0 - 2.0 * h)
        elif k_star > 0:
            grad[:n - k_star] += (1.0 - h[k_star:])
            grad[k_star:] -= h[:n - k_star]
        else:
            m = -k_star
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-15
                    if active_m.any():
                        f_prime = np.max(g[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-15:
                    best_trial_c5 = float(approx_max) * dx
                    best_i = i
                    best_j = j

        if best_i >= 0:
            delta_star = 0.5 * (max(h[best_i] - 1.0, -h[best_j]) + min(h[best_i], 1.0 - h[best_j]))
            g = change[best_j] - change[best_i]
            corr_now = np.correlate(h, 1.0 - h, mode='full')
            a_d = max(h[best_i] - 1.0, -h[best_j])
            b_d = min(h[best_i], 1.0 - h[best_j])
            for _ in range(20):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr_now + mid_d * g
                f_mid = np.max(vals)
                active_m = vals >= f_mid - 1e-14
                if active_m.any():
                    f_prime = np.max(g[active_m])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            final_delta = 0.5 * (a_d + b_d)
            h[best_i] -= final_delta
            h[best_j] += final_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            p1 = int(best_i)
            p2 = int(best_j)
            for p in [p1, p2]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5


def multi_shift_cd(h_init, n, target, time_limit, n_active=4, n_cand=60):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(20000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[0]]
        weights = np.exp(-80.0 * (c_max - corr[top_shifts]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            if k == 0:
                grad[:] += wi * (1.0 - 2.0 * h)
            elif k > 0:
                grad[:n - k] += wi * (1.0 - h[k:])
                grad[k:] -= wi * h[:n - k]
            else:
                m = -k
                grad[m:] += wi * (1.0 - h[:n - m])
                grad[:n - m] -= wi * h[m:]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j

        if best_i >= 0:
            h[best_i] -= best_trial_c5 / dx  # placeholder, recompute
            # Redo: find the actual delta
            i, j = best_i, best_j
            g = change[j] - change[i]
            a_d = max(h[i] - 1.0, -h[j])
            b_d = min(h[i], 1.0 - h[j])
            for _ in range(18):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_m = vals >= f_mid - 1e-14
                if active_m.any():
                    f_prime = np.max(g[active_m])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            final_delta = 0.5 * (a_d + b_d)
            h[i] -= final_delta
            h[j] += final_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [i, j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def _smooth_obj(x):
    corr = fft_corr(x)
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = fft_corr(x)
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        return np.zeros(len(x))
    weights /= wsum
    grad = compute_grad(weights, x, len(x))
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha=800, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 25.0 - (time.time() - start_time))

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    best_n = n
    best_target = target

    rng = np.random.RandomState(seed)

    # Phase 1: FISTA at initial resolution
    if tleft() > 120:
        alphas = [50, 200, 500, 2000, 8000]
        h_f, sc_f = fista_optimize(best_h, best_n, best_target,
                                   time_limit=min(250, tleft() * 0.6),
                                   alpha_schedule=alphas,
                                   iters_per_stage=3000)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f

    # Phase 2: Nesterov subgradient
    if tleft() > 100:
        h_ng, sc_ng = nesterov_subgrad(best_h, best_n, best_target,
                                       time_limit=min(150, tleft() * 0.5),
                                       max_iters=80000)
        if sc_ng < best_score:
            best_score = sc_ng
            best_h = h_ng

    # Phase 3: Coordinate descent polish
    if tleft() > 100:
        h_cd, sc_cd = coordinate_descent(best_h, best_n, best_target,
                                         time_limit=min(200, tleft() * 0.5),
                                         max_sweeps=40000,
                                         n_cand_start=30, n_cand_end=120,
                                         max_no_improve=800)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 4: Multi-shift CD
    if tleft() > 80:
        h_ms, sc_ms = multi_shift_cd(best_h, best_n, best_target,
                                     time_limit=min(120, tleft() * 0.5),
                                     n_active=5, n_cand=80)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms

    # Phase 5: Try higher resolutions
    for n_new in [500, 700, 900, 1200]:
        if tleft() < 250:
            break
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        h_cur = h_interp.copy()
        sc_cur = compute_c5(h_cur)
        cur_n = n_new
        cur_target = target_new

        # FISTA at new resolution
        if tleft() > 150:
            alphas = [100, 300, 1000, 5000, 20000]
            h_f2, sc_f2 = fista_optimize(h_cur, cur_n, cur_target,
                                         time_limit=min(200, tleft() * 0.4),
                                         alpha_schedule=alphas,
                                         iters_per_stage=2000)
            if sc_f2 < sc_cur:
                sc_cur = sc_f2
                h_cur = h_f2

        # CD at new resolution
        if tleft() > 100:
            h_cd2, sc_cd2 = coordinate_descent(h_cur, cur_n, cur_target,
                                               time_limit=min(180, tleft() * 0.4),
                                               max_sweeps=30000,
                                               n_cand_start=30, n_cand_end=100,
                                               max_no_improve=600)
            if sc_cd2 < sc_cur:
                sc_cur = sc_cd2
                h_cur = h_cd2

        # Multi-shift CD
        if tleft() > 60:
            h_ms2, sc_ms2 = multi_shift_cd(h_cur, cur_n, cur_target,
                                           time_limit=min(80, tleft() * 0.3),
                                           n_active=6, n_cand=90)
            if sc_ms2 < sc_cur:
                sc_cur = sc_ms2
                h_cur = h_ms2

        if sc_cur < best_score:
            best_score = sc_cur
            best_h = h_cur
            best_n = cur_n
            best_target = cur_target

    # Phase 6: Final SLSQP polish
    if tleft() > 60:
        h_sp, sc_sp = slsqp_round(best_h, best_n, best_target,
                                  [2000, 10000, 50000], 1500, 1e-15)
        if sc_sp < best_score:
            best_score = sc_sp
            best_h = h_sp

    # Phase 7: Final CD
    if tleft() > 50:
        h_cd3, sc_cd3 = coordinate_descent(best_h, best_n, best_target,
                                           time_limit=min(80, tleft() * 0.6),
                                           max_sweeps=20000,
                                           n_cand_start=50, n_cand_end=150,
                                           max_no_improve=400)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    # Phase 8: Random restarts
    if tleft() > 80:
        for noise_scale in [0.01, 0.03, 0.07, 0.15]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            h_r, sc_r = fista_optimize(h_pert, best_n, best_target,
                                       time_limit=min(40, tleft() * 0.4),
                                       alpha_schedule=[200, 1000, 5000],
                                       iters_per_stage=1500)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r
            if tleft() > 30 and sc_r < best_score + 0.001:
                h_cd_r, sc_cdr = coordinate_descent(h_r, best_n, best_target,
                                                    time_limit=min(30, tleft() * 0.4),
                                                    max_sweeps=10000,
                                                    n_cand_start=40, n_cand_end=120,
                                                    max_no_improve=200)
                if sc_cdr < best_score:
                    best_score = sc_cdr
                    best_h = h_cd_r

    # Phase 9: Final SLSQP
    if tleft() > 40:
        h_sp2, sc_sp2 = slsqp_round(best_h, best_n, best_target,
                                    [10000, 50000, 200000], 2000, 1e-15)
        if sc_sp2 < best_score:
            best_score = sc_sp2
            best_h = h_sp2

    # Phase 10: Final CD
    if tleft() > 20:
        h_cd4, sc_cd4 = coordinate_descent(best_h, best_n, best_target,
                                           time_limit=min(15, tleft() * 0.5),
                                           max_sweeps=8000,
                                           n_cand_start=60, n_cand_end=150,
                                           max_no_improve=100)
        if sc_cd4 < best_score:
            best_score = sc_cd4
            best_h = h_cd4

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
