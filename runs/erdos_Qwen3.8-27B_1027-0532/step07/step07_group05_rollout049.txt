def compute_c5(h):

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_from_corr(h, n, top_lags, weights):
    """Compute gradient of weighted sum of top overlaps."""
    grad = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    for idx in range(len(top_lags)):
        k = int(top_lags[idx]) - mid
        w = weights[idx]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad -= np.mean(grad)  # Remove mean to respect sum constraint approximately
    return grad

def fista_step(h, n, target, alpha, step_size, max_iters, time_limit, start_time):

    h_prev = h.copy()
    t = 1.0
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)

        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        c_max = np.max(corr)
        scaled = alpha * (corr - c_max)
        exp_vals = np.exp(scaled)
        w_sum = np.sum(exp_vals)
        weights = exp_vals / w_sum
        
        top_idx = np.argsort(corr)[-8:]
        grad = compute_grad_from_corr(h_curr, n, top_idx, weights[top_idx])

        y = h_curr - step * grad

        h_curr = h_momentum

        if sc < best_score:

            best_h = h_curr.copy()
        
        t = t_new

def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    
    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_fun(x):
        return np.sum(x) - _g_target
    
    def cons_jac(x):
        return np.ones(_g_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def build_change_row(p, h, n):
    """Build the p-th row of the change matrix correctly."""
    n_corr = 2 * n - 1
    row = np.zeros(n_corr)
    # First term: positions [p, p+n-1], values (1-h[n-1-offset]) for offset 0..n-1
    row[p:p+n] += (1.0 - h[::-1])
    # Second term: positions [n-1-p, 2n-2-p], values -h[offset] for offset 0..n-1
    row[n-1-p:n-1-p+n] -= h
    return row

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        change[p, p:p+n] += (1.0 - h[::-1])
        change[p, n-1-p:2*n-1-p] -= h
    return change

def smart_cd(h_init, n, target, time_limit, start_time, max_sweeps,
             n_active=3, n_cand_start=20, n_cand_end=50,
             max_no_improve=300, bisection_iters=16):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum
        
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        max_pairs = 300
        pairs = []
        if n_h * n_l <= max_pairs:
            for ii in range(n_h):
                for jj in range(n_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pairs.append((iv, jv))
        else:
            # Stratified sampling
            n_sample = max_pairs
            for _ in range(n_sample):
                ii = np.random.randint(n_h)
                jj = np.random.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pairs.append((iv, jv))
        
        for i, j in pairs:
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue
            
            g = change[j] - change[i]
            
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            
            # Update correlation incrementally
            corr += best_delta * (change[best_j] - change[best_i])
            
            # Update change rows
            change[best_i, :] = build_change_row(best_i, h, n)
            change[best_j, :] = build_change_row(best_j, h, n)
    
    return best_h, best_c5

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
