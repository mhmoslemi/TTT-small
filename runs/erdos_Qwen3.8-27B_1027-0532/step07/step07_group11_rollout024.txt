```python
def projected_gradient(h_init, n, target, time_limit, alpha=1000, eta=0.01, max_iter=50000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_val = c_max * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Softmax weights
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500, 0)
        p = np.exp(shifted)
        p /= p.sum()
        
        # Gradient
        grad = np.zeros(n)
        for s_idx in range(len(corr)):
            if p[s_idx] < 1e-15:
                continue
            k = s_idx - mid
            w = p[s_idx]
            if k >= 0:
                # corr[idx] = sum_i h[i](1-h[i+k]) for i in [0, n-1-k]
                # d/dh[j] = (1-h[j+k]) for j in [0, n-1-k] - h[j-k] for j in [k, n-1]
                v1 = np.arange(n - k)  # j where j+k < n
                grad[v1] += w * (1.0 - h[v1 + k])
                v2 = np.arange(k, n)  # j where j-k >= 0
                grad[v2] -= w * h[v2 - k]
            else:
                k_abs = -k
                v1 = np.arange(n - k_abs)
                grad[v1] += w * (1.0 - h[v1 + k_abs])
                v2 = np.arange(k_abs, n)
                grad[v2] -= w * h[v2 - k_abs]
        
        # Center gradient (for the equality constraint)
        grad -= np.mean(grad)
        
        # Line search
        eta_cur = eta
        for ls in range(10):
            h_try = h - eta_cur * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < c5_val * (1 - 1e-4) or eta_cur < 1e-10:
                break
            eta_cur *= 0.5
        
        h = h_try
        eta = eta_cur
    
    return best_h, best_c5
```

```python
    grad = np.zeros(n)
    mid = n - 1
    for s in range(-(n-1), n):
        w = p[mid + s]
        if w < 1e-15:
            continue
        if s >= 0:
            # (1-h[j+s]) for j in [0, n-1-s]
            grad[:n-s] += w * (1.0 - h[s:])
            # -h[j-s] for j in [s, n-1]
            grad[s:] -= w * h[:n-s]
        else:
            s_abs = -s
            grad[:n-s_abs] += w * (1.0 - h[s_abs:])
            grad[s_abs:] -= w * h[:n-s_abs]
    return grad
```

```python
def projected_gradient_fast(h_init, n, target, time_limit, alpha=5000, eta0=0.01):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = eta0
    
    m = len(h)  # = n
    corr_len = 2*n - 1
    
    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_val = c_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Softmax weights
        diff = corr - c_max
        diff = np.clip(diff, -500.0/alpha, 0.0)
        p = np.exp(alpha * diff)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(corr_len)
            p[np.argmax(corr)] = 1.0
        else:
            p /= p_sum
        
        # Gradient
        grad = np.zeros(n)
        for s in range(-(n-1), n):
            w = p[mid + s]
            if w < 1e-14:
                continue
            if s >= 0:
                grad[:n-s] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:n-s]
            else:
                s_abs = -s
                grad[:n-s_abs] += w * (1.0 - h[s_abs:])
                grad[s_abs:] -= w * h[:n-s_abs]
        
        # Center for constraint
        grad -= np.mean(grad)
        
        # Line search (backtracking)
        cur_obj = c5_val
        for ls in range(20):
            h_try = h - eta * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc <= cur_obj * (1.0 - 1e-5) or sc < cur_obj:
                break
            eta *= 0.5
        else:
            eta = max(eta * 0.5, 1e-10)
            continue
        
        h = h_try
        eta = min(eta * 1.05, 0.1)  # Slightly increase after successful step
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def projected_gradient(h_init, n, target, time_limit, alphas, eta0=0.005):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = eta0

    for alpha in alphas:
        if time.time() - start > time_limit - 5:
            break
        h_local = h.copy()
        eta_local = eta
        for it in range(500000):
            if time.time() - start > time_limit - 5:
                break
            corr = np.correlate(h_local, 1.0 - h_local, mode='full')
            c_max = float(np.max(corr))
            c5_val = c_max * dx
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h_local.copy()

            diff = corr - c_max
            diff = np.clip(diff, -500.0 / alpha, 0.0)
            p = np.exp(alpha * diff)
            p_sum = p.sum()
            if p_sum < 1e-300:
                p = np.zeros(len(corr))
                p[int(np.argmax(corr))] = 1.0
            else:
                p /= p_sum

            grad = np.zeros(n)
            for s in range(-(n - 1), n):
                w = p[mid + s]
                if w < 1e-13:
                    continue
                if s >= 0:
                    grad[:n - s] += w * (1.0 - h_local[s:])
                    grad[s:] -= w * h_local[:n - s]
                else:
                    sa = -s
                    grad[:n - sa] += w * (1.0 - h_local[sa:])
                    grad[sa:] -= w * h_local[:n - sa]

            grad -= np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break

            accepted = False
            eta_try = eta_local
            for ls in range(20):
                h_try = h_local - eta_try * grad
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < c5_val - 1e-15:
                    h_local = h_try
                    accepted = True
                    break
                eta_try *= 0.5
            if not accepted:
                eta_local = max(eta_try * 0.5, 1e-12)
                if eta_local < 1e-12:
                    break
            else:
                eta_local = min(eta_local * 1.1, 0.05)

        eta = max(eta_local, 1e-6)
        h = h_local.copy()

    return best_h, best_c5


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd_advanced(h_init, n, target, time_limit, max_sweeps=50000,
                     n_cand_start=30, n_cand_end=120, n_shifts=3,
                     max_no_improve=500, anneal_temp=0.0, anneal_decay=0.999):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = np.sum(shift_weights)
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 1000.0
                    if np.random.random() < np.exp(-delta_obj / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 2:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.95, 1e-10)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_out, target_dst)


def multi_shift_subgradient(h_init, n, target, time_limit, n_steps=50000, n_weights=5, step_scale=0.004):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        p = np.exp(corr * 80.0)
        p /= p.sum()

        grad = np.zeros(n)
        top_k = np.argsort(p)[-n_weights:]
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
            eta = min(eta * 1.02, 0.05)
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.9

    return best_h, best_c5
```
