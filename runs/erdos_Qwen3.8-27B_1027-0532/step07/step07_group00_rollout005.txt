        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def nesterov_smoothed(h_init, n, target, time_limit, max_iter=3000,
                       alpha_start=50.0, alpha_end=10000.0,
                       momentum=0.9, step_init=0.02):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 800:
            break

        progress = min(float(it) / max(max_iter, 1), 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        y = h + momentum * (h - h_prev)
        y = project_feasible(y, target)

        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        weights = weights / w_sum

        grad = np.zeros(n)
        top_k = min(50, n_corr)
        top_idx = np.argsort(weights)[-top_k:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - y[idxs[v1] - k])
            grad[v2] -= w * y[idxs[v2] + k]
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        grad = grad / grad_norm

        h_try = h - step * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            no_improve = 0
            h_prev = h.copy()
            h = h_try
            step = min(step * 1.02, 0.1)
        elif sc < compute_c5(h):
            h_prev = h.copy()
            h = h_try
            no_improve += 1
        else:
            step *= 0.8
            no_improve += 1

    return best_h, best_c5


def pair_cd_improved(h_init, n, target, time_limit, max_sweeps=20000,
                     n_cand_start=25, n_cand_end=80, max_no_improve=300,
                     weight_beta=6.0, top_k_shifts=40):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        grad = np.zeros(n)
        top_idx = np.argsort(weights)[-top_k_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        found = False
        for ii in range(n_h):
            if found:
                break
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    found = True
                    break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def multi_pair_cd(h_init, n, target, time_limit, max_sweeps=15000,
                  n_pairs_per_sweep=3, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        c_max = float(np.max(corr))
        weights = np.exp(5.0 * (corr - c_max) / max(c_max, 1e-10))
        grad = np.zeros(n)
        top_k = min(30, n_corr)
        top_idx = np.argsort(weights)[-top_k:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        n_cand = min(60, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_h = len(high_set)
        n_l = len(low_set)
        moves = []
        for ii in range(min(n_pairs_per_sweep, n_h)):
            best_i = -1
            best_j = -1
            best_delta = 0.0
            best_trial = c5_val

            start_h = (ii * 3) % n_h
            start_l = (ii * 7) % n_l
            for ii2 in range(n_h):
                i = int(high_set[(start_h + ii2) % n_h])
                for jj2 in range(n_l):
                    j = int(low_set[(start_l + jj2) % n_l])
                    if i == j:
                        continue
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo < 1e-12:
                        continue
                    g = change[j] - change[i]
                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(16):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-14
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g) * dx
                    if approx_max < best_trial:
                        best_trial = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
                        break

            if best_i >= 0:
                moves.append((best_i, best_j, best_delta))
                corr = np.correlate(h - best_delta * (np.eye(n)[best_i] - np.eye(n)[best_j]), 1.0 - (h - best_delta * (np.eye(n)[best_i] - np.eye(n)[best_j])), mode='full')

        if moves:
            for (i, j, d) in moves:
                h[i] -= d
                h[j] += d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for (i, j, d) in moves:
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)

    return best_h, best_c5


def penalty_subgradient(h_init, n, target, time_limit, max_iter=5000,
                         t_target=None, beta=50.0, step_init=0.01):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    no_improve = 0

    if t_target is None:
        t_target = best_c5 - 0.005

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 600:
            t_target = best_c5 - 0.002
            no_improve = 0
            if it > 3000:
                break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0

        active = corr * dx > t_target
        if not active.any():
            no_improve += 1
            continue

        n_active = int(np.sum(active))
        grad = np.zeros(n)
        for s_idx in range(n_corr):
            if not active[s_idx]:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(n_active, 1)
        grad -= np.mean(grad)

        h_try = h - step * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val:
            h = h_try
            step = min(step * 1.03, 0.08)
        else:
            step *= 0.85
            no_improve += 1

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturb(h, n, target, seed_val, pattern='sin'):
    rng = np.random.RandomState(seed_val)
    x = np.arange(n) / n
    if pattern == 'sin':
        freq = rng.choice([2, 3, 5, 7, 11])
        phase = rng.uniform(0, 2 * np.pi)
        pert = 0.03 * np.sin(2 * np.pi * freq * x + phase)
    elif pattern == 'tri':
        freq = rng.choice([2, 3, 4])
        pert = 0.02 * np.abs(np.sin(np.pi * freq * x)) - 0.01
    else:
        pert = rng.randn(n) * 0.02
    h_new = h + pert
    return project_feasible(h_new, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Nesterov smoothed gradient at initial resolution
    if tleft() > 120:
        h_nest, sc_nest = nesterov_smoothed(best_h, n0, target0,
                                            time_limit=min(100, tleft() * 0.7),
                                            max_iter=3000,
                                            alpha_start=50.0, alpha_end=20000.0,
                                            momentum=0.85, step_init=0.015)
        if sc_nest < best_score:
            best_score = sc_nest
            best_h = h_nest

    # Phase 2: SLSQP polish
    if tleft() > 80:
        h_pol, sc_pol = slsqp_round(best_h, n0, target0,
                                    [100, 500, 2000, 10000],
                                    400, 1e-14)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Phase 3: Improved pair CD (larger candidate sets)
    if tleft() > 150:
        h_cd, c5_cd = pair_cd_improved(best_h, n0, target0,
                                       time_limit=min(120, tleft() * 0.6),
                                       max_sweeps=30000,
                                       n_cand_start=30, n_cand_end=80,
                                       max_no_improve=400,
                                       weight_beta=6.0,
                                       top_k_shifts=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Interpolate to finer grid
    n1 = 1000
    target1 = n1 / 2.0
    if tleft() > 350:
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 250:
            h_pol1, sc_pol1 = slsqp_round(h1, n1, target1,
                                          [50, 200, 1000, 5000],
                                          300, 1e-14)
            if sc_pol1 < best_score:
                best_score = sc_pol1
                best_h = h_pol1
            else:
                best_h = h_pol1

        if tleft() > 180:
            h_cd1, c5_cd1 = pair_cd_improved(best_h, n1, target1,
                                             time_limit=min(150, tleft() * 0.6),
                                             max_sweeps=25000,
                                             n_cand_start=30, n_cand_end=70,
                                             max_no_improve=350,
                                             weight_beta=6.0,
                                             top_k_shifts=40)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        if tleft() > 80:
            h_pol1b, sc1b = slsqp_round(best_h, n1, target1,
                                        [5000, 20000],
                                        400, 1e-14)
            if sc1b < best_score:
                best_score = sc1b
                best_h = h_pol1b

    # Phase 5: Penalty subgradient (focus on active constraints)
    if tleft() > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pen, c5_pen = penalty_subgradient(best_h, n_cur, target_cur,
                                            time_limit=min(80, tleft() * 0.6),
                                            max_iter=4000,
                                            t_target=best_score - 0.003,
                                            step_init=0.008)
        if c5_pen < best_score:
            best_score = c5_pen
            best_h = h_pen

    # Phase 6: Multi-start with structured perturbations
    if tleft() > 200:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        patterns = ['sin', 'tri', 'rand']
        for p_i, pat in enumerate(patterns):
            if tleft() < 60:
                break
            h_pert = structured_perturb(best_h, n_cur, target_cur,
                                        seed + p_i * 100, pattern=pat)
            h_ref, sc_ref = slsqp_round(h_pert, n_cur, target_cur,
                                        [200, 1000, 5000],
                                        300, 1e-14)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 50:
                h_cd_ref, c5_cd_ref = pair_cd_improved(best_h, n_cur, target_cur,
                                                       time_limit=min(30, tleft() * 0.4),
                                                       max_sweeps=10000,
                                                       n_cand_start=25, n_cand_end=60,
                                                       max_no_improve=200,
                                                       weight_beta=6.0)
                if c5_cd_ref < best_score:
                    best_score = c5_cd_ref
                    best_h = h_cd_ref

    # Phase 7: Nesterov at current resolution with higher alpha
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_nest2, sc_nest2 = nesterov_smoothed(best_h, n_cur, target_cur,
                                              time_limit=min(60, tleft() * 0.7),
                                              max_iter=2500,
                                              alpha_start=500.0, alpha_end=50000.0,
                                              momentum=0.9, step_init=0.008)
        if sc_nest2 < best_score:
            best_score = sc_nest2
            best_h = h_nest2

    # Phase 8: Pair CD with very high beta (focus on critical shifts)
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cd2, c5_cd2 = pair_cd_improved(best_h, n_cur, target_cur,
                                         time_limit=min(60, tleft() * 0.6),
                                         max_sweeps=15000,
                                         n_cand_start=40, n_cand_end=90,
                                         max_no_improve=300,
                                         weight_beta=10.0,
                                         top_k_shifts=20)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 9: Try even finer grid if time allows
    if tleft() > 250:
        n_cur = len(best_h)
        if n_cur < 1500:
            n2 = 1500
            target2 = n2 / 2.0
            h2 = interpolate_to(best_h, n2, target2)

            if tleft() > 200:
                h_pol2h, sc2h = slsqp_round(h2, n2, target2,
                                            [100, 500, 2000],
                                            300, 1e-14)
                if tleft() > 150:
                    h_cd2h, c5_cd2h = pair_cd_improved(h_pol2h, n2, target2,
                                                       time_limit=min(130, tleft() * 0.6),
                                                       max_sweeps=15000,
                                                       n_cand_start=30, n_cand_end=70,
                                                       max_no_improve=300,
                                                       weight_beta=6.0)
                    if c5_cd2h < best_score:
                        best_score = c5_cd2h
                        best_h = h_cd2h
                    if tleft() > 50:
                        h_pol3, sc3 = slsqp_round(best_h, n2, target2,
                                                  [5000, 20000],
                                                  400, 1e-14)
                        if sc3 < best_score:
                            best_score = sc3
                            best_h = h_pol3

    # Phase 10: Final polish
    if tleft() > 50:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final_cd, c5_final = pair_cd_improved(best_h, n_cur, target_cur,
                                                time_limit=min(35, tleft() * 0.6),
                                                max_sweeps=10000,
                                                n_cand_start=40, n_cand_end=80,
                                                max_no_improve=200,
                                                weight_beta=8.0)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Phase 11: Final SLSQP
    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin, sc_fin = slsqp_round(best_h, n_cur, target_cur,
                                    [50000, 100000],
                                    400, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
