import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)

def _grad_smooth(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    # Gradient of sum(weights * corr) w.r.t. x
    # corr[k] = sum_i x[i] * (1 - x[i+k]) for appropriate k
    # d corr[k] / d x[p] = (1 - x[p-k]) - x[p+k] (with boundary handling)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for ki in range(n_corr):
        if weights[ki] < 1e-15:
            continue
        k = ki - mid
        # corr[ki] = sum over valid i of x[i] * (1 - x[i+k])
        # d/dx[p] = (1 - x[p-k]) if p-k is valid index in second factor
        #           - x[p+k] if p+k is valid index in second factor
        if k >= 0:
            # corr = sum_{i=0}^{n-1-k} x[i] * (1-x[i+k])
            # d/dx[p] for p in [0, n-1-k]: (1 - x[p+k])
            # d/dx[p] for p in [k, n-1]: -x[p-k]
            v1 = (idxs >= 0) & (idxs <= n - 1 - k)
            v2 = (idxs >= k) & (idxs <= n - 1)
            grad[v1] += weights[ki] * (1.0 - x[idxs[v1] + k])
            grad[v2] -= weights[ki] * x[idxs[v2] - k]
        else:
            k_abs = -k
            # corr = sum_{i=k_abs}^{n-1} x[i] * (1-x[i-k_abs])
            # d/dx[p] for p in [k_abs, n-1]: (1 - x[p-k_abs])
            # d/dx[p] for p in [0, n-1-k_abs]: -x[p+k_abs]
            v1 = (idxs >= k_abs) & (idxs <= n - 1)
            v2 = (idxs >= 0) & (idxs <= n - 1 - k_abs)
            grad[v1] += weights[ki] * (1.0 - x[idxs[v1] - k_abs])
            grad[v2] -= weights[ki] * x[idxs[v2] + k_abs]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

def projected_subgrad(h_init, n, target, time_limit, n_iter=5000, step_scale=0.01):
    """Projected subgradient method with momentum."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 0.9  # momentum
    
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Subgradient: pick random active constraint
        top_k = np.argmax(corr)
        k = top_k - (n - 1)
        
        # Gradient of corr[top_k] w.r.t. h
        grad = np.zeros(n)
        idxs = np.arange(n)
        if k >= 0:
            v1 = (idxs >= 0) & (idxs <= n - 1 - k)
            v2 = (idxs >= k) & (idxs <= n - 1)
            grad[v1] += (1.0 - h[idxs[v1] + k])
            grad[v2] -= h[idxs[v2] - k]
        else:
            k_abs = -k
            v1 = (idxs >= k_abs) & (idxs <= n - 1)
            v2 = (idxs >= 0) & (idxs <= n - 1 - k_abs)
            grad[v1] += (1.0 - h[idxs[v1] - k_abs])
            grad[v2] -= h[idxs[v2] + k_abs]
        grad -= np.mean(grad)  # project out the sum constraint direction
        
        # Adaptive step size
        step = step_scale / (1.0 + 0.01 * it)
        v = beta * v - step * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        h = h_new
    
    return best_h, best_c5

            v1 = (idxs >= k_abs) & (idxs <= n - 1)
            v2 = (idxs >= 0) & (idxs <= n - 1 - k_abs)
            grad[v1] += wk * (1.0 - x[idxs[v1] - k_abs])
            grad[v2] -= wk * x[idxs[v2] + k_abs]
    return grad


def _smooth_jac_global(x):
    return _smooth_jac(x, _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,), jac=_smooth_jac_global,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change


def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]


def enhanced_cd(h_init, n, target, time_limit, max_sweeps=20000,
                n_active=5, n_cand_start=20, n_cand_end=60,
                max_no_improve=500, bisection_iters=12,
                equalize_weight=0.35):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break
        top_idx = np.argsort(corr)[-n_active:]
        top_vals = corr[top_idx]
        top_mean = np.mean(top_vals)
        excess = np.maximum(top_vals - top_mean, 1e-10)
        weights = excess / np.sum(excess)
        uniform = np.ones(n_active) / n_active
        weights = (1.0 - equalize_weight) * uniform + equalize_weight * weights
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            if k >= 0:
                v1 = (idxs >= 0) & (idxs <= n - 1 - k)
                v2 = (idxs >= k) & (idxs <= n - 1)
                grad[v1] += w * (1.0 - h[idxs[v1] + k])
                grad[v2] -= w * h[idxs[v2] - k]
            else:
                k_abs = -k
                v1 = (idxs >= k_abs) & (idxs <= n - 1)
                v2 = (idxs >= 0) & (idxs <= n - 1 - k_abs)
                grad[v1] += w * (1.0 - h[idxs[v1] - k_abs])
                grad[v2] -= w * h[idxs[v2] + k_abs]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        max_pairs = 250
        if n_h * n_l <= max_pairs:
            pair_list = []
            for ii in range(n_h):
                for jj in range(n_l):
                    i_val = int(high_set[ii])
                    j_val = int(low_set[jj])
                    if i_val != j_val:
                        pair_list.append((i_val, j_val))
        else:
            pair_list = []
            n_strata = min(25, max_pairs)
            for s in range(n_strata):
                ii = int(s * n_h / n_strata)
                jj = int(s * n_l / n_strata)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))
            rng_local = np.random.RandomState(sweep + n)
            while len(pair_list) < max_pairs:
                ii = rng_local.randint(n_h)
                jj = rng_local.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))
        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue
            g = change[j] - change[i]
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial - 1e-15:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
    return best_h, best_c5


def projected_subgrad_momentum(h_init, n, target, time_limit,
                                n_iter=8000, step_scale=0.005, beta=0.85):
    start = time.time()
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx_local
        dx_local = 2.0 / n
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        n_active = min(5, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        top_vals = corr[top_idx]
        top_mean = np.mean(top_vals)
        excess = np.maximum(top_vals - top_mean, 1e-10)
        weights = excess / np.sum(excess)
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            if k >= 0:
                v1 = (idxs >= 0) & (idxs <= n - 1 - k)
                v2 = (idxs >= k) & (idxs <= n - 1)
                grad[v1] += w * (1.0 - h[idxs[v1] + k])
                grad[v2] -= w * h[idxs[v2] - k]
            else:
                k_abs = -k
                v1 = (idxs >= k_abs) & (idxs <= n - 1)
                v2 = (idxs >= 0) & (idxs <= n - 1 - k_abs)
                grad[v1] += w * (1.0 - h[idxs[v1] - k_abs])
                grad[v2] -= w * h[idxs[v2] + k_abs]
        grad -= np.mean(grad)
        step = step_scale / (1.0 + 0.005 * it)
        v = beta * v - step * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 80:
        h_cd, c5_cd = enhanced_cd(best_h, n0, target0,
                                   time_limit=min(40, tleft() * 0.4),
                                   max_sweeps=6000, n_active=5,
                                   n_cand_start=15, n_cand_end=40,
                                   max_no_improve=250, bisection_iters=12,
                                   equalize_weight=0.4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n0, target0,
                            [100, 500, 2000],
                            500, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        tl = min(120, tleft() * 0.45)
        h_cd1, c5_cd1 = enhanced_cd(h1, n1, target1,
                                     time_limit=tl,
                                     max_sweeps=15000, n_active=5,
                                     n_cand_start=20, n_cand_end=55,
                                     max_no_improve=400, bisection_iters=12,
                                     equalize_weight=0.35)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1
            n0 = n1
            target0 = target1

    if tleft() > 120:
        h_pol = slsqp_round(best_h, n0, target0,
                            [500, 2000, 10000],
                            600, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 300:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        tl = min(180, tleft() * 0.45)
        h_cd2, c5_cd2 = enhanced_cd(h2, n2, target2,
                                     time_limit=tl,
                                     max_sweeps=20000, n_active=7,
                                     n_cand_start=25, n_cand_end=70,
                                     max_no_improve=500, bisection_iters=12,
                                     equalize_weight=0.35)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
            n0 = n2
            target0 = target2

    if tleft() > 150:
        h_sub, sc_sub = projected_subgrad_momentum(best_h, n0, target0,
                                                    time_limit=min(80, tleft() * 0.5),
                                                    n_iter=6000, step_scale=0.004, beta=0.85)
        if sc_sub < best_score:
            best_score = sc_sub
            best_h = h_sub

    if tleft() > 100:
        h_pol = slsqp_round(best_h, n0, target0,
                            [2000, 10000, 50000],
                            500, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 120:
        for scale in [0.02, 0.05, 0.08]:
            if tleft() < 70:
                break
            h_pert = best_h + rng.randn(n0) * scale
            h_pert = project_feasible(h_pert, target0)
            h_re, sc_re = enhanced_cd(h_pert, n0, target0,
                                       time_limit=min(40, tleft() * 0.35),
                                       max_sweeps=8000, n_active=5,
                                       n_cand_start=25, n_cand_end=60,
                                       max_no_improve=300, bisection_iters=12,
                                       equalize_weight=0.4)
            if sc_re < best_score:
                best_score = sc_re
                best_h = h_re

    if tleft() > 80:
        h_cd3, c5_cd3 = enhanced_cd(best_h, n0, target0,
                                     time_limit=min(50, tleft() * 0.5),
                                     max_sweeps=15000, n_active=7,
                                     n_cand_start=35, n_cand_end=80,
                                     max_no_improve=400, bisection_iters=12,
                                     equalize_weight=0.5)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 60:
        h_final = slsqp_round(best_h, n0, target0,
                               [10000, 50000],
                               400, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if tleft() > 40:
        h_cd4, c5_cd4 = enhanced_cd(best_h, n0, target0,
                                     time_limit=min(25, tleft() * 0.5),
                                     max_sweeps=10000, n_active=7,
                                     n_cand_start=40, n_cand_end=90,
                                     max_no_improve=300, bisection_iters=10,
                                     equalize_weight=0.5)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
