    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])

    return change

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

    cur_score = compute_c5(h_cur)
    for alpha in alphas:

            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new
        except Exception:

    return h_cur, cur_score

**Use a "warm start" for the binsearch**: For most pairs, the optimal delta is small (the current solution is already near-optimal for that direction). We can check if delta=0 is already optimal (i.e., the gradient in that direction is non-negative), and skip the binsearch if so.

Even better: **Pre-compute which pairs have negative directional derivative** and only binsearch those.

The directional derivative for pair (i, j) at delta=0 is:

This is max_{s: corr[s] = max(corr)} g[s] (the max of g over the active set).

So for each pair, we just need to check if max(g[active_set]) < 0 (meaning the direction is improving). We can pre-compute the active set and then for each pair, just compute max(g[active_set]) which is much faster than full binsearch.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def smooth_max_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        weights = np.zeros(len(corr))
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= wsum
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-14:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def pair_cd_fast(h_init, n, target, time_limit, max_sweeps=100000,
                 n_cand_start=50, n_cand_end=150, max_no_improve=500,
                 n_binsearch=20, top_k_soft=5, soft_weight=0.3,
                 stochastic_frac=0.03, seed=123):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        c_max = float(np.max(corr))
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        
        # Compute gradient with top-k soft weighting
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        if soft_weight > 0 and top_k_soft > 1:
            top_k_idx = np.argsort(corr)[-top_k_soft:]
            top_k_vals = corr[top_k_idx]
            top_k_max = top_k_vals.max()
            shifted_k = 10.0 * (top_k_vals - top_k_max)
            soft_w = np.exp(shifted_k)
            soft_w /= soft_w.sum()
            
            grad = np.zeros(n)
            for ii in range(len(top_k_idx)):
                s = int(top_k_idx[ii])
                w = soft_w[ii]
                k = s - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k])
            # Blend with max gradient
            grad_max = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad_max[v1] += (1.0 - h[idxs[v1] - k_star])
            grad_max[v2] -= h[idxs[v2] + k_star])
            grad = (1.0 - soft_weight) * grad_max + soft_weight * grad
        else:
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star])
        
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        # Pre-filter promising pairs using active set
        active_thresh = 1e-10
        active = np.where(corr >= c_max - active_thresh)[0]
        n_active = len(active)
        
        # Compute directional derivatives for all candidate pairs
        # g = change[j] - change[i], dir_deriv = max(g[active])
        # Vectorize:
        high_ch = change[high_set][:, active]  # (n_h, n_active)
        low_ch = change[low_set][:, active]  # (n_l, n_active)
        # For pair (ii, jj): diff = low_ch[jj] - high_ch[ii], dir_deriv = max(diff)
        # Broadcasting: (n_h, n_l, n_active) → max over axis 2 → (n_h, n_l)
        diffs = low_ch[None, :, :].astype(float) - high_ch[:, None, :].astype(float)
        dir_derivs = np.max(diffs, axis=2)
        
        # Find promising pairs (dir_deriv < 0)
        promising = np.argwhere(dir_derivs < -1e-12)
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        # Try promising pairs (limit to top 200 by most negative dir_deriv)
        if len(promising) > 0:
            flat_idx = promising[:, 0] * len(low_set) + promising[:, 1]
            flat_vals = dir_derivs.flat[flat_idx]
            # Sort by most negative
            order = np.argsort(flat_vals)[:min(200, len(flat_vals))]
            for idx in order:
                ii = int(idx // len(low_set))
                jj = int(idx % len(low_set))
                i = int(high_set[ii])
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        # Also try some random pairs
        n_random = max(5, int(stochastic_frac * current_ncand * current_ncand))
        for _ in range(n_random):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-12:
                continue
            g = change[j] - change[i]
            # Quick check: is this direction improving?
            g_active = g[active]
            if np.max(g_active) >= -1e-12:
                continue
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(n_binsearch):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_m = vals >= f_mid - 1e-14
                if active_m.any():
                    f_prime = np.max(g[active_m])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            # No improving pair found - perturb
            if consecutive_no_improve > 50 and consecutive_no_improve % 30 == 0:
                n_pert = max(5, n // 20)
                idxs_pert = rng.choice(n, size=n_pert, replace=False)
                h[idxs_pert] += rng.randn(n_pert) * 0.03
                h = project_feasible(h, target)
                # Recompute change matrix for perturbed indices
                for ip in idxs_pert:
                    update_change_row(change, int(ip), h, n, shifts)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
