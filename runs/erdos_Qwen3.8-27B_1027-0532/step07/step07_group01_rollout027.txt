from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradients(h, n):
    """Compute gradients of all f_k at current h. Returns (n, 2n-1) array."""
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    grads = np.zeros((n, 2*n-1))
    for s_idx, k in enumerate(shifts):
        g = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        g[v1] += (1.0 - h[idxs[v1] - k])
        g[v2] -= h[idxs[v2] + k]
        grads[:, s_idx] = g
    return grads

    n = len(h_init)

    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    # Precompute masks for each shift
    shift_masks = []
    for k in range(-(n-1), n):
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        shift_masks.append((v1, v2))
    
    alphas = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]
    
    for iteration in range(max_inner_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            break
        
        # Build LP using all shifts
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Build A_ub and b_ub
        A_ub = np.zeros((n_corr, n + 1))
        b_ub = np.zeros(n_corr)
        
        for s_idx, k in enumerate(range(-(n-1), n)):
            v1, v2 = shift_masks[s_idx]
            g = np.zeros(n)
            g[v1] += (1.0 - h[idxs[v1] - k])
            g[v2] -= h[idxs[v2] + k]
            
            A_ub[s_idx, :n] = g
            A_ub[s_idx, n] = -1.0
            b_ub[s_idx] = np.dot(g, h) - corr[s_idx]
        
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})

            continue

            h_trial = project_feasible(h_trial, target)

                h = h_trial

                h = project_feasible(h, target)

def subgradient_step(h, target, n_top=10, step_scale=1.0):
    """One step of projected subgradient descent."""
    n = len(h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Top shifts
    top_k = np.argsort(corr)[-n_top:]
    top_corr = corr[top_k]
    top_sum = np.sum(top_corr)
    if top_sum < 1e-15:
        return h
    
    # Weighted gradient
    grad = np.zeros(n)
    for k_idx in top_k:
        w = corr[k_idx] / top_sum
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    
    # Step size (standard: c/sqrt(t))
    # For now, use fixed step
    step = step_scale / np.sqrt(n)
    h_new = h - step * grad
    h_new = project_feasible(h_new, target)
    return h_new

    rng = np.random.RandomState(12345)

    shift_masks = []
    for k in range(-(n - 1), n):
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        shift_masks.append((v1, v2))

    alphas = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]
    max_constraints = min(n_corr, 1500) if not use_all_shifts else n_corr

    for iteration in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            break
        if no_improve > 100 and no_improve % 50 == 0:
            h = best_h + rng.randn(n) * 0.02
            h = project_feasible(h, target)
            no_improve = 0

        if max_constraints < n_corr:
            top_idx = np.argsort(corr)[-max_constraints:]
        else:
            top_idx = np.arange(n_corr)

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        nc = len(top_idx)
        A_ub = np.zeros((nc, n + 1))
        b_ub = np.zeros(nc)

        for ci, s_idx in enumerate(top_idx):
            k = s_idx - mid
            v1, v2 = shift_masks[s_idx]
            g = np.zeros(n)
            g[v1] += (1.0 - h[idxs[v1] - k])
            g[v2] -= h[idxs[v2] + k]
            A_ub[ci, :n] = g
            A_ub[ci, n] = -1.0
            b_ub[ci] = np.dot(g, h) - corr[s_idx]

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 3.0})
        except Exception:
            continue

        if not res.success:
            if no_improve > 50:
                break
            continue

        h_lp = project_feasible(res.x[:n], target)

        improved = False
        for alpha in alphas:
            h_trial = h + alpha * (h_lp - h)
            h_trial = project_feasible(h_trial, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_val - 1e-16:
                h = h_trial
                improved = True
                break

        if not improved and no_improve > 150:
            h = best_h + rng.randn(n) * 0.03
            h = project_feasible(h, target)
            no_improve = 0

    return best_h, best_c5


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=80, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(777)
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > 200 and consecutive_no_improve % 80 == 0:
            h = best_h + rng.randn(n) * 0.015
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        current_ncand = max(current_ncand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def multi_shift_cd(h_init, n, target, time_limit, max_sweeps=20000, n_cand=40, n_shifts=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts_arr = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 400:
            break

        top_shifts = np.argsort(corr)[-n_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts_arr)
            update_change_row(change, best_j, h, n, shifts_arr)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def _smooth_obj(x, alpha, n, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


_g_n = 0
_g_target = 0.0


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, n, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 120:
        h_slp, sc_slp = slp_optimize(best_h, current_target,
                                     time_limit=min(200, tleft() * 0.6))
        if sc_slp < best_score:
            best_score = sc_slp
            best_h = h_slp

    if tleft() > 80:
        h_cd, sc_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(100, tleft() * 0.5),
                              max_sweeps=30000,
                              n_cand_start=20, n_cand_end=80,
                              max_no_improve=400)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 100:
        h_slp2, sc_slp2 = slp_optimize(best_h, current_target,
                                       time_limit=min(120, tleft() * 0.5))
        if sc_slp2 < best_score:
            best_score = sc_slp2
            best_h = h_slp2

    if tleft() > 60:
        h_pol, sc_pol = slsqp_round(best_h, current_n, current_target,
                                    [100, 500, 2000, 8000, 30000],
                                    800, 1e-15)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    if tleft() > 250:
        n1 = 640
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_slp1, sc_slp1 = slp_optimize(h1, target1,
                                       time_limit=min(180, tleft() * 0.55))
        if sc_slp1 < best_score:
            best_score = sc_slp1
            best_h = h_slp1
            current_n = n1
            current_target = target1

    if current_n == 640 and tleft() > 80:
        h_cd1, sc_cd1 = pair_cd(best_h, current_n, current_target,
                                time_limit=min(100, tleft() * 0.5),
                                max_sweeps=20000,
                                n_cand_start=20, n_cand_end=70,
                                max_no_improve=400)
        if sc_cd1 < best_score:
            best_score = sc_cd1
            best_h = h_cd1

    if tleft() > 60:
        h_slp3, sc_slp3 = slp_optimize(best_h, current_target,
                                       time_limit=min(80, tleft() * 0.5))
        if sc_slp3 < best_score:
            best_score = sc_slp3
            best_h = h_slp3

    if tleft() > 200:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_slp2, sc_slp2b = slp_optimize(h2, target2,
                                        time_limit=min(150, tleft() * 0.5),
                                        use_all_shifts=False)
        if sc_slp2b < best_score:
            best_score = sc_slp2b
            best_h = h_slp2b
            current_n = n2
            current_target = target2

    if current_n == 800 and tleft() > 60:
        h_cd2, sc_cd2b = pair_cd(best_h, current_n, current_target,
                                 time_limit=min(80, tleft() * 0.5),
                                 max_sweeps=15000,
                                 n_cand_start=15, n_cand_end=60,
                                 max_no_improve=300)
        if sc_cd2b < best_score:
            best_score = sc_cd2b
            best_h = h_cd2

    if tleft() > 150:
        h_ms, sc_ms = multi_shift_cd(best_h, current_n, current_target,
                                     time_limit=min(120, tleft() * 0.6),
                                     max_sweeps=15000, n_cand=40, n_shifts=5)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms

    if tleft() > 200:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.07, 0.12]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_slp_p, sc_slp_p = slp_optimize(h_pert, current_target,
                                             time_limit=min(60, tleft() * 0.4))

            if tleft() > 80:

                                          max_sweeps=15000,

                                          max_no_improve=400)
        if sc_final_cd < best_score:

            best_h = h_final_cd

    if tleft() > 40:
        h_final_pol, sc_final_pol = slsqp_round(best_h, current_n, current_target,
                                                [5000, 20000, 100000],
                                                800, 1e-15)
        if sc_final_pol < best_score:
            best_score = sc_final_pol
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
