```python
def compute_c5(h):

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1

    change = np.zeros((n, n_corr))
    for p in range(n):
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)

def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = idxs + k < n
        wm = idxs - k >= 0
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15, time_limit=None, start_time=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def sa_coordinate_descent(h_init, n, target, time_limit, max_sweeps=30000,
                          n_cand_start=30, n_cand_end=150, n_active=3,
                          max_no_improve=400, init_temp=0.001, cooldown=0.9995):
    """Coordinate descent with simulated annealing and multi-active gradient."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = init_temp
    rng = np.random.RandomState(42)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        # Compute multi-active gradient
        top_idx = np.argsort(corr)[-n_active:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Adaptive candidate set
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-15:
                    best_trial_c5 = float(approx_max) * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            # Simulated annealing: try random non-improving move
            if temp > 1e-8:
                i_rand = rng.randint(n)
                j_rand = rng.randint(n)
                if i_rand != j_rand and h[i_rand] > h[j_rand]:
                    delta_rand = min(h[i_rand] - h[j_rand], 0.1) * rng.random()
                    if delta_rand > 1e-6:
                        h_try = h.copy()
                        h_try[i_rand] -= delta_rand
                        h_try[j_rand] += delta_rand
                        h_try = np.clip(h_try, 0.0, 1.0)
                        h_try = project_feasible(h_try, target)
                        sc_try = compute_c5(h_try)
                        delta_sc = sc_try - c5_val
                        if delta_sc < 0 or (temp > 1e-8 and rng.random() < np.exp(-delta_sc / temp)):
                            h = h_try
                            update_change_row(change, i_rand, h, n, shifts)
                            update_change_row(change, j_rand, h, n, shifts)
                        temp *= cooldown

    return best_h, best_c5

def gradient_polish(h_init, n, target, time_limit, max_steps=50000,
                    base_lr=0.03, n_active=8, lr_decay=0.9999):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.9
                if lr < 1e-7:
                    break

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[0]]
        weights = np.exp(-100.0 * (c_max - corr[top_shifts]))
        ws = np.sum(weights)
        if ws < 1e-30:
            weights = np.ones(len(top_shifts)) / len(top_shifts)
        else:
            weights /= ws

        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-16:
            h = h_try
            cur_c5 = sc
        else:
            lr *= 0.5
            if lr < 1e-8:
                break

    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = idxs + k < n
        wm = idxs - k >= 0
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def time_remaining(start_time, budget_s):
    return max(0.0, budget_s - 30.0 - (time.time() - start_time))


def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def cd_with_restarts(h_init, n, target, time_limit, max_sweeps=30000,
                     n_cand_start=30, n_cand_end=150, n_active=3,
                     max_no_improve=300, init_temp=0.0, cooldown=0.9995,
                     n_restarts=4, restart_noise=0.05):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(12345)

    n_restarts_eff = max(1, n_restarts)
    per_restart_limit = time_limit / n_restarts_eff
    sweep_per_restart = max_sweeps // n_restarts_eff

    for restart in range(n_restarts_eff):
        if time.time() - start > time_limit:
            break
        if restart > 0:
            h = best_h + rng.randn(n) * restart_noise * (restart * 0.5)
            h = project_feasible(h, target)

        consecutive_no_improve = 0
        temp = init_temp
        change = compute_change_matrix(h, n)

        for sweep in range(sweep_per_restart):
            if time.time() - start > time_limit:
                break

            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_val = float(np.max(corr)) * dx

            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1

            if consecutive_no_improve > max_no_improve:
                break

            top_idx = np.argsort(corr)[-n_active:]
            grad = np.zeros(n)
            for s_idx in top_idx:
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += (1.0 - h[idxs[v1] - k])
                grad[v2] -= h[idxs[v2] + k]
            grad -= np.mean(grad)

            progress = min(float(sweep) / max(sweep_per_restart, 1), 1.0)
            current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
            current_ncand = min(current_ncand, n // 3)

            sorted_idx = np.argsort(grad)
            low_set = sorted_idx[:current_ncand]
            high_set = sorted_idx[-current_ncand:]

            best_trial_c5 = c5_val
            best_i = -1
            best_j = -1
            best_delta = 0.0

            n_h = len(high_set)
            n_l = len(low_set)
            for ii in range(n_h):
                i = int(high_set[ii])
                for jj in range(n_l):
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo < 1e-13:
                        continue

                    g = change[j] - change[i]
                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(16):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-15
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-13:
                            b_d = mid_d
                        else:
                            a_d = mid_d

                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-13:
                        continue

                    approx_max = np.max(corr + delta_star * g)
                    if approx_max < best_trial_c5 / dx - 1e-15:
                        best_trial_c5 = float(approx_max) * dx
                        best_i = i
                        best_j = j
                        best_delta = delta_star

            if best_i >= 0:
                h[best_i] -= best_delta
                h[best_j] += best_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, best_i, h, n, shifts)
                update_change_row(change, best_j, h, n, shifts)
            else:
                if temp > 1e-10:
                    i_rand = rng.randint(n)
                    j_rand = rng.randint(n)
                    if i_rand != j_rand and h[i_rand] > h[j_rand]:
                        delta_rand = min(h[i_rand] - h[j_rand], 0.05) * rng.random()
                        if delta_rand > 1e-6:
                            h_try = h.copy()
                            h_try[i_rand] -= delta_rand
                            h_try[j_rand] += delta_rand
                            h_try = np.clip(h_try, 0.0, 1.0)
                            h_try = project_feasible(h_try, target)
                            sc_try = compute_c5(h_try)
                            delta_sc = sc_try - c5_val
                            if delta_sc < 0 or rng.random() < np.exp(-max(delta_sc, 0.0) / temp):
                                h = h_try
                                update_change_row(change, i_rand, h, n, shifts)
                                update_change_row(change, j_rand, h, n, shifts)
                            temp *= cooldown

    return best_h, best_c5


def gradient_polish(h_init, n, target, time_limit, max_steps=50000,
                    base_lr=0.03, n_active=8, lr_min=1e-7):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 4000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.9
                if lr < lr_min:
                    break

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[0]]
        weights = np.exp(-80.0 * (c_max - corr[top_shifts]))
        ws = np.sum(weights)
        if ws < 1e-30:
            weights = np.ones(len(top_shifts)) / len(top_shifts)
        else:
            weights /= ws

        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-16:
            h = h_try
            cur_c5 = sc
        else:
            lr *= 0.5
            if lr < lr_min:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
