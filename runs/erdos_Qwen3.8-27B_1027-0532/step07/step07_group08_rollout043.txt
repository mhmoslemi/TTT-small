    n = len(h)

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def improved_cd(h_init, n, target, time_limit, seed=42):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    change = compute_change_matrix(h, n)

    T = 0.015
    T_min = 0.002
    T_max = 0.12

    consecutive_no_improve = 0
    max_no_improve = 500
    rng = np.random.RandomState(seed)

    for sweep in range(300000):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve < 40:
            T = max(T_min, T * 0.997)
        elif consecutive_no_improve > 150:
            T = min(T_max, T * 1.008)

        c_max = np.max(corr)
        shifted = (corr - c_max) / T
        shifted = np.clip(shifted, -500, 0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros(n_corr)
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights /= w_sum

        grad = change.T @ weights
        grad -= np.mean(grad)

        progress = min(float(sweep) / 40000.0, 1.0)
        n_cand = int(30 + 50 * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_mv = 2

        for _mv in range(n_mv):
            if time.time() - start > time_limit:
                break

            if _mv > 0:
                grad = change.T @ weights
                grad -= np.mean(grad)
                sorted_idx = np.argsort(grad)
                low_set = sorted_idx[:n_cand]
                high_set = sorted_idx[-n_cand:]

            best_trial_c5 = c5_val
            best_i = -1
            best_j = -1
            best_delta = 0.0

            n_h = len(high_set)
            n_l = len(low_set)

            step = max(1, n_cand // 15)
            for ii in range(0, n_h, step):
                i = int(high_set[ii])
                for jj in range(0, n_l, step):
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo < 1e-12:
                        continue

                    g = change[j] - change[i]

                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(10):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-14
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d

                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue

                    approx_max = np.max(corr + delta_star * g) * dx
                    if approx_max < best_trial_c5:
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star

            if best_i >= 0:
                h[best_i] -= best_delta
                h[best_j] += best_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, best_i, h, n)
                update_change_row(change, best_j, h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
            else:
                break

        if consecutive_no_improve > max_no_improve // 3 and sweep % 80 == 0:
            scale = 0.015 + 0.04 * min(1.0, consecutive_no_improve / max_no_improve)
            n_pert = max(5, n // 20)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 0.003:
                h = h_pert
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                consecutive_no_improve = max(0, consecutive_no_improve - 80)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 60:
        deadline = time.time() + min(40, tleft() * 0.4)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [50, 200, 1000, 5000, 20000],
                                600, 1e-14, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 350:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
        current_n = n1
        current_target = target1

        if tleft() > 250:
            deadline = time.time() + min(80, tleft() * 0.3)
            h_pol1, sc_pol1 = slsqp_round(best_h, current_n, current_target,
                                          [200, 1000, 5000, 15000],
                                          700, 1e-14, deadline=deadline)
            if sc_pol1 < best_score:
                best_score = sc_pol1
                best_h = h_pol1

        if tleft() > 180:
            h_cd1, c5_cd1 = improved_cd(best_h, current_n, current_target,
                                         time_limit=min(130, tleft() * 0.55),
                                         seed=seed + 1)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

    if tleft() > 400:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
        current_n = n2
        current_target = target2

        if tleft() > 300:
            deadline = time.time() + min(80, tleft() * 0.25)
            h_pol2, sc_pol2 = slsqp_round(best_h, current_n, current_target,
                                          [500, 2000, 8000, 25000],
                                          900, 1e-14, deadline=deadline)
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_h = h_pol2

        if tleft() > 250:
            h_cd2, c5_cd2 = improved_cd(best_h, current_n, current_target,
                                         time_limit=min(200, tleft() * 0.6),
                                         seed=seed + 2)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    if tleft() > 250:
        h_cd3, c5_cd3 = improved_cd(best_h, current_n, current_target,
                                     time_limit=min(200, tleft() * 0.6),
                                     seed=seed + 3)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 180:
        deadline = time.time() + min(60, tleft() * 0.35)
        h_pol3, sc_pol3 = slsqp_round(best_h, current_n, current_target,
                                       [5000, 15000, 40000],
                                       1000, 1e-14, deadline=deadline)
        if sc_pol3 < best_score:
            best_score = sc_pol3
            best_h = h_pol3

    if tleft() > 180:
        for attempt in range(3):
            if tleft() < 50:
                break
            noise_scale = [0.015, 0.04, 0.07][attempt]
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = improved_cd(h_pert, current_n, current_target,
                                         time_limit=min(45, tleft() * 0.35),
                                         seed=seed + 20 + attempt)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 60:
        deadline = time.time() + min(40, tleft() * 0.5)
        h_final, sc_final = slsqp_round(best_h, current_n, current_target,
                                         [10000, 30000, 80000],
                                         1000, 1e-14, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    if tleft() > 40:
        h_cd_f, c5_f = improved_cd(best_h, current_n, current_target,
                                    time_limit=min(25, tleft() * 0.5),
                                    seed=seed + 99)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
