**New idea: Frank-Wolfe (conditional gradient) on the smooth max approximation**

The problem is:
minimize f(x) = max_s corr_s(x) subject to 0 ≤ x ≤ 1, sum(x) = n/2

The Frank-Wolfe method works as follows:
1. At current point x, compute the linearization: min_{y in feasible set} grad(f(x))^T (y - x)
2. The minimizer y* is found by projecting the gradient onto the feasible set
3. Take a step: x_new = (1-t)*x + t*y*

For our constraint set (simplex-like with box constraints), the projection is:
- We want y that minimizes grad^T y subject to 0 ≤ y ≤ 1, sum(y) = n/2
- This means: set y[i] = 1 for the n/2 indices with the smallest grad[i], y[i] = 0 otherwise
- (That's the bang-bang solution to the linear subproblem)

This is a very different approach from CD and might escape local optima better!

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        step = diff / mask.sum()
        # Don't overshoot
        if diff > 0:
            step = min(step, 1.0 - h[mask].min())
        else:
            step = max(step, h[mask].min())
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(h)

def frank_wolfe(h_start, n, target, time_limit, alphas=(50, 200, 1000, 5000, 20000)):
    """Frank-Wolfe (conditional gradient) for smooth max approximation."""
    start = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        # Number of iterations for this alpha
        n_iter = max(50, int(2000 / (alpha / 50)))
        n_iter = min(n_iter, 5000)
        
        for it in range(n_iter):
            if time.time() - start > time_limit:
                break
            # Compute gradient of smooth max
            grad = smooth_max_grad(h, alpha)
            # Center gradient (since sum constraint means constant shift doesn't matter)
            grad_centered = grad - np.mean(grad)
            
            # Frank-Wolfe linear minimization oracle:
            # Minimize grad_centered^T y subject to 0 <= y <= 1, sum(y) = target
            # Optimal: y[i] = 1 for the `target` smallest grad_centered[i], 0 otherwise
            sorted_idx = np.argsort(grad_centered)
            y = np.zeros(n)
            y[sorted_idx[:int(target)]] = 1.0
            
            # Step size (use backtracking or 2/(t+2) rule)
            t = it + 1
            gamma = 2.0 / (t + 2.0)
            h_new = (1.0 - gamma) * h + gamma * y
            
            # Project to feasible set
            h_new = project_feasible(h_new, target)
            
            # Check improvement
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            
            h = h_new
    
    return best_h, best_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def _cons_fun(x):

def _cons_jac(x):
    return np.ones(_g_n)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def bang_bang_rounding(h, target):
    """Round h to bang-bang (0 or 1) values, maintaining the sum constraint."""
    n = len(h)
    # Sort by value, set top `target` to 1, rest to 0
    # But use a smarter approach: threshold at 0.5, then adjust
    h_bb = (h >= 0.5).astype(float)
    # Adjust to get exactly target ones
    n_ones = int(np.sum(h_bb))
    if n_ones > target:
        # Need to turn off some: turn off the ones with lowest h values
        ones_mask = h_bb == 1.0
        sorted_ones = np.where(ones_mask)[0]
        sorted_by_h = sorted_ones[np.argsort(h[sorted_ones])]
        excess = n_ones - target
        h_bb[sorted_by_h[:excess]] = 0.0
    elif n_ones < target:
        # Need to turn on some: turn on the ones with highest h values
        zeros_mask = h_bb == 0.0
        sorted_zeros = np.where(zeros_mask)[0]
        sorted_by_h = sorted_zeros[np.argsort(-h[sorted_zeros])]
        deficit = target - n_ones
        h_bb[sorted_by_h[:deficit]] = 1.0
    return h_bb

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        step = diff / mask.sum()
        if diff > 0:
            step = min(step, max(0.0, 1.0 - h[mask].min()) + 1e-15)
        else:
            step = max(step, -h[mask].min() + 1e-15)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def smooth_max_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(h)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_grad(x, alpha):
    return smooth_max_grad(x, alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

        for s in active:
            k = s - mid
            v1a = (idxs - k >= 0) & (idxs - k < n)
            v2a = (idxs + k >= 0) & (idxs + k < n)
            grad_active[v1a] += (1.0 - h[idxs[v1a] - k])
            grad_active[v2a] -= h[idxs[v2a] + k]
        combined = 0.4 * grad + 0.6 * grad_active

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        ones_sorted = ones[np.argsort(-combined[ones])]
        zeros_sorted = zeros[np.argsort(combined[zeros])]
        nc = min(n_cand, len(ones_sorted), len(zeros_sorted))
        ones_top = ones_sorted[:nc]
        zeros_top = zeros_sorted[:nc]

        best_new_max = c5_val
        best_i = -1
        best_j = -1

        for i in ones_top:
            for j in zeros_top:
                g = change[j] - change[i]
                new_max = float(np.max(corr + g))
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_i = int(i)
                    best_j = int(j)

        if best_i < 0 and consecutive_no_improve > 50:
            full_nc = min(150, len(ones), len(zeros))
            ones_full = ones_sorted[:full_nc]
            zeros_full = zeros_sorted[:full_nc]
            for i in ones_full:
                for j in zeros_full:
                    g = change[j] - change[i]
                    new_max = float(np.max(corr + g))
                    if new_max < best_new_max - 1e-14:
                        best_new_max = new_max
                        best_i = int(i)
                        best_j = int(j)

        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=150, max_no_improve=500,
            n_binsearch=20, stochastic_frac=0.05, seed=123):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]

        corr_max = corr[k_star_idx]
        threshold = corr_max - 0.08 * corr_max
        active = np.where(corr >= threshold)[0]
        grad_active = np.zeros(n)
        for s in active:
            k = s - mid
            v1a = (idxs - k >= 0) & (idxs - k < n)
            v2a = (idxs + k >= 0) & (idxs + k < n)
            grad_active[v1a] += (1.0 - h[idxs[v1a] - k])
            grad_active[v2a] -= h[idxs[v2a] + k]
        combined = 0.3 * grad + 0.7 * grad_active
        combined -= np.mean(combined)

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(combined)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        n_random = int(stochastic_frac * len(high_set) * len(low_set))
        if n_random > 0:
            for _ in range(n_random):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0 and best_delta > 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        elif consecutive_no_improve % 50 == 0 and consecutive_no_improve > 50:
            idxs_pert = rng.choice(n, size=max(10, n // 15), replace=False)
            h[idxs_pert] += rng.randn(len(idxs_pert)) * 0.03
            h = project_feasible(h, target)
            for idx in idxs_pert:
                update_change_row(change, int(idx), h, n, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Bang-bang CD (fast, exploits structure)
    if tleft() > 150:
        h_bb_init = bang_bang_rounding(best_h, int(target0))
        h_bbc, c5_bbc = bangbang_cd(h_bb_init, n0, target0,
                                     time_limit=min(120, tleft() * 0.35),
                                     max_sweeps=100000, n_cand=100,
                                     seed=seed, perturbation_interval=300)
        if c5_bbc < best_score:
            best_score = c5_bbc
            best_h = h_bbc

    # Phase 2: Frank-Wolfe on smooth max
    if tleft() > 120:
        h_fw, sc_fw = frank_wolfe(best_h, n0, target0,
                                   time_limit=min(100, tleft() * 0.3),
                                   alphas=(50, 200, 1000, 5000, 20000))
        if sc_fw < best_score:
            best_score = sc_fw
            best_h = h_fw

    # Phase 3: Enhanced pair CD with active set gradient
    if tleft() > 150:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                               time_limit=min(180, tleft() * 0.45),
                               max_sweeps=50000,
                               n_cand_start=40, n_cand_end=150,
                               max_no_improve=500,
                               n_binsearch=20,
                               stochastic_frac=0.05, seed=42)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP polish
    if tleft() > 80:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                        [500, 2000, 10000, 50000],
                                        2000, 1e-15)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Phase 5: Bang-bang rounding + bang-bang CD again
    if tleft() > 150:
        h_bb2 = bang_bang_rounding(best_h, int(target0))
        h_bbc2, c5_bbc2 = bangbang_cd(h_bb2, n0, target0,
                                       time_limit=min(130, tleft() * 0.4),
                                       max_sweeps=100000, n_cand=120,
                                       seed=seed + 1000, perturbation_interval=200)
        if c5_bbc2 < best_score:
            best_score = c5_bbc2
            best_h = h_bbc2

    # Phase 6: Another pair CD round with different seed/params
    if tleft() > 130:
        h_cd2, c5_cd2 = pair_cd(best_h, n0, target0,
                                 time_limit=min(120, tleft() * 0.4),
                                 max_sweeps=50000,
                                 n_cand_start=60, n_cand_end=200,
                                 max_no_improve=600,
                                 n_binsearch=22,
                                 stochastic_frac=0.08, seed=137)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 7: Try higher resolution
    if tleft() > 250:
        n1 = int(n0 * 1.5)
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if tleft() > 180:
            h1_bb = bang_bang_rounding(h1, int(target1))
            h1_bbc, c5_1b = bangbang_cd(h1_bb, n1, int(target1),
                                         time_limit=min(120, tleft() * 0.4),
                                         max_sweeps=50000, n_cand=80,
                                         seed=seed + 2000, perturbation_interval=250)
            if c5_1b < best_score:
                best_score = c5_1b
                best_h = h1_bbc
                n0 = n1
                target0 = target1
        if tleft() > 100:
            h1_cd, c5_1c = pair_cd(best_h, n0, target0,
                                    time_limit=min(80, tleft() * 0.35),
                                    max_sweeps=30000,
                                    n_cand_start=30, n_cand_end=100,
                                    max_no_improve=400,
                                    n_binsearch=18,
                                    stochastic_frac=0.03, seed=251)
            if c5_1c < best_score:
                best_score = c5_1c
                best_h = h1_cd

    # Phase 8: Final SLSQP
    if tleft() > 60:
        h_fp, sc_fp = slsqp_optimize(best_h, n0, target0,
                                      [10000, 50000, 200000],
                                      2000, 1e-15)
        if sc_fp < best_score:
            best_score = sc_fp
            best_h = h_fp

    # Phase 9: Final bang-bang polish
    if tleft() > 50:
        h_fb = bang_bang_rounding(best_h, int(target0))
        h_fbc, c5_fbc = bangbang_cd(h_fb, n0, target0,
                                     time_limit=min(40, tleft() * 0.5),
                                     max_sweeps=50000, n_cand=150,
                                     seed=seed + 5000, perturbation_interval=150)
        if c5_fbc < best_score:
            best_score = c5_fbc
            best_h = h_fbc

    # Phase 10: Final CD sweep
    if tleft() > 30:
        h_fcd, c5_fcd = pair_cd(best_h, n0, target0,
                                 time_limit=min(25, tleft() * 0.5),
                                 max_sweeps=20000,
                                 n_cand_start=80, n_cand_end=200,
                                 max_no_improve=200,
                                 n_binsearch=20,
                                 stochastic_frac=0.02, seed=501)
        if c5_fcd < best_score:
            best_score = c5_fcd
            best_h = h_fcd

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
