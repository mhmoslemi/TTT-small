        val_y = smooth_max_val(corr_y, alpha)
        for _ in range(15):
            z = y - step * grad_y
            h_new = project_feasible(z, target)
            corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            val_new = smooth_max_val(corr_new, alpha)
            if val_new <= val_y - 0.5 * step * np.dot(grad_y, z - y):
                break
            step *= 0.5
        else:
            h_new = project_feasible(z, target)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        h = h_new
        t = t_new
        
        # Update best (using actual C5, not smooth max)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Early termination
        if it > 100 and np.linalg.norm(grad_y) < 1e-10 * alpha:
            break
    
    return best_h, best_c5

def pair_cd_annealed(h_init, n, target, time_limit, 
                     max_sweeps=50000, n_cand_start=20, n_cand_end=60,
                     top_k_eval=40, max_no_improve=500,
                     temp_start=0.01, temp_end=0.0):

    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Build change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        # Annealing temperature
        progress = min(float(sweep) / max_sweeps, 1.0)
        temp = temp_start * (temp_end / temp_start) ** progress if temp_start > 0 else 0.0
        
        k_star_idx = int(np.argmax(corr))
        
        # Compute gradient from top shifts
        top_shifts = np.argsort(corr)[-3:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)
        
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]
        
        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            
            g = change[j] - change[i]
            
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(11):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            # Annealing: accept worse moves with some probability
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
            elif temp > 1e-10 and approx_max < c5_val + temp:
                # Accept with some probability (simplified)
                if np.random.random() < np.exp(-(approx_max - c5_val) / max(temp, 1e-10)):
                    if approx_max < best_trial_c5:
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:

                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_optimize(h_start, n, target, alphas, maxiter=600):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'args': (target,)}]
    
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        try:
            res = minimize(_obj_func_alpha, h_cur, 
                          args=(alpha, target),
                          jac=_obj_grad_alpha,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def _cons_fun(x, target):
    return np.sum(x) - target

def _obj_func_alpha(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_val(corr, alpha)

def _obj_grad_alpha(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    weights = smooth_max_weights(corr, alpha)
    return grad_from_weights(x, weights)

            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx

            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
            elif temp > 1e-12:
                delta_obj = approx_max - c5_val
                if delta_obj < temp * 3:
                    accept_prob = np.exp(-delta_obj / max(temp, 1e-12))
                    if rng.random() < accept_prob and approx_max < best_trial_c5 + temp:
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def _cons_fn(x, target):
    return np.sum(x) - target

def _slsqp_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_val(corr, alpha)

def _slsqp_grad(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    weights = smooth_max_weights(corr, alpha)
    return grad_from_weights(x, weights)

def slsqp_optimize(h_start, n, target, alphas, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        cons = [{'type': 'eq', 'fun': _cons_fn, 'args': (target,)}]
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha, target),
                           jac=_slsqp_grad, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def multi_restart_cd(h_init, n, target, time_limit, n_restarts=3, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    for r in range(n_restarts):
        tl = time_limit - (time.time() - start)
        if tl < 60:
            break
        per_restart = tl / (n_restarts - r)

        if r == 0:
            h_try = h_init.copy()
        else:
            scale = 0.05 * (1 + r * 0.5)
            h_try = h_init + rng.randn(n) * scale
            h_try = project_feasible(h_try, target)

        h_res, c5_res = pair_cd_annealed(h_try, n, target, time_limit=min(per_restart * 0.6, 120),
                                          max_sweeps=20000, n_cand_start=25, n_cand_end=65,
                                          top_k_eval=50, max_no_improve=400,
                                          temp_start=0.008, temp_end=0.0, n_shifts_grad=3,
                                          rng=rng)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

        tl = time_limit - (time.time() - start)
        if tl > 30:
            h_pol = slsqp_optimize(best_h, n, target, [1000, 5000, 20000], maxiter=400)
            sc = compute_c5(h_pol)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_pol

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: FISTA at original resolution
    tl = _time_left(start_time, budget_s)
    if tl > 150:
        h_f, c5_f = fista_optimize(best_h, n0, target0,
                                    time_limit=min(120, tl * 0.5),
                                    alpha_start=50, alpha_end=30000,
                                    max_iters=8000)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    # Phase 2: SLSQP at original resolution
    tl = _time_left(start_time, budget_s)
    if tl > 80:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [500, 2000, 8000, 20000], maxiter=600)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 3: Move to n=800
    tl = _time_left(start_time, budget_s)
    if tl > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
        else:
            h1 = best_h.copy()  # stay at original
            sc1 = best_score

        n_cur = len(h1)
        t_cur = n_cur / 2.0

        tl = _time_left(start_time, budget_s)
        if tl > 250:
            h_f2, c5_f2 = fista_optimize(h1, n_cur, t_cur,
                                          time_limit=min(120, tl * 0.4),
                                          alpha_start=100, alpha_end=20000,
                                          max_iters=8000)
            if c5_f2 < best_score:
                best_score = c5_f2
                best_h = h_f2
                n_cur = len(best_h)
                t_cur = n_cur / 2.0

        tl = _time_left(start_time, budget_s)
        if tl > 180:
            h_cd, c5_cd = pair_cd_annealed(h1 if n_cur == n1 else best_h,
                                            n_cur, t_cur,
                                            time_limit=min(150, tl * 0.65),
                                            max_sweeps=40000,
                                            n_cand_start=25, n_cand_end=70,
                                            top_k_eval=50, max_no_improve=800,
                                            temp_start=0.004, temp_end=0.0,
                                            n_shifts_grad=3, rng=rng)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                n_cur = len(best_h)
                t_cur = n_cur / 2.0

        tl = _time_left(start_time, budget_s)
        if tl > 60:
            h_pol2 = slsqp_optimize(best_h, n_cur, t_cur,
                                     [2000, 8000, 20000], maxiter=700)
            sc2 = compute_c5(h_pol2)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

    # Phase 4: Multi-restart at current resolution
    tl = _time_left(start_time, budget_s)
    n_cur = len(best_h)
    t_cur = n_cur / 2.0
    if tl > 200:
        h_mr, c5_mr = multi_restart_cd(best_h, n_cur, t_cur,
                                        time_limit=min(180, tl * 0.7),
                                        n_restarts=3, rng=rng)
        if c5_mr < best_score:
            best_score = c5_mr
            best_h = h_mr

    # Phase 5: Try n=1000
    tl = _time_left(start_time, budget_s)
    if tl > 200:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

        tl = _time_left(start_time, budget_s)
        if tl > 150:
            h_cd2, c5_cd2 = pair_cd_annealed(h2, n2, target2,
                                              time_limit=min(120, tl * 0.65),
                                              max_sweeps=30000,
                                              n_cand_start=25, n_cand_end=65,
                                              top_k_eval=50, max_no_improve=500,
                                              temp_start=0.003, temp_end=0.0,
                                              n_shifts_grad=2, rng=rng)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

        tl = _time_left(start_time, budget_s)
        if tl > 50:
            h_pol3 = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                     [5000, 15000], maxiter=500)
            sc3 = compute_c5(h_pol3)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    # Phase 6: Final FISTA + CD at best resolution
    tl = _time_left(start_time, budget_s)
    n_cur = len(best_h)
    t_cur = n_cur / 2.0
    if tl > 100:
        h_f3, c5_f3 = fista_optimize(best_h, n_cur, t_cur,
                                      time_limit=min(70, tl * 0.5),
                                      alpha_start=500, alpha_end=80000,
                                      max_iters=6000)
        if c5_f3 < best_score:
            best_score = c5_f3
            best_h = h_f3

    tl = _time_left(start_time, budget_s)
    n_cur = len(best_h)
    t_cur = n_cur / 2.0
    if tl > 60:
        h_final_cd, c5_fc = pair_cd_annealed(best_h, n_cur, t_cur,
                                              time_limit=min(45, tl * 0.6),
                                              max_sweeps=20000,
                                              n_cand_start=40, n_cand_end=90,
                                              top_k_eval=70, max_no_improve=300,
                                              temp_start=0.0, temp_end=0.0,
                                              n_shifts_grad=3, rng=rng)
        if c5_fc < best_score:
            best_score = c5_fc
            best_h = h_final_cd

    # Phase 7: Final SLSQP
    tl = _time_left(start_time, budget_s)
    n_cur = len(best_h)
    t_cur = n_cur / 2.0
    if tl > 40:
        h_final = slsqp_optimize(best_h, n_cur, t_cur,
                                  [10000, 40000], maxiter=400)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
