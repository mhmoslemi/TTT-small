    dx = 2.0 / n
    n_corr = 2 * n - 1
    start = time.time()
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    change = compute_change_matrix(h, n)
    shifts = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = np.max(corr) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
        
        # Compute gradient for active shift
        k_star = int(np.argmax(corr)) - (n-1)
        idxs = np.arange(n)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Build candidate pairs
        pairs_i = []
        pairs_j = []
        for i in high_set:
            for j in low_set:
                if i != j:
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo > 1e-12:
                        pairs_i.append(i)
                        pairs_j.append(j)
        
        if not pairs_i:
            continue
        
        pairs_i = np.array(pairs_i)
        pairs_j = np.array(pairs_j)
        delta_los = np.maximum(h[pairs_i] - 1.0, -h[pairs_j])
        delta_his = np.minimum(h[pairs_i], 1.0 - h[pairs_j])

        G = change[pairs_j] - change[pairs_i]  # shape (n_pairs, n_corr)

        a = delta_los.copy()
        b = delta_his.copy()
        for _ in range(14):

            vals = corr[None, :] + mid[:, None] * G  # (n_pairs, n_corr)
            f = vals.max(axis=1)
            active = vals >= f[:, None] - 1e-14
            f_prime = np.where(active.any(axis=1), (G * active).max(axis=1), 0.0)
            # Where f_prime > 0, the function is increasing, so reduce b
            dec = f_prime > 1e-12
            b[dec] = mid[dec]
            a[~dec] = mid[~dec]
        
        optimal_delta = 0.5 * (a + b)

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    change[p, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] = -h[p - shifts[neg_mask]]
    return change

    n_src = len(h_src)

    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_descent(h_init, n, target, time_limit, max_iters=200000,
                        epsilon=5.0, n_active_max=8):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    start = time.time()
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    step_scale = 0.02
    decay = 0.9999

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            step_scale = min(step_scale * 1.05, 0.1)
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                step_scale *= 0.95

        c_max = np.max(corr)
        active_mask = corr >= c_max - epsilon
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > n_active_max:
            active_idx = active_idx[np.argsort(corr[active_idx])[-n_active_max:]]

        subgrad = np.zeros(n)
        for s_idx in active_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            subgrad[v1] += (1.0 - h[idxs[v1] - k])
            subgrad[v2] -= h[idxs[v2] + k]

        subgrad -= np.mean(subgrad)
        grad_norm = np.linalg.norm(subgrad)
        if grad_norm < 1e-12:
            break

        step = step_scale * dx / (grad_norm * dx + 1e-8)
        step = min(step, 0.05)

        h_new = h - step * subgrad
        h_new = project_feasible(h_new, target)

        h = h_new

    return best_h, best_c5

def _lsq_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_refine(h_init, n, target, time_limit, alphas=None, maxiter_per_alpha=400):
    global _g_n, _g_target
    if alphas is None:
        alphas = [200, 1000, 5000]
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    start = time.time()
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(_lsq_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

    n_corr = 2 * n - 1
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(123)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 4000:
            break

        progress = min(step / max(1, max_steps), 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        dlo = max(h[i] - 1.0, -h[j])
        dhi = min(h[i], 1.0 - h[j])
        if dhi - dlo < 1e-12:
            continue

        delta = (dhi - dlo) * rng.rand() + dlo
        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new_c5 = float(np.max(corr + delta * g)) * dx
        delta_c = approx_new_c5 - c5_val

        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            cur_c5 = approx_new_c5
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 30.0 - (time.time() - start_time))

    h_raw = np.asarray(initial_h_values, dtype=float)
    n0 = len(h_raw)
    target0 = n0 / 2.0

    best_h = project_feasible(h_raw, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick vectorized CD at initial resolution
    if tleft() > 80:
        h1, sc1 = vectorized_cd(best_h, n0, target0,
                                time_limit=min(100.0, tleft() * 0.8),
                                n_cand_start=20, n_cand_end=45,
                                max_sweeps=15000, max_no_improve=300)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: Subgradient to break symmetry
    if tleft() > 60:
        h2, sc2 = subgradient_descent(best_h, n0, target0,
                                      time_limit=min(50.0, tleft() * 0.6),
                                      max_iters=100000, epsilon=3.0)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Interpolate to n=800, CD
    if tleft() > 200:
        n2 = 800
        target2 = n2 / 2.0
        h3 = interpolate_to(best_h, n2, target2)
        h3, sc3 = vectorized_cd(h3, n2, target2,
                                time_limit=min(250.0, tleft() * 0.8),
                                n_cand_start=25, n_cand_end=55,
                                max_sweeps=25000, max_no_improve=400)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3
            n0 = n2
            target0 = target2

    # Phase 4: Multi-active CD at n=800
    if tleft() > 100:
        h4, sc4 = multi_active_cd(best_h, n0, target0,
                                  time_limit=min(100.0, tleft() * 0.7),
                                  n_cand=40, n_active=4,
                                  max_sweeps=12000, max_no_improve=300)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Interpolate to n=1600, CD
    if tleft() > 200:
        n5 = 1600
        target5 = n5 / 2.0
        h5 = interpolate_to(best_h, n5, target5)
        h5, sc5 = vectorized_cd(h5, n5, target5,
                                time_limit=min(220.0, tleft() * 0.8),
                                n_cand_start=20, n_cand_end=40,
                                max_sweeps=18000, max_no_improve=350)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5
            n0 = n5
            target0 = target5

    # Phase 6: SA escape at current resolution
    if tleft() > 80:
        n_sa = min(n0, 800)
        h6 = interpolate_to(best_h, n_sa, n_sa / 2.0)
        h6, sc6 = sa_escape(h6, n_sa, n_sa / 2.0,
                            time_limit=min(70.0, tleft() * 0.7),
                            max_steps=15000, t_start=0.025, t_end=0.0002, n_cand=45)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6
            n0 = n_sa
            target0 = target2 if n_sa == 800 else n_sa / 2.0
            target0 = n_sa / 2.0

    # Phase 7: SLSQP refinement at n=800
    if tleft() > 80:
        n7 = 800
        target7 = n7 / 2.0
        h7 = interpolate_to(best_h, n7, target7)
        h7, sc7 = slsqp_refine(h7, n7, target7,
                               time_limit=min(80.0, tleft() * 0.7),
                               alphas=[500, 2000, 8000], maxiter_per_alpha=500)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7
            n0 = n7
            target0 = target7

    # Phase 8: Final CD at best resolution
    if tleft() > 50:
        h8, sc8 = vectorized_cd(best_h, n0, target0,
                                time_limit=min(60.0, tleft() * 0.8),
                                n_cand_start=30, n_cand_end=55,
                                max_sweeps=10000, max_no_improve=200)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: Final SLSQP polish
    if tleft() > 40:
        n9 = min(n0, 800)
        target9 = n9 / 2.0
        h9 = interpolate_to(best_h, n9, target9)
        h9, sc9 = slsqp_refine(h9, n9, target9,
                               time_limit=min(35.0, tleft() * 0.8),
                               alphas=[8000, 30000], maxiter_per_alpha=400)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9
            n0 = n9
            target0 = target9

    # Final: rapid CD to squeeze out last improvements
    if tleft() > 20:
        hf, scf = vectorized_cd(best_h, n0, target0,
                                time_limit=min(20.0, tleft() * 0.8),
                                n_cand_start=40, n_cand_end=60,
                                max_sweeps=5000, max_no_improve=100)
        if scf < best_score:
            best_score = scf
            best_h = hf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
