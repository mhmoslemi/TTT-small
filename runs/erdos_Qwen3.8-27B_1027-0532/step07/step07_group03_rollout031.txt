import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new

def compute_corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad

        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def bundle_method(h_init, n, target, time_limit, max_iters=5000):
    """Bundle method for min max of linear approximations."""
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    bundle_g = []
    bundle_rhs = []  # intercept terms
    
    max_bundle = 200
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
        
        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])
        g = compute_corr_gradient(h, k_star)
        
        # q_i(h) = f_i + g_i^T (h - h_i) is a linear under-estimator of f_i(h)
        # We want: min_t max_i (f_i + g_i^T (h - h_i))
        # Linearize: g_i^T h - g_i^T h_i + f_i <= t
        # i.e., g_i^T h - t <= g_i^T h_i - f_i
        
        bundle_g.append(g)
        # rhs = g^T h_i - f_i (we need g^T h - t <= rhs for all i)
        bundle_rhs.append(float(g @ h) - f_val)
        
        if len(bundle_g) > max_bundle:
            keep = max_bundle // 2
            bundle_g = [bundle_g[0]] + bundle_g[-(keep-1):]
            bundle_rhs = [bundle_rhs[0]] + bundle_rhs[-(keep-1):]
        
        n_b = len(bundle_g)
        # Variables: h (n) + t (1)
        # Minimize t
        # Subject to:
        #   g_i^T h - t <= rhs_i for all i
        #   sum(h) = target
        #   0 <= h_i <= 1, t >= 0
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_ub = np.zeros((n_b, n + 1))
        b_ub = np.zeros(n_b)
        for i in range(n_b):
            A_ub[i, :n] = bundle_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = bundle_rhs[i]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad

def compute_corr_gradient_multi(h, k_stars, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(k_stars, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

def cutting_plane_method(h_init, n, target, time_limit, max_iters=50000, tol=1e-10):

            cuts_h = cuts_h[-keep:]
            no_improve = 0

        f_val = float(corr[k_star])

        threshold = corr[k_star] - 1e-8

            cuts_h = [cuts_h[0]] + cuts_h[-(keep-1):]

            if not res.success:

                if not res.success:
                    break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            
            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

def compute_corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad

def compute_corr_gradient_multi(h, k_stars, weights):
    n = len(h)

    idxs = np.arange(n)

        grad[v1] += w * (1.0 - h[idxs[v1] + k])

                if sc800_cd < best_score:
                    best_score = sc800_cd
                    best_h = h800_cd

    if tleft() > 400:
        n2 = 1200
        target2 = n2 / 2.0
        h1200 = interpolate_to(best_h, n2, target2)
        sc1200 = compute_c5(h1200)
        if sc1200 < best_score:
            best_score = sc1200
            best_h = h1200
            best_n = n2
            best_target = target2
        if tleft() > 320:
            h1200_cp, sc1200_cp = cutting_plane_method(h1200, n2, target2,
                                                        time_limit=min(250, tleft() * 0.5),
                                                        max_iters=30000, tol=1e-10)
            if sc1200_cp < best_score:
                best_score = sc1200_cp
                best_h = h1200_cp
            if tleft() > 130:
                h1200_cd, sc1200_cd = interceptive_cd(best_h, best_n, best_target,
                                                       time_limit=min(100, tleft() * 0.4),
                                                       max_sweeps=20000,
                                                       n_cand_start=40, n_cand_end=120,
                                                       max_no_improve=400)
                if sc1200_cd < best_score:
                    best_score = sc1200_cd
                    best_h = h1200_cd

    if tleft() > 60:
        h_cd2, sc_cd2 = interceptive_cd(best_h, best_n, best_target,
                                         time_limit=min(50, tleft() * 0.6),
                                         max_sweeps=30000,
                                         n_cand_start=50, n_cand_end=140,
                                         max_no_improve=300)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    if tleft() > 40:
        h_sls2, sc_sls2 = slsqp_round(best_h, best_n, best_target,
                                       [5000, 20000], 700, 1e-14)
        if sc_sls2 < best_score:
            best_score = sc_sls2
            best_h = h_sls2

    if tleft() > 60:
        rng = np.random.RandomState(seed)
        for ns in [0.01, 0.03]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(best_n) * ns
            h_pert = project_feasible(h_pert, best_target)
            h_ref, sc_ref = cutting_plane_method(h_pert, best_n, best_target,
                                                 time_limit=min(30, tleft() * 0.3),
                                                 max_iters=8000, tol=1e-9)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 20:
        h_cd3, sc_cd3 = interceptive_cd(best_h, best_n, best_target,
                                         time_limit=min(15, tleft() * 0.5),
                                         max_sweeps=15000,
                                         n_cand_start=60, n_cand_end=150,
                                         max_no_improve=200)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
