- Compute the "expected" gradient: E_λ[∇g_k(h)]
- Take a step in the negative gradient direction
- Update λ based on which constraints are most violated

            grad[v2] -= lam[m] * h[idxs[v2] - k]

        h_new = project_feasible(h_new, target)

        # Increase weight of constraints that are most violated

        lam_new = lam * np.exp(eta * corr_new / max(1e-10, corr_new.max()))
        lam_new = lam_new / lam_new.sum()
        lam = lam_new
        
        h = h_new
        h = project_feasible(h, target)
    
    return best_h, best_score

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def get_gradient_at_shift(h, k):
    n = len(h)
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]

Actually, one more important consideration: the `np.correlate` function. Let me verify the indexing.

np.correlate(a, v, mode='full') computes:
c[k] = sum_i a[i+k] * conj(v[i])

where the output has length len(a) + len(v) - 1.

For a = h (length n) and v = 1-h (length n):
c[k] = sum_i h[i+k] * (1-h[i]) for i where both indices are valid

The output has length 2n-1, indexed from 0 to 2n-2.
c[k] corresponds to shift (k - (n-1)).

So c[n-1] = sum_i h[i] * (1-h[i]) = sum_i h[i] - sum_i h[i]^2 = n/2 - sum h[i]^2

And c[n-1 + s] = sum_i h[i+s] * (1-h[i]) for appropriate i, which is the overlap at shift s.

OK so the C₅ value is max(c) * dx where dx = 2/n.

import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def get_gradient_at_shift(h, k):
    n = len(h)
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad


def cutting_plane_v2(h_init, n, target, time_limit, max_iters=50000, max_cuts=250):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    cuts_g = []
    cuts_f = []
    cuts_h = []
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 8000 and len(cuts_g) > 30:
            cuts_g = cuts_g[:5] + cuts_g[-25:]
            cuts_f = cuts_f[:5] + cuts_f[-25:]
            cuts_h = cuts_h[:5] + cuts_h[-25:]
            no_improve = 0
        k_max = int(np.argmax(corr))
        corr_max = corr[k_max]
        if no_improve > 3000:
            base_tol = 0.01 * corr_max
        elif no_improve > 1500:
            base_tol = 0.005 * corr_max
        elif no_improve > 500:
            base_tol = 0.002 * corr_max
        else:
            base_tol = max(1e-7, 0.001 * corr_max)
        active_mask = corr >= corr_max - base_tol
        active_lags = np.where(active_mask)[0]
        max_new = min(len(active_lags), 8)
        if len(active_lags) > max_new:
            sorted_act = active_lags[np.argsort(corr[active_lags])[-max_new:]]
        else:
            sorted_act = active_lags
        for la in sorted_act:
            k = int(la) - mid
            g = get_gradient_at_shift(h, k)
            cuts_g.append(g)
            cuts_f.append(float(corr[la]))
            cuts_h.append(h.copy())
        if len(cuts_g) > max_cuts:
            keep_old = min(8, len(cuts_g) // 5)
            keep_new = max_cuts - keep_old
            cuts_g = cuts_g[:keep_old] + cuts_g[-keep_new:]
            cuts_f = cuts_f[:keep_old] + cuts_f[-keep_new:]
            cuts_h = cuts_h[:keep_old] + cuts_h[-keep_new:]
        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                continue
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            t_val = float(res.x[n])
            gap = abs(sc / dx - t_val)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
            if gap < 1e-10:
                break
        except Exception:
            continue
    return best_h, best_score


def swap_local_search(h_init, n, target, time_limit,
                      n_cand_start=15, n_cand_end=60,
                      max_sweeps=50000, max_no_improve=4000):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    idxs = np.arange(n)
    s_off = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = get_gradient_at_shift(h, k_star)
        grad -= np.mean(grad)
        corr_sorted = np.sort(corr)[::-1]
        if len(corr_sorted) > 1 and corr_sorted[1] >= corr_sorted[0] - 0.005 * corr_sorted[0]:
            k2_idx = int(np.argsort(corr)[::-1][1])
            k2 = k2_idx - mid
            grad2 = get_gradient_at_shift(h, k2)
            grad = 0.5 * grad + 0.5 * grad2
            grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g_vec = np.zeros(2 * n - 1)
                for s in range(-(n - 1), n):
                    val = 0.0
                    if 0 <= i + s < n:
                        val -= (1.0 - h[i + s])
                    if 0 <= i - s < n:
                        val += h[i - s]
                    if 0 <= j + s < n:
                        val += (1.0 - h[j + s])
                    if 0 <= j - s < n:
                        val -= h[j - s]
                    g_vec[s + s_off] = val
                a_d, b_d = delta_lo, delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g_vec[active])
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            no_improve += 5
    return best_h, best_c5


def block_local_search(h_init, n, target, time_limit,
                       block_size=3, max_sweeps=15000, max_no_improve=2000):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    s_off = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = get_gradient_at_shift(h, k_star)
        grad -= np.mean(grad)
        bs = block_size
        n_blocks = n - bs + 1
        block_scores = np.zeros(n_blocks)
        for p in range(n_blocks):
            block_scores[p] = grad[p:p + bs].sum()
        top_k = 12
        bot_k = 12
        top_blocks = np.argsort(block_scores)[-top_k:]
        bot_blocks = np.argsort(block_scores)[:bot_k]
        best_trial = c5_val
        best_move = None
        for bi in top_blocks:
            for bj in bot_blocks:
                if abs(bi - bj) < bs:
                    continue
                g_vec = np.zeros(2 * n - 1)
                for s in range(-(n - 1), n):
                    val = 0.0
                    for p in range(bi, bi + bs):
                        if 0 <= p + s < n:
                            val -= (1.0 - h[p + s])
                        if 0 <= p - s < n:
                            val += h[p - s]
                    for p in range(bj, bj + bs):
                        if 0 <= p + s < n:
                            val += (1.0 - h[p + s])
                        if 0 <= p - s < n:
                            val -= h[p - s]
                    g_vec[s + s_off] = val
                max_delta = min(h[bi:bi + bs].min(), (1.0 - h[bj:bj + bs].max())) * bs
                if max_delta < 1e-12:
                    continue
                a_d, b_d = 0.0, max_delta
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + (mid_d / bs) * g_vec
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max((g_vec / bs)[active])
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue
                approx_max = np.max(corr + (delta_star / bs) * g_vec) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_move = (bi, bj, delta_star)
        if best_move is not None:
            bi, bj, delta = best_move
            h[bi:bi + bs] -= delta / bs
            h[bj:bj + bs] += delta / bs
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            no_improve += 3
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_subgradient(h_init, n, target, time_limit, max_iters=100000):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    h_curr = h_init.copy().astype(float)
    h_prev = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)
    momentum = 0.85

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * (it + 1))) / 2.0
        y = h_curr + ((t_k - 1.0) / t_k) * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        alpha = 1.0 / np.sqrt(it + 1) * 2.0
        corr_y = compute_corr(y)
        k_star = int(np.argmax(corr_y))
        k = k_star - mid
        grad = get_gradient_at_shift(y, k)
        grad -= np.mean(grad)
        h_new = y - alpha * grad / max(1.0, np.linalg.norm(grad))
        h_new = project_feasible(h_new, target)
        h_prev = h_curr.copy()
        h_curr = h_new
        sc = compute_c5(h_curr)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_curr.copy()
    return best_h, best_score

        if sc_final < best_score:
            best_score = sc_final
            best_h = h_sls3

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
