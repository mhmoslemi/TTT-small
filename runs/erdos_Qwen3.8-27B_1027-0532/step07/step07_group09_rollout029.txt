    """
    FISTA minimization of smooth approximation of C5 + penalty for constraint.
    """
    n = len(h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h.copy()
    y = h.copy()
    t_k = 1.0
    L = alpha_smooth + penalty_lambda / n  # Lipschitz constant estimate
    
    for k in range(n_steps):
        h = y.copy()
        
        # Compute gradient of smooth objective
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha_smooth * (corr - c_max), -500.0, 0.0)
        softmax_weights = np.exp(shifted) / np.sum(np.exp(shifted))
        
        grad = np.zeros(n)
        for ki, wi in enumerate(softmax_weights):
            if wi < 1e-10:
                continue
            k_shift = int(ki) - mid
            v1 = (idxs - k_shift >= 0) & (idxs - k_shift < n)
            v2 = (idxs + k_shift >= 0) & (idxs + k_shift < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k_shift])
            grad[v2] -= wi * h[idxs[v2] + k_shift]
        
        # Add penalty gradient
        grad += 2 * penalty_lambda * (h - target / n)
        
        # FISTA update
        step = 1.0 / L
        y_new = np.maximum(0.0, np.minimum(1.0, y - step * grad))
        
        # Project onto sum constraint (approximately)
        # Actually, the penalty handles this, so we just clip
        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t_k**2))
        y = y_new + ((t_k - 1) / t_new) * (y_new - h)
        h = y_new
        t_k = t_new

            h = h_new

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    p_idx = np.arange(n)[:, None]
    k_idx = shifts[None, :]
    pk = p_idx + k_idx
    pmk = p_idx - k_idx
    valid_pk = (pk >= 0) & (pkg := pk) & (pk < n)
    valid_pmk = (pmk >= 0) & (pmk < n)
    h_at_pk = np.zeros((n, n_corr))
    h_at_pk[valid_pk] = h[pk[valid_pk]]
    h_at_pmk = np.zeros((n, n_corr))
    h_at_pmk[valid_pmk] = h[pmk[valid_pmk]]
    change = valid_pk.astype(float) - h_at_pk - h_at_pmk
    return change


def update_change_row(change, p, h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    pk = p + shifts
    pmk = p - shifts
    valid_pk = (pk >= 0) & (pk < n)
    valid_pmk = (pmk >= 0) & (pmk < n)
    change[p, :] = 0.0
    change[p, valid_pk] += (1.0 - h[pk[valid_pk]])
    change[p, valid_pmk] -= h[pmk[valid_pmk]]


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def fista_optimize(h_init, n, target, time_limit, alphas, steps_per_alpha):
    start = time.time()
    h = h_init.copy().astype(float)
    penalty = 3.0
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    mid = n - 1
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        y = h.copy()
        t_k = 1.0
        L = alpha * 2.5 + penalty + 1.0
        step_size = 1.0 / L
        for step in range(steps_per_alpha):
            if time.time() - start > time_limit:
                break
            corr = np.correlate(y, 1.0 - y, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            exp_vals = np.exp(shifted)
            sum_exp = np.sum(exp_vals)
            w = exp_vals / sum_exp
            grad = np.zeros(n)
            for ki in range(n_corr):
                wi = w[ki]
                if wi < 1e-10:
                    continue
                k = int(ki) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[idxs[v1]] += wi * (1.0 - y[idxs[v1] - k])
                grad[idxs[v2]] -= wi * y[idxs[v2] + k]
            grad += penalty * (y - target / n)
            h_new = np.clip(y - step_size * grad, 0.0, 1.0)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
            h = h_new
            t_k = t_new
        h = project_feasible(h, target)
    return h
