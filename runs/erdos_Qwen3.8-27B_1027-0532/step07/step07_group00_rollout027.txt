def slp_iteration(h0, n, dx):
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    
    # Compute A matrix (n_corr x n)
    A = np.zeros((n_corr, n))
    for k_idx in range(n_corr):
        k = k_idx - mid
        # A_k(i) = (1-h0[i+k]) if i+k in [0,n) minus h0[i-k] if i-k in [0,n)
        for i in range(n):
            if 0 <= i+k < n:
                A[k_idx, i] += (1 - h0[i+k])
            if 0 <= i-k < n:
                A[k_idx, i] -= h0[i-k]
    
    # Compute RHS for each constraint
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    rhs = np.zeros(n_corr)
    for k_idx in range(n_corr):
        rhs[k_idx] = corr0[k_idx] - A[k_idx] @ h0  # constant term
    
    # LP: minimize t
    # subject to: A @ h - (1/dx) * t <= rhs  (for each k)
    #             sum(h) = n/2
    #             0 <= h <= 1
    
    # Variables: [h (n vars), t (1 var)]
    # Objective: [0, ..., 0, 1] (minimize t)
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A @ h - (1/dx) * t <= rhs
    A_ub = np.hstack([A, -np.ones((n_corr, 1)) / dx])
    b_ub = rhs
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h[i] <= 1, t unbounded below (but we're minimizing so it'll be positive)
    bounds = [(0, 1)] * n + [(None, None)]
    
    from scipy.optimize import linprog
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        h_new = result.x[:n]
        t_new = result.x[n] * dx
        return h_new, t_new
    return h0, float('inf')

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def weighted_pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
                     n_cand_start=20, n_cand_end=70, max_no_improve=350,
                     weight_beta=5.0, use_anneal=False, T0=0.0003):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(12345)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        T = T0 * 0.01 ** progress if use_anneal else 0.0

        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        grad = np.zeros(n)
        top_k_shifts = min(40, n_corr)
        top_idx = np.argsort(weights)[-top_k_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        found = False
        for ii in range(n_h):
            if found:
                break
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    found = True
                    break
                elif T > 1e-12 and approx_max - c5_val < T * 3:
                    if rng.random() < np.exp(-(approx_max - c5_val) / T):
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
                        found = True
                        break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def adaptive_gradient_cd(h_init, n, target, time_limit, max_iter=20000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = compute_change_matrix(h, n)

    top_k_schedule = [1, 2, 3, 5, 8, 15, 30, 50, 100]
    k_idx = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        c_max = float(np.max(corr))
        threshold = c_max * 0.95
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        n_active = len(active)

        grad = np.zeros(n)
        for s_idx in active:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(n_active, 1)
        grad -= np.mean(grad)

        n_cand = min(50, n // 4)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        found = False

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            if found:
                break
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = d_lo
                b_d = d_hi
                for _ in range(16):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g
                    f_m = np.max(vals)
                    act = vals >= f_m - 1e-14
                    fp = np.max(g[act]) if act.any() else 0.0
                    if fp > 1e-12:
                        b_d = m_d
                    else:
                        a_d = m_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    found = True
                    break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            no_improve += 50

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_perturb_search(h_start, n, target, time_limit, n_starts=3, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_start.copy()
    best_c = compute_c5(h_start)

    for s in range(n_starts):
        if time.time() - start > time_limit - 30:
            break
        noise_scales = [0.005, 0.02, 0.05, 0.1]
        ns = noise_scales[s % len(noise_scales)]
        h_pert = best_h + rng.randn(n) * ns
        h_pert = project_feasible(h_pert, target)
        rem = max(10, time_limit - (time.time() - start) - 30)
        h_s, c_s = slp_optimize(h_pert, n, time_limit=rem * 0.6, max_iter=200)
        if c_s < best_c:
            best_c = c_s
            best_h = h_s
    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLP at initial resolution (main workhorse)
    if tleft() > 200:
        h_s, c_s = slp_optimize(best_h, n0, time_limit=min(250, tleft() * 0.7),
                                max_iter=800, init_damping=0.4)
        if c_s < best_score:
            best_score = c_s
            best_h = h_s

    # Phase 2: Adaptive gradient CD
    if tleft() > 150:
        h_ag, c_ag = adaptive_gradient_cd(best_h, n0, target0,
                                          time_limit=min(200, tleft() * 0.6),
                                          max_iter=20000)
        if c_ag < best_score:
            best_score = c_ag
            best_h = h_ag

    # Phase 3: SLSQP polish
    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [200, 1000, 5000, 20000],
                                500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: SLP again (second round, often finds further improvement)
    if tleft() > 150:
        h_s2, c_s2 = slp_optimize(best_h, n0, target0,
                                  time_limit=min(180, tleft() * 0.6),
                                  max_iter=600, init_damping=0.3)
        if c_s2 < best_score:
            best_score = c_s2
            best_h = h_s2

    # Phase 5: Weighted pair CD
    if tleft() > 120:
        h_cd, c5_cd = weighted_pair_cd(best_h, n0, target0,
                                       time_limit=min(150, tleft() * 0.55),
                                       max_sweeps=25000,
                                       n_cand_start=20, n_cand_end=60,
                                       max_no_improve=300,
                                       weight_beta=6.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Interpolate to n=1000 and optimize
    n1 = 1000
    target1 = n1 / 2.0
    if tleft() > 350:
        h1 = interpolate_to(best_h, n1, target1)
        best_h_n1 = h1
        best_score_n1 = compute_c5(h1)

        if tleft() > 280:
            h_s1, c_s1 = slp_optimize(h1, n1, time_limit=min(200, tleft() * 0.6),
                                      max_iter=500, init_damping=0.4)
            if c_s1 < best_score_n1:
                best_score_n1 = c_s1
                best_h_n1 = h_s1

        if tleft() > 200:
            h_ag1, c_ag1 = adaptive_gradient_cd(best_h_n1, n1, target1,
                                                time_limit=min(150, tleft() * 0.5),
                                                max_iter=15000)
            if c_ag1 < best_score_n1:
                best_score_n1 = c_ag1
                best_h_n1 = h_ag1

        if tleft() > 120:
            h_pol1, sc1 = slsqp_round(best_h_n1, n1, target1,
                                      [500, 3000, 10000],
                                      400, 1e-14)
            if sc1 < best_score_n1:
                best_score_n1 = sc1
                best_h_n1 = h_pol1

        if tleft() > 80:
            h_cd1, c5_cd1 = weighted_pair_cd(best_h_n1, n1, target1,
                                             time_limit=min(100, tleft() * 0.5),
                                             max_sweeps=15000,
                                             n_cand_start=20, n_cand_end=55,
                                             max_no_improve=250,
                                             weight_beta=6.0)
            if c5_cd1 < best_score_n1:
                best_score_n1 = c5_cd1
                best_h_n1 = h_cd1

        if best_score_n1 < best_score:
            best_score = best_score_n1
            best_h = best_h_n1

    # Phase 7: Multi-perturbation SLP search
    if tleft() > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_mp, c_mp = multi_perturb_search(best_h, n_cur, target_cur,
                                          time_limit=min(120, tleft() * 0.7),
                                          n_starts=4, seed=seed)
        if c_mp < best_score:
            best_score = c_mp
            best_h = h_mp

    # Phase 8: Final SLP polish
    if tleft() > 80:
        n_cur = len(best_h)
        h_fin_s, c_fin_s = slp_optimize(best_h, n_cur,
                                        time_limit=min(80, tleft() * 0.7),
                                        max_iter=300, init_damping=0.2)
        if c_fin_s < best_score:
            best_score = c_fin_s
            best_h = h_fin_s

    # Phase 9: Final SLSQP
    if tleft() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin, sc_fin = slsqp_round(best_h, n_cur, target_cur,
                                    [10000, 50000],
                                    400, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 10: Final pair CD
    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fcd, c_fcd = weighted_pair_cd(best_h, n_cur, target_cur,
                                        time_limit=min(25, tleft() * 0.7),
                                        max_sweeps=10000,
                                        n_cand_start=30, n_cand_end=70,
                                        max_no_improve=150,
                                        weight_beta=8.0)
        if c_fcd < best_score:
            best_score = c_fcd
            best_h = h_fcd

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
