- The negative part for shift k: h shifted left by k, masked to [0, n)

Actually, the gradient with respect to x (the first n-1 variables) is:
- For i < n-1: ∂f/∂x[i] = ∂f/∂h[i] - ∂f/∂h[n-1]

- It uses quasi-Newton approximation which is cheaper than SLSQP's
- It's generally faster per iteration

Let me also add a **subgradient projection method** as a complementary approach:

- Take a step in the negative subgradient direction

```python
import numpy as np
from scipy.optimize import minimize, minimize_lsq
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def project_sum_only(h, target_sum):
    """Project onto sum(h) = target_sum without box constraints."""
    return h + (target_sum - np.sum(h)) / len(h)

def lbfobs_smooth(x, n, target, alpha):
    """Smooth log-sum-exp objective with parameterization for sum constraint."""
    h = np.empty(n)
    h[:n-1] = x
    h[n-1] = target - np.sum(x)
    if h[n-1] < -1e-6 or h[n-1] > 1.0 + 1e-6:
        # Penalty for infeasibility
        penalty = 100.0 * ((max(0, -h[n-1]))**2 + (max(0, h[n-1]-1))**2)
        h = np.clip(h, 0.0, 1.0)
    else:
        penalty = 0.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return c_max + log_sum / alpha + penalty

def lbfobs_smooth_grad(x, n, target, alpha):
    h = np.empty(n)
    h[:n-1] = x
    h[n-1] = target - np.sum(x)
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    weights /= wsum
    
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad_full = np.zeros(n)
    for k_idx in range(n_corr):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        w = weights[k_idx]
        v2 = (idxs + k >= 0) & (idxs + k < n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        grad_full[v2] += w * (1.0 - h[idxs[v2] + k])
        grad_full[v1] -= w * h[idxs[v1] - k]
    
    # Gradient with respect to x (first n-1 variables)
    # h[n-1] = target - sum(x), so dh[n-1]/dx[i] = -1
    grad = grad_full[:n-1].copy() - grad_full[n-1]
    return grad

def lbfgs_round(h_start, n, target, alphas, maxiter_per_alpha=5000):
    """Run L-BFGS-B with parameterization, for increasing alpha."""
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    x = h_cur[:n-1].copy()
    
    for alpha in alphas:
        bounds_lbf = [(0.0, 1.0)] * (n - 1)
        try:
            res = minimize(lbfobs_smooth, x, jac=lbfobs_smooth_grad,
                          args=(n, target, alpha),
                          method='L-BFGS-B', bounds=bounds_lbf,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-20, 'gtol': 1e-15})
            x_new = res.x
            h_new = np.empty(n)
            h_new[:n-1] = x_new
            h_new[n-1] = target - np.sum(x_new)
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
                x = h_cur[:n-1].copy()
        except Exception:
            pass
    return h_cur, cur_score

def nesterov_subgradient(h_init, n, target, time_limit, n_active=10, 
                          base_lr=0.01, momentum=0.9):
    """Nesterov-accelerated subgradient method for the smooth max."""
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    v = np.zeros(n)  # momentum variable
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = base_lr
    no_improve = 0
    
    for step in range(500000):
        if time.time() - start > time_limit:
            break
        if no_improve > 10000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Soft max weights over top n_active shifts
        sorted_idx = np.argsort(corr)[::-1]
        top_k = sorted_idx[:n_active]
        c_top = corr[top_k[0]]
        w = np.zeros(2*n-1)
        for idx in top_k:
            w[idx] = np.exp(-10.0 * (c_top - corr[idx]))
        wsum = np.sum(w)
        if wsum < 1e-15:
            w[top_k[0]] = 1.0
        else:
            w /= wsum
        
        # Compute gradient
        grad = np.zeros(n)
        for k_idx in top_k:
            wi = w[k_idx]
            if wi < 1e-10:
                continue
            k = int(k_idx) - mid
            v2 = (idxs + k >= 0) & (idxs + k < n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            grad[v2] += wi * (1.0 - h[idxs[v2] + k])
            grad[v1] -= wi * h[idxs[v1] - k]
        
        # Center the gradient (remove mean to preserve sum constraint)
        grad -= np.mean(grad)
        
        # Nesterov step
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        # Check improvement, reduce lr if needed
        sc = compute_c5(h_new)
        if sc < c5_val + 1e-12:  # allow slight increase for momentum
            h = h_new
        else:
            lr *= 0.95
            v *= 0.5
            if lr < 1e-10:
                break
    
    return best_h, best_c5

def pair_cd_fast(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand=60, patience=300):
    """Fast pairwise coordinate descent with change matrix."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    def update_row(p):
        change[p, :] = 0.0
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            break
        
        # Focus on top shifts
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        # Compute gradient at the active shift
        grad = np.zeros(n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        grad[v2] += (1.0 - h[idxs[v2] + k_star])
        grad[v1] -= h[idxs[v1] - k_star]
        grad -= np.mean(grad)
        
        # Candidate selection
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        nh = len(high_set)
        nl = len(low_set)
        
        for ii in range(nh):
            i = int(high_set[ii])
            hi = h[i]
            for jj in range(nl):
                j = int(low_set[jj])
                if i == j:
                    continue
                hj = h[j]
                delta_lo = max(-hj + 1e-15, -1e-15)  # h[i] decreases, h[j] increases
                delta_hi = min(hi - 1e-15, 1.0 - hj + 1e-15)
                if delta_hi - delta_lo < 1e-14:
                    continue
                
                g = change[j] - change[i]
                
                # Binary search for optimal delta
                a_d, b_d = delta_lo, delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-14:
                    continue
                
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_trial - 1e-15:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_row(best_i)
            update_row(best_j)
    
    return best_h, best_c5

def top_k_focus_slsqp(h_start, n, target, top_k, alpha, maxiter=3000):
    """SLSQP that only smooths over the top-k shifts."""
    h = h_start.copy()
    cur_score = compute_c5(h)
    
    def obj(x):
        hh = project_sum_only(x, target)
        hh = np.clip(hh, 0, 1)
        corr = np.correlate(hh, 1.0 - hh, mode='full')
        # Get top k
        sorted_indices = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[sorted_indices]
        c_max = top_vals[0]
        shifted = np.clip(alpha * (top_vals - c_max), -600.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def obj_grad(x):
        hh = np.clip(project_sum_only(x, target), 0, 1)
        corr = np.correlate(hh, 1.0 - hh, mode='full')
        sorted_indices = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[sorted_indices]
        c_max = top_vals[0]
        shifted = np.clip(alpha * (top_vals - c_max), -600.0, 0.0)
        w_top = np.exp(shifted)
        wsum = np.sum(w_top)
        
        idxs = np.arange(n)
        mid = n - 1
        grad = np.zeros(n)
        for si in range(top_k):
            k_idx = sorted_indices[si]
            wi = w_top[si] / wsum
            k = int(k_idx) - mid
            v2 = (idxs + k >= 0) & (idxs + k < n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            grad[v2] += wi * (1.0 - hh[idxs[v2] + k])
            grad[v1] -= wi * hh[idxs[v1] - k]
        return grad
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            return h_new, sc
    except Exception:
        pass
    return h, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def lbf_obs(x, n, target, alpha):
    h = np.empty(n)
    h[:n - 1] = x
    h[n - 1] = target - np.sum(x)
    if h[n - 1] < -0.05 or h[n - 1] > 1.05:
        pen = 50.0 * (max(0.0, -h[n - 1]) ** 2 + max(0.0, h[n - 1] - 1.0) ** 2)
    else:
        pen = 0.0
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / alpha + pen)


def lbf_obs_grad(x, n, target, alpha):
    h = np.empty(n)
    h[:n - 1] = x
    h[n - 1] = target - np.sum(x)
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-15:
        return np.zeros(n - 1)
    weights /= wsum
    mid = n - 1
    idxs = np.arange(n)
    grad_full = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        w = weights[k_idx]
        v2 = (idxs + k >= 0) & (idxs + k < n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        grad_full[v2] += w * (1.0 - h[idxs[v2] + k])
        grad_full[v1] -= w * h[idxs[v1] - k]
    grad = grad_full[:n - 1].copy() - grad_full[n - 1]
    return grad


def lbfgs_round(h_start, n, target, alphas, maxiter=4000):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    x = h_cur[:n - 1].copy()
    bounds_lbf = [(0.0, 1.0)] * (n - 1)
    for alpha in alphas:
        try:
            res = minimize(lbf_obs, x, jac=lbf_obs_grad,
                           args=(n, target, alpha),
                           method='L-BFGS-B', bounds=bounds_lbf,
                           options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
            x_new = res.x
            h_new = np.empty(n)
            h_new[:n - 1] = x_new
            h_new[n - 1] = target - np.sum(x_new)
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
                x = h_cur[:n - 1].copy()
        except Exception:
            pass
    return h_cur, cur_score

def pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
            n_cand=70, patience=400):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    def update_row(p):
        change[p, :] = 0.0
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        grad[v2] += (1.0 - h[idxs[v2] + k_star])
        grad[v1] -= h[idxs[v1] - k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        cur_ncand = int(n_cand * (0.5 + 0.5 * progress))
        cur_ncand = max(10, min(cur_ncand, n // 3))

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:cur_ncand]
        high_set = sorted_idx[-cur_ncand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        nh = len(high_set)

                        f_prime = 0.0

                        a_d = mid_d

                approx = float(np.max(corr + delta_star * g)) * dx

        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
