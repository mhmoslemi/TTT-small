    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = T * (corr - corr.max())
    p = np.exp(shifted)
    p /= p.sum()
    
    # Gradient of smooth max
    n_corr = len(corr)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(n_corr):
        if p[s] < 1e-10:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += p[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= p[s] * h[idxs[v2] + k]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def vectorized_pair_cd(h_init, n, target, time_limit,
                       n_cand_i=20, n_cand_j=100, n_bisect=12,
                       max_sweeps=50000, max_no_improve=500,
                       stochastic_frac=0.03, seed=42):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    idxs = np.arange(n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                if consecutive_no_improve % 50 == 0:
                    npert = max(1, n // 20)
                    idxs_p = rng.choice(n, size=npert, replace=False)
                    h[idxs_p] += rng.randn(npert) * 0.04
                    h = project_feasible(h, target)
                elif consecutive_no_improve > max_no_improve * 2:
                    break
                continue

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        cur_i = max(10, int(n_cand_i * (1.0 + 0.5 * progress)))
        cur_j = max(50, int(n_cand_j * (1.0 + 0.5 * progress)))
        cur_i = min(cur_i, n // 4)
        cur_j = min(cur_j, n // 2)

        sorted_idx = np.argsort(grad)
        high_cands = sorted_idx[-cur_i:]
        low_cands = sorted_idx[:cur_j]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_cands)):
            i = int(high_cands[ii])
            g_matrix = change[low_cands] - change[i]
            lo_arr = np.maximum(h[i] - 1.0, -h[low_cands])
            hi_arr = np.minimum(h[i], 1.0 - h[low_cands])
            valid = (hi_arr - lo_arr) > 1e-12
            n_valid = int(valid.sum())
            if n_valid < 1:

            g_valid = g_matrix[valid]

                mid_d = 0.5 * (lo_v + hi_v)
                vals = corr[None, :] + mid_d[:, None] * g_valid
                f_mid = vals.max(axis=1)
                active = vals >= (f_mid[:, None] - 1e-14)
                g_clipped = np.where(active, g_valid, -1e30)
                f_prime = g_clipped.max(axis=1)
                update = f_prime > 1e-12
                lo_v[update] = mid_d[update]
                hi_v[~update] = mid_d[~update]

            deltas = 0.5 * (lo_v + hi_v)
            vals_final = corr[None, :] + deltas[:, None] * g_valid
            scores = vals_final.max(axis=1) * dx

            best_local = float(scores.min())
            if best_local < best_trial_c5:
                best_local_idx = int(scores.argmin())
                best_trial_c5 = best_local
                best_i = i
                best_j = int(low_cands[valid][best_local_idx])
                best_delta = float(deltas[best_local_idx])

        n_random = max(1, int(stochastic_frac * cur_i * cur_j))
        for _ in range(n_random):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-12:
                continue
            g = change[j] - change[i]
            a_d, b_d = delta_lo, delta_hi
            for _ in range(n_bisect):
                m_d = 0.5 * (a_d + b_d)
                vals = corr + m_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = float(np.max(g[active]))
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = m_d
                else:
                    a_d = m_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx = float(np.max(corr + delta_star * g) * dx)
            if approx < best_trial_c5:
                best_trial_c5 = approx
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def multi_temperature_slsqp(h_start, n, target, time_limit, seed=42):
    global _g_n, _g_target
    start = time.time()
    _g_n = n
    _g_target = target

            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
