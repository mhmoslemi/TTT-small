    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_subgradient(h, beta=20.0, n_top=30):
    """Compute weighted subgradient of max_k c_k(h)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    mid = n - 1
    
    # Get top shifts by correlation
    top_idx = np.argsort(corr)[-n_top:]
    top_corr = corr[top_idx]
    
    # Exponential weights
    shifted = beta * (top_corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for i, s_idx in enumerate(top_idx):
        w = weights[i]
        if w < 1e-10:
            continue
        k = int(s_idx) - mid
        # Term 1: (1-h[j+k]) for j where j+k is valid
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1+k:hi1+k])
        # Term 2: -h[j-k] for j where j-k is valid
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] += -w * h[lo2-k:hi2-k]
    
    return grad, corr, c_max, top_idx

def subgradient_descent(h_init, n, target, time_limit, beta_start=10, beta_end=100,
                         step_start=0.01, n_top=30, momentum=0.9):
    """Subgradient descent with projection and momentum"""
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)  # momentum
    
    n_iter = 0
    step = step_start
    
    while time.time() - start < time_limit:
        n_iter += 1
        progress = min(n_iter / 5000.0, 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        step = step_start * (0.5 ** (n_iter // 1000))
        
        grad, corr, c_max, top_idx = weighted_subgradient(h, beta, n_top)
        dx = 2.0 / n
        c5_val = c_max * dx
        
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Remove mean to maintain sum constraint (approximately)
        grad -= np.mean(grad)
        
        # Momentum step
        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5_val + 1e-14:
            h = h_new
        else:
            # Reject step, reduce momentum
            v = 0.5 * v
    
    return best_h, best_c5
