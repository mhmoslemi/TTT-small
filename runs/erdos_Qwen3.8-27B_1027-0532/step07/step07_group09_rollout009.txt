def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def bisect_delta(corr, g, delta_min, delta_max, n_bisect=12):
    a_d = delta_min
    b_d = delta_max
    for _ in range(n_bisect):
        mid_d = 0.5 * (a_d + b_d)
        vals = corr + mid_d * g
        f_mid = np.max(vals)
        active_mask = vals >= f_mid - 1e-14
        if active_mask.any():
            f_prime = np.max(g[active_mask])
        else:
            f_prime = 0.0
        if f_prime > 1e-12:
            b_d = mid_d
        else:
            a_d = mid_d
    return 0.5 * (a_d + b_d)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def bisect_delta(corr, g, delta_min, delta_max, n_bisect=12):
    a_d = delta_min
    b_d = delta_max
    for _ in range(n_bisect):
        mid_d = 0.5 * (a_d + b_d)
        vals = corr + mid_d * g
        f_mid = np.max(vals)
        active_mask = vals >= f_mid - 1e-13
        if active_mask.any():
            f_prime = np.max(g[active_mask])
        else:
            f_prime = 0.0
        if f_prime > 1e-12:
            b_d = mid_d
        else:
            a_d = mid_d
    return 0.5 * (a_d + b_d)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def cd_optimize(h_init, n, target, time_limit, max_sweeps=20000,
                n_cand_start=20, n_cand_end=55, max_no_improve=300,
                triple_prob=0.35, n_bisect=12):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)
    rng = np.random.RandomState(12345)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-15:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = min(8, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].astype(float)
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)
        if ncand < 5:
            ncand = 5

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=np.int32)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.vstack([change[p, :] for p in all_cands])

        best_trial = c5_val
        best_move = None

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            ci = int(cand_map[i])
            hr = G[ci, :]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                cj = int(cand_map[j])
                lr = G[cj, :]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue

                g = lr - hr
                delta_star = bisect_delta(corr, g, delta_min, delta_max, n_bisect)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_move = (0, i, j, k, delta_star)

        if rng.random() < triple_prob:
            if n_h >= 2 and n_l >= 1:
                k_idx = rng.randint(0, n_l)
                k = int(low_set[k_idx])
                ck = int(cand_map[k])
                kr = G[ck, :]
                h_k = h[k]

                for ii in range(n_h):
                    i = int(high_set[ii])
                    ci = int(cand_map[i])
                    h_i = h[i]
                    hr_i = G[ci, :]
                    for jj in range(ii + 1, n_h):
                        j = int(high_set[jj])
                        cj = int(cand_map[j])
                        h_j = h[j]
                        hr_j = G[cj, :]

                        d_min = max(-h_i, -h_j, (h_k - 1.0) / 2.0)
                        d_max = min(1.0 - h_i, 1.0 - h_j, h_k / 2.0)
                        if d_max - d_min < 1e-12:
                            continue

                        g = hr_i + hr_j - 2.0 * kr
                        delta_star = bisect_delta(corr, g, d_min, d_max, n_bisect)
                        if abs(delta_star) < 1e-12:
                            continue

                        approx_max = np.max(corr + delta_star * g)
                        if approx_max < best_trial:
                            best_trial = approx_max
                            best_move = (1, i, j, k, delta_star)

            if n_l >= 2 and n_h >= 1:
                k_idx = rng.randint(0, n_h)
                k = int(high_set[k_idx])
                ck = int(cand_map[k])
                kr = G[ck, :]
                h_k = h[k]

                for ii in range(n_l):
                    i = int(low_set[ii])
                    ci = int(cand_map[i])
                    h_i = h[i]
                    lr_i = G[ci, :]
                    for jj in range(ii + 1, n_l):
                        j = int(low_set[jj])
                        cj = int(cand_map[j])
                        h_j = h[j]
                        lr_j = G[cj, :]

                        d_min = max(-h_k / 2.0, h_i - 1.0, h_j - 1.0)
                        d_max = min((1.0 - h_k) / 2.0, h_i, h_j)
                        if d_max - d_min < 1e-12:
                            continue

                        g = 2.0 * kr - lr_i - lr_j
                        delta_star = bisect_delta(corr, g, d_min, d_max, n_bisect)
                        if abs(delta_star) < 1e-12:
                            continue

                        approx_max = np.max(corr + delta_star * g)
                        if approx_max < best_trial:
                            best_trial = approx_max
                            best_move = (2, k, i, j, delta_star)

        if best_move is not None:
            mtype = best_move[0]
            if mtype == 0:
                _, i, j, _, delta = best_move
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
            elif mtype == 1:
                _, i, j, k, delta = best_move
                h[i] += delta
                h[j] += delta
                h[k] -= 2.0 * delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
                update_change_row(change, k, h, n, shifts)
            elif mtype == 2:
                _, k, i, j, delta = best_move
                h[k] += 2.0 * delta
                h[i] -= delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, k, h, n, shifts)
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_step(h, n, target, lr, n_active_tol=1e-6):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    tol = max(1e-10, n_active_tol * max_corr)
    active = np.where(corr >= max_corr - tol)[0]

    grad = np.zeros(n)
    for k_idx in active:
        k = int(k_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad /= len(active)
    grad -= np.mean(grad)

    h_new = project_feasible(h - lr * grad, target)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 50:
        h_pol = slsqp_optimize(best_h, n0, target0, [200, 1000, 5000], 500, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 130:
        h_cd, c5_cd = cd_optimize(best_h, n0, target0,
                                   time_limit=min(110, remaining * 0.7),
                                   max_sweeps=30000,
                                   n_cand_start=15, n_cand_end=45,
                                   max_no_improve=400,
                                   triple_prob=0.4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 50:
        h_pol = slsqp_optimize(best_h, n0, target0, [2000, 8000, 20000], 700, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 150:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_cd, c5_cd = cd_optimize(h1, n1, target1,
                                   time_limit=min(120, remaining * 0.65),
                                   max_sweeps=20000,
                                   n_cand_start=20, n_cand_end=55,
                                   max_no_improve=350,
                                   triple_prob=0.4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            n0 = n1
            target0 = target1

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 50:
        h_pol = slsqp_optimize(best_h, n0, target0, [5000, 15000, 40000], 600, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 200:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_cd, c5_cd = cd_optimize(h2, n2, target2,
                                   time_limit=min(140, remaining * 0.65),
                                   max_sweeps=12000,
                                   n_cand_start=15, n_cand_end=40,
                                   max_no_improve=300,
                                   triple_prob=0.35)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            n0 = n2
            target0 = target2

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 50:
        h_pol = slsqp_optimize(best_h, n0, target0, [10000, 30000], 500, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 80:
        rng = np.random.RandomState(seed)
        scales = [0.01, 0.02, 0.04, 0.06]
        for ns in scales:
            rem = budget_s - 30.0 - (time.time() - start_time)
            if rem < 40:
                break
            h_pert = best_h + rng.randn(len(best_h)) * ns
            h_pert = project_feasible(h_pert, target0)
            h_ref = slsqp_optimize(h_pert, n0, target0, [500, 3000, 10000], 400, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 80:
        rem2 = budget_s - 30.0 - (time.time() - start_time)
        if rem2 > 60:
            h_sg = best_h.copy()
            for it in range(5000):
                if time.time() - start_time > budget_s - 40:
                    break
                progress = min(it / 5000.0, 1.0)
                lr = 0.003 * (1.0 - 0.9 * progress) ** 2
                h_new = subgradient_step(h_sg, n0, target0, lr)
                sc_new = compute_c5(h_new)
                if sc_new < compute_c5(h_sg) - 1e-15:
                    h_sg = h_new
                else:
                    lr *= 0.5
                    h_new = subgradient_step(h_sg, n0, target0, lr)
                    sc_new = compute_c5(h_new)
                    if sc_new < compute_c5(h_sg) - 1e-15:
                        h_sg = h_new
            sc = compute_c5(h_sg)
            if sc < best_score:
                best_score = sc
                best_h = h_sg

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 50:
        h_cd, c5_cd = cd_optimize(best_h, n0, target0,
                                   time_limit=min(35, remaining * 0.6),
                                   max_sweeps=8000,
                                   n_cand_start=25, n_cand_end=60,
                                   max_no_improve=150,
                                   triple_prob=0.5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 25:
        h_final = slsqp_optimize(best_h, n0, target0, [10000, 30000, 80000], 400, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
