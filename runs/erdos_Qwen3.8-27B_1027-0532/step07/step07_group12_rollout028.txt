import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    # Project h to [0,1]^n with sum = target_sum
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        # Need to increase sum: add to entries < 1
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        # Need to decrease sum: remove from entries > 0
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    per_entry = diff / len(idx)
    h[idx] = np.clip(h[idx] + per_entry, 0.0, 1.0)
    # Handle residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] = np.clip(h[idx] + diff / len(idx), 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_grad(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    weights = e / e.sum()
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    for m in np.where(weights > 1e-12)[0]:
        k = int(m) - mid
        w = weights[m]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += w * (1.0 - x[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pm in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            try:
                res = minimize(lse_obj, h_cur, args=(alpha, target, penalty * pm),
                             jac=lse_grad, method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, n_cand=50, n_shifts=1, bisect_iters=12):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    no_improve = 0

    for _ in range(1000000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        if cur_max * dx < best_c5 - 1e-13:
            best_c5 = cur_max * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 250:
                break

        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top = np.argsort(corr)[-n_shifts:]
            aw = 30.0
            w = np.exp(aw * (corr[top] - cur_max))
            w /= w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]

        sg = np.argsort(-grad)
        hi = sg[:n_cand]
        lo_set = sg[-n_cand:]
        cands = np.unique(np.append(hi, lo_set))
        nc = len(cands)

        profs = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(cands[ci])
            ks = np.arange(-(n - 1), n)
            i1 = m - ks
            i2 = m + ks
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            if np.any(v1):
                profs[ci, v1] += h[i1[v1]]
            if np.any(v2):
                profs[ci, v2] += h[i2[v2]]

        cp = {}
        for p in range(nc):
            cp[int(cands[p])] = p

        best_imp = 0.0
        best_i = -1
        best_j = -1
        best_d = 0.0

        for i in hi:
            hv = float(h[i])
            if hv < 1e-12:
                continue
            ci = cp[int(i)]
            for j in lo_set:
                if int(i) == int(j):
                    continue
                dm = min(hv, 1.0 - float(h[j]))
                if dm < 1e-12:
                    continue
                g = profs[ci] - profs[cp[int(j)]]

                lo_d = 0.0
                hi_d = dm
                for _b in range(bisect_iters):
                    md = 0.5 * (lo_d + hi_d)
                    vals = corr + md * g
                    fm = float(np.max(vals))
                    act = vals >= fm - 1e-10
                    fp = float(np.max(g[act]))
                    if fp > 1e-12:
                        hi_d = md
                    else:
                        lo_d = md

                ds = 0.5 * (lo_d + hi_d)
                if ds < 1e-12:
                    continue
                imp = cur_max - float(np.max(corr + ds * g))
                if imp > best_imp:
                    best_imp = imp
                    best_i = int(i)
                    best_j = int(j)
                    best_d = ds

        if best_i >= 0 and best_imp > 1e-14:
            h[best_i] -= best_d
            h[best_j] += best_d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    return project_feasible(np.interp(x_new, x_old, h_src), target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - t0)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0

    best_h = project_feasible(h0, target)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS warm-up
    if tleft() > 70:
        hp, sc = lbfgs_phase(best_h, n, target, [200, 1000, 5000], 300, min(55, tleft() * 0.3), 100.0)
        if sc < best_score:
            best_score, best_h = sc, hp

    # Phase 2: Intensive pair CD (main workhorse)
    if tleft() > 250:
        ct = min(300, tleft() * 0.3)
        hcd, c5cd = pair_cd(best_h, n, target, ct, n_cand=50, n_shifts=1, bisect_iters=12)
        if c5cd < best_score:
            best_score, best_h = c5cd, hcd

    # Phase 3: L-BFGS high alpha
    if tleft() > 70:
        hp, sc = lbfgs_phase(best_h, n, target, [5000, 20000], 200, min(55, tleft() * 0.25), 500.0)
        if sc < best_score:
            best_score, best_h = sc, hp

    # Phase 4: Pair CD multi-shift
    if tleft() > 160:
        ct = min(150, tleft() * 0.25)
        hcd, c5cd = pair_cd(best_h, n, target, ct, n_cand=60, n_shifts=3, bisect_iters=10)
        if c5cd < best_score:
            best_score, best_h = c5cd, hcd

    # Phase 5: L-BFGS very high alpha
    if tleft() > 60:
        hp, sc = lbfgs_phase(best_h, n, target, [100000, 500000], 200, min(50, tleft() * 0.3), 5000.0)
        if sc < best_score:
            best_score, best_h = sc, hp

    # Phase 6: Pair CD single shift, more candidates
    if tleft() > 120:
        ct = min(110, tleft() * 0.3)
        hcd, c5cd = pair_cd(best_h, n, target, ct, n_cand=80, n_shifts=1, bisect_iters=10)
        if c5cd < best_score:
            best_score, best_h = c5cd, hcd

    # Phase 7: Higher resolution
    if tleft() > 280:
        n2 = 1200
        t2 = n2 / 2.0
        h2 = interp_to(best_h, n2, t2)
        hp, sc = lbfgs_phase(h2, n2, t2, [100, 1000, 5000], 200, min(80, tleft() * 0.25), 100.0)
        if sc < best_score:
            best_score, best_h = sc, hp
            n, target = n2, t2
        if tleft() > 100:
            hcd, c5cd = pair_cd(best_h, n, target, min(90, tleft() * 0.35), n_cand=40, n_shifts=1, bisect_iters=10)
            if c5cd < best_score:
                best_score, best_h = c5cd, hcd

    # Phase 8: Perturbation + re-optimize
    if tleft() > 120:
        rng = np.random.RandomState(seed)
        nu = len(best_h)
        tu = nu / 2.0
        for scale in [0.015, 0.008, 0.004]:
            if tleft() < 50:
                break
            hp2 = project_feasible(best_h + rng.randn(nu) * scale, tu)
            hq, sq = lbfgs_phase(hp2, nu, tu, [5000, 50000], 150, min(30, tleft() * 0.2), 500.0)
            if sq < best_score:
                if tleft() > 40:
                    hcd, c5cd = pair_cd(hq, nu, tu, min(30, tleft() * 0.25), n_cand=40, n_shifts=1, bisect_iters=10)
                    if c5cd < best_score:
                        best_score, best_h = c5cd, hcd
                    else:
                        best_score, best_h = sq, hq
                else:
                    best_score, best_h = sq, hq

    # Phase 9: Final L-BFGS polish
    if tleft() > 30:
        nu = len(best_h)
        tu = nu / 2.0
        hp, sc = lbfgs_phase(best_h, nu, tu, [500000, 2000000], 100, min(25, tleft() * 0.5), 50000.0)
        if sc < best_score:
            best_score, best_h = sc, hp

    # Phase 10: Final pair CD
    if tleft() > 20:
        nu = len(best_h)
        tu = nu / 2.0
        hcd, c5cd = pair_cd(best_h, nu, tu, min(15, tleft() * 0.6), n_cand=30, n_shifts=1, bisect_iters=10)
        if c5cd < best_score:
            best_score, best_h = c5cd, hcd

    nu = len(best_h)
    tu = nu / 2.0
    hf = project_feasible(best_h, tu)
    final_score = compute_c5(hf)
    return hf, final_score, nu
