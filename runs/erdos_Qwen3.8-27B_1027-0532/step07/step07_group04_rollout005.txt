C[j0] = sum_k w_k * 1[0<=j0+k<n]  (sum of weights where j0+k is in range)

Then: ∂f/∂h[j0] = dx * (A[j0] - B[j0] - h[j0] * C[j0])

Now:

  This is a convolution of w and h!

So A and C are prefix sums of w (with appropriate indexing), and B is a convolution.

With numpy, the convolution B can be computed via np.convolve or FFT in O(n log n).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr
    m = np.max(scaled)
    lse = m + np.log(np.sum(np.exp(scaled - m)))
    val = lse / alpha * dx
    sm = np.sum(h) - n / 2.0
    val += penalty * sm * sm
    return val

def smooth_max_grad(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    scaled = alpha * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    w = exp_s / np.sum(exp_s)
    shifts = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for j in range(n):
        mask1 = (shifts >= j - (n - 1)) & (shifts <= j)
        mask2 = (shifts >= -j) & (shifts <= n - 1 - j)
        idx1 = shifts[mask1] + (n - 1)
        idx2 = shifts[mask2] + (n - 1)
        A = np.sum(w[idx1])
        B_val = np.sum(w[idx1] * h[j - shifts[mask1]])
        C = np.sum(w[idx2])
        grad[j] = dx * (A - B_val - h[j] * C)
    sm = np.sum(h) - n / 2.0
    grad += 2.0 * penalty * sm
    return grad

def lbfgsb_round(h_start, n, target, alphas, maxiter, penalty, time_limit):
    start = time.time()
    best_h = h_start.copy()
    best_score = compute_c5(best_h)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        h_cur = best_h.copy()
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha, penalty),
                           jac=lambda x: smooth_max_grad(x, alpha, penalty),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass
    return best_h, best_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def fast_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand=35, n_shifts=3, max_no_improve=400, bisect_iters=12):
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-n_shifts:]
        c_max = corr[np.argmax(corr)]
        weights = np.exp(-8.0 * (c_max - corr[top_shifts]) / max(c_max, 1e-10))
        weights = weights / np.sum(weights)

        grad = np.zeros(n)
        for wi, s_idx in enumerate(top_shifts):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += weights[wi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= weights[wi] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisect_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def sa_cd(h_init, n, target, time_limit, max_steps=50000,
          t_start=0.03, t_end=0.0002, n_cand=40, seed=42):
    dx = 2.0 / n
    start = time.time()
    rng = np.random.RandomState(seed)
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 800:
            break

        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        if rng.rand() < 0.7:
            i = int(high_set[rng.randint(len(high_set))])
            j = int(low_set[rng.randint(len(low_set))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue

        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-12:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx * 0.1, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

            if approx_new_c5 < best_c5 - 1e-14:
                best_c5 = approx_new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    target_dst = n_dst / 2.0
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    # Phase 5: Multi-start CD for exploration

        h_ms, c5_ms = multi_start_cd(best_h, n0, target0, rng,
                                     n_restarts=3,
                                     time_per_restart=min(30, tleft() / 5.0),
                                     seed_base=200)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 6: SA for escape
    if tleft() > 100:
        h_sa, c5_sa = sa_cd(best_h, n0, target0,
                            time_limit=min(80, tleft() * 0.6),
                            max_steps=30000,
                            t_start=0.02, t_end=0.0001,
                            n_cand=35, seed=seed + 500)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 7: CD refinement
    if tleft() > 100:
        h_cd3, c5_cd3 = fast_cd(best_h, n0, target0,
                                time_limit=min(90, tleft() * 0.6),
                                max_sweeps=30000, n_cand=35, n_shifts=4,
                                max_no_improve=300, bisect_iters=12)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 8: L-BFGS-B polish
    if tleft() > 50:
        h_pol2, sc2 = lbfgsb_round(best_h, n0, target0,
                                    [500, 2000, 8000], 200, 1e4,
                                    time_limit=min(40, tleft() * 0.7))
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 9: Final CD
    if tleft() > 30:
        h_cf, c5_cf = fast_cd(best_h, n0, target0,
                              time_limit=min(25, tleft() * 0.7),
                              max_sweeps=10000, n_cand=40, n_shifts=4,
                              max_no_improve=150, bisect_iters=10)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    # Phase 10: Try higher resolution if time permits
    if tleft() > 150 and n0 < 1000:
        n3 = 1000
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3)
        h_cd4, c5_cd4 = fast_cd(h3, n3, target3,
                                time_limit=min(100, tleft() * 0.6),
                                max_sweeps=20000, n_cand=25, n_shifts=3,
                                max_no_improve=250, bisect_iters=10)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4
            n0 = n3
            target0 = target3

        if tleft() > 40:
            h_cf2, c5_cf2 = fast_cd(best_h, n0, target0,
                                    time_limit=min(30, tleft() * 0.6),
                                    max_sweeps=10000, n_cand=30, n_shifts=3,
                                    max_no_improve=200, bisect_iters=10)
            if c5_cf2 < best_score:
                best_score = c5_cf2
                best_h = h_cf2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
