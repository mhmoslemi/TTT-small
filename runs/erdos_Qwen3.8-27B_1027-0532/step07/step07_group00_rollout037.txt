        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with analytical gradient at initial resolution
    if tleft() > 80:
        _set_alpha(100.0)
        h1, s1 = slsqp_optimize2(best_h, target0, [100, 500, 2000], 300, 1e-15)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: Weighted pair coordinate descent
    if tleft() > 200:
        h2, s2 = weighted_pair_cd(best_h, n0, target0,
                                  time_limit=min(160, tleft() * 0.5),
                                  max_sweeps=25000,
                                  n_cand_start=20, n_cand_end=60,
                                  max_no_improve=350,
                                  weight_beta=6.0)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: SLSQP high alpha at initial resolution
    if tleft() > 120:
        h3, s3 = slsqp_optimize2(best_h, target0, [5000, 20000, 100000], 500, 1e-15)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Interpolate to n=1600 and optimize
    n1 = 1600
    target1 = n1 / 2.0
    if tleft() > 500:
        h1 = interpolate_to(best_h, n1, target1)
        sc_interp = compute_c5(h1)

        if tleft() > 400:
            h4, s4 = slsqp_optimize2(h1, target1, [200, 1000, 5000, 20000], 400, 1e-15)
            if s4 < best_score:
                best_score = s4
                best_h = h4

        if tleft() > 250:
            h5, s5 = weighted_pair_cd(best_h, n1, target1,
                                      time_limit=min(200, tleft() * 0.5),
                                      max_sweeps=15000,
                                      n_cand_start=15, n_cand_end=50,
                                      max_no_improve=300,
                                      weight_beta=6.0)
            if s5 < best_score:
                best_score = s5
                best_h = h5

        if tleft() > 100:
            h6, s6 = slsqp_optimize2(best_h, target1, [50000, 200000, 500000], 400, 1e-15)
            if s6 < best_score:
                best_score = s6
                best_h = h6

    # Phase 5: Multi-start exploration at fine resolution
    if tleft() > 200:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)

        for trial in range(4):
            if tleft() < 100:
                break
            noise_scale = 0.02 + 0.02 * trial
            freq = 3 + 2 * trial

            h_pert = best_h.copy()
            h_pert = h_pert + noise_scale * np.sin(2.0 * np.pi * freq * np.arange(n_cur) / n_cur)
            h_pert = h_pert + noise_scale * 0.5 * rng.randn(n_cur)
            h_pert = project_feasible(h_pert, target_cur)

            h_trial, s_trial = slsqp_optimize2(h_pert, target_cur, [500, 5000, 50000], 300, 1e-15)
            if s_trial < best_score:
                best_score = s_trial
                best_h = h_trial

            if tleft() > 60:
                h_cd_trial, s_cd = weighted_pair_cd(best_h, n_cur, target_cur,
                                                    time_limit=min(40, tleft() * 0.4),
                                                    max_sweeps=8000,
                                                    n_cand_start=20, n_cand_end=50,
                                                    max_no_improve=200,
                                                    weight_beta=7.0)
                if s_cd < best_score:
                    best_score = s_cd
                    best_h = h_cd_trial

    # Phase 6: Final SLSQP polish
    if tleft() > 50:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin, s_fin = slsqp_optimize2(best_h, target_cur, [200000, 1000000], 400, 1e-16)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
