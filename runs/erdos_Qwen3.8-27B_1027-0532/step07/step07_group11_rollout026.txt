    idxs = np.arange(n)
    mid = n - 1

    m = len(active_shifts)
    A = np.zeros((m + 1, n))  # m constraint rows + 1 equality row
    b = np.zeros(m + 1)
    
    for i, s_idx in enumerate(active_shifts):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)

        else:

                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 500.0
                    if np.random.random() < np.exp(-max(0.0, delta_obj) / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 3:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= 0.999
        else:
            temp = max(temp * 0.93, 1e-12)

    return best_h, best_c5


def subgradient_multi(h_init, n, target, time_limit, n_steps=20000, n_weights=5, step_scale=0.005):
    start_t = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start_t > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        p = np.exp(corr * 80.0)
        p /= p.sum()

        grad = np.zeros(n)
        top_k = np.argsort(p)[-n_weights:]
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.92

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def binary_push(h, n, target, threshold=0.3):
    """Push values towards 0 or 1 to create more binary structure"""
    h_new = h.copy()
    n_below = np.sum(h_new < 0.5)
    n_above = np.sum(h_new >= 0.5)
    target_below = int(np.round(target * (1.0 - threshold)))
    for i in range(len(h_new)):
        if h_new[i] < threshold:
            h_new[i] = 0.0
        elif h_new[i] > 1.0 - threshold:
            h_new[i] = 1.0
    h_new = project_feasible(h_new, target)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    # Phase 1: Quick SLSQP polish with analytical gradient
    if tleft() > 100:
        h_pol, sc = slsqp_opt(best_h, cur_n, cur_target,
                              [50, 200, 1000, 5000], 800)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Main optimization with LP + pair CD
    if tleft() > 400:
        t_limited = min(350, tleft() * 0.6)
        h_cd, c5_cd = main_optimize(best_h, cur_n, cur_target,
                                    time_limit=t_limited,
                                    max_sweeps=3000,
                                    n_cand_start=20, n_cand_end=60,
                                    n_active_lp=5, lp_every=100,
                                    lp_step=0.03, max_no_improve=300,
                                    use_anneal=True, temp_init=0.015)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP refinement
    if tleft() > 80:
        h_pol, sc = slsqp_opt(best_h, cur_n, cur_target,
                              [2000, 10000], 1200)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Grid refinement to n=800
    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if tleft() > 250:
            h_cd1, c5_cd1 = main_optimize(h1, n1, target1,
                                          time_limit=min(200, tleft() * 0.55),
                                          max_sweeps=2000,
                                          n_cand_start=15, n_cand_end=50,
                                          n_active_lp=5, lp_every=150,
                                          lp_step=0.02, max_no_improve=250,
                                          use_anneal=False, temp_init=0.0)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                cur_n = n1
                cur_target = target1
            elif sc1 < best_score:
                best_score = sc1
                best_h = h1
                cur_n = n1
                cur_target = target1

    # Phase 5: Subgradient on current grid
    if tleft() > 80:
        h_sg, c5_sg = subgradient_multi(best_h, cur_n, cur_target,
                                        time_limit=min(60, tleft() * 0.5),
                                        n_steps=15000, n_weights=5,
                                        step_scale=0.004)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: Another LP+CD round
    if tleft() > 150:
        h_cd2, c5_cd2 = main_optimize(best_h, cur_n, cur_target,
                                      time_limit=min(120, tleft() * 0.6),
                                      max_sweeps=2000,
                                      n_cand_start=30, n_cand_end=80,
                                      n_active_lp=6, lp_every=80,
                                      lp_step=0.02, max_no_improve=200,
                                      use_anneal=True, temp_init=0.005)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 7: Final SLSQP polish
    if tleft() > 60:
        h_final_pol, sc = slsqp_opt(best_h, cur_n, cur_target,
                                    [5000, 20000, 50000], 1000)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    # Phase 8: Try grid n=1200 if time permits
    if tleft() > 300:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 200:
            h_cd3, c5_cd3 = main_optimize(h2, n2, target2,
                                          time_limit=min(150, tleft() * 0.55),
                                          max_sweeps=1500,
                                          n_cand_start=15, n_cand_end=40,
                                          n_active_lp=5, lp_every=200,
                                          lp_step=0.02, max_no_improve=200,
                                          use_anneal=False, temp_init=0.0)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
                cur_n = n2
                cur_target = target2

    # Phase 9: Final SLSQP on best grid
    if tleft() > 50:
        h_final_pol, sc = slsqp_opt(best_h, cur_n, cur_target,
                                    [10000, 30000], 800)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    # Phase 10: Final quick CD
    if tleft() > 30:
        h_final_cd, c5_final = main_optimize(best_h, cur_n, cur_target,
                                             time_limit=min(20, tleft() * 0.7),
                                             max_sweeps=1000,
                                             n_cand_start=40, n_cand_end=80,
                                             n_active_lp=5, lp_every=200,
                                             lp_step=0.01, max_no_improve=100,
                                             use_anneal=False, temp_init=0.0)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
