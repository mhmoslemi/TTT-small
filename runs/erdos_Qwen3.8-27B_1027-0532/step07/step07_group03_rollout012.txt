import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target_sum) > 1e-10:
        h += (target_sum - np.sum(h)) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad


def compute_corr_gradient_multi(h, k_stars, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(k_stars, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

                a_d, b_d = d_lo, d_hi
                for _ in range(12):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = m_d
                    else:
                        a_d = m_d
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) < 1e-13:
                    continue
                approx = np.max(corr + d_star * g_vec) * dx
                if approx < best_trial - 1e-15:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = d_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pi, pj = best_i, best_j
            for p in (pi, pj):
                change[p, :] = 0.0
                pm = (p + shifts >= 0) & (p + shifts < n)
                nm = (p - shifts >= 0) & (p - shifts < n)
                change[p, pm] += (1.0 - h[p + shifts[pm]])
                change[p, nm] -= h[p - shifts[nm]]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def nesterov_subgrad(h_init, n, target, time_limit, max_iters=30000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    h_curr = h_init.copy().astype(float)
    h_prev = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)
    t_k = 1.0
    f_opt_est = compute_c5(h_curr)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = h_curr + ((t_k - 1.0) / t_k) * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)

        corr_y = compute_corr(y)
        c_max = np.max(corr_y)

        active_mask = corr_y >= c_max - 1e-6
        n_act = int(active_mask.sum())
        if n_act == 1:
            k_active = int(np.argmax(corr_y))
            g = compute_corr_gradient(y, k_active)
        else:
            acts = np.where(active_mask)[0]
            w = np.ones(len(acts)) / len(acts)
            g = compute_corr_gradient_multi(y, acts, w)
        g -= np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        f_curr = c_max * dx
        step = (f_curr - f_opt_est) / (g_norm * g_norm + 1e-12)
        step = np.clip(step, 1e-8, 5.0)

        h_new = y - step * g
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < best_score - 1e-15:
            best_score = sc
            best_h = h_new.copy()
            f_opt_est = sc
        else:
            f_opt_est = min(f_opt_est, sc)

        h_prev = h_curr.copy()
        h_curr = h_new

    return best_h, best_score

    if tleft() > 60:

            best_score = s2

                best_n = n1
                cur_h = h_cd1
                cur_sc = s_cd1

        if tleft() > 120:

                                                time_limit=min(110, tleft() * 0.45),
                                                max_iters=6000, tol=1e-9)
            if s_cp1 < best_score:

                best_h = h_cp1
                best_n = n1
                cur_h = h_cp1
                cur_sc = s_cp1

        if tleft() > 80:
            h_cd2, s_cd2 = fast_cd(cur_h, n1, target1,
                                   time_limit=min(70, tleft() * 0.4),
                                   max_sweeps=25000,
                                   n_cand_start=35, n_cand_end=5,

            if s_cd2 < best_score:
