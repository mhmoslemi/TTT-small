from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

One more idea: **use the "exact" line search for pair moves**

Instead of binary search, we can find the exact optimal delta by:
1. Computing all breakpoints of the piecewise-linear function g(delta) = max_k (corr[k] + delta * change[k])
2. The minimum is at one of the breakpoints or at a boundary
3. Breakpoints occur where two lines cross: (corr[k1] - corr[k2]) / (change[k2] - change[k1])

But there are O(n²) breakpoints (one for each pair of shifts), which is too many.

However, we only need to consider the top ~50 shifts (since the max is determined by them). So there are O(50²) = 2500 breakpoints, which is manageable!

1. Take the top 50 shifts
2. For each pair of shifts (k1, k2) among these 50, compute the crossing point
3. Filter crossings to [delta_lo, delta_hi]
4. Evaluate g at all crossings and at boundaries
5. Return the delta with minimum g

This is O(50² + 50 * 50) = O(5000) per pair, compared to O(10 * 50) = O(500) for 10-step binary search. So binary search is still faster for a single pair. But the exact method gives the true optimum, while binary search might miss it slightly.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < len(h))
    neg_mask = (p - shifts >= 0) & (p - shifts < len(h))
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def fast_cd(h_init, n, target, time_limit, max_sweeps=500000,
            n_cand=50, top_k_delta=40, n_grad_shifts=4,
            reheat_interval=200, reheat_scale=0.05, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stale = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > max_no_improve:
                break

        # Reheat
        if stale > reheat_interval and stale % 50 == 0:
            rng = np.random.RandomState(sweep * 17 + 3)
            n_pert = max(5, n // 40)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 1e-5:
                h = h_pert
                change = compute_change_matrix(h, n)
                stale = 0
                continue

        # Top shifts for delta
        top_idx = np.argsort(corr)[-top_k_delta:]
        top_corr = corr[top_idx]

        # Gradient for candidate selection
        grad_shifts = np.argsort(corr)[-n_grad_shifts:]
        gw = corr[grad_shifts]
        gw = gw / gw.sum()
        
        grad = np.zeros(n)
        for s_i, w in zip(grad_shifts, gw):
            k = int(s_i) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_d = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            ch_i = change[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue

                g = change[j] - ch_i
                # Quick rejection
                if g[mid] > 0:  # mid is the zero-lag, but let's use argmax
                    pass  # allow it
                # Actually use the k_star check
                k_star = int(np.argmax(corr))
                if g[k_star] > 0.15 * (np.abs(g).max() + 1e-15):
                    continue

                # Binary search on top shifts
                a = d_lo
                b = d_hi
                g_top = g[top_idx]
                for _ in range(12):
                    m = 0.5 * (a + b)
                    vals = top_corr + m * g_top
                    fm = np.max(vals)
                    act = vals >= fm - 1e-14
                    if act.any():
                        fp = np.max(g_top[act])
                    else:
                        fp = 0.0
                    if fp > 1e-12:
                        b = m
                    else:
                        a = m

                d_star = 0.5 * (a + b)
                if abs(d_star) < 1e-12:
                    continue

                # Full verification
                approx = np.max(corr + d_star * g) * dx
                if approx < best_trial - 1e-12:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_d = d_star

        if best_i >= 0:
            h[best_i] -= best_d
            h[best_j] += best_d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, shifts)
            update_change_row(change, best_j, h, shifts)

    return best_h, best_c5

    n_src = len(h_src)
    if n_src == n_dst:

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def fast_cd(h_init, n, target, time_limit, max_sweeps=500000,
            n_cand_start=30, n_cand_end=60, top_k_delta=40, n_grad_shifts=4,
            reheat_interval=200, reheat_scale=0.04, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stale = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > max_no_improve:
                break

        if stale > reheat_interval and stale % 40 == 0:
            rng = np.random.RandomState(sweep * 19 + 11)
            n_pert = max(4, n // 50)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 1e-5:
                h = h_pert
                change = compute_change_matrix(h, n)
                stale = 0
                continue

        top_idx = np.argsort(corr)[-top_k_delta:]
        top_corr = corr[top_idx].copy()

        k_star = int(np.argmax(corr))

        grad_shifts = np.argsort(corr)[-n_grad_shifts:]
        gw = corr[grad_shifts].copy()
        gw_sum = gw.sum()
        if gw_sum > 0:
            gw = gw / gw_sum
        else:
            gw = np.ones(n_grad_shifts) / n_grad_shifts

        grad = np.zeros(n)
        for s_i in range(len(grad_shifts)):
            k = int(grad_shifts[s_i]) - mid

        grad -= np.mean(grad)

        n_cand = min(n_cand, n // 4)

        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_d = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            ch_i = change[i]
            hi_val = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(hi_val - 1.0, -h[j])
                d_hi = min(hi_val, 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue

                g = change[j] - ch_i
                if g[k_star] > 0:
                    continue

                g_top = g[top_idx]
                a = d_lo
                b = d_hi
                for _ in range(12):
                    m = 0.5 * (a + b)
                    vals = top_corr + m * g_top
                    fm = np.max(vals)
                    act = vals >= fm - 1e-14
                    if act.any():
                        fp = np.max(g_top[act])
                    else:
                        fp = 0.0
                    if fp > 1e-12:
                        b = m
                    else:
                        a = m

                d_star = 0.5 * (a + b)
                if abs(d_star) < 1e-12:
                    continue

                approx = np.max(corr + d_star * g) * dx
                if approx < best_trial - 1e-12:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_d = d_star

        if best_i >= 0:
            h[best_i] -= best_d
            h[best_j] += best_d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, shifts, n)
            update_change_row(change, best_j, h, shifts, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
