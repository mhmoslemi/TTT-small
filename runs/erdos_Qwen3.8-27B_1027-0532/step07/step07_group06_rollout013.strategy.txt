def compute_c5(h):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(x, target):
    # Maintains integral constraint
    s = np.sum(x)
    while True:
        diff = target - s
        if abs(diff) < 1e-14:
            return x.astype(float)
        x_new = x.copy()
        # Simple projection using linear step
        if diff > 0:
            x_new[x < 1.0] += diff/np.sum(~(x==1.0))
        else:
            x_new[x > 0.0] += diff/np.sum(x)
        x_new = np.clip(x_new, 0, 1)
        s = np.sum(x_new)
        diff = target - s

def surrogate_objective(h, alpha=0.1, grid_size=400):
    # Smoothed surrogate with projection
    h_proj = project_feasible(h, grid_size/2)
    c = np.correlate(h_proj, 1-h_proj, mode='full')
    c_max = np.max(c)
    if c_max < 1e-15:
        return 0.0
    scaled = alpha * (c - c_max)
    np.clip(scaled, -500.0, 0.0, out=scaled)
    val = np.log(np.sum(np.exp(scaled)) / alpha + 1e-15)
    return float(val + c_max)

def gradient_estimate(h):
    # Finite differences gradient
    h_forward = h.copy()
    h_backward = h.copy()
    delta = 1e-4
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # Forward
        h_forward[i] = min(max(h[i]+delta, 0.0), 1.0)
        f_plus = surrogate_objective(h_forward)
        # Backward
        h_backward[i] = min(max(h[i]-delta, 0.0), 1.0)
        f_minus = surrogate_objective(h_backward)
        grad[i] = (f_plus - f_minus) / (2*delta)
    return grad
