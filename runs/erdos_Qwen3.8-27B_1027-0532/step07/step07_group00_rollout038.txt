def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP initial polish
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [80, 400, 1500, 8000],
                                400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: LAM optimization
    if tleft() > 300:
        h_lam, sc_lam = lame_optimize(best_h, n0, target0,
                                      time_limit=min(200, tleft() * 0.5),
                                      max_iter=250)
        if sc_lam < best_score:
            best_score = sc_lam
            best_h = h_lam

    # Phase 3: SLSQP after LAM
    if tleft() > 100:
        h_pol2, sc2 = slsqp_round(best_h, n0, target0,
                                  [500, 3000, 15000],
                                  400, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 4: Pair CD
    if tleft() > 200:
        h_cd, c5_cd = pair_cd_optimize(best_h, n0, target0,
                                       time_limit=min(180, tleft() * 0.5),
                                       max_sweeps=20000,
                                       n_cand_start=15, n_cand_end=55,
                                       max_no_improve=350,
                                       weight_beta=5.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Subgradient refinement
    if tleft() > 100:
        h_sg, sc_sg = subgradient_optimize(best_h, n0, target0,
                                           time_limit=min(80, tleft() * 0.5),
                                           n_iter=15000, step_init=0.003)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 6: Interpolate to higher resolution
    n1 = 1000
    target1 = n1 / 2.0
    if tleft() > 350:
        h1 = interpolate_to(best_h, n1, target1)
        h1_pol, sc1 = slsqp_round(h1, n1, target1,
                                  [100, 500, 3000, 10000],
                                  350, 1e-14)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1_pol
        else:
            best_h = h1_pol if compute_c5(h1_pol) < compute_c5(best_h) else best_h

        if tleft() > 250:
            h1_lam, sc1_lam = lame_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                            time_limit=min(180, tleft() * 0.5),
                                            max_iter=200)
            if sc1_lam < best_score:
                best_score = sc1_lam
                best_h = h1_lam

        if tleft() > 150:
            h1_cd, c5_1cd = pair_cd_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                             time_limit=min(120, tleft() * 0.5),
                                             max_sweeps=15000,
                                             n_cand_start=15, n_cand_end=50,
                                             max_no_improve=300,
                                             weight_beta=6.0)
            if c5_1cd < best_score:
                best_score = c5_1cd
                best_h = h1_cd

        if tleft() > 60:
            h1_pol2, sc1p2 = slsqp_round(best_h, len(best_h), len(best_h) / 2.0,
                                         [5000, 20000, 80000],
                                         500, 1e-14)
            if sc1p2 < best_score:
                best_score = sc1p2
                best_h = h1_pol2

    # Phase 7: Multi-start perturbation
    if tleft() > 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.1]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, target_cur)
            h_ref, sc_ref = slsqp_round(h_pert, n_cur, target_cur,
                                        [200, 2000, 10000],
                                        300, 1e-14)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 60:
                h_cd_ref, c5_ref = pair_cd_optimize(best_h, n_cur, target_cur,
                                                    time_limit=min(30, tleft() * 0.4),
                                                    max_sweeps=8000,
                                                    n_cand_start=15, n_cand_end=45,
                                                    max_no_improve=150,
                                                    weight_beta=5.0)
                if c5_ref < best_score:
                    best_score = c5_ref
                    best_h = h_cd_ref

    # Phase 8: Final LAM + SLSQP polish
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final_lam, sc_fl = lame_optimize(best_h, n_cur, target_cur,
                                           time_limit=min(50, tleft() * 0.5),
                                           max_iter=100)
        if sc_fl < best_score:
            best_score = sc_fl
            best_h = h_final_lam

    if tleft() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin, sc_fin = slsqp_round(best_h, n_cur, target_cur,
                                    [20000, 100000],
                                    400, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Final pair CD polish
    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin_cd, c5_fin_cd = pair_cd_optimize(best_h, n_cur, target_cur,
                                               time_limit=min(25, tleft() * 0.6),
                                               max_sweeps=6000,
                                               n_cand_start=25, n_cand_end=60,
                                               max_no_improve=150,
                                               weight_beta=8.0)
        if c5_fin_cd < best_score:
            best_score = c5_fin_cd
            best_h = h_fin_cd

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
