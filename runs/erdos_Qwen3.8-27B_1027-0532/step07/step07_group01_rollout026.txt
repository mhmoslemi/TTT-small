def fast_cd_step(h, corr, corr_max, change, n, dx, n_cand):
    n_corr = 2 * n - 1
    mid = n - 1
    
    # Compute gradient at the active shift
    k_star = int(np.argmax(corr)) - mid
    
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    grad -= np.mean(grad)
    
    # Pick candidate sets
    sorted_idx = np.argsort(grad)
    low_set = sorted_idx[:n_cand]
    high_set = sorted_idx[-n_cand:]
    
    best_improvement = 0
    best_i = -1
    best_j = -1
    best_delta = 0.0
    
    for i in high_set:
        if h[i] < 1e-14:
            continue
        for j in low_set:
            if h[j] > 1.0 - 1e-14:
                continue
            if i == j:
                continue
            g = change[j] - change[i]
            max_delta = min(h[i], 1.0 - h[j])
            
            # Find max delta that keeps all corr values <= corr_max
            # corr + delta * g <= corr_max for all positions
            # delta * g[p] <= corr_max - corr[p] for all p
            # For g[p] > 0: delta <= (corr_max - corr[p]) / g[p]
            pos_mask = g > 1e-15
            if pos_mask.any():
                ratios = (corr_max - corr[pos_mask]) / g[pos_mask]
                delta_upper = np.min(ratios)
            else:
                delta_upper = max_delta
            
            delta_upper = min(delta_upper, max_delta)
            if delta_upper <= 1e-14:
                continue
            
            # Take a fraction of delta_upper for safety
            delta = 0.99 * delta_upper
            
            # Verify improvement

            if new_max + dx < corr_max + dx - 1e-12:  # Wait, corr is already in units of h values

                    best_improvement = improvement

                    best_delta = delta
    
    return best_i, best_j, best_delta

    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def fast_vectorized_cd(h_init, n, target, time_limit,
                       n_cand_start=30, n_cand_end=60,
                       max_sweeps=80000,
                       use_multi_shift=True, n_active=5,
                       momentum_beta=0.85):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    momentum = np.zeros(n)
    sweep_count = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        sweep_count += 1

        corr = np.correlate(h, 1.0 - h, mode='full')
        corr_max = float(np.max(corr))
        c5_val = corr_max * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 1000:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        if use_multi_shift and n_active > 1:
            top_shifts = np.argsort(corr)[-n_active:]
            top_sum = np.sum(corr[top_shifts])
            if top_sum < 1e-15:
                top_sum = 1.0
            grad = np.zeros(n)
            for s_idx in top_shifts:
                w = corr[s_idx] / top_sum
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        else:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]

        grad -= np.mean(grad)
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad

        sorted_idx = np.argsort(momentum)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        if current_ncand < 2:
            continue

        g_all = change[low_set[None, :], :] - change[high_set[:, None], :]

        safe_g = np.where(g_all > 1e-15, g_all, 1.0)
        ratios = (corr_max - corr[None, None, :]) / safe_g
        ratios = np.where(g_all > 1e-15, ratios, np.inf)
        delta_corr = np.min(ratios, axis=2)

        max_delta_bounds = np.minimum(h[high_set][:, None], (1.0 - h[low_set])[None, :])
        delta_final = np.minimum(delta_corr, max_delta_bounds)
        delta_final = np.clip(delta_final, 0.0, None)

        mask_valid = delta_final > 1e-13
        if not mask_valid.any():
            continue

        effective_delta = np.where(mask_valid, delta_final, 0.0)
        new_max_all = np.max(corr[None, None, :] + effective_delta[:, :, None] * g_all, axis=2)
        improvement_all = corr_max - new_max_all
        improvement_all = np.where(mask_valid, improvement_all, -1.0)

        best_imp = float(np.max(improvement_all))
        if best_imp > 1e-14:
            flat_idx = int(np.argmax(improvement_all))
            hi_idx = flat_idx // len(low_set)
            lo_idx = flat_idx % len(low_set)
            i = int(high_set[hi_idx])
            j = int(low_set[lo_idx])
            delta = float(effective_delta[hi_idx, lo_idx])

            if delta > 1e-14 and i != j:
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
                momentum[i] = 0.0
                momentum[j] = 0.0

    return best_h, best_c5


def subgradient_step(h, n, target, alpha, n_active=10):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_max = float(np.max(corr))
    active = np.where(corr >= corr_max - 1e-10)[0]
    active = active[-n_active:]

    grad = np.zeros(n)
    for s_idx in active:
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad /= len(active)
    grad -= np.mean(grad)

    h_new = h - alpha * grad
    h_new = project_feasible(h_new, target)
    return h_new


def binary_push(h, target, threshold=0.3):
    n = len(h)
    h2 = h.copy()
    h2[h2 < threshold] = 0.0
    h2[h2 > 1.0 - threshold] = 1.0
    return project_feasible(h2, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 120:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [50, 200, 1000, 5000, 20000],
                                600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 300:
        h_cd, c5_cd = fast_vectorized_cd(best_h, current_n, current_target,
                                         time_limit=min(200, tleft() * 0.5),
                                         n_cand_start=25, n_cand_end=70,
                                         max_sweeps=60000,
                                         use_multi_shift=True, n_active=5,
                                         momentum_beta=0.85)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [10000, 50000, 200000],
                                1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 350:
        n_hi = 800
        target_hi = n_hi / 2.0
        h_hi = interpolate_to(best_h, n_hi, target_hi)
        sc_hi = compute_c5(h_hi)

        if tleft() > 200:
            h_cd_hi, c5_hi = fast_vectorized_cd(h_hi, n_hi, target_hi,
                                                time_limit=min(180, tleft() * 0.45),
                                                n_cand_start=20, n_cand_end=55,
                                                max_sweeps=40000,
                                                use_multi_shift=True, n_active=5,
                                                momentum_beta=0.85)
            if c5_hi < best_score:
                best_score = c5_hi
                best_h = h_cd_hi
                current_n = n_hi
                current_target = target_hi

        if current_n == n_hi and tleft() > 60:
            h_pol_hi, sc_hi_pol = slsqp_round(best_h, n_hi, target_hi,
                                              [10000, 50000, 200000],
                                              800, 1e-15)
            if sc_hi_pol < best_score:
                best_score = sc_hi_pol
                best_h = h_pol_hi

    if tleft() > 150:
        h_cd2, c5_cd2 = fast_vectorized_cd(best_h, current_n, current_target,
                                           time_limit=min(100, tleft() * 0.5),
                                           n_cand_start=40, n_cand_end=80,
                                           max_sweeps=50000,
                                           use_multi_shift=True, n_active=8,
                                           momentum_beta=0.9)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 200:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.1, 0.2]:
            if tleft() < 100:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = fast_vectorized_cd(h_pert, current_n, current_target,
                                               time_limit=min(50, tleft() * 0.25),
                                               n_cand_start=30, n_cand_end=60,
                                               max_sweeps=20000,
                                               use_multi_shift=True, n_active=5,
                                               momentum_beta=0.85)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 80:
        h_bin = binary_push(best_h, current_target, threshold=0.25)
        h_bin_cd, c5_bin = fast_vectorized_cd(h_bin, current_n, current_target,
                                              time_limit=min(50, tleft() * 0.4),
                                              n_cand_start=30, n_cand_end=70,
                                              max_sweeps=20000,
                                              use_multi_shift=False, n_active=1,
                                              momentum_beta=0.9)
        if c5_bin < best_score:
            best_score = c5_bin
            best_h = h_bin_cd

    if tleft() > 80:
        h_cd3, c5_cd3 = fast_vectorized_cd(best_h, current_n, current_target,
                                           time_limit=min(60, tleft() * 0.6),
                                           n_cand_start=50, n_cand_end=90,
                                           max_sweeps=40000,
                                           use_multi_shift=True, n_active=10,
                                           momentum_beta=0.92)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 40:
        h_final, sc_final = slsqp_round(best_h, current_n, current_target,
                                        [50000, 200000, 500000],
                                        600, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
