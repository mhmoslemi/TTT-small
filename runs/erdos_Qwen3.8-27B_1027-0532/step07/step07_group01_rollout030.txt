    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_pnorm(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max < 1e-15:
        return 0.0
    normalized = corr / c_max
    val = np.sum(normalized ** p) ** (1.0 / p)
    return float(c_max * val)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def slsqp_round_pnorm(h_start, n, target, p_values, maxiter_per_p, ftol=1e-15):

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=150, max_no_improve=500,
            restart_every=200, restart_scale=0.005, n_shifts=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(99999)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_every and consecutive_no_improve % 30 == 0:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0

        # Multi-shift gradient
        top_n = min(n_shifts, n_corr)
        top_shifts = np.argsort(corr)[-top_n:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def fw_step(h, n, target, nhw=3):
    """Frank-Wolfe style step: find the polytope vertex that most reduces the linearized objective"""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(np.max(corr)) * dx

    # Get top shifts for subgradient
    top_n = min(nhw, n_corr)
    top_shifts = np.argsort(corr)[-top_n:]
    top_sum = np.sum(corr[top_shifts])
    if top_sum < 1e-15:
        top_sum = 1.0

    # Compute subgradient (weighted)
    grad = np.zeros(n)
    for s_idx in top_shifts:
        w = corr[s_idx] / top_sum
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad -= np.mean(grad)

    # Find the vertex of the polytope {0<=x<=1, sum(x)=target} that minimizes <grad, x>
    # This is: set x_i = 0 if grad_i > 0, x_i = 1 if grad_i < 0,
    # then adjust to meet the sum constraint
    sorted_g = np.argsort(grad)
    # We want to minimize sum(grad_i * x_i)
    # So set x_i = 1 for the smallest grad_i, x_i = 0 for largest
    # Need exactly target ones (approximately)
    n_ones = int(round(target))
    if n_ones > n:
        n_ones = n
    # Set the n_ones smallest gradient positions to 1, rest to 0
    vertex = np.zeros(n)
    vertex[sorted_g[:n_ones]] = 1.0

    # Frank-Wolfe step
    # Try different step sizes
    best_c5 = c5_val
    best_step = 0.0
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        h_trial = (1.0 - t) * h + t * vertex
        h_trial = project_feasible(h_trial, target)
        sc = compute_c5(h_trial)
        if sc < best_c5:
            best_c5 = sc
            best_step = t

    if best_step > 0:
        h_new = (1.0 - best_step) * h + best_step * vertex
        h_new = project_feasible(h_new, target)
        return h_new, compute_c5(h_new)
    return h, c5_val

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)

        if abs(diff) < 1e-15:

    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=150, max_no_improve=500,
            restart_every=200, restart_scale=0.005, n_shifts=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(424242)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_every and consecutive_no_improve % 30 == 0:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0

        top_n = min(n_shifts, n_corr)
        top_shifts = np.argsort(corr)[-top_n:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def fw_step(h, n, target, nhw=3):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = float(np.max(corr)) * dx

    top_n = min(nhw, n_corr)
    top_shifts = np.argsort(corr)[-top_n:]
    top_sum = np.sum(corr[top_shifts])
    if top_sum < 1e-15:
        top_sum = 1.0

    grad = np.zeros(n)
    for s_idx in top_shifts:
        w = corr[s_idx] / top_sum
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad -= np.mean(grad)

    sorted_g = np.argsort(grad)
    n_ones = int(round(target))
    if n_ones > n:
        n_ones = n
    if n_ones < 0:
        n_ones = 0
    vertex = np.zeros(n)
    vertex[sorted_g[:n_ones]] = 1.0

    best_c5 = c5_val
    best_step = 0.0
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
        h_trial = (1.0 - t) * h + t * vertex
        h_trial = project_feasible(h_trial, target)
        sc = compute_c5(h_trial)
        if sc < best_c5:
            best_c5 = sc
            best_step = t

    if best_step > 0:
        h_new = (1.0 - best_step) * h + best_step * vertex
        h_new = project_feasible(h_new, target)
        return h_new, compute_c5(h_new)
    return h, c5_val


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 80:
        h_pol, sc = slsqp_round_lse(best_h, current_n, current_target,
                                     [30, 100, 500, 2000, 10000],
                                     800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 80:
        h_pn, sc_pn = slsqp_round_pnorm(best_h, current_n, current_target,
                                         [8, 16, 32, 64, 128],
                                         600, 1e-15)
        if sc_pn < best_score:
            best_score = sc_pn
            best_h = h_pn

    if tleft() > 250:
        h_cd, c5_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(250, tleft() * 0.55),
                              max_sweeps=80000,
                              n_cand_start=30, n_cand_end=150,
                              max_no_improve=600, restart_scale=0.005,
                              n_shifts=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_fw = best_h.copy()
        for _ in range(40):
            if tleft() < 20:
                break
            h_try, sc_try = fw_step(h_fw, current_n, current_target, nhw=5)
            if sc_try < best_score:
                best_score = sc_try
                best_h = h_try.copy()
                h_fw = h_try.copy()

    if tleft() > 60:
        h_pol2, sc2 = slsqp_round_pnorm(best_h, current_n, current_target,
                                         [32, 128, 512, 2048],
                                         800, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 350:
        n1 = 560
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 120:
            h_pn1, sc1 = slsqp_round_pnorm(h1, n1, target1,
                                            [16, 64, 256],
                                            600, 1e-15)
            sc_check = compute_c5(h_pn1)
            if tleft() > 100:
                h_cd1, c5_cd1 = pair_cd(h_pn1, n1, target1,
                                        time_limit=min(100, tleft() * 0.4),
                                        max_sweeps=40000,
                                        n_cand_start=25, n_cand_end=120,
                                        max_no_improve=500, n_shifts=5)
                if c5_cd1 < best_score:
                    best_score = c5_cd1
                    best_h = h_cd1
                    current_n = n1
                    current_target = target1
            if current_n == n1 and tleft() > 50:
                h_pol1, sc1b = slsqp_round_lse(h_pn1, n1, target1,
                                                [5000, 50000, 500000],
                                                1000, 1e-15)
                if sc1b < best_score:
                    best_score = sc1b
                    best_h = h_pol1

    if tleft() > 200:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.003, 0.008, 0.02, 0.05, 0.12]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = slsqp_round_lse(h_pert, current_n, current_target,
                                            [200, 1000, 5000],
                                            500, 1e-15)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 70:
                h_cd_p, c5_p = pair_cd(h_ref, current_n, current_target,
                                       time_limit=min(40, tleft() * 0.3),
                                       max_sweeps=12000,
                                       n_cand_start=30, n_cand_end=100,
                                       max_no_improve=300, n_shifts=5)
                if c5_p < best_score:
                    best_score = c5_p
                    best_h = h_cd_p

    if tleft() > 100:
        h_cd2, c5_cd2 = pair_cd(best_h, current_n, current_target,
                                time_limit=min(90, tleft() * 0.6),
                                max_sweeps=50000,
                                n_cand_start=40, n_cand_end=180,
                                max_no_improve=600, restart_scale=0.003,
                                n_shifts=7)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 40:
        h_fp, sc_fp = slsqp_round_pnorm(best_h, current_n, current_target,
                                         [128, 512, 2048],
                                         1000, 1e-16)
        if sc_fp < best_score:
            best_score = sc_fp
            best_h = h_fp

    if tleft() > 20:
        h_ult, sc_ult = slsqp_round_lse(best_h, current_n, current_target,
                                         [200000, 1000000, 5000000],
                                         800, 1e-16)
        if sc_ult < best_score:
            best_score = sc_ult
            best_h = h_ult

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
