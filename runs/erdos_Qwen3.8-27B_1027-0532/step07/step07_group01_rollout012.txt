    """Solve the trust region QP and return the step direction d."""
    d = cp.Variable(n)
    t = cp.Variable()
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    c_max = np.max(corr)
    
    # For each active k, compute gradient
    constraints = []
    for k_val in active_k_values:
        k = k_val - mid
        idxs = np.arange(n)
        grad = np.zeros(n)
        pos_mask = (idxs + k >= 0) & (idxs + k < n)
        neg_mask = (idxs - k >= 0) & (idxs - k < n)
        grad[pos_mask] += (1.0 - h[idxs[pos_mask] + k])

        # i.e., grad^T d <= t - g_k(h)

    constraints.append(d <= 1.0 - h)

    prob = cp.Problem(objective, constraints)

    if d.value is not None:
        return d.value
    return np.zeros(n)

    pos = (idxs + k >= 0) & (idxs + k < n)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_k(h, k):
    n = len(h)
    idxs = np.arange(n)
    grad = np.zeros(n)
    pos = (idxs + k >= 0) & (idxs + k < n)
    neg = (idxs - k >= 0) & (idxs - k < n)
    grad[pos] += (1.0 - h[idxs[pos] + k])
    grad[neg] -= h[idxs[neg] - k]
    return grad

def trust_region_solve(h, corr, n, n_active=30, mu=0.001):
    mid = n - 1
    c_max = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1]
    active_idx = sorted_idx[:n_active]
    active_mask = corr >= c_max - 0.03 * max(c_max, 1e-6)
    active_idx = np.where(active_mask)[0]
    if len(active_idx) > 60:
        active_idx = sorted_idx[:60]
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = []
    for ki in active_idx:
        k = int(ki) - mid
        grad = compute_grad_k(h, k)
        constraints.append(grad @ d <= t - corr[ki])
    constraints.append(cp.sum(d) == 0)
    constraints.append(d >= -h - 1e-12)
    constraints.append(d <= 1.0 - h + 1e-12)
    constraints.append(t >= -2.0)
    prob = cp.Problem(cp.Minimize(t + 0.5 * mu * cp.sum_squares(d)), constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000, eps=1e-7)
    except Exception:
        return np.zeros(n)
    if d.value is None:
        return np.zeros(n)
    return np.asarray(d.value).flatten()

def trust_region_optimize(h_init, n, target, time_limit, n_active=25, mu_init=0.01):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    mu = mu_init
    mu_decay = 0.85
    rng = np.random.RandomState(123)
    for it in range(20000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr)) * (2.0 / n)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            noise_scale = 0.005 + 0.01 * (no_improve % 5)
            h = best_h + rng.randn(n) * noise_scale
            h = project_feasible(h, target)
            mu = mu_init
            no_improve = 0
            continue
        d = trust_region_solve(h, corr, n, n_active=n_active, mu=mu)
        if np.linalg.norm(d) < 1e-10:
            mu *= mu_decay
            no_improve += 5
            continue
        alpha = 1.0
        improved = False
        for _ in range(15):
            h_trial = project_feasible(h + alpha * d, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5 - 1e-15:
                h = h_trial
                improved = True
                break
            alpha *= 0.5
        if not improved:
            mu *= mu_decay
        else:
            mu = max(mu * 0.95, 1e-6)
    return best_h, best_c5

def smooth_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter=600):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def multi_shift_cd(h_init, n, target, time_limit, max_sweeps=30000,
                   n_cand_start=20, n_cand_end=80, n_shifts=5,
                   max_no_improve=400, restart_every=200):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(999)
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        if consecutive_no_improve > restart_every and consecutive_no_improve % 40 == 0:
            scale = 0.01 + 0.005 * (consecutive_no_improve % 4)
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0
        top_shifts = np.argsort(corr)[-n_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0
        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)
