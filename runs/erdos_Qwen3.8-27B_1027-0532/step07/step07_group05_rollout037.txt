def project_feasible(h, target_sum):

def smooth_max(corr, alpha):

def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha):
    """SLSQP with smooth max."""
    bounds = [(0.0, 1.0)] * n
    
    def objective(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, alpha)
    
    def objective_grad(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        exp_vals = np.exp(alpha * (corr - np.max(corr)))
        exp_vals = np.clip(exp_vals, 0, np.exp(0))
        weights = exp_vals / np.sum(exp_vals)
        # Gradient of smooth max
        idxs = np.arange(n)
        mid = n - 1
        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            w = weights[k_idx]
            if w < 1e-15:
                continue
            k = k_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - x[idxs[v1] - k])
            grad[v2] -= w * x[idxs[v2] + k]
        return grad
    
    def constraint_fun(x):
        return np.sum(x) - target
    
    def constraint_jac(x):
        return np.ones(n)
    
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(objective, h_cur, args=(alpha,),
                          jac=lambda x, a=alpha: objective_grad(x, a),
                          method='SLSQP', bounds=bounds,

    start = time.time()

    mid = n - 1
    idxs = np.arange(n)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    lr = 0.1  # Initial learning rate
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        # Smooth gradient
        exp_vals = np.exp(alpha * (corr - np.max(corr)))
        exp_vals = np.clip(exp_vals, 0, 1.0)
        w_sum = np.sum(exp_vals)
        if w_sum < 1e-15:
            w_sum = 1.0
        weights = exp_vals / w_sum
        
        grad = np.zeros(n)
        # Only process terms with significant weight
        active = weights > 1e-10
        for k_idx in np.where(active)[0]:

            k = k_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        grad -= np.mean(grad)
        
        # Backtracking line search
        improved = False
        for ftol_factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * ftol_factor
            h_new = h - step * grad
            h_new = project_feasible(h_new, target)

                if c5_new < best_c5:

                    best_h = h.copy()
                improved = True
                break
        
        if not improved:
            lr *= 0.5
            if lr < 1e-10:
                break

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
