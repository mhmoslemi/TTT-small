import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad

def compute_corr_gradient_multi(h, k_stars, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(k_stars, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

def multi_lag_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=20, n_cand_end=100, n_top_lags=3, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        # Get top lags
        sorted_lags = np.argsort(corr)[::-1]
        top_lags = sorted_lags[:n_top_lags]
        
        # Weight by how close each is to the max
        max_corr = corr[sorted_lags[0]]
        weights = np.zeros(n_top_lags)
        for i, lag in enumerate(top_lags):
            weights[i] = corr[lag] / max_corr if max_corr > 0 else 1.0
        weights = weights / weights.sum()

        k_star_idx = int(top_lags[0])
        k_star = k_star_idx - mid

        # Multi-lag gradient
        grad = compute_corr_gradient_multi(h, top_lags, weights)
        grad -= np.mean(grad)

        # Also consider single worst lag gradient
        grad_single = compute_corr_gradient(h, k_star_idx)
        grad_single -= np.mean(grad_single)

        # Use whichever gives better direction (try both)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        best_trial_c5 = c5_val
        best_move = None

        for grad_use in [grad, grad_single]:
            sorted_idx = np.argsort(grad_use)
            low_set = sorted_idx[:current_ncand]
            high_set = sorted_idx[-current_ncand:]

            n_h = len(high_set)
            n_l = len(low_set)
            for ii in range(n_h):
                i = int(high_set[ii])
                for jj in range(n_l):
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo < 1e-12:
                        continue

                    g_vec = change[j] - change[i]
                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(12):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g_vec
                        f_mid = np.max(vals)
                        active_m = vals >= f_mid - 1e-14
                        if active_m.any():
                            f_prime = np.max(g_vec[active_m])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d

                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue

                    approx_max = np.max(corr + delta_star * g_vec) * dx
                    if approx_max < best_trial_c5:
                        best_trial_c5 = approx_max
                        best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change matrix
            change[i, :] = 0.0
            change[j, :] = 0.0
            pos_i = (i + shifts >= 0) & (i + shifts < n)
            neg_i = (i - shifts >= 0) & (i - shifts < n)
            change[i, pos_i] += (1.0 - h[i + shifts[pos_i]])
            change[i, neg_i] -= h[i - shifts[neg_i]]
            pos_j = (j + shifts >= 0) & (j + shifts < n)
            neg_j = (j - shifts >= 0) & (j - shifts < n)
            change[j, pos_j] += (1.0 - h[j + shifts[pos_j]])
            change[j, neg_j] -= h[j - shifts[neg_j]]

    return best_h, best_c5

def cutting_plane_hybrid(h_init, n, target, time_limit, max_iters=20000, tol=1e-10):
    """Enhanced cutting plane with adaptive cut management."""
    start = time.time()
    mid = n - 1
    dx = 2.0 / n

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_b = []  # b_i = g_i^T h_i - f_i (so constraint is g^T h - t <= b_i)
    
    max_cuts = 300
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            # Keep only most violated cuts
            if len(cuts_g) > 20:
                violations = []
                for i in range(len(cuts_g)):
                    v = cuts_g[i] @ h - cuts_b[i] + 1.0  # t = 1 for upper bound
                    violations.append(v)
                violations = np.array(violations)
                keep = np.argsort(violations)[::max(1, len(violations)//40)]
                cuts_g = [cuts_g[i] for i in keep]
                cuts_b = [cuts_b[i] for i in keep]
            no_improve = 0

        # Add cuts for multiple active lags
        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])

        active_mask = corr >= corr[k_star] - 1e-5
        active_lags = np.where(active_mask)[0]
        n_active = len(active_lags)

        if n_active == 1:
            g = compute_corr_gradient(h, k_star)
            cuts_g.append(g)
            cuts_b.append(float(g @ h) - f_val)
        else:
            # Add cuts for top few lags
            top_n = min(5, n_active)
            for lag in active_lags[:top_n]:
                g = compute_corr_gradient(h, int(lag))
                cuts_g.append(g)
                cuts_b.append(float(g @ h) - float(corr[lag]))

        if len(cuts_g) > max_cuts:
            # Keep oldest + most recently added
            keep = max_cuts // 2
            cuts_g = cuts_g[:5] + cuts_g[-keep:]
            cuts_b = cuts_b[:5] + cuts_b[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = cuts_b[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                continue
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            continue

    return best_h, best_score

def nesterov_subgrad(h_init, n, target, time_limit, max_iters=50000):
    """Nesterov accelerated subgradient method with Polyak averaging."""
    start = time.time()
    dx = 2.0 / n
    mid = n - 1

    h_curr = h_init.copy().astype(float)
    h_avg = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)

    # Estimate upper bound
    f_star = compute_c5(h_init) * 1.1  # initial upper bound estimate
    best_ub = f_star

    idxs = np.arange(n)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        # Step size
        step = 0.1 / np.sqrt(1.0 + it * 0.01)
        
        # Nesterov momentum
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it / 4.0)) / 2.0
        y = h_curr + ((t_k - 1.0) / t_k) * (h_curr - h_avg if it > 0 else np.zeros(n))
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)

        # Compute subgradient at y
        corr_y = compute_corr(y)
        c_max = float(np.max(corr_y))
        
        # Use smooth approximation for gradient
        alpha = 500.0 + 100.0 * it / max_iters * 4900
        shifted = np.clip(alpha * (corr_y - c_max), -500.0, 0.0)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-30:
            p = np.zeros(len(corr_y))
            k_star = int(np.argmax(corr_y))
            p[k_star] = 1.0
        else:
            p = p / p_sum

        # Compute gradient from probability distribution
        grad = np.zeros(n)
        for m in range(len(corr_y)):
            if p[m] < 1e-15:
                continue
            k = m - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += p[m] * (1.0 - y[idxs[v1] + k])
            grad[v2] -= p[m] * y[idxs[v2] - k]

        # Gradient step
        h_new = y - step * grad
        h_new = project_feasible(h_new, target)

        # Update average (Polyak)
        f_new = compute_c5(h_new)
        h_avg = h_avg + (h_new - h_avg) / (it + 1)
        h_avg = project_feasible(h_avg, target)

        h_curr = h_new

        # Track best
        if f_new < best_score - 1e-14:
            best_score = f_new
            best_h = h_curr.copy()

        # Also check average
        f_avg = compute_c5(h_avg)
        if f_avg < best_score - 1e-14:
            best_score = f_avg
            best_h = h_avg.copy()

    return best_h, best_score

def smooth_slq(h_init, n, target, alphas, maxiter_per_alpha):
    """SLSQP with smooth log-sum-exp objective."""
    dx = 2.0 / n
    
    def obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

    def cons_fun(x):
        return np.sum(x) - target

    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]

    for alpha in alphas:
        try:
            res = minimize(obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def random_perturb(h, n, target, rng, scale):
    h_pert = h + rng.randn(n) * scale
    return project_feasible(h_pert, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def nesterov_avg(h_init, n, target, time_limit, max_iters):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h_curr = h_init.copy().astype(float)
    h_prev = h_curr.copy()
    h_avg = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 8.0 * it + 1.0))
        y = h_curr + ((t_k - 1.0) / t_k) * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)

        alpha = 300.0 + 4700.0 * it / max(max_iters, 1)
        corr_y = compute_corr(y)
        c_max = np.max(corr_y)
        shifted = np.clip(alpha * (corr_y - c_max), -500.0, 0.0)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-30:
            p = np.zeros(len(corr_y))
            p[int(np.argmax(corr_y))] = 1.0
        else:
            p = p / p_sum

        grad = np.zeros(n)
        for m in range(len(corr_y)):
            if p[m] < 1e-14:
                continue
            k = m - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += p[m] * (1.0 - y[idxs[v1] + k])
            grad[v2] -= p[m] * y[idxs[v2] - k]

        step = 0.05 / (1.0 + 0.005 * it)
        h_new = y - step * grad
        h_new = project_feasible(h_new, target)

        h_avg = h_avg + (h_new - h_avg) / (it + 1)
        h_avg = project_feasible(h_avg, target)

        h_prev = h_curr.copy()
        h_curr = h_new

        f_new = compute_c5(h_new)
        if f_new < best_score - 1e-14:
            best_score = f_new
            best_h = h_curr.copy()
        f_avg = compute_c5(h_avg)
        if f_avg < best_score - 1e-14:
            best_score = f_avg
            best_h = h_avg.copy()

    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):

    if tleft() > 50:
        h_cp_f, sc_cp_f = cutting_plane_enhanced(best_h, n0, target0,
                                                   min(40, tleft() * 0.5), 10000, 1e-11)
        if sc_cp_f < best_score:
            best_score = sc_cp_f
            best_h = h_cp_f

    if tleft() > 30:
        h_cd_f, sc_cd_f = multi_lag_cd(best_h, n0, target0,
                                         min(25, tleft() * 0.5), 15000,
                                         30, 80, 3, 150)
        if sc_cd_f < best_score:
            best_score = sc_cd_f
            best_h = h_cd_f

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
