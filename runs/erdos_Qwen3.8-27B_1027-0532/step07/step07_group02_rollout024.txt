def subgradient_step(h, n, alpha_warm, n_iter):
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Compute gradient with softmax weighting
        shifted = corr - np.max(corr)
        weights = np.exp(shifted / alpha_warm)
        weights /= np.sum(weights)
        
        # Compute weighted gradient
        grad = np.zeros(n)
        mid = n - 1
        for s in range(2*n - 1):
            if weights[s] < 1e-10:
                continue
            k = s - mid
            if k >= 0:
                # corr[s] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
                # d/dh[p] of h[i]*(1-h[i+k]) is (1-h[i+k]) if i=p, -h[i] if i+k=p
                for i in range(n - k):
                    grad[i] += weights[s] * (1.0 - h[i+k])
                    grad[i+k] -= weights[s] * h[i]
            elif k < 0:
                # corr[s] = sum_{i=0}^{n-1+k} h[i-k]*(1-h[i])
                # Actually for negative k, let k' = -k > 0
                # corr = sum_{i=k'}^{n-1} h[i]*(1-h[i-k'])
                # = sum_{j=0}^{n-1-k'} h[j+k']*(1-h[j])
                k_neg = -k
                for j in range(n - k_neg):
                    grad[j+k_neg] += weights[s] * (1.0 - h[j])
                    grad[j] -= weights[s] * h[j+k_neg]
        
        # Center the gradient
        grad -= np.mean(grad)
        
        # Adaptive step size
        step = alpha_warm / (np.linalg.norm(grad) + 1e-10)
        
        # Take step
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        h = h_new
    
    return best_h, best_c5

def compute_weighted_grad(h, weights, n):
    mid = n - 1
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(2*n - 1):
        w = weights[s]
        if w < 1e-10:
            continue
        k = s - mid
        if k >= 0:
            # corr[s] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dh[i] = (1-h[i+k]) for i in [0, n-1-k]
            # d/dh[i+k] = -h[i] for i in [0, n-1-k], i.e., h[j] for j in [k, n-1]
            grad[:n-k] += w * one_minus_h[k:]
            grad[k:] -= w * h[:n-k]
        else:
            k_abs = -k
            # corr[s] = sum_{i=0}^{n-1-k_abs} h[i+k_abs]*(1-h[i])
            # d/dh[i+k_abs] = (1-h[i]) for i in [0, n-1-k_abs]
            # d/dh[i] = -h[i+k_abs] for i in [0, n-1-k_abs]
            grad[k_abs:] += w * one_minus_h[:n-k_abs]
            grad[:n-k_abs] -= w * h[k_abs:]
    return grad

            update_change_row(change, best_j, h, n, shifts)
        else:
            if stall > 50 and stall % 40 == 0:
                npert = max(5, n // 25)
                ip = rng.choice(n, size=npert, replace=False)
                h[ip] += rng.randn(npert) * perturb_scale
                h = project_feasible(h, target)
                change = compute_change_matrix(h, n)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    t0 = time.time()
    def tl():
        return budget_s - 15.0 - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    h = project_feasible(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    if tl() > 80:
        hp, sp = slsqp_optimize(best_h, n, target, [100, 500, 2000], 800, 1e-16)
        if sp < best_c5:
            best_c5 = sp
            best_h = hp

    if tl() > 80:
        hf, sf = frank_wolfe_steps(best_h, n, target, 60, seed=seed)
        if sf < best_c5:
            best_c5 = sf
            best_h = hf
        hc, sc = local_search(best_h, n, target, min(60, tl() * 0.5),
                              10000, 60, 20, seed + 1, max_stall=300)
        if sc < best_c5:
            best_c5 = sc
            best_h = hc

    if tl() > 120:
        hc, sc = local_search(best_h, n, target, min(150, tl() * 0.5),
                              30000, 80, 22, seed + 2, max_stall=500,
                              perturb_scale=0.03)
        if sc < best_c5:
            best_c5 = sc
            best_h = hc

    if tl() > 60:
        hp, sp = slsqp_optimize(best_h, n, target, [2000, 8000, 30000], 1500, 1e-16)
        if sp < best_c5:
            best_c5 = sp
            best_h = hp

    if tl() > 100:
        hc, sc = local_search(best_h, n, target, min(120, tl() * 0.6),
                              25000, 100, 22, seed + 3, max_stall=400,
                              perturb_scale=0.05)
        if sc < best_c5:
            best_c5 = sc
            best_h = hc

    if tl() > 200:
        n2 = 600
        tgt2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, tgt2)
        if tl() > 150:
            hc2, sc2 = local_search(h2, n2, tgt2, min(120, tl() * 0.5),
                                     20000, 60, 20, seed + 4, max_stall=350)
            if sc2 < best_c5:
                best_c5 = sc2
                best_h = hc2
                n, target = n2, tgt2
        if tl() > 60:
            hp2, sp2 = slsqp_optimize(best_h, n, target, [1000, 5000], 1000, 1e-16)
            if sp2 < best_c5:
                best_c5 = sp2
                best_h = hp2

    if tl() > 150:
        n3 = 800
        tgt3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, tgt3)
        if tl() > 100:
            hc3, sc3 = local_search(h3, n3, tgt3, min(80, tl() * 0.5),
                                     15000, 50, 18, seed + 5, max_stall=300)
            if sc3 < best_c5:
                best_c5 = sc3
                best_h = hc3
                n, target = n3, tgt3
        if tl() > 40:
            hp3, sp3 = slsqp_optimize(best_h, n, target, [2000, 10000], 800, 1e-16)
            if sp3 < best_c5:
                best_c5 = sp3
                best_h = hp3

    if tl() > 80:
        hc, sc = local_search(best_h, n, target, min(60, tl() * 0.7),
                              15000, 120, 24, seed + 6, max_stall=400,
                              perturb_scale=0.02)
        if sc < best_c5:
            best_c5 = sc
            best_h = hc

    if tl() > 30:
        hp, sp = slsqp_optimize(best_h, n, target, [20000, 100000], 2000, 1e-16)
        if sp < best_c5:
            best_c5 = sp
            best_h = hp

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
