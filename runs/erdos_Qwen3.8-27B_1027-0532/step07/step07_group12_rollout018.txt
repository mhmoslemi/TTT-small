from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def block_pair_cd(h_init, n, target, time_limit, block_size=3, n_cand=80, max_sweeps=50000):
    """Block coordinate descent: move mass between blocks of consecutive points."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    no_improve = 0
    idxs = np.arange(n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Compute gradient for the dominant shift
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        
        # Block-based candidate selection

        n_blocks = n // block_size
        block_grads = np.array([grad[b*block_size:(b+1)*block_size].mean() for b in range(n_blocks)])
        
        # Pick top and bottom blocks
        top_blocks = np.argsort(-block_grads)[:n_cand]
        bot_blocks = np.argsort(block_grads)[:n_cand]
        
        # Within selected blocks, pick individual points
        high_points = set()
        low_points = set()
        for b in top_blocks:
            high_points.update(range(int(b)*block_size, min((int(b)+1)*block_size, n)))
        for b in bot_blocks:
            low_points.update(range(int(b)*block_size, min((int(b)+1)*block_size, n)))
        
        high_set = np.array(sorted(high_points))[:min(100, len(high_points))]
        low_set = np.array(sorted(low_points))[:min(100, len(low_points))]
        
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        if nc < 2:
            continue
        
        # Compute profile for each candidate
        # profile[c, m] = change in corr[m] when we increase h at position c by 1
        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):

            ks = np.arange(-(n-1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        
        cand_pos = {int(all_cands[p]): p for p in range(nc)}
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        # Try all pairs
        for i in high_set:

            for j in low_set:

                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                
                # Binary search for optimal delta
                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        
        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def multi_shift_gradient(h, n, n_shifts=5, alpha_w=30.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    mid = n - 1
    idxs = np.arange(n)
    top_idx = np.argsort(corr)[-n_shifts:]
    w = np.exp(alpha_w * (corr[top_idx] - cur_max))
    w = w / w.sum()
    grad = np.zeros(n)
    for wi, si in zip(w, top_idx):
        k = int(si) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    return grad, corr

def pair_cd(h_init, n, target, time_limit, n_cand=100, max_sweeps=100000, n_shifts=1):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if sweep % 50 == 0 and time.time() - start > time_limit * 0.95:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            w = np.exp(40.0 * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        if nc < 2:
            continue

        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

def lse_optimize(h_init, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen in [penalty, penalty * 10.0]:
            if time.time() - start > time_limit:
                break
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
            except Exception:
                pass

    return best_h, best_c5

def subgradient_phase(h_init, n, target, time_limit, max_steps=50000, n_shifts=5, base_lr=0.05):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        grad, corr = multi_shift_gradient(h, n, n_shifts=n_shifts, alpha_w=25.0)
        cur_c5 = float(np.max(corr)) * dx
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 1500:
                break

        grad -= (2.0 / n) * np.sum(grad)
        improved = False
        decay = 1.0 / (1.0 + 0.003 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < cur_c5 - 1e-15:
                h = h_try
                improved = True
                break
        if not improved and step > 3000:
            break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        hcd, c5cd = pair_cd(best_h, n, target, time_limit=tl,
                           n_cand=200, max_sweeps=200000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 9: Multi-resolution - go to n=1000
    if tleft() > 300:
        n2 = 1000
        target2 = n2 / 2.0
        h_interp = interpolate_to(best_h, n2, target2)
        tl = min(100, tleft() * 0.25)
        hp, sc = lse_optimize(h_interp, n2, target2,
                             alphas=[50, 500, 5000, 50000],
                             maxiter=200, time_limit=tl, penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n2
            target = target2
        if tleft() > 120:
            tl = min(100, tleft() * 0.3)
            hcd, c5cd = pair_cd(best_h, n, target, time_limit=tl,
                               n_cand=100, max_sweeps=100000, n_shifts=1)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
                n = n2
                target = target2

    # Phase 10: Try alternative starting points
    if tleft() > 150:
        rng = np.random.RandomState(seed)
        alternatives = []
        # Golden ratio construction
        hg = make_golden_h(n)
        alternatives.append(hg)
        # Structured sinusoidal
        hs = make_structured_h(n, freq=5)
        alternatives.append(hs)
        # Random binary
        hr = make_random_h(n, seed=seed + 1)
        alternatives.append(hr)

        for alt in alternatives:
            if tleft() < 60:
                break
            # Quick optimize the alternative
            tl = min(40, tleft() * 0.2)
            ha, sca = pair_cd(alt, n, target, time_limit=tl,
                             n_cand=60, max_sweeps=30000, n_shifts=1)
            if sca < best_score:
                best_score = sca
                best_h = ha
                n = len(best_h)
                target = n / 2.0
                # Polish
                if tleft() > 50:
                    tl2 = min(40, tleft() * 0.3)
                    hcd2, c5cd2 = pair_cd(best_h, n, target, time_limit=tl2,
                                         n_cand=100, max_sweeps=50000, n_shifts=1)
                    if c5cd2 < best_score:
                        best_score = c5cd2
                        best_h = hcd2

    # Phase 11: Final pair CD polishing
    if tleft() > 40:
        tl = min(30, tleft() * 0.5)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=tl,
                           n_cand=80, max_sweeps=50000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 12: Final L-BFGS polish
    if tleft() > 30:
        tl = min(20, tleft() * 0.6)
        hp, sc = lse_optimize(best_h, n, target,
                             alphas=[500000, 2000000],
                             maxiter=100, time_limit=tl, penalty=50000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
