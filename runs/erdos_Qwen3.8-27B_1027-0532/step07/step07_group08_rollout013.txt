import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, corr, n, active_shifts, weights):
    """Compute subgradient of C5 at h given active shifts and weights."""
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s_idx, w in zip(active_shifts, weights):
        k = int(s_idx) - mid
        if k == 0:
            continue  # gradient is zero for shift 0 (h_i(1-h_i))
        # grad_i += w * [(1-h_{i+k}) - h_{i-k}]
        # First term: (1-h_{i+k}), valid when 0 <= i+k < n, i.e., -k <= i < n-k
        lo1, hi1 = max(0, -k), min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1+k:hi1+k])
        # Second term: -h_{i-k}, valid when 0 <= i-k < n, i.e., k <= i < n+k
        lo2, hi2 = max(0, k), min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2-k:hi2-k]
    return grad

    stale = 0

        else:

        active = np.where(corr >= threshold)[0]

        total = active_corr.sum()
        if total < 1e-10:

        grad = compute_gradient(h, corr, n, active, weights)

        v = np.zeros(n)

        if np.max(np.abs(d)) < 1e-10:
            break

        a = np.correlate(h, d, mode='full') + np.correlate(d, 1.0 - h, mode='full')
        b = np.correlate(d, d, mode='full')
        
        if use_gps:

            gr = (np.sqrt(5) - 1) / 2
            t_lo, t_hi = 0.0, 1.0
            t1 = t_lo + gr * (t_hi - t_lo)
            t2 = t_lo + (1-gr) * (t_hi - t_lo)
            f1 = float(np.max(corr - t1*a - t1*t1*b)) * dx
            f2 = float(np.max(corr - t2*a - t2*t2*b)) * dx
            for _ in range(30):

                    t_hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = t_lo + gr * (t_hi - t_lo)
                    f1 = float(np.max(corr - t1*a - t1*t1*b)) * dx
                else:
                    t_lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = t_lo + (1-gr) * (t_hi - t_lo)
                    f2 = float(np.max(corr - t2*a - t2*t2*b)) * dx
            t_star = (t_lo + t_hi) / 2
        else:

            t_star = 2.0 / (iter_count + 2)
        
        h_new = (1.0 - t_star) * h + t_star * v
        h_new = project_feasible(h_new, target)
