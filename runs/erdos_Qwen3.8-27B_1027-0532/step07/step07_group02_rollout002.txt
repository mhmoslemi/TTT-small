        if best_i < 0:
            continue  # no improvement found with screening delta
        
        # Phase 2: Binary search for optimal delta on the best pair
        gi = g[best_i]
        gj = g[best_j]
        delta_dir = gj - gi  # corr changes by delta * delta_dir
        
        delta_lo = max(-(1.0 - h[best_i]), -h[best_j])  # h[best_i] - delta >= 0, h[best_j] + delta <= 1
        # Wait: h[best_i] decreases by delta, so delta <= h[best_i]
        # h[best_j] increases by delta, so delta <= 1 - h[best_j]
        # And delta >= 0 (we're decreasing h[best_i] and increasing h[best_j])
        # Actually from the gradient: grad[best_i] > 0 (in high_set) means we want to decrease h[best_i]
        # grad[best_j] < 0 (in low_set) means we want to increase h[best_j]
        # So: h[best_i] -= delta, h[best_j] += delta, with delta > 0
        delta_lo = 0.0
        delta_hi = min(h[best_i], 1.0 - h[best_j])

            continue

            delta_mid = 0.5 * (delta_lo + delta_hi)
            test_corr = corr + delta_mid * delta_dir
            test_max = np.max(test_corr)
            # Check subderivative: if test_max is at the max, the derivative w.r.t. delta
            # is delta_dir at the argmax position
            active = test_corr >= test_max - 1e-12
            if active.any():
                deriv = np.max(delta_dir[active])
            else:
                deriv = 0.0
            if deriv > 1e-12:
                delta_hi = delta_mid
            else:
                delta_lo = delta_mid
        
        delta_star = 0.5 * (delta_lo + delta_hi)
        if delta_star < 1e-12:
            continue
        
        # Apply the update
        h[best_i] -= delta_star
        h[best_j] += delta_star
        
        # Update corr
        corr = corr + delta_star * delta_dir
        
        # Update g matrix
        # h[best_i] decreased by delta_star
        # For all s: g[best_i - k_s](s) += delta_star (if valid)
        # For all s: g[best_i + k_s](s) += delta_star (if valid)
        # h[best_j] increased by delta_star
        # For all s: g[best_j - k_s](s) -= delta_star (if valid)
        # For all s: g[best_j + k_s](s) -= delta_star (if valid)

        # g[p](s) = (1-h[p+k_s]) - h[p-k_s]
        # When h[best_i] decreases by d:
        #   g[best_i - k_s](s) gets +d in the (1-h[...]) term if best_i - k_s + k_s = best_i
        #     i.e., g[p](s) where p + k_s = best_i, so p = best_i - k_s
        #   g[best_i + k_s](s) gets +d in the -h[...] term if best_i + k_s - k_s = best_i
        #     i.e., g[p](s) where p - k_s = best_i, so p = best_i + k_s
        # When h[best_j] increases by d:
        #   g[best_j - k_s](s) gets -d in the (1-h[...]) term
        #   g[best_j + k_s](s) gets -d in the -h[...] term
        
        # For h[best_i] -= delta_star (decrease by delta_star):
        #   (1 - h[p+k_s]) term: affected when p+k_s = best_i, i.e., p = best_i - k_s
        #     h[best_i] decreased, so (1-h[best_i]) increased by delta_star
        #     g[best_i - k_s](s) += delta_star for valid s
        #   -h[p-k_s] term: affected when p-k_s = best_i, i.e., p = best_i + k_s
        #     h[best_i] decreased, so -h[best_i] increased by delta_star
        #     g[best_i + k_s](s) += delta_star for valid s
        
        # For h[best_j] += delta_star (increase by delta_star):
        #   (1 - h[p+k_s]) term: affected when p+k_s = best_j, i.e., p = best_j - k_s
        #     h[best_j] increased, so (1-h[best_j]) decreased by delta_star
        #     g[best_j - k_s](s) -= delta_star for valid s
        #   -h[p-k_s] term: affected when p-k_s = best_j, i.e., p = best_j + k_s
        #     h[best_j] increased, so -h[best_j] decreased by delta_star

_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    g = np.zeros((n, n_corr))
    for p in range(n):
        mask1 = (p + shifts >= 0) & (p + shifts < n)
        g[p, mask1] += (1.0 - h[p + shifts[mask1]])
        mask2 = (p - shifts >= 0) & (p - shifts < n)
        g[p, mask2] -= h[p - shifts[mask2]]
    return g

def update_change_after_swap(g, h, i, j, delta, n, shifts):
    for s in range(len(shifts)):
        k_s = shifts[s]
        p1 = i - k_s
        if 0 <= p1 < n:
            g[p1, s] += delta
        p2 = i + k_s
        if 0 <= p2 < n:
            g[p2, s] += delta
        p3 = j - k_s
        if 0 <= p3 < n:
            g[p3, s] -= delta
        p4 = j + k_s
        if 0 <= p4 < n:
            g[p4, s] -= delta

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

        new_corr = corr + delta_star * delta_dir
        new_c5 = float(np.max(new_corr)) * dx
        delta_e = new_c5 - c5_val
        if delta_e < 0 or rng.random() < np.exp(-delta_e / max(temp, 1e-8)):
            h[i] -= delta_star
            h[j] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = new_corr
            update_change_after_swap(g, h, i, j, delta_star, n, shifts)
            cur_c5 = new_c5
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def adaptive_screen_cd(h_init, n, target, time_limit, seed=42,
                       base_n_cand=80, base_screen=0.01,
                       max_no_improve=350, binsearch_iter=25):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    g = build_change_matrix(h, n)
    consecutive_no_improve = 0
    screen_d = base_screen
    n_cand_cur = base_n_cand
    while time.time() - start < time_limit:
        c5_raw = float(np.max(corr))
        c5_val = c5_raw * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            screen_d = max(0.005, screen_d * 0.7)
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            n_pert = n // 10
            idxs_pert = rng.choice(n, size=n_pert, replace=False)
            h[idxs_pert] += rng.randn(n_pert) * 0.03
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            g = build_change_matrix(h, n)
            consecutive_no_improve = 0
            screen_d = 0.03
            n_cand_cur = min(base_n_cand + 40, n // 2)
            continue
        if consecutive_no_improve > max_no_improve // 2:
            n_cand_cur = min(n_cand_cur + 10, n // 2)
            screen_d = max(0.005, screen_d * 1.5)
        k_star_idx = int(np.argmax(corr))
        grad = -g[:, k_star_idx].copy()
        grad -= np.mean(grad)
        nc = min(n_cand_cur, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:nc]
        high_set = sorted_idx[-nc:]
        best_trial = c5_raw
        best_i = -1
        best_j = -1
        g_low = g[low_set]
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            base = corr - screen_d * g[i]
            matrix = base[None, :] + screen_d * g_low
            maxes = matrix.max(axis=1)
            min_idx = int(np.argmin(maxes))
            if maxes[min_idx] < best_trial:
                best_trial = maxes[min_idx]
                best_i = i
                best_j = int(low_set[min_idx])
        if best_i < 0:
            if consecutive_no_improve % 100 == 0:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i != j:
                    delta_dir = g[j] - g[i]
                    delta_hi_r = min(h[i], 1.0 - h[j])
                    if delta_hi_r > 1e-10:
                        delta_r = delta_hi_r * 0.5 * rng.random()
                        test_corr = corr + delta_r * delta_dir
                        if np.max(test_corr) < c5_raw + 0.001:
                            h[i] -= delta_r
                            h[j] += delta_r
                            h = np.clip(h, 0.0, 1.0)
                            h = project_feasible(h, target)
                            corr = test_corr
                            update_change_after_swap(g, h, i, j, delta_r, n, shifts)
            continue
        delta_dir = g[best_j] - g[best_i]
        delta_lo = 0.0
        delta_hi = min(h[best_i], 1.0 - h[best_j])
        if delta_hi - delta_lo < 1e-13:
            continue
        for _ in range(binsearch_iter):
            delta_mid = 0.5 * (delta_lo + delta_hi)
            test_corr = corr + delta_mid * delta_dir
            test_max = np.max(test_corr)
            active = test_corr >= test_max - 1e-12
            if active.any():
                deriv = np.max(delta_dir[active])
            else:
                deriv = 0.0
            if deriv > 1e-12:
                delta_hi = delta_mid
            else:
                delta_lo = delta_mid
        delta_star = 0.5 * (delta_lo + delta_hi)
        if delta_star < 1e-12:
            continue
        h[best_i] -= delta_star
        h[best_j] += delta_star
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        corr = corr + delta_star * delta_dir
        update_change_after_swap(g, h, best_i, best_j, delta_star, n, shifts)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init_raw = np.array(initial_h_values, dtype=float)
    n0 = len(h_init_raw)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init_raw, target0)
    best_score = compute_c5(best_h)

    if tleft() > 80:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                       [100, 500, 2000], 500, 1e-15)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    if tleft() > 300:
        h_cd, sc_cd = adaptive_screen_cd(best_h, n0, target0,
                                         time_limit=min(500, tleft() * 0.5),
                                         seed=42, base_n_cand=80,
                                         base_screen=0.01,
                                         max_no_improve=400,
                                         binsearch_iter=25)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 200:
        h_ms, sc_ms = multi_shift_cd(best_h, n0, target0,
                                     time_limit=min(200, tleft() * 0.35),
                                     seed=137, n_cand=70, n_shifts=3,
                                     screen_delta=0.01, max_no_improve=300)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms

    if tleft() > 150:
        h_cd2, sc_cd2 = adaptive_screen_cd(best_h, n0, target0,

                                           seed=256, base_n_cand=100,

                                           max_no_improve=500,
                                           binsearch_iter=28)
        if sc_cd2 < best_score:

            best_h = h_cd2

    if tleft() > 100:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                       [5000, 20000, 80000], 2000, 1e-15)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    if tleft() > 120:
        h_sa, sc_sa = simulated_annealing_cd(best_h, n0, target0,
                                             time_limit=min(100, tleft() * 0.4),
                                             seed=789, n_cand=80,
                                             screen_delta=0.01,
                                             t_start=0.5, t_end=0.01)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    if tleft() > 80:
        h_cd3, sc_cd3 = fast_cd(best_h, n0, target0,
                                time_limit=min(70, tleft() * 0.5),
                                seed=501, n_cand=120,
                                screen_delta=0.005,
                                max_no_improve=600, binsearch_iter=28)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    if tleft() > 250:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 200:
            h_cd1, sc_cd1 = adaptive_screen_cd(h1, n1, target1,
                                               time_limit=min(180, tleft() * 0.4),
                                               seed=314, base_n_cand=60,
                                               base_screen=0.01,
                                               max_no_improve=300,
                                               binsearch_iter=22)
            if sc_cd1 < best_score:
                best_score = sc_cd1
                best_h = h_cd1
        if tleft() > 80:
            h_pol1, sc_pol1 = slsqp_optimize(best_h, len(best_h), len(best_h)/2,
                                             [2000, 10000, 40000], 1500, 1e-15)
            if sc_pol1 < best_score:
                best_score = sc_pol1
                best_h = h_pol1

    if tleft() > 200:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 150:
            h_cd2b, sc_cd2b = adaptive_screen_cd(h2, n2, target2,
                                                 time_limit=min(130, tleft() * 0.4),
                                                 seed=628, base_n_cand=50,
                                                 base_screen=0.01,
                                                 max_no_improve=250,
                                                 binsearch_iter=20)
            if sc_cd2b < best_score:
                best_score = sc_cd2b
                best_h = h_cd2b
        if tleft() > 60:
            h_pol2, sc_pol2 = slsqp_optimize(best_h, len(best_h), len(best_h)/2,
                                             [5000, 20000], 1000, 1e-15)
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_h = h_pol2

    if tleft() > 60:
        h_final_cd, sc_final_cd = fast_cd(best_h, len(best_h), len(best_h)/2,
                                          time_limit=min(50, tleft() * 0.5),

    final_h = project_feasible(best_h, final_n / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, final_n
