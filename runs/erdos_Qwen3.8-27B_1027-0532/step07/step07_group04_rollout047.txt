from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_corr(h, k_star):
    """Compute gradient of corr[k_star] with respect to h."""
    n = len(h)
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    return grad

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = alpha * corr * dx
    scaled -= np.max(scaled)  # for numerical stability
    exp_scaled = np.exp(scaled)
    SUM = np.sum(exp_scaled)
    obj = (np.max(scaled) + np.log(SUM)) / alpha  # = max + log(SUM/alpha)
    
    # Gradient: softmax weights times grad of each corr[k]
    weights = exp_scaled / SUM  # softmax
    # weights[k] * dx is the weight for corr[k]
    # We need sum_k weights[k] * grad_corr[k] * dx
    # But grad of obj w.r.t. x is sum_k weights[k] * grad(corr[k]) * dx / ... 
    # Actually: d/dx [log(sum exp(alpha*corr[k]*dx))/alpha]
    # = (1/alpha) * sum_k exp(alpha*corr[k]*dx)/sum_j exp(...) * alpha*dx * d(corr[k])/dx
    # = dx * sum_k weights[k] * grad_corr[k]
    
    mid = n - 1
    grad = np.zeros(n)
    # Only need to compute for shifts with significant weight
    # For efficiency, only use top-k shifts
    k_threshold = np.sum(weights > 1e-10)
    if k_threshold > 20:
        top_idx = np.argsort(weights)[-20:]
    else:
        top_idx = np.where(weights > 1e-10)[0]

        if w < 1e-12:

        v1 = (idxs - k >= 0) & (idxs - k < n)

        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    
    return obj, grad

                       max_iters=5000, lr_start=0.01):

    start = time.time()

            break

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

            mask = h < 1.0 - 1e-15
        else:

            h = np.clip(h + diff / n, 0.0, 1.0)
            break

        h = np.clip(h, 0.0, 1.0)

def compute_change_matrix(h, n):

        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        k_star_idx = int(np.argmax(corr))
        g_active = change[high_set][:, k_star_idx] - change[low_set][:, k_star_idx]
        d_lo = np.maximum(h[high_set][:, None] - 1.0, -h[low_set][None, :])
        d_hi = np.minimum(h[high_set][:, None], 1.0 - h[low_set][None, :])
        valid = (g_active > 1e-10) & (d_hi - d_lo > 1e-12)

        if not valid.any():
            consecutive_no_improve += 5
            continue

        n_h = len(high_set)
        g_vals = np.where(valid, g_active, -np.inf)
        flat_idx = np.argsort(g_vals.ravel())[::-1]
        checked = 0
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for fidx in flat_idx:
            if checked >= max_pairs_check:
                break
            ii = int(fidx) % n_h
            jj = int(fidx) // n_h
            if not valid[ii, jj]:
                continue
            checked += 1
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = float(d_lo[ii, jj])
            delta_hi = float(d_hi[ii, jj])
            g_vec = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(14):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g_vec
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    fp_max = np.max(g_vec[active_mask])
                    fp_min = np.min(g_vec[active_mask])
                else:
                    fp_max = 0.0
                    fp_min = 0.0
                if fp_min > 1e-12:
                    b_d = mid_d
                elif fp_max < -1e-12:
                    a_d = mid_d
                else:
                    break

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g_vec) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts_arr)
            update_change_row(change, best_j, h, n, shifts_arr)

    return best_h, best_c5

def sa_cd(h_init, n, target, time_limit, max_steps=20000,
          t_start=0.03, t_end=0.0002, n_cand=50, seed=123):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 500:
            break

        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue

        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue

        g_vec = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g_vec) * dx
        delta_c = approx_new_c5 - c5_val

        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

            if approx_new_c5 < best_c5 - 1e-14:
                best_c5 = approx_new_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0

    # Phase 1: Quick SLSQP on initial grid
    tl = tleft()
    if tl > 60:
        h_pol, sc = slsqp_refine(best_h, n0, target0,
                                  [200, 1000, 5000],
                                  600, 1e-14,
                                  time_limit=min(50, tl * 0.7), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD on n=600
    tl = tleft()
    if tl > 200:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_cd, c5_cd = fast_cd(h1, n1, target1,
                               time_limit=min(180, tl * 0.7),
                               max_sweeps=80000,
                               n_cand_start=25, n_cand_end=70,
                               max_no_improve=400, max_pairs_check=250)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            best_n = n1
            target0 = target1

    # Phase 3: SLSQP on n=600
    tl = tleft()
    if tl > 80:
        h_pol, sc = slsqp_refine(best_h, best_n, target0,
                                  [5000, 20000, 80000],
                                  800, 1e-14,
                                  time_limit=min(70, tl * 0.6), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: CD on n=800
    tl = tleft()
    if tl > 250:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_cd2, c5_cd2 = fast_cd(h2, n2, target2,
                                 time_limit=min(220, tl * 0.7),
                                 max_sweeps=60000,
                                 n_cand_start=30, n_cand_end=80,
                                 max_no_improve=400, max_pairs_check=250)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
            best_n = n2
            target0 = target2

    # Phase 5: Multi-shift CD
    tl = tleft()
    if tl > 100:
        h_ms, c5_ms = multi_shift_cd(best_h, best_n, target0,
                                      time_limit=min(80, tl * 0.6),
                                      max_sweeps=15000, n_cand=50,
                                      n_shifts=4, max_pairs_check=200)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 6: SLSQP refinement
    tl = tleft()
    if tl > 80:
        h_pol, sc = slsqp_refine(best_h, best_n, target0,
                                  [20000, 100000, 300000],
                                  1000, 1e-14,
                                  time_limit=min(70, tl * 0.6), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: CD on n=1000
    tl = tleft()
    if tl > 250:
        n3 = 1000
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        h_cd3, c5_cd3 = fast_cd(h3, n3, target3,
                                 time_limit=min(220, tl * 0.7),
                                 max_sweeps=50000,
                                 n_cand_start=30, n_cand_end=75,
                                 max_no_improve=400, max_pairs_check=220)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
            best_n = n3
            target0 = target3

    # Phase 8: SA for escape
    tl = tleft()
    if tl > 80:
        h_sa, c5_sa = sa_cd(best_h, best_n, target0,
                             time_limit=min(60, tl * 0.6),
                             max_steps=20000,
                             t_start=0.03, t_end=0.0002,
                             n_cand=50, seed=777)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 9: Rapid CD after SA
    tl = tleft()
    if tl > 80:
        h_cd9, c5_cd9 = fast_cd(best_h, best_n, target0,
                                 time_limit=min(70, tl * 0.7),
                                 max_sweeps=30000,
                                 n_cand_start=50, n_cand_end=90,
                                 max_no_improve=250, max_pairs_check=300)
        if c5_cd9 < best_score:
            best_score = c5_cd9
            best_h = h_cd9

    # Phase 10: Final SLSQP
    tl = tleft()
    if tl > 60:
        h_pol, sc = slsqp_refine(best_h, best_n, target0,
                                  [100000, 500000],
                                  800, 1e-14,
                                  time_limit=min(50, tl * 0.7), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 11: Final CD polish
    tl = tleft()
    if tl > 30:
        h_cf, c5_cf = fast_cd(best_h, best_n, target0,
                               time_limit=min(25, tl * 0.8),
                               max_sweeps=10000,
                               n_cand_start=60, n_cand_end=100,
                               max_no_improve=150, max_pairs_check=200, n_bisect=16)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
