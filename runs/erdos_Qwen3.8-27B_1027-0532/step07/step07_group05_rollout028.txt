```python
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_gradient(h, alpha):
    """Compute gradient of log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    # Softmax weights
    shifted = alpha * corr * dx - np.max(alpha * corr * dx)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    # Gradient: sum_k w_k * grad_c_k
    # grad_c_k[p] = (1-h[p+k])*[valid] - h[p-k]*[valid]
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if weights[idx] < 1e-12:
            continue
        k = idx - mid
        w = weights[idx]
        # h[p] as first factor: p in valid range, p+k valid
        if k >= 0:
            # i from 0 to n-1-k, so p from 0 to n-1-k
            p_range = np.arange(0, n - k)
            if len(p_range) > 0:
                grad[p_range] += w * (1.0 - h[p_range + k])
            # h[p] as second factor: p = i+k, i from 0 to n-1-k, so p from k to n-1
            p_range2 = np.arange(k, n)
            if len(p_range2) > 0:
                grad[p_range2] -= w * h[p_range2 - k]
        else:
            k_abs = -k
            # i from k_abs to n-1, so p from k_abs to n-1
            p_range = np.arange(k_abs, n)
            if len(p_range) > 0:
                grad[p_range] += w * (1.0 - h[p_range + k])
            # h[p] as second factor: p = i+k = i-k_abs, i from k_abs to n-1, so p from 0 to n-1-k_abs
```

```python
def lp_direction(h0, n, target):
    """Solve LP linearization to get a descent direction."""
    from scipy.optimize import linprog
    corr0 = compute_corr(h0)
    dx = 2.0 / n
    n_corr = len(corr0)
    mid = n - 1
    
    # Compute gradients of all c_k at h0
    # A[k, p] = (∇c_k)[p]
    # G[k] = c_k(h0)
    # LP: min t s.t. G[k] + A[k,:] · (h - h0) ≤ t, 0≤h≤1, sum(h)=target
    
    # Simplify: G[k] - A[k,:] · h0 + A[k,:] · h ≤ t
    # Let b[k] = G[k] - A[k,:] · h0
    # A[k,:] · h - t ≤ -b[k]
    
    # Variables: h[0],...,h[n-1], t (total n+1 variables)
    # c = [0, ..., 0, 1] (minimize t)
    
    # Constraints:
    # A[k,:] · h - t ≤ -b[k] for each k (n_corr constraints)
    # sum(h) = target (equality)
    # 0 ≤ h[p] ≤ 1 (bounds)
    
    # Build A constraint matrix: shape (n_corr, n+1)
    A_eq_row = np.zeros(n + 1)
    A_eq_row[:n] = 1.0
    
    # Inequality constraints: A_ineq @ x ≤ b_ineq
    A_ineq = np.zeros((n_corr, n + 1))
    b_ineq = np.zeros(n_corr)
    
    for k in range(n_corr):
        # Compute ∇c_k at h0
        lag = k - mid
        if lag >= 0:
            pk = n - lag
            grad_k = np.zeros(n)
            grad_k[:pk] = (1.0 - h0[lag:lag+pk])
            grad_k[lag:] -= h0[:pk]
        else:
            ka = -lag
            pk = n - ka
            grad_k = np.zeros(n)
            grad_k[ka:] = (1.0 - h0[:pk])
            grad_k[:pk] -= h0[ka:]
        
        b_ineq[k] = -corr0[k] + np.dot(grad_k, h0)
        A_ineq[k, :n] = grad_k
        A_ineq[k, n] = -1.0  # -t
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, 1)] * n + [(None, None)]
    A_eq = A_eq_row.reshape(1, -1)
    b_eq = np.array([target])
    
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h_src, n_dst, target_dst):
    x_old = np.linspace(0.0, 1.0, len(h_src))
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_new = np.interp(x_new, x_old, h_src)
    return project_feasible(h_new, target_dst)

def smooth_obj_val(corr, dx, alpha):
    a = alpha * corr * dx
    return float(np.max(a)) / alpha + np.log(np.sum(np.exp(a - np.max(a)))) / alpha

def compute_smooth_grad(corr, h, alpha, dx):
    n = len(h)
    mid = n - 1
    n_corr = len(corr)
    a = alpha * corr * dx
    a_shift = a - np.max(a)
    ex = np.exp(a_shift)
    w = ex / np.sum(ex)
    
    grad = np.zeros(n)
    for m in range(n_corr):
        wm = w[m]
        if wm < 5e-4:
            continue
        k = m - mid
        if k >= 0:
            pk = n - k
            grad[:pk] += wm * (1.0 - h[k:k+pk])
            grad[k:] -= wm * h[:pk]
        else:
            ka = -k
            pk = n - ka
            grad[ka:] += wm * (1.0 - h[:pk])
            grad[:pk] -= wm * h[ka:]
    grad *= dx
    return grad

def grad_descent_phase(h0, n, target, alphas, iters_per_alpha, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    dx = 2.0 / n
    
    for alpha in alphas:
        for it in range(iters_per_alpha):
            if time.time() - start_time > time_limit:
                return best_h, best_score
            
            corr = compute_corr(h)
            obj = smooth_obj_val(corr, dx, alpha)
            grad = compute_smooth_grad(corr, h, alpha, dx)
            grad -= np.mean(grad)
            
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-14:
                break
            
            step = 1.0 / max(g_norm, 1e-10)
            improved = False
            for _ in range(30):
                h_new = project_feasible(h - step * grad, target)
                corr_new = compute_corr(h_new)
                obj_new = smooth_obj_val(corr_new, dx, alpha)
                if obj_new < obj - 1e-5 * step * g_norm:
                    improved = True
                    break
                step *= 0.5
            
            if improved:
                h = h_new
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
            else:
                break  # Can't improve at this step size
    
    return best_h, best_score
```

```python
def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

def cd_phase(h0, n, target, time_limit, start_time, n_active=3, n_cand=30, max_sweeps=20000):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    change = compute_change_matrix(h, n)
    max_no_improve = 500
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break
        
        # Find top active lags
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum
        
        # Compute gradient (direction of steepest ascent of max)
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Select candidate pairs
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        # Try pairs
        n_h = len(high_set)
        n_l = len(low_set)
        max_pairs = 300
        pair_list = []
        if n_h * n_l <= max_pairs:
            for ii in range(n_h):
                for jj in range(n_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pair_list.append((iv, jv))
        else:
            # Sample pairs
            for _ in range(max_pairs):
                ii = int(np.random.randint(n_h))
                jj = int(np.random.randint(n_l))
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))
        
        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue
            
            g = change[j] - change[i]
            
            # Bisection to find optimal delta
            a_d = 0.0
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
    
    return best_h, best_c5
```

```python
def lp_escape(h0, n, target, n_trials=5, seed=42):

    rng = np.random.RandomState(seed)
    best_h = h0.copy()

    for trial in range(n_trials):
        # Add small perturbation
        h_pert = h0 + rng.randn(n) * 0.01
        h_pert = project_feasible(h_pert, target)
        
        # Solve LP
        corr0 = compute_corr(h_pert)

        mid = n - 1

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        b_ub = np.zeros(n_corr)
        
        for k in range(n_corr):
            lag = k - mid
            grad_k = np.zeros(n)
            if lag >= 0:
                pk = n - lag
                grad_k[:pk] = (1.0 - h_pert[lag:lag+pk])
                grad_k[lag:] -= h_pert[:pk]
            else:
                ka = -lag
                pk = n - ka
                grad_k[ka:] = (1.0 - h_pert[:pk])

            b_ub[k] = -corr0[k] + np.dot(grad_k, h_pert)
            A_ub[k, :n] = grad_k
            A_ub[k, n] = -1.0
        
        bounds = [(0, 1)] * n + [(None, None)]
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
        except Exception:
            continue
        
        if not res.success:
            continue
        
        h_lp = res.x[:n]
        h_lp = project_feasible(h_lp, target)
        
        # Line search from h_pert to h_lp
        best_local = h_pert
        best_local_score = compute_c5(h_pert)
        for frac in [0.25, 0.5, 0.75, 1.0]:
            h_try = project_feasible(h_pert + frac * (h_lp - h_pert), target)
            sc = compute_c5(h_try)
            if sc < best_local_score:
                best_local_score = sc
                best_local = h_try
        
        if best_local_score < best_score:
            best_score = best_local_score
            best_h = best_local.copy()
            h0 = best_h.copy()  # Continue from improved point
    
    return best_h, best_score
```

            best_score = sc_new
            best_h = h_new
    
    # Phase 7: Final CD
    if tleft() > 60:
        h_new, sc_new = cd_phase(best_h, n0, target0, tleft() - 10, start_time,
                                  n_active=3, n_c
