def smoothed_fw(h_init, n, target, time_limit, alpha_start=10, alpha_end=10000, n_alphas=20):
    start = time.time()
    h = h_init.copy().astype(float)
    n_corr = 2*n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    # Log-spaced alpha schedule
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    
    for alpha in alphas:
        phase_start = time.time()
        # Each alpha phase gets equal time
        phase_time = time_limit / n_alphas
        
        for it in range(1000000):
            if time.time() - phase_start > phase_time:
                break
            if time.time() - start > time_limit:
                break
            
            # Compute correlations
            corr = np.correlate(h, 1.0-h, mode='full')
            
            # Compute weights
            max_corr = np.max(corr)
            weights = np.exp(alpha * (corr - max_corr))
            weights /= weights.sum()
            
            # Compute gradient
            grad = np.zeros(n)
            for s_idx in range(n_corr):
                if weights[s_idx] < 1e-12:
                    continue
                k = s_idx - mid
                if k >= 0:
                    # ∂g_k/∂h[j] = (1-h[j+k]) for j in [0, n-1-k] - h[j-k] for j in [k, n-1]
                    v1 = idxs[:n-k]  # j in [0, n-1-k]
                    v2 = idxs[k:]    # j in [k, n-1]
                    grad[v1] += weights[s_idx] * (1.0 - h[v1 + k])
                    grad[v2] -= weights[s_idx] * h[v2 - k]
                elif k < 0:
                    m = -k
                    v1 = idxs[m:]    # j in [m, n-1]
                    v2 = idxs[:n-m]  # j in [0, n-1-m]
                    grad[v1] += weights[s_idx] * (1.0 - h[v1 - m])
                    grad[v2] -= weights[s_idx] * h[v2 + m]
                else:
                    # k = 0: ∂g_0/∂h[j] = 1 - 2h[j]
                    grad += weights[s_idx] * (1.0 - 2.0 * h)
            
            # Frank-Wolfe: find v* = argmin <grad, v> s.t. 0≤v≤1, sum v = target
            # v* has 1 where grad is smallest
            sorted_idx = np.argsort(grad)
            v_star = np.zeros(n)
            v_star[sorted_idx[:int(target)]] = 1.0
            # Since target = n/2 (integer for even n), this is exact

            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = project_feasible(h_trial, target)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)

        diff = target_sum - np.sum(h)

            break

        else:

            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

def project_simplex_box(h, target):
    # Project h onto {v: 0<=v<=1, sum(v) = target}
    # using bisection on the threshold
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)

                n_alphas=30, n_iters_per_alpha=200000):

        phase_start = time.time()
        
        for it in range(n_iters_per_alpha):
            if time.time() - phase_start > phase_time:
                break
            if time.time() - start > time_limit:
                break
            
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = np.max(corr)

            exp_vals = np.exp(alpha * (corr - max_corr))
            w_sum = np.sum(exp_vals)
            if w_sum < 1e-300:
                weights = np.zeros(n_corr)
                weights[np.argmax(corr)] = 1.0
            else:
                weights = exp_vals / w_sum

            grad = np.zeros(n)
            for s_idx in range(n_corr):
                wi = weights[s_idx]
                if wi < 1e-14:
                    continue
                k = s_idx - mid
                if k > 0:

                    # g_k = sum_{j=0}^{n-1-k} h[j](1-h[j+k])
                    # d/dh[j] = (1-h[j+k]) for j in [0, n-1-k]
                    #           - h[j-k] for j in [k, n-1]
                    j1 = idxs[:n - k]
                    grad[j1] += wi * (1.0 - h[j1 + k])
                    j2 = idxs[k:]
                    grad[j2] -= wi * h[j2 - k]
                elif k < 0:
                    m = -k
                    # g_{-m} = sum_{j=m}^{n-1} h[j](1-h[j-m])
                    # d/dh[j] = (1-h[j-m]) for j in [m, n-1]
                    #           - h[j+m] for j in [0, n-1-m]
                    j1 = idxs[m:]
                    grad[j1] += wi * (1.0 - h[j1 - m])
                    j2 = idxs[:n - m]
                    grad[j2] -= wi * h[j2 + m]
                else:
                    # k = 0: g_0 = sum h[j](1-h[j]), d/dh[j] = 1 - 2h[j]
                    grad += wi * (1.0 - 2.0 * h)

            sorted_idx = np.argsort(grad)
            v_star = np.zeros(n)
            v_star[sorted_idx[:int(target)]] = 1.0
            
            # Step size
            gamma = 2.0 / (it + 2.0)
            h_trial = (1.0 - gamma) * h + gamma * v_star
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = project_feasible(h_trial, target)
            h = h_trial
    
    return h

import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def project_simplex_box(h, target):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):

    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def fw_method(h_init, n, target, time_limit, alpha_start=2.0, alpha_end=50000.0, n_alphas=20):

    n_corr = 2 * n - 1

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        it = 0
        phase_start = time.time()

        increasing_factor = 1.0 + 0.5 * (alphas[list(alphas).index(alpha)] / alpha_start - 1.0) / 10.0
        max_phase = base_time * min(increasing_factor, 3.0)
        while time.time() - phase_start < max_phase:

                break

            t = t_new
    return h

def multi_shift_cd(h_init, n, target, time_limit, max_sweeps=20000, n_cand=50, n_shifts=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts_arr = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 400:
            break
        top_shifts = np.argsort(corr)[-n_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0
        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts_arr)
            update_change_row(change, best_j, h, n, shifts_arr)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [30, 100, 300, 1000, 3000, 10000],
                                600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 300:
        h_fw = fw_method(best_h, current_n, current_target,
                         time_limit=min(250, tleft() * 0.5),
                         alpha_start=2.0, alpha_end=20000.0, n_alphas=18)
        sc_fw = compute_c5(h_fw)
        if sc_fw < best_score:
            best_score = sc_fw
            best_h = h_fw

    if tleft() > 250:
        h_cd, c5_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(200, tleft() * 0.55),
                              max_sweeps=35000,
                              n_cand_start=20, n_cand_end=80,
                              max_no_improve=500)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 200:
        h_apg = apg_method(best_h, current_n, current_target,
                           time_limit=min(150, tleft() * 0.5),
                           alpha_schedule=[20, 50, 100, 300, 800, 2000, 5000, 15000, 40000])
        sc_apg = compute_c5(h_apg)
        if sc_apg < best_score:
            best_score = sc_apg
            best_h = h_apg

    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [5000, 20000, 80000, 200000],
                                1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 350:
        n1 = 500
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 250:
            h_fw1 = fw_method(h1, n1, target1,
                              time_limit=min(120, tleft() * 0.35),
                              alpha_start=3.0, alpha_end=10000.0, n_alphas=12)
            sc_fw1 = compute_c5(h_fw1)
            if sc_fw1 < best_score:
                best_score = sc_fw1
                best_h = h_fw1
                current_n = n1
                current_target = target1
        if current_n == n1:
            if tleft() > 150:
                h_cd1, c5_cd1 = pair_cd(best_h, n1, target1,
                                        time_limit=min(120, tleft() * 0.4),
                                        max_sweeps=20000,
                                        n_cand_start=15, n_cand_end=55,
                                        max_no_improve=400)
                if c5_cd1 < best_score:
                    best_score = c5_cd1
                    best_h = h_cd1
            if tleft() > 80:
                h_pol1, sc1 = slsqp_round(best_h, n1, target1,
                                          [5000, 20000, 80000],
                                          800, 1e-15)
                if sc1 < best_score:
                    best_score = sc1
                    best_h = h_pol1

    if tleft() > 180:
        h_ms, c5_ms = multi_shift_cd(best_h, current_n, current_target,
                                     time_limit=min(130, tleft() * 0.5),
                                     max_sweeps=15000, n_cand=45, n_shifts=5)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if tleft() > 120:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.10]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [100, 500, 2000, 8000],
                                        400, 1e-15)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 80:
                h_cd_p, c5_p = pair_cd(h_ref, current_n, current_target,
                                       time_limit=min(50, tleft() * 0.3),
                                       max_sweeps=8000,
                                       n_cand_start=25, n_cand_end=65,
                                       max_no_improve=300)
                if c5_p < best_score:
                    best_score = c5_p
                    best_h = h_cd_p

    if tleft() > 100:
        h_final_cd, c5_final = pair_cd(best_h, current_n, current_target,
                                       time_limit=min(80, tleft() * 0.7),
                                       max_sweeps=12000,
                                       n_cand_start=30, n_cand_end=90,
                                       max_no_improve=400)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if tleft() > 60:
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                             [10000, 50000, 200000, 500000],
                                             1000, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
