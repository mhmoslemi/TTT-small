    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h

    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:

        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):

            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def fourier_perturb(h, n_modes, amplitude, rng):
    n = len(h)
    x = np.linspace(0.0, 2.0, n)
    perturbation = np.zeros(n)
    for k in range(1, n_modes + 1):
        a = rng.uniform(-amplitude, amplitude)
        b = rng.uniform(-amplitude, amplitude)
        perturbation += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return project_feasible(h + perturbation, n / 2.0)

        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new2
            target = target_new2

    # Phase 8: Pair CD at highest resolution
    if tleft() > 120:
        cd_time = min(110, tleft() * 0.2)
        hcd, c5cd = pair_cd_fast(best_h, n, target, time_limit=cd_time, n_cand=35, max_sweeps=200000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 9: Subgradient for fine-tuning
    if tleft() > 50:
        hsg, c5sg = subgradient_phase(best_h, n, target,
                                      time_limit=min(40, tleft() * 0.15),
                                      max_steps=100000, n_shifts=5, base_lr=0.06)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 10: Fourier perturbation restart
    if tleft() > 120:
        rng = np.random.RandomState(seed)
        n_use = len(best_h)
        target_use = n_use / 2.0
        for attempt in range(3):
            if tleft() < 50:
                break
            h_pert = fourier_perturb(best_h, n_modes=5, amplitude=0.03, rng=rng)
            hp, sc = lbfgs_phase(h_pert, n_use, target_use,
                                alphas=[1000, 10000, 50000],
                                maxiter=200, time_limit=min(25, tleft() * 0.15),
                                penalty=500.0)
            if sc < best_score and tleft() > 40:
                hcd, c5cd = pair_cd_fast(hp, n_use, target_use,
                                         time_limit=min(35, tleft() * 0.2),
                                         n_cand=30, max_sweeps=100000)
                if c5cd < best_score:
                    best_score = c5cd
                    best_h = hcd
            elif sc < best_score:
                best_score = sc
                best_h = hp

    # Phase 11: Final L-BFGS-B
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_phase(best_h, n_use, target_use,
                            alphas=[500000, 2000000],
                            maxiter=200, time_limit=min(30, tleft() * 0.5),
                            penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 12: Final pair CD polish
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd_fast(best_h, n_use, target_use,
                                 time_limit=min(20, tleft() * 0.7),
                                 n_cand=25, max_sweeps=50000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
