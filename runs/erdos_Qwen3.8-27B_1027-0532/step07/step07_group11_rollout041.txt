                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd_enhanced(h_init, n, target, time_limit, max_sweeps=60000,
                     n_cand_start=15, n_cand_end=80,
                     n_shifts_start=2, n_shifts_end=6,
                     max_no_improve=350, anneal_temp=0.03, anneal_decay=0.9995,
                     restart_interval=300, restart_scale=0.04):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp
    rng = np.random.RandomState(12345)

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_n_shifts = int(n_shifts_start + (n_shifts_end - n_shifts_start) * progress)
        current_n_shifts = max(1, min(current_n_shifts, n - 1))

        top_shifts = np.argsort(corr)[-current_n_shifts:]
        shift_weights = np.exp(corr[top_shifts] * 30.0)
        sw_sum = np.sum(shift_weights)
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        grad = np.zeros(n)
        density = np.zeros(n)
        for si in range(current_n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
            density[v1] += w
            density[v2] += w
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        score_high = grad * (1.0 + 2.0 * density)
        score_low = -grad * (1.0 + 2.0 * density)

        sorted_high = np.argsort(-score_high)
        sorted_low = np.argsort(-score_low)

        high_set = sorted_high[:current_ncand]
        low_set = sorted_low[:current_ncand]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 2000.0
                    if np.random.random() < np.exp(-delta_obj / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 3:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.93, 1e-10)
            if restart_interval > 0 and consecutive_no_improve % restart_interval == 0:
                h_pert = best_h + rng.randn(n) * restart_scale
                h_pert = project_feasible(h_pert, target)
                sc_pert = compute_c5(h_pert)
                if sc_pert < best_c5 * 1.01:
                    h = h_pert.copy()
                    change = compute_change_matrix(h, n)
                    changes_since_rebuild = 0
                    consecutive_no_improve = 0
            else:
                h = best_h.copy()
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0

    return best_h, best_c5


def subgradient_multi(h_init, n, target, time_limit, n_steps=40000, n_weights=4, step_scale=0.004):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        p = np.exp(corr * 40.0)
        p /= p.sum()

        grad = np.zeros(n)
        top_k = np.argsort(p)[-n_weights:]
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = float(step) / max(n_steps, 1)
        current_eta = eta * max(0.3, 1.0 - 0.7 * progress)

        h_try = h - current_eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - 0.5 * current_eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.92

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    # Phase 1: Initial SLSQP at n=400
    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [30, 100, 500, 2000],
                                400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Main CD at n=400 (most time here)
    if tleft() > 350:
        h_cd, c5_cd = pair_cd_enhanced(best_h, cur_n, cur_target,
                                       time_limit=min(300, tleft() * 0.55),
                                       max_sweeps=60000,
                                       n_cand_start=15, n_cand_end=80,
                                       n_shifts_start=2, n_shifts_end=6,
                                       max_no_improve=350,
                                       anneal_temp=0.03, anneal_decay=0.9995,
                                       restart_interval=250, restart_scale=0.04)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP refinement at n=400
    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [1000, 5000, 20000],
                                800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Additional CD at n=400 with different params
    if tleft() > 150:
        h_cd2, c5_cd2 = pair_cd_enhanced(best_h, cur_n, cur_target,
                                         time_limit=min(120, tleft() * 0.55),
                                         max_sweeps=40000,
                                         n_cand_start=30, n_cand_end=100,
                                         n_shifts_start=3, n_shifts_end=8,
                                         max_no_improve=300,
                                         anneal_temp=0.01, anneal_decay=0.999,
                                         restart_interval=0, restart_scale=0.0)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 5: Subgradient at n=400
    if tleft() > 80:
        h_sg, c5_sg = subgradient_multi(best_h, cur_n, cur_target,
                                        time_limit=min(60, tleft() * 0.5),
                                        n_steps=30000, n_weights=4,
                                        step_scale=0.005)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: SLSQP at n=400 (final)
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [5000, 20000, 50000],
                                1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: Interpolate to n=800
    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 200:
            h_pol1, sc_pol1 = slsqp_round(h1, n1, target1,
                                          [50, 200, 1000],
                                          400, 1e-14)
            if sc_pol1 < sc1:
                h1 = h_pol1
                sc1 = sc_pol1

        if sc1 < best_score or tleft() > 180:
            if tleft() > 180:
                h_cd1, c5_cd1 = pair_cd_enhanced(h1, n1, target1,
                                                 time_limit=min(150, tleft() * 0.55),
                                                 max_sweeps=30000,
                                                 n_cand_start=10, n_cand_end=40,
                                                 n_shifts_start=2, n_shifts_end=5,
                                                 max_no_improve=250,
                                                 anneal_temp=0.015, anneal_decay=0.999,
                                                 restart_interval=0, restart_scale=0.0)
                if c5_cd1 < best_score:
                    best_score = c5_cd1
                    best_h = h_cd1
                    cur_n = n1
                    cur_target = target1
                elif sc1 < best_score:
                    best_score = sc1
                    best_h = h1
                    cur_n = n1
                    cur_target = target1

    # Phase 8: SLSQP at current resolution (high alpha)
    if tleft() > 120:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [2000, 10000, 30000],
                                800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 9: Interpolate to n=1200
    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 150:
            h_pol2, sc_pol2 = slsqp_round(h2, n2, target2,
                                          [50, 200, 1000, 5000],
                                          500, 1e-14)
            if sc_pol2 < sc2:
                h2 = h_pol2
                sc2 = sc_pol2

        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            cur_n = n2
            cur_target = target2
        elif tleft() > 120:
            h_cd2, c5_cd2 = pair_cd_enhanced(h2, n2, target2,
                                             time_limit=min(100, tleft() * 0.5),
                                             max_sweeps=15000,
                                             n_cand_start=8, n_cand_end=30,
                                             n_shifts_start=2, n_shifts_end=4,
                                             max_no_improve=200,
                                             anneal_temp=0.0, anneal_decay=1.0,
                                             restart_interval=0, restart_scale=0.0)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                cur_n = n2
                cur_target = target2

    # Phase 10: Final SLSQP polish
    if tleft() > 80:
        h_final_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                      [5000, 20000, 80000],
                                      1200, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    # Phase 11: Stochastic restarts
    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.06, 0.1]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(cur_n) * noise_scale
            h_pert = project_feasible(h_pert, cur_target)
            h_ref, sc = slsqp_round(h_pert, cur_n, cur_target,
                                    [200, 1000, 5000],
                                    400, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 12: Final CD polish
    if tleft() > 50:
        h_final_cd, c5_final = pair_cd_enhanced(best_h, cur_n, cur_target,
                                                time_limit=min(35, tleft() * 0.7),
                                                max_sweeps=10000,
                                                n_cand_start=40, n_cand_end=80,
                                                n_shifts_start=4, n_shifts_end=8,
                                                max_no_improve=150,
                                                anneal_temp=0.0, anneal_decay=1.0,
                                                restart_interval=0, restart_scale=0.0)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Phase 13: Absolute final SLSQP
    if tleft() > 30:
        h_abs, sc = slsqp_round(best_h, cur_n, cur_target,
                                [10000, 50000],
                                600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_abs

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
