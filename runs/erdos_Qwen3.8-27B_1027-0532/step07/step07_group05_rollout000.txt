import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

def soft_max_val(corr, alpha):
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    return max_c + float(np.log(np.sum(np.exp(shifted))) / alpha)

def soft_max_weights(corr, alpha):
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    exp_s = np.exp(shifted)
    return exp_s / float(np.sum(exp_s))

def grad_from_weights(h, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    active = np.where(weights > 1e-9)[0]
    for ki in active:
        k = int(ki) - mid
        w = weights[ki]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad

def momentum_gd(h_init, n, target, max_iters, alphas, beta, base_step, time_limit):
    start = time.time()
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        frac = float(it) / max_iters
        alpha = alphas[0] + (alphas[1] - alphas[0]) * frac
        corr = compute_corr(h)
        smv = soft_max_val(corr, alpha)
        smv_dx = smv * (2.0 / n)
        if smv_dx < best_score - 1e-14:
            best_score = smv_dx
            best_h = h.copy()
        w = soft_max_weights(corr, alpha)
        g = grad_from_weights(h, w)
        g -= np.mean(g)
        step = base_step / (1.0 + 0.5 * frac)
        v = beta * v + step * g
        h_new = h - v
        h_new = project_feasible(h_new, target)
        if not np.all(np.isfinite(h_new)):
            break
        h = h_new
    return best_h, best_score

def enhanced_cd(h_init, n, target, time_limit, max_sweeps, n_active, n_cand_start, n_cand_end, max_no_improve, bisection_iters):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    mid = n - 1
    idxs = np.arange(n)
    rng = np.random.RandomState(12345)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)

            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue
            g = change[j] - change[i]
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)

                best_j = j

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    target_sum = target
    n_pts = n
    for alpha in alphas:
        g_n = n_pts
        g_target = target_sum
        try:
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - g_target, 'jac': lambda x: np.ones(g_n)}]
            res = minimize(lambda x: soft_max_val(compute_corr(x), alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur
