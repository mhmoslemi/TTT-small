import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def vectorized_cd(h_init, n, target, time_limit, max_sweeps=30000,
                  n_cand_start=20, n_cand_end=80, max_no_improve=300,
                  n_top_shifts=3, bisection_iters=22):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    k_vals = np.arange(-(n - 1), n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    h_pad = np.zeros(3 * n - 1)
    h_pad[n - 1:2 * n - 1] = h

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-n_top_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep + 1) / max(1, max_sweeps), 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        n_h = len(high_set)
        n_l = len(low_set)

        all_cands = np.concatenate([high_set, low_set]).astype(int)
        n_cands = len(all_cands)

        pk = all_cands[:, None] + k_vals[None, :] + (n - 1)
        mk = all_cands[:, None] - k_vals[None, :] + (n - 1)
        valid_pk = (pk >= 0) & (pk < 3 * n - 1)
        valid_mk = (mk >= 0) & (mk < 3 * n - 1)

        profiles = np.where(valid_pk, h_pad[np.clip(pk, 0, 3 * n - 2)], 0.0)
        profiles += np.where(valid_mk, h_pad[np.clip(mk, 0, 3 * n - 2)], 0.0)

        high_prof = profiles[:n_h, :]
        low_prof = profiles[n_h:, :]

        changes = high_prof[:, None, :] - low_prof[None, :, :]

        h_i_vals = h[high_set]
        h_j_vals = h[low_set]
        delta_lo = np.maximum(h_i_vals[:, None] - 1.0, -h_j_vals[None, :])
        delta_hi = np.minimum(h_i_vals[:, None], 1.0 - h_j_vals[None, :])
        valid = delta_hi - delta_lo > 1e-12

        if not valid.any():
            consecutive_no_improve += 1
            continue

        a_d = delta_lo.copy()
        b_d = delta_hi.copy()

        for _ in range(bisection_iters):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr[None, None, :] + mid_d[:, :, None] * changes
            f_mid = vals.max(axis=2)
            active = vals >= f_mid[:, :, None] - 1e-14
            f_prime = np.where(active, changes, -np.inf).max(axis=2)
            pos = f_prime > 1e-12
            b_d[pos] = mid_d[pos]
            a_d[~pos] = mid_d[~pos]

        d_star = 0.5 * (a_d + b_d)
        vals_f = corr[None, None, :] + d_star[:, :, None] * changes
        obj = vals_f.max(axis=2) * dx
        obj[~valid] = np.inf
        obj[np.abs(d_star) < 1e-13] = np.inf

        best_idx = np.unravel_index(np.argmin(obj), obj.shape)
        if obj[best_idx] >= c5_val - 1e-15:
            consecutive_no_improve += 1
            continue

        i = int(high_set[best_idx[0]])
        j = int(low_set[best_idx[1]])
        d_apply = d_star[best_idx]

        h[i] -= d_apply
        h[j] += d_apply
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        h_pad[n - 1:2 * n - 1] = h

    return best_h, best_c5

                                     [5000, 20000],
                                     500, 1e-14)
        if sc_bp < best_score:
            best_score = sc_bp
            best_h = h_bpol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
