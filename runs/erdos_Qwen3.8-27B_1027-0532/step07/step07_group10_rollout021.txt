import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def project_box_sum(z, target_sum):
    n = len(z)
    lo = z.min() - 1.5
    hi = z.max() + 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h_proj < 1.0 - 1e-10
        else:
            mask = h_proj > 1e-10
        if mask.any():
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def build_shift_masks(n):
    masks = []
    mid = n - 1
    idxs = np.arange(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        masks.append((k, v1, v2))
    return masks

def smooth_gradient(h, alpha, shift_masks):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        return c_max, np.zeros(n)
    weights /= w_sum
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-13:
            continue
        k, v1, v2 = shift_masks[k_idx]
        w = weights[k_idx]
        grad[v1] += w * (1.0 - h[idxs_v1(h, k, v1)])
        grad[v2] -= w * h[idxs_v2(h, k, v2)]
    return c_max, grad

def idxs_v1(h, k, v1):
    idxs = np.arange(len(h))
    return idxs[v1] - k

def idxs_v2(h, k, v2):
    idxs = np.arange(len(h))
    return idxs[v2] + k

def smooth_gradient_fast(h, alpha, masks):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        return c_max, np.zeros(n)
    weights /= w_sum
    grad = np.zeros(n)
    idxs = np.arange(n)
    for k_idx in range(len(corr)):

        k, v1, v2 = masks[k_idx]
        w = weights[k_idx]

            grad[v1] += w * (1.0 - h[idxs[v1] - k])
        if v2.any():

def nesterov_optimize(h_init, n, target, time_limit, alpha, max_steps, base_lr):
    start = time.time()
    masks = build_shift_masks(n)
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_k = 1.0
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 8000:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_next
        y = h + beta * (h - h_prev)

        c_max, grad = smooth_gradient_fast(y, alpha, masks)
        h_try = project_box_sum(y - lr * grad, target)

        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                lr *= 0.9
                if lr < 1e-9:
                    break

        h_prev = h
        h = h_try
        t_k = t_next

    return best_h, best_c5

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def coordinate_descent(h_init, n, target, time_limit, n_cand_start, n_cand_end, max_sweeps, max_no_improve):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-14:
                    best_trial_c5 = float(approx_max) * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    dx = 2.0 / n
    masks = build_shift_masks(n)
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

        def obj_grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
            weights = np.exp(shifted)
            weights /= np.sum(weights)
            g = np.zeros(n)
            idxs = np.arange(n)
            for k_idx in range(len(corr)):
                if weights[k_idx] < 1e-12:
                    continue
                k = k_idx - (n - 1)
                w = weights[k_idx]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                g[v1] += w * (1.0 - x[idxs[v1] - k])
                g[v2] -= w * x[idxs[v2] + k])
            return g

        def cons_fun(x):
            return np.sum(x) - target

        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h_cur, jac=obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_sharpen(h_src, n_dst, target_dst, sharpness=3.0):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    h_out = project_feasible(h_out, target_dst)
    if sharpness > 1.0:
        h_centered = h_out - 0.5
        h_pushed = np.tanh(sharpness * h_centered) / np.tanh(sharpness * 0.5) * 0.5 + 0.5
        h_pushed = np.clip(h_pushed, 0.0, 1.0)
        h_pushed = project_feasible(h_pushed, target_dst)
        return h_pushed
    return h_out

        h_ms, sc_ms = multi_shift_cd(best_h, best_n, best_tgt,
                                     time_limit=min(50, tleft() * 0.6),
                                     n_cand=40, n_active=5)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms

    if tleft() > 50:
        h_pol, sc_pol = slsqp_round(best_h, best_n, best_tgt,
                                    [500, 2000, 8000], 800, 1e-15)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    n1 = 800
    if tleft() > 300:
        tgt1 = n1 / 2.0
        h1 = interpolate_sharpen(best_h, n1, tgt1, sharpness=2.0)
        sc1 = compute_c5(h1)

        if tleft() > 250:
            h1_ns, sc1_ns = nesterov_optimize(h1, n1, tgt1,
                                              time_limit=min(60, tleft() * 0.5),
                                              alpha=200, max_steps=30000, base_lr=0.06)
            if sc1_ns < sc1:
                h1 = h1_ns
                sc1 = sc1_ns

        if tleft() > 150:
            h1_cd, sc1_cd = coordinate_descent(h1, n1, tgt1,
                                               time_limit=min(100, tleft() * 0.6),
                                               n_cand_start=25, n_cand_end=80,
                                               max_sweeps=25000, max_no_improve=400)
            if sc1_cd < best_score:
                best_score = sc1_cd
                best_h = h1_cd
                best_n = n1
                best_tgt = tgt1
                h1 = h1_cd

        if tleft() > 80:
            h1_ms, sc1_ms = multi_shift_cd(h1, n1, tgt1,
                                           time_limit=min(60, tleft() * 0.6),
                                           n_cand=50, n_active=6)
            if sc1_ms < best_score:
                best_score = sc1_ms
                best_h = h1_ms
                best_n = n1
                best_tgt = tgt1
                h1 = h1_ms

        if tleft() > 50:
            h1_pol, sc1_pol = slsqp_round(h1, n1, tgt1,
                                          [2000, 10000], 1000, 1e-15)
            if sc1_pol < best_score:
                best_score = sc1_pol
                best_h = h1_pol
                best_n = n1
                best_tgt = tgt1

    n2 = 1200
    if tleft() > 250:
        tgt2 = n2 / 2.0
        h2 = interpolate_sharpen(best_h, n2, tgt2, sharpness=2.5)
        sc2 = compute_c5(h2)

        if tleft() > 200:
            h2_ns, sc2_ns = nesterov_optimize(h2, n2, tgt2,
                                              time_limit=min(60, tleft() * 0.5),
                                              alpha=300, max_steps=20000, base_lr=0.04)
            if sc2_ns < sc2:
                h2 = h2_ns
                sc2 = sc2_ns

        if tleft() > 120:
            h2_cd, sc2_cd = coordinate_descent(h2, n2, tgt2,
                                               time_limit=min(100, tleft() * 0.6),
                                               n_cand_start=30, n_cand_end=90,
                                               max_sweeps=20000, max_no_improve=350)
            if sc2_cd < best_score:
                best_score = sc2_cd
                best_h = h2_cd
                best_n = n2
                best_tgt = tgt2
                h2 = h2_cd

        if tleft() > 60:
            h2_ms, sc2_ms = multi_shift_cd(h2, n2, tgt2,
                                           time_limit=min(50, tleft() * 0.5),
                                           n_cand=60, n_active=7)
            if sc2_ms < best_score:
                best_score = sc2_ms
                best_h = h2_ms
                best_n = n2
                best_tgt = tgt2

        if tleft() > 50:
            h2_pol, sc2_pol = slsqp_round(h2, n2, tgt2,
                                          [3000, 10000], 800, 1e-15)
            if sc2_pol < best_score:
                best_score = sc2_pol
                best_h = h2_pol
                best_n = n2
                best_tgt = tgt2

    if tleft() > 80:
        h_cd_fin, sc_cd_fin = coordinate_descent(best_h, best_n, best_tgt,
                                                 time_limit=min(60, tleft() * 0.6),
                                                 n_cand_start=40, n_cand_end=100,
                                                 max_sweeps=15000, max_no_improve=300)
        if sc_cd_fin < best_score:
            best_score = sc_cd_fin
            best_h = h_cd_fin

    if tleft() > 50:
        h_ns_fin, sc_ns_fin = nesterov_optimize(best_h, best_n, best_tgt,
                                                time_limit=min(35, tleft() * 0.6),
                                                alpha=5000, max_steps=10000, base_lr=0.01)
        if sc_ns_fin < best_score:
            best_score = sc_ns_fin
            best_h = h_ns_fin

    if tleft() > 40:
        h_pol_fin, sc_pol_fin = slsqp_round(best_h, best_n, best_tgt,
                                            [10000, 50000], 1500, 1e-15)
        if sc_pol_fin < best_score:
            best_score = sc_pol_fin
            best_h = h_pol_fin

    if tleft() > 40:
        rng = np.random.RandomState(seed)
        for ns in [0.03, 0.08, 0.15]:
            if tleft() < 25:
                break
            h_pert = best_h + rng.randn(best_n) * ns
            h_pert = project_feasible(h_pert, best_tgt)
            h_ref, sc_ref = nesterov_optimize(h_pert, best_n, best_tgt,
                                              time_limit=min(15, tleft() * 0.5),
                                              alpha=500, max_steps=5000, base_lr=0.05)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 25:
        h_cd_last, sc_cd_last = coordinate_descent(best_h, best_n, best_tgt,
                                                   time_limit=min(15, tleft() * 0.6),
                                                   n_cand_start=50, n_cand_end=100,
                                                   max_sweeps=5000, max_no_improve=150)
        if sc_cd_last < best_score:
            best_score = sc_cd_last
            best_h = h_cd_last

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, best_n
