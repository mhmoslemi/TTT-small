```python
def frank_wolfe(h_init, n, target, time_limit, max_iters=100000):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    mid = n - 1
    
    for t in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Compute correlation and find active set
        corr = compute_corr(h)
        c_max = np.max(corr)
        c5_val = c_max * (2.0/n)
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
        
        # Compute subgradient (average of active constraints)
        active = corr >= c_max - 1e-10
        n_active = active.sum()
        g = np.zeros(n)
        idxs = np.arange(n)
        for k_idx in np.where(active)[0]:
            k = k_idx - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            g[v1] += (1.0 - h[idxs[v1] + k])
            g[v2] -= h[idxs[v2] - k]
        g /= n_active
        
        # Linear minimization oracle
        sorted_idx = np.argsort(g)
        s = np.zeros(n)
        s[sorted_idx[:int(target)]] = 1.0
        # Handle fractional case
        frac = target - int(target)
        if frac > 1e-10:
            boundary_idx = sorted_idx[int(target)]
            s[boundary_idx] = frac
        
        # FW step size
        gamma = 2.0 / (t + 2)
        h = (1 - gamma) * h + gamma * s
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    
    return best_h, best_score
```

Actually, I want to try one more thing: **proximal gradient method with the log-sum-exp smoothing**.

The idea:
1. Start with a large smoothing parameter α (smooth objective)
2. Use accelerated proximal gradient (FISTA) to minimize the smoothed objective
3. Gradually decrease α (making the objective less smoothed, more like the true max)
4. At each α value, the FISTA method converges faster for smoother functions

The smoothed objective is:
f_α(h) = (1/α) log(∑_k exp(α c_k(h)))

The gradient is:
∇f_α(h) = (1/∑_k exp(α c_k)) ∑_k exp(α c_k) ∇c_k(h)

This is a weighted average of the gradients, with weights proportional to exp(α c_k). As α → ∞, this becomes the gradient of the max (i.e., the subgradient).

The proximal step (for the constraints 0 ≤ h ≤ 1, ∑h = n/2) is:
prox_L(z) = argmin_{s ∈ [0,1]^n, ∑s = n/2} L||s - z||²

This is the "Euclidean projection onto the probability simplex scaled by n/2" intersected with [0,1]. Since [0,1] contains the simplex (when n/2 < n), the projection is just the simplex projection.

The simplex projection can be computed in O(n log n) using the algorithm in Duchi et al. (2012).

Let me implement FISTA with the simplex projection:

```python
def simplex_project(z, target):
    """Project z onto {s: 0 <= s <= 1, sum(s) = target}."""
    # First, project onto {s: s >= 0, sum(s) = target} (simplex)
    # Then clip to [0, 1] and re-project
    n = len(z)
    s = np.sort(z)[::-1]
    cssv = np.cumsum(s)
    rho = np.argmax(s * np.arange(1, n+1) - (cssv - target) > 0)
    theta = (cssv[rho] - target) / (rho + 1.0)
    result = np.maximum(z - theta, 0.0)
    # Clip to [0, 1]
    result = np.clip(result, 0.0, 1.0)
    # Re-project to satisfy sum constraint
    return project_feasible(result, target)
```

```python
def fista_smooth(h_init, n, target, time_limit, alphas=[10, 50, 200, 1000, 5000, 20000], iters_per_alpha=5000):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    mid = n - 1
    idxs = np.arange(n)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        
        y = h.copy()  # FISTA momentum variable
        t_k = 1.0
        
        for it in range(iters_per_alpha):
            if time.time() - start > time_limit:
                break
            
            # Compute gradient of smoothed objective at y
            corr_y = compute_corr(y)
            logits = alpha * (corr_y - np.max(corr_y))  # shift for numerical stability
            p = np.exp(logits)
            p /= p.sum()
            
            # Gradient = weighted average of ∇c_k
            grad = np.zeros(n)
            for m in range(len(corr_y)):
                if p[m] < 1e-15:
                    continue
                k = m - mid
                v1 = (idxs + k >= 0) & (idxs + k < n)
                v2 = (idxs - k >= 0) & (idxs - k < n)
                grad[v1] += p[m] * (1.0 - y[idxs[v1] + k])
                grad[v2] -= p[m] * y[idxs[v2] - k]
            grad *= alpha  # gradient of f_alpha
            
            # Lipschitz constant estimate (upper bound)
            # ||∇c_k||_inf ≤ 2 (each entry is at most 2 in magnitude)
            # ||∇f_alpha||_inf ≤ alpha * 2 (weighted average doesn't increase max)
            L = 2.0 * alpha + 1.0
            step = 1.0 / L
            
            # Gradient step
            step_vec = y - step * grad
            step_vec = project_feasible(step_vec, target)
            
            # FISTA update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            h_new = step_vec + ((t_k - 1.0) / t_new) * (step_vec - h)
            h_new = project_feasible(h_new, target)
            t_k = t_new
            h = h_new
            
            # Track best
            sc = compute_c5(h)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h.copy()
        
        # After each alpha, update y for next round
        y = h.copy()
    
    return best_h, best_score
```

```python
def compute_smoothed_gradient(h, alpha):
    """Compute gradient of log-sum-exp smoothed objective efficiently."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    logits = alpha * (corr - np.max(corr))
    p = np.exp(logits)
    p /= p.sum()
    
    mid = n - 1
    # p has length 2n-1, index j corresponds to lag j - mid
    # We need: grad[m] = alpha * (1 - sum_k p[k] * h[m+k] - sum_k p[k] * h[m-k])
    # where the sums are over valid k
    
    # Using convolution:
    # sum_k p[k] * h[m+k] = (p_conv * h)[m] where p_conv is the convolution kernel
    
    # p is indexed by j = 0, ..., 2n-2, corresponding to lag j - mid
    # For the first sum: sum over valid k of p[k+mid] * h[m+k]
    # Let w[j] = p[j] for j = 0, ..., 2n-2 (these are the weights for lags j-mid)
    # sum_k w[k+mid] * h[m+k] for k = -(n-1), ..., n-1, m+k in [0,n-1]
    # = sum_{m'} w[m' - m + mid] * h[m'] ... this is getting confusing
    
    # Let me just use the direct O(n^2) method for now, it's fine for n=400
    idxs = np.arange(n)
    grad = np.ones(n) * alpha
    
    for j in range(2 * n - 1):
        if p[j] < 1e-15:
            continue
        k = j - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] -= alpha * p[j] * h[idxs[v1] + k]
        grad[v2] -= alpha * p[j] * h[idxs[v2] - k]
    
    return grad
```

```python
def compute_corr_fast(h, g):
    """Compute cross-correlation using FFT."""
    n = len(h)
    ng = len(g)
    size = n + ng - 1
    f_h = np.fft.rfft(h, size)
    f_g = np.fft.rfft(g, size)
    conv = np.fft.irfft(f_h * np.conj(f_g), size)
    return conv
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = project_feasible(h_start.copy(), target)
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def fista_smoothed(h_init, n, target, time_limit, alphas=None, iters_per_alpha=8000):
    start = time.time()
    if alphas is None:
        alphas = [20, 80, 300, 1200, 5000, 20000, 80000]
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    mid = n - 1
    idxs = np.arange(n)
    n_corr = 2 * n - 1

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        y = h.copy()
        t_k = 1.0
        L = 2.0 * alpha + 1.0
        step_size = 1.0 / L
        local_best = best_score

        for it in range(iters_per_alpha):
            if time.time() - start > time_limit:
                break
            corr_y = compute_corr(y)
            logits = alpha * (corr_y - np.max(corr_y))
            p = np.exp(logits)
            p_sum = p.sum()
            if p_sum < 1e-300:
                break
            p = p / p_sum

            n_significant = max(1, int(np.sum(p > 1e-12)))
            if n_significant < n_corr:
                thresh = np.sort(p)[::-1][min(n_significant, n_corr - 1)] * 0.1
                sig_mask = p > thresh
            else:
                sig_mask = p > 0

            grad = np.zeros(n)
            for j in range(n_corr):
                if not sig_mask[j]:
                    continue
                k = j - mid
                v1 = (idxs + k >= 0) & (idxs + k < n)
                v2 = (idxs - k >= 0) & (idxs - k < n)
                grad[v1] += p[j] * (1.0 - y[idxs[v1] + k])
                grad[v2] -= p[j] * y[idxs[v2] - k]
            grad *= alpha

            step_vec = y - step_size * grad
            step_vec = project_feasible(step_vec, target)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            h_new = step_vec + ((t_k - 1.0) / t_new) * (step_vec - h)
            h_new = project_feasible(h_new, target)
            t_k = t_new
            h = h_new

            if it % 200 == 0:
                sc = compute_c5(h)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                if sc < local_best - 1e-6:
                    local_best = sc
                    L = max(L * 0.8, 2.0 * alpha + 1.0)
                    step_size = 1.0 / L
                elif sc > local_best + 1e-4:
                    L = L * 1.5
                    step_size = 1.0 / L

        sc = compute_c5(h)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h.copy()
        y = h.copy()

    return best_h, best_score


def frank_wolfe_method(h_init, n, target, time_limit, max_iters=300000):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    mid = n - 1
    idxs = np.arange(n)
    int_target = int(target)
    frac_target = target - int_target

    no_improve = 0
    for t in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c_max = np.max(corr)
        c5_val = c_max * (2.0 / n)

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 10000:
                break

        tol_active = max(c_max * 5e-7, 1e-12)
        active = corr >= c_max - tol_active
        active_idx = np.where(active)[0]

        g = np.zeros(n)
        for k_idx in active_idx:
            k = int(k_idx) - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            g[v1] += (1.0 - h[idxs[v1] + k])
            g[v2] -= h[idxs[v2] - k]
        g /= len(active_idx)

        sorted_idx = np.argsort(g)
        s = np.zeros(n)
        s[sorted_idx[:int_target]] = 1.0
        if frac_target > 1e-10:
            s[sorted_idx[int_target]] = frac_target

        gamma = 2.0 / (t + 2.0)
        h_new = (1.0 - gamma) * h + gamma * s
        h_new = project_feasible(h_new, target)

        sc_new = compute_c5(h_new)
        if sc_new > c5_val + 1e-10:
            gamma = gamma * 0.5
            h_new = (1.0 - gamma) * h + gamma * s
            h_new = project_feasible(h_new, target)
        h = h_new

    if compute_c5(h) < best_score:
        best_score = compute_c5(h)
        best_h = h.copy()
    return best_h, best_score


def cutting_plane_method(h_init, n, target, time_limit, max_iters=8000, tol=1e-9, n_cuts_keep=200):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_f = []
    cuts_h = []

    no_improve = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 2000:
            cuts_g = cuts_g[-20:]
            cuts_f = cuts_f[-20:]
            cuts_h = cuts_h[-20:]
            no_improve = 0

        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])
        active_mask = corr >= corr[k_star] - 1e-8
        active_lags = np.where(active_mask)[0]

        n_active_add = min(len(active_lags), 3)
        for k_idx in active_lags[:n_active_add]:
            k = int(k_idx) - mid
            g = np.zeros(n)
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            g[v1] += (1.0 - h[idxs[v1] + k])
            g[v2] -= h[idxs[v2] - k]
            cuts_g.append(g.copy())
            cuts_f.append(float(corr[k_idx]))
            cuts_h.append(h.copy())

        if len(cuts_g) > n_cuts_keep:
            keep = n_cuts_keep // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break

    return best_h, best_score


def coord_descent_hybrid(h_init, n, target, time_limit, max_sweeps=20000,
                         n_cand=40, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pi = best_i
            pj = best_j
            change[pi, :] = 0.0
            change[pj, :] = 0.0
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    sc = compute_c5(h)
    if sc < best_c5:
        best_c5 = sc
        best_h = h.copy()
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                             max_sweeps=15000, n_cand=60,
                                             max_no_improve=200)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for ns in [0.03, 0.06, 0.01]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n0) * ns
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc_ref = cutting_plane_method(h_pert, n0, target0,
                                                 time_limit=min(30, tleft() * 0.4),
                                                 max_iters=3000, tol=1e-9, n_cuts_keep=100)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 40:
        h_fista2, sc_f2 = fista_smoothed(best_h, n0, target0,
                                         time_limit=min(30, tleft() * 0.5),
                                         alphas=[10000, 50000, 200000],
                                         iters_per_alpha=3000)
        if sc_f2 < best_score:
            best_score = sc_f2
            best_h = h_fista2

    if tleft() > 30:
        h_final, sc_final = slsqp_round(best_h, n0, target0,
                                        [20000, 80000], 600, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
```
