    h_padded = np.pad(h, radius, mode='edge')
    h_blur = np.convolve(h_padded, kernel, mode='valid')
    return h_blur[:n]

_g_n = 0
_g_alpha = 0.0
_g_target = 0.0

def _smooth_obj(x):
    corr = compute_corr(x)
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    n = len(x)
    dx = 2.0 / n
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha) * dx

def _smooth_obj_grad(x):
    corr = compute_corr(x)
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        weights = np.zeros(len(corr))
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= wsum
    n = len(x)
    mid = n - 1
    dx = 2.0 / n
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(len(corr)):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[idxs[v1]] += w * (1.0 - x[idxs[v1] + k])
        grad[idxs[v2]] -= w * x[idxs[v2] - k]
    return grad * dx

def lbfgs_optimize(h_start, n, target, alphas, maxiter_per_alpha=2000):
    global _g_n, _g_alpha, _g_target
    _g_n = n
    _g_target = target
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur,
                           jac=_smooth_obj_grad,
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=120, max_no_improve=400,
            n_binsearch=18, stochastic_frac=0.05, seed=123):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs + k_star >= 0) & (idxs + k_star < n)
        v2 = (idxs - k_star >= 0) & (idxs - k_star < n)
        grad[idxs[v1]] += (1.0 - h[idxs[v1] + k_star])
        grad[idxs[v2]] -= h[idxs[v2] - k_star]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        n_pairs = n_h * n_l
        n_random = int(stochastic_frac * n_pairs)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if n_random > 0:
            for _ in range(n_random):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def smooth_restart(h_best, n, target, time_limit, sigmas, seed=777):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    for sigma in sigmas:
        if time.time() - start > time_limit:
            break
        remaining = time_limit - (time.time() - start)
        if remaining < 60:
            break
        h_blur = gaussian_blur(h_best, sigma)
        h_pert = h_blur + rng.randn(n) * 0.01
        h_pert = project_feasible(h_pert, target)
        alloc = remaining / (len(sigmas) + 1)
        h_cd, c5_cd = pair_cd(h_pert, n, target, time_limit=min(alloc, 80),
                               max_sweeps=10000, n_cand_start=20, n_cand_end=80,
                               max_no_improve=200, n_binsearch=16,
                               stochastic_frac=0.05, seed=int(rng.randint(100000)))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    return best_h, best_c5

def perturb_restart(h_best, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    scales = [0.02, 0.05, 0.1, 0.15, 0.2]
    while time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        if remaining < 40:
            break
        scale = scales[int(rng.randint(len(scales)))]
        h_pert = h_best + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        alloc = min(remaining / 4.0, 100)
        h_cd, c5_cd = pair_cd(h_pert, n, target, time_limit=alloc,
                               max_sweeps=8000, n_cand_start=20, n_cand_end=70,
                               max_no_improve=150, n_binsearch=15,
                               stochastic_frac=0.08, seed=int(rng.randint(100000)))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        if time.time() - start > time_limit * 0.85:
            break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
