import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, alpha):
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.minimum(shifted, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(corr, h, n, alpha):
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.minimum(shifted, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        # Fallback to hard max
        idx = int(np.argmax(corr))
        weights = np.zeros(len(corr))
        weights[idx] = 1.0
    else:
        weights /= wsum
    n_corr = len(corr)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(n_corr):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad

                   n_iters_per_alpha=500, step_scale=0.5, seed=42):

    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    n_alphas = 20
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        t = 1
        h_curr = best_h.copy()
        h_prev = best_h.copy()
        
        for it in range(n_iters_per_alpha):
            if time.time() - start > time_limit:
                break
            y = h_curr + (t - 1) / (t + 2) * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)
            y = project_feasible(y, target)
            
            corr = compute_corr(y)
            grad = smooth_max_grad(corr, y, n, alpha)
            
            # Adaptive step size
            step = step_scale / alpha  # rough estimate
            
            h_new = y - step * grad
            h_new = project_feasible(h_new, target)
            
            # Check improvement
            f_curr = smooth_max(corr, alpha)
            corr_new = compute_corr(h_new)
            f_new = smooth_max(corr_new, alpha)
            
            if f_new <= f_curr:
                h_prev = h_curr
                h_curr = h_new
                t += 1
            else:

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=100, max_no_improve=400,
            n_binsearch=20, stochastic_frac=0.05, seed=123):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        n_pairs = n_h * n_l
        n_random = max(1, int(stochastic_frac * n_pairs))
        for _ in range(n_random):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-12:
                continue
            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(n_binsearch):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        elif consecutive_no_improve > 100 and consecutive_no_improve % 80 == 0:
            n_pert = max(5, n // 20)
            idxs_pert = rng.choice(n, size=n_pert, replace=False)
            h[idxs_pert] += rng.randn(n_pert) * 0.03
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
    return best_h, best_c5

def anneal_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                   n_cand=80, temp_start=0.01, temp_end=0.001,
                   n_binsearch=18, max_no_improve=500, seed=456):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        progress = min(float(sweep) / max_sweeps, 1.0)
        temp = temp_start * (temp_end / temp_start) ** progress
        corr_shifted = (corr - np.max(corr)) / (temp * dx)
        weights = np.exp(corr_shifted)
        wsum = np.sum(weights)
        if wsum < 1e-30:
            weights = np.zeros(len(corr))
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights /= wsum
        grad = np.zeros(n)
        for s in range(len(corr)):
            w = weights[s]
            if w < 1e-10:
                continue
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                accept_prob = 1.0
                if approx_max >= best_trial_c5:
                    diff = (approx_max - best_trial_c5) / (temp * dx)
                    accept_prob = np.exp(-diff)
                if rng.rand() < accept_prob and approx_max < best_trial_c5 + 0.001:
                    if approx_max < best_trial_c5:
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def fista_smooth(h_init, n, target, time_limit, alpha_schedule=None,
                 step_scale=0.5, max_iter_per_alpha=20000, seed=42):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    if alpha_schedule is None:
        alpha_schedule = np.geomspace(30, 200000, 15)
    for alpha in alpha_schedule:
        if time.time() - start > time_limit:
            break
        h_curr = best_h.copy()
        h_prev = best_h.copy()
        t = 1
        step = step_scale / np.sqrt(alpha)
        stale = 0
        for it in range(max_iter_per_alpha):
            if time.time() - start > time_limit:
                break
            momentum = (t - 1.0) / (t + 2.0)
            y = h_curr + momentum * (h_curr - h_prev)
            y = project_feasible(y, target)
            corr = np.correlate(y, 1.0 - y, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            weights = np.exp(shifted)
            wsum = np.sum(weights)
            if wsum < 1e-30:
                weights = np.zeros(len(corr))
                weights[int(np.argmax(corr))] = 1.0
            else:
                weights /= wsum
            idxs = np.arange(n)
            mid = n - 1
            grad = np.zeros(n)
            for s in range(len(corr)):
                w = weights[s]
                if w < 1e-12:
                    continue
                k = s - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - y[idxs[v1] - k])
                grad[v2] -= w * y[idxs[v2] + k]
            h_prev = h_curr
            h_curr = project_feasible(y - step * grad, target)
            t += 1
            score = float(np.max(np.correlate(h_curr, 1.0 - h_curr, mode='full'))) * dx
            if score < best_score - 1e-14:
                best_score = score
                best_h = h_curr.copy()
                stale = 0
            else:
                stale += 1
            if stale > 3000:
                break
    return best_h, best_score

def subgradient_anneal(h_init, n, target, time_limit, seed=789):
    start = time.time()
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    step = 0.01
    alpha = 500
    for it in range(200000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-30:
            weights = np.zeros(len(corr))
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights /= wsum
        grad = np.zeros(n)
        for s in range(len(corr)):
            w = weights[s]
            if w < 1e-10:
                continue
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        score_new = compute_c5(h_new)
        accept = score_new < c5_val
        if not accept:
            temp = max(0.001, 0.01 * (1 - it / 200000.0))
            if rng.rand() < np.exp(-(score_new - c5_val) / temp):
                accept = True
        if accept:
            h = h_new
        step = max(0.001, step * 0.9999)
        if it % 5000 == 4999:
            alpha = min(alpha * 1.5, 500000)
        if it % 10000 == 9999 and score_new >= c5_val:
            n_pert = max(5, n // 15)
            idxs_pert = rng.choice(n, size=n_pert, replace=False)
            h[idxs_pert] += rng.randn(n_pert) * 0.02
            h = project_feasible(h, target)
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def make_structured_init(n, target, pattern='bump', seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    if pattern == 'bump':
        center = 1.0 + rng.randn() * 0.2
        width = 0.3 + rng.rand() * 0.3
        h = np.exp(-((x - center) / width) ** 2)
    elif pattern == 'two_bump':
        c1 = 0.5 + rng.randn() * 0.1
        c2 = 1.5 + rng.randn() * 0.1
        w = 0.25 + rng.rand() * 0.15
        h = np.exp(-((x - c1) / w) ** 2) + np.exp(-((x - c2) / w) ** 2)
    elif pattern == 'plateau':
        w = 0.4 + rng.rand() * 0.3
        c = 0.8 + rng.rand() * 0.4
        h = np.exp(-((x - c) / w) ** 4)
    elif pattern == 'random':
        h = rng.rand(n) * 0.5 + 0.25
    else:
        h = np.ones(n) * 0.5
    return project_feasible(h, target)

def run(seed=42, budget_s=1300, **kwargs):

                                    step_scale=0.35, max_iter_per_alpha=8000, seed=seed + 40)
        if _remaining() > 100:
            h3, sc3 = pair_cd(h3, n3, target3, time_limit=min(70, _remaining() * 0.35),
                               max_sweeps=12000, n_cand_start=25, n_cand_end=80,
                               max_no_improve=250, n_binsearch=16,
                               stochastic_frac=0.05, seed=seed + 41)
        if _remaining() > 40:
            h3, sc3 = slsqp_optimize(h3, n3, target3, [5000, 20000], 1000, 1e-15)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3
            n = n3
            target = target3

    if _remaining() > 60:
        h_final_cd, sc_final_cd = pair_cd(best_h, len(best_h), len(best_h) / 2.0,
                                           time_limit=min(40, _remaining() * 0.5),
                                           max_sweeps=10000, n_cand_start=50, n_cand_end=140,
                                           max_no_improve=200, n_binsearch=20,
                                           stochastic_frac=0.05, seed=seed + 50)
        if sc_final_cd < best_score:
            best_score = sc_final_cd
            best_h = h_final_cd

    if _remaining() > 30:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_final, sc_final = slsqp_optimize(best_h, n_cur, t_cur,
                                            [50000, 200000], 2000, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
