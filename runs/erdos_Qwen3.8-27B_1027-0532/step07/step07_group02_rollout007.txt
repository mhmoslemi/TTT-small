**Better: Use the "slope" information to jump directly to the optimum.**

At δ, let A(δ) = argmax_s (c[s] + δ*g[s]). The function f is linear between breakpoints, with slope = max_{s∈A(δ)} g[s] (if we're at a non-breakpoint) or in the range [min_{s∈A} g[s], max_{s∈A} g[s]] (if at a breakpoint).

If f'(δ) < 0 (all active slopes are negative), f is decreasing, so we want to increase δ.
If f'(δ) > 0 (all active slopes are positive), f is increasing, so we want to decrease δ.
If f'(δ) straddles 0, we're at the optimum.

The key insight: if at δ, the active set A has all g[s] < 0, then f is decreasing at δ, and we should increase δ. The next breakpoint (where a new s becomes active) is at:
δ_next = min_{s ∉ A} (c[s] + δ*g[s] = c[s_A] + δ*g[s_A]) 
= min_{s ∉ A} (c[s] - c[s_A]) / (g[s_A] - g[s])

for the s_A ∈ A that gives the minimum δ_next.

This is a "linear programming in 1D" approach. It should find the exact optimum in O(n_corr) time (one pass to find the active set, one pass to find the next breakpoint).

```python
def find_optimal_delta(corr, g, delta_lo, delta_hi):
    """Find δ in [delta_lo, delta_hi] that minimizes max_s (corr[s] + δ*g[s])."""
    # Start at delta_lo
    delta = delta_lo
    for iteration in range(100):  # max iterations
        vals = corr + delta * g
        f_val = np.max(vals)
        active = vals >= f_val - 1e-12
        g_active = g[active]
        g_max = np.max(g_active)
        g_min = np.min(g_active)
        
        if g_min <= 0 <= g_max:
            # We're at the optimum
            break
        elif g_max < 0:
            # f is decreasing, increase delta
            # Find next breakpoint
            c_max = f_val
            # For each s not active, find where corr[s] + delta*g[s] = c_max
            # delta_new = (c_max - corr[s]) / g[s] (if g[s] < g_max, i.e., g[s] is more negative)
            # We want the smallest delta_new > delta
            inactive = ~active
            # delta_new = (c_max - corr[s]) / g[s] for g[s] < 0 (these will increase the perspective)
            # Actually: corr[s] + delta_new * g[s] = c_max => delta_new = (c_max - corr[s]) / g[s]
            # For this to be > delta, we need (c_max - corr[s]) / g[s] > delta
            # Since g[s] < 0 (for inactive s that could become active), this means:
            # c_max - corr[s] < delta * g[s] (dividing by negative g[s] reverses inequality)
            # c_max - corr[s] < delta * g[s] => corr[s] + delta * g[s] > c_max
            # But inactive means corr[s] + delta * g[s] < c_max, contradiction!
            # So actually, for g[s] < 0, corr[s] + delta_new * g[s] = c_max gives delta_new < delta
            # (since g[s] < 0, increasing delta decreases corr[s] + delta*g[s])
            # We want the ACTIVE set to change, i.e., some inactive s to become active
            # s becomes active when corr[s] + delta_new * g[s] >= c_max
            # For g[s] > g_max (more positive), as delta increases, corr[s]+delta*g[s] increases
            # So: delta_new = (c_max - corr[s]) / g[s] for g[s] > 0
            # And we need delta_new > delta, which means (c_max - corr[s])/g[s] > delta
            # => c_max - corr[s] > delta * g[s] (for g[s] > 0)
            # => c_max > corr[s] + delta*g[s] which is true since s is inactive
            # So delta_new = (c_max - corr[s]) / g[s] > delta for g[s] > 0
            # We want the MINIMUM such delta_new
            pos_mask = inactive & (g > 0)
            if pos_mask.any():
                delta_candidates = (c_max - corr[pos_mask]) / g[pos_mask]
                delta_next = np.min(delta_candidates)
                delta = min(max(delta_next, delta_lo), delta_hi)
            else:
                # No positive g among inactive, so f is linear with negative slope forever
                delta = delta_hi
                break
        elif g_min > 0:
            # f is increasing, decrease delta
            inactive = ~active
            neg_mask = inactive & (g < 0)
            if neg_mask.any():
                delta_candidates = (f_val - corr[neg_mask]) / g[neg_mask]
                delta_next = np.max(delta_candidates)
                delta = max(min(delta_next, delta_hi), delta_lo)
            else:
                delta = delta_lo
                break
    return delta
```

```python
    change = np.zeros((n, n_corr))

        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    neg_mask = (p - shifts >= 0) & (p - shifts < n)

    change[p, neg_mask] -= h[p - shifts[neg_mask]]
```

```python
def find_min_max_linear(corr, g, lo, hi):
    """Find δ ∈ [lo, hi] minimizing max_s (corr[s] + δ*g[s]).
    This is a 1D convex piecewise-linear optimization."""
    n_corr = len(corr)
    
    # Evaluate at lo
    delta = lo
    for _ in range(200):  # max iterations
        vals = corr + delta * g
        f = np.max(vals)
        active = vals >= f - 1e-12
        g_active = g[active]
        
        if np.min(g_active) <= 0 <= np.max(g_active):
            break  # optimal
        elif np.max(g_active) < 0:
            # Need to increase delta
            # Find the smallest δ' > δ where some inactive s becomes active
            inactive = ~active
            # s becomes active when corr[s] + δ'*g[s] >= f
            # δ' >= (f - corr[s]) / g[s] (if g[s] > 0)
            # or δ' <= (f - corr[s]) / g[s] (if g[s] < 0), but that's < delta since g[s]<0 and f-corr[s]>0
            pos_inactive = inactive & (g > 0)
            if pos_inactive.any():
                candidates = (f - corr[pos_inactive]) / g[pos_inactive]
                next_delta = np.min(candidates)
            else:
                next_delta = hi
            delta = min(max(next_delta, delta), hi)
        else:  # np.min(g_active) > 0
            # Need to decrease delta
            inactive = ~active
            neg_inactive = inactive & (g < 0)
            if neg_inactive.any():
                candidates = (f - corr[neg_inactive]) / g[neg_inactive]
                next_delta = np.max(candidates)
            else:
                next_delta = lo
            delta = max(min(next_delta, delta), lo)
    
    return delta
```

For the "top-K" gradient: instead of using the gradient of just the worst shift, use a weighted average of the top K worst shifts. This gives a better direction for the pair CD because it accounts for multiple near-active shifts.

```python
def compute_topk_gradient(h, k_star_indices, n):
    """Compute weighted gradient from top K active shifts."""
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in k_star_indices:
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad -= np.mean(grad)
    return grad
```
