def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)

        h = np.clip(h, 0.0, 1.0)

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)])

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue

                g = G[lr] - G[hr]

                a_d = delta_min
                b_d = delta_max
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def subgradient_cd(h_init, n, target, time_limit, max_iter=20000, lr_start=0.002):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_corr = np.max(corr)
        tol = max(1e-10, 5e-8 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)

        progress = min(it / max_iter, 1.0)
        lr = lr_start * (1.0 - 0.9 * progress) ** 2

        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val - 1e-15:
            h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    if tleft() > 30:
        h_pol = slsqp_round(best_h, n0, target0,
                            [200, 1000, 5000],
                            300, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n1 = 900
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)

    if tleft() > 350:
        h_cd, c5_cd = vectorized_pair_cd(h1, n1, target1,
                                         time_limit=min(300, tleft() * 0.55),
                                         max_sweeps=50000,
                                         n_cand_start=30, n_cand_end=65,
                                         max_no_improve=400,
                                         n_active_max=8,
                                         bisection_iters=12)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            best_n = n1
            best_target = target1

    if tleft() > 60:
        h_pol = slsqp_round(best_h, best_n, best_target,
                            [1000, 5000],
                            500, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        h_cd2, c5_cd2 = vectorized_pair_cd(best_h, best_n, best_target,
                                           time_limit=min(200, tleft() * 0.55),
                                           max_sweeps=40000,
                                           n_cand_start=40, n_cand_end=75,
                                           max_no_improve=350,
                                           n_active_max=10,
                                           bisection_iters=13)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 60:
        h_pol2 = slsqp_round(best_h, best_n, best_target,
                             [5000, 20000],
                            1000, 1e-14)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 120:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.02, 0.04]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            h_r, c5_r = vectorized_pair_cd(h_pert, best_n, best_target,
                                           time_limit=min(35, tleft() * 0.2),
                                           max_sweeps=8000,
                                           n_cand_start=35, n_cand_end=65,
                                           max_no_improve=200,
                                           n_active_max=8,
                                           bisection_iters=12)
            sc_r = compute_c5(h_r)
            if sc_r < best_score:
                h_r2 = slsqp_round(h_r, best_n, best_target,
                                   [1000, 5000],
                                   300, 1e-14)
                sc_r2 = compute_c5(h_r2)
                if sc_r2 < best_score:
                    best_score = sc_r2
                    best_h = h_r2
                elif sc_r < best_score:
                    best_score = sc_r
                    best_h = h_r

    if tleft() > 80:
        h_sg, c5_sg = subgradient_cd(best_h, best_n, best_target,
                                     time_limit=min(40, tleft() * 0.4),
                                     max_iter=15000, lr_start=0.0015)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 60:
        h_cd_final, c5_final = vectorized_pair_cd(best_h, best_n, best_target,
                                                  time_limit=min(50, tleft() * 0.5),
                                                  max_sweeps=25000,
                                                  n_cand_start=50, n_cand_end=85,
                                                  max_no_improve=300,
                                                  n_active_max=10,
                                                  bisection_iters=14)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 30:
        h_final = slsqp_round(best_h, best_n, best_target,
                              [10000, 50000],
                              500, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
