```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_obj(x, alpha, top_k=10):
    """Smooth max using log-sum-exp on top-k correlations."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Use top-k for better conditioning
    sorted_corr = np.sort(corr)[::-1]
    top = sorted_corr[:top_k]
    scaled = alpha * (top - top[0])
    return float(top[0] + np.log(np.sum(np.exp(scaled))) / alpha)

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def lbfgsb_smooth(h_start, n, target, alphas, top_k=10, maxiter=2000):
    """L-BFGS-B optimization with log-sum-exp smoothing."""
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        def obj(x):
            return logsumexp_obj(x, alpha, top_k)
        def obj_grad(x):
            # Compute gradient analytically
            one_minus_x = 1.0 - x
            corr = np.correlate(x, one_minus_x, mode='full')
            sorted_idx = np.argsort(corr)[::-1][:top_k]
            top_vals = corr[sorted_idx]
            scaled = alpha * (top_vals - top_vals[0])
            weights = np.exp(scaled) / np.sum(np.exp(scaled))
            # Gradient of each corr[k] w.r.t. x
            mid = n - 1
            grad = np.zeros(n)
            idxs = np.arange(n)
            for si in range(top_k):
                k = int(sorted_idx[si]) - mid
                w = weights[si]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
                grad[v2] -= w * x[idxs[v2] + k]
            return grad
        try:
            res = minimize(obj, h_cur, jac=obj_grad, method='L-BFGS-B',
                          bounds=bounds, constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
```

- Given p, minimize Σ_k p_k * corr[k] over h (this is a smooth non-convex problem, solvable with gradient descent)
- Update p_k ∝ exp(corr[k] / τ) (softmax update)

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x, alpha, top_k=5):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx_sorted = np.argsort(corr)[::-1]
    top = corr[idx_sorted[:top_k]]
    scaled = alpha * (top - top[0])
    return float(top[0] + np.log(np.sum(np.exp(scaled))) / alpha)


def smooth_obj_grad(x, alpha, top_k=5):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx_sorted = np.argsort(corr)[::-1][:top_k]
    top_vals = corr[idx_sorted]
    scaled = alpha * (top_vals - top_vals[0])
    exp_vals = np.exp(scaled)
    weights = exp_vals / exp_vals.sum()
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for si in range(top_k):
        k = int(idx_sorted[si]) - mid
        w = weights[si]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, top_k=5, maxiter=800):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, top_k),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=120, n_shifts=3,
            max_no_improve=500, anneal_temp=0.0, anneal_decay=0.999):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = np.sum(shift_weights)
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = approx_max - c5_val
                    if np.random.random() < np.exp(-delta_obj / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 2:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.95, 1e-12)

    return best_h, best_c5


def subgradient_topk(h_init, n, target, time_limit, n_steps=50000,
                     n_weights=5, decay_rate=0.9999, step_scale=0.004):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        top_k = np.argsort(corr)[-n_weights:]
        top_vals = corr[top_k]
        scaled = top_vals - top_vals[0]
        weights = np.exp(scaled * 10.0)
        weights /= weights.sum()

        grad = np.zeros(n)
        for si in range(n_weights):
            k = int(top_k[si]) - mid
            w = weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
            eta = min(eta * 1.01, step_scale * 2)
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.9
                if step % 1000 == 0 and step > 0:
                    eta = max(step_scale * 0.5, 1e-5)

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, block_sizes=(3, 5, 8), max_sweeps=5000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for bs in block_sizes:
        if time.time() - start > time_limit:
            break
        if no_improve > 800:
            break

        for sweep in range(max_sweeps):
            if time.time() - start > time_limit:
                break
            if no_improve > 300:
                break

            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_val = float(np.max(corr)) * dx
            if c5_val < best_c5 - 1e-13:
                best_c5 = c5_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid

            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
            grad -= np.mean(grad)

            n_blocks = n // bs
            if n_blocks < 4:
                break
            block_grads = np.zeros(n_blocks)
            for b in range(n_blocks):
                block_grads[b] = np.mean(grad[b * bs:(b + 1) * bs])

            sorted_blocks = np.argsort(block_grads)
            n_low = min(8, n_blocks // 3)
            n_high = min(8, n_blocks // 3)
            low_blocks = sorted_blocks[:n_low]
            high_blocks = sorted_blocks[-n_high:]

            best_trial = c5_val
            best_move = None

            for hb in high_blocks:
                for lb in low_blocks:
                    if hb == lb:
                        continue
                    hi_s = hb * bs
                    hi_e = (hb + 1) * bs
                    lo_s = lb * bs
                    lo_e = (lb + 1) * bs
                    max_move = min(np.sum(h[hi_s:hi_e]),
                                   np.sum(1.0 - h[lo_s:lo_e]))
                    if max_move < 1e-10:
                        continue
                    for frac in [0.3, 0.6, 1.0]:
                        delta = frac * max_move / bs
                        h_try = h.copy()
                        h_try[hi_s:hi_e] -= delta
                        h_try[lo_s:lo_e] += delta
                        h_try = np.clip(h_try, 0.0, 1.0)
                        h_try = project_feasible(h_try, target)
                        sc = compute_c5(h_try)
                        if sc < best_trial - 1e-14:
                            best_trial = sc
                            best_move = h_try.copy()

            if best_move is not None:
                h = best_move
            else:
                break

    return best_h, best_c5


def structured_perturb(h_init, n, target, time_limit, n_modes=8):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    x = np.linspace(0, 2, n)
    rng = np.random.RandomState(123)

    for m in range(1, n_modes + 1):
        if time.time() - start > time_limit:
            break
        for amp in [0.05, 0.1, 0.15]:
            for phase in [0.0, np.pi / 2, np.pi]:
                pert = amp * np.sin(2 * np.pi * m * x / 2.0 + phase)
                h_try = project_feasible(h_init + pert, target)
                sc = compute_c5(h_try)
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h_try.copy()
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def exponential_weights_descent(h_init, n, target, time_limit, n_steps=30000,
                                temp_param=1.0, step_scale=0.003):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        scaled = (corr - corr.max()) / temp_param
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(n_corr) / n_corr
        else:
            p /= p_sum

        top_k = np.argsort(p)[-min(8, n_corr):]
        grad = np.zeros(n)
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            if w < 1e-6:
                continue
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.92

        if step % 5000 == 0 and step > 0:
            temp_param *= 1.5

    return best_h, best_c5

    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
```
