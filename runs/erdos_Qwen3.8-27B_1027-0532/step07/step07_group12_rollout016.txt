import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

# LSE objective and gradient with adaptive alpha
def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0, 1000.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12,
                                       'maxls': 50})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-10:
                    break
            except Exception:
                pass

    return best_h, best_c5

# Crossover between two solutions
def crossover(h1, h2, target, n_points, alpha=0.5):
    h_new = (1 - alpha) * h1 + alpha * h2
    return project_feasible(h_new, target)

# Interpolation
def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:

        else:

        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0, 1000.0, 10000.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12, 'maxls': 50})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-10:
                    if pen_mult >= 100.0:
                        break
            except Exception:
                pass
    return best_h, best_c5

def pair_cd_fast(h_init, n, target, time_limit, n_cand=25, n_shifts=2):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    mid = n - 1
    idxs = np.arange(n)
    ks = np.arange(-(n - 1), n)
    for sweep in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = 25.0
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]
        sorted_grad = np.argsort(-grad)
        high_set = sorted_grad[:n_cand]
        low_set = sorted_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            idx1 = m - ks
            idx2 = m + ks
            val1 = np.zeros(n_corr)
            val2 = np.zeros(n_corr)
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            val1[valid1] = 1.0 - h[idx1[valid1]]
            if ks[valid2] != 0:
                val2[valid2] = -h[idx2[valid2]]
            profiles[ci] = val1 + val2
        cand_pos = {int(all_cands[p]): p for p in range(nc)}
        argmax_pos = int(np.argmax(corr))
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            hi_val = float(h[i])
            if hi_val < 1e-10:
                continue
            ci = cand_pos[i]
            prof_i = profiles[ci]
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-10:
                    continue
                cj = cand_pos[j]
                g = prof_i - profiles[cj]
                if g[argmax_pos] > 1e-10:
                    continue
                lo = 0.0
                hi_d = delta_max
                for _ in range(13):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active_m = vals >= (f_mid - 1e-14)
                    f_prime = float(np.max(g[active_m]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

                if i_pos == j_pos:

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def make_periodic_init(n, target, period, phase=0.0):
    t = np.arange(n)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * t / period + phase)
    return project_feasible(h, target)

def make_block_init(n, target, block_size):
    h = np.zeros(n)
    for i in range(0, n, 2 * block_size):
        h[i:i + block_size] = 1.0
    return project_feasible(h, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    def tleft():
        return max(0.0, budget_s - 35.0 - (time.time() - start_time))
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)
    solution_pool = [(best_h.copy(), best_score)]
    if tleft() > 80:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[50, 200, 1000],
                            maxiter=300, time_limit=min(40, tleft() * 0.3),
                            penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
        solution_pool.append((hp, sc))
    if tleft() > 150:
        cd_time = min(140, tleft() * 0.18)
        hcd, c5cd = pair_cd_fast(best_h, n, target, time_limit=cd_time, n_cand=25, n_shifts=2)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd
        solution_pool.append((hcd, c5cd))
    if tleft() > 80:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[5000, 20000, 100000],
                            maxiter=300, time_limit=min(60, tleft() * 0.25),
                            penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
        solution_pool.append((hp, sc))
    if tleft() > 120:
        cd_time = min(110, tleft() * 0.18)
        hcd, c5cd = pair_cd_multidelta(best_h, n, target, time_limit=cd_time, n_cand=30, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd
        solution_pool.append((hcd, c5cd))
    if tleft() > 130:
        diverse_starts = []
        for period in [11, 13, 17, 19, 23, 29, 31, 37, 41, 47, 53, 61, 67, 71, 79, 83]:
            diverse_starts.append(make_periodic_init(n, target, period))
        for bs in [15, 25, 35, 50, 75]:
            diverse_starts.append(make_block_init(n, target, bs))
        h_rand = rng.uniform(0, 1, n)
        diverse_starts.append(project_feasible(h_rand, target))
        n_try = min(8, len(diverse_starts))
        for si in range(n_try):
            if tleft() < 60:
                break
            cand = diverse_starts[si].copy()
            hp, sc = lbfgs_phase(cand, n, target,
                                alphas=[200, 2000, 20000],
                                maxiter=150, time_limit=min(15, tleft() * 0.15),
                                penalty=200.0)
            if sc < best_score:
                best_score = sc
                best_h = hp
                solution_pool.append((hp, sc))
    if tleft() > 80:
        solution_pool.sort(key=lambda x: x[1])
        for _ in range(3):
            if tleft() < 40 or len(solution_pool) < 2:
                break
            h1 = solution_pool[0][0]
            h2 = solution_pool[min(1, len(solution_pool) - 1)][0]
            for alpha in [0.3, 0.5, 0.7]:
                if tleft() < 30:
                    break
                h_cross = project_feasible((1.0 - alpha) * h1 + alpha * h2, target)
                hp, sc = lbfgs_phase(h_cross, n, target,
                                    alphas=[5000, 50000],
                                    maxiter=100, time_limit=min(12, tleft() * 0.1),
                                    penalty=500.0)
                if sc < best_score:
                    best_score = sc
                    best_h = hp
    if tleft() > 100:
        cd_time = min(100, tleft() * 0.2)
        hcd, c5cd = pair_cd_fast(best_h, n, target, time_limit=cd_time, n_cand=35, n_shifts=3)
        if c5cd < best_score:
            best_score = sc if False else c5cd
            best_h = hcd
        solution_pool.append((hcd, c5cd))
    if tleft() > 200:
        for n_new in [800, 1000]:
            if tleft() < 150:
                break
            target_new = n_new / 2.0
            h_interp = interpolate_to(best_h, n_new, target_new)
            hp, sc = lbfgs_phase(h_interp, n_new, target_new,
                                alphas=[100, 1000, 10000, 100000],
                                maxiter=200, time_limit=min(80, tleft() * 0.3),
                                penalty=100.0)
            if sc < best_score:
                best_score = sc
                best_h = hp
                n = n_new
                target = target_new
                solution_pool.append((hp, sc))
                if tleft() > 80:
                    hcd, c5cd = pair_cd_fast(best_h, n, target, time_limit=min(70, tleft() * 0.3),
                                             n_cand=25, n_shifts=2)
                    if c5cd < best_score:
                        best_score = c5cd
                        best_h = hcd
    if tleft() > 100:
        cd_time = min(90, tleft() * 0.25)
        hcd, c5cd = pair_cd_multidelta(best_h, n, target, time_limit=cd_time, n_cand=40, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_phase(best_h, n_use, target_use,
                            alphas=[500000, 2000000, 10000000],
                            maxiter=200, time_limit=min(50, tleft() * 0.4),
                            penalty=50000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd_fast(best_h, n_use, target_use,
                                  time_limit=min(30, tleft() * 0.6),
                                  n_cand=30, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_phase(best_h, n_use, target_use,
                            alphas=[10000000, 50000000],
                            maxiter=100, time_limit=min(15, tleft() * 0.6),
                            penalty=500000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
