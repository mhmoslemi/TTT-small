def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,

def subgradient_step(h, n, target, lr):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    tol = max(1e-10, 1e-6 * max_corr)
    active = np.where(corr >= max_corr - tol)[0]
    
    grad = np.zeros(n)
    for k_idx in active:
        k = int(k_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad /= len(active)
    grad -= np.mean(grad)
    
    h_new = h - lr * grad
    return project_feasible(h_new, target)
