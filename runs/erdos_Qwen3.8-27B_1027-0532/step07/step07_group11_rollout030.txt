def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        else:

        if not mask.any():
            break

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k in range(len(weights)):
        if weights[k] < 1e-10:
            continue
        shift = k - mid
        v1 = (idxs - shift >= 0) & (idxs - shift < n)
        v2 = (idxs + shift >= 0) & (idxs + shift < n)
        grad[v1] += weights[k] * (1.0 - x[idxs[v1] - shift])
        grad[v2] -= weights[k] * x[idxs[v2] + shift]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alpha_schedule, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alpha_schedule:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alpha_schedule, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alpha_schedule:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    return change, shifts


def update_change_row(change, p, h, shifts):
    n = len(h)
    change[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos] += (1.0 - h[p + shifts[pos]])
    change[p, neg] -= h[p - shifts[neg]]


def fast_pair_cd(h_init, n, target, time_limit,
                 n_cand=60, n_shifts=4, max_sweeps=200000,
                 max_no_improve=3000, anneal_temp=0.005,
                 anneal_decay=0.9998, top_k_eval=8):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    temp = anneal_temp

    change, shifts = build_change_matrix(h, n)
    changes_count = 0
    rebuild_thresh = n // 3

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        sw = corr[top_shifts]
        sw_sum = sw.sum()
        if sw_sum < 1e-15:
            sw_sum = 1.0
        sw = sw / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            k = int(top_shifts[si]) - mid
            w = sw[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        top_corr_vals = corr[np.argsort(corr)[-top_k_eval:]]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_d = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            hi_val = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(hi_val - 1.0, -h[j])
                d_hi = min(hi_val, 1.0 - h[j])
                if d_hi - d_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a = d_lo
                b = d_hi
                for _ in range(14):
                    m = 0.5 * (a + b)
                    if m < 1e-14:
                        a = m
                        continue
                    vals = corr + m * g
                    f_max = float(np.max(vals))
                    active = vals >= f_max - 1e-12
                    if active.any():
                        f_prime = float(np.max(g[active]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b = m
                    else:
                        a = m
                d_star = 0.5 * (a + b)
                if abs(d_star) < 1e-13:
                    continue

                approx_c5 = float(np.max(corr + d_star * g)) * dx

                accept = False
                if approx_c5 < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-12:
                    delta_obj = (approx_c5 - c5_val) * 500.0
                    if delta_obj < 50.0 and np.random.random() < np.exp(-delta_obj / temp):
                        accept = True

                if accept and approx_c5 < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_c5
                    best_i = i
                    best_j = j
                    best_d = d_star

        if best_i >= 0:
            h[best_i] -= best_d
            h[best_j] += best_d
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            if abs(diff) > 1e-12:
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
            update_change_row(change, best_i, h, shifts)
            update_change_row(change, best_j, h, shifts)
            changes_count += 2
            if changes_count > rebuild_thresh:
                change, shifts = build_change_matrix(h, n)
                changes_count = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.999, 1e-14)

    return best_h, best_c5


def multi_grad_descent(h_init, n, target, time_limit, n_steps=50000,
                       step_scale=0.003, n_weights=4):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale
    best_eta = eta

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        p = np.exp(corr * 80.0)
        p /= p.sum()
        top_k = np.argsort(p)[-n_weights:]

        grad = np.zeros(n)
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val - 1e-15:
            h = h_try
            best_eta = eta
            eta = min(eta * 1.02, 0.01)
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
                eta = max(eta * 0.7, 1e-6)
            else:
                eta = max(eta * 0.9, 1e-6)

        if step % 2000 == 0 and step > 0:
            eta = max(step_scale * 0.5, best_eta * 0.5)

    return best_h, best_c5


def block_cd_fast(h_init, n, target, time_limit, block_size=8,
                  max_sweeps=20000, n_train=6):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    n_blocks = n // block_size

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        block_grads = np.zeros(n_blocks)
        for b in range(n_blocks):
            block_grads[b] = np.mean(grad[b * block_size:(b + 1) * block_size])

        lo_blocks = np.argsort(block_grads)[:n_train]
        hi_blocks = np.argsort(block_grads)[-n_train:]

        best_trial = c5_val
        best_move = None

        for hb in hi_blocks:
            for lb in lo_blocks:
                if hb == lb:
                    continue
                hs = int(hb) * block_size
                he = hs + block_size
                ls = int(lb) * block_size
                le = ls + block_size
                max_mv = min(np.sum(h[hs:he]), np.sum(1.0 - h[ls:le]))
                if max_mv < 1e-10:
                    continue
                for frac in [0.3, 0.6, 1.0]:
                    delta = frac * max_mv / block_size
                    h_copy = h.copy()
                    h_copy[hs:he] -= delta
                    h_copy[ls:le] += delta
                    h_copy = project_feasible(h_copy, target)
                    sc = compute_c5(h_copy)
                    if sc < best_trial - 1e-14:
                        best_trial = sc
                        best_move = h_copy.copy()

        if best_move is not None:
            h = best_move

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def randomized_restarts(h_init, n, target, time_limit, n_restarts=8,
                        scales=None):
    start = time.time()
    if scales is None:
        scales = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.15, 0.2]
    rng = np.random.RandomState(12345)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    per_restart = time_limit / max(n_restarts, 1)

    for r in range(n_restarts):
        if time.time() - start > time_limit:
            break
        s = scales[r % len(scales)]
        h_pert = h_init + rng.randn(n) * s
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pert.copy()

    return best_h, best_c5

        h_sg, c5_sg = multi_grad_descent(best_h, cur_n, cur_target,
                                         time_limit=min(60, tleft() * 0.7),
                                         n_steps=30000, step_scale=0.004,
                                         n_weights=4)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 60:
        h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                   [5000, 15000], 800)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 500:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
        cur_n = n1
        cur_target = target1

        if tleft() > 400:
            h_pol1, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                        [100, 500, 2000], 500)
            if sc < best_score:
                best_score = sc
                best_h = h_pol1

        if tleft() > 350:
            h_cd1, c5_cd1 = fast_pair_cd(best_h, cur_n, cur_target,
                                         time_limit=min(330, tleft() * 0.9),
                                         n_cand=50, n_shifts=3,
                                         max_sweeps=200000, max_no_improve=3000,
                                         anneal_temp=0.002, anneal_decay=0.9999)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

    if tleft() > 100:
        h_cd2, c5_cd2 = fast_pair_cd(best_h, cur_n, cur_target,
                                     time_limit=min(80, tleft() * 0.8),
                                     n_cand=80, n_shifts=5,
                                     max_sweeps=150000, max_no_improve=2000,
                                     anneal_temp=0.001, anneal_decay=0.99995)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 150:
        h_restarts, sc_r = randomized_restarts(best_h, cur_n, cur_target,
                                               time_limit=min(20, tleft() * 0.3),
                                               n_restarts=10)
        if sc_r < best_score:
            best_score = sc_r
            best_h = h_restarts
        h_pol_r, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                     [500, 3000, 10000], 600)
        if sc < best_score:
            best_score = sc
            best_h = h_pol_r

    if tleft() > 80:
        h_blk, sc_blk = block_cd_fast(best_h, cur_n, cur_target,
                                      time_limit=min(60, tleft() * 0.7),
                                      block_size=4, max_sweeps=10000,
                                      n_train=8)
        if sc_blk < best_score:
            best_score = sc_blk
            best_h = h_blk

    if tleft() > 50:
        h_final_cd, c5_fin = fast_pair_cd(best_h, cur_n, cur_target,
                                          time_limit=min(35, tleft() * 0.8),
                                          n_cand=100, n_shifts=5,
                                          max_sweeps=100000, max_no_improve=1500,
                                          anneal_temp=0.0, anneal_decay=1.0)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_final_cd

    if tleft() > 30:
        h_pol_f, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                     [20000, 50000], 600)
        if sc < best_score:
            best_score = sc
            best_h = h_pol_f

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
