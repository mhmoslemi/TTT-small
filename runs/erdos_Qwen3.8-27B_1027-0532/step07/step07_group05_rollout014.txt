```python
import cvxpy as cp

def solve_sove_direction(h, c, g_matrix, active_lags, step_size):
    n = len(h)
    d = cp.Variable(n)
    
    # Objective: minimize max of active lags
    t = cp.Variable()
    constraints = [
        t >= c[k] + g_matrix[:, k] @ d for k in active_lags
    ]
    # Step size constraint
    constraints += [cp.norm(d, 2) <= step_size]
    # Box constraints
    constraints += [h + d >= 0, h + d <= 1]
    # Integral constraint
    constraints += [cp.sum(d) == 0]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.SCS)  # or ECOS
    
    return d.value
```

Now let me implement the key components:

**Fast CD with vectorized pair evaluation:**

The key insight is to vectorize the bisection over multiple candidate pairs. Instead of doing bisection for each pair separately (O(n_corr) per pair per bisection step), we can:
1. For each candidate pair (i, j), compute the "change vector" g = change[j] - change[i]
2. For a given delta, the new correlation is c + delta * g
3. We want to find the delta that minimizes max(c + delta * g)
4. This is equivalent to finding where the maximum of the lines {c[k] + delta * g[k]} changes

The minimum of max_k (c[k] + delta * g[k]) over delta can be found by:
- The function f(delta) = max_k (c[k] + delta * g[k]) is a convex piecewise-linear function
- Its minimum is where the subgradient contains 0, i.e., where there exist k1, k2 with g[k1] < 0 < g[k2] (or g[k] = 0) and c[k1] + delta*g[k1] = c[k2] + delta*g[k2]
- delta* = (c[k2] - c[k1]) / (g[k1] - g[k2]) for the "crossing" pair

This is much faster than bisection! We can compute delta* in O(n_corr) time by finding the crossing point.

Actually, for a single pair, the optimal delta is:
delta* = min( delta_max, argmin_delta max_k (c[k] + delta * g[k]) )

where delta_max is the maximum feasible delta (limited by box constraints).

The argmin can be found by:
- Consider only k where g[k] ≠ 0
- For g[k] > 0: the line c[k] + delta*g[k] is increasing, so it's only relevant for small delta

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def fast_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_active=5, n_cand_start=20, n_cand_end=60,
            max_no_improve=500, bisection_iters=14):

    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break
        
        # Find active lags
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Candidate selection
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Try multiple pairs
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)

        max_pairs = 300
        if n_h * n_l <= max_pairs:

                        if int(high_set[ii]) != int(low_set[jj])]
        else:

                ii = rng.randint(n_h)
                jj = rng.randint(n_l)

            # Bisection for optimal delta
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
    
    return best_h, best_c5
```

```python
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        if n_h * n_l <= max_pairs:
            pair_list = []
            for ii in range(n_h):
                for jj in range(n_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pair_list.append((iv, jv))
        else:
            pair_list = []
            for frac in np.linspace(0.0, 1.0, 12, endpoint=False):
                hi = int(frac * (n_h - 1))
                lo = int(frac * (n_l - 1))
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pair_list.append((iv, jv))
            rng = np.random.RandomState(sweep % 100000)
            for _ in range(max_pairs - len(pair_list)):
                ii = rng.randint(n_h)
                jj = rng.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))

        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g = change[j] - change[i]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5

def smooth_max_gradient(h, n, alpha, n_top=20):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_top:]

    exp_vals = np.exp(alpha * (corr[top_idx] - np.max(corr[top_idx])))
    weights = exp_vals / np.sum(exp_vals)

    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = int(top_idx[idx]) - mid
        w = weights[idx]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad -= np.mean(grad)
    return grad, float(np.max(corr)) * dx

def smooth_gradient_descent(h_init, n, target, time_limit, alphas=None, max_iters=20000):
    if alphas is None:
        alphas = [50, 200, 500, 2000, 10000]
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        step = 0.5 / alpha
        for it in range(max_iters):
            if time.time() - start > time_limit:
                break
            grad, c5_val = smooth_max_gradient(h, n, alpha)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break
            direction = -grad / grad_norm

            for attempt in range(20):
                h_trial = h + step * direction
                h_trial = project_feasible(h_trial, target)
                sc = compute_c5(h_trial)
                if sc < best_c5 - 1e-13:
                    h = h_trial
                    best_c5 = sc
                    best_h = h.copy()
                    no_improve = 0
                    break
                else:
                    step *= 0.7
            else:
                step = 0.5 / alpha
                no_improve += 1
                if no_improve > 100:
                    break
    return best_h, best_c5

def multi_cd(h_init, n, target, time_limit, max_sweeps=50000,
             n_active=5, n_cand=40, max_no_improve=300,
             bisection_iters=14, max_pairs=200, block_size=3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_move = None

        if sweep % 5 == 0:
            for i in high_set[:min(20, len(high_set))]:
                for j in low_set[:min(20, len(low_set))]:
                    i, j = int(i), int(j)
                    if i == j:
                        continue
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi < 1e-12:
                        continue
                    g = change[j] - change[i]
                    a_d, b_d = 0.0, delta_hi
                    for _ in range(bisection_iters):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active = vals >= f_mid - 1e-14
                        if active.any():
                            f_prime = np.max(g[active])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g) * dx
                    if approx_max < best_trial:
                        best_trial = approx_max
                        best_move = (i, j, delta_star, None)
        else:
            n_h = len(high_set)
            n_l = len(low_set)
            max_p = max_pairs
            if n_h * n_l <= max_p:
                pair_list = [(int(high_set[ii]), int(low_set[jj]))
                            for ii in range(n_h) for jj in range(n_l)
                            if int(high_set[ii]) != int(low_set[jj])]
            else:
                pair_list = []
                for frac in np.linspace(0.0, 1.0, 10, endpoint=False):
                    hi = int(frac * (n_h - 1))
                    lo = int(frac * (n_l - 1))
                    pair_list.append((int(high_set[hi]), int(low_set[lo])))
                for _ in range(max_p - len(pair_list)):
                    ii = np.random.randint(n_h)
                    jj = np.random.randint(n_l)
                    pair_list.append((int(high_set[ii]), int(low_set[jj])))

            for i, j in pair_list:
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d, b_d = 0.0, delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_move = (i, j, delta_star, None)

        if best_move is not None:
            i, j, delta_star, _ = best_move
            h[i] -= delta_star
            h[j] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n)
            update_change_row(change, j, h, n)

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
```
