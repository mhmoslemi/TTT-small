import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    Z = np.sum(exp_shifted)
    weights = exp_shifted / Z
    
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(len(corr)):
        w = weights[s]
        if w < 1e-10:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    # Add gradient of c_max (which is subgradient from the argmax direction)
    # Actually c_max is the max of corr, so its "gradient" contribution is
    # the gradient of the max component. But since we're using the smooth approx,
    # the gradient of the full expression is:
    # d/dx [c_max + log(sum(exp(alpha*(corr-c_max))))/alpha]
    # = grad(c_max) + (1/alpha) * sum_k weights_k * alpha * grad(corr_k) - (1/alpha)*sum_k weights_k * alpha * grad(c_max)
    # = grad(c_max) + sum_k weights_k * grad(corr_k) - grad(c_max) * sum_k weights_k
    # = grad(c_max) + sum_k weights_k * grad(corr_k) - grad(c_max)
    # = sum_k weights_k * grad(corr_k)
    # So the gradient is just the weighted sum of correlation gradients!
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def pair_cd_v3(h_init, n, target, time_limit, max_sweeps=50000,
               n_cand_start=20, n_cand_end=100, max_no_improve=500,
               n_active_shifts=5, reheat_interval=100, reheat_scale=0.05,
               use_momentum=False, momentum_beta=0.5):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0
    
    change = compute_change_matrix(h, n)
    
    # Momentum state
    h_momentum = np.zeros(n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        # Reheat when stuck
        if sweeps_since_improve > reheat_interval and sweeps_since_improve % 30 == 0:
            rng = np.random.RandomState(sweep * 13 + 7)
            n_pert = max(10, n // 20)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 1e-5:
                h = h_pert
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                continue
        
        # Get top active shifts with weights
        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            # Use temperature-based weighting (sharper at start, flatter later)
            progress = min(float(sweep) / max(1, max_sweeps), 1.0)
            temp = 1.0 + 3.0 * progress  # Gets flatter over time
            weights = (corr[top_k] / corr_top_sum) ** (1.0 / temp)
            weights = weights / weights.sum()
        
        # Compute gradient for these shifts
        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        if use_momentum:
            h_momentum = momentum_beta * h_momentum + (1 - momentum_beta) * grad
            grad_eff = h_momentum
        else:
            grad_eff = grad
        
        grad_eff -= np.mean(grad_eff)
        
        # Adaptive candidate count
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        current_ncand = max(current_ncand, 10)
        
        sorted_idx = np.argsort(grad_eff)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        k_star_idx = int(np.argmax(corr))
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                
                # Skip if this direction would increase the max shift too much
                g_star = g[k_star_idx]
                if g_star > 0.5 * np.max(np.abs(g)) and g_star > 0:
                    # This direction increases the active shift - skip
                    # But only if it's significantly worse than other components
                    if g_star > 0.1:
                        continue
                
                # Bisection for optimal step
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def try_structural_variants(h, n, target, n_variants=5):
    """Try some structural transformations that might help."""
    results = []
    results.append((h.copy(), compute_c5(h)))
    
    # Shift by 1
    h_shift = np.roll(h, 1)
    h_shift = project_feasible(h_shift, target)
    results.append((h_shift, compute_c5(h_shift)))
    
    # Flip
    h_flip = 1.0 - h
    h_flip = project_feasible(h_flip, target)
    results.append((h_flip, compute_c5(h_flip)))
    
    # Reverse
    h_rev = h[::-1].copy()
    h_rev = project_feasible(h_rev, target)
    results.append((h_rev, compute_c5(h_rev)))
    
    # Scale and reproject
    h_scaled = h * (target / h.sum())
    h_scaled = project_feasible(h_scaled, target)
    results.append((h_scaled, compute_c5(h_scaled)))
    
    return results

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def subgradient_cd_v2(h_init, n, target, time_limit, alpha0=0.4, alpha_min=0.002):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    stale = 0

    for step in range(300000):
        if time.time() - start > time_limit:
            break
        if alpha < alpha_min:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
            alpha = min(alpha * 1.05, alpha0)
        else:
            stale += 1
            alpha *= 0.97
            if stale > 600:
                break

        c_max = float(np.max(corr))
        temp = 0.08 * c_max + 0.005
        weights = np.exp((corr - c_max) / temp)
        w_sum = weights.sum()
        if w_sum < 1e-10:
            active = np.array([int(np.argmax(corr))])
            w_active = np.array([1.0])
        else:
            weights = weights / w_sum
            active = np.where(weights > 1e-4)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(corr))])
                w_active = np.array([1.0])
            else:
                w_active = weights[active]
                w_active = w_active / w_active.sum()

        grad = np.zeros(n)
        n_act = len(active)
        for idx_s in range(n_act):
            s = int(active[idx_s])
            w = w_active[idx_s]
            if w < 1e-5:
                continue
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        grad -= np.mean(grad)

        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: SLSQP polish at n=400
    if tleft() > 80:
        deadline = time.time() + min(55, tleft() * 0.3)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000, 10000, 50000],
                                600, 1e-14, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD at n=400
    if tleft() > 150:
        h_cd, c5_cd = pair_cd_v3(best_h, current_n, current_target,
                                 time_limit=min(80, tleft() * 0.35),
                                 max_sweeps=15000,
                                 n_cand_start=15, n_cand_end=70,
                                 max_no_improve=300,
                                 n_active_shifts=5,
                                 reheat_interval=60, reheat_scale=0.04,
                                 bisection_iters=12, filter_threshold=0.3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Interpolate to n=800 and optimize
    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            current_n = n1
            current_target = target1

        if tleft() > 250:
            h_cd1, c5_cd1 = pair_cd_v3(h1, n1, target1,
                                       time_limit=min(180, tleft() * 0.45),
                                       max_sweeps=30000,
                                       n_cand_start=15, n_cand_end=75,
                                       max_no_improve=400,
                                       n_active_shifts=5,
                                       reheat_interval=80, reheat_scale=0.04,
                                       bisection_iters=12, filter_threshold=0.3)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                current_n = n1
                current_target = target1

        if tleft() > 80:
            deadline = time.time() + min(55, tleft() * 0.35)
            h_pol1, sc1_pol = slsqp_round(best_h, current_n, current_target,
                                          [2000, 10000, 50000, 200000],
                                          800, 1e-14, deadline=deadline)
            if sc1_pol < best_score:
                best_score = sc1_pol
                best_h = h_pol1

    # Phase 4: More CD at n=800 with different parameters
    if tleft() > 150:
        h_cd2, c5_cd2 = pair_cd_v3(best_h, current_n, current_target,
                                   time_limit=min(100, tleft() * 0.35),
                                   max_sweeps=20000,
                                   n_cand_start=25, n_cand_end=90,
                                   max_no_improve=350,
                                   n_active_shifts=7,
                                   reheat_interval=60, reheat_scale=0.03,
                                   bisection_iters=14, filter_threshold=0.2)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 5: Subgradient at n=800
    if tleft() > 80:
        h_sg, c5_sg = subgradient_cd_v2(best_h, current_n, current_target,
                                        time_limit=min(50, tleft() * 0.3),
                                        alpha0=0.3, alpha_min=0.005)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: Interpolate to n=1600
    if tleft() > 300:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            current_n = n2
            current_target = target2

        if tleft() > 200:
            h_cd3, c5_cd3 = pair_cd_v3(h2, n2, target2,
                                       time_limit=min(150, tleft() * 0.4),
                                       max_sweeps=15000,
                                       n_cand_start=12, n_cand_end=60,
                                       max_no_improve=300,
                                       n_active_shifts=5,
                                       reheat_interval=80, reheat_scale=0.03,
                                       bisection_iters=12, filter_threshold=0.35)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
                current_n = n2
                current_target = target2

        if tleft() > 60:
            deadline = time.time() + min(40, tleft() * 0.35)
            h_pol2, sc2_pol = slsqp_round(best_h, current_n, current_target,
                                          [5000, 20000, 100000],
                                          600, 1e-14, deadline=deadline)
            if sc2_pol < best_score:
                best_score = sc2_pol
                best_h = h_pol2

    # Phase 7: Multi-start perturbation
    if tleft() > 120:
        for attempt in range(5):
            if tleft() < 40:
                break
            noise_scale = 0.02 + 0.03 * attempt
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = pair_cd_v3(h_pert, current_n, current_target,
                                       time_limit=min(18, tleft() * 0.2),
                                       max_sweeps=5000,
                                       n_cand_start=15, n_cand_end=50,
                                       max_no_improve=80,
                                       n_active_shifts=5,
                                       reheat_interval=40, reheat_scale=0.03,
                                       bisection_iters=10, filter_threshold=0.3)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 8: Final CD polish
    if tleft() > 50:
        h_cd_f, c5_f = pair_cd_v3(best_h, current_n, current_n / 2.0,
                                  time_limit=min(35, tleft() * 0.5),
                                  max_sweeps=10000,
                                  n_cand_start=20, n_cand_end=80,
                                  max_no_improve=200,
                                  n_active_shifts=5,
                                  reheat_interval=50, reheat_scale=0.02,
                                  bisection_iters=14, filter_threshold=0.25)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    # Phase 9: Final SLSQP
    if tleft() > 30:
        n_sl = min(current_n, 1200)
        h_sl = interpolate_to(best_h, n_sl, n_sl / 2.0) if current_n != n_sl else best_h.copy()
        deadline = time.time() + min(25, tleft() * 0.6)
        h_final, sc_final = slsqp_round(h_sl, n_sl, n_sl / 2.0,
                                        [10000, 50000, 200000],
                                        600, 1e-14, deadline=deadline)
        sc_final = compute_c5(h_final)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
