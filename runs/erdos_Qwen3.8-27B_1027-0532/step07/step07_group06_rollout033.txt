    v2 = (idxs + k >= 0) & (idxs + k < n)
    grad[v1] += (1.0 - h[idxs[v1] - k])
    grad[v2] -= h[idxs[v2] + k]
    return grad

    """Compute gradient using active set (shifts near the max)"""
    corr = compute_corr(h)

        active = corr >= (c_max - epsilon)
    else:

        active = np.zeros(len(corr), dtype=bool)
        active[top_idx] = True

    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.ones(active.sum()) / active.sum()
    else:

        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]

            h_new = h - step_size * grad
            h_new = project_feasible(h_new, target)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        h = h_new + ((t - 1.0) / t_new) * (h_new - h_prev)
        h = project_feasible(h, target)
        t = t_new
        
        c5_cur = compute_c5(h)
        if c5_cur < best_c5 - tol:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
            # Adapt L
            L = max(L * 0.95, n * dx * 0.5)
        else:
            no_improve += 1
            if no_improve > 2000:
                break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_lse_loss(x, alpha):
    corr = compute_corr(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def pnorm_loss(x, p):
    corr = compute_corr(x)
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_opt(h_start, n, target, alphas, maxiter, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    func = pnorm_loss if use_pnorm else smooth_lse_loss
    
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        
        grad -= np.mean(grad)
        
        # Adaptive candidate count
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)

        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:

                    else:
                        a_d = mid_d

                if abs(delta_star) < 1e-12:
                    continue

                if approx_max < best_trial_c5:

                    best_i = i
                    best_j = j
                    best_delta = delta_star

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(best_i)
            update_change_row(best_j)
    
    return best_h, best_c5

def gradient_descent_pnorm(h_init, n, target, time_limit, start_time,
                           p_schedule=(2, 4, 8, 16), max_iters_per_p=20000,
                           lr_init=0.01):
    """Projected gradient descent with p-norm surrogate"""
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for p in p_schedule:
        if time.time() - start_time > time_limit:
            break
        
        h_cur = h.copy()
        best_h_phase = h_cur.copy()
        best_c5_phase = compute_c5(h_cur)
        lr = lr_init / np.sqrt(n)
        no_improve = 0
        
        for it in range(max_iters_per_p):
            if time.time() - start_time > time_limit:
                break
            if no_improve > 5000:
                break
            
            corr = compute_corr(h_cur)
            c_max = np.max(corr)
            
            if c_max < 1e-15:
                break
            
            # Compute gradient of p-norm
            # E_p = (sum_k (max(corr_k,0))^p)^(1/p)
            # dE_p/dh_i = (1/E_p^(p-1)) * sum_k max(corr_k,0)^(p-1) * dcorr_k/dh_i
            positive_corr = np.maximum(corr, 0.0)
            pp = positive_corr ** (p - 1)
            pp_sum = pp.sum()
            if pp_sum < 1e-30:
                break
            
            weights = pp / pp_sum
            # Scale by E_p
            E_p = float(c_max * (np.sum((positive_corr / c_max) ** p)) ** (1.0 / p))
            scale = E_p ** (1 - p) * p  # = p / E_p^(p-1)
            
            # Compute weighted gradient
            mid = n - 1
            idxs = np.arange(n)
            grad = np.zeros(n)
            
            # Only process shifts with significant weight
            active = weights > 1e-10
            for s_idx in np.where(active)[0]:
                w = weights[s_idx] * scale
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
                grad[v2] -= w * h_cur[idxs[v2] + k]
            
            grad -= np.mean(grad)
            
            # Projected gradient step
            h_trial = h_cur - lr * grad
            h_trial = project_feasible(h_trial, target)
            
            c5_trial = compute_c5(h_trial)
            if c5_trial < compute_c5(h_cur) - 1e-15:
                h_cur = h_trial
                no_improve = 0
                if c5_trial < best_c5_phase - 1e-14:
                    best_c5_phase = c5_trial
                    best_h_phase = h_cur.copy()
                # Decay learning rate slowly
                lr *= 0.999
            else:
                no_improve += 1
                lr *= 0.99
        
        h = best_h_phase
        sc = compute_c5(h)
        if sc < best_c5:
            best_c5 = sc
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_lse_loss(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def pnorm_loss(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_opt(h_start, n, target, alphas, maxiter, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    func = pnorm_loss if use_pnorm else smooth_lse_loss
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def pnorm_gradient_descent(h_init, n, target, time_limit, start_time,
                           p_schedule=(2, 4, 8, 16, 32), max_iters_per_p=15000,
                           lr_base=0.05):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)

    for p in p_schedule:
        if time.time() - start_time > time_limit:

        h_cur = h.copy()
        best_phase = h_cur.copy()
        best_c5_phase = compute_c5(h_cur)
        lr = lr_base / np.sqrt(n)
        no_improve = 0

        for it in range(max_iters_per_p):
            if time.time() - start_time > time_limit:
                break
            if no_improve > 4000:
                break

            corr = compute_corr(h_cur)
            c_max = np.max(corr)
            if c_max < 1e-15:
                break

            positive_corr = np.maximum(corr, 0.0)
            if p == 2:
                weights = positive_corr
            else:
                weights = positive_corr ** (p - 1)
            wsum = weights.sum()
            if wsum < 1e-30:
                break
            weights = weights / wsum

            E_p = float(c_max * (np.sum((positive_corr / c_max) ** p)) ** (1.0 / p))
            scale = p / (E_p ** (p - 1)) if E_p > 1e-30 else 1.0

            active = weights > 1e-12
            grad = np.zeros(n)
            for s_idx in np.where(active)[0]:
                w = weights[s_idx] * scale
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
                grad[v2] -= w * h_cur[idxs[v2] + k]

            grad -= np.mean(grad)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            h_trial = h_cur - lr * grad
            h_trial = project_feasible(h_trial, target)
            c5_trial = compute_c5(h_trial)

                    best_c5_phase = c5_trial

                lr = min(lr * 1.001, lr_base)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                   use_adaptive_shifts=True)
        if sc5 < best_score:

            n0 = n1
            target0 = target1

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 80:

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 80:
        h8, sc8 = stochastic_refine(best_h, n0, target0,
                                    start_time + min(80, remaining - 20), start_time,
                                    max_steps=12000, base_scale=0.02)

            best_h = h8

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 60:
        h9, sc9 = slsqp_opt(best_h, n0, target0,

            best_score = sc9
            best_h = h9

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.06]:

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
