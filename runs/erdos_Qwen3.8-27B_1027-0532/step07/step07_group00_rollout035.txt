import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha, n, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_init, n, target, alphas, maxiter_per_alpha, time_limit):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(lambda x: _smooth_obj(x, alpha, n, target), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_init, n, target, alphas, maxiter, time_limit):

    _g_target = target
    start = time.time()
    bounds = [(0.0, 1.0)] * n

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_init, n, target, alphas, maxiter, time_limit):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def frank_wolfe(h_init, n, target, time_limit, n_iter=5000, beta=25.0, top_k=25):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for t in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 1500:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        c_max = float(np.max(corr))
        a_k = min(top_k, n_corr)
        active_idx = np.argsort(corr)[-a_k:]

        weights = np.exp(beta * (corr[active_idx] - c_max))
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.zeros(a_k)
            weights[np.argmax(corr[active_idx])] = 1.0
        else:
            weights /= w_sum

        grad = np.zeros(n)
        for ii in range(a_k):
            w = weights[ii]
            if w < 1e-15:
                continue
            k = int(active_idx[ii]) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        sorted_idx = np.argsort(grad)
        v_fw = np.zeros(n)
        v_fw[sorted_idx[:int(target)]] = 1.0
        frac = target - int(target)
        if frac > 1e-12:
            v_fw[sorted_idx[int(target)]] = frac

        best_gamma = 0.0
        best_trial = c5_val
        for gamma in [1.0, 0.5, 0.3, 0.15, 0.05]:
            h_try = (1.0 - gamma) * h + gamma * v_fw
            h_try = project_feasible(h_try, target)
            trial_c5 = compute_c5(h_try)
            if trial_c5 < best_trial:
                best_trial = trial_c5
                best_gamma = gamma

        if best_gamma > 0:
            h = (1.0 - best_gamma) * h + best_gamma * v_fw
            h = project_feasible(h, target)
        else:
            h[no_improve % n] += 0.001 * ((no_improve % 2) * 2 - 1)
            h = project_feasible(h, target)

    return best_h, best_c5


def pair_cd(h_init, n, target, time_limit, max_sweeps=20000,
            n_cand_start=15, n_cand_end=50, max_no_improve=300,
            weight_beta=5.0, use_anneal=False, T0=0.0003):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(99999)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    n_corr_len = 2 * n - 1
    change = np.zeros((n, n_corr_len))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        T = T0 * 0.01 ** progress if use_anneal else 0.0

        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        grad = np.zeros(n)
        top_k_shifts = min(25, n_corr_len)
        top_idx = np.argsort(weights)[-top_k_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break
                elif T > 1e-12 and approx_max - c5_val < T * 3:
                    if rng.random() < np.exp(-(approx_max - c5_val) / T):
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
                        break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5


def discrete_local_search(h_init, n, target, time_limit, max_sweeps=5000):
    dx = 2.0 / n
    n_corr_len = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = np.where(h_init > 0.5, 1.0, 0.0)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target:
        excess = len(ones) - int(target)
        to_flip = ones[np.random.RandomState(42).permutation(excess)]
        h[to_flip] = 0.0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    elif len(ones) < target:
        deficit = int(target) - len(ones)
        to_flip = zeros[np.random.RandomState(42).permutation(deficit)]
        h[to_flip] = 1.0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        ones_list = np.where(h == 1.0)[0]
        zeros_list = np.where(h == 0.0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break

        best_i = -1
        best_j = -1
        best_gain = 0.0
        for i in ones_list:
            for j in zeros_list:
                gain = grad[i] - grad[j]
                if gain > best_gain:
                    best_gain = gain
                    best_i = int(i)
                    best_j = int(j)

        if best_i >= 0 and best_gain > 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
        else:
            rng_s = np.random.RandomState(sweep)
            if len(ones_list) > 0 and len(zeros_list) > 0:
                i = int(ones_list[rng_s.randint(len(ones_list))])
                j = int(zeros_list[rng_s.randint(len(zeros_list))])
                h[i] = 0.0
                h[j] = 1.0

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
