def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

            break

        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for k_idx in active_shifts:
            k = k_idx - mid
            # f_k(h) = sum_i h[i]*(1-h[i+k]) for valid i
            if k == 0:
                # f_0 = sum h[i] - sum h[i]^2 = target - sum h[i]^2
                # f_0 ≤ t  =>  target - sum h[i]^2 ≤ t  =>  sum h[i]^2 ≥ target - t
                # This is a SOCP constraint: ||h||_2 ≥ target - t
                # Or equivalently: t ≥ target - ||h||^2

                # f_k ≤ t  =>  sum h[i] - h^T A_k h / 2 ≤ t

                # A_k is the matrix with (A_k)_{i,i+k} = (A_k)_{i+k,i} = 1 for i=0..n-1-k

                # But A_k is NOT PSD in general!
                # So this is a non-convex QP

            else:

Actually, I want to try one more idea: **using the FFT to compute the correlation more efficiently for the local search**.

For n=400, np.correlate takes about 50 microseconds (estimating). The change matrix approach avoids recomputing the correlation from scratch, but we still need to compute the max of the updated correlation. This is O(n) per evaluation.

The FFT approach would be: compute FFT of h and FFT of (1-h), multiply, inverse FFT. This gives the full correlation in O(n log n) time. But for n=400, the overhead of FFT setup might make it slower than the direct approach.

Let me also consider: what if I use a **different discretization**? Instead of uniform spacing, what if I use a non-uniform grid that clusters points near the "interesting" regions of h? This is unlikely to help for a step function optimization, so I'll skip it.

One more idea: **exploit the symmetry h(x) → h(2-x)**. The solution should be symmetric (or close to it). I can enforce this by optimizing only half the solution and mirroring. This halves the effective dimensionality.

But wait, the correlation is NOT symmetric under h(x) → h(2-x) in general. Let me check:
- If h' (x) = h(2-x), then corr'(k) = sum h'(x) * (1-h'(x+k)) dx = sum h(2-x) * (1-h(2-x-k)) dx
  Let u = 2-x: = sum h(u) * (1-h(u-k)) du (with appropriate limits)
  = sum h(u) * (1-h(u-k)) = corr(-k)

So the correlation spectrum is symmetric: corr'(k) = corr(-k). This means max_k corr'(k) = max_k corr(k). So the symmetry IS preserved!

This means I can enforce h[i] = h[n-1-i] (palindromic constraint) without loss of optimality. This halves the degree of freedom and might help the optimization converge faster.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_local_search(h_init, n, target, time_limit, max_sweeps=500000,
                      sa_temp=0.3, sa_decay=0.99995, n_cand=150,
                      n_shifts=5, seed=42, rebuild_every=50):
    start = time.time()
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    change = compute_change_matrix(h, n)
    changes = 0
    temp = sa_temp
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 30000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = shift_weights.sum()
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        targets = sorted_idx[:n_cand]
        sources = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_s = len(sources)
        for ii in range(n_s):
            i = int(sources[ii])
            delta = h[i]
            if delta < 1e-10:
                continue
            g = change[targets] - change[i]
            new_max = np.max(corr[None, :] + delta * g, axis=1)
            new_c5 = new_max * dx
            bidx = int(np.argmin(new_c5))
            if new_c5[bidx] < best_trial_c5 - 1e-14:
                best_trial_c5 = float(new_c5[bidx])
                best_i = i
                best_j = int(targets[bidx])
                best_delta = delta
            elif temp > 1e-10:
                threshold = best_trial_c5 + 0.002
                close_mask = new_c5 < threshold
                if close_mask.any():
                    close_idx = np.where(close_mask)[0]
                    ridx = close_idx[rng.integers(len(close_idx))]
                    do = (float(new_c5[ridx]) - c5_val) * 300.0
                    if do / temp < 50.0 and rng.random() < np.exp(-do / temp):
                        best_i = i
                        best_j = int(targets[ridx])
                        best_delta = delta
                        best_trial_c5 = float(new_c5[ridx])

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes += 2
            if changes > rebuild_every:
                change = compute_change_matrix(h, n)
                changes = 0
            temp *= sa_decay
        else:
            temp = max(temp * 0.995, 1e-12)

    return best_h, best_c5


def subgradient_multi(h_init, n, target, time_limit, n_steps=50000,
                      n_weights=5, step_scale=0.005):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        p = np.exp(corr * 80.0)
        p /= p.sum()
        top_k = np.argsort(p)[-n_weights:]

        grad = np.zeros(n)
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.9

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def binary_init(n, seed=123):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    # Try to arrange: use a low-discrepancy-like placement
    positions = np.sort(ones)
    # Re-arrange using a "spread out" pattern
    target_positions = (np.arange(n // 2) * 2.0 + 0.5).astype(int)
    for ii in range(n // 2):
        h[positions[ii]] = 0.0
        pos = min(target_positions[ii], n - 1)
        h[pos] = 1.0
    return h


def multi_start_local_search(h_init, n, target, time_limit, n_starts=3, seed=42):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    per_start = max(10, (time_limit - 5) // n_starts)

    for s in range(n_starts):
        if time.time() - start > time_limit - 10:
            break
        if s == 0:
            h_cur = h_init.copy()
        else:
            rng = np.random.RandomState(seed + s * 1000)
            scale = 0.03 + 0.03 * s
            h_cur = h_init + rng.randn(n) * scale
            h_cur = project_feasible(h_cur, target)
        tl = min(per_start, time_limit - (time.time() - start) - 5)
        if tl < 10:
            break
        h_ls, c5_ls = fast_local_search(h_cur, n, target, tl,
                                        max_sweeps=200000, sa_temp=0.2,
                                        sa_decay=0.99995, n_cand=150,
                                        n_shifts=5, seed=seed + s * 7)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls

    return best_h, best_c5
