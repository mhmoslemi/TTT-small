            continue
        k = k_idx - mid
        w = p[k_idx]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    grad_avg = np.mean(grad)
    return float(c_max + np.log(p_sum) / alpha), grad - np.mean(grad)


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol,
                                    'disp': False})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        change[p, mid:p + 1 + mid] += (1.0 - h[:p + 1][::-1])
        start2 = mid - p
        end2 = start2 + (n - p)
        change[p, start2:end2] -= h[p:]
    return change


def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    change[p, mid:p + 1 + mid] += (1.0 - h[:p + 1][::-1])
    start2 = mid - p
    end2 = start2 + (n - p)
    change[p, start2:end2] -= h[p:]


def cd_optimize(h_init, n, target, time_limit, max_sweeps=20000,
                n_active=4, n_cand_start=10, n_cand_max=60,
                max_no_improve=300, bisection_iters=14,
                top_pairs=100, random_frac=0.1):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    rng = np.random.RandomState(12345)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        progress = float(sweep) / max_sweeps
        n_cand = int(n_cand_start + (n_cand_max - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        top_k = min(n_active, n_corr)
        top_idx = np.argsort(corr)[-top_k:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(top_k):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_h = len(high_set)
        n_l = len(low_set)
        n_deterministic = max(10, min(top_pairs - int(top_pairs * random_frac), 50))

        pairs = []
        for frac_idx in range(n_deterministic):
            frac = frac_idx / max(n_deterministic - 1, 1)
            hi = int(frac * (n_h - 1))
            lo = int(frac * (n_l - 1))
            iv = int(high_set[hi])
            jv = int(low_set[lo])
            if iv != jv:
                pairs.append((iv, jv))

        n_random = int(top_pairs * random_frac)
        for _ in range(n_random):
            ii = rng.randint(n_h)
            jj = rng.randint(n_l)
            iv = int(high_set[ii])
            jv = int(low_set[jj])
            if iv != jv:
                pairs.append((iv, jv))

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i_val, j_val in pairs:
            if i_val == j_val:
                continue
            delta_hi = min(h[i_val], 1.0 - h[j_val])
            if delta_hi < 1e-13:
                continue

            g = change[j_val] - change[i_val]
            g_max_active = np.max(g[corr >= np.max(corr) - 1e-13])

            if g_max_active <= 1e-14:
                delta_star = delta_hi
            else:
                a_d = 0.0
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)

            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i_val
                best_j = j_val
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5


def accelerated_smooth_step(h, n, target, alpha, n_iters=200, time_limit=5.0):
    start = time.time()
    global _g_alpha
    _g_alpha = alpha
    h_y = h.copy()
    h_curr = h.copy()
    c5_curr = compute_c5(h_curr)
    best_h = h_curr.copy()
    best_c5 = c5_curr
    t_k = 1.0

    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h_curr)
        grad -= np.mean(grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h_y = project_feasible(h_curr + (t_k / t_new) * (h_y - h_curr), target)
        lr = 1.0 / (np.max(np.abs(grad)) + 1e-10)
        lr = min(lr * 0.5, 0.5 / alpha)
        h_next = project_feasible(h_y - lr * grad, target)
        if compute_c5(h_next) > c5_curr * 1.01 + 0.01:
            lr *= 0.1
            h_next = project_feasible(h_y - lr * grad, target)
        c5_next = compute_c5(h_next)
        h_curr = h_next
        t_k = t_new
        c5_curr = c5_next
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    n_curr = n0
    target_curr = target0

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, n_curr, target_curr,
                                [100, 500, 2000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 60:
        h_cd, sc = cd_optimize(best_h, n_curr, target_curr,
                                time_limit=min(50, tleft() * 0.35),
                                max_sweeps=10000, n_active=4,
                                n_cand_start=12, n_cand_max=40,
                                max_no_improve=250, bisection_iters=12,
                                top_pairs=80, random_frac=0.15)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 40:
        _g_alpha = 5000.0
        h_acc, sc = accelerated_smooth_step(best_h, n_curr, target_curr,
                                            alpha=5000.0, n_iters=100,
                                            time_limit=min(30, tleft() * 0.4))
        if sc < best_score:
            best_score = sc
            best_h = h_acc

    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        tl = min(180, tleft() * 0.35)
        h1, sc1 = slsqp_round(h1, n1, target1,
                               [500, 2000, 8000], 400, 1e-14)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            n_curr = n1
            target_curr = target1

    if tleft() > 150:
        h_cd1, sc_cd1 = cd_optimize(best_h, n_curr, target_curr,
                                     time_limit=min(120, tleft() * 0.35),
                                     max_sweeps=20000, n_active=5,
                                     n_cand_start=15, n_cand_max=55,
                                     max_no_improve=350, bisection_iters=13,
                                     top_pairs=100, random_frac=0.1)
        if sc_cd1 < best_score:
            best_score = sc_cd1
            best_h = h_cd1

    if tleft() > 80:
        _g_alpha = 10000.0
        h_acc1, sc = accelerated_smooth_step(best_h, n_curr, target_curr,
                                             alpha=10000.0, n_iters=150,
                                             time_limit=min(60, tleft() * 0.35))
        if sc < best_score:
            best_score = sc
            best_h = h_acc1

    if tleft() > 60:
        h_pol2, sc2 = slsqp_round(best_h, n_curr, target_curr,
                                   [10000, 50000], 600, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for scale in [0.02, 0.05, 0.1]:
            if tleft() < 70:
                break
            h_pert = best_h + rng.randn(n_curr) * scale
            h_pert = project_feasible(h_pert, target_curr)
            h_re, sc_re = cd_optimize(h_pert, n_curr, target_curr,
                                       time_limit=min(40, tleft() * 0.25),
                                       max_sweeps=6000, n_active=4,
                                       n_cand_start=15, n_cand_max=50,
                                       max_no_improve=200, bisection_iters=12,
                                       top_pairs=80, random_frac=0.1)
            if sc_re < best_score:
                best_score = sc_re
                best_h = h_re

    if tleft() > 80:
        h_cd2, sc_cd2 = cd_optimize(best_h, n_curr, target_curr,
                                     time_limit=min(60, tleft() * 0.4),
                                     max_sweeps=15000, n_active=6,
                                     n_cand_start=25, n_cand_max=65,
                                     max_no_improve=300, bisection_iters=13,
                                     top_pairs=120, random_frac=0.1)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    if tleft() > 40:
        h_pol3, sc3 = slsqp_round(best_h, n_curr, target_curr,
                                   [50000, 200000], 400, 1e-15)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    if tleft() > 30:
        h_cd3, sc_cd3 = cd_optimize(best_h, n_curr, target_curr,
                                     time_limit=min(25, tleft() * 0.5),
                                     max_sweeps=8000, n_active=5,
                                     n_cand_start=35, n_cand_max=70,
                                     max_no_improve=150, bisection_iters=12,
                                     top_pairs=100, random_frac=0.1)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
