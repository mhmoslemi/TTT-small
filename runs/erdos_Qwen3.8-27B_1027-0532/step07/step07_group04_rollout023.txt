def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def update_change_row(change, p, h, n, shifts):
    n_corr = 2 * n - 1
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        update_change_row(change, p, h, n, shifts)
    return change

                  n_cand_start=40, n_cand_end=80, active_tol=0.0):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    start = time.time()
    
    change = compute_change_matrix(h, n)
    
    total_iters = 0
    while time.time() - start < time_limit and no_improve < max_no_improve:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        T = float(np.max(corr))
        
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Determine active set
        if active_tol > 0:
            active_mask = corr >= T - active_tol
        else:
            # Use a relative tolerance
            active_mask = corr >= T - max(0.01 * dx * T, 0.1)
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])
        # Cap active set size
        if len(active_idx) > 20:
            top_order = np.argsort(corr[active_idx])[::-1][:20]
            active_idx = active_idx[top_order]

        grad = np.zeros(n)
        for s_idx in active_idx:

                # d/dh[p] = (1 - 2*h[p])
                grad += (1.0 - 2.0 * h)
            else:
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += (1.0 - h[idxs[v1] - k])
                grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        g_max = np.max(diff, axis=2)  # (n_l, n_h)

        case1 = g_max < -1e-10
        case2 = ~case1 & (g_min > 1e-10)
        
        approx[case1] = T + delta_hi[case1] * g_max[case1]
        approx[case2] = T + delta_lo[case2] * g_min[case2]

        best_flat_idx = int(np.argmin(approx_flat))
        best_l, best_h_idx = np.unravel_index(best_flat_idx, approx.shape)

            continue
        
        best_i = int(high_set[best_h_idx])
        best_j = int(low_set[best_l])

        g_full = change[best_j] - change[best_i]
        d_lo = max(h[best_i] - 1.0, -h[best_j])
        d_hi = min(h[best_i], 1.0 - h[best_j])
        
        if d_hi - d_lo < 1e-14:
            total_iters += 1
            continue
        
        a_d = d_lo
        b_d = d_hi
        for _ in range(20):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr + mid_d * g_full
            f_mid = np.max(vals)
            am = vals >= f_mid - 1e-14
            if am.any():
                f_prime = np.max(g_full[am])
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                b_d = mid_d
            else:
                a_d = mid_d
        
        delta_star = 0.5 * (a_d + b_d)
        
        if abs(delta_star) < 1e-12:
            total_iters += 1
            continue

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        update_change_row(change, p, h, n, shifts)
    return change

def smooth_obj(x, alpha, target, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14,
                 time_limit=None, start_time=None):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha, target, n), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

        c5_val = float(np.max(corr)) * dx
        T = float(np.max(corr))

        active_mask = corr >= T - max(0.5, 0.003 * T)
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        for s_idx in active_idx[:8]:
            k = int(s_idx) - mid
            if k == 0:
                grad += (1.0 - 2.0 * h)
            else:
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += (1.0 - h[idxs[v1] - k])
                grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        d_lo = max(h[i] - 1.0, -h[j])
        d_hi = min(h[i], 1.0 - h[j])
        if d_hi - d_lo < 1e-12:
            continue

        delta = (d_hi - d_lo) * rng.rand() + d_lo
        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new = np.max(corr + delta * g) * dx
        delta_c = approx_new - c5_val

        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temp > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temp * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

            if approx_new < best_c5 - 1e-14:
                best_c5 = approx_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    remaining = lambda: budget_s - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick CD on n=400
    if remaining() > 100:
        h1, s1, _ = vectorized_cd(best_h, n0, target0,
                                  time_limit=min(60, remaining() * 0.4),
                                  max_no_improve=400,
                                  n_cand_start=25, n_cand_end=60)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: SLSQP polish on n=400
    if remaining() > 60:
        h2, s2 = slsqp_refine(best_h, n0, target0,
                              [500, 2000, 5000],
                              500, 1e-14,
                              time_limit=min(50, remaining() * 0.4),
                              start_time=start_time)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Interpolate to n=500, intensive CD
    n1 = 500
    target1 = n1 / 2.0
    h_500 = interpolate_to(best_h, n1, target1)
    if remaining() > 200:
        h3, s3, _ = vectorized_cd(h_500, n1, target1,
                                   time_limit=min(120, remaining() * 0.5),
                                   max_no_improve=600,
                                   n_cand_start=30, n_cand_end=70)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Multi-shift CD on n=500
    if remaining() > 100:
        h4, s4 = multi_shift_cd(best_h, n1, target1,
                                 time_limit=min(70, remaining() * 0.4),
                                 max_no_improve=400, n_cand=50, n_shifts=5)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: SLSQP on n=500
    if remaining() > 60:
        h5, s5 = slsqp_refine(best_h, n1, target1,
                              [2000, 8000, 20000],
                              800, 1e-14,
                              time_limit=min(50, remaining() * 0.4),
                              start_time=start_time)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Interpolate to n=600, intensive CD
    n2 = 600
    target2 = n2 / 2.0
    h_600 = interpolate_to(best_h, n2, target2)
    if remaining() > 250:
        h6, s6, _ = vectorized_cd(h_600, n2, target2,
                                   time_limit=min(150, remaining() * 0.5),
                                   max_no_improve=800,
                                   n_cand_start=30, n_cand_end=80)
        if s6 < best_score:
            best_score = s6
            best_h = h6
            n1 = n2
            target1 = target2

    # Phase 7: SA escape on current grid
    if remaining() > 80:
        h7, s7 = sa_cd(best_h, n1, target1,
                       time_limit=min(60, remaining() * 0.4),
                       max_steps=30000, t_start=0.025, t_end=0.0002,
                       n_cand=50, seed=seed)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: CD after SA
    if remaining() > 80:
        h8, s8, _ = vectorized_cd(best_h, n1, target1,
                                   time_limit=min(60, remaining() * 0.5),
                                   max_no_improve=500,
                                   n_cand_start=40, n_cand_end=80)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    # Phase 9: Multi-shift on current grid
    if remaining() > 70:
        h9, s9 = multi_shift_cd(best_h, n1, target1,
                                 time_limit=min(50, remaining() * 0.5),
                                 max_no_improve=400, n_cand=50, n_shifts=4)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    # Phase 10: Try n=700 if time permits
    n3 = 700
    target3 = n3 / 2.0
    if remaining() > 300:
        h_700 = interpolate_to(best_h, n3, target3)
        h10, s10, _ = vectorized_cd(h_700, n3, target3,
                                     time_limit=min(180, remaining() * 0.5),
                                     max_no_improve=800,
                                     n_cand_start=30, n_cand_end=70)
        if s10 < best_score:
            best_score = s10
            best_h = h10
            n1 = n3
            target1 = target3

    # Phase 11: Final SLSQP
    if remaining() > 50:
        h11, s11 = slsqp_refine(best_h, n1, target1,
                                [5000, 20000, 50000],
                                1000, 1e-14,
                                time_limit=min(40, remaining() * 0.6),
                                start_time=start_time)
        if s11 < best_score:
            best_score = s11
            best_h = h11

    # Phase 12: Final rapid CD
    if remaining() > 30:
        h12, s12, _ = vectorized_cd(best_h, n1, target1,
                                     time_limit=min(25, remaining() * 0.7),
                                     max_no_improve=300,
                                     n_cand_start=50, n_cand_end=100)
        if s12 < best_score:
            best_score = s12
            best_h = h12

    # Phase 13: One more multi-shift
    if remaining() > 20:
        h13, s13 = multi_shift_cd(best_h, n1, target1,
                                   time_limit=min(15, remaining() * 0.7),
                                   max_no_improve=200, n_cand=40, n_shifts=4)
        if s13 < best_score:
            best_score = s13
            best_h = h13

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
