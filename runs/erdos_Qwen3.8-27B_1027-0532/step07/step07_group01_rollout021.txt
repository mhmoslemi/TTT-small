            h = h_new

Now let me think about the accelerated subgradient method more carefully. The key issue is: what's the right step size?

For a subgradient method minimizing $f(h)$ where $f$ is $L$-smooth:
- Step size: $\alpha_t = \frac{1}{L \cdot t}$ (diminishing)
- Or: $\alpha_t = \frac{R}{L \cdot t}$ where $R$ is the initial distance to the optimum

For the smooth max approximation with temperature $\alpha$, the Lipschitz constant is bounded by $\alpha \cdot \max |gradient of corr|$ which is about $\alpha \cdot 2n$ (roughly). So $L \approx 2\alpha n$.

With Nesterov acceleration, the convergence rate improves from $O(1/t)$ to $O(1/t^2)$.

Let me implement a practical version:

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def multi_shift_cd(h_init, n, target, time_limit, max_sweeps=20000, n_cand=40, n_shifts=4):
    start = time.time()
    dx = 2.0 / n
    shifts_arr = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 300:
            break

        top_shifts = np.argsort(corr)[-n_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts_arr)
            update_change_row(change, best_j, h, n, shifts_arr)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [15, 40, 100, 400, 1500, 6000, 20000],
                                1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 80:
        h_lb, sc_lb = lbfgsb_round(best_h, current_n, current_target,
                                   [50, 200, 1000, 5000, 20000, 80000],
                                   1500)
        if sc_lb < best_score:
            best_score = sc_lb
            best_h = h_lb

    if tleft() > 100:
        h_cg, sc_cg = accelerated_subgrad(best_h, current_n, current_target,
                                          time_limit=min(80, tleft() * 0.5),
                                          alpha_start=20.0, alpha_end=80000.0,
                                          max_steps=50000)
        if sc_cg < best_score:
            best_score = sc_cg
            best_h = h_cg

    if tleft() > 120:
        h_cd, c5_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(120, tleft() * 0.5),
                              max_sweeps=50000,
                              n_cand_start=15, n_cand_end=70,
                              max_no_improve=600,
                              restart_every=200, restart_scale=0.012)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [3000, 10000, 40000, 100000],
                                1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 60:
        h_ms, c5_ms = multi_shift_cd(best_h, current_n, current_target,
                                     time_limit=min(50, tleft() * 0.5),
                                     max_sweeps=15000, n_cand=35, n_shifts=4)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if tleft() > 350:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 50:
            h_pol1, sc1 = slsqp_round(h1, n1, target1,
                                      [30, 100, 500, 2000, 8000, 30000],
                                      800, 1e-15)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1
                current_n = n1
                current_target = target1
        if current_n == n1 and tleft() > 60:
            h_lb1, sc_lb1 = lbfgsb_round(best_h, n1, target1,
                                         [100, 500, 2000, 10000, 50000],
                                         1200)
            if sc_lb1 < best_score:
                best_score = sc_lb1
                best_h = h_lb1
        if current_n == n1 and tleft() > 100:
            h_cd1, c5_cd1 = pair_cd(best_h, n1, target1,
                                    time_limit=min(100, tleft() * 0.4),
                                    max_sweeps=30000,
                                    n_cand_start=15, n_cand_end=65,
                                    max_no_improve=500,
                                    restart_every=180, restart_scale=0.01)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
        if current_n == n1 and tleft() > 50:
            h_ag1, sc_ag1 = accelerated_subgrad(best_h, n1, target1,
                                                time_limit=min(40, tleft() * 0.4),
                                                alpha_start=20.0, alpha_end=50000.0,
                                                max_steps=30000)
            if sc_ag1 < best_score:
                best_score = sc_ag1
                best_h = h_ag1
        if current_n == n1 and tleft() > 40:
            h_pol1b, sc1b = slsqp_round(best_h, n1, target1,
                                        [10000, 50000, 150000],
                                        1200, 1e-15)
            if sc1b < best_score:

                best_h = h_pol1b

    if tleft() > 350:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if tleft() > 50:
            h_pol2, sc2a = slsqp_round(h2, n2, target2,
                                       [20, 80, 400, 2000, 8000, 30000],

            if sc_lb2 < best_score:

        if current_n == n2 and tleft() > 100:
            h_cd2, c5_cd2 = pair_cd(best_h, n2, target2,
                                    time_limit=min(100, tleft() * 0.4),

    if tleft() > 150:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.008, 0.02, 0.05, 0.1]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,

    if tleft() > 60:

                                       max_no_improve=400)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if tleft() > 30:
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                            [15000, 60000, 200000],
                                            1000, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
