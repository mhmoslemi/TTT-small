import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _time_left(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _obj_func(x):
    return smooth_obj(x, _g_alpha)


def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new

    return h_cur


def build_change_matrix(h, n):
    n_corr = 2 * n - 1

    change = np.zeros((n, n_corr))

        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])

    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(11):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            if trust < 0.01:
                break
            trust = min(trust * 2.0, 0.3)
            no_improve = 0
            continue

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        threshold = c5_val - max(0.008 * dx, 0.003)
        active_candidates = np.where(corr * dx >= threshold)[0]
        top_idx = np.argsort(corr)[-max_active:]
        active_set = list(set(list(active_candidates) + list(top_idx)))
        if len(active_set) > max_active * 2:
            active_set = list(top_idx)

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_rows = []
        b_rows = []

        for k_idx in active_set:
            k = int(k_idx) - mid
            g = np.zeros(n)
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            g[v1] += (1.0 - h[idxs[v1] + k])
            g[v2] -= h[idxs[v2] - k]
            rhs_val = float(np.dot(g, h)) - corr[k_idx]
            row = np.zeros(n + 1)
            row[:n] = g
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(rhs_val)

        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        lower = np.maximum(0.0, h - trust)
        upper = np.minimum(1.0, h + trust)
        lower = np.append(lower, 0.0)
        upper = np.append(upper, 5.0)

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lower, upper)),
                          method='highs')
        except Exception:
            trust *= 0.7
            continue

        if not res.success:
            trust *= 0.7
            if trust < 0.005:
                break
            continue

        h_lp = res.x[:n]
        direction = h_lp - h
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-10:
            trust *= 0.7
            if trust < 0.005:
                break
            continue

        alpha_list = [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]
        improved = False
        for alpha in alpha_list:
            h_trial = h + alpha * direction
            h_trial = project_feasible(h_trial, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_val - 1e-14:
                h = h_trial
                improved = True
                if alpha >= 0.7:
                    trust = min(trust * 1.1, 0.5)
                else:
                    trust = min(trust * 0.95, 0.3)
                break
        if not improved:
            trust = max(trust * 0.6, 0.005)

    return best_h, best_c5


def projected_gradient_refine(h_init, n, target, time_limit, max_iter=10000,
                              init_lr=0.02, decay=0.9995):
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    idxs = np.arange(n)
    lr = init_lr
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            if lr < 1e-6:
                break
            lr *= 5.0
            no_improve = 0
            continue

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_k = np.argsort(corr)[-5:]
        grad = np.zeros(n)
        for s_idx in top_k:
            k = int(s_idx) - (n - 1)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_k))
        grad -= np.sum(grad) / n

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)

        if c5_try < c5_val - 1e-15:
            h = h_try
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= decay

        lr = max(lr, 1e-8)

    return best_h, best_c5


def bundle_refine(h_init, n, target, time_limit, max_iter=2000):
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    mid = n - 1
    no_improve = 0
    cuts = []

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 300:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star = int(np.argmax(corr))
        if len(cuts) < 20:
            already = any(abs(int(k) - k_star) < 1 for k in cuts)
            if not already:
                cuts.append(k_star)
        else:
            cuts[-1] = k_star

        active_threshold = c5_val - 0.003 * dx
        active = [k for k in cuts if corr[k] * dx >= active_threshold]
        if len(active) < 3:
            top3 = np.argsort(corr)[-3:]
            active = list(set(active + list(top3)))

        used = active[-12:]

        grad_sum = np.zeros(n)
        for k_idx in used:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_sum[v1] += (1.0 - h[idxs[v1] - k])
            grad_sum[v2] -= h[idxs[v2] + k]
        grad_sum /= float(len(used))
        grad_sum -= np.sum(grad_sum) / n

        step = 0.05 * max(0.01, 1.0 - it / max_iter)
        h_try = h - step * grad_sum
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)

        if c5_try < c5_val - 1e-14:
            h = h_try
        else:
            h_try = h - step * 0.5 * grad_sum
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val - 1e-14:
                h = h_try

    return best_h, best_c5
