```python
    # h: array of length n
    # w: array of length 2n-1 (weights for each correlation)
    # Returns: gradient of Σ w[m]*corr[m] with respect to h
    grad = np.zeros(n)
    mid = n - 1
    for m in range(2*n-1):
        if abs(w[m]) < 1e-12:
            continue
        lag = m - mid
        # Positive part: (1-h[p-lag]) for p in [max(0,lag), min(n, n+lag))
        p_start = max(0, lag)
        p_end = min(n, n + lag)
        if p_end > p_start:
            grad[p_start:p_end] += w[m] * (1 - h[p_start-lag : p_end-lag])
        # Negative part: -h[p+lag] for p in [max(0,-lag), min(n, n-lag))
        q_start = max(0, -lag)
        q_end = min(n, n - lag)
        if q_end > q_start:
            grad[q_start:q_end] -= w[m] * h[q_start+lag : q_end+lag]
    return grad
```

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:

        else:
            mask = h > 1e-15

            break

    mid = n - 1
    for m in range(2 * n - 1):
        if abs(w[m]) < 1e-12:
            continue
        lag = m - mid
        p_start = max(0, lag)
        p_end = min(n, n + lag)
        if p_end > p_start:
            grad[p_start:p_end] += w[m] * (1.0 - h[p_start - lag: p_end - lag])
        q_start = max(0, -lag)
        q_end = min(n, n - lag)
        if q_end > q_start:
            grad[q_start:q_end] -= w[m] * h[q_start + lag: q_end + lag]
    return grad

def proj_grad_optimize(h_init, n, target, time_limit, beta_start=20.0, beta_end=2000.0,
                       n_iters=20000, step_start=0.05, step_end=0.0001):
    dx = 2.0 / n
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        
        progress = it / n_iters
        beta = beta_start * (beta_end / beta_start) ** progress
        step = step_start * (step_end / step_start) ** progress
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            break
        
        # Compute exponential weights
        shifted = beta * (corr - corr.max())
        shifted = np.clip(shifted, -500.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
        else:
            w /= w_sum
        
        # Compute gradient
        grad = compute_smoothed_gradient(h, w, n)
        grad -= np.mean(grad)
        
        # Gradient step with projection
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        
        # Check if it improved
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val + 1e-14:
            h = h_new
        else:
            # Try smaller step
            h_new2 = h - 0.5 * step * grad
            h_new2 = project_feasible(h_new2, target)
            new_c5_2 = compute_c5(h_new2)
            if new_c5_2 < c5_val + 1e-14:
                h = h_new2
            else:
                no_improve += 1
    
    return best_h, best_c5

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change
