    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    
    corr = compute_corr(h)
    max_corr = np.max(corr)

    P = np.arange(n)[:, None]  # shape (n, 1)
    S = np.arange(n_corr)[None, :]  # shape (1, n_corr)

    neg_mask = (pm >= 0) & (pm < n)  # shape (n, n_corr)

    change += pos_mask * (1.0 - h[pq_clipped])
    change -= neg_mask * h[pm_clipped]

                best_score = sc
                best_h = h_new
        except Exception:
            pass
    return best_h, best_score


def recompute_change(h, n):
    n_corr = 2 * n - 1
    offsets = np.arange(-(n - 1), n)
    P = np.arange(n)[:, None]
    S = offsets[None, :].astype(float)
    pq = P + S
    pm = P - S
    pos_mask = (pq >= 0) & (pq < n)
    neg_mask = (pm >= 0) & (pm < n)
    pq_c = np.clip(pq, 0, n - 1).astype(np.intp)
    pm_c = np.clip(pm, 0, n - 1).astype(np.intp)
    change = np.zeros((n, n_corr))
    change += pos_mask * (1.0 - h[pq_c])
    change -= neg_mask * h[pm_c]
    return change


def fast_cd(h_init, n, target, time_limit, max_sweeps=80000, ncand_min=20, ncand_max=100):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    max_corr = float(np.max(corr))
    change = recompute_change(h, n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        active_thresh = max_corr - 1e-8
        active_mask = corr >= active_thresh
        n_active = int(active_mask.sum())
        if n_active <= 5:
            active_idxs = np.where(active_mask)[0]
        else:
            active_idxs = np.argsort(corr)[-5:]

        grad = np.zeros(n)
        for ai in active_idxs:
            k = int(ai) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(1, len(active_idxs))
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        ncand = int(ncand_min + (ncand_max - ncand_min) * progress)
        ncand = max(10, min(ncand, n // 4))

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        n_h = len(high_set)
        n_l = len(low_set)
        best_new_max = max_corr
        best_i = -1
        best_j = -1
        best_delta = 0.0

        batch_sz = max(4, min(16, n_h // 2))
        for bs in range(0, n_h, batch_sz):
            be = min(bs + batch_sz, n_h)
            bh = high_set[bs:be]

            diff = change[low_set][None, :, :] - change[bh][:, None, :]
            delta_max = np.minimum(h[bh][:, None], 1.0 - h[low_set][None, :])

            g_active = diff[:, :, int(np.argmax(corr))]
            valid = (g_active < -1e-12) & (delta_max > 1e-12)
            if not valid.any():
                continue

            num = max_corr - corr
            with np.errstate(divide='ignore', invalid='ignore'):
                eq_ratio = np.where(diff > 1e-12, num[None, None, :] / diff, np.inf)
            eq_pt = np.min(eq_ratio, axis=2)
            eq_pt = np.clip(eq_pt, 0.0, delta_max)

            cands = np.stack([
                eq_pt,
                delta_max,
                np.clip(eq_pt * 0.5, 0.0, delta_max),
                np.clip(eq_pt * 1.5, 0.0, delta_max),
                np.clip(eq_pt * 2.5, 0.0, delta_max),
            ], axis=2)

            new_vals = corr[None, None, :] + cands * diff
            f_vals = np.max(new_vals, axis=2)
            f_vals[~valid] = np.inf

            if np.all(np.isinf(f_vals)):
                continue

            min_f = np.min(f_vals)
            if min_f < best_new_max - 1e-14:
                bi, bl = np.unravel_index(np.argmin(f_vals), f_vals.shape)
                bci = int(np.argmin(f_vals[bi, bl]))
                best_new_max = float(min_f)
                best_i = int(bh[bi])
                best_j = int(low_set[bl])
                best_delta = float(cands[bi, bl, bci])

        if best_i < 0 or best_delta < 1e-12:
            no_improve += 1
            if no_improve > 300:
                break
            continue

        no_improve = 0
        h[best_i] -= best_delta
        h[best_j] += best_delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        corr = compute_corr(h)
        max_corr = float(np.max(corr))
        change = recompute_change(h, n)

        c5_now = max_corr * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0

    if tleft() > 80:
        h_s, sc = slsqp_refine(best_h, best_n, best_n / 2.0,
                                [100, 500, 2000], 400, min(70, tleft() * 0.7))
        if sc < best_score:
            best_score = sc
            best_h = h_s

    if tleft() > 250:
        t_alloc = min(300, tleft() * 0.55)
        h_cd, sc_cd = fast_cd(best_h, best_n, best_n / 2.0,
                               time_limit=t_alloc, max_sweeps=80000,
                               ncand_min=25, ncand_max=100)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 80:
        h_s2, sc2 = slsqp_refine(best_h, best_n, best_n / 2.0,
                                  [2000, 8000, 20000], 600, min(70, tleft() * 0.6))
        if sc2 < best_score:
            best_score = sc2
            best_h = h_s2

    if tleft() > 200:
        t_alloc2 = min(250, tleft() * 0.5)
        h_cd2, sc_cd2 = fast_cd(best_h, best_n, best_n / 2.0,
                                 time_limit=t_alloc2, max_sweeps=60000,
                                 ncand_min=30, ncand_max=120)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    if tleft() > 500:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 400:
            h1_cd, sc1_cd = fast_cd(h1, n1, target1,
                                     time_limit=min(250, tleft() * 0.45),
                                     max_sweeps=50000, ncand_min=20, ncand_max=80)
            if sc1_cd < sc1:
                sc1 = sc1_cd
                h1 = h1_cd

        if tleft() > 150:
            h1_s, sc1_s = slsqp_refine(h1, n1, target1,
                                        [1000, 5000, 15000], 400,
                                        min(120, tleft() * 0.3))
            if sc1_s < sc1:
                sc1 = sc1_s
                h1 = h1_s

        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            best_n = n1

    if tleft() > 300 and best_n == n0:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 250:
            h1_cd, sc1_cd = fast_cd(h1, n1, target1,
                                     time_limit=min(200, tleft() * 0.5),
                                     max_sweeps=40000, ncand_min=20, ncand_max=70)
            if sc1_cd < sc1:
                sc1 = sc1_cd
                h1 = h1_cd

        if tleft() > 100:
            h1_s, sc1_s = slsqp_refine(h1, n1, target1,
                                        [1000, 5000], 400,
                                        min(80, tleft() * 0.3))
            if sc1_s < sc1:
                sc1 = sc1_s
                h1 = h1_s

        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            best_n = n1

    if tleft() > 100:
        t_final = min(80, tleft() * 0.5)
        h_f, sc_f = fast_cd(best_h, best_n, best_n / 2.0,
                             time_limit=t_final, max_sweeps=30000,
                             ncand_min=30, ncand_max=110)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f

    if tleft() > 50:
        h_fs, sc_fs = slsqp_refine(best_h, best_n, best_n / 2.0,
                                    [5000, 20000, 60000], 500,
                                    min(40, tleft() * 0.6))
        if sc_fs < best_score:
            best_score = sc_fs
            best_h = h_fs

    if tleft() > 30:
        rng = np.random.RandomState(seed)
        for ns in [0.03, 0.08]:
            if tleft() < 25:
                break
            h_p = best_h + rng.randn(best_n) * ns
            h_p = project_feasible(h_p, best_n / 2.0)
            t_r = min(20, tleft() * 0.4)
            h_r, sc_r = fast_cd(h_p, best_n, best_n / 2.0,
                                 time_limit=t_r, max_sweeps=20000,
                                 ncand_min=30, ncand_max=90)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, best_n
