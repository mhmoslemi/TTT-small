    # So: f_s(h_cur) + grad_s @ h - grad_s @ h_cur <= t

            valid2 = np.arange(0, n - s_neg)

    return h, None

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for p in range(n):
        pos_mask = (idxs - shifts >= 0) & (idxs - shifts < n)
        neg_mask = (idxs + shifts >= 0) & (idxs + shifts < n)
        change[p, pos_mask] += (1.0 - h[idxs[pos_mask] - shifts[pos_mask]])
        change[p, neg_mask] -= h[idxs[neg_mask] + shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    idxs = np.arange(n)
    change[p, :] = 0.0
    pos_mask = (idxs - shifts >= 0) & (idxs - shifts < n)
    neg_mask = (idxs + shifts >= 0) & (idxs + shifts < n)
    change[p, pos_mask] += (1.0 - h[idxs[pos_mask] - shifts[pos_mask]])
    change[p, neg_mask] -= h[idxs[neg_mask] + shifts[neg_mask]]
    return change

def cd_search(h_init, n, target, time_limit, max_sweeps=30000,
              n_cand_start=40, n_cand_end=120, max_no_improve=400,
              n_top_shifts=1, shift_weight=5.0):
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        c_max_val = float(np.max(corr))
        top_shifts = np.argsort(corr)[-n_top_shifts:]
        weights = np.exp(-shift_weight * (c_max_val - corr[top_shifts]) / max(c_max_val, 1e-10))
        weights = weights / np.sum(weights)

        grad = np.zeros(n)
        for wi, s_idx in enumerate(top_shifts):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += weights[wi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= weights[wi] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_prime_check(g, vals, f_mid):
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def f_prime_check(g, vals, f_mid):
    active_mask = vals >= f_mid - 1e-14
    if active_mask.any():
        return np.max(g[active_mask]) > 1e-12
    return False

def lp_refine(h, n, target, time_limit, n_active_max=30, max_iters=200):
    """LP-based linearization refinement."""
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    mid = n - 1
    h_cur = h.copy().astype(float)
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h_cur)
        c_max_val = float(np.max(corr))
        c5_val = c_max_val * dx

        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h_cur.copy()

        gap = c_max_val - np.percentile(corr, 50.0)
        threshold = c_max_val - max(gap * 0.1, 1e-6)
        active_mask = corr >= threshold
        active_shifts = np.where(active_mask)[0]

        if len(active_shifts) > n_active_max:
            active_shifts = active_shifts[np.argsort(corr[active_shifts])[-n_active_max:]]

        n_act = len(active_shifts)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_ub = np.zeros((n_act, n + 1))
        b_ub = np.zeros(n_act)

        for idx, s_idx in enumerate(active_shifts):
            s = int(s_idx) - mid
            grad = np.zeros(n)
            if s >= 0:
                if n - s > 0:
                    valid = np.arange(0, n - s)
                    grad[valid] += (1.0 - h_cur[valid + s])
                if s < n:
                    valid2 = np.arange(s, n)
                    grad[valid2] -= h_cur[valid2 - s]
            else:
                s_neg = -s
                if s_neg < n:
                    valid = np.arange(s_neg, n)
                    grad[valid] += (1.0 - h_cur[valid - s_neg])
                if n - s_neg > 0:
                    valid2 = np.arange(0, n - s_neg)
                    grad[valid2] -= h_cur[valid2 + s_neg]

            A_ub[idx, :n] = grad
            A_ub[idx, n] = -1.0
            f_s_val = corr[s_idx]
            b_ub[idx] = grad @ h_cur - f_s_val

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds = [(0.0, 1.0)] * n + [(None, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        except Exception:
            break

        if not res.success:
            break

        h_lp = res.x[:n]
        h_lp = project_feasible(h_lp, target)
        sc_lp = compute_c5(h_lp)

        if sc_lp < c5_val - 1e-14:
            alpha = 0.5
            h_trial = (1 - alpha) * h_cur + alpha * h_lp
            h_trial = project_feasible(h_trial, target)
            sc_trial = compute_c5(h_trial)
            if sc_trial < best_c5:
                h_cur = h_trial
            else:
                h_cur = h_lp
        else:
            h_cur = h_lp

    return best_h, best_c5

def sa_01_search(h_init, n, target, time_limit, max_steps=50000,
                 t_start=0.03, t_end=0.0001, n_cand=80):
    """Simulated annealing on 0-1 projections."""
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(123)

    h_01 = (h_init > 0.5).astype(float)
    n_ones = int(np.sum(h_01))
    ones = np.where(h_01 == 1)[0]
    zeros = np.where(h_01 == 0)[0]
    # Ensure exactly n/2 ones
    while n_ones < n // 2:
        j = int(zeros[rng.randint(len(zeros))])
        h_01[j] = 1.0
        n_ones += 1
        zeros = np.where(h_01 == 0)[0]
    while n_ones > n // 2:
        i = int(ones[rng.randint(len(ones))])
        h_01[i] = 0.0
        n_ones -= 1
        ones = np.where(h_01 == 1)[0]
        zeros = np.where(h_01 == 0)[0]

    h = h_01.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 1000:
            break

        progress = min(step / max(max_steps, 1), 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue
        if h[i] < 0.01 or h[j] > 0.99:
            continue

        g = change[j] - change[i]
        delta = 0.5
        approx_new_c5 = np.max(corr + delta * g) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= 1.0
            h[j] += 1.0
            h = np.clip(h, 0.0, 1.0)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            cur_c5 = approx_new_c5

            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def smooth_obj(x, alpha, n, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h, n, target, alphas, maxiter_per_alpha, ftol=1e-14,
                 time_limit=None, start_time=None):
    dx = 2.0 / n
    start = time.time() if start_time is None else start_time
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha, n, target), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            h_lp1, c5_lp1 = lp_refine(best_h, n0, target0,
                                      time_limit=min(60, tleft() * 0.4),
                                      n_active_max=25, max_iters=80)
            if c5_lp1 < best_score:
                best_score = c5_lp1
                best_h = h_lp1

    # Phase 7: SA escape on current grid
    if tleft() > 80:
        h_sa, c5_sa = sa_01_search(best_h, n0, target0,
                                   time_limit=min(60, tleft() * 0.5),
                                   max_steps=30000,
                                   t_start=0.03, t_end=0.0001,
                                   n_cand=70)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 8: CD after SA
    if tleft() > 60:
        h_cd2, c5_cd2 = cd_search(best_h, n0, target0,
                                  time_limit=min(50, tleft() * 0.5),
                                  max_sweeps=15000,
                                  n_cand_start=60, n_cand_end=120,
                                  max_no_improve=250,
                                  n_top_shifts=3, shift_weight=3.0)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 9: Try n=800
    if tleft() > 200:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_cd2b, c5_cd2b = cd_search(h2, n2, target2,
                                    time_limit=min(100, tleft() * 0.4),
                                    max_sweeps=20000,
                                    n_cand_start=35, n_cand_end=90,
                                    max_no_improve=350,
                                    n_top_shifts=2, shift_weight=4.0)
        if c5_cd2b < best_score:
            best_score = c5_cd2b
            best_h = h_cd2b
            n0 = n2
            target0 = target2

        if tleft() > 60:
            h_lp2, c5_lp2 = lp_refine(best_h, n0, target0,
                                      time_limit=min(50, tleft() * 0.4),
                                      n_active_max=20, max_iters=60)
            if c5_lp2 < best_score:
                best_score = c5_lp2
                best_h = h_lp2

    # Phase 10: High-alpha SLSQP
    if tleft() > 80:
        h_pol2, sc2 = slsqp_refine(best_h, n0, target0,
                                   [20000, 80000],
                                   1200, 1e-14,
                                   time_limit=tleft() * 0.6, start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 11: Final CD
    if tleft() > 40:
        h_cd3, c5_cd3 = cd_search(best_h, n0, target0,
                                  time_limit=min(30, tleft() * 0.6),
                                  max_sweeps=10000,
                                  n_cand_start=80, n_cand_end=120,
                                  max_no_improve=200,
                                  n_top_shifts=3, shift_weight=3.0)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 12: Final SLSQP
    if tleft() > 30:
        h_final, sc_final = slsqp_refine(best_h, n0, target0,
                                         [50000, 200000],
                                         800, 1e-15,
                                         time_limit=tleft() * 0.7, start_time=start_time)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
