**Another improvement**: Instead of just looking at top-3 shifts, use ALL shifts with a smooth weighting. The gradient of the smooth max is:
∇_h smooth_max(c) ∝ softmax(c/α) · ∇c / α

This gives a direction that accounts for ALL shifts, not just the top few.

def smooth_max_gradient(h, alpha):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    shifted = alpha * (corr - np.max(corr))
    weights = np.exp(shifted)
    weights /= weights.sum()
    # Gradient
    grad = np.zeros(n)
    for k_idx in np.where(weights > 1e-10)[0]:
        k = k_idx - mid
        w = weights[k_idx]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad, corr

def softmax_gradient_descent(h_init, n, target, time_limit, n_steps=5000,
                             alpha_start=100.0, alpha_end=2000.0,
                             step_scale=0.01):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h_cur = h_init.copy().astype(float)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h_cur.copy()

        alpha = alpha_start * (alpha_end / alpha_start) ** (step / max(n_steps, 1))
        shifted = alpha * (corr - np.max(corr))
        weights = np.exp(shifted)
        weights /= weights.sum()

        active = weights > 1e-8
        n_active = int(active.sum())
        if n_active > 20:
            top_idx = np.argsort(weights)[-20:]
            active = np.zeros(len(weights), dtype=bool)
            active[top_idx] = True
            weights = np.where(active, weights, 0.0)
            weights /= weights.sum()

        grad = np.zeros(n)
        active_indices = np.where(active)[0]
        for k_idx in active_indices:
            k = int(k_idx) - mid
            w = weights[k_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
            grad[v2] -= w * h_cur[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h_cur - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h_cur = h_try
        else:
            h_try2 = h_cur - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h_cur = h_try2
            else:
                eta *= 0.9

        if step % 1000 == 0 and step > 0:
            eta = max(step_scale * max(0.2, 1.0 - step / n_steps), 1e-5)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]

        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)
        grad /= n_shifts

        ncand = min(n_cand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial = c5_val
        best_move = None

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                delta_lo = max(h[i] - 1.0, -h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d, b_d = delta_lo, delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active_mask]) if active_mask.any() else 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_trial - 1e-14:
                    best_trial = approx
                    best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            changes += 2
            if changes > n // 4:
                change = compute_change_matrix(h, n)
                changes = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    # Phase 1: Initial SLSQP polish
    if tleft() > 120:
        h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                   [100, 500, 2000], 500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Pair CD with change matrix
    if tleft() > 300:
        time_alloc = min(250, tleft() * 0.5)
        h_cd, c5_cd = pair_cd(best_h, cur_n, cur_target,
                              time_limit=time_alloc,
                              max_sweeps=50000,
                              n_cand_start=40, n_cand_end=160,
                              n_shifts=5, max_no_improve=500,
                              anneal_temp=0.008)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Softmax gradient descent
    if tleft() > 150:
        time_alloc = min(120, tleft() * 0.5)
        h_sg, c5_sg = softmax_gradient_descent(best_h, cur_n, cur_target,
                                               time_limit=time_alloc,
                                               n_steps=8000,
                                               alpha_start=80.0,
                                               alpha_end=3000.0,
                                               step_scale=0.012)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 4: SLSQP refinement
    if tleft() > 100:
        h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                   [2000, 8000, 20000], 800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Diversification - try structural variants
    if tleft() > 200:
        h_div, c5_div = diversify_restart(best_h, cur_n, cur_target,
                                          time_limit=30, n_starts=8)
        if c5_div < best_score:
            best_score = c5_div
            best_h = h_div

        # Optimize the diversified solution
        if tleft() > 150:
            h_cd2, c5_cd2 = pair_cd(best_h, cur_n, cur_target,
                                    time_limit=min(100, tleft() * 0.5),
                                    max_sweeps=30000,
                                    n_cand_start=50, n_cand_end=180,
                                    n_shifts=6, max_no_improve=300,
                                    anneal_temp=0.003)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 6: Try higher resolution
    if tleft() > 400:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 300:
            h_cd1, c5_cd1 = pair_cd(h1, n1, target1,
                                    time_limit=min(200, tleft() * 0.5),
                                    max_sweeps=40000,
                                    n_cand_start=30, n_cand_end=120,
                                    n_shifts=4, max_no_improve=400,
                                    anneal_temp=0.005)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                cur_n = n1
                cur_target = target1
            elif sc1 < best_score:
                best_score = sc1
                best_h = h1
                cur_n = n1
                cur_target = target1

        if cur_n == n1 and tleft() > 150:
            h_pol1, sc1 = slsqp_optimize(best_h, cur_n, cur_target,
                                         [1000, 5000], 600, 1e-15)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1

    # Phase 7: More pair CD with different parameters
    if tleft() > 200:
        h_cd3, c5_cd3 = pair_cd(best_h, cur_n, cur_target,
                                time_limit=min(150, tleft() * 0.55),
                                max_sweeps=40000,
                                n_cand_start=60, n_cand_end=200,
                                n_shifts=7, max_no_improve=350,
                                anneal_temp=0.002)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 8: Multi-shift CD variant
    if tleft() > 150:
        h_ms, c5_ms = multi_shift_cd(best_h, cur_n, cur_target,
                                     time_limit=min(120, tleft() * 0.5),
                                     n_shifts=6, n_cand=100,
                                     max_no_improve=250)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 9: Random restarts with SLSQP
    if tleft() > 200:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.06, 0.1, 0.15]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(cur_n) * noise_scale
            h_pert = project_feasible(h_pert, cur_target)
            h_ref, sc = slsqp_optimize(h_pert, cur_n, cur_target,
                                       [500, 2000, 8000], 600, 1e-15)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 10: Final pair CD
    if tleft() > 100:
        h_final_cd, c5_final = pair_cd(best_h, cur_n, cur_target,
                                       time_limit=min(70, tleft() * 0.6),
                                       max_sweeps=25000,
                                       n_cand_start=80, n_cand_end=220,
                                       n_shifts=8, max_no_improve=250,
                                       anneal_temp=0.0)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Phase 11: Final SLSQP
    if tleft() > 40:
        h_final_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                         [5000, 15000, 30000], 1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    # Final verification
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    # Safety check
    if not np.isfinite(final_score) or final_score > 0.5:
        final_h = project_feasible(h_init, n0 / 2.0)
        final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
