def compute_c5(h):

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        grad = grad / grad_norm  # normalized
        
        # Step
        t = it + 1
        lr = lr_start / np.sqrt(t)
        
        h_new = h_y - lr * grad
        h_new = project_feasible(h_new, target)
        
        # Nesterov update
        theta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        h_y = (theta * h_new + (1.0 - t) * h_y) / (theta + 1.0)
        h_y = project_feasible(h_y, target)
        h_x = h_new
    
    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, block_size=8, max_sweeps=5000):
    """Block coordinate descent: optimize blocks of consecutive coordinates."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    n_blocks = n // block_size
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        # Random block
        b = int(np.random.randint(n_blocks))
        start_idx = b * block_size
        end_idx = min(start_idx + block_size, n)
        
        # Optimize the block: minimize max(corr) by adjusting h[start_idx:end_idx]
        # Use a simple approach: for each coordinate in the block, try small perturbations
        for p in range(start_idx, end_idx):
            # Try moving in both directions
            for direction in [-1, 1]:
                for step in [0.01, 0.005, 0.001]:
                    delta = direction * step
                    if h[p] + delta < 0 or h[p] + delta > 1:
                        continue
                    # Compensate sum constraint by adjusting another coordinate
                    # Find coordinate with opposite gradient
                    h_trial = h.copy()
                    h_trial[p] += delta
                    # Find best compensation
                    # Simple: spread the -delta uniformly over all other coords
                    h_trial_excl_p = h_trial.copy()
                    h_trial_excl_p[p] = 0  # exclude from adjustment
                    # Actually, let's just project
                    h_trial = project_feasible(h_trial, target)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < c5_val - 1e-14:
                        h = h_trial
                        c5_val = c5_trial
                        break
                else:
                    continue
                break
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    np.random.seed(seed)
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP polish at original resolution
    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, n0, target0, [100, 500, 2000], 500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD at original resolution
    if tleft() > 60:
        h_cd, c5_cd = enhanced_cd(best_h, n0, target0,
                                   time_limit=min(40, tleft() * 0.25),
                                   max_sweeps=15000,
                                   n_active=4,
                                   n_cand_start=20, n_cand_end=40,
                                   max_no_improve=300, bisection_iters=12)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Upscale to 800 and optimize
    if tleft() > 200:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        tl = min(120, tleft() * 0.4)
        h_cd1, c5_cd1 = enhanced_cd(h1, n1, target1,
                                     time_limit=tl,
                                     max_sweeps=30000,
                                     n_active=4,
                                     n_cand_start=25, n_cand_end=55,
                                     max_no_improve=500, bisection_iters=12)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1
            n0 = n1
            target0 = target1

    # Phase 4: SLSQP at higher resolution
    if tleft() > 150:
        h_pol, sc = slsqp_round(best_h, n0, target0, [500, 2000, 10000], 800, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Upscale to 1200
    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        tl = min(150, tleft() * 0.4)
        h_cd2, c5_cd2 = enhanced_cd(h2, n2, target2,
                                     time_limit=tl,
                                     max_sweeps=40000,
                                     n_active=5,
                                     n_cand_start=30, n_cand_end=60,
                                     max_no_improve=500, bisection_iters=12)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
            n0 = n2
            target0 = target2

    # Phase 6: SLSQP at even higher resolution
    if tleft() > 120:
        h_pol, sc = slsqp_round(best_h, n0, target0, [2000, 10000, 50000], 600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: More CD with different parameters
    if tleft() > 100:
        h_cd3, c5_cd3 = enhanced_cd(best_h, n0, target0,
                                     time_limit=min(80, tleft() * 0.4),
                                     max_sweeps=25000,
                                     n_active=6,
                                     n_cand_start=40, n_cand_end=80,
                                     max_no_improve=400, bisection_iters=12)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 8: Subgradient with averaging
    if tleft() > 80:
        h_sub, c5_sub = subgradient_method(best_h, n0, target0,
                                            time_limit=min(60, tleft() * 0.3),
                                            n_iter=10000, lr_start=0.05)
        if c5_sub < best_score:
            best_score = c5_sub
            best_h = h_sub

    # Phase 9: Final SLSQP polish
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0, [10000, 50000], 500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 10: Final CD
    if tleft() > 40:
        h_cd4, c5_cd4 = enhanced_cd(best_h, n0, target0,
                                     time_limit=min(30, tleft() * 0.5),
                                     max_sweeps=15000,
                                     n_active=5,
                                     n_cand_start=50, n_cand_end=100,
                                     max_no_improve=200, bisection_iters=10)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    # Phase 11: Try perturbation restarts
    if tleft() > 60:
        rng = np.random.RandomState(seed + 1)
        for scale in [0.01, 0.02, 0.05]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(n0) * scale
            h_pert = project_feasible(h_pert, target0)
            h_re, sc_re = enhanced_cd(h_pert, n0, target0,
                                       time_limit=min(30, tleft() * 0.3),
                                       max_sweeps=10000,
                                       n_active=4,
                                       n_cand_start=25, n_cand_end=50,
                                       max_no_improve=200, bisection_iters=10)
            if sc_re < best_score:
                best_score = sc_re
                best_h = h_re

    final_h = project_feasible(best_h, len(best_h) / 2.0)
