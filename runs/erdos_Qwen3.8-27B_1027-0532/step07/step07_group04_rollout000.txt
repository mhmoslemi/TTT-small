import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        for k_idx in range(n_corr):
            k = k_idx - mid
            val = 0.0
            if 0 <= p + k < n:
                val += (1.0 - h[p + k])
            if 0 <= p - k < n:
                val -= h[p - k]
            change[p, k_idx] = val
    return change

def update_change_rows(change, h, n, rows):
    n_corr = 2 * n - 1
    mid = n - 1
    for p in rows:
        for k_idx in range(n_corr):
            k = k_idx - mid
            val = 0.0
            if 0 <= p + k < n:
                val += (1.0 - h[p + k])
            if 0 <= p - k < n:
                val -= h[p - k]
            change[p, k_idx] = val

_g_n = 0
_g_target = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_quick(h_init, n, target, time_limit, alphas=(500, 2000, 8000), maxiter=500):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    start_t = time.time()
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def fast_cd_topk(h_init, n, target, time_limit, max_sweeps=500000,
                 n_cand=60, top_k=20, max_no_improve=300, verify_every=20):
    dx = 2.0 / n
    start_time = time.time()
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change_matrix(h, n)

    d_deltas = np.array([0.2, 0.35, 0.5, 0.65, 0.8, 1.0])

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')

        if sweep % verify_every == 0:
            c5_val = float(np.max(corr)) * dx
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            c5_val = float(np.max(corr)) * dx
            no_improve += 1

        if no_improve > max_no_improve:
            break

        top_idxs = np.argsort(corr)[-top_k:]
        c_max = corr[np.argmax(corr)]
        weights = np.exp(-4.0 * (c_max - corr[top_idxs]) / max(c_max, 1e-10))
        weights /= weights.sum()

        grad = np.zeros(n)
        for wi, s_idx in enumerate(np.atleast_1d(top_idxs)):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += weights[wi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= weights[wi] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        ncand = min(n_cand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial = c5_val
        best_i, best_j, best_delta = -1, -1, 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            ch_i = change[i]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                h_j = h[j]
                d_lo = max(h_i - 1.0, -h_j)
                d_hi = min(h_i, 1.0 - h_j)
                if d_hi - d_lo < 1e-12:
                    continue
                g = change[j] - ch_i
                for frac in d_deltas:
                    delta = d_lo + frac * (d_hi - d_lo)
                    if abs(delta) < 1e-12:
                        continue
                    approx_max = np.max(corr[top_idxs] + delta * g[top_idxs]) * dx
                    if approx_max < best_trial - 1e-15:
                        best_trial = approx_max
                        best_i, best_j, best_delta = i, j, delta

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_rows(change, h, n, [best_i, best_j])

    return best_h, best_c5

def sa_cd(h_init, n, target, time_limit, max_steps=50000,
          t_start=0.03, t_end=0.0002, n_cand=50, top_k=15, seed=42):
    dx = 2.0 / n
    start_time = time.time()
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 800:
            break

        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        top_idxs = np.argsort(corr)[-top_k:]
        c_max = corr[np.argmax(corr)]
        weights = np.exp(-4.0 * (c_max - corr[top_idxs]) / max(c_max, 1e-10))
        weights /= weights.sum()

        grad = np.zeros(n)
        for wi, s_idx in enumerate(np.atleast_1d(top_idxs)):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += weights[wi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= weights[wi] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        ncand = min(n_cand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        d_lo = max(h[i] - 1.0, -h[j])
        d_hi = min(h[i], 1.0 - h[j])
        if d_hi - d_lo < 1e-12:
            continue

        delta = d_lo + rng.rand() * (d_hi - d_lo)
        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new = np.max(corr[top_idxs] + delta * g[top_idxs]) * dx
        delta_c = approx_new - c5_val

        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-12:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_rows(change, h, n, [i, j])
            if approx_new < best_c5 - 1e-14:
                best_c5 = approx_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

                                   max_sweeps=100000, n_cand=70, top_k=20,
                                   max_no_improve=100)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
