    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

        grad -= np.mean(grad)

        # with some randomness
        sorted_idx = np.argsort(grad)
        n_high = max(10, n_cand)

            else:

        else:

        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))

        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue

        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue

        g_vec = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g_vec) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            ratio = -delta_c / max(temperature, 1e-15)
            if ratio < 500 and rng.rand() < np.exp(-ratio):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            cur_c5 = approx_new_c5

            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def slsqp_smooth(h_init, n, target, alpha, maxiter, time_limit, start_time, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_vals = np.exp(shifted)
        log_sum = np.log(np.sum(exp_vals))
        softmax_weights = exp_vals / np.sum(exp_vals)
        g_corr = softmax_weights
        idxs = np.arange(n)
        mid = n - 1
        grad = np.zeros(n)
        for s_idx in range(len(g_corr)):
            w = g_corr[s_idx]
            if w < 1e-10:
                continue
            k = s_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - x[idxs[v1] - k])
            grad[v2] -= w * x[idxs[v2] + k]
        return grad * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_init.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() - start_time > time_limit:
            return h_init, compute_c5(h_init)
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_init, compute_c5(h_init)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: CD at n=500
    if tleft() > 120:
        n1 = 500
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
        h_cd1, c5_1 = cd_optimize(h1, n1, target1,
                                  time_limit=min(110, tleft() * 0.7),
                                  max_sweeps=15000,
                                  n_cand_start=50, n_cand_end=140,
                                  max_no_improve=350, n_bisect=20,
                                  multi_top=3, rng_seed=seed)
        if c5_1 < best_score:
            best_score = c5_1
            best_h = h_cd1

    # Phase 2: SA on n=500
    if tleft() > 60:
        n1 = 500
        target1 = n1 / 2.0
        h_sa, c5_sa = sa_gradient(best_h, n1, target1,
                                  time_limit=min(50, tleft() * 0.6),
                                  max_steps=15000,
                                  t_start=0.02, t_end=0.0002,
                                  n_cand=100, rng_seed=seed + 1)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 3: CD again on n=500
    if tleft() > 60:
        h_cd2, c5_2 = cd_optimize(best_h, 500, 250.0,
                                  time_limit=min(50, tleft() * 0.6),
                                  max_sweeps=10000,
                                  n_cand_start=80, n_cand_end=150,
                                  max_no_improve=250, n_bisect=20,
                                  multi_top=4, rng_seed=seed + 2)
        if c5_2 < best_score:
            best_score = c5_2
            best_h = h_cd2

    # Phase 4: Scale to n=800, CD
    if tleft() > 180:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
        h_cd3, c5_3 = cd_optimize(h2, n2, target2,
                                  time_limit=min(160, tleft() * 0.7),
                                  max_sweeps=12000,
                                  n_cand_start=60, n_cand_end=180,
                                  max_no_improve=350, n_bisect=20,
                                  multi_top=3, rng_seed=seed + 3)
        if c5_3 < best_score:
            best_score = c5_3
            best_h = h_cd3

    # Phase 5: SA on n=800
    if tleft() > 60:
        h_sa2, c5_sa2 = sa_gradient(best_h, 800, 400.0,
                                    time_limit=min(50, tleft() * 0.5),
                                    max_steps=12000,
                                    t_start=0.015, t_end=0.0001,
                                    n_cand=120, rng_seed=seed + 4)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2

    # Phase 6: CD on n=800
    if tleft() > 80:
        h_cd4, c5_4 = cd_optimize(best_h, 800, 400.0,
                                  time_limit=min(70, tleft() * 0.6),
                                  max_sweeps=8000,
                                  n_cand_start=100, n_cand_end=200,
                                  max_no_improve=250, n_bisect=20,
                                  multi_top=4, rng_seed=seed + 5)
        if c5_4 < best_score:
            best_score = c5_4
            best_h = h_cd4

    # Phase 7: Scale to n=1000, CD
    if tleft() > 200:
        n3 = 1000
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        sc3 = compute_c5(h3)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3
        h_cd5, c5_5 = cd_optimize(h3, n3, target3,
                                  time_limit=min(180, tleft() * 0.7),
                                  max_sweeps=8000,
                                  n_cand_start=60, n_cand_end=180,
                                  max_no_improve=300, n_bisect=18,
                                  multi_top=3, rng_seed=seed + 6)
        if c5_5 < best_score:
            best_score = c5_5
            best_h = h_cd5

    # Phase 8: SA on n=1000
    if tleft() > 50:
        h_sa3, c5_sa3 = sa_gradient(best_h, 1000, 500.0,
                                    time_limit=min(40, tleft() * 0.5),
                                    max_steps=10000,
                                    t_start=0.01, t_end=0.0001,
                                    n_cand=120, rng_seed=seed + 7)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3

    # Phase 9: Final CD polish
    if tleft() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cd6, c5_6 = cd_optimize(best_h, n_cur, target_cur,
                                  time_limit=min(35, tleft() * 0.7),
                                  max_sweeps=6000,
                                  n_cand_start=100, n_cand_end=200,
                                  max_no_improve=200, n_bisect=18,
                                  multi_top=4, rng_seed=seed + 8)
        if c5_6 < best_score:
            best_score = c5_6
            best_h = h_cd6

    # Phase 10: SLSQP polish if time remains
    if tleft() > 50 and len(best_h) <= 800:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sl, sc_sl = slsqp_smooth(best_h, n_cur, target_cur,
                                   alpha=2000, maxiter=500,
                                   time_limit=min(40, tleft() * 0.7),
                                   start_time=start_time)
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    # Final CD with whatever time is left
    if tleft() > 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cf, c5_cf = cd_optimize(best_h, n_cur, target_cur,
                                  time_limit=min(15, tleft() * 0.7),
                                  max_sweeps=4000,
                                  n_cand_start=120, n_cand_end=200,
                                  max_no_improve=100, n_bisect=16,
                                  rng_seed=seed + 9)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
