def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        else:
            mask = h > 1e-15
        if not mask.any():

def compute_change_matrix(h, n):

    at lag k when h[p] is increased by 1 (keeping others fixed)."""
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        # When h[p] increases by delta:
        # corr_k changes for each k by: delta * (1-h[p-k]) - delta * h[p+k]
        # where the first term is valid when p-k is in [0,n)
        # and the second when p+k is in [0,n)

        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

import cvxpy as cp

def linearized_lp(h, n, target, max_lags=None):
    """Solve the linearized relaxation of the Erdős problem."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    n_corr = 2 * n - 1
    
    # Select top lags to constrain (for tractability)
    if max_lags is not None and max_lags < n_corr:
        top_idx = np.argsort(corr)[-max_lags:]
    else:
        top_idx = np.arange(n_corr)
    
    # Variables
    delta = cp.Variable(n)
    t = cp.Variable()
    
    # Objective
    obj = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # t >= linearized correlation at each selected lag
    for k_idx in top_idx:
        k = k_idx - mid
        # Gradient of corr_k with respect to h
        # corr_k = sum_i h[i] * (1-h[i+k])
        # d/dh[p] corr_k = sum of contributions where p appears
        # p appears as "first" index: (1-h[p+k]) if p+k < n
        # p appears as "second" index: -h[p-k] if p-k >= 0
        
        # Build gradient vector
        grad_k = np.zeros(n)
        for p in range(n):
            if p + k < n and p + k >= 0:
                grad_k[p] += (1.0 - h[p + k])
            if p - k >= 0 and p - k < n:
                grad_k[p] -= h[p - k]
        
        # Linear constraint: corr[k_idx] + grad_k @ delta <= t (all * dx)
        constraints.append(corr[k_idx] * dx + cp.multiply(grad_k, delta) @ 1 <= t)
    
    # Box constraints on h + delta
    constraints.append(h + delta >= 0)
    constraints.append(h + delta <= 1)
    
    # Sum constraint
    constraints.append(cp.sum(h + delta) == target)
    
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.GLOP)  # or CBC, GLPK
    except:
        try:
            prob.solve(solver=cp.CBC)
        except:
            try:
                prob.solve(solver=cp.GLPK)
            except:
                return h, None
    
    if delta.value is not None:
        h_new = np.clip(h + delta.value, 0, 1)
        h_new = project_feasible(h_new, target)
        return h_new, compute_c5(h_new)
    
    return h, compute_c5(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

    actual_c5 = compute_c5(h_new)
    
    return h_new, actual_c5

    return best_h, best_c5


def coordinate_descent(h_init, n, target, time_limit,
                       max_sweeps=30000, n_active=5, n_cand=40,
                       max_no_improve=300, bisection_iters=12):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        max_pairs = 300
        if n_h * n_l <= max_pairs:
            pair_list = []
            for ii in range(n_h):
                for jj in range(n_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pair_list.append((iv, jv))
        else:
            pair_list = []
            for frac in np.linspace(0.0, 1.0, min(max_pairs, 20)):
                hi = int(frac * (n_h - 1)) if n_h > 1 else 0
                lo = int(frac * (n_l - 1)) if n_l > 1 else 0
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pair_list.append((iv, jv))
            rng = np.random.RandomState(sweep)
            while len(pair_list) < max_pairs:
                ii = rng.randint(n_h)
                jj = rng.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))

        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g = change[j] - change[i]
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial - 1e-14:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 100:
        h_lp, c5_lp = lp_trust_region(best_h, n, target,
                                       time_limit=min(200, tleft() * 0.4),
                                       initial_radius=0.15, min_radius=0.005)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 100:
        h_cd, c5_cd = coordinate_descent(best_h, n, target,
                                          time_limit=min(100, tleft() * 0.25),
                                          max_sweeps=15000, n_active=5,
                                          n_cand=40, max_no_improve=250)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 200:
        h_lp2, c5_lp2 = lp_trust_region(best_h, n, target,
                                         time_limit=min(150, tleft() * 0.3),
                                         initial_radius=0.1, min_radius=0.003)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [100, 500], 400, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        c5_2 = compute_c5(h2)

        h_lp3, c5_lp3 = lp_trust_region(h2, n2, target2,
                                         time_limit=min(200, tleft() * 0.35),
                                         initial_radius=0.12, min_radius=0.004)
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3
            n = n2
            target = target2
        else:
            if c5_2 < best_score:
                best_score = c5_2
                best_h = h2
                n = n2
                target = target2

    if tleft() > 100:
        h_cd2, c5_cd2 = coordinate_descent(best_h, n, target,
                                            time_limit=min(100, tleft() * 0.3),
                                            max_sweeps=20000, n_active=5,
                                            n_cand=50, max_no_improve=300)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 120:
        h_lp4, c5_lp4 = lp_trust_region(best_h, n, target,
                                         time_limit=min(120, tleft() * 0.3),
                                         initial_radius=0.08, min_radius=0.002)
        if c5_lp4 < best_score:
            best_score = c5_lp4
            best_h = h_lp4

    if tleft() > 80:
        h_cd3, c5_cd3 = coordinate_descent(best_h, n, target,
                                            time_limit=min(60, tleft() * 0.3),
                                            max_sweeps=15000, n_active=7,
                                            n_cand=60, max_no_improve=200)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 80:
        h_pol2 = slsqp_round(best_h, n, target, [500, 2000], 500, 1e-14)
        sc = compute_c5(h_pol2)
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    if tleft() > 80:
        h_lp5, c5_lp5 = lp_trust_region(best_h, n, target,
                                         time_limit=min(60, tleft() * 0.3),
                                         initial_radius=0.05, min_radius=0.001)
        if c5_lp5 < best_score:
            best_score = c5_lp5
            best_h = h_lp5

    if tleft() > 50:
        rng = np.random.RandomState(seed)
        for scale in [0.02, 0.05]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(len(best_h)) * scale
            h_pert = project_feasible(h_pert, target)
            h_re, sc_re = coordinate_descent(h_pert, len(best_h), target,
                                             time_limit=min(30, tleft() * 0.25),
                                             max_sweeps=8000, n_active=5,
                                             n_cand=40, max_no_improve=150)
            if sc_re < best_score:
                best_score = sc_re
                best_h = h_re

    if tleft() > 40:
        h_cd4, c5_cd4 = coordinate_descent(best_h, n, target,
                                            time_limit=min(30, tleft() * 0.5),
                                            max_sweeps=10000, n_active=5,
                                            n_cand=50, max_no_improve=150)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    if tleft() > 30:
        h_final = slsqp_round(best_h, n, target, [2000], 300, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
