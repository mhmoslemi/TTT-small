            break

def smooth_obj_val(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)

        w = weights[k_idx]
        v1 = (idxs - k >= 0) & (idxs - k < n)

        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    grad -= np.mean(grad)
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

def nesterov_smooth(h_init, n, target, time_limit, alpha_start=200, alpha_end=50000,

    h = h_init.copy().astype(float)

        progress = min(step / max(n_steps, 1), 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

            best_h = h_new.copy()

            change[p, :] = 0.0
            change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
            change[p, neg_mask] -= h[p - shifts[neg_mask]]
            p = best_j
            pos_mask = (p + shifts >= 0) & (p + shifts < n)
            neg_mask = (p - shifts >= 0) & (p - shifts < n)
            change[p, :] = 0.0
            change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
            change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, block_size=4, n_cand=40, max_sweeps=10000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 200:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_block = sorted_idx[:block_size]
        high_block = sorted_idx[-block_size:]

        # Try moving mass from one in high_block to one in low_block
        # with a small perturbation of neighboring elements
        best_trial_c5 = c5_val
        best_update = None

        for i in high_block:
            for j in low_block:
                i, j = int(i), int(j)
                if i == j:
                    continue
                # Compute change vector for moving delta from i to j
                # Track which corr values change and how
                change_vec = np.zeros(n_corr)
                # Effect on i
                pos_i = (i + shifts >= 0) & (i + shifts < n)
                neg_i = (i - shifts >= 0) & (i - shifts < n)
                change_vec[pos_i] -= (1.0 - h[i + shifts[pos_i]])
                change_vec[neg_i] += h[i - shifts[neg_i]]
                # Effect on j
                pos_j = (j + shifts >= 0) & (j + shifts < n)
                neg_j = (j - shifts >= 0) & (j - shifts < n)
                change_vec[pos_j] += (1.0 - h[j + shifts[pos_j]])
                change_vec[neg_j] -= h[j - shifts[neg_j]]

                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * change_vec
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(change_vec[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * change_vec) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_update = (i, j, delta_star)

        if best_update is not None:
            i, j, delta = best_update
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def gradient_descent_active(h_init, n, target, time_limit, max_steps=50000,
                            base_lr=0.02, n_active=10, lr_decay=0.9999):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.92
                if lr < 1e-7:
                    break

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[0]]
        weights = np.exp(-60.0 * (c_max - corr[top_shifts]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
        else:
            lr *= 0.5
            if lr < 1e-8:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 25.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    # Phase 1: Quick polish at original resolution
    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, best_n, best_target, [200, 1000, 5000], 800)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Nesterov momentum at original resolution
    if tleft() > 80:
        h_nest, sc = nesterov_smooth(best_h, best_n, best_target,
                                     time_limit=min(60, tleft() * 0.4),
                                     alpha_start=100, alpha_end=20000,
                                     n_steps=30000, momentum=0.9)
        if sc < best_score:
            best_score = sc
            best_h = h_nest

    # Phase 3: CD at original resolution
    if tleft() > 60:
        h_cd, sc = weighted_grad_cd(best_h, best_n, best_target,
                                    time_limit=min(40, tleft() * 0.3),
                                    n_active=8, n_cand_start=30, n_cand_end=80,
                                    max_no_improve=300, max_sweeps=15000)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 4: Upscale to n=800
    n1 = 800
    if tleft() > 300:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        # SLSQP warmup at new resolution
        if tleft() > 250:
            h1_pol, sc1p = slsqp_round(h1, n1, target1, [100, 500, 2000], 600)
            if sc1p < sc1:
                h1 = h1_pol
                sc1 = sc1p

        # Nesterov momentum - main workhorse at high res
        if tleft() > 200:
            h1_nest, sc1n = nesterov_smooth(h1, n1, target1,
                                            time_limit=min(180, tleft() * 0.4),
                                            alpha_start=200, alpha_end=100000,
                                            n_steps=60000, momentum=0.92)
            if sc1n < sc1:
                h1 = h1_nest
                sc1 = sc1n

        # Weighted CD at high res
        if tleft() > 150:
            h1_cd, sc1c = weighted_grad_cd(h1, n1, target1,
                                           time_limit=min(120, tleft() * 0.4),
                                           n_active=10, n_cand_start=40, n_cand_end=120,
                                           max_no_improve=400, max_sweeps=25000)
            if sc1c < sc1:
                h1 = h1_cd
                sc1 = sc1c

        # Gradient descent at high res
        if tleft() > 100:
            h1_gd, sc1g = gradient_descent_active(h1, n1, target1,
                                                  time_limit=min(60, tleft() * 0.3),
                                                  max_steps=40000, base_lr=0.02,
                                                  n_active=10)
            if sc1g < sc1:
                h1 = h1_gd
                sc1 = sc1g

        # Block CD at high res
        if tleft() > 80:
            h1_bc, sc1b = block_cd(h1, n1, target1,
                                   time_limit=min(50, tleft() * 0.3),
                                   block_size=5, n_cand=50, max_sweeps=8000)
            if sc1b < sc1:
                h1 = h1_bc
                sc1 = sc1b

        # Final SLSQP polish at high res
        if tleft() > 80:
            h1_fin, sc1f = slsqp_round(h1, n1, target1, [5000, 20000], 1500)
            if sc1f < sc1:
                h1 = h1_fin
                sc1 = sc1f

        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            best_n = n1
            best_target = target1

    # Phase 5: Try even higher resolution if time permits
    n2 = 1200
    if tleft() > 400:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 350:
            h2_pol, sc2p = slsqp_round(h2, n2, target2, [200, 1000], 500)
            if sc2p < sc2:
                h2 = h2_pol
                sc2 = sc2p

        if tleft() > 250:
            h2_nest, sc2n = nesterov_smooth(h2, n2, target2,
                                            time_limit=min(200, tleft() * 0.4),
                                            alpha_start=200, alpha_end=80000,
                                            n_steps=50000, momentum=0.92)
            if sc2n < sc2:
                h2 = h2_nest
                sc2 = sc2n

        if tleft() > 150:
            h2_cd, sc2c = weighted_grad_cd(h2, n2, target2,
                                           time_limit=min(120, tleft() * 0.4),
                                           n_active=12, n_cand_start=50, n_cand_end=150,
                                           max_no_improve=300, max_sweeps=20000)
            if sc2c < sc2:
                h2 = h2_cd
                sc2 = sc2c

        if tleft() > 80:
            h2_gd, sc2g = gradient_descent_active(h2, n2, target2,
                                                  time_limit=min(50, tleft() * 0.3),
                                                  max_steps=30000, base_lr=0.015,
                                                  n_active=12)
            if sc2g < sc2:
                h2 = h2_gd
                sc2 = sc2g

        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            best_n = n2
            best_target = target2

    # Phase 6: Final refinement on best solution found
    if tleft() > 100:
        h_fin, sc_fin = slsqp_round(best_h, best_n, best_target, [10000, 50000], 2000)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    if tleft() > 80:
        h_cd_fin, sc_cd = weighted_grad_cd(best_h, best_n, best_target,
                                           time_limit=min(60, tleft() * 0.5),
                                           n_active=12, n_cand_start=50, n_cand_end=150,
                                           max_no_improve=200, max_sweeps=15000)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd_fin

    # Phase 7: Random restart perturbations
    if tleft() > 60:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.05]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            h_ref, sc = slsqp_round(h_pert, best_n, best_target, [500, 5000], 500)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 8: Last CD sweep
    if tleft() > 30:
        h_cd_last, sc_last = weighted_grad_cd(best_h, best_n, best_target,
                                              time_limit=min(20, tleft() * 0.5),
                                              n_active=15, n_cand_start=60, n_cand_end=150,
                                              max_no_improve=100, max_sweeps=8000)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_cd_last

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
