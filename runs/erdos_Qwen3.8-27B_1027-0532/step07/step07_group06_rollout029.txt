import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_simplex_box(h, target_sum):
    """Project h onto {x : 0<=x<=1, sum(x)=target_sum} using bisection on shift."""
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h - mid, 0.0, 1.0)

def subgradient_c5(h):
    """Compute subgradient of C5(h) = max_k corr[k] * dx at the active k."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    k = k_star - (n - 1)

    # = (1-h[i+k]) - h[i-k] (with boundary handling)
    # More precisely:
    # corr[k] = sum_{i: 0<=i<n, 0<=i+k<n} h[i](1-h[i+k])
    # d/dh[p] = (1-h[p+k]) if 0<=p+k<n, else 0
    #            + (-h[p-k]) if 0<=p-k<n, else 0

        valid1 = np.arange(n - k)
        g[valid1] += (1.0 - h[valid1 + k])
        valid2 = np.arange(k, n)
        g[valid2] -= h[valid2 - k]

        g[valid1] += (1.0 - h[valid1 + kk])
        valid2 = np.arange(kk, n)
        g[valid2] -= h[valid2 - kk]
    return g * dx, float(np.max(corr)) * dx

def proximal_subgradient(h_init, n, target, time_limit, start_time, 
                          max_iters=500000, alpha_init=0.1, decay=0.9999,
                          restart_every=5000):
    """Proximal subgradient method with restarts."""
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    
    alpha = alpha_init
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if it > 0 and it % restart_every == 0:
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_h = h.copy()
            # Restart: go back to best, increase step size slightly
            h = best_h.copy()
            alpha = alpha_init * (1.0 + 0.1 * (it / restart_every))

        best_score = final_score

    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def pair_cd(h_init, n, target, time_limit, start_time,

        change[p, neg_mask] -= h[p - shifts[neg_mask]]

                v2 = idxs[idxs >= k_star]

def subgradient_active(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star_idx = int(np.argmax(corr))
    k = k_star_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        valid_a = np.arange(n - k)
        g[valid_a] += (1.0 - h[valid_a + k])
        valid_b = np.arange(k, n)
        g[valid_b] -= h[valid_b - k]
    else:
        kk = -k
        valid_a = np.arange(n - kk)
        g[valid_a] += (1.0 - h[valid_a + kk])
        valid_b = np.arange(kk, n)
        g[valid_b] -= h[valid_b - kk]
    return g * dx, float(np.max(corr)) * dx

def proximal_subgrad(h_init, n, target, time_limit, start_time,
                     max_iters=1000000, alpha_init=0.08, decay=0.99998,
                     restart_every=5000, n_active=1):
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    alpha = alpha_init
    cur_score = best_score

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if it > 0 and it % restart_every == 0:
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_h = h.copy()
            h = best_h.copy()
            alpha = alpha_init * (1.0 + 0.05 * (it // restart_every))

        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_score = float(np.max(corr)) * dx

        if n_active <= 1:
            k_star_idx = int(np.argmax(corr))
            k = k_star_idx - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                va = np.arange(n - k)
                g[va] += (1.0 - h[va + k])
                vb = np.arange(k, n)
                g[vb] -= h[vb - k]
            else:
                kk = -k
                va = np.arange(n - kk)
                g[va] += (1.0 - h[va + kk])
                vb = np.arange(kk, n)
                g[vb] -= h[vb - kk]
        else:
            top = np.argsort(corr)[-n_active:]
            weights = corr[top].astype(float)
            wsum = weights.sum()
            if wsum < 1e-15:
                weights = np.ones(n_active) / n_active
            else:
                weights = weights / wsum
            g = np.zeros(n)
            for si, w in zip(top, weights):
                kk = int(si) - (n - 1)
                if kk >= 0:
                    va = np.arange(n - kk)
                    g[va] += w * (1.0 - h[va + kk])
                    vb = np.arange(kk, n)
                    g[vb] -= w * h[vb - kk]
                else:
                    kk2 = -kk
                    va = np.arange(n - kk2)
                    g[va] += w * (1.0 - h[va + kk2])
                    vb = np.arange(kk2, n)
                    g[vb] -= w * h[vb - kk2]

        g *= dx
        h_new = h - alpha * g
        h = project_simplex_box(h_new, target)
        alpha *= decay

    final_score = compute_c5(h)
    if final_score < best_score:
        best_h = h.copy()
        best_score = final_score
    return best_h, best_score

def lse_func(h, alpha, target):
    n = len(h)
    dx = 2.0 / n
    hp = project_feasible(h, target)
    corr = np.correlate(hp, 1.0 - hp, mode='full')
    cm = np.max(corr)
    shifted = np.clip(alpha * (corr - cm), -500.0, 0.0)
    return float(cm + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def obj(x):
            return lse_func(x, alpha, target)
        try:
            res = minimize(obj, h_cur, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def pair_cd(h_init, n, target, time_limit, start_time,
            max_sweeps=30000, n_cand_start=15, n_cand_end=60,
            max_no_improve=400, n_top_shifts=1, bisection_iters=16):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = np.zeros((n, n_corr))
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max_no_improve:
            break

        if n_top_shifts <= 1:
            k_star = int(np.argmax(corr)) - mid
            g = np.zeros(n)
            if k_star >= 0:
                va = np.arange(n - k_star)
                g[va] += (1.0 - h[va + k_star])
                vb = np.arange(k_star, n)
                g[vb] -= h[vb - k_star]
            else:
                kk = -k_star
                va = np.arange(n - kk)
                g[va] += (1.0 - h[va + kk])
                vb = np.arange(kk, n)
                g[vb] -= h[vb - kk]
        else:
            top = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top].astype(float)
            ws = weights.sum()
            if ws < 1e-15:
                weights = np.ones(n_top_shifts) / n_top_shifts
            else:
                weights = weights / ws
            g = np.zeros(n)
            for si, w in zip(top, weights):
                kk = int(si) - mid
                if kk >= 0:
                    va = np.arange(n - kk)
                    g[va] += w * (1.0 - h[va + kk])
                    vb = np.arange(kk, n)
                    g[vb] -= w * h[vb - kk]
                else:
                    kk2 = -kk
                    va = np.arange(n - kk2)
                    g[va] += w * (1.0 - h[va + kk2])
                    vb = np.arange(kk2, n)
                    g[vb] -= w * h[vb - kk2]
        g -= np.mean(g)

        prog = min(float(sweep) / max(max_sweeps, 1), 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * prog)
        ncand = min(ncand, n // 3)
        ncand = max(ncand, 5)

        sidx = np.argsort(g)
        low_set = sidx[:ncand]
        high_set = sidx[-ncand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                dlo = max(h[i] - 1.0, -h[j])
                dhi = min(h[i], 1.0 - h[j])
                if dhi - dlo < 1e-12:
                    continue
                gg = change[j] - change[i]
                a_d, b_d = dlo, dhi
                for _ in range(bisection_iters):
                    m = 0.5 * (a_d + b_d)
                    vals = corr + m * gg
                    fm = np.max(vals)
                    am = vals >= fm - 1e-14
                    fp = np.max(gg[am]) if am.any() else 0.0
                    if fp > 1e-12:
                        b_d = m
                    else:
                        a_d = m
                ds = 0.5 * (a_d + b_d)
                if abs(ds) < 1e-12:
                    continue
                am = np.max(corr + ds * gg) * dx
                if am < best_trial:
                    best_trial = am
                    best_i = i
                    best_j = j
                    best_delta = ds

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                pm = (p + shifts >= 0) & (p + shifts < n)
                nm = (p - shifts >= 0) & (p - shifts < n)
                change[p, :] = 0.0
                change[p, pm] += (1.0 - h[p + shifts[pm]])
                change[p, nm] -= h[p - shifts[nm]]

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def stochastic_kick(h_init, n, target, time_limit, start_time,
                    max_steps=30000, base_scale=0.02):
    rng = np.random.RandomState(12345)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 5000:
            break
        scale = base_scale * max(0.03, 1.0 - step / float(max_steps))
        npert = int(rng.choice([1, 2, 3, 5, 8]))
        npert = min(npert, n)
        perm = rng.permutation(n)
        idxs = perm[:npert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(npert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 100:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 150:
        tl = start_time + min(150, remaining - 30)
        h_ps, sc_ps = proximal_subgrad(best_h, n0, target0, tl, start_time,
                                       max_iters=2000000, alpha_init=0.06,
                                       decay=0.99997, restart_every=8000, n_active=2)
        if sc_ps < best_score:
            best_score = sc_ps
            best_h = h_ps

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 100:
        tl = start_time + min(100, remaining - 30)
        h_sl, sc_sl = slsqp_round(best_h, n0, target0,
                                  [100, 500, 2000], 500, tl, start_time)
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 200:
        tl = start_time + min(200, remaining - 30)
        h_cd, sc_cd = pair_cd(best_h, n0, target0, tl, start_time,
                              max_sweeps=60000, n_cand_start=15, n_cand_end=55,
                              max_no_improve=500, n_top_shifts=3, bisection_iters=18)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 250:
        n1 = 800
        target1 = n1 / 2.0
        h_int = interpolate_to(best_h, n1, target1)
        tl = start_time + min(80, remaining - 100)
        h_h, sc_h = slsqp_round(h_int, n1, target1, [200, 1000], 400, tl, start_time)
        if sc_h < best_score:
            best_score = sc_h
            best_h = h_h
            n0, target0 = n1, target1
        remaining = budget_s - 30.0 - (time.time() - start_time)
        if remaining > 150:
            tl2 = start_time + min(150, remaining - 30)
            h_cd2, sc_cd2 = pair_cd(best_h, n0, target0, tl2, start_time,
                                    max_sweeps=40000, n_cand_start=12, n_cand_end=45,
                                    max_no_improve=350, n_top_shifts=2, bisection_iters=16)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_h = h_cd2

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 120:
        tl = start_time + min(120, remaining - 20)
        h_cd3, sc_cd3 = pair_cd(best_h, n0, target0, tl, start_time,
                                max_sweeps=40000, n_cand_start=25, n_cand_end=70,
                                max_no_improve=300, n_top_shifts=1, bisection_iters=20)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 80:
        tl = start_time + min(80, remaining - 20)
        h_fin, sc_fin = slsqp_round(best_h, n0, target0,
                                    [5000, 15000], 600, tl, start_time)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 60:
        tl = start_time + min(60, remaining - 15)
        h_sk, sc_sk = stochastic_kick(best_h, n0, target0, tl, start_time,
                                      max_steps=20000, base_scale=0.015)
        if sc_sk < best_score:
            best_score = sc_sk
            best_h = h_sk

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 50:
        rng = np.random.RandomState(seed)
        for ns in [0.012, 0.03]:
            remaining = budget_s - 30.0 - (time.time() - start_time)
            if remaining < 40:
                break
            h_pert = best_h + rng.randn(n0) * ns
            h_pert = project_feasible(h_pert, target0)
            tl = start_time + min(35, remaining - 15)
            h_pk, sc_pk = slsqp_round(h_pert, n0, target0,
                                      [500, 3000], 400, tl, start_time)
            if sc_pk < best_score:
                best_score = sc_pk
                best_h = h_pk

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 40:
        tl = start_time + min(40, remaining - 10)
        h_f2, sc_f2 = pair_cd(best_h, n0, target0, tl, start_time,
                              max_sweeps=20000, n_cand_start=35, n_cand_end=80,
                              max_no_improve=150, n_top_shifts=1, bisection_iters=20)
        if sc_f2 < best_score:
            best_score = sc_f2
            best_h = h_f2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
