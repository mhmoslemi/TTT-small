    """Build matrix M where M[i,k] = d corr[k] / d h[i]"""
    n_corr = 2 * n - 1
    mid = n - 1
    M = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for p in range(n):
        # corr[k] = sum_i h[i](1-h[i+k]) for valid indices
        # d corr[k] / d h[p] = (1 - h[p+k]) if p+k valid, for the h[i] term
        #                      - h[p-k] if p-k valid, for the (1-h[i+k]) term (when i+k = p, i.e., i = p-k)
        lags_pos = np.arange(0, n - p)  # p + k < n, so k in [0, n-1-p]
        # corr at lag k (k >= 0): h[p] appears as h[i] with i=p, so k goes up to n-1-p
        # In full correlation, lag k (k>=0) corresponds to index mid+k
        M[p, mid + lags_pos] += (1.0 - h[p + lags_pos])
        
        lags_neg = np.arange(0, p + 1)  # p - k >= 0, so k in [0, p]
        # h[p] appears as h[i+k] with i+k=p, i=p-k, so k goes from 0 to p
        # In full correlation, lag -k (k>=0) corresponds to index mid-k
        M[p, mid - lags_neg] -= h[p - lags_neg]
    return M

def compute_gradient(h, n, weights):
    """Compute grad = M^T @ weights efficiently using the structure."""
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    
    # Instead of building M, compute directly
    # grad[i] = sum_k weights[k] * (1-h[i+k])_valid - sum_k weights[k] * h[i-k]_valid
    
    # More efficient: for each lag k with non-zero weight, add contribution
    active = np.abs(weights) > 1e-15
    active_k = np.where(active)[0]
    
    for k_idx in active_k:
        k = int(k_idx) - mid  # actual lag
        w = weights[k_idx]
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    
    return grad

Actually, one more idea: **Use the structure of the correlation to speed things up**.

The correlation at lag k is:
c_k = Σ_{i} h[i](1-h[i+k]) (for valid i)

This can be computed as:
c_k = Σ_i h[i] - Σ_i h[i]h[i+k] (over valid range)

Let H_k = Σ_i h[i]h[i+k] (autocorrelation at lag k)
Let S_k = Σ_i h[i] (over the valid range for lag k)

Then c_k = S_k - H_k.

Now, H_k is the autocorrelation, which can be computed via FFT in O(n log n) for all k simultaneously!

S_k is just a partial sum, which can be computed in O(n) using a prefix sum.

So the full correlation vector can be computed in O(n log n) instead of O(n²) (which is what np.correlate does).

Wait, np.correlate already does this efficiently (using FFT internally for large arrays). So this isn't a new optimization.

∂H_k/∂h[i] = h[i+k] + h[i-k] (symmetric)
So the gradient of H at lag k is the "dilated" version of h.

        n_cand = min(n_cand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        if n_h * n_l <= max_pairs:
            pairs = []
            for ii in range(n_h):
                for jj in range(n_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pairs.append((iv, jv))
        else:
            pairs = []
            for frac in np.linspace(0.0, 1.0, min(max_pairs, 50)):
                hi = int(frac * (n_h - 1)) if n_h > 1 else 0
                lo = int(frac * (n_l - 1)) if n_l > 1 else 0
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv and (iv, jv) not in pairs:
                    pairs.append((iv, jv))
            np.random.shuffle(pairs)
            pairs = pairs[:max_pairs]

        for i, j in pairs:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue
            g = change[j] - change[i]
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx = np.max(corr + delta_star * g) * dx
            if approx < best_trial - 1e-15:
                best_trial = approx
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5


def lbfgsb_round(h0, n, target, alpha, time_limit, maxiter=2000):
    start = time.time()
    n_corr = 2 * n - 1

    def obj_func(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_val(corr, alpha)

    def grad_func(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        w = smooth_weights(corr, alpha)
        return compute_grad_weighted(x, n, w)

    bounds = [(0.0, 1.0)] * n
    cons_sum = np.sum(h0) - target

    try:
        result = fmin_l_bfgs_b(obj_func, h0.copy(), fprime=grad_func,
                                bounds=bounds, maxiter=maxiter,
                                factr=1e12, pgtol=1e-12,
                                callback=None)
    except Exception:
        return h0, compute_c5(h0)

    h_new = project_feasible(result[0], target)
    sc = compute_c5(h_new)
    return h_new, sc


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick FISTA at initial resolution
    if tleft() > 60:
        for alpha in [100, 500, 2000]:
            if tleft() < 30:
                break
            tl = min(40, tleft() * 0.3)
            h_f, sc_f = fista_smooth(best_h, n0, target0, alpha,
                                       max_iter=5000, time_limit=tl)
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_f

    # Phase 2: Coordinate descent at initial resolution
    if tleft() > 80:
        tl = min(120, tleft() * 0.4)
        h_cd, sc_cd = enhanced_cd(best_h, n0, target0, time_limit=tl,
                                   max_sweeps=20000, n_active=4,
                                   n_cand_start=20, n_cand_end=50,
                                   max_no_improve=400, bisection_iters=14)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 3: Scale to n=800
    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        n_cur = n1
        target_cur = target1

        # FISTA at higher resolution
        if tleft() > 200:
            for alpha in [200, 1000, 5000]:
                if tleft() < 60:
                    break
                tl = min(80, tleft() * 0.3)
                h_f, sc_f = fista_smooth(h1, n_cur, target_cur, alpha,
                                           max_iter=10000, time_limit=tl)
                if sc_f < best_score:
                    best_score = sc_f
                    best_h = h_f
                    h1 = h_f

        # CD at higher resolution
        if tleft() > 150:
            tl = min(200, tleft() * 0.45)
            h_cd, sc_cd = enhanced_cd(best_h, n_cur, target_cur, time_limit=tl,
                                       max_sweeps=30000, n_active=4,
                                       n_cand_start=25, n_cand_end=65,
                                       max_no_improve=600, bisection_iters=14)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd

        # L-BFGS-B
        if tleft() > 120:
            tl = min(100, tleft() * 0.35)
            h_lb, sc_lb = lbfgsb_round(best_h, n_cur, target_cur,
                                        alpha=5000, time_limit=tl, maxiter=3000)
            if sc_lb < best_score:
                best_score = sc_lb
                best_h = h_lb

    # Phase 4: Further CD refinement
    if tleft() > 120:
        tl = min(150, tleft() * 0.5)
        h_cd, sc_cd = enhanced_cd(best_h, n_cur, target_cur, time_limit=tl,
                                   max_sweeps=40000, n_active=5,
                                   n_cand_start=30, n_cand_end=80,
                                   max_no_improve=500, bisection_iters=14)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 5: Multiple restarts
    if tleft() > 200:
        scales = [0.02, 0.05, 0.10, 0.15]
        for scale in scales:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(n_cur) * scale
            h_pert = project_feasible(h_pert, target_cur)
            tl1 = min(40, tleft() * 0.2)
            h_f, sc_f = fista_smooth(h_pert, n_cur, target_cur, 2000,
                                       max_iter=5000, time_limit=tl1)
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_f
            tl2 = min(60, tleft() * 0.25)
            h_cd, sc_cd = enhanced_cd(best_h, n_cur, target_cur, time_limit=tl2,
                                       max_sweeps=15000, n_active=4,
                                       n_cand_start=25, n_cand_end=60,
                                       max_no_improve=300, bisection_iters=12)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd

    # Phase 6: Try symmetric constraint approach
    if tleft() > 150:
        m = n_cur // 2
        g_init = best_h[:m].copy()
        # Build symmetric h from g
        def g_to_h(g):
            h_sym = np.zeros(n_cur)
            h_sym[:m] = g
            h_sym[m:] = 1.0 - g
            return h_sym

        g_cur = g_init.copy()
        # Optimize g using FISTA (no sum constraint needed)
        tl_sym = min(80, tleft() * 0.3)
        h_sym_start = g_to_h(g_cur)
        h_sym_opt, sc_sym = fista_smooth(h_sym_start, n_cur, target_cur, 3000,
                                           max_iter=8000, time_limit=tl_sym)
        # Extract g from symmetric h
        g_opt = h_sym_opt[:m].copy()
        h_sym_final = g_to_h(g_opt)
        h_sym_final = project_feasible(h_sym_final, target_cur)
        sc_sym2 = compute_c5(h_sym_final)
        if sc_sym2 < best_score:
            best_score = sc_sym2
            best_h = h_sym_final

    # Phase 7: Final polish with high-alpha FISTA + CD
    if tleft() > 80:
        for alpha in [10000, 50000]:
            if tleft() < 30:
                break
            tl = min(50, tleft() * 0.35)
            h_f, sc_f = fista_smooth(best_h, n_cur, target_cur, alpha,
                                       max_iter=15000, time_limit=tl)
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_f

    if tleft() > 50:
        tl = min(40, tleft() * 0.6)
        h_cd, sc_cd = enhanced_cd(best_h, n_cur, target_cur, time_limit=tl,
                                   max_sweeps=20000, n_active=6,
                                   n_cand_start=40, n_cand_end=100,
                                   max_no_improve=300, bisection_iters=14)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 30:
        tl = min(25, tleft() * 0.7)
        h_lb, sc_lb = lbfgsb_round(best_h, n_cur, target_cur,
                                    alpha=50000, time_limit=tl, maxiter=1000)
        if sc_lb < best_score:
            best_score = sc_lb
            best_h = h_lb

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
