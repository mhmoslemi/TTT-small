def subgradient_multi_v2(h_init, n, target, time_limit, n_steps=30000, 
                         n_weights=10, step_scale=0.005, momentum=0.5):

    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale
    vel = np.zeros(n)

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        # Soft-max weights
        log_c = corr - np.max(corr)
        p = np.exp(log_c * 8.0)
        p /= p.sum()

        # Weighted gradient from top shifts
        top_k = np.argsort(p)[-min(n_weights, n_corr):]
        grad = np.zeros(n)
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            if w < 1e-10:
                continue
            pk = idxs - k
            pk2 = idxs + k
            mask1 = (pk >= 0) & (pk < n)
            mask2 = (pk2 >= 0) & (pk2 < n)
            contrib = np.zeros(n)
            contrib[mask1] += (1.0 - h[pk[mask1]])
            contrib[mask2] -= h[pk2[mask2]]
            grad += w * contrib
        grad -= np.mean(grad)

        # Momentum update
        vel = momentum * vel - eta * grad
        h_try = h + vel
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                vel = np.zeros(n)
                h = h_try2
                eta *= 0.95
            else:
                vel = np.zeros(n)
                eta *= 0.9

    return best_h, best_c5

    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)

    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)

            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        threshold = n_active_frac * float(np.max(corr))
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        wts = corr[active]
        wsum = wts.sum()
        if wsum < 1e-15:
            wts = np.ones(len(active)) / len(active)
        else:
            wts = wts / wsum

        grad = np.zeros(n)
        for ai in range(len(active)):
            grad += wts[ai] * change[:, active[ai]]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        total_pairs = n_h * n_l

        if total_pairs > max_pairs_per_sweep:
            rng = np.random
            n_h_sub = int(np.sqrt(max_pairs_per_sweep))
            n_l_sub = max(1, max_pairs_per_sweep // max(n_h_sub, 1))
            h_idx = rng.choice(n_h, min(n_h, n_h_sub), replace=False)
            l_idx = rng.choice(n_l, min(n_l, n_l_sub), replace=False)
            for ii in h_idx:
                for jj in l_idx:
                    i = int(high_set[ii])
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo < 1e-12:
                        continue
                    g = change[j] - change[i]
                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(13):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        am = vals >= f_mid - 1e-14
                        fp = np.max(g[am]) if am.any() else 0.0
                        if fp > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g) * dx
                    accept = False
                    if approx_max < best_trial_c5 - 1e-14:
                        accept = True
                    elif temp > 1e-10:
                        delta_obj = (approx_max - c5_val) * 1000.0
                        if np.random.random() < np.exp(-delta_obj / temp):
                            accept = True
                    if accept and approx_max < best_trial_c5 + 1e-10:
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
        else:
            for ii in range(n_h):
                i = int(high_set[ii])
                for jj in range(n_l):
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo < 1e-12:
                        continue
                    g = change[j] - change[i]
                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(13):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        am = vals >= f_mid - 1e-14
                        fp = np.max(g[am]) if am.any() else 0.0
                        if fp > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g) * dx
                    accept = False
                    if approx_max < best_trial_c5 - 1e-14:
                        accept = True
                    elif temp > 1e-10:
                        delta_obj = (approx_max - c5_val) * 1000.0
                        if np.random.random() < np.exp(-delta_obj / temp):
                            accept = True
                    if accept and approx_max < best_trial_c5 + 1e-10:
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, n_corr)
            update_change_row(change, best_j, h, n, n_corr)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 3:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.98, 1e-12)

    return best_h, best_c5


def subgradient_momentum(h_init, n, target, time_limit, n_steps=50000,
                         step_scale=0.006, momentum=0.4, decay_beta=8.0):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale
    vel = np.zeros(n)

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        log_c = corr - np.max(corr)
        p = np.exp(log_c * decay_beta)
        p /= p.sum()

        top_k = np.argsort(p)[-15:]
        grad = np.zeros(n)
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            if w < 1e-10:
                continue
            pk = idxs - k
            pk2 = idxs + k
            m1 = (pk >= 0) & (pk < n)
            m2 = (pk2 >= 0) & (pk2 < n)
            contrib = np.zeros(n)
            contrib[m1] += (1.0 - h[pk[m1]])
            contrib[m2] -= h[pk2[m2]]
            grad += w * contrib
        grad -= np.mean(grad)

        vel = momentum * vel - eta * grad
        h_try = h + vel
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                vel = np.zeros(n)
                h = h_try2
            else:
                vel = np.zeros(n)
                eta *= 0.92

        if step % 2000 == 1999 and step > 0:
            eta = max(step_scale * (1.0 - 0.3 * step / n_steps), step_scale * 0.1)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def perturbed_restart(h_best, n, target, rng, scale):
    h_pert = h_best + rng.randn(n) * scale
    h_pert = project_feasible(h_pert, target)
    return h_pert


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [30, 100, 400], 500, 1e-13)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            cur_n = n1
            cur_target = target1

        if tleft() > 150:
            h_cd, c5_cd = pair_cd_v2(best_h, cur_n, cur_target,
                                      time_limit=min(150, tleft() * 0.6),
                                      max_sweeps=50000,
                                      n_cand_start=40, n_cand_end=130,
                                      n_active_frac=0.92,
                                      max_no_improve=400,
                                      anneal_temp=0.004, anneal_decay=0.9997,
                                      max_pairs_per_sweep=350)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                cur_n = n1
                cur_target = target1

    if tleft() > 70:
        h_sg, c5_sg = subgradient_momentum(best_h, cur_n, cur_target,
                                            time_limit=min(55, tleft() * 0.5),
                                            n_steps=50000, step_scale=0.007,
                                            momentum=0.35, decay_beta=7.0)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 70:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [500, 2000, 5000], 800, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 120:
        h_cd2, c5_cd2 = pair_cd_v2(best_h, cur_n, cur_target,
                                    time_limit=min(100, tleft() * 0.65),
                                    max_sweeps=35000,
                                    n_cand_start=60, n_cand_end=150,
                                    n_active_frac=0.90,
                                    max_no_improve=350,
                                    anneal_temp=0.0, anneal_decay=1.0,
                                    max_pairs_per_sweep=450)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 220:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            cur_n = n2
            cur_target = target2
        if tleft() > 170:
            h_cd3, c5_cd3 = pair_cd_v2(best_h, cur_n, cur_target,
                                        time_limit=min(150, tleft() * 0.6),
                                        max_sweeps=25000,
                                        n_cand_start=30, n_cand_end=90,
                                        n_active_frac=0.93,
                                        max_no_improve=300,
                                        anneal_temp=0.0, anneal_decay=1.0,
                                        max_pairs_per_sweep=300)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
                cur_n = n2
                cur_target = target2

    if tleft() > 50:
        h_sg2, c5_sg2 = subgradient_momentum(best_h, cur_n, cur_target,
                                              time_limit=min(40, tleft() * 0.6),
                                              n_steps=30000, step_scale=0.005,
                                              momentum=0.3, decay_beta=9.0)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2

    if tleft() > 60:
        h_final, sc = slsqp_round(best_h, cur_n, cur_target,
                                   [5000, 15000, 30000], 800, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if tleft() > 80:
        for noise_scale in [0.015, 0.03, 0.06]:
            if tleft() < 40:
                break
            h_pert = perturbed_restart(best_h, cur_n, cur_target, rng, noise_scale)
            h_ref, sc = slsqp_round(h_pert, cur_n, cur_target,
                                     [200, 1000, 4000], 500, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 35:
        h_fin, c5_fin = pair_cd_v2(best_h, cur_n, cur_target,
                                    time_limit=min(25, tleft() * 0.7),
                                    max_sweeps=12000,
                                    n_cand_start=80, n_cand_end=200,
                                    n_active_frac=0.88,
                                    max_no_improve=200,
                                    anneal_temp=0.0, anneal_decay=1.0,
                                    max_pairs_per_sweep=500)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
