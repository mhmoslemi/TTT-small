import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_alpha(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj_alpha, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_grad_shift(h, k, n):
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    grad[v1] += (1.0 - h[idxs[v1] - k])
    grad[v2] -= h[idxs[v2] + k]
    return grad


def weighted_grad_from_shifts(h, n, weights):
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for s_idx in range(n_corr):
        w = weights[s_idx]
        if w < 1e-10:
            continue
        k = int(s_idx) - mid
        if k == 0:
            grad[:] = 0.0
            continue
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad -= np.mean(grad)
    return grad


def softmax_weights(corr, beta):
    c_max = np.max(corr)
    shifted = beta * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    if total < 1e-300:
        return np.full(len(corr), 1.0 / len(corr))
    return exps / total


def nesterov_subgrad(h_init, n, target, time_limit, beta_start=50.0, beta_end=200.0,
                     lr_start=0.003, momentum_start=0.8, momentum_end=0.95,
                     max_iters=50000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0

    for t in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        progress = min(float(t) / max(max_iters, 1), 1.0)
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start / (1.0 + 0.01 * t)
        mu = momentum_start + (momentum_end - momentum_start) * progress

        y = h + mu * (h - h_prev)
        y = project_feasible(y, target)

        corr = np.correlate(y, 1.0 - y, mode='full')
        weights = softmax_weights(corr, beta)

        top_k = min(10, n_corr)
        top_idx = np.argsort(weights)[-top_k:]
        active_weights = np.zeros(n_corr)
        active_weights[top_idx] = weights[top_idx]
        active_weights /= active_weights.sum()

        grad = np.zeros(n)
        for s_idx in top_idx:
            w = active_weights[s_idx]
            if w < 1e-10:
                continue
            k = int(s_idx) - mid
            if k == 0:
                continue
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - y[idxs[v1] - k])
            grad[v2] -= w * y[idxs[v2] + k]
        grad -= np.mean(grad)

        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)

        if score < best_score - 1e-15:
            best_score = score
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        h_prev = h
        h = h_new

    return best_h, best_score


def improved_pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
                     n_cand_start=20, n_cand_end=60, max_no_improve=250,
                     weight_beta=6.0, top_k_shifts=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(42)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)

        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        top_idx = np.argsort(weights)[-top_k_shifts:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, :] = 0.0
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5


def p_power_descent(h_init, n, target, time_limit, p=12, n_iter=10000, step_init=0.004):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    no_improve = 0

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_k = min(15, n_corr)
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = corr[s_idx] ** (p - 1)
            if w < 1e-20:
                continue
            k = int(s_idx) - mid
            if k == 0:
                continue
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            step *= 0.5
            continue
        grad /= grad_norm

        h_try = h - step * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val - 1e-15:
            h = h_try
            step = min(step * 1.02, 0.03)
        else:
            step *= 0.8

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def sinusoidal_perturb(h_base, n, target, frequency, amplitude):
    idxs = np.arange(n)
    h_pert = h_base + amplitude * np.sin(2.0 * np.pi * frequency * idxs / n)
    return project_feasible(h_pert, target)
