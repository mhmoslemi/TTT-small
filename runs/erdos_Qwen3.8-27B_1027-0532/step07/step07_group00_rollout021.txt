        else:
            step *= 0.9

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def sharpen_and_reopt(h, n, target, time_limit, sharp_power=20):
    h_sharp = h ** sharp_power
    h_sharp = project_feasible(h_sharp, target)
    return h_sharp


def structured_restart(h, n, target, rng, mode):
    if mode == 0:
        return project_feasible(h[::-1].copy(), target)
    elif mode == 1:
        h_pert = h + rng.randn(n) * 0.02
        return project_feasible(h_pert, target)
    elif mode == 2:
        h_pert = h.copy()
        blk = rng.randint(5, 20)
        i0 = rng.randint(0, n - blk)
        j0 = rng.randint(0, n - blk)
        if abs(i0 - j0) < blk:
            j0 = (j0 + n // 2) % (n - blk)
        h_pert[i0:i0 + blk], h_pert[j0:j0 + blk] = h[j0:j0 + blk].copy(), h[i0:i0 + blk].copy()
        return project_feasible(h_pert, target)
    elif mode == 3:
        freq = rng.randint(2, 10)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 0.01 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        return project_feasible(h + mod, target)
    else:
        return project_feasible(h + rng.randn(n) * 0.05, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP polish at initial resolution
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [200, 1000, 5000, 20000],
                                600, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Power-sum SLSQP (different landscape)
    if tleft() > 80:
        h_pp, sc_pp = slsqp_power_round(best_h, n0, target0,
                                        [30, 60, 120],
                                        400, 1e-14)
        if sc_pp < best_score:
            best_score = sc_pp
            best_h = h_pp

    # Phase 3: Improved CD (unweighted, focused on active set)
    if tleft() > 150:
        h_cd1, c5_1 = improved_cd(best_h, n0, target0,
                                  time_limit=min(120, tleft() * 0.5),
                                  max_sweeps=50000, n_top=8,
                                  no_improve_patience=300,
                                  use_weighted=False, block_swap_prob=0.05)
        if c5_1 < best_score:
            best_score = c5_1
            best_h = h_cd1

    # Phase 4: Weighted CD (different weighting)
    if tleft() > 120:
        h_cd2, c5_2 = improved_cd(best_h, n0, target0,
                                  time_limit=min(100, tleft() * 0.5),
                                  max_sweeps=50000, n_top=10,
                                  no_improve_patience=250,
                                  use_weighted=True, weight_beta=7.0,
                                  block_swap_prob=0.03)
        if c5_2 < best_score:
            best_score = c5_2
            best_h = h_cd2

    # Phase 5: Subgradient descent
    if tleft() > 80:
        h_sg, c5_sg = subgradient_descent(best_h, n0, target0,
                                          time_limit=min(60, tleft() * 0.5),
                                          n_iter=10000, step_init=0.008)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: Sharpen and re-optimize
    if tleft() > 100:
        h_sharp = sharpen_and_reopt(best_h, n0, target0, tleft(), sharp_power=15)
        h_cd3, c5_3 = improved_cd(h_sharp, n0, target0,
                                  time_limit=min(60, tleft() * 0.5),
                                  max_sweeps=30000, n_top=8,
                                  no_improve_patience=200)
        if c5_3 < best_score:
            best_score = c5_3
            best_h = h_cd3
        if tleft() > 30:
            h_pol_sh, sc_sh = slsqp_round(best_h, n0, target0,
                                          [5000, 20000], 400, 1e-14)
            if sc_sh < best_score:
                best_score = sc_sh
                best_h = h_pol_sh

    # Phase 7: Higher resolution n=1000
    n1 = 1000
    target1 = n1 / 2.0
    if tleft() > 300:
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 250:
            h_pol1, sc1 = slsqp_round(h1, n1, target1,
                                      [100, 500, 2000, 8000],
                                      500, 1e-14)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1
            else:
                best_h = h_pol1

            if tleft() > 180:
                h_cd4, c5_4 = improved_cd(best_h, n1, target1,
                                          time_limit=min(150, tleft() * 0.5),
                                          max_sweeps=40000, n_top=8,
                                          no_improve_patience=250,
                                          use_weighted=True, weight_beta=6.0)
                if c5_4 < best_score:
                    best_score = c5_4
                    best_h = h_cd4

            if tleft() > 60:
                h_cd5, c5_5 = improved_cd(best_h, n1, target1,
                                          time_limit=min(50, tleft() * 0.6),
                                          max_sweeps=30000, n_top=10,
                                          no_improve_patience=200,
                                          use_weighted=False)
                if c5_5 < best_score:
                    best_score = c5_5
                    best_h = h_cd5

            if tleft() > 40:
                h_pol1b, sc1b = slsqp_round(best_h, n1, target1,
                                            [10000, 50000], 500, 1e-14)
                if sc1b < best_score:
                    best_score = sc1b
                    best_h = h_pol1b

    # Phase 8: Multi-restart with structured perturbations
    if tleft() > 120:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)
        for mode in range(5):
            if tleft() < 50:
                break
            h_restart = structured_restart(best_h, n_cur, target_cur, rng, mode)
            h_r, sc_r = improved_cd(h_restart, n_cur, target_cur,
                                    time_limit=min(30, tleft() * 0.4),
                                    max_sweeps=20000, n_top=8,
                                    no_improve_patience=150)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r
            if tleft() > 30:
                h_r2, sc_r2 = slsqp_round(h_r, n_cur, target_cur,
                                          [2000, 10000], 300, 1e-14)
                if sc_r2 < best_score:
                    best_score = sc_r2
                    best_h = h_r2

    # Phase 9: Try even higher resolution if time allows
    if tleft() > 250:
        n_cur = len(best_h)
        if n_cur < 1200:
            n2 = 1200
            target2 = n2 / 2.0
            h2 = interpolate_to(best_h, n2, target2)
            if tleft() > 200:
                h_pol2, sc2 = slsqp_round(h2, n2, target2,
                                          [100, 500, 2000],
                                          400, 1e-14)
                if tleft() > 120:
                    h_cd6, c5_6 = improved_cd(h_pol2, n2, target2,
                                              time_limit=min(100, tleft() * 0.5),
                                              max_sweeps=30000, n_top=8,
                                              no_improve_patience=200,
                                              use_weighted=True, weight_beta=6.0)
                    if c5_6 < best_score:
                        best_score = c5_6
                        best_h = h_cd6
                    if tleft() > 50:
                        h_pol3, sc3 = slsqp_round(best_h, n2, target2,
                                                  [5000, 20000], 400, 1e-14)
                        if sc3 < best_score:
                            best_score = sc3
                            best_h = h_pol3

    # Phase 10: Final intensive CD polish
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin_cd, c5_fin = improved_cd(best_h, n_cur, target_cur,
                                       time_limit=min(50, tleft() * 0.6),
                                       max_sweeps=50000, n_top=12,
                                       no_improve_patience=200,
                                       use_weighted=True, weight_beta=8.0,
                                       block_swap_prob=0.05)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin_cd

    # Phase 11: Final SLSQP
    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin, sc_fin = slsqp_round(best_h, n_cur, target_cur,
                                    [20000, 50000, 100000],
                                    500, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
