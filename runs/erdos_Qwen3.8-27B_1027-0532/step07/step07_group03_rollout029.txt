                continue

            idxs = np.arange(n)
            g_i = np.zeros(2 * n - 1)
            g_j = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                k = m - mid
                v1i = (i + k >= 0) & (i + k < n)
                v2i = (i - k >= 0) & (i - k < n)
                v1j = (j + k >= 0) & (j + k < n)
                v2j = (j - k >= 0) & (j - k < n)
                if v1i:
                    g_i[m] += (1.0 - h[i + k])
                if v2i:
                    g_i[m] -= h[i - k]
                if v1j:
                    g_j[m] += (1.0 - h[j + k])
                if v2j:
                    g_j[m] -= h[j - k]

            g_diff = g_j - g_i
            a_d = 0.0
            b_d = delta_max
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g_diff
                f_mid = np.max(vals)
                active_m = vals >= f_mid - 1e-14
                if active_m.any():
                    f_prime = np.max(g_diff[active_m])
                else:
                    f_prime = 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if delta_star > 1e-14:
                approx_max = np.max(corr + delta_star * g_diff) * dx
                cur_max = np.max(corr) * dx
                if approx_max < best_score - 1e-14:
                    h[i] -= delta_star
                    h[j] += delta_star
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    corr = compute_corr(h)
                    improved = True
                    if float(np.max(corr)) * dx < best_score - 1e-14:
                        best_score = float(np.max(corr)) * dx
                        best_h = h.copy()

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_active_subgrad(h_init, n, target, time_limit, max_iters=20000,
                         n_active_max=20, base_lr=0.02):
    start = time.time()
    mid = n - 1
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    avg_h = h.copy()
    best_avg_score = compute_c5(avg_h)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        lr = base_lr / np.sqrt(it + 1)
        corr = compute_corr(h)
        c_max = np.max(corr)
        sorted_corr = np.sort(corr)[::-1]
        n_act = min(n_active_max, int(np.sum(corr >= sorted_corr[min(n_active_max, len(corr)-1)])))
        n_act = max(n_act, 1)
        threshold_idx = np.argsort(corr)[::-1][:n_act]

        grad = np.zeros(n)
        for m in threshold_idx:
            w = 1.0 / n_act
            k = int(m) - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] + k])
            grad[v2] -= w * h[idxs[v2] - k]]

        h = h - lr * grad
        h = project_feasible(h, target)
        sc = compute_c5(h)

        avg_h = (it * avg_h + h) / (it + 1)
        avg_sc = compute_c5(avg_h)
        if avg_sc < best_avg_score - 1e-14:
            best_avg_score = avg_sc
            best_h = avg_h.copy()
        if sc < best_avg_score - 1e-14:
            best_avg_score = sc
            best_h = h.copy()
        if sc < best_score - 1e-14 or avg_sc < best_score - 1e-14:
            best_score = min(sc, avg_sc)
            best_h = h.copy() if sc < avg_sc else avg_h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0

    # Phase 1: Quick SLSQP polish
    if tleft() > 180:
        h_sls, sc = slsqp_round(best_h, n0, target0,
                                [50, 200, 1000, 5000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_sls

    # Phase 2: Nesterov smoothed gradient (main new method)
    if tleft() > 250:
        h_nes, sc_nes = nesterov_smoothed(best_h, n0, target0,
                                          time_limit=min(200, tleft() * 0.5),
                                          alpha_start=50.0, alpha_end=30000.0,
                                          max_iters=40000, momentum=0.82)
        if sc_nes < best_score:
            best_score = sc_nes
            best_h = h_nes

    # Phase 3: Multi-active subgradient with averaging
    if tleft() > 200:
        h_ms, sc_ms = multi_active_subgrad(best_h, n0, target0,
                                           time_limit=min(150, tleft() * 0.4),
                                           max_iters=20000,
                                           n_active_max=15, base_lr=0.03)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms

    # Phase 4: Cutting plane
    if tleft() > 200:
        h_cp, sc_cp = cutting_plane_method(best_h, n0, target0,
                                           time_limit=min(200, tleft() * 0.5),
                                           max_iters=10000, tol=1e-9)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp

    # Phase 5: Interceptive CD
    if tleft() > 180:
        h_cd, sc_cd = interceptive_cd(best_h, n0, target0,
                                      time_limit=min(150, tleft() * 0.5),
                                      max_sweeps=25000,
                                      n_cand_start=15, n_cand_end=55,
                                      max_no_improve=400)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 6: Another Nesterov pass with different params
    if tleft() > 150:
        h_nes2, sc_nes2 = nesterov_smoothed(best_h, n0, target0,
                                            time_limit=min(120, tleft() * 0.5),
                                            alpha_start=200.0, alpha_end=50000.0,
                                            max_iters=30000, momentum=0.88)
        if sc_nes2 < best_score:
            best_score = sc_nes2
            best_h = h_nes2

    # Phase 7: Cutting plane again
    if tleft() > 130:
        h_cp2, sc_cp2 = cutting_plane_method(best_h, n0, target0,
                                             time_limit=min(120, tleft() * 0.5),
                                             max_iters=8000, tol=1e-10)
        if sc_cp2 < best_score:
            best_score = sc_cp2
            best_h = h_cp2

    # Phase 8: Interceptive CD with finer settings
    if tleft() > 120:
        h_cd2, sc_cd2 = interceptive_cd(best_h, n0, target0,
                                        time_limit=min(100, tleft() * 0.5),
                                        max_sweeps=20000,
                                        n_cand_start=25, n_cand_end=70,
                                        max_no_improve=300)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 9: Multi-resolution - try higher n
    if tleft() > 200:
        for n_try in [500, 600]:
            if tleft() < 180:
                break
            target_try = n_try / 2.0
            h_try = interpolate_to(best_h, n_try, target_try)
            sc_try = compute_c5(h_try)
            if sc_try < best_score:
                best_score = sc_try
                best_h = h_try
                best_n = n_try
            if tleft() > 150:
                h_nes3, sc_n3 = nesterov_smoothed(h_try, n_try, target_try,
                                                  time_limit=min(120, tleft() * 0.4),
                                                  alpha_start=30.0, alpha_end=20000.0,
                                                  max_iters=25000, momentum=0.85)
                if sc_n3 < best_score:
                    best_score = sc_n3
                    best_h = h_nes3
                    best_n = n_try
            if tleft() > 80:
                h_cd3, sc_cd3 = interceptive_cd(h_try, n_try, target_try,
                                                time_limit=min(80, tleft() * 0.4),
                                                max_sweeps=15000,
                                                n_cand_start=10, n_cand_end=40,
                                                max_no_improve=300)
                if sc_cd3 < best_score:
                    best_score = sc_cd3
                    best_h = h_cd3
                    best_n = n_try

    # Phase 10: SLSQP polish at current best n
    if tleft() > 100:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_sls2, sc_sls2 = slsqp_round(best_h, n_cur, tgt_cur,
                                      [2000, 8000, 30000], 600, 1e-14)
        if sc_sls2 < best_score:
            best_score = sc_sls2
            best_h = h_sls2

    # Phase 11: Final cutting plane
    if tleft() > 60:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_cp3, sc_cp3 = cutting_plane_method(best_h, n_cur, tgt_cur,
                                             time_limit=min(50, tleft() * 0.5),
                                             max_iters=5000, tol=1e-10)
        if sc_cp3 < best_score:
            best_score = sc_cp3
            best_h = h_cp3

    # Phase 12: Perturbation + local search
    if tleft() > 80:
        rng = np.random.RandomState(seed)
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        for ns in [0.03, 0.08, 0.01]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n_cur) * ns
            h_pert = project_feasible(h_pert, tgt_cur)
            h_ref, sc_ref = nesterov_smoothed(h_pert, n_cur, tgt_cur,
                                              time_limit=min(40, tleft() * 0.3),
                                              alpha_start=100.0, alpha_end=20000.0,
                                              max_iters=15000, momentum=0.85)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 13: Final interceptive CD
    if tleft() > 40:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_cd4, sc_cd4 = interceptive_cd(best_h, n_cur, tgt_cur,
                                        time_limit=min(35, tleft() * 0.5),
                                        max_sweeps=10000,
                                        n_cand_start=30, n_cand_end=80,
                                        max_no_improve=200)
        if sc_cd4 < best_score:
            best_score = sc_cd4
            best_h = h_cd4

    # Final SLSQP
    if tleft() > 25:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_final_sls, sc_final = slsqp_round(best_h, n_cur, tgt_cur,
                                            [5000, 20000], 400, 1e-14)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_sls

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
