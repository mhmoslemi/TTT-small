import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v1 - k]] if False else h[idxs[v2] - k]
    return grad


def compute_corr_gradient_multi(h, k_stars, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(k_stars, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def cutting_plane_method(h_init, n, target, time_limit, max_iters=10000, tol=1e-10):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_f = []
    cuts_h = []

    max_cuts = 200
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:

            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            if len(cuts_g) > 15:
                cuts_g = cuts_g[-15:]
                cuts_f = cuts_f[-15:]
                cuts_h = cuts_h[-15:]
            no_improve = 0

        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])

        # Use top M active lags with adaptive threshold
        active_mask = corr >= corr[k_star] - 1e-6
        n_active = int(active_mask.sum())
        
        if n_active == 1:
            g = compute_corr_gradient(h, k_star)
        else:
            active_lags = np.where(active_mask)[0]
            # Weight by correlation value (higher correlation gets more weight)

            w = w / w.sum()
            g = compute_corr_gradient_multi(h, active_lags, w)

        cuts_g.append(g)
        cuts_f.append(f_val)
        cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            # Keep the first cut and the most recent ones
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break

    return best_h, best_score


def interceptive_cd(h_init, n, target, time_limit, max_sweeps=30000,
                    n_cand_start=20, n_cand_end=80, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        n_h = min(int(n_cand_start + (n_cand_end - n_cand_start) * min(float(sweep) / max_sweeps, 1.0)), n // 2)

        # Compute gradient to select candidates
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_h]
        high_set = sorted_idx[-n_h:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_high = len(high_set)
        n_low = len(low_set)
        for ii in range(n_high):
            i = int(high_set[ii])
            for jj in range(n_low):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i, :] = 0.0
            change[best_j, :] = 0.0
            pi = best_i
            pj = best_j
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def active_set_refinement(h_init, n, target, time_limit, max_iters=500):
    """Refine by trying to equalize top correlations."""
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()

        # Find top correlations
        sorted_corr = np.sort(corr)[::-1]
        # Determine active set: correlations within 1e-4 of max
        thresh = sorted_corr[0] - 1e-4 * sorted_corr[0]
        active_lags = np.where(corr >= thresh)[0]
        
        if len(active_lags) < 2:
            break

        # Try to reduce the spread among active correlations
        # Compute gradient that reduces the maximum
        g = compute_corr_gradient_multi(h, active_lags, np.ones(len(active_lags)) / len(active_lags))
        g -= np.mean(g)  # Keep sum invariant

        # Line search
        best_step = 0.0
        best_val = c5_val
        for step in [0.01, 0.005, 0.001, 0.0005, 0.0001, -0.001, -0.005]:
            h_trial = project_feasible(h + step * g, target)
            sc = compute_c5(h_trial)
            if sc < best_val:
                best_val = sc
                best_step = step

        if best_step == 0.0:
            break

        h = project_feasible(h + best_step * g, target)

    return best_h, best_score

    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(k_stars, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def cutting_plane_method(h_init, n, target, time_limit, max_iters=10000, tol=1e-10):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_f = []
    cuts_h = []

    max_cuts = 250
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 1500:
            if len(cuts_g) > 12:
                cuts_g = cuts_g[-12:]
                cuts_f = cuts_f[-12:]
                cuts_h = cuts_h[-12:]
            no_improve = 0

        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])

        active_mask = corr >= corr[k_star] - 5e-7
        n_active = int(active_mask.sum())

        if n_active <= 1:
            g = compute_corr_gradient(h, k_star)
        else:
            active_lags = np.where(active_mask)[0]
            w = np.ones(len(active_lags)) / len(active_lags)
            g = compute_corr_gradient_multi(h, active_lags, w)

        cuts_g.append(g)
        cuts_f.append(f_val)
        cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break

    return best_h, best_score


def interceptive_cd(h_init, n, target, time_limit, max_sweeps=30000,
                    n_cand_start=20, n_cand_end=80, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_h = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_h = min(n_h, n // 2)

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_h]
        high_set = sorted_idx[-n_h:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_high = len(high_set)
        n_low = len(low_set)
        for ii in range(n_high):
            i = int(high_set[ii])
            for jj in range(n_low):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i, :] = 0.0
            change[best_j, :] = 0.0
            pi = best_i
            pj = best_j
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def equalize_top_corrs(h_init, n, target, time_limit, max_iters=300, n_top=5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()

        sorted_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[sorted_idx]
        spread = top_vals[0] - top_vals[-1]

        if spread < 1e-12:
            break

        w = np.ones(len(sorted_idx))
        g = compute_corr_gradient_multi(h, sorted_idx, w / len(sorted_idx))
        g -= np.mean(g)

        best_step = 0.0
        best_sc = c5_val
        for step in [0.005, 0.002, 0.001, 0.0005, 0.0001, -0.001, -0.002, -0.0005]:
            h_trial = project_feasible(h + step * g, target)
            sc = compute_c5(h_trial)
            if sc < best_sc:
                best_sc = sc
                best_step = step

        if best_step == 0.0:
            break

        h = project_feasible(h + best_step * g, target)

    return best_h, best_score


def gradient_descent_avoid(h_init, n, target, time_limit, max_iters=50000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    idxs = np.arange(n)

    step_size = 0.02
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if it % 500 == 0:
            step_size *= 0.95

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()

        k_star = int(np.argmax(corr))
        g = compute_corr_gradient(h, k_star)
        g -= np.mean(g)

        h_new = project_feasible(h - step_size * g, target)
        sc_new = compute_c5(h_new)

        if sc_new < c5_val + 1e-12:
            h = h_new
        else:
            h_new = project_feasible(h - 0.5 * step_size * g, target)
            sc_new2 = compute_c5(h_new)
            if sc_new2 < c5_val + 1e-12:
                h = h_new
            else:
                step_size *= 0.8

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP at n0
    if tleft() > 80:
        h_sls, sc = slsqp_round(best_h, n0, target0, [200, 1000, 5000], 500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_sls

    # Phase 2: Cutting plane at n0
    if tleft() > 180:
        alloc = min(150, tleft() * 0.4)
        h_cp, sc_cp = cutting_plane_method(best_h, n0, target0,
                                           time_limit=alloc,
                                           max_iters=10000, tol=1e-11)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp

    # Phase 3: Interceptive CD at n0
    if tleft() > 150:
        alloc = min(130, tleft() * 0.4)
        h_cd, sc_cd = interceptive_cd(best_h, n0, target0,
                                      time_limit=alloc,
                                      max_sweeps=25000,
                                      n_cand_start=15, n_cand_end=60,
                                      max_no_improve=300)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 4: Equalize top correlations
    if tleft() > 50:
        h_eq, sc_eq = equalize_top_corrs(best_h, n0, target0,
                                         time_limit=min(40, tleft() * 0.4),
                                         max_iters=200, n_top=4)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq

    # Phase 5: Try n=800
    n1 = 800
    target1 = n1 / 2.0
    best_h_800 = None
    best_score_800 = None

    if tleft() > 350:
        h800 = interpolate_to(best_h, n1, target1)
        sc800 = compute_c5(h800)
        best_h_800 = h800.copy()
        best_score_800 = sc800

        if tleft() > 150:
            h800_cp, sc800_cp = cutting_plane_method(h800, n1, target1,
                                                      time_limit=min(150, tleft() * 0.35),
                                                      max_iters=8000, tol=1e-11)
            if sc800_cp < best_score_800:
                best_score_800 = sc800_cp
                best_h_800 = h800_cp

        if tleft() > 150:
            h800_cd, sc800_cd = interceptive_cd(best_h_800, n1, target1,
                                                 time_limit=min(150, tleft() * 0.35),
                                                 max_sweeps=20000,
                                                 n_cand_start=10, n_cand_end=45,
                                                 max_no_improve=250)
            if sc800_cd < best_score_800:
                best_score_800 = sc800_cd
                best_h_800 = h800_cd

        if tleft() > 80:
            h800_sls, sc800_s = slsqp_round(best_h_800, n1, target1,
                                             [3000, 10000], 500, 1e-14)
            if sc800_s < best_score_800:
                best_score_800 = sc800_s
                best_h_800 = h800_sls

        if best_score_800 < best_score:
            best_score = best_score_800
            best_h = best_h_800.copy()

    # Phase 6: Second cutting plane round on current best
    cur_n = len(best_h)
    cur_target = cur_n / 2.0
    if tleft() > 120:
        alloc = min(100, tleft() * 0.5)
        h_cp2, sc_cp2 = cutting_plane_method(best_h, cur_n, cur_target,
                                              time_limit=alloc,
                                              max_iters=8000, tol=1e-11)
        if sc_cp2 < best_score:
            best_score = sc_cp2
            best_h = h_cp2

    # Phase 7: Interceptive CD refinement
    if tleft() > 100:
        cur_n = len(best_h)
        cur_target = cur_n / 2.0
        alloc = min(90, tleft() * 0.5)
        h_cd2, sc_cd2 = interceptive_cd(best_h, cur_n, cur_target,
                                        time_limit=alloc,
                                        max_sweeps=20000,
                                        n_cand_start=20, n_cand_end=70,
                                        max_no_improve=300)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 8: SLSQP polish
    if tleft() > 60:
        cur_n = len(best_h)
        cur_target = cur_n / 2.0
        h_sls2, sc2 = slsqp_round(best_h, cur_n, cur_target,
                                   [5000, 20000], 600, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_sls2

    # Phase 9: Perturbation restarts
    if tleft() > 80:
        rng = np.random.RandomState(seed)
        cur_n = len(best_h)
        cur_target = cur_n / 2.0
        for pert_size in [0.015, 0.04, 0.07]:
            if tleft() < 35:
                break
            h_pert = best_h + rng.randn(cur_n) * pert_size
            h_pert = project_feasible(h_pert, cur_target)
            h_ref, sc_ref = cutting_plane_method(h_pert, cur_n, cur_target,
                                                  time_limit=min(35, tleft() * 0.4),
                                                  max_iters=4000, tol=1e-9)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 10: Gradient descent avoid as complementary local search
    if tleft() > 50:
        cur_n = len(best_h)
        cur_target = cur_n / 2.0
        h_gd, sc_gd = gradient_descent_avoid(best_h, cur_n, cur_target,
                                              time_limit=min(40, tleft() * 0.5),
                                              max_iters=30000)
        if sc_gd < best_score:
            best_score = sc_gd
            best_h = h_gd

    # Phase 11: Final CD
    if tleft() > 35:
        cur_n = len(best_h)
        cur_target = cur_n / 2.0
        h_cd3, sc3 = interceptive_cd(best_h, cur_n, cur_target,
                                      time_limit=min(28, tleft() * 0.6),
                                      max_sweeps=10000,
                                      n_cand_start=30, n_cand_end=80,
                                      max_no_improve=150)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_cd3

    # Phase 12: Final SLSQP
    if tleft() > 25:
        cur_n = len(best_h)
        cur_target = cur_n / 2.0
        h_fin, sc_fin = slsqp_round(best_h, cur_n, cur_target,
                                     [30000], 400, 1e-15)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
