import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

_g_alpha = 100.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    # corr[k] = sum_i h[i] * (1-h[i+k]) for appropriate k
    # This is equivalent to correlation of h with (1-h)
    # Using FFT: corr = ifft(fft(h, 2*n) * fft(1-h, 2*n))
    N = 2 * n
    H = fft(h, N)
    O = fft(1.0 - h, N)
    full_corr = np.real(ifft(H * np.conj(O)))
    # The 'full' mode correlation of a with b (length n) is:
    # full_corr[k] = sum_i a[i] * b[i+k] for k = -(n-1), ..., n-1
    # Actually np.correlate(h, 1-h, 'full') gives:
    # c[k] = sum_i h[i] * (1-h)[i+k] where the output has length 2n-1
    # Index 0 corresponds to shift -(n-1), index n-1 to shift 0, etc.
    # Let me just use np.correlate for correctness
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _time_left(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def smooth_max_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_val(corr, alpha)

def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _obj(x):
    return smooth_obj(x, _g_alpha)

def _grad(x):
    return smooth_obj_grad(x, _g_alpha)

def _cons(x):

def slsqp_optimize(h_start, target, alphas, maxiter=500, ftol=1e-14):

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=25, n_cand_end=70, top_k_eval=50,
            max_no_improve=800, n_shifts_grad=3,
            use_multishift_score=False, n_shifts_score=3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        k_star_idx = int(top_shifts[-1])

        # Compute gradient
        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] if use_multishift_score else 1.0
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, max(10, n // 4))
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        # Score pairs by how much they reduce the max shift
        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        # If multishift scoring, also consider other top shifts
        if use_multishift_score and n_shifts_score > 1:
            for s_idx in top_shifts[:-1]:
                ch_h = change[high_set, int(s_idx)]
                ch_l = change[low_set, int(s_idx)]
                score_mat += (ch_l[None, :] - ch_h[:, None])

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            # Binary search for optimal delta
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5

def block_cd(h_init, n, target, time_limit, block_size=8, max_sweeps=20000,
             max_no_improve=500):
    """Block coordinate descent: optimize small blocks of variables jointly."""
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)

        # Find the most "costly" positions for the active shift
        # grad for shift k_star: d/dh[p] corr[k_star]
        # corr[k_star] = sum_i h[i]*(1-h[i+k_star]) for valid i
        # = sum_i h[i] - sum_i h[i]*h[i+k_star]
        # d/dh[p] = (1 - h[p+k_star]) - h[p-k_star] (with appropriate bounds)
        
        grad_k = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad_k[v1] += (1.0 - h[idxs[v1] - k_star])
        grad_k[v2] -= h[idxs[v2] + k_star]
        grad_k -= np.mean(grad_k)

        # Select block: positions with most negative gradient (decreasing them helps)
        # and most positive gradient (increasing them helps)
        sorted_idx = np.argsort(grad_k)
        block_low = sorted_idx[:block_size]
        block_high = sorted_idx[-block_size:]
        block = np.unique(np.concatenate([block_low, block_high]))
        
        if len(block) < 2:
            continue

        # Optimize the block using scipy minimize
        n_block = len(block)
        block_vals = h[block].copy()
        target_block = block_vals.sum()  # Keep sum constant

        def block_obj(x):
            h_try = h.copy()
            h_try[block] = x
            c = np.correlate(h_try, 1.0 - h_try, mode='full')
            return np.max(c) * dx

        def block_obj_smooth(x):
            h_try = h.copy()
            h_try[block] = x
            c = np.correlate(h_try, 1.0 - h_try, mode='full')
            alpha = 500.0
            cm = np.max(c)
            s = np.clip(alpha * (c - cm), -500.0, 0.0)
            return cm + np.log(np.sum(np.exp(s))) / alpha

        def block_grad(x):
            h_try = h.copy()
            h_try[block] = x
            c = np.correlate(h_try, 1.0 - h_try, mode='full')
            alpha = 500.0
            cm = np.max(c)
            s = np.clip(alpha * (c - cm), -500.0, 0.0)
            e = np.exp(s)
            w = e / np.sum(e)
            g = np.zeros_like(h_try)
            for ki in range(len(c)):
                wi = w[ki]
                if wi < 1e-10:
                    continue
                k = ki - (n - 1)
                vi = (idxs - k >= 0) & (idxs - k < n)
                vj = (idxs + k >= 0) & (idxs + k < n)
                g[vi] += wi * (1.0 - h_try[vi - k])
                g[vj] -= wi * h_try[vj + k]
            return g[block]

        bounds = [(0.0, 1.0)] * n_block
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_block}]
        
        try:
            res = minimize(block_obj_smooth, block_vals, jac=block_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-12})
            h_new = h.copy()
            h_new[block] = project_feasible(res.x, target_block)
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < c5_val - 1e-14:
                h = h_new
        except Exception:
            pass

    return best_h, best_c5

def l_bfgs_optimize(h_start, target, alphas, maxiter=1000):
    """Use L-BFGS-B with penalty for the equality constraint."""
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        _g_alpha = alpha
        
        def pen_obj(x):
            base = smooth_obj(x, _g_alpha)
            pen = 1e4 * (np.sum(x) - target) ** 2
            return base + pen
        
        def pen_grad(x):
            base_g = smooth_obj_grad(x, _g_alpha)
            pen_g = 2e4 * (np.sum(x) - target)
            return base_g + pen_g
        
        try:
            res = minimize(pen_obj, h_cur, jac=pen_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fourier_perturb(h, n, rng, num_modes=8, scale=0.1):
    """Add Fourier perturbation to h."""
    x = np.linspace(0, 2, n, endpoint=False)
    perturb = np.zeros(n)
    for k in range(1, num_modes + 1):
        freq = k / 2.0  # frequency in cycles per unit
        a = rng.randn() * scale
        b = rng.randn() * scale
        perturb += a * np.sin(2 * np.pi * freq * x) + b * np.cos(2 * np.pi * freq * x)
    return project_feasible(h + perturb, n / 2.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):

        if abs(diff) < 1e-14:
            break

            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)

        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]

                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_move = (i, j, delta_star)

        if best_move:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, i, shifts, n)
            update_change_row(change, h, j, shifts, n)

    return best_h, best_c5
