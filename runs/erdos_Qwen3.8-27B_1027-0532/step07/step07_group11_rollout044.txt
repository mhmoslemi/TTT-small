Actually, h(x)(1-h(x+k)) ≥ 0 always since both factors are in [0,1]. And the sum (which is the correlation) is also ≥ 0. So we can use the power-sum directly.

But wait, np.correlate(h, 1-h) can have negative values? Let's see:
corr[k] = sum_i h[i] * (1-h[i+k])
Since 0 ≤ h[i] ≤ 1 and 0 ≤ (1-h[i+k]) ≤ 1, each term h[i]*(1-h[i+k]) ≥ 0.
So corr[k] ≥ 0 for all k. Good.

So the power-sum approximation works:
smooth_max_p = (sum corr_k^p)^(1/p)

For large p, this approaches the max.

The gradient of this with respect to h[j] is:
∂smooth_max_p/∂h[j] = (1/p) * (sum corr_k^p)^(1/p - 1) * sum_k p * corr_k^(p-1) * ∂corr_k/∂h[j]
= (sum corr_k^p)^(1/p - 1) * sum_k corr_k^(p-1) * ∂corr_k/∂h[j]

This is a weighted sum of the gradients of individual correlations, with weights proportional to corr_k^(p-1).

For large p, the weights concentrate on the maximum correlations, so this behaves like the subgradient of the max.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _smooth_ps(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max < 1e-15:
        return 0.0
    scaled = corr / c_max
    return float(c_max * (np.mean(scaled**p))**(1.0/p))


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_run(h_start, n, target, obj_func, param, maxiter, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(obj_func, h_start, args=(param,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_start.copy(), compute_c5(h_start)


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts, pos_mask_cache, neg_mask_cache):
    change[p, :] = 0.0
    pm = pos_mask_cache[p]
    nm = neg_mask_cache[p]
    change[p, pm] += (1.0 - h[p + shifts[pm]])
    change[p, nm] -= h[p - shifts[nm]]
    return change


def build_masks(n):
    shifts = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1
    pos_caches = []
    neg_caches = []
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        pos_caches.append(pm)
        neg_caches.append(nm)
    return shifts, pos_caches, neg_caches


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=80, n_shifts=3,
            max_no_improve=400, anneal_temp=0.01, anneal_decay=0.9995):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts, pos_caches, neg_caches = build_masks(n)

    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            if temp > 0.001:
                h = project_feasible(h + np.random.randn(n) * 0.02, target)
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
                consecutive_no_improve = 0
                temp *= 0.5
                continue
            else:
                break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        sw_sum = np.sum(corr[top_shifts])
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = corr[top_shifts] / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):

            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:

                if accept and approx_max < best_trial_c5 + 1e-10:

                    best_i = i
                    best_j = j
                    best_delta = delta_star

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

        else:
            temp = max(temp * 0.95, 1e-10)

def interpolate_to(h_src, n_dst, target_dst):

        if tleft() > 60:
            h_sg, c5_sg = subgradient_multi(best_h, cur_n, cur_target,

    target2 = n2 / 2.0
