from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:

                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

            corr = np.correlate(h, 1.0 - h, mode='full')
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5

                grad[v1] += weights[idx] * (1.0 - y[idxs[v1] - k_lag])
                grad[v2] -= weights[idx] * y[idxs[v2] + k_lag]
            
            # Subtract mean for sum constraint (Lagrangian relaxation)
            grad -= np.mean(grad)
            
            # Step size
            step = 2.0 / (alpha + k + 1.0)
            
            # Nesterov update
            new_momentum = (1 + 1.0 / (k + 1)) / (1 + 1.0 / k) if k > 1 else 1.0
            y_new = y - step * grad
            h_new = new_momentum * y_new - (new_momentum - 1) * h
            h_new = project_feasible(h_new, target)
            
            # Check improvement
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            h = h_new
            y = h_new
            
            # After some iterations, add penalty for sum constraint violation
            if np.abs(np.sum(h) - target) > 1e-6:
                h = project_feasible(h, target)
                y = h
    
    return best_h, best_c5

    corr = np.correlate(h, 1.0 - h, mode='full')

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        if subgrad_escape and consecutive_no_improve > 40 and consecutive_no_improve % 40 == 0:
            top_idx = np.argsort(corr)[-n_active:]
            w_sum = np.sum(corr[top_idx])
            if w_sum <= 0:
                w_sum = 1.0
            weights = corr[top_idx] / w_sum
            grad = np.zeros(n)
            for idx in range(n_active):
                k = int(top_idx[idx]) - mid
                w = weights[idx]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
            grad -= np.mean(grad)
            for step_mult in [0.3, 0.8, 1.5, 3.0]:
                step = step_mult * 0.003 / (1.0 + 0.003 * consecutive_no_improve)
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, target)
                corr_trial = np.correlate(h_trial, 1.0 - h_trial, mode='full')
                if np.max(corr_trial) * dx < c5_val:
                    h = h_trial
                    corr = corr_trial
                    change = compute_change_matrix(h, n)
                    break

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        nh = len(high_set)
        nl = len(low_set)

        pair_list = []
        if nh * nl <= n_pairs:
            for ii in range(nh):
                for jj in range(nl):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pair_list.append((iv, jv))
        else:
            for t in range(n_pairs):
                frac_h = t / max(n_pairs - 1, 1)
                frac_l = (t + 0.5) / max(n_pairs, 1)
                ih = int(frac_h * (nh - 1)) if nh > 1 else 0
                il = int(frac_l * (nl - 1)) if nl > 1 else 0
                iv = int(high_set[ih])
                jv = int(low_set[il])
                if iv != jv:
                    pair_list.append((iv, jv))
            for _ in range(15):
                iv = int(high_set[np.random.randint(nh)])
                jv = int(low_set[np.random.randint(nl)])
                if iv != jv:
                    pair_list.append((iv, jv))

        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g = change[j] - change[i]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5


def nesterov_subgrad(h_init, n, target, time_limit, n_active=4, max_iter=50000):
    start = time.time()
    mid = n - 1
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    y = h.copy()
    beta = 0.0
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for k in range(1, max_iter + 1):
        if time.time() - start > time_limit:
            break
        if k % 500 == 0 and k > 1:
            beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0

        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            lag = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - lag >= 0) & (idxs - lag < n)
            v2 = (idxs + lag >= 0) & (idxs + lag < n)
            grad[v1] += w * (1.0 - y[idxs[v1] - lag])
            grad[v2] -= w * y[idxs[v2] + lag]
        grad -= np.mean(grad)

        step = 0.002 / (1.0 + 0.001 * k)
        y_new = y - step * grad
        h_new = (1.0 + 1.0 / (k + 1.0)) * y_new - (1.0 / (k + 1.0)) * h
        h_new = project_feasible(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        h = h_new
        y = h_new

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 120:
        h_pol = slsqp_polish(best_h, n0, target0,
                             [100, 500, 2000, 8000], 400, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 180:
        h_cd, c5_cd = enhanced_cd(best_h, n0, target0,
                                  time_limit=min(120, tleft() * 0.4),
                                  max_sweeps=15000,
                                  n_active=4,
                                  n_cand_start=20, n_cand_end=45,
                                  max_no_improve=400, bisection_iters=13,
                                  n_pairs=500)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_ng, sc_ng = nesterov_subgrad(best_h, n0, target0,
                                       time_limit=min(50, tleft() * 0.3),
                                       n_active=4, max_iter=30000)
        if sc_ng < best_score:
            best_score = sc_ng
            best_h = h_ng

    if tleft() > 50:
        h_pol2 = slsqp_polish(best_h, n0, target0,
                              [8000, 30000, 100000], 300, 1e-14)
        sc = compute_c5(h_pol2)
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    if tleft() > 400:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 200:
            h1_pol = slsqp_polish(h1, n1, target1,
                                  [500, 3000, 12000], 300, 1e-13)
            sc = compute_c5(h1_pol)
            if sc < best_score:
                best_score = sc
                best_h = h1_pol
            else:
                h1_pol = h1

        if tleft() > 200:
            h_cd1, c5_cd1 = enhanced_cd(h1_pol, n1, target1,
                                        time_limit=min(150, tleft() * 0.45),
                                        max_sweeps=20000,
                                        n_active=4,
                                        n_cand_start=25, n_cand_end=55,
                                        max_no_improve=500, bisection_iters=13,

            if c5_cd1 < best_score:

                target0 = target1

    if tleft() > 100:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_cd2, c5_cd2 = enhanced_cd(best_h, n_cur, tgt_cur,
                                    time_limit=min(80, tleft() * 0.4),

                                         time_limit=min(40, tleft() * 0.4),

        if sc_ng2 < best_score:
            best_score = sc_ng2
