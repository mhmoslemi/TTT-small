import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)

    e = np.exp(shifted)

    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)

    if np.any(active_mask):

            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)

                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:

            except Exception:

        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 250:
                break
        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = 30.0
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def momentum_subgrad(h_init, n, target, time_limit, rng, max_steps=500000, n_shifts=7, base_lr=0.05, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 20.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        decay = 1.0 / (1.0 + 0.002 * step)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult * decay
            v = momentum * v - lr * grad
            h_try = project_feasible(h + v, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved and step > 5000:
            v *= 0.5
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 40:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[100, 500, 2000],
                            maxiter=400, time_limit=min(30, tleft() * 0.3),
                            penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 150:
        hc, sc = cyclic_cd(best_h, n, target, time_limit=min(120, tleft() * 0.3), rng=rng, max_sweeps=300000)
        if sc < best_score:
            best_score = sc
            best_h = hc

    if tleft() > 100:
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(80, tleft() * 0.3), rng=rng, n_cand=40, max_sweeps=50000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 100:
        hc, sc = cyclic_cd(best_h, n, target, time_limit=min(80, tleft() * 0.3), rng=rng, max_sweeps=200000)
        if sc < best_score:
            best_score = sc
            best_h = hc

    if tleft() > 60:
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(50, tleft() * 0.3), rng=rng, n_cand=50, max_sweeps=30000, n_shifts=3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 50:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[5000, 20000, 100000],
                            maxiter=300, time_limit=min(40, tleft() * 0.35),
                            penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 80:
        hc, sc = cyclic_cd(best_h, n, target, time_limit=min(60, tleft() * 0.4), rng=rng, max_sweeps=200000)
        if sc < best_score:
            best_score = sc
            best_h = hc

    if tleft() > 50:
        hsg, c5sg = momentum_subgrad(best_h, n, target, time_limit=min(40, tleft() * 0.4), rng=rng,
                                      max_steps=200000, n_shifts=10, base_lr=0.04, momentum=0.92)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    if tleft() > 200:
        n_new = 800
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = lbfgs_phase(h_interp, n_new, target_new,
                            alphas=[50, 500, 5000],
                            maxiter=300, time_limit=min(60, tleft() * 0.3),
                            penalty=100.0)
        if tleft() > 100:
            hc, sc2 = cyclic_cd(hp, n_new, target_new, time_limit=min(50, tleft() * 0.3), rng=rng, max_sweeps=150000)
            if sc2 < sc:
                sc = sc2
                hp = hc
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new

    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd(best_h, n_use, target_use, time_limit=min(60, tleft() * 0.4), rng=rng, n_cand=35, max_sweeps=40000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd
        if tleft() > 50:
            hc, sc = cyclic_cd(best_h, len(best_h), len(best_h) / 2.0, time_limit=min(40, tleft() * 0.5), rng=rng, max_sweeps=100000)
            if sc < best_score:
                best_score = sc
                best_h = hc

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_phase(best_h, n_use, target_use,
                            alphas=[100000, 500000],
                            maxiter=200, time_limit=min(30, tleft() * 0.7),
                            penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hc, sc = cyclic_cd(best_h, n_use, target_use, time_limit=min(20, tleft() * 0.8), rng=rng, max_sweeps=50000)
        if sc < best_score:
            best_score = sc
            best_h = hc

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
