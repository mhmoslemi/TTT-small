import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    a = _g_alpha
    shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / a)


def _slsqp_cons(x):
    return np.sum(x) - _g_target


def _slsqp_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() > deadline:
            break
        _g_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            if res.x is not None:
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    cur_score = sc
                    h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def build_change_matrix(h, n):
    shifts = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, shifts):
    n = len(h)
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            best_h = h.copy()

            sweeps_since_improve += 1

            break

            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 2e-5:
                h = h_pert
                change = build_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                consecutive_no_improve = 0
                continue

        k_star_idx = int(np.argmax(corr))
        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(len(top_k)) / len(top_k)
        else:
            weights = corr[top_k] / corr_top_sum

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:

                if g[k_star_idx] > 1e-12:

                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)

                                       [500, 3000, 15000, 80000],
                                       800, 1e-14, deadline=deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    if tleft() > 350:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            current_n = n2
            current_target = target2
        if tleft() > 250:
            h_cd, c5_cd = pair_cd(h2, n2, target2,
                                  time_limit=min(230, tleft() * 0.55),
                                  max_sweeps=40000,
                                  n_cand_start=15, n_cand_end=65,
                                  max_no_improve=450,
                                  n_active_shifts=5,
                                  reheat_interval=90, reheat_scale=0.03)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                current_n = n2
                current_target = target2
        if tleft() > 80:
            deadline = time.time() + min(60, tleft() * 0.45)
            h_pol, sc = slsqp_optimize(best_h, current_n, current_target,
                                       [1000, 5000, 25000, 100000],
                                       1000, 1e-14, deadline=deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    if tleft() > 150:
        h_cd, c5_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(130, tleft() * 0.6),
                              max_sweeps=35000,
                              n_cand_start=20, n_cand_end=80,
                              max_no_improve=500,
                              n_active_shifts=6,
                              reheat_interval=80, reheat_scale=0.025)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 120:
        for attempt in range(5):
            if tleft() < 45:
                break
            noise_scale = 0.04 + 0.03 * attempt
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(25, tleft() * 0.25)
            h_ref, sc_ref = slsqp_optimize(h_pert, current_n, current_target,
                                           [200, 1000, 5000, 20000],
                                           600, 1e-14, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 100:
        h_cc, c5_cc = cyc_cd(best_h, current_n, current_target,
                             time_limit=min(80, tleft() * 0.5),
                             n_cycles=80)
        if c5_cc < best_score:
            best_score = c5_cc
            best_h = h_cc

    if tleft() > 60:
        deadline = time.time() + min(45, tleft() * 0.5)
        h_final, sc_final = slsqp_optimize(best_h, current_n, current_target,
                                           [5000, 20000, 80000, 300000],
                                           1000, 1e-14, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    if tleft() > 20:
        h_cd_final, c5_final = pair_cd(best_h, current_n, current_target,
                                       time_limit=min(15, tleft() * 0.5),
                                       max_sweeps=10000,
                                       n_cand_start=30, n_cand_end=100,
                                       max_no_improve=200,
                                       n_active_shifts=6,
                                       reheat_interval=60, reheat_scale=0.015)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
