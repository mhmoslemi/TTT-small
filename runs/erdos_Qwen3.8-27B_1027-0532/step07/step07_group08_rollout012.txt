import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def slp_step(h_cur, n, target, trust_radius, c_bound):
    """
    One SLP step: solve LP to find best h within trust region of h_cur.
    Linearizes corr[k] around h_cur.
    """
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    
    # Compute current correlations
    corr_cur = compute_corr(h_cur)
    c_cur = float(np.max(corr_cur))
    
    # Build linear coefficients for each correlation
    # corr[k] = sum_i h[i+k] * (1-h[i]) for valid (i, i+k)
    # Linearize: corr[k] ≈ sum_i [(1-h_cur[i]) * h[i+k] - h_cur[i+k] * h[i]] + const[k]
    # Reindex for h[j]:
    #   coefficient of h[j] in corr[k]: (1-h_cur[j-k]) if 0<=j-k<n, plus (-h_cur[j+k]) if 0<=j+k<n
    #   const[k]: sum over valid i of h_cur[i+k]*(h_cur[i] - 1 + h_cur[i]) = sum h_cur[i+k]*(2*h_cur[i]-1)
    # Wait, let me recompute const.
    # f(h) at h_cur: sum_i h_cur[i+k] * (1 - h_cur[i])
    # Linear: sum_i [(1-h_cur[i])*h[i+k] - h_cur[i+k]*h[i]] + correction
    # The correction is: f(h_cur) - sum_i [(1-h_cur[i])*h_cur[i+k] - h_cur[i+k]*h_cur[i]]
    #             = sum_i h_cur[i+k]*(1-h_cur[i]) - sum_i [(1-h_cur[i])*h_cur[i+k] - h_cur[i+k]*h_cur[i]]
    #             = sum_i h_cur[i+k] - sum_i h_cur[i+k]*h_cur[i] - sum_i h_cur[i+k] + sum_i h_cur[i+k]*h_cur[i] + sum_i h_cur[i+k]*h_cur[i]
    #             = sum_i h_cur[i+k]*h_cur[i]
    
    # So corr[k] ≈ sum_j coeff[k,j] * h[j] + sum_i_valid h_cur[i+k]*h_cur[i]
    # where coeff[k,j] = (1-h_cur[j-k]) * [0<=j-k<n] + (-h_cur[j+k]) * [0<=j+k<n]
    
    # Build the matrix A where A[k, j] = coeff[k,j]
    n_corr = 2 * n - 1
    A = np.zeros((n_corr, n))
    const = np.zeros(n_corr)
    
    for k in range(n_corr):
        shift = k - mid
        # corr[k] = sum_i h[i+shift] * (1 - h[i]) for valid i
        # valid i: 0 <= i < n and 0 <= i+shift < n
        i_start = max(0, -shift)
        i_end = min(n, n - shift)
        if i_start >= i_end:
            const[k] = 0
            continue
        i_vals = np.arange(i_start, i_end)
        # coeff of h[j] where j = i+shift: (1-h_cur[i])
        # coeff of h[j] where j = i: -h_cur[i+shift]
        j_up = i_vals + shift  # indices in h for the "up" term
        j_dn = i_vals  # indices in h for the "down" term
        A[k, j_up] += (1 - h_cur[i_vals])
        A[k, j_dn] += (-h_cur[j_up])
        const[k] = np.sum(h_cur[j_up] * h_cur[i_vals])
    
    # LP: minimize t (which represents c_cur in "corr units")
    # subject to:
    #   A[k,:] @ h + const[k] <= t for all k  (i.e., corr[k] <= t)
    #   sum(h) = target
    #   h_cur[j] - r <= h[j] <= h_cur[j] + r (trust region)
    #   0 <= h[j] <= 1
    # t is free (or t >= 0)
    
    # Variables: [h(0), ..., h(n-1), t]
    # Objective: minimize t -> c = [0, ..., 0, 1]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A[k,:] @ h + const[k] - t <= 0
    # i.e., A[k,:] @ h - t <= -const[k]
    A_ineq = np.hstack([A, -np.ones((n_corr, 1))])
    b_ineq = -const
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: max(h_cur - r, 0) <= h[j] <= min(h_cur + r, 1), t >= c_cur - epsilon
    # Actually, to allow improvement, let t be free below c_cur
    lb = np.maximum(h_cur - trust_radius, 0.0)
    ub = np.minimum(h_cur + trust_radius, 1.0)
    lb_full = np.append(lb, -np.inf)  # t can be anything (but will be bounded by constraints)
    ub_full = np.append(ub, np.inf)
    
    # Actually, let's bound t to avoid numerical issues
    lb_full = np.append(lb, c_cur - 1.0)
    ub_full = np.append(ub, c_cur + 1.0)
    
    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                      A_eq=A_eq, b_eq=b_eq, bounds=list(zip(lb_full, ub_full)),
                      method='highs')
        if res.status == 0:
            h_new = res.x[:n]
            t_new = res.x[-1]
            return h_new, t_new
    except Exception:
        pass
    return None, None

def slp_optimize(h_init, n, target, time_limit, r_init=0.05, r_max=0.5):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c = compute_c5(h)
    r = r_init
    stale = 0
    
    for it in range(50000):
        if time.time() - start > time_limit:
            break
        
        h_trial, t_trial = slp_step(h, n, target, r)
        if h_trial is None:
            r *= 0.5
            if r < 1e-6:
                break
            continue
        
        h_trial = project_feasible(h_trial, target)
        c_trial = compute_corr(h_trial)
        actual_max = float(np.max(c_trial))
        
        if actual_max < float(np.max(compute_corr(h))) - 1e-10:
            # Improve
            h = h_trial
            stale = 0
            r = min(r * 1.3, r_max)
            c5_trial = actual_max * (2.0 / n)
            if c5_trial < best_c:
                best_c = c5_trial
                best_h = h.copy()
        else:
            stale += 1
            r *= 0.5
            if r < 1e-6:
                break
            if stale > 200:
                break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import linprog, minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def build_slp_matrix(h0, n):
    n_corr = 2 * n - 1
    mid = n - 1
    A = np.zeros((n_corr, n))
    consts = np.zeros(n_corr)
    h0_sq = h0 * h0
    for k in range(n_corr):
        s = k - mid
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_start >= i_end:
            continue
        i_vals = np.arange(i_start, i_end)
        j_up = i_vals + s
        A[k, j_up] += (1.0 - h0[i_vals])
        A[k, i_vals] += (-h0[j_up])
        consts[k] = np.sum(h0[j_up] * (h0[i_vals] - 1.0)) + np.sum(h0[j_up] * h0[i_vals])
        consts[k] = np.sum(h0[j_up] * h0[i_vals])
    return A, consts


def slp_optimize(h_init, time_limit, r_init=0.03, r_max=0.3, max_iters=20000):
    start = time.time()
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    r = r_init
    stale = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if r < 1e-7:
            break
        if stale > 500:
            break

        A, consts = build_slp_matrix(h, n)
        n_corr = 2 * n - 1
        c_cur = float(np.max(compute_corr(h)))

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ineq = np.hstack([A, -np.ones((n_corr, 1))])
        b_ineq = -consts
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        lb = np.maximum(h - r, 0.0)
        ub = np.minimum(h + r, 1.0)
        lb_full = np.append(lb, c_cur - 0.5)
        ub_full = np.append(ub, c_cur + 0.5)

        h_trial = None
        try:
            res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb_full, ub_full)),
                          method='highs')
            if res.status == 0:
                h_trial = res.x[:n]
        except Exception:
            pass

        if h_trial is None:
            r *= 0.5
            stale += 10
            continue

        h_trial = project_feasible(h_trial, target)
        c5_trial = compute_c5(h_trial)
        c5_cur = compute_c5(h)

        if c5_trial < c5_cur - 1e-12:
            h = h_trial
            stale = 0
            r = min(r * 1.2, r_max)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
        else:
            stale += 1
            r *= 0.6

    return best_h, best_c5


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=15, n_cand_end=70, max_no_improve=400,
            n_active_shifts=3, reheat_interval=150, reheat_scale=0.04):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0

    def compute_change_row(p):
        row = np.zeros(2 * n - 1)
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        row[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        row[neg_mask] -= h[p - shifts[neg_mask]]
        return row

    change = np.array([compute_change_row(p) for p in range(n)])

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if sweeps_since_improve > reheat_interval and sweeps_since_improve % 40 == 0:
            rng = np.random.RandomState(sweep * 7 + 3)
            n_pert = max(5, n // 40)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 1e-6:
                h = h_pert
                change = np.array([compute_change_row(p) for p in range(n)])
                corr = compute_corr(h)
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                continue

        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            weights = corr[top_k] / corr_top_sum

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star_idx = int(np.argmax(corr))
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i] = compute_change_row(best_i)
            change[best_j] = compute_change_row(best_j)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
