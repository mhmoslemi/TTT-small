    A_eq = np.zeros((1, n + 1))

        t_new = res.x[n]
    else:
        h_new = h
        t_new = t_bound
    
    return h_new, t_new

from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    mid = n - 1
    idxs = np.arange(n)
    
    A_ub = np.zeros((n_corr, n + 1))
    b_ub = np.zeros(n_corr)
    
    for m in range(n_corr):
        k = m - mid
        vk1 = (idxs + k >= 0) & (idxs + k < n)
        vk2 = (idxs - k >= 0) & (idxs - k < n)
        grad = np.zeros(n)
        if np.any(vk1):
            grad[vk1] += (1.0 - h[idxs[vk1] + k])
        if np.any(vk2):
            grad[vk2] -= h[idxs[vk2] - k]
        
        A_ub[m, :n] = grad
        A_ub[m, n] = -1.0
        b_ub[m] = corr[m] - grad @ h
    
    return A_ub, b_ub, corr

def lp_solve(h, n, target, max_iter=200, step_frac=1.0):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        A_ub, b_ub, corr = build_lp_matrices(h, n)

            else:

The function f(h) = max_k c_k(h) is the max of smooth functions, so it's smooth almost everywhere (except at points where two c_k's are tied at the max). We can use a subgradient method:

At point h, pick a subgradient of f at h. A subgradient of max_k c_k(h) is any convex combination of the gradients of the active c_k's.

This is a QP! And it's convex.

The subgradient is: ∇f(h) = ∑_k λ_k ∇c_k(h) where λ_k > 0 only for active k's and ∑λ_k = 1.

For simplicity, use λ_k that's proportional to exp(β * c_k) (softmax over the top few).

def proximal_gradient_step(h, n, target, lr, n_active=10, beta=50):
    """One proximal gradient step for the max problem"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    mid = n - 1
    cur_max = np.max(corr)
    
    # Softmax weights on top shifts
    top = np.argsort(corr)[-n_active:]
    w = np.exp(beta * (corr[top] - cur_max))
    w = w / w.sum()
    
    # Compute weighted subgradient
    idxs = np.arange(n)
    grad = np.zeros(n)
    for wi, m in zip(w, top):
        k = m - mid
        vk1 = (idxs + k >= 0) & (idxs + k < n)
        vk2 = (idxs - k >= 0) & (idxs - k < n)

            grad[vk1] += wi * (1.0 - h[idxs[vk1] + k])
        if np.any(vk2):
            grad[vk2] -= wi * h[idxs[vk2] - k]
    
    # Proximal gradient step: minimize 0.5/lr * ||h_new - h||^2 + grad · h_new
    # subject to sum(h_new) = target, 0 <= h_new <= 1
    # This is a QP. Use scipy.

Hmm, but the simple projection might not be optimal. Let me use a proper QP solver.

Actually, for the proximal step:
minimize 0.5/β * ||h - z||² + g^T h
s.t. 1^T h = target, 0 ≤ h ≤ 1

where z = h_old, g = subgradient.

This is a QP with a separable objective (plus the equality constraint). The KKT conditions give:
h_i - z_i + β * g_i + β * μ + β * (ν_i - ν_i') = 0
where μ is the multiplier for the equality, ν_i ≥ 0 for h_i ≥ 0, ν_i' ≥ 0 for h_i ≤ 1.

So h_i = z_i - β*g_i - β*μ + β*ν_i - β*ν_i'
With complementary slackness: if 0 < h_i < 1, then h_i = z_i - β*g_i - β*μ.

This is like a "soft thresholding" with a shift. We can solve it by bisection on μ:
For a given μ, h_i = clip(z_i - β*g_i - β*μ, 0, 1)
Then adjust μ so that ∑h_i = target.

This is very fast! Just a bisection on one scalar parameter.

Let me implement this:

```python
def proximal_step(h, grad, lr, target):
    """Optimal proximal gradient step with box and equality constraints"""
    n = len(h)
    # h_new = clip(h - lr * grad - lr * mu, 0, 1) where mu is chosen so sum = target
    # Bisection on mu
    
    lo = -100.0
    hi = 100.0
    
    def sum_at_mu(mu):
        return np.sum(np.clip(h - lr * grad - lr * mu, 0.0, 1.0))
    
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = sum_at_mu(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    mu = 0.5 * (lo + hi)
    h_new = np.clip(h - lr * grad - lr * mu, 0.0, 1.0)
    return h_new
```

This is O(n) per bisection iteration, very fast!

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_corr_and_grads(h, n, shift_indices=None):
    """Compute correlation and gradients for given shifts"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    mid = n - 1
    idxs = np.arange(n)
    
    if shift_indices is None:
        shift_indices = np.arange(n_corr)
    
    grads = np.zeros((len(shift_indices), n))
    for i, m in enumerate(shift_indices):
        k = int(m) - mid
        vk1 = (idxs + k >= 0) & (idxs + k < n)
        vk2 = (idxs - k >= 0) & (idxs - k < n)
        if np.any(vk1):
            grads[i, vk1] += (1.0 - h[idxs[vk1] + k])
        if np.any(vk2):
            grads[i, vk2] -= h[idxs[vk2] - k]
    
    return corr, grads

def lp_full(h, n, target):
    """Solve full LP linearization. Returns new h or None if failed."""
    corr, grads = get_corr_and_grads(h, n)
    n_corr = len(corr)
    
    A_ub = np.zeros((n_corr, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = corr - grads @ h
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None

def proximal_gradient_phase(h_init, n, target, time_limit, max_iter=50000):
    """Proximal gradient with adaptive parameters"""
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * (2.0 / n)
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Adaptive beta: increase over time
        beta = min(20.0 + 0.5 * it, 200.0)
        n_active = min(5 + it // 500, 20)
        
        top = np.argsort(corr)[-n_active:]
        w = np.exp(beta * (corr[top] - cur_max))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, m in zip(w, top):
            k = int(m) - mid
            vk1 = (idxs + k >= 0) & (idxs + k < n)
            vk2 = (idxs - k >= 0) & (idxs - k < n)

                grad[vk1] += wi * (1.0 - h[idxs[vk1] + k])
            if np.any(vk2):

def _prox_step(h, grad, lr, target):
    """Optimal proximal step: minimize 0.5/lr||x-z||^2 + g^T x s.t. sum(x)=target, 0<=x<=1"""
    n = len(h)
    z = h - lr * grad  # unconstrained minimizer before projection
    
    # Bisection on mu: x_i = clip(z_i - lr*mu, 0, 1), choose mu s.t. sum(x) = target
    lo = -100.0
    hi = 100.0

        s = np.sum(np.clip(z - lr * mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    
    mu = 0.5 * (lo + hi)
    return np.clip(z - lr * mu, 0.0, 1.0)

            h = h_lp

        elif c5_lp < c5_cur:

            no_improve += 1
        else:

                    h = h_try

                    break
            else:
                no_improve += 1
        
        if no_improve > 200:

    return best_h, best_c5

Let me also think about whether the LP approach can get "stuck." The LP linearizes at the current point and finds the best point in the linearized feasible region. def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def build_and_solve_lp(h, n, target, active_shifts=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    mid = n - 1
    idxs = np.arange(n)
    
    if active_shifts is None:
        active_shifts = np.arange(n_corr)
    
    n_act = len(active_shifts)
    A_ub = np.zeros((n_act, n + 1))
    b_ub = np.zeros(n_act)
    
    for ai, m in enumerate(active_shifts):
        k = int(m) - mid
        vk1 = (idxs + k >= 0) & (idxs + k < n)
        vk2 = (idxs - k >= 0) & (idxs - k < n)
        if np.any(vk1):
            A_ub[ai, vk1] += (1.0 - h[idxs[vk1] + k])
        if np.any(vk2):
            A_ub[ai, vk2] -= h[idxs[vk2] - k]
        A_ub[ai, n] = -1.0
        b_ub[ai] = corr[m] - A_ub[ai, :n] @ h

    A_eq = np.zeros((1, n + 1))

    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
