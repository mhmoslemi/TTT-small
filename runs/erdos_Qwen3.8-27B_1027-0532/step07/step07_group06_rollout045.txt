def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

    _g_target = target

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    cur_score = compute_c5(h_cur)

    if use_pnorm:
        func = pnorm_obj
    else:
        func = soft_max_obj

    for param in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)

        best_h = h2

    # Phase 3: Pair CD with top-3 shifts
    t3 = min(total_deadline, start_time + 0.62 * budget_s)
    h3, sc3 = pair_cd_improved(best_h, n0, target0, t3,
                                max_sweeps=25000, n_cand_start=15, n_cand_end=55,
                                n_top_shifts=3, bisection_iters=18, max_no_improve=350)
    if sc3 < best_score:
        best_score = sc3
        best_h = h3

    # Phase 4: Block CD
    t4 = min(total_deadline, start_time + 0.72 * budget_s)
    h4, sc4 = block_cd(best_h, n0, target0, t4, max_sweeps=20000, block_sizes=(2, 3, 5))
    if sc4 < best_score:
        best_score = sc4
        best_h = h4

    # Phase 5: SLSQP p-norm refinement
    t5 = min(total_deadline, start_time + 0.80 * budget_s)
    h5, sc5 = slsqp_round(best_h, n0, target0, [4, 8, 16, 32], 600, t5, use_pnorm=True)
    if sc5 < best_score:
        best_score = sc5
        best_h = h5

    # Phase 6: Multi-restart to escape local optima
    t6 = min(total_deadline, start_time + 0.90 * budget_s)
    h6, sc6 = multi_restart(best_h, n0, target0, t6, n_restarts=4)
    if sc6 < best_score:
        best_score = sc6
        best_h = h6

    # Phase 7: Higher resolution
    remaining = total_deadline - time.time()
    if remaining > 80:
        n1 = 600
        target1 = n1 / 2.0
        h_int = interpolate_to(best_h, n1, target1)
        t7 = min(total_deadline, start_time + 0.93 * budget_s)
        h7, sc7 = pair_cd_improved(h_int, n1, target1, t7,
                                    max_sweeps=15000, n_cand_start=12, n_cand_end=45,
                                    n_top_shifts=3, bisection_iters=18, max_no_improve=300)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 8: Final SLSQP polish
    t8 = total_deadline
    n_final = len(best_h)
    target_final = n_final / 2.0
    h8, sc8 = slsqp_round(best_h, n_final, target_final, [2000, 8000], 400, t8, use_pnorm=False)
    if sc8 < best_score:
        best_score = sc8
        best_h = h8

    # Phase 9: Final pair CD
    if time.time() < total_deadline - 10:
        h9, sc9 = pair_cd_improved(best_h, n_final, target_final, total_deadline,
                                    max_sweeps=10000, n_cand_start=25, n_cand_end=60,
                                    n_top_shifts=1, bisection_iters=20, max_no_improve=200)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
