One more thought: the "exact" line search formula. For a pair move with difference vector g = profile_j - profile_i:

The new correlation is: new_corr[l] = corr[l] + δ * g[l]

We want the largest δ ∈ [0, delta_max] such that max_l new_corr[l] < cur_max.

This is equivalent to: for all l, corr[l] + δ*g[l] < cur_max
→ δ < (cur_max - corr[l]) / g[l] for all l where g[l] > 0

So δ_threshold = min_{l: g[l] > 0} (cur_max - corr[l]) / g[l]

If no such l exists (all g[l] ≤ 0), then δ_threshold = ∞, and δ* = delta_max.
Otherwise, δ* = min(delta_max, δ_threshold).

But wait, this gives the δ where the new max is EXACTLY cur_max (no improvement). For an improvement, we need the new max < cur_max.

The improvement is: cur_max - max_l (corr[l] + δ* * g[l])

If δ* = δ_threshold (the binding constraint), then at the binding l*, new_corr[l*] = cur_max, so improvement = 0. That's not useful.

The issue is: at δ = 0, the max is cur_max. As δ increases, some components go up and some go down. The max will be cur_max (or higher) for small δ if any active component has g ≥ 0.

For an improvement, we need:
- All active components (those with corr[l] = cur_max) must have g[l] < 0 (so they decrease)
- AND no inactive component can reach cur_max before all active components drop below it

If all active components have g < 0, then for small δ > 0, the max is still cur_max (determined by the active components, which are decreasing). Wait no - if ALL active components have g < 0, then for any δ > 0, they are all below cur_max. The new max is then determined by the next-highest components.

So the improvement at δ is: cur_max - max_l (corr[l] + δ*g[l])

For small δ, the max is determined by either:
- The active components (now at cur_max + δ*g[l] < cur_max)
- The next-highest components (at corr[l] + δ*g[l])

The improvement is maximized at the δ where one of the "rising" components (g[l] > 0) catches up to the "falling" active components.

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_profile(h, m):
    n = len(h)
    profile = np.zeros(2 * n - 1)
    idx1 = m + np.arange(2 * n - 1) - (n - 1)
    idx2 = m - (np.arange(2 * n - 1) - (n - 1))
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    profile[v1] += h[idx1[v1]]
    profile[v2] += (1.0 - h[idx2[v2]])
    return profile

def multi_shift_gradient(h, corr, n_shifts=5, alpha_w=30.0):
    n = len(h)
    cur_max = float(np.max(corr))
    top_idx = np.argsort(corr)[-n_shifts:]
    w = np.exp(alpha_w * (corr[top_idx] - cur_max))
    w = w / w.sum()
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    for wi, si in zip(w, top_idx):
        k = int(si) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= wi * h[idxs[v2] + k]
    return grad

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start_t = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        for pen in [penalty * 1.0, penalty * 10.0, penalty * 100.0]:
            if time.time() - start_t > time_limit:
                break
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_aspiration(h, target):

    if c5_bin < c5_orig:

                               n_cand=50, max_sweeps=500000, n_shifts=5, alpha_w=30.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 3: Multi-pair moves
    if tleft() > 150:
        mp_time = min(120, tleft() * 0.25)
        hmp, c5mp = multi_pair_cd(best_h, n, target, time_limit=mp_time,
                                  K=4, n_shifts=5, alpha_w=30.0, max_iters=100000)
        if c5mp < best_score:
            best_score = c5mp
            best_h = hmp

    # Phase 4: More pair CD with different parameters
    if tleft() > 200:
        cd_time = min(200, tleft() * 0.3)
        hcd, c5cd = pair_cd_v2(best_h, n, target, time_limit=cd_time,
                               n_cand=80, max_sweeps=500000, n_shifts=8, alpha_w=50.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 5: Block moves
    if tleft() > 120:
        bm_time = min(100, tleft() * 0.25)
        hbm, c5bm = block_move_cd(best_h, n, target, time_limit=bm_time,
                                  block_size=2, n_cand=30, max_sweeps=30000, n_shifts=5)
        if c5bm < best_score:
            best_score = c5bm
            best_h = hbm

    # Phase 6: L-BFGS with high alpha
    if tleft() > 100:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[5000, 20000, 100000],
                            maxiter=300, time_limit=min(70, tleft() * 0.3),
                            penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 7: Pair CD again (refined)
    if tleft() > 150:
        cd_time = min(150, tleft() * 0.3)
        hcd, c5cd = pair_cd_v2(best_h, n, target, time_limit=cd_time,
                               n_cand=100, max_sweeps=500000, n_shifts=3, alpha_w=20.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 8: Binary aspiration
    if tleft() > 80:
        h_bin, c5_bin = binary_aspiration(best_h, target)
        if c5_bin < best_score:
            best_score = c5_bin
            best_h = h_bin
            hp, sc = lbfgs_phase(best_h, n, target,
                                alphas=[100, 1000, 10000],
                                maxiter=200, time_limit=min(40, tleft() * 0.4),
                                penalty=500.0)
            if sc < best_score:
                best_score = sc
                best_h = hp

    # Phase 9: Higher resolution
    if tleft() > 250:
        n_new = 1200
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = lbfgs_phase(h_interp, n_new, target_new,
                            alphas=[50, 500, 5000],
                            maxiter=200, time_limit=min(80, tleft() * 0.25),
                            penalty=100.0)
        if tleft() > 100:
            hcd, c5cd = pair_cd_v2(hp, n_new, target_new,
                                   time_limit=min(100, tleft() * 0.3),
                                   n_cand=40, max_sweeps=100000, n_shifts=5, alpha_w=30.0)
            if c5cd < sc:
                sc = c5cd
                hp = hcd
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new

    # Phase 10: Final intensive pair CD at best resolution
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(60, tleft() * 0.6)
        hcd, c5cd = pair_cd_v2(best_h, n_use, target_use, time_limit=cd_time,
                               n_cand=60, max_sweeps=500000, n_shifts=5, alpha_w=30.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 11: Final L-BFGS
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_phase(best_h, n_use, target_use,
                            alphas=[50000, 200000],
                            maxiter=150, time_limit=min(30, tleft() * 0.7),
                            penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 12: Final pair CD
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd_v2(best_h, n_use, target_use, time_limit=min(18, tleft() * 0.7),
                               n_cand=40, max_sweeps=100000, n_shifts=3, alpha_w=20.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
