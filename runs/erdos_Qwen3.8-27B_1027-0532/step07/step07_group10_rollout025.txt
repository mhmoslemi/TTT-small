def vectorized_cd_sweep(h, corr, change, n_corr, shifts, n_cand, n_binary_iters):
    n = len(h)
    idxs = np.arange(n)
    mid = n - 1
    
    # Compute gradient for active set
    k_star = int(np.argmax(corr)) - mid
    grad = np.zeros(n)
    v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    grad -= np.mean(grad)
    
    sorted_idx = np.argsort(grad)
    low_set = sorted_idx[:n_cand]
    high_set = sorted_idx[-n_cand:]
    
    # Build G matrix: G[ii, jj] = change[low_set[jj]] - change[high_set[ii]]
    # Shape: (n_h, n_l, n_corr)
    G = change[low_set][None, :, :] - change[high_set][:, None, :]
    
    # Delta bounds for each pair
    # delta_lo[ii,jj] = max(h[high_set[ii]] - 1, -h[low_set[jj]])
    # delta_hi[ii,jj] = min(h[high_set[ii]], 1 - h[low_set[jj]])
    h_high = h[high_set][:, None]  # (n_h, 1)
    h_low = h[low_set][None, :]  # (1, n_l)
    
    delta_lo = np.maximum(h_high - 1.0, -h_low)  # (n_h, n_l)
    delta_hi = np.minimum(h_high, 1.0 - h_low)  # (n_h, n_l)
    
    # Exclude i == j
    # (we need to check which (ii,jj) pairs have high_set[ii] == low_set[jj])
    
    # Binary search for optimal delta
    a = delta_lo.copy()
    b = delta_hi.copy()
    valid = (b - a) > 1e-13
    
    for _ in range(n_binary_iters):
        mid_d = 0.5 * (a + b)  # (n_h, n_l)
        # values = corr[None,None,:] + mid_d[:,:,None] * G  # (n_h, n_l, n_corr)
        # This might be too much memory for large n_corr
        # For n_h=n_l=50, n_corr=2400: 50*50*2400 = 6M floats = 48MB. OK.
        vals = corr[None, None, :] + mid_d[:, :, None] * G
        f_mid = vals.max(axis=2)  # (n_h, n_l)
        
        # f_prime at each pair: max of G over active set
        active_mask = vals >= f_mid[:, :, None] - 1e-14  # (n_h, n_l, n_corr)
        f_prime = np.where(active_mask, G, -np.inf).max(axis=2)  # (n_h, n_l)
        
        update = f_prime > 1e-13
        a = np.where(update & valid, mid_d, a)
        b = np.where(update & valid, b, mid_d)
    
    delta_star = 0.5 * (a + b)
    
    # Find best pair
    vals_final = corr[None, None, :] + delta_star[:, :, None] * G
    f_final = vals_final.max(axis=2)
    
    # Exclude invalid pairs
    i_arr = high_set[:, None]  # (n_h, 1)
    j_arr = low_set[None, :]  # (1, n_l)
    same_mask = (i_arr == j_arr) | ~valid
    f_final[same_mask] = np.inf
    
    best_idx = np.unravel_index(np.argmin(f_final), f_final.shape)
    if f_final[best_idx] < np.max(corr):
        i = int(high_set[best_idx[0]])
        j = int(low_set[best_idx[1]])
        delta = delta_star[best_idx]
        return i, j, delta
    return None

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)

def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = (idxs + k) < n
        wm = (idxs - k) >= 0
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                  n_binary=12):

    n_corr = 2 * n - 1

    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:

        if consecutive_no_improve > max_no_improve:
            break
        
        # Noise injection for diversification
        if noise_interval > 0 and sweep > 0 and sweep % noise_interval == 0 and consecutive_no_improve > 50:
            h = h + np.random.randn(n) * noise_scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)

        # Multi-active gradient

        c_max = corr[top_k[0]]

            grad[v1] += wi * (1.0 - h[idxs[v1] - k])

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = (idxs + k) < n
        wm = (idxs - k) >= 0
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15, time_limit=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

            h2_cd, c5_2cd = vectorized_cd(best_h, best_n, best_target,
                                          time_limit=cd_limit2, max_sweeps=40000,
                                          n_cand_start=40, n_cand_end=90,
                                          n_active=5, noise_interval=200,
                                          noise_scale=0.03, max_no_improve=500,
                                          n_binary=11, alpha_w=40.0)
            if c5_2cd < best_score:
                best_score = c5_2cd
                best_h = h2_cd
                best_n = n2
                best_target = target2

        if tleft() > 40:
            h2_gd, c5_2gd = gradient_polish(best_h, best_n, best_target,
                                            time_limit=min(30, tleft() * 0.3),
                                            max_steps=15000, base_lr=0.015, n_active=8)
            if c5_2gd < best_score:
                best_score = c5_2gd
                best_h = h2_gd

    if tleft() > 80:
        slsqp_limit = min(60, tleft() * 0.5)
        h_fin, sc_fin = slsqp_round(best_h, best_n, best_target,
                                    [5000, 20000], 1000, 1e-15, time_limit=slsqp_limit)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    if tleft() > 60:
        cd_limit3 = min(100, tleft() * 0.5)
        h3_cd, c5_3cd = vectorized_cd(best_h, best_n, best_target,
                                      time_limit=cd_limit3, max_sweeps=20000,
                                      n_cand_start=60, n_cand_end=120,
                                      n_active=6, noise_interval=300,
                                      noise_scale=0.02, max_no_improve=400,
                                      n_binary=14, alpha_w=60.0)
        if c5_3cd < best_score:
            best_score = c5_3cd
            best_h = h3_cd

    if tleft() > 30:
        h4_gd, c5_4gd = gradient_polish(best_h, best_n, best_target,
                                        time_limit=min(20, tleft() * 0.6),
                                        max_steps=10000, base_lr=0.01, n_active=10)
        if c5_4gd < best_score:
            best_score = c5_4gd
            best_h = h4_gd

    if tleft() > 25:
        h5, sc5 = slsqp_round(best_h, best_n, best_target,
                              [20000, 50000], 800, 1e-15, time_limit=min(20, tleft() * 0.8))
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    if tleft() > 15:
        rng_final = np.random.RandomState(seed)
        for ns in [0.01, 0.03]:
            if tleft() < 10:
                break
            h_pert = best_h + rng_final.randn(best_n) * ns
            h_pert = project_feasible(h_pert, best_target)
            h_ref, sc_ref = slsqp_round(h_pert, best_n, best_target,
                                        [2000, 10000], 400, 1e-14, time_limit=min(8, tleft() * 0.4))
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
