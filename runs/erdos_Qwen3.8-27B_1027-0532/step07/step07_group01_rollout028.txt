import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def fast_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=100, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(777)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            # Perturb and restart
            h = best_h + rng.randn(n) * 0.01
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0
            if best_c5 > 0.381:  # Give up if we can't do better
                break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        # Vectorized pair evaluation
        n_h = len(high_set)
        n_l = len(low_set)
        
        # For each pair (i, j), g = change[j] - change[i]
        # We want to find delta that minimizes max(corr + delta * g)
        # The optimal delta is where the derivative of max(corr + delta*g) is 0
        # i.e., max(g[active_set]) = 0 where active_set = {k: corr[k] + delta*g[k] = max}
        
        # Strategy: for each pair, the objective is a piecewise linear function of delta.
        # The minimum is either at a "kink" point or at the boundary.
        # Kink points occur when corr[a] + delta*g[a] = corr[b] + delta*g[b]
        # i.e., delta = (corr[b] - corr[a]) / (g[a] - g[b])
        
        # For speed, use bisection (as in the original) but vectorize over pairs.
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        # Vectorized bisection over all pairs
        # Build the change vectors for all pairs
        # g_matrix[pair_idx] = change[low_set[jj]] - change[high_set[ii]]
        
        # Actually this would be n_h * n_l * n_corr which is too large for n=400, ncand=100
        # 100*100*799 = 7.99M - this is actually fine!
        
        if n_h * n_l * n_corr < 20_000_000:
            # Build g matrix: shape (n_h * n_l, n_corr)
            # g[ii*n_l + jj] = change[low_set[jj]] - change[high_set[ii]]
            ch_high = change[high_set]  # (n_h, n_corr)
            ch_low = change[low_set]  # (n_l, n_corr)
            # For pair (ii, jj): g = ch_low[jj] - ch_high[ii]
            # Reshape: g_matrix = ch_low[None, :, :].repeat(n_h, 0) - ch_high[:, None, :].repeat(n_l, 1)
            # Shape: (n_h, n_l, n_corr)
            g_matrix = ch_low[None, :, :] - ch_high[:, None, :]  # (n_h, n_l, n_corr)
            
            # For bisection, we need to find the best delta for each pair
            # delta range: for pair (ii, jj), delta in [max(h[i]-1, -h[j]), min(h[i], 1-h[j])]
            # Let's use a common range for simplicity: [-h[high_set[ii]], min(1-h[low_set[jj]], h[high_set[ii]])]
            # Actually the delta is the amount we decrease h[i] and increase h[j]
            # So delta > 0, and delta <= min(h[i], 1-h[j])
            
            delta_lo = np.maximum(h[high_set][:, None] - 1.0, -h[low_set][None, :])  # (n_h, n_l)
            delta_hi = np.minimum(h[high_set][:, None], 1.0 - h[low_set][None, :])  # (n_h, n_l)
            
            # Bisection
            for _ in range(14):
                mid_d = 0.5 * (delta_lo + delta_hi)  # (n_h, n_l)
                vals = corr[None, None, :] + mid_d[:, :, None] * g_matrix  # (n_h, n_l, n_corr)
                f_mid = np.max(vals, axis=2)  # (n_h, n_l)
                active_mask = vals >= f_mid[:, :, None] - 1e-14  # (n_h, n_l, n_corr)
                f_prime = np.max(np.where(active_mask, g_matrix, -np.inf), axis=2)  # (n_h, n_l)
                delta_hi = np.where(f_prime > 1e-12, mid_d, delta_hi)
                delta_lo = np.where(f_prime <= 1e-12, mid_d, delta_lo)
            
            delta_star = 0.5 * (delta_lo + delta_hi)  # (n_h, n_l)
            
            # Compute objective at delta_star for each pair
            vals_final = corr[None, None, :] + delta_star[:, :, None] * g_matrix
            obj = np.max(vals_final, axis=2) * dx  # (n_h, n_l)
            
            # Filter out invalid pairs
            obj = np.where(delta_star > 1e-13, obj, np.inf)
            
            min_idx = np.unravel_index(np.argmin(obj), obj.shape)
            min_val = obj[min_idx]
            
            if min_val < best_trial_c5:
                best_i = int(high_set[min_idx[0]])
                best_j = int(low_set[min_idx[1]])
                best_delta = float(delta_star[min_idx])
                best_trial_c5 = min_val
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def vectorized_cd(h_init, n, target, time_limit, max_sweeps=50000,
                  n_cand_start=30, n_cand_end=100, max_no_improve=400,
                  bisection_iters=13):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(12345)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            h = best_h + rng.randn(n) * 0.015
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        n_h = len(high_set)
        n_l = len(low_set)

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        if n_h * n_l <= 16000:
            ch_high = change[high_set]
            ch_low = change[low_set]
            g_matrix = ch_low[None, :, :] - ch_high[:, None, :]

            delta_lo = np.maximum(h[high_set][:, None] - 1.0, -h[low_set][None, :])
            delta_hi = np.minimum(h[high_set][:, None], 1.0 - h[low_set][None, :])

            for _ in range(bisection_iters):
                mid_d = 0.5 * (delta_lo + delta_hi)
                vals = corr[None, None, :] + mid_d[:, :, None] * g_matrix
                f_mid = np.max(vals, axis=2)
                active_mask = vals >= f_mid[:, :, None] - 1e-14
                f_prime = np.max(np.where(active_mask, g_matrix, -np.inf), axis=2)
                delta_hi = np.where(f_prime > 1e-12, mid_d, delta_hi)
                delta_lo = np.where(f_prime <= 1e-12, mid_d, delta_lo)

            delta_star = 0.5 * (delta_lo + delta_hi)
            vals_final = corr[None, None, :] + delta_star[:, :, None] * g_matrix
            obj = np.max(vals_final, axis=2) * dx
            obj = np.where(delta_star > 1e-13, obj, np.inf)

            flat_idx = np.argmin(obj)
            min_val = obj.flat[flat_idx]
            if min_val < best_trial_c5:
                mi, mj = np.unravel_index(flat_idx, obj.shape)
                best_i = int(high_set[mi])
                best_j = int(low_set[mj])
                best_delta = float(delta_star[mi, mj])
        else:
            for ii in range(n_h):
                i = int(high_set[ii])
                for jj in range(n_l):
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    d_lo = max(h[i] - 1.0, -h[j])
                    d_hi = min(h[i], 1.0 - h[j])
                    if d_hi - d_lo < 1e-13:
                        continue
                    g = change[j] - change[i]
                    a_d, b_d = d_lo, d_hi
                    for _ in range(bisection_iters):
                        m_d = 0.5 * (a_d + b_d)
                        vals = corr + m_d * g
                        f_m = np.max(vals)
                        active = vals >= f_m - 1e-14
                        if active.any():
                            fp = np.max(g[active])
                        else:
                            fp = 0.0
                        if fp > 1e-12:
                            b_d = m_d
                        else:
                            a_d = m_d
                    d_star = 0.5 * (a_d + b_d)
                    if abs(d_star) < 1e-13:
                        continue
                    approx = np.max(corr + d_star * g) * dx
                    if approx < best_trial_c5:
                        best_trial_c5 = approx
                        best_i, best_j, best_delta = i, j, d_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def binary_search_improve(h_init, n, target, time_limit):

def change_j_minus_i(h, n, i, j, shifts):
    g = np.zeros(2 * n - 1)
    for s in shifts:
        if 0 <= i + s < n and 0 <= j + s < n:
            g[s + n - 1] += h[j + s] - h[i + s]
        if 0 <= i - s < n and 0 <= j - s < n:
            g[n - 1 - s] -= h[j - s] - h[i - s]
    return g


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)

        return budget_s - 15.0 - (time.time() - start_time)

    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [50, 200, 1000, 5000, 20000],
                                600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 300:
        h_cd, c5_cd = vectorized_cd(best_h, current_n, current_target,
                                    time_limit=min(350, tleft() * 0.6),
                                    max_sweeps=60000,
                                    n_cand_start=30, n_cand_end=90,
                                    max_no_improve=400)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [5000, 20000, 80000, 200000],
                                1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        h_cd2, c5_cd2 = vectorized_cd(best_h, current_n, current_target,
                                      time_limit=min(250, tleft() * 0.6),
                                      max_sweeps=40000,
                                      n_cand_start=40, n_cand_end=110,
                                      max_no_improve=500)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 100:
        h_ms, c5_ms = multi_grad_cd(best_h, current_n, current_target,
                                    time_limit=min(120, tleft() * 0.5),
                                    max_sweeps=20000, n_cand=50, n_shifts=4)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if tleft() > 200:
        n1 = 480
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 150:
            h_pol1, sc1 = slsqp_round(h1, n1, target1,
                                      [100, 500, 2000, 10000, 50000],
                                      600, 1e-15)
            if tleft() > 80:
                h_cd1, c5_cd1 = vectorized_cd(h_pol1, n1, target1,
                                              time_limit=min(100, tleft() * 0.5),
                                              max_sweeps=20000,
                                              n_cand_start=25, n_cand_end=70,
                                              max_no_improve=300)
                if c5_cd1 < best_score:
                    best_score = c5_cd1
                    best_h = h_cd1
                    current_n = n1
                    current_target = target1

    if tleft() > 150:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.10, 0.20]:
            if tleft() < 70:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [100, 500, 2000, 8000],
                                        400, 1e-15)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 80:
                h_cd_p, c5_p = vectorized_cd(h_ref, current_n, current_target,
                                             time_limit=min(60, tleft() * 0.3),
                                             max_sweeps=10000,
                                             n_cand_start=30, n_cand_end=80,
                                             max_no_improve=250)
                if c5_p < best_score:
                    best_score = c5_p
                    best_h = h_cd_p

    if tleft() > 120:
        h_final_cd, c5_final = vectorized_cd(best_h, current_n, current_target,
                                            time_limit=min(110, tleft() * 0.6),
                                            max_sweeps=30000,
                                            n_cand_start=40, n_cand_end=120,
                                            max_no_improve=500)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if tleft() > 40:
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                             [10000, 50000, 200000, 500000],
                                             800, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    if tleft() > 30:
        h_last_cd, c5_last = vectorized_cd(best_h, current_n, current_target,
                                           time_limit=min(25, tleft() * 0.7),
                                           max_sweeps=10000,
                                           n_cand_start=50, n_cand_end=130,
                                           max_no_improve=200)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_last_cd

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
