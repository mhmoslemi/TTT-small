    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(12345)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        _update_change_row(change, p, h, n, shifts)

    total_perturbs = 0
    max_perturbs = 20 if perturb_every > 0 else 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            if perturb_every > 0 and total_perturbs < max_perturbs:
                # Perturb and restart
                h = best_h.copy()
                n_pert = int(n * 0.1)
                perm = rng.permutation(n)
                idxs_p = perm[:n_pert]
                h[idxs_p] += rng.randn(n_pert) * perturb_scale
                h = project_feasible(h, target)
                for p in idxs_p:
                    _update_change_row(change, p, h, n, shifts)
                total_perturbs += 1
                no_improve = 0
                continue
            else:
                break

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_shifts].astype(float)
            wsum = np.sum(weights)
            if wsum < 1e-15:
                weights = np.ones(n_top_shifts) / n_top_shifts
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            _update_change_row(change, best_i, h, n, shifts)
            _update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def stochastic_kick(h_init, n, target, deadline, max_steps=15000, base_scale=0.02):
    rng = np.random.RandomState(777)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() > deadline:
            break
        if no_improve > 2000:
            break

        scale = base_scale * max(0.05, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        perm = rng.permutation(n)
        idxs = perm[:n_pert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 50:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5

def grad_descent_lse(h_init, n, target, deadline, alpha=1000, lr=0.01, momentum=0.9, max_iter=100000):
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    velocity = np.zeros(n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        weights = np.exp(np.clip(alpha * (corr - c_max), -500, 0))
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            break
        
        # Gradient of LSE
        grad = np.zeros(n)
        for k_idx, w in enumerate(weights):
            if w / w_sum < 1e-10:
                continue
            k = k_idx - (n - 1)
            idxs = np.arange(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        grad -= np.mean(grad)  # Project out the constraint direction

        h_new = h + velocity
        h_new = project_feasible(h_new, target)

        if sc < best_score - 1e-13:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, deadline):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(_lse_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-14:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def init_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_after_move(change, n, i, delta_i, j, delta_j):
    q = np.arange(n)
    mid = n - 1
    cols_anti_i = (i - q) + mid
    cols_diag_i = (q - i) + mid
    cols_anti_j = (j - q) + mid
    cols_diag_j = (q - j) + mid
    np.add.at(change, (q, cols_anti_i), delta_i)
    np.add.at(change, (q, cols_diag_i), delta_i)
    np.add.at(change, (q, cols_anti_j), delta_j)
    np.add.at(change, (q, cols_diag_j), delta_j)


def pair_cd_fast(h_init, n, target, deadline, max_sweeps=80000,
                 n_cand_start=15, n_cand_end=100,

                 bisection_iters=18, max_perturbs=15, perturb_scale=0.015,
                 seed=999):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    rng = np.random.RandomState(seed)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    total_perturbs = 0

    change = init_change_matrix(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            if total_perturbs < max_perturbs:
                h = best_h.copy()

                idxs_p = perm[:n_pert]
                h[idxs_p] += rng.randn(n_pert) * perturb_scale
                h = project_feasible(h, target)
                change = init_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full').copy()
                total_perturbs += 1
                no_improve = 0
                continue
            else:
                break

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_shifts].astype(float)
            wsum = np.sum(weights)
            if wsum < 1e-15:

            else:

            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():

                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:

                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr += best_delta * (change[best_j] - change[best_i])
            update_change_after_move(change, n, best_i, -best_delta, best_j, best_delta)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

                                        max_steps=15000, base_scale=0.012, seed=555)
        if sc_st < best_score:
            best_score = sc_st
            best_h = h_st

    remaining = budget_s - 15.0 - (time.time() - start_time)
    if remaining > 60:
        h_fin, sc_fin = slsqp_refine(best_h, n0, target0,
                                     [10000, 50000], 800,
                                     start_time + min(60, remaining - 5))
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    remaining = budget_s - 15.0 - (time.time() - start_time)
    if remaining > 50:
        cd_end3 = start_time + min(50, remaining - 3)
        h_cd3, sc_cd3 = pair_cd_fast(best_h, n0, target0, cd_end3,
                                     max_sweeps=50000, n_cand_start=30, n_cand_end=120,
                                     max_no_improve=300, n_top_shifts=1,
                                     bisection_iters=20, max_perturbs=5,
                                     perturb_scale=0.008, seed=777)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
