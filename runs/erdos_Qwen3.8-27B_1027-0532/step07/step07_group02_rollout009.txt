    c_max = np.max(corr)

    # Take step

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

    rng = np.random.RandomState(seed)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)

        if use_softmax:
            temp = temp_start + (temp_end - temp_start) * progress
            corr_shifted = (corr - np.max(corr)) / temp
            weights = np.exp(corr_shifted)
            wsum = np.sum(weights)
            if wsum < 1e-30:
                weights = np.zeros(n_corr)
                weights[int(np.argmax(corr))] = 1.0
            else:
                weights /= wsum
            grad = np.zeros(n)
            for s in range(n_corr):
                w = weights[s]
                if w < 1e-10:
                    continue
                k = s - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        else:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        # Evaluate pairs
        # To speed up, first compute rough scores, then do line search on top candidates
        rough_scores = np.zeros((n_h, n_l))
        for ii in range(n_h):
            for jj in range(n_l):
                i = int(high_set[ii])
                j = int(low_set[jj])

                    rough_scores[ii, jj] = -np.inf
                    continue
                # Rough score: directional derivative at active constraints
                rough_scores[ii, jj] = -np.dot(weights, change[j] - change[i]) if use_softmax else -change[i, int(np.argmax(corr))] + change[j, int(np.argmax(corr))]

            if delta_hi - delta_lo < 1e-12:

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(n_binsearch):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:

                else:
                    a_d = mid_d

            if abs(delta_star) < 1e-12:

            approx_max = np.max(corr + delta_star * g) * dx

                best_i = i

                best_delta = delta_star

        for _ in range(n_random):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-12:

                else:
                    f_prime = 0.0

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:

            for p in idxs_pert:

            h = project_feasible(h, target)
            for p in idxs_pert:
                update_change_row(change, int(p), h, shifts, n)

    return best_h, best_c5

def cd_optimize_fast(h_init, n, target, time_limit, seed=42,
                     n_cand_start=30, n_cand_end=80,
                     max_sweeps=50000, max_no_improve=400,
                     n_binsearch=12, stochastic_frac=0.03):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        active_idx = int(np.argmax(corr))
        k_star = active_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        n_h = len(high_set)
        n_l = len(low_set)

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                if g[active_idx] > 1e-10:
                    continue
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid * dx >= best_trial_c5 + 1e-15:
                        b_d = mid_d
                        continue
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, shifts, n)
            update_change_row(change, best_j, h, shifts, n)
        elif consecutive_no_improve > 60 and consecutive_no_improve % 30 == 0:
            n_pert = max(5, n // 30)
            idxs_pert = rng.choice(n, size=n_pert, replace=False)
            h[idxs_pert] += rng.randn(n_pert) * 0.02
            h = project_feasible(h, target)
            for p in idxs_pert:
                update_change_row(change, int(p), h, shifts, n)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_start_cd(h_best, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    scales = [0.005, 0.015, 0.03, 0.06, 0.1, 0.15]
    attempts = 0
    max_attempts = 20

    while time.time() - start < time_limit and attempts < max_attempts:
        scale = scales[rng.randint(len(scales))]
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        remaining = time_limit - (time.time() - start)
        if remaining < 30:
            break
        alloc = min(remaining * 0.4, 60)
        h_cd, c5_cd = cd_optimize_fast(h_pert, n, target, time_limit=alloc,
                                        seed=int(rng.randint(100000)),
                                        n_cand_start=30, n_cand_end=80,
                                        max_no_improve=200, n_binsearch=12)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            attempts = 0
        else:
            attempts += 1

    return best_h, best_c5
