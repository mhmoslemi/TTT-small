        #     i+k=j, i=j-k: contributes -x[j-k] (if j-k valid)
        # So d corr[s]/dx[j] = 1 - x[j+k] - x[j-k] (with appropriate boundary handling)

        # and let SLSQP use finite differences (slower but simpler).
        return None  # Let SLSQP use numerical gradient

    # Let me implement the gradient properly.
    
    def obj_grad_full(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            return smooth_max(corr, alpha), np.zeros(n)
        weights = weights / w_sum
        
        # corr[s] = sum_i x[i] * (1 - x[i+k_s]) where k_s = s - (n-1)
        # d corr[s] / d x[j] = (1 - x[j+k_s]) - x[j-k_s]  (with boundary handling)
        #   where x[j+k_s] is included only if 0 <= j+k_s < n
        #   and x[j-k_s] is included only if 0 <= j-k_s < n
        
        grad = np.zeros(n)
        idxs = np.arange(n)
        mid = n - 1
        # For efficiency, only process significant weights
        significant = weights > 1e-10

            w = weights[s]
            k = int(s) - mid
            # d corr[s]/dx[j] = (1-x[j+k]) - x[j-k]
            # For j where j+k is in [0,n): term (1-x[j+k])
            # For j where j-k is in [0,n): term -x[j-k]
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += w * (1.0 - x[idxs[v1] + k])
            grad[v2] -= w * x[idxs[v2] - k]
        
        val = smooth_max(corr, alpha)
        return val, grad
    
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            def _obj(x):
                return _obj_grad_full(x, alpha)[0]
            def _grad(x):
                return _obj_grad_full(x, alpha)[1]
            
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                     'jac': lambda x: np.ones(n)}]
            res = minimize(_obj, h_cur, jac=_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, cur_score

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradients(h, n):
    """Compute gradients of all correlations w.r.t. h."""
    # corr[s] = sum_i h[i] * (1 - h[i+k]) where k = s - (n-1)
    # d corr[s] / d h[j] = (1-h[j+k]) - h[j-k] (with boundary handling)
    mid = n - 1
    idxs = np.arange(n)
    grads = np.zeros((2*n-1, n))
    for s in range(2*n-1):
        k = s - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grads[s, v1] += (1.0 - h[idxs[v1] + k])
        grads[s, v2] -= h[idxs[v2] - k]
    return grads

def lp_direction(h, active_set, n):
    """Find descent direction using LP."""
    # Compute gradients for active set only
    mid = n - 1
    idxs = np.arange(n)
    g = np.zeros((len(active_set), n))
    for idx, s in enumerate(active_set):
        k = int(s) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        g[idx, v1] += (1.0 - h[idxs[v1] + k])
        g[idx, v2] -= h[idxs[v2] - k]
    
    # LP: min t
    # s.t. g[k] · d <= t for k in active_set
    #      sum d = 0
    #      -1 <= d[i] <= 1
    
    # Variables: d[0..n-1], t
    # c = [0, ..., 0, 1] (minimize t)
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    # Inequality constraints: A_ub @ [d; t] <= b_ub
    # g[k] · d - t <= 0 for each k
    A_ub = np.zeros((len(active_set), n + 1))
    A_ub[:, :n] = g
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(len(active_set))
    
    # Equality: sum d = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -1 <= d[i] <= 1, t <= 10 (upper bound on t)
    bounds = [(-1.0, 1.0)] * n + [(-10.0, 10.0)]
    
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def line_search(h, d, target, c5_current, max_steps=50):
    """Find best step size for direction d."""
    best_alpha = 0.0
    best_c5 = c5_current
    alpha = 0.1
    for _ in range(max_steps):
        if alpha < 1e-6:
            break
        h_new = h + alpha * d
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_alpha = alpha
            alpha *= 0.7
        else:
            alpha *= 0.5
    return best_alpha, best_c5, h if best_alpha == 0 else project_feasible(h + best_alpha * d, target)

Actually, I just realized something important. Let me re-examine the problem:

The overlap is ∫₀² h(x)(1-h(x+k)) dx.

But this integral is only over the region where both x and x+k are in [0,2]. For k > 0, this is x ∈ [0, 2-k]. For k < 0, this is x ∈ [-k, 2].

The length of the integration domain is 2-|k|. So for large |k|, the integral is automatically small (short domain). The maximum is typically achieved at small |k|.

This means the problem is "dominated" by the behavior of h at small shifts. The function h needs to be "self-similar" at small scales.

A function that is self-similar at all small scales is a constant (or a function with very low frequency content). But we need ∫h = 1, so the average is 1/2. A constant h = 1/2 gives:
c[k] = (1/2)(1/2) * (2-|k|) = (2-|k|)/4

The max is at k=0: c[0] = 1/2. That's 0.5, much worse than 0.38.

So we need non-constant h. The optimal h should balance:
- k=0: want h to be "peaky" (bimodal, close to 0 or 1)
- k=small: want h to be "smooth" (similar values nearby)

import numpy as np
from scipy.optimize import minimize, linprog
import time

_global_alpha = [100.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_global_alpha[0] * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / _global_alpha[0]

def _smooth_grad(x):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_global_alpha[0] * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return np.zeros(n)
    weights = weights / w_sum
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        if weights[s] < 1e-10:
            continue
        k = s - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += weights[s] * (1.0 - x[idxs[v1] + k])
        grad[v2] -= weights[s] * x[idxs[v2] - k]

def slsqp_polish(h_init, n, target, alphas, maxiter, deadline=None):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)

        try:

                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def sca_optimize(h_init, n, target, time_limit, max_iter=2000, tol_frac=0.02,
                 use_all_constraints_after=100):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_val = float(np.max(corr)) * (2.0 / n)

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        c_max = np.max(corr)
        tol = tol_frac * c_max if it < use_all_constraints_after else 0.0
        active = np.where(corr >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        n_act = len(active)
        A_ub = np.zeros((n_act, n + 1))
        b_ub = np.zeros(n_act)

        for ii, s in enumerate(active):
            k = int(s) - mid
            g = np.zeros(n)
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            g[v1] += (1.0 - h_cur[idxs[v1] + k])
            g[v2] -= h_cur[idxs[v2] - k]
            A_ub[ii, :n] = g
            A_ub[ii, n] = -1.0
            b_ub[ii] = np.dot(g, h_cur) - corr[s]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        c_vec = np.zeros(n + 1)
        c_vec[n] = 1.0
        bounds = [(0.0, 1.0)] * n + [(-50.0, 50.0)]

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        except Exception:
            break

        if res.status != 0:
            break

        h_lp = res.x[:n]
        h_lp = project_feasible(h_lp, target)

        best_alpha = 0.0
        best_trial = c5_val
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            h_trial = (1.0 - alpha) * h_cur + alpha * h_lp
            h_trial = project_feasible(h_trial, target)
            trial_c5 = compute_c5(h_trial)
            if trial_c5 < best_trial - 1e-14:
                best_trial = trial_c5
                best_alpha = alpha

        if best_alpha == 0.0:
            if it > 50:
                break
            continue

        h_cur = (1.0 - best_alpha) * h_cur + best_alpha * h_lp
        h_cur = project_feasible(h_cur, target)

    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=10, n_cand_end=50, max_no_improve=300,
            n_active_shifts=3, reheat_interval=100, reheat_scale=0.03):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0

    change = np.zeros((n, 2 * n - 1))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

            rng = np.random.RandomState(sweep * 13 + 7)
            n_pert = max(5, n // 50)

            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 1e-5:
                h = h_pert
                for p in range(n):
                    pos_mask = (p + shifts >= 0) & (p + shifts < n)
                    neg_mask = (p - shifts >= 0) & (p - shifts < n)
                    change[p, :] = 0.0
                    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                    change[p, neg_mask] -= h[p - shifts[neg_mask]]
                sweep_idx = int(np.argmax(np.correlate(h, 1.0 - h, mode='full')))
                sweeps_since_improve = 0
                continue

        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts

            grad[v2] -= w * h[idxs[v2] - k]

                    else:
                        a_d = mid_d

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 250:
            deadline = min(180, tleft() * 0.45)
            h_sca2, sc_sca2 = sca_optimize(best_h, n2, target2, time_limit=deadline,
                                            max_iter=2500, tol_frac=0.02)
            if sc_sca2 < best_score:
                best_score = sc_sca2
                best_h = h_sca2
                current_n = n2
                current_target = target2

        if tleft() > 100:
            deadline = time.time() + min(50, tleft() * 0.3)
            h_pol2, sc2 = slsqp_polish(best_h, current_n, current_target,
                                        [2000, 10000, 30000], 800, deadline=deadline)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2
                current_n = n2
                current_target = target2

    # Phase 5: SCA again at current resolution
    if tleft() > 200:
        deadline = min(120, tleft() * 0.45)
        h_sca3, sc_sca3 = sca_optimize(best_h, current_n, current_target,
                                        time_limit=deadline, max_iter=2000,
                                        tol_frac=0.01)
        if sc_sca3 < best_score:
            best_score = sc_sca3
            best_h = h_sca3

    # Phase 6: Pair CD for fine-tuning
    if tleft() > 150:
        h_cd, c5_cd = pair_cd(best_h, current_n, current_target,
                               time_limit=min(100, tleft() * 0.5),
                               max_sweeps=30000,
                               n_cand_start=15, n_cand_end=60,
                               max_no_improve=400,
                               n_active_shifts=4,
                               reheat_interval=100, reheat_scale=0.02)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: 1-h symmetry multi-start
    if tleft() > 150:
        h_inv = project_feasible(1.0 - best_h, current_target)
        sc_inv = compute_c5(h_inv)
        if sc_inv < best_score:
            best_score = sc_inv
            best_h = h_inv

        if tleft() > 100:
            deadline = min(60, tleft() * 0.4)
            h_sca_inv, sc_inv = sca_optimize(best_h, current_n, current_target,
                                              time_limit=deadline, max_iter=1500,
                                              tol_frac=0.02)
            if sc_inv < best_score:
                best_score = sc_inv
                best_h = h_sca_inv

    # Phase 8: Random perturbation + SCA
    if tleft() > 120:
        for attempt in range(3):
            if tleft() < 50:
                break
            noise_scale = [0.02, 0.05, 0.08][attempt]
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            deadline = min(40, tleft() * 0.35)
            h_ref, sc_ref = sca_optimize(h_pert, current_n, current_target,
                                          time_limit=deadline, max_iter=1000,
                                          tol_frac=0.03)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 9: Final SLSQP
    if tleft() > 40:
        deadline = time.time() + min(25, tleft() * 0.5)
        h_final, sc_final = slsqp_polish(best_h, current_n, current_target,
                                          [5000, 20000, 80000], 600, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 10: Final SCA polish
    if tleft() > 30:
        deadline = min(20, tleft() * 0.6)
        h_final2, sc_final2 = sca_optimize(best_h, current_n, current_target,
                                            time_limit=deadline, max_iter=500,
                                            tol_frac=0.005)
        if sc_final2 < best_score:
            best_score = sc_final2
            best_h = h_final2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
