    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP at initial resolution
    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000], 350, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: FISTA gradient descent
    if tleft() > 250:
        h_f, sc_f = fista_gradient(best_h, n0, target0,
                                   time_limit=min(120, tleft() * 0.35),
                                   alpha_start=100, alpha_end=50000,
                                   top_k=40, max_iter=10000,
                                   step_init=0.003, step_max=0.03)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f

    # Phase 3: Fast pair CD (main workhorse)
    if tleft() > 300:
        h_cd, c5_cd = fast_pair_cd(best_h, n0, target0,
                                   time_limit=min(280, tleft() * 0.5),
                                   max_sweeps=40000,
                                   n_cand_start=12, n_cand_end=28,
                                   max_no_improve=400,
                                   weight_beta=5.0, n_bisect=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Multi-start perturbation at current resolution
    if tleft() > 150:
        h_ms, sc_ms = multi_start_overlap(best_h, n0, target0,
                                          n_starts=3, time_per_start=25, seed=seed)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms
        if tleft() > 80:
            h_cd2, c5_cd2 = fast_pair_cd(best_h, n0, target0,
                                         time_limit=min(70, tleft() * 0.4),
                                         max_sweeps=15000,
                                         n_cand_start=14, n_cand_end=24,
                                         max_no_improve=200,
                                         weight_beta=6.0, n_bisect=14)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 5: Interpolate to higher resolution
    n1 = 1200
    target1 = n1 / 2.0
    if tleft() > 400:
        h1 = interpolate_to(best_h, n1, target1)
        cur_n = n1
        cur_target = target1

        if tleft() > 350:
            h_pol1, sc1 = slsqp_round(h1, cur_n, cur_target,
                                      [100, 500], 300, 1e-14)
            if sc1 < best_score:
                best_score = sc1
            best_h = h_pol1

        if tleft() > 250:
            h_f1, sc_f1 = fista_gradient(best_h, cur_n, cur_target,
                                         time_limit=min(100, tleft() * 0.3),
                                         alpha_start=100, alpha_end=30000,
                                         top_k=40, max_iter=6000,
                                         step_init=0.002, step_max=0.03)
            if sc_f1 < best_score:
                best_score = sc_f1
                best_h = h_f1

        if tleft() > 200:
            h_cd1, c5_cd1 = fast_pair_cd(best_h, cur_n, cur_target,
                                         time_limit=min(180, tleft() * 0.5),
                                         max_sweeps=25000,
                                         n_cand_start=12, n_cand_end=25,
                                         max_no_improve=300,
                                         weight_beta=5.0, n_bisect=14)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        if tleft() > 60:
            h_pol1b, sc1b = slsqp_round(best_h, cur_n, cur_target,
                                        [5000, 20000], 350, 1e-14)
            if sc1b < best_score:
                best_score = sc1b
                best_h = h_pol1b

    # Phase 6: Annealed pair CD to escape local minima
    if tleft() > 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_an, c5_an = fast_pair_cd(best_h, n_cur, target_cur,
                                   time_limit=min(120, tleft() * 0.5),
                                   max_sweeps=15000,
                                   n_cand_start=15, n_cand_end=35,
                                   max_no_improve=400,
                                   weight_beta=4.0, n_bisect=14,
                                   use_anneal=True, T0=0.0006)
        if c5_an < best_score:
            best_score = c5_an
            best_h = h_an

    # Phase 7: Final SLSQP high-alpha polish
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pol2, sc2 = slsqp_round(best_h, n_cur, target_cur,
                                  [10000, 50000], 500, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 8: Final fast pair CD
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin, c5_fin = fast_pair_cd(best_h, n_cur, target_cur,
                                     time_limit=min(50, tleft() * 0.6),
                                     max_sweeps=10000,
                                     n_cand_start=20, n_cand_end=40,
                                     max_no_improve=200,
                                     weight_beta=8.0, n_bisect=16)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    # Phase 9: Final SLSQP
    if tleft() > 35:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin2, sc_fin2 = slsqp_round(best_h, n_cur, target_cur,
                                      [100000], 300, 1e-14)
        if sc_fin2 < best_score:
            best_score = sc_fin2
            best_h = h_fin2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
