For a binary vector h with exactly n/2 ones, a swap (i: 1→0, j: 0→1) changes the correlation as follows:

For shift k (represented as index k_idx = k + (n-1) in the full correlation):
Δcorr[k_idx] = -(h[i-k_valid] terms) + (h[j-k_valid] terms)

More precisely, for the correlation c[k] = Σ_x h[x](1-h[x-k]) (using the convention where k is the shift):
When we change h[i] from 1 to 0: c[k] decreases by (1-h[i-k]) for valid k

So Δc[k] = (1-h[j-k]) - (1-h[i-k]) = h[i-k] - h[j-k] (for valid k where both i-k and j-k are in range)

This is a simple formula! For each shift k, the change is just h[i-k] - h[j-k] (where defined).

So if I maintain the current correlation, I can compute the new correlation after a swap in O(n) time.

The key question is: which swap (i,j) to try? I want the one that most reduces max(c).

A greedy approach:
1. Find k* = argmax(c)
2. Find the swap (i,j) that most reduces c[k*] while not increasing other c[k] too much
3. The change in c[k*] is h[i-k*] - h[j-k*]
4. To minimize c[k*], we want h[i-k*] = 0 and h[j-k*] = 1
5. So i should be at a position where h[i-k*]=0 (i.e., i-k* is a zero position) and j where h[j-k*]=1

This gives us a structured way to find good swaps!

Let me formalize:
- Let k* be the shift with maximum correlation
- We want to find i (currently 1) and j (currently 0) such that:
  - h[i-k*] = 0 (so the swap reduces c[k*] by 1)
  - h[j-k*] = 1 (so the swap doesn't increase c[k*])
  - The swap doesn't increase other correlations too much

The condition h[i-k*] = 0 means i = z + k* for some zero position z.
The condition h[j-k*] = 1 means j = o + k* for some one position o.

So the candidates are:
- i ∈ {z + k* : z is a zero position} ∩ {positions where h=1}
- j ∈ {o + k* : o is a one position} ∩ {positions where h=0}

For each such pair, the change in the full correlation vector is:
Δc[k] = h[i-k] - h[j-k] (for valid k)

We want to minimize the maximum of c + Δc.

    """Compute gradient of smooth max approximation."""
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(corr))
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= s

    """Compute gradient of weighted sum of correlations."""
    n = len(h)
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        wi = weights[k_idx]
        if wi < 1e-10:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    return grad

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def binary_cd(h_bin, n, time_limit, max_stalls=300, n_eval=30, top_k_shifts=3):
    start_time = time.time()
    target = n // 2
    c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    dx = 2.0 / n
    best_c5 = float(np.max(c)) * dx
    best_h = h_bin.copy()
    stalls = 0
    
    for it in range(10000000):
        if time.time() - start_time > time_limit:
            break
        
        c5_val = float(np.max(c)) * dx
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h_bin.copy()
            stalls = 0
        else:
            stalls += 1
            if stalls > max_stalls:
                break
        
        # Get top shifts
        top_idxs = np.argsort(c)[-top_k_shifts:]
        k_star_idx = int(top_idxs[-1])
        k_star = k_star_idx - (n - 1)
        
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find candidates that reduce the max shift
        # For k_star: Δc[k_star] = h[i-k_star] - h[j-k_star]
        # We want this to be negative (reduce c[k_star])
        # So we want h[i-k_star]=0 and h[j-k_star]=1
        
        i_cands = []
        for i in ones:
            ip = i - k_star
            if 0 <= ip < n and h_bin[ip] == 0:
                i_cands.append(i)
        
        j_cands = []
        for j in zeros:
            jp = j - k_star
            if 0 <= jp < n and h_bin[jp] == 1:
                j_cands.append(j)
        
        if not i_cands or not j_cands:
            # Fallback: use gradient-based selection
            grad = np.zeros(n)
            for s_idx in top_idxs:
                k = int(s_idx) - (n - 1)
                idxs = np.arange(n)
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += (1.0 - h_bin[idxs[v1] - k].astype(float))
                grad[v2] -= h_bin[idxs[v2] + k].astype(float)
            i_cands = list(np.argsort(grad)[-30:])
            j_cands = list(np.argsort(grad)[:30])
        
        # Evaluate a subset of pairs
        n_i = len(i_cands)
        n_j = len(j_cands)
        n_pairs = min(n_eval, n_i * n_j)
        
        if n_pairs == 0:
            stalls += 1
            continue
        
        rng = np.random.RandomState(it % 2**31)
        best_new_max = float(np.max(c))
        best_i = -1
        best_j = -1
        
        if n_i * n_j <= n_eval:
            pairs = [(i_cands[a], j_cands[b]) for a in range(n_i) for b in range(n_j)]
        else:
            pairs = [(i_cands[rng.randint(n_i)], j_cands[rng.randint(n_j)]) for _ in range(n_pairs)]
        
        for i, j in pairs:
            if i == j:
                continue
            # Compute change in correlation
            # Δc[k+n-1] = h[i-k] - h[j-k] for valid i-k, j-k
            delta = np.zeros(2 * n - 1)
            for k in range(k_star - n + 1, k_star + n):
                if k < -(n-1) or k > n-1:
                    continue
                val = 0.0
                ik = i - k
                jk = j - k
                if 0 <= ik < n:
                    val += h_bin[ik]
                if 0 <= jk < n:
                    val -= h_bin[jk]
                if val != 0:
                    delta[k + n - 1] = val
            
            new_max = float(np.max(c + delta))
            if new_max < best_new_max:
                best_new_max = new_max
                best_i = i
                best_j = j
        
        if best_i >= 0:
            h_bin[best_i] = 0
            h_bin[best_j] = 1
            c = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        else:
            stalls += 5
    
    return best_h, best_c5

def continuous_cd(h_init, n, target, time_limit, max_stalls=500, n_cand=40, top_k=40, n_shifts_grad=3):
    start_time = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stalls = 0
    
    for sweep in range(10000000):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stalls = 0
        else:
            stalls += 1
            if stalls > max_stalls:
                break
        
        k_star_idx = int(np.argmax(corr))
        
        # Gradient from top shifts
        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Score pairs by change in c[k_star]
        # For pair (i in high, j in low): Δc[k_star] = (1-h[j-k_star]) - (1-h[i-k_star]) = h[i-k_star] - h[j-k_star]
        # Wait, I need to be more careful.
        # c[k] = Σ_x h[x](1-h[x-k])
        # If we decrease h[i] by δ and increase h[j] by δ:
        # Δc[k] = -δ(1-h[i-k]) + δ(1-h[j-k]) = δ(h[i-k] - h[j-k])
        
        ch_i = np.zeros(len(high_set))
        ch_j = np.zeros(len(low_set))
        for idx, i in enumerate(high_set):
            ip = i - (int(top_shifts[-1]) - mid)  # i - k_star
            ch_i[idx] = h[ip] if 0 <= ip < n else 0.0
        for idx, j in enumerate(low_set):
            jp = j - (int(top_shifts[-1]) - mid)
            ch_j[idx] = h[jp] if 0 <= jp < n else 0.0
        
        # Score: for pair (i,j), Δc[k_star] = δ * (h[i-k_star] - h[j-k_star])
        # We want this negative, so h[i-k_star] < h[j-k_star]
        # The "greedy score" for reducing k_star is: h[j-k_star] - h[i-k_star]
        score_mat = ch_j[None, :] - ch_i[:, None]  # shape (n_high, n_low)
        # We want large score (reduces k_star)
        
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, -n_to_eval)[-n_to_eval:]
        
        # For each candidate pair, do a bisection line search
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for tf in top_flat:
            ii = int(tf // len(low_set))
            jj = int(tf % len(low_set))
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            
            # Compute the gradient of c w.r.t. delta for this pair
            # Δc[k] = δ * ((1-h[j-k]) - (1-h[i-k])) = δ * (h[i-k] - h[j-k])
            g = np.zeros(n_corr)
            k_star = int(top_shifts[-1]) - mid
            for k in range(-(n-1), n):
                ik = i - k
                jk = j - k
                val = 0.0
                if 0 <= ik < n:
                    val += h[ik]
                if 0 <= jk < n:
                    val -= h[jk]
                g[k + mid] = val
            
            # Bisection: find maximum δ such that max(c + δ*g) is minimized
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-12
                f_prime = np.max(g[active])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            
            approx_max = float(np.max(corr + delta_star * g)) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha, _g_target
    start_time = time.time()
    
    def time_left():
        return budget_s - 30.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    
    # Phase 1: Quick SLSQP at n0
    if time_left() > 60:
        h_pol = slsqp_optimize(best_h, n0, target0, [200, 1000, 5000], 400, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
    
    # Phase 2: Binary local search at n0
    if time_left() > 100:
        h_bin = (best_h > 0.5).astype(float)
        # Fix to have exactly n0/2 ones
        n_ones = int(np.sum(h_bin))
        if n_ones != n0 // 2:
            vals_sorted = np.argsort(best_h)
            if n_ones < n0 // 2:
                for idx in vals_sorted[::-1][n0//2 - n_ones:]:
                    h_bin[idx] = 1
            else:
                for idx in vals_sorted[:n_ones - n0//2]:
                    h_bin[idx] = 0
        h_bin = project_feasible(h_bin, target0)
        
        tl = time_left()
        h_b, c5_b = binary_cd(h_bin.astype(int).astype(float), n0, min(80, tl*0.5),
                             max_stalls=200, n_eval=40, top_k_shifts=3)
        # Convert binary to near-continuous
        h_cont = h_b.copy()
        h_cont[h_cont > 0.5] = 0.99
        h_cont[h_cont < 0.5] = 0.01
        h_cont = project_feasible(h_cont, target0)
        sc = compute_c5(h_cont)
        if sc < best_score:
            best_score = sc
            best_h = h_cont
    
    # Phase 3: Continuous CD at n0
    if time_left() > 80:
        tl = time_left()
        h_cd, c5_cd = continuous_cd(best_h, n0, target0, min(60, tl*0.6),
                                    max_stalls=400, n_cand=40, top_k=50, n_shifts_grad=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
    
    # Phase 4: SLSQP refinement at n0
    if time_left() > 50:
        h_pol = slsqp_optimize(best_h, n0, target0, [2000, 10000, 50000], 600, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
    
    # Phase 5: Upscale to n1
    if time_left() > 250:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
        n0 = n1
        target0 = target1
        
        # Binary search at n1
        if time_left() > 150:
            h_bin1 = (best_h > 0.5).astype(float)
            n_ones1 = int(np.sum(h_bin1))
            if n_ones1 != n1 // 2:
                vals_sorted = np.argsort(best_h)
                if n_ones1 < n1 // 2:
                    for idx in vals_sorted[::-1][n1//2 - n_ones1:]:
                        h_bin1[idx] = 1
                else:
                    for idx in vals_sorted[:n_ones1 - n1//2]:
                        h_bin1[idx] = 0
            
            tl = time_left()
            h_b1, c5_b1 = binary_cd(h_bin1.astype(int).astype(float), n1, min(100, tl*0.5),
                                    max_stalls=300, n_eval=50, top_k_shifts=3)
            h_cont1 = h_b1.copy()
            h_cont1[h_cont1 > 0.5] = 0.99
