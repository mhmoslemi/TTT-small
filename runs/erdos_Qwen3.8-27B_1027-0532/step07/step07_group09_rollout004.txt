def top_shift_equalization(h_init, n, target, time_limit, k_top=5, n_iter=500):
    """Try to equalize the top k shifts by solving a small LP at each step"""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))
        
        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()

        # Get top k shifts
        top_k_idx = np.argsort(corr)[-k_top:]
        c_top = corr[top_k_idx]
        c_max = np.max(c_top)
        
        # If already well-equalized, try reducing the common level
        spread = c_max - np.min(c_top)
        if spread < 1e-6:
            # All top shifts are equal, try to reduce them
            # Gradient of the average of top k
            avg_grad = np.zeros(n)
            for ki in top_k_idx:
                k = int(ki) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                avg_grad[v1] += (1.0 - h[idxs[v1] - k])
                avg_grad[v2] -= h[idxs[v2] + k]
            avg_grad /= k_top
            avg_grad -= np.mean(avg_grad)
            
            # Move in negative gradient direction
            step = 0.001
            h_new = project_feasible(h - step * avg_grad, target)
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5:
                h = h_new
            else:
                step *= 0.5
                h_new = project_feasible(h - step * avg_grad, target)
                if compute_c5(h_new) < best_c5:
                    h = h_new
        else:
            # Try to reduce the spread: increase the lower top-k shifts, decrease the max
            # Direction: for the max shift, move in -grad direction
            #           for other top-k shifts, move in -grad direction too
            # Combined direction that reduces all of them
            combined_grad = np.zeros(n)
            for ki in top_k_idx:
                k = int(ki) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                combined_grad[v1] += (1.0 - h[idxs[v1] - k])
                combined_grad[v2] -= h[idxs[v2] + k]
            combined_grad /= k_top
            combined_grad -= np.mean(combined_grad)
            
            # Line search
            for step in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                h_new = project_feasible(h - step * combined_grad, target)
                new_c5 = compute_c5(h_new)
                if new_c5 * dx < c5_val * dx - 1e-15:
                    h = h_new
                    break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):

        if abs(diff) < 1e-14:
            break

            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10

            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
