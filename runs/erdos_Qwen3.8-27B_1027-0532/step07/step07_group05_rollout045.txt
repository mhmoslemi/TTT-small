            grad[:n-ka] += w * (1.0 - h[ka:])

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def enhanced_cd(h_init, n, target, time_limit, max_sweeps=20000,
                n_active=3, n_cand_start=15, n_cand_end=50,
                max_no_improve=300, bisection_iters=12):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)

    def compute_change_row(p):
        row = np.zeros(n_corr)
        lags = np.arange(0, p + 1)
        row[mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        row[mid + lags2] -= h[p + lags2]
        return row

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        max_pairs = 150

        if n_h * n_l <= max_pairs:
            pairs = []
            for ii in range(n_h):
                for jj in range(n_l):
                    i_val = int(high_set[ii])
                    j_val = int(low_set[jj])
                    if i_val != j_val:
                        pairs.append((i_val, j_val))
        else:
            pairs = []
            for frac in np.linspace(0.0, 1.0, 12):
                hi = int(frac * (n_h - 1))
                lo = int(frac * (n_l - 1))
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pairs.append((iv, jv))
            while len(pairs) < max_pairs:
                ii = np.random.randint(n_h)
                jj = np.random.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pairs.append((iv, jv))

            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)

                if active_mask.any():
                    f_prime = np.max(g[active_mask])

                best_i = i
                best_j = j
                best_delta = delta_star

            h[best_j] += best_delta

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    _g_n = n
    _g_target = target
    _g_alpha_idx = [0]

    def _smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def _cons_fun(x):
        return np.sum(x) - _g_target

    def _cons_jac(x):
        return np.ones(_g_n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

        h_sub2, c5_sub2 = fast_subgradient(best_h, n0, target0,
                                            time_limit=min(150, tleft() * 0.5),
                                            alpha_start=50.0, alpha_end=200000.0,
                                            momentum=0.86, c5_every=5)
        if c5_sub2 < best_score:
            best_score = c5_sub2
            best_h = h_sub2

    if tleft() > 80:
        h_cd5, c5_cd5 = enhanced_cd(best_h, n0, target0,
                                     time_limit=min(60, tleft() * 0.5),
                                     max_sweeps=12000,
                                     n_active=4,
                                     n_cand_start=25, n_cand_end=55,
                                     max_no_improve=250, bisection_iters=10)
        if c5_cd5 < best_score:
            best_score = c5_cd5
            best_h = h_cd5

    if tleft() > 150:
        rng = np.random.RandomState(seed + 1)
        for scale in [0.05, 0.1, 0.15]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n0) * scale
            h_pert = project_feasible(h_pert, target0)
            h_re, sc_re = fast_subgradient(h_pert, n0, target0,
                                            time_limit=min(50, tleft() * 0.3),
                                            alpha_start=30.0, alpha_end=50000.0,
                                            momentum=0.84, c5_every=5)
            if sc_re < best_score:
                best_score = sc_re
                best_h = h_re

    if tleft() > 80:
        h_cd6, c5_cd6 = enhanced_cd(best_h, n0, target0,
                                     time_limit=min(60, tleft() * 0.6),
                                     max_sweeps=15000,
                                     n_active=3,
                                     n_cand_start=30, n_cand_end=60,
                                     max_no_improve=200, bisection_iters=10)
        if c5_cd6 < best_score:
            best_score = c5_cd6
            best_h = h_cd6

    if tleft() > 60:
        _g_n = n0
        _g_target = target0

        def _smooth_obj_final(x, alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

        def _cons_fun_f(x):
            return np.sum(x) - _g_target

        def _cons_jac_f(x):
            return np.ones(_g_n)

        bounds = [(0.0, 1.0)] * n0
        cons = [{'type': 'eq', 'fun': _cons_fun_f, 'jac': _cons_jac_f}]
        try:
            res = minimize(_smooth_obj_final, best_h, args=(10000.0,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target0)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
