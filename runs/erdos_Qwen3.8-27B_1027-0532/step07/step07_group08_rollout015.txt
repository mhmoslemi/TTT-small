def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(30):

            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    change[p, neg_mask] -= h[p - shifts[neg_mask]]

For a given t, check if there exists h such that:
- sum_i h[i](1-h[i+k]) ≤ t for all k
- sum h = n/2, 0 ≤ h ≤ 1

This feasibility check is still non-convex, but I can relax it by dropping some constraints.

Alternatively, I can use **penalty methods**: minimize the sum of violations of the overlap constraints.

f(h) ≈ f(h_prev) + ∇f(h_prev)^T (h - h_prev) + (h-h_prev)^T Q (h-h_prev)/2

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h[:] = target_sum / n
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted))) / alpha


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def projected_gradient(h_init, n, target, time_limit, deadline=None,
                       lr0=0.03, decay=0.9995, max_iter=200000,
                       n_active_frac=0.15, reheat_interval=800, reheat_scale=0.03):
    start_time = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    change = compute_change_matrix(h, n)
    lr = lr0
    stale = 0
    no_improve_count = 0

    for it in range(max_iter):
        if time.time() > time_limit:
            break
        if deadline is not None and time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        c5_val = c_max * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
            no_improve_count = 0
        else:
            stale += 1
            no_improve_count += 1

        if no_improve_count > 5000:
            if it % reheat_interval == 0:
                rng = np.random.RandomState(it * 13 + 7)
                n_pert = max(10, n // 30)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                h[pert_idx] += rng.randn(n_pert) * reheat_scale
                h = project_feasible(h, target)
                change = compute_change_matrix(h, n)
                no_improve_count = 0
                stale = 0
            else:
                break

        # Compute weighted gradient focusing on top shifts
        spread = c_max - float(np.min(corr))
        if spread < 1e-8:
            w = np.zeros(n_corr)
            w[np.argmax(corr)] = 1.0
        else:
            alpha_w = 30.0 / max(spread, 0.005)
            w = np.exp(alpha_w * (corr - c_max))
            n_top = max(3, int(n_corr * n_active_frac))
            top_idx = np.argsort(w)[-n_top:]
            w_other = np.zeros(n_corr)
            w_other[top_idx] = w[top_idx]
            w = w_other
            s = w.sum()
            if s > 0:
                w = w / s
            else:
                w[np.argmax(corr)] = 1.0

        grad = change.T @ w
        grad -= np.mean(grad)

        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)

        changed = np.where(np.abs(h_new - h) > 1e-10)[0]
        for idx in changed:
            update_change_row(change, idx, h_new, n, shifts)

        h = h_new
        lr *= decay

    return best_h, best_c5


def pair_cd(h_init, n, target, time_limit, deadline=None,
            max_sweeps=50000, n_cand_start=15, n_cand_end=80,
            max_no_improve=500, n_active_shifts=3,
            reheat_interval=150, reheat_scale=0.04):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    sweeps_since_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):

            break

        if c5_val < best_c5 - 1e-13:

            no_improve += 1
            sweeps_since_improve += 1

        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star_idx = int(np.argmax(corr))
        n_h = len(high_set)
        n_l = len(low_set)

                    best_j = j

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

def frank_wolfe_steps(h_init, n, target, time_limit, deadline=None,
                      max_iter=100000, gamma0=1.0, gamma_decay=0.9999):
    start_time = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    gamma = gamma0
    no_improve = 0

    for it in range(max_iter):
        if time.time() > time_limit:
            break
        if deadline is not None and time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        c5_val = c_max * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20000:
                break

        # Compute subgradient of max: pick active shifts (top fraction)
        spread = c_max - float(np.min(corr))
        if spread < 1e-8:
            n_act = 1
        else:
            n_act = max(3, int(n_corr * 0.1))
        top_k = np.argsort(corr)[-n_act:]
        w_act = corr[top_k]
        s_w = w_act.sum()
        if s_w > 0:
            w_act = w_act / s_w
        else:
            w_act = np.ones(n_act) / n_act

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, w_act):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        # Frank-Wolfe: find vertex g that minimizes grad·g
        # Vertex of {0<=g<=1, sum(g)=n/2} is: g[i]=1 for i in S, 0 otherwise, |S|=n/2
        # Min grad·g = sum of bottom n/2 entries of grad
        sorted_g = np.argsort(grad)
        S = set(sorted_g[:n // 2].tolist())
        g = np.zeros(n)
        for idx in S:
            g[idx] = 1.0

        # Step: h_new = (1-gamma)*h + gamma*g
        h_new = (1.0 - gamma) * h + gamma * g
        h_new = project_feasible(h_new, target)
        h = h_new
        gamma *= gamma_decay

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
