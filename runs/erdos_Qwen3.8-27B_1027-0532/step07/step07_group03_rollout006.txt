import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_gradient(h, k_star, n):
    mid = n - 1
    k = k_star - mid
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad

def get_gradient_weighted(h, k_stars, weights, n):
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(k_stars, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

def local_search_cd(h_init, n, target, time_limit, max_sweeps=50000,

    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)

    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    def recompute_change_row(i):
        pi = int(i)
        change[pi, :] = 0.0
        pos_i = (pi + shifts >= 0) & (pi + shifts < n)
        neg_i = (pi - shifts >= 0) & (pi - shifts < n)
        change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
        change[pi, neg_i] -= h[pi - shifts[neg_i]]

            if sc1_cp < best_score:
                best_score = sc1_cp
                best_h = h1_cp
                best_n = n1
                target0 = target1
            else:
                best_h = h1_cp if sc1_cp < sc1 else h1
                best_n = n1
                target0 = target1
            sc1 = compute_c5(best_h)

        if tleft() > 150:
            h1_ls, sc1_ls = local_search_cd(best_h, best_n, target0,
                                            time_limit=min(130, tleft() * 0.4),
                                            max_sweeps=30000,
                                            n_cand_start=8, n_cand_end=40,
                                            max_no_improve=200)
            if sc1_ls < best_score:
                best_score = sc1_ls
                best_h = h1_ls

        if tleft() > 80:
            h1_sls, sc1_sls = slsqp_refine(best_h, best_n, target0,
                                           time_limit=min(70, tleft() * 0.4),
                                           alphas=[1000, 5000], maxiter=500)
            if sc1_sls < best_score:
                best_score = sc1_sls
                best_h = h1_sls

    # Phase 6: Multi-start perturbation
    if tleft() > 120:
        rng = np.random.RandomState(seed)
        for attempt in range(4):
            if tleft() < 40:
                break
            noise_scale = 0.03 * (1.0 + 0.3 * attempt)
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_prt, sc_prt = local_search_cd(h_pert, best_n, target0,
                                            time_limit=min(25, tleft() * 0.25),
                                            max_sweeps=8000,
                                            n_cand_start=10, n_cand_end=30,
                                            max_no_improve=100)
            if sc_prt < best_score:
                best_score = sc_prt
                best_h = h_prt

    # Phase 7: Final local search refinement
    if tleft() > 60:
        h_fin, sc_fin = local_search_cd(best_h, best_n, target0,
                                        time_limit=min(50, tleft() * 0.5),
                                        max_sweeps=20000,
                                        n_cand_start=15, n_cand_end=70,
                                        max_no_improve=150)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 8: Final SLSQP polish
    if tleft() > 40:
        h_pol, sc_pol = slsqp_refine(best_h, best_n, target0,
                                     time_limit=min(30, tleft() * 0.6),
                                     alphas=[5000, 20000], maxiter=400)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    final_n = len(final_h)

    return final_h, final_score, final_n
