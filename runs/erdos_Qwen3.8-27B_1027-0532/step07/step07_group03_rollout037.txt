import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / len(h), 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def tile_template(template, p, n):
    return np.tile(template, n // p + 1)[:n]

def optimize_template(p, n_full, time_budget):
    target = p / 2.0
    best = np.full(p, 0.5)
    best_sc = compute_c5(tile_template(best, p, n_full))

    def obj(x, alpha):
        pc = np.zeros(p)
        for r in range(p):
            pc[r] = np.sum(x * (1.0 - np.roll(x, r)))
        m = np.max(pc)
        shifted = np.clip(alpha * (pc - m), -500.0, 0.0)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    bounds = [(0.0, 1.0)] * p
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(p)}]

    start = time.time()
    alphas = [30, 100, 300, 1000, 3000, 10000]
    for alpha in alphas:
        if time.time() - start > time_budget:
            break
        for rst in range(4):
            if time.time() - start > time_budget:
                break
            x0 = np.full(p, 0.5)
            if rst > 0:
                rng = np.random.RandomState(p * 1000 + rst)
                x0 = np.clip(x0 + rng.randn(p) * 0.2, 0, 1)
                x0 = project_feasible(x0, target)
            try:
                res = minimize(obj, x0, args=(alpha,), method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 400, 'ftol': 1e-15})
                x_new = project_feasible(res.x, target)
                h_full = tile_template(x_new, p, n_full)
                sc = compute_c5(h_full)
                if sc < best_sc:
                    best_sc = sc
                    best = x_new.copy()
            except Exception:
                pass

        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def cf(x):
        return np.sum(x) - target

    def cj(x):
        return np.ones(n)

        try:
            res = minimize(obj, h, args=(alpha,), method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,

                          options={'maxiter': maxiter, 'ftol': 1e-14})

            sc = compute_c5(h_new)
            if sc < best_sc:

                best = h_new.copy()

def coord_descent(h_init, n, target, time_limit, max_sweeps=20000,

                best_h = project_feasible(tile_template(tmpl, p, n0), target0)

    if tleft() > 150:
        h_fw, sc_fw = frank_wolfe(best_h, n0, target0,
                                   time_limit=min(120, tleft() * 0.35), n_active=1)
        if sc_fw < best_score:
            best_score = sc_fw
            best_h = h_fw

    if tleft() > 200:
        h_sls, sc_sls = smooth_slsqp(best_h, n0, target0,
                                      [100, 500, 2000], maxiter=800)
        if sc_sls < best_score:
            best_score = sc_sls
            best_h = h_sls

    if tleft() > 250:
        h_cp, sc_cp = cutting_plane(best_h, n0, target0,
                                     time_limit=min(200, tleft() * 0.5),
                                     max_iters=6000, tol=1e-9)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp

    if tleft() > 200:
        h_sls2, sc2 = smooth_slsqp(best_h, n0, target0,
                                    [5000, 20000, 80000], maxiter=1000)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_sls2

    if tleft() > 200:
        h_cd, sc_cd = coord_descent(best_h, n0, target0,
                                     time_limit=min(180, tleft() * 0.5),
                                     max_sweeps=20000, n_cand_start=15, n_cand_end=80)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 150:
        h_fw2, sc_fw2 = frank_wolfe(best_h, n0, target0,
                                     time_limit=min(100, tleft() * 0.3), n_active=3)
        if sc_fw2 < best_score:
            best_score = sc_fw2
            best_h = h_fw2

    if tleft() > 150:
        h_cp2, sc_cp2 = cutting_plane(best_h, n0, target0,
                                       time_limit=min(120, tleft() * 0.4),
                                       max_iters=5000, tol=1e-10)
        if sc_cp2 < best_score:
            best_score = sc_cp2
            best_h = h_cp2

    if tleft() > 120:
        h_sls3, sc3 = smooth_slsqp(best_h, n0, target0,
                                    [50000, 200000], maxiter=800)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_sls3

    if tleft() > 100:
        n_coarse = 100
        target_c = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_c)
        h_coarse_fb, sc_fb = frank_wolfe(h_coarse, n_coarse, target_c,
                                          time_limit=min(30, tleft() * 0.2), n_active=1)
        if sc_fb < best_score:
            best_score = sc_fb
            best_h = interpolate_to(h_coarse_fb, n0, target0)

    if tleft() > 80:
        h_cd2, sc_cd2 = coord_descent(best_h, n0, target0,
                                       time_limit=min(60, tleft() * 0.4),
                                       max_sweeps=15000, n_cand_start=20, n_cand_end=100)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    if tleft() > 60:
        h_cp3, sc_cp3 = cutting_plane(best_h, n0, target0,
                                       time_limit=min(40, tleft() * 0.4),
                                       max_iters=3000, tol=1e-10)
        if sc_cp3 < best_score:
            best_score = sc_cp3
            best_h = h_cp3

    if tleft() > 40:
        h_fin, sc_fin = smooth_slsqp(best_h, n0, target0,
                                      [100000, 500000], maxiter=500)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
