```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
```

```python
def frank_wolfe(h_init, n, target, time_limit, n_active=5, max_iter=200000):

    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find active shifts
        top_idx = np.argsort(corr)[-n_active:]
        c_max = corr[top_idx].max()
        weights = np.exp((corr[top_idx] - c_max) / (dx * 0.01))
        weights /= weights.sum()
        
        # Gradient
        grad = np.zeros(n)
        for s_idx, w in zip(top_idx, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        # Frank-Wolfe oracle
        sorted_idx = np.argsort(grad)
        s = np.zeros(n)
        s_int = int(target)
        s[sorted_idx[:s_int]] = 1.0
        frac = target - s_int
        if frac > 1e-10:
            s[sorted_idx[s_int]] = frac
        
        # Line search
        best_gamma = 2.0 / (t + 2.0)
        # Try a few step sizes
        for gamma in [1.0, 0.5, 0.3, 0.2, 2.0/(t+2.0)]:
            h_trial = (1 - gamma) * h + gamma * s

            h_trial = project_feasible(h_trial, target)

            if sc < c5_val:

                break
        
        h = (1 - best_gamma) * h + best_gamma * s
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    
    return best_h, best_c5
```

```python
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_eq(x):
    return np.sum(x) - _g_target

def cons_eq_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h_start, n, target, alphas, maxiter, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    h_cur = h_start.copy()
    cur = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                cur = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur

def compute_change(h, n):
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    change = np.zeros((n, 2*n-1))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    return change

def update_row(change, p, h, n):
    shifts = np.arange(-(n-1), n)
    change[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos] += (1.0 - h[p + shifts[pos]])
    change[p, neg] -= h[p - shifts[neg]]
```

**Alternative: Analytical delta computation**

For a given pair (i,j), the optimal delta minimizes f(δ) = max_s (c_s + δ * g_s) where c_s = corr[s] and g_s = G[i,j,s].

The minimum occurs at δ* where there exist s1 ≠ s2 such that:
c_{s1} + δ* * g_{s1} = c_{s2} + δ* * g_{s2} = max_s (c_s + δ* * g_s)
and g_{s1} < 0 < g_{s2} (or one of them is 0).

So δ* = (c_{s2} - c_{s1}) / (g_{s1} - g_{s2})

This is the intersection of the two lines that form the upper envelope at the minimum.

To find this analytically:
1. Compute the upper envelope of the lines {c_s + δ * g_s}
2. Find the vertex (minimum) of this envelope within [d_lo, d_hi]

The upper envelope of n lines can be computed in O(n log n) using the convex hull trick, but that's complex.

A simpler approach: the minimum of max_s (c_s + δ * g_s) is the smallest λ such that c_s + δ * g_s ≤ λ for all s. This is equivalent to:
δ * g_s ≤ λ - c_s for all s

If g_s > 0: δ ≤ (λ - c_s) / g_s
If g_s < 0: δ ≥ (λ - c_s) / g_s  (note: dividing by negative reverses inequality)
If g_s = 0: need c_s ≤ λ

So for a given λ, the feasible δ range is:
[max_{s: g_s<0} (λ - c_s)/g_s, min_{s: g_s>0} (λ - c_s)/g_s]

And we need this range to be non-empty. The minimum λ is when this range just closes.

This is essentially a 1D bisection on λ, which is the same complexity as the δ bisection. So no gain.

4. Update h, repeat
