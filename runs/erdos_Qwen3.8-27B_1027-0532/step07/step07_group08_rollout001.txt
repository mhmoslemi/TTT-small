import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

# Global variables for SLSQP
_g_n = 0
_g_target = 0.0

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if alpha < 10:
        return float(c_max)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def cd_refined(h_init, n, target, time_limit, max_sweeps=50000,
               n_cand_start=10, n_cand_end=80, max_no_improve=500,
               n_active_shifts_start=2, n_active_shifts_end=5,
               reheat_interval=80, reheat_scale=0.05, top_k_pairs=3):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if sweeps_since_improve > reheat_interval and sweeps_since_improve % 30 == 0:
            rng = np.random.RandomState(sweep * 13 + 7)
            n_pert = max(5, n // 30)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 2e-5:
                h = h_pert
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                continue

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        n_active = n_active_shifts_start + int((n_active_shifts_end - n_active_shifts_start) * progress)
        n_active = min(n_active, 6)

        top_k = np.argsort(corr)[-n_active:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active) / n_active
        else:
            weights = corr[top_k] / corr_top_sum

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star_idx = int(np.argmax(corr))
        n_h = len(high_set)
        n_l = len(low_set)
        n_pairs = min(n_h * n_l, 3000)
        pair_count = 0

        for ii in range(n_h):
            if pair_count >= n_pairs:
                break
            i = int(high_set[ii])
            for jj in range(n_l):
                if pair_count >= n_pairs:
                    break
                j = int(low_set[jj])
                if i == j:
                    continue
                pair_count += 1

                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def golden_section_cd(h_init, n, target, time_limit, max_sweeps=30000,
                      n_cand=40, max_no_improve=400,
                      n_active_shifts=3, reheat_interval=100, reheat_scale=0.04):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0

    change = compute_change_matrix(h, n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if sweeps_since_improve > reheat_interval and sweeps_since_improve % 25 == 0:
            rng = np.random.RandomState(sweep * 17 + 11)
            n_pert = max(8, n // 25)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 3e-5:
                h = h_pert
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                continue

        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            weights = corr[top_k] / corr_top_sum

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        n_pairs = min(n_h * n_l, 2500)
        pair_count = 0

        for ii in range(n_h):
            if pair_count >= n_pairs:
                break
            i = int(high_set[ii])
            for jj in range(n_l):
                if pair_count >= n_pairs:
                    break
                j = int(low_set[jj])
                if i == j:
                    continue
                pair_count += 1

                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]

                a = delta_lo
                b = delta_hi
                c1 = a + (1.0 / phi) * (b - a)
                c2 = a + (1.0 - 1.0 / phi) * (b - a)
                f1 = float(np.max(corr + c1 * g))
                f2 = float(np.max(corr + c2 * g))

                for _ in range(18):
                    if f1 < f2:
                        b = c2
                        c2 = c1
                        f2 = f1
                        c1 = a + (1.0 / phi) * (b - a)
                        f1 = float(np.max(corr + c1 * g))
                    else:
                        a = c1
                        c1 = c2
                        f1 = f2
                        c2 = a + (1.0 - 1.0 / phi) * (b - a)
                        f2 = float(np.max(corr + c2 * g))

                delta_star = 0.5 * (a + b)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def subgradient_method(h_init, n, target, time_limit, alpha0=0.2, alpha_min=0.003):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    stale = 0

    for step in range(300000):
        if time.time() - start > time_limit:
            break
        if alpha < alpha_min:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
            alpha = min(alpha * 1.08, alpha0)
        else:
            stale += 1
            alpha *= 0.97
            if stale > 600:
                break

        c_max = float(np.max(corr))
        threshold = c_max - 0.08 * max(c_max, 0.01)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        n_act = len(active)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = 1.0 / n_act
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: SLSQP at n=400 with aggressive alpha schedule
    if tleft() > 100:
        deadline = time.time() + min(60, tleft() * 0.35)
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [50, 200, 1000, 5000, 20000, 80000],
                                   600, 1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD at n=400 (quick)
    if tleft() > 200:
        h_cd, c5_cd = cd_refined(best_h, n0, target0,
                                 time_limit=min(60, tleft() * 0.2),
                                 max_sweeps=15000,
                                 n_cand_start=10, n_cand_end=40,
                                 max_no_improve=300,
                                 n_active_shifts_start=2, n_active_shifts_end=4,
                                 reheat_interval=60, reheat_scale=0.04)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Interpolate to n=800
    if tleft() > 400:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
        current_n = n1
        current_target = target1

        if tleft() > 300:
            deadline = time.time() + min(80, tleft() * 0.25)
            h_pol1, sc_pol1 = slsqp_optimize(best_h, n1, target1,
                                             [100, 500, 2000, 10000, 50000],
                                             800, 1e-15, deadline=deadline)
            if sc_pol1 < best_score:
                best_score = sc_pol1
                best_h = h_pol1

        if tleft() > 200:
            h_cd1, c5_cd1 = golden_section_cd(best_h, n1, target1,
                                              time_limit=min(150, tleft() * 0.4),
                                              max_sweeps=20000,
                                              n_cand=35,
                                              max_no_improve=400,
                                              n_active_shifts=3,
                                              reheat_interval=90, reheat_scale=0.05)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

    # Phase 4: More CD at n=800 with different parameters
    if tleft() > 250:
        h_cd2, c5_cd2 = cd_refined(best_h, current_n, current_target,
                                   time_limit=min(120, tleft() * 0.4),
                                   max_sweeps=25000,
                                   n_cand_start=15, n_cand_end=60,
                                   max_no_improve=500,
                                   n_active_shifts_start=3, n_active_shifts_end=5,
                                   reheat_interval=70, reheat_scale=0.045)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 5: Interpolate to n=1200
    if tleft() > 350:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
        current_n = n2
        current_target = target2

        if tleft() > 250:
            deadline = time.time() + min(80, tleft() * 0.25)
            h_pol2, sc_pol2 = slsqp_optimize(best_h, n2, target2,
                                             [200, 1000, 5000, 20000, 80000],
                                             600, 1e-15, deadline=deadline)
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_h = h_pol2

        if tleft() > 150:
            h_cd3, c5_cd3 = golden_section_cd(best_h, n2, target2,
                                              time_limit=min(120, tleft() * 0.5),
                                              max_sweeps=15000,
                                              n_cand=30,
                                              max_no_improve=400,
                                              n_active_shifts=3,
                                              reheat_interval=80, reheat_scale=0.04)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

    # Phase 6: Multi-start perturbation
    if tleft() > 150:
        for attempt in range(5):
            if tleft() < 50:
                break
            noise_scales = [0.02, 0.04, 0.08, 0.12, 0.18]
            noise_scale = noise_scales[attempt]
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(25, tleft() * 0.3)
            h_ref, sc_ref = slsqp_optimize(h_pert, current_n, current_target,
                                           [100, 500, 2000, 10000, 50000],
                                           400, 1e-15, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 7: Subgradient method
    if tleft() > 80:
        h_sg, c5_sg = subgradient_method(best_h, current_n, current_target,
                                         time_limit=min(50, tleft() * 0.5),
                                         alpha0=0.15, alpha_min=0.004)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: Final CD polish
    if tleft() > 100:
        h_cd_f, c5_f = golden_section_cd(best_h, current_n, current_target,
                                         time_limit=min(70, tleft() * 0.55),
                                         max_sweeps=12000,
                                         n_cand=45,
                                         max_no_improve=300,
                                         n_active_shifts=4,
                                         reheat_interval=60, reheat_scale=0.025)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    # Phase 9: Final SLSQP polish with high alpha
    if tleft() > 40:
        deadline = time.time() + min(30, tleft() * 0.6)
        h_final_pol, sc_final = slsqp_optimize(best_h, current_n, current_target,
                                               [5000, 20000, 100000, 300000],
                                               1000, 1e-15, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
