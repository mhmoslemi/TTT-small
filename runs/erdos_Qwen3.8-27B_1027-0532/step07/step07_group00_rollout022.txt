import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def prox_feasible(v, target_sum):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(v - tau, 0.0, 1.0)


def compute_gradient_all(h, p, n):
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(n_corr):
        w = p[s]
        if w < 1e-14:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad


def nesterov_smoothed(h_init, n, target, alpha_schedule, steps_per_alpha, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for alpha in alpha_schedule:
        if time.time() - start > time_limit:
            break
        v = h.copy()
        c = 0.0
        step = 0.5 / (alpha + 1.0)
        for st in range(steps_per_alpha):
            if time.time() - start > time_limit:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_val = float(np.max(corr)) * dx
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > steps_per_alpha:
                break
            corr_scaled = corr * dx
            max_c = corr_scaled.max()
            log_w = np.clip(alpha * (corr_scaled - max_c), -500.0, 0.0)
            exp_w = np.exp(log_w)
            p = exp_w / exp_w.sum()
            grad = compute_gradient_all(h, p, n)
            grad -= np.mean(grad)
            c_new = (1.0 + np.sqrt(1.0 + 4.0 * c * c)) / 2.0
            v = h + (c / c_new) * (h - v)
            h_trial = prox_feasible(v - step * grad, target)
            f_trial = compute_c5(h_trial)
            if f_trial > c5_val + 0.01:
                step *= 0.5
                h_trial = prox_feasible(v - step * grad, target)
            else:
                step = min(step * 1.03, 2.0 / (alpha + 1.0))
            h = h_trial
            c = c_new
    return best_h, best_c5


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def weighted_pair_cd_v2(h_init, n, target, time_limit, max_sweeps=20000,

    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        tw = min(top_k, n_corr)

        grad = np.zeros(n)

            k = int(s_idx) - mid

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_i = -1
        best_j = -1
        best_delta = 0.0

                a_d = delta_lo

def binary_anneal(h_init, n, target, time_limit, max_sweeps=50000, T0=0.05, T1=0.001):
    start = time.time()
    dx = 2.0 / n
    rng = np.random.RandomState(42)

    h_bin = (h_init > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target)
    n_ones = int(np.sum(h_bin))
    if n_ones != target:
        idx = np.where(h_bin > 0.5)[0]
        if n_ones > target:
            to_zero = idx[:n_ones - target]
            h_bin[to_zero] = 0.0
        else:
            to_one = np.where(h_bin <= 0.5)[0][:target - n_ones]
            h_bin[to_one] = 1.0

    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    no_improve = 0
    n_target = int(target)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break
        progress = min(float(sweep) / max_sweeps, 1.0)
        T = T0 * (T1 / T0) ** progress

        c5_val = compute_c5(h_bin)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h_bin[i] < h_bin[j]:
            i, j = j, i
        delta_lo = max(h_bin[i] - 1.0, -h_bin[j])
        delta_hi = min(h_bin[i], 1.0 - h_bin[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        delta = rng.uniform(delta_lo, delta_hi)
        h_trial = h_bin.copy()
        h_trial[i] -= delta
        h_trial[j] += delta
        h_trial = np.clip(h_trial, 0.0, 1.0)
        h_trial = project_feasible(h_trial, target)
        c5_trial = compute_c5(h_trial)
        if c5_trial < c5_val or rng.random() < np.exp(-(c5_trial - c5_val) / T):
            h_bin = h_trial

    return best_h, best_c5


def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha=400, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)

    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(99)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    change = compute_change_matrix(h, n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star = int(np.argmax(corr)) - mid
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        n_cand = 30
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_found = False
        for _ in range(10):
            i = int(high_set[rng.randint(n_cand)])
            j = int(low_set[rng.randint(n_cand)])
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-12:
                continue
            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(14):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                f_prime = np.max(g[active]) if active.any() else 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx = np.max(corr + delta_star * g) * dx
            if approx < c5_val - 1e-14:
                h[i] -= delta_star
                h[j] += delta_star
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
                best_found = True
                break
        if not best_found:
            no_improve += 5

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Nesterov smoothed descent at initial resolution
    if tleft() > 120:
        h_ne, sc_ne = nesterov_smoothed(best_h, n0, target0,
                                        [30, 100, 500, 2000, 8000],
                                        steps_per_alpha=800,
                                        time_limit=min(100, tleft() * 0.7))
        if sc_ne < best_score:
            best_score = sc_ne
            best_h = h_ne

    # Phase 2: Weighted pair CD
    if tleft() > 150:
        h_cd, c5_cd = weighted_pair_cd_v2(best_h, n0, target0,
                                          time_limit=min(130, tleft() * 0.6),
                                          max_sweeps=25000,
                                          n_cand_start=20, n_cand_end=60,
                                          max_no_improve=400,
                                          weight_beta=6.0, top_k=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP polish
    if tleft() > 100:
        h_sp, sc_sp = slsqp_polish(best_h, n0, target0,
                                   [500, 2000, 10000],
                                   maxiter_per_alpha=500, ftol=1e-14)
        if sc_sp < best_score:
            best_score = sc_sp
            best_h = h_sp

    # Phase 4: Multi-point CD (randomized pairs focused on top shift)
    if tleft() > 120:
        h_mp, c5_mp = multi_point_cd(best_h, n0, target0,
                                     time_limit=min(100, tleft() * 0.6),
                                     n_points=3, max_sweeps=15000)
        if c5_mp < best_score:
            best_score = c5_mp
            best_h = h_mp

    # Phase 5: Nesterov again with different alpha schedule
    if tleft() > 150:
        h_ne2, sc_ne2 = nesterov_smoothed(best_h, n0, target0,
                                          [100, 500, 2000, 10000, 30000],
                                          steps_per_alpha=600,
                                          time_limit=min(130, tleft() * 0.6))
        if sc_ne2 < best_score:
            best_score = sc_ne2
            best_h = h_ne2

    # Phase 6: Binary annealing for structure discovery
    if tleft() > 150:
        h_ba, c5_ba = binary_anneal(best_h, n0, target0,
                                    time_limit=min(120, tleft() * 0.7),
                                    max_sweeps=40000, T0=0.08, T1=0.0005)
        if c5_ba < best_score:
            best_score = c5_ba
            best_h = h_ba
        # Polish the binary result
        if tleft() > 80:
            h_pol_b, sc_pol_b = slsqp_polish(h_ba, n0, target0,
                                             [200, 1000, 5000],
                                             maxiter_per_alpha=400, ftol=1e-14)
            if sc_pol_b < best_score:
                best_score = sc_pol_b
                best_h = h_pol_b
            if tleft() > 60:
                h_cd_b, c5_cd_b = weighted_pair_cd_v2(h_pol_b, n0, target0,
                                                      time_limit=min(50, tleft() * 0.6),
                                                      max_sweeps=10000,
                                                      n_cand_start=20, n_cand_end=50,
                                                      max_no_improve=250,
                                                      weight_beta=6.0, top_k=35)
                if c5_cd_b < best_score:
                    best_score = c5_cd_b
                    best_h = h_cd_b

    # Phase 7: Interpolate to finer grid
    n1 = 1000
    target1 = n1 / 2.0
    if tleft() > 300:
        h1 = interpolate_to(best_h, n1, target1)
        s1 = compute_c5(h1)
        n_cur = n1
        target_cur = target1
        h_cur = h1
        if s1 < best_score:
            best_score = s1
            best_h = h1

        if tleft() > 250:
            h_ne1, sc_ne1 = nesterov_smoothed(h_cur, n_cur, target_cur,
                                              [50, 200, 1000, 5000],
                                              steps_per_alpha=500,
                                              time_limit=min(180, tleft() * 0.6))
            if sc_ne1 < best_score:
                best_score = sc_ne1
                best_h = h_ne1
                h_cur = h_ne1

        if tleft() > 150:
            h_cd1, c5_cd1 = weighted_pair_cd_v2(h_cur, n_cur, target_cur,
                                                time_limit=min(130, tleft() * 0.6),
                                                max_sweeps=15000,
                                                n_cand_start=20, n_cand_end=55,
                                                max_no_improve=300,
                                                weight_beta=6.0, top_k=40)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                h_cur = h_cd1

        if tleft() > 80:
            h_sp1, sc_sp1 = slsqp_polish(h_cur, n_cur, target_cur,
                                         [2000, 10000, 30000],
                                         maxiter_per_alpha=400, ftol=1e-14)
            if sc_sp1 < best_score:
                best_score = sc_sp1
                best_h = h_sp1

    # Phase 8: Multi-start perturbation
    if tleft() > 200:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.10]:
            if tleft() < 100:
                break
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, target_cur)
            h_ref, sc_ref = slsqp_polish(h_pert, n_cur, target_cur,
                                         [100, 500, 3000],
                                         maxiter_per_alpha=300, ftol=1e-14)
            if tleft() > 60:
                h_cd_ref, c5_cd_ref = weighted_pair_cd_v2(h_ref, n_cur, target_cur,
                                                          time_limit=min(50, tleft() * 0.4),
                                                          max_sweeps=8000,
                                                          n_cand_start=20, n_cand_end=50,
                                                          max_no_improve=200,
                                                          weight_beta=6.0, top_k=35)
                if c5_cd_ref < best_score:
                    best_score = c5_cd_ref
                    best_h = h_cd_ref
            elif sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 9: Final Nesterov + CD polish
    if tleft() > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin_ne, sc_fin_ne = nesterov_smoothed(best_h, n_cur, target_cur,
                                                [5000, 20000, 50000],
                                                steps_per_alpha=400,
                                                time_limit=min(60, tleft() * 0.5))
        if sc_fin_ne < best_score:
            best_score = sc_fin_ne
            best_h = h_fin_ne

    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin_cd, c5_fin_cd = weighted_pair_cd_v2(best_h, n_cur, target_cur,
                                                  time_limit=min(50, tleft() * 0.7),
                                                  max_sweeps=10000,
                                                  n_cand_start=30, n_cand_end=60,
                                                  max_no_improve=200,
                                                  weight_beta=8.0, top_k=50)
        if c5_fin_cd < best_score:
            best_score = c5_fin_cd
            best_h = h_fin_cd

    # Phase 10: Final SLSQP
    if tleft() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final_sp, sc_final_sp = slsqp_polish(best_h, n_cur, target_cur,
                                               [20000, 50000, 100000],
                                               maxiter_per_alpha=300, ftol=1e-15)
        if sc_final_sp < best_score:
            best_score = sc_final_sp
            best_h = h_final_sp

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
