    h_opt = np.clip(h_opt, 0, 1)
    return h_opt

def run_until_improvement(h_init, best_score, iterations=100, tolerance=1e-6):
    """Perform iterative optimization with adaptive stopping"""
    h_cur = h_init.copy()
    score_cur = best_score
    
    for _ in range(iterations):
        # Perform local optimization
        h_new = cutting_plane_method(h_cur, len(h_cur), len(h_cur)/2, time_limit=50)[0]
        h_new = project_feasible(h_new, len(h_new)/2)
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new solution
        score_new = compute_c5(h_new)
        if score_new < score_cur - tolerance:
            score_cur = score_new
            h_cur = h_new.copy()
        
        # Early termination if convergence
        if abs(score_cur - previous_score) < tolerance:
            break
        
        previous_score = score_cur
        
    return h_cur, score_cur

def main_optimization(time_budget=1300):

    remaining_time = time_budget - time.time() + start_time
