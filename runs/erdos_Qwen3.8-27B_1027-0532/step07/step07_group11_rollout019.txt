                approx_max = np.max(corr + delta_star * g_vec) * dx

                accept = False
                if approx_max < best_trial - 1e-15:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 2000.0
                    if np.random.random() < np.exp(-delta_obj / max(temp, 1e-15)):
                        accept = True

                if accept and approx_max < best_trial + 1e-12:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(ch, best_i, h, n, shifts)
            update_change_row(ch, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 2:
                ch = change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.93, 1e-12)

    return best_h, best_c5


def subgrad_weighted(h_init, n, target, time_limit, n_steps=50000,
                     n_active=4, step_scale=0.004, decay=0.9998):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        top_k = np.argsort(corr)[-n_active:]
        sw = corr[top_k]
        sw_sum = sw.sum()
        if sw_sum < 1e-15:
            sw = np.ones(n_active) / n_active
        else:
            sw = sw / sw_sum

        grad = np.zeros(n)
        for si in range(n_active):
            k = int(top_k[si]) - mid
            w = sw[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            continue
        grad = grad / grad_norm

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
            eta *= 1.02
            eta = min(eta, step_scale * 3)
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.92
        eta = max(eta, 1e-6)

    return best_h, best_c5


def binary_local_search(h_init, n, target, time_limit, n_iters=20000):
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    idxs = np.arange(n)
    mid = n - 1

    ones_mask = h > 0.5
    if ones_mask.sum() != n // 2:
        sorted_h = np.argsort(h)
        ones_mask[sorted_h[-n // 2:]] = True
        ones_mask[sorted_h[:n - n // 2]] = False

    top_k = np.argsort(np.correlate(h, 1.0 - h, mode='full'))[-3:]
    active_shifts = [int(s) - mid for s in top_k]

    for it in range(n_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = floats(np.max(corr)) * (2.0 / n)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        top_k = np.argsort(corr)[-4:]
        active_shifts = [int(s) - mid for s in top_k]

        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(~ones_mask)[0]
        if len(ones_idx) < 2 or len(zeros_idx) < 2:
            break

        gain = np.zeros(len(ones_idx))
        for ak in active_shifts:
            for oi in range(len(ones_idx)):
                p = ones_idx[oi]
                q = p + ak
                if 0 <= q < n and ones_mask[q]:
                    gain[oi] += 1.0
                q2 = p - ak
                if 0 <= q2 < n and ones_mask[q2]:
                    gain[oi] += 1.0

        remove_candidate = ones_idx[np.argmax(gain)]

        gain_z = np.zeros(len(zeros_idx))
        for ak in active_shifts:
            for zi in range(len(zeros_idx)):
                p = zeros_idx[zi]
                q = p + ak
                if 0 <= q < n and ones_mask[q]:
                    gain_z[zi] += 1.0
                q2 = p - ak
                if 0 <= q2 < n and ones_mask[q2]:
                    gain_z[zi] += 1.0

        add_candidate = zeros_idx[np.argmax(gain_z)]

        if gain_z[np.argmax(gain_z)] > gain[np.argmax(gain)] - 0.5:
            ones_mask[remove_candidate] = False
            ones_mask[add_candidate] = True
            h[remove_candidate] = 0.0
            h[add_candidate] = 1.0

    best_h = project_feasible(best_h, target)
    return project_feasible(h, target), best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [100, 500, 2000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 180:
        h_cd, c5_cd = pair_cd_v2(best_h, cur_n, cur_target,
                                 time_limit=min(150, tleft() * 0.55),
                                 max_sweeps=50000,
                                 n_cand_start=20, n_cand_end=130,
                                 n_shifts=4, max_no_improve=350,
                                 anneal_temp=0.015, anneal_decay=0.9996,
                                 bisect_iters=12)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 50:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [2000, 8000], 600, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:

    if tleft() > 300:

        if tleft() > 250:

                                       n_shifts=4, max_no_improve=350,
                                       anneal_temp=0.008, anneal_decay=0.9997,
                                       bisect_iters=12)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                cur_n = n1
                cur_target = target1
            elif sc1 < best_score:
                best_score = sc1
                best_h = h1
                cur_n = n1
                cur_target = target1

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [3000, 12000], 800, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_cd2, c5_cd2 = pair_cd_v2(best_h, cur_n, cur_target,
                                   time_limit=min(120, tleft() * 0.6),
                                   max_sweeps=30000,
                                   n_cand_start=30, n_cand_end=120,
                                   n_shifts=5, max_no_improve=300,
                                   anneal_temp=0.005, anneal_decay=0.9998,
                                   bisect_iters=12)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if tleft() > 200:
            h_cd2b, c5_cd2b = pair_cd_v2(h2, n2, target2,
                                         time_limit=min(180, tleft() * 0.55),

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
