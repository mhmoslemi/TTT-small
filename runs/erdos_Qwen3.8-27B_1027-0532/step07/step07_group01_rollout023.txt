def smoothed_subgrad_cd(h, n, target, time_limit, alpha_start=100, alpha_end=10000, 
                        n_iters=50000, ftol=1e-12):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    t = 1.0  # FISTA momentum
    h_prev = h.copy()
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        if it % 10000 == 0:
            alpha = alpha_start * (alpha_end / alpha_start) ** (it / n_iters)
        else:
            alpha = alpha_start * (alpha_end / alpha_start) ** (it / n_iters)
        
        # FISTA update
        if it > 0:
            t_prev = t
            t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = h + (t_prev - 1.0) / (t + 1.0) * (h - h_prev)
        else:
            y = h.copy()
            t = 1.0
        
        # Compute smooth gradient
        corr = np.correlate(y, 1.0 - y, mode='full')
        shifted = alpha * (corr - np.max(corr))
        shifted = np.clip(shifted, -500, 0)
        p = np.exp(shifted)
        p /= p.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for s_idx in range(n_corr):
            if p[s_idx] < 1e-15:
                continue
            k = s_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += p[s_idx] * (1.0 - y[idxs[v1] - k])
            grad[v2] -= p[s_idx] * y[idxs[v2] + k]
        
        # Project gradient (zero mean for sum constraint)
        grad -= np.mean(grad)
        
        # Step size (backtracking or fixed)
        # Lipschitz bound: L ≈ alpha * max_k ||∇g_k||^2 * dx
        # ||∇g_k||^2 ≤ 4n (each component is in [-1,1], n components)
        # So L ≈ alpha * 4 * n * dx = alpha * 8
        L = alpha * 8.0
        step = 1.0 / L
        
        # Gradient step
        h_new = y - step * grad
        h_prev = h.copy()
        
        # Project onto feasible set
        h = project_feasible(h_new, target)
        
        # Update best
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

For all pairs (i, j) where i ∈ high_set, j ∈ low_set:
g[i,j] = change[j] - change[i]

decrease[i,j] = max(corr) - max(new_corr[i,j])

I want the pair with the largest decrease.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})

            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def fast_cd(h_init, n, target, time_limit, n_cand=60, step_init=0.03,

        else:

                if step < 1e-5:
                    break
                max_no_improve = int(max_no_improve * 1.5)
                no_improve = 0
        
        # Compute gradient at active shift
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

            continue

        best_j = -1

                f_mid = np.max(vals)

        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        c5_val = np.max(corr) * dx
        
        if c5_val < best_c5 - 1e-14:

            best_h = h.copy()

        shifted = alpha * (corr - np.max(corr))
        shifted = np.clip(shifted, -500.0, 0.0)
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            break
        p /= p_sum
        
        # Weighted gradient
        grad = np.zeros(n)
        # Only process significant weights
        thresh = 1e-8
        active = p > thresh
        for s_idx in np.where(active)[0]:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        grad -= np.mean(grad)
        
        # Adaptive step
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        new_c5 = compute_c5(h_new)
        
        if new_c5 < c5_val - 1e-15:
            h = h_new
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.7
            if step < 1e-6:
                break
    
    return best_h, best_c5

    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_cd(h_init, n, target, time_limit, n_cand=60, step_init=0.03,
            screen_k=10, multi_step_scales=None):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    if multi_step_scales is None:
        multi_step_scales = [1.0, 3.0, 10.0]

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = step_init
    no_improve = 0
    rng = np.random.RandomState(12345)

    change = compute_change_matrix(h, n)

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        corr_max = np.max(corr)
        c5_val = corr_max * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            step = min(step * 1.02, 0.15)
        else:
            no_improve += 1
            if no_improve > 500:
                step *= 0.5
                if step < 5e-6:
                    break
                no_improve = 0

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_h = len(high_set)
        n_l = len(low_set)

        candidates = []
        for ss in multi_step_scales:
            s = step * ss
            for ii in range(n_h):
                i = int(high_set[ii])
                if h[i] < 1e-10:
                    continue
                ci = change[i]
                for jj in range(n_l):
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    max_delta = min(h[i], 1.0 - h[j])
                    delta = min(s, max_delta)
                    if delta < 1e-12:
                        continue
                    g = change[j] - ci
                    new_max = np.max(corr + delta * g)
                    decrease = corr_max - new_max
                    if decrease > 1e-10:
                        candidates.append((decrease, i, j, max_delta))

        if not candidates:
            if no_improve > 200 and no_improve % 100 == 50:
                ri = int(rng.randint(n_h + n_l))
                if ri < n_h:
                    i = int(high_set[ri])
                else:
                    i = int(low_set[ri - n_h])
                if h[i] > 1e-10:
                    j = int(low_set[0]) if h[i] > 0.5 else int(high_set[0])
                    if i != j and h[i] > h[j]:
                        delta = min(0.01, h[i], 1.0 - h[j])
                        if delta > 1e-10:
                            h[i] -= delta
                            h[j] += delta
                            h = project_feasible(np.clip(h, 0, 1), target)
                            update_change_row(change, i, h, n, shifts)
                            update_change_row(change, j, h, n, shifts)
            step *= 0.8
            continue

        candidates.sort(reverse=True)
        top = candidates[:screen_k]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_dec = 0.0

        for _, i, j, max_delta in top:
            delta_lo = 0.0
            delta_hi = max_delta
            ci = change[i]
            g = change[j] - ci
            for _ in range(22):
                mid_d = 0.5 * (delta_lo + delta_hi)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-11:
                    delta_hi = mid_d
                else:
                    delta_lo = mid_d
            delta_star = 0.5 * (delta_lo + delta_hi)
            if delta_star < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            dec = corr_max * dx - approx_max
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
                best_dec = dec

        if best_i >= 0 and best_dec > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            step *= 0.75

    return best_h, best_c5


def subgradient_pass(h_init, n, target, time_limit, alpha_start=50, alpha_end=5000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.05
    it = 0

    while time.time() - start < time_limit:
        it += 1
        progress = min(float(it) / 15000.0, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = np.max(corr) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        shifted = alpha * (corr - np.max(corr))
        shifted = np.clip(shifted, -500.0, 0.0)
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            break
        p /= p_sum

        grad = np.zeros(n)
        active = p > 1e-7
        active_idxs = np.where(active)[0]
        for s_idx in active_idxs:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break

        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        new_c5 = compute_c5(h_new)

        if new_c5 < c5_val - 1e-15:
            h = h_new
            step = min(step * 1.03, 0.3)
        else:
            step *= 0.7
            if step < 1e-7:
                break

    return best_h, best_c5


def multi_shift_cd(h_init, n, target, time_limit, n_cand=50, n_shifts=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    step = 0.02

    change = compute_change_matrix(h, n)

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        corr_max = np.max(corr)
        c5_val = corr_max * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                step *= 0.5
                if step < 5e-6:
                    break
                no_improve = 0

        top_shifts = np.argsort(corr)[-n_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[int(s_idx)] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        candidates = []
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            if h[i] < 1e-10:
                continue
            ci = change[i]
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                max_delta = min(h[i], 1.0 - h[j])
                delta = min(step, max_delta)
                if delta < 1e-12:
                    continue
                g = change[j] - ci
                new_max = np.max(corr + delta * g)
                decrease = corr_max - new_max
                if decrease > 1e-10:
                    candidates.append((decrease, i, j, max_delta))

        if not candidates:
            step *= 0.7
            continue

        candidates.sort(reverse=True)
        top = candidates[:8]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for _, i, j, max_delta in top:
            delta_lo = 0.0
            delta_hi = max_delta
            ci = change[i]
            g = change[j] - ci
            for _ in range(20):
                mid_d = 0.5 * (delta_lo + delta_hi)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-11:
                    delta_hi = mid_d
                else:
                    delta_lo = mid_d
            delta_star = 0.5 * (delta_lo + delta_hi)
            if delta_star < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            step *= 0.75

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [50, 200, 1000, 5000],
                                500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        h_sg, sc_sg = subgradient_pass(best_h, current_n, current_target,
                                        time_limit=min(80, tleft() * 0.35),
                                        alpha_start=50, alpha_end=5000)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if tleft() > 350:
        h_cd, c5_cd = fast_cd(best_h, current_n, current_target,
                              time_limit=min(380, tleft() * 0.55),
                              n_cand=55, step_init=0.03, screen_k=12,
                              multi_step_scales=[1.0, 4.0])
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol2, sc2 = slsqp_round(best_h, current_n, current_target,
                                   [5000, 20000, 80000],
                                   800, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 250:
        h_cd2, c5_cd2 = fast_cd(best_h, current_n, current_target,
                                 time_limit=min(220, tleft() * 0.55),
                                 n_cand=70, step_init=0.01, screen_k=15,
                                 multi_step_scales=[1.0, 3.0])
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 120:
        h_ms, c5_ms = multi_shift_cd(best_h, current_n, current_target,
                                      time_limit=min(100, tleft() * 0.5),
                                      n_cand=45, n_shifts=5)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if tleft() > 350:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        best_h_hi = h1
        best_score_hi = sc1

        if tleft() > 100:
            h1_sg, sc1_sg = subgradient_pass(h1, n1, target1,
                                              time_limit=min(60, tleft() * 0.25),
                                              alpha_start=50, alpha_end=3000)
            if sc1_sg < best_score_hi:
                best_score_hi = sc1_sg
                best_h_hi = h1_sg

        if tleft() > 150:
            h1_cd, c5_1_cd = fast_cd(best_h_hi, n1, target1,
                                      time_limit=min(140, tleft() * 0.45),
                                      n_cand=45, step_init=0.02, screen_k=8,
                                      multi_step_scales=[1.0, 3.0])
            if c5_1_cd < best_score_hi:
                best_score_hi = c5_1_cd
                best_h_hi = h1_cd

        if best_score_hi < best_score:
            best_score = best_score_hi
            best_h = best_h_hi
            current_n = n1
            current_target = target1
            if tleft() > 50:
                h_pol3, sc3 = slsqp_round(best_h, n1, target1,
                                           [10000, 50000],
                                           600, 1e-15)
                if sc3 < best_score:
                    best_score = sc3
                    best_h = h_pol3

    if tleft() > 100:
        h_final_cd, c5_final = fast_cd(best_h, current_n, current_target,
                                        time_limit=min(80, tleft() * 0.6),
                                        n_cand=65, step_init=0.005, screen_k=12,
                                        multi_step_scales=[1.0, 2.0])
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if tleft() > 40:
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                             [50000, 200000],
                                             600, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
