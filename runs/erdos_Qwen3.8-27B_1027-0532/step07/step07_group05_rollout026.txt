import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n):
    h = np.clip(h, 0.0, 1.0)
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        k_min = max(0, p - (n - 1))
        k_max = min(n - 1, p + (n - 1))
        for q in range(k_min, k_max + 1):
            k = q - p
            change[p, mid + k] = (1.0 - h[q]) - h[2 * p - q] if 0 <= 2 * p - q < n else (1.0 - h[q])
    # Recompute properly
    change = np.zeros((n, n_corr))
    for p in range(n):
        for k in range(-(n - 1 - p), p + 1):
            i1 = p - k
            i2 = p + k
            if 0 <= i1 < n and 0 <= i2 < n:
                change[p, mid + k] = (1.0 - h[i1]) - h[i2]
    return change

def make_change_matrix_v2(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    p = np.arange(n)
    for k in range(-(n - 1), n):
        i1 = p - k
        i2 = p + k
        valid = (i1 >= 0) & (i1 < n) & (i2 >= 0) & (i2 < n)
        if valid.any():
            cols = np.full(valid.sum(), mid + k, dtype=int)
            rows = np.where(valid)[0]
            change[rows, mid + k] = (1.0 - h[i1[valid]]) - h[i2[valid]]
    return change

def update_change_after_swap(change, h, i, j, n):
    n_corr = 2 * n - 1
    mid = n - 1
    # h[i] goes 1->0 (delta=-1), h[j] goes 0->1 (delta=+1)
    rows = np.arange(n)
    # For position i with delta=-1:
    # change[q, mid+(q-i)] -= -1 → += 1
    # change[q, mid+(i-q)] += -1 → -= 1
    cols1 = mid + rows - i
    cols2 = mid + i - rows
    valid1 = (cols1 >= 0) & (cols1 < n_corr)
    valid2 = (cols2 >= 0) & (cols2 < n_corr)
    change[rows[valid1], cols1[valid1]] += 1.0
    change[rows[valid2], cols2[valid2]] -= 1.0
    # For position j with delta=+1:
    cols3 = mid + rows - j
    cols4 = mid + j - rows
    valid3 = (cols3 >= 0) & (cols3 < n_corr)
    valid4 = (cols4 >= 0) & (cols4 < n_corr)
    change[rows[valid3], cols3[valid3]] -= 1.0
    change[rows[valid4], cols4[valid4]] += 1.0

def binary_ls(h_init, n, time_limit, n_active=4, n_cand=40, n_zero_sample=60,
              max_iters=500000, kick_interval=500, kick_size=3, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1

    h = h_init.copy()
    corr = compute_corr(h).astype(float)
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()

    change = make_change_matrix_v2(h, n)

    no_improve = 0
    total_iters = 0

    for it in range(max_iters):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        total_iters += 1

        # Periodic kick to escape local optima
        if kick_interval > 0 and it > 0 and it % kick_interval == 0 and no_improve > 30:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) >= kick_size and len(zeros) >= kick_size:
                swap_ids = rng.choice(ones, kick_size, replace=False)
                swap_jds = rng.choice(zeros, kick_size, replace=False)
                for si, sj in zip(swap_ids, swap_jds):
                    h[si] = 0.0
                    h[sj] = 1.0
                    corr += -change[si] + change[sj]
                    update_change_after_swap(change, h, si, sj, n)
            no_improve = 0
            continue

        # Find top lags
        top_idx = np.argsort(corr)[-n_active:]
        max_val = float(corr[top_idx[-1]])

        if max_val <= 0:
            break

        # Get positions of top lags
        top_lag_vals = corr[top_idx]
        # Second max (for when the max position gets reduced)
        sorted_corr = np.sort(corr)[::-1]
        second_max = float(sorted_corr[1]) if n_corr > 1 else 0.0

        # Generate candidates: positions i where h[i]=1 that contribute to top lags
        candidate_ones = set()
        for lag_pos in top_idx:
            k = int(lag_pos) - mid
            if k == 0:
                continue
            if k > 0:
                mask = (h[:n - k] == 1.0) & (h[k:] == 0.0)
                contrib = np.where(mask)[0]
            else:
                kk = -k
                mask = (h[kk:] == 1.0) & (h[:n - kk] == 0.0)
                contrib = np.where(mask)[0] + kk
            if len(contrib) > 0:
                if len(contrib) > n_cand:
                    contrib = contrib[rng.choice(len(contrib), n_cand, replace=False)]
                candidate_ones.update(contrib.tolist())

        candidate_ones = list(candidate_ones)
        if not candidate_ones:
            no_improve += 1
            if no_improve > 200:
                break
            continue

        # Sample zeros
        zeros = np.where(h == 0.0)[0]
        if len(zeros) > n_zero_sample:
            zero_sample = zeros[rng.choice(len(zeros), n_zero_sample, replace=False)]
        else:
            zero_sample = zeros

        best_pair = None
        best_new_max = max_val

        # Evaluate candidate swaps
        for i in candidate_ones:
            if h[i] != 1.0:
                continue
            ci = change[i]
            for j in zero_sample:
                if j == i:
                    continue
                # Delta = -ci + change[j]
                delta = -ci + change[j]
                nz = np.where(delta != 0)[0]
                if len(nz) == 0:
                    continue
                # New values at affected positions
                new_at_nz = corr[nz] + delta[nz]
                nz_max = float(np.max(new_at_nz))
                # New overall max: if the current max position is in nz, use second_max
                if nz[-1] in top_idx or any(pos in nz for pos in top_idx[:1]):
                    # Check if any top position is affected
                    affected_top = [t for t in top_idx if t in set(nz.tolist())]
                    if affected_top:
                        non_affected_max = second_max
                    else:
                        non_affected_max = max_val
                else:
                    non_affected_max = max_val
                new_max = max(non_affected_max, nz_max)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_pair = (i, j)

        if best_pair is None or best_new_max >= max_val:
            no_improve += 1
            if no_improve > 500:
                break
            continue

        # Apply swap
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        corr += -change[i] + change[j]
        update_change_after_swap(change, h, i, j, n)
        no_improve = 0

        new_c5 = float(np.max(corr)) * dx
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h.copy()

    return best_h, best_c5

def _g_n = 0
def _g_target = 0.0
def _cons_fn(x):
    return float(np.sum(x) - _g_target)
def _cons_jac_fn(x):
    return np.ones(_g_n)

def slsqp_polish(h, n, alphas, maxiter, time_limit):
    global _g_n, _g_target
    start = time.time()
    _g_n = n
    _g_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fn, 'jac': _cons_jac_fn}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        def obj(x, a=alpha):
            c = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(c)
            shifted = np.clip(a * (c - cmax), -500.0, 0.0)
            return float(cmax + np.sum(np.exp(shifted)) / a)
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_feasible(res.x, n / 2.0)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_binary(h_src, n_dst):
    n_src = len(h_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_binary(h_interp, n_dst)
