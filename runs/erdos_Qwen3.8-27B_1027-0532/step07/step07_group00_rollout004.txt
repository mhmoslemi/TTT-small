_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-11
        else:
            mask = h > 1e-11
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_cd(h_init, n, target, time_limit, max_sweeps=30000,
            n_cand_start=25, n_cand_end=65, max_no_improve=400,
            weight_beta=6.0, top_k_shifts=40, use_anneal=False, T0=0.0004):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(42)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        T = T0 * 0.005 ** progress if use_anneal else 0.0

        c_max = float(np.max(corr))
        denom = max(c_max, 1e-10)
        weights = np.exp(weight_beta * (corr - c_max) / denom)

        grad = np.zeros(n)
        tk = min(top_k_shifts, n_corr)
        top_idx = np.argsort(weights)[-tk:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(22):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break
                elif T > 1e-12 and approx_max - c5_val < T * 4:
                    if rng.random() < np.exp(-(approx_max - c5_val) / T):
                        best_trial_c5 = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
                        break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def lag_focused_cd(h_init, n, target, time_limit, max_sweeps=20000,
                   n_cand=40, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_move = None

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

    return best_h, best_c5


def spectral_perturb(h_init, n, rng, n_modes=8, scale=0.3):
    h = h_init.copy().astype(float)
    H = np.fft.rfft(h)
    freqs = np.arange(len(H))
    for _ in range(n_modes):
        fi = rng.randint(2, len(H) // 2)
        phase = rng.uniform(0, 2 * np.pi)
        H[fi] += scale * np.exp(1j * phase)
        H[-fi % len(H)] = H[fi].conjugate() if fi < len(H) else H[fi]
    h_new = np.fft.irfft(H, n=n)
    h_new = project_feasible(h_new, n / 2.0)
    return h_new


def dilate_perturb(h_init, n, rng, factor_range=(0.97, 1.03)):
    h = h_init.copy().astype(float)
    f = rng.uniform(*factor_range)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = (x_old * f) % 1.0
    h_new = np.interp(x_new, x_old, h, period=1.0)
    h_new = project_feasible(h_new, n / 2.0)
    return h_new


def shift_perturb(h_init, n, rng):
    h = h_init.copy().astype(float)
    s = rng.randint(1, n // 4)
    h_new = np.roll(h, s)
    h_new = project_feasible(h_new, n / 2.0)
    return h_new


def p_power_gradient(h_init, n, target, time_limit, p=8, n_iter=6000, step_init=0.004):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    no_improve = 0

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 800:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_k = min(25, n_corr)
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = corr[s_idx] ** (p - 1)
            if w < 1e-20:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - step * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val - 1e-15:
            h = h_try
            step = min(step * 1.03, 0.04)
        else:
            step *= 0.88
            if step < 1e-6:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Main aggressive CD
    if tleft() > 350:
        h_cd, c5_cd = fast_cd(best_h, n0, target0,
                              time_limit=min(300, tleft() * 0.5),
                              max_sweeps=30000,
                              n_cand_start=20, n_cand_end=60,
                              max_no_improve=400,
                              weight_beta=6.0, top_k_shifts=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Lag-focused CD
    if tleft() > 250:
        h_lf, c5_lf = lag_focused_cd(best_h, n0, target0,
                                     time_limit=min(200, tleft() * 0.5),
                                     max_sweeps=20000, n_cand=45,
                                     max_no_improve=300)
        if c5_lf < best_score:
            best_score = c5_lf
            best_h = h_lf

    # Phase 4: SLSQP + CD refinement
    if tleft() > 200:
        h_pol2, sc2 = slsqp_round(best_h, n0, target0,
                                  [2000, 10000, 50000], 500, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2
        if tleft() > 100:
            h_cd2, c5_cd2 = fast_cd(best_h, n0, target0,
                                    time_limit=min(80, tleft() * 0.5),
                                    max_sweeps=15000,
                                    n_cand_start=25, n_cand_end=65,
                                    max_no_improve=250,
                                    weight_beta=8.0, top_k_shifts=50)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 5: Multi-start with structured perturbations
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    rng = np.random.RandomState(seed)

    perturbations = []
    for _ in range(3):
        perturbations.append(spectral_perturb(best_h, n_cur, rng, n_modes=6, scale=0.2))
    for _ in range(2):
        perturbations.append(dilate_perturb(best_h, n_cur, rng))
    perturbations.append(shift_perturb(best_h, n_cur, rng))

    per_perturb_time = min(80, tleft() / max(len(perturbations), 1) * 0.7)
    for pert_h in perturbations:
        if tleft() < 60:
            break
        h_p, sc_p = slsqp_round(pert_h, n_cur, target_cur,
                                [200, 2000], 300, 1e-14)
        if tleft() > 40:
            h_cd_p, c5_cd_p = fast_cd(h_p, n_cur, target_cur,
                                      time_limit=min(per_perturb_time, tleft() * 0.4),
                                      max_sweeps=10000,
                                      n_cand_start=20, n_cand_end=50,
                                      max_no_improve=200,
                                      weight_beta=7.0, top_k_shifts=35)
            if c5_cd_p < best_score:
                best_score = c5_cd_p
                best_h = h_cd_p
        elif sc_p < best_score:
            best_score = sc_p
            best_h = h_p

    # Phase 6: p-power gradient on best
    if tleft() > 120:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pp, c5_pp = p_power_gradient(best_h, n_cur, target_cur,
                                       time_limit=min(100, tleft() * 0.5),
                                       p=10, n_iter=5000, step_init=0.003)
        if c5_pp < best_score:
            best_score = c5_pp
            best_h = h_pp

    # Phase 7: Annealed CD to escape local minima
    if tleft() > 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_an, c5_an = fast_cd(best_h, n_cur, target_cur,
                              time_limit=min(120, tleft() * 0.5),
                              max_sweeps=15000,
                              n_cand_start=30, n_cand_end=60,
                              max_no_improve=500,
                              weight_beta=4.0, top_k_shifts=30,
                              use_anneal=True, T0=0.0006)
        if c5_an < best_score:
            best_score = c5_an
            best_h = h_an

    # Phase 8: Final SLSQP with high alpha
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin, sc_fin = slsqp_round(best_h, n_cur, target_cur,
                                    [10000, 50000, 100000], 500, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 9: Final CD polish
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final_cd, c5_final = fast_cd(best_h, n_cur, target_cur,
                                       time_limit=min(50, tleft() * 0.6),
                                       max_sweeps=10000,
                                       n_cand_start=30, n_cand_end=70,
                                       max_no_improve=150,
                                       weight_beta=10.0, top_k_shifts=50)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Phase 10: Try higher resolution if time permits
    if tleft() > 200:
        n_cur = len(best_h)
        if n_cur < 1000:
            n2 = 1000
            target2 = n2 / 2.0
            h2 = interpolate_to(best_h, n2, target2)
            sc2 = compute_c5(h2)
            if sc2 < best_score + 0.001:
                h2_pol, sc2_pol = slsqp_round(h2, n2, target2,
                                              [200, 1000, 5000], 300, 1e-14)
                if tleft() > 80:
                    h2_cd, c5_2cd = fast_cd(h2_pol, n2, target2,
                                            time_limit=min(120, tleft() * 0.5),
                                            max_sweeps=10000,
                                            n_cand_start=20, n_cand_end=50,
                                            max_no_improve=200,
                                            weight_beta=6.0, top_k_shifts=40)
                    if c5_2cd < best_score:
                        best_score = c5_2cd
                        best_h = h2_cd
                elif sc2_pol < best_score:
                    best_score = sc2_pol
                    best_h = h2_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
