            if 0 <= p - k < n:

def project_feasible(h, target_sum):

            update_change_rows(change, [i, j], h, n)

def compute_weighted_grad(x, weights, n):
    """Compute the weighted gradient: sum_k w_k * ∂f_k/∂x"""
    mid = n - 1
    idxs = np.arange(n)
    g = np.zeros(n)
    
    for k_idx in np.where(weights > 1e-12)[0]:
        k = k_idx - mid
        v1 = (idxs + k < n)
        v2 = (idxs - k >= 0)
        g[v1] += weights[k_idx] * (1.0 - x[idxs[v1] + k])
        g[v2] -= weights[k_idx] * x[idxs[v2] - k]
    
    return g
