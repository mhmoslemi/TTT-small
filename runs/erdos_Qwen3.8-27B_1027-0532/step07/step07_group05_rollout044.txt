        h = np.clip(h, 0.0, 1.0)

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

    _g_target = target

                                  time_limit=min(50, tleft() * 0.25),
                                  max_sweeps=8000,
                                  n_active=5,
                                  n_cand_start=20, n_cand_end=50,
                                  max_no_improve=200, bisection_iters=12)
            if scd < best_score:
                best_score = scd
                best_h = hd
                # Follow up with SLSQP on the improved point
                if tleft() > 60:
                    hp, scp = slsqp_optimize(best_h, n0, target0,
                                             [5000, 20000],
                                             400, 1e-14)
                    if scp < best_score:
                        best_score = scp
                        best_h = hp

    # Phase 8: Final CD
    if tleft() > 80:
        h_cd3, c5_cd3 = enhanced_cd(best_h, n0, target0,
                                    time_limit=min(60, tleft() * 0.5),
                                    max_sweeps=12000,
                                    n_active=5,
                                    n_cand_start=35, n_cand_end=70,
                                    max_no_improve=200, bisection_iters=14)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 9: Final SLSQP polish
    if tleft() > 50:
        h_fin, sc_fin = slsqp_optimize(best_h, n0, target0,
                                       [10000, 50000],
                                       400, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
