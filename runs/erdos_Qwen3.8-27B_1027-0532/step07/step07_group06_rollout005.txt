def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _cons_fun(x):
    return np.sum(x) - _g_target

    return np.ones(_g_n)

    corr = compute_corr(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    corr = compute_corr(x)
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))


def slsqp_optimize(h_start, n, target, alphas, maxiter, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            if use_pnorm:
                res = minimize(obj_pnorm, h_cur, args=(param,),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-16})
            else:
                res = minimize(obj_lse, h_cur, args=(param,),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 5000:
            break

        scale = base_scale * max(0.03, 1.0 - 0.8 * step / max(max_steps, 1))
        n_pert = int(rng.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        perm = rng.permutation(n)
        idxs = perm[:n_pert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 120:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def multi_start_refine(h_init, n, target, time_limit, start_time, num_restarts=3):
    rng = np.random.RandomState(999)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    for r in range(num_restarts):
        elapsed = time.time() - start_time
        remaining = time_limit - elapsed
        if remaining < 60:
            break

        scale = 0.05 * (r + 1)
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)

        slsqp_time = min(remaining - 30, 80)
        h_r, sc_r = slsqp_optimize(h_pert, n, target,
                                   [200, 2000, 10000],
                                   500, start_time + slsqp_time, start_time, use_pnorm=False)
        if sc_r > best_c5 - 1e-12:
            cd_time = min(remaining - 20, 120)
            h_r, sc_r = paired_cd(h_r, n, target, start_time + cd_time, start_time,
                                  max_sweeps=20000, n_cand_start=15, n_cand_end=50,
                                  max_no_improve=300, n_top_shifts=3, bisection_iters=20)

        if sc_r < best_c5 - 1e-14:
            best_c5 = sc_r
            best_h = h_r

    return best_h, best_c5
