import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_max_corr(h):
    """Compute gradient of the maximum correlation term."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star_idx = int(np.argmax(corr))
    k_star = k_star_idx - (n - 1)
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    return grad, corr

def projected_subgrad_nesterov(h_init, n, target, time_limit, 
                                eta0=0.1, momentum_decay=0.9):

    prev_h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_full(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_topk(x, alpha, k):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(corr)[-k:]
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr[idx] - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_full(x):
    return np.sum(x) - _g_target


def _cons_jac_full(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol,
                time_limit, start_time, top_k=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_full, 'jac': _cons_jac_full}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            if top_k is not None:
                k = min(top_k, 2 * n - 1)
                res = minimize(smooth_obj_topk, h_cur, args=(alpha, k),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            else:
                res = minimize(smooth_obj_full, h_cur, args=(alpha,),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def _update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def fast_pair_cd(h_init, n, target, time_limit, max_sweeps,

    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx_global

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_l = len(low_set)

                if j == i:
                    continue
                if h[j] > 0.95:
                    continue

                    continue

                a_d = delta_lo

                    f_prime = np.max(g[active_mask])
                    if f_prime > 1e-10:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx_global
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            _update_change_row(change, best_i, h, n, shifts)
            _update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        top_idx = np.argsort(corr)[-top_k:]
        c_max = corr[np.argmax(corr)]
        weights = np.exp(-4.0 * (c_max - corr[top_idx]) / max(c_max, 1e-10))
        weights = weights / np.sum(weights)

        grad = np.zeros(n)
        idxs = np.arange(n)
        mid = n - 1
        for wi, s_idx in enumerate(top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += weights[wi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= weights[wi] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        eta = eta0 / (1.0 + 0.01 * t)
        v = momentum * v - eta * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, dx_global
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(budget_s - 35.0 - (time.time() - start_time), 0.0)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP on initial grid with top-k
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [500, 2000, 8000], 800, 1e-14,
                                min(60, tleft() * 0.7), start_time, top_k=20)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD on initial grid
    if tleft() > 120:
        h_cd, sc = fast_pair_cd(best_h, n0, target0,
                                min(100, tleft() * 0.7), 50000,
                                25, 60, 5000, n_bisect=10)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 3: Free variable refinement
    if tleft() > 50:
        h_fr, sc = free_var_refine(best_h, n0, target0,
                                   min(35, tleft() * 0.6), threshold=0.02, alpha=8000)
        if sc < best_score:
            best_score = sc
            best_h = h_fr

    # Phase 4: Interpolate to n=600 and CD
    if tleft() > 250:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)

        h_cd1, sc1 = fast_pair_cd(h1, n1, target1,
                                  min(180, tleft() * 0.65), 40000,
                                  25, 70, 5000, n_bisect=10)
        if sc1 < best_score:
            best_score = sc1
            best_h = h_cd1
            n0 = n1
            target0 = target1

    # Phase 5: Multi-shift CD
    if tleft() > 100:
        h_ms, sc = multi_shift_cd(best_h, n0, target0,
                                  min(80, tleft() * 0.6), 15000,
                                  45, 5)
        if sc < best_score:
            best_score = sc
            best_h = h_ms

    # Phase 6: SLSQP refinement on finer grid
    if tleft() > 100:
        h_pol1, sc = slsqp_round(best_h, n0, target0,
                                 [5000, 20000], 1200, 1e-14,
                                 min(80, tleft() * 0.6), start_time, top_k=10)
        if sc < best_score:
            best_score = sc
            best_h = h_pol1

    # Phase 7: More CD
    if tleft() > 120:
        h_cd2, sc = fast_pair_cd(best_h, n0, target0,
                                 min(100, tleft() * 0.6), 30000,
                                 40, 80, 4000, n_bisect=10)
        if sc < best_score:
            best_score = sc
            best_h = h_cd2

    # Phase 8: Free variable refinement
    if tleft() > 50:
        h_fr1, sc = free_var_refine(best_h, n0, target0,
                                    min(35, tleft() * 0.6), threshold=0.01, alpha=10000)
        if sc < best_score:
            best_score = sc
            best_h = h_fr1

    # Phase 9: Try n=800
    if tleft() > 300:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        h_cd3, sc3 = fast_pair_cd(h2, n2, target2,
                                  min(200, tleft() * 0.55), 30000,
                                  25, 70, 4000, n_bisect=10)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_cd3
            n0 = n2
            target0 = target2

        if tleft() > 60:
            h_fr2, sc2 = free_var_refine(best_h, n0, target0,
                                         min(40, tleft() * 0.5), threshold=0.01, alpha=10000)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_fr2

    # Phase 10: SA for escape
    if tleft() > 80:
        h_sa, sc = sa_pair_cd(best_h, n0, target0,
                              min(60, tleft() * 0.6), 25000,
                              0.03, 0.0005, 50)
        if sc < best_score:
            best_score = sc
            best_h = h_sa
            if tleft() > 40:
                h_fr3, sc2 = free_var_refine(best_h, n0, target0,
                                             min(30, tleft() * 0.5), threshold=0.01, alpha=8000)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_fr3

    # Phase 11: Final SLSQP
    if tleft() > 60:
        h_pol2, sc = slsqp_round(best_h, n0, target0,
                                 [20000, 80000], 800, 1e-15,
                                 min(50, tleft() * 0.6), start_time, top_k=5)
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    # Phase 12: Final rapid CD
    if tleft() > 40:
        h_cf, sc = fast_pair_cd(best_h, n0, target0,
                                min(30, tleft() * 0.6), 15000,
                                50, 80, 1000, n_bisect=12)
        if sc < best_score:
            best_score = sc
            best_h = h_cf

    # Phase 13: Final free var
    if tleft() > 20:
        h_ff, sc = free_var_refine(best_h, n0, target0,
                                   min(15, tleft() * 0.6), threshold=0.005, alpha=20000)
        if sc < best_score:
            best_score = sc
            best_h = h_ff

    # Final check: also try the original n=400 version
    if tleft() > 30:
        h_400 = interpolate_to(best_h, n0, n0 / 2.0)
        sc_400 = compute_c5(h_400)
        if sc_400 < best_score:
            best_score = sc_400
            best_h = h_400
            n0 = n0

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
