import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step_v = diff / n
            h = np.clip(h + step_v, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_eq_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def power_mean_cd(h_init, n, target, time_limit, n_steps=8000, p_start=6, p_end=200,
                  base_step=0.004):
    start = time.time()
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = base_step

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        progress = min(step / max(n_steps - 1, 1), 1.0)
        p = p_start * (p_end / p_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        S = np.sum(corr ** p)
        if S < 1e-30:
            break
        coeff = S ** (1.0 / p - 1.0)

        grad = np.zeros(n)
        for k in range(n_corr):
            shift = k - mid
            w = coeff * corr[k] ** (p - 1)
            if abs(w) < 1e-18:
                continue
            v1 = (idxs - shift >= 0) & (idxs - shift < n)
            v2 = (idxs + shift >= 0) & (idxs + shift < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - shift])
            grad[v2] -= w * h[idxs[v2] + shift]

        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')))

        if sc <= c5_val + 1e-14:
            h = h_try
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = float(np.max(np.correlate(h_try2, 1.0 - h_try2, mode='full')))
            if sc2 <= c5_val + 1e-14:
                h = h_try2
            else:
                eta *= 0.9

        if step % 1000 == 0 and step > 0:
            eta = base_step * max(0.3, 1.0 - step / (2.0 * n_steps))

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturbations(h, n, target, rng, n_perts=8):
    results = []
    for r in range(n_perts):
        h_p = h.copy()
        pert_type = r % 4
        if pert_type == 0:
            scale = 0.03 + 0.02 * r
            h_p = h + rng.randn(n) * scale
        elif pert_type == 1:
            freq = 1 + r
            t = np.arange(n) / n
            h_p = h + 0.05 * np.sin(2 * np.pi * freq * t + rng.randn())
        elif pert_type == 2:
            block_size = max(n // (4 + r), 4)
            n_blocks = n // block_size
            for b in range(n_blocks):
                delta = rng.randn() * 0.04
                start_i = b * block_size
                end_i = start_i + block_size
                h_p[start_i:end_i] += delta
        else:
            n_swaps = 3 + r
            for _ in range(n_swaps):
                i, j = rng.randint(0, n, size=2)
                h_p[i], h_p[j] = h_p[j], h_p[i]
        h_p = project_feasible(h_p, target)
        results.append(h_p)
    return results


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    # Phase 1: Quick SLSQP refinement at initial resolution
    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                   [100, 500, 2000], 800, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Power-mean gradient descent for broader exploration
    if tleft() > 100:
        h_pm, sc_pm = power_mean_cd(best_h, cur_n, cur_target,
                                    time_limit=min(80, tleft() * 0.6),
                                    n_steps=6000, p_start=4, p_end=100,
                                    base_step=0.006)
        if sc_pm < best_score:
            best_score = sc_pm
            best_h = h_pm

    # Phase 3: CD with change matrix
    if tleft() > 150:
        h_cd, c5_cd = pair_cd_fast(best_h, cur_n, cur_target,
                                   time_limit=min(180, tleft() * 0.65),
                                   max_sweeps=25000,
                                   n_cand_start=20, n_cand_end=70,
                                   n_shifts=3, max_no_improve=350,
                                   anneal_temp=0.005, anneal_decay=0.9993)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP with high alpha
    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                   [5000, 20000, 50000], 1200, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Structured perturbations + SLSQP refinement
    if tleft() > 200:
        perts = structured_perturbations(best_h, cur_n, cur_target, rng, n_perts=6)
        for h_p in perts:
            if tleft() < 60:
                break
            sc_p = compute_c5(h_p)
            h_ref, sc_ref = slsqp_optimize(h_p, cur_n, cur_target,
                                           [500, 5000], 600, 1e-14)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 6: Upsample to higher resolution
    if tleft() > 400:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 300:
            h_cd1, c5_cd1 = pair_cd_fast(h1, n1, target1,
                                         time_limit=min(200, tleft() * 0.6),
                                         max_sweeps=20000,
                                         n_cand_start=15, n_cand_end=50,
                                         n_shifts=3, max_no_improve=300,
                                         anneal_temp=0.0, anneal_decay=1.0)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                cur_n = n1
                cur_target = target1
            elif sc1 < best_score:
                best_score = sc1
                best_h = h1
                cur_n = n1
                cur_target = target1

        if tleft() > 80:
            h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                       [2000, 10000], 800, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 7: Another round of CD + polish at current resolution
    if tleft() > 150:
        h_cd2, c5_cd2 = pair_cd_fast(best_h, cur_n, cur_target,
                                     time_limit=min(100, tleft() * 0.6),
                                     max_sweeps=15000,
                                     n_cand_start=40, n_cand_end=100,
                                     n_shifts=4, max_no_improve=250,
                                     anneal_temp=0.003, anneal_decay=0.999)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 8: Final SLSQP with very high alpha
    if tleft() > 60:
        h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                   [10000, 50000, 100000], 1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 9: Try upsample to n=1200 if time permits
    if tleft() > 350:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 250:
            h_cd2, c5_cd2 = pair_cd_fast(h2, n2, target2,
                                         time_limit=min(180, tleft() * 0.6),
                                         max_sweeps=15000,
                                         n_cand_start=12, n_cand_end=40,
                                         n_shifts=2, max_no_improve=250,
                                         anneal_temp=0.0, anneal_decay=1.0)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                cur_n = n2
                cur_target = target2
            elif sc2 < best_score:
                best_score = sc2
                best_h = h2
                cur_n = n2
                cur_target = target2

        if tleft() > 60:
            h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                       [5000, 20000], 600, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 10: Final micro-tuning with power mean
    if tleft() > 50:
        h_pm2, sc_pm2 = power_mean_cd(best_h, cur_n, cur_target,
                                      time_limit=min(30, tleft() * 0.7),
                                      n_steps=3000, p_start=20, p_end=500,
                                      base_step=0.002)
        if sc_pm2 < best_score:
            best_score = sc_pm2
            best_h = h_pm2

    # Phase 11: Final SLSQP polish
    if tleft() > 30:
        h_pol, sc = slsqp_optimize(best_h, cur_n, cur_target,
                                   [50000, 200000], 600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
