    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(12345)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        # Gradient-based pair selection using smooth max
        alpha_sel = 100.0
        _, grad = smooth_max_and_grad(h, alpha_sel)
        grad_centered = grad - np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        sorted_idx = np.argsort(grad_centered)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        # Also include top correlation lags for direct targeting
        top_corr_idx = np.argsort(corr)[-n_active:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        # Build candidate pairs
        candidates = []
        n_h = len(high_set)
        n_l = len(low_set)

        max_pairs = 300
        if n_h * n_l <= max_pairs:
            for ii in range(n_h):
                for jj in range(n_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        candidates.append((iv, jv))
        else:
            step = max(1, n_h // 15)
            for ii in range(0, n_h, step):
                for jj in range(0, n_l, max(1, n_l // 15)):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        candidates.append((iv, jv))
            random_n = min(max_pairs - len(candidates), 100)
            for _ in range(max(0, random_n)):
                iv = int(high_set[rng.randint(n_h)])
                jv = int(low_set[rng.randint(n_l)])
                if iv != jv:
                    candidates.append((iv, jv))

        for i, j in candidates:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-13:
                continue

            # Compute change vector on the fly: cv_j - cv_i
            # h[i] decreases by delta, h[j] increases by delta
            # Change in corr[k] = -delta * [cv_i at k] + delta * [cv_j at k]
            # where cv_p[k] is the change in corr when h[p] increases by 1

            # cv for point p: how corr changes when h[p] increases
            # For lag k (output index k+mid):
            #   h[p] as first factor (h[n+k]): contributes (1-h[p-k]) if 0<=p-k<n -> at output index (p-k)+k+mid... 
            #   Actually: corr at lag k = sum_i h[i+k]*(1-h[i])
            #   If we increase h[p] by 1:
            #     - p appears as i+k (first factor): i = p-k, contributes (1-h[p-k]) if 0<=p-k<n
            #     - p appears as i (in 1-h[i]): i = p, contributes -h[p+k] if 0<=p+k<n
            #   So cv_p at lag k = (1-h[p-k])*1(0<=p-k<n) - h[p+k]*1(0<=p+k<n)
            #   Output index = k + mid

            # For the pair (i, j): move delta from i to j
            # corr_new = corr - delta*cv_i + delta*cv_j = corr + delta*(cv_j - cv_i)

            # cv_j - cv_i at each lag k:
            # = [(1-h[j-k])*1(0<=j-k<n) - h[j+k]*1(0<=j+k<n)] - [(1-h[i-k])*1(0<=i-k<n) - h[i+k]*1(0<=i+k<n)]

            # Let's compute this efficiently
            change = np.zeros(n_corr)

            # cv_j contribution
            jk_min = max(0, j - n + 1)
            jk_max = min(n - 1, j)
            for kk in range(jk_min, jk_max + 1):
                pm = j - kk
                if 0 <= pm < n:
                    change[kk + mid] += (1.0 - h[pm])
            jk2_min = max(0, -j)
            jk2_max = min(n - 1 - j, n - 1)
            for kk in range(jk2_min, jk2_max + 1):
                pp = j + kk
                if 0 <= pp < n:
                    change[kk + mid] -= h[pp]

            # cv_i contribution (negated)
            ik_min = max(0, i - n + 1)
            ik_max = min(n - 1, i)
            for kk in range(ik_min, ik_max + 1):
                pm = i - kk
                if 0 <= pm < n:
                    change[kk + mid] -= (1.0 - h[pm])
            ik2_min = max(0, -i)
            ik2_max = min(n - 1 - i, n - 1)
            for kk in range(ik2_min, ik2_max + 1):
                pp = i + kk
                if 0 <= pp < n:
                    change[kk + mid] += h[pp]

            # Bisection for best delta
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * change
                f_mid = np.max(vals)
                if f_mid > c5_corr_max + 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            c5_corr_max = np.max(corr)
            # Re-do bisection properly
            a_d = 0.0
            b_d = delta_hi
            current_max = np.max(corr)
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * change
                f_mid = np.max(vals)
                if f_mid > current_max + 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-14:
                # Try tiny step to see if it helps
                delta_star = min(1e-8, 0.5 * delta_hi)
                if delta_star < 1e-14:
                    continue

            approx_max = np.max(corr + delta_star * change) * dx
            if approx_max < best_trial - 1e-15:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0 and best_delta > 1e-15:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def vectorized_cd(h_init, n, target, time_limit, max_sweeps=15000,
                  n_active=3, n_cand_start=10, n_cand_end=40,
                  max_no_improve=250, bisection_iters=14):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(999)

    # Precompute change vectors for all points (memory: n * n_corr)
    # For n=800, n_corr=1599: 800*1599 = 1.28M floats = 10MB. OK.
    change_mat = np.zeros((n, n_corr))
    for p in range(n):
        # cv_p at lag k (output index k+mid):
        # (1-h[p-k])*1(0<=p-k<n) - h[p+k]*1(0<=p+k<n)
        for k in range(max(0, p - n + 1), min(n - 1, p) + 1):
            pm = p - k
            if 0 <= pm < n:
                change_mat[p, k + mid] += (1.0 - h[pm])
        for k in range(max(0, -p), min(n - 1 - p, n - 1) + 1):
            pp = p + k
            if 0 <= pp < n:
                change_mat[p, k + mid] -= h[pp]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        current_max = float(np.max(corr))

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        # Gradient-based selection
        _, grad = smooth_max_and_grad(h, 100.0)
        grad_c = grad - np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        sorted_idx = np.argsort(grad_c)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        max_pairs = 250

        if n_h * n_l <= max_pairs:
            pairs = [(int(high_set[ii]), int(low_set[jj]))
                     for ii in range(n_h) for jj in range(n_l)
                     if int(high_set[ii]) != int(low_set[jj])]
        else:
            pairs = []
            step_h = max(1, n_h // 20)
            step_l = max(1, n_l // 20)
            for ii in range(0, n_h, step_h):
                for jj in range(0, n_l, step_l):
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pairs.append((iv, jv))
            rem = max_pairs - len(pairs)
            for _ in range(max(0, rem)):
                iv = int(high_set[rng.randint(n_h)])
                jv = int(low_set[rng.randint(n_l)])
                if iv != jv:
                    pairs.append((iv, jv))

        for i, j in pairs:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-13:
                continue

            g = change_mat[j] - change_mat[i]
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                f_mid = np.max(corr + mid_d * g)
                if f_mid > current_max + 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-14:
                continue

            approx_max = float(np.max(corr + delta_star * g)) * dx
            if approx_max < best_trial - 1e-15:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0 and best_delta > 1e-15:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change matrix rows
            update_row(change_mat, best_i, h, n, mid)
            update_row(change_mat, best_j, h, n, mid)

    return best_h, best_c5

def update_row(change_mat, p, h, n, mid):
    change_mat[p, :] = 0.0
    for k in range(max(0, p - n + 1), min(n - 1, p) + 1):
        pm = p - k
        if 0 <= pm < n:
            change_mat[p, k + mid] += (1.0 - h[pm])
    for k in range(max(0, -p), min(n - 1 - p, n - 1) + 1):
        pp = p + k
        if 0 <= pp < n:
            change_mat[p, k + mid] -= h[pp]

def fista_smooth(h_init, n, target, alpha, time_limit, max_iter=50000, lr_init=0.01):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    t_k = 1.0
    lr = lr_init

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        f_val, grad = smooth_max_and_grad(h, alpha)
        grad_c = grad - np.mean(grad)

        h_step = h - lr * grad_c
        h_proj = project_feasible(h_step, target)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = h_proj + ((t_k - 1.0) / t_new) * (h_proj - h_prev)
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        h_prev = h_proj
        t_k = t_new

        # Adaptive step size
        if it % 100 == 99:
            sc = compute_c5(h)
            if sc < best_f - 1e-12:
                best_f = sc
                best_h = h.copy()
            # Decrease lr if not improving
            lr = max(lr * 0.95, 0.0001)

    sc = compute_c5(best_h)
    return best_h, min(sc, compute_c5(project_feasible(h, target)))

def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha, _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 0: SLSQP at n=400 with analytical gradient
    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [50, 200, 500, 2000],
                                300, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 1: Quick CD at n=400
    if tleft() > 100:
        h_cd, sc_cd = vectorized_cd(best_h, n0, target0,
                                     time_limit=min(40, tleft() * 0.25),
                                     max_sweeps=8000,
                                     n_cand_start=8, n_cand_end=25,
                                     max_no_improve=200, bisection_iters=12)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd
            n0 = len(h_cd)
            target0 = n0 / 2.0

    # Phase 2: Upscale to n=800
    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h1, sc1 = slsqp_round(h1, n1, target1,
                              [200, 1000, 3000],
                              400, 1e-14)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            n0 = n1
            target0 = target1

    # Phase 3: Main CD at n=800
    if tleft() > 300:
        tl = min(350, tleft() * 0.4)
        h_cd1, sc_cd1 = vectorized_cd(best_h, n0, target0,
                                       time_limit=tl,
                                       max_sweeps=25000,
                                       n_cand_start=12, n_cand_end=45,
                                       max_no_improve=300, bisection_iters=14)
        if sc_cd1 < best_score:
            best_score = sc_cd1
            best_h = h_cd1

    # Phase 4: SLSQP polishing with high alpha
    if tleft() > 150:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [3000, 10000, 30000],
                                500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Another CD round
    if tleft() > 150:
        tl = min(120, tleft() * 0.35)
        h_cd2, sc_cd2 = vectorized_cd(best_h, n0, target0,
                                       time_limit=tl,
                                       max_sweeps=15000,
                                       n_cand_start=15, n_cand_end=50,
                                       max_no_improve=250, bisection_iters=14)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 6: FISTA with smooth max
    if tleft() > 120:
        tl = min(80, tleft() * 0.3)
        h_fista, sc_f = fista_smooth(best_h, n0, target0,
                                      alpha=5000, time_limit=tl,
                                      max_iter=50000, lr_init=0.005)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_fista

    # Phase 7: Final SLSQP
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [10000, 50000, 100000],
                                400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 8: Final CD
    if tleft() > 60:
        tl = min(40, tleft() * 0.5)
        h_cd3, sc_cd3 = vectorized_cd(best_h, n0, target0,
                                       time_limit=tl,
                                       max_sweeps=10000,
                                       n_cand_start=20, n_cand_end=55,
                                       max_no_improve=200, bisection_iters=14)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    # Phase 9: Final SLSQP polish
    if tleft() > 40:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [50000, 200000],
                                300, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
