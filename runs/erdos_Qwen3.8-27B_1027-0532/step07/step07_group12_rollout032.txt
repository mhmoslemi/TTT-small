def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass
    return best_h, best_c5

        velocity = beta * velocity + (1 - beta) * grad

def structured_perturb(h, n, target, rng, mode):
    h_new = h.copy()
    if mode == 'block_flip':
        a = rng.randint(0, n // 3)
        b = a + rng.randint(n // 10, n // 3)
        b = min(b, n)
        h_new[a:b] = 1.0 - h_new[a:b]
    elif mode == 'bump':
        center = rng.randint(0, n)
        width = rng.randint(n // 10, n // 5)
        height = rng.choice([-1, 1]) * rng.uniform(0.05, 0.15)
        x = np.arange(n)
        bump = height * np.exp(-0.5 * ((x - center) / width) ** 2)
        h_new += bump
    elif mode == 'lowfreq':
        for _ in range(rng.randint(1, 4)):
            freq = rng.randint(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.08)
            x = np.arange(n)
            h_new += amp * np.sin(2 * np.pi * freq * x / n + phase)
    elif mode == 'smooth_noise':
        noise = rng.randn(n)
        # Low-pass filter
        kernel = np.ones(5) / 5
        noise = np.convolve(noise, kernel, mode='same')
        h_new += noise * 0.03
    return project_feasible(h_new, target)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass
    return best_h, best_c5

def subgradient_momentum(h_init, n, target, time_limit, max_steps=40000, n_shifts=5, base_lr=0.03):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    velocity = np.zeros(n)
    beta = 0.9
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 15.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= (np.sum(grad) / n)
        velocity = beta * velocity + (1.0 - beta) * grad
        decay = 1.0 / (1.0 + 0.01 * step)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * velocity
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-16:
                h = h_try
                improved = True
                break
        if not improved and step > 3000:
            break
    return best_h, best_c5

def structured_perturb(h, n, target, rng, mode):
    h_new = h.copy()
    if mode == 'block_flip':
        a = rng.randint(0, n // 3)
        length = rng.randint(n // 12, n // 4)
        b = min(a + length, n)
        h_new[a:b] = 1.0 - h_new[a:b]
    elif mode == 'bump':
        center = rng.randint(0, n)
        width = max(1, rng.randint(n // 15, n // 6))
        height = rng.choice([-1.0, 1.0]) * rng.uniform(0.04, 0.12)
        x = np.arange(n)
        bump = height * np.exp(-0.5 * ((x - center) / width) ** 2)
        h_new += bump
    elif mode == 'lowfreq':
        x = np.arange(n)
        for _ in range(rng.randint(1, 4)):
            freq = rng.randint(1, 5)
            phase = rng.uniform(0.0, 2.0 * np.pi)
            amp = rng.uniform(0.02, 0.06)
            h_new += amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    elif mode == 'smooth_noise':
        noise = rng.randn(n)
        kernel = np.ones(7) / 7.0
        noise = np.convolve(noise, kernel, mode='same')
        h_new += noise * 0.025
    return project_feasible(h_new, target)

                                      binary_iters=12, no_improve_limit=600,
                                      alpha_w=25.0)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 100:
        sa_time = min(80, tleft() * 0.3)
        hsa, c5sa = simulate_annealing_cd(best_h, n, target, time_limit=sa_time,
                                           n_cand=30, n_shifts=2)
        if c5sa < best_score:
            best_score = c5sa
            best_h = hsa

    if tleft() > 60:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[100000, 500000],
                            maxiter=200, time_limit=min(40, tleft() * 0.25),
                            penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 180:
        rng = np.random.RandomState(seed)
        n_use = len(best_h)
        target_use = n_use / 2.0
        modes = ['block_flip', 'bump', 'lowfreq', 'smooth_noise', 'block_flip', 'bump']
        for mode in modes:
            if tleft() < 70:
                break
            h_pert = structured_perturb(best_h, n_use, target_use, rng, mode)
            hp, sc = lbfgs_phase(h_pert, n_use, target_use,
                                alphas=[500, 5000, 20000],
                                maxiter=150, time_limit=min(20, tleft() * 0.15),
                                penalty=500.0)
            if sc < best_score:
                best_score = sc
                best_h = hp
                if tleft() > 80:
                    hcd, c5cd = pair_cd_enhanced(best_h, n_use, target_use,
                                                  time_limit=min(50, tleft() * 0.2),
                                                  n_cand_start=25, n_cand_end=45,
                                                  max_sweeps=50000, n_shifts=1,
                                                  binary_iters=12)
                    if c5cd < best_score:
                        best_score = c5cd
                        best_h = hcd

    if tleft() > 50:
        hsg, c5sg = subgradient_momentum(best_h, n, target, time_limit=min(35, tleft() * 0.3),
                                          max_steps=25000, n_shifts=7, base_lr=0.025)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    if tleft() > 40:
        cd_time = min(35, tleft() * 0.5)
        hcd, c5cd = pair_cd_enhanced(best_h, n, target, time_limit=cd_time,
                                      n_cand_start=45, n_cand_end=55,
                                      max_sweeps=50000, n_shifts=1,
                                      binary_iters=16, no_improve_limit=300)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 25:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[500000, 2000000, 10000000],
                            maxiter=150, time_limit=min(20, tleft() * 0.6),
                            penalty=50000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 20:
        hcd, c5cd = pair_cd_enhanced(best_h, n, target, time_limit=min(15, tleft() * 0.7),
                                      n_cand_start=55, n_cand_end=55,
                                      max_sweeps=20000, n_shifts=1,
                                      binary_iters=18, no_improve_limit=200)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
