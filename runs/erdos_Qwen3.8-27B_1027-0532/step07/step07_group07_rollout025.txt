import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _time_left(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-8:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _obj_func(x):
    return smooth_obj(x, _g_alpha)

def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_alpha, _g_target
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def soft_gradient(h, corr, beta):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    shifted = corr - np.max(corr)
    weights = np.exp(beta * shifted)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad

    n_corr = 2 * n - 1

    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Soft gradient
        grad = soft_gradient(h, corr, beta)
        grad -= np.mean(grad)

        # Candidate selection
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        # Active shift (max correlation)
        k_star_idx = int(np.argmax(corr))

        # Score matrix
        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)

        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval - 1)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:

            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-14:
                continue

            g = change[j] - change[i]

            # Bisection for optimal delta
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-14:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

            # Update change matrix for changed rows

            # Also need to update rows that interact with best_i and best_j

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _time_left(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-8:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _obj_func(x):
    return smooth_obj(x, _g_alpha)


def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-15):
    global _g_alpha, _g_target
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def compute_soft_gradient(h, corr, beta):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    shifted = corr - np.max(corr)
    weights = np.exp(beta * shifted)
    total = np.sum(weights)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        k = k_max - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
        return grad
    weights /= total
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad

        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 1:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval - 1)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-14:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-14:
                continue

            approx_max = np.max(corr + delta_star * g) * (2.0 / n)
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def multi_shift_cd(h_init, n, target, time_limit, max_sweeps=30000,
                   n_cand=100, top_k_eval=80, n_active_shifts=5,
                   max_no_improve=500, bisection_iters=18):
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    dx = 2.0 / n

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        if sweep > 0 and sweep % 40 == 0:
            change = build_change_matrix(h, n)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_shifts = np.argsort(corr)[-n_active_shifts:]
        k_star_idx = int(top_shifts[-1])

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] ** 2
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        n_cand_eff = min(n_cand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand_eff]
        high_set = sorted_idx[-n_cand_eff:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 1:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval - 1)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-14:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-14:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-15:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_search(h_init, n, target, time_limit, max_steps=30000,
                      base_scale=0.025):
    start = time.time()
    rng = np.random.RandomState(999)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        scale = base_scale * max(0.15, 1.0 - step / max_steps)
        n_pert = int(rng.choice([3, 5, 8, 12, 15, 20]))
        n_pert = min(n_pert, n)
        idxs_p = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs_p] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 800:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def perturbed_restart(h_init, n, target, time_limit, noise_scales=None):
    start = time.time()
    if noise_scales is None:
        noise_scales = [0.03, 0.08, 0.15, 0.25]
    rng = np.random.RandomState(42)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    for ns in noise_scales:
        tl = time_limit - (time.time() - start)
        if tl < 40:
            break
        h_pert = h_init + rng.randn(n) * ns
        h_pert = project_feasible(h_pert, target)
        h_opt, sc = slsqp_optimize(h_pert, n, target,
                                   [200, 1000, 5000],
                                   400, 1e-15)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_opt

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_alpha, _g_target
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP polish at initial resolution
    tl = _time_left(start_time, budget_s)
    if tl > 100:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [50, 200, 1000, 5000],
                                   600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Move to higher resolution
    tl = _time_left(start_time, budget_s)
    if tl > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

        tl = _time_left(start_time, budget_s)
        if tl > 280:
            h_cd, c5_cd = refined_cd(h1, n1, target1,
                                     time_limit=min(220, tl * 0.55),
                                     max_sweeps=50000,
                                     n_cand=130, top_k_eval=100,
                                     beta=40.0,
                                     max_no_improve=700,
                                     bisection_iters=20,
                                     rebuild_interval=25)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        tl = _time_left(start_time, budget_s)
        if tl > 80:
            h_pol2, sc2 = slsqp_optimize(best_h, n1, target1,
                                         [500, 2000, 8000],
                                         800, 1e-15)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

    # Phase 3: Multi-shift CD
    tl = _time_left(start_time, budget_s)
    if tl > 200:
        h_ms, c5_ms = multi_shift_cd(best_h, n1, target1,
                                     time_limit=min(170, tl * 0.65),
                                     max_sweeps=30000,
                                     n_cand=120, top_k_eval=90,
                                     n_active_shifts=4,
                                     max_no_improve=600,
                                     bisection_iters=18)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 4: Stochastic search
    tl = _time_left(start_time, budget_s)
    if tl > 120:
        h_ss, c5_ss = stochastic_search(best_h, n1, target1,
                                        time_limit=min(80, tl * 0.55),
                                        max_steps=25000,
                                        base_scale=0.02)
        if c5_ss < best_score:
            best_score = c5_ss
            best_h = h_ss

    # Phase 5: Another refined CD with different params
    tl = _time_left(start_time, budget_s)
    if tl > 150:
        h_cd2, c5_cd2 = refined_cd(best_h, n1, target1,
                                   time_limit=min(120, tl * 0.65),
                                   max_sweeps=35000,
                                   n_cand=160, top_k_eval=120,
                                   beta=100.0,
                                   max_no_improve=800,
                                   bisection_iters=22,
                                   rebuild_interval=20)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 6: Try higher resolution n=1200
    tl = _time_left(start_time, budget_s)
    if tl > 300:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            n1 = n2
            target1 = target2

        tl = _time_left(start_time, budget_s)
        if tl > 200:
            h_cd3, c5_cd3 = refined_cd(h2, n2, target2,
                                       time_limit=min(160, tl * 0.6),
                                       max_sweeps=25000,
                                       n_cand=120, top_k_eval=90,
                                       beta=50.0,
                                       max_no_improve=500,
                                       bisection_iters=20,
                                       rebuild_interval=30)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

        tl = _time_left(start_time, budget_s)
        if tl > 60:
            h_pol3, sc3 = slsqp_optimize(best_h, n1, target1,
                                         [2000, 10000],
                                         600, 1e-15)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    # Phase 7: Perturbed restarts
    tl = _time_left(start_time, budget_s)
    if tl > 100:
        h_pr, sc_pr = perturbed_restart(best_h, n1, target1,
                                        time_limit=min(80, tl * 0.6),
                                        noise_scales=[0.05, 0.12, 0.20])
        if sc_pr < best_score:
            best_score = sc_pr
            best_h = h_pr

    # Phase 8: Final CD
    tl = _time_left(start_time, budget_s)
    if tl > 80:
        h_cd_final, c5_final = refined_cd(best_h, n1, target1,
                                          time_limit=min(60, tl * 0.7),
                                          max_sweeps=30000,
                                          n_cand=150, top_k_eval=100,
                                          beta=80.0,
                                          max_no_improve=400,
                                          bisection_iters=22,
                                          rebuild_interval=20)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    # Phase 9: Final SLSQP
    tl = _time_left(start_time, budget_s)
    if tl > 50:
        h_final_pol, sc_final = slsqp_optimize(best_h, n1, target1,
                                              [5000, 20000, 80000],
                                              600, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    # Final
    n_out = len(best_h)
    final_h = project_feasible(best_h, n_out / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, n_out
