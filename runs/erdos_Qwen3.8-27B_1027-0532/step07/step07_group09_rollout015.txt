        h = np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def sparse_grad(h, corr, K=20):
    n = len(h)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    top_idx = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        k = int(k_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad /= K
    grad -= np.mean(grad)
    return grad

def nesterov_optimize(h_init, n, target, time_limit, K=20, lr_start=0.01, max_iter=50000):
    start = time.time()
    dx = 2.0 / n
    
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    momentum = np.zeros(n)
    mu = 0.9  # momentum parameter
    
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break
        
        # Nesterov lookahead
        h_y = h + mu * (h - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        
        corr = compute_corr(h_y)
        grad = sparse_grad(h_y, corr, K)
        
        # Adaptive learning rate
        progress = min(it / max_iter, 1.0)
        lr = lr_start * (1.0 - 0.9 * progress)
        
        h_new = project_feasible(h - lr * grad, target)
        
        if np.sum((h_new - h)**2) < 1e-20:
            no_improve += 100
        else:
            no_improve += 1
        
        h_prev = h.copy()
        h = h_new
        
        sc = compute_c5(h)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h.copy()
            no_improve = 0
            # Reset momentum on improvement
            mu = 0.9
    
    return best_h, best_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])

        else:
            w = np.ones(n_active) / n_active

        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

            update_change_row(change, best_j, h, n, shifts)

Actually, I realize I should also try a different approach to the Nesterov/method: instead of using the sparse gradient (top-K), I can use the **exact subgradient** of the max function. The subgradient of max(f_1, ..., f_m) at a point is the convex hull of the gradients of the active functions. A valid subgradient is the average of the active function gradients.

For our problem:
- f_k(h) = overlap[k] = ∑ h[i](1-h[i+k])
- ∇f_k(h)[p] = (1-h[p+k])*[valid] - h[p-k]*[valid]
- The active set is {k : f_k = max}
- A subgradient is the average of ∇f_k over active k

This is what the parent's `subgradient_cd` does. The issue is that it's a subgradient, not a gradient, so standard convergence results don't apply directly. But with Nesterov acceleration and a good learning rate schedule, it might work well.

Let me implement a proper Nesterov-accelerated projected subgradient method:

```python
def nesterov_subgradient(h_init, n, target, time_limit, max_iter=100000, K_active=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    
    # Nesterov parameters
    t = 1.0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 10000:
            break
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mu = (t - 1.0) / t_new
        h_y = h + mu * (h - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        
        # Compute correlation at h_y
        corr = np.correlate(h_y, 1.0 - h_y, mode='full')
        max_corr = np.max(corr)
        
        # Active set (top K)
        tol = 1e-6 * max_corr
        active = np.where(corr >= max_corr - tol)[0]
        if len(active) > K_active:
            active = np.argsort(corr)[-K_active:]
        
        # Subgradient
        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h_y[idxs[v1] - k])
            grad[v2] -= h_y[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)
        
        # Learning rate: O(1/sqrt(t))
        lr = 0.5 / np.sqrt(t_new)
        
        h_new = project_feasible(h - lr * grad, target)
        
        h_prev = h.copy()
        h = h_new
        t = t_new
        
        sc = compute_c5(h)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_score
```

The problem with this is that the learning rate O(1/sqrt(t)) gets very small quickly, and the method converges slowly (O(1/sqrt(T)) for non-smooth functions). For our problem with T ~ 20000 iterations, the improvement would be very small.

A better approach might be to use a **fixed but decreasing learning rate** with periodic restarts. Or use **FISTA** (Fast Iterative Shrinkage-Thresholding Algorithm) style.

Actually, for non-convex problems, Nesterov acceleration doesn't have the same guarantees. It might help in practice but could also cause oscillations.

def project_feasible(h, target_sum):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change
