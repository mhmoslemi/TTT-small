_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def correlate_gradient(h, k_stars, weights):
    """Compute weighted sum of gradients of corr[k] for given lags."""
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(k_stars, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)

        h[mask] += diff / mask.sum()

def get_active_gradient(h, active_lags, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(active_lags, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_G_ALPHA * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _G_ALPHA)

def _slsqp_eq_fun(x):
    return np.sum(x) - _G_TARGET

def _slsqp_eq_jac(x):
    return np.ones(_G_N)

def slsqp_refine(h_start, n, target, alphas, maxiter):
    global _G_N, _G_TARGET, _G_ALPHA
    _G_N = n
    _G_TARGET = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _G_ALPHA = alpha
        cons = [{'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac}]
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def cutting_plane(h_init, n, target, time_limit, max_iters=20000, max_cuts=200, tol=1e-10):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    cuts_g = []
    cuts_f = []
    cuts_h = []
    no_improve = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            cuts_g = cuts_g[-5:]
            cuts_f = cuts_f[-5:]
            cuts_h = cuts_h[-5:]
            no_improve = 0
        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])
        active_mask = corr >= corr[k_star] - 1e-6
        active_lags = np.where(active_mask)[0]
        n_act = len(active_lags)
        w = np.ones(n_act) / n_act
        g = get_active_gradient(h, active_lags, w)
        cuts_g.append(g)
        cuts_f.append(f_val)
        cuts_h.append(h.copy())
        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]
        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
            t_val = float(res.x[n])
            gap = float(np.max(compute_corr(h))) - t_val
            if gap < tol:
                break
        except Exception:
            break
    return best_h, best_score

def coordinate_descent(h_init, n, target, time_limit, max_sweeps=50000,
                       n_cand_start=20, n_cand_end=80):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 1000:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pi, pj = best_i, best_j
            change[pi, :] = 0.0
            change[pj, :] = 0.0
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

**Key idea: The problem might benefit from a "dual" perspective.**

Instead of minimizing max_k corr_k(h), we can think of it as a zero-sum game:
- Player 1 (minimizer) chooses h
- Player 2 (maximizer) chooses k
- Payoff: corr_k(h)

At equilibrium, Player 1's optimal strategy is h*, and Player 2's optimal strategy is a distribution over lags. The value of the game is C₅.

This suggests using a game-theoretic approach:
1. Start with a distribution p over lags
2. Given p, find h that minimizes E_p[corr_k(h)] (this is a quadratic optimization, still non-convex)
3. Given h, update p to put mass on the lags that maximize corr_k(h)
4. Iterate

Step 2 is still hard, but we can approximate it. If we fix the distribution p, the objective is:
E_p[corr_k] = sum_k p_k * sum_i h[i]*(1-h[i+k])
= sum_i h[i] - sum_k p_k * sum_i h[i]*h[i+k]
= sum_i h[i] - h^T (sum_k p_k S_k) h

The second term is a quadratic form. If Q = sum_k p_k S_k is positive semidefinite, then the objective is concave (we're maximizing a concave function), which is easy. But Q is generally not PSD.

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_G_ALPHA * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _G_ALPHA)

def _slsqp_eq_fun(x):
    return np.sum(x) - _G_TARGET

def _slsqp_eq_jac(x):
    return np.ones(_G_N)

def slsqp_refine(h_start, n, target, alphas, maxiter):
    global _G_N, _G_TARGET, _G_ALPHA
    _G_N = n
    _G_TARGET = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _G_ALPHA = alpha
        cons = [{'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac}]
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def cutting_plane(h_init, n, target, time_limit, max_iters=20000, max_cuts=200, tol=1e-10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    cuts_g = []
    cuts_f = []
    cuts_h = []
    no_improve = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 5000:
            cuts_g = cuts_g[-5:]
            cuts_f = cuts_f[-5:]
            cuts_h = cuts_h[-5:]
            no_improve = 0
        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])
        active_mask = corr >= corr[k_star] - 1e-6
        active_lags = np.where(active_mask)[0]
        n_act = len(active_lags)
        w = np.ones(n_act) / n_act
        g = get_active_gradient(h, active_lags, w)
        cuts_g.append(g)
        cuts_f.append(f_val)
        cuts_h.append(h.copy())
        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]
        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
            t_val = float(res.x[n])
            gap = float(np.max(compute_corr(h))) - t_val
            if gap < tol:
                break
        except Exception:
            break
    return best_h, best_score

def coordinate_descent(h_init, n, target, time_limit, max_sweeps=50000,
                       n_cand_start=20, n_cand_end=80):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 1500:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pi, pj = best_i, best_j
            change[pi, :] = 0.0
            change[pj, :] = 0.0
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def smooth_grad_descent(h_init, n, target, time_limit, alpha_start=200.0,
                        alpha_end=50000.0, max_iters=20000, step_init=0.01):
    start = time.time()
    dx = 2.0 / n
    h_curr = h_init.copy().astype(float)
    h_prev = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)
    t_k = 1.0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_curr + ((t_k - 1.0) / t_k) * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        progress = min(float(it) / max_iters, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        corr_y = compute_corr(y)
        c_max = np.max(corr_y)
        shifted = np.clip(alpha * (corr_y - c_max), -500.0, 0.0)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            p = np.ones(len(corr_y)) / len(corr_y)
        else:
            p = p / p_sum
        idxs = np.arange(n)
        mid = n - 1
        grad = np.zeros(n)
        active_p = p > 1e-15
        for m in range(len(corr_y)):
            if not active_p[m]:
                continue
            k = m - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += p[m] * (1.0 - y[idxs[v1] + k])
            grad[v2] -= p[m] * y[idxs[v2] - k]
        step = step_init / (1.0 + 0.01 * it)
        h_new = y - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        h_prev = h_curr.copy()
        h_curr = h_new
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_curr.copy()
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_TARGET, _G_ALPHA
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return max(0.0, budget_s - 30.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0

    if _tl() > 80:
        n1 = 800
        target1 = n1 / 2.0
        h800_interp = interpolate_to(best_h, n1, target1)
        h800_repeat = np.repeat(best_h, 2)
        h800_repeat = project_feasible(h800_repeat, target1)
        sc_interp = compute_c5(h800_interp)
        sc_repeat = compute_c5(h800_repeat)
        if sc_interp <= sc_repeat:
            h800 = h800_interp
        else:
            h800 = h800_repeat
        sc800 = compute_c5(h800)
        if sc800 < best_score:
            best_score = sc800
            best_h = h800
            best_n = n1
        if _tl() > 60:
            h800, sc = slsqp_refine(h800, n1, target1, [300, 1500], 500)
            if sc < best_score:
                best_score = sc
                best_h = h800

    if _tl() > 200:
        tl = min(400.0, _tl() * 0.45)
        h_cp, sc_cp = cutting_plane(best_h, best_n, best_n / 2.0, time_limit=tl,
                                    max_iters=20000, max_cuts=200, tol=1e-10)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp

    if _tl() > 150:
        tl = min(200.0, _tl() * 0.45)
        h_cd, sc_cd = coordinate_descent(best_h, best_n, best_n / 2.0, time_limit=tl,
                                         max_sweeps=40000, n_cand_start=25, n_cand_end=90)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if _tl() > 100:
        h_sls, sc_sls = slsqp_refine(best_h, best_n, best_n / 2.0,
                                      [5000, 20000], 700)
        if sc_sls < best_score:
            best_score = sc_sls
            best_h = h_sls

    if _tl() > 120:
        tl = min(100.0, _tl() * 0.5)
        h_sg, sc_sg = smooth_grad_descent(best_h, best_n, best_n / 2.0,
                                          time_limit=tl, alpha_start=100.0,
                                          alpha_end=30000.0, max_iters=10000,
                                          step_init=0.005)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if _tl() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h1200 = interpolate_to(best_h, n2, target2)
        sc1200 = compute_c5(h1200)
        if _tl() > 180:
            h1200, sc = cutting_plane(h1200, n2, target2,
                                       time_limit=min(180.0, _tl() * 0.5),
                                       max_iters=15000, max_cuts=150, tol=1e-10)
            sc = compute_c5(h1200)
            if sc < best_score:
                best_score = sc
                best_h = h1200
                best_n = n2
            if _tl() > 60:
                h1200, sc2 = coordinate_descent(h1200, n2, target2,
                                                 time_limit=min(50.0, _tl() * 0.5),
                                                 max_sweeps=15000, n_cand_start=20,
                                                 n_cand_end=60)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h1200

    if _tl() > 100:
        h_cp2, sc_cp2 = cutting_plane(best_h, best_n, best_n / 2.0,
                                      time_limit=min(100.0, _tl() * 0.5),
                                      max_iters=15000, max_cuts=150, tol=1e-10)
        if sc_cp2 < best_score:
            best_score = sc_cp2
            best_h = h_cp2

    if _tl() > 120:
        for attempt in range(4):
            if _tl() < 30:
                break
            noise_scale = 0.02 + 0.01 * attempt
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_n / 2.0)
            tl = min(45.0, _tl() * 0.35)
            h_ref, sc_ref = cutting_plane(h_pert, best_n, best_n / 2.0,
                                          time_limit=tl, max_iters=8000,
                                          max_cuts=100, tol=1e-9)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
                if _tl() > 30:
                    h_ref2, sc_ref2 = coordinate_descent(h_ref, best_n, best_n / 2.0,
                                                         time_limit=min(20.0, _tl() * 0.4),
                                                         max_sweeps=8000, n_cand_start=30,
                                                         n_cand_end=70)
                    if sc_ref2 < best_score:
                        best_score = sc_ref2
                        best_h = h_ref2

    if _tl() > 50:
        h_cd2, sc_cd2 = coordinate_descent(best_h, best_n, best_n / 2.0,
                                           time_limit=min(40.0, _tl() * 0.6),
                                           max_sweeps=20000, n_cand_start=40, n_cand_end=100)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    if _tl() > 30:
        h_fin, sc_fin = slsqp_refine(best_h, best_n, best_n / 2.0,
                                      [10000, 50000], 400)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    if _tl() > 20:
        h_cp_fin, sc_cp_fin = cutting_plane(best_h, best_n, best_n / 2.0,
                                            time_limit=min(15.0, _tl() * 0.7),
                                            max_iters=10000, max_cuts=100, tol=1e-11)
        if sc_cp_fin < best_score:
            best_score = sc_cp_fin
            best_h = h_cp_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
