import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = idxs + k < n
        wm = idxs - k >= 0
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def momentum_grad_descent(h_init, n, target, time_limit, max_steps=50000,
                          base_lr=0.05, momentum=0.9, n_active_schedule=None,
                          n_active_start=1, n_active_end=16):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = base_lr
    no_improve = 0
    
    if n_active_schedule is None:
        n_active_schedule = [1, 2, 4, 8, 16, 32, 64]
    
    schedule_len = len(n_active_schedule)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Determine n_active based on progress
        progress = min(step / max(max_steps, 1), 1.0)
        idx = min(int(progress * schedule_len), schedule_len - 1)
        n_active = n_active_schedule[idx]
        n_active = min(n_active, 2 * n - 1)
        
        # Get top active shifts
        top_shifts = np.argsort(corr)[-n_active:]
        
        # Compute weighted gradient
        c_max = corr[top_shifts[-1]]
        # Use softmax weights over active shifts
        diffs = c_max - corr[top_shifts]
        weights = np.exp(-10.0 * diffs)
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Momentum update
        v = momentum * v - lr * grad
        h_try = h + v
        h_try = project_feasible(h_try, target)
        
        sc = compute_c5(h_try)
        if sc < c5_val - 1e-12:
            h = h_try
        else:
            # Backtrack
            v *= momentum  # keep momentum but reduce step
            h_try = h - lr * grad  # no momentum, just gradient
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < c5_val - 1e-12:
                h = h_try
            else:
                lr *= 0.5
                if lr < 1e-8:
                    break
    
    return best_h, best_c5

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def improved_cd(h_init, n, target, time_limit, max_sweeps=30000,
                n_cand_start=30, n_cand_end=100, max_no_improve=300,
                n_active_cd=3):
    """Coordinate descent with multi-active gradient and improved candidate selection."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        # Multi-active gradient
        top_shifts = np.argsort(corr)[-n_active_cd:]
        c_max = corr[top_shifts[-1]]
        diffs = c_max - corr[top_shifts]
        weights = np.exp(-20.0 * diffs)
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                
                g = change[j] - change[i]
                
                # Binary search for optimal delta
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    
    return best_h, best_c5

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-n_active_cd:]
        c_max_val = corr[top_shifts[-1]]
        diffs = c_max_val - corr[top_shifts]
        weights = np.exp(-25.0 * diffs)
        wsum = np.sum(weights)
        if wsum < 1e-30:
            weights = np.ones(len(top_shifts)) / len(top_shifts)
        else:
            weights /= wsum

        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_perturb_restarts(h_best, n, target, time_limit, seed=42, n_rounds=6):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_score = compute_c5(h_best)

    for rnd in range(n_rounds):
        if time.time() - start > time_limit:
            break
        scale = 0.03 + 0.04 * rnd
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)

        h_ref, sc = slsqp_round(h_pert, n, target, [500, 5000], 800, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

        h_cd, sc_cd = improved_cd(h_ref, n, target,
                                  time_limit=min(20, max(0, time_limit - (time.time() - start))),
                                  max_sweeps=3000, n_cand_start=40, n_cand_end=80,
                                  max_no_improve=200)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    # Phase 1: SLSQP at initial resolution
    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, best_n, best_target,
                                [100, 500, 2000], 800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Momentum GD
    if tleft() > 60:
        h_gd, sc_gd = momentum_gd(best_h, best_n, best_target,
                                  time_limit=min(45, tleft() * 0.4))
        if sc_gd < best_score:
            best_score = sc_gd
            best_h = h_gd

    # Phase 3: Improved CD at initial resolution
    if tleft() > 80:
        h_cd, sc_cd = improved_cd(best_h, best_n, best_target,
                                  time_limit=min(90, tleft() * 0.5),
                                  max_sweeps=25000,
                                  n_cand_start=30, n_cand_end=90,
                                  max_no_improve=400, n_active_cd=3)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 4: SLSQP polish at initial resolution
    if tleft() > 50:
        h_pol2, sc2 = slsqp_round(best_h, best_n, best_target,
                                  [5000, 20000], 1000, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 5: Escalate to n=600
    n1 = 600
    if tleft() > 200:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 180:
            h1_pol, sc1p = slsqp_round(h1, n1, target1,
                                       [100, 500, 2000], 600, 1e-15)
            if sc1p < compute_c5(h1):
                h1 = h1_pol

        if tleft() > 140:
            h1_gd, sc1g = momentum_gd(h1, n1, target1,
                                      time_limit=min(40, tleft() * 0.3))
            if sc1g < best_score:
                best_score = sc1g
                best_h = h1_gd
                best_n = n1
                best_target = target1
            else:
                h1 = h1_gd

        if tleft() > 120:
            h1_cd, sc1c = improved_cd(h1, n1, target1,
                                      time_limit=min(100, tleft() * 0.5),
                                      max_sweeps=20000,
                                      n_cand_start=35, n_cand_end=100,
                                      max_no_improve=400, n_active_cd=4)
            if sc1c < best_score:
                best_score = sc1c
                best_h = h1_cd
                best_n = n1
                best_target = target1
            else:
                h1 = h1_cd

        if tleft() > 60:
            h1_foc, sc1f = focused_slsqp(h1, n1, target1,
                                         n_focus=5, alphas=[2000, 10000],
                                         maxiter=1200)
            if sc1f < best_score:
                best_score = sc1f
                best_h = h1_foc
                best_n = n1
                best_target = target1

    # Phase 6: Escalate to n=800
    n2 = 800
    if tleft() > 300:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        if tleft() > 250:
            h2_pol, sc2p = slsqp_round(h2, n2, target2,
                                       [100, 500, 2000], 600, 1e-15)
            if sc2p < compute_c5(h2):
                h2 = h2_pol

        if tleft() > 180:
            h2_gd, sc2g = momentum_gd(h2, n2, target2,
                                      time_limit=min(60, tleft() * 0.3))
            if sc2g < best_score:
                best_score = sc2g
                best_h = h2_gd
                best_n = n2
                best_target = target2
            else:
                h2 = h2_gd

        if tleft() > 150:
            h2_cd, sc2c = improved_cd(h2, n2, target2,
                                      time_limit=min(130, tleft() * 0.6),
                                      max_sweeps=25000,
                                      n_cand_start=40, n_cand_end=110,
                                      max_no_improve=500, n_active_cd=4)
            if sc2c < best_score:
                best_score = sc2c
                best_h = h2_cd
                best_n = n2
                best_target = target2
            else:
                h2 = h2_cd

        if tleft() > 80:
            h2_foc, sc2f = focused_slsqp(h2, n2, target2,
                                         n_focus=5, alphas=[5000, 20000],
                                         maxiter=1500)
            if sc2f < best_score:
                best_score = sc2f
                best_h = h2_foc
                best_n = n2
                best_target = target2

        if tleft() > 80:
            h2_cd2, sc2c2 = improved_cd(best_h, best_n, best_target,
                                        time_limit=min(70, tleft() * 0.5),
                                        max_sweeps=20000,
                                        n_cand_start=50, n_cand_end=120,
                                        max_no_improve=400, n_active_cd=5)
            if sc2c2 < best_score:
                best_score = sc2c2
                best_h = h2_cd2

    # Phase 7: Final SLSQP polish
    if tleft() > 60:
        h_fin, sc_fin = slsqp_round(best_h, best_n, best_target,
                                    [10000, 50000, 200000], 1500, 1e-15)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 8: More CD
    if tleft() > 80:
        h_cd_f, sc_cf = improved_cd(best_h, best_n, best_target,
                                    time_limit=min(80, tleft() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_start=60, n_cand_end=130,
                                    max_no_improve=300, n_active_cd=5)
        if sc_cf < best_score:
            best_score = sc_cf
            best_h = h_cd_f

    # Phase 9: Perturbation restarts
    if tleft() > 100:
        h_pr, sc_pr = multi_perturb_restarts(best_h, best_n, best_target,
                                             time_limit=min(90, tleft() * 0.6),
                                             seed=seed, n_rounds=5)
        if sc_pr < best_score:
            best_score = sc_pr
            best_h = h_pr

    # Phase 10: Final focused SLSQP
    if tleft() > 50:
        h_ff, sc_ff = focused_slsqp(best_h, best_n, best_target,
                                    n_focus=8, alphas=[50000, 200000],
                                    maxiter=2000)
        if sc_ff < best_score:
            best_score = sc_ff
            best_h = h_ff

    # Phase 11: Last moment CD
    if tleft() > 30:
        h_last, sc_last = improved_cd(best_h, best_n, best_target,
                                      time_limit=min(20, tleft() * 0.7),
                                      max_sweeps=5000,
                                      n_cand_start=70, n_cand_end=130,
                                      max_no_improve=100, n_active_cd=5)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
