def compute_c5(h):

def project_feasible(h, target_sum):

            h = np.clip(h + step, 0.0, 1.0)
            break

        h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def build_change_matrix(h, n):
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=40, n_cand_end=160, n_shifts=4,
            max_no_improve=400, anneal_temp=0.0, anneal_decay=0.999):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    change = build_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = np.sum(shift_weights)
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):

            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                b_d = delta_hi
                for _ in range(14):

                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 1000.0
                    if np.random.random() < np.exp(-delta_obj / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 2:
                change = build_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.95, 1e-10)

    return best_h, best_c5


def subgradient_momentum(h_init, n, target, time_limit, n_steps=20000,
                         step_scale=0.004, momentum=0.7, n_weights=5):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale
    vel = np.zeros(n)

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        p = np.exp(corr * 80.0)
        p /= p.sum()

        grad = np.zeros(n)
        top_k = np.argsort(p)[-n_weights:]
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        vel = momentum * vel + (1.0 - momentum) * grad
        h_try = h - eta * vel
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
            eta = min(eta * 1.02, step_scale * 3)
        else:
            h_try2 = h - 0.4 * eta * vel
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.88
                vel = 0.5 * vel

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturb(h_base, n, target, rng, mode=0):
    if mode == 0:
        scale = 0.04 + 0.06 * rng.random()
        h_pert = h_base + rng.randn(n) * scale
    elif mode == 1:
        freq = 3 + int(rng.random() * 8)
        phase = rng.random() * 2 * np.pi
        t = np.linspace(0, 2 * np.pi * freq, n)
        scale = 0.03 + 0.05 * rng.random()
        h_pert = h_base + scale * np.sin(t + phase)
    else:
        n_blocks = 4 + int(rng.random() * 6)
        block_size = n // n_blocks
        scales = rng.randn(n_blocks) * 0.05
        h_pert = h_base.copy()
        for b in range(n_blocks):
            h_pert[b * block_size:(b+1) * block_size] += scales[b]
    return project_feasible(h_pert, target)
