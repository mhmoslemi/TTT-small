**Approach: Use CVXPY with a relaxation**

The problem is: minimize max_k R(k) subject to sum(h) = n/2, 0 <= h <= 1.

R(k) = sum_i h[i] - sum_i h[i]*h[i+k]

The term sum_i h[i]*h[i+k] is quadratic. But if we replace it with a linear approximation at the current point, we get a linear program.

**Sequential Linear Programming (SLP)**:
1. At current h*, linearize R(k) around h*:
   R(k) ≈ R(k; h*) + sum_p (dR(k)/dh[p]) * (h[p] - h*[p])
   
   where dR(k)/dh[p] = (1-h*[p+k]) - h*[p-k]

2. Solve: min t subject to R(k; h*) + grad_k * (h - h*) <= t, sum(h) = n/2, 0 <= h <= 1
   This is a linear program!

3. Update h* and repeat.

This is a cutting-plane / linearization approach. Each iteration solves an LP, which is fast with efficient solvers.

import cvxpy as cp

def slp_step(h_cur, n, target, n_cuts=50):
    """One step of sequential linear programming."""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    c5_val = np.max(corr) * dx
    
    # Select top n_cuts correlations for the LP
    top_idxs = np.argsort(corr)[-n_cuts:]
    
    # Linearize each selected R(k) around h_cur
    # R(k) = sum_i h[i]*(1-h[i+k])
    # dR(k)/dh[p] = (1 - h_cur[p+k]) - h_cur[p-k] (with boundary handling)
    
    # Build the LP
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target
    ]
    
    for k_idx in top_idxs:
        k = k_idx - mid
        # Linearization of R(k) at h_cur
        R_k_val = corr[k_idx]
        grad_k = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad_k[v1] = (1.0 - h_cur[idxs[v1] - k])
        grad_k[v2] -= h_cur[idxs[v2] + k]
        
        # R(k) + grad_k dot (h - h_cur) <= t/dx
        # R(k) - grad_k dot h_cur + grad_k dot h <= t/dx
        R_k_lin = R_k_val - np.dot(grad_k, h_cur) + np.dot(grad_k, h_var)
        constraints.append(R_k_lin <= t_var / dx)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.GLOP)  # or CLP, or CBC
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_cur, c5_val
    
    if h_var.value is not None:
        h_new = project_feasible(np.array(h_var.value).flatten(), target)
        new_c5 = compute_c5(h_new)
        return h_new, new_c5
    
    return h_cur, c5_val

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

_gn = 0
_gt = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slp_iteration(h, n, target, n_cuts, trust_radius):
    """One SLP iteration: solve LP and return new h."""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Select top correlations
    if n_cuts is not None and n_cuts < n_corr:
        top_idxs = np.argsort(corr)[-n_cuts:]
    else:
        top_idxs = np.arange(n_corr)
    
    # Build gradients
    # For each k in top_idxs, compute grad_k
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h - trust_radius,
        h_var <= h + trust_radius
    ]
    
    for k_idx in top_idxs:
        k = int(k_idx) - mid
        R_k = corr[k_idx]
        
        # Gradient of R(k) w.r.t. h
        grad_k = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad_k[v1] = (1.0 - h[idxs[v1] - k])
        grad_k[v2] -= h[idxs[v2] + k]
        
        # Linearization: R(k) + grad_k dot (h_new - h) <= t/dx
        # R(k) - grad_k dot h + grad_k dot h_new <= t/dx
        offset = R_k - np.dot(grad_k, h)
        constraints.append(cp.sum(grad_k * h_var) + offset <= t_var / dx)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    # Try solvers
    for solver in [cp.GLOP, cp.CBC, cp.GLPK]:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None:
                h_new = project_feasible(np.array(h_var.value).flatten(), target)
                return h_new
        except Exception:
            continue
    
    return None

                 n_cuts=200, trust_start=0.5, trust_min=0.01):
    """SLP optimization with adaptive trust region."""
    start = time.time()
    h = h_init.copy().astype(float)
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    trust = trust_start
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        h_new = slp_iteration(h, n, target, n_cuts, trust)
        
        if h_new is None:
            break
        
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            # Increase trust region slightly
            trust = min(trust * 1.05, trust_start)
        else:
            h = h  # stay
            no_improve += 1
            # Decrease trust region
            trust = max(trust * 0.9, trust_min)
        
        if no_improve > 100:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _slp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _slp_cons(x):
    return np.sum(x) - _gt


def _slp_jac(x):
    return np.ones(_gn)


def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _gn, _gt
    _gn = n
    _gt = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slp_cons, 'jac': _slp_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_slp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0

    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def slp_solve_lp(h, n, target, n_cuts, trust_radius):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    corr = np.correlate(h, 1.0 - h, mode='full')

    if n_cuts is not None and n_cuts < n_corr:
        top_idxs = np.argsort(corr)[-n_cuts:]
    else:
        top_idxs = np.arange(n_corr)

    h_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        h_var >= h - trust_radius,
        h_var <= h + trust_radius
    ]

    for k_idx in top_idxs:
        k = int(k_idx) - mid
        R_k = corr[k_idx]
        grad_k = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad_k[v1] = (1.0 - h[idxs[v1] - k])
        grad_k[v2] -= h[idxs[v2] + k]
        offset = R_k - np.dot(grad_k, h)
        constraints.append(np.dot(grad_k, h_var) + offset <= t_var / dx)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    for solver in [cp.GLOP, cp.CBC, cp.GLPK]:
        try:
            prob.solve(solver=solver, verbose=False)
            if h_var.value is not None:
                return np.array(h_var.value).flatten()
        except Exception:
            continue
    return None


def slp_optimize(h_init, n, target, time_limit, max_iters=5000,
                 n_cuts=100, trust_start=0.3, trust_min=0.005):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    trust = trust_start
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        h_lp = slp_solve_lp(h, n, target, n_cuts, trust)
        if h_lp is None:
            break

        h_lp = project_feasible(h_lp, target)
        new_c5 = compute_c5(h_lp)

        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_lp.copy()
            h = h_lp
            no_improve = 0
            trust = min(trust * 1.03, trust_start)
        else:
            direction = h_lp - h
            norm_d = np.linalg.norm(direction)
            if norm_d < 1e-12:
                trust = max(trust * 0.85, trust_min)
                no_improve += 1
            else:
                direction = direction / norm_d * min(norm_d, trust)
                lo_s, hi_s = 0.0, 1.0
                best_step = 0.0
                best_step_c5 = compute_c5(h)
                for _ in range(8):
                    mid_s = 0.5 * (lo_s + hi_s)
                    h_trial = project_feasible(h + mid_s * direction, target)
                    trial_c5 = compute_c5(h_trial)
                    if trial_c5 < best_step_c5:
                        best_step_c5 = trial_c5
                        best_step = mid_s
                        lo_s = mid_s
                    else:
                        hi_s = mid_s
                if best_step > 1e-6:
                    h = project_feasible(h + best_step * direction, target)
                    if compute_c5(h) < best_c5 - 1e-12:
                        best_c5 = compute_c5(h)
                        best_h = h.copy()

                    else:
                        no_improve += 1
                        trust = max(trust * 0.9, trust_min)
                else:
                    no_improve += 1
                    trust = max(trust * 0.8, trust_min)

def subgrad_momentum(h_init, n, target, time_limit, max_iter=20000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    mu = 0.85
    beta = 30.0
    step = 0.03
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        corr_scaled = corr * dx * beta
        corr_scaled -= np.max(corr_scaled)
        weights = np.exp(corr_scaled)
        weights /= np.sum(weights)

        grad = np.zeros(n)
        active_idxs = np.where(weights > 1e-10)[0]
        for k_idx in active_idxs:
            k = int(k_idx) - mid
            w = weights[k_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        v = mu * v - step * grad
        h_new = project_feasible(h + v, target)
        new_c5 = compute_c5(h_new)

        if new_c5 < best_c5 - 1e-13:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            beta = min(beta * 1.01, 3000.0)
        else:
            h = h_new
            no_improve += 1
            step *= 0.97
            beta = max(beta * 0.995, 10.0)

        if it % 200 == 0:
            v *= 0.5
        if no_improve > 500:
            break

    return best_h, best_c5

        h_slp, sc_slp = slp_optimize(best_h, current_n, current_target,
                                     time_limit=min(200, tleft() * 0.5),
                                     max_iters=3000, n_cuts=80,
                                     trust_start=0.2, trust_min=0.01)
        if sc_slp < best_score:
            best_score = sc_slp
            best_h = h_slp

    if tleft() > 200:
        h_cd, sc_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(180, tleft() * 0.5),
                              max_sweeps=40000,
                              n_cand_start=20, n_cand_end=60,
                              max_no_improve=500)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 200:
        h_ms, sc_ms = multi_k_cd(best_h, current_n, current_target,
                                 time_limit=min(180, tleft() * 0.5),
                                 max_sweeps=25000, n_cand=45,
                                 n_top_k=5, max_no_improve=400)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms

    if tleft() > 350:
        n1 = 480
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            current_n = n1
            current_target = target1

        if tleft() > 250:
            h_slp1, sc_slp1 = slp_optimize(best_h, current_n, current_target,
                                           time_limit=min(200, tleft() * 0.55),
                                           max_iters=2500, n_cuts=80,
                                           trust_start=0.15, trust_min=0.01)
            if sc_slp1 < best_score:
                best_score = sc_slp1
                best_h = h_slp1

        if tleft() > 150:
            h_cd1, sc_cd1 = pair_cd(best_h, current_n, current_target,
                                    time_limit=min(130, tleft() * 0.5),
                                    max_sweeps=25000,
                                    n_cand_start=15, n_cand_end=55,
                                    max_no_improve=400)
            if sc_cd1 < best_score:
                best_score = sc_cd1
                best_h = h_cd1

        if tleft() > 100:
            h_sls1, sc_sls1 = slsqp_refine(best_h, current_n, current_target,
                                           [2000, 10000, 50000],
                                           600, 1e-15)
            if sc_sls1 < best_score:
                best_score = sc_sls1
                best_h = h_sls1

    if tleft() > 250:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.10, 0.20]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_slp_p, sc_slp_p = slp_optimize(h_pert, current_n, current_target,
                                             time_limit=min(60, tleft() * 0.3),
                                             max_iters=1500, n_cuts=60,
                                             trust_start=0.15, trust_min=0.01)
            if sc_slp_p < best_score:
                best_score = sc_slp_p
                best_h = h_slp_p
            if tleft() > 100:
                h_cd_p, sc_cd_p = pair_cd(h_slp_p, current_n, current_target,
                                          time_limit=min(50, tleft() * 0.3),
                                          max_sweeps=10000,
                                          n_cand_start=25, n_cand_end=50,
                                          max_no_improve=300)
                if sc_cd_p < best_score:
                    best_score = sc_cd_p
                    best_h = h_cd_p

    if tleft() > 200:
        n2 = 560
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if tleft() > 150:
            h_slp2, sc_slp2 = slp_optimize(h2, n2, target2,
                                           time_limit=min(120, tleft() * 0.5),
                                           max_iters=2000, n_cuts=70,
                                           trust_start=0.12, trust_min=0.01)
            if sc_slp2 < best_score:
                best_score = sc_slp2
                best_h = h_slp2
                current_n = n2
                current_target = target2
            if tleft() > 80:
                h_cd2, sc_cd2 = pair_cd(best_h, current_n, current_target,
                                        time_limit=min(70, tleft() * 0.5),
                                        max_sweeps=15000,
                                        n_cand_start=15, n_cand_end=45,
                                        max_no_improve=300)
                if sc_cd2 < best_score:
                    best_score = sc_cd2
                    best_h = h_cd2

    if tleft() > 100:
        h_sls_f, sc_sls_f = slsqp_refine(best_h, current_n, current_target,
                                         [5000, 20000, 100000],
                                         800, 1e-15)
        if sc_sls_f < best_score:
            best_score = sc_sls_f
            best_h = h_sls_f

    if tleft() > 80:
        h_cd_f, sc_cd_f = pair_cd(best_h, current_n, current_target,
                                  time_limit=min(60, tleft() * 0.7),
                                  max_sweeps=20000,
                                  n_cand_start=30, n_cand_end=80,
                                  max_no_improve=400)
        if sc_cd_f < best_score:
            best_score = sc_cd_f
            best_h = h_cd_f

    if tleft() > 40:
        h_sub_f, sc_sub_f = subgrad_momentum(best_h, current_n, current_target,
                                             time_limit=min(30, tleft() * 0.6))
        if sc_sub_f < best_score:
            best_score = sc_sub_f
            best_h = h_sub_f

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
