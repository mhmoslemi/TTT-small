        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break

    return best_h, best_score


def compute_corr_grad_single(h, k_star):
    n = len(h)
    mid = n - 1
    k = int(k_star) - mid
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad


def compute_corr_grad_multi(h, k_stars, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_star, w in zip(k_stars, weights):
        k = int(k_star) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad


def interceptive_cd(h_init, n, target, time_limit, max_sweeps=30000,
                    n_cand_start=20, n_cand_end=80, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    _rebuild_change(h, n, shifts, idxs, change)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        active_mask = corr >= corr[k_star_idx] - 1e-7
        if active_mask.sum() <= 3:
            grad = compute_corr_grad_single(h, k_star_idx)
        else:
            active_lags = np.where(active_mask)[0]
            w = np.ones(len(active_lags)) / len(active_lags)
            grad = compute_corr_grad_multi(h, active_lags, w)
        grad = grad - np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        current_ncand = max(current_ncand, 10)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-13
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pi = best_i
            pj = best_j
            change[pi, :] = 0.0
            change[pj, :] = 0.0
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    return best_h, best_c5


def _rebuild_change(h, n, shifts, idxs, change):
    change[:] = 0.0
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]


def interceptive_cd_aperiodic(g_init, n_full, time_limit, max_sweeps=20000,
                               n_cand_start=15, n_cand_end=60, max_no_improve=300):
    nh = n_full // 2
    dx = 2.0 / n_full
    n_corr_full = 2 * n_full - 1
    shifts_full = np.arange(-(n_full - 1), n_full)
    idxs_full = np.arange(n_full)

    g = g_init.copy().astype(float)
    h = np.concatenate([g, 1.0 - g])
    best_c5 = compute_c5(h)
    best_g = g.copy()
    consecutive_no_improve = 0

    change_full = np.zeros((n_full, n_corr_full))
    for p in range(n_full):
        pos_mask = (p + shifts_full >= 0) & (p + shifts_full < n_full)
        neg_mask = (p - shifts_full >= 0) & (p - shifts_full < n_full)
        change_full[p, pos_mask] += (1.0 - h[p + shifts_full[pos_mask]])
        change_full[p, neg_mask] -= h[p - shifts_full[neg_mask]]

    dir_g = np.zeros((nh, n_corr_full))
    for i in range(nh):
        dir_g[i] = change_full[i] - change_full[i + nh]

    for sweep in range(max_sweeps):
        if time.time() - start_cd() > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_g = g.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n_full - 1)

        grad_g = np.zeros(nh)
        for i in range(nh):
            grad_g[i] = np.dot(dir_g[i], corr)

        sorted_idx = np.argsort(grad_g)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, nh // 2)
        current_ncand = max(current_ncand, 5)

        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(0.0, -(g[j]))
                delta_hi = min(g[i], 1.0 - g[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = dir_g[j] - dir_g[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-13
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            g[best_i] -= best_delta
            g[best_j] += best_delta
            g = np.clip(g, 0.0, 1.0)
            h = np.concatenate([g, 1.0 - g])
            di = best_i
            dj = best_j
            change_full[di, :] = 0.0
            change_full[di + nh, :] = 0.0
            change_full[dj, :] = 0.0
            change_full[dj + nh, :] = 0.0
            for p in [di, di + nh, dj, dj + nh]:
                pos_mask = (p + shifts_full >= 0) & (p + shifts_full < n_full)
                neg_mask = (p - shifts_full >= 0) & (p - shifts_full < n_full)
                change_full[p, pos_mask] += (1.0 - h[p + shifts_full[pos_mask]])
                change_full[p, neg_mask] -= h[p - shifts_full[neg_mask]]
            for i in range(nh):
                dir_g[i] = change_full[i] - change_full[i + nh]

    return best_g, best_c5

def nesterov_cd(h_init, n, target, time_limit, max_iters=50000, step_scale=0.5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()

        k_star = int(np.argmax(corr))
        k = k_star - mid

        active_mask = corr >= corr[k_star] - 1e-6
        if active_mask.sum() <= 5:
            grad = compute_corr_grad_single(h, k_star)
        else:
            active_lags = np.where(active_mask)[0]
            w = np.ones(len(active_lags)) / len(active_lags)
            grad = compute_corr_grad_multi(h, active_lags, w)

        grad = grad - np.mean(grad)
        step = step_scale / (1.0 + it * 0.01)

        h_new = h - step * grad
        h_new = project_feasible(h_new, target)

        h = h_new
        pi_ch = np.where(np.abs(np.diff(h, prepend=h[0])) > 1e-10)[0]
        for p in pi_ch:
            pos_mask = (p + shifts >= 0) & (p + shifts < n)
            neg_mask = (p - shifts >= 0) & (p - shifts < n)
            change[p, :] = 0.0
            change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
            change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 120:
        h_cp, sc = cutting_plane_method(best_h, n0, target0,
                                         time_limit=min(100, tleft() * 0.5),
                                         max_iters=8000, tol=1e-9)
        if sc < best_score:
            best_score = sc
            best_h = h_cp

    if tleft() > 100:
        h_cd, sc = interceptive_cd(best_h, n0, target0,
                                    time_limit=min(100, tleft() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_start=15, n_cand_end=60,
                                    max_no_improve=300)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    if tleft() > 100:
        h_nes, sc_nes = nesterov_cd(best_h, n0, target0,
                                     time_limit=min(80, tleft() * 0.5),
                                     max_iters=30000, step_scale=0.3)
        if sc_nes < best_score:
            best_score = sc_nes
            best_h = h_nes

    if tleft() > 50:
        h_cd2, sc2 = interceptive_cd(best_h, n0, target0,

                                      n_cand_start=30, n_cand_end=80,

        if sc2 < best_score:

        best_h_800 = h800

    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
