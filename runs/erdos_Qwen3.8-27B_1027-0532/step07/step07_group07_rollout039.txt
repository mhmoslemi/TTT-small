```python
def nag_optimize(h_init, n, target, time_limit, alpha_start=100, alpha_end=10000, 
                 lr_start=0.01, lr_end=0.001, max_iter=50000):
    """Nesterov Accelerated Gradient with projection"""
    start = time.time()
    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 1.0
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break
            
        progress = it / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Nesterov extrapolation
        beta = (t - 1) / (t + 1)
        y = h + beta * (h - h_prev)
        y = project_feasible(y, target)
        
        # Gradient at y
        grad = smooth_max_grad(y, alpha)
        
        # Gradient step
        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)
        
        # Evaluate
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        t = (1 + np.sqrt(1 + 4 * t * t)) / 2
        h_prev = h
        h = h_new
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def sparse_grad(h, alpha, n_top=10):
    """Compute sparse gradient using only top n_top shifts."""
    n = len(h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Get top n_top shifts
    top_idx = np.argpartition(corr, -n_top)[-n_top:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    
    # Compute weights for these shifts
    scaled = alpha * dx * corr[top_idx]
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    weights = exp_vals / np.sum(exp_vals)
    
    idxs = np.arange(n)
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = int(top_idx[idx]) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad *= dx
    return grad

def adam_optimize(h_init, n, target, time_limit, alpha_start=100, alpha_end=5000,
                  n_top=10, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                  max_iter=100000):
    """Adam optimizer with sparse gradient and increasing alpha."""
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break
        
        t += 1
        progress = min(it / max_iter, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr_t = lr * (1.0 - 0.5 * progress)
        
        grad = sparse_grad(h, alpha, n_top)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                h = best_h.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
        
        h = h_new
    
    return best_h, best_c5
```

Let me define the move as:

- Select m random indices to decrease (call them I- = {j_1, ..., j_m})
- Move δ from I- to I+ (i.e., h[i_k] += δ/m, h[j_k] -= δ/m for all k)
- Choose δ randomly from [0, δ_max]

This maintains the sum constraint (to first order) and the box constraint (if δ is small enough).

After the move, project to [0,1] and to the exact sum constraint.

For the temperature schedule, I'll use:
T(t) = T0 * exp(-t / τ)

where T0 is chosen so that initially ~50% of moves are accepted, and τ is chosen so that T reaches a very small value at the end of the SA phase.

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def _time_left(t0, budget):
    return budget - 30.0 - (time.time() - t0)


def _obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = _g_alpha * dx * corr
    m = np.max(scaled)
    return m + np.log(np.sum(np.exp(scaled - m))) / _g_alpha


def _grad(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = _g_alpha * dx * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    s = np.sum(exp_s)
    if s < 1e-300:
        return np.zeros(n)
    wts = exp_s / s
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    g = np.zeros(n)
    for ki in range(n_corr):
        w = wts[ki]
        if w < 1e-12:
            continue
        k = ki - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        g[v1] += w * (1.0 - x[idxs[v1] - k])
        g[v2] -= w * x[idxs[v2] + k]
    g *= dx
    return g


def _cons(x):
    return np.sum(x) - _g_target


def slsqp(h0, n, tgt, alphas, maxiter=500):
    global _g_alpha, _g_target
    _g_target = tgt
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons}]
    h = h0.copy()
    sc = compute_c5(h)
    for a in alphas:
        _g_alpha = a
        try:
            r = minimize(_obj, h, jac=_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(r.x, tgt)
            s = compute_c5(h_new)
            if s < sc:
                sc = s
                h = h_new
        except Exception:
            pass
    return h, sc


def adam_opt(h0, n, tgt, t0, budget, alpha_lo=50.0, alpha_hi=20000.0,
             lr0=0.03, n_top=8, max_it=80000):
    start = time.time()
    h = h0.copy()
    best = h.copy()
    best_sc = compute_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stuck = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    for it in range(max_it):
        if time.time() - start > 120 or _time_left(t0, budget) < 40:
            break
        if stuck > 6000:
            break
        t += 1
        prog = min(it / max_it, 1.0)
        alpha = alpha_lo * (alpha_hi / alpha_lo) ** prog
        lr = lr0 * (1.0 - 0.5 * prog)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        top = np.argpartition(corr, -n_top)[-n_top:]
        sc_ = alpha * dx * corr[top]
        mm = np.max(sc_)
        ew = np.exp(sc_ - mm)
        ww = ew / np.sum(ew)
        g = np.zeros(n)
        for ii in range(n_top):
            w = ww[ii]
            if w < 1e-12:
                continue
            k = int(top[ii]) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v1] += w * (1.0 - h[idxs[v1] - k])
            g[v2] -= w * h[idxs[v2] + k]
        g *= dx
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, tgt)
        s = compute_c5(h_new)
        if s < best_sc - 1e-15:
            best_sc = s
            best = h_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 2000:
                h = best.copy()
                m[:] = 0.0
                v[:] = 0.0
                stuck = 0
        h = h_new
    return best, best_sc


def build_change(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]
    return change


def upd_change(change, h, p, shifts, n):
    change[p, :] = 0.0
    pm = (p + shifts >= 0) & (p + shifts < n)
    nm = (p - shifts >= 0) & (p - shifts < n)
    change[p, pm] += (1.0 - h[p + shifts[pm]])
    change[p, nm] -= h[p - shifts[nm]]


def pair_cd(h0, n, tgt, t0, budget, time_lim, n_cand_s=30, n_cand_e=70,

def interp_to(h_src, n_dst, tgt_dst):
    n_s = len(h_src)
    if n_s == n_dst:
        return project_feasible(h_src.copy(), tgt_dst)
    xn = np.linspace(0.0, 1.0, n_dst)
    xo = np.linspace(0.0, 1.0, n_s)
    ho = np.interp(xn, xo, h_src)
    return project_feasible(ho, tgt_dst)

                best_h = h8b
        tl = _time_left(t0, budget_s)
        if tl > 150:
            h8c, s8c = adam_opt(h8b, n8, tgt8, t0, budget_s,
                                alpha_lo=100, alpha_hi=50000, lr0=0.02, n_top=10)
            if s8c < best_sc:
                best_sc = s8c
                best_h = h8c
        tl = _time_left(t0, budget_s)
        if tl > 200:
            h8d, s8d = pair_cd(best_h, n8, tgt8, t0, budget_s,
                               min(180, tl * 0.7), n_cand_s=30, n_cand_e=75,
                               top_k=50, n_sh=3, max_noimp=700)
            if s8d < best_sc:
                best_sc = s8d
                best_h = h8d
        tl = _time_left(t0, budget_s)
        if tl > 80:
            h8e, s8e = multi_block_cd(best_h, n8, tgt8, t0, budget_s,
                                      min(60, tl * 0.6), block_size=3,
                                      n_cand=40, top_k=30, max_sweeps=15000)
            if s8e < best_sc:
                best_sc = s8e
                best_h = h8e
        tl = _time_left(t0, budget_s)
        if tl > 60:
            h8f, s8f = slsqp(best_h, n8, tgt8, [10000, 50000], 400)
            if s8f < best_sc:
                best_sc = s8f
                best_h = h8f

    tl = _time_left(t0, budget_s)
    if tl > 200:
        n12 = 1200
        tgt12 = n12 / 2.0
        h12 = interp_to(best_h, n12, tgt12)
        sc12 = compute_c5(h12)
        if sc12 < best_sc:
            best_sc = sc12
            best_h = h12
            best_n = n12
        tl = _time_left(t0, budget_s)
        if tl > 100:
            h12b, s12b = adam_opt(h12, n12, tgt12, t0, budget_s,
                                  alpha_lo=100, alpha_hi=50000, lr0=0.015, n_top=10)
            if s12b < best_sc:
                best_sc = s12b
                best_h = h12b
        tl = _time_left(t0, budget_s)
        if tl > 150:
            h12c, s12c = pair_cd(best_h, n12, tgt12, t0, budget_s,
                                 min(130, tl * 0.7), n_cand_s=25, n_cand_e=60,
                                 top_k=40, n_sh=3, max_noimp=500)
            if s12c < best_sc:
                best_sc = s12c
                best_h = h12c
        tl = _time_left(t0, budget_s)
        if tl > 50:
            h12d, s12d = slsqp(best_h, n12, tgt12, [20000, 80000], 300)
            if s12d < best_sc:
                best_sc = s12d
                best_h = h12d

    tl = _time_left(t0, budget_s)
    if tl > 80:
        h_ss, s_ss = stochastic_hill(best_h, best_n, best_n / 2.0,
                                     t0, budget_s, min(60, tl * 0.6))
        if s_ss < best_sc:
            best_sc = s_ss
            best_h = h_ss

    tl = _time_left(t0, budget_s)
    if tl > 60:
        rng = np.random.RandomState(seed)
        for ns in [0.03, 0.08, 0.15]:
            tl = _time_left(t0, budget_s)
            if tl < 40:
                break
            hp = best_h + rng.randn(best_n) * ns
            hp = project_feasible(hp, best_n / 2.0)
            hr, sr = slsqp(hp, best_n, best_n / 2.0, [500, 2000, 10000], 400)
            if sr < best_sc:
                best_sc = sr
                best_h = hr

    tl = _time_left(t0, budget_s)
    if tl > 40:
        h_cd, s_cd = pair_cd(best_h, best_n, best_n / 2.0, t0, budget_s,
                             min(35, tl * 0.7), n_cand_s=40, n_cand_e=90,
                             top_k=60, n_sh=3, max_noimp=300)
        if s_cd < best_sc:
            best_sc = s_cd
            best_h = h_cd

    tl = _time_left(t0, budget_s)
    if tl > 30:
        h_fin, s_fin = slsqp(best_h, best_n, best_n / 2.0,
                             [50000, 200000], 300)
        if s_fin < best_sc:
            best_sc = s_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_sc = compute_c5(final_h)
    return final_h, final_sc, len(final_h)
```
