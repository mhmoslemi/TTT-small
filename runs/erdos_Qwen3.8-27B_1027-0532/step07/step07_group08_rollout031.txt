import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1

        change[p, neg_mask] -= h[p - shifts[neg_mask]]

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):

            best_h = h_sym

    # Phase 2: Asymmetric SLSQP
    if tleft() > 50:
        deadline = time.time() + min(35, tleft() * 0.3)
        h_asym, sc_asym = slsqp_round(best_h, n0, target0,
                                       [200, 1000, 5000, 20000],
                                       500, 1e-14, deadline=deadline)
        if sc_asym < best_score:
            best_score = sc_asym
            best_h = h_asym

    # Phase 3: CD at n=400
    if tleft() > 250:
        h_cd, sc_cd = fast_cd(best_h, n0, target0,
                               time_limit=min(150, tleft() * 0.5),
                               n_cand=28, n_active=4,
                               max_sweeps=10000, reheat_every=70,
                               reheat_scale=0.03, max_stall=250)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 4: SLSQP after CD
    if tleft() > 50:
        deadline = time.time() + min(35, tleft() * 0.3)
        h_p, sc_p = slsqp_sym_round(best_h, n0, target0,
                                     [1000, 5000, 20000, 80000],
                                     600, 1e-14, deadline=deadline)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p

    # Phase 5: Interpolate to n=800
    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        current_n = n1
        current_target = target1

        if tleft() > 250:
            h_cd1, sc1 = fast_cd(h1, n1, target1,
                                  time_limit=min(160, tleft() * 0.5),
                                  n_cand=25, n_active=4,
                                  max_sweeps=8000, reheat_every=60,
                                  reheat_scale=0.025, max_stall=220)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_cd1

        if tleft() > 50:
            deadline = time.time() + min(35, tleft() * 0.3)
            h_p1, sc_p1 = slsqp_sym_round(best_h, current_n, current_target,
                                           [2000, 10000, 50000],
                                           500, 1e-14, deadline=deadline)
            if sc_p1 < best_score:
                best_score = sc_p1
                best_h = h_p1

    # Phase 6: More CD at current resolution
    if tleft() > 180:
        h_cd2, sc_cd2 = fast_cd(best_h, current_n, current_target,
                                 time_limit=min(110, tleft() * 0.55),
                                 n_cand=30, n_active=5,
                                 max_sweeps=8000, reheat_every=50,
                                 reheat_scale=0.02, max_stall=200)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 7: Try n=1200
    if tleft() > 300:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        current_n = n2
        current_target = target2

        if tleft() > 200:
            h_cd2, sc_cd2 = fast_cd(h2, n2, target2,
                                     time_limit=min(130, tleft() * 0.5),
                                     n_cand=22, n_active=4,
                                     max_sweeps=6000, reheat_every=50,
                                     reheat_scale=0.025, max_stall=200)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_h = h_cd2

        if tleft() > 50:
            deadline = time.time() + min(35, tleft() * 0.4)
            h_p2, sc_p2 = slsqp_sym_round(best_h, current_n, current_target,
                                           [5000, 20000],
                                           400, 1e-14, deadline=deadline)
            if sc_p2 < best_score:
                best_score = sc_p2
                best_h = h_p2

    # Phase 8: Multi-start perturbation
    if tleft() > 120:
        scales = [0.015, 0.04, 0.07]
        for attempt in range(3):
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(current_n) * scales[attempt]
            h_pert = project_feasible(h_pert, current_target)
            h_try, sc_try = fast_cd(h_pert, current_n, current_target,
                                     time_limit=min(25, tleft() * 0.25),
                                     n_cand=25, n_active=3,
                                     max_sweeps=5000, reheat_every=40,
                                     reheat_scale=0.02, max_stall=150)
            if tleft() > 30:
                deadline = time.time() + min(15, tleft() * 0.3)
                h_try2, sc_try2 = slsqp_sym_round(h_try, current_n, current_target,
                                                   [2000, 10000],
                                                   300, 1e-14, deadline=deadline)
                sc_try2 = compute_c5(h_try2)
            else:
                h_try2 = h_try
                sc_try2 = compute_c5(h_try)
            if sc_try2 < best_score:
                best_score = sc_try2
                best_h = h_try2

    # Phase 9: Final symmetric SLSQP
    if tleft() > 40:
        deadline = time.time() + min(30, tleft() * 0.6)
        h_final, sc_final = slsqp_sym_round(best_h, current_n, current_target,
                                             [5000, 20000, 80000, 200000],
                                             600, 1e-14, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 10: Final asymmetric SLSQP
    if tleft() > 30:
        deadline = time.time() + min(20, tleft() * 0.6)
        h_final2, sc_final2 = slsqp_round(best_h, current_n, current_target,
                                           [10000, 50000, 150000],
                                           500, 1e-14, deadline=deadline)
        if sc_final2 < best_score:
            best_score = sc_final2
            best_h = h_final2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
