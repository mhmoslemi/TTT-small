def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-8)[0]
    else:
        idx = np.where(h > 1e-8)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

Actually, I realize I should be more careful about the Bisection. The function f(v) = max_m [corr_m + (v-old)*B_m] (for m≠mid) and corr_mid + (v-old)(1-v-old) (for m=mid) is NOT necessarily convex because of the k=0 term.

For the bisection to work, I need the subgradient to be monotonically non-decreasing in v. For the linear parts (m≠mid), the subgradient is B[active] which is constant for each "region" of v. As v increases, the active m changes, and the subgradient changes. For the max of linear functions, the subgradient IS monotonically non-decreasing (this is a property of convex functions).

For the k=0 term, the subgradient is 1-2v, which IS monotonically decreasing in v. So if k=0 is the active constraint, the subgradient might not be monotonically non-decreasing, and bisection might fail.

To handle this, I'll modify the bisection: instead of purely bisection, I'll do a "sweep" of the candidate v values and pick the best. Since the function is a max of O(n) lines (plus one parabola), the minimum must occur at:
- v = 0
- v = 1
- An intersection of two lines (for m1 ≠ m2, m1, m2 ≠ mid): (v-old)*B[m1] + corr[m1] = (v-old)*B[m2] + corr[m2], so v = old + (corr[m2]-corr[m1])/(B[m1]-B[m2])
- An intersection of a line (m≠mid) and the parabola (mid): corr[m] + (v-old)*B[m] = corr[mid] + (v-old)(1-v-old)
- The vertex of the parabola: v = 0.5 (if it's in [0,1] and is the minimum of the parabola)

Finding all these candidates is O(n²) which is too slow per coordinate.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-8)[0]
    else:
        idx = np.where(h > 1e-8)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def exact_cd(h_init, target, max_sweeps, time_limit, seed=42, recompute_every=20):
    n = len(h_init)
    dx = 2.0 / n
    mid = n - 1
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    start = time.time()
    no_improve = 0
    rng = np.random.RandomState(seed)
    
    # Precompute shift indices
    ks_all = np.arange(2 * n - 1) - mid
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        # Periodic recomputation to avoid drift
        if sweep % recompute_every == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
        
        cur_c5 = float(np.max(corr)) * dx
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        order = rng.permutation(n)
        
        for j in order:
            old_v = h[j]
            
            ip = j + ks_all
            im = j - ks_all
            B = np.zeros(2 * n - 1)
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            B[vp] += (1.0 - h[ip[vp]])
            B[vm] -= h[im[vm]]
            
            # Bisection
            lo_v = 0.0
            hi_v = 1.0
            for _ in range(35):
                mid_v = 0.5 * (lo_v + hi_v)
                delta = mid_v - old_v
                vals = corr + delta * B
                vals[mid] = corr[mid] + delta * (1.0 - mid_v - old_v)
                active = int(np.argmax(vals))
                if active == mid:
                    sg = 1.0 - 2.0 * mid_v
                else:
                    sg = B[active]
                if sg > 0:
                    hi_v = mid_v
                else:
                    lo_v = mid_v
            
            v_best = 0.5 * (lo_v + hi_v)
            
            # Evaluate candidates
            cands = [0.0, 1.0, v_best, old_v]
            best_cv = float('inf')
            best_v = old_v
            for v in cands:
                delta = v - old_v
                vals = corr + delta * B
                vals[mid] = corr[mid] + delta * (1.0 - v - old_v)
                cv = float(np.max(vals))
                if cv < best_cv:
                    best_cv = cv
                    best_v = v
            
            if best_v != old_v:
                h[j] = best_v
                delta = best_v - old_v
                corr += delta * B
                corr[mid] = np.sum(h * (1.0 - h))
        
        h = project_feasible(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
    
    return best_h, best_c5

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-8)[0]
    else:
        idx = np.where(h > 1e-8)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)

def lse_objective(x, alpha, target, penalty):

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
