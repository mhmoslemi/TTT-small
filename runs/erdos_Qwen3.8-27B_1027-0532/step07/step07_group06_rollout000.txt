import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    # Project h to [0,1]^n with sum = target_sum
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            h = np.clip(h + diff/n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def topk_obj(h, k, alpha):
    # Smooth top-k objective
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use log-sum-exp on the top k
    sorted_corr = np.sort(corr)[::-1]
    top_k = sorted_corr[:k]
    max_top = np.max(top_k)
    shifted = np.clip(alpha * (top_k - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lse_obj, h_cur, args=(alpha,),
                         method='SLSQP', bounds=bounds,
                         constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                      'jac': lambda x: np.ones(n)}],
                         options={'maxiter': maxiter, 'ftol': 1e-15})

            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _topk_lse_obj(x, k, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argpartition(-corr, k)[:k]
    top_vals = corr[idx]
    c_max = np.max(top_vals)
    shifted = np.clip(alpha * (top_vals - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_opt(h_start, n, target, alphas, maxiter, time_limit, start_time, use_topk=False, topk=5):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    if use_topk:
        func = _topk_lse_obj
    else:
        func = _lse_obj

    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            if use_topk:
                res = minimize(func, h_cur, args=(topk, param),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-14})
            else:
                res = minimize(func, h_cur, args=(param,),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new

    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)

def _compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                    else:

                    best_j = j

        h_fin3, sc_fin3 = slsqp_opt(best_h, n0, target0,
                                    [50000], 400,
                                    start_time + min(25, remaining() - 10), start_time)
        if sc_fin3 < best_score:
            best_score = sc_fin3
            best_h = h_fin3

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
