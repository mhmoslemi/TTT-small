import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, shift_k):
    n = len(h)
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - shift_k >= 0) & (idxs - shift_k < n)
    v2 = (idxs + shift_k >= 0) & (idxs + shift_k < n)
    if np.any(v1):
        g[v1] += (1.0 - h[idxs[v1] - shift_k])
    if np.any(v2):
        g[v2] -= h[idxs[v2] + shift_k]
    return g

def slp_optimize(h_init, target, time_limit, M=5, max_iters=3000, start_time=None):

    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    if start_time is None:
        start_time = time.time()
    
    no_improve = 0
    slp_M = 1  # start with M=1, increase gradually
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Adaptively increase M
        if it == 50:
            slp_M = 3
        elif it == 150:
            slp_M = 5
        elif it == 300:
            slp_M = 8
        elif it == 500:
            slp_M = min(15, 2 * n // 10)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        # Select top M shifts
        top_m_idx = np.argsort(corr)[-slp_M:]
        mid = n - 1
        
        # Build LP
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub_list = []
        b_ub_list = []
        
        for mi in top_m_idx:
            k = int(mi) - mid
            g = compute_gradient(h, k)
            rhs = float(np.dot(g, h)) - float(corr[mi])
            row = np.zeros(n + 1)
            row[:n] = g
            row[n] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(rhs)
        
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        
        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([target])

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
        except Exception:
            break
        
        if not res.success:
            no_improve += 5
            if no_improve > 250:
                break
            continue
        
        h_lp = res.x[:n]
        
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            h_try = h + alpha * (h_lp - h)
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break

            no_improve += 1
            # Try a smaller step or different M
            if no_improve > 50 and slp_M > 1:
                slp_M = max(1, slp_M // 2)
    
    return best_h, best_c5

def pair_cd(h_init, target, time_limit, n_cand=30, max_sweeps=50000, n_shifts=1, start_time=None):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    if start_time is None:
        start_time = time.time()
    
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 400:
                break
        
        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = compute_gradient(h, k_star)
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = 50.0
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                grad += wi * compute_gradient(h, k)
        
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        
        # Precompute profiles
        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        
        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                
                lo = 0.0
                hi_d = delta_max
                for _ in range(16):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:

                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    for m in range(len(weights)):
        if weights[m] < 1e-12:
            continue
        k = m - mid
        w = weights[m]
        grad += w * compute_gradient(x, k)
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, target, alphas, maxiter, time_limit, penalty=100.0, start_time=None):
    n = len(h_start)
    if start_time is None:
        start_time = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start_time > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
            except Exception:
                pass
    
    return best_h, best_c5
