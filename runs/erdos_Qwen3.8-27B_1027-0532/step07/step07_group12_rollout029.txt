import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def make_block_solution(n, frac=0.5, offset=0.0):
    h = np.zeros(n)
    n_on = int(n * frac)
    start = int(n * offset)
    h[start:start+n_on] = 1.0 / frac
    return project_feasible(h, n / 2.0)

    h = 0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0)
    return project_feasible(h, n / 2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):

    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        w = w / w.sum()
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= (2.0 / n) * np.sum(grad)
        improved = False
        decay = 1.0 / (1.0 + 0.003 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved and step > 3000:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 0: Quick L-BFGS refinement of initial
    if tleft() > 50:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[200, 2000],
                            maxiter=300, time_limit=min(35, tleft() * 0.15),
                            penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 1: Binary local search to find good discrete starting point
    if tleft() > 400:
        bl_time = min(60, tleft() * 0.08)
        h_bin, c5_bin = binary_local_search(n, bl_time, seed, max_steps=50000)
        h_bin_smooth = project_feasible(h_bin, target)
        hp, sc = lbfgs_phase(h_bin_smooth, n, target,
                            alphas=[500, 5000],
                            maxiter=200, time_limit=min(25, tleft() * 0.05),
                            penalty=200.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Low-resolution diverse exploration
    if tleft() > 350:
        n_low = 200
        target_low = n_low / 2.0
        rng = np.random.RandomState(seed + 12345)
        starts = []
        starts.append(interpolate_to(best_h, n_low, target_low))
        starts.append(make_block_solution(n_low, 0.5, 0.0))
        starts.append(make_block_solution(n_low, 0.65, 0.1))
        starts.append(make_block_solution(n_low, 0.4, 0.35))
        starts.append(make_wave_solution(n_low, 1.0, 0.35))
        starts.append(make_wave_solution(n_low, 2.0, 0.25))
        starts.append(make_wave_solution(n_low, 3.0, 0.2))
        for s in range(4):
            starts.append(random_solution(n_low, seed + 1000 + s * 77))
        for h_s in starts:
            if tleft() < 120:
                break
            hp, sc = lbfgs_phase(h_s, n_low, target_low,
                                alphas=[100, 500, 2000],
                                maxiter=150, time_limit=min(15, tleft() * 0.05),
                                penalty=100.0)
            if tleft() > 100:
                hcd, sc2 = pair_cd(hp, n_low, target_low, time_limit=min(12, tleft() * 0.04),
                                    n_cand=20, max_sweeps=30000, n_shifts=1)
                if sc2 < sc:
                    hp, sc = hcd, sc2
            h_full = interpolate_to(hp, n, target)
            sc_full = compute_c5(h_full)
            if sc_full < best_score:
                best_score = sc_full
                best_h = h_full

    # Phase 3: Main pair CD at full resolution
    if tleft() > 250:
        cd_time = min(280, tleft() * 0.3)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time,
                            n_cand=35, max_sweeps=300000, n_shifts=1, boundary_frac=0.15)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 4: L-BFGS high alpha refinement
    if tleft() > 80:
        hp, sc = lbfgs_phase(best_h, n, target,
                            alphas=[5000, 50000],
                            maxiter=300, time_limit=min(60, tleft() * 0.15),
                            penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 5: Pair CD multi-shift
    if tleft() > 130:
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(100, tleft() * 0.18),
                            n_cand=40, max_sweeps=150000, n_shifts=3, boundary_frac=0.1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 6: Try n=500 resolution
    if tleft() > 300:
        n_alt = 500
        target_alt = n_alt / 2.0
        h_interp = interpolate_to(best_h, n_alt, target_alt)
        hp, sc = lbfgs_phase(h_interp, n_alt, target_alt,
                            alphas=[200, 2000, 20000],
                            maxiter=250, time_limit=min(60, tleft() * 0.12),
                            penalty=200.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_alt
            target = target_alt
        if tleft() > 150:
            hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(80, tleft() * 0.2),
                                 n_cand=30, max_sweeps=100000, n_shifts=1)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

    # Phase 7: Try n=700 resolution
    if tleft() > 300:
        n_alt2 = 700
        target_alt2 = n_alt2 / 2.0
        h_interp2 = interpolate_to(best_h, n_alt2, target_alt2)
        hp2, sc2 = lbfgs_phase(h_interp2, n_alt2, target_alt2,
                               alphas=[200, 2000, 20000],
                               maxiter=250, time_limit=min(60, tleft() * 0.12),
                               penalty=200.0)
        if sc2 < best_score:
            best_score = sc2
            best_h = hp2
            n = n_alt2
            target = target_alt2
        if tleft() > 120:
            hcd2, c5cd2 = pair_cd(best_h, n, target, time_limit=min(70, tleft() * 0.18),
                                   n_cand=30, max_sweeps=80000, n_shifts=1)
            if c5cd2 < best_score:
                best_score = c5cd2
                best_h = hcd2

    # Phase 8: Subgradient with more shifts
    if tleft() > 60:
        hsg, c5sg = subgradient_phase(best_h, n, target, time_limit=min(40, tleft() * 0.1),
                                       max_steps=100000, n_shifts=7, base_lr=0.06)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 9: Random perturbation + re-optimize
    if tleft() > 120:
        rng2 = np.random.RandomState(seed + 42)
        n_use = len(best_h)
        target_use = n_use / 2.0
        for trial in range(4):
            if tleft() < 50:
                break
            noise_scale = 0.01 + 0.01 * trial
            h_pert = best_h + rng2.randn(n_use) * noise_scale
            h_pert = project_feasible(h_pert, target_use)
            hp3, sc3 = lbfgs_phase(h_pert, n_use, target_use,
                                   alphas=[2000, 20000],
                                   maxiter=150, time_limit=min(20, tleft() * 0.12),
                                   penalty=500.0)
            if sc3 < best_score and tleft() > 40:
                hcd3, sc4 = pair_cd(hp3, n_use, target_use, time_limit=min(25, tleft() * 0.15),
                                     n_cand=25, max_sweeps=40000, n_shifts=1)
                if sc4 < best_score:
                    best_score = sc4
                    best_h = hcd3
            elif sc3 < best_score:
                best_score = sc3
                best_h = hp3

    # Phase 10: Final aggressive L-BFGS
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp4, sc5 = lbfgs_phase(best_h, n_use, target_use,
                               alphas=[500000, 2000000],
                               maxiter=200, time_limit=min(35, tleft() * 0.4),
                               penalty=50000.0)
        if sc5 < best_score:
            best_score = sc5
            best_h = hp4

    # Phase 11: Final pair CD polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd4, c5cd4 = pair_cd(best_h, n_use, target_use, time_limit=min(25, tleft() * 0.6),
                               n_cand=30, max_sweeps=40000, n_shifts=1)
        if c5cd4 < best_score:
            best_score = c5cd4
            best_h = hcd4

    # Final feasibility and scoring
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
