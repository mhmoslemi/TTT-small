The correlation at shift s is: c(s) = sum_{i: i+s valid} h[i] * (1 - h[i+s])

If we change h[i] -> h[i] - δ and h[j] -> h[j] + δ, then:
c_new(s) = c_old(s) + δ * [contribution from j - contribution from i]

The "contribution from p" to c(s) is:
- If p appears as the first index (i=p): (1-h[p+s])
- If p appears as the second index (i+s=p, i=p-s): -h[p-s]

So the change in c(s) from changing h[p] by δ_p is:
δ_p * [(1-h[p+s]) * 1_{p+s valid} - h[p-s] * 1_{p-s valid}]

This is exactly δ_p * change[p, s_index] where s_index corresponds to shift s.

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14,
                time_limit=None, start_time=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def vectorized_cd(h_init, n, target, time_limit, max_sweeps=200000,
                  n_cand=60, n_deltas=10, recompute_every=50):

    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    
    # Geometric delta values
    delta_vals = np.array([0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0])
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > 300:
            break
        
        # Multi-shift gradient
        top_shifts = np.argsort(corr)[-4:]
        grad = np.zeros(n)
        for si in top_shifts:
            k = int(si) - mid
            w = 1.0 if si == top_shifts[-1] else 0.2
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        n_h = len(high_set)
        n_l = len(low_set)
        if n_h < 2 or n_l < 2:
            break
        
        # Vectorized pair evaluation
        ch_high = change[high_set]  # (n_h, n_corr)
        ch_low = change[low_set]  # (n_l, n_corr)
        g = ch_low[None, :, :] - ch_high[:, None, :]  # (n_h, n_l, n_corr)
        
        best_delta_c5 = c5_val
        best_i_local = -1
        best_j_local = -1
        best_delta_val = 0.0
        
        for dv in delta_vals:
            new_max = np.max(corr[None, None, :] + dv * g, axis=2)
            flat_idx = int(np.argmin(new_max))
            ii = flat_idx // n_l
            jj = flat_idx % n_l
            val = new_max[ii, jj] * dx
            if val < best_delta_c5:
                best_delta_c5 = val
                best_i_local = int(high_set[ii])
                best_j_local = int(low_set[jj])
                best_delta_val = dv
        
        if best_i_local >= 0 and best_delta_val > 0:
            h[best_i_local] -= best_delta_val
            h[best_j_local] += best_delta_val
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            
            # Update corr incrementally
            g_move = change[best_j_local] - change[best_i_local]
            corr = corr + best_delta_val * g_move
            
            # Update change matrix
            update_change_row(change, best_i_local, h, n, shifts)
            update_change_row(change, best_j_local, h, n, shifts)
            
            # Periodic recompute
            if sweep % recompute_every == 0:
                corr = np.correlate(h, 1.0 - h, mode='full').copy()
    
    return best_h, best_c5

    change = compute_change_matrix(h, n)

        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 800:
            break
        
        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress
        
        c5_val = float(np.max(corr)) * dx
        
        # Gradient
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue
        
        # Random delta in feasible range
        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-10:
            continue
        
        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx
        
        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True
        
        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            
            corr = corr + delta * g
            update_change_row(change, i, h, n, shifts)

                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1
    
    return best_h, best_c5

        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14,
                time_limit=None, start_time=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def vectorized_cd(h_init, n, target, time_limit, max_sweeps=200000,
                  n_cand=60, recompute_every=50, no_improve_limit=400):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    delta_vals = np.array([0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0, 2.0])

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > no_improve_limit:
            break

        top_shifts = np.argsort(corr)[-4:]
        grad = np.zeros(n)
        for si in top_shifts:
            k = int(si) - mid
            w = 1.0 if si == top_shifts[-1] else 0.2
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_h = len(high_set)
        n_l = len(low_set)
        if n_h < 2 or n_l < 2:
            break

        ch_high = change[high_set]
        ch_low = change[low_set]
        g = ch_low[None, :, :] - ch_high[:, None, :]

        best_delta_c5 = c5_val
        best_i_local = -1
        best_j_local = -1
        best_delta_val = 0.0

        for dv in delta_vals:
            new_max = np.max(corr[None, None, :] + dv * g, axis=2)
            flat_idx = int(np.argmin(new_max))
            ii = flat_idx // n_l
            jj = flat_idx % n_l
            val = new_max[ii, jj] * dx
            if val < best_delta_c5:
                best_delta_c5 = val
                best_i_local = int(high_set[ii])
                best_j_local = int(low_set[jj])
                best_delta_val = dv

        if best_i_local >= 0 and best_delta_val > 0:
            h[best_i_local] -= best_delta_val
            h[best_j_local] += best_delta_val
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

            g_move = change[best_j_local] - change[best_i_local]
            corr = corr + best_delta_val * g_move

            update_change_row(change, best_i_local, h, n, shifts)
            update_change_row(change, best_j_local, h, n, shifts)

            if sweep % recompute_every == 0:
                corr = np.correlate(h, 1.0 - h, mode='full').copy()

    return best_h, best_c5

def sa_cd(h_init, n, target, time_limit, max_steps=50000,
          t_start=0.03, t_end=0.0002, n_cand=50):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(123)

    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 800:
            break

        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        c5_val = float(np.max(corr)) * dx

        top_shifts = np.argsort(corr)[-3:]
        grad = np.zeros(n)
        for si in top_shifts:
            k = int(si) - mid
            w = 1.0 if si == top_shifts[-1] else 0.3
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-10:
            continue

        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

            corr = corr + delta * g
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

            if step % 100 == 0:
                corr = np.correlate(h, 1.0 - h, mode='full').copy()

            cur_c5 = float(np.max(corr)) * dx
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_cd_refine(h_init, n, target, time_limit, n_phases=3):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    change = compute_change_matrix(h, n)
    consecutive_no_improve = 0

    cand_sizes = [40, 60, 80]
    delta_sets = [
        np.array([0.001, 0.005, 0.02, 0.08, 0.3, 1.0]),
        np.array([0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0]),
        np.array([0.0005, 0.002, 0.007, 0.02, 0.07, 0.2, 0.7]),
    ]

    for phase in range(n_phases):
        phase_time = (time_limit - (time.time() - start)) / (n_phases - phase)
        if phase_time < 10:
            break
        n_cand = min(cand_sizes[phase % len(cand_sizes)], n // 3)
        delta_vals = delta_sets[phase % len(delta_sets)]

        sweep_count = 0
        while time.time() - start < time_limit - 5:
            sweep_count += 1
            if consecutive_no_improve > 300:
                break

            c5_val = float(np.max(corr)) * dx
            if c5_val < best_c5 - 1e-13:
                best_c5 = c5_val
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1

            top_shifts = np.argsort(corr)[-5:]
            grad = np.zeros(n)
            for si in top_shifts:
                k = int(si) - mid
                w = 1.0 if si == top_shifts[-1] else 0.15
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
            grad -= np.mean(grad)

            sorted_idx = np.argsort(grad)
            low_set = sorted_idx[:n_cand]
            high_set = sorted_idx[-n_cand:]

            n_h = len(high_set)
            n_l = len(low_set)
            if n_h < 2 or n_l < 2:
                break

            ch_high = change[high_set]
            ch_low = change[low_set]
            g_mat = ch_low[None, :, :] - ch_high[:, None, :]

            best_d_c5 = c5_val
            best_i = -1
            best_j = -1
            best_d = 0.0

            for dv in delta_vals:
                new_max = np.max(corr[None, None, :] + dv * g_mat, axis=2)
                fi = int(np.argmin(new_max))
                ii = fi // n_l
                jj = fi % n_l
                val = new_max[ii, jj] * dx
                if val < best_d_c5:
                    best_d_c5 = val
                    best_i = int(high_set[ii])
                    best_j = int(low_set[jj])
                    best_d = dv

            if best_i >= 0 and best_d > 0:
                h[best_i] -= best_d
                h[best_j] += best_d
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                g_move = change[best_j] - change[best_i]
                corr = corr + best_d * g_move
                update_change_row(change, best_i, h, n, shifts)
                update_change_row(change, best_j, h, n, shifts)
                if sweep_count % 40 == 0:
                    corr = np.correlate(h, 1.0 - h, mode='full').copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 50:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 300, 1000],
                                400, 1e-14,
                                time_limit=min(35, tleft() * 0.5), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 130:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        h_cd, c5_cd = multi_cd_refine(h1, n1, target1,
                                       time_limit=min(110, tleft() * 0.55),
                                       n_phases=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 160:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_cd2, c5_cd2 = multi_cd_refine(h2, n2, target2,
                                         time_limit=min(140, tleft() * 0.55),
                                         n_phases=3)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 70:
        h_sa, c5_sa = sa_cd(best_h, len(best_h), len(best_h) / 2.0,
                            time_limit=min(55, tleft() * 0.5),
                            max_steps=50000, t_start=0.03, t_end=0.0002,
                            n_cand=50)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    if tleft() > 80:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_cd5, c5_cd5 = vectorized_cd(best_h, n_cur, t_cur,
                                       time_limit=min(65, tleft() * 0.5),
                                       max_sweeps=150000, n_cand=60,
                                       recompute_every=40, no_improve_limit=350)
        if c5_cd5 < best_score:
            best_score = c5_cd5
            best_h = h_cd5

    if tleft() > 55:
        h_pol6, sc6 = slsqp_round(best_h, len(best_h), len(best_h) / 2.0,
                                   [3000, 10000, 30000],
                                   600, 1e-14,
                                   time_limit=min(45, tleft() * 0.5), start_time=start_time)
        if sc6 < best_score:
            best_score = sc6
            best_h = h_pol6

    if tleft() > 50:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_cd7, c5_cd7 = vectorized_cd(best_h, n_cur, t_cur,
                                       time_limit=min(40, tleft() * 0.5),
                                       max_sweeps=100000, n_cand=55,
                                       recompute_every=35, no_improve_limit=300)
        if c5_cd7 < best_score:
            best_score = c5_cd7
            best_h = h_cd7

    if tleft() > 40:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_sa8, c5_sa8 = sa_cd(best_h, n_cur, t_cur,
                              time_limit=min(30, tleft() * 0.5),
                              max_steps=40000, t_start=0.02, t_end=0.0001,
                              n_cand=40)
        if c5_sa8 < best_score:
            best_score = c5_sa8
            best_h = h_sa8

    if tleft() > 35:
        h_fp, sc_fp = slsqp_round(best_h, len(best_h), len(best_h) / 2.0,
                                   [10000, 50000, 100000],
                                   500, 1e-14,
                                   time_limit=min(25, tleft() * 0.5), start_time=start_time)
        if sc_fp < best_score:
            best_score = sc_fp
            best_h = h_fp

    if tleft() > 20:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_cf, c5_cf = vectorized_cd(best_h, n_cur, t_cur,
                                     time_limit=min(15, tleft() * 0.5),
                                     max_sweeps=50000, n_cand=40,
                                     recompute_every=25, no_improve_limit=200)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
