from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

            try:

def pair_cd(h_init, n, target, time_limit, n_cand=50, max_sweeps=100000, n_shifts=1):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = 50.0
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]
        
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        
        # Build profiles
        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        
        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        # Search over pairs
        # Limit pairs to avoid O(n_cand^2) being too slow
        # Use a subset of high x low pairs
        hi_list = [i for i in high_set if h[int(i)] > 1e-10]
        lo_list = [j for j in low_set if h[int(j)] < 1.0 - 1e-10]
        
        if len(hi_list) * len(lo_list) > 2500:
            # Subsample
            hi_list = hi_list[::max(1, len(hi_list)//50)][:50]
            lo_list = lo_list[::max(1, len(lo_list)//50)][:50]
        
        for i in hi_list:
            if int(i) not in cand_pos:
                continue
            hi_val = float(h[int(i)])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in lo_list:
                if int(i) == int(j):
                    continue
                if int(j) not in cand_pos:
                    continue
                hj_val = float(h[int(j)])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                
                # Binary search for optimal delta
                lo = 0.0
                hi_d = delta_max
                for _ in range(20):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        
        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def subgradient_step(h, n, target, n_shifts=5, lr=0.01):
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    
    top_idx = np.argsort(corr)[-n_shifts:]
    alpha_w = 30.0
    w = np.exp(alpha_w * (corr[top_idx] - cur_max))
    w = w / w.sum()
    
    grad = np.zeros(n)
    for wi, si in zip(w, top_idx):
        k = int(si) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= wi * h[idxs[v2] + k]
    
    grad *= dx
    # Center the gradient to maintain sum constraint
    grad -= np.mean(grad)
    
    return h - lr * grad, grad

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_optimize(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 30})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-9:
                    break
            except Exception:
                pass
    
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, n_cand=40, max_sweeps=100000, n_shifts=1, biniters=20):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 600:
                break
        
        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = 50.0
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]
        
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        
        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        
        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        hi_list = [i for i in high_set if h[int(i)] > 1e-10]
        lo_list = [j for j in low_set if h[int(j)] < 1.0 - 1e-10]
        
        # Limit pairs for speed
        max_pairs = 4000
        if len(hi_list) * len(lo_list) > max_pairs:
            step_h = max(1, len(hi_list) // 40)
            step_l = max(1, len(lo_list) // 40)
            hi_list = hi_list[::step_h][:40]
            lo_list = lo_list[::step_l][:40]
        
        for i in hi_list:
            if int(i) not in cand_pos:
                continue
            hi_val = float(h[int(i)])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in lo_list:
                if int(i) == int(j):
                    continue
                if int(j) not in cand_pos:
                    continue
                delta_max = min(hi_val, 1.0 - float(h[int(j)]))
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                
                lo = 0.0
                hi_d = delta_max
                for _ in range(biniters):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        
        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def subgradient_phase(h_init, n, target, time_limit, max_steps=50000, n_shifts=5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        
        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        
        grad *= dx
        grad -= np.mean(grad)
        
        improved = False
        decay = 1.0 / (1.0 + 0.003 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = 0.05 * lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        
        if not improved and step > 5000:
            break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-14
    if np.any(active_mask):

        for m in active_idx:

            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 30})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-9:
                    break
            except Exception:
                pass
    return best_h, best_c5

            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        hi_list = [i for i in high_set if h[int(i)] > 1e-10]
        lo_list = [j for j in low_set if h[int(j)] < 1.0 - 1e-10]
        max_pairs = 5000
        if len(hi_list) * len(lo_list) > max_pairs:
            step_h = max(1, len(hi_list) // 50)
            step_l = max(1, len(lo_list) // 50)
            hi_list = hi_list[::step_h][:50]
            lo_list = lo_list[::step_l][:50]
        for i in hi_list:
            if int(i) not in cand_pos:
                continue
            hi_val = float(h[int(i)])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in lo_list:
                if int(i) == int(j):
                    continue
                if int(j) not in cand_pos:
                    continue
                delta_max = min(hi_val, 1.0 - float(h[int(j)]))
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(biniters):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def subgradient_phase(h_init, n, target, time_limit, max_steps=50000, n_shifts=5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        decay = 1.0 / (1.0 + 0.003 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = 0.05 * lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved and step > 5000:
            break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: Quick L-BFGS warmup
    if tleft() > 80:
        hp, sc = lbfgs_optimize(best_h, n, target,
                                alphas=[50, 200, 1000],
                                maxiter=200, time_limit=min(40, tleft()*0.15),
                                penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Multi-resolution exploration at lower n
    if tleft() > 250:
        for n_res in [120, 200, 350]:
            if tleft() < 70:
                break
            target_res = n_res / 2.0
            h_res = interpolate_to(best_h, n_res, target_res)
            hp, sc = lbfgs_optimize(h_res, n_res, target_res,
                                    alphas=[100, 1000, 10000, 100000],
                                    maxiter=200, time_limit=min(35, tleft()*0.1),
                                    penalty=100.0)
            h_back = interpolate_to(hp, n, target)
            sc_back = compute_c5(h_back)
            if sc_back < best_score:
                hp2, sc2 = lbfgs_optimize(h_back, n, target,
                                          alphas=[5000, 50000, 500000],
                                          maxiter=150, time_limit=min(20, tleft()*0.06),
                                          penalty=500.0)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = hp2
                hcd2, c5cd2 = pair_cd(best_h, n, target, time_limit=min(15, tleft()*0.05),
                                       n_cand=30, max_sweeps=20000, n_shifts=1, biniters=20)
                if c5cd2 < best_score:
                    best_score = c5cd2
                    best_h = hcd2

    # Phase 3: Main pair CD - long
    if tleft() > 200:
        cd_time = min(280, tleft() * 0.3)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time, n_cand=40, max_sweeps=200000, n_shifts=1, biniters=22)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 4: High-alpha L-BFGS
    if tleft() > 100:
        hp, sc = lbfgs_optimize(best_h, n, target,
                                alphas=[5000, 50000, 500000, 5000000],
                                maxiter=300, time_limit=min(70, tleft()*0.25),
                                penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 5: Pair CD multi-shift
    if tleft() > 150:
        cd_time = min(150, tleft() * 0.25)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time, n_cand=50, max_sweeps=150000, n_shifts=3, biniters=22)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 6: Very high alpha L-BFGS
    if tleft() > 80:
        hp, sc = lbfgs_optimize(best_h, n, target,
                                alphas=[5000000, 20000000, 100000000],
                                maxiter=250, time_limit=min(50, tleft()*0.4),
                                penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 7: Subgradient
    if tleft() > 50:
        hsg, c5sg = subgradient_phase(best_h, n, target, time_limit=min(35, tleft()*0.4),
                                       max_steps=30000, n_shifts=7)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 8: Higher resolution attempt
    if tleft() > 200:
        n_new = 800
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = lbfgs_optimize(h_interp, n_new, target_new,
                                alphas=[100, 1000, 10000, 100000, 1000000],
                                maxiter=200, time_limit=min(70, tleft()*0.3),
                                penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new
        if tleft() > 80:
            hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(60, tleft()*0.3),
                                 n_cand=30, max_sweeps=80000, n_shifts=1, biniters=20)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

    # Phase 9: Perturbation restart
    if tleft() > 70:
        rng = np.random.RandomState(seed)
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert = best_h + rng.randn(n_use) * 0.008
        h_pert = project_feasible(h_pert, target_use)
        hp, sc = lbfgs_optimize(h_pert, n_use, target_use,
                                alphas=[5000, 50000, 500000],
                                maxiter=150, time_limit=min(20, tleft()*0.3),
                                penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
            if tleft() > 30:
                hcd, c5cd = pair_cd(best_h, n_use, target_use, time_limit=min(20, tleft()*0.4),
                                     n_cand=25, max_sweeps=20000, n_shifts=1, biniters=20)
                if c5cd < best_score:
                    best_score = c5cd
                    best_h = hcd

    # Phase 10: Final L-BFGS squeeze
    if tleft() > 35:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_optimize(best_h, n_use, target_use,
                                alphas=[200000000, 1000000000],
                                maxiter=150, time_limit=min(25, tleft()*0.6),
                                penalty=10000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 11: Final pair CD
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd(best_h, n_use, target_use, time_limit=min(15, tleft()*0.7),
                             n_cand=25, max_sweeps=20000, n_shifts=1, biniters=20)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
