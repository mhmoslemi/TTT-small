                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]

        grad -= np.mean(grad)
        # Track moving averages of grad sign
        mom_inv = (1 - momentum) * mom_inv + momentum * np.sign(grad)
        grad_eff = grad + 0.5 * mom_inv

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)

        sorted_idx = np.argsort(grad_eff)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid < 1e-15:
                        break
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-11:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            no_improve += 3

    return best_h, best_c5

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        h = np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0

    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)

    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    no_improve = 0
    rng = np.random.RandomState(31337)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:
            break

        top_idx = np.argsort(corr)[-n_active:]
        top_sum = np.sum(corr[top_idx])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s in top_idx:
            w = corr[int(s)] / top_sum
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_trial = h - lr * grad
        h_trial = project_feasible(h_trial, target)
        h = h_trial
        lr *= lr_decay

        if no_improve > 1500 and no_improve % 200 == 0:
            h = best_h + rng.randn(n) * 0.005
            h = project_feasible(h, target)
            lr = lr_init * 0.3
            no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 120:
        h_pol, sc = slsqp_opt(best_h, current_n, current_target,
                              [30, 80, 200, 500, 1500, 5000, 15000, 50000, 150000, 500000],
                              1000)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 300:
        h_cd, c5_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(280, tleft() * 0.55),
                              max_sweeps=50000,
                              n_cand_start=20, n_cand_end=90,
                              top_m=1, momentum=0.3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        h_pol, sc = slsqp_opt(best_h, current_n, current_target,
                              [10000, 50000, 200000, 800000, 2000000],
                              1500)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        h_ms, c5_ms = multi_cd(best_h, current_n, current_target,
                               time_limit=min(90, tleft() * 0.6),
                               max_sweeps=15000, n_cand=50, n_shifts=3,
                               momentum=0.4)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if tleft() > 350:
        for n_try in [480, 520]:
            if tleft() < 200:
                break
            target_try = n_try / 2.0
            h_try = interpolate_to(best_h, n_try, target_try)
            if tleft() > 180:
                h_cd_t, c5_t = pair_cd(h_try, n_try, target_try,
                                       time_limit=min(150, tleft() * 0.5),
                                       max_sweeps=25000,
                                       n_cand_start=15, n_cand_end=65,
                                       top_m=1, momentum=0.3)
                if c5_t < best_score:
                    best_score = c5_t
                    best_h = h_cd_t
                    current_n = n_try
                    current_target = target_try
            if current_n == n_try and tleft() > 80:
                h_pol_t, sc_t = slsqp_opt(best_h, n_try, target_try,
                                          [5000, 20000, 100000, 500000],
                                          1000)
                if sc_t < best_score:
                    best_score = sc_t
                    best_h = h_pol_t

    if tleft() > 250:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.1, 0.2]:
            if tleft() < 100:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = slsqp_opt(h_pert, current_n, current_target,
                                      [100, 500, 2000, 10000, 50000],
                                      600)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 120:
                h_cd_p, c5_p = pair_cd(h_ref, current_n, current_target,
                                       time_limit=min(80, tleft() * 0.35),
                                       max_sweeps=12000,
                                       n_cand_start=25, n_cand_end=70,
                                       top_m=2, momentum=0.35)
                if c5_p < best_score:
                    best_score = c5_p
                    best_h = h_cd_p

    if tleft() > 100:
        h_sgd, c5_sgd = subgradient_descent(best_h, current_n, current_target,
                                            time_limit=min(80, tleft() * 0.6),
                                            lr_init=0.003, lr_decay=0.9998,
                                            max_steps=30000, n_active=3)
        if c5_sgd < best_score:
            best_score = c5_sgd
            best_h = h_sgd

    if tleft() > 80:
        h_final_cd, c5_final = pair_cd(best_h, current_n, current_target,
                                       time_limit=min(70, tleft() * 0.7),
                                       max_sweeps=20000,
                                       n_cand_start=30, n_cand_end=100,
                                       top_m=1, momentum=0.25)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if tleft() > 50:
        h_final_pol, sc_final = slsqp_opt(best_h, current_n, current_target,
                                          [50000, 200000, 800000, 3000000, 10000000],
                                          1200)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
