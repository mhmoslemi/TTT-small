        bounds = [(-h_cur[p], 1.0 - h_cur[p]) for p in range(n)]
        bounds.append((None, None))
        
        try:

        if res.status != 0:
            break
        
        delta = res.x[:n]
        
        # Line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:

def pair_cd_fast(h_init, n, target, time_limit, max_sweeps=20000,
                 n_cand=30, max_no_improve=200, weight_beta=5.0):
    """Fast weighted pair coordinate descent"""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))
        
        # Compute weighted gradient
        grad = np.zeros(n)
        top_k_shifts = min(20, n_corr)
        top_idx = np.argsort(weights)[-top_k_shifts:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-6:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_move = None
        
        for i in high_set:
            for j in low_set:
                i, j = int(i), int(j)
                if i == j:
                    continue
                delta_lo = max(-h[j], h[i] - 1.0)
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                a_d, b_d = delta_lo, delta_hi
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active_mask]) if active_mask.any() else 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_move = (i, j, delta_star)
                    break
            if best_move is not None:
                break
        
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [i, j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

def periodic_try(n, target, n_periods=20):
    """Try periodic constructions and return the best one"""
    best_h = None
    best_score = float('inf')
    
    for period in range(2, n_periods + 2):
        if n % period != 0:
            continue
        n_rep = n // period
        # We need to find values v[0..period-1] such that:
        # sum(v) = target / n_rep
        # and we minimize max_k sum_i v[i]*(1-v[(i+k)%period]) * (2/period)
        
        # Initialize with a continuous function and optimize
        v0 = np.ones(period) * (target / (n_rep * period))
        
        def periodic_obj(v):
            v = np.clip(v, 0, 1)
            # Compute C5 for period-periodic function
            max_corr = 0
            for k in range(period):
                corr_k = np.sum(v * (1 - np.roll(v, -k)))
                max_corr = max(max_corr, corr_k)
            # Scale: actual C5 = (n_rep * max_corr) * (2/n) = max_corr * 2/period
            return max_corr * 2.0 / period
        
        from scipy.optimize import minimize as sc_minimize
        # Try a few initializations
        for init_type in range(3):
            if init_type == 0:
                v_init = np.ones(period) * (target / (n_rep * period))
            elif init_type == 1:
                v_init = np.where(np.arange(period) % 2 == 0, 0.8, 0.2)
                v_init = v_init * (target / (n_rep * period)) / v_init.sum()
            else:
                v_init = np.cos(np.arange(period) * 2 * np.pi / period) * 0.3 + 0.5
                v_init = np.clip(v_init, 0.01, 0.99)
                v_init = v_init / v_init.sum() * (target / (n_rep * period))
            
            try:
                res = sc_minimize(periodic_obj, v_init, bounds=[(0,1)]*period,
                                method='Nelder-Mead', options={'maxiter': 10000, 'xatol': 1e-8})
                v_opt = np.clip(res.x, 0, 1)
                # Construct full h
                h_p = np.tile(v_opt, n_rep)
                h_p = project_feasible(h_p, target)
                sc = compute_c5(h_p)
                if sc < best_score:
                    best_score = sc
                    best_h = h_p.copy()
            except Exception:
                pass
    
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_m = (p + shifts >= 0) & (p + shifts < n)
        neg_m = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_m] += (1.0 - h[p + shifts[pos_m]])
        change[p, neg_m] -= h[p - shifts[neg_m]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))

        grad = np.zeros(n)
        top_k_s = min(25, n_corr)
        top_idx = np.argsort(weights)[-top_k_s:]
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        cur_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        cur_ncand = min(cur_ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:cur_ncand]
        high_set = sorted_idx[-cur_ncand:]

        best_trial = c5_val
        best_move = None

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            found = False
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(-h[j], h[i] - 1.0)
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d, b_d = d_lo, d_hi
                for _ in range(14):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g
                    f_mid = np.max(vals)
                    am = vals >= f_mid - 1e-14
                    fp = np.max(g[am]) if am.any() else 0.0
                    if fp > 1e-12:
                        b_d = m_d
                    else:
                        a_d = m_d
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) < 1e-12:
                    continue
                approx = np.max(corr + d_star * g) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_move = (i, j, d_star)
                    found = True
                    break
            if found:
                break

        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [i, j]:
                change[p, :] = 0.0
                pos_m = (p + shifts >= 0) & (p + shifts < n)
                neg_m = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_m] += (1.0 - h[p + shifts[pos_m]])
                change[p, neg_m] -= h[p - shifts[neg_m]]

    return best_h, best_c5


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj_p, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def periodic_optimize(n, target, period):
    if n % period != 0:
        return None, None
    m = n // period
    target_v = target / m

    def period_obj(v):
        v = np.clip(v, 0.0, 1.0)
        max_corr = 0.0
        for r in range(period):
            v_shifted = np.roll(v, r)
            corr_r = np.sum(v * (1.0 - v_shifted))
            if corr_r > max_corr:
                max_corr = corr_r
        return max_corr * 2.0 / period

    best_v = None
    best_sc = float('inf')

    rng = np.random.RandomState(777)
    for trial in range(8):
        if trial == 0:
            v0 = np.full(period, target_v / period)
        elif trial == 1:
            v0 = np.where(np.arange(period) % 2 == 0, 0.7, 0.3)
            v0 = v0 * (target_v / v0.sum())
        elif trial == 2:
            v0 = 0.5 + 0.3 * np.cos(2.0 * np.pi * np.arange(period) / period)
            v0 = np.clip(v0, 0.05, 0.95)
            v0 = v0 * (target_v / v0.sum())
        else:
            v0 = np.ones(period) * (target_v / period)
            v0 += rng.randn(period) * 0.1
            v0 = np.clip(v0, 0.01, 0.99)
            v0 = v0 * (target_v / v0.sum())

        try:
            res = minimize(period_obj, v0, method='Nelder-Mead',
                           options={'maxiter': 20000, 'xatol': 1e-10, 'fatol': 1e-12})
            v_opt = np.clip(res.x, 0.0, 1.0)
            sc = period_obj(v_opt)
            # Verify with full computation
            h_full = np.tile(v_opt, m)
            h_full = project_feasible(h_full, target)
            sc_full = compute_c5(h_full)
            if sc_full < best_sc:
                best_sc = sc_full
                best_v = h_full.copy()
        except Exception:
            pass

    return best_v, best_sc


def nesterov_subgrad(h_init, n, target, time_limit, n_iter=5000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_curr = 1.0
    lr_base = 0.02

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_y = float(np.max(corr)) * dx

        top_k = 10
        top_idx = np.argsort(corr)[-top_k:]
        c_max = corr[top_idx[0]]

        grad = np.zeros(n)
        for s_idx in top_idx:
            w = np.exp(20.0 * (corr[s_idx] - c_max) / max(c_max, 1e-10))
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - y[idxs[v1] - k])
            grad[v2] -= w * y[idxs[v2] + k]
        grad_sum = grad.sum()
        grad -= grad_sum / n

        t_curr = it + 1
        beta = (1.0 - 1.0 / (t_curr + 1)) / 2.0
        lr = lr_base / (t_curr ** 0.5)

        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)

        c5_x = compute_c5(h_new)
        if c5_x < best_c5:
            best_c5 = c5_x
            best_h = h_new.copy()

        y = h_new + beta * (h_new - h)
        y = project_feasible(y, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 18.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Pair CD initial polish
    if tleft() > 120:
        h_cd, sc_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(70, tleft() * 0.4),
                              max_sweeps=25000, n_cand_start=20, n_cand_end=50,
                              max_no_improve=250, weight_beta=5.0)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 2: LP active set
    if tleft() > 150:
        h_lp, sc_lp = lp_active_set(best_h, n0, target0,
                                     time_limit=min(80, tleft() * 0.35),
                                     top_k_start=15, max_iters=300)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    # Phase 3: Pair CD after LP
    if tleft() > 120:
        h_cd2, sc_cd2 = pair_cd(best_h, n0, target0,
                                 time_limit=min(60, tleft() * 0.35),
                                 max_sweeps=20000, n_cand_start=25, n_cand_end=55,
                                 max_no_improve=200, weight_beta=6.0)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 4: LP with more active shifts
    if tleft() > 120:
        h_lp2, sc_lp2 = lp_active_set(best_h, n0, target0,
                                       time_limit=min(70, tleft() * 0.35),
                                       top_k_start=25, max_iters=250)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_h = h_lp2

    # Phase 5: Nesterov subgradient
    if tleft() > 100:
        h_ns, sc_ns = nesterov_subgrad(best_h, n0, target0,
                                        time_limit=min(50, tleft() * 0.3),
                                        n_iter=3000)
        if sc_ns < best_score:
            best_score = sc_ns
            best_h = h_ns

    # Phase 6: SLSQP polish
    if tleft() > 80:
        h_pol, sc_pol = slsqp_round(best_h, n0, target0,
                                     [100, 500, 2000, 8000], 400, 1e-14)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Phase 7: Pair CD with higher beta
    if tleft() > 80:
        h_cd3, sc_cd3 = pair_cd(best_h, n0, target0,
                                 time_limit=min(50, tleft() * 0.4),
                                 max_sweeps=20000, n_cand_start=30, n_cand_end=60,
                                 max_no_improve=180, weight_beta=8.0)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    # Phase 8: LP again
    if tleft() > 60:
        h_lp3, sc_lp3 = lp_active_set(best_h, n0, target0,
                                       time_limit=min(40, tleft() * 0.4),
                                       top_k_start=20, max_iters=200)
        if sc_lp3 < best_score:
            best_score = sc_lp3
            best_h = h_lp3

    # Phase 9: Periodic multi-start
    if tleft() > 150:
        for period in [6, 8, 10, 12, 14, 16]:
            if tleft() < 100:
                break
            h_per, sc_per = periodic_optimize(n0, target0, period)
            if h_per is not None:
                # Refine the periodic solution
                h_ref, sc_ref = pair_cd(h_per, n0, target0,
                                         time_limit=min(30, tleft() * 0.15),
                                         max_sweeps=8000, n_cand_start=20,
                                         n_cand_end=40, max_no_improve=100,
                                         weight_beta=5.0)
                h_lp_per, sc_lp_per = lp_active_set(h_ref, n0, target0,
                                                     time_limit=min(20, tleft() * 0.1),
                                                     top_k_start=12, max_iters=80)
                if sc_lp_per < best_score:
                    best_score = sc_lp_per
                    best_h = h_lp_per

    # Phase 10: Final alternating rounds
    for rnd in range(4):
        if tleft() < 50:
            break
        h_a, sc_a = pair_cd(best_h, n0, target0,
                             time_limit=min(30, tleft() * 0.25),
                             max_sweeps=12000, n_cand_start=30, n_cand_end=55,
                             max_no_improve=120, weight_beta=7.0)
        if sc_a < best_score:
            best_score = sc_a
            best_h = h_a
        if tleft() < 30:
            break
        h_b, sc_b = lp_active_set(best_h, n0, target0,
                                   time_limit=min(20, tleft() * 0.2),
                                   top_k_start=15, max_iters=100)
        if sc_b < best_score:
            best_score = sc_b
            best_h = h_b

    # Phase 11: Final SLSQP
    if tleft() > 30:
        h_fin, sc_fin = slsqp_round(best_h, n0, target0,
                                     [5000, 20000, 50000], 400, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 12: Final pair CD
    if tleft() > 20:
        h_fin2, sc_fin2 = pair_cd(best_h, n0, target0,
                                   time_limit=min(15, tleft() * 0.7),
                                   max_sweeps=8000, n_cand_start=40, n_cand_end=60,
                                   max_no_improve=80, weight_beta=9.0)
        if sc_fin2 < best_score:
            best_score = sc_fin2
            best_h = h_fin2

    final_h = project_feasible(best_h, n0 / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, n0
