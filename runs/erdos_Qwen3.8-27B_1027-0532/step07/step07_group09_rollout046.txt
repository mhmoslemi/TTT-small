from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_row(p, h, n):
    """Compute the change vector for position p."""
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros(n_corr)
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[neg_mask] -= h[p - shifts[neg_mask]]
    return change

def zero_one_local_search(h_init, n, target, time_limit, max_iter=50000):
    """Local search on the 0-1 hypercube."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    # Threshold to 0-1
    h = (h_init > 0.5).astype(float)
    # Fix the sum
    n_ones = int(np.sum(h))
    target_ones = int(target)
    if n_ones > target_ones:
        # Need to flip some 1s to 0s
        ones_idx = np.where(h == 1)[0]
        rng = np.random.RandomState(123)
        to_flip = rng.choice(ones_idx, n_ones - target_ones, replace=False)
        h[to_flip] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(h == 0)[0]
        rng = np.random.RandomState(123)
        to_flip = rng.choice(zeros_idx, target_ones - n_ones, replace=False)
        h[to_flip] = 1
    
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    
    # Precompute change matrix
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        change[p, :] = compute_change_row(p, h, n)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        score = c_max * dx
        
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find positions to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Use active constraints to guide search
        n_active = min(5, n_corr)
        tol = max(1e-10, 0.01 * c_max)
        active = np.where(corr >= c_max - tol)[0]
        
        # Compute gradient-like scores
        # For each position p, how much would increasing/decreasing it help?
        # If we decrease h[p] (flip 1→0), Δcorr = -change[p]
        # If we increase h[p] (flip 0→1), Δcorr = +change[p]
        
        # Score for "wanting to decrease": sum over active lags of -change[p, active]
        # (we want this to be positive, meaning decreasing helps)
        dec_score = np.zeros(n)
        inc_score = np.zeros(n)
        for a in active:
            dec_score -= change[:, a]
            inc_score += change[:, a]
        
        # Among ones, pick the one with highest dec_score (most beneficial to flip to 0)
        # Among zeros, pick the one with highest inc_score (most beneficial to flip to 1)
        # But we need dec_score[i] and inc_score[j] to be compatible
        
        # Try top candidates
        top_k = min(20, len(ones))
        top_l = min(20, len(zeros))
        
        best_dec_ones = ones[np.argsort(dec_score[ones])[-top_k:]]
        best_inc_zeros = zeros[np.argsort(inc_score[zeros])[-top_l:]]
        
        best_improvement = 0
        best_i = -1
        best_j = -1
        
        for i in best_dec_ones:
            for j in best_inc_zeros:
                delta = change[j] - change[i]
                new_max = np.max(corr + delta)
                improvement = c_max - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i = i
                    best_j = j
        
        if best_i >= 0 and best_improvement > 1e-12:
            h[best_i] = 0
            h[best_j] = 1
            change[best_i, :] = compute_change_row(best_i, h, n)
            change[best_j, :] = compute_change_row(best_j, h, n)
    
    return best_h, best_score

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_polish(h_start, n, target, alphas, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            return c_max + np.log(np.sum(np.exp(shifted))) / alpha
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def to_zero_one(h, n, rng):
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target_ones = n // 2
    if n_ones > target_ones:
        ones = np.where(h == 1)[0]
        to_flip = rng.choice(ones, n_ones - target_ones, replace=False)
        h[to_flip] = 0
    elif n_ones < target_ones:
        zeros = np.where(h == 0)[0]
        to_flip = rng.choice(zeros, target_ones - n_ones, replace=False)
        h[to_flip] = 1
    return h

def compute_change_row(p, h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros(n_corr)
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[neg_mask] -= h[p - shifts[neg_mask]]
    return change

def zero_one_sa(h_init, n, time_limit, rng, max_iter=100000,
                t_start=0.5, t_end=0.01, patience=300):
    """Simulated annealing on 0-1 configurations."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    target_ones = n // 2
    
    h = to_zero_one(h_init, n, rng)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    
    change = np.zeros((n, n_corr))
    for p in range(n):
        change[p, :] = compute_change_row(p, h, n)
    
    no_improve = 0
    T = t_start
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > patience:
            break
        
        progress = it / max_iter
        T = t_start + (t_end - t_start) * progress
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        delta = change[j] - change[i]
        new_corr = corr + delta
        new_c_max = float(np.max(new_corr))
        
        delta_energy = new_c_max - c_max
        if delta_energy <= 0 or rng.random() < np.exp(-delta_energy / T):
            # Accept
            h[i] = 0
            h[j] = 1
            corr = new_corr
            c_max = new_c_max
            change[i, :] = compute_change_row(i, h, n)
            change[j, :] = compute_change_row(j, h, n)
            
            new_score = c_max * dx
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_score

        top_zeros = zeros[np.argsort(pressure_inc[zeros])[-k:]]

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_polish(h_start, n, target, alphas, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        def _obj(x, _alpha=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(_alpha * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / _alpha)
        try:
            res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(len(x))

_g_target = 0.0

    n_corr = 2 * n - 1
