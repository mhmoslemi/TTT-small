    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _obj_func(x):
    return _smooth_obj_impl(x, _g_alpha, _g_n)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change


def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]


def bisection_step(corr, g, max_delta, n_iter=12):
    a = 0.0
    b = max_delta
    for _ in range(n_iter):
        mid_d = 0.5 * (a + b)
        vals = corr + mid_d * g
        f_mid = np.max(vals)
        active = vals >= f_mid - 1e-14
        if active.any():
            f_prime = np.max(g[active])
        else:
            f_prime = 0.0
        if f_prime > 1e-12:
            b = mid_d
        else:
            a = mid_d
    return 0.5 * (a + b)


def enhanced_cd_fast(h_init, n, target, time_limit, max_sweeps=20000,
                     n_active=3, n_cand_start=15, n_cand_end=60,
                     max_no_improve=300, bisection_iters=10,
                     block_size=1):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        c_max_val = np.max(corr)
        active_threshold = c_max_val - 1e-10
        active_mask = corr >= active_threshold
        n_act = np.sum(active_mask)
        if n_act == 0:
            top_idx = np.argsort(corr)[-n_active:]
        else:
            top_idx = np.where(active_mask)[0]
            if len(top_idx) > n_active * 3:
                top_idx = np.argsort(corr)[-n_active * 3:]

        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(len(top_idx)):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_move = None

        n_h = len(high_set)
        n_l = len(low_set)

        if block_size <= 1:
            max_evaluate = 300
            if n_h * n_l <= max_evaluate:
                pairs = [(int(high_set[ii]), int(low_set[jj]))
                         for ii in range(n_h) for jj in range(n_l)
                         if high_set[ii] != low_set[jj]]
            else:
                pairs = []
                step_h = max(1, n_h // 15)
                step_l = max(1, n_l // 15)
                for ii in range(0, n_h, step_h):
                    for jj in range(0, n_l, step_l):
                        iv = int(high_set[ii])
                        jv = int(low_set[jj])
                        if iv != jv:
                            pairs.append((iv, jv))
                for _ in range(min(max_evaluate - len(pairs), 50)):
                    ii = int(np.random.randint(n_h))
                    jj = int(np.random.randint(n_l))
                    pairs.append((int(high_set[ii]), int(low_set[jj])))

            for i, j in pairs:
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue
                g = change[j] - change[i]
                delta_star = bisection_step(corr, g, delta_hi, bisection_iters)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_move = ([(i)], [(j)], delta_star)
        else:
            bs = min(block_size, n_h, n_l)
            donors = high_set[:bs]
            recipients = low_set[:bs]
            g_block = np.zeros(n_corr)
            for d in donors:
                g_block -= change[d]
            for r in recipients:
                g_block += change[r]
            max_deltas = np.minimum(h[donors], 1.0 - h[recipients])
            max_delta = np.min(max_deltas)
            if max_delta > 1e-12:
                delta_star = bisection_step(corr, g_block, max_delta, bisection_iters)
                if delta_star > 1e-12:
                    approx_max = np.max(corr + delta_star * g_block) * dx
                    if approx_max < best_trial:
                        best_trial = approx_max
                        best_move = ([int(d) for d in donors], [int(r) for r in recipients], delta_star)

            if n_h >= 2 * bs and n_l >= 2 * bs:
                donors2 = high_set[bs:2*bs]
                recipients2 = low_set[bs:2*bs]
                g_block2 = np.zeros(n_corr)
                for d in donors2:
                    g_block2 -= change[d]
                for r in recipients2:
                    g_block2 += change[r]
                max_deltas2 = np.minimum(h[donors2], 1.0 - h[recipients2])
                max_delta2 = np.min(max_deltas2)
                if max_delta2 > 1e-12:
                    delta_star2 = bisection_step(corr, g_block2, max_delta2, bisection_iters)
                    if delta_star2 > 1e-12:
                        approx_max2 = np.max(corr + delta_star2 * g_block2) * dx
                        if approx_max2 < best_trial:
                            best_trial = approx_max2
                            best_move = ([int(d) for d in donors2], [int(r) for r in recipients2], delta_star2)

        if best_move is not None:
            donors, recipients, delta = best_move
            for d in donors:
                h[d] -= delta
            for r in recipients:
                h[r] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for d in donors:
                update_change_row(change, d, h, n)
            for r in recipients:
                update_change_row(change, r, h, n)

    return best_h, best_c5


def projected_gradient_step(h, n, target, lr=0.01, n_active=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    top_idx = np.argsort(corr)[-n_active:]
    idxs = np.arange(n)
    grad = np.zeros(n)
    w_sum = np.sum(corr[top_idx])
    if w_sum <= 0:
        w_sum = 1.0
    weights = corr[top_idx] / w_sum
    for idx in range(n_active):
        k = int(top_idx[idx]) - (n - 1)
        w = weights[idx]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad -= np.mean(grad)
    h_new = h - lr * grad
    h_new = project_feasible(h_new, target)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP at initial resolution
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n0, target0, [100, 500, 2000], 400)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD at initial resolution
    if tleft() > 100:
        h_cd, c5_cd = enhanced_cd_fast(best_h, n0, target0,
                                       time_limit=min(60, tleft() * 0.3),
                                       max_sweeps=15000, n_active=3,
                                       n_cand_start=15, n_cand_end=40,
                                       max_no_improve=300, bisection_iters=11,
                                       block_size=1)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Upscale to 800
    if tleft() > 250:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        tl1 = min(150, tleft() * 0.4)
        h_cd1, c5_cd1 = enhanced_cd_fast(h1, n1, target1,
                                          time_limit=tl1,
                                          max_sweeps=25000, n_active=3,
                                          n_cand_start=20, n_cand_end=60,
                                          max_no_improve=400, bisection_iters=11,
                                          block_size=2)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1
            n0 = n1
            target0 = target1
        else:
            h_cd1b, c5_cd1b = enhanced_cd_fast(h1, n1, target1,
                                                time_limit=min(60, tleft() * 0.2),
                                                max_sweeps=15000, n_active=5,
                                                n_cand_start=30, n_cand_end=70,
                                                max_no_improve=300, bisection_iters=10,
                                                block_size=1)
            if c5_cd1b < best_score:
                best_score = c5_cd1b
                best_h = h_cd1b
                n0 = n1
                target0 = target1

    # SLSQP polish at current resolution
    if tleft() > 120:
        h_pol = slsqp_round(best_h, n0, target0, [5000, 20000], 600)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: CD with block moves at current resolution
    if tleft() > 150:
        h_cd2, c5_cd2 = enhanced_cd_fast(best_h, n0, target0,
                                          time_limit=min(100, tleft() * 0.4),
                                          max_sweeps=20000, n_active=5,
                                          n_cand_start=25, n_cand_end=70,
                                          max_no_improve=350, bisection_iters=11,
                                          block_size=3)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 5: Try higher resolution (1200)
    if tleft() > 350:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        tl2 = min(250, tleft() * 0.5)
        h_cd2, c5_cd2 = enhanced_cd_fast(h2, n2, target2,
                                          time_limit=tl2,
                                          max_sweeps=30000, n_active=3,
                                          n_cand_start=25, n_cand_end=70,
                                          max_no_improve=500, bisection_iters=10,
                                          block_size=2)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
            n0 = n2
            target0 = target2

    # SLSQP polish
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n0, target0, [20000, 100000], 500)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: More CD refinement
    if tleft() > 120:
        h_cd3, c5_cd3 = enhanced_cd_fast(best_h, n0, target0,
                                          time_limit=min(80, tleft() * 0.4),
                                          max_sweeps=20000, n_active=5,
                                          n_cand_start=35, n_cand_end=80,
                                          max_no_improve=300, bisection_iters=12,
                                          block_size=2)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 7: Projected gradient exploration + CD
    if tleft() > 150:
        for pg_iter in range(30):
            if tleft() < 80:
                break
            h_pg = projected_gradient_step(best_h, n0, target0, lr=0.005, n_active=5)
            h_re, sc_re = enhanced_cd_fast(h_pg, n0, target0,
                                            time_limit=min(30, tleft() * 0.15),
                                            max_sweeps=8000, n_active=3,
                                            n_cand_start=20, n_cand_end=50,
                                            max_no_improve=200, bisection_iters=10,
                                            block_size=1)
            if sc_re < best_score:
                best_score = sc_re
                best_h = h_re

    # Phase 8: Random perturbations
    if tleft() > 120:
        for scale in [0.02, 0.05, 0.08]:
            if tleft() < 70:
                break
            h_pert = best_h + rng.randn(n0) * scale
            h_pert = project_feasible(h_pert, target0)
            h_re, sc_re = enhanced_cd_fast(h_pert, n0, target0,
                                            time_limit=min(40, tleft() * 0.25),
                                            max_sweeps=10000, n_active=3,
                                            n_cand_start=25, n_cand_end=60,
                                            max_no_improve=250, bisection_iters=10,
                                            block_size=2)
            if sc_re < best_score:
                best_score = sc_re
                best_h = h_re

    # Final SLSQP polish
    if tleft() > 50:
        h_final = slsqp_round(best_h, n0, target0, [50000, 200000], 400)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final CD
    if tleft() > 30:
        h_final, c5_final = enhanced_cd_fast(best_h, n0, target0,
                                              time_limit=min(20, tleft() * 0.5),
                                              max_sweeps=10000, n_active=5,
                                              n_cand_start=40, n_cand_end=80,
                                              max_no_improve=200, bisection_iters=10,
                                              block_size=2)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
