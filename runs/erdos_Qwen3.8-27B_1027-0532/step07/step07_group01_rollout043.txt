def compute_subgradient(h, n, active_shifts):
    """Compute subgradient of max_k f_k at h, averaged over active shifts."""
    dx = 2.0 / n
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k in active_shifts:
        # f_k = sum_x h(x)(1-h(x+k)) for valid x
        # df_k/dh(x) = (1-h(x+k)) for x+k valid, and -h(x-k) for x-k valid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad /= len(active_shifts)
    # Subtract mean to handle the equality constraint (Lagrange multiplier)
    grad -= np.mean(grad)
    return grad
```

    B[:n] = one_minus_h
    
    # Convolution: (P * B)[i] = sum_k P[k] * B[i-k]
    # We want: sum_k p[k] * (1-h[i+k]) = sum_j (1-h[j]) * p[j-i] (letting j=i+k)
    # = sum_j B[j] * P[(j-i)+(n-1)]... 
    # This is getting too confusing. Let me just use direct computation but with the trick
    # of only computing over significant p values.
    
    # Actually, for the FFT approach:
    # Define p_ext as a sequence of length 2n-1: p_ext[k] = p[k] for k=0,...,2n-2
    # This represents the weights for shifts from -(n-1) to (n-1).
    # 
    # grad1[i] = sum_{k: 0<=i+k<n} p_ext[k+(n-1)] * (1-h[i+k])
    # Let j = i+k, so k = j-i, and k+(n-1) = j-i+n-1
    # grad1[i] = sum_{j: 0<=j<n, 0<=j-i+n-1<2n-1} p_ext[j-i+n-1] * (1-h[j])
    # = sum_{j: 0<=j<n, i-n+1<=j<=i+n-1} p_ext[j-(i-n+1)] * (1-h[j])
    
    # This is a cross-correlation! 
    # Let's define q[m] = p_ext[m] for m=0,...,2n-2 (length 2n-1)
    # grad1[i] = sum_j (1-h[j]) * q[j - (i - n + 1)]
    # = cross_corr(1-h, q)[j - (i-n+1) summed over j]
    # = (q_rev * (1-h))[i - n + 1] where * is linear convolution
    # = (q_rev * (1-h))[i] shifted by (n-1)
    
    # Hmm, let me just use numpy correlate directly:
    # np.correlate(a, v) = sum_n a[n+len(v)-1] * conj(v[n])
    # Actually numpy docs: c[k] = sum_n a[n+k] * conj(v[n]) for the 'full' mode
    
    # Let me try a different approach. Just use the direct computation but vectorize over i:
    
    # For each i, grad1[i] = sum over k such that 0<=i+k<n of p_ext[k+n-1] * (1-h[i+k])
    # = sum over j such that 0<=j<n and j=i+k (k=j-i) of p_ext[j-i+n-1] * (1-h[j])
    # Valid when 0<=j-i+n-1<2n-1, i.e., i-n+1<=j<=i+n-1, and 0<=j<n
    
    # I'll just use the loop over significant p values for high alpha
    sig_mask = p > 1e-10
    sig_indices = np.where(sig_mask)[0]
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in sig_indices:
        k = s - (n - 1)
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += p[s] * (1.0 - x[idxs[v1] + k])
        grad[v2] -= p[s] * x[idxs[v2] - k]
    grad -= np.mean(grad)
    return value, grad
```

Let me also think about the block CD. For a block of B coordinates {i_1, ..., i_B}, we want to find the optimal values for these B coordinates given the others are fixed.

The objective is max_k C_k, where C_k = sum_x h(x)(1-h(x+k)) dx.

For the B variable coordinates, C_k becomes a quadratic function in these B variables (plus a constant). We want to minimize the max of these quadratic functions.

For B=2 or 3, we can use a small-scale optimization (e.g., scipy minimize on the B variables). For larger B, we'd need to be more careful.

Let me also consider a different angle: **mirror descent / entropic regularization**.

Instead of Euclidean projection, use the Bregman divergence with the negative entropy. This gives a multiplicative update:
h_new[i] = h[i] * exp(-lr * grad[i])

Then renormalize to satisfy the sum constraint. This naturally keeps h in [0,1] (if h starts in [0,1]) and might lead to better solutions.

Actually, the multiplicative weights update is:
h_new[i] ∝ h[i] * exp(-lr * grad[i])

Then project to {sum = target}. This is like exponentiated gradient / Follow-the-Regularized-Leader with entropic regularization.

This might work well because:
- It naturally respects the [0,1] constraint (exponential of something is always positive)
- It can make large changes to coordinates that need them
- It's related to the multiplicative weights algorithm which has good theoretical guarantees

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value_grad(x, alpha):
    """Compute LSE value and gradient, with mean subtraction for sum constraint."""
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    p = exp_vals / total
    value = c_max + np.log(total) / alpha
    
    idxs = np.arange(n)
    grad = np.zeros(n)
    sig = p > 1e-12
    sig_idx = np.where(sig)[0]
    for s in sig_idx:
        k = s - (n - 1)
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += p[s] * (1.0 - x[idxs[v1] + k])
        grad[v2] -= p[s] * x[idxs[v2] - k]
    grad -= np.mean(grad)
    return value, grad

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    
    def cons_fun(x):
        return np.sum(x) - _g_target
    
    def cons_jac(x):
        return np.ones(_g_n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
        
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
```

- Optimize over the block while keeping others fixed
- Use a small-scale optimizer for the block

```python
def block_cd(h_init, n, target, time_limit, block_size=3, max_sweeps=5000):
    start_t = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    
    # Precompute shift array
    shifts = np.arange(-(n-1), n)
    
    no_improve = 0
    rng = np.random.RandomState(55)
    
    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx  # dx = 2.0/n
        dx = 2.0 / n
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        if no_improve > 200:
            h = best_h + rng.randn(n) * 0.01
            h = project_feasible(h, target)
            no_improve = 0
        
        # Compute gradient
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        
        # Select block: B coordinates with most extreme gradient values
        sorted_idx = np.argsort(grad)
        # Pick a mix of low (should increase) and high (should decrease)
        half = block_size // 2
        low_block = sorted_idx[:half]
        high_block = sorted_idx[-(block_size-half):]
        block = np.concatenate([low_block, high_block])
        
        # Optimize over the block
        def block_obj(block_vals):
            h_temp = h.copy()
            for i, bv in zip(block, block_vals):
                h_temp[i] = bv
            h_temp = np.clip(h_temp, 0.0, 1.0)
            corr_temp = np.correlate(h_temp, 1.0 - h_temp, mode='full')
            return np.max(corr_temp)
        
        # Constraints: sum of block vals + sum of rest = target
        rest_sum = target - np.sum(h[np.setdiff1d(idxs, block)])
        # bounds for each block variable
        bnds = [(0.0, 1.0)] * block_size
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - rest_sum}]
        
        try:
            res = minimize(block_obj, h[block], method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': 100, 'ftol': 1e-12})
            if res.success or res.fun < block_obj(h[block]):
                h_new = h.copy()
                h_new[block] = res.x
                h_new = project_feasible(h_new, target)
                new_score = compute_c5(h_new)
                if new_score < c5_val:
                    h = h_new
        except Exception:
            pass
    
    return best_h, best_score
```

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):

    n = len(h)

        h = np.clip(h, 0.0, 1.0)

def lse_value_grad(x, alpha):

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)

    sig_idx = np.where(p > 1e-10)[0]
