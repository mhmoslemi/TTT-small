import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff/n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds_lp = [(0.0, 1.0)] * n + [(-10.0, None)]
    
    A_ub = np.zeros((n_corr, n + 1))
    b_ub = np.zeros(n_corr)
    
    for it in range(n_iters):
        if time_left_fn is not None and time_left_fn() < 10:
            break
        
        # Build linearization
        const = np.zeros(n_corr)
        for m in range(n_corr):
            k = m - mid
            i_min = max(0, -k)
            i_max = min(n - 1, n - 1 - k)
            if i_min > i_max:
                A_ub[m] = 0
                b_ub[m] = 0
                continue
            c = np.zeros(n)
            ct = 0.0
            i_idx = np.arange(i_min, i_max + 1)

            c[i_idx] -= h[j_idx]

                         A_eq=A_eq, b_eq=b_eq,

            h_new = project_feasible(h_new, target)

            else:
                h = h_new
        except Exception:
            break
    
    return best_h, best_score

        j2_lo = max(0, -k)
        j2_hi = min(n - 1, n - 1 - k)
        c[j2_lo:j2_hi + 1] -= h[j2_lo + k:j2_hi + k + 1]
        b[m] = -float(np.dot(h[j2_lo:j2_hi + 1], h[j2_lo + k:j2_hi + k + 1]))
        A[m] = c
    return A, b

def sca_lp(h_init, n, target, n_iters, time_fn, damping=1.0):
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds_lp = [(0.0, 1.0)] * n + [(-10.0, None)]
    rng = np.random.RandomState(7)
    prev_score = best_score
    stall = 0
    for it in range(n_iters):
        if time_fn() < 15:
            break
        A, b = build_sca_matrices(h, n)
        A_ub = np.hstack([A, -np.ones((n * 2 - 1, 1))])
        b_ub = b
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                         A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs')
        except Exception:
            break
        if not res.success:
            break
        h_lp = res.x[:n].copy()
        t_lp = float(res.x[n])
        h_new = (1.0 - damping) * h + damping * h_lp
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_new.copy()
        actual_max = float(np.max(compute_corr(h_lp)))
        if abs(actual_max - t_lp) < 1e-4 * max(1.0, abs(t_lp)):
            stall += 1
            if stall >= 3:
                h = best_h + 0.005 * rng.randn(n)
                h = project_feasible(h, target)
                stall = 0
            else:
                h = best_h.copy()
        else:
            h = h_new
            stall = 0
        if sc >= prev_score - 1e-12 and stall > 5:
            h = best_h + 0.01 * rng.randn(n)
            h = project_feasible(h, target)
            stall = 0
        prev_score = sc
    return best_h, best_score

def interceptive_cd(h_init, n, target, time_limit, max_sweeps=20000,
                    n_cand_start=15, n_cand_end=60, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = float(np.max(vals))
                    if f_prime := float(np.max(g_vec[vals >= f_mid - 1e-14])) > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx = float(np.max(corr + delta_star * g_vec)) * dx
                if approx < best_trial_c5:
                    best_trial_c5 = approx
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pi, pj = best_i, best_j
            for p in (pi, pj):
                change[p, :] = 0.0
                pos_m = (p + shifts >= 0) & (p + shifts < n)
                neg_m = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_m] += (1.0 - h[p + shifts[pos_m]])
                change[p, neg_m] -= h[p - shifts[neg_m]]
    return best_h, best_c5

def binary_hill_climb(h_init, n, target, time_limit, max_rounds=200):
    """Round to binary, then hill-climb"""
    start = time.time()
    h_bin = (h_init > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target)
    best_h = h_bin.copy()
    best_score = compute_c5(h_bin)
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in ones.copy():
            if time.time() - start > time_limit:
                break
            for j in zeros.copy():
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                sc = compute_c5(h_bin)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                if sc < best_score - 1e-14:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                    best_score = sc
                    best_h = h_bin.copy()
                    ones = np.where(h_bin > 0.5)[0]
                    zeros = np.where(h_bin < 0.5)[0]
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_score

def subgradient_smooth(h_init, n, target, time_limit, max_iters=5000,
                       alpha_start=100, alpha_end=10000):
    """Subgradient method with smooth max approximation"""
    start = time.time()
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = 0.7
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max_iters)
        t_k = 1.0 + (1.0 + momentum) ** (0.5 + it * (1.0 - momentum) / max_iters)
        y = h + ((t_k - 1.0) / t_k) * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = compute_corr(y)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            p = np.zeros(len(corr))
            p[np.argmax(corr)] = 1.0
        else:
            p /= p_sum
        mid = n - 1
        idxs = np.arange(n)
        grad = np.zeros(n)
        for m in range(len(corr)):
            if p[m] < 1e-12:
                continue
            k = m - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += p[m] * (1.0 - y[idxs[v1] + k])
            grad[v2] -= p[m] * y[idxs[v2] - k]
        step = 2.0 / (alpha * np.sqrt(np.dot(grad, grad) + 1e-30))
        h_new = y - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        h_prev = h.copy()
        h = h_new
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score

import numpy as np
from scipy.optimize import linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
