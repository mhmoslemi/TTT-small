def swap_delta(h, a, b, n):
    """Compute change in corr when h[a]:1->0, h[b]:0->1."""
    m = n - 1
    n_corr = 2 * n - 1
    delta = np.zeros(n_corr)
    
    # h' = h with h'[a]=0, h'[b]=1
    # Term 1: -(1-h'(a+m-k)) for valid k
    # a+m-k in [0, n-1] => k in [a+m-n+1, a+m]
    k_lo = max(0, a + m - n + 1)
    k_hi = min(n_corr - 1, a + m)
    idx = a + m - np.arange(k_lo, k_hi + 1)
    h_prime_at_idx = h[idx].copy()
    h_prime_at_idx[idx == a] = 0
    h_prime_at_idx[idx == b] = 1
    delta[k_lo:k_hi+1] -= (1 - h_prime_at_idx)

    k_lo = max(0, b + m - n + 1)
    k_hi = min(n_corr - 1, b + m)

    h_prime_at_idx = h[idx].copy()
    h_prime_at_idx[idx == a] = 0

    delta[k_lo:k_hi+1] += (1 - h_prime_at_idx)
    
    # Term 3: +h'(a+k-m) for valid k
