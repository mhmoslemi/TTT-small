    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    mid = n - 1
    c_max = float(np.max(corr))

        weights = np.exp(10.0 * (corr - c_max) / max(c_max, 1e-10))
    else:

        weights = np.zeros(n_corr)
        weights[top_idx] = 1.0

        k = int(s_idx) - mid

        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    
    # Center the gradient (for the equality constraint)
    grad -= np.mean(grad)
    return grad * dx, corr, c_max * dx

            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

            best_h = h_new.copy()
            no_improve = 0

        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        y = project_feasible(y, target)
        t_k = t_new
        h = h_new

        best_move = None
        for ii in range(n_cand):
            i = int(high_set[n_cand - 1 - ii])
            for jj in range(n_cand):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                delta_star = 0.5 * (d_lo + d_hi)
                approx = float(np.max(corr + delta_star * g)) * dx
                if approx < best_trial - 1e-12:
                    best_trial = approx
                    best_move = (i, j, delta_star)
        if best_move is not None:
            i, j, d = best_move
            g = change[j] - change[i]
            a, b = max(h[i] - 1.0, -h[j]), min(h[i], 1.0 - h[j])
            for _ in range(6):
                mid_d = 0.5 * (a + b)
                vals = corr + mid_d * g
                f_mid = float(np.max(vals))
                amask = vals >= f_mid - 1e-13
                if amask.any():
                    fp = float(np.max(g[amask]))
                else:
                    fp = 0.0
                if fp > 1e-11:
                    b = mid_d
                else:
                    a = mid_d
            best_delta = 0.5 * (a + b)
            if abs(best_delta) > 1e-12:
                h[i] -= best_delta
                h[j] += best_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
    return best_h, best_c5

    global _g_n, _g_target
