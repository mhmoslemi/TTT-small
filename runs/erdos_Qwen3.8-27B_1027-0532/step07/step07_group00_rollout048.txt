def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_init(n, freq=3):
    """Create a structured initial guess with oscillation."""
    x = np.linspace(0, 2, n)
    # Multi-frequency structure
    h = 0.5 + 0.3 * np.sin(np.pi * freq * x) + 0.15 * np.sin(np.pi * 2 * freq * x + 0.5)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)

def random_structured_init(n, rng):
    """Random structured initial guess."""

def subgradient_step(h, n, target, step_size, n_active=5):
    """One step of subgradient method on the smooth approximation."""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    
    # Find active shifts
    threshold = c_max - 0.05 * c_max
    active = np.where(corr >= threshold)[0]
    if len(active) > n_active:
        active = active[np.argsort(corr[active])[-n_active:]]
    
    grad = np.zeros(n)
    for s_idx in active:
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad -= np.mean(grad)
    
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, target)
    return h_new, compute_c5(h_new)

def adaptive_subgradient(h_init, n, target, time_limit, n_iter=10000):
    """Adaptive subgradient with momentum."""
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    beta = 0.9
    step = 0.01
    no_improve = 0
    
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break
        
        h_new, sc = subgradient_step(h, n, target, step)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_new.copy()
            no_improve = 0
            step = min(step * 1.02, 0.05)
        elif sc < compute_c5(h) + 1e-10:
            no_improve += 1
            step *= 0.95
        
        momentum = beta * momentum + (1 - beta) * (h_new - h)
        h = h_new + 0.1 * momentum
        h = project_feasible(h, target)
    
    return best_h, best_c5

    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_line_search(corr, g, delta_lo, delta_hi, n_bisect=18):
    a_d = delta_lo
    b_d = delta_hi
    for _ in range(n_bisect):
        mid_d = 0.5 * (a_d + b_d)
        vals = corr + mid_d * g
        f_mid = np.max(vals)
        active_mask = vals >= f_mid - 1e-14
        if active_mask.any():
            f_prime = np.max(g[active_mask])
        else:
            f_prime = 0.0
        if f_prime > 1e-12:
            b_d = mid_d
        else:
            a_d = mid_d
    return 0.5 * (a_d + b_d)


def improved_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                     n_cand=30, max_no_improve=500, n_active=8,
                     use_anneal=False, T0=0.0002, beta_anneal=0.97):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(9999)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    T = T0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if use_anneal:
                T *= beta_anneal

        if consecutive_no_improve > max_no_improve:
            break

        c_max = float(np.max(corr))
        threshold = c_max - 0.02 * c_max
        active_mask = corr >= threshold
        active_count = int(np.sum(active_mask))

        if active_count > n_active:
            top_active = np.argsort(corr)[::-1][:n_active]
            grad = np.zeros(n)
            for s_idx in top_active:
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += (1.0 - h[idxs[v1] - k])
                grad[v2] -= h[idxs[v2] + k]
        else:
            grad = np.zeros(n)
            for s_idx in np.where(active_mask)[0]:
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += (1.0 - h[idxs[v1] - k])
                grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = min(len(high_set), n_cand)
        n_l = min(len(low_set), n_cand)

        for ii in range(n_h):
            i = int(high_set[n_h - 1 - ii])
            found_improvement = False
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                delta_star = fast_line_search(corr, g, delta_lo, delta_hi)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    found_improvement = True
                    if best_trial_c5 < c5_val - 0.005 * c5_val:
                        break
            if found_improvement and best_trial_c5 < c5_val - 0.01 * c5_val:
                break

        if use_anneal and best_i < 0 and T > 1e-10:
            for _trial in range(5):
                i = int(high_set[rng.randint(n_h)])
                j = int(low_set[rng.randint(n_l)])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                delta_star = fast_line_search(corr, g, delta_lo, delta_hi)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if rng.random() < np.exp(-(approx_max - c5_val) / T):
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def batch_cd(h_init, n, target, time_limit, max_sweeps=20000,
             batch_size=4, n_cand=35, max_no_improve=350):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        c_max = float(np.max(corr))
        weights = np.exp(6.0 * (corr - c_max) / max(c_max, 1e-10))
        top_k = min(20, n_corr)
        top_idx = np.argsort(weights)[-top_k:]

        grad = np.zeros(n)
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-6:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        used = set()
        moves = []
        for i_wrap in range(min(batch_size * 2, len(high_set))):
            if len(moves) >= batch_size:
                break
            i = int(high_set[len(high_set) - 1 - i_wrap])
            if i in used:
                continue
            best_j = -1
            best_approx = c5_val
            for j_wrap in range(min(batch_size * 2, len(low_set))):
                j = int(low_set[j_wrap])
                if j in used or j == i:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                delta_star = fast_line_search(corr, g, delta_lo, delta_hi, 14)
                if abs(delta_star) < 1e-12:
                    continue
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_approx:
                    best_approx = approx
                    best_j = j
            if best_j >= 0 and best_approx < c5_val:
                moves.append((i, best_j))
                used.add(i)
                used.add(best_j)

        if moves:
            for (i, j) in moves:
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                cur_corr = np.correlate(h, 1.0 - h, mode='full')
                g = change[j] - change[i]
                delta_star = fast_line_search(cur_corr, g, delta_lo, delta_hi)
                if abs(delta_star) > 1e-12:
                    h[i] -= delta_star
                    h[j] += delta_star
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    update_change_row(change, i, h, n, shifts)
                    update_change_row(change, j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def antisymmetric_init(n, freq=2):
    half = n // 2
    x = np.linspace(0, 1, half)
    h_half = 0.5 + 0.4 * np.sin(np.pi * freq * x) + 0.1 * np.sin(2 * np.pi * freq * x)
    h_half = np.clip(h_half, 0.0, 1.0)
    h = np.concatenate([h_half, 1.0 - h_half])
    if len(h) < n:
        h = np.concatenate([h, np.zeros(n - len(h))])
    elif len(h) > n:
        h = h[:n]
    return project_feasible(h, n / 2.0)


def staircase_init(n, n_steps=4):
    h = np.zeros(n)
    for i in range(n):
        block = int(i * n_steps / n)
        h[i] = (block % 2 == 0)
    return project_feasible(h.astype(float), n / 2.0)


def smooth_wave_init(n, freq=3, phase=0.0):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.35 * np.sin(np.pi * freq * x + phase) + 0.15 * np.cos(2 * np.pi * freq * x + 0.7)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 18.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    rng = np.random.RandomState(seed)

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP on initial
    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, n0, target0, [300, 1500], 300)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Improved pair CD
    if tleft() > 180:
        h_cd, sc_cd = improved_pair_cd(best_h, n0, target0,
                                       time_limit=min(160, tleft() * 0.6),
                                       max_sweeps=40000, n_cand=25,
                                       max_no_improve=400, n_active=6)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 3: Try diverse starting points at coarse resolution
    n_coarse = 200
    target_coarse = n_coarse / 2.0
    candidates = []

    if tleft() > 350:
        # Generate diverse candidates
        h_a = antisymmetric_init(n_coarse, freq=2)
        h_b = antisymmetric_init(n_coarse, freq=3)
        h_c = staircase_init(n_coarse, n_steps=6)
        h_d = smooth_wave_init(n_coarse, freq=4, phase=0.5)
        h_e = smooth_wave_init(n_coarse, freq=5, phase=1.2)
        h_f = np.interp(np.linspace(0, 1, n_coarse), np.linspace(0, 1, n0), best_h)
        h_f = project_feasible(h_f, target_coarse)

        for h_cand in [h_a, h_b, h_c, h_d, h_e, h_f]:
            if tleft() < 80:
                break
            sc = compute_c5(h_cand)
            if tleft() > 80:
                h_opt, sc_opt = slsqp_optimize(h_cand, n_coarse, target_coarse, [200, 800], 200)
                if sc_opt < sc:
                    sc = sc_opt
                    h_cand = h_opt
            if tleft() > 60:
                h_cd_c, sc_cd_c = improved_pair_cd(h_cand, n_coarse, target_coarse,
                                                   time_limit=min(40, tleft() * 0.3),
                                                   max_sweeps=10000, n_cand=15,
                                                   max_no_improve=200, n_active=5)
                if sc_cd_c < sc:
                    sc = sc_cd_c
                    h_cand = h_cd_c
            candidates.append((sc, h_cand))

        candidates.sort(key=lambda x: x[0])

        # Take the best coarse candidate and refine at full resolution
        if candidates and tleft() > 250:
            best_coarse_score, best_coarse_h = candidates[0]
            h_fine = interpolate_to(best_coarse_h, n0, target0)
            sc_fine = compute_c5(h_fine)

            if tleft() > 180:
                h_pol_f, sc_pf = slsqp_optimize(h_fine, n0, target0, [300, 1500, 5000], 300)
                if sc_pf < sc_fine:
                    sc_fine = sc_pf
                    h_fine = h_pol_f

            if tleft() > 130:
                h_cd_f, sc_cf = improved_pair_cd(h_fine, n0, target0,
                                                 time_limit=min(120, tleft() * 0.5),
                                                 max_sweeps=20000, n_cand=30,
                                                 max_no_improve=300, n_active=8)
                if sc_cf < sc_fine:
                    sc_fine = sc_cf
                    h_fine = h_cd_f

            if sc_fine < best_score:
                best_score = sc_fine
                best_h = h_fine

        # Also try second-best coarse candidate
        if len(candidates) > 1 and tleft() > 200:
            sc2, h2 = candidates[1]
            if sc2 < best_score * 1.005:
                h_fine2 = interpolate_to(h2, n0, target0)
                if tleft() > 150:
                    h_pol_f2, sc_pf2 = slsqp_optimize(h_fine2, n0, target0, [500, 2000], 250)
                    if sc_pf2 < compute_c5(h_fine2):
                        h_fine2 = h_pol_f2
                        sc_pf2 = sc_pf2
                if tleft() > 80:
                    h_cd_f2, sc_cf2 = improved_pair_cd(h_fine2, n0, target0,
                                                       time_limit=min(70, tleft() * 0.5),
                                                       max_sweeps=15000, n_cand=25,
                                                       max_no_improve=250, n_active=6)
                    if sc_cf2 < sc_pf2:
                        sc_pf2 = sc_cf2
                        h_fine2 = h_cd_f2
                if sc_pf2 < best_score:
                    best_score = sc_pf2
                    best_h = h_fine2

    # Phase 4: Batch CD
    if tleft() > 120:
        h_bc, sc_bc = batch_cd(best_h, n0, target0,
                               time_limit=min(100, tleft() * 0.5),
                               max_sweeps=15000, batch_size=4,
                               n_cand=30, max_no_improve=300)
        if sc_bc < best_score:
            best_score = sc_bc
            best_h = h_bc

    # Phase 5: Annealed CD for escape
    if tleft() > 100:
        h_an, sc_an = improved_pair_cd(best_h, n0, target0,
                                       time_limit=min(90, tleft() * 0.5),
                                       max_sweeps=20000, n_cand=35,
                                       max_no_improve=500, n_active=10,
                                       use_anneal=True, T0=0.0004, beta_anneal=0.985)
        if sc_an < best_score:
            best_score = sc_an
            best_h = h_an

    # Phase 6: Perturbation + re-optimize
    if tleft() > 130:
        for idx, noise_scale in enumerate([0.015, 0.04, 0.08]):
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)

            if tleft() > 50:
                h_pol_p, sc_p = slsqp_optimize(h_pert, n0, target0, [500, 2000], 250)
                if sc_p < compute_c5(h_pert):
                    h_pert = h_pol_p

            if tleft() > 40:
                h_cd_p, sc_cd_p = improved_pair_cd(h_pert, n0, target0,
                                                   time_limit=min(35, tleft() * 0.4),
                                                   max_sweeps=10000, n_cand=20,
                                                   max_no_improve=150, n_active=6)
                if sc_cd_p < compute_c5(h_pert):
                    h_pert = h_cd_p
                    sc_cd_p = sc_cd_p

            sc_final_p = compute_c5(h_pert)
            if sc_final_p < best_score:
                best_score = sc_final_p
                best_h = h_pert

    # Phase 7: Structured perturbation (frequency-targeted)
    if tleft() > 80:
        x = np.linspace(0, 2, n0)
        for freq_p in [2.0, 3.0, 5.0]:
            if tleft() < 40:
                break
            h_pert = best_h + 0.03 * np.sin(np.pi * freq_p * x)
            h_pert = project_feasible(h_pert, target0)
            h_pol_p, sc_p = slsqp_optimize(h_pert, n0, target0, [1000, 5000], 250)
            if sc_p < best_score:
                best_score = sc_p
                best_h = h_pol_p
                if tleft() > 40:
                    h_cd_p, sc_cd_p = improved_pair_cd(best_h, n0, target0,
                                                       time_limit=min(30, tleft() * 0.4),
                                                       max_sweeps=8000, n_cand=25,
                                                       max_no_improve=150, n_active=6)
                    if sc_cd_p < best_score:
                        best_score = sc_cd_p
                        best_h = h_cd_p

    # Phase 8: Final intensive CD
    if tleft() > 60:
        h_final_cd, sc_final_cd = improved_pair_cd(best_h, n0, target0,
                                                   time_limit=min(50, tleft() * 0.7),
                                                   max_sweeps=20000, n_cand=50,
                                                   max_no_improve=200, n_active=5)
        if sc_final_cd < best_score:
            best_score = sc_final_cd
            best_h = h_final_cd

    # Phase 9: Final SLSQP
    if tleft() > 30:
        h_fin, sc_fin = slsqp_optimize(best_h, n0, target0, [10000, 50000], 300)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 10: One last CD sweep
    if tleft() > 15:
        h_last, sc_last = improved_pair_cd(best_h, n0, target0,
                                           time_limit=min(12, tleft() * 0.8),
                                           max_sweeps=5000, n_cand=40,
                                           max_no_improve=100, n_active=5)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
