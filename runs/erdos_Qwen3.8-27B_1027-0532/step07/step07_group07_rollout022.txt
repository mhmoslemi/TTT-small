import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fft_correlate_full(a, b):
    """Compute full cross-correlation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(np.conj(A) * B, fft_size)[:size]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_method(h_init, n, target, time_limit, m_lags=5, 
                       init_step=0.1, max_iter=50000,
                       smoothing=0.0):
    """Subgradient method with adaptive step size."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = init_step
    h_prev = h.copy()
    corr_prev = None
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Compute correlation via FFT
        corr = fft_correlate_full(h, 1.0 - h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find top-m lags
        top_idx = np.argpartition(corr, -m_lags)[-m_lags:]
        
        # Compute subgradient
        grad = np.zeros(n)
        for idx in top_idx:
            k = int(idx) - (n - 1)
            # grad_k[j] = (1 - h[j-k]) * 1_{0<=j-k<n} - h[j+k] * 1_{0<=j+k<n}
            for j in range(n):
                g = 0.0
                if 0 <= j - k < n:
                    g += (1.0 - h[j - k])
                if 0 <= j + k < n:
                    g -= h[j + k]
                grad[j] += g
        grad /= m_lags
        
        # Add smoothing if desired
        if smoothing > 0:
            if n > 2:
                smooth_grad = np.zeros(n)
                smooth_grad[1:-1] = (h[2:] - 2*h[1:-1] + h[:-2]) * smoothing
                grad += smooth_grad
        
        # Barzilai-Borwein step size
        if corr_prev is not None:
            s = h - h_prev
            y = -grad  # negative gradient
            sy = np.dot(s, y)
            ys = np.dot(y, s)
            if abs(ys) > 1e-12 and sy > 0:
                bb_step = sy / ys
                step = 0.5 * step + 0.5 * np.clip(bb_step, 0.01 * init_step, 10 * init_step)
        
        h_prev = h.copy()
        corr_prev = corr
        
        # Update
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        
        # Backtracking if no improvement
        new_c5 = compute_c5(h_new)
        if new_c5 > c5_val * 1.01 + 1e-6:
            step *= 0.7
            h_new = h - step * grad
            h_new = project_feasible(h_new, target)
        
        h = h_new
        
        # Decay step size slowly
        step *= 0.9995
    
    return best_h, best_c5

from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fft_correlate_full(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(np.conj(A) * B, fft_size)[:size]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_for_lag(h, k, n):
    """Compute gradient of overlap at lag k w.r.t. h."""
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    else:
        kp = -k
        grad[:n-kp] += (1.0 - h[kp:])
        grad[kp:] -= h[:n-kp]
    return grad

def subgradient_refine(h_init, n, target, time_limit, m_lags=5, 
                       smax=50, init_step=None, smooth_coeff=0.0):

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    if init_step is None:
        init_step = 1.0 / (m_lags * dx) * 0.5
    
    step = init_step

        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:

            best_h = h.copy()
            no_improve = 0

        top_idx = np.argpartition(corr, -m_lags)[-m_lags:]

def compute_c5(h):

def project(h, target):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fft_corr(a, b):
    n = len(a)
    size = 2 * n - 1
    fs = 1
    while fs < size:
        fs *= 2
    A = np.fft.rfft(a, fs)
    B = np.fft.rfft(b, fs)
    return np.fft.irfft(np.conj(A) * B, fs)[:size]

def project(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = h < 1.0 - 1e-13 if d > 0 else h > 1e-13
        if not m.any():
            h[:] = np.clip(h + d/n, 0, 1)
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0, 1)
    return h

def build_change(h, n):
    """Build the change matrix: change[p, q] = ∂corr[q]/∂h[p]"""
    n_corr = 2*n - 1
    change = np.zeros((n, n_corr))
    shifts = np.arange(-(n-1), n)
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos] += (1.0 - h[p + shifts[pos]])
    change[p, neg] -= h[p - shifts[neg]]

def pair_cd(h0, n, target, tlimit, max_sweeps=50000, n_cand=50, 
            top_k=50, n_shifts=3, max_no_imp=1000, bisect_it=20):

    n_corr = 2*n - 1

    idxs = np.arange(n)
    mid = n - 1
    
    h = h0.copy()
    bc = compute_c5(h)
    bh = h.copy()
    no_imp = 0
    
    change = build_change(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > tlimit:
            break
        
        corr = fft_corr(h, 1-h)
        c = corr.max() * dx
        
        if c < bc - 1e-14:
            bc, bh, no_imp = c, h.copy(), 0
        else:
            no_imp += 1
            if no_imp > max_no_imp:
                break
        
        k_star = int(np.argmax(corr))
        
        # Gradient from top n_shifts lags
        top = np.argsort(corr)[-n_shifts:]
        g = np.zeros(n)
        for si in top:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v1] += (1.0 - h[idxs[v1] - k])
            g[v2] -= h[idxs[v2] + k]
        g -= g.mean()
        
        si = np.argsort(g)
        low = si[:n_cand]
        high = si[-n_cand:]
        
        # Score matrix: how much does moving from high to low improve corr at k_star?
        ch_h = change[high, k_star]
        ch_l = change[low, k_star]
        scores = ch_l[None, :] - ch_h[:, None]
        
        nl = len(low)
        flat = scores.ravel()
        ne = min(top_k, flat.size)
        if ne <= 0:
            continue
        top_flat = np.argpartition(flat, ne)[:ne]
        
        best_c = c
        best_i = best_j = -1
        best_d = 0.0
        
        for tf in top_flat:
            ii = int(tf // nl)
            jj = int(tf % nl)
            i = int(high[ii])
            j = int(low[jj])
            if i == j:

            dlo = max(h[i] - 1.0, -h[j])
            dhi = min(h[i], 1.0 - h[j])
            if dhi - dlo < 1e-13:
                continue
            
            gd = change[j] - change[i]
            
            a, b = dlo, dhi
            for _ in range(bisect_it):
                mid_d = 0.5*(a+b)
                vals = corr + mid_d * gd

                am = vals >= fm - 1e-13

                if fp > 1e-13:
                    b = mid_d
                else:
                    a = mid_d

            am = (corr + ds*gd).max() * dx

                best_c = am

            h[best_j] += best_d
            h = np.clip(h, 0, 1)
            h = project(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)
    
    return bh, bc
