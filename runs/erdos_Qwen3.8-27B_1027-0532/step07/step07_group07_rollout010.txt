    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)

    n = len(h)

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    val = float(c_max + np.log(s) / alpha)
    if s < 1e-300:
        return val, np.zeros(len(corr)), np.array([], dtype=int)

    active_idx = np.argpartition(weights, -n_active)[-n_active:]
    active_idx = active_idx[weights[active_idx] > 1e-12]
    return val, weights[active_idx], active_idx

    """Compute gradient given pre-computed weights and active indices."""
    mid = n - 1
    idxs = np.arange(n)

        w = weights[i]
        if w < 1e-14:

        k = int(active_indices[i]) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad

                momentum = (t - 1.0) / t_next
                y = h + momentum * (h - h_prev)
            else:

                t_next = 1.5

            if c5_new < best_c5 - 1e-15:

                best_h = h_new.copy()

            # Simple acceptance: accept if true objective improved
            # This is not standard FISTA but works in practice

            h = h_new

            if c5_new < compute_c5(h_prev) - 1e-15:
                eta = min(eta * 1.05, 0.1)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:

            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

# Global state for SLSQP callbacks
_g_alpha = 1.0
_g_target = 0.0

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)

def _slsqp_grad(x):
    n = len(x)
    n_corr = 2 * n - 1
    mid = n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _slsqp_cons(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_alpha, _g_target
    _g_target = target
    bounds = [(0.0, 1.0)] * n

    h_cur = h_start.copy()

                h_cur = h_new

    return h_cur, cur_score
