        top_idx = np.argsort(corr)[-n_shifts:]
        c_max_val = float(corr[np.argmax(corr)])
        temp = max(c_max_val - float(corr[top_idx].min()), 1e-8)
        weights = np.exp((corr[top_idx] - c_max_val) / temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wi, si in enumerate(top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += weights[wi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= weights[wi] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                m_d = 0.5 * (d_lo + d_hi)
                approx = np.max(corr + m_d * g) * dx
                if approx >= best_trial:
                    continue
                a = d_lo
                b = d_hi
                for _ in range(10):
                    m = 0.5 * (a + b)
                    vals = corr + m * g
                    fm = np.max(vals)
                    am = vals >= fm - 1e-14
                    if am.any():
                        fp = np.max(g[am])
                    else:
                        fp = 0.0
                    if fp > 1e-12:
                        b = m
                    else:
                        a = m
                d_star = 0.5 * (a + b)
                if abs(d_star) < 1e-12:
                    continue
                trial = np.max(corr + d_star * g) * dx
                if trial < best_trial:
                    best_trial = trial
                    best_i = i
                    best_j = j
                    best_delta = d_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def sa_escape(h_init, n, target, time_limit, max_steps, t_start, t_end, n_cand, seed=123):
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = get_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 800:
            break

        progress = min(step / max(max_steps, 1), 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        d_lo = max(h[i] - 1.0, -h[j])
        d_hi = min(h[i], 1.0 - h[j])
        if d_hi - d_lo < 1e-12:
            continue

        delta = (d_hi - d_lo) * rng.rand() + d_lo
        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new = np.max(corr + delta * g) * dx
        delta_c = approx_new - c5_val

        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            if approx_new < best_c5 - 1e-14:
                best_c5 = approx_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)

    if tleft() > 60:

                                min(50, tleft() * 0.7), start_time)

    if tleft() > 350:

    if tleft() > 100:

            best_score = c5_ms
            best_h = h_ms

    if tleft() > 80:
        h_sa, c5_sa = sa_escape(best_h, n0, target0,
                                min(60, tleft() * 0.5), 20000,
                                t_start=0.03, t_end=0.0002, n_cand=70,

        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    if tleft() > 300:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        tl2 = min(250, tleft() * 0.6)
        h_cd3, c5_3 = pair_cd_v3(h2, n2, target2, tl2,
                                  max_sweeps=40000, n_cand_min=40, n_cand_max=100,
                                  k_top_min=4, k_top_max=8, max_no_improve=350)
        if c5_3 < best_score:
            best_score = c5_3
            best_h = h_cd3
            n0 = n2
            target0 = target2

            h_pol3, sc3 = slsqp_round(best_h, n0, target0, [2000, 8000], 600, 1e-14,
                                      min(70, tleft() * 0.5), start_time)
            if sc3 < best_score:

    if tleft() > 100:
        h_pol4, sc4 = slsqp_round(best_h, n0, target0, [5000, 15000, 50000], 1000, 1e-14,
                                  min(80, tleft() * 0.7), start_time)

    final_h = project_feasible(best_h, len(best_h) / 2.0)
