```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def alternating_lp(h_init, n, target, max_iters=40, lambda_param=1.0, time_limit=200):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for k in range(-(n-1), n):
            if k >= 0:
                i_indices = np.arange(0, n - k)
                h_old_subset = h[i_indices + k]
            else:
                i_indices = np.arange(-k, n)
                h_old_subset = h[i_indices + k]
            
            if len(i_indices) == 0:
                continue
            
            # constraint: sum(h[i] * h_old[i+k]) * dx >= 1 - t
            constraints.append(cp.sum(h_var[i_indices] * h_old_subset) * dx + t_var >= 1.0)
        
        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        
        if h_var.value is not None and not np.any(np.isnan(h_var.value)):
            h_new = h_var.value
            h_blend = (1.0 - lambda_param) * h + lambda_param * h_new
            h = project_feasible(h_blend, target)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c5
```

```python
def alternating_lp(h_init, n, target, max_iters=40, time_limit=200, top_m=None):
    """
    Alternating LP method for the Erdős overlap problem.
    
    At each iteration, fix h_old and solve:
      min t
      s.t. dx * sum_i h[i]*(1-h_old[i+k]) <= t  for k in selected shifts
           0 <= h[i] <= 1
           sum h[i] = target
    """
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Precompute the shift structure
    shifts_info = []  # list of (k, i_indices)
    for k in range(-(n-1), n):
        if k >= 0:
            i_idx = np.arange(0, n - k)
        else:
            i_idx = np.arange(-k, n)
        if len(i_idx) > 0:
            shifts_info.append((k, i_idx))
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Select shifts to use
        if top_m is not None and top_m < len(shifts_info):
            # Compute current c_k values and select top M
            corr = np.correlate(h, 1.0 - h, mode='full')
            # corr is indexed from 0 to 2n-2, corresponding to shifts -(n-1) to (n-1)
            mid = n - 1
            # Select top M shifts by c_k value
            top_idx = np.argsort(corr)[-top_m:]
            selected_k = set(int(idx - mid) for idx in top_idx)
            use_shifts = [(k, ii) for k, ii in shifts_info if k in selected_k]
        else:
            use_shifts = shifts_info
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        for k, i_idx in use_shifts:
            h_old_subset = h[i_idx + k]
            # coeff = 1 - h_old[i+k]

                break

            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Check for convergence
        if it > 0 and abs(compute_c5(h) - best_c5) < 1e-8:
            # Compute actual improvement
            pass
    
    return best_h, best_c5
```

```python
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_vars = n + 1  # h[0..n-1] + t
    mid = n - 1
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Compute current c_k values
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Select shifts
        if top_m is not None and top_m < 2*n - 1:
            top_idx = np.argsort(corr)[-top_m:]
        else:
            top_idx = np.arange(2*n - 1)

        # Variables: [h[0], ..., h[n-1], t]
        # Objective: minimize t → c = [0, 0, ..., 0, 1]
        c_obj = np.zeros(n_vars)
        c_obj[-1] = 1.0
        
        A_ub_rows = []
        b_ub_vals = []
        
        for idx in top_idx:
            k = int(idx) - mid
            if k >= 0:

            else:

            row = np.zeros(n_vars)

        b_ub = np.array(b_ub_vals)

        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Bounds: 0 <= h[i] <= 1, t >= 0
        bounds = [(0, 1)] * n + [(0, None)]
        
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.success:
            h_new = res.x[:n]

            h = project_feasible(h_new, target)
            c5 = compute_c5(h)

                best_h = h.copy()
```

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)

        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    x_new = np.linspace(0.0, 1.0, n_dst)
