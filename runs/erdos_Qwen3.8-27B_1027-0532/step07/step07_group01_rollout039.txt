```python
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):

    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def line_search_delta(corr, g, delta_lo, delta_hi, n_bisect=20):
    """Find delta in [delta_lo, delta_hi] minimizing max(corr + delta*g)"""
    a_d = delta_lo
    b_d = delta_hi
    for _ in range(n_bisect):
        mid_d = 0.5 * (a_d + b_d)
        vals = corr + mid_d * g
        f_mid = np.max(vals)
        active_mask = vals >= f_mid - 1e-14
        if active_mask.any():
            f_prime = np.max(g[active_mask])
        else:
            f_prime = 0.0
        if f_prime > 1e-12:
            b_d = mid_d
        else:
            a_d = mid_d
    return 0.5 * (a_d + b_d)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha + penalty * (np.sum(x) - target)**2)

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    logits = alpha * (corr - c_max)
    logits = np.clip(logits, -500.0, 0.0)
    w = np.exp(logits)
    s = np.sum(w)

        w = np.zeros_like(logits)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s

    grad = np.zeros(n)

            grad[k:] -= wk * x[:n-k]

            grad[kp:] += wk * (1.0 - x[:n-kp])
            grad[:n-kp] -= wk * x[kp:]

        if time.time() - start > time_limit:

            sc_old = compute_c5(h)
            if sc < sc_old:
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _smooth_grad(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    logits = alpha * (corr - c_max)
    logits = np.clip(logits, -500.0, 0.0)
    w = np.exp(logits)
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(logits)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    n_corr = len(w)
    mid = n - 1
    grad = np.zeros(n)
    for k_off in range(n_corr):
        wk = w[k_off]
        if wk < 1e-15:
            continue
        k = k_off - mid
        if k >= 0:
            grad[:n - k] += wk * (1.0 - x[k:])
            grad[k:] -= wk * x[:n - k]
        else:
            kp = -k
            grad[kp:] += wk * (1.0 - x[:n - kp])
            grad[:n - kp] -= wk * x[kp:]
    grad -= np.mean(grad)
    return grad


def lbfgs_optimize(h_start, n, target, alphas, maxiter_each=1500, time_limit=60.0):
    h = h_start.copy()
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        remaining = time_limit - (time.time() - start)
        if remaining < 5:
            break
        try:
            res = minimize(_smooth_obj, h, args=(alpha,),
                           jac=_smooth_grad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            sc_old = compute_c5(h)
            if sc < sc_old:
                h = h_new
        except Exception:
            pass
    return h, compute_c5(h)


def _slsqp_cons_fun(x):
    return np.sum(x) - _lbfgs_target


def _slsqp_cons_jac(x):
    return np.ones(_lbfgs_n)


_lbfgs_n = 0
_lbfgs_target = 0.0


def slsqp_optimize(h_start, n, target, alphas, maxiter_each=500, time_limit=60.0):
    global _lbfgs_n, _lbfgs_target
    _lbfgs_n = n
    _lbfgs_target = target
    h = h_start.copy()
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    for alpha in alphas:

        try:
            res = minimize(_smooth_obj, h, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,

            sc_old = compute_c5(h)
            if sc < sc_old:

        except Exception:
            pass

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
            n_cand_start=30, n_cand_end=70, max_no_improve=500,
            n_bisect=22, n_top_try=25, restart_scale=0.015):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(777)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0
            continue

        max_corr_val = float(np.max(corr))
        active_set = np.where(corr >= max_corr_val - 1e-10)[0]

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        if k_star >= 0:
            grad[:n - k_star] += (1.0 - h[k_star:])
            grad[k_star:] -= h[:n - k_star]
        else:
            kp = -k_star
            grad[kp:] += (1.0 - h[:n - kp])
            grad[:n - kp] -= h[kp:]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        candidates = []
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            hi_val = h[i]
            if hi_val < 1e-12:
                continue
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_max = min(hi_val, 1.0 - h[j])
                if delta_max < 1e-12:
                    continue
                g_active = change[j, active_set] - change[i, active_set]
                fo = -np.max(g_active) * delta_max
                if fo > 1e-12:
                    candidates.append((fo, i, j, delta_max))

        candidates.sort(key=lambda x: x[0], reverse=True)
        best_trial_c5 = c5_val
        best_move = None
        n_to_try = min(n_top_try, len(candidates))

        for ci in range(n_to_try):
            fo, i, j, delta_max = candidates[ci]
            g = change[j] - change[i]
            a_d = 0.0
            b_d = delta_max
            for _ in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_m = vals >= f_mid - 1e-14
                if active_m.any():
                    f_prime = np.max(g[active_m])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
        else:
            consecutive_no_improve += 1

    return best_h, best_c5


def multi_shift_pair_cd(h_init, n, target, time_limit, max_sweeps=15000,
                        n_cand=40, n_shifts=5, n_bisect=20):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(999)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 300:
            h = best_h + rng.randn(n) * 0.01
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0
            continue

        top_shifts = np.argsort(corr)[-n_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            if k >= 0:
                grad[:n - k] += w * (1.0 - h[k:])
                grad[k:] -= w * h[:n - k]
            else:
                kp = -k
                grad[kp:] += w * (1.0 - h[:n - kp])
                grad[:n - kp] -= w * h[kp:]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        max_corr_val = float(np.max(corr))
        active_set = np.where(corr >= max_corr_val - 1e-8)[0]

        candidates = []
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            hi_val = h[i]
            if hi_val < 1e-12:
                continue
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_max = min(hi_val, 1.0 - h[j])
                if delta_max < 1e-12:
                    continue
                g_active = change[j, active_set] - change[i, active_set]
                fo = -np.max(g_active) * delta_max
                if fo > 1e-12:
                    candidates.append((fo, i, j, delta_max))

        candidates.sort(key=lambda x: x[0], reverse=True)
        best_trial_c5 = c5_val
        best_move = None
        n_to_try = min(20, len(candidates))

        for ci in range(n_to_try):
            fo, i, j, delta_max = candidates[ci]
            g = change[j] - change[i]
            a_d = 0.0
            b_d = delta_max
            for _ in range(n_bisect):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_m = vals >= f_mid - 1e-14
                if active_m.any():
                    f_prime = np.max(g[active_m])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def time_left():
        return budget_s - 18.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: L-BFGS-B at initial resolution
    if time_left() > 100:
        tl = min(80, time_left() - 10)
        h1, sc1 = lbfgs_optimize(best_h, current_n, current_target,
                                  alphas=[100, 500, 2000, 8000, 30000, 100000],
                                  maxiter_each=2000, time_limit=tl)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: SLSQP polish at initial resolution
    if time_left() > 80:
        tl = min(60, time_left() - 10)
        h2, sc2 = slsqp_optimize(best_h, current_n, current_target,
                                  alphas=[10000, 50000, 200000, 1000000],
                                  maxiter_each=600, time_limit=tl)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Pair CD at initial resolution
    if time_left() > 120:
        tl = min(100, time_left() - 10)
        h3, sc3 = pair_cd(best_h, current_n, current_target,
                           time_limit=tl, max_sweeps=25000,
                           n_cand_start=25, n_cand_end=70,
                           max_no_improve=400, n_bisect=22, n_top_try=30)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: Upsample to 2x resolution and optimize
    n2 = current_n * 2
    target2 = n2 / 2.0
    if time_left() > 350:
        h2 = interpolate_to(best_h, n2, target2)
        sc_int = compute_c5(h2)

        if time_left() > 280:
            tl = min(90, time_left() - 30)
            h2b, sc2b = lbfgs_optimize(h2, n2, target2,
                                        alphas=[200, 1000, 5000, 20000, 80000, 300000],
                                        maxiter_each=1500, time_limit=tl)
            if sc2b < best_score:
                best_score = sc2b
                best_h = h2b
                current_n = n2
                current_target = target2

        if time_left() > 200 and current_n == n2:
            tl = min(60, time_left() - 20)
            h2c, sc2c = slsqp_optimize(best_h, current_n, current_target,
                                        alphas=[20000, 100000, 500000, 2000000],
                                        maxiter_each=500, time_limit=tl)
            if sc2c < best_score:
                best_score = sc2c
                best_h = h2c

        if time_left() > 200 and current_n == n2:
            tl = min(170, time_left() - 20)
            h2d, sc2d = pair_cd(best_h, current_n, current_target,
                                 time_limit=tl, max_sweeps=20000,
                                 n_cand_start=30, n_cand_end=65,
                                 max_no_improve=350, n_bisect=22, n_top_try=25)
            if sc2d < best_score:
                best_score = sc2d
                best_h = h2d

        if time_left() > 100 and current_n == n2:
            tl = min(70, time_left() - 15)
            h2e, sc2e = lbfgs_optimize(best_h, current_n, current_target,
                                        alphas=[50000, 200000, 1000000],
                                        maxiter_each=1500, time_limit=tl)
            if sc2e < best_score:
                best_score = sc2e
                best_h = h2e

    # Phase 5: Multi-shift pair CD
    if time_left() > 120 and current_n == n2:
        tl = min(90, time_left() - 15)
        h5, sc5 = multi_shift_pair_cd(best_h, current_n, current_target,
                                       time_limit=tl, max_sweeps=12000,
                                       n_cand=35, n_shifts=5, n_bisect=20)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: Try upsampling to 4x if time allows
    n3 = current_n * 2
    target3 = n3 / 2.0
    if time_left() > 350 and current_n == n2:
        h3 = interpolate_to(best_h, n3, target3)
        if time_left() > 280:
            tl = min(80, time_left() - 30)
            h3b, sc3b = lbfgs_optimize(h3, n3, target3,
                                        alphas=[500, 2000, 10000, 50000, 200000],
                                        maxiter_each=1000, time_limit=tl)
            if sc3b < best_score:
                best_score = sc3b
                best_h = h3b
                current_n = n3
                current_target = target3

        if time_left() > 180 and current_n == n3:
            tl = min(150, time_left() - 20)
            h3c, sc3c = pair_cd(best_h, current_n, current_target,
                                 time_limit=tl, max_sweeps=12000,
                                 n_cand_start=25, n_cand_end=55,
                                 max_no_improve=300, n_bisect=20, n_top_try=20)
            if sc3c < best_score:
                best_score = sc3c
                best_h = h3c

        if time_left() > 80 and current_n == n3:
            tl = min(60, time_left() - 15)
            h3d, sc3d = lbfgs_optimize(best_h, current_n, current_target,
                                        alphas=[100000, 500000, 2000000],
                                        maxiter_each=1000, time_limit=tl)
            if sc3d < best_score:
                best_score = sc3d
                best_h = h3d

    # Phase 7: Randomized restarts at current resolution
    if time_left() > 150:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.008, 0.02, 0.05]:
            if time_left() < 80:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            tl = min(40, time_left() - 30)
            h_ref, sc_ref = lbfgs_optimize(h_pert, current_n, current_target,
                                            alphas=[500, 2000, 10000, 50000],
                                            maxiter_each=800, time_limit=tl)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if time_left() > 60:
                tl2 = min(35, time_left() - 20)
                h_cd_p, sc_cd_p = pair_cd(h_ref, current_n, current_target,
                                           time_limit=tl2, max_sweeps=8000,
                                           n_cand_start=25, n_cand_end=60,
                                           max_no_improve=250, n_bisect=20, n_top_try=20)
                if sc_cd_p < best_score:
                    best_score = sc_cd_p
                    best_h = h_cd_p

    # Phase 8: Final pair CD
    if time_left() > 50:
        tl = min(35, time_left() - 8)
        h8, sc8 = pair_cd(best_h, current_n, current_target,
                           time_limit=tl, max_sweeps=10000,
                           n_cand_start=30, n_cand_end=80,
                           max_no_improve=200, n_bisect=24, n_top_try=30)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: Final L-BFGS-B polish
    if time_left() > 25:
        tl = min(18, time_left() - 5)
        h9, sc9 = lbfgs_optimize(best_h, current_n, current_target,
                                  alphas=[200000, 1000000, 5000000],
                                  maxiter_each=2000, time_limit=tl)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
