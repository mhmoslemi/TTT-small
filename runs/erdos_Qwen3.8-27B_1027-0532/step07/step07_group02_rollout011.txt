```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_all(h, n):
    """Compute full change matrix."""
    shifts = np.arange(-(n-1), n)
    n_corr = 2*n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    res[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    res[neg_mask] -= h[p - shifts[neg_mask]]
    return res

def opt_single_delta(corr, ch_row, h_val):
    """Find optimal delta for single coordinate. h[i] -> h[i] + delta.
    Minimize max_k (corr[k] + delta * ch_row[k])
    s.t. -h_val <= delta <= 1 - h_val.

    dhi = 1.0 - h_val

        return 0.0

        hi_s = np.max(ch_row[active0])

        elif hi_s < -1e-12:

        else:

    g = ch_j - ch_i
```

The Erdős problem asks: what's the minimum possible value of max_k ∫ h(x)(1-h(x+k)) dx?

For h being a uniform distribution on [0,1] (h=1 on [0,1], 0 on [1,2]):
- For k=0: ∫₀¹ 1*(1-1) dx + ∫₁² 0*(1-0) dx = 0
- For k=1: ∫₀¹ 1*(1-h(x+1)) dx = ∫₀¹ 1*1 dx = 1 (since h(x+1)=0 for x∈[0,1])
  Wait, h is defined on [0,2]. h(x+1) for x∈[0,1] is h on [1,2] which is 0.
  So ∫₀¹ 1*(1-0) dx = 1.
  But we also have ∫₁² 0*(1-h(x+1)) dx = 0.
  Total: 1. That's bad.

Actually, the integral is over the domain where both h(x) and h(x+k) are defined. For h supported on [0,1]:
- h(x) is 1 on [0,1], 0 on [1,2]
- h(x+k) is 1 on [-k, 1-k], 0 on [1-k, 2-k] (shifted)

The overlap ∫ h(x)(1-h(x+k)) dx is over x where both are defined, i.e., x ∈ [0,2] ∩ [-k, 2-k].

For k > 0: x ∈ [0, 2-k]
For k < 0: x ∈ [-k, 2]

This is getting complicated. The discrete version handles this automatically through the correlation.

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_row(p, h, n, shifts):
    n_corr = 2 * n - 1
    res = np.zeros(n_corr)
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    res[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    res[neg_mask] -= h[p - shifts[neg_mask]]
    return res

def compute_change_all(h, n):
    shifts = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        change[p] = compute_change_row(p, h, n, shifts)
    return change

def opt_single_delta(corr, ch_row, h_val, n_bisect=45):
    dlo = -h_val
    dhi = 1.0 - h_val
    if dhi - dlo < 1e-14:
        return 0.0
    for _ in range(n_bisect):
        mid = 0.5 * (dlo + dhi)
        vals = corr + mid * ch_row
        fm = np.max(vals)
        active = vals >= fm - 1e-13
        if active.any():
            lo_s = np.min(ch_row[active])
            hi_s = np.max(ch_row[active])
        else:
            lo_s = hi_s = 0.0
        if lo_s > 1e-12:
            dhi = mid
        elif hi_s < -1e-12:
            dlo = mid
        else:
            return mid
    return 0.5 * (dlo + dhi)

def opt_pair_delta(corr, ch_i, ch_j, h_i, h_j, n_bisect=45):
    g = ch_j - ch_i
    dlo = max(h_i - 1.0, -h_j)
    dhi = min(h_i, 1.0 - h_j)
    if dhi - dlo < 1e-14:
        return 0.0
    for _ in range(n_bisect):
        mid = 0.5 * (dlo + dhi)
        vals = corr + mid * g
        fm = np.max(vals)
        active = vals >= fm - 1e-13
        if active.any():
            lo_s = np.min(g[active])
            hi_s = np.max(g[active])
        else:
            lo_s = hi_s = 0.0
        if lo_s > 1e-12:
            dhi = mid
        elif hi_s < -1e-12:
            dlo = mid
        else:
            return mid
    return 0.5 * (dlo + dhi)

def soft_gradient(h, corr, alpha):
    n = len(h)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        weights = np.zeros(n_corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    grad = np.zeros(n)
    for s in range(n_corr):
        w = weights[s]
        if w < 1e-10:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    grad -= np.mean(grad)
    return grad

def single_coord_cd(h_init, n, target, time_limit, seed=42,
                    n_bisect=45, recompute_freq=80, use_soft_order=True):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1
    rng = np.random.RandomState(seed)

    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()

    no_improve = 0
    pass_count = 0

    while time.time() - start < time_limit:
        change = compute_change_all(h, n)
        if use_soft_order:
            sg = soft_gradient(h, corr, 30.0)
            order = np.argsort(sg)
            flat = (sg.max() - sg.min()) < 1e-10
            if flat:
                order = rng.permutation(n)
        else:
            order = rng.permutation(n)

        pass_improved = False
        coords_done = 0
        for i in order:
            if time.time() - start > time_limit:
                break
            delta = opt_single_delta(corr, change[i], h[i], n_bisect)
            if abs(delta) > 1e-11:
                h[i] += delta
                h[i] = min(max(h[i], 0.0), 1.0)
                actual_delta = h[i] - (h[i] - delta)
                corr += actual_delta * change[i]
                pass_improved = True
            coords_done += 1
            if coords_done % recompute_freq == 0:
                change = compute_change_all(h, n)

        pass_count += 1
        if pass_improved:
            cur_c5 = float(np.max(corr)) * dx
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if pass_count % 20 == 0:
            corr_v = np.correlate(h, 1.0 - h, mode='full')
            if np.max(np.abs(corr - corr_v)) > 1e-6:
                corr = corr_v
            cur_c5 = float(np.max(corr)) * dx
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0

        if no_improve > 50:
            break

    h = project_feasible(h, target)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, seed=42,
            n_cand_start=50, n_cand_end=130, max_sweeps=30000,
            n_bisect=35, stochastic_frac=0.05, alpha_soft=20.0,
            max_no_improve=250):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)

    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    change = compute_change_all(h, n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break

        sg = soft_gradient(h, corr, alpha_soft)
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        cur_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        cur_ncand = min(cur_ncand, n // 2)
        sorted_idx = np.argsort(sg)
        low_set = sorted_idx[:cur_ncand]
        high_set = sorted_idx[-cur_ncand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        n_pairs = n_h * n_l
        n_random = max(1, int(stochastic_frac * n_pairs))

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta = opt_pair_delta(corr, change[i], change[j], h[i], h[j], n_bisect)
                if abs(delta) < 1e-11:
                    continue
                approx_max = np.max(corr + delta * (change[j] - change[i])) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta

        for _ in range(n_random):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            delta = opt_pair_delta(corr, change[i], change[j], h[i], h[j], n_bisect)
            if abs(delta) < 1e-11:
                continue
            approx_max = np.max(corr + delta * (change[j] - change[i])) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in (best_i, best_j):
                change[p] = compute_change_row(p, h, n, shifts)
            for p in range(n):
                if p != best_i and p != best_j:
                    ka = (n - 1) + (best_i - p)
                    kb = (n - 1) + (best_j - p)
                    if 0 <= ka < n_corr:
                        change[p][ka] -= best_delta
                    if 0 <= kb < n_corr:
                        change[p][kb] += best_delta
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_restart(h_best, n, target, time_limit, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    scales = [0.02, 0.05, 0.1, 0.15, 0.25]
    while time.time() - start < time_limit:
        if time_limit - (time.time() - start) < 40:
            break
        scale = scales[rng.randint(len(scales))]
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        remaining = time_limit - (time.time() - start)
        h_cd, c5_cd = single_coord_cd(h_pert, n, target,
                                       time_limit=min(remaining * 0.4, 60),
                                       seed=int(rng.randint(100000)))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    return best_h, best_c5
```
