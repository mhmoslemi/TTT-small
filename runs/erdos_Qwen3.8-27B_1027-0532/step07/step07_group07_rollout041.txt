    n = len(h)

    sms = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    penalty = mu * (np.sum(x) - target_sum) ** 2

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def lse_grad(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    if s < 1e-300:
        return np.zeros(n)
    w = exp_s / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        wi = w[k_idx]
        if wi < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - x[idxs[v1] - k])
        grad[v2] -= wi * x[idxs[v2] + k]
    return grad


def _obj_fn(x):
    return lse_obj(x, _g_alpha)


def _grad_fn(x):
    return lse_grad(x, _g_alpha)


def _cons_fn(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_alpha, _g_target
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fn}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_fn, h_cur, jac=_grad_fn,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def lbfgsb_penalty(h_start, n, target, alphas, mu_schedule, maxiter=2000):
    h_cur = h_start.copy().astype(float)
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha, mu in zip(alphas, mu_schedule):
        def obj_pen(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
            val += mu * (np.sum(x) - target) ** 2
            return float(val)

        def grad_pen(x):
            g = lse_grad(x, alpha)
            g += 2.0 * mu * (np.sum(x) - target)
            return g

        try:
            res = minimize(obj_pen, h_cur, jac=grad_pen,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'gtol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def subgradient_polish(h_start, n, target, time_limit, n_iter=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.02
    idxs = np.arange(n)
    n_corr = 2 * n - 1
    mid = n - 1

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        k_star = int(np.argmax(corr))
        k = k_star - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        g = np.zeros(n)
        g[v1] += (1.0 - h[idxs[v1] - k])
        g[v2] -= h[idxs[v2] + k]
        g -= np.mean(g)

        norm_g = np.linalg.norm(g)
        if norm_g < 1e-14:
            break

        h_new = h - step * g
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < c5_val:
            h = h_new
        else:
            step *= 0.95
            if step < 1e-8:
                step = 0.01
        if it % 200 == 0:
            step = min(step * 1.05, 0.05)

    return best_h, best_c5


def focused_cd(h_init, n, target, time_limit, max_sweeps=30000,
               n_cand=50, top_k=40, max_no_improve=500):
    start = time.time()

    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)

        grad[v2] -= h[idxs[v2] + k_star]

        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_l = change[low_set, k_star_idx]
        score_mat = ch_l[None, :] - ch_h[:, None]

        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_i = -1
        best_j = -1
        best_delta = 0.0

            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            g = change[j] - change[i]

                vals = corr + mid_d * g
                f_mid = np.max(vals)

                f_prime = np.max(g[active])
                if f_prime > 1e-13:

                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h_fin = slsqp_optimize(best_h, n0, target0,
                               [100000, 500000, 1000000],
                               600, 1e-14)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
