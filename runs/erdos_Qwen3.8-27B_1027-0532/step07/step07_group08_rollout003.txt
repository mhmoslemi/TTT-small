            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            diff_sum = target - np.sum(h)
            if abs(diff_sum) > 1e-14:
                h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            no_move_count = 0
        else:
            no_move_count += 1
            if no_move_count > 200:
                break

    return best_h, best_c5


def subgradient_method(h_init, n, target, time_limit, alpha0=0.5, alpha_min=0.002):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    stale = 0

    for step in range(300000):
        if time.time() - start > time_limit:
            break
        if alpha < alpha_min:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
            alpha = min(alpha * 1.15, alpha0)
        else:
            stale += 1
            alpha *= 0.97
            if stale > 800:
                break

        c_max = float(np.max(corr))
        threshold = c_max - 0.08 * max(c_max, 0.01)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        n_act = len(active)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = 1.0 / n_act
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def downsample_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: SLSQP at n=400 with wide alpha range
    if tleft() > 60:
        deadline = time.time() + min(55, tleft() * 0.45)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [50, 200, 1000, 5000, 20000],
                                800, 1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Very high alpha SLSQP
    if tleft() > 50:
        deadline = time.time() + min(40, tleft() * 0.4)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [50000, 100000, 200000],
                                1000, 1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 3: CD at n=400
    if tleft() > 200:
        h_cd, c5_cd = cd_optimized(best_h, n0, target0,
                                   time_limit=min(150, tleft() * 0.55),
                                   n_cand_start=12, n_cand_end=70,
                                   n_active_shifts=2, max_sweeps=50000,
                                   max_no_improve=400, reheat_interval=120,
                                   reheat_scale=0.03)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: SLSQP polish after CD
    if tleft() > 60:
        deadline = time.time() + min(45, tleft() * 0.5)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [10000, 50000, 200000],
                                1200, 1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Subgradient at n=400
    if tleft() > 80:
        h_sg, c5_sg = subgradient_method(best_h, n0, target0,
                                         time_limit=min(50, tleft() * 0.5),
                                         alpha0=0.3, alpha_min=0.003)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: More CD at n=400
    if tleft() > 150:
        h_cd, c5_cd = cd_optimized(best_h, n0, target0,
                                   time_limit=min(100, tleft() * 0.55),
                                   n_cand_start=20, n_cand_end=90,
                                   n_active_shifts=3, max_sweeps=40000,
                                   max_no_improve=500, reheat_interval=100,
                                   reheat_scale=0.02)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Interpolate to n=800
    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 200:
            h_cd1, c5_cd1 = cd_optimized(h1, n1, target1,
                                         time_limit=min(150, tleft() * 0.5),
                                         n_cand_start=15, n_cand_end=60,
                                         n_active_shifts=2, max_sweeps=30000,
                                         max_no_improve=350, reheat_interval=100,
                                         reheat_scale=0.03)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                current_n = n1
                current_target = target1

        if tleft() > 80:
            deadline = time.time() + min(60, tleft() * 0.45)
            h_pol1, sc1 = slsqp_round(best_h, current_n, current_target,
                                      [10000, 50000, 150000],
                                      1000, 1e-15, deadline=deadline)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1

        # Downsample back to n=400 if we went to 800
        if current_n == n1 and tleft() > 50:
            h_back = downsample_to(best_h, n0, target0)
            sc_back = compute_c5(h_back)
            if sc_back < best_score:
                best_score = sc_back
                best_h = h_back
                current_n = n0
                current_target = target0

    # Phase 8: Multi-start perturbation
    if tleft() > 150:
        for attempt in range(3):
            if tleft() < 50:
                break
            noise_scale = [0.02, 0.05, 0.1][attempt]
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(30, tleft() * 0.3)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [10000, 50000, 200000],
                                        600, 1e-15, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 9: Final CD
    if tleft() > 100:
        h_cd_f, c5_f = cd_optimized(best_h, current_n, current_target,
                                    time_limit=min(60, tleft() * 0.55),
                                    n_cand_start=25, n_cand_end=100,
                                    n_active_shifts=1, max_sweeps=30000,
                                    max_no_improve=250, reheat_interval=80,
                                    reheat_scale=0.015)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    # Phase 10: Final SLSQP
    if tleft() > 35:
        deadline = time.time() + min(25, tleft() * 0.6)
        h_final, sc_final = slsqp_round(best_h, current_n, current_target,
                                        [50000, 150000, 500000],
                                        1000, 1e-15, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Ensure we return at the original n if possible, or best n
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    if len(final_h) != n0 and tleft() > 5:
        h_at_n0 = downsample_to(final_h, n0, target0)
        sc_at_n0 = compute_c5(h_at_n0)
        if sc_at_n0 <= final_score + 1e-10:
            final_h = h_at_n0
            final_score = sc_at_n0

    return final_h, final_score, len(final_h)
