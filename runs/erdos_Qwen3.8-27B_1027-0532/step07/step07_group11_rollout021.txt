    pos_mask = (idxs + shifts >= 0) & (idxs + shifts < n)
    neg_mask = (idxs - shifts >= 0) & (idxs - shifts < n)
    change[p, pos_mask] += (1.0 - h[idxs[pos_mask] + shifts[pos_mask]])
    change[p, neg_mask] -= h[idxs[neg_mask] - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
            n_cand_start=40, n_cand_end=100, n_shifts=3,
            max_no_improve=400, anneal_temp=0.01, anneal_decay=0.9995,
            reheat_every=200, reheat_factor=1.5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if sweep > 0 and sweep % reheat_every == 0 and consecutive_no_improve > reheat_every // 2:
            temp = max(temp * reheat_factor, 0.005)

        top_shifts = np.argsort(corr)[-n_shifts:]
        sw_sum = np.sum(corr[top_shifts])
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = corr[top_shifts] / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = float(np.max(g[active_mask]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 2000.0
                    if np.random.random() < np.exp(-min(delta_obj, 500.0) / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts, idxs)
            update_change_row(change, best_j, h, n, shifts, idxs)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 2:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.97, 1e-10)

    return best_h, best_c5


def subgradient_multi(h_init, n, target, time_limit, n_steps=20000, n_weights=4, step_scale=0.004):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        p = np.exp(corr * 60.0)
        p /= p.sum()

        grad = np.zeros(n)
        top_k = np.argsort(p)[-n_weights:]
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
            eta = min(eta * 1.05, step_scale * 2)
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.9

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    try:
        if tleft() > 120:
            h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                    [100, 500, 2000],
                                    800, 1e-13)
            if sc < best_score:
                best_score = sc
                best_h = h_pol
    except Exception:
        pass

    if tleft() > 300:
        try:
            h_cd, c5_cd = pair_cd(best_h, cur_n, cur_target,
                                  time_limit=min(250, tleft() * 0.5),
                                  max_sweeps=50000,
                                  n_cand_start=40, n_cand_end=110,
                                  n_shifts=4, max_no_improve=500,
                                  anneal_temp=0.015, anneal_decay=0.9995,
                                  reheat_every=150, reheat_factor=1.8)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
        except Exception:
            pass

    if tleft() > 100:
        try:
            h_sg, c5_sg = subgradient_multi(best_h, cur_n, cur_target,
                                            time_limit=min(80, tleft() * 0.5),
                                            n_steps=25000, n_weights=4,
                                            step_scale=0.005)
            if c5_sg < best_score:
                best_score = c5_sg
                best_h = h_sg
        except Exception:
            pass

    if tleft() > 80:
        try:
            h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                    [5000, 15000],
                                    1200, 1e-13)
            if sc < best_score:
                best_score = sc
                best_h = h_pol
        except Exception:
            pass

    if tleft() > 400:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if tleft() > 250:
            try:
                h_cd1, c5_cd1 = pair_cd(h1, n1, target1,
                                         time_limit=min(250, tleft() * 0.55),
                                         max_sweeps=40000,
                                         n_cand_start=30, n_cand_end=90,
                                         n_shifts=3, max_no_improve=400,
                                         anneal_temp=0.01, anneal_decay=0.9997,
                                         reheat_every=200, reheat_factor=1.5)
                if c5_cd1 < best_score:
                    best_score = c5_cd1
                    best_h = h_cd1
                    cur_n = n1
                    cur_target = target1
                elif sc1 < best_score:
                    best_score = sc1
                    best_h = h1
                    cur_n = n1
                    cur_target = target1
            except Exception:
                pass

    if tleft() > 100:
        try:
            h_sg2, c5_sg2 = subgradient_multi(best_h, cur_n, cur_target,
                                              time_limit=min(70, tleft() * 0.5),
                                              n_steps=20000, n_weights=5,
                                              step_scale=0.003)
            if c5_sg2 < best_score:
                best_score = c5_sg2
                best_h = h_sg2
        except Exception:
            pass

    if tleft() > 100:
        try:
            h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                    [10000, 30000],
                                    2000, 1e-13)
            if sc < best_score:
                best_score = sc
                best_h = h_pol
        except Exception:
            pass

    if tleft() > 350:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if tleft() > 250:
            try:
                h_cd2, c5_cd2 = pair_cd(h2, n2, target2,
                                         time_limit=min(220, tleft() * 0.55),
                                         max_sweeps=30000,
                                         n_cand_start=25, n_cand_end=70,
                                         n_shifts=2, max_no_improve=350,
                                         anneal_temp=0.005, anneal_decay=0.9998,
                                         reheat_every=150, reheat_factor=1.4)
                if c5_cd2 < best_score:
                    best_score = c5_cd2
                    best_h = h_cd2
                    cur_n = n2
                    cur_target = target2
                elif sc2 < best_score:
                    best_score = sc2
                    best_h = h2
                    cur_n = n2
                    cur_target = target2
            except Exception:
                pass

    if tleft() > 60:
        try:
            h_cd3, c5_cd3 = pair_cd(best_h, cur_n, cur_target,
                                     time_limit=min(50, tleft() * 0.6),
                                     max_sweeps=15000,
                                     n_cand_start=60, n_cand_end=140,
                                     n_shifts=5, max_no_improve=200,
                                     anneal_temp=0.003, anneal_decay=0.999,
                                     reheat_every=100, reheat_factor=1.3)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
        except Exception:
            pass

    if tleft() > 50:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.06, 0.1, 0.15]:
            if tleft() < 35:
                break
            h_pert = best_h + rng.randn(cur_n) * noise_scale
            h_pert = project_feasible(h_pert, cur_target)
            try:
                h_ref, sc = slsqp_round(h_pert, cur_n, cur_target,
                                        [500, 2000, 8000],
                                        600, 1e-13)
                if sc < best_score:
                    best_score = sc
                    best_h = h_ref
            except Exception:
                pass

    if tleft() > 50:
        try:
            h_final_cd, c5_final = pair_cd(best_h, cur_n, cur_target,
                                            time_limit=min(35, tleft() * 0.7),
                                            max_sweeps=12000,
                                            n_cand_start=80, n_cand_end=160,
                                            n_shifts=5, max_no_improve=150,
                                            anneal_temp=0.0, anneal_decay=1.0,
                                            reheat_every=9999, reheat_factor=1.0)
            if c5_final < best_score:
                best_score = c5_final
                best_h = h_final_cd
        except Exception:
            pass

    if tleft() > 30:
        try:
            h_final_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                          [20000, 50000, 100000],
                                          1500, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_final_pol
        except Exception:
            pass

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
