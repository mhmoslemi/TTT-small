import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def subgradient_at_max(h, corr, top_k=5):
    """Compute a subgradient of max_k f_k at h, using top k shifts."""
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    
    # Get top k shifts by correlation value
    top_shifts = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for s_idx in top_shifts:
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    
    # Weight by correlation values (softmax-like)
    # Actually, for subgradient, we can use any convex combination
    # Let's use uniform weight for simplicity (or weighted by corr values)
    # Actually the above already sums over top k, which is like uniform weight
    grad /= float(len(top_shifts))
    return grad

def projected_subgradient_descent(h_init, n, target, time_limit, top_k=5, 
                                    max_iter=50000, ftol=1e-16):
    """Projected subgradient descent with backtracking line search."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    # Initialize step size
    step = 0.1
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient
        grad = subgradient_at_max(h, corr, top_k)
        
        # Remove mean to maintain sum constraint
        grad -= np.mean(grad)
        
        # Line search (backtracking)
        alpha = step
        found = False
        for _ in range(20):
            h_try = h - alpha * grad
            h_try = project_feasible(h_try, target)
            new_c5 = compute_c5(h_try)
            if new_c5 <= c5_val - 1e-16:
                h = h_try
                cur_c5 = new_c5
                found = True
                step = min(step * 1.2, 1.0)
                break
            alpha *= 0.7
        
        if not found:
            step *= 0.5
            if step < 1e-10:
                break
    
    return best_h, best_c5

import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _obj_func(x):
    return smooth_obj(x, _g_alpha)


def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=80, top_k_eval=60,
            max_no_improve=800, n_shifts_grad=3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        total_w = 0.0
        for s_idx in top_shifts:
            w = corr[s_idx]
            total_w += w
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        if total_w > 0:
            grad /= total_w
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5 - 1e-16:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5

    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    return best_h, best_c5


def subgradient_step(h_init, n, target, time_limit, top_k=5,
                     momentum=0.7, max_iter=20000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    step = 0.05
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_shifts = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-12:
            break

        alpha = step / (1.0 + it * 0.001)
        v = momentum * v - alpha * grad
        h_try = h + v
        h_try = project_feasible(h_try, target)
        new_c5 = compute_c5(h_try)

        if new_c5 <= c5_val + 1e-15:
            h = h_try
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.8
            if step < 1e-10:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def tikhonov_refine(h_init, n, target, time_limit, max_iter=5000,
                    gamma=0.1):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.02
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 1000:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        active_threshold = np.max(corr) - 0.02 * np.max(corr)
        active_shifts = np.where(corr >= active_threshold)[0]

        if len(active_shifts) == 0:
            active_shifts = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        for s_idx in active_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(active_shifts))
        grad -= np.mean(grad)

        # Tikhonov regularization: penalize large steps
        smooth_grad = np.convolve(grad, np.ones(5) / 5.0, mode='same')
        grad = (1.0 - gamma) * grad + gamma * smooth_grad
        grad -= np.mean(grad)

        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-12:
            break

        alpha = step
        for _ in range(8):
            h_try = h - alpha * grad
            h_try = project_feasible(h_try, target)
            new_c5 = compute_c5(h_try)
            if new_c5 <= c5_val + 1e-15:
                h = h_try
                step = min(step * 1.1, 0.3)
                break
            alpha *= 0.6

        else:
            step *= 0.7
            if step < 1e-11:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def _time_left():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP polish at initial resolution
    tl = _time_left()
    if tl > 60:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [100, 500, 2000],
                               400, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Go to n=800
    tl = _time_left()
    if tl > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            cur_n = n1
            cur_target = target1
        else:
            cur_n = n1
            cur_target = target1
            best_h = h1
            best_score = sc1

        # Multi-pair CD at n=800
        tl = _time_left()
        if tl > 150:
            h_mp, c5_mp = multi_pair_cd(best_h, cur_n, cur_target,
                                        time_limit=min(130, tl * 0.7),
                                        max_sweeps=40000, n_cand=50,
                                        top_k_eval=100, n_pairs_per_sweep=3,
                                        max_no_improve=600, n_shifts_grad=3)
            if c5_mp < best_score:
                best_score = c5_mp
                best_h = h_mp

        # Single-pair CD for fine-tuning
        tl = _time_left()
        if tl > 100:
            h_cd, c5_cd = pair_cd(best_h, cur_n, cur_target,
                                  time_limit=min(90, tl * 0.75),
                                  max_sweeps=50000,
                                  n_cand_start=30, n_cand_end=70,
                                  top_k_eval=60, max_no_improve=800,
                                  n_shifts_grad=3)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        # SLSQP polish
        tl = _time_left()
        if tl > 50:
            h_pol = slsqp_optimize(best_h, cur_n, cur_target,
                                   [500, 2000, 8000],
                                   800, 1e-15)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 3: Go to n=1200
    tl = _time_left()
    if tl > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        cur_n = n2
        cur_target = target2
        sc2 = compute_c5(h2)
        best_h = h2
        best_score = sc2

        # Pair CD
        tl = _time_left()
        if tl > 150:
            h_cd, c5_cd = pair_cd(best_h, cur_n, cur_target,
                                  time_limit=min(130, tl * 0.75),
                                  max_sweeps=60000,
                                  n_cand_start=25, n_cand_end=70,
                                  top_k_eval=60, max_no_improve=800,
                                  n_shifts_grad=3)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        # Multi-pair CD
        tl = _time_left()
        if tl > 100:
            h_mp, c5_mp = multi_pair_cd(best_h, cur_n, cur_target,
                                        time_limit=min(80, tl * 0.75),
                                        max_sweeps=25000, n_cand=50,
                                        top_k_eval=80, n_pairs_per_sweep=3,
                                        max_no_improve=500, n_shifts_grad=3)
            if c5_mp < best_score:
                best_score = c5_mp
                best_h = h_mp

        # Tikhonov refine
        tl = _time_left()
        if tl > 60:
            h_tr, c5_tr = tikhonov_refine(best_h, cur_n, cur_target,
                                          time_limit=min(50, tl * 0.7),
                                          max_iter=3000, gamma=0.3)
            if c5_tr < best_score:
                best_score = c5_tr
                best_h = h_tr

        # SLSQP polish
        tl = _time_left()
        if tl > 50:
            h_pol = slsqp_optimize(best_h, cur_n, cur_target,
                                   [1000, 5000, 20000],
                                   1000, 1e-15)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 4: Try n=1600 if time permits
    tl = _time_left()
    if tl > 200:
        n3 = 1600
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        cur_n = n3
        cur_target = target3
        sc3 = compute_c5(h3)
        best_h = h3
        best_score = sc3

        tl = _time_left()
        if tl > 120:
            h_cd, c5_cd = pair_cd(best_h, cur_n, cur_target,
                                  time_limit=min(100, tl * 0.75),
                                  max_sweeps=50000,
                                  n_cand_start=20, n_cand_end=60,
                                  top_k_eval=50, max_no_improve=600,
                                  n_shifts_grad=3)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        tl = _time_left()
        if tl > 40:
            h_pol = slsqp_optimize(best_h, cur_n, cur_target,
                                   [1000, 5000],
                                   600, 1e-15)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 5: Final polish at current resolution
    tl = _time_left()
    if tl > 40:
        h_cd_final, c5_final = pair_cd(best_h, cur_n, cur_target,
                                       time_limit=min(30, tl * 0.7),
                                       max_sweeps=20000,
                                       n_cand_start=40, n_cand_end=90,
                                       top_k_eval=70, max_no_improve=300,
                                       n_shifts_grad=4)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    tl = _time_left()
    if tl > 30:
        h_final_pol = slsqp_optimize(best_h, cur_n, cur_target,
                                     [5000, 20000],
                                     600, 1e-15)
        sc = compute_c5(h_final_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
