import cvxpy as cp
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def active_set_lp(h_cur, n, target, max_iters=200, tol=1e-8, time_limit=60):
    """
    Active-set LP / outer approximation method.
    At each step, linearize the active constraints and solve an LP.
    """

    dx = 2.0 / n
    h = h_cur.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    active_shifts = []  # list of shift indices (into corr array)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()

        threshold = c_max_raw * 0.95
        active_mask = corr >= threshold
        if np.sum(active_mask) < 3:
            # At least top 3
            top_idx = np.argsort(corr)[-3:]
            active_mask = np.zeros(len(corr), dtype=bool)
            active_mask[top_idx] = True
        
        n_active = int(np.sum(active_mask))
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        # Compute linearized f_k for each active shift
        mid = n - 1
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

            # f_k(h) ≈ f_k(h) + grad_k · (h - h_cur)
            # grad_k[j] = (1 - h_cur[(j-k) mod n]) - h_cur[(j+k) mod n])  (times dx, but we can absorb)

            # f_k(h) = sum_j h[j] * (1 - h[(j-k) mod n]) * dx
            # Linearization: f_k(h_cur) + sum_j [(1-h_cur[(j-k)mod n]) - h_cur[(j+k)mod n]] * (h[j]-h_cur[j]) * dx
            
            # Compute gradient
            grad = np.zeros(n)
            idxs = np.arange(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] = (1.0 - h_cur[idxs[v1] - k])
            grad[v2] -= h_cur[idxs[v2] + k]
            
            f_k_cur = corr[s]
            
            # Linear constraint: f_k_cur + dx * grad · (h_var - h_cur) <= t_var
            # => dx * grad · h_var <= t_var - f_k_cur + dx * grad · h_cur
            lhs = dx * cp.dot(grad, h_var)
            rhs = t_var - f_k_cur + dx * np.dot(grad, h_cur)
            constraints.append(lhs <= rhs)
        
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
            if h_var.value is not None and t_var.value is not None:
                h_new = h_var.value.copy()
                h_new = project_feasible(h_new, target)
                c5_check = compute_c5(h_new)
                # Only accept if it improves the actual (nonlinear) objective
                if c5_check < c5_val + tol:
                    h = h_new

                    # Take a fraction of the step
                    for frac in [0.5, 0.25, 0.1]:
                        h_trial = h + frac * (h_new - h)

                        if c5_trial < c5_val:

                            break
                    else:

                        if it > 50:

            else:

        except Exception:
            break
        
        # Convergence check
        if it > 10 and abs(c5_val - best_c5) < 1e-10:
            break
    
    return best_h, best_c5

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():

            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient_active(h, active_shifts, n):
    """Compute averaged gradient over active shifts."""
    dx = 2.0 / n
    grad = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    n_act = len(active_shifts)
    for s in active_shifts:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        w = dx / n_act
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad

def fw_vertex(grad, n, target):
    """Find the FW vertex: set h[j]=1 for bottom n/2 gradient values."""
    sorted_idx = np.argsort(grad)
    v = np.zeros(n)
    v[sorted_idx[:int(target)]] = 1.0
    return v

def fw_line_search(h, v, n, target, n_evals=20):

    best_alpha = 0.0
    best_f = compute_c5(h)

    alphas = np.linspace(0.0, 1.0, n_evals)

        h_trial = project_feasible(h_trial, target)
        f_vals.append(compute_c5(h_trial))
    f_vals = np.array(f_vals)
    best_i = int(np.argmin(f_vals))
    best_alpha = alphas[best_i]

                best_alpha = a_mid
                a_lo = a_mid
            else:
                a_hi = a_mid

    h_new = project_feasible(h_new, target)
    return h_new, best_alpha, best_f

For the continuous case, the optimal h is likely close to binary (since the objective is bilinear and the feasible set is a polytope, the optimum of a bilinear function over a polytope is at a vertex... wait, that's not quite right because we have a MAX of bilinear functions).

Actually, for a single bilinear function f(h) = h^T A h (where A is not symmetric), the minimum over the polytope is at a vertex. But for max of multiple bilinear functions, the optimum might not be at a vertex.

However, in practice, the optimal h for this type of problem tends to be close to binary. This suggests that we should:
1. Find a good binary (or near-binary) solution
2. Then polish with continuous methods

Let me try a different approach:

**Approach: Greedy binary construction + continuous polish**

1. Start with a random binary vector with exactly n/2 ones
2. Use local search (flip individual bits) to minimize C₅
3. Then interpolate to higher resolution
4. Polish with continuous methods

For the binary local search:
- Flipping h[i] from 0 to 1 (or vice versa) changes the correlation in a predictable way
- We can compute the change in C₅ efficiently

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_active_gradient(h, active_shifts, n, corr_vals):
    dx = 2.0 / n
    grad = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    temps = np.log(corr_vals - corr_vals.min() + 1e-10)
    temps = temps - temps.max()
    weights = np.exp(temps)
    weights = weights / weights.sum()
    for s, w in zip(active_shifts, weights):
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad * dx

def fw_line_search(h, v, n, target, c5_start, n_coarse=12, n_fine=8):
    dx = 2.0 / n
    best_alpha = 0.0
    best_f = c5_start
    alphas = np.linspace(0.01, 1.0, n_coarse)
    best_i = 0
    best_f_coarse = c5_start
    for i, a in enumerate(alphas):
        h_trial = (1.0 - a) * h + a * v
        h_trial = project_feasible(h_trial, target)
        f = compute_c5(h_trial)
        if f < best_f_coarse:
            best_f_coarse = f
            best_i = i
    if 0 < best_i < n_coarse - 1:
        a_lo = alphas[best_i - 1]
        a_hi = alphas[best_i + 1]
    elif best_i == 0:
        a_lo = 0.0
        a_hi = alphas[1]
    else:
        a_lo = alphas[-2]
        a_hi = 1.0
    for _ in range(n_fine):
        a_mid = 0.5 * (a_lo + a_hi)
        h_mid = (1.0 - a_mid) * h + a_mid * v
        h_mid = project_feasible(h_mid, target)
        f_mid = compute_c5(h_mid)
        if f_mid < best_f_coarse:
            best_f_coarse = f_mid
            best_alpha = a_mid
            a_lo = a_mid
        else:
            a_hi = a_mid
    if best_f_coarse >= c5_start:
        return h, 0.0, c5_start
    h_new = (1.0 - best_alpha) * h + best_alpha * v
    h_new = project_feasible(h_new, target)
    return h_new, best_alpha, best_f_coarse

def active_set_fw(h_init, n, target, time_limit, max_iters=20000,
                  active_frac=0.04, min_active=5, max_active=30,
                  reheat_interval=300, reheat_scale=0.04, momentum=False):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 1.0
    no_improve = 0
    h_mom = h.copy()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if (it + 1) % 200 == 0 and time.time() - start > time_limit * 0.95:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > reheat_interval:
            rng = np.random.RandomState(it * 17 + 3)
            n_pert = max(5, n // 25)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            if compute_c5(h_pert) < c5_val + 0.005:
                h = h_pert
            no_improve = 0
            continue
        
        if no_improve > reheat_interval * 3:
            break
        
        c_max = float(np.max(corr))
        n_active = max(min_active, min(max_active, int(len(corr) * active_frac)))
        top_idx = np.argsort(corr)[-n_active:]
        
        grad = compute_active_gradient(h, top_idx, n, corr[top_idx])
        
        sorted_idx = np.argsort(grad)
        v = np.zeros(n)
        v[sorted_idx[:int(target)]] = 1.0
        
        if np.allclose(v, h, atol=1e-12):
            break
        
        if momentum and it > 10:
            h_mom = 0.7 * h_mom + 0.3 * h
            grad_mom = compute_active_gradient(h_mom, top_idx, n, corr[top_idx])
            sorted_mom = np.argsort(grad_mom)
            v_mom = np.zeros(n)
            v_mom[sorted_mom[:int(target)]] = 1.0
            v = 0.5 * v + 0.5 * v_mom
        
        h_new, alpha_used, f_new = fw_line_search(h, v, n, target, c5_val)
        h = h_new
    
    return best_h, best_c5

def pair_cd_fast(h_init, n, target, time_limit, max_sweeps=20000,
                 n_cand_start=10, n_cand_end=50, max_no_improve=300,
                 n_active_shifts=4, reheat_interval=200, reheat_scale=0.03):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        if consecutive_no_improve > reheat_interval and consecutive_no_improve % 50 == 0:
            rng = np.random.RandomState(sweep * 11 + 5)
            n_pert = max(5, n // 30)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc = compute_c5(h_pert)
            if sc < c5_val + 0.003:
                h = h_pert
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                for p in range(n):
                    pos_mask = (p + shifts >= 0) & (p + shifts < n)
                    neg_mask = (p - shifts >= 0) & (p - shifts < n)
                    change[p, :] = 0.0
                    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                    change[p, neg_mask] -= h[p - shifts[neg_mask]]
                consecutive_no_improve = 0
                continue
        
        top_k = np.argsort(corr)[-n_active_shifts:]
        grad = np.zeros(n)
        for s_idx in top_k:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= n_active_shifts
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        k_star_idx = int(np.argmax(corr))
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                delta_lo = max(h[i] - 1.0, -h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d, b_d = delta_lo, delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active_m]) if active_m.any() else 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    
    def cons_fun(x):
        return np.sum(x) - _g_target
    
    def cons_jac(x):
        return np.ones(_g_n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            def smooth_obj(x, a=alpha):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
                return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
            
            res = minimize(smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def _g_smooth_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
