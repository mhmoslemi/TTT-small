For the pair (i,j) with move h[i] -= delta, h[j] += delta:
The correlation at shift k becomes:
C_k(delta) = C_k(0) + delta * (dC_k/dh[i] * (-1) + dC_k/dh[j] * 1) = C_k(0) + delta * g_k

where g_k = dC_k/dh[j] - dC_k/dh[i] (the change in correlation k per unit delta).

So each C_k is linear in delta! The max of linear functions is a convex piecewise-linear function of delta. We want to find the delta that minimizes this.

The minimum of a convex piecewise-linear function occurs at a "kink point" where two lines cross, or at a boundary (delta = 0, delta = h[i], or delta = 1-h[j]).

So instead of bisection, we could:
1. Find all the "break points" of the piecewise-linear function
2. Evaluate C₅ at each break point
3. Take the best one

The break points occur where C_k(delta) = C_m(delta) for some k, m, i.e., where delta = (C_m(0) - C_k(0)) / (g_k - g_m) (assuming g_k ≠ g_m).

There are O(n²) such break points (since there are O(n) shifts), but we can be smarter: we only need to find the minimum, which occurs at one of the break points where the subgradient changes sign.

This is equivalent to finding the point where the "upper envelope" has a minimum. For a convex piecewise-linear function with O(n) pieces, this can be done in O(n log n) time using a convex hull trick.

    return s_max, grad

def proj_grad(h_init, target, start_time, time_limit, 
              alpha0=50, alpha_mult=2.0, alpha_max=100000,
              steps_per_alpha=100, base_lr=0.5):
    h = h_init.copy()
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    alpha = alpha0
    while time.time() - start_time < time_limit:
        # Run steps_per_alpha iterations at current alpha
        for it in range(steps_per_alpha):
            if time.time() - start_time > time_limit:
                break
            _, grad = smooth_max_and_grad(h, alpha)
            lr = base_lr / (1.0 + alpha / 200.0)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target)
            h = h_new
        
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        
        # Check if we should increase alpha
        # (if progress is slow, focus more on the max)
        alpha = min(alpha * alpha_mult, alpha_max)
    
    return best_h, best_score

def pair_cd(h_init, n, target, start_time, time_limit,
            max_sweeps=20000, n_cand=30, max_no_improve=300, n_top=3):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix (will be updated)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Compute gradient from top shifts
        if n_top <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top:]
            weights = corr[top_shifts] / np.sum(corr[top_shifts])
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        
        grad -= np.mean(grad)
        
        # Select candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                a_d, b_d = delta_lo, delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_i, best_j, best_delta = i, j, delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            np.clip(h, 0.0, 1.0, out=h)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_m = (p + shifts >= 0) & (p + shifts < n)
                neg_m = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_m] += (1.0 - h[p + shifts[pos_m]])
                change[p, neg_m] -= h[p - shifts[neg_m]]
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_refine(h_init, n, target, start_time, time_limit):
    # Use SLSQP with high-alpha smooth max for final polish
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    h = h_init.copy()
    best_score = compute_c5(h)
    
    for alpha in [2000, 10000]:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj(x):
                _, _ = smooth_max_and_grad(x, alpha)
                # Recompute for correct value
                corr = np.correlate(x, 1.0 - x, mode='full')
                f = corr * (2.0 / len(x))
                f_max = np.max(f)
                shifted = np.clip(alpha * (f - f_max), -500, 0)
                return f_max + np.log(np.sum(np.exp(shifted))) / alpha
            
            def obj_grad(x):
                _, g = smooth_max_and_grad(x, alpha)
                return g
            
            res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h = h_new
        except Exception:
            pass
    
    return h, best_score
