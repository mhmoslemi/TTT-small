    c_max = np.max(corr)
    c_min_active = c_max - rel_tol * max(c_max, 1e-10)
    active = corr >= c_min_active
    if not active.any():
        active = np.zeros(n_corr, dtype=bool)
        active[int(np.argmax(corr))] = True
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in np.where(active)[0]:
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    n_active = np.count_nonzero(active)
    grad /= n_active
    grad -= np.mean(grad)
    return grad

def pair_cd(h_init, n, target, time_limit, max_sweeps=80000,
            n_cand_base=60, max_no_improve=500,
            n_binsearch=20, seed=42, use_subgrad=True,
            stochastic_frac=0.03):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        if use_subgrad:
            tol = 0.0005 + 0.002 * min(1.0, float(consecutive_no_improve) / 200.0)
            grad = active_subgradient(corr, h, n, rel_tol=tol)
        else:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
            grad -= np.mean(grad)
        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        n_cand = int(n_cand_base + 60 * progress)
        n_cand = min(n_cand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_pairs = n_cand * n_cand
        n_rand = max(1, int(stochastic_frac * n_pairs)) if sweep > 50 else 0
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if n_rand > 0:
            for _ in range(n_rand):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def slsqp_smooth(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            from scipy.optimize import minimize
            def _obj(x, a=alpha):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
                return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
            def _cons_fun(x):
                return np.sum(x) - _g_target
            def _cons_jac(x):
                return np.ones(_g_n)
            res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, cur_score

def structured_perturb(h, n, target, rng):
    strategy = rng.randint(5)
    h_pert = h.copy()
    if strategy == 0:
        freq = rng.randint(2, 12)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2, n)
        h_pert += 0.05 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    elif strategy == 1:
        freq = rng.randint(1, 6)
        x = np.linspace(0, 2, n)
        h_pert += 0.08 * np.cos(2 * np.pi * freq * x / 2.0)
    elif strategy == 2:
        mask = rng.choice(n, size=n // 4, replace=False)
        h_pert[mask] += rng.randn(len(mask)) * 0.04
    elif strategy == 3:
        half = n // 2
        h_pert[:half] += 0.03
        h_pert[half:] -= 0.03
    else:
        h_pert += rng.randn(n) * 0.03
    return project_feasible(h_pert, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    rng = np.random.RandomState(seed)
    if tleft() > 60:
        h_pol, sc_pol = slsqp_smooth(best_h, n0, target0, [100, 500, 2000], 500)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol
    if tleft() > 200:
        h_cd, c5_cd = pair_cd(best_h, n0, target0, time_limit=min(180, tleft() * 0.6),
                               max_sweeps=80000, n_cand_base=50,
                               max_no_improve=600, n_binsearch=20,
                               seed=42, use_subgrad=True, stochastic_frac=0.02)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
    if tleft() > 150:
        h_cd2, c5_cd2 = pair_cd(best_h, n0, target0, time_limit=min(120, tleft() * 0.6),
                                 max_sweeps=60000, n_cand_base=70,
                                 max_no_improve=500, n_binsearch=20,
                                 seed=137, use_subgrad=True, stochastic_frac=0.05)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
    if tleft() > 100:
        h_pol, sc_pol = slsqp_smooth(best_h, n0, target0, [5000, 20000], 1500)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol
    if tleft() > 120:
        n_restarts = 3
        while tleft() > 60 and n_restarts > 0:
            h_pert = structured_perturb(best_h, n0, target0, rng)
            alloc = min(40, tleft() * 0.7 / max(1, n_restarts))
            h_r, c5_r = pair_cd(h_pert, n0, target0, time_limit=alloc,
                                 max_sweeps=30000, n_cand_base=60,
                                 max_no_improve=300, n_binsearch=18,
                                 seed=int(rng.randint(100000)),
                                 use_subgrad=True, stochastic_frac=0.05)
            if c5_r < best_score:
                best_score = c5_r
                best_h = h_r
            n_restarts -= 1
    if tleft() > 80:
        h_cd3, c5_cd3 = pair_cd(best_h, n0, target0, time_limit=min(100, tleft() * 0.7),
                                 max_sweeps=80000, n_cand_base=40,
                                 max_no_improve=400, n_binsearch=20,
                                 seed=999, use_subgrad=True, stochastic_frac=0.03)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
    if tleft() > 50:
        h_pol, sc_pol = slsqp_smooth(best_h, n0, target0, [20000, 80000], 2000)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol
    if tleft() > 200:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 150:
            h_cd1, c5_cd1 = pair_cd(h1, n1, target1, time_limit=min(130, tleft() * 0.6),
                                     max_sweeps=40000, n_cand_base=40,
                                     max_no_improve=400, n_binsearch=18,
                                     seed=251, use_subgrad=True, stochastic_frac=0.02)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                n0 = n1
                target0 = target1
        if tleft() > 50:
            h_pol1, sc_pol1 = slsqp_smooth(best_h, n0, target0, [5000, 20000], 1000)
            if sc_pol1 < best_score:
                best_score = sc_pol1
                best_h = h_pol1
    if tleft() > 60:
        h_cd_f, c5_f = pair_cd(best_h, n0, target0, time_limit=min(50, tleft() * 0.7),
                                max_sweeps=30000, n_cand_base=55,
                                max_no_improve=300, n_binsearch=20,
                                seed=501, use_subgrad=True, stochastic_frac=0.03)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f
    if tleft() > 30:
        h_final, sc_final = slsqp_smooth(best_h, n0, target0, [50000, 200000], 2000)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
