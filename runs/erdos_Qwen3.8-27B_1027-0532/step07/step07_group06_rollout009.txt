import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x, target, n):
    return np.sum(x) - target

def _cons_jac(x, target, n):
    return np.ones(n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, deadline, use_pnorm=False):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _cons_fun(x, target, n),
             'jac': lambda x: _cons_jac(x, target, n)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for param in alphas:
        if time.time() > deadline:
            break
        try:

                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

    return h_cur, cur_score

def project_feasible(h, target_sum):

                j = rng.randint(n)
                if i == j:
                    continue
                tmp = h_try[i]
                h_try[i] = h_try[j]
                h_try[j] = tmp
            else:
                i = rng.randint(n - 3)
                h_try[i:i+4] = h_try[i:i+4] + rng.randn(4) * 0.02

            h_try = project_feasible(h_try, target)
            new_c5 = compute_c5(h_try)
            delta_obj = new_c5 - cur_c5

            if delta_obj < 0 or rng.rand() < np.exp(-delta_obj / max(temp, 1e-8)):
                h = h_try
                cur_c5 = new_c5
                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
