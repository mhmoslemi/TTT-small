    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        
        dx = 2.0 / n
        # Compute correlation
        ones = 1.0 - h
        corr = np.correlate(h, ones, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Compute gradients for all shifts
        # grad[k_offset, j] = df_k/dh[j] where k_offset is the correlate index
        n_corr = 2 * n - 1
        grad = np.zeros((n_corr, n))
        
        for m in range(n_corr):
            k = m - (n - 1)
            # valid_first: max(0, -k) to min(n-1, n-1-k)
            lo = max(0, -k)
            hi = min(n-1, n-1-k)
            if lo > hi:
                continue
            # First term: 1 for j in [lo, hi]
            grad[m, lo:hi+1] = 1.0
            # Second term: -h[j+k] for 0 ≤ j+k < n
            lo2 = max(0, -k)
            hi2 = min(n-1, n-1-k)
            # j+k ranges from lo2+k to hi2+k, which is max(k,0) to min(n-1, n-1-k+k) = n-1
            # So j+k is in [max(k,0), n-1] when j is in [lo2, hi2]
            # h[j+k] for j in [lo2, hi2]:
            grad[m, lo2:hi2+1] -= h[lo2+k:hi2+k+1]
            # Third term: -h[j-k] for 0 ≤ j-k < n
            lo3 = max(0, k)
            hi3 = min(n-1, n-1+k)
            # j-k ranges from lo3-k to hi3-k
            # j-k is in [0, n-1] when j is in [max(0,k), n-1]
            # h[j-k] for j in [lo3, hi3]:
            grad[m, lo3:hi3+1] -= h[lo3-k:hi3-k+1]

        #      0 ≤ h ≤ 1
        #      sum(h) = target

        # = corr[m]*dx + dx * grad[m] @ (h - h₀)
        # We want: corr[m]*dx + dx * grad[m] @ (h - h₀) ≤ t
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == target]

            constraints.append(linearized_overlap <= t_var)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h[np.arange(n)] += diff / n
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_matrix(h, n):
    """Compute the gradient matrix for all shifts.
    Returns array of shape (2*n-1, n) where grad[m, j] = df_m/dh[j]
    (without the dx factor).
    """
    n_corr = 2 * n - 1
    grad = np.zeros((n_corr, n))
    idx = np.arange(n)
    
    for m in range(n_corr):
        k = m - (n - 1)
        # valid_first: j such that 0 ≤ j < n and 0 ≤ j+k < n
        lo = max(0, -k)
        hi = min(n - 1, n - 1 - k)
        if lo > hi:
            continue
        # First term: 1 for j in [lo, hi]
        grad[m, lo:hi+1] = 1.0
        # Second term: -h[j+k] for j in [lo, hi] (j+k is in [0, n-1])
        grad[m, lo:hi+1] -= h[lo+k:hi+k+1]
        # Third term: -h[j-k] for j such that 0 ≤ j-k < n
        lo3 = max(0, k)
        hi3 = min(n - 1, n - 1 + k)
        if lo3 <= hi3:
            grad[m, lo3:hi3+1] -= h[lo3-k:hi3-k+1]
    
    return grad

    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                # Perturbation
                rng = np.random.RandomState(sweep + 1)
                n_pert = max(5, n // 50)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                h_pert = h.copy()
                h_pert[pert_idx] += rng.randn(n_pert) * 0.03
                h_pert = project_feasible(h_pert, target)
                if compute_c5(h_pert) < c5_val + 0.005:
                    h = h_pert
                    for p in pert_idx:
                        pos_mask = (p + shifts >= 0) & (p + shifts < n)
                        neg_mask = (p - shifts >= 0) & (p - shifts < n)
                        change[p, :] = 0
                        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                        change[p, neg_mask] -= h[p - shifts[neg_mask]]
                no_improve = 0
                continue
        
        # Find top active shifts
        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            weights = corr[top_k] / corr_top_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Select candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Find best pair
        k_star_idx = int(np.argmax(corr))
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                
                # Binary search for best delta
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, :] = 0
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    return best_h, best_c5


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=_cons_list(),
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def _cons_list():
    return [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
