    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(777)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > 200 and consecutive_no_improve % 40 == 0:
            scale = 0.01 + 0.02 * (consecutive_no_improve // 200)
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0

        if n_active == 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            weights = np.array([1.0])
            active_ks = np.array([k_star])
        else:
            top_idx = np.argsort(corr)[-n_active:]
            top_sum = np.sum(corr[top_idx])
            if top_sum < 1e-15:
                top_sum = 1.0
            weights = corr[top_idx] / top_sum
            active_ks = np.array([int(s) - mid for s in top_idx])

        grad = np.zeros(n)
        for w, k in zip(weights, active_ks):
            if k > 0:
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
            elif k < 0:
                ka = -k
                v1 = (idxs + ka >= 0) & (idxs + ka < n)
                v2 = (idxs - ka >= 0) & (idxs - ka < n)
                grad[v1] += w * (1.0 - h[idxs[v1] + ka])
                grad[v2] -= w * h[idxs[v2] - ka]
            else:
                grad += w * (1.0 - 2.0 * h)
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        max_pairs = 2500
        if n_h * n_l > max_pairs:
            n_h_use = min(n_h, max_pairs // max(n_l, 1))
            n_l_use = min(n_l, max_pairs // max(n_h, 1))
            if n_h_use < 1:
                n_h_use = 1
            if n_l_use < 1:
                n_l_use = 1
            high_set = high_set[-n_h_use:]
            low_set = low_set[:n_l_use]
            n_h = len(high_set)
            n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            ci = change[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-14:
                    continue
                g = change[j] - ci
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-14:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, block_size=4, n_iterations=150):
    global _g_block, _g_h_fixed, _g_block_target
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    idxs = np.arange(n)
    rng = np.random.RandomState(555)

    for it in range(n_iterations):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        top3 = np.argsort(corr)[-3:]
        top_sum = np.sum(corr[top3])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s in top3:
            w = corr[s] / top_sum
            k = int(s) - mid
            if k > 0:
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
            elif k < 0:
                ka = -k
                v1 = (idxs + ka >= 0) & (idxs + ka < n)
                v2 = (idxs - ka >= 0) & (idxs - ka < n)
                grad[v1] += w * (1.0 - h[idxs[v1] + ka])
                grad[v2] -= w * h[idxs[v2] - ka]
            else:
                grad += w * (1.0 - 2.0 * h)
        grad -= np.mean(grad)

        abs_grad = np.abs(grad)
        sorted_g = np.argsort(abs_grad)[::-1]
        n_select = max(block_size + 3, n // 5)
        candidates = sorted_g[:n_select]

        start_idx = int(rng.randint(0, max(1, len(candidates) - block_size + 1)))
        block = np.sort(candidates[start_idx:start_idx + block_size])

        if len(np.unique(block)) < block_size:
            continue

        _g_block = block
        _g_h_fixed = h.copy()

        fixed_sum = np.sum(h) - np.sum(h[block])
        _g_block_target = target - fixed_sum

        if _g_block_target < 1e-10 or _g_block_target > block_size - 1e-10:
            continue

        x0 = h[block].copy()
        bounds_b = [(0.0, 1.0)] * block_size
        cons_b = [{'type': 'eq', 'fun': _block_cons, 'jac': _block_cons_jac}]

        try:
            res = minimize(_block_obj, x0, method='SLSQP',
                          bounds=bounds_b, constraints=cons_b,
                          options={'maxiter': 400, 'ftol': 1e-14})
            h_new = h.copy()
            h_new[block] = np.clip(res.x, 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [50, 200, 1000, 5000, 20000],
                                600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 280:
        h_cd, c5_cd = pair_cd(best_h, current_n, current_target,
                              time_limit=min(300, tleft() * 0.6),
                              max_sweeps=30000,
                              n_cand_start=18, n_cand_end=80,
                              max_no_improve=500,
                              n_active=2, bisection_iters=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 150:
        h_blk, c5_blk = block_cd(best_h, current_n, current_target,
                                 time_limit=min(140, tleft() * 0.6),
                                 block_size=4, n_iterations=120)
        if c5_blk < best_score:
            best_score = c5_blk
            best_h = h_blk

    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [5000, 20000, 80000, 300000],
                                1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        h_cd2, c5_cd2 = pair_cd(best_h, current_n, current_target,
                                time_limit=min(200, tleft() * 0.6),
                                max_sweeps=25000,
                                n_cand_start=30, n_cand_end=90,
                                max_no_improve=400,
                                n_active=3, bisection_iters=15)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 300:
        n1 = 440
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 200:
            h_cd1, c5_cd1 = pair_cd(h1, n1, target1,
                                    time_limit=min(150, tleft() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_start=15, n_cand_end=65,
                                    max_no_improve=400,
                                    n_active=2, bisection_iters=14)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                current_n = n1
                current_target = target1
        if current_n == n1 and tleft() > 80:
            h_pol1, sc1 = slsqp_round(best_h, n1, target1,
                                       [5000, 20000, 80000],
                                       800, 1e-15)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1
            if tleft() > 80:
                h_blk1, c5_blk1 = block_cd(best_h, n1, target1,
                                           time_limit=min(70, tleft() * 0.5),
                                           block_size=4, n_iterations=80)
                if c5_blk1 < best_score:
                    best_score = c5_blk1
                    best_h = h_blk1

    if tleft() > 200:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.015, 0.04, 0.08, 0.12]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [200, 2000, 10000],
                                        500, 1e-15)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if tleft() > 80:
                h_cd_p, c5_p = pair_cd(h_ref, current_n, current_target,
                                       time_limit=min(50, tleft() * 0.35),

                                       n_cand_start=25, n_cand_end=70,
                                       max_no_improve=300,
                                       n_active=2, bisection_iters=14)
                if c5_p < best_score:
                    best_score = c5_p
                    best_h = h_cd_p

    if tleft() > 80:
        h_blk2, c5_blk2 = block_cd(best_h, current_n, current_target,
                                   time_limit=min(70, tleft() * 0.7),
                                   block_size=5, n_iterations=60)
        if c5_blk2 < best_score:

    if tleft() > 60:
        h_final_cd, c5_final = pair_cd(best_h, current_n, current_target,

                                            [50000, 200000, 500000],
                                            800, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
