def compute_gradient_top(h, n, top_idx, weights):
    """Compute weighted sum of gradients for top shifts."""
    grad = np.zeros(n)
    mid = n - 1
    for idx, w in zip(top_idx, weights):
        k = int(idx) - mid  # actual shift
        for j in range(n):
            term = 0.0
            # h[j] as the "first" factor: i=j, need j+k valid
            if 0 <= j + k < n:
                term += (1.0 - h[j + k])
            # h[j] as the "second" factor: i+k=j, i=j-k, need j-k valid
            if 0 <= j - k < n:
                term -= h[j - k]
            grad[j] += w * term
    return grad

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if c_max < 1e-15:
        return 0.0
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x, target):
    return np.sum(x) - target

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_refine(h_start, target, alphas, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (target,)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    def obj(x, alpha):
        return smooth_obj(x, alpha)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max < 1e-15:
        return 0.0
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x, target):
    return np.sum(x) - target

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_refine(h_start, target, alphas, maxiter, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (target,)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def obj(x, a=alpha):
            return smooth_obj(x, a, target)
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_dm(h_init, n, target, time_limit, start_time,
                   n_top=5, lr_init=0.005, momentum=0.9,
                   max_iter=200000, patience=8000):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while it < max_iter:
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr)) * dx if (dx := 2.0 / n) else 0

        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            break

        top_idx = np.argsort(corr)[-n_top:]
        weights = corr[top_idx].astype(float)
        w_sum = weights.sum()
        if w_sum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights = weights / w_sum

        grad = np.zeros(n)
        for t in range(n_top):
            k = int(top_idx[t]) - mid
            w = weights[t]
            if k >= 0:
                if k > 0:
                    grad[k:] += w * (1.0 - h[:n - k])
                    grad[:n - k] -= w * h[k:]
                else:
                    grad[:] += w * (1.0 - h)
            else:
                m = -k
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]

        grad -= np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target)

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(np.max(corr_new)) * (2.0 / n)

        if c5_new < c5 - 1e-15:
            h = h_new
            lr = min(lr * 1.01, 0.05)
        else:
            v[:] = 0.0
            lr *= 0.85

        it += 1

    return best_h, best_c5

def pair_cd_fast(h_init, n, target, time_limit, start_time,
                 max_sweeps=20000, n_cand=30, max_no_improve=300,
                 n_top_shifts=3, bisection_iters=14):
    dx = 2.0 / n
    mid = n - 1
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-n_top_shifts:]
        weights = corr[top_shifts].astype(float)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(n_top_shifts) / n_top_shifts
        else:
            weights = weights / wsum

        grad = np.zeros(n)
        for t in range(n_top_shifts):
            k = int(top_shifts[t]) - mid
            w = weights[t]
            if k >= 0:
                if k > 0:
                    grad[k:] += w * (1.0 - h[:n - k])
                    grad[:n - k] -= w * h[k:]
                else:
                    grad[:] += w * (1.0 - h)
            else:
                m = -k
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]

        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(n_h):
            i = int(high_set[ii])
            hi = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                hj = h[j]
                delta_lo = max(-hj, hi - 1.0)
                delta_hi = min(hi, 1.0 - hj)
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(2 * n - 1)
                if i - j >= 0:
                    g_off = (i - j)
                    k_val = i - j
                    for idx in range(n - k_val):
                        g[idx + mid] += (1.0 - h[idx + k_val]) * 0 + h[idx + k_val] * 0
                # Recompute: change in corr when we do h[i]-=d, h[j]+=d
                # corr[k] changes by: -d*(1-h[i+k]) + d*h[i-k] - d*(1-h[j+k]) + d*h[j-k] (with bounds)
                # Actually: corr_k = sum h[n+k](1-h[n])
                # If h[i] decreases by d: corr_k changes by -d*(1-h[n]) where n+k=i => n=i-k, and -d*h[n+k](1) where n=i => contributes -d to the (1-h[n]) part... 
                # Let me think differently.
                # corr_k = sum_n h[n+k]*(1-h[n])
                # ∂corr_k/∂h[i] = (1-h[i-k])*1[i-k valid as n] ... = (1-h[i-k]) if i>=k>=0 or appropriate
                # Actually from our gradient formula:
                # ∂corr_k/∂h[i] for shift k: 
                #   = (1-h[i-k]) * 1[i-k in range for the first term position] - h[i+k] * 1[...]
                # This is getting complicated. Let me just compute the change directly.
                # 
                # If we change h[i] -= d and h[j] += d:
                # New corr_k = old_corr_k + d * (∂corr_k/∂h[j] - ∂corr_k/∂h[i])
                # So the change vector is d * (grad_k[j] - grad_k[i]) for each k
                # 
                # For efficiency, compute the difference gradient:
                # diff_k = grad_at_j - grad_at_i for each shift k
                # This is O(n) to compute (one per shift).
                # 
                # But we already computed the full gradient from top shifts. The per-shift gradient
                # at a specific point is O(1). So for each (i,j) pair, computing the change vector
                # is O(n) (one entry per shift).
                # 
                # Total: O(n_cand^2 * n) per sweep, which for n_cand=30, n=800 is 720000. Fine.

                # Compute change in correlation for this pair
                # For each shift k, the change is d * (gk_j - gk_i) where gk is grad of corr_k
                # gk_i = (1-h[i-k])*1[i-k valid] - h[i+k]*1[i+k valid] (for k>=0)
                # This is the same as our gradient formula evaluated at point i for shift k.
                
                # Let's vectorize: for all shifts at once
                # Shift k ranges from -(n-1) to n-1
                shifts = np.arange(-(n-1), n)
                # For point i and shift k: grad contribution = (1-h[idx1])*mask1 - h[idx2]*mask2
                # where idx1 = i-k (valid if 0<=i-k<n and appropriate for the shift direction)
                # This is complex. Let me just compute it directly for the top shifts.
                
                # Actually for the bisection, we only need to evaluate max of (corr + d * change_vec)
                # where change_vec[k] = grad_k[j] - grad_k[i]
                # We can compute this for all k in O(n) using the correlation structure.
                
                # For shift k >= 0: corr_k = sum_{n=0}^{n-1-k} h[n+k](1-h[n])
                # change_k = [h[j+k](1-h[j]) - h[j+k](1-h[j]-d) ... ]
                # Hmm, let me think of it as:
                # New h: h'[i] = h[i]-d, h'[j] = h[j]+d, rest same
                # New corr_k = sum_n h'[n+k](1-h'[n])
                # = sum_n h[n+k](1-h[n]) + d * (contribution from h[j] + d) - d * (contribution from h[i] - d)
                # 
                # The change at shift k is:
                # d * [(1-h[j-k])*1[j-k valid as n for shift k as second arg] + h[j+k]*1[...] ... ]
                # This is just d * (grad_k[j] - grad_k[i])
                
                # For the bisection, I need: max_k (corr[k] + d * (gk[j] - gk[i]))
                # where gk[i] = ∂corr_k/∂h[i]
                
                # Let me compute the change vector for all k:
                # Using numpy: for each valid shift k, compute gk[i] and gk[j]
                # gk[i] for k>=0: (1-h[i-k]) if i>=k else 0, minus h[i+k] if i<n-k else 0
                # gk[i] for k<0 (m=-k): (1-h[i+m]) if i<n-m else 0, minus h[i-m] if i>=m else 0
                
                # Vectorized over all k:
                ki = shifts  # array of all shifts
                # For point i:
                # term1: (1-h[i-k]) where i-k is valid index [0,n)
                idx_ik = i - ki  # i - k for each k
                mask_ik = (idx_ik >= 0) & (idx_ik < n)
                term1_i = np.where(mask_ik, 1.0 - h[idx_ik], 0.0)
                # But we also need the shift-specific validity: for k>=0, need i >= k (i.e., i-k >= 0 and i-k <= n-1-k)
                # Actually, the condition for shift k>=0 is that n=i-k must satisfy 0 <= n <= n-1-k, i.e., k <= i <= n-1
                # For k<0 (m=-k): n=i must satisfy m <= i <= n-1, i.e., i >= m = -k
                # So the valid mask for the "first term" (1-h[i-k]):
                #   k >= 0: k <= i (i.e., i-k >= 0) AND i <= n-1 (always true) AND i-k <= n-1-k (i <= n-1, always)
                #   Actually I think the simple condition 0 <= i-k < n AND the shift-range condition is what we need.
                # For k >= 0: valid if 0 <= i-k AND i-k <= n-1-k, i.e., k <= i <= n-1
                # For k < 0 (m=-k): valid if 0 <= i-k < n AND i >= m, i.e., i >= -k AND 0 <= i+k < n
                # Hmm, this is the same as what I had before. Let me just use the general mask.
                
                # Actually, I realize the simplest approach: just compute the change vector directly.
                # change_at_shift_k = corr_after[k] - corr_before[k] (per unit d)
                # = grad_k[j] - grad_k[i]
                
                # For speed, let me precompute for points i and j:
                # grad_k[i] for all k:
                #   Using the formula: for k>=0: (1-h[i-k])*1[k<=i] - h[i+k]*1[i<n-k]
                #   for k<0, m=-k: (1-h[i+m])*1[i<n-m] - h[i-m]*1[i>=m]
                
                # Vectorized:
                # For point i:
                gk_i = np.zeros(2*n-1)
                gk_j = np.zeros(2*n-1)
                
                # For shift k (0-indexed in shifts array, actual shift = shifts[t]):
                for t in range(n_top_shifts):
                    k = int(top_shifts[t]) - mid
                    if k >= 0:
                        if k == 0:
                            gk_i[t + (n-1) - (n-1)] = 0  # placeholder
                        # grad at i for shift k:
                        # (1-h[i-k]) if i >= k, minus h[i+k] if i < n-k
                        val_i = 0.0
                        if i >= k and i < n:
                            val_i += (1.0 - h[i - k])
                        if i < n - k:
                            val_i -= h[i + k]
                        val_j = 0.0
                        if j >= k and j < n:
                            val_j += (1.0 - h[j - k])
                        if j < n - k:
                            val_j -= h[j + k]
                        gk_i[t] = val_i
                        gk_j[t] = val_j
                    else:
                        m = -k
                        val_i = 0.0
                        if i < n - m:
                            val_i += (1.0 - h[i + m])
                        if i >= m:
                            val_i -= h[i - m]
                        val_j = 0.0
                        if j < n - m:
                            val_j += (1.0 - h[j + m])
                        if j >= m:
                            val_j -= h[j - m]
                        gk_i[t] = val_i
                        gk_j[t] = val_j
                
                # change at top shift t: d * (gk_j[t] - gk_i[t])
                change_top = gk_j[:n_top_shifts] - gk_i[:n_top_shifts]
                
                # For the bisection, we need max over ALL shifts of (corr[k] + d * change[k])
                # But we only computed change for top shifts. For non-top shifts, the change is small
                # (since they're far from the max). So approximate: max ≈ max(top + d*change_top, rest_max)
                # where rest_max = max of corr over non-top shifts (unchanged by d to first order)
                
                # Actually, for correctness, let me compute the change for ALL shifts.
                # This is O(n) per (i,j) pair, which with n_cand^2 pairs gives O(n_cand^2 * n) per sweep.
                # For n_cand=30, n=800: 900*800 = 720000. Fine.
                
                # Compute change for all shifts:
                change_all = np.zeros(2*n-1)
                for t in range(2*n-1):
                    k = t - (n-1)
                    if k >= 0:
                        # grad at i: (1-h[i-k])*1[i>=k] - h[i+k]*1[i<n-k]
                        vi = 0.0
                        if k <= i:
                            vi += (1.0 - h[i-k])
                        if i < n - k:
                            vi -= h[i+k]
                        # grad at j:
                        vj = 0.0
                        if k <= j:
                            vj += (1.0 - h[j-k])
                        if j < n - k:
                            vj -= h[j+k]
                    else:
                        m = -k
                        vi = 0.0
                        if i < n - m:
                            vi += (1.0 - h[i+m])
                        if i >= m:
                            vi -= h[i-m]
                        vj = 0.0
                        if j < n - m:
                            vj += (1.0 - h[j+m])
                        if j >= m:
                            vj -= h[j-m]
                    change_all[t] = vj - vi
                
                # Bisection to find optimal d
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * change_all
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(change_all[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * change_all) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def stochastic_kick(h_init, n, target, time_limit, start_time, n_restarts=5, scale=0.03):
    rng = np.random.RandomState(123)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    for r in range(n_restarts):
        if time.time() - start_time > time_limit:
            break
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        h_sub, sc_sub = subgradient_dm(h_pert, n, target,
                                       min(time_limit, start_time + 30), start_time,
                                       n_top=5, lr_init=0.003, momentum=0.85,
                                       max_iter=30000, patience=3000)
        if sc_sub < best_c5:
            best_c5 = sc_sub
            best_h = h_sub
    
    return best_h, best_c5

        if sc11 < best_score:

            best_h = h11

    # Phase 12: Stochastic restarts
    if remaining() > 80:
        t_end = min(80, remaining() - 10)
        h12, sc12 = stochastic_kick(best_h, n0, target0, start_time + t_end, start_time,
                                     n_restarts=4, scale=0.02)
        if sc12 < best_score:
            best_score = sc12
            best_h = h12

    # Phase 13: Final SLSQP
    if remaining() > 40:
        t_end = min(40, remaining() - 5)
        h13, sc13 = slsqp_refine(best_h, target0, [5000, 20000], 300, start_time + t_end, start_time)
        if sc13 < best_score:
            best_score = sc13
            best_h = h13

    # Phase 14: Final pair CD
    if remaining() > 40:
        t_end = min(40, remaining() - 5)
        h14, sc14 = pair_cd_fast(best_h, n0, target0, start_time + t_end, start_time,
                                  max_sweeps=8000, n_cand=30, max_no_improve=150,
                                  n_top_shifts=1, bisection_iters=16)
        if sc14 < best_score:
            best_score = sc14
            best_h = h14

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
