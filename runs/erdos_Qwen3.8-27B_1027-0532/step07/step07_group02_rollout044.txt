Let me also add a new technique: **Subgradient descent with Polyak step size**

The idea:
- Maintain a running upper bound UB
- At each iteration, compute the subgradient g of C₅ at the current point
- Take a step in the -g direction with step size: eta = (UB - f(x)) / ||g||^2
- Project back to feasible set

This has good convergence properties for convex problems and can work well for non-convex problems near the optimum.

Actually, since the problem is non-convex, let me use a simpler approach: **gradient descent with backtracking line search**.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-16):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if alpha < 0:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol,
                                    'maxls': 50})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])

def update_change_row(change, p, h, n, shifts):

    neg_mask = (p - shifts >= 0) & (p - shifts < n)

def pair_cd_fast(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=30, n_cand_end=100, max_no_improve=300,
                 n_binsearch=14, seed=123):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            if consecutive_no_improve % 100 == 0 and consecutive_no_improve > 100:
                npert = max(5, n // 20)
                idxs_pert = rng.choice(n, size=npert, replace=False)
                h[idxs_pert] += rng.randn(npert) * 0.05
                h = project_feasible(h, target)
                # Recompute change for perturbed indices
                for ip in idxs_pert:
                    update_change_row(change, int(ip), h, n, shifts)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def projected_subgrad_descent(h_init, n, target, time_limit, lr=0.01, 
                               momentum=0.9, seed=42):
    """Subgradient descent with projection and momentum."""
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    v = np.zeros(n)  # momentum vector
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    
    # Adaptive learning rate
    current_lr = lr
    decay = 0.999
    
    while time.time() - start < time_limit:
        # Compute subgradient (gradient of the max correlation)
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        k = k_star - mid
        
        grad = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
        
        # Also add small contributions from near-max correlations
        top_k = np.argsort(corr)[-5:]
        for tk in top_k:
            if tk == k_star:
                continue
            weight = (corr[tk] - corr[k_star]) * 10  # small weight for near-max
            if weight > 0:
                kk = tk - mid
                w1 = (idxs - kk >= 0) & (idxs - kk < n)
                w2 = (idxs + kk >= 0) & (idxs + kk < n)
                grad[w1] += weight * (1.0 - h[idxs[w1] - kk])
                grad[w2] -= weight * h[idxs[w2] + kk]
        
        # Center gradient (to maintain sum constraint)
        grad -= np.mean(grad)
        
        # Momentum update
        v = momentum * v - current_lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        # Evaluate
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
            h = h_new
        else:
            no_improve += 1
            # Random restart if stuck
            if no_improve > 200:
                h = best_h + rng.randn(n) * 0.02
                h = project_feasible(h, target)
                v = np.zeros(n)
                no_improve = 0
        
        current_lr *= decay
    
    return best_h, best_c5

One more thought: I should also try the "Frank-Wolfe" approach as an alternative to pair CD. The Frank-Wolfe method for this problem would be:
1. Compute gradient g of C₅ (or smoothed version)
2. Find h_LINE: the point in the feasible set minimizing <g, h>
   - This is: set h_LINE[i] = 1 for the n/2 indices with smallest g[i], 0 otherwise
3. Update: h = (1-t)*h + t*h_LINE

This is much cheaper per iteration than pair CD (O(n log n) per step vs O(n_cand^2 * n_binsearch * n)).

Let me add this as an additional optimization method.

```python
def frank_wolfe(h_init, n, target, time_limit, alpha=1000, 
                step_type='diminishing', seed=42):
    """Frank-Wolfe (conditional gradient) on the hypersimplex."""
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    t = 0  # iteration counter
    
    while time.time() - start < time_limit:
        t += 1
        
        # Compute gradient (use smoothed version)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for s in range(n_corr):
            w = weights[s]
            if w < 1e-10:
                continue
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        # Frank-Wolfe LINE step: minimize <grad, h> over hypersimplex
        # Set h_LINE[i] = 1 for target indices with smallest grad[i]
        sorted_idx = np.argsort(grad)
        h_line = np.zeros(n)
        h_line[sorted_idx[:target]] = 1.0
        
        # Step size
        if step_type == 'diminishing':
            step = 2.0 / (t + 2)
        elif step_type == 'fixed':
            step = 0.1
        else:  # 'full'
            step = 1.0
        
        h_new = (1 - step) * h + step * h_line
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        if no_improve > 500:
            break
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
