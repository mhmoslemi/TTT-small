```python
def subgradient_bb(h_init, n, target, time_limit, max_iter=50000):
    """Subgradient method with Barzilai-Borwein step size and momentum."""
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Momentum
    momentum = 0.9
    v = np.zeros(n)
    
    # BB step size tracking
    h_prev = h.copy()
    grad_prev = None
    bb_step = 0.01
    
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_val = c_max * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active set (shifts within tolerance of max)
        tol = c_max * 0.01 + 1e-6
        active = np.where(corr >= c_max - tol)[0]
        
        # Compute gradient of the smooth approximation
        alpha = 100.0 + 50.0 * no_improve  # increasing temperature
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for k_idx in active:
            w = weights[k_idx]
            if w < 1e-10:
                continue
            k = int(k_idx) - mid
            v1_m = (idxs - k >= 0) & (idxs - k < n)
            v1_p = (idxs + k >= 0) & (idxs + k < n)
            grad[v1_m] += w * (1.0 - h[idxs[v1_m] - k])
            grad[v1_p] -= w * h[idxs[v1_p] + k]
        
        # Center the gradient (to maintain the sum constraint)
        grad -= np.mean(grad)
        
        # BB step size
        if grad_prev is not None:
            s = h - h_prev
            y = grad - grad_prev
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            if yy > 1e-12 and abs(sy) > 1e-12:
                bb_step = np.dot(s, s) / abs(sy)
                bb_step = np.clip(bb_step, 1e-6, 0.1)
        
        h_prev = h.copy()
        grad_prev = grad.copy()
        
        # Momentum step
        v = momentum * v - bb_step * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        # Line search (simple backtracking)
        step_scale = 1.0
        for ls in range(20):
            h_try = h + step_scale * v
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < c5_val:
                h = h_try
                break
            step_scale *= 0.5
        
        if step_scale < 0.001:
            h = h_new  # accept the projected step anyway
        
    return best_h, best_c5
```

```python
def augmented_lagrangian_smooth(h_init, n, target, time_limit, 
                                  base_alpha=100, alpha_growth=1.5,
                                  rho=1.0, max_alpha_iters=50):
    """Augmented Lagrangian method with smooth max approximation."""
    start = time.time()
    
    alpha = base_alpha
    lam = 0.0
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha_iter in range(max_alpha_iters):
        if time.time() - start > time_limit:
            break
        
        # Inner loop: minimize augmented objective
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            smooth_max = c_max + np.log(np.sum(np.exp(shifted))) / alpha
            sum_viol = np.sum(x) - target
            return smooth_max + rho/2.0 * sum_viol**2 + lam * sum_viol
        
        def obj_grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            weights = np.exp(shifted)
            weights /= np.sum(weights)
            
            mid = n - 1
            idxs = np.arange(n)
            grad = np.zeros(n)
            active = weights > 1e-10
            for k_idx in np.where(active)[0]:
                k = int(k_idx) - mid
                w = weights[k_idx]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
                grad[v2] -= w * x[idxs[v2] + k]
            
            sum_viol = np.sum(x) - target
            grad += (rho * sum_viol + lam)
            
            return grad
        
        # Use L-BFGS-B
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 200, 'ftol': 1e-15})
        h_new = res.x
        
        # Check sum constraint
        sum_new = np.sum(h_new)
        if abs(sum_new - target) > 0.1:
            h_new = project_feasible(h_new, target)
        
        # Update best
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        
        # Update Lagrange multiplier
        sum_diff = np.sum(h_new) - target
        lam += rho * sum_diff
        
        # Increase alpha (make smooth approximation sharper)
        alpha *= alpha_growth
        rho *= 2.0
        
        h = h_new
    
    return best_h, best_score
```

```python
def subgradient_heavy_ball(h_init, n, target, time_limit, max_iter=100000,
                           alpha_start=0.01, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    alpha = alpha_start
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_val = c_max * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                alpha *= 0.95
        
        # Active set
        tol = max(c_max * 0.02, 1e-4)
        active = np.where(corr >= c_max - tol)[0]
        n_active = len(active)
        
        # Average gradient over active set
        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= n_active
        grad -= np.mean(grad)  # maintain sum constraint
        
        # Heavy ball step
        v = momentum * v - alpha * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        # Check improvement
        sc_new = compute_c5(h_new)
        if sc_new < c5_val * (1 + 1e-6):
            h = h_new
        else:
            # Try smaller step
            h_new2 = h + 0.5 * v
            h_new2 = project_feasible(h_new2, target)
            sc_new2 = compute_c5(h_new2)
            if sc_new2 < c5_val * (1 + 1e-8):
                h = h_new2
            else:
                alpha *= 0.9
                v *= 0.5
    
    return best_h, best_c5
```

```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def _smth_val_grad(x, alpha):
    n = _g_n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    weights /= wsum
    val = c_max + np.log(wsum) / alpha
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for ki in active:
        k = int(ki) - mid
        w = weights[ki]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return val, grad

def _aobj(x):
    val, grad = _smth_val_grad(x, _g_alpha)
    sv = np.sum(x) - _g_target
    return val + _g_rho * 0.5 * sv * sv + _g_lam * sv

def _agrad(x):
    val, grad = _smth_val_grad(x, _g_alpha)
    sv = np.sum(x) - _g_target
    grad += _g_rho * sv + _g_lam
    return grad

def aug_lag(h_init, n, target, time_limit, base_alpha=50.0, alpha_growth=1.4,
            rho_start=0.1, rho_growth=2.0, max_outer=40, max_inner=300):
    global _g_n, _g_alpha, _g_rho, _g_lam, _g_target
    _g_n = n
    _g_target = target
    start = time.time()
    alpha = base_alpha
    rho = rho_start
    lam = 0.0
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_sc = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for outer in range(max_outer):
        if time.time() - start > time_limit:
            break
        _g_alpha = alpha
        _g_rho = rho
        _g_lam = lam
        try:
            res = minimize(_aobj, h, jac=_agrad, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': max_inner, 'ftol': 1e-18})
            h_new = res.x
        except Exception:
            break
        sv = np.sum(h_new) - target
        if abs(sv) < 0.05:
            h = h_new
        else:
            h = project_feasible(h_new, target)
        sc = compute_c5(h)
        if sc < best_sc:
            best_sc = sc
            best_h = h.copy()
        lam = lam + rho * sv
        rho *= rho_growth
        alpha *= alpha_growth
        if alpha > 5e5:
            break
    return best_h, best_sc

def multi_cd(h_init, n, target, time_limit, n_cand=40, n_active=3, max_sweeps=20000):

    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_imp = 0
    # Build change matrix
    n_corr = 2*n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5v = float(np.max(corr)) * dx
        if c5v < best_c5 - 1e-14:
            best_c5 = c5v
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 500:
            break
        
        # Top active shifts
        top = np.argsort(corr)[-n_active:]
        grad = np.zeros(n)
        for si in top:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(top)
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        lows = sorted_idx[:n_cand]
        highs = sorted_idx[-n_cand:]
        
        best_trial = c5v
        best_i = -1
        best_j = -1
        best_d = 0.0
        
        nh = len(highs)
        nl = len(lows)
        for ii in range(nh):
            i = int(highs[ii])
            for jj in range(nl):
                j = int(lows[jj])
                if i == j:
                    continue
                dlo = max(h[i] - 1.0, -h[j])
                dhi = min(h[i], 1.0 - h[j])
                if dhi - dlo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = dlo
                b_d = dhi
                for _ in range(16):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g
                    fm = np.max(vals)
                    am = vals >= fm - 1e-14
                    if am.any():
                        fp = np.max(g[am])
                    else:
                        fp = 0.0
                    if fp > 1e-12:
                        b_d = m_d
                    else:
                        a_d = m_d
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) < 1e-12:
                    continue
                am2 = np.max(corr + d_star * g) * dx
                if am2 < best_trial:
                    best_trial = am2
                    best_i = i
                    best_j = j
                    best_d = d_star
        
        if best_i >= 0:
            h[best_i] -= best_d
            h[best_j] += best_d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pm = (p + shifts >= 0) & (p + shifts < n)
                nm = (p - shifts >= 0) & (p - shifts < n)
                change[p, pm] += (1.0 - h[p + shifts[pm]])
                change[p, nm] -= h[p - shifts[nm]]
    
    return best_h, best_c5

def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
                    best_j = j
                    best_d = d_star
        if best_i >= 0:
            h[best_i] -= best_d
            h[best_j] += best_d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_chg_row(change, best_i, h, n, shifts)
            update_chg_row(change, best_j, h, n, shifts)
    return best_h, best_c5


def subgrad_hb(h_init, n, target, time_limit, max_iter=80000, alpha0=0.005, momentum=0.92):
    st = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_imp = 0
    alpha = alpha0
    for it in range(max_iter):
        if time.time() - st > time_limit:
            break
        if no_imp > 8000:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5v = c_max * dx
        if c5v < best_c5 - 1e-15:
            best_c5 = c5v
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                alpha = alpha * 0.95
        tol = max(c_max * 0.02, 1e-4)
        active = np.where(corr >= c_max - tol)[0]
        grad = np.zeros(n)
        for ki in active:
            k = int(ki) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad = grad / len(active)
        grad -= np.mean(grad)
        v = momentum * v - alpha * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c5v * (1.0 + 1e-7):
            h = h_new
        else:
            h_half = h + 0.5 * v
            h_half = project_feasible(h_half, target)
            sc_half = compute_c5(h_half)
            if sc_half < c5v * (1.0 + 1e-9):
                h = h_half
            else:
                alpha = alpha * 0.9
                v = v * 0.5
    return best_h, best_c5


def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
