To find this exactly:
1. Compute the values v_k = corr[k] and slopes s_k = g[k] for all k
2. We want to find δ that minimizes max_k (v_k + δ * s_k)
3. This is equivalent to finding the smallest δ such that the "upper envelope" of the lines {v_k + δ*s_k} is minimized

The upper envelope is a convex piecewise-linear function. Its minimum is at a vertex, which is the intersection of two lines.

Algorithm:
1. Sort lines by their value at δ=0: v_k
2. The active line at δ=0 is the one with max v_k
3. As δ increases, the active line changes when another line crosses it
4. Find the δ where the "second-highest" line first becomes the max (this is a candidate minimum)
5. Check if this is the true minimum (the slope of the upper envelope should be 0 or change from negative to positive at this point)

import numpy as np

# Global variables for SLSQP callbacks
_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def enhanced_cd_v2(h_init, n, target, time_limit, max_sweeps=30000,
                   n_active_start=3, n_active_end=8,
                   n_cand_start=30, n_cand_end=60,
                   max_no_improve=300, bisection_iters=14):
    """Enhanced coordinate descent with adaptive parameters."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        # Adaptive n_active
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_active = int(n_active_start + (n_active_end - n_active_start) * progress)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)
        
        # Active lags: top n_active by correlation value
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        # Compute gradient (subgradient approximation)
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Select candidate points
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        # Try all pairs (or a good subset)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        total_pairs = n_h * n_l
        
        # If too many pairs, sample; otherwise try all
        if total_pairs <= 300:
            pairs = [(int(high_set[ii]), int(low_set[jj])) 
                     for ii in range(n_h) for jj in range(n_l) 
                     if high_set[ii] != low_set[jj]]
        else:
            # Stratified sampling
            n_samples = min(total_pairs, 300)
            pairs = []
            for s in range(n_samples):
                ii = int(s * n_h / n_samples)
                jj = int(s * n_l / n_samples)
                pairs.append((int(high_set[ii]), int(low_set[jj])))
            # Add some random
            for s in range(50):
                ii = np.random.randint(n_h)
                jj = np.random.randint(n_l)
                pairs.append((int(high_set[ii]), int(low_set[jj])))
        
        for i, j in pairs:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g = change[j] - change[i]

            # Bisection for optimal delta
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5

def simulated_annealing(h_init, n, target, time_limit, t_start=0.1, t_end=0.001, 
                        n_iter_max=500000, cooling='exp'):
    """Simulated annealing on near-binary solutions."""
    start = time.time()
    h = h_init.copy().astype(float)
    target_sum = target
    best_h = h.copy()
    best_score = compute_c5(h)
    
    # Pre-compute the change matrix for fast evaluation
    # When we flip positions i and j (decrease h[i] by δ, increase h[j] by δ):
    # Δcorr[k] = δ * (change[j,k] - change[i,k])
    
    # For SA, we'll do small random perturbations
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    rng = np.random.RandomState(12345)
    
    curr_score = best_score
    t = t_start
    
    # For efficiency, track the correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    change = compute_change_matrix(h, n)
    
    for it in range(n_iter_max):
        if time.time() - start > time_limit:
            break
        
        # Update temperature
        if cooling == 'exp':
            t = t_start * (t_end / t_start) ** (it / max(n_iter_max, 1))
        else:
            t = t_start / (1.0 + 0.01 * it)
        
        # Random move: pick two random positions, exchange mass
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Choose step size
        delta = rng.uniform(0.01, 0.05)
        
        # Ensure feasibility after move
        delta = min(delta, h[i], 1.0 - h[j])
        if delta < 1e-6:
            continue
        
        g = change[j] - change[i]
        new_corr = corr + delta * g
        new_score = np.max(new_corr) * dx
        
        delta_score = new_score - curr_score
        
        if delta_score < 0 or rng.uniform() < np.exp(-delta_score / max(t, 1e-10)):
            # Accept
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target_sum)
            curr_score = new_score  # approximate
            corr = new_corr  # approximate (not exact after projection)
            update_change_row(change, i, h, n)
            update_change_row(change, j, h, n)
            
            # Recompute exact score periodically
            if it % 100 == 0:
                corr = np.correlate(h, 1.0 - h, mode='full')
                curr_score = float(np.max(corr)) * dx
                if curr_score < best_score:
                    best_score = curr_score
                    best_h = h.copy()
        else:
            # Reject
            pass
    
    return best_h, best_score

def thue_morse_h(n):
    """Generate Thue-Morse sequence of length n, mapped to [0,1]."""
    # Thue-Morse: t(n) = number of 1s in binary representation of n (mod 2)
    # This has exactly n/2 ones for even n
    tm = np.zeros(n)
    for i in range(n):
        # Count bits
        x = i
        cnt = 0
        while x:
            cnt += x & 1
            x >>= 1
        tm[i] = cnt % 2
    return tm

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change


def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]


def enhanced_cd(h_init, n, target, time_limit, max_sweeps=30000,

                n_cand_start=30, n_cand_end=65,

    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:

        else:
            consecutive_no_improve += 1

                break

        n_active = int(n_active_start + (n_active_end - n_active_start) * progress)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0

        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        best_i = -1
        best_j = -1

        n_l = len(low_set)

                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g = change[j] - change[i]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)

                else:

                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)

            approx_max = np.max(corr + delta_star * g) * dx

                best_j = j

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

def simulated_annealing(h_init, n, target, time_limit, t_start=0.08, t_end=0.0005,

    dx = 2.0 / n
    n_corr = 2 * n - 1

    rng = np.random.RandomState(rng_seed)
    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score

    corr = np.correlate(h, 1.0 - h, mode='full')
    change = compute_change_matrix(h, n)

    for it in range(n_iter_max):
        if time.time() - start > time_limit:
            break
        if it % 2000 == 0 and it > 0:
            frac = it / max(n_iter_max, 1)
            t = t_start * (t_end / t_start) ** frac

        if it % 50 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            curr_score = float(np.max(corr)) * dx
            if curr_score < best_score:
                best_score = curr_score
                best_h = h.copy()

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(0.005, 0.04)
        delta = min(delta, h[i], 1.0 - h[j])
        if delta < 1e-6:
            continue

        g = change[j] - change[i]
        new_corr = corr + delta * g
        new_score = float(np.max(new_corr)) * dx
        delta_score = new_score - curr_score

        if delta_score < 0 or rng.uniform() < np.exp(-delta_score / max(t, 1e-10)):

            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = new_corr
            curr_score = new_score
            update_change_row(change, i, h, n)
            update_change_row(change, j, h, n)

    return best_h, best_score


def thue_morse_h(n):
    tm = np.zeros(n)
    for i in range(n):
        x = i
        cnt = 0
        while x:
            cnt += x & 1
            x >>= 1
        tm[i] = cnt % 2
    return tm


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                      500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if tleft() > 35:
        h_cd3, c5_cd3 = enhanced_cd(best_h, n0, target0,
                                     time_limit=min(20, tleft() * 0.6),
                                     max_sweeps=8000,
                                     n_active_start=6, n_active_end=10,
                                     n_cand_start=50, n_cand_end=80,
                                     max_no_improve=120,
                                     bisection_iters=14)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
