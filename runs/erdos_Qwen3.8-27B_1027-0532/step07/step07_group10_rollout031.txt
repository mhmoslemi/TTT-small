    lr = base_lr
    no_improve = 0

            break
        if no_improve > 10000:
            break
        
        corr = np.correlate(h, 1-h, mode='full')
        c5 = np.max(corr) * dx

            best_score = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_corr = np.max(corr)
        active = np.where(corr >= max_corr - 1e-10)[0]
        
        # Compute gradient (averaged over active shifts)
        mid = n - 1
        grad = np.zeros(n)
        idxs = np.arange(n)

            h = h_try
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_score

from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)

def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = idxs + k
        wv = idxs - k
        mask_p = (wp >= 0) & (wp < n)
        mask_m = (wv >= 0) & (wv < n)
        grad[mask_p] += weights[k_idx] * (1.0 - x[wp[mask_p]])
        grad[mask_m] -= weights[k_idx] * x[wv[mask_m]]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):

    _g_n = n
    _g_target = target

    cur_score = compute_c5(h_cur)

def nesterov_subgrad(h_init, n, target, time_limit, base_lr=0.01, 
                     n_active=3, momentum_init=0.9):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 1
    lr = base_lr
    no_improve = 0
    
    while time.time() - start < time_limit:
        if no_improve > 50000:
            break
        
        y = h + (t / (t + 1 + np.sqrt(1 + 4*t*t))) * (h - h_prev)
        # Actually the standard FISTA momentum is:
        # beta_t = (1 - sqrt(1 + 4*t*t)) / (2*t)
        # y = h + beta_t * (h - h_prev)
        beta_t = (1.0 - np.sqrt(1.0 + 4.0*t*t)) / (2.0*t)
        y = h + beta_t * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        c5_y = np.max(corr_y) * dx
        
        # Top active shifts
        top_idx = np.argsort(corr_y)[-n_active:]
        
        grad = np.zeros(n)
        for si in top_idx:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - y[idxs[v1] - k])
            grad[v2] -= y[idxs[v2] + k]
        grad /= n_active
        grad -= np.mean(grad)
        
        # Gradient step
        h_new = y - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if c5_new <= c5_y + 1e-10:
            # Accept
            h_prev = h
            h = h_new
            t += 1
            lr = max(base_lr / np.sqrt(t), 1e-6)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)

    weights /= np.sum(weights)

                if delta_hi - delta_lo < 1e-13:

                g = low_rows[jj] - high_rows[ii]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-15:
                    best_trial_c5 = float(approx_max) * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def proj_grad_min(h_init, n, target, time_limit, base_lr=0.05, n_active=5,
                  decay_start=10000, decay_period=5000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = base_lr
    no_improve = 0

    for step in range(1000000):
        if time.time() - start > time_limit:
            break
        if no_improve > decay_start and lr < 1e-7:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idx = np.argsort(corr)[-n_active:]
        max_corr = corr[top_idx[-1]]
        weights = np.exp(-15.0 * (max_corr - corr[top_idx]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_idx):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad /= n_active
        grad -= np.mean(grad)

        h_try = h - lr * grad
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, target)

        c5_try = compute_c5(h_try)
        if c5_try < c5_val - 1e-15:
            h = h_try
        else:
            lr *= 0.5
            if no_improve > decay_start:
                lr = base_lr * np.exp(-no_improve / decay_period)
            if lr < 1e-8:
                break

    return best_h, best_c5


def nesterov_cd(h_init, n, target, time_limit, base_lr=0.03, n_active=4):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    h_prev = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 1
    lr = base_lr
    no_improve = 0

    for step in range(1000000):
        if time.time() - start > time_limit:
            break
        if no_improve > 50000 and lr < 1e-7:
            break

        beta = (1.0 - np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * t)
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr_y = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr_y)[-n_active:]
        max_corr = corr_y[top_idx[-1]]
        weights = np.exp(-10.0 * (max_corr - corr_y[top_idx]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_idx):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - y[idxs[v1] - k])
            grad[v2] -= wi * y[idxs[v2] + k]
        grad /= n_active
        grad -= np.mean(grad)

        h_new = y - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if c5_new <= compute_c5(y) + 1e-10:
            h_prev = h
            h = h_new
            t += 1
            lr = max(base_lr / np.sqrt(t), 1e-7)
        else:
            lr *= 0.5
            h_prev = h
            if lr < 1e-8:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_active_cd(h_init, n, target, time_limit, n_cand=40, n_active=4):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    for sweep in range(30000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 400:
            break

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[-1]]
        weights = np.exp(-20.0 * (c_max - corr[top_shifts]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, s_idx in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad /= n_active
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        high_rows = np.zeros((n_cand, n_corr))
        low_rows = np.zeros((n_cand, n_corr))
        for idx in range(n_cand):
            p = int(high_set[idx])
            pos_mask = (p + shifts >= 0) & (p + shifts < n)
            neg_mask = (p - shifts >= 0) & (p - shifts < n)
            high_rows[idx, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
            high_rows[idx, neg_mask] = -h[p - shifts[neg_mask]]
        for idx in range(n_cand):
            p = int(low_set[idx])
            pos_mask = (p + shifts >= 0) & (p + shifts < n)
            neg_mask = (p - shifts >= 0) & (p - shifts < n)
            low_rows[idx, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
            low_rows[idx, neg_mask] = -h[p - shifts[neg_mask]]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(n_cand):
            i = int(high_set[ii])
            delta_lo_i = h[i] - 1.0
            delta_hi_i = h[i]
            for jj in range(n_cand):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(delta_lo_i, -h[j])
                delta_hi = min(delta_hi_i, 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = low_rows[jj] - high_rows[ii]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(11):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-15:
                    best_trial_c5 = float(approx_max) * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, best_n, target0,
                                [80, 300, 1000], 600, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        h_cd, c5_cd = fast_pair_cd(best_h, best_n, target0,
                                   time_limit=min(80, tleft() * 0.5),
                                   max_sweeps=15000,
                                   n_cand_start=20, n_cand_end=60,
                                   max_no_improve=300)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n1 = 800
    if tleft() > 200:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 150:
            h1_cd, c5_1cd = fast_pair_cd(h1, n1, target1,
                                         time_limit=min(200, tleft() * 0.55),
                                         max_sweeps=25000,
                                         n_cand_start=25, n_cand_end=70,
                                         max_no_improve=400)
            if c5_1cd < best_score:
                best_score = c5_1cd
                best_h = h1_cd
                best_n = n1
                h1 = best_h

        if tleft() > 100:
            h1_pg, c5_1pg = proj_grad_min(h1, n1, target1,
                                          time_limit=min(80, tleft() * 0.5),
                                          base_lr=0.04, n_active=5)
            if c5_1pg < best_score:
                best_score = c5_1pg
                best_h = h1_pg
                best_n = n1
                h1 = best_h

        if tleft() > 80:
            h1_cd2, c5_1cd2 = multi_active_cd(h1, n1, target1,
                                              time_limit=min(60, tleft() * 0.6),
                                              n_cand=35, n_active=4)
            if c5_1cd2 < best_score:
                best_score = c5_1cd2
                best_h = h1_cd2
                best_n = n1

        if tleft() > 60:
            h1_sls, sc1s = slsqp_round(best_h, best_n, target1,
                                       [500, 2000], 800, 1e-14)
            if sc1s < best_score:
                best_score = sc1s
                best_h = h1_sls
                best_n = n1

    n2 = 1000
    if tleft() > 300:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        if tleft() > 250:
            h2_cd, c5_2cd = fast_pair_cd(h2, n2, target2,
                                         time_limit=min(250, tleft() * 0.55),
                                         max_sweeps=20000,
                                         n_cand_start=30, n_cand_end=80,
                                         max_no_improve=400)
            if c5_2cd < best_score:
                best_score = c5_2cd
                best_h = h2_cd
                best_n = n2
                h2 = best_h

        if tleft() > 120:
            h2_pg, c5_2pg = proj_grad_min(h2, n2, target2,
                                          time_limit=min(100, tleft() * 0.5),
                                          base_lr=0.03, n_active=6)
            if c5_2pg < best_score:
                best_score = c5_2pg
                best_h = h2_pg
                best_n = n2
                h2 = best_h

        if tleft() > 80:
            h2_cd2, c5_2cd2 = multi_active_cd(h2, n2, target2,
                                              time_limit=min(60, tleft() * 0.6),
                                              n_cand=40, n_active=5)
            if c5_2cd2 < best_score:
                best_score = c5_2cd2
                best_h = h2_cd2
                best_n = n2

        if tleft() > 60:
            h2_sls, sc2s = slsqp_round(best_h, best_n, target2,
                                       [1000, 5000], 1000, 1e-14)
            if sc2s < best_score:
                best_score = sc2s
                best_h = h2_sls
                best_n = n2

    if tleft() > 80:
        h_nest, c5_nest = nesterov_cd(best_h, best_n, best_n / 2.0,
                                      time_limit=min(60, tleft() * 0.6),
                                      base_lr=0.02, n_active=4)
        if c5_nest < best_score:
            best_score = c5_nest
            best_h = h_nest

    if tleft() > 80:
        h_sls_final, sc_f = slsqp_round(best_h, best_n, best_n / 2.0,
                                        [3000, 10000, 30000], 1500, 1e-14)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_sls_final

    if tleft() > 60:
        h_cd_final, c5_cf = fast_pair_cd(best_h, best_n, best_n / 2.0,
                                         time_limit=min(50, tleft() * 0.7),
                                         max_sweeps=15000,
                                         n_cand_start=40, n_cand_end=90,
                                         max_no_improve=300)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cd_final

    if tleft() > 80:
        for noise_scale in [0.01, 0.03, 0.08]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_n / 2.0)
            h_ref, sc = slsqp_round(h_pert, best_n, best_n / 2.0,
                                    [300, 2000, 10000], 500, 1e-13)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 40:
        h_pg_last, c5_pl = proj_grad_min(best_h, best_n, best_n / 2.0,
                                         time_limit=min(30, tleft() * 0.6),
                                         base_lr=0.02, n_active=8)
        if c5_pl < best_score:
            best_score = c5_pl
            best_h = h_pg_last

    if tleft() > 30:
        h_cd_last, c5_cl = fast_pair_cd(best_h, best_n, best_n / 2.0,
                                        time_limit=min(20, tleft() * 0.7),
                                        max_sweeps=8000,
                                        n_cand_start=50, n_cand_end=100,
                                        max_no_improve=150)
        if c5_cl < best_score:
            best_score = c5_cl
            best_h = h_cd_last

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)
    if final_score < best_score:
        best_score = final_score
        best_h = final_h

    return best_h, best_score, best_n
