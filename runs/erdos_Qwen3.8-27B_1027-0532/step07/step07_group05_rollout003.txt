For the CD method, I'll make some improvements over the previous version:
1. Use the smoothed gradient (with softmax weights) instead of just top-n
2. Try more candidate pairs
3. Use a better step size (not just bisection on the max, but actual line search)

Actually, for the CD method, the key operation is: given the current correlation vector and a pair (i,j) where we decrease h[i] by δ and increase h[j] by δ, find the optimal δ.

The change in correlation is:
Δcorr = δ * (change[j] - change[i])

where change[i] is the vector of changes to corr when h[i] is increased by 1 (it's the i-th row of the change matrix).

The new max is max(corr + δ * g) where g = change[j] - change[i].

To find the optimal δ ∈ [0, δ_max], we need to minimize max(corr + δ*g) over δ.

Since max(corr + δ*g) is a convex piecewise-linear function of δ, the minimum occurs at a breakpoint or at the boundary. The breakpoints occur where two correlation values are equal:
corr[a] + δ*g[a] = corr[b] + δ*g[b]
δ = (corr[b] - corr[a]) / (g[a] - g[b]) (when g[a] ≠ g[b])

So the optimal δ is either 0, δ_max, or one of the breakpoints within [0, δ_max].

We can find this by:
1. Computing all breakpoints (O(n²) for n correlation values)
2. Filtering to those in [0, δ_max]
3. Evaluating the objective at each candidate point
4. Taking the minimum

For n_corr = 2n-1 ≈ 800 (for n=400), computing all breakpoints is O(800²) = 640K, which is a bit much for doing this every iteration. But we only need to do it for the top ~20 correlation values (the ones near the maximum), reducing it to O(20²) = 400 per iteration.

Actually, a simpler approach: since the function is convex and piecewise linear, we can use the fact that the minimum of a convex PL function over an interval is at a "kink" point. We can find the kink points by:
1. Sorting the values corr + δ*g at δ=0 and δ=δ_max
2. The kink points are where the ordering changes

But this is still complex. The bisection approach from the previous code is actually quite efficient and works well. Let me stick with it but make a few improvements.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(x, target_sum):
    n = len(x)
    x = np.clip(x.astype(float), -10.0, 10.0)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) > 1e-14:
        free = (y > 1e-10) & (y < 1.0 - 1e-10)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y


def compute_smoothed_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    corr_max = np.max(corr)
    shifted = np.clip(alpha * (corr - corr_max), -50.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for m in active:
        L = m - (n - 1)
        w = dx * weights[m]
        if L >= 0:
            if n - L > 0:
                grad[:n - L] += w * (1.0 - h[L:])
            if L < n:
                grad[L:] -= w * h[:n - L]
        else:
            if n + L > 0:
                grad[-L:] += w * (1.0 - h[:n + L])
            if n + L > 0:
                grad[:n + L] -= w * h[-L:]
    return grad, weights, corr


def proximal_gradient_optimize(h_init, target, time_limit, alpha_start=50.0, alpha_end=5000.0, n_alphas=8, lr_init=0.05, max_iter=50000):
    n = len(h_init)
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    for alpha in alphas:
        lr = lr_init
        momentum = np.zeros(n)
        beta_mom = 0.9
        for it in range(max_iter):
            if time.time() - start > time_limit:
                break
            grad, _, _ = compute_smoothed_grad(h, alpha)
            momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
            h_trial = h - lr * momentum
            h_trial = project_feasible(h_trial, target)
            score = compute_c5(h_trial)
            if score < best_score - 1e-15:
                best_score = score
                best_h = h_trial.copy()
                h = h_trial
                lr = min(lr * 1.05, 0.5)
            else:
                lr *= 0.7
                if lr < 1e-8:
                    break
    return best_h, best_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change


def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]


def enhanced_cd(h_init, n, target, time_limit, max_sweeps=20000,
                n_active=3, n_cand_start=20, n_cand_end=60,
                max_no_improve=300, bisection_iters=12):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        max_pairs = 400
        if n_h * n_l <= max_pairs:
            pair_list = []
            for ii in range(n_h):
                for jj in range(n_l):
                    i_val = int(high_set[ii])
                    j_val = int(low_set[jj])
                    if i_val != j_val:
                        pair_list.append((i_val, j_val))
        else:
            pair_list = []
            for frac in [0.0, 0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.85, 1.0]:
                hi = int(frac * (n_h - 1))
                lo = int(frac * (n_l - 1))
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pair_list.append((iv, jv))
            while len(pair_list) < max_pairs:
                ii = int(np.random.randint(n_h))
                jj = int(np.random.randint(n_l))
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))

        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue

            g = change[j] - change[i]

            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur
