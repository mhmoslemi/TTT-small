def momentum_subgradient(h_init, n, target, time_limit, n_steps=50000, 
                         step_scale=0.003, momentum=0.9, n_top=5):
    """Subgradient descent with Nesterov momentum."""
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Compute weighted subgradient
        top_idx = np.argsort(corr)[-n_top:]
        weights = corr[top_idx]
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for si in range(n_top):
            s_idx = int(top_idx[si])
            k = s_idx - mid
            w = weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Nesterov momentum
        v = momentum * v + grad
        h_try = h - eta * v
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        
        if sc < c5_val:
            h = h_try
        else:
            # Try without momentum (just gradient)
            h_try2 = h - eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
                v = 0.5 * v  # reduce momentum
            else:
                eta *= 0.95
                v *= 0.5
        
        # Periodically reset momentum
        if step % 1000 == 0 and step > 0:
            v *= 0.5
        # Adapt lr
        if step % 2000 == 0 and step > 0:
            eta = max(step_scale * max(0.2, 1.0 - step / (2.0 * n_steps)), 1e-7)
    
    return best_h, best_c5

def trust_constr_refine(h_start, n, target, alphas, maxiter_per_alpha=500):
    """Use trust-constr method which handles constraints natively."""
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:

                return smooth_objective(x, alpha)
            
            from scipy.optimize import LinearConstraint, trust_constr
            A = np.ones((1, n))
            lin_cons = LinearConstraint(A, target, target)
            res = trust_constr(obj, h_cur, bounds=bounds, constraints=[lin_cons],
                               maxiter=maxiter_per_alpha, 
                               gtol=1e-10, xtol=1e-12)
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def block_exchange(h_init, n, target, time_limit, block_size=4, max_sweeps=20000,

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

The key insight from the previous solution is the **change matrix** approach:

def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):

        h = np.clip(h, 0.0, 1.0)

def compute_change_matrix(h, n):
    """Pre-compute how each coordinate affects each correlation shift."""
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_objective(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)


def constraint_fun(x):
    return np.sum(x) - _g_target


def constraint_jac(x):
    return np.ones(_g_n)


def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_objective, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def efficient_cd(h_init, n, target, time_limit,
                 max_sweeps=100000, n_cand_start=15, n_cand_end=80,
                 max_no_improve=500, n_top_shifts=3,
                 use_all_shifts=False, softmax_alpha=150.0,
                 n_binary_search=12, max_pairs=25,
                 random_escape_interval=200, escape_scale=0.02):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    rng = np.random.RandomState(12345)

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if use_all_shifts:
            corr_shifted = (corr - np.max(corr)) * softmax_alpha
            exp_corr = np.exp(np.clip(corr_shifted, -500, 0))
            weights_full = exp_corr / np.sum(exp_corr)
            threshold = weights_full.max() * 5e-4
            active = np.where(weights_full > threshold)[0]
            active = active[:30]
            grad = np.zeros(n)
            for s_idx in active:
                k = int(s_idx) - mid
                w = weights_full[s_idx]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
            grad -= np.mean(grad)
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            shift_weights = corr[top_shifts]
            sw_sum = shift_weights.sum()
            if sw_sum < 1e-15:
                sw_sum = 1.0
            shift_weights = shift_weights / sw_sum
            grad = np.zeros(n)
            for si in range(n_top_shifts):
                s_idx = int(top_shifts[si])
                k = s_idx - mid
                w = shift_weights[si]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
            grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        n_check = min(current_ncand, max_pairs)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_check]
        high_set = sorted_idx[-n_check:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                quick_max = np.max(corr + delta_hi * g)
                if quick_max * dx >= best_trial:
                    continue

                lo_d = delta_lo
                hi_d = delta_hi
                for _ in range(n_binary_search):
                    mid_d = 0.5 * (lo_d + hi_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid * dx < best_trial - 1e-14:
                        lo_d = mid_d
                    else:
                        hi_d = mid_d

                delta_star = 0.5 * (lo_d + hi_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial - 1e-14:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 3:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
        else:
            if no_improve > 0 and no_improve % random_escape_interval == 0:
                idx1 = rng.randint(n)
                idx2 = rng.randint(n)
                tries = 0
                while idx2 == idx1 and tries < 10:
                    idx2 = rng.randint(n)
                    tries += 1
                if idx1 != idx2:
                    d = escape_scale * min(h[idx1], 1.0 - h[idx2])
                    if d > 1e-10:
                        h[idx1] -= d
                        h[idx2] += d
                        h = np.clip(h, 0.0, 1.0)
                        h = project_feasible(h, target)
                        update_change_row(change, idx1, h, n, shifts)
                        update_change_row(change, idx2, h, n, shifts)
                        changes_since_rebuild += 2
                        if changes_since_rebuild > n // 3:
                            change = compute_change_matrix(h, n)
                            changes_since_rebuild = 0

    return best_h, best_c5


def momentum_subgrad(h_init, n, target, time_limit, n_steps=50000,
                     step_scale=0.003, momentum=0.9, n_top=5,
                     use_softmax=False, softmax_alpha=200.0):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        if use_softmax:
            corr_sh = (corr - np.max(corr)) * softmax_alpha
            exp_c = np.exp(np.clip(corr_sh, -500, 0))
            w_full = exp_c / np.sum(exp_c)
            thr = w_full.max() * 5e-4
            active = np.where(w_full > thr)[0][:n_top]
            w_vals = w_full[active]
        else:
            active = np.argsort(corr)[-n_top:]
            w_vals = corr[active]
            w_vals = w_vals / w_vals.sum()

        grad = np.zeros(n)
        for si in range(len(active)):
            s_idx = int(active[si])
            k = s_idx - mid
            w = w_vals[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        v_new = momentum * v + grad
        h_try = h - eta * v_new
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
            v = v_new
        else:
            h_try2 = h - eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
                v *= 0.5
            else:
                eta *= 0.97
                v *= 0.8

        if step % 2000 == 0 and step > 0:
            eta = max(step_scale * max(0.3, 1.0 - step / (2.0 * n_steps)), 1e-7)
            v *= 0.3

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def binary_fix_refine(h_init, n, target, threshold=0.05, time_limit=30.0):
    start = time.time()
    h = h_init.copy().astype(float)
    fixed_zero = h < threshold
    fixed_one = h > 1.0 - threshold
    free = ~(fixed_zero | fixed_one)

    if free.sum() < 10 or free.sum() >= n:
        return h, compute_c5(h)

    h_fixed = h.copy()
    h_fixed[fixed_zero] = 0.0
    h_fixed[fixed_one] = 1.0

    free_idx = np.where(free)[0]
    n_free = len(free_idx)
    free_target = target - np.sum(h_fixed[~free])

    if free_target < 0 or free_target > n_free:
        return h, compute_c5(h)

    x0 = h[free].copy()
    bounds_free = [(0.0, 1.0)] * n_free

    def obj_free(x):
        h_temp = h_fixed.copy()
        h_temp[free] = x
        return smooth_objective(h_temp, 2000.0)

    def cons_free(x):
        return x.sum() - free_target

    def jac_free(x):
        return np.ones(n_free)

    global _g_n, _g_target
    _g_n = n_free
    _g_target = free_target

    try:
        res = minimize(obj_free, x0, method='SLSQP', bounds=bounds_free,
                       constraints=[{'type': 'eq', 'fun': cons_free, 'jac': jac_free}],
                       options={'maxiter': 300, 'ftol': 1e-12},
                       args=())
        h_new = h_fixed.copy()
        h_new[free] = res.x
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        sc_old = compute_c5(h)
        if sc_new < sc_old:
            return h_new, sc_new
    except Exception:
        pass

    return h, compute_c5(h)

                                          time_limit=min(80, tleft() * 0.4),
                                          max_sweeps=30000,
                                          n_cand_start=10, n_cand_end=30,
                                          max_no_improve=300,
                                          n_top_shifts=2,
                                          use_all_shifts=True, softmax_alpha=80.0,
                                          max_pairs=15, n_binary_search=10)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

        if tleft() > 80:
            h_pol3, sc3 = slsqp_refine(best_h, cur_n, cur_target,
                                        [200, 1000, 5000], 600, 1e-14)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    if tleft() > 60:
        h_cd3, c5_cd3 = efficient_cd(best_h, cur_n, cur_target,
                                      time_limit=min(40, tleft() * 0.5),
                                      max_sweeps=30000,
                                      n_cand_start=20, n_cand_end=60,
                                      max_no_improve=250,
                                      n_top_shifts=4,
                                      use_all_shifts=True, softmax_alpha=100.0,
                                      max_pairs=25, n_binary_search=12)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 30:
        h_bf2, sc_bf2 = binary_fix_refine(best_h, cur_n, cur_target,
                                           threshold=0.02, time_limit=20.0)
        if sc_bf2 < best_score:
            best_score = sc_bf2
            best_h = h_bf2

    if tleft() > 30:
        h_final, sc_final = slsqp_refine(best_h, cur_n, cur_target,
                                          [10000, 50000], 1000, 1e-14)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
