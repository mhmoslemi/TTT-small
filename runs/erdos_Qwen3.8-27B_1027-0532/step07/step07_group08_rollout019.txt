import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def simulated_annealing(h_init, n, target, time_limit,

        if time.time() - start_time > time_limit:
            break

        progress = step / total_steps
        # Geometric cooling
        temp = temp_start * (temp_end / temp_start) ** progress
        n_pert = max(n_pert_end, int(n_pert_start * (1 - progress) + n_pert_end * progress))

        idx = rng.choice(n, size=n_pert, replace=False)

            current_c5 = trial_c5
            if current_c5 < best_c5:

                best_h = h.copy()
        elif temp > 1e-12:
            ratio = np.exp(-delta / temp)
            if rng.random() < ratio:
                h = h_trial
                current_c5 = trial_c5

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=20, n_cand_end=80, max_no_improve=300,
            n_active_shifts=5, reheat_interval=100, reheat_scale=0.03,
            n_bisect=8):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        if sweeps_since_improve > reheat_interval and sweeps_since_improve % 12 == 0:
            rng = np.random.RandomState(sweep * 31 + 17)
            n_pert = max(5, n // 30)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 0.002:
                h = h_pert
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                continue
        c_max = float(np.max(corr))
        threshold = c_max - 0.15 * max(c_max, 0.001)
        active_shifts = np.where(corr >= threshold)[0]
        if len(active_shifts) > n_active_shifts:
            active_shifts = np.argsort(corr)[-n_active_shifts:]
        n_act = len(active_shifts)
        w_active = corr[active_shifts]
        w_sum = np.sum(w_active)
        if w_sum < 1e-15:
            weights = np.ones(n_act) / n_act
        else:
            weights = w_active / w_sum
        grad = np.zeros(n)
        for s_idx, w in zip(active_shifts, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        k_star_idx = int(np.argmax(corr))
        multi_score = np.zeros(n)
        for s_idx, w in zip(active_shifts, weights):
            multi_score += w * change[:, s_idx]
        n_h = len(high_set)
        n_l = len(low_set)
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                if g[k_star_idx] > 0:
                    continue
                ms_delta = multi_score[j] - multi_score[i]
                if ms_delta > 0.25 * abs(g[k_star_idx]) + 1e-10:
                    continue
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    am = vals >= f_mid - 1e-14
                    if am.any():
                        f_prime = np.max(g[am])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-14:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5


def simulated_annealing(h_init, n, target, time_limit,
                        temp_start=0.005, temp_end=0.00005,
                        n_pert_start=10, n_pert_end=2, seed=777):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    total_steps = 20000
    for step in range(total_steps):
        if time.time() - start_time > time_limit:
            break
        progress = step / total_steps
        temp = temp_start * (temp_end / temp_start) ** progress
        n_pert = max(n_pert_end, int(n_pert_start * (1 - progress) + n_pert_end * progress))
        idx = rng.choice(n, size=n_pert, replace=False)
        h_trial = h.copy()
        h_trial[idx] += rng.randn(n_pert) * temp * 2.0
        h_trial = project_feasible(h_trial, target)
        trial_c5 = compute_c5(h_trial)
        delta = trial_c5 - current_c5
        if delta < 0:
            h = h_trial
            current_c5 = trial_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        elif temp > 1e-12:
            ratio = np.exp(-delta / temp)
            if rng.random() < ratio:
                h = h_trial
                current_c5 = trial_c5
    return best_h, best_c5


def block_anneal(h_init, n, target, time_limit, seed=456):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    total_steps = 12000
    for step in range(total_steps):
        if time.time() - start_time > time_limit:
            break
        progress = step / total_steps
        temp = 0.004 * (0.0001 / 0.004) ** progress
        block_size = rng.randint(3, max(4, n // 15))
        start_idx = rng.randint(0, n - block_size)
        h_trial = h.copy()
        h_trial[start_idx:start_idx + block_size] += rng.randn(block_size) * temp
        h_trial += rng.randn(n) * temp * 0.2
        h_trial = project_feasible(h_trial, target)
        trial_c5 = compute_c5(h_trial)
        delta = trial_c5 - current_c5
        if delta < 0:
            h = h_trial
            current_c5 = trial_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        elif temp > 1e-12:
            ratio = np.exp(-delta / temp)
            if rng.random() < ratio:
                h = h_trial
                current_c5 = trial_c5
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
