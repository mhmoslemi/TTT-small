    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _power_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.sum(corr ** p) ** (1.0 / p))


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, method='ls'):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            if method == 'ls':
                res = minimize(_smooth_obj, h_cur, args=(alpha,),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            else:
                res = minimize(_power_obj, h_cur, args=(alpha,),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, shifts):
    change[p, :] = 0.0

    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def interpolate_to(h_src, n_dst, target_dst):

                update_change_row(change, best_bs_j + tt, h, shifts)

    return best_h, best_c5


def subgradient_step(h, n, target, lr, n_active_shifts=5, tol=1e-8):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    tol_val = max(tol, tol * max_corr)
    active = np.where(corr >= max_corr - tol_val)[0]
    n_act = min(len(active), n_active_shifts)
    active = active[np.argsort(corr[active])[-n_act:]]

    grad = np.zeros(n)
    for k_idx in active:
        k = int(k_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad /= len(active)
    grad -= np.mean(grad)

    h_new = project_feasible(h - lr * grad, target)
    return h_new
