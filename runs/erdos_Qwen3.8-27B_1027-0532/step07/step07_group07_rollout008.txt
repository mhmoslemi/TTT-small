    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14, time_limit=None, start_time=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        def obj(x):
            return smooth_obj(x, alpha)
        def grad(x):
            return smooth_obj_grad(x, alpha)
        try:
            res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def block_cd(h_start, n, target, time_limit, block_size=40, n_passes=5, alpha=500):
    start = time.time()
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_blocks = (n + block_size - 1) // block_size
    
    for sweep in range(n_passes):
        if time.time() - start > time_limit:
            break
        for b in range(n_blocks):
            if time.time() - start > time_limit:
                break
            lo = b * block_size
            hi = min(lo + block_size, n)
            bs = hi - lo
            blk_target = target - (np.sum(h) - np.sum(h[lo:hi]))
            
            def blk_obj(x):
                h_trial = h.copy()
                h_trial[lo:hi] = x
                return smooth_obj(h_trial, alpha)
            
            def blk_grad(x):
                h_trial = h.copy()
                h_trial[lo:hi] = x
                return smooth_obj_grad(h_trial, alpha)[lo:hi]
            
            def blk_con(x):
                return np.sum(x) - blk_target
            
            try:
                res = minimize(blk_obj, h[lo:hi].copy(), jac=blk_grad, method='SLSQP',
                              bounds=[(0.0, 1.0)] * bs,
                              constraints=[{'type': 'eq', 'fun': blk_con}],
                              options={'maxiter': 100, 'ftol': 1e-12})
                h[lo:hi] = project_feasible(res.x, blk_target)
            except Exception:
                pass
            h = project_feasible(h, target)
            c5_val = compute_c5(h)
            if c5_val < best_c5 - 1e-15:
                best_c5 = c5_val
                best_h = h.copy()
    return best_h, best_c5

def pair_cd_improved(h_init, n, target, time_limit, max_sweeps=50000,
                      n_cand_start=30, n_cand_end=80, top_k_eval=60,
                      max_no_improve=500, n_shifts_grad=3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    change = build_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        
        # Multi-shift weighted gradient
        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx]  # weight by correlation value
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        # Score using weighted combination of top shifts
        score_mat = np.zeros((len(high_set), len(low_set)))
        for s_idx in top_shifts:
            w = corr[s_idx]
            ch_h = change[high_set, int(s_idx)]
            ch_l = change[low_set, int(s_idx)]
            score_mat += w * (ch_l[None, :] - ch_h[:, None])
        
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for tf in top_flat:
            ii = int(tf // len(low_set))
            jj = int(tf % len(low_set))
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            
            g = change[j] - change[i]
            
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def stochastic_search(h_init, n, target, time_limit, max_steps=20000):
    start = time.time()
    rng = np.random.RandomState(777)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break
        
        scale = 0.03 * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.choice([3, 5, 10, 15]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        
        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 150:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0
    
    return best_h, best_c5
