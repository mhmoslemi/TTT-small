def project_feasible(h, target_sum):

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_grad_smooth(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        if w[s] < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[s] * h[idxs[v2] + k]
    return grad


def frank_wolfe(h_init, n, target, time_limit, alpha0=50.0, alpha_final=5000.0,
                max_iter=20000, use_ls=True):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    int_target = int(target)
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        alpha = alpha0 * (alpha_final / alpha0) ** (float(t) / max(1, max_iter))
        g = compute_grad_smooth(h, alpha)
        sorted_idx = np.argsort(g)
        s = np.zeros(n)
        s[sorted_idx[:int_target]] = 1.0
        if use_ls:
            gamma = 1.0
            f_curr = compute_c5(h)
            for _ in range(15):
                h_trial = h + gamma * (s - h)
                h_trial = project_feasible(h_trial, target)
                f_trial = compute_c5(h_trial)
                if f_trial <= f_curr - 1e-16:
                    break
                gamma *= 0.5
            h = h_trial
        else:
            gamma = 2.0 / (t + 2.0)
            h = h + gamma * (s - h)
            h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5

        s_vec = h - h_prev

        yy = np.dot(y_vec, y_vec)
        ss = np.dot(s_vec, s_vec)
        if yy > 1e-14:
            if sy > 1e-14:
                step = sy / yy
            else:
                step = ss / yy
        else:
            step = 1.0 / alpha
        step = np.clip(step, 0.005, 5.0)
        h_new = h - step * g
        h_new = np.clip(h_new, 0.0, 1.0)

        if c5_val < best_c5:

        g_prev = g
        h = h_new

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                                   time_limit=min(50, tleft() * 0.5),
                                   max_sweeps=10000,
                                   n_cand_start=30, n_cand_end=90,
                                   max_no_improve=250,
                                   n_active_shifts=5,
                                   reheat_interval=60, reheat_scale=0.02)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    if tleft() > 40:
        deadline = time.time() + min(25, tleft() * 0.5)
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                             [5000, 20000, 50000],
                                             600, 1e-14, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
