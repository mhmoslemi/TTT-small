import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=40, n_cand_end=200, n_shifts=5,
            max_no_improve=800, anneal_temp=0.0, anneal_decay=0.999):
    """Improved coordinate descent with vectorized evaluation."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        # Use top shifts with softmax weighting
        top_k = min(n_shifts, n_corr)
        top_shifts = np.argsort(corr)[-top_k:]
        # Softmax weighting for smoother gradients
        max_c = corr[top_shifts].max()
        weights = np.exp((corr[top_shifts] - max_c) * 5.0)
        weights /= weights.sum()

        # Compute gradient from weighted shifts
        grad = np.zeros(n)
        for si in range(top_k):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Determine candidate sets
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        # Vectorized evaluation of all pairs
        n_h = len(high_set)
        n_l = len(low_set)
        
        if n_h * n_l > 10000:
            # Too many pairs, subsample
            rng = np.random.RandomState(sweep)
            if n_h > 100:
                high_set = high_set[rng.choice(n_h, 100, replace=False)]
                n_h = 100
            if n_l > 100:
                low_set = low_set[rng.choice(n_l, 100, replace=False)]
                n_l = 100

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                
                # Bisection for optimal delta
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 500.0
                    if np.random.random() < np.exp(-max(delta_obj, -500) / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 2:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.97, 1e-10)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new

    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    cur_n = n0
    cur_target = target0

    # Phase 1: Quick SLSQP polish at original resolution
    tl = time_remaining(start_time, budget_s)
    if tl > 100:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [50, 200, 1000], 500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Coordinate descent at original resolution
    tl = time_remaining(start_time, budget_s)
    if tl > 200:
        h_cd, c5_cd = pair_cd(best_h, cur_n, cur_target,
                              time_limit=min(120, tl * 0.45),
                              max_sweeps=50000,
                              n_cand_start=40, n_cand_end=160,
                              n_shifts=4, max_no_improve=800,
                              anneal_temp=0.015, anneal_decay=0.9995)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Multi-shift CD
    tl = time_remaining(start_time, budget_s)
    if tl > 100:
        h_mc, c5_mc = multi_shift_cd(best_h, cur_n, cur_target,
                                     time_limit=min(60, tl * 0.4),
                                     max_sweeps=20000,
                                     n_cand=80, n_top=8)
        if c5_mc < best_score:
            best_score = c5_mc
            best_h = h_mc

    # Phase 4: SLSQP polish
    tl = time_remaining(start_time, budget_s)
    if tl > 80:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [3000, 8000], 800, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Move to higher resolution n=800
    tl = time_remaining(start_time, budget_s)
    if tl > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        tl = time_remaining(start_time, budget_s)
        if tl > 280:
            h_pol1, sc_pol1 = slsqp_round(h1, n1, target1,
                                          [100, 500, 2000], 400, 1e-14)
            if sc_pol1 < sc1:
                h1 = h_pol1
                sc1 = sc_pol1

        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            cur_n = n1
            cur_target = target1

        # Extended CD at higher resolution
        tl = time_remaining(start_time, budget_s)
        if tl > 250:
            h_cd1, c5_cd1 = pair_cd(h1, cur_n, cur_target,
                                    time_limit=min(200, tl * 0.55),
                                    max_sweeps=40000,
                                    n_cand_start=30, n_cand_end=120,
                                    n_shifts=4, max_no_improve=600,
                                    anneal_temp=0.005, anneal_decay=0.9997)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        # Multi-shift CD at higher res
        tl = time_remaining(start_time, budget_s)
        if tl > 80:
            h_mc1, c5_mc1 = multi_shift_cd(best_h, cur_n, cur_target,
                                           time_limit=min(50, tl * 0.4),
                                           max_sweeps=15000,
                                           n_cand=80, n_top=6)
            if c5_mc1 < best_score:
                best_score = c5_mc1
                best_h = h_mc1

    # Phase 6: SLSQP polish at current resolution
    tl = time_remaining(start_time, budget_s)
    if tl > 100:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [8000, 20000], 1200, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: Try even higher resolution n=1200
    tl = time_remaining(start_time, budget_s)
    if tl > 350:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        tl = time_remaining(start_time, budget_s)
        if tl > 280:
            h_pol2, sc_pol2 = slsqp_round(h2, n2, target2,
                                          [200, 1000], 500, 1e-14)
            if sc_pol2 < sc2:
                h2 = h_pol2
                sc2 = sc_pol2

        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            cur_n = n2
            cur_target = target2

        tl = time_remaining(start_time, budget_s)
        if tl > 200:
            h_cd2, c5_cd2 = pair_cd(h2, cur_n, cur_target,
                                    time_limit=min(180, tl * 0.55),
                                    max_sweeps=30000,
                                    n_cand_start=25, n_cand_end=100,
                                    n_shifts=5, max_no_improve=500,
                                    anneal_temp=0.0, anneal_decay=1.0)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

        tl = time_remaining(start_time, budget_s)
        if tl > 60:
            h_pol2b, sc_pol2b = slsqp_round(best_h, cur_n, cur_target,
                                            [5000, 15000], 800, 1e-14)
            if sc_pol2b < best_score:
                best_score = sc_pol2b
                best_h = h_pol2b

    # Phase 8: Stochastic restarts
    tl = time_remaining(start_time, budget_s)
    if tl > 150:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.02, 0.04, 0.06, 0.1]:
            tl = time_remaining(start_time, budget_s)
            if tl < 60:
                break
            h_pert = best_h + rng.randn(cur_n) * noise_scale
            h_pert = project_feasible(h_pert, cur_target)
            sc_pert = compute_c5(h_pert)

            if sc_pert < best_score + 0.005:
                h_ref, sc_ref = fast_cd(h_pert, cur_n, cur_target,
                                        time_limit=min(25, tl * 0.25),
                                        max_sweeps=8000, n_cand=60, n_shifts=3)
                tl = time_remaining(start_time, budget_s)
                if tl > 40:
                    h_ref, sc_ref = slsqp_round(h_ref, cur_n, cur_target,
                                                [500, 2000, 8000], 600, 1e-14)
                if sc_ref < best_score:
                    best_score = sc_ref
                    best_h = h_ref

    # Phase 9: Final multi-shift CD
    tl = time_remaining(start_time, budget_s)
    if tl > 80:
        h_final_mc, c5_final_mc = multi_shift_cd(best_h, cur_n, cur_target,
                                                 time_limit=min(50, tl * 0.5),
                                                 max_sweeps=20000,
                                                 n_cand=100, n_top=10)
        if c5_final_mc < best_score:
            best_score = c5_final_mc
            best_h = h_final_mc

    # Phase 10: Final pair CD
    tl = time_remaining(start_time, budget_s)
    if tl > 50:
        h_final_cd, c5_final = pair_cd(best_h, cur_n, cur_target,
                                       time_limit=min(35, tl * 0.55),
                                       max_sweeps=15000,
                                       n_cand_start=60, n_cand_end=180,
                                       n_shifts=5, max_no_improve=200,
                                       anneal_temp=0.0, anneal_decay=1.0)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Phase 11: Final SLSQP with very high alpha
    tl = time_remaining(start_time, budget_s)
    if tl > 30:
        h_final_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                      [15000, 40000], 800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    # Phase 12: One more CD pass on the result
    tl = time_remaining(start_time, budget_s)
    if tl > 25:
        h_extra, c5_extra = fast_cd(best_h, cur_n, cur_target,
                                    time_limit=min(15, tl * 0.5),
                                    max_sweeps=8000, n_cand=80, n_shifts=4)
        if c5_extra < best_score:
            best_score = c5_extra
            best_h = h_extra

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
