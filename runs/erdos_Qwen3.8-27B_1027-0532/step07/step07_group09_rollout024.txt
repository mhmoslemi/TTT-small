import cvxpy as cp

def sqp_step(h, n, target):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute gradients for all shifts
    # Gradient of corr_k with respect to h[i]:
    # corr_k = sum_j h[j] * (1 - h[j + (k - mid)])
    # d corr_k / d h[i] = (1 - h[i + (k-mid)]) - h[i - (k-mid)] (when indices valid)
    
    # This is the same as the "change matrix" concept
    
    # Build the LP
    delta = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        cp.sum(h + delta) == target,
        h + delta >= 0,
        h + delta <= 1
    ]
    
    # For each shift, add constraint: corr[k] + grad_k · delta <= t
    # grad_k[i] = (1 - h[i+k_rel]) - h[i-k_rel] where k_rel = k - mid
    
    for k_idx in range(n_corr):
        k_rel = k_idx - mid
        grad = np.zeros(n)
        v1 = (idxs + k_rel >= 0) & (idxs + k_rel < n)
        v2 = (idxs - k_rel >= 0) & (idxs - k_rel < n)
        grad[v1] += (1.0 - h[idxs[v1] + k_rel])
        grad[v2] -= h[idxs[v2] - k_rel]
        constraints.append(corr[k_idx] + cp.sum(grad * delta) <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS)  # or SOLVER of choice
    
    return delta.value, t.value
```

from scipy.optimize import linprog

def solve_sqp_lp(h, n, target, delta_max):
    # Variables: [δ_0, ..., δ_{n-1}, t]
    # Minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Bounds on δ: max(-h[i], -delta_max) ≤ δ[i] ≤ min(1-h[i], delta_max)
    lb = np.maximum(-h, -delta_max)
    ub = np.minimum(1.0 - h, delta_max)
    bounds = list(zip(lb, ub)) + [(None, None)]  # t is unbounded
    
    # Equality: sum(δ) = target - sum(h)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [target - np.sum(h)]
    
    # Inequality: grad_k · δ - t ≤ -corr[k] for each k
    # i.e., [grad_k, -1] · [δ, t] ≤ -corr[k]
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    A_ub = np.zeros((n_corr, n + 1))
    b_ub = np.zeros(n_corr)
    
    for k_idx in range(n_corr):
        k_rel = k_idx - mid
        grad = np.zeros(n)
        v1 = (idxs + k_rel >= 0) & (idxs + k_rel < n)
        v2 = (idxs - k_rel >= 0) & (idxs - k_rel < n)
        grad[v1] += (1.0 - h[idxs[v1] + k_rel])
        grad[v2] -= h[idxs[v2] - k_rel]
        A_ub[k_idx, :n] = grad
        A_ub[k_idx, n] = -1.0
        b_ub[k_idx] = -corr[k_idx]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        delta_sol = result.x[:n]
        return delta_sol
    return None
```

    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

        top_idx = np.argsort(corr)[-top_k:]
    else:
        top_idx = np.arange(n_corr)
    
    n_active = len(top_idx)
    
    # Variables: [δ_0, ..., δ_{n-1}, t], total n+1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Bounds
    lb = np.maximum(-h, -delta_max)
    ub = np.minimum(1.0 - h, delta_max)
    bounds = list(zip(lb, ub)) + [(None, None)]
    
    # Equality: sum(δ) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Inequality constraints: grad_k · δ - t ≤ -corr[k]
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)
    
    for i, k_idx in enumerate(top_idx):
        k_rel = int(k_idx) - mid
        grad = np.zeros(n)
        v1 = (idxs + k_rel >= 0) & (idxs + k_rel < n)
        v2 = (idxs - k_rel >= 0) & (idxs - k_rel < n)
        if v1.any():
            grad[v1] += (1.0 - h[idxs[v1] + k_rel])
        if v2.any():
            grad[v2] -= h[idxs[v2] - k_rel]
        A_ub[i, :n] = grad
        A_ub[i, n] = -1.0
        b_ub[i] = -corr[k_idx]
    
    return c_obj, A_ub, b_ub, A_eq, b_eq, bounds

import numpy as np
from scipy.optimize import linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def build_sqp_lp(h, n, target, delta_max, top_k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    if top_k is not None and top_k < n_corr:
        top_idx = np.argsort(corr)[-top_k:]
    else:
        top_idx = np.arange(n_corr)

    n_active = len(top_idx)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    lb = np.maximum(-h, -delta_max)
    ub = np.minimum(1.0 - h, delta_max)
    bounds = list(zip(lb, ub)) + [(None, None)]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)

    for i in range(n_active):
        k_idx = int(top_idx[i])
        k_rel = k_idx - mid
        grad = np.zeros(n)
        v1 = (idxs + k_rel >= 0) & (idxs + k_rel < n)
        v2 = (idxs - k_rel >= 0) & (idxs - k_rel < n)
        if v1.any():
            grad[v1] += (1.0 - h[idxs[v1] + k_rel])
        if v2.any():
            grad[v2] -= h[idxs[v2] - k_rel]
        A_ub[i, :n] = grad
        A_ub[i, n] = -1.0
        b_ub[i] = -corr[k_idx]

    return c_obj, A_ub, b_ub, A_eq, b_eq, bounds


def sqp_optimize(h_init, n, target, time_limit, top_k=40, delta_init=0.02):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    delta_max = delta_init
    no_improve = 0
    it = 0

    while time.time() - start < time_limit:
        it += 1
        if no_improve > 400:
            break

        c_obj, A_ub, b_ub, A_eq, b_eq, bounds = build_sqp_lp(h, n, target, delta_max, top_k)

        result = None
        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
        except Exception:
            try:
                result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                 bounds=bounds, method='interior-point')
            except Exception:
                delta_max *= 0.7
                no_improve += 30
                continue

        if result is None or not result.success:
            delta_max *= 0.7
            no_improve += 5
            if delta_max < 1e-8:
                break
            continue

        delta_sol = result.x[:n]
        h_new = h + delta_sol
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            delta_max = min(delta_max * 1.04, 0.08)
        else:
            delta_max *= 0.82
            no_improve += 1
            if delta_max < 1e-8:
                break

    return best_h, best_c5


def pair_cd(h_init, n, target, time_limit, max_sweeps=20000, n_cand_start=15, n_cand_end=50, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = min(5, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = G[cand_map[i]]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = G[cand_map[j]]
                h_j = h[j]

                d_max = min(h_i, 1.0 - h_j)
                d_min = max(h_i - 1.0, -h_j)
                if d_max - d_min < 1e-12:
                    continue

                g = lr - hr

                a_d = d_min
                b_d = d_max
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for idx in [best_i, best_j]:
                change[idx, :] = 0.0
                pos_mask = (idx + shifts >= 0) & (idx + shifts < n)
                neg_mask = (idx - shifts >= 0) & (idx - shifts < n)
                change[idx, pos_mask] += (1.0 - h[idx + shifts[pos_mask]])
                change[idx, neg_mask] -= h[idx - shifts[neg_mask]]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
