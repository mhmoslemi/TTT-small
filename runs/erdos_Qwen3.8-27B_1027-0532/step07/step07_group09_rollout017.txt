    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = min(4, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)])

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]
                h_j = h[j]
                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue
                g = G[lr] - G[hr]
                a_d = delta_min
                b_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def trust_region_step(h, n, target, alpha, max_iter=200, lr_init=None):
    dx = 2.0 / n
    if lr_init is None:
        lr_init = 1.0 / (n * alpha * 5.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init

    for _ in range(max_iter):
        grad, corr = compute_gradient(h, alpha, n)
        grad -= np.mean(grad)
        h_new = project_feasible(h - lr * grad, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.2, 1.0 / (2.0 * n))
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return best_h, best_c5, lr


def fista_with_adaptive(h_init, n, target, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    y = h.copy()
    t_mom = 1.0
    no_improve = 0
    stuck_count = 0

    alpha_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000, 8000, 15000]
    alpha_idx = 0
    lr = 1.0 / (20.0 * n)

    while time.time() - start < time_limit:
        alpha = alpha_schedule[alpha_idx]
        grad, corr = compute_gradient(y, alpha, n)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            stuck_count += 1
            if stuck_count > 5:
                if alpha_idx < len(alpha_schedule) - 1:
                    alpha_idx += 1
                    y = h.copy()
                    t_mom = 1.0
                    stuck_count = 0
                    no_improve = 0
                    lr = 1.0 / (20.0 * n * alpha_schedule[alpha_idx] / alpha_schedule[alpha_idx - 1])
                else:
                    break
            continue

        h_new = project_feasible(y - lr * grad, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
            stuck_count = 0
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            y = h_new + (t_mom - 1.0) / t_new * (h_new - h)
            t_mom = t_new
        else:
            no_improve += 1
            if no_improve > 30:
                no_improve = 0
                stuck_count += 1
                lr *= 0.7
                t_mom = 1.0
                y = h_new.copy()
                if stuck_count > 4:
                    stuck_count = 0
                    if alpha_idx < len(alpha_schedule) - 1:
                        alpha_idx += 1
                        lr = 1.0 / (20.0 * n)
            h = h_new
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_step(h, n, target, lr, n_active=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    idxs = np.arange(n)
    max_corr = np.max(corr)
    tol = max(1e-10, 1e-8 * max_corr)
    active = np.where(corr >= max_corr - tol)[0]
    active = active[:n_active]
    grad = np.zeros(n)
    for k_idx in active:
        k = int(k_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad /= len(active)
    grad -= np.mean(grad)
    return project_feasible(h - lr * grad, target)


def multi_scale_run(seed, budget_s):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        h_pol = slsqp_round(best_h, n0, target0, [50, 200, 1000], 300, 1e-13)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_fid, c5_fid = fista_with_adaptive(best_h, n0, target0, time_limit=min(100, tleft() * 0.5))
        if c5_fid < best_score:
            best_score = c5_fid
            best_h = h_fid

    if tleft() > 80:
        h_cd, c5_cd = vectorized_pair_cd(best_h, n0, target0, time_limit=min(60, tleft() * 0.6),
                                         max_sweeps=15000, n_cand_start=15, n_cand_end=45,
                                         max_no_improve=300)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol = slsqp_round(best_h, n0, target0, [500, 2000, 5000, 10000], 400, 1e-13)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n1 = 800
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)

    if tleft() > 200:
        h_fid1, c5_fid1 = fista_with_adaptive(h1, n1, target1, time_limit=min(120, tleft() * 0.5))
        if c5_fid1 < best_score:
            best_score = c5_fid1
            best_h = h_fid1
            n0 = n1
            target0 = target1

    if tleft() > 100:
        h_cd1, c5_cd1 = vectorized_pair_cd(best_h, n0, target0, time_limit=min(80, tleft() * 0.55),
                                           max_sweeps=12000, n_cand_start=20, n_cand_end=55,
                                           max_no_improve=350)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1

    if tleft() > 80:
        h_pol1 = slsqp_round(best_h, n0, target0, [2000, 5000, 10000, 20000], 500, 1e-13)
        sc1 = compute_c5(h_pol1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h_pol1

    n2 = 1200
    target2 = n2 / 2.0
    h2 = interpolate_to(best_h, n2, target2)

    if tleft() > 200:
        h_fid2, c5_fid2 = fista_with_adaptive(h2, n2, target2, time_limit=min(100, tleft() * 0.45))
        if c5_fid2 < best_score:
            best_score = c5_fid2
            best_h = h_fid2
            n0 = n2
            target0 = target2

    if tleft() > 80:
        h_cd2, c5_cd2 = vectorized_pair_cd(best_h, n0, target0, time_limit=min(60, tleft() * 0.55),
                                           max_sweeps=8000, n_cand_start=25, n_cand_end=60,
                                           max_no_improve=250)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 60:
        h_pol2 = slsqp_round(best_h, n0, target0, [5000, 10000, 30000], 400, 1e-13)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.05]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref = slsqp_round(h_pert, n0, target0, [500, 2000, 5000], 300, 1e-13)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 100:
        h_fid3, c5_fid3 = fista_with_adaptive(best_h, n0, target0, time_limit=min(60, tleft() * 0.5))
        if c5_fid3 < best_score:
            best_score = c5_fid3
            best_h = h_fid3

    if tleft() > 50:
        h_cd3, c5_cd3 = vectorized_pair_cd(best_h, n0, target0, time_limit=min(30, tleft() * 0.5),
                                           max_sweeps=5000, n_cand_start=30, n_cand_end=70,
                                           max_no_improve=150)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    if tleft() > 40:
        h_final = slsqp_round(best_h, n0, target0, [10000, 30000, 80000], 300, 1e-13)
        sc_final = compute_c5(h_final)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
