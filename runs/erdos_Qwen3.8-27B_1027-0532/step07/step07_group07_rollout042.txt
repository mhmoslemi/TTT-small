import numpy as np
from scipy.optimize import minimize
import time

# Global variables for SLSQP callbacks
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _time_left(start_time, budget_s):
    return budget_s - 20.0 - (time.time() - start_time)

def smooth_obj_and_grad(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    obj = c_max + np.log(s) / alpha
    if s < 1e-300:
        return obj, np.zeros(n)
    weights = exp_shifted / s
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in np.where(weights > 1e-8)[0]:
        w = weights[k_idx]
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return float(obj), grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _smooth_obj_func(x):
    return smooth_obj_and_grad(x, _g_alpha)[0]

def _smooth_obj_grad(x):
    return smooth_obj_and_grad(x, _g_alpha)[1]

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_target, _g_alpha
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj_func, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def momentum_smooth(h_init, n, target, time_limit, alphas=[100, 500, 2000, 8000],
                    momentum=0.9, lr_decay=0.95, max_iter=2000,
                    proj_every=50):
    """Gradient descent with Nesterov momentum on smooth approximation."""
    start = time.time()
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        lr = 1.0 / alpha * 10
        for it in range(max_iter):
            if time.time() - start > time_limit:
                break
            obj, grad = smooth_obj_and_grad(h, alpha)
            # Nesterov step
            v = momentum * v - lr * grad
            h_new = h + v
            # Center the update to maintain sum
            h_new -= np.sum(h_new) - target
            h_new = np.clip(h_new, 0.0, 1.0)
            if (it + 1) % proj_every == 0:
                h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new.copy()
            h = h_new
            lr *= lr_decay
    return best_h, best_c5

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change
