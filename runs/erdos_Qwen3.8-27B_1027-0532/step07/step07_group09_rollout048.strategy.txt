def project_feasible(x, n):
    """
    Projects onto [0,1]^n and enforces integral=1.
    Returns: x, normalized to satisfy constraints
    """
    # Clip to [0,1]
    x_clip = np.clip(x, 0, 1)
    total_int = np.sum(x_clip)
    target = n/2.0  # from integral constraint

    # Adjust integral by scaling
    if abs(total_int - target) > 1e-10:
        scale_factor = target / total_int
        x_scale = x_clip * scale_factor
        # Recalculate integral and adjust
        total_scaled = np.sum(x_scale)
        if abs(total_scaled) < 1e-10:
            # Fallback distribution
            x_scale = np.ones(n)/n * scale_factor
        else:
            x_scale = x_scale / total_scaled * target
    
    # Fine-tune boundary cases
    x_final = np.clip(x_scale * 1.0001, 0, 1)
    return x_clip if np.sum(x_final) < target else x_scale

def compute_gradient(x, n):
    """
    Returns ∇C5(x) with projected gradient momentum
    """
    corr = np.correlate(x, 1-x, mode='full')
    k_opt = np.argmax(corr)  # Which correlation provides the maximum overlap

    for i in range(n):
        idx_plus = (i + k_opt) % n
        idx_minus = (i - k_opt) % n
        grad[i] = (1.0 - x[idx_plus]) - x[idx_minus]

    ft = np.fft.fft(x)
    freq_energy = np.sum(np.abs(ft)**2)
    if freq_energy > 1e6:  # High frequency noise
        grad = grad * 0.8  # Damp gradient
        
    return grad
