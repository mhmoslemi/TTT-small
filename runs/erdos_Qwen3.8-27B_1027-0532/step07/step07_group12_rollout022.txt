    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_omega(v, target):
    """Exact Euclidean projection onto {h: 0<=h<=1, sum(h)=target}"""
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(v - lam, 0.0, 1.0)

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_omega(v, target):
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(v - lam, 0.0, 1.0)

def lse_obj_and_grad(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    log_sum = np.log(total_e)
    obj = (c_max + log_sum / alpha) * dx + penalty * (np.sum(x) - target) ** 2
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-10
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return obj, grad

def lse_obj_func(x, alpha, target, penalty):
    o, _ = lse_obj_and_grad(x, alpha, target, penalty)
    return o

def lse_grad_func(x, alpha, target, penalty):
    _, g = lse_obj_and_grad(x, alpha, target, penalty)
    return g

def lbfgs_phase(h_init, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_obj_func, h_cur, args=(alpha, target, pen),
                              jac=lse_grad_func, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_omega(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new

                    break
            except Exception:
                pass
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, n_cand=30, max_sweeps=50000, n_shifts=1):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)

            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = 50.0
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]
        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star
        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = project_omega(h, target)
    return best_h, best_c5

def nesterov_phase(h_init, n, target, time_limit, n_top=5, base_lr=0.08, momentum_decay=0.95):

    velocity = np.zeros(n)
    no_improve = 0
    for step in range(500000):
        if time.time() - start > time_limit:
            break
        decay = 1.0 / (1.0 + 0.002 * step)
        corr = np.correlate(x_k, 1.0 - x_k, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = x_k.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        top_idx = np.argsort(corr)[-n_top:]

        w = np.exp(alpha_w * (corr[top_idx] - cur_max))

        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - x_k[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * x_k[idxs[v2] + k]
        grad *= dx
        grad = grad - np.mean(grad)
        lr = base_lr * decay
        velocity = momentum_decay * velocity - lr * grad
        x_new = project_omega(x_k + velocity, target)
        beta = (1.0 + np.sqrt(1.0 + 4.0 * step * step)) / 2.0
        x_prev = x_k
        x_k = x_new
        if step % 500 == 0:
            c5_check = compute_c5(x_k)

def binary_local_search(h_init, n, target, time_limit, max_swaps=5000, n_tries=50):

            h_bin[ones[-1]] = 0.0
            ones = ones[:-1]
        else:
            h_bin[zeros[-1]] = 1.0

def interpolate_to(h_src, n_dst, target_dst):

    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_omega(h_out, target_dst)

                h_blend = (1.0 - v) * best_h + v * h_interp
                h_blend = project_omega(h_blend, best_n / 2.0)
                hp, sc = lbfgs_phase(h_blend, best_n, best_n / 2.0,
                                      alphas=[1000, 10000],
                                      maxiter=200, time_limit=min(20, tleft() * 0.2),
                                      penalty=200.0)
                if sc < best_score:
                    best_score = sc
                    best_h = hp

    # Phase 11: Try n=2400 if lots of time left
    if tleft() > 400:
        n_very_hi = 2400
        target_vh = n_very_hi / 2.0
        h_vh = interpolate_to(best_h, n_very_hi, target_vh)
        hpl, sc = lbfgs_phase(h_vh, n_very_hi, target_vh,
                              alphas=[100, 1000, 10000],
                              maxiter=200, time_limit=min(150, tleft() * 0.35),
                              penalty=200.0)
        if sc < best_score:
            best_score = sc
            best_h = hpl
            best_n = n_very_hi
        if tleft() > 150:
            hcd, c5cd = pair_cd(best_h, best_n, best_n / 2.0,
                                 time_limit=min(120, tleft() * 0.3),
                                 n_cand=30, max_sweeps=100000, n_shifts=1)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

    # Phase 12: Multi-start with different initializations
    if tleft() > 200:
        n_use = best_n
        target_use = n_use / 2.0
        # Block pattern
        h_block = np.zeros(n_use)
        h_block[:n_use // 4] = 1.0
        h_block[3 * n_use // 4:] = 1.0
        h_block = project_omega(h_block, target_use)
        hp, sc = lbfgs_phase(h_block, n_use, target_use,
                              alphas=[100, 1000, 10000],
                              maxiter=200, time_limit=min(60, tleft() * 0.25),
                              penalty=200.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
        # Periodic pattern
        n_per = 4
        h_per = np.tile(np.array([1.0, 1.0, 0.0, 0.0]), n_use // n_per)[:n_use]
        h_per = project_omega(h_per, target_use)
        hp, sc = lbfgs_phase(h_per, n_use, target_use,
                              alphas=[100, 1000, 10000],
                              maxiter=200, time_limit=min(60, tleft() * 0.25),
                              penalty=200.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 13: Final polish
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_phase(best_h, n_use, target_use,
                              alphas=[100000, 500000],
                              maxiter=200, time_limit=min(30, tleft() * 0.5),
                              penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd(best_h, n_use, target_use,
                             time_limit=min(20, tleft() * 0.6),
                             n_cand=25, max_sweeps=30000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_omega(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
