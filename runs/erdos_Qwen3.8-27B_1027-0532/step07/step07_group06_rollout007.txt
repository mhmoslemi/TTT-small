def projected_subgrad(h_init, n, target, time_limit, start_time, 
                      K=5, max_iter=50000, use_momentum=False):
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    momentum_vec = np.zeros(n)
    beta = 0.9
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find top-K lags
        mid = n - 1
        top_k = np.argsort(corr)[-K:]
        
        # Compute combined gradient
        idxs = np.arange(n)
        grad = np.zeros(n)
        for k_idx in top_k:
            k = int(k_idx) - mid
            valid1 = (idxs + k >= 0) & (idxs + k < n)
            valid2 = (idxs - k >= 0) & (idxs - k < n)
            grad[valid1] += (1.0 - h[idxs[valid1] + k])
            grad[valid2] -= h[idxs[valid2] - k]
        
        # Project to sum=0
        grad -= np.mean(grad)
        
        # Momentum
        if use_momentum:
            momentum_vec = beta * momentum_vec + (1-beta) * grad
            direction = -momentum_vec
        else:
            direction = -grad
        
        # Find max step for feasibility
        pos_mask = direction < -1e-14
        neg_mask = direction > 1e-14
        alpha_max = np.inf
        if pos_mask.any():
            alpha_max = min(alpha_max, np.min(-h[pos_mask] / direction[pos_mask]))
        if neg_mask.any():
            alpha_max = min(alpha_max, np.min((1.0 - h[neg_mask]) / direction[neg_mask]))
        alpha_max = max(alpha_max, 0.0)
        
        if alpha_max < 1e-14:
            momentum_vec *= 0.5
            continue
        
        # Line search: golden section
        a, b = 0.0, alpha_max
        gr = (np.sqrt(5) - 1) / 2  # golden ratio
        
        def f_alpha(alpha):
            h_trial = h + alpha * direction
            h_trial = np.clip(h_trial, 0.0, 1.0)
            corr_trial = np.correlate(h_trial, 1.0 - h_trial, mode='full')
            return float(np.max(corr_trial))
        
        # Golden section search

        d = a + gr * (b - a)

                d = c

                fc = fd

        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_lse, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

        if no_improve > patience:

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, :] = 0.0
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - 15.0 - elapsed()

    # Phase 1: Projected subgradient at n=400 (45% of time)
    if remaining() > 100:
        t_lim = start_time + min(0.45 * budget_s, remaining() - 10)
        h1, sc1 = projected_subgrad(best_h, n0, target0, t_lim, start_time,
                                     K_start=2, K_end=12, max_iter=500000,
                                     momentum_beta=0.82, search_points=6)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 1b: Second round of projected subgradient with different params
    if remaining() > 80:
        t_lim2 = start_time + min(0.2 * budget_s, remaining() - 10)
        h1b, sc1b = projected_subgrad(best_h, n0, target0, t_lim2, start_time,
                                       K_start=5, K_end=20, max_iter=300000,
                                       momentum_beta=0.9, search_points=10)
        if sc1b < best_score:
            best_score = sc1b
            best_h = h1b

    # Phase 2: SLSQP refinement
    if remaining() > 60:
        t_lim3 = start_time + min(60, remaining() - 10)
        h2, sc2 = slsqp_round(best_h, n0, target0, [200, 1000, 5000],
                               600, t_lim3, start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Pair CD at n=400
    if remaining() > 150:
        t_lim4 = start_time + min(150, remaining() - 10)
        h3, sc3 = pair_cd(best_h, n0, target0, t_lim4, start_time,
                           max_sweeps=30000, n_cand_start=15, n_cand_end=55,
                           max_no_improve=350, n_top_shifts=3, bisection_iters=16)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: Interpolate to n=800 and optimize
    if remaining() > 200:
        n1 = 800
        target1 = n1 / 2.0
        h_int = interpolate_to(best_h, n1, target1)
        t_lim5 = start_time + min(0.2 * budget_s, remaining() - 10)
        h4, sc4 = projected_subgrad(h_int, n1, target1, t_lim5, start_time,
                                     K_start=3, K_end=15, max_iter=200000,
                                     momentum_beta=0.85, search_points=6)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4
            n0 = n1
            target0 = target1

    # Phase 5: Pair CD at higher resolution
    if remaining() > 150:
        t_lim6 = start_time + min(150, remaining() - 10)
        h5, sc5 = pair_cd(best_h, n0, target0, t_lim6, start_time,
                           max_sweeps=20000, n_cand_start=12, n_cand_end=45,
                           max_no_improve=300, n_top_shifts=3, bisection_iters=16)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: SLSQP at higher resolution
    if remaining() > 60:
        t_lim7 = start_time + min(60, remaining() - 10)
        h6, sc6 = slsqp_round(best_h, n0, target0, [2000, 10000],
                               800, t_lim7, start_time)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: More projected subgradient with different params
    if remaining() > 100:
        t_lim8 = start_time + min(0.15 * budget_s, remaining() - 10)
        h7, sc7 = projected_subgrad(best_h, n0, target0, t_lim8, start_time,
                                     K_start=8, K_end=25, max_iter=200000,
                                     momentum_beta=0.88, search_points=8)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 8: Final pair CD polishing
    if remaining() > 80:
        t_lim9 = start_time + min(0.08 * budget_s, remaining() - 5)
        h8, sc8 = pair_cd(best_h, n0, target0, t_lim9, start_time,
                           max_sweeps=15000, n_cand_start=20, n_cand_end=60,
                           max_no_improve=200, n_top_shifts=1, bisection_iters=18)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: Final SLSQP
    if remaining() > 40:
        t_lim10 = start_time + min(0.05 * budget_s, remaining() - 3)
        h9, sc9 = slsqp_round(best_h, n0, target0, [5000, 20000],
                               500, t_lim10, start_time)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
