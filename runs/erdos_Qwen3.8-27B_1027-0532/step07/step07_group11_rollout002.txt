                        dd = d[s1] - d[s2]
                        if abs(dd) < 1e-15:
                            continue

                        if delta_lo + 1e-12 < delta < delta_hi - 1e-12:

                            # Check this is actually the max

                                best_d = delta

                    h[j] -= best_d
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    best_delta_c5 = best_f * dx
                    best_move = True
        
        if not best_move:
            continue
    
    return best_h, best_c5

# Vectorized 1D minimax solver
def solve_1d_minimax(corr_vals, deltas, lo, hi):
    """
    Find δ ∈ [lo, hi] that minimizes max_s (corr_vals[s] + δ * deltas[s])
    corr_vals, deltas: arrays of length M
    """
    M = len(corr_vals)
    
    # Evaluate at boundaries
    f_lo = np.max(corr_vals + lo * deltas)
    f_hi = np.max(corr_vals + hi * deltas)
    best_f = min(f_lo, f_hi)
    best_d = lo if f_lo <= f_hi else hi
    
    # Find all pairwise intersections
    # delta_s1_s2 = (corr_vals[s2] - corr_vals[s1]) / (deltas[s1] - deltas[s2])
    d_diff = deltas[None, :] - deltas[:, None]  # [s1, s2] -> deltas[s2] - deltas[s1]
    c_diff = corr_vals[:, None] - corr_vals[None, :]  # [s1, s2] -> corr_vals[s1] - corr_vals[s2]
    
    # delta = c_diff / d_diff gives the intersection point
    # But we need: corr_vals[s1] + delta * deltas[s1] = corr_vals[s2] + delta * deltas[s2]
    # => delta * (deltas[s1] - deltas[s2]) = corr_vals[s2] - corr_vals[s1]
    # => delta = (corr_vals[s2] - corr_vals[s1]) / (deltas[s1] - deltas[s2])
    # = -c_diff / d_diff (with my definitions above)
    
    # Actually let me redo:
    # f_s(δ) = c_s + δ * d_s
    # Intersection of s1, s2: c_1 + δ*d_1 = c_2 + δ*d_2
    # δ = (c_2 - c_1) / (d_1 - d_2)
    
    d1 = deltas[:, None]  # M x 1
    d2 = deltas[None, :]  # 1 x M
    c1 = corr_vals[:, None]
    c2 = corr_vals[None, :]
    
    denom = d1 - d2  # M x M
    numer = c2 - c1  # M x M
    # delta = numer / denom
    with np.errstate(divide='ignore', invalid='ignore'):
        delta_all = numer / denom
    
    # Only consider valid entries (upper triangle, denom != 0, in range)
    tri_mask = np.triu(np.ones((M, M), dtype=bool), k=1)
    valid = tri_mask & (np.abs(denom) > 1e-15) & (delta_all > lo) & (delta_all < hi)
    
    valid_deltas = delta_all[valid]
    
    if len(valid_deltas) == 0:

    min_idx = np.argmin(f_candidates)
    if f_candidates[min_idx] < best_f:
        best_f = f_candidates[min_idx]
        best_d = valid_deltas[min_idx]
    
    return best_d, best_f

The full correlation has length 2n-1, with indices from 0 to 2n-2.

The first term depends only on the positions of h that are shifted, and the second is a "self-overlap" term.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def solve_minimax_1d(c_vals, d_vals, lo, hi):
    M = len(c_vals)
    f_lo = float(np.max(c_vals + lo * d_vals))
    f_hi = float(np.max(c_vals + hi * d_vals))
    best_f = min(f_lo, f_hi)
    best_d = lo if f_lo <= f_hi else hi
    denom = d_vals[:, None] - d_vals[None, :]
    numer = c_vals[None, :] - c_vals[:, None]
    with np.errstate(divide='ignore', invalid='ignore'):
        delta_all = numer / denom
    tri = np.triu(np.ones((M, M), dtype=bool), k=1)
    valid = tri & (np.abs(denom) > 1e-15) & (delta_all > lo + 1e-12) & (delta_all < hi - 1e-12)
    if valid.any():
        vd = delta_all[valid]
        f_cand = np.max(c_vals[None, :] + vd[:, None] * d_vals[None, :], axis=1)
        mi = int(np.argmin(f_cand))
        if f_cand[mi] < best_f:
            best_f = float(f_cand[mi])
            best_d = float(vd[mi])
    return best_d, best_f


def pair_cd_exact(h_init, n, target, time_limit, max_sweeps=100000,
                  n_top_shifts=30, n_cand=40, max_no_improve=4000,

        top_idx = np.argsort(corr)[-n_top_shifts:]
        top_corr = corr[top_idx].copy()
        top_k = (top_idx - mid).astype(int)

        grad = np.zeros(n)
        for s in range(len(top_idx)):
            k = top_k[s]
            w = top_corr[s]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]]

        sorted_idx = np.argsort(grad)
        n_c = min(n_cand, n // 4)
        low_set = sorted_idx[:n_c]
        high_set = sorted_idx[-n_c:]

        moves_applied = 0

            if best_f < corr_max - 1e-10:
                h[i] += best_d
                h[j] -= best_d
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                actual_c5 = compute_c5(h)
                if actual_c5 < c5_val - 1e-14:
                    moves_applied += 1
                else:
                    h = h_init.copy() if False else h

                    h[i] += best_d * 0.5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_multi(h_init, n, target, time_limit, n_steps=30000, n_weights=5, step_scale=0.004):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        p = np.exp(corr * 80.0)
        p /= p.sum()

        grad = np.zeros(n)
        top_k = np.argsort(p)[-n_weights:]
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
            eta = min(eta * 1.02, 0.02)
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
