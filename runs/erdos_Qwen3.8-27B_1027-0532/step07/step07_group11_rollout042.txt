        if abs(diff) < 1e-14:

        if diff > 0:

        else:

            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()

    c_max = np.max(corr)
    alpha = _g_alpha
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

def compute_c5(h):

    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    cur_n = n0
    cur_target = target0

    # Phase 1: Quick SLSQP at n0
    if time.time() < deadline - 80:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [100, 500, 2000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Interpolate to n=800
    n1 = 800
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)
    best_h = h1
    best_score = compute_c5(h1)
    cur_n = n1
    cur_target = target1

    # Phase 3: SLSQP at n1 (moderate alpha)
    if time.time() < deadline - 400:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [50, 200, 500], 600, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Pair CD at n1
    if time.time() < deadline - 250:
        tl = min(180, deadline - time.time() - 70)
        h_cd, c5_cd = pair_cd(best_h, cur_n, cur_target,
                              time_limit=tl,
                              max_sweeps=60000,
                              n_cand_start=40, n_cand_end=130,
                              n_shifts=4, max_no_improve=300,
                              anneal_temp=0.01, anneal_decay=0.9995)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP with higher alpha
    if time.time() < deadline - 200:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [1000, 3000, 8000], 600, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Gradient projection method
    if time.time() < deadline - 150:
        tl = min(100, deadline - time.time() - 50)
        h_gp, c5_gp = gradient_projection(best_h, cur_n, cur_target,
                                          time_limit=tl, n_steps=80000, step_scale=0.003)
        if c5_gp < best_score:
            best_score = c5_gp
            best_h = h_gp

    # Phase 7: More pair CD with different parameters
    if time.time() < deadline - 120:
        tl = min(100, deadline - time.time() - 40)
        h_cd, c5_cd = pair_cd(best_h, cur_n, cur_target,
                              time_limit=tl,
                              max_sweeps=50000,
                              n_cand_start=60, n_cand_end=160,
                              n_shifts=5, max_no_improve=250,
                              anneal_temp=0.0, anneal_decay=1.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Stochastic restarts
    if time.time() < deadline - 100:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.04, 0.1, 0.2]:
            if time.time() > deadline - 40:
                break
            h_pert = best_h + rng.randn(cur_n) * noise_scale
            h_pert = project_feasible(h_pert, cur_target)
            h_ref, sc = slsqp_round(h_pert, cur_n, cur_target,
                                    [100, 500, 2000], 300, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 9: Final pair CD
    if time.time() < deadline - 50:
        tl = min(40, deadline - time.time() - 15)
        if tl > 5:
            h_cd, c5_cd = pair_cd(best_h, cur_n, cur_target,
                                  time_limit=tl,
                                  max_sweeps=20000,
                                  n_cand_start=100, n_cand_end=200,
                                  n_shifts=6, max_no_improve=150,
                                  anneal_temp=0.0, anneal_decay=1.0)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 10: Final SLSQP polish
    if time.time() < deadline - 5:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [5000, 15000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
