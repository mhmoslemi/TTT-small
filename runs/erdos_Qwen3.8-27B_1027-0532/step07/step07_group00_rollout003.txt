```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    """

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    mid = n - 1

        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = np.max(corr) * dx

            # 2) j = n: contributes -h[j+m] if 0 <= j+m < n
```

```python
    Game-theoretic approach:
    - Outer: multiplicative weights over shifts
    - Inner: projected gradient on h
    """

    n_corr = 2 * n - 1

    best_h = h.copy()

    lr_h = 0.01
    
    for it in range(n_iters):

            break

        # Compute gradient of the weighted sum

        weighted_obj = np.dot(p, corr) * dx

                idx2 = np.arange(n) + shift
```

```python
def structural_perturbation(h, n, rng):
    """Apply a random structural perturbation to h."""
    h_out = h.copy()
    move_type = rng.randint(4)
    
    if move_type == 0:  # Segment reversal
        start = rng.randint(0, n - 20)
        length = rng.randint(10, min(50, n - start))
        h_out[start:start+length] = h_out[start:start+length][::-1]
    
    elif move_type == 1:  # Block swap
        block_len = rng.randint(5, min(30, n // 4))
        pos1 = rng.randint(0, n - block_len)
        pos2 = rng.randint(0, n - block_len)
        if abs(pos1 - pos2) < block_len:
            return h_out
        # Swap blocks
        temp = h_out[pos1:pos1+block_len].copy()
        h_out[pos1:pos1+block_len] = h_out[pos2:pos2+block_len]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_logsumexp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    return float(max_c + np.log(np.sum(np.exp(alpha * (corr - max_c)))) / alpha)

def smooth_logsumexp_grad(h, alpha):
    """
    Gradient of (1/alpha) * log(sum_k exp(alpha * corr[k]))
    where corr[k] = sum_n h[n+k]*(1-h[n]) for valid n
    
    Grad_j = (sum_k exp(alpha*corr[k]) * d(corr[k])/dh[j]) / (sum_k exp(alpha*corr[k]))
    """
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    weights = np.exp(alpha * (corr - max_c))
    total_w = np.sum(weights)
    
    grad = np.zeros(n)
    mid = n - 1
    n_corr = len(corr)
    
    # Only process shifts with significant weight
    threshold = 1e-8 * total_w
    for k in range(n_corr):
        w = weights[k]
        if w < threshold:
            continue
        shift = k - mid
        if shift == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        else:
            # d(corr[k])/dh[j]:
            # corr[k] = sum_{n: valid} h[n+shift] * (1-h[n])
            # d/dh[j] = (1-h[j-shift])*I(0<=j-shift<n) - h[j+shift]*I(0<=j+shift<n)
            # Note: when shift=0, both terms contribute to the same j: (1-h[j]) - h[j] = 1-2h[j]
            
            # Term 1: j-shift valid means 0 <= j-shift < n
            j_vals = np.arange(n)
            n_vals = j_vals - shift  # n = j - shift
            valid1 = (n_vals >= 0) & (n_vals < n)
            grad[valid1] += w * (1.0 - h[n_vals[valid1]])
            
            # Term 2: j+shift valid means 0 <= j+shift < n
            n2_vals = j_vals + shift  # this is the "other" index
            valid2 = (n2_vals >= 0) & (n2_vals < n)
            grad[valid2] -= w * h[n2_vals[valid2]]
    
    grad /= total_w
    return grad
```
