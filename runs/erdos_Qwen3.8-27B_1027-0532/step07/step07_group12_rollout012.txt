import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        mask = (result > 1e-15) & (result < 1.0 - 1e-15)
        if np.any(mask):
            result[mask] += residual / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-14
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, n_cand=40, max_sweeps=100000, n_shifts=1, adaptive_shifts=False):

            elif gap < 0.2 * cur_max:

            else:

                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)

        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue

                lo = 0.0
                hi_d = delta_max
                for _ in range(12):

                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:

                    else:
                        lo = mid_d

                    best_i = int(i)
                    best_j = int(j)

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

                                   n_cand=35, max_sweeps=50000, n_shifts=1, adaptive_shifts=True)
                if c5cd < best_score:
                    best_score = c5cd
                    best_h = hcd

    # Phase 7: Higher resolution attempt
    if tleft() > 200:
        n_new = 800
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = lbfgs_phase(h_interp, n_new, target_new,
                            alphas=[200, 2000, 20000, 200000],
                            maxiter=300, time_limit=min(80, tleft() * 0.25),
                            penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new
            if tleft() > 80:
                hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(100, tleft() * 0.35),
                                   n_cand=40, max_sweeps=100000, n_shifts=1, adaptive_shifts=True)
                if c5cd < best_score:
                    best_score = c5cd
                    best_h = hcd

    # Phase 8: Final LSE with very high alpha
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_phase(best_h, n_use, target_use,
                            alphas=[500000, 5000000, 50000000],
                            maxiter=200, time_limit=min(40, tleft() * 0.5),
                            penalty=100000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 9: Final pair CD
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd(best_h, n_use, target_use,
                           time_limit=min(30, tleft() * 0.8),
                           n_cand=30, max_sweeps=50000, n_shifts=1, adaptive_shifts=True)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
