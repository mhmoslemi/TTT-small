    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = alpha * corr * dx
    s -= np.max(s)

    m2 = (Q >= 0) & (Q < n)
    t2 = np.where(m2, h[np.clip(Q, 0, n-1)], 0.0) @ w

        if not np.any(mask):
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_grad_cache(n):
    shifts = np.arange(-(n - 1), n)
    p = np.arange(n)
    P = p[:, None] + shifts[None, :]
    m1 = (P >= 0) & (P < n)
    Pc = np.clip(P, 0, n - 1)
    Q = p[:, None] - shifts[None, :]
    m2 = (Q >= 0) & (Q < n)
    Qc = np.clip(Q, 0, n - 1)
    return Pc, m1, Qc, m2

def lse_grad_cached(h, alpha, dx, cache):
    Pc, m1, Qc, m2 = cache
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = alpha * corr * dx
    s -= np.max(s)
    w = np.exp(s)
    w /= np.sum(w)
    t1 = np.where(m1, 1.0 - h[Pc], 0.0) @ w
    t2 = np.where(m2, h[Qc], 0.0) @ w
    return dx * (t1 - t2), corr

def pd_lse(h_init, time_limit, alpha_start=100.0, alpha_end=30000.0, max_iters=80000):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    cache = make_grad_cache(n)
    start_time = time.time()
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    mu = 0.8
    best_h = h.copy()
    best_score = compute_c5(h)
    no_imp = 0
    step = 1.0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_imp > 12000:
            break
        prog = min(it / float(max_iters), 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** prog
        grad, corr = lse_grad_cached(h, alpha, dx, cache)
        gnorm = np.linalg.norm(grad)
        step = min(step * 1.02, 1.0 / (gnorm + 1e-12))
        v = mu * v - step * grad
        h_new = project_feasible(h + v, target)
        score = compute_c5(h_new)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                v *= 0.5
        h = h_new
    return best_h, best_score

def cd_refine(h_init, n, target, time_limit, n_cand=60, max_sweeps=30000):
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_imp = 0
    change = np.zeros((n, n_corr))
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_imp > 800:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                dlo = max(h[i] - 1.0, -h[j])
                dhi = min(h[i], 1.0 - h[j])
                if dhi - dlo < 1e-12:
                    continue
                g = change[j] - change[i]
                a, b = dlo, dhi
                for _ in range(14):
                    m = 0.5 * (a + b)
                    vals = corr + m * g
                    fm = np.max(vals)
                    am = vals >= fm - 1e-14
                    fp = np.max(g[am]) if am.any() else 0.0
                    if fp > 1e-12:
                        b = m
                    else:
                        a = m
                ds = 0.5 * (a + b)
                if abs(ds) < 1e-12:
                    continue
                am = np.max(corr + ds * g) * dx
                if am < best_trial:
                    best_trial = am
                    best_i = i
                    best_j = j
                    best_delta = ds
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pi = best_i
            change[pi, :] = 0.0
            pm = (pi + shifts >= 0) & (pi + shifts < n)
            nm = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pm] += (1.0 - h[pi + shifts[pm]])
            change[pi, nm] -= h[pi - shifts[nm]]
            pj = best_j
            change[pj, :] = 0.0
            pm2 = (pj + shifts >= 0) & (pj + shifts < n)
            nm2 = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pm2] += (1.0 - h[pj + shifts[pm2]])
            change[pj, nm2] -= h[pj - shifts[nm2]]
    return best_h, best_c5

def sa_escape(h_init, n, target, time_limit, n_cand=50, t_start=0.03, t_end=0.0002):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(123)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_imp = 0
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]
    max_steps = 30000
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_imp > 2000:
            break
        prog = min(step / max_steps, 1.0)
        temp = t_start * (t_end / t_start) ** prog
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        k_idx = int(np.argmax(corr))
        k_star = k_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        si = np.argsort(grad)
        low_set = si[:n_cand]
        high_set = si[-n_cand:]
        i = int(high_set[rng.randint(n_cand)])
        j = int(low_set[rng.randint(n_cand)])
        if i == j:
            continue
        dlo = max(h[i] - 1.0, -h[j])
        dhi = min(h[i], 1.0 - h[j])
        if dhi - dlo < 1e-12:
            continue
        delta = dlo + (dhi - dlo) * rng.rand()
        if abs(delta) < 1e-12:
            continue
        g = change[j] - change[i]
        new_c5 = float(np.max(corr + delta * g)) * dx
        dc = new_c5 - c5_val
        accept = dc < -1e-14 or (temp > 1e-10 and rng.rand() < np.exp(-max(dc, 0.0) / max(temp * dx, 1e-15)))
        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[i, :] = 0.0
            pm = (i + shifts >= 0) & (i + shifts < n)
            nm = (i - shifts >= 0) & (i - shifts < n)
            change[i, pm] += (1.0 - h[i + shifts[pm]])
            change[i, nm] -= h[i - shifts[nm]]
            change[j, :] = 0.0
            pm2 = (j + shifts >= 0) & (j + shifts < n)
            nm2 = (j - shifts >= 0) & (j - shifts < n)
            change[j, pm2] += (1.0 - h[j + shifts[pm2]])
            change[j, nm2] -= h[j - shifts[nm2]]
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    best_h = project_feasible(h0, target0)
    best_score = compute_c5(best_h)

    # Phase 1: PD on n=600
    if tleft() > 180:
        n1 = 600
        h1 = interpolate_to(best_h, n1, n1 / 2.0)
        h1, s1 = pd_lse(h1, min(150, tleft() * 0.6), 100.0, 20000.0, 60000)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: CD on n=600
    if tleft() > 80:
        h2, s2 = cd_refine(best_h, len(best_h), len(best_h) / 2.0, min(70, tleft() * 0.5), n_cand=50, max_sweeps=20000)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: PD on n=1000
    if tleft() > 300:
        n3 = 1000
        h3 = interpolate_to(best_h, n3, n3 / 2.0)
        h3, s3 = pd_lse(h3, min(250, tleft() * 0.65), 200.0, 50000.0, 100000)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: CD on n=1000
    if tleft() > 100:
        n_cur = len(best_h)
        h4, s4 = cd_refine(best_h, n_cur, n_cur / 2.0, min(80, tleft() * 0.5), n_cand=60, max_sweeps=25000)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: SA escape on n=1000
    if tleft() > 100:
        n_cur = len(best_h)
        h5, s5 = sa_escape(best_h, n_cur, n_cur / 2.0, min(80, tleft() * 0.5), n_cand=50)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: CD again after SA
    if tleft() > 60:
        n_cur = len(best_h)
        h6, s6 = cd_refine(best_h, n_cur, n_cur / 2.0, min(50, tleft() * 0.5), n_cand=60, max_sweeps=15000)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Try n=1200
    if tleft() > 350:
        n7 = 1200
        h7 = interpolate_to(best_h, n7, n7 / 2.0)
        h7, s7 = pd_lse(h7, min(280, tleft() * 0.6), 300.0, 60000.0, 80000)
        if s7 < best_score:
            best_score = s7
            best_h = h7
        if tleft() > 80:
            h7b, s7b = cd_refine(best_h, len(best_h), len(best_h) / 2.0, min(60, tleft() * 0.5), n_cand=60, max_sweeps=20000)
            if s7b < best_score:
                best_score = s7b
                best_h = h7b

    # Phase 8: Final PD refinement
    if tleft() > 60:
        n_cur = len(best_h)
        h8, s8 = pd_lse(best_h, min(50, tleft() * 0.6), 5000.0, 100000.0, 30000)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    # Phase 9: Final CD
    if tleft() > 30:
        n_cur = len(best_h)
        h9, s9 = cd_refine(best_h, n_cur, n_cur / 2.0, min(25, tleft() * 0.6), n_cand=70, max_sweeps=10000)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
