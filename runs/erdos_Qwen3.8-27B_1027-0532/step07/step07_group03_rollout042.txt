def frank_wolfe(h_init, n, target, time_limit, max_iters=100000, active_tol=0.01):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for t in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c_max = np.max(corr)
        c5_val = float(c_max) * (2.0 / n)
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
        
        # Compute subgradient (weighted average of active shifts)
        active = corr >= c_max - active_tol * (2.0/n)
        # Actually, let me use the raw correlation values
        active = corr >= c_max - active_tol  # in correlation units
        
        # Compute gradient as weighted average
        mid = n - 1
        idxs = np.arange(n)
        grad = np.zeros(n)
        active_lags = np.where(active)[0]
        weights = np.ones(len(active_lags)) / len(active_lags)
        
        for k_star, w in zip(active_lags, weights):
            k = int(k_star) - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] + k])
            grad[v2] -= w * h[idxs[v2] - k]
        
        # Linear oracle: find x in P that minimizes <grad, x>
        # P = {x : 0 ≤ x ≤ 1, sum(x) = n/2}
        # Solution: x[i] = 1 where grad[i] is smallest
        sorted_idx = np.argsort(grad)
        x_star = np.zeros(n)
        x_star[sorted_idx[:int(target)]] = 1.0
        
        # Step size
        gamma = 2.0 / (t + 2.0)
        
        # Update
        h = (1.0 - gamma) * h + gamma * x_star
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr_gradient(h, k_shift):
    """Compute gradient of correlation at shift k_shift (positive means h is shifted right)"""
    n = len(h)
    idxs = np.arange(n)
    grad = np.zeros(n)
    # corr = sum_i h[i] * (1 - h[i + k_shift]) for valid i
    # d/dh[j] = sum over i where i+k_shift=j: (1-h[i]) - sum over i where i=j: h[i+k_shift]
    # For shift s = k_shift:
    # Term 1: h[j] appears as h[i] when i=j, contributes (1-h[j+s]) if j+s < n
    # Term 2: h[j] appears as h[i+s] when i=j-s, contributes -h[j-s] if j-s >= 0
    v1 = (idxs + k_shift >= 0) & (idxs + k_shift < n)
    v2 = (idxs - k_shift >= 0) & (idxs - k_shift < n)
    grad[v1] += (1.0 - h[idxs[v1] + k_shift])
    grad[v2] -= h[idxs[v2] - k_shift]
    return grad

def compute_corr_gradient_multi(h, k_shifts, weights):
    n = len(h)
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_shift, w in zip(k_shifts, weights):
        k = int(k_shift)
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

def frank_wolfe(h_init, n, target, time_limit, max_iters=50000, active_tol_frac=0.05):
    """Frank-Wolfe method for the min-max problem."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    mid = n - 1
    idxs = np.arange(n)
    
    for t in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c_max = np.max(corr)
        c5_val = float(c_max) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
        
        # Active set: shifts with correlation within tolerance of max
        tol = active_tol_frac * c_max
        active = corr >= c_max - tol
        active_lags = np.where(active)[0]
        
        # Compute subgradient
        if len(active_lags) == 1:
            k = int(active_lags[0]) - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad = np.zeros(n)
            grad[v1] += (1.0 - h[idxs[v1] + k])
            grad[v2] -= h[idxs[v2] - k]
        else:
            n_active = len(active_lags)
            grad = np.zeros(n)
            for lag in active_lags:
                k = int(lag) - mid
                v1 = (idxs + k >= 0) & (idxs + k < n)
                v2 = (idxs - k >= 0) & (idxs - k < n)
                grad[v1] += (1.0 - h[idxs[v1] + k]) / n_active
                grad[v2] -= h[idxs[v2] - k]) / n_active
        
        # Linear oracle: minimize <grad, x> s.t. 0 ≤ x ≤ 1, sum(x) = target
        # Solution: x[i] = 1 where grad[i] is smallest
        sorted_idx = np.argsort(grad)
        x_star = np.zeros(n)
        x_star[sorted_idx[:int(target)]] = 1.0
        
        # Check if x_star = h (convergence)
        if np.sum(np.abs(x_star - h)) < 1e-10:
            break
        
        # Step size: use 2/(t+2) with a cap
        gamma = min(2.0 / (t + 2.0), 1.0)
        
        # Update
        h = (1.0 - gamma) * h + gamma * x_star
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    
    return best_h, best_score

def subgradient_with_averaging(h_init, n, target, time_limit, max_iters=100000, 
                                 lr_init=1.0, active_tol_frac=0.1):
    """Subgradient method with Polyak averaging."""
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    h_avg = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for t in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c_max = np.max(corr)
        c5_val = float(c_max) * dx
        
        # Update average
        h_avg = (t * h_avg + h) / (t + 1)
        h_avg_feasible = project_feasible(h_avg, target)
        c5_avg = compute_c5(h_avg_feasible)
        if c5_avg < best_score - 1e-14:
            best_score = c5_avg
            best_h = h_avg_feasible.copy()
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
        
        # Subgradient
        tol = active_tol_frac * c_max
        active = corr >= c_max - tol
        active_lags = np.where(active)[0]
        
        grad = np.zeros(n)
        for lag in active_lags:
            k = int(lag) - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += (1.0 - h[idxs[v1] + k])
            grad[v2] -= h[idxs[v2] - k]
        grad /= len(active_lags)
        
        # Step size: decrease over time
        lr = lr_init / (1.0 + 0.01 * t)
        
        # Subgradient step
        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    
    return best_h, best_score

def cutting_plane_v2(h_init, n, target, time_limit, max_iters=5000, tol=1e-9, max_cuts=200):
    """Improved cutting plane method."""
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    cuts_g = []
    cuts_f = []
    cuts_h = []
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 2000:
            # Reset cuts
            cuts_g = [cuts_g[0]] if cuts_g else []
            cuts_f = [cuts_f[0]] if cuts_f else []
            cuts_h = [cuts_h[0]] if cuts_h else []
            no_improve = 0
        
        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])
        
        # Active set for gradient
        active_mask = corr >= corr[k_star] - 1e-6
        active_lags = np.where(active_mask)[0]
        
        if len(active_lags) == 1:
            k = int(active_lags[0]) - mid
            v1 = (np.arange(n) + k >= 0) & (np.arange(n) + k < n)
            v2 = (np.arange(n) - k >= 0) & (np.arange(n) - k < n)
            g = np.zeros(n)
            g[v1] += (1.0 - h[np.arange(n)[v1] + k])
            g[v2] -= h[np.arange(n)[v2] - k]
        else:
            g = np.zeros(n)
            for lag in active_lags:
                k = int(lag) - mid
                idxs = np.arange(n)
                v1 = (idxs + k >= 0) & (idxs + k < n)
                v2 = (idxs - k >= 0) & (idxs - k < n)
                g[v1] += (1.0 - h[idxs[v1] + k]) / len(active_lags)
                g[v2] -= h[idxs[v2] - k]) / len(active_lags)
        
        cuts_g.append(g)
        cuts_f.append(f_val)
        cuts_h.append(h.copy())
        
        if len(cuts_g) > max_cuts:
            # Keep the first cut and the most recent half
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]
        
        # Solve LP
        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t
        
        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                continue
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            
            t_val = float(res.x[n])
            gap = sc / dx - t_val
            
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break
            
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            continue
    
    return best_h, best_score

def interceptive_cd(h_init, n, target, time_limit, max_sweeps=20000,
                    n_cand_start=20, n_cand_end=80, max_no_improve=300):
    """Interceptive coordinate descent (improved)."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        # Gradient for active shift
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        # Select candidates
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            
            # Update change matrix
            change[best_i, :] = 0.0
            change[best_j, :] = 0.0
            pi = best_i
            pj = best_j
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]
    
    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def sturmian_start(n, alpha=None):
    """Generate a Sturmian/Beatty sequence as starting point."""

        alpha = (1.0 + np.sqrt(5.0)) / 2.0
    x = np.arange(n) * alpha
    h = (x % 1.0 < 0.5).astype(float)
    # Adjust to exact sum
    target = n / 2.0
    h = project_feasible(h, target)
    return h

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            break
        
        # Find top 3 active shifts
        top_lags = np.argsort(corr)[-3:]
        top_lags = top_lags[top_lags < n_corr]
        
        # Weighted gradient
        weights = np.exp(5.0 * (corr[top_lags] - corr[top_lags].max()))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for lag, w in zip(top_lags, weights):
            k = int(lag) - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] + k])
            grad[v2] -= w
