```python
def compute_grad_logsumexp(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # softmax
    shifted = alpha * (corr - np.max(corr))
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # grad[i] = sum_k weights[k] * ((1-h[i+k]) - h[i-k]) with valid index checks
    # shifts = arange(-(n-1), n), corr index = shift + (n-1)
    shifts = np.arange(-(n-1), n)
    grad = np.zeros(n)
    
    for k_idx in range(len(shifts)):
        k = shifts[k_idx]
        w = weights[k_idx]
        if w < 1e-15:
            continue
        v1 = (np.arange(n) - k >= 0) & (np.arange(n) - k < n)
        v2 = (np.arange(n) + k >= 0) & (np.arange(n) + k < n)
        grad[v1] += w * (1.0 - h[np.arange(n)[v1] - k])
        grad[v2] -= w * h[np.arange(n)[v2] + k]
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_logsumexp(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (corr - np.max(corr))
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    idxs = np.arange(n)
    shifts = np.arange(-(n-1), n)
    grad = np.zeros(n)
    
    for k_idx in range(len(shifts)):
        k = int(shifts[k_idx])
        w = weights[k_idx]
        if w < 1e-15:
            continue
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if v1.any():
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
        if v2.any():
            grad[v2] -= w * h[idxs[v2] + k]
    
    return grad

def proj_grad_descent(h_init, n, target, time_limit, start_time, 
                      alphas=None, bg_gamma=0.95, max_iter=100000):
    if alphas is None:
        alphas = [10, 50, 200, 1000, 5000, 20000, 100000]
    
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        lr = 0.5 / (alpha * dx * n)  # rough Lipschitz estimate
        lr = min(lr, 0.5)
        
        for it in range(max_iter):
            if time.time() - start_time > time_limit:
                break
            
            grad = compute_grad_logsumexp(h, alpha)
            grad -= np.mean(grad)  # projection onto sum constraint
            
            # Projected gradient step
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target)
            
            # Backtracking line search
            c_curr = compute_c5(h)
            c_new = compute_c5(h_new)
            lam = 1.0
            while c_new > c_curr - 1e-15 * lam * np.sum(grad**2) and lam > 1e-6:
                lam *= bg_gamma
                h_try = h - lr * lam * grad
                h_try = project_feasible(h_try, target)
                c_new = compute_c5(h_try)
                if c_new < c_new + 1e-30:  # prevent infinite loop
                    break
            h_new = h - lr * lam * grad
            h_new = project_feasible(h_new, target)
            
            if compute_c5(h_new) < c_curr - 1e-16:
                h = h_new
                cur_c5 = compute_c5(h)
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                break  # stalled for this alpha
    
    return best_h, best_c5
```

```python
        L = alpha * dx * 2 * n  # rough upper bound
        step = 1.0 / L

        y = h.copy()
        t = 1.0
        prev_c5 = compute_c5(x)
```

```python
def find_optimal_delta(corr, g, delta_lo, delta_hi, tolerance=1e-10):
    """Find delta that minimizes max_k (corr[k] + delta * g[k]) for delta in [delta_lo, delta_hi]"""
    c_max = np.max(corr)
    
    # Active set: correlations near the max
    active = np.where(corr >= c_max - tolerance * c_max - 1e-14)[0]
    
    # Evaluate at boundaries
    candidates = [delta_lo, delta_hi]
    
    # Evaluate at breakpoints
    # For each pair of active k's, find where they cross
    n_act = len(active)
    for a in range(n_act):
        for b in range(a+1, n_act):
            ga = g[active[a]]
            gb = g[active[b]]
            if abs(ga - gb) < 1e-15:
                continue
            delta_cross = (corr[active[b]] - corr[active[a]]) / (ga - gb)
            if delta_lo <= delta_cross <= delta_hi:
                candidates.append(delta_cross)
    
    # Also, for each active k, the breakpoint with the max k
    k_star = argmax(corr)
    for a in active:
        if a == k_star:
            continue
        ga = g[a]
        gs = g[k_star]
        if abs(ga - gs) < 1e-15:
            continue
        delta_cross = (corr[k_star] - corr[a]) / (ga - gs)
        if delta_lo <= delta_cross <= delta_hi:
            candidates.append(delta_cross)
    
    # Evaluate f at all candidates
    best_delta = delta_lo
    best_val = np.inf
    for d in candidates:
        val = np.max(corr + d * g)
        if val < best_val:
            best_val = val
            best_delta = d
    
    return best_delta, best_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

def compute_grad_lse(h, alpha):
    """Compute gradient of log-sum-exp objective."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = np.clip(alpha * (corr - np.max(corr)), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    idxs = np.arange(n)
    shifts = np.arange(-(n-1), n)
    grad = np.zeros(n)
    
    for k_idx in range(len(shifts)):
        k = int(shifts[k_idx])
        w = weights[k_idx]
        if w < 1e-15:
            continue
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if v1.any():
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
        if v2.any():
            grad[v2] -= w * h[idxs[v2] + k]
    
    return grad

def fista_lse(h_init, n, target, time_limit, start_time,
              alphas=None, max_iter_per_alpha=20000, rtol=1e-14):
    """FISTA with log-sum-exp smooth approximation."""
    if alphas is None:
        alphas = [10, 50, 200, 1000, 5000, 20000]
    
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        # Lipschitz constant: ||grad|| is bounded by ~2*alpha*dx*n
        # Use a conservative estimate
        L = 2.0 * alpha * dx * n + 1e-10
        step = 1.0 / L
        
        x = h.copy()
        y = h.copy()
        t = 1.0
        prev_obj = _smooth_lse(x, alpha)
        stagnation = 0
        
        for it in range(max_iter_per_alpha):
            if time.time() - start_time > time_limit:
                break
            
            gy = compute_grad_lse(y, alpha)
            gy -= np.mean(gy)
            
            x_new = y - step * gy
            x_new = project_feasible(x_new, target)
            
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = x_new + (t - 1.0) / t_new * (x_new - x)
            y = np.clip(y, 0.0, 1.0)
            x = x_new
            t = t_new
            
            obj_new = _smooth_lse(x, alpha)
            
            # Check improvement in actual C5
            c5_curr = compute_c5(x)
            if c5_curr < best_c5 - 1e-14:
                best_c5 = c5_curr
                best_h = x.copy()
            
            if abs(prev_obj - obj_new) < rtol * max(1.0, abs(prev_obj)):
                stagnation += 1
            else:
                stagnation = 0
            prev_obj = obj_new
            
            if stagnation > 200:
                break
    
    return best_h, best_c5


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def pair_cd_v2(h_init, n, target, time_limit, start_time,
               max_sweeps=50000, n_cand_start=15, n_cand_end=60,
               max_no_improve=500, n_top_shifts=1, bisection_iters=14):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_shifts].astype(float)
            wsum = np.sum(weights)
            if wsum < 1e-15:
                weights = np.ones(n_top_shifts) / n_top_shifts
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time, use_pnorm=False):
    _g_n_global = [n]
    _g_target_global = [target]
    
    def _cons_fun(x):
        return np.sum(x) - _g_target_global[0]
    
    def _cons_jac(x):
        return np.ones(_g_n_global[0])
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for param in alphas:
        if time.time() - start_time > time_limit:

                return _pnorm_obj(x, param)
        else:
            def func(x):
                return _smooth_lse(x, param)
        
        try:
            res = minimize(func, h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score
```

For the correlation corr_k = sum_i h[i]*(1-h[i+k]), the second derivative w.r.t. h[i], h[j] is:

```python
def pair_cd_vectorized(h, n, target, shifts, corr, grad, c, bisection_iters=14):
    """Vectorized pair CD step: find the best (i, j, delta) triple."""
    idxs = np.arange(n)
    mid = n - 1
    
    # Compute gradient for top shift
    k_star = int(np.argmax(corr)) - mid
    v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    grad = np.zeros(n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    grad -= np.mean(grad)
    
    sorted_idx = np.argsort(grad)
    low_set = sorted_idx[:c]
    high_set = sorted_idx[-c:]
    
    # Vectorized bisection
    n_h = len(high_set)
    n_l = len(low_set)
    
    # Delta bounds
    h_i = h[high_set][:, None]  # (n_h, 1)
    h_j = h[low_set][None, :]   # (1, n_l)
    delta_lo = np.maximum(h_i - 1.0, -h_j)  # (n_h, n_l)
    delta_hi = np.minimum(h_i, 1.0 - h_j)   # (n_h, n_l)
    
    valid = delta_hi - delta_lo > 1e-12  # (n_h, n_l)
    
    if not valid.any():
        return None
    
    # Change vectors for all pairs
    # g_matrix[p, q, k] = change[low_set[q], k] - change[high_set[p], k]
    change_low = change[low_set]   # (n_l, n_corr)
    change_high = change[high_set]  # (n_h, n_corr)
    
    # For bisection, we need corr + delta * g for each pair
    # g for pair (p, q) = change_low[q] - change_high[p]
    # shape: (n_h, n_l, n_corr)
    
    # This might be too large. Let's do it in chunks.
    chunk_size = min(n_h, 10)
    
    best_trial = float(np.max(corr))
    best_p, best_q, best_delta = -1, -1, 0.0
    
    for p_start in range(0, n_h, chunk_size):
        p_end = min(p_start + chunk_size, n_h)
        ch = change_high[p_start:p_end]  # (chunk, n_corr)
        
        # g for all pairs in this chunk
        g = ch[None, :, None]  # (1, chunk, n_corr)
        # Actually, g[p, q, k] = change_low[q, k] - change_high[p, k]
        # = change_low[None, :, None] - change_high[:, None, :]
        g = change_low[None, :, :] - ch[:, None, :]  # (chunk, n_l, n_corr)
        
        a_d = delta_lo[p_start:p_end].copy()  # (chunk, n_l)
        b_d = delta_hi[p_start:p_end].copy()
        
        for _ in range(bisection_iters):
            mid_d = 0.5 * (a_d + b_d)  # (chunk, n_l)
            vals = corr[None, None, :] + mid_d[:, :, None] * g  # (chunk, n_l, n_corr)
            f_mid = vals.max(axis=2)  # (chunk, n_l)
            active = vals >= f_mid[:, :, None] - 1e-14
            f_prime = np.where(active, g, -np.inf).max(axis=2)  # (chunk, n_l)
            
            a_d = np.where(f_prime > 1e-12, a_d, mid_d)
            b_d = np.where(f_prime > 1e-12, b_d, mid_d)
        
        delta_star = 0.5 * (a_d + b_d)
        approx_max = (corr[None, None, :] + delta_star[:, :, None] * g).max(axis=2) * (2.0/n)
        
        # Find best in this chunk
        mask = valid[p_start:p_end]
        approx_max[~mask] = np.inf
        flat_idx = np.argmin(approx_max)
        p_local = flat_idx // n_l
        q_local = flat_idx % n_l
        if approx_max[flat_idx] < best_trial:
            best_trial = approx_max[flat_idx]
            best_p = p_start + p_local
            best_q = q_local
            best_delta = delta_star[p_local, q_local]
    
    if best_p < 0:
        return None
    
    i = int(high_set[best_p])
    j = int(low_set[best_q])
    return i, j, best_delta
```
