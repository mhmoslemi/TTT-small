import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, alpha, target, pen_penalty):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    smax = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    penalty = pen_penalty * (np.sum(x) - target) ** 2
    return smax + penalty

def soft_max_grad(x, alpha, target, pen_penalty):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        weights = np.ones(n_corr) / n_corr
    else:
        weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            v1 = np.arange(0, n - k)
            v2 = np.arange(k, n)
            grad[v1] += w * (1.0 - x[v2])
            grad[v2] -= w * x[v1]
        else:
            k_pos = -k
            v1 = np.arange(k_pos, n)
            v2 = np.arange(0, n - k_pos)
            grad[v1] += w * (1.0 - x[v2])
            grad[v2] -= w * x[v1]
    # Add gradient of penalty
    grad += 2.0 * pen_penalty * (np.sum(x) - target)
    return grad

def lbfgsb_optimize(h_start, n, target, alphas, maxiter, pen_start=1.0):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        pen = pen_start
        # Increase penalty if sum constraint is badly violated
        for pen_iter in range(5):
            res = minimize(
                lambda x: soft_max_obj(x, alpha, target, pen),
                h_cur,
                jac=lambda x: soft_max_grad(x, alpha, target, pen),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
            # Check if sum constraint is satisfied
            if abs(np.sum(h_cur) - target) < 1e-10:
                break
            pen *= 10.0
    
    return h_cur

# Global state for L-BFGS-B callbacks
_g_alpha = 100.0
_g_target = 0.0
_g_penalty = 1.0

def _lbfgs_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    smax = c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha
    s = np.sum(x) - _g_target
    return smax + _g_penalty * s * s

def _lbfgs_grad(x):
    n = len(x)
    n_corr = 2 * n - 1
    mid = n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s_total = np.sum(exp_shifted)
    if s_total < 1e-300:
        weights = np.ones(n_corr) / n_corr
    else:
        weights = exp_shifted / s_total

The standard Nesterov accelerated gradient method for minimizing f(x) with Lipschitz gradient L:
- x_{k+1} = x_k - (1/L) ∇f(x_k)
- y_{k+1} = x_{k+1} + (t_k - 1)/(t_{k+1}) (x_{k+1} - x_k)
- t_{k+1} = (1 + sqrt(1 + 4 t_k²))/2

But our function is not smooth (it's a max of smooth functions). The smooth max approximation makes it smooth, but with a large Lipschitz constant.

For the non-smooth max, we can use the "accelerated proximal gradient" or "FISTA" method, but the prox is just projection onto the feasible set.

- Compute subgradient at y_k: pick k* = argmax g_k(y_k), compute ∇g_{k*}(y_k)
- x_{k+1} = proj_C(y_k - alpha_k ∇g_{k*}(y_k))
- y_{k+1} = x_{k+1} + (t_k/(t_{k+1}))(x_{k+1} - x_k)
- t_{k+1} = (1 + sqrt(1 + 4t_k²))/2
- alpha_k = beta / t_{k+1} (for some step size beta)

The convergence rate is O(1/k²) for the function values, which is good.

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 100.0
_g_target = 0.0
_g_penalty = 1.0


def _lbfgs_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    smax = c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha
    s = np.sum(x) - _g_target
    return smax + _g_penalty * s * s


def _lbfgs_grad(x):
    n = len(x)
    n_corr = 2 * n - 1
    mid = n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s_total = np.sum(exp_shifted)
    if s_total < 1e-300:
        weights = np.ones(n_corr) / n_corr
    else:
        weights = exp_shifted / s_total
    top_m = min(40, n_corr)
    top_idx = np.argsort(weights)[-top_m:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = int(k_idx) - mid
        if k >= 0:
            v1 = np.arange(0, n - k)
            grad[v1] += w * (1.0 - x[v1 + k])
            grad[v1 + k] -= w * x[v1]
        else:
            kp = -k
            v1 = np.arange(kp, n)
            grad[v1] += w * (1.0 - x[v1 - kp])
            grad[v1 - kp] -= w * x[v1]
    grad += 2.0 * _g_penalty * (np.sum(x) - _g_target)
    return grad


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(120):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def lbfgs_optimize(h_start, target, alphas, maxiter_per_alpha=300, n_top=40):
    global _g_alpha, _g_target, _g_penalty
    n = len(h_start)
    _g_target = target
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        _g_alpha = float(alpha)
        for pen_iter in range(6):
            _g_penalty = float(1.0 + pen_iter * 50.0)
            try:
                res = minimize(
                    _lbfgs_obj, h_cur, jac=_lbfgs_grad,
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': maxiter_per_alpha, 'ftol': 1e-18, 'gtol': 1e-14}
                )
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    cur_score = sc
                    h_cur = h_new
            except Exception:
                break
            if abs(np.sum(h_cur) - target) < 1e-10:
                break
    return h_cur


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0

        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])

def subgradient_method(h_init, n, target, time_limit, max_iters=3000):
    start_t = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1

    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star_idx = int(np.argmax(corr))
        k = k_star_idx - mid
        subgrad = np.zeros(n)
        if k >= 0:
            v1 = np.arange(0, n - k)
            subgrad[v1] += (1.0 - h[v1 + k])
            subgrad[v1 + k] -= h[v1]
        else:
            kp = -k
            v1 = np.arange(kp, n)
            subgrad[v1] += (1.0 - h[v1 - kp])
            subgrad[v1 - kp] -= h[v1]
        step_size = 2.0 / (n * (it + 1) ** 0.6)
        h_new = h - step_size * subgrad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < compute_c5(h):
            h = h_new
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new.copy()

    return best_h, best_c5
