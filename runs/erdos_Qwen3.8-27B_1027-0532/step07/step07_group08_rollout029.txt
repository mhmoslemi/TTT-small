```python
def smooth_max_grad(h, p, dx):
    """Gradient of smooth max approximation, projected onto sum=0 subspace"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    corr_clipped = np.clip(corr, -500.0/p, 500.0/p)  # avoid overflow
    log_weights = p * corr_clipped - np.max(p * corr_clipped)
    weights = np.exp(log_weights)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    for k in range(2*n - 1):
        s = k - (n-1)
        w = weights[k] * dx
        if abs(w) < 1e-15:
            continue
        v1 = (idxs + s >= 0) & (idxs + s < n)
        v2 = (idxs - s >= 0) & (idxs - s < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + s])
        grad[v2] -= w * h[idxs[v2] - s]
    
    # Project gradient onto sum=0 subspace
    grad -= np.mean(grad)
    return grad
```

```python
def outer_approximation(h_init, n, target, time_limit, max_active=30, trust_radius=0.5):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    active_set = set()
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        c5 = float(np.max(corr))
        if c5 < best_score:
            best_score = c5
            best_h = h.copy()
        
        # Update active set
        top_k = np.argsort(corr)[-max_active:].tolist()
        
        # Solve LP with trust region
        # Variables: h[0..n-1], t
        # Minimize t
        # Subject to:
        #   g_k(h)·h - t ≤ c_k(h) - g_k(h)·h for k in top_k  [linearized constraint]
        #   sum(h) = target
        #   0 ≤ h ≤ 1
        #   ||h - h_current||_inf ≤ trust_radius  [trust region]
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = []
        b_ub = []
        idxs = np.arange(n)
        
        for k in top_k:
            s = k - (n-1)
            g = np.zeros(n)
            v1 = (idxs + s >= 0) & (idxs + s < n)
            v2 = (idxs - s >= 0) & (idxs - s < n)
            g[v1] = (1.0 - h[idxs[v1] + s])
            g[v2] -= h[idxs[v2] - s]
            
            f_k = float(corr[k]) / (2.0/n)  # unscaled corr
            b_k = f_k - np.dot(g, h)
            
            row = np.zeros(n + 1)
            row[:n] = g
            row[-1] = -1.0
            A_ub.append(row)
            b_ub.append(b_k)
        
        # Trust region: h - trust_radius ≤ h_current ≤ h + trust_radius
        # This gives 2n more constraints
        for i in range(n):
            row_low = np.zeros(n + 1)
            row_low[i] = -1.0
            A_ub.append(row_low)
            b_ub.append(-h[i] + trust_radius)
            
            row_high = np.zeros(n + 1)
            row_high[i] = 1.0
            A_ub.append(row_high)
            b_ub.append(h[i] + trust_radius)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.success:
            h_new = res.x[:n]
            h_new = project_feasible(h_new, target)
            score_new = compute_c5(h_new)
            
            if score_new < c5:
                # Accept: expand trust region
                h = h_new
                trust_radius = min(trust_radius * 1.5, 1.0)
            else:
                # Reject: shrink trust region
                trust_radius *= 0.5
                if trust_radius < 0.01:
                    break
        else:
            trust_radius *= 0.5
            if trust_radius < 0.01:
                break
    
    return best_h, best_score
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0
_g_lambda = 0.0
_g_p = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(x):
    n = _g_n
    p = _g_p
    lam = _g_lambda
    target = _g_target
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = p * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    obj = max_c + np.log(np.sum(np.exp(shifted))) / p
    penalty = lam * (np.sum(x) - target) ** 2
    return obj + penalty


def smooth_max_grad(x):
    n = _g_n
    p = _g_p
    lam = _g_lambda
    target = _g_target
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    max_c = np.max(corr)
    shifted = p * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k] * dx
        if abs(w) < 1e-15:

        v1 = (idxs + s >= 0) & (idxs + s < n)
        v2 = (idxs - s >= 0) & (idxs - s < n)
        if v1.any():
            grad[v1] += w * (1.0 - x[idxs[v1] + s])
        if v2.any():
            grad[v2] -= w * x[idxs[v2] - s]
    
    # Add penalty gradient
    grad += 2.0 * lam * (np.sum(x) - target)
    return grad

    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for p, lam in zip(p_list, lam_list):

def compute_c5(h):
    n = len(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:

            mask = h > 1e-15

            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break

        h = np.clip(h, 0.0, 1.0)

    n = _g_n

    lam = _g_lam

    max_c = np.max(corr)
    shifted = p * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    obj = max_c + np.log(np.sum(np.exp(shifted))) / p
    penalty = lam * (np.sum(x) - target) ** 2
    return obj + penalty


def smooth_max_grad(x):
    n = _g_n
