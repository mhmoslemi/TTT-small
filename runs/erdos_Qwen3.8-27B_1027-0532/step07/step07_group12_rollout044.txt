    dx = 2.0 / n

    obj += penalty * (np.sum(x) - target) ** 2

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    threshold = np.sort(corr)[-k]

    # For smoothness, use a soft version
    alpha = 50.0

    below = corr < threshold
    w[below] = np.exp(alpha * (corr[below] - threshold))
    
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)

        wi = w[int(m)]
        v1 = (idxs - ki >= 0) & (idxs - ki < n)
        v2 = (idxs + ki >= 0) & (idxs + ki < n)

        if np.any(v2):

    grad += 2.0 * penalty * (np.sum(x) - target)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-10
    active_idx = np.where(active_mask)[0]
    for m in active_idx:
        k = int(m) - mid
        w = weights[int(m)]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += w * (1.0 - x[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def hinge_objective(x, threshold, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    excess = np.maximum(0.0, corr - threshold)
    obj = np.sum(excess ** 2) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def hinge_gradient(x, threshold, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    excess = np.maximum(0.0, corr - threshold)
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active = np.where(excess > 1e-8)[0]
    for m in active:
        k = int(m) - mid
        w = 2.0 * excess[int(m)]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += w * (1.0 - x[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_run(h_start, n, target, alpha, maxiter, time_limit, penalty=100.0):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_objective, h_start, args=(alpha, target, penalty),
                      jac=lse_gradient, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

def lbfgs_hinge_run(h_start, n, target, threshold, maxiter, time_limit, penalty=100.0):
    start = time.time()

def pair_cd(h_init, n, target, time_limit, n_cand=40, max_sweeps=100000, n_shifts=1):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if sweep % 50 == 0 and time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = 30.0
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)

        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]
                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def subgradient_polish(h_init, n, target, time_limit, max_steps=50000, n_shifts=8, base_lr=0.03):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step % 100 == 0 and time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 1500:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 15.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= (2.0 / n) * np.sum(grad)

        improved = False
        decay = 1.0 / (1.0 + 0.003 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved and step > 3000:
            break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h_new = lbfgs_run(h_cur, n, target, alpha, maxiter_each, time_limit=0, penalty=penalty)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def tleft():
        return budget_s - 30.0 - elapsed()

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS with moderate alphas at n=600
    if tleft() > 120:
        hp, sc = multi_alpha_lbfgs(best_h, n, target,
                                   alphas=[50, 200, 1000],
                                   maxiter_each=300,
                                   time_limit=min(80, tleft() * 0.4),
                                   penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Pair CD at n=600
    if tleft() > 100:
        cd_time = min(90, tleft() * 0.3)
        hcd, c5cd = pair_cd(best_h, n, target, time_limit=cd_time, n_cand=40, max_sweeps=200000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 3: Hinge-based L-BFGS
    if tleft() > 80:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        threshold = float(np.percentile(corr, 99.0))
        hp = lbfgs_hinge_run(best_h, n, target, threshold, maxiter=200, time_limit=40, penalty=50.0)
        sc = compute_c5(hp)
        if sc < best_score:
            best_score = sc
            best_h = hp
        if tleft() > 40:
            hcd, c5cd = pair_cd(best_h, n, target, time_limit=min(30, tleft() * 0.5), n_cand=50, max_sweeps=100000, n_shifts=1)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

    # Phase 4: Subgradient polish at n=600
    if tleft() > 60:
        hsg, c5sg = subgradient_polish(best_h, n, target, time_limit=min(50, tleft() * 0.4),
                                        max_steps=50000, n_shifts=6, base_lr=0.04)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 5: Upscale to n=800
    if tleft() > 400:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 300:
            hp2, sc2 = multi_alpha_lbfgs(h2, n2, target2,
                                         alphas=[100, 500, 2000, 10000],
                                         maxiter_each=200,
                                         time_limit=min(150, tleft() * 0.4),
                                         penalty=100.0)
            if sc2 < best_score:
                best_score = sc2
                best_h = hp2
                n = n2
                target = target2
            else:
                h2 = hp2
                sc2 = compute_c5(h2)

        if tleft() > 150:
            hcd2, c5cd2 = pair_cd(h2, n2, target2, time_limit=min(100, tleft() * 0.35),
                                   n_cand=35, max_sweeps=100000, n_shifts=1)
            if c5cd2 < best_score:
                best_score = c5cd2
                best_h = hcd2
                n = n2
                target = target2
            elif c5cd2 < sc2:
                h2 = hcd2

        if tleft() > 80:
            hsg2, c5sg2 = subgradient_polish(h2, n2, target2, time_limit=min(60, tleft() * 0.3),
                                              max_steps=30000, n_shifts=8, base_lr=0.03)
            if c5sg2 < best_score:
                best_score = c5sg2
                best_h = hsg2
                n = n2
                target = target2

    # Phase 6: Try n=1000 if time permits
    if tleft() > 350:
        n3 = 1000
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)

        if tleft() > 250:
            hp3, sc3 = multi_alpha_lbfgs(h3, n3, target3,
                                         alphas=[200, 1000, 5000],
                                         maxiter_each=150,
                                         time_limit=min(120, tleft() * 0.4),
                                         penalty=200.0)
            if sc3 < best_score:
                best_score = sc3
                best_h = hp3
                n = n3
                target = target3
            else:
                h3 = hp3

        if tleft() > 100:
            hcd3, c5cd3 = pair_cd(h3, n3, target3, time_limit=min(80, tleft() * 0.3),
                                   n_cand=30, max_sweeps=50000, n_shifts=1)
            if c5cd3 < best_score:
                best_score = c5cd3
                best_h = hcd3
                n = n3
                target = target3

    # Phase 7: Binary local search attempt
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hb, c5b = binary_local_search(best_h, n_use, target_use, time_limit=min(60, tleft() * 0.5))
        if c5b < best_score:
            best_score = c5b
            best_h = hb

    # Phase 8: Restart with perturbation
    if tleft() > 150:
        n_use = len(best_h)
        target_use = n_use / 2.0
        for noise_scale in [0.01, 0.03, 0.05]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n_use) * noise_scale
            h_pert = project_feasible(h_pert, target_use)
            hp, sc = multi_alpha_lbfgs(h_pert, n_use, target_use,
                                       alphas=[500, 5000],
                                       maxiter_each=150,
                                       time_limit=min(30, tleft() * 0.3),
                                       penalty=200.0)
            if sc < best_score:
                best_score = sc
                best_h = hp
                if tleft() > 40:
                    hcd, c5cd = pair_cd(hp, n_use, target_use, time_limit=min(30, tleft() * 0.4),
                                         n_cand=30, max_sweeps=50000, n_shifts=1)
                    if c5cd < best_score:
                        best_score = c5cd
                        best_h = hcd

    # Phase 9: Final pair CD polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd(best_h, n_use, target_use, time_limit=min(30, tleft() * 0.6),
                             n_cand=40, max_sweeps=50000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 10: Final subgradient
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hsg, c5sg = subgradient_polish(best_h, n_use, target_use, time_limit=min(20, tleft() * 0.7),
                                        max_steps=20000, n_shifts=10, base_lr=0.02)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Final feasible projection and scoring
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
