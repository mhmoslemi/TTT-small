import time

    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)

    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h_start.copy()

def cutting_plane(h_init, n, target, max_iters=50, max_active=30, tol=1e-8):
    h = h_init.copy()
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        
        # Find active set
        threshold = c_max - tol
        active = np.where(corr >= threshold)[0]
        if len(active) > max_active:
            # Keep only top max_active
            active = np.argsort(corr)[-max_active:]
        
        # Compute gradients for active lags
        n_corr = 2 * n - 1
        mid = n - 1
        n_active = len(active)
        
        A = np.zeros((n_active, n + 1))
        b = np.zeros(n_active)
        
        for idx, k in enumerate(active):
            s = k - mid  # shift
            grad = np.zeros(n)
            if s >= 0:
                valid_i = np.arange(n - s)
                grad[valid_i] += (1.0 - h[valid_i + s])
                valid_i2 = np.arange(s, n)
                grad[valid_i2] -= h[valid_i2 - s]
            else:
                s_abs = -s
                valid_i = np.arange(n - s_abs)
                grad[valid_i] += (1.0 - h[valid_i + s_abs])
                valid_i2 = np.arange(s_abs, n)
                grad[valid_i2] -= h[valid_i2 - s_abs]
            
            A[idx, :n] = grad
            A[idx, n] = -1.0
            b[idx] = float(np.dot(grad, h)) - corr[k]
        
        # Equality constraint: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Objective: minimize t
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        bounds = [(0.0, 1.0)] * n + [(None, None)]
        
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        h_new = project_feasible(res.x[:n], target)
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        
        if c5_new < c5_old - 1e-12:
            h = h_new
        else:
            break
    
    return h, compute_c5(h)

from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new

def compute_corr_grad(h, s, n):
    """Compute gradient of corr_s with respect to h."""
    s = abs(s)
    if s == 0:
        # corr_0 = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
        # d/dh[i] = 1 - 2*h[i]
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    # Terms where h[i] appears as h[j] (j < n-s): contributes (1-h[i+s])
    valid1 = np.arange(0, n - s)
    grad[valid1] += (1.0 - h[valid1 + s])
    # Terms where h[i] appears as h[j+s] (j >= s): contributes -h[i-s]
    valid2 = np.arange(s, n)
    grad[valid2] -= h[valid2 - s]
    return grad

def cutting_plane(h_init, n, target, max_iters=50, max_active=30, tol=1e-10):
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        
        # Find active set: lags with corr >= c_max - tol
        threshold = c_max * (1.0 - 1e-6) - 1e-10
        active = np.where(corr >= threshold)[0]
        if len(active) > max_active:
            active = np.argsort(corr)[-max_active:]
        
        n_active = len(active)
        if n_active == 0:
            break
        
        # Build LP
        A = np.zeros((n_active, n + 1))
        b = np.zeros(n_active)
        
        for idx, k in enumerate(active):
            s = k - mid
            grad = compute_corr_grad(h, s, n)
            A[idx, :n] = grad
            A[idx, n] = -1.0
            b[idx] = float(np.dot(grad, h)) - corr[k]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        bounds_lp = [(0.0, 1.0)] * n + [(None, None)]
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
        except Exception:
            break
        
        if not res.success:
            break
        
        h_new = project_feasible(res.x[:n], target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try a smaller step
            alpha = 0.5
            h_trial = h + alpha * (h_new - h)
            h_trial = project_feasible(h_trial, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-13:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial
            else:
                break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

        best_j = -1

                    delta_max = min(h[i_val], 1.0 - h[j_val])
                    if delta_max < 1e-12:

                    g = change[j_val] - change[i_val]
                    # Bisection
                    a_d = 0.0
                    b_d = delta_max
                    for _ in range(bisection_iters):

                        f_mid = np.max(vals)

                        if active_mask.any():

                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:

                        best_j = j_val

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            # Fix sum
            diff = target - np.sum(h)
            if abs(diff) > 1e-12:
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
            
            # Update corr incrementally
            corr += best_delta * (change[best_j] - change[best_i])
            
            # Update change matrix for affected rows
            for p in [best_i, best_j]:
                for k in range(n_corr):
                    s_k = abs(k - mid)
                    val = 0.0
                    if s_k == 0:

                        if p >= s_k:
                            val -= h[p - s_k]
                    change[p, k] = val
    
    return best_h, best_c5

Actually, option 4 is interesting. The function f(delta) = max_k (corr[k] + delta * g[k]) is a piecewise linear function (maximum of linear functions). We want the smallest delta such that f(delta) ≤ f(0) (or the delta that minimizes f).

Since f is convex in delta (it's a max of linear functions), and we want to minimize it over delta ∈ [0, delta_max], the optimal delta is either:
- 0 (if f is non-decreasing at 0)
- delta_max (if f is non-increasing at delta_max)
- A "kink" point (where two lines cross)

The kink points are at delta = (corr[k1] - corr[k2]) / (g[k2] - g[k1]) for pairs (k1, k2) where g[k1] ≠ g[k2].

But finding the exact minimum of a max of linear functions is equivalent to solving a small LP, which is what we're already doing implicitly with the bisection.

A faster approach: since we only care about the top few lags (the ones near the maximum), we can restrict the bisection to those. Specifically:
- The maximum of corr is at lags T = top_idx
- After the update, the maximum can only be at lags where corr[k] + delta*g[k] is large
- For small delta, this is still the top lags
- So we can check: max over top_idx of (corr[k] + delta*g[k])

This reduces the bisection cost from O(n_corr) to O(n_active) per step, which is much faster!

import numpy as np
from scipy.optimize import minimize, linprog

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_corr_grad(h, s, n):
    s = abs(s)
    if s == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    valid1 = np.arange(0, n - s)
    grad[valid1] += (1.0 - h[valid1 + s])
    valid2 = np.arange(s, n)
    grad[valid2] -= h[valid2 - s]
    return grad


def cutting_plane(h_init, n, target, max_iters=50, max_active=25, tol_frac=1e-6):
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1

    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))

        threshold = c_max * (1.0 - tol_frac)
        active = np.where(corr >= threshold)[0]
        if len(active) > max_active:
            active = np.argsort(corr)[-max_active:]

        n_active = len(active)
        if n_active == 0:
            break

        A = np.zeros((n_active, n + 1))
        b = np.zeros(n_active)

        for idx, k in enumerate(active):
            s = int(k) - mid
            grad = compute_corr_grad(h, s, n)
            A[idx, :n] = grad
            A[idx, n] = -1.0
            b[idx] = float(np.dot(grad, h)) - corr[k]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds_lp = [(0.0, 1.0)] * n + [(None, None)]

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
        except Exception:
            break

        if not res.success:
            break

        h_new = project_feasible(res.x[:n], target)
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)

        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new

        rng2 = np.random.RandomState(seed + 100)
        h_bin = np.zeros(n0)
        half = n0 // 2
        idxs_bin = rng2.permutation(n0)
        h_bin[idxs_bin[:half]] = 1.0
        h_bin = project_feasible(h_bin, target0)
        h_cp, c5_cp = cutting_plane(h_bin, n0, target0, max_iters=40, max_active=25, tol_frac=1e-4)
        c5_cp_final = compute_c5(h_cp)
        if c5_cp_final < best_score:
            best_score = c5_cp_final
            best_h = h_cp

    # Stage 8: Final refinement
    if tleft() > 60:
        h_cd, c5_cd = fast_cd(best_h, n0, target0, time_limit=min(30, tleft() * 0.4),
                              max_sweeps=20000, n_active=5, n_cand=35,
                              max_no_improve=150, bisection_iters=8)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 40:
        h_cp, c5_cp = cutting_plane(best_h, n0, target0, max_iters=30, max_active=30, tol_frac=1e-9)
        if c5_cp < best_score:
            best_score = c5_cp
            best_h = h_cp

    if tleft() > 30:
        h_pol, sc = slsqp_optimize(best_h, n0, target0, [5000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
