```python
def project_feasible(h, target_sum):

            break

        h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
```

```python
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find active constraints (top correlations)
        top_k = min(20, n_corr)
        top_indices = np.argsort(corr)[-top_k:]
        
        # Build linearization
        new_constraints = []
        for idx in top_indices:
            k = int(idx) - mid
            c_val = corr[idx]
            grad = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] = (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
            # Linearization: c_val + grad·(h_new - h) <= t
            # => grad·h_new <= t - c_val + grad·h
            rhs = -c_val + np.dot(grad, h)
            new_constraints.append((grad, rhs))
        
        # Solve LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
        for grad, rhs in new_constraints:
            cons.append(grad @ h_var - t_var <= rhs)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.CBC)
        except:
            try:
                prob.solve(solver=cp.GLPK)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        new_score = compute_c5(h_new)
        
        if new_score < c5_val - 1e-10:
            h = h_new
        else:
            # No improvement, try with more constraints or break
            if it > 50:
                break
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def cd_optimize(h_init, n, target, time_limit, max_sweeps, n_cand_start, n_cand_end, max_no_improve, rng):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            pert_scale = min(0.1, 0.02 * (consecutive_no_improve / max_no_improve))
            h = best_h + rng.randn(n) * pert_scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0
            continue

        top_k = min(4, n_corr)
        top_shifts = np.argsort(corr)[-top_k:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

            c_val = corr[idx]
            grad = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] = (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
            rhs = -c_val + np.dot(grad, h)
            new_grads.append(grad)
            new_rhs.append(rhs)

        if it < 5:
            cut_grads = new_grads
            cut_rhs = new_rhs
        else:
            for g, r in zip(new_grads, new_rhs):
                cut_grads.append(g)
                cut_rhs.append(r)
            if len(cut_grads) > 40:
                h_test = h.copy()
                scores = np.array([np.dot(g, h_test) - r for g, r in zip(cut_grads, cut_rhs)])
                keep = np.argsort(-scores)[:40]
                cut_grads = [cut_grads[i] for i in keep]
                cut_rhs = [cut_rhs[i] for i in keep]

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons_list = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
        for g, r in zip(cut_grads, cut_rhs):
            cons_list.append(cp.multiply(g, h_var).sum() - t_var <= r)

        prob = cp.Problem(cp.Minimize(t_var), cons_list)
        solved = False
        for solver in [cp.CBC, cp.GLPK, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if h_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        new_score = compute_c5(h_new)

        if new_score < c5_val - 1e-10:
            h = h_new
        elif it > 30:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    rng = np.random.RandomState(seed)
    current_n = n0
    current_target = target0

    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [100, 500, 2000, 10000, 50000], 600, 1e-15)

            best_score = sc
            best_h = h_pol

    if tleft() > 120:

            best_score = sc_cd
            best_h = h_cd

        h_cp, sc_cp = cutting_plane_optimize(best_h, current_n, current_target,

    if tleft() > 50:

    if tleft() > 350:

                    best_score = sc_cd1
                    best_h = h_cd1

            if tleft() > 80:
                h_cp1, sc_cp1 = cutting_plane_optimize(best_h, current_n, current_target,
                                                        min(50, tleft() * 0.6), 60, 15)
                if sc_cp1 < best_score:
                    best_score = sc_cp1
                    best_h = h_cp1

            if tleft() > 50:
                h_pol1, sc1b = slsqp_round(best_h, current_n, current_target,
                                            [10000, 50000, 200000], 500, 1e-15)
                if sc1b < best_score:
                    best_score = sc1b
                    best_h = h_pol1

    if tleft() > 350:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score * 1.02:
            h2a, sc2a = slsqp_round(h2, n2, target2,
                                     [100, 1000, 5000], 400, 1e-15)
            if sc2a < best_score:
                best_score = sc2a
                best_h = h2a
                current_n = n2
                current_target = target2

            if tleft() > 150:
                h_cd2, sc_cd2 = cd_optimize(best_h, current_n, current_target,
                                            min(150, tleft() * 0.55), 12000, 20, 55, 500, rng)

                    best_score = sc_cd2
                    best_h = h_cd2

                h_cp2, sc_cp2 = cutting_plane_optimize(best_h, current_n, current_target,

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
```
