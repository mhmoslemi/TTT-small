import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def topk_objective(x, target, k, penalty):
    """Minimize sum of top-k correlation values."""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Sum of top k values
    top_vals = np.sort(corr)[::-1][:k]
    obj = float(np.sum(top_vals)) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def topk_gradient(x, target, k, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    sorted_idx = np.argsort(corr)[::-1]
    top_k_idx = sorted_idx[:k]
    # Gradient is sum of gradients of top-k functions
    grad = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    for m in top_k_idx:
        mk = int(m) - mid
        v1 = (idxs - mk >= 0) & (idxs - mk < n)
        v2 = (idxs + mk >= 0) & (idxs + mk < n)
        if np.any(v1):
            grad[v1] += (1.0 - x[idxs[v1] - mk])
        if np.any(v2):
            grad[v2] -= x[idxs[v2] + mk]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):

    dx = 2.0 / n

def lbfgs_optimize(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0, use_topk=False, topk=5):

    bounds = [(0.0, 1.0)] * n

                                  method='L-BFGS-B', bounds=bounds,
                                  options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
                else:

                h_new = project_feasible(res.x, target)

                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass

                grad[v2] -= wi * h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))

            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]

                profiles[ci, valid2] += h[idx2[valid2]]

                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue

                g = profiles[ci] - profiles[cj]

                hi_d = delta_max

                        hi_d = mid_d
                    else:
                        lo = mid_d

                if delta_star < 1e-10:

                improve = cur_max - new_max

                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

def binarize_and_refine(h_init, n, target, time_limit):

    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Binarize
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Fix sum
    target_int = int(round(target))
    num_ones = int(np.sum(h_bin))
    if num_ones > target_int:
        # Remove excess ones (those closest to 0.5)
        ones_idx = np.where(h_bin == 1.0)[0]
        sort_by_h = np.argsort(h[ones_idx])
        remove = sort_by_h[:num_ones - target_int]
        h_bin[ones_idx[remove]] = 0.0
    elif num_ones < target_int:
        zero_idx = np.where(h_bin == 0.0)[0]
        sort_by_h = np.argsort(-h[zero_idx])
        add = sort_by_h[:target_int - num_ones]
        h_bin[zero_idx[add]] = 1.0

    h_cur = h_bin.copy()
    c5_cur = compute_c5(h_cur)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur.copy()

    # Combinatorial local search: try flips
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    while time.time() - start < time_limit * 0.4 and no_improve < 200:
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        cur_max = float(np.max(corr))

        improved = False
        for i in range(n):
            if time.time() - start > time_limit * 0.4:
                break
            old_val = h_cur[i]
            new_val = 1.0 - old_val
            if new_val < 0 or new_val > 1:
                continue

            # corr changes at positions n-1 + (i - j) for all j where h[j] changes
            # Actually, let's just compute the new corr

            new_corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                improved = True
                no_improve = 0
            else:
                h_cur[i] = old_val
                no_improve += 1
        if not improved and no_improve >= 200:
            break
        if not improved:
            no_improve += 100

    if compute_c5(h_cur) < best_c5:

def subgradient_phase(h_init, n, target, time_limit, max_steps=50000, n_shifts=5, base_lr=0.05):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 1000:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 25.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= (2.0 / n) * np.sum(grad)

        improved = False
        decay = 1.0 / (1.0 + 0.003 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1]:

            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

            break

def interpolate_to(h_src, n_dst, target_dst):

def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]

    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2

            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(20):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-15)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-15:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-11:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def subgradient_phase(h_init, n, target, time_limit, max_steps=50000, n_shifts=5, base_lr=0.05):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800:
                break
        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= (2.0 / n) * np.sum(grad)
        improved = False
        decay = 1.0 / (1.0 + 0.004 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved and step > 800:
            break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 80:
        hp, sc = lbfgs_optimize(best_h, n, target,
            alphas=[50, 200, 1000], maxiter=400, time_limit=min(60, tleft() * 0.3),
            penalty=200.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 100:
        hcd, c5cd = pair_cd_v2(best_h, n, target,
            time_limit=min(90, tleft() * 0.3), n_cand=25, max_sweeps=80000, n_top=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 80:
        hp, sc = lbfgs_optimize(best_h, n, target,
            alphas=[2000, 10000, 50000], maxiter=400, time_limit=min(60, tleft() * 0.25),
            penalty=1000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 120:
        hcd, c5cd = pair_cd_v2(best_h, n, target,
            time_limit=min(130, tleft() * 0.3), n_cand=40, max_sweeps=100000, n_top=3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 60:
        hp, sc = lbfgs_optimize(best_h, n, target,
            alphas=[50000, 200000, 1000000], maxiter=300, time_limit=min(50, tleft() * 0.3),
            penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 120:
        hcd, c5cd = pair_cd_v2(best_h, n, target,
            time_limit=min(130, tleft() * 0.3), n_cand=55, max_sweeps=80000, n_top=5)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 60:
        hsg, c5sg = subgradient_phase(best_h, n, target,
            time_limit=min(40, tleft() * 0.4), max_steps=30000, n_shifts=7, base_lr=0.06)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    if tleft() > 180:
        n_new = 800
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = lbfgs_optimize(h_interp, n_new, target_new,
            alphas=[100, 1000, 10000, 100000], maxiter=300,
            time_limit=min(70, tleft() * 0.3), penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new
        if tleft() > 80:
            hcd, c5cd = pair_cd_v2(best_h, n, target,
                time_limit=min(90, tleft() * 0.4), n_cand=40, max_sweeps=60000, n_top=3)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

    if tleft() > 150:
        rng = np.random.RandomState(seed)
        n_use = len(best_h)
        target_use = n_use / 2.0
        for attempt in range(4):
            if tleft() < 40:
                break
            noise_scale = 0.01 + 0.01 * attempt
            h_pert = best_h + rng.randn(n_use) * noise_scale
            h_pert = project_feasible(h_pert, target_use)
            hp, sc = lbfgs_optimize(h_pert, n_use, target_use,
                alphas=[5000, 50000], maxiter=200,
                time_limit=min(25, tleft() * 0.15), penalty=500.0)
            if sc < best_score:
                best_score = sc
                best_h = hp
                if tleft() > 50:
                    hcd, c5cd = pair_cd_v2(hp, n_use, target_use,
                        time_limit=min(35, tleft() * 0.2), n_cand=30, max_sweeps=30000, n_top=3)
                    if c5cd < best_score:
                        best_score = c5cd
                        best_h = hcd

    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_optimize(best_h, n_use, target_use,
            alphas=[500000, 2000000, 10000000], maxiter=200, time_limit=min(40, tleft() * 0.5),
            penalty=20000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = pair_cd_v2(best_h, n_use, target_use,
            time_limit=min(25, tleft() * 0.8), n_cand=50, max_sweeps=40000, n_top=5)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
