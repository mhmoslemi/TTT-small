    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h0.copy()
    best_sc = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(alpha)
        try:
            res = minimize(lse_obj, h_cur, jac=lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_sc


def projected_pg(h0, n, target, deadline, alpha_schedule, max_iters=5000):
    global _g_alpha
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if no_improve > 800:
            break
        alpha = alpha_schedule[min(it, len(alpha_schedule) - 1)]
        _g_alpha = float(alpha)
        val = compute_c5(h)
        if val < best_val - 1e-15:
            best_val = val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        grad = lse_grad(h)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 0.5 / gnorm
        improved = False
        for _ in range(25):
            h_try = h - step * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < val - 1e-12 * step:
                h = h_try
                improved = True
                break
            step *= 0.5
        if not improved:
            h = best_h.copy()
            val = best_val
    return best_h, best_val


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def pair_cd(h0, n, target, deadline, max_sweeps=50000,
            n_cand_start=30, n_cand_end=80, top_k_eval=50,
            max_no_improve=1000, n_shifts_grad=3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h0.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        k_star_idx = int(top_shifts[-1])

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx]
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h = change[high_set, k_star_idx]
        ch_l = change[low_set, k_star_idx]
        score_mat = ch_l[None, :] - ch_h[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(10):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fourier_perturb(h, n, target, rng, epsilon):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    freq = rng.choice(np.array([1, 2, 3, 4, 5, 6, 7, 8, 10, 12], dtype=float))
    phase = rng.uniform(0, 2.0 * np.pi)
    pert = epsilon * np.sin(np.pi * freq * x + phase)
    h_new = h + pert
    return project_feasible(h_new, target)


def block_cyclic_shift(h, n, target, rng):
    b = int(rng.choice([4, 6, 8, 10, 12, 16, 20]))
    b = min(b, n // 3)
    start = int(rng.randint(0, n - b + 1))
    h_new = h.copy()
    h_new[start:start + b] = np.roll(h[start:start + b], 1)
    return project_feasible(h_new, target)


def stochastic_hill_climb(h0, n, target, deadline, rng, max_steps=10000):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    no_improve = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        if no_improve > 2000:
            break
        move_type = rng.randint(3)
        if move_type == 0:
            scale = 0.03 * max(0.2, 1.0 - step / max_steps)
            n_pert = int(rng.choice([3, 5, 8, 12, 20]))
            n_pert = min(n_pert, n)
            idxs = rng.choice(n, size=n_pert, replace=False)
            h_try = h.copy()
            h_try[idxs] += rng.randn(n_pert) * scale
        elif move_type == 1:
            h_try = fourier_perturb(h, n, target, rng,
                                     0.02 * max(0.3, 1.0 - step / max_steps))
        else:
            h_try = block_cyclic_shift(h, n, target, rng)
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < cur_val - 1e-14:
            h = h_try
            cur_val = sc
            no_improve = 0
            if sc < best_val - 1e-14:
                best_val = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 300:
                h = best_h.copy()
                cur_val = best_val
                no_improve = 0
    return best_h, best_val


def run(seed=42, budget_s=1300, **kwargs):
    global _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    def time_left():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: SLSQP at initial resolution
    if time_left() > 80:
        deadline = start_time + 70
        h_p, sc_p = slsqp_optimize(best_h, n0, target0,
                                    [100, 500, 2000, 8000], 500, deadline)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p

    # Phase 2: Upsample to n=1000
    if time_left() > 500:
        n1 = 1000
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            n0 = n1
            target0 = target1

    # Phase 3: Projected gradient at n=1000
    if time_left() > 400:
        deadline = start_time + 150
        alpha_sched = [200, 500, 1000, 2000, 5000, 10000, 20000]
        h_pg, sc_pg = projected_pg(best_h, n0, target0, deadline,
                                    alpha_sched, max_iters=3000)
        if sc_pg < best_score:
            best_score = sc_pg
            best_h = h_pg

    # Phase 4: Pair CD at current resolution
    if time_left() > 250:
        deadline = start_time + 220
        h_cd, sc_cd = pair_cd(best_h, n0, target0, deadline,
                               max_sweeps=60000,
                               n_cand_start=30, n_cand_end=80,
                               top_k_eval=50,
                               max_no_improve=1500,
                               n_shifts_grad=3)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 5: SLSQP polish
    if time_left() > 150:
        deadline = start_time + 120
        h_p, sc_p = slsqp_optimize(best_h, n0, target0,
                                    [1000, 5000, 20000, 50000], 800, deadline)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p

    # Phase 6: Stochastic hill climb with structured moves
    if time_left() > 180:
        deadline = start_time + 160
        h_sh, sc_sh = stochastic_hill_climb(best_h, n0, target0,
                                             deadline, rng, max_steps=8000)
        if sc_sh < best_score:
            best_score = sc_sh
            best_h = h_sh

    # Phase 7: Pair CD second round
    if time_left() > 200:
        deadline = start_time + 180
        h_cd2, sc_cd2 = pair_cd(best_h, n0, target0, deadline,
                                 max_sweeps=50000,
                                 n_cand_start=40, n_cand_end=90,
                                 top_k_eval=60,
                                 max_no_improve=1200,
                                 n_shifts_grad=3)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 8: Try upsample to n=1600
    if time_left() > 350:
        n2 = 1600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            n0 = n2
            target0 = target2

        if time_left() > 200:
            deadline = start_time + 180
            h_cd3, sc_cd3 = pair_cd(best_h, n0, target0, deadline,
                                     max_sweeps=40000,
                                     n_cand_start=30, n_cand_end=70,
                                     top_k_eval=50,
                                     max_no_improve=1000,
                                     n_shifts_grad=3)
            if sc_cd3 < best_score:
                best_score = sc_cd3
                best_h = h_cd3

        if time_left() > 80:
            deadline = start_time + 70
            h_p3, sc_p3 = slsqp_optimize(best_h, n0, target0,
                                          [2000, 10000, 30000], 600, deadline)
            if sc_p3 < best_score:
                best_score = sc_p3
                best_h = h_p3

    # Phase 9: Final pair CD
    if time_left() > 120:
        deadline = start_time + 100
        h_cd_f, sc_cd_f = pair_cd(best_h, n0, target0, deadline,
                                   max_sweeps=40000,
                                   n_cand_start=40, n_cand_end=100,
                                   top_k_eval=70,
                                   max_no_improve=800,
                                   n_shifts_grad=4)
        if sc_cd_f < best_score:
            best_score = sc_cd_f
            best_h = h_cd_f

    # Phase 10: Final SLSQP
    if time_left() > 50:
        deadline = start_time + 40
        h_final, sc_final = slsqp_optimize(best_h, n0, target0,
                                            [5000, 20000, 80000], 400, deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 11: Very short final pair CD
    if time_left() > 30:
        deadline = start_time + 22
        h_cd_last, sc_cd_last = pair_cd(best_h, n0, target0, deadline,
                                         max_sweeps=10000,
                                         n_cand_start=50, n_cand_end=100,
                                         top_k_eval=80,
                                         max_no_improve=200,
                                         n_shifts_grad=4)
        if sc_cd_last < best_score:
            best_score = sc_cd_last
            best_h = h_cd_last

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
