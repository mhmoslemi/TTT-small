import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    start = time.time()
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),

            sc = compute_c5(h_new)

                h_cur = h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15, time_limit=300):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    start = time.time()
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def find_optimal_delta(corr, g, delta_lo, delta_hi, n_binsearch):
    if delta_hi - delta_lo < 1e-12:
        return 0.0
    a_d = delta_lo
    b_d = delta_hi
    for _ in range(n_binsearch):
        mid_d = 0.5 * (a_d + b_d)
        vals = corr + mid_d * g
        f_mid = np.max(vals)
        active_mask = vals >= f_mid - 1e-14
        if active_mask.any():
            f_prime = np.max(g[active_mask])
        else:
            f_prime = 0.0
        if f_prime > 1e-12:
            b_d = mid_d
        else:
            a_d = mid_d
    return 0.5 * (a_d + b_d)

            best_h = h_bin_cd

    # Phase 6: SLSQP after potential improvement
    if tleft() > 100:
        h_pol, sc_pol = slsqp_optimize(best_h, n0, target0,
                                        [10000, 50000, 200000],
                                        3000, 1e-15, time_limit=min(90, tleft()))
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Phase 7: Coarse-to-fine n=600
    if tleft() > 350:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 250:
            h_cd1, c5_cd1 = pair_cd_fast(h1, n1, target1,
                                          time_limit=min(220, tleft() * 0.6),
                                          max_sweeps=35000,
                                          n_cand=50, max_no_improve=400,
                                          n_binsearch=16, stochastic_frac=0.03,
                                          seed=251, alpha_smooth=100.0)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
        if tleft() > 120:
            h_pol1, sc_pol1 = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                              [5000, 30000, 100000],
                                              2000, 1e-15, time_limit=min(100, tleft()))
            if sc_pol1 < best_score:
                best_score = sc_pol1
                best_h = h_pol1

    # Phase 8: Coarse-to-fine n=800
    if tleft() > 250:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 180:
            h_cd2, c5_cd2 = pair_cd_fast(h2, n2, target2,
                                          time_limit=min(160, tleft() * 0.6),
                                          max_sweeps=25000,
                                          n_cand=40, max_no_improve=350,
                                          n_binsearch=14, stochastic_frac=0.02,
                                          seed=389, alpha_smooth=80.0)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
        if tleft() > 80:
            h_pol2, sc_pol2 = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                              [10000, 50000],
                                              2000, 1e-15, time_limit=min(70, tleft()))
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_h = h_pol2

    # Phase 9: Final pair CD
    if tleft() > 80:
        cur_n = len(best_h)
        cur_target = cur_n / 2.0
        h_cd_final, c5_final = pair_cd_fast(best_h, cur_n, cur_target,
                                             time_limit=min(70, tleft() * 0.75),
                                             max_sweeps=20000,
                                             n_cand=80, max_no_improve=300,
                                             n_binsearch=18, stochastic_frac=0.05,
                                             seed=501, alpha_smooth=200.0)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    # Phase 10: Final SLSQP
    if tleft() > 30:
        cur_n = len(best_h)
        cur_target = cur_n / 2.0
        h_final, sc_final = slsqp_optimize(best_h, cur_n, cur_target,
                                            [50000, 200000],
                                            2000, 1e-15, time_limit=min(25, tleft()))
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
