def cutting_plane_cvx(h_init, n, target, time_limit, max_iters=5000, tol=1e-10):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    cuts_g = []
    cuts_f = []
    cuts_h = []
    
    max_cuts = 200
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
        
        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])
        
        # Get gradient of the active constraint
        k = k_star - mid
        idxs = np.arange(n)
        g = np.zeros(n)
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        g[v1] += (1.0 - h[idxs[v1] + k])
        g[v2] -= h[idxs[v2] - k]
        
        cuts_g.append(g)
        cuts_f.append(f_val)
        cuts_h.append(h.copy())
        
        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]
        
        n_cuts = len(cuts_g)
        
        # Solve LP: min t s.t. g_i^T h - t <= g_i^T h_i - f_i, sum(h) = target, 0 <= h <= 1
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for i in range(n_cuts):
            constraints.append(cuts_g[i] @ h_var - t_var <= cuts_g[i] @ cuts_h[i] - cuts_f[i])
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=cp.GLOP)
        except:
            try:
                prob.solve(solver=cp.HIGHS)
            except:
                prob.solve(solver=cp.ECOS)
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        
        t_val = float(t_var.value)
        gap = sc / dx - t_val
        
        if gap < tol:
            h = h_new
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h.copy()
            break
        
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_score

def project_feasible(h, target_sum):

    n = len(h)
    for _ in range(100):

        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14

            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_for_lag(h, k):
    """Compute gradient of f_k(h) = sum_i h[i]*(1-h[i+k]) with respect to h"""
    n = len(h)
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    g[v1] += (1.0 - h[idxs[v1] + k])
    g[v2] -= h[idxs[v2] - k]
    return g

def cutting_plane_multi(h_init, n, target, time_limit, max_iters=10000, 
                         n_cuts_keep=300, n_new_cuts=5, tol=1e-10):
    """Cutting plane method with multiple cuts per iteration"""
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    cuts_g = []  # gradients
    cuts_c = []  # constants (g^T h - f)
    
    for it in range(max_iters):

            break

        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()

        max_corr = np.max(corr)
        active_threshold = max_corr - 1e-6
        active_lags = np.where(corr >= active_threshold)[0]

        new_cuts_added = 0
        for k_star in active_lags[:n_new_cuts]:

            g = gradient_for_lag(h, k)

                scores.append(slack)
            scores = np.array(scores)
            # Keep the most active (tightest) cuts

            cuts_g = [cuts_g[i] for i in keep_idx]
            cuts_c = [cuts_c[i] for i in keep_idx]

        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = cuts_c[i]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds_lp = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
        
        try:

                best_h = h_new.copy()
            h = h_new

                break
        except Exception as e:
            continue
    
    return best_h, best_score

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_for_lag(h, k):
    n = len(h)
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    g[v1] += (1.0 - h[idxs[v1] + k])
    g[v2] -= h[idxs[v2] - k]
    return g

def cutting_plane(h_init, n, target, time_limit, max_iters=5000,
                  max_cuts=200, n_new_cuts=3, tol=1e-10):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    cuts_g = []
    cuts_c = []
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
        
        # Find active lags
        max_corr = float(np.max(corr))
        # Add cuts for the top active lags
        sorted_lags = np.argsort(corr)[::-1]
        n_to_add = min(n_new_cuts, len(sorted_lags))
        
        for idx in range(n_to_add):
            k_star = int(sorted_lags[idx])
            if corr[k_star] < max_corr - 0.01:  # Only add if close to max
                break
            k = k_star - mid
            f_val = float(corr[k_star])
            g = gradient_for_lag(h, k)
            c_val = float(g @ h) - f_val
            cuts_g.append(g)
            cuts_c.append(c_val)
        
        # Manage cut set
        if len(cuts_g) > max_cuts:
            slacks = np.array([cuts_c[i] - cuts_g[i] @ h for i in range(len(cuts_g))])
            # Keep the tightest (most negative slack) cuts
            keep = np.argsort(slacks)[:max_cuts]
            cuts_g = [cuts_g[i] for i in keep]
            cuts_c = [cuts_c[i] for i in keep]
        
        # Solve LP
        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_ub_rows = []
        b_ub_rows = []
        for i in range(n_cuts):
            row = np.zeros(n + 1)
            row[:n] = cuts_g[i]
            row[n] = -1.0
            A_ub_rows.append(row)
            b_ub_rows.append(cuts_c[i])
        
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_rows)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds_lp = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                continue
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            
            t_val = float(res.x[n])
            gap = sc / dx - t_val
            
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
            
            if gap < tol:
                break
        except Exception:
            continue
    
    return best_h, best_score

def coord_descent_polish(h_init, n, target, time_limit, max_sweeps=10000):
    """Fast coordinate descent: try single-coordinate moves"""
    start = time.time()
    mid = n - 1
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    # Precompute effect vectors for each coordinate
    # effect[i] = change in corr when h[i] increases by 1
    shifts = np.arange(-(n - 1), n)
    effect = np.zeros((n, n_corr))
    for i in range(n):
        pos = (i + shifts >= 0) & (i + shifts < n)
        neg = (i - shifts >= 0) & (i - shifts < n)
        effect[i, pos] += (1.0 - h[i + shifts[pos]])
        effect[i, neg] -= h[i - shifts[neg]]
    
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            break
        
        # Try each coordinate: find best delta
        best_move = None
        best_c5 = c5_val
        
        for i in range(n):
            # Find best delta for coordinate i
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-12:
                continue
            
            # Binary search for best delta
            for _ in range(10):
                mid_d = 0.5 * (lo + hi)
                if abs(mid_d) < 1e-12:
                    break
                new_corr = corr + mid_d * effect[i]
                new_max = float(np.max(new_corr)) * dx
                if new_max < best_c5:
                    lo = mid_d
                else:
                    hi = mid_d
            
            best_delta = 0.5 * (lo + hi)
            if abs(best_delta) < 1e-12:
                continue
            
            new_corr = corr + best_delta * effect[i]
            new_c5 = float(np.max(new_corr)) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_move = (i, best_delta)
        
        if best_move is not None:
            i, delta = best_move
            h[i] += delta
            h[i] = np.clip(h[i], 0.0, 1.0)
            # Update effect for changed coordinate
            # (skip for speed - will be recomputed next iteration)
    
    return best_h, best_score

def binary_local_search(h_init, n, target, time_limit, max_iters=100000):
    """Local search on binary solutions (h[i] in {0,1})"""
    start = time.time()
    
    # Round to binary
    h = (h_init > 0.5).astype(float)
    # Adjust to satisfy sum constraint
    # Need exactly n/2 ones
    n_ones = int(np.sum(h))
    target_ones = n // 2
    
    # Adjust
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    
    if n_ones > target_ones:
        # Remove some ones
        excess = n_ones - target_ones
        # Remove ones at random positions
        for _ in range(excess):
            idx = np.random.choice(ones_idx)
            h[idx] = 0
            ones_idx = ones_idx[ones_idx != idx]
    elif n_ones < target_ones:
        # Add some ones
        deficit = target_ones - n_ones
        for _ in range(deficit):
            idx = np.random.choice(zeros_idx)
            h[idx] = 1
    
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Try flipping a random pair (one 1→0 and one 0→1)
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        i = int(np.random.choice(ones_idx))
        j = int(np.random.choice(zeros_idx))
        
        h[i] = 0
        h[j] = 1
        sc = compute_c5(h)
        
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
        else:
            # Revert (or accept with some probability for simulated annealing)
            h[i] = 1
            h[j] = 0
    
    return best_h, best_score

One more thought: I should also try **subgradient descent with Nesterov acceleration** as a complement to the cutting plane method. The subgradient method is simpler and might find different local optima.

For the subgradient method:
- At each iteration, compute the active lag k*
- Compute the subgradient g = ∇f_{k*}(h)
- Take a step: h_new = proj(h - α * g)
- Use a diminishing step size: α_t = α_0 / sqrt(t) or α_0 / t

With Nesterov acceleration:
- y_t = h_t + β_t * (h_t - h_{t-1})
- h_{t+1} = proj(y_t - α_t * g(y_t))
- β_t = (1 - sqrt(μ)) / (1 + sqrt(μ)) where μ is the strong convexity parameter (0 in our case, so β_t → 1)

The issue with subgradient descent is that it's slow (O(1/sqrt(t)) convergence). But it's simple and can be run for a long time.

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_for_lag(h, k):
    n = len(h)
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    g[v1] += (1.0 - h[idxs[v1] + k])
    g[v2] -= h[idxs[v2] - k]
    return g

def gradient_multi(h, k_stars, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for ks, w in zip(k_stars, weights):
        k = int(ks) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

        cuts_c = list(warm_cuts[1])
    else:
        cuts_g = []
        cuts_c = []

    no_improve = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)

        if c5_val < best_score - 1e-14:

            no_improve = 0

        if no_improve > 2000:
            if len(cuts_g) > 20:
                slacks = np.array([cuts_c[i] - cuts_g[i] @ h for i in range(len(cuts_g))])
                keep = np.argsort(slacks)[:20]
                cuts_g = [cuts_g[i] for i in keep]
                cuts_c = [cuts_c[i] for i in keep]
            no_improve = 0

        max_corr = float(np.max(corr))

def subgrad_descent(h_init, n, target, time_limit, alpha_init=0.01, max_iters=20000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1

    h_curr = h_init.copy().astype(float)
    h_prev = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it)) / 2.0
        beta = (t_k - 1.0) / t_k
        y = h_curr + beta * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr_y = compute_corr(y)
        k_star = int(np.argmax(corr_y))
        k = k_star - mid
        g = gradient_for_lag(y, k)
        g -= np.mean(g)

        alpha = alpha_init / (1.0 + 0.1 * it)
        h_new = y - alpha * g
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        h_prev = h_curr.copy()
        h_curr = h_new
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_curr.copy()

    return best_h, best_score

        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        # Also try subgrad from perturbed start
        if tleft() > 80:
            rng = np.random.RandomState(seed + 1)
            h_pert = best_h + rng.randn(n0) * 0.03
            h_pert = project_feasible(h_pert, target0)
            tl2 = min(50, tleft() - 40)
            h_sg2, sc_sg2 = subgrad_descent(h_pert, n0, target0, tl2,
                                             alpha_init=0.008, max_iters=30000)
            if sc_sg2 < best_score:
                best_score = sc_sg2
                best_h = h_sg2

    # Phase 4: Random restarts with cutting plane
    rng = np.random.RandomState(seed)
    n_restarts = 6
    if tleft() > n_restarts * 40:
        time_per_restart = min(50, (tleft() - 120) / max(n_restarts, 1))
        if time_per_restart > 20:
            perturbations = [0.02, 0.05, 0.1, 0.15, 0.08, 0.12]
            for r in range(n_restarts):
                if tleft() < time_per_restart + 80:
                    break
                sigma = perturbations[r % len(perturbations)]
                if r % 2 == 0:
                    h_pert = best_h + rng.randn(n0) * sigma
                else:
                    h_pert = best_h.copy()
                    block_sz = n0 // 5
                    s_idx = rng.randint(0, n0 - block_sz)
                    h_pert[s_idx:s_idx + block_sz] = rng.rand(block_sz)
                h_pert = project_feasible(h_pert, target0)

                tl_r = min(time_per_restart, tleft() - 80)
                if tl_r < 20:
                    break
                h_cp_r, sc_r, _, _ = cutting_plane(h_pert, n0, target0, tl_r,
                                                    max_iters=5000, max_cuts=200,
                                                    n_new_cuts=5, tol=1e-11)
                if sc_r < best_score:
                    best_score = sc_r
                    best_h = h_cp_r

    # Phase 5: Pair coordinate descent polish
    if tleft() > 120:
        tl = min(100, tleft() - 50)
        h_cd, sc_cd = pair_cd(best_h, n0, target0, tl,
                               max_sweeps=8000, n_cand_start=15, n_cand_end=45,
                               max_no_improve=200)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 6: Multi-scale refinement (n=800)
    if tleft() > 280:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1_init = compute_c5(h1)

        if tleft() > 220:
            tl1 = min(170, tleft() - 100)
            h1_cp, sc1, cg1, cc1 = cutting_plane(h1, n1, target1, tl1,
                                                   max_iters=10000, max_cuts=250,
                                                   n_new_cuts=6, tol=1e-12)
            if sc1 < best_score:
                best_score = sc1
                best_h = h1_cp
                n0 = n1
                target0 = target1

        if tleft() > 80:
            tl_cd = min(60, tleft() - 40)
            h1_cd, sc1_cd = pair_cd(best_h, n0, target0, tl_cd,
                                     max_sweeps=4000, n_cand_start=10, n_cand_end=30,
                                     max_no_improve=150)
            if sc1_cd < best_score:
                best_score = sc1_cd
                best_h = h1_cd

    # Phase 7: Final cutting plane polish
    if tleft() > 80:
        tl = min(60, tleft() - 30)
        h_fin, sc_fin, _, _ = cutting_plane(best_h, n0, target0, tl,
                                             max_iters=5000, max_cuts=200,
                                             n_new_cuts=8, tol=1e-13)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 8: Final coordinate descent
    if tleft() > 40:
        tl = min(30, tleft() - 20)
        h_last, sc_last = pair_cd(best_h, n0, target0, tl,
                                   max_sweeps=3000, n_cand_start=20, n_cand_end=40,
                                   max_no_improve=100)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
