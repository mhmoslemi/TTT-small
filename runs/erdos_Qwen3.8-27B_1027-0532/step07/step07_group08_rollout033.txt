import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def fast_subgradient(h_init, n, target, time_limit, alpha0=0.5, alpha_min=0.001, 
                     n_active_frac=0.1, patience=200):
    """Fast subgradient method with adaptive step size."""
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    stale = 0
    
    for step in range(500000):
        if time.time() - start > time_limit:
            break
        if alpha < alpha_min:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
            alpha = min(alpha * 1.05, alpha0)
        else:
            stale += 1
            alpha *= 0.97
            if stale > patience:
                break
        
        # Use top active shifts with exponential weighting
        c_max = float(np.max(corr))
        # Select shifts above threshold
        threshold = c_max - 0.1 * max(c_max, 0.01)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        
        # Weight by excess
        excess = corr[active] - (c_max - 0.1 * max(c_max, 0.01))
        excess = np.maximum(excess, 0)
        if excess.sum() < 1e-15:
            weights = np.ones(len(active)) / len(active)
        else:
            weights = excess / excess.sum()
        
        grad = np.zeros(n)
        for s_idx, w in zip(active, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new
    
    return best_h, best_c5

import numpy as np

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

    return project_feasible(h_out, target_dst)

    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 45.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 60:
        h_sg, sc_sg = fast_subgradient(best_h, n0, target0,
                                       time_limit=min(40, tleft() * 0.3),
                                       alpha0=0.25, alpha_min=0.003, patience=120)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if tleft() > 60:
        deadline = time.time() + min(45, tleft() * 0.35)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [100, 500, 2000, 8000],
                                400, 1e-14, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 350:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        current_n = n1
        current_target = target1
        if tleft() > 250:
            h_cd, c5_cd = cd_diverse(h1, n1, target1,
                                     time_limit=min(220, tleft() * 0.55),
                                     max_sweeps=60000,
                                     n_cand_start=15, n_cand_end=80,
                                     max_no_improve=450,
                                     n_active_shifts=3,
                                     reheat_interval=70, reheat_scale=0.05)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    if tleft() > 70:
        deadline = time.time() + min(55, tleft() * 0.35)
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [500, 3000, 15000],
                                800, 1e-14, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_cd2, c5_cd2 = cd_diverse(best_h, current_n, current_target,
                                   time_limit=min(150, tleft() * 0.5),
                                   max_sweeps=45000,
                                   n_cand_start=25, n_cand_end=100,
                                   max_no_improve=400,
                                   n_active_shifts=4,
                                   reheat_interval=60, reheat_scale=0.03)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 50:
        deadline = time.time() + min(40, tleft() * 0.4)
        h_pol2, sc2 = slsqp_round(best_h, current_n, current_target,
                                  [2000, 10000, 40000],
                                  1200, 1e-14, deadline=deadline)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 120:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        current_n = n2
        current_target = target2
        if tleft() > 100:
            h_cd3, c5_cd3 = cd_diverse(h2, n2, target2,
                                       time_limit=min(80, tleft() * 0.5),
                                       max_sweeps=30000,
                                       n_cand_start=12, n_cand_end=50,
                                       max_no_improve=300,
                                       n_active_shifts=3,
                                       reheat_interval=80, reheat_scale=0.04)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd2 = h_cd3

    if tleft() > 40:
        deadline = time.time() + min(30, tleft() * 0.5)
        h_pol3, sc3 = slsqp_round(best_h, current_n, current_target,
                                  [5000, 20000, 60000],
                                  1000, 1e-14, deadline=deadline)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    if tleft() > 80:
        for attempt in range(3):
            if tleft() < 30:
                break
            noise_scale = [0.015, 0.04, 0.08][attempt]
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(25, tleft() * 0.3)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [500, 3000, 12000],
                                        600, 1e-14, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 50:
        h_cd4, c5_cd4 = cd_diverse(best_h, current_n, current_target,
                                   time_limit=min(35, tleft() * 0.5),
                                   max_sweeps=25000,
                                   n_cand_start=30, n_cand_end=90,

    if tleft() > 30:
        deadline = time.time() + min(20, tleft() * 0.6)
        h_final, sc_final = slsqp_round(best_h, current_n, current_target,
                                        [10000, 40000, 80000],
                                        1000, 1e-14, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
