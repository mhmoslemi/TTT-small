This is a great insight. The problem is equivalent to finding h with unit mass (integral=1) and h ∈ [0,1] that maximizes the minimum of its autocorrelation.

Now, for the autocorrelation A_k = sum_i h_i h_{i+k}, note that:
- A_0 = sum_i h_i^2 (this is the largest, since h_i^2 ≤ h_i and sum h_i is fixed)
- As |k| → ∞, A_k → 0 (for well-behaved h)

So we want to make the "non-trivial" autocorrelation values (small |k|) as large as possible.

This suggests that the optimal h should have a "spread out" structure - not concentrated, but also not perfectly periodic.

Now, for the algorithm:

Since we want to maximize min_k A_k, this is a max-min problem. We can reformulate as:
maximize t
subject to A_k ≥ t for all k
sum_i h_i = n/2
0 ≤ h_i ≤ 1

This is a non-convex problem (A_k is quadratic in h), but we can use iterative methods.

**Algorithm: Max-Min Autocorrelation**
1. Start with initial h
2. Compute A_k for all k
3. Find t = min_k A_k
4. Compute gradient of t with respect to h (subgradient from the minimizing k)
5. Take a step in the direction that increases t
6. Project onto feasible set
7. Repeat

This is essentially what the subgradient method in the previous code does, but framed differently.

2. Try to find h such that A_k ≥ t for all k
3. If not possible, decrease t
4. Use projected gradient or Newton's method for the inner problem

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        # Need to add mass - distribute to points with room
        room = 1.0 - h
        total_room = room.sum()
        if total_room < 1e-14:
            return h
        h += room * (diff / total_room)
    else:
        # Need to remove mass - distribute from points with mass
        total_mass = h.sum()
        if total_mass < 1e-14:
            return h
        h += h * (diff / total_mass)
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            h[idx] += residual / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass

    return best_h, best_c5

def autocorr_leveling(h_init, n, target, time_limit, n_active=5, max_steps=50000):
    """Max-min autocorrelation leveling algorithm."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break
        
        # Find top n_active shifts
        top_idx = np.argsort(corr)[-n_active:]
        # Weighted gradient (softer than hard max)
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        
        # Project gradient to satisfy sum constraint
        grad -= np.mean(grad)
        
        # Adaptive step size with backtracking
        decay = 1.0 / (1.0 + 0.01 * step)
        base_lr = 0.1 * decay
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        
        if not improved and step > 5000:
            break

    return best_h, best_c5

def pair_cd_improved(h_init, n, target, time_limit, n_cand=30, max_sweeps=50000, n_shifts=1, use_momentum=False):
    """Improved pair CD with optional momentum."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    momentum = np.zeros(n) if use_momentum else None
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = 50.0
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]
        
        if use_momentum:
            momentum = 0.9 * momentum + 0.1 * grad
            grad_use = momentum
        else:
            grad_use = grad

        sorted_by_grad = np.argsort(-grad_use)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)

        # Precompute profiles for efficiency
        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                # Bisection for optimal delta
                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def block_move_cd(h_init, n, target, time_limit, block_size=5, n_cand=20, max_sweeps=30000):
    """Block-based coordinate descent."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    n_blocks = n // block_size
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        k_star = int(np.argmax(corr)) - mid
        
        # Compute gradient
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        
        # Group into blocks, compute block scores
        block_scores = np.zeros(n_blocks)
        for b in range(n_blocks):
            s = b * block_size
            e = min(s + block_size, n)
            block_scores[b] = grad[s:e].sum()
        
        # Select top and bottom blocks
        top_blocks = np.argsort(-block_scores)[:n_cand]
        bot_blocks = np.argsort(block_scores)[:n_cand]
        
        best_improve = 0.0
        best_move = None
        
        for bi in top_blocks:
            si = bi * block_size
            ei = min(si + block_size, n)
            for bj in bot_blocks:
                sj = bj * block_size
                ej = min(sj + block_size, n)
                if bi == bj:
                    continue
                
                # Try moving mass from block i to block j
                # Total mass to move = average of block i, normalized
                mass_avail = h[si:ei].sum()
                mass_room = block_size - h[sj:ej].sum()
                delta_total = min(mass_avail, mass_room) * 0.5
                if delta_total < 1e-10:
                    continue
                
                # Distribution: proportional removal from block i, proportional addition to block j
                rem = h[si:ei] * (delta_total / (mass_avail + 1e-15))
                add = np.ones(ej - sj) * (delta_total / (ej - sj))
                
                # Apply and check
                h_new = h.copy()
                h_new[si:ei] -= rem
                h_new[sj:ej] += add
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_feasible(h_new, target)
                
                c5_new = compute_c5(h_new)
                improve = c5_val - c5_new
                if improve > best_improve:
                    best_improve = improve
                    best_move = (si, ei, sj, ej, rem.copy(), add.copy())
        
        if best_move is not None and best_improve > 1e-13:
            si, ei, sj, ej, rem, add = best_move
            h[si:ei] -= rem
            h[sj:ej] += add
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        room = 1.0 - h
        total_room = room.sum()
        if total_room < 1e-14:
            return h
        h += room * (diff / total_room)
    else:
        total_mass = h.sum()
        if total_mass < 1e-14:
            return h
        h += h * (diff / total_mass)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-8)[0]
        else:
            idx = np.where(h > 1e-8)[0]
        if len(idx) > 0:
            h[idx] += residual / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    log_sum = np.log(np.sum(e))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)

                h_cur = h_new

                    best_improve = improve
                    best_move = (si, ei, sj, ej, rem.copy(), add.copy())
        if best_move is not None and best_improve > 1e-13:
            si, ei, sj, ej, rem, add = best_move
            h[si:ei] -= rem
            h[sj:ej] += add
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def coarse_to_fine(h_init, n_init, target_init, n_fine, target_fine, time_limit):
    start = time.time()
    h_coarse = h_init.copy()
    target_coarse = target_init
    n_c = len(h_coarse)
    hp, sc = lbfgs_phase(h_coarse, n_c, target_coarse, alphas=[500, 5000],
                         maxiter=200, time_limit=min(30, time_limit * 0.25), penalty=500.0)
    hcd, sc2 = pair_cd_fast(hp, n_c, target_coarse, time_limit=min(20, time_limit * 0.2),
                            n_cand=25, max_sweeps=30000, n_shifts=1)
    if sc2 < sc:
        hp = hcd
        sc = sc2
    h_fine = interpolate_to(hp, n_fine, target_fine)
    hp_fine, sc_fine = lbfgs_phase(h_fine, n_fine, target_fine, alphas=[500, 5000, 50000],
                                    maxiter=300, time_limit=min(40, time_limit * 0.35), penalty=500.0)
    hcd_fine, sc_cd = pair_cd_fast(hp_fine, n_fine, target_fine,
                                    time_limit=min(30, time_limit * 0.25),
                                    n_cand=30, max_sweeps=30000, n_shifts=1)
    if sc_cd < sc_fine:
        hp_fine = hcd_fine
        sc_fine = sc_cd
    return hp_fine, sc_fine

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)
    work_n = n
    work_target = target

    # Phase 1: Upsample to 1200 and quick optimize
    n1 = 1200
    t1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, t1)
    if tleft() > 100:
        hp1, sc1 = lbfgs_phase(h1, n1, t1, alphas=[200, 2000],
                               maxiter=200, time_limit=min(40, tleft() * 0.15), penalty=200.0)
        hcd1, sc1b = pair_cd_fast(hp1, n1, t1, time_limit=min(25, tleft() * 0.1),
                                   n_cand=30, max_sweeps=30000, n_shifts=1)
        if sc1b < sc1:
            hp1 = hcd1
            sc1 = sc1b
        if sc1 < best_score:
            best_score = sc1
            best_h = hp1
            work_n = n1
            work_target = t1

    # Phase 2: L-BFGS at working resolution
    if tleft() > 130:
        hp, sc = lbfgs_phase(best_h, work_n, work_target,
                             alphas=[1000, 5000, 50000],
                             maxiter=400, time_limit=min(80, tleft() * 0.25),
                             penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 3: Autocorrelation leveling
    if tleft() > 130:
        hlev, sc = autocorr_leveling(best_h, work_n, work_target,
                                     time_limit=min(90, tleft() * 0.22),
                                     n_active=3, max_steps=40000)
        if sc < best_score:
            best_score = sc
            best_h = hlev

    # Phase 4: Pair CD main
    if tleft() > 160:
        hcd, sc = pair_cd_fast(best_h, work_n, work_target,
                               time_limit=min(120, tleft() * 0.3),
                               n_cand=40, max_sweeps=80000, n_shifts=1)
        if sc < best_score:
            best_score = sc
            best_h = hcd

    # Phase 5: L-BFGS high alpha
    if tleft() > 100:
        hp, sc = lbfgs_phase(best_h, work_n, work_target,
                             alphas=[50000, 500000, 2000000],
                             maxiter=300, time_limit=min(60, tleft() * 0.2),
                             penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 6: Block CD
    if tleft() > 100:
        hbm, sc = block_cd(best_h, work_n, work_target,
                           time_limit=min(70, tleft() * 0.2),
                           block_size=4, n_cand=15, max_sweeps=15000)
        if sc < best_score:
            best_score = sc
            best_h = hbm

    # Phase 7: Pair CD with multi-shift
    if tleft() > 120:
        hcd, sc = pair_cd_fast(best_h, work_n, work_target,
                               time_limit=min(90, tleft() * 0.22),
                               n_cand=45, max_sweeps=60000, n_shifts=3)
        if sc < best_score:
            best_score = sc
            best_h = hcd

    # Phase 8: Try higher resolution 2000
    if tleft() > 250:
        n2 = 2000
        t2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, t2)
        hp2, sc2 = lbfgs_phase(h2, n2, t2, alphas=[500, 5000],
                               maxiter=200, time_limit=min(50, tleft() * 0.15),
                               penalty=200.0)
        hcd2, sc2b = pair_cd_fast(hp2, n2, t2, time_limit=min(40, tleft() * 0.12),
                                   n_cand=30, max_sweeps=40000, n_shifts=1)
        if sc2b < sc2:
            hp2 = hcd2
            sc2 = sc2b
        if sc2 < best_score:
            best_score = sc2
            best_h = hp2
            work_n = n2
            work_target = t2
        if tleft() > 100:
            hlev2, sc2c = autocorr_leveling(best_h, work_n, work_target,
                                            time_limit=min(60, tleft() * 0.2),
                                            n_active=3, max_steps=30000)
            if sc2c < best_score:
                best_score = sc2c
                best_h = hlev2
            if tleft() > 60:
                hcd3, sc2d = pair_cd_fast(best_h, work_n, work_target,
                                           time_limit=min(50, tleft() * 0.2),
                                           n_cand=30, max_sweeps=40000, n_shifts=1)
                if sc2d < best_score:
                    best_score = sc2d
                    best_h = hcd3

    # Phase 9: Random restarts
    if tleft() > 120:
        rng = np.random.RandomState(seed)
        n_use = len(best_h)
        target_use = n_use / 2.0
        for trial in range(4):
            if tleft() < 30:
                break
            noise_scales = [0.02, 0.04, 0.06, 0.08]
            ns = noise_scales[trial % 4]
            h_pert = best_h + rng.randn(n_use) * ns
            h_pert = project_feasible(h_pert, target_use)
            hp_r, sc_r = lbfgs_phase(h_pert, n_use, target_use,
                                     alphas=[5000, 50000],
                                     maxiter=150, time_limit=min(20, tleft() * 0.1),
                                     penalty=500.0)
            if sc_r < best_score and tleft() > 40:
                hcd_r, sc_r2 = pair_cd_fast(hp_r, n_use, target_use,
                                            time_limit=min(20, tleft() * 0.1),
                                            n_cand=25, max_sweeps=20000, n_shifts=1)
                if sc_r2 < best_score:
                    best_score = sc_r2
                    best_h = hcd_r
                elif sc_r < best_score:
                    best_score = sc_r
                    best_h = hp_r
            elif sc_r < best_score:
                best_score = sc_r
                best_h = hp_r

    # Phase 10: Coarse-to-fine from scratch
    if tleft() > 200:
        n_c2 = 400
        t_c2 = n_c2 / 2.0
        h_c2 = interpolate_to(best_h, n_c2, t_c2)
        n_f2 = 1200
        t_f2 = n_f2 / 2.0
        h_cf, sc_cf = coarse_to_fine(h_c2, n_c2, t_c2, n_f2, t_f2,
                                     time_limit=min(120, tleft() * 0.35))
        if sc_cf < best_score:
            best_score = sc_cf
            best_h = h_cf
            work_n = n_f2
            work_target = t_f2
            if tleft() > 50:
                hcd_cf, sc_cf2 = pair_cd_fast(best_h, work_n, work_target,
                                              time_limit=min(40, tleft() * 0.3),
                                              n_cand=30, max_sweeps=30000, n_shifts=1)
                if sc_cf2 < best_score:
                    best_score = sc_cf2
                    best_h = hcd_cf

    # Phase 11: Final L-BFGS
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_phase(best_h, n_use, target_use,
                             alphas=[500000, 5000000],
                             maxiter=200, time_limit=min(30, tleft() * 0.5),
                             penalty=50000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 12: Final pair CD
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, sc = pair_cd_fast(best_h, n_use, target_use,
                               time_limit=min(20, tleft() * 0.6),
                               n_cand=25, max_sweeps=30000, n_shifts=1)
        if sc < best_score:
            best_score = sc
            best_h = hcd

    # Final
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
