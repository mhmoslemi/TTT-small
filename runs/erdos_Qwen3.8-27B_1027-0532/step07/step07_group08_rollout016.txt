def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15

            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_for_shift(h, k, n):
    """Compute gradient of f_k(h) = sum_i h[i]*(1-h[i+k]) with respect to h."""
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    grad[v1] += (1.0 - h[idxs[v1] - k])
    grad[v2] -= h[idxs[v2] + k]
    return grad

def active_set_gradient(h, n, n_active=5, eps=0.05):
    """Compute weighted gradient using active set of shifts."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    threshold = c_max - eps * max(c_max, 0.01)
    active = np.where(corr >= threshold)[0]

            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
        else:
            stale += 1

        if stale > 300:
            break

        c_max = float(np.max(corr))
        threshold = c_max - eps * max(c_max, 0.01)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        n_act = min(len(active), n_active)
        top = active[np.argsort(corr[active])[-n_act:]]

        w = 1.0 / n_act
        grad = np.zeros(n)
        for s in top:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        h_new = h - alpha * grad
        h_new = project_feasible(h_new, target)
        h = h_new

        alpha = alpha * 0.9995
        if step % 100 == 0:
            alpha = max(alpha, 0.02)
        if stale % 50 == 0:
            alpha = max(alpha * 0.9, 0.01)

    return best_h, best_c5


def pair_cd(h_init, n, target, time_limit, max_sweeps=40000,
            n_cand_start=15, n_cand_end=70, max_no_improve=500,
            n_active_shifts=3, reheat_interval=120, reheat_scale=0.04, seed=42):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0

    def compute_change_matrix(h_local):
        n_corr = 2 * n - 1
        change = np.zeros((n, n_corr))
        for p in range(n):
            pos_mask = (p + shifts >= 0) & (p + shifts < n)
            neg_mask = (p - shifts >= 0) & (p - shifts < n)
            change[p, pos_mask] += (1.0 - h_local[p + shifts[pos_mask]])
            change[p, neg_mask] -= h_local[p - shifts[neg_mask]]
        return change

    def update_change_row(change, p, h_local):
        change[p, :] = 0.0
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h_local[p + shifts[pos_mask]])
        change[p, neg_mask] -= h_local[p - shifts[neg_mask]]
        return change

    change = compute_change_matrix(h)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if sweeps_since_improve > reheat_interval and sweeps_since_improve % 50 == 0:
            n_pert = max(5, n // 35)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 1e-6:
                h = h_pert
                change = compute_change_matrix(h)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                continue

        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            weights = corr[top_k] / corr_top_sum

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star_idx = int(np.argmax(corr))
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                if g[k_star_idx] > 0.3 * np.max(np.abs(g)) + 1e-10:
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h)
            update_change_row(change, best_j, h)

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, block_size=8, max_iters=5000, seed=42):
    """Block coordinate descent: optimize over blocks of h."""
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_blocks = n // block_size
    remainder = n % block_size

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        for b in range(n_blocks):
            s = b * block_size
            e = s + block_size
            if e > n:
                e = n
            if e - s < 3:
                break

            block = h[s:e].copy()
            bounds = [(0.0, 1.0)] * (e - s)

            def obj(x, s=s, e=e):
                h_trial = h.copy()
                h_trial[s:e] = x
                h_trial = project_feasible(h_trial, target)
                return -compute_c5(h_trial)

            try:
                res = minimize(obj, block, method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-10})
                h_new = h.copy()
                h_new[s:e] = res.x
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h_new.copy()
                if sc <= compute_c5(h) + 1e-12:
                    h = h_new
            except Exception:
                pass

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def diverse_start(h_base, n, target, seed, method=0):
    """Generate diverse starting points from a base."""
    rng = np.random.RandomState(seed)
    if method == 0:
        return project_feasible(h_base + rng.randn(n) * 0.05, target)
    elif method == 1:
        return fourier_perturb(h_base.copy(), 0.04, rng, max_freq=max(3, n // 6))
    else:
        n_pert = n // 5
        idx = rng.choice(n, size=n_pert, replace=False)
        h_new = h_base.copy()
        h_new[idx] += rng.randn(n_pert) * 0.08
        return project_feasible(h_new, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: SLSQP polish at n=400
    if tleft() > 100:
        deadline = time.time() + min(60, tleft() * 0.35)
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [80, 300, 1000, 5000],
                                400, 1e-14, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Active gradient at n=400
    if tleft() > 80:
        h_ag, sc_ag = active_grad_descent(best_h, n0, target0,
                                          time_limit=min(40, tleft() * 0.4),
                                          n_active=5, eps=0.04, alpha0=0.12)
        if sc_ag < best_score:
            best_score = sc_ag
            best_h = h_ag

    # Phase 3: Annealing at n=400
    if tleft() > 80:
        h_anc, sc_anc = annealing_phase(best_h, n0, target0,
                                        time_limit=min(40, tleft() * 0.4), seed=seed)
        if sc_anc < best_score:
            best_score = sc_anc
            best_h = h_anc

    # Phase 4: Pair CD at n=400
    if tleft() > 60:
        h_cd, sc_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(35, tleft() * 0.45),
                              max_sweeps=20000, n_cand_start=12, n_cand_end=50,
                              max_no_improve=300, n_active_shifts=3,
                              reheat_interval=100, reheat_scale=0.03, seed=seed)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 5: Interpolate to n=800
    if tleft() > 300:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            current_n = n1
            current_target = target1

        # Annealing at n=800
        if tleft() > 150:
            h_anc1, sc_anc1 = annealing_phase(best_h, current_n, current_target,
                                              time_limit=min(60, tleft() * 0.35), seed=seed + 1)
            if sc_anc1 < best_score:
                best_score = sc_anc1
                best_h = h_anc1

        # Active gradient at n=800
        if tleft() > 100:
            h_ag1, sc_ag1 = active_grad_descent(best_h, current_n, current_target,
                                                time_limit=min(50, tleft() * 0.35),
                                                n_active=6, eps=0.03, alpha0=0.1)
            if sc_ag1 < best_score:
                best_score = sc_ag1
                best_h = h_ag1

        # Pair CD at n=800
        if tleft() > 120:
            h_cd1, sc_cd1 = pair_cd(best_h, current_n, current_target,
                                    time_limit=min(80, tleft() * 0.4),
                                    max_sweeps=30000, n_cand_start=15, n_cand_end=65,
                                    max_no_improve=400, n_active_shifts=3,
                                    reheat_interval=120, reheat_scale=0.035, seed=seed + 2)
            if sc_cd1 < best_score:
                best_score = sc_cd1
                best_h = h_cd1

        # SLSQP polish at n=800
        if tleft() > 70:
            deadline = time.time() + min(45, tleft() * 0.4)
            h_pol1, sc_pol1 = slsqp_round(best_h, current_n, current_target,
                                          [1000, 5000, 20000],
                                          600, 1e-14, deadline=deadline)
            if sc_pol1 < best_score:
                best_score = sc_pol1
                best_h = h_pol1

    # Phase 6: Multi-start at current resolution
    if tleft() > 200:
        for attempt in range(3):
            if tleft() < 80:
                break
            h_start = diverse_start(best_h, current_n, current_target,
                                    seed + 10 + attempt, method=attempt % 3)
            h_try, sc_try = active_grad_descent(h_start, current_n, current_target,
                                                time_limit=min(30, tleft() * 0.3),
                                                n_active=5, eps=0.04, alpha0=0.1)
            if sc_try < best_score:
                best_score = sc_try
                best_h = h_try

    # Phase 7: Interpolate to n=1200
    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            current_n = n2
            current_target = target2

        if tleft() > 150:
            h_anc2, sc_anc2 = annealing_phase(best_h, current_n, current_target,
                                              time_limit=min(50, tleft() * 0.35), seed=seed + 5)
            if sc_anc2 < best_score:
                best_score = sc_anc2
                best_h = h_anc2

        if tleft() > 100:
            h_cd2, sc_cd2 = pair_cd(best_h, current_n, current_target,
                                    time_limit=min(60, tleft() * 0.45),
                                    max_sweeps=20000, n_cand_start=12, n_cand_end=55,
                                    max_no_improve=350, n_active_shifts=3,
                                    reheat_interval=100, reheat_scale=0.03, seed=seed + 6)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_h = h_cd2

        if tleft() > 60:
            deadline = time.time() + min(40, tleft() * 0.4)
            h_pol2, sc_pol2 = slsqp_round(best_h, current_n, current_target,
                                          [2000, 10000, 30000],
                                          500, 1e-14, deadline=deadline)
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_h = h_pol2

    # Phase 8: Final combined polish
    if tleft() > 80:
        h_final_cd, sc_fc = pair_cd(best_h, current_n, current_target,
                                    time_limit=min(40, tleft() * 0.45),
                                    max_sweeps=15000, n_cand_start=20, n_cand_end=70,
                                    max_no_improve=300, n_active_shifts=4,
                                    reheat_interval=80, reheat_scale=0.02, seed=seed + 20)
        if sc_fc < best_score:
            best_score = sc_fc
            best_h = h_final_cd

    # Phase 9: Final SLSQP
    if tleft() > 40:
        deadline = time.time() + min(25, tleft() * 0.5)
        h_final, sc_final = slsqp_round(best_h, current_n, current_target,
                                        [5000, 20000, 80000],
                                        600, 1e-14, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Final projection and evaluation
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
