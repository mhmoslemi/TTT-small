def project_feasible(h, target_sum):

            h = np.clip(h + step, 0.0, 1.0)
            break

        h = np.clip(h, 0.0, 1.0)

def build_change_matrix(h, n):

                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h_try[i:i+block_len], h_try[j:j+block_len] = h_try[j:j+block_len].copy(), h_try[i:i+block_len].copy()
        # Also try partial: just flip the values

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0
_g_power = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _time_left(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def smooth_obj_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad_lse(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-8:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def smooth_obj_power(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.mean(np.abs(corr) ** p) ** (1.0 / p))


def smooth_obj_grad_power(x, p):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    pm = np.mean(np.abs(corr) ** p) ** (1.0 / p)
    if pm < 1e-30:
        return np.zeros(n)
    weights = np.abs(corr) ** (p - 1) * np.sign(corr) * (pm ** (1 - p)) / n_corr

        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        accept = False
        if best_i >= 0:
            accept = True
        elif T > 1e-5 and rng.random() < 0.05:
            i_rand = int(rng.randint(n))
            j_rand = int(rng.randint(n))
            if i_rand != j_rand:
                d_lo = max(h[i_rand] - 1.0, -h[j_rand])
                d_hi = min(h[i_rand], 1.0 - h[j_rand])
                if d_hi - d_lo > 1e-3:
                    d_rand = 0.5 * (d_lo + d_hi)
                    g_r = change[j_rand] - change[i_rand]
                    approx_r = np.max(corr + d_rand * g_r) * dx
                    delta_obj = approx_r - c5_val
                    if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
                        best_i = i_rand
                        best_j = j_rand
                        best_delta = d_rand
                        accept = True

        if accept and best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            s_check = np.sum(h)
            if abs(s_check - target) > 1e-10:
                h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def cd_multishift_score(h_init, n, target, time_limit, max_sweeps=40000,
                        n_cand=60, top_k_eval=70, n_shifts=4,
                        max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_shift_indices = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shift_indices]
        sw_sum = np.sum(shift_weights)
        if sw_sum < 1e-30:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        grad = np.zeros(n)
        for s_idx in top_shift_indices:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += shift_weights[list(top_shift_indices).index(s_idx)] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= shift_weights[list(top_shift_indices).index(s_idx)] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_top = change[high_set][:, top_shift_indices]
        ch_l_top = change[low_set][:, top_shift_indices]

        score_h = ch_h_top @ shift_weights
        score_l = ch_l_top @ shift_weights
        score_mat = score_l[None, :] - score_h[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            s_check = np.sum(h)
            if abs(s_check - target) > 1e-10:
                h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def perturbation_search(h_init, n, target, time_limit, max_steps=30000):
    start = time.time()
    rng = np.random.RandomState(777)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        scale = 0.04 * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.choice([2, 3, 5, 8, 12]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 200:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def block_swap_search(h_init, n, target, time_limit, max_steps=15000):
    start = time.time()
    rng = np.random.RandomState(4242)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 1500:
            break

        block_len = int(rng.choice([2, 3, 5, 8, 12, 16]))
        block_len = min(block_len, n // 4)
        i = int(rng.randint(0, n - block_len))
        j = int(rng.randint(0, n - block_len))
        if abs(i - j) < block_len:
            no_improve += 1
            continue

        h_try = h.copy()
        tmp = h_try[i:i + block_len].copy()
        h_try[i:i + block_len] = h_try[j:j + block_len]
        h_try[j:j + block_len] = tmp
        h_try = project_feasible(h_try, target)
        sc1 = compute_c5(h_try)

        h_try2 = h.copy()
        h_try2[i:i + block_len] = 1.0 - h[i:i + block_len]
        h_try2 = project_feasible(h_try2, target)
        sc2 = compute_c5(h_try2)

        if sc1 < best_c5 - 1e-15:
            best_c5 = sc1
            best_h = h_try.copy()
            h = h_try.copy()
            no_improve = 0
        elif sc2 < best_c5 - 1e-15:
            best_c5 = sc2
            best_h = h_try2.copy()
            h = h_try2.copy()
            no_improve = 0
        else:
            no_improve += 1
            if rng.random() < 0.03:
                scale = 0.03
                n_pert = int(rng.choice([3, 5, 8]))
                pidxs = rng.choice(n, size=n_pert, replace=False)
                h_try3 = h.copy()
                h_try3[pidxs] += rng.randn(n_pert) * scale
                h_try3 = project_feasible(h_try3, target)
                sc3 = compute_c5(h_try3)
                if sc3 < best_c5 - 1e-15:
                    best_c5 = sc3
                    best_h = h_try3
                    h = h_try3.copy()
                    no_improve = 0

    return best_h, best_c5
