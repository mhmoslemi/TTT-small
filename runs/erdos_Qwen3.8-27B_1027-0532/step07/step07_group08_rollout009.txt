c[m] = ∑_n h[n+m] * (1-h)[n] for valid n and n+m.

With m ranging from -(n-1) to n-1 (indexed as 0 to 2n-2 in the output).

So c[m] = ∑_n h[n+m] * (1-h[n]) where 0 ≤ n < n and 0 ≤ n+m < n.

Let me use the indexing: let the output index be `idx` from 0 to 2n-2. Then the shift is k = idx - (n-1).

c[idx] = ∑_n h[n + idx - (n-1)] * (1-h[n]) for valid indices.

Let k = idx - (n-1). Then:
c[n-1+k] = ∑_n h[n+k] * (1-h[n]) for n and n+k both in [0, n-1].

OK so for a given shift k, the correlation is:
C(k) = ∑_{n: n∈[0,n-1], n+k∈[0,n-1]} h[n+k] * (1-h[n])

∂C(k)/∂h[i] = 
- If i = n+k for some valid n (i.e., n = i-k is valid): contributes (1-h[i-k])
- If i = n for some valid n with n+k valid (i.e., i+k is valid): contributes -h[i+k]... 

Wait, let me redo this. C(k) = ∑_n h[n+k] * (1-h[n]).

∂C(k)/∂h[i] = ∂/∂h[i] ∑_n h[n+k] * (1-h[n])

h[i] can appear as:
1. h[n+k] where n+k = i, so n = i-k: contribution is (1-h[i-k]) if 0 ≤ i-k < n and 0 ≤ i < n (both valid)
2. h[n] where n = i: contribution is -h[i+k] if 0 ≤ i < n and 0 ≤ i+k < n (both valid)

So: ∂C(k)/∂h[i] = (1-h[i-k]) * [0≤i-k<n and 0≤i<n] - h[i+k] * [0≤i<n and 0≤i+k<n]

Simplifying (since 0≤i<n is always true for valid i):
∂C(k)/∂h[i] = (1-h[i-k]) * [0≤i-k<n] - h[i+k] * [0≤i+k<n]

This is what I'll use.

Now, for the "targeted smoothing" approach:
- Find k* = argmax_k C(k)
- Compute grad[i] = ∂C(k*)/∂h[i] = (1-h[i-k*]) * [0≤i-k*<n] - h[i+k*] * [0≤i+k*<n]
- Update: h[i] -= α * grad[i] (since we want to decrease C(k*))
- Project to feasible set

To make this more robust, I'll use a weighted combination of the top few shifts.

```python
import numpy as np
from scipy.optimize import minimize
import time

# Global variables for SLSQP callbacks
_g_n = 0
_g_target = 0.0
_g_alpha = 1.0

def compute_c5(h):
    """Compute C5 = max_k correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff/n, 0.0, 1.0)
            break
        step = diff / mask.sum()
        # Clip step to avoid overshooting
        if diff > 0:
            max_step = np.min(1.0 - h[mask])
            step = min(step, max_step)
        else:
            min_step = np.min(h[mask])
            step = max(step, -min_step)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, alpha):
    """Smooth approximation of max(corr) using log-sum-exp"""
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500, 0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alpha_list, maxiter, deadline):
    """Run SLSQP with increasing alpha for smooth max"""
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    
    for alpha in alpha_list:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

def compute_gradient_multi(h, n, top_k=3):
    """Compute gradient as weighted average of top-k shift gradients"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    idxs = np.arange(n)
    
    # Get top-k shifts
    top_indices = np.argsort(corr)[-top_k:]
    weights = corr[top_indices]
    weight_sum = np.sum(weights)
    if weight_sum < 1e-10:
        weights = np.ones(top_k) / top_k
    else:
        weights = weights / weight_sum
    
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        k = int(idx) - mid
        # ∂C(k)/∂h[i] = (1-h[i-k]) * [0≤i-k<n] - h[i+k] * [0≤i+k<n]
        if k >= 0:
            # i-k valid when i >= k, i+k valid when i < n-k
            valid_src = idxs >= k  # i-k >= 0
            valid_dst = idxs < n - k  # i+k < n
            grad[valid_src] += w * (1.0 - h[idxs[valid_src] - k])
            grad[valid_dst] -= w * h[idxs[valid_dst] + k]
        elif k < 0:
            # i-k valid when i < n+k (since i-k < n means i < n+k)
            # i+k valid when i >= -k
            valid_src = idxs < n + k  # i-k < n
            valid_dst = idxs >= -k  # i+k >= 0
            grad[valid_src] += w * (1.0 - h[idxs[valid_src] - k])
            grad[valid_dst] -= w * h[idxs[valid_dst] + k]
    
    return grad, corr

def targeted_smoothing_cd(h_init, n, target, time_limit, n_active=3,
                           alpha0=0.05, alpha_min=0.001, momentum=0.5,
                           max_iter=500000, reheat_every=200, reheat_scale=0.02):
    """CD using targeted smoothing on active shifts"""
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    alpha = alpha0
    stale = 0
    rng = np.random.RandomState(12345)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0/n)
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
            # Slightly increase alpha on improvement
            alpha = min(alpha * 1.02, alpha0)
        else:
            stale += 1
            if stale > reheat_every:
                alpha *= 0.95
                if alpha < alpha_min:
                    # Reheat with perturbation
                    n_pert = max(10, n // 20)
                if alpha < alpha_min * 0.5:
                    # Apply perturbation and reset
                    idx_pert = rng.choice(n, size=n_pert, replace=False)
                    h[idx_pert] += rng.randn(n_pert) * reheat_scale
                    h = project_feasible(h, target)
                    alpha = alpha0 * 0.5
                    vel = np.zeros(n)
                    stale = 0
                    continue
        
        # Compute gradient
        grad, _ = compute_gradient_multi(h, n, top_k=n_active)
        # Center the gradient to preserve sum
        grad -= np.mean(grad)
        
        # Momentum update
        vel = momentum * vel + alpha * grad
        h_new = h - vel  # Descend (we want to decrease C)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    """Interpolate h from its current resolution to n_dst points"""
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    # Linear interpolation
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def pair_cd(h_init, n, target, time_limit, n_cand=40, max_sweeps=20000,
            max_no_improve=300, reheat_interval=100, reheat_scale=0.03):
    """Pair coordinate descent"""
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    rng = np.random.RandomState(67890)
    
    # Precompute change matrix
    change = np.zeros((n, 2*n-1))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        
        if no_improve > reheat_interval and no_improve % 30 == 0:
            n_pert = max(5, n // 30)
            pert = rng.choice(n, size=n_pert, replace=False)
            h[pert] += rng.randn(n_pert) * reheat_scale
            h = project_feasible(h, target)
            # Recompute change
            for p in pert:
                pos = (p + shifts >= 0) & (p + shifts < n)
                neg = (p - shifts >= 0) & (p - shifts < n)
                change[p, :] = 0
                change[p, pos] += (1.0 - h[p + shifts[pos]])
                change[p, neg] -= h[p - shifts[neg]]
            no_improve = reheat_interval + 10  # Give it more room
            continue
        
        # Gradient for move selection
        k_star = int(np.argmax(corr))
        grad = np.zeros(n)
        k = k_star - mid
        if k >= 0:
            valid_src = idxs >= k
            valid_dst = idxs < n - k
            grad[valid_src] += (1.0 - h[idxs[valid_src] - k])
            grad[valid_dst] -= h[idxs[valid_dst] + k]
        else:
            valid_src = idxs < n + k
            valid_dst = idxs >= -k
            grad[valid_src] += (1.0 - h[idxs[valid_src] - k])
            grad[valid_dst] -= h[idxs[valid_dst] + k]
        grad -= np.mean(grad)
        
        # Select candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_trial = c5_val
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                # Binary search for optimal delta
                a_d, b_d = d_lo, d_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid >= c5_val - 1e-15:
                        active_mask = vals >= f_mid - 1e-14
                        f_prime = np.max(g[active_mask]) if active_mask.any() else 0
                    else:
                        f_prime = 0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                change[p, :] = 0
                pos = (p + shifts >= 0) & (p + shifts < n)
                neg = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos] += (1.0 - h[p + shifts[pos]])
                change[p, neg] -= h[p - shifts[neg]]
    
    return best_h, best_c5
```

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    for alpha in alpha_list:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
