```python
def accel_gradient(h_init, n, target, time_limit, alpha, max_iter=50000):
    """Accelerated gradient descent on smooth max with projection."""
    start = time.time()
    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 1.0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Compute gradient of smooth max
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        w = np.exp(shifted) / np.sum(np.exp(shifted))
        
        # Gradient of smooth max = Σ_k w_k * ∂corr_k/∂h
        grad = np.zeros(n)
        mid = n - 1
        idxs = np.arange(n)
        for k_idx in range(len(corr)):
            if w[k_idx] < 1e-10:
                continue
            k = k_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[k_idx] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[k_idx] * h[idxs[v2] + k]
        
        grad -= np.mean(grad)  # Project onto sum-preserving direction
        
        # Nesterov momentum
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        beta = (t - 1) / t_new
        h_y = h + beta * (h - h_prev)
        
        # Step
        lr = 1.0 / (alpha * alpha * n)  # Lipschitz constant bound
        h_new = project_feasible(h_y - lr * grad, target)
        
        prev_c5 = compute_c5(h)
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        h_prev = h
        h = h_new
        t = t_new
        
        # Adaptive learning rate
        if new_c5 < prev_c5 - 1e-15:
            lr *= 1.1
        else:
            lr *= 0.9
    
    return best_h, best_c5
```

Wait, the Nesterov acceleration requires the objective to be smooth with a known Lipschitz gradient constant. The smooth max with parameter alpha has gradient Lipschitz constant related to alpha (the max of weights times the max of ∂²corr/∂h², which is bounded by 1). So the Lipschitz constant is approximately alpha (since the weights sum to 1 and the second derivatives of corr are bounded by 1).

Actually, the Hessian of corr_k with respect to h is a matrix with entries in {-1, 0, 1} (since corr_k is bilinear in h). The spectral norm of this Hessian is bounded by O(n). So the Lipschitz constant of the gradient of the smooth max is bounded by alpha * O(n).

This means the step size should be ~ 1/(alpha * n) which is very small for large alpha and large n. This makes the gradient method very slow for sharp smooth max.

This suggests that SLSQP (which uses second-order information) is more appropriate for the sharp regime, while gradient methods work better for the smooth regime.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-12):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-16:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        fracs = [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.5, 0.8]
        for frac in fracs:
            d = frac * delta_max
            h_trial = h.copy()
            h_trial[i_src] -= d
            h_trial[i_dst] += d
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = project_feasible(h_trial, target)
            trial_max = float(np.max(np.correlate(h_trial, 1.0 - h_trial, mode='full')))
            if trial_max < best_trial:
                best_trial = trial_max
                best_delta = d

        if best_delta > 1e-12 and best_trial < c5_val - 1e-15:
            h[i_src] -= best_delta
            h[i_dst] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def multi_pair_cd(h_init, n, target, time_limit, max_sweeps=30000, n_cand=25):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    max_no_improve = 1500

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-15:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        si = int(np.argmax(corr))
        k = si - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_d = 0.0

        for ii in range(min(n_cand, len(high_set))):
            i = int(high_set[ii])
            for jj in range(min(n_cand, len(low_set))):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-12:
                    continue
                d = 0.05 * delta_max
                h_trial = h.copy()
                h_trial[i] -= d
                h_trial[j] += d
                h_trial = np.clip(h_trial, 0.0, 1.0)
                h_trial = project_feasible(h_trial, target)
                trial_max = float(np.max(np.correlate(h_trial, 1.0 - h_trial, mode='full')))
                if trial_max < best_trial:
                    best_trial = trial_max
                    best_i = i
                    best_j = j
                    best_d = d

        if best_i >= 0 and best_trial < c5_val - 1e-15:
            h[best_i] -= best_d
            h[best_j] += best_d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def gradient_polish(h_init, n, target, time_limit, alpha=5000, max_iter=50000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    lr = 1.0 / (alpha * np.sqrt(n))

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        w = np.exp(shifted)
        w = w / w.sum()

        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            if w[k_idx] < 1e-8:
                continue
            k = k_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[k_idx] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[k_idx] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)

        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improve = 0
            h = h_new
            lr *= 1.05
        else:
            no_improve += 1
            lr *= 0.85
            h = project_feasible(h + 0.5 * lr * grad, target)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 120:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [100, 500, 2000, 8000], 1000, 1e-12)
        if sc < best_score:
            best_score = sc
            best_h = h_pol.copy()

    n1 = 800
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)

    if tleft() > 180:
        h1, sc1 = slsqp_optimize(h1, n1, target1,
                                  [500, 2000, 8000], 1200, 1e-12)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1.copy()

    if tleft() > 120:
        h_cd, c5_cd = fast_focused_cd(best_h, n1, target1,
                                       time_limit=min(100, tleft() * 0.6),
                                       max_sweeps=40000, top_k=2)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd.copy()

    if tleft() > 100:
        h_pol, sc = slsqp_optimize(best_h, n1, target1,
                                    [20000, 50000, 100000], 1500, 1e-13)
        if sc < best_score:
            best_score = sc
            best_h = h_pol.copy()

    if tleft() > 100:
        h_mp, c5_mp = multi_pair_cd(best_h, n1, target1,
                                     time_limit=min(80, tleft() * 0.6),
                                     max_sweeps=20000, n_cand=20)
        if c5_mp < best_score:
            best_score = c5_mp
            best_h = h_mp.copy()

    if tleft() > 80:
        h_gp, c5_gp = gradient_polish(best_h, n1, target1,
                                       time_limit=min(60, tleft() * 0.7),
                                       alpha=5000, max_iter=30000)
        if c5_gp < best_score:
            best_score = c5_gp
            best_h = h_gp.copy()

    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 180:
            h2, sc2 = slsqp_optimize(h2, n2, target2,
                                      [2000, 8000, 30000], 800, 1e-12)
            if sc2 < best_score:
                best_score = sc2
                best_h = h2.copy()
        if tleft() > 80:
            h_cd2, c5_cd2 = fast_focused_cd(best_h, n2, target2,
                                             time_limit=min(60, tleft() * 0.6),
                                             max_sweeps=25000, top_k=2)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2.copy()
        if tleft() > 60:
            h_pol2, sc2b = slsqp_optimize(best_h, n2, target2,
                                           [50000, 150000], 1000, 1e-13)
            if sc2b < best_score:
                best_score = sc2b
                best_h = h_pol2.copy()

    if tleft() > 60:
        h_pol3, sc3 = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                      [100000, 300000], 800, 1e-13)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3.copy()

    if tleft() > 40:
        h_cd3, c5_cd3 = fast_focused_cd(best_h, len(best_h), len(best_h) / 2.0,
                                         time_limit=min(30, tleft() * 0.7),
                                         max_sweeps=15000, top_k=3)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3.copy()

    rng = np.random.RandomState(seed)
    if tleft() > 50:
        h_pert = best_h + rng.randn(len(best_h)) * 0.01
        h_pert = project_feasible(h_pert, len(best_h) / 2.0)
        h_rp, sc_rp = slsqp_optimize(h_pert, len(best_h), len(best_h) / 2.0,
                                      [2000, 10000, 50000], 600, 1e-13)
        if sc_rp < best_score:
            best_score = sc_rp
            best_h = h_rp.copy()

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
