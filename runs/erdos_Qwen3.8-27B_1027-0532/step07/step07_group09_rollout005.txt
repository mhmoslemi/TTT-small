        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = min(top_k, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = w.sum()
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)])

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]
                h_j = h[j]
                delta_max = min(h_i, 1.0 - h_j)
                if delta_max < 1e-12:
                    continue
                g = G[lr] - G[hr]
                a_d = 0.0
                b_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid < best_trial - 1e-14:
                        a_d = mid_d
                    else:
                        b_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial - 1e-14:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_row(change, best_i, h, n, shifts)
            update_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def triple_cd(h_init, n, target, time_limit, max_steps=8000,
              n_cand=25, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    change, shifts = build_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idx = np.argsort(corr)[-3:]
        w = corr[top_idx].copy()
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_move = None

        n_try = min(12, len(high_set))
        n_low_try = min(10, len(low_set))

        for ii in range(n_try):
            k_idx = int(high_set[ii])
            gk = change[k_idx, :]
            for aa in range(n_low_try):
                for bb in range(aa + 1, n_low_try):
                    i_idx = int(low_set[aa])
                    j_idx = int(low_set[bb])
                    g = gk - change[i_idx, :] - change[j_idx, :]
                    delta_max = min(h[i_idx], h[j_idx], (1.0 - h[k_idx]) / 2.0)
                    if delta_max < 1e-12:
                        continue
                    a_d = 0.0
                    b_d = delta_max
                    for _ in range(10):
                        mid_d = 0.5 * (a_d + b_d)
                        if np.max(corr + mid_d * g) < best_trial - 1e-14:
                            a_d = mid_d
                        else:
                            b_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g)
                    if approx_max < best_trial - 1e-14:
                        best_trial = approx_max
                        best_move = (i_idx, j_idx, k_idx, delta_star, 1)

        n_try2 = min(8, len(low_set))
        n_high_try = min(8, len(high_set))
        for ii in range(n_try2):
            k_idx = int(low_set[ii])
            gk = change[k_idx, :]
            for aa in range(n_high_try):
                for bb in range(aa + 1, n_high_try):
                    i_idx = int(high_set[aa])
                    j_idx = int(high_set[bb])
                    g = change[i_idx, :] + change[j_idx, :] - gk
                    delta_max = min((1.0 - h[i_idx]) / 2.0, (1.0 - h[j_idx]) / 2.0, h[k_idx])
                    if delta_max < 1e-12:
                        continue
                    a_d = 0.0
                    b_d = delta_max
                    for _ in range(10):
                        mid_d = 0.5 * (a_d + b_d)
                        if np.max(corr + mid_d * g) < best_trial - 1e-14:
                            a_d = mid_d
                        else:
                            b_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = np.max(corr + delta_star * g)
                    if approx_max < best_trial - 1e-14:
                        best_trial = approx_max
                        best_move = (i_idx, j_idx, k_idx, delta_star, -1)

        if best_move is not None:
            i_idx, j_idx, k_idx, delta_star, sign = best_move
            if sign == 1:
                h[i_idx] -= delta_star
                h[j_idx] -= delta_star
                h[k_idx] += 2.0 * delta_star
            else:
                h[i_idx] += delta_star
                h[j_idx] += delta_star
                h[k_idx] -= 2.0 * delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_row(change, i_idx, h, n, shifts)
            update_row(change, j_idx, h, n, shifts)
            update_row(change, k_idx, h, n, shifts)

    return best_h, best_c5


def momentum_pg(h_init, n, target, time_limit, max_iter=30000,
                lr_start=0.004, mom=0.85):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 6000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_corr = np.max(corr)
        tol = max(1e-8, 0.02 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(len(active), 1)
        grad -= np.mean(grad)

        progress = min(it / max_iter, 1.0)
        lr = lr_start * (1.0 - 0.92 * progress) ** 1.5
        v = mom * v - lr * grad
        h_new = project_feasible(h + v, target)
        new_c5 = compute_c5(h_new)
        if new_c5 <= c5_val:
            h = h_new
        else:
            v *= 0.5
            h_half = project_feasible(h + v, target)
            if compute_c5(h_half) <= c5_val:
                h = h_half
            else:
                v *= 0.5

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 50:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [80, 400, 2000, 10000],
                                   300, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(100, tleft() * 0.5),
                              max_sweeps=20000,
                              n_cand_start=20, n_cand_end=55,
                              max_no_improve=250, top_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [500, 2000, 10000, 50000],
                                   400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 60:
        h_tri, c5_tri = triple_cd(best_h, n0, target0,
                                   time_limit=min(50, tleft() * 0.5),
                                   max_steps=6000, n_cand=25,
                                   max_no_improve=300)
        if c5_tri < best_score:
            best_score = c5_tri
            best_h = h_tri

    if tleft() > 40:
        h_mom, c5_mom = momentum_pg(best_h, n0, target0,
                                     time_limit=min(30, tleft() * 0.5),
                                     max_iter=15000, lr_start=0.003, mom=0.85)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    n1 = 800
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)
    sc1 = compute_c5(h1)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, n1, target1,
                                   [100, 500, 3000],
                                   300, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_cd, c5_cd = pair_cd(best_h, n1, target1,
                              time_limit=min(120, tleft() * 0.55),
                              max_sweeps=15000,
                              n_cand_start=20, n_cand_end=55,
                              max_no_improve=300, top_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol, sc = slsqp_optimize(best_h, n1, target1,
                                   [500, 2000, 10000],
                                   500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 50:
        h_tri, c5_tri = triple_cd(best_h, n1, target1,
                                   time_limit=min(40, tleft() * 0.5),
                                   max_steps=4000, n_cand=30,
                                   max_no_improve=250)
        if c5_tri < best_score:
            best_score = c5_tri
            best_h = h_tri

    if tleft() > 50:
        h_mom, c5_mom = momentum_pg(best_h, n1, target1,
                                     time_limit=min(35, tleft() * 0.5),
                                     max_iter=15000, lr_start=0.003, mom=0.85)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    n2 = 1200
    target2 = n2 / 2.0
    h2 = interpolate_to(best_h, n2, target2)
    sc2 = compute_c5(h2)
    if sc2 < best_score:
        best_score = sc2
        best_h = h2
        n0 = n2
        target0 = target2
    else:
        h2 = interpolate_to(best_h, n1, target1)
        n0 = n1
        target0 = target1
        best_h = h2

    if tleft() > 100:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(80, tleft() * 0.55),
                              max_sweeps=12000,
                              n_cand_start=18, n_cand_end=50,
                              max_no_improve=250, top_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [200, 1000, 5000],
                                   400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 40:
        h_mom, c5_mom = momentum_pg(best_h, n0, target0,
                                     time_limit=min(30, tleft() * 0.5),
                                     max_iter=10000, lr_start=0.002, mom=0.85)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.015, 0.04]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc = slsqp_optimize(h_pert, n0, target0,
                                       [200, 1000, 5000],
                                       300, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 50:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(35, tleft() * 0.6),
                              max_sweeps=8000,
                              n_cand_start=25, n_cand_end=60,
                              max_no_improve=200, top_k=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 40:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [5000, 20000],
                                   400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
