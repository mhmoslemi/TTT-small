def project_simplex(h, target):

And ∇corr_k[j] = (1-h[j+k]) - h[j-k] (with appropriate boundary conditions)

This is what I had before. Good.

But wait, there's a subtlety. The constraint is sum(h) = target, so the effective gradient for the constrained problem is:
∇f(h) - lambda * 1 (where lambda is chosen so that the projected step maintains the constraint)

In the projected gradient method, we handle this by projecting onto the feasible set after each step. The gradient doesn't need to be modified.

However, for the line search, we should evaluate the objective (C₅) at the projected point, not the smooth objective. This ensures we're actually minimizing C₅, not just the smooth approximation.

One more optimization: for the line search, instead of computing the full correlation from scratch, I can use:
- If the step is small, corr_new ≈ corr_old + step * grad_corr
- Where grad_corr[k] = change[j] - change[i] for the specific pair (in CD)
- But for gradient descent, the step direction is the full gradient, so I need to compute the new h and then the new corr.

For efficiency, let me just compute the full correlation. It's fast enough with numpy.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_simplex(h, target):
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        mask = (proj > 1e-14) & (proj < 1.0 - 1e-14)
        if mask.any():
            proj[mask] += diff / mask.sum()
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj


def get_smooth_data(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    weights /= w_sum
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if v1.any():
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
        if v2.any():
            grad[v2] -= w * h[idxs[v2] + k]
    return corr, grad


def pg_optimize(h_init, target, time_limit, alphas, lr_init=0.005, max_no_improve=200):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n = len(h)
    dx = 2.0 / n
    
    for alpha in alphas:
        no_improve = 0
        lr = lr_init
        while no_improve < max_no_improve:
            if time.time() - start > time_limit:
                return best_h, best_c5
            corr, grad = get_smooth_data(h, alpha)
            grad -= np.mean(grad)
            c5_now = float(np.max(corr)) * dx
            
            step = lr
            improved = False
            for _ in range(20):
                h_try = project_simplex(h - step * grad, target)
                corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                c5_try = float(np.max(corr_try)) * dx
                if c5_try < c5_now - 1e-16:
                    h = h_try
                    improved = True
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    break
                step *= 0.5
            if not improved:
                lr = max(step, 1e-12)
                no_improve += 1
                if lr < 1e-12:
                    break
    return best_h, best_c5


def compute_change(h, n):
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        if pos.any():
            change[p, pos] += (1.0 - h[p + shifts[pos]])
        if neg.any():
            change[p, neg] -= h[p - shifts[neg]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    if pos.any():
        change[p, pos] += (1.0 - h[p + shifts[pos]])
    if neg.any():
        change[p, neg] -= h[p - shifts[neg]]


def cd_optimize(h_init, target, time_limit, n_cand=40, max_no_improve=200):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    no_improve = 0
    change = compute_change(h, n)
    
    while no_improve < max_no_improve:
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d, b_d = d_lo, d_hi
                for _ in range(12):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        fp = np.max(g[active])
                    else:
                        fp = 0.0
                    if fp > 1e-13:
                        b_d = m_d
                    else:
                        a_d = m_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx = np.max(corr + delta_star * g) * dx
                if approx < best_trial - 1e-16:
                    best_trial = approx
                    best_i, best_j, best_delta = i, j, delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            np.clip(h, 0.0, 1.0, out=h)
            h = project_simplex(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_simplex(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_simplex(h_out, target_dst)
