import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def fft_toeplitz_matvec(w_nonneg, h):
    n = len(h)
    if n == 1:
        return w_nonneg[0] * h
    kernel = np.zeros(2 * n - 1)
    kernel[n - 1] = w_nonneg[0]
    kernel[:n - 1] = w_nonneg[1:n][::-1]
    kernel[n:] = w_nonneg[1:n]
    size = 1
    while size < 3 * n - 2:
        size *= 2
    H = np.fft.rfft(h, size)
    K = np.fft.rfft(kernel, size)
    conv = np.fft.irfft(H * K, size)
    return conv[n - 1:2 * n - 1]

def mw_projected_gradient(h_init, n, target, time_limit, eta=1.0, inner_steps=30, lr_init=0.08, max_outer=500000):
    start = time.time()
    h = h_init.copy().astype(float)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1

    w_full = np.ones(n_corr) / n_corr
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    no_improve = 0
    patience = 20000

    for outer in range(max_outer):
        if time.time() - start > time_limit:
            break

        w_sym = np.zeros(n)
        for k in range(n):
            wk = w_full[mid + k]
            wk_neg = w_full[mid - k] if mid - k >= 0 else 0.0
            w_sym[k] = 0.5 * (wk + wk_neg)

        for _ in range(inner_steps):
            bh = fft_toeplitz_matvec(w_sym, h)
            grad = 2.0 * bh - 1.0
            h_new = h + lr * grad
            h_new = project_feasible(h_new, target)
            h = h_new

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.005, 0.2)
        else:
            no_improve += 1
            if no_improve > patience:
                h = best_h + 0.05 * np.random.randn(n)
                h = project_feasible(h, target)
                w_full = np.ones(n_corr) / n_corr
                no_improve = 0
                lr = lr_init
                if time.time() - start > time_limit * 0.9:
                    break

        C = corr * dx
        scaled = eta * (C - np.mean(C))
        scaled = np.clip(scaled, -30.0, 30.0)
        w_full = w_full * np.exp(scaled)
        w_full = w_full / np.sum(w_full)

    return best_h, best_c5

def binary_round(h, target):
    n = len(h)
    n_ones = int(round(target))
    sorted_idx = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[n_ones:]] = 1.0
    return h_bin

def binary_local_search(h_init, n, target, time_limit, max_sweeps=50000):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1

    h = binary_round(h_init, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    corr = np.correlate(h, 1.0 - h, mode='full')
    shifts = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if sweep % 50 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_val = float(np.max(corr)) * dx
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h.copy()

        active_thresh = np.max(corr) * dx - 0.01
        active = np.where(corr * dx >= active_thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        n_ones = int(round(target))
        improved = False

        for i in range(n):
            if time.time() - start > time_limit:
                break
            if h[i] > 0.5:
                delta = -1.0
            else:
                delta = 1.0

            if h[i] > 0.5 and n_ones == 0:
                continue
            if h[i] <= 0.5 and n_ones == n:
                continue

            delta_c = np.zeros(n_corr)
            for k in active:
                ki = int(k) - mid
                ik = i + ki
                ik_minus = i - ki
                val = 0.0
                if 0 <= ik < n:
                    val += (1.0 - h[ik])
                if 0 <= ik_minus < n:
                    val -= h[ik_minus]
                delta_c[k] = delta * val

            new_max = np.max(corr[active] + delta_c[active]) * dx
            if new_max < best_c5 - 1e-14:
                h[i] += delta
                h[i] = np.clip(h[i], 0.0, 1.0)
                if h[i] > 0.5:
                    n_ones += 1
                else:
                    n_ones -= 1
                corr[active] += delta_c[active]
                best_c5 = new_max
                best_h = h.copy()
                improved = True

        if sweep % 50 == 0 and not improved and sweep > 100:
            break

    return best_h, best_c5

def active_set_gd(h_init, n, target, time_limit, n_active=8, lr_init=0.02):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    no_improve = 0

    for step in range(200000):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.8
                if lr < 1e-7:
                    break

        top_idx = np.argsort(corr)[-n_active:]
        c_max = corr[top_idx[0]]
        weights = np.exp(-40.0 * (c_max - corr[top_idx]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_idx):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val - 1e-15:
            h = h_try
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.9

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_smooth(h_start, n, target, alphas, maxiter=800):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    A_eq = np.ones((1, n))
    b_eq = np.array([target])

    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full') * dx
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

        def obj_grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full') * dx
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
            weights = np.exp(shifted)
            weights /= np.sum(weights)
            n_local = len(x)
            mid_local = n_local - 1
            idxs_local = np.arange(n_local)
            grad = np.zeros(n_local)
            for k_idx in range(len(corr)):
                if weights[k_idx] < 1e-12:
                    continue
                k = k_idx - mid_local
                wp = idxs_local + k < n_local
                wm = idxs_local - k >= 0
                grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
                grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
            return grad

        try:
            res = minimize(obj, h_cur, jac=obj_grad, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, cur_score
