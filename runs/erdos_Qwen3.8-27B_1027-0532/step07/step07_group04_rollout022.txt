def compute_change_matrix(h, n):

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n_ones):
    n = len(h)
    idx = np.argsort(-h)
    result = np.zeros(n)
    result[idx[:n_ones]] = 1.0
    return result

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        for m_val in range(-mid, mid + 1):
            t = m_val + mid
            a = p + m_val
            if 0 <= a < n:
                change[p, t] += (1.0 - h[a])
            b = p - m_val
            if 0 <= b < n:
                change[p, t] -= h[b]
    return change

def update_change_after_move(change, x, delta, n):
    mid = n - 1
    n_corr = 2 * n - 1
    for p in range(n):
        t1 = x + mid - p
        t2 = mid + p - x
        if 0 <= t1 < n_corr:
            change[p, t1] -= delta
        if 0 <= t2 < n_corr:
            change[p, t2] -= delta

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

    h_bin = to_binary(h_init, int(target0))
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    # Phase 1: Binary LS at initial resolution with ILS
    if tleft() > 120:
        alloc = min(180, tleft() * 0.30)
        h_b1, s_b1 = fast_binary_ls(h_bin, n0, alloc, n_cand=80, max_no_improve=80, max_its=3)
        if s_b1 < best_score:
            best_score = s_b1
            best_h = h_b1

    # Phase 2: Multi-gradient binary LS
    if tleft() > 80:
        alloc = min(80, tleft() * 0.25)
        h_m1, s_m1 = multi_grad_binary_ls(best_h, n0, alloc, n_cand=60, n_shifts=4, max_no_improve=60)
        if s_m1 < best_score:
            best_score = s_m1
            best_h = h_m1

    # Phase 3: SA for diversity
    if tleft() > 100:
        alloc = min(100, tleft() * 0.25)
        h_sa, s_sa = sa_binary(best_h, n0, alloc, t_start=0.06, t_end=0.001, n_cand=50, rng_seed=seed)
        if s_sa < best_score:
            best_score = s_sa
            best_h = h_sa

    # Phase 4: SLSQP refinement
    if tleft() > 60:
        h_s1, s_s1 = slsqp_refine(best_h, n0, target0, [1000, 5000], 400,
                                   min(60, tleft() * 0.4), start_time)
        if s_s1 < best_score:
            best_score = s_s1
            best_h = h_s1

    # Phase 5: Go to n=600
    if tleft() > 200:
        n1 = 600
        target1 = n1 / 2.0
        h_interp = interpolate_to(best_h, n1, target1)
        h_bin1 = to_binary(h_interp, int(target1))

        alloc = min(180, tleft() * 0.35)
        h_b2, s_b2 = fast_binary_ls(h_bin1, n1, alloc, n_cand=100, max_no_improve=60, max_its=3)
        if s_b2 < best_score:
            best_score = s_b2
            best_h = h_b2

        if tleft() > 60:
            h_m2, s_m2 = multi_grad_binary_ls(best_h, n1, min(60, tleft() * 0.25),
                                               n_cand=80, n_shifts=4, max_no_improve=50)
            if s_m2 < best_score:
                best_score = s_m2
                best_h = h_m2

        if tleft() > 60:
            h_s2, s_s2 = slsqp_refine(best_h, n1, target1, [3000, 10000], 400,
                                       min(60, tleft() * 0.4), start_time)
            if s_s2 < best_score:
                best_score = s_s2
                best_h = h_s2

    # Phase 6: Go to n=800
    if tleft() > 250:
        n2 = 800
        target2 = n2 / 2.0
        h_interp2 = interpolate_to(best_h, n2, target2)
        h_bin2 = to_binary(h_interp2, int(target2))

        alloc = min(180, tleft() * 0.35)
        h_b3, s_b3 = fast_binary_ls(h_bin2, n2, alloc, n_cand=90, max_no_improve=50, max_its=3)
        if s_b3 < best_score:
            best_score = s_b3
            best_h = h_b3

        if tleft() > 80:
            h_m3, s_m3 = multi_grad_binary_ls(best_h, n2, min(60, tleft() * 0.25),
                                               n_cand=70, n_shifts=4, max_no_improve=50)
            if s_m3 < best_score:
                best_score = s_m3
                best_h = h_m3

        if tleft() > 60:
            h_s3, s_s3 = slsqp_refine(best_h, n2, target2, [5000, 20000], 400,
                                       min(60, tleft() * 0.4), start_time)
            if s_s3 < best_score:
                best_score = s_s3
                best_h = h_s3

    # Phase 7: Final binary LS on best
    if tleft() > 80:
        n_cur = len(best_h)
        h_bin7 = to_binary(best_h, int(n_cur / 2.0))
        alloc = min(80, tleft() * 0.35)
        h_b4, s_b4 = fast_binary_ls(h_bin7, n_cur, alloc, n_cand=80, max_no_improve=40, max_its=2)
        if s_b4 < best_score:
            best_score = s_b4
            best_h = h_b4

    # Phase 8: Final SLSQP
    if tleft() > 40:
        h_fp, s_fp = slsqp_refine(best_h, len(best_h), len(best_h) / 2.0,
                                   [10000, 50000], 300,
                                   min(35, tleft() * 0.6), start_time)
        if s_fp < best_score:
            best_score = s_fp
            best_h = h_fp

    # Phase 9: Very final binary LS
    if tleft() > 30:
        n_cur = len(best_h)
        h_bin9 = to_binary(best_h, int(n_cur / 2.0))
        h_b5, s_b5 = fast_binary_ls(h_bin9, n_cur, min(25, tleft() * 0.6),
                                     n_cand=60, max_no_improve=30, max_its=1)
        if s_b5 < best_score:
            best_score = s_b5
            best_h = h_b5

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
