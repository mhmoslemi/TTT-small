    return float(np.max(corr)) * dx

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):

    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)

            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)

            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

                if i == j:
                    continue

                d_hi = min(h[i], 1.0 - h[j])

                    continue
                
                g = change[j] - change[i]
                a_d, b_d = d_lo, d_hi

                    vals = corr + m * g
                    fm = np.max(vals)
                    am = vals >= fm - 1e-14
                    fp = np.max(g[am]) if am.any() else 0.0
                    if fp > 1e-12:
                        b_d = m
                    else:
                        a_d = m
                
                delta = 0.5 * (a_d + b_d)
                if abs(delta) < 1e-12:
                    continue
                
                approx = np.max(corr + delta * g) * dx
                accept = False
                if approx < best_trial - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    if np.random.random() < np.exp(-(approx - c5_val) * 500.0 / temp):
                        accept = True
                
                if accept:
                    best_trial = approx
                    best_i, best_j, best_delta = i, j, delta
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                change[p, :] = 0.0

                nm = (p - shifts >= 0) & (p - shifts < n)
                change[p, pm] += (1.0 - h[p + shifts[pm]])
                change[p, nm] -= h[p - shifts[nm]]
            temp *= 0.999
        else:
            temp = max(temp * 0.95, 1e-10)

def frank_wolfe(h_init, n, target, time_limit, max_iter=50000):

    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute subgradient (use top-3 shifts)
        top_k = np.argsort(corr)[-3:]
        w = corr[top_k].copy()
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(3) / 3.0
        else:
            w /= w_sum
        
        grad = np.zeros(n)
        for si in range(3):
            k = int(top_k[si]) - mid
            wi = w[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # FW step: find s = argmin grad^T s subject to 0<=s<=1, sum(s)=n/2
        sorted_grad = np.sort(grad)
        # s_i = 0 if grad_i > tau, s_i = 1 if grad_i < tau
        # Need sum(s) = n/2
        # If we set s_i = 1 for the n/2 smallest gradient entries:
        n_half = n // 2
        s = np.zeros(n)
        s_idx = np.argsort(grad)
        s[s_idx[:n_half]] = 1.0
        
        # Line search
        gamma = 2.0 / (it + 2)
        h_new = (1.0 - gamma) * h + gamma * s
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < c5_val - 1e-15:
            h = h_new
        else:
            # Try smaller step

            h_new2 = (1.0 - gamma2) * h + gamma2 * s
            h_new2 = np.clip(h_new2, 0.0, 1.0)
            h_new2 = project_feasible(h_new2, target)

                sc3 = compute_c5(h_new3)
                if sc3 < c5_val - 1e-15:
                    h = h_new3
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            grad[v2] -= w * h[idxs[v2] + k]

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)

    _g_n = n
    _g_target = target

    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:

                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def _build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]
    return change, shifts


def _update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pm = (p + shifts >= 0) & (p + shifts < n)
    nm = (p - shifts >= 0) & (p - shifts < n)
    change[p, pm] += (1.0 - h[p + shifts[pm]])
    change[p, nm] -= h[p - shifts[nm]]
    return change


def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand=50, n_shifts=3, max_no_improve=400,
            n_bisect=8, anneal_temp=0.0, anneal_decay=0.999,
            rebuild_every=None):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    if rebuild_every is None:
        rebuild_every = max(n // 4, 50)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    temp = anneal_temp

    change, shifts = _build_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        sw = corr[top_shifts].copy()
        sw_sum = sw.sum()
        if sw_sum < 1e-15:
            sw = np.ones(n_shifts) / n_shifts
        else:
            sw /= sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = sw[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = d_lo
                b_d = d_hi
                for _ in range(n_bisect):
                    m = 0.5 * (a_d + b_d)
                    vals = corr + m * g
                    fm = np.max(vals)
                    am = vals >= fm - 1e-14
                    fp = np.max(g[am]) if am.any() else 0.0
                    if fp > 1e-12:
                        b_d = m
                    else:
                        a_d = m

                delta = 0.5 * (a_d + b_d)
                if abs(delta) < 1e-12:
                    continue

                approx = np.max(corr + delta * g) * dx
                accept = False
                if approx < best_trial - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx - c5_val) * 1000.0
                    if delta_obj < 500.0 and np.random.random() < np.exp(-delta_obj / temp):
                        accept = True

                if accept:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = delta

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            _update_change_row(change, best_i, h, n, shifts)
            _update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > rebuild_every:
                change, shifts = _build_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.95, 1e-10)

    return best_h, best_c5


def subgradient_multi(h_init, n, target, time_limit, n_steps=30000,
                      n_weights=5, step_scale=0.004, softmax_temp=100.0):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        p = np.exp(corr / max(softmax_temp, 1.0))
        p /= p.sum()

        grad = np.zeros(n)
        top_idx = np.argsort(p)[-n_weights:]
        for s_idx in top_idx:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < c5_val:
            h = h_try
        else:
            h_try2 = h - 0.5 * eta * grad
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < c5_val:
                h = h_try2
            else:
                eta *= 0.92

        if step % 1000 == 0 and step > 0:
            eta = max(step_scale * max(0.2, 1.0 - 0.5 * step / n_steps), 1e-6)

    return best_h, best_c5


def frank_wolfe(h_init, n, target, time_limit, max_iter=50000, n_top=3):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 800:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_k = np.argsort(corr)[-n_top:]
        w = corr[top_k].copy()
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.ones(n_top) / n_top
        else:
            w /= w_sum

        grad = np.zeros(n)
        for si in range(n_top):
            k = int(top_k[si]) - mid
            wi = w[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        n_half = n // 2
        s_idx = np.argsort(grad)
        s = np.zeros(n)
        s[s_idx[:n_half]] = 1.0

        gamma = 2.0 / (it + 2) * 0.3
        gamma = min(gamma, 0.15)
        h_new = (1.0 - gamma) * h + gamma * s
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        sc = compute_c5(h_new)
        if sc < c5_val - 1e-15:
            h = h_new
        else:
            gamma2 = gamma * 0.4
            h_new2 = (1.0 - gamma2) * h + gamma2 * s
            h_new2 = np.clip(h_new2, 0.0, 1.0)
            h_new2 = project_feasible(h_new2, target)
            sc2 = compute_c5(h_new2)
            if sc2 < c5_val - 1e-15:
                h = h_new2

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_refine(h_init, n, target, time_limit, rng, n_starts=4,
                       scales=None):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    if scales is None:
        scales = [0.03, 0.06, 0.1, 0.15]
    for r in range(n_starts):
        if time.time() - start > time_limit:
            break
        scale = scales[r % len(scales)]
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pert.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP at original resolution
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [30, 100, 500, 2000],
                                400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Pair CD at original resolution
    if tleft() > 150:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(100, tleft() * 0.5),
                              max_sweeps=40000, n_cand=35, n_shifts=3,
                              max_no_improve=400, n_bisect=8,
                              anneal_temp=0.02, anneal_decay=0.9995)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Interpolate to n=800
    n1 = 800
    target1 = n1 / 2.0
    cur_n = n0
    cur_target = target0
    if tleft() > 150:
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        cur_n = n1
        cur_target = target1
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 4: Intensive pair CD at n=800 (main phase)
    if tleft() > 250:
        h_cd1, c5_cd1 = pair_cd(best_h, cur_n, cur_target,
                                time_limit=min(280, tleft() * 0.6),
                                max_sweeps=80000, n_cand=45, n_shifts=3,
                                max_no_improve=500, n_bisect=7,
                                anneal_temp=0.0, anneal_decay=1.0)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1

    # Phase 5: SLSQP at n=800
    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [500, 2000, 8000],
                                600, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Subgradient with softmax
    if tleft() > 80:
        h_sg, c5_sg = subgradient_multi(best_h, cur_n, cur_target,
                                        time_limit=min(60, tleft() * 0.5),
                                        n_steps=25000, n_weights=5,
                                        step_scale=0.005, softmax_temp=80.0)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 7: Pair CD with wider candidate set
    if tleft() > 120:
        h_cd2, c5_cd2 = pair_cd(best_h, cur_n, cur_target,
                                time_limit=min(100, tleft() * 0.6),
                                max_sweeps=40000, n_cand=70, n_shifts=4,
                                max_no_improve=300, n_bisect=6,
                                anneal_temp=0.005, anneal_decay=0.999)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 8: Frank-Wolfe for diversity
    if tleft() > 60:
        h_fw, c5_fw = frank_wolfe(best_h, cur_n, cur_target,
                                  time_limit=min(50, tleft() * 0.5),
                                  max_iter=30000, n_top=3)
        if c5_fw < best_score:
            best_score = c5_fw
            best_h = h_fw

    # Phase 9: SLSQP polish
    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, cur_n, cur_target,
                                [3000, 10000],
                                500, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 10: Higher resolution attempt
    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2_init = compute_c5(h2)
        if tleft() > 180:
            h_cd3, c5_cd3 = pair_cd(h2, n2, target2,
                                    time_limit=min(180, tleft() * 0.65),
                                    max_sweeps=40000, n_cand=30, n_shifts=3,
                                    max_no_improve=400, n_bisect=6,
                                    anneal_temp=0.0, anneal_decay=1.0)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
            elif sc2_init < best_score:
                best_score = sc2_init
                best_h = h2

    # Phase 11: Stochastic restarts with SLSQP
    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.08]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, len(best_h) / 2.0)
            h_ref, sc = slsqp_round(h_pert, len(best_h), len(best_h) / 2.0,
                                    [100, 500, 2000],
                                    400, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 12: Final pair CD
    if tleft() > 50:
        h_final_cd, c5_final = pair_cd(best_h, len(best_h), len(best_h) / 2.0,
                                       time_limit=min(35, tleft() * 0.7),
                                       max_sweeps=20000, n_cand=80, n_shifts=5,
                                       max_no_improve=200, n_bisect=6)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Phase 13: Final SLSQP
    if tleft() > 30:
        h_fin, sc = slsqp_round(best_h, len(best_h), len(best_h) / 2.0,
                                [5000, 15000],
                                400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
