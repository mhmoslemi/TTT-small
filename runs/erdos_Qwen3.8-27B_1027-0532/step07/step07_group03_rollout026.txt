def binary_local_search(h_init, time_limit, max_sweeps=200, verbose=False):

    dx = 2.0 / n
    n_shifts = 2 * n - 1
    
    # Start from binary
    h = (h_init > 0.5).astype(float)
    # Fix sum to exactly n/2
    # If sum > n/2, change some 1s to 0s (starting from "worse" positions)
    # If sum < n/2, change some 0s to 1s
    target = n // 2
    s = int(np.sum(h))
    if s > target:
        # Need to flip (s - target) ones to zeros
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(ones)
        h[ones[:s-target]] = 0
    elif s < target:
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(zeros)
        h[zeros[:target-s]] = 1
    
    corr = compute_corr(h)
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    
    start = time.time()
    no_improve = 0

            break

            if no_improve > 5:

4. We add a new cut at h_new

The LP solution might be "optimistic" (think the objective is lower than it really is), but we always verify with the true objective. The method converges because we're always moving to points with (hopefully) lower true objective.

The issue is convergence speed. If the LP is too optimistic, we might take many iterations before finding a good point.

**Improvement: Use quadratic cuts instead of linear cuts.**
Instead of g^T h ≤ t + ..., use:
f_k(h_i) + ∇f_k(h_i)^T (h - h_i) - (1/2)(h-h_i)^T B_k (h-h_i) ≤ t
where B_k is some positive semi-definite matrix (like the negative of the Hessian, clipped to be PSD).

But this makes the subproblem a QP instead of LP, which is slower per iteration but might converge faster.

Actually, for our problem, the Hessian of f_k is:
H_k[j,l] = -δ(j+k, l) - δ(j-k, l) (for valid indices)

This is NOT negative semi-definite (it's actually the negative of an adjacency matrix, which has both positive and negative eigenvalues). So we can't simply use -H_k as a convex curvature.

One option: use B_k = L_k * I where L_k is the Lipschitz constant of ∇f_k. This gives a valid lower bound:
f_k(h) ≥ f_k(h_i) + ∇f_k(h_i)^T (h - h_i) - (L_k/2)||h - h_i||²

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_C_matrix(h, n):
    n_shifts = 2 * n - 1
    mid = n - 1
    C = np.zeros((n, n_shifts))
    for k_idx in range(n_shifts):
        k = k_idx - mid
        if k >= 0:
            p1 = np.arange(k, n)
            C[p1, k_idx] += h[p1 - k]
            if k > 0:
                p2 = np.arange(n - k)
                C[p2, k_idx] += h[p2 + k]
        else:
            k_neg = -k
            p1 = np.arange(n - k_neg)
            C[p1, k_idx] += h[p1 + k_neg]
            if k_neg > 0:
                p2 = np.arange(k_neg, n)
                C[p2, k_idx] += h[p2 - k_neg]
    return C


def update_C_after_swap(C, i, j, n):
    n_shifts = 2 * n - 1
    mid = n - 1
    for k_idx in range(n_shifts):
        k = k_idx - mid
        if 0 <= i + k < n:
            C[i + k, k_idx] -= 1.0
        if 0 <= i - k < n:
            C[i - k, k_idx] -= 1.0
        if 0 <= j + k < n:
            C[j + k, k_idx] += 1.0
        if 0 <= j - k < n:
            C[j - k, k_idx] += 1.0


def binary_local_search(h_init, time_limit, max_sweeps=500, rng_seed=42):
    n = len(h_init)
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    target = n // 2

    h = (h_init > 0.5).astype(float)
    s = int(np.sum(h))
    rng = np.random.RandomState(rng_seed)
    if s > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:s - target]] = 0.0
    elif s < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:target - s]] = 1.0

    corr = compute_corr(h)

    no_improve = 0
    max_no_improve = 50

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break

        best_max = float(np.max(corr))
        best_i = -1
        best_j = -1

        for i in ones:
            d = corr + C[i]
            vals = d[None, :] - C[zeros]
            max_vals = np.max(vals, axis=1)
            min_idx = int(np.argmin(max_vals))
            if max_vals[min_idx] < best_max - 1e-10:
                best_max = max_vals[min_idx]
                best_i = int(i)
                best_j = int(zeros[min_idx])

        if best_i < 0:
            no_improve += 1
            if no_improve > max_no_improve:
                break
            continue

        old_ci = C[best_i].copy()
        old_cj = C[best_j].copy()

        h[best_i] = 0.0
        h[best_j] = 1.0
        corr = corr + old_ci - old_cj
        corr = np.maximum(corr, 0.0)

        update_C_after_swap(C, best_i, best_j, n)

        new_c5 = float(np.max(corr)) * dx
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

def interpolate_to(h_src, n_dst, target_dst):

            best_h = h_cp800

    if tleft() > 120:
        h_cd800, sc_cd800 = interceptive_cd(best_h, n1, target1,
                                             time_limit=min(120, tleft() * 0.5),
                                             max_sweeps=15000,
                                             n_cand_start=20, n_cand_end=70,
                                             max_no_improve=250)
        if sc_cd800 < best_score:
            best_score = sc_cd800
            best_h = h_cd800

    # Phase 9: Multi-restart with perturbations
    if tleft() > 150:
        rng = np.random.RandomState(seed + 100)
        perturb_scales = [0.01, 0.03, 0.05]
        time_per_perturb = min(40, tleft() / (len(perturb_scales) + 1))
        for pscale in perturb_scales:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(len(best_h)) * pscale
            h_pert = project_feasible(h_pert, len(best_h) / 2.0)
            h_ref, sc_ref = cutting_plane_method(h_pert, len(best_h), len(best_h) / 2.0,
                                                 time_limit=min(40, tleft() * 0.3),
                                                 max_iters=3000, tol=1e-9)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 10: Binary local search on best so far
    if tleft() > 100:
        h_bin2, sc_bin2 = binary_local_search(best_h, time_limit=min(100, tleft() * 0.5),
                                              max_sweeps=200, rng_seed=seed + 200)
        if sc_bin2 < best_score:
            best_score = sc_bin2
            best_h = h_bin2

    # Phase 11: Final SLSQP polish
    if tleft() > 50:
        h_final_sls, sc_final = slsqp_round(best_h, len(best_h), len(best_h) / 2.0,
                                            [10000, 50000, 100000], 500, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_sls

    # Phase 12: Final interceptive CD
    if tleft() > 30:
        h_cd_final, sc_cd_final = interceptive_cd(best_h, len(best_h), len(best_h) / 2.0,
                                                  time_limit=min(25, tleft() * 0.6),
                                                  max_sweeps=10000,
                                                  n_cand_start=30, n_cand_end=80,
                                                  max_no_improve=150)
        if sc_cd_final < best_score:
            best_score = sc_cd_final
            best_h = h_cd_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
