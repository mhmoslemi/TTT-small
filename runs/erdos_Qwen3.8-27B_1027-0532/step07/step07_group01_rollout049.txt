def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:

                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def subgradient_cd(h_init, n, target, time_limit, max_iters=5000):
    """Subgradient method with momentum."""
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    momentum = np.zeros(n)
    beta = 0.9
    
    # Adaptive step size
    step_scale = 0.05
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Subgradient: use weighted average over top shifts
        top_k = 5
        top_idxs = np.argsort(corr)[-top_k:]
        top_sum = np.sum(corr[top_idxs])
        if top_sum < 1e-15:
            top_sum = 1.0
        
        grad = np.zeros(n)
        for s_idx in top_idxs:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Adaptive step size based on gradient magnitude
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-10:
            break
        alpha = step_scale / (1.0 + 0.01 * it)
        
        # Momentum update
        momentum = beta * momentum - alpha * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, target)
        
        # Check if we improved
        sc = compute_c5(h_new)
        if sc < c5_val + 0.001:  # Allow small increases
            h = h_new
        else:
            momentum *= 0.5  # Dampen momentum if we got worse
            h = h_new if sc < best_c5 + 0.005 else h
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _ls_fun(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_ls_fun, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(screen_iter):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-13
                    f_prime = np.max(g[active_mask]) if active_mask.any() else 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_screen = 0.5 * (a_d + b_d)
                if abs(delta_screen) > 1e-12:
                    est_c5 = float(np.max(corr + delta_screen * g) * dx)
                    candidates.append((est_c5, i, j, delta_lo, delta_hi, g))

        if not candidates:
            continue

        candidates.sort(key=lambda x: x[0])

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for c5_est, i, j, d_lo, d_hi, g in candidates[:top_k]:
            if c5_est >= best_trial_c5:
                break
            a_d = d_lo
            b_d = d_hi
            for _ in range(full_iter):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask]) if active_mask.any() else 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = float(np.max(corr + delta_star * g) * dx)
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, block_size=3, max_sweeps=10000,
             n_cand=30, full_iter=14):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(99)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 300:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        rng.shuffle(low_set)
        rng.shuffle(high_set)

        best_trial_c5 = c5_val
        best_move = None

        for trial in range(8):
            S = high_set[trial * block_size:(trial + 1) * block_size]
            T = low_set[trial * block_size:(trial + 1) * block_size]
            S = S[:block_size]
            T = T[:block_size]
            if len(S) < block_size or len(T) < block_size:
                continue

            g_block = np.zeros(2 * n - 1)
            for p in S:
                g_block -= change[p]
            for p in T:
                g_block += change[p]

            d_max = min(min(h[S]), min(1.0 - h[T]))
            d_min = max(max(h[S] - 1.0), max(-h[T]))
            if d_max - d_min < 1e-13:
                continue

            a_d = d_min
            b_d = d_max
            for _ in range(full_iter):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g_block
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g_block[active_mask]) if active_mask.any() else 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = float(np.max(corr + delta_star * g_block) * dx)
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_move = (S, T, delta_star)

        if best_move is not None:
            S, T, delta_star = best_move
            for p in S:
                h[p] -= delta_star
            for p in T:
                h[p] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in S:
                update_change_row(change, p, h, n, shifts)
            for p in T:
                update_change_row(change, p, h, n, shifts)

    return best_h, best_c5


def subgradient_refine(h_init, n, target, time_limit, max_iters=3000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    momentum = np.zeros(n)
    beta = 0.85
    step_base = 0.03

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        top_k = 7
        top_idxs = np.argsort(corr)[-top_k:]
        top_sum = np.sum(corr[top_idxs])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_idxs:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break

        alpha = step_base / (1.0 + 0.005 * it)
        momentum = beta * momentum - alpha * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, target)

        sc = compute_c5(h_new)
        if sc < c5_val + 0.002:
            h = h_new
        else:
            momentum *= 0.3

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def remaining():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: Initial SLSQP
    if remaining() > 70:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [50, 200, 1000, 5000, 20000],
                                500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Screened pair CD
    if remaining() > 200:
        h_cd, c5_cd = pair_cd_screened(best_h, current_n, current_target,
                                        time_limit=min(200, remaining() * 0.55),
                                        max_sweeps=50000,
                                        n_cand_start=25, n_cand_end=80,
                                        max_no_improve=500,
                                        screen_iter=4, full_iter=16, top_k=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Block CD
    if remaining() > 80:
        h_blk, c5_blk = block_cd(best_h, current_n, current_target,
                                  time_limit=min(70, remaining() * 0.5),
                                  block_size=3, max_sweeps=8000, n_cand=30)
        if c5_blk < best_score:
            best_score = c5_blk
            best_h = h_blk

    # Phase 4: Subgradient
    if remaining() > 60:
        h_sub, sc_sub = subgradient_refine(best_h, current_n, current_target,
                                            time_limit=min(50, remaining() * 0.5))
        if sc_sub < best_score:
            best_score = sc_sub
            best_h = h_sub

    # Phase 5: SLSQP refinement
    if remaining() > 60:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [5000, 20000, 80000],
                                800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Interpolate to n=500
    if remaining() > 350:
        n1 = 500
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            current_n = n1
            current_target = target1

        if remaining() > 220:
            h_cd1, c5_cd1 = pair_cd_screened(h1, n1, target1,
                                              time_limit=min(180, remaining() * 0.5),
                                              max_sweeps=30000,
                                              n_cand_start=20, n_cand_end=65,
                                              max_no_improve=400,
                                              screen_iter=4, full_iter=16, top_k=12)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                current_n = n1
                current_target = target1

        if current_n == n1 and remaining() > 70:
            h_pol1, sc1b = slsqp_round(best_h, n1, target1,
                                       [5000, 20000, 80000],
                                       600, 1e-15)
            if sc1b < best_score:
                best_score = sc1b
                best_h = h_pol1

    # Phase 7: Random restarts
    if remaining() > 200:
        rng = np.random.RandomState(seed)
        for attempt in range(5):
            if remaining() < 70:
                break
            noise_scale = 0.02 + 0.03 * attempt
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [200, 2000, 10000],
                                        400, 1e-15)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            if remaining() > 100:
                h_cd_p, c5_p = pair_cd_screened(h_ref, current_n, current_target,
                                                 time_limit=min(50, remaining() * 0.3),
                                                 max_sweeps=8000,
                                                 n_cand_start=25, n_cand_end=60,
                                                 max_no_improve=300,
                                                 screen_iter=3, full_iter=14, top_k=10)
                if c5_p < best_score:
                    best_score = c5_p
                    best_h = h_cd_p

    # Phase 8: Final pair CD
    if remaining() > 80:
        h_final_cd, c5_final = pair_cd_screened(best_h, current_n, current_target,
                                                 time_limit=min(70, remaining() * 0.75),
                                                 max_sweeps=20000,
                                                 n_cand_start=30, n_cand_end=90,
                                                 max_no_improve=400,
                                                 screen_iter=4, full_iter=18, top_k=20)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Phase 9: Final SLSQP
    if remaining() > 40:
        h_final_pol, sc_final = slsqp_round(best_h, current_n, current_target,
                                             [20000, 100000, 300000],
                                             700, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
