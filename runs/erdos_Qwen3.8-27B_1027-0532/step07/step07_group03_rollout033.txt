```python
import numpy as np
from scipy.optimize import minimize, linprog, approx_fprime
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def lprop_correlation(h, n, target, alpha, smooth_penalty=0.0):
    """L-BFGS-B optimization with smooth max objective + optional smoothing penalty"""
    dx = 2.0 / n
    
    def objective(x):
        # Handle the equality constraint by penalizing it
        sum_penalty = (np.sum(x) - target) ** 2 * 1e6
        corr = np.correlate(x, 1.0 - x, mode='full')
        sm = smooth_max_value(corr, alpha)
        # Optional smoothing penalty
        if smooth_penalty > 0:
            diffs = np.diff(x)
            sm += smooth_penalty * np.sum(diffs**2)
        return sm * dx + sum_penalty
    
    def grad(x):
        # Numerical gradient would be too slow for n=400
        # Use analytical gradient for the smooth max
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        p = np.exp(shifted)
        p = p / p.sum()
        
        # Gradient of corr[k] w.r.t. h[i]
        mid = n - 1
        idxs = np.arange(n)
        g = np.zeros(n)
        for m in range(2*n-1):
            if p[m] < 1e-15:
                continue
            k = m - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            g[v1] += p[m] * (1.0 - x[idxs[v1] + k])
            g[v2] -= p[m] * x[idxs[v2] - k]
        
        if smooth_penalty > 0:
            # Gradient of sum((x[i+1]-x[i])^2) w.r.t. x
            g_smooth = np.zeros(n)
            g_smooth[1:] += 2 * (x[1:] - x[:-1])
            g_smooth[:-1] -= 2 * (x[1:] - x[:-1])
            g += smooth_penalty * g_smooth
        
        # Constraint gradient
        g += 2e6 * (np.sum(x) - target)
        
        return g * dx
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(objective, h, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10})
    h_new = project_feasible(res.x, target)
    return h_new, compute_c5(h_new)
```

```python
def block_cd(h_init, n, target, time_limit, block_size=10, max_sweeps=1000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    n_blocks = n // block_size + (1 if n % block_size > 0 else 0)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        for b in range(n_blocks):
            if time.time() - start > time_limit:
                break
            i_start = b * block_size
            i_end = min((b+1) * block_size, n)
            block = h[i_start:i_end].copy()
            block_size_actual = i_end - i_start
            
            # Optimize this block
            def block_obj(x_b):
                h_trial = h.copy()
                h_trial[i_start:i_end] = x_b
                # Fix sum constraint: adjust the rest
                needed = target - np.sum(x_b)
                rest = h_trial[:i_start] + h_trial[i_end:]  # this doesn't work directly
                # Actually, let's just penalize the sum

                corr = np.correlate(h_trial, 1.0 - h_trial, mode='full')
                return smooth_max_value(corr, 200) * dx + sum_pen
            
            # Use L-BFGS-B on this small subproblem

            res = minimize(block_obj, block, method='L-BFGS-B',
                          bounds=bounds_b, options={'maxiter': 100, 'ftol': 1e-12})
            h[i_start:i_end] = np.clip(res.x, 0.0, 1.0)
        
        # Re-project
        h = project_feasible(h, target)
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    p = np.exp(shifted)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros_like(corr)
        p[np.argmax(corr)] = 1.0
    else:
        p = p / s
    return p

def corr_gradient(h, weights):
    """Compute gradient of weighted sum of correlations w.r.t. h.
    weights has length 2n-1, corresponding to corr indices 0..2n-2.
    """
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    g = np.zeros(n)
    for m in range(2*n-1):
        if weights[m] < 1e-15:
            continue
        k = m - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        g[v1] += weights[m] * (1.0 - h[idxs[v1] + k])
        g[v2] -= weights[m] * h[idxs[v2] - k]
    return g

def lbfgsb_optimize(h_init, n, target, alpha, maxiter=500, smooth_reg=0.0):
    """Optimize using L-BFGS-B with smooth max objective."""
    dx = 2.0 / n
    
    def obj_grad(x):
        x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x, 1.0 - x, mode='full')
        p = smooth_max_weights(corr, alpha)
        val = smooth_max_val(corr, alpha) * dx
        
        g = corr_gradient(x, p)
        
        if smooth_reg > 0:
            gs = np.zeros(n)
            if n > 1:
                diff = np.diff(x)
                gs[1:] += 2.0 * diff
                gs[:-1] -= 2.0 * diff
            g += smooth_reg * gs
            val += smooth_reg * np.sum((np.diff(x))**2)
        
        # Soft constraint for sum
        sum_err = np.sum(x) - target
        val += 1e4 * sum_err**2
        g += 2e4 * sum_err
        
        return val, g * dx
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h_init, jac=True, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                               'gtol': 1e-10, 'iprint': -1, 'maxls': 50})
        h_new = project_feasible(res.x, target)
    except Exception:
        h_new = h_init.copy()
    return h_new, compute_c5(h_new)

def gradient_descent_with_line_search(h_init, n, target, time_limit, 
                                       alpha=5000, lr_init=0.001, lr_decay=0.98,
                                       max_iters=20000, momentum=0.9):
    """Faster-than-linear gradient descent with momentum and backtracking line search."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_score - 1e-15:
            best_score = c5_val
            best_h = h.copy()
        
        p = smooth_max_weights(corr, alpha)
        g = corr_gradient(h, p)
        
        # Project gradient onto feasible direction (zero sum)
        g = g - np.mean(g)
        
        # Momentum update
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        # Check if this is an improvement
        new_score = compute_c5(h_new)
        if new_score < c5_val + 1e-14:
            h = h_new
        else:
            # Backtrack
            v = 0.5 * v
            h_new = h + v
            h_new = project_feasible(h_new, target)
            h = h_new
        
        lr *= lr_decay
    
    return best_h, best_score

def smart_cd(h_init, n, target, time_limit, max_sweeps=50000,
             n_cand=50, max_no_improve=200):
    """Coordinate descent with smart pair selection."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Find active lags
        c_max = np.max(corr)
        active_threshold = c_max - 0.01 * c_max  # within 1% of max
        active_mask = corr >= active_threshold
        n_active = int(active_mask.sum())
        
        # Gradient based on active set
        if n_active == 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            active_lags = np.where(active_mask)[0]
            w = np.ones(len(active_lags)) / len(active_lags)
            grad = np.zeros(n)
            for k_idx, w_val in zip(active_lags, w):
                k = int(k_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w_val * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w_val * h[idxs[v2] + k]
        
        grad -= np.mean(grad)
        
        # Select candidate pairs
        # high_set: indices where we want to decrease h (high grad)
        # low_set: indices where we want to increase h (low grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                
                g_vec = change[j] - change[i]
                # Binary search for optimal delta
                a_d, b_d = delta_lo, delta_hi
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                
                approx_max = float(np.max(corr + delta_star * g_vec)) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    p = np.exp(shifted)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros_like(corr)
        p[np.argmax(corr)] = 1.0
    else:
        p = p / s
    return p


def smooth_max_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def corr_gradient(h, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    g = np.zeros(n)
    for m in range(2 * n - 1):

        k = m - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        g[v1] += weights[m] * (1.0 - h[idxs[v1] + k])
        g[v2] -= weights[m] * h[idxs[v2] - k]
    return g


def lbfgsb_round(h_init, n, target, alphas, maxiter_each=400):
    dx = 2.0 / n
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        def obj_grad(x):
            x = np.clip(x, 0.0, 1.0)
            corr = np.correlate(x, 1.0 - x, mode='full')
            p = smooth_max_weights(corr, alpha)
            val = smooth_max_val(corr, alpha) * dx
            g = corr_gradient(x, p)
            sum_err = np.sum(x) - target
            val += 1e4 * sum_err * sum_err
            g += 2e4 * sum_err
            return val, g * dx

        bounds = [(0.0, 1.0)] * n
        try:

                                    'gtol': 1e-11, 'iprint': -1, 'maxls': 40})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new

def momentum_grad_descent(h_init, n, target, time_limit,
                          alpha=5000, lr_init=0.0005, lr_decay=0.995,
                          momentum=0.85, max_iters=30000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-15:
            best_score = c5_val
            best_h = h.copy()

        p = smooth_max_weights(corr, alpha)
        g = corr_gradient(h, p)
        g = g - np.mean(g)

        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, target)

        new_score = compute_c5(h_new)
        if new_score < c5_val + 1e-14:
            h = h_new
        else:
            v = 0.5 * v
            h_new = h + v
            h_new = project_feasible(h_new, target)
            h = h_new

        lr *= lr_decay

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
