    mid = n - 1

        # Affected correlation indices

        m_hi = min(2*n - 2, i + n - 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def ccd_optimize(h_init, time_limit, max_sweeps=5000):
    h = h_init.copy().astype(float)
    n = len(h)
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    start = time.time()
    best_c5 = float(np.max(corr)) * (2.0 / n)
    best_h = h.copy()
    
    # Precompute affected ranges
    aff_lo = np.zeros(n, dtype=int)
    aff_hi = np.zeros(n, dtype=int)
    for i in range(n):
        aff_lo[i] = max(0, min(mid - i, i))
        aff_hi[i] = min(2 * mid, max(2 * mid - i, i + mid))
    
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 100:
            break
        
        total_change = 0.0
        for i in range(n):
            old_val = h[i]
            m_lo = aff_lo[i]
            m_hi = aff_hi[i]
            m_range = np.arange(m_lo, m_hi + 1)
            k_vals = m_range - mid
            
            valid1 = (i + k_vals >= 0) & (i + k_vals < n)
            valid2 = (i - k_vals >= 0) & (i - k_vals < n)
            
            a = np.zeros(len(m_range))
            if np.any(valid1):
                a[valid1] += (1.0 - h[i + k_vals[valid1]])
            if np.any(valid2):
                a[valid2] -= h[i - k_vals[valid2]]
            
            rest = corr[m_range] - old_val * a
            
            # Bisection
            lo_x, hi_x = 0.0, 1.0
            if hi_x - lo_x > 1e-15:
                for _ in range(35):
                    mid_x = 0.5 * (lo_x + hi_x)
                    vals = a * mid_x + rest
                    f_max = np.max(vals)
                    active = vals >= f_max - 1e-15
                    min_s = np.min(a[active])
                    max_s = np.max(a[active])
                    if min_s >= -1e-14:
                        hi_x = mid_x
                    elif max_s <= 1e-14:
                        lo_x = mid_x
                    else:
                        lo_x = hi_x = mid_x
                        break
            
            new_val = 0.5 * (lo_x + hi_x)
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                total_change += abs(delta)
                h[i] = new_val
                corr[m_range] += delta * a
        
        # Project
        h = project_feasible(h, n / 2.0)
        # Recompute corr after projection (since projection changed values)
        if abs(np.sum(h) - n / 2.0) > 1e-10:
            corr = np.correlate(h, 1.0 - h, mode='full')
        
        c5_val = float(np.max(corr)) * (2.0 / n)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if total_change < 1e-12:
            no_improve += 50
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def ccd_optimize(h_init, time_limit, max_sweeps=8000, project_every=5):
    h = h_init.copy().astype(float)
    n = len(h)
    target = n / 2.0
    mid = n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    start = time.time()
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()

    aff_lo = np.empty(n, dtype=np.int64)
    aff_hi = np.empty(n, dtype=np.int64)
    for i in range(n):
        aff_lo[i] = max(0, min(mid - i, i))
        aff_hi[i] = min(2 * mid, max(2 * mid - i, i + mid))

    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 80:
            break

        for i in range(n):
            old_val = h[i]
            m_lo = aff_lo[i]
            m_hi = aff_hi[i]
            m_range = np.arange(m_lo, m_hi + 1)
            k_vals = m_range - mid

            valid1 = (i + k_vals >= 0) & (i + k_vals < n)
            valid2 = (i - k_vals >= 0) & (i - k_vals < n)

            a = np.zeros(len(m_range))
            if np.any(valid1):
                a[valid1] += (1.0 - h[i + k_vals[valid1]])
            if np.any(valid2):
                a[valid2] -= h[i - k_vals[valid2]]

            rest = corr[m_range] - old_val * a

            lo_x, hi_x = 0.0, 1.0
            for _ in range(38):
                mid_x = 0.5 * (lo_x + hi_x)
                vals = a * mid_x + rest
                f_max = np.max(vals)
                active = vals >= f_max - 1e-15
                min_s = np.min(a[active])
                max_s = np.max(a[active])
                if min_s >= -1e-14:
                    hi_x = mid_x
                elif max_s <= 1e-14:
                    lo_x = mid_x
                else:
                    lo_x = hi_x = mid_x
                    break

            new_val = 0.5 * (lo_x + hi_x)
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                h[i] = new_val
                corr[m_range] += delta * a

        if (sweep + 1) % project_every == 0:
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd_advanced(h_init, n, target, time_limit, max_sweeps=30000,
                     n_cand_start=25, n_cand_end=100, n_shifts=3,
                     max_no_improve=400, anneal_temp=0.0, anneal_decay=0.999):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    temp = anneal_temp

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        top_shifts = np.argsort(corr)[-n_shifts:]
        shift_weights = corr[top_shifts]
        sw_sum = np.sum(shift_weights)
        if sw_sum < 1e-15:
            sw_sum = 1.0
        shift_weights = shift_weights / sw_sum

        grad = np.zeros(n)
        for si in range(n_shifts):
            s_idx = int(top_shifts[si])
            k = s_idx - mid
            w = shift_weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                accept = False
                if approx_max < best_trial_c5 - 1e-14:
                    accept = True
                elif temp > 1e-10:
                    delta_obj = (approx_max - c5_val) * 1000.0
                    if np.random.random() < np.exp(-delta_obj / temp):
                        accept = True

                if accept and approx_max < best_trial_c5 + 1e-10:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 2:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.95, 1e-10)

    return best_h, best_c5


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def block_cd(h_init, n, target, time_limit, block_size=4, max_sweeps=2000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        n_blocks = n // block_size
        block_grads = np.zeros(n_blocks)
        for b in range(n_blocks):
            block_grads[b] = np.mean(grad[b * block_size:(b + 1) * block_size])

        sorted_blocks = np.argsort(block_grads)
        low_blocks = sorted_blocks[:6]
        high_blocks = sorted_blocks[-6:]

        best_trial = c5_val
        best_move = None

        for hb in high_blocks:
            for lb in low_blocks:
                if hb == lb:
                    continue
                hi_start = hb * block_size
                hi_end = (hb + 1) * block_size
                lo_start = lb * block_size
                lo_end = (lb + 1) * block_size

                max_move = min(np.sum(h[hi_start:hi_end]),
                               np.sum(1.0 - h[lo_start:lo_end]))
                if max_move < 1e-10:
                    continue

                for frac in [0.3, 0.5, 0.7, 1.0]:
                    delta = frac * max_move / block_size
                    h_try = h.copy()
                    h_try[hi_start:hi_end] -= delta
                    h_try[lo_start:lo_end] += delta
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_trial - 1e-14:
                        best_trial = sc
                        best_move = (h_try.copy(), sc)

        if best_move is not None:
            h = best_move[0]

    return best_h, best_c5
