For a pair (i,j) with change δ:
- corr_k(δ) = corr_k(0) + δ * g_k where g_k = ∂corr_k/∂h at the change
- We want to find δ that minimizes max_k corr_k(δ)

The optimal δ is where the maximum is achieved by at least two k values simultaneously (a "saddle" point), or at a boundary.

We can find this by:
1. For each pair of active k values (k1, k2), find the δ where corr_k1(δ) = corr_k2(δ)
2. Check if at that δ, no other correlation exceeds the common value
3. Pick the best such δ

This is more efficient than bisection when the number of active lags is small.

def optimal_delta(corr, g, h_i, h_j):
    """Find optimal delta for moving mass from i to j.
    corr: current correlation values
    g: change in correlation per unit delta
    h_i: current value at i (will decrease)
    h_j: current value at j (will increase)
    """
    delta_max = min(h_i, 1.0 - h_j)
    if delta_max < 1e-15:
        return 0.0
    
    # Find candidates: boundaries and tie points
    candidates = [0.0, delta_max]
    
    # Top correlations
    n_corr = len(corr)
    top_k = np.argsort(corr)[-10:]  # top 10 by current value
    
    # For each pair of top correlations, find where they tie
    for a in range(len(top_k)):
        for b in range(a+1, len(top_k)):
            ka, kb = top_k[a], top_k[b]
            da, db = g[ka], g[kb]
            if abs(da - db) > 1e-15:
                # corr[ka] + delta*da = corr[kb] + delta*db
                # delta = (corr[kb] - corr[ka]) / (da - db)
                delta_tie = (corr[kb] - corr[ka]) / (da - db)
                if 0 < delta_tie < delta_max:
                    candidates.append(delta_tie)
    
    # Evaluate at each candidate
    best_delta = 0.0
    best_val = np.max(corr)
    
    for d in candidates:
        new_corr = corr + d * g
        val = np.max(new_corr)
        if val < best_val:
            best_val = val
            best_delta = d
    
    # Also check: for each top k, the delta where it equals the second-highest
    # (this is already covered by the tie points)
    
    return best_delta

                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-14:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    h_test = h.copy()
                    h_test[i] -= delta_star
                    h_test[j] += delta_star
                    h_test = project_feasible(h_test, target)
                    exact_c5 = compute_c5(h_test)
                    if exact_c5 < best_trial:
                        best_trial = exact_c5
                        best_i = i
                        best_j = j
                        best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = project_feasible(h, target)
            change[best_i] = compute_change_row(h, best_i, n)
            change[best_j] = compute_change_row(h, best_j, n)
    
    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    dx = 2.0 / n
    
    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
