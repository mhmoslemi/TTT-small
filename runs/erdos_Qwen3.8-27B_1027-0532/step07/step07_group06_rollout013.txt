def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max < 1e-15:
        return 0.0
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def lse_grad_vec(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if c_max < 1e-15:
        return np.zeros(n)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = weights[s + n - 1]
        if w < 1e-10:
            continue
        m1 = (idxs >= s) & (idxs < s + n)
        m2 = (idxs >= -s) & (idxs < n - s)
        grad[m1] += w * (1.0 - h[idxs[m1] - s])
        grad[m2] -= w * h[idxs[m2] + s]
    return grad

def pair_cd(h_init, n, target, time_limit, start_time,
            max_sweeps=50000, n_cand_start=10, n_cand_end=60,
            max_no_improve=500, n_top_shifts=1, bisection_iters=18,
            step_frac=1.0, momentum_beta=0.0, perturb_every=0):
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    corr = np.correlate(h, 1.0 - h, mode='full')

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            raw_w = corr[top_shifts] - float(corr[top_shifts].min()) + 1.0
            weights = raw_w / raw_w.sum()
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        if momentum_beta > 0.0 and sweep > 0:
            if not hasattr(pair_cd, '_mom_grad'):
                pair_cd._mom_grad = grad.copy()
            else:
                pair_cd._mom_grad = momentum_beta * pair_cd._mom_grad + (1.0 - momentum_beta) * grad
            grad_use = pair_cd._mom_grad
        else:
            grad_use = grad

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)
        current_ncand = max(current_ncand, 4)

        sorted_idx = np.argsort(grad_use)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = delta_lo
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * (h[j] - h[i]) * np.zeros(len(corr))
                    g_vec = np.zeros(len(corr))
                    for kk in range(len(corr)):
                        lag = kk - mid
                        if 0 <= i - lag < n:
                            g_vec[kk] += (1.0 - h[i - lag])
                        if 0 <= j - lag < n:
                            g_vec[kk] -= (1.0 - h[j - lag])
                        if 0 <= i + lag < n:
                            g_vec[kk] -= h[i + lag]
                        if 0 <= j + lag < n:
                            g_vec[kk] += h[j + lag]
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g_vec[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star * step_frac

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5


def pair_cd_fast(h_init, n, target, time_limit, start_time,
                 max_sweeps=50000, n_cand_start=10, n_cand_end=55,
                 max_no_improve=400, n_top_shifts=1, bisection_iters=18,
                 step_frac=1.0, momentum_alpha=0.0):
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    corr = np.correlate(h, 1.0 - h, mode='full')
    mom_grad = np.zeros(n)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_k = np.argsort(corr)[-n_top_shifts:]
            raw_w = corr[top_k] - float(corr[top_k].min()) + 1.0
            weights = raw_w / raw_w.sum()
            grad = np.zeros(n)
            for s_idx, w in zip(top_k, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        if momentum_alpha > 0.0:
            mom_grad = momentum_alpha * mom_grad + (1.0 - momentum_alpha) * grad
            grad_use = mom_grad
        else:
            grad_use = grad

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)
        current_ncand = max(current_ncand, 4)

        sorted_idx = np.argsort(grad_use)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_gvec = None

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = np.zeros(n_corr)
                for lag in range(-(n-1), n):
                    val = 0.0
                    if 0 <= i - lag < n:
                        val += (1.0 - h[i - lag])
                    if 0 <= j - lag < n:
                        val -= (1.0 - h[j - lag])
                    if 0 <= i + lag < n:
                        val -= h[i + lag]
                    if 0 <= j + lag < n:
                        val += h[j + lag]
                    g_vec[lag + mid] = val

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g_vec[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star * step_frac
                    best_gvec = g_vec

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lse_obj, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def adam_refine(h_init, n, target, time_limit, start_time,
                lr=0.005, alpha=200, beta1=0.9, beta2=0.999, eps=1e-8,
                max_steps=5000):
    h = h_init.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    t = 0
    no_improve = 0

    while time.time() - start_time < time_limit:
        if t >= max_steps:
            break
        if no_improve > max_steps // 2:
            break
        t += 1

        grad = lse_grad_vec(h, alpha)
        grad -= np.mean(grad)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr_t = lr / (1.0 + 0.01 * t)
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < cur_score - 1e-15:
            h = h_new
            cur_score = sc
            no_improve = 0
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 200:
                h = best_h.copy()
                cur_score = best_score
                no_improve = 0
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_refine(h_init, n, target, time_limit, start_time,
                      max_steps=30000, base_scale=0.02):
    rng = np.random.RandomState(12345)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 5000:
            break

        scale = base_scale * max(0.05, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 4, 6]))
        n_pert = min(n_pert, n)
        perm = rng.permutation(n)
        idxs = perm[:n_pert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 150:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5
