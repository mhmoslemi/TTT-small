def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

                           method='SLSQP', bounds=bounds, constraints=cons,

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=15, n_cand_end=70, max_no_improve=400,
            weight_beta_start=3.0, weight_beta_end=10.0,
            active_tol=1e-10, restart_interval=200, restart_scale=0.005):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(777)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:

            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_interval and consecutive_no_improve % restart_interval == 0:
            h += rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            for p in range(n):
                update_change_row(change, p, h, n, shifts)
            continue

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        weight_beta = weight_beta_start + (weight_beta_end - weight_beta_start) * progress

        c_max = float(np.max(corr))
        active_mask = corr >= c_max - max(active_tol, c_max * 1e-8)

        top_k = min(40, n_corr)
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = np.exp(weight_beta * (corr[s_idx] - c_max) / max(c_max, 1e-10))
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)

        high_set = sorted_idx[-current_ncand:]

        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-12:
                    continue

                g = change[j] - change[i]
                g_active_max = np.max(g[active_mask]) if active_mask.any() else 0.0

                if g_active_max < -1e-12:
                    delta_star = delta_max
                    approx_max = float(np.max(corr + delta_star * g)) * dx
                else:
                    a_d = 0.0
                    b_d = delta_max
                    for _ in range(22):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        am = vals >= f_mid - 1e-14

                            f_prime = np.max(g[am])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d
                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue
                    approx_max = float(np.max(corr + delta_star * g)) * dx

                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish
    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, n0, target0,
                                [200, 1000, 5000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Main fast CD (the workhorse)
    if tleft() > 300:
        tl = min(400, tleft() * 0.55)
        h_cd, c5_cd = fast_cd(best_h, n0, target0,
                              time_limit=tl,
                              max_sweeps=50000,
                              n_cand_start=20, n_cand_end=70,
                              max_no_improve=500,
                              weight_beta_start=3.0, weight_beta_end=12.0,
                              active_tol=1e-10,
                              restart_interval=250, restart_scale=0.004)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Momentum CD
    if tleft() > 200:
        tl = min(180, tleft() * 0.4)
        h_mom, c5_mom = momentum_cd(best_h, n0, target0,
                                    time_limit=tl,
                                    max_sweeps=20000,
                                    n_cand=45, beta=0.7, weight_beta=6.0)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    # Phase 4: SLSQP polish after CD
    if tleft() > 100:
        h_pol2, sc2 = slsqp_round(best_h, n0, target0,
                                  [2000, 10000, 30000], 500, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 5: Second CD pass with different parameters
    if tleft() > 200:
        tl = min(180, tleft() * 0.4)
        h_cd2, c5_cd2 = fast_cd(best_h, n0, target0,
                                time_limit=tl,
                                max_sweeps=40000,
                                n_cand_start=25, n_cand_end=80,
                                max_no_improve=400,
                                weight_beta_start=5.0, weight_beta_end=15.0,
                                active_tol=5e-11,
                                restart_interval=300, restart_scale=0.003)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 6: Double-step CD for deeper local search
    if tleft() > 120:
        tl = min(100, tleft() * 0.4)
        h_ds, c5_ds = double_step_cd(best_h, n0, target0,
                                     time_limit=tl,
                                     max_sweeps=8000,
                                     n_cand=30, weight_beta=6.0)
        if c5_ds < best_score:
            best_score = c5_ds
            best_h = h_ds

    # Phase 7: Multi-start with structured perturbations
    if tleft() > 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)
        for attempt in range(3):
            if tleft() < 80:
                break
            if attempt == 0:
                h_pert = best_h + rng.randn(n_cur) * 0.02
            elif attempt == 1:
                shift_amt = rng.randint(5, 20)
                h_pert = np.roll(best_h, shift_amt)
            else:
                h_pert = 1.0 - best_h
                h_pert = project_feasible(h_pert, target_cur)
                h_pert += rng.randn(n_cur) * 0.01

            h_pert = project_feasible(h_pert, target_cur)

            if tleft() > 60:
                h_ref, sc_ref = slsqp_round(h_pert, n_cur, target_cur,
                                            [500, 5000], 300, 1e-14)
                if tleft() > 40:
                    h_ref2, c5_ref2 = fast_cd(h_ref, n_cur, target_cur,
                                              time_limit=min(35, tleft() * 0.4),
                                              max_sweeps=10000,
                                              n_cand_start=20, n_cand_end=50,
                                              max_no_improve=200,
                                              weight_beta_start=4.0,
                                              weight_beta_end=8.0)
                    if c5_ref2 < best_score:
                        best_score = c5_ref2
                        best_h = h_ref2
                elif sc_ref < best_score:
                    best_score = sc_ref
                    best_h = h_ref

    # Phase 8: Try coarser resolution and upsample
    if tleft() > 200 and len(best_h) > 400:
        n_coarse = max(200, len(best_h) // 3)
        target_coarse = n_coarse / 2.0
        h_c = interpolate_to(best_h, n_coarse, target_coarse)
        if tleft() > 150:
            h_c_pol, sc_c = slsqp_round(h_c, n_coarse, target_coarse,
                                        [500, 2000], 300, 1e-13)
            if tleft() > 100:
                h_c_cd, c5_c_cd = fast_cd(h_c_pol, n_coarse, target_coarse,
                                          time_limit=min(60, tleft() * 0.4),
                                          max_sweeps=20000,
                                          n_cand_start=10, n_cand_end=35,
                                          max_no_improve=200,
                                          weight_beta_start=3.0,
                                          weight_beta_end=8.0)
                h_up = interpolate_to(h_c_cd, len(best_h), target0)
                sc_up = compute_c5(h_up)
                if sc_up < best_score:
                    best_score = sc_up
                    best_h = h_up

    # Phase 9: Final CD polish
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        tl = min(60, tleft() * 0.5)
        h_final, c5_final = fast_cd(best_h, n_cur, target_cur,
                                    time_limit=tl,
                                    max_sweeps=20000,
                                    n_cand_start=30, n_cand_end=70,
                                    max_no_improve=300,
                                    weight_beta_start=6.0,
                                    weight_beta_end=12.0)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final

    # Phase 10: Final SLSQP
    if tleft() > 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin, sc_fin = slsqp_round(best_h, n_cur, target_cur,
                                    [10000, 50000, 100000], 400, 1e-14)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
