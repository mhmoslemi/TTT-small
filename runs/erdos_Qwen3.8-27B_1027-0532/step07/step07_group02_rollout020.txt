                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if stochastic_frac > 0:
            n_pairs = len(high_set) * len(low_set)
            n_random = int(stochastic_frac * n_pairs)
            for _ in range(n_random):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_binsearch):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def block_lp_cd(h_init, n, target, time_limit, max_sweeps=20000,
                block_size=4, n_cand=50, seed=42):
    """Block coordinate descent using LP for the subproblem."""
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 300:
            break
        
        # Compute softmax-weighted gradient
        temp = 3.0
        corr_shifted = (corr - np.max(corr)) / temp
        weights = np.exp(corr_shifted)
        wsum = np.sum(weights)
        if wsum < 1e-30:
            weights = np.zeros(n_corr)
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights /= wsum
        
        grad = np.zeros(n)
        for s in range(n_corr):
            w = weights[s]
            if w < 1e-10:
                continue
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Select block: mix of extreme gradient variables and random
        sorted_idx = np.argsort(grad)
        low_candidates = sorted_idx[:n_cand]
        high_candidates = sorted_idx[-n_cand:]
        
        B = block_size
        # Select half from low, half from high
        n_low = B // 2
        n_high = B - n_low
        
        if rng.rand() < 0.1:
            # Random block
            block = rng.choice(n, size=B, replace=False)
        else:
            low_sel = rng.choice(low_candidates, size=n_low, replace=False)
            high_sel = rng.choice(high_candidates, size=n_high, replace=False)
            block = np.concatenate([low_sel, high_sel])
            # Remove duplicates
            block = np.unique(block)
            if len(block) < 2:
                continue
            B = len(block)
        
        # Build LP
        # Variables: [t, d_1, ..., d_B]
        # Minimize t
        # Subject to: t - sum_k d_k * change[block[k]][s] >= corr[s] for all s
        #             -h[block[k]] <= d_k <= 1 - h[block[k]]
        #             sum d_k = 0
        # 
        # In scipy form: A_ub @ x <= b_ub
        # For each s: -t + sum_k d_k * change[block[k]][s] <= -corr[s]
        change_block = change[block]  # shape (B, n_corr)
        
        # A_ub: (n_corr + 2*B) x (B+1)
        A_ub = np.zeros((n_corr + 2 * B, B + 1))
        b_ub = np.zeros(n_corr + 2 * B)
        
        # Correlation constraints
        A_ub[:n_corr, 0] = -1.0
        A_ub[:n_corr, 1:] = change_block  # (B, n_corr) transposed
        b_ub[:n_corr] = -corr
        
        # Upper bounds on d_k
        for k in range(B):
            A_ub[n_corr + k, 1 + k] = 1.0
            b_ub[n_corr + k] = 1.0 - h[block[k]]
        # Lower bounds on d_k (as -d_k <= h[block[k]])
        for k in range(B):
            A_ub[n_corr + B + k, 1 + k] = -1.0
            b_ub[n_corr + B + k] = h[block[k]]
        
        # Equality: sum d_k = 0
        A_eq = np.zeros((1, B + 1))
        A_eq[0, 1:] = 1.0
        b_eq = np.array([0.0])
        
        # Objective: minimize t
        c_obj = np.zeros(B + 1)
        c_obj[0] = 1.0
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          method='highs')
            if res.success:
                d_opt = res.x[1:]
                h_new = h.copy()
                for k in range(len(block)):
                    h_new[block[k]] += d_opt[k]
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < c5_val - 1e-15:
                    for k in range(len(block)):
                        update_change_row(change, int(block[k]), h_new, n, shifts)
                    h = h_new
                    # Don't update best here, will check at top of loop
                elif rng.rand() < 0.01:
                    # Occasionally accept non-improving moves (simulated annealing)
                    h = h_new
                    for k in range(len(block)):
                        update_change_row(change, int(block[k]), h, n, shifts)
        except Exception:
            pass
        
        if consecutive_no_improve > 100 and consecutive_no_improve % 50 == 0:
            # Perturbation
            idxs_pert = rng.choice(n, size=n // 8, replace=False)
            h[idxs_pert] += rng.randn(len(idxs_pert)) * 0.03
            h = project_feasible(h, target)
            for idx in idxs_pert:
                update_change_row(change, int(idx), h, n, shifts)
    
    return best_h, best_c5

def nesterov_smooth(h_init, n, target, time_limit, alpha=10000.0,
                    lr_init=0.01, max_iters=5000, seed=42):
    """Nesterov accelerated gradient on smoothed max objective."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)
    
    h = h_init.copy().astype(float)
    y = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    beta = 1.0
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        mu = (1.0 - beta) / beta
        y = h + mu * (y - h) if it > 0 else h.copy()
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)
        
        # Compute gradient at y
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr_y)
        shifted = np.clip(alpha * (corr_y - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum < 1e-30:
            weights = np.zeros(n_corr)
            weights[int(np.argmax(corr_y))] = 1.0
        else:
            weights /= wsum
        
        grad = np.zeros(n)
        for s in range(n_corr):
            w = weights[s]
            if w < 1e-12:
                continue
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - y[idxs[v1] - k])
            grad[v2] -= w * y[idxs[v2] + k]
        
        # Adaptive step size (backtracking)
        lr = lr_init / (1.0 + it * 0.01)
        for _ in range(10):
            h_new = y - lr * grad
            h_new = project_feasible(h_new, target)
            f_new = _smooth_obj(h_new, alpha)
            f_y = _smooth_obj(y, alpha) + lr * np.dot(grad, h_new - y)
            if f_new <= f_y + 1e-10:
                break
            lr *= 0.5
        
        h = h_new
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
