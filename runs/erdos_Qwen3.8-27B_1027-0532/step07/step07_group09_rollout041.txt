import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15

            break

        h = np.clip(h, 0.0, 1.0)

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    from scipy.interpolate import CubicSpline, PchipInterpolator
    x_old = np.linspace(0.0, 1.0, n_src)

            h_out = np.interp(x_new, x_old, h_src)
    else:
        h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_g_n = 0
_g_target = 0.0

    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol,
                                    'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

    h_cur = h.copy().astype(float)
    v = np.zeros(n)
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    max_iter = 50000
    lr = 0.01
    for it in range(max_iter):
        # Nesterov lookahead
        y = h_cur + momentum * v
        
        # Compute gradient at y
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        tol = max(1e-10, 1e-6 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]
        
        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - y[idxs[v1] - k])
            grad[v2] -= y[idxs[v2] + k]
        grad /= max(len(active), 1)
        grad -= np.mean(grad)

        h_new = project_feasible(h_cur + v, target)
        
        # Line search
        score = compute_c5(h_new)
        if score < best_score - 1e-15:

            best_h = h_new.copy()
        
        if score < compute_c5(h_cur) - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.8
            v = momentum * v
    
    return best_h, best_score

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    n_src = len(h_src)
