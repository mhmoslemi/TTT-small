                    d_candidate = 0.5 * (delta_lo + delta_hi)
                
                # Try a few candidates around d_candidate
                candidates = [d_candidate, 0.5 * d_candidate, 0.75 * d_candidate, 
                             0.25 * d_candidate, delta_hi * 0.5]
                candidates = [c for c in candidates if delta_lo < c < delta_hi - 1e-14]
                
                for d in candidates:
                    approx_max = np.max(corr + d * g) * dx
                    if approx_max < best_trial_c5:

                        best_i = i
                        best_j = j
                        best_delta = d

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def aggressive_cd(h_init, n, target, time_limit, max_sweeps=60000,
                  n_cand_min=25, n_cand_max=130, n_shifts=5,
                  max_no_improve=1000, restart_every=150):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(99999)

    change = compute_change_matrix(h, n)
    c5_improve_count = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            c5_improve_count += 1
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_every and consecutive_no_improve % 40 == 0:
            scale = 0.015 * (1.0 + 0.002 * (consecutive_no_improve - restart_every))
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0

        top_shifts = np.argsort(corr)[-n_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(consecutive_no_improve) / float(max_no_improve), 1.0)
        current_ncand = int(n_cand_min + (n_cand_max - n_cand_min) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        corr_max = float(np.max(corr))
        corr_sorted = np.sort(corr)[::-1]
        second_max = float(corr_sorted[1]) if n_corr > 1 else 0.0
        gap = corr_max - second_max

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-14:
                    continue
                g = change[j] - change[i]

                active_mask = corr >= corr_max - 1e-8
                g_active = g[active_mask]
                if g_active.size > 0:
                    g_amax = float(np.max(g_active))
                    if g_amax > 1e-12:
                        d_est = gap / g_amax * 0.9
                        d_est = max(delta_lo + 1e-12, min(d_est, delta_hi))
                    else:
                        d_est = 0.5 * (delta_lo + delta_hi)
                else:
                    d_est = 0.5 * (delta_lo + delta_hi)

                candidates = np.array([d_est, 0.5 * d_est, 0.75 * d_est,
                                       0.25 * d_est, 1.25 * d_est, 0.5 * delta_hi])
                candidates = candidates[(candidates > delta_lo + 1e-14) &
                                        (candidates < delta_hi - 1e-14)]
                if candidates.size == 0:
                    continue

                vals = corr[None, :] + candidates[:, None] * g[None, :]
                row_max = np.max(vals, axis=1) * dx
                best_local = int(np.argmin(row_max))
                approx_max = row_max[best_local]

                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = float(candidates[best_local])

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)
