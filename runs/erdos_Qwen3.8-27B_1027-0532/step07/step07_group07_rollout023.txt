def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)

        grad[v1] += w * (1.0 - x[idxs[v1] - k])

    return grad

    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

    return smooth_obj(x, _g_alpha)

def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def pair_cd_improved(h_init, n, target, time_limit, max_sweeps=200000,
                     n_cand_start=40, n_cand_end=120, top_k_eval=60,
                     max_no_improve=2000, n_shifts_grad=3):
    """Improved pair coordinate descent with:
    - Multi-shift gradient for ranking
    - Vectorized bisection
    - Progressive candidate set growth
    """
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Compute gradient from top shifts
        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        # Rank candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        # Score pairs using the max shift
        k_star_idx = int(np.argmax(corr))
        ch_h = change[high_set, k_star_idx]
        ch_l = change[low_set, k_star_idx]
        score_mat = ch_l[None, :] - ch_h[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval - 1)[:n_to_eval] if n_to_eval < flat_scores.size else np.arange(flat_scores.size)

        # Vectorized bisection for top candidates
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_eval = min(n_to_eval, 80)  # cap for speed
        top_flat = top_flat[:n_eval]

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            # Bisection
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

        # Periodically evaluate true C5
        if sweep % 100 == 0:
            true_c5 = compute_c5(h)
            if true_c5 < best_c5 - 1e-14:
                best_c5 = true_c5
                best_h = h.copy()

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        else:

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_eval_this = min(n_to_eval, 100)
        top_flat = top_flat[:n_eval_this]

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(13):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def fista_smooth(h_init, n, target, alpha, time_limit, max_iter=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    t_k = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    L = alpha * 2.0 / dx * 1.5
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        y = h + (t_k - 1.0) / (t_k + 1.0) * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        grad = smooth_obj_grad(y, alpha)
        step = 1.0 / L
        h_new = y - step * grad
        h_new = project_feasible(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        h_prev = h
        h = h_new
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol
                n0 = n1
                target0 = target1

    tl = _time_left(start_time, budget_s)
    if tl > 350:
        n2 = 2000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

        tl = _time_left(start_time, budget_s)
        if tl > 200:
            h_cd2, c5_cd2 = pair_cd_improved(h2, n2, target2,

                                              n_cand_start=50, n_cand_end=150,
                                              top_k_eval=100, max_no_improve=5000,
