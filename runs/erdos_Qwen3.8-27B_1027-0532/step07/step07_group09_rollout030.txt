    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)

    mid = n - 1
    k = k_idx - mid
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)

def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

        grad[v1] += w * (1.0 - x[idxs[v1] + k])

    return grad

    _g_target = target

    cur_score = compute_c5(h_cur)

        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,), jac=lambda x, a: _smooth_obj_grad(x, a),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

def lp_global_step(h, n, target, corr, top_k=15, max_iters=50):
    n_corr = 2 * n - 1
    mid = n - 1
    c5_corr_target = float(np.max(corr)) + 0.01

    A_rows = []
    b_vals = []

    top_idx = np.argsort(corr)[-top_k:]
    for k_idx in top_idx:
        k = int(k_idx) - mid
        grad = compute_grad_at_shift(h, k, n)
        rhs = c5_corr_target - corr[k_idx] + np.dot(grad, h)
        A_rows.append(np.concatenate([grad, [-1.0]]))
        b_vals.append(rhs)

    if len(A_rows) < 5:
        return None

    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(-1.0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_sol = project_feasible(res.x[:n], target)
            return h_sol
    except Exception:
        pass
    return None


def nesterov_subgrad(h_init, n, target, time_limit, max_iter=50000, lr_start=0.005):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    x = h_init.copy().astype(float)
    y = h_init.copy().astype(float)
    best_h = h_init.copy().astype(float)
    best_c5 = compute_c5(best_h)
    no_improve = 0
    beta_prev = 1.0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_corr = np.max(corr)
        tol = max(1e-10, 1e-6 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        n_act = len(active)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - y[idxs[v1] - k])
            grad[v2] -= y[idxs[v2] + k]
        grad /= n_act
        grad -= np.mean(grad)

        progress = min(it / max_iter, 1.0)
        lr = lr_start * (1.0 - 0.9 * progress) ** 1.5

        x_new = project_feasible(y - lr * grad, target)
        sqrt_t = np.sqrt(float(it + 1))
        beta = (1.0 + sqrt_t) / 2.0
        theta = (beta_prev - 1.0) / beta
        y_new = x_new + theta * (x_new - x)
        y_new = project_feasible(y_new, target)
        x = x_new
        y = y_new
        beta_prev = beta

    return best_h, best_c5


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        pc = 0
        v = i
        while v > 0:
            pc += v & 1
            v >>= 1
        h[i] = 1.0 if (pc % 2 == 1) else 0.0
    return project_feasible(h, n / 2.0)


def golden_ratio_init(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return project_feasible(h, n / 2.0)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

        return budget_s - 40.0 - (time.time() - start_time)
