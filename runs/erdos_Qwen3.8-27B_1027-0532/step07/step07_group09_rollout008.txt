import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

        threshold = max_corr - 0.05 * max_corr
        active = np.where(corr >= threshold)[0]
        n_act = min(len(active), 20)
        active = active[np.argsort(corr[active])[-n_act:]]

        grad = np.zeros(n)
        for si in active:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= n_act
        grad -= np.mean(grad)

        ncand = min(n_cand, n // 4)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)])
        h_hi = h[high_set]
        h_lo = h[low_set]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]

                delta_max = min(float(h_hi[ii]), 1.0 - float(h_lo[jj]))
                delta_min = max(float(h_hi[ii]) - 1.0, -float(h_lo[jj]))
                if delta_max - delta_min < 1e-12:
                    continue

                g_vec = G[lr] - G[hr]
                a_d = delta_min
                b_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g_vec[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec)
                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, mid)
            update_change_row(change, best_j, h, n, mid)

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, max_sweeps=15000,
             block_sizes=(2, 3, 4, 5, 6, 8), max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    rng = np.random.RandomState(12345)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        idxs = np.arange(n)

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        bs = block_sizes[sweep % len(block_sizes)]
        n_pos = n - bs + 1
        if n_pos <= 0:
            continue

        src_start = rng.randint(0, n_pos)
        block_grad = grad[src_start:src_start + bs].sum()

        search_range = min(80, n_pos)
        offsets = rng.permutation(n_pos)[:search_range]
        best_search_grad = -np.inf
        best_target_pos = -1
        for sp in offsets:
            if sp == src_start:
                continue
            bg = grad[sp:sp + bs].sum()
            if bg > best_search_grad:
                best_search_grad = bg
                best_target_pos = int(sp)

        if best_target_pos < 0 or src_start == best_target_pos:
            continue

        if block_grad > best_search_grad:
            s_start, d_start = src_start, best_target_pos
        else:
            s_start, d_start = best_target_pos, src_start

        src_block = h[s_start:s_start + bs]
        dst_block = h[d_start:d_start + bs]

        delta_max = min(float(src_block.max()), float((1.0 - dst_block).max()))
        delta_min = max(float(src_block.min()) - 1.0, float(-dst_block.min()))
        delta_max = min(delta_max, float(src_block.max()))
        delta_min = max(delta_min, -float(src_block.min()))

        if delta_max - delta_min < 1e-12:
            continue

        g = np.zeros(n_corr)
        for t in range(bs):
            g += change[d_start + t, :] - change[s_start + t, :]

        a_d = delta_min
        b_d = delta_max
        for _ in range(12):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr + mid_d * g
            f_mid = np.max(vals)
            active_mask = vals >= f_mid - 1e-14
            if active_mask.any():
                f_prime = np.max(g[active_mask])
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                b_d = mid_d
            else:
                a_d = mid_d

        delta_star = 0.5 * (a_d + b_d)
        if abs(delta_star) < 1e-12:
            continue

        approx_max = np.max(corr + delta_star * g)
        if approx_max < c5_val - 1e-14:
            h[s_start:s_start + bs] -= delta_star
            h[d_start:d_start + bs] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for t in range(bs):
                update_change_row(change, s_start + t, h, n, mid)
                update_change_row(change, d_start + t, h, n, mid)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_step(h_init, n, target, time_limit, max_iter=20000, lr_start=0.005):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 4000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_corr = np.max(corr)
        tol = max(1e-8, 1e-4 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]
        n_act = min(len(active), 25)
        active = active[np.argsort(corr[active])[-n_act:]]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)

        progress = min(it / max_iter, 1.0)
        lr = lr_start * (1.0 - 0.97 * progress ** 0.7)

        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val + 1e-15:
            h = h_new

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 50:
        h_pol = slsqp_round(best_h, n0, target0,
                            [200, 1000, 5000, 20000],
                            600, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n1 = 1000
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)
    best_h = h1
    best_score = compute_c5(best_h)
    n0 = n1
    target0 = target1

    if tleft() > 500:
        h_cd, c5_cd = aggressive_cd(best_h, n0, target0,
                                    time_limit=min(300, tleft() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_min=30, n_cand_max=80,
                                    max_no_improve=500,
                                    n_active_min=5, n_active_max=15,
                                    bisection_iters=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        h_pol = slsqp_round(best_h, n0, target0,
                            [1000, 5000, 20000, 80000],
                            800, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_mc, c5_mc = multi_active_cd(best_h, n0, target0,
                                      time_limit=min(200, tleft() * 0.55),
                                      max_sweeps=15000,
                                      n_cand=70, max_no_improve=400)
        if c5_mc < best_score:
            best_score = c5_mc
            best_h = h_mc

    if tleft() > 100:
        h_pol = slsqp_round(best_h, n0, target0,
                            [2000, 10000, 50000],
                            800, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_bm, c5_bm = block_cd(best_h, n0, target0,
                               time_limit=min(120, tleft() * 0.55),
                               max_sweeps=12000,
                               block_sizes=(2, 3, 4, 5, 6, 8, 10),
                               max_no_improve=500)
        if c5_bm < best_score:
            best_score = c5_bm
            best_h = h_bm

    if tleft() > 100:
        h_cd2, c5_cd2 = aggressive_cd(best_h, n0, target0,
                                      time_limit=min(80, tleft() * 0.55),
                                      max_sweeps=10000,
                                      n_cand_min=40, n_cand_max=90,
                                      max_no_improve=300,
                                      n_active_min=8, n_active_max=20,
                                      bisection_iters=14)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n0, target0,
                            [5000, 20000, 80000],
                            600, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        h_sg, c5_sg = subgradient_step(best_h, n0, target0,
                                       time_limit=min(60, tleft() * 0.5),
                                       max_iter=15000, lr_start=0.004)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 120:
        n2 = 1400
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 100:
            h_cd3, c5_cd3 = aggressive_cd(h2, n2, target2,
                                          time_limit=min(80, tleft() * 0.55),
                                          max_sweeps=8000,
                                          n_cand_min=25, n_cand_max=60,
                                          max_no_improve=300,
                                          n_active_min=5, n_active_max=12,
                                          bisection_iters=12)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
                n0 = n2
                target0 = target2
        if tleft() > 50:
            h_pol3 = slsqp_round(best_h, n0, target0,
                                 [2000, 10000, 40000],
                                 500, 1e-14)
            sc3 = compute_c5(h_pol3)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    if tleft() > 60:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.06]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref = slsqp_round(h_pert, n0, target0,
                                [500, 2000, 10000],
                                400, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 60:
        h_cd_fin, c5_fin = aggressive_cd(best_h, n0, target0,
                                         time_limit=min(40, tleft() * 0.6),
                                         max_sweeps=6000,
                                         n_cand_min=50, n_cand_max=100,
                                         max_no_improve=200,
                                         n_active_min=10, n_active_max=20,
                                         bisection_iters=14)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_cd_fin

    if tleft() > 30:
        h_final = slsqp_round(best_h, n0, target0,
                              [10000, 40000, 150000],
                              500, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
