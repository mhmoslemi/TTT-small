                    # When we do h[i] -= d, h[j] += d:
                    # delta_corr[s] = -d * d(corr[s])/d(h[i]) + d * d(corr[s])/d(h[j])
                    #              = d * (d(corr[s])/d(h[j]) - d(corr[s])/d(h[i]))
                    #              = d * ((1-h[j+k]-h[j-k]) - (1-h[i+k]-h[i+k]))
                    #              = d * (h[i+k] + h[i-k] - h[j+k] - h[j-k])
                    #
                    # So we want this to be negative (decrease corr[s]) for the max shift.
                    # delta_corr[k_star] = d * (h[i+k] + h[i-k] - h[j+k] - h[j-k])
                    # We want h[i+k] + h[i-k] < h[j+k] + h[j-k], i.e., val_i < val_j
                    # which is what we checked above (val_j > val_i).
                    #
                    # The rate of decrease at k_star is: (val_j - val_i) per unit delta
                    # But we also need to check other shifts don't increase too much.
                    # For a first-order check, the max change across all shifts is what matters.
                    # But that's expensive. Let's just check k_star and use bisection.
                    
                    # For bisection, we need to check if the new max corr (after delta) is < current max
                    # This requires computing the full corr, which is O(n)
                    # With 10 bisection steps × O(n) per step = 10n = 8000 ops per pair
                    # With 30×30 = 900 pairs, that's 7.2M ops per sweep. Too slow.
                    
                    # Better: use the linear approximation for the candidate check,
                    # and only do full evaluation for the best candidate.
                    pass
                
                # Use linear approximation: delta_corr[k_star] ≈ d * (val_i - val_j) < 0
                # Also check that no other active shift increases too much
                # For simplicity, just check k_star and use a small step
                # The step is limited by delta_hi
                # Use a fraction of the max possible step for safety
                max_d = delta_hi * 0.8
                approx_new_max = c5_val - max_d * (val_j - val_i) * dx
                if approx_new_max < best_trial:
                    # Verify with actual computation (expensive but only for best candidate)
                    h_trial = h.copy()
                    h_trial[i] -= max_d
                    h_trial[j] += max_d
                    h_trial = np.clip(h_trial, 0.0, 1.0)
                    h_trial = project_feasible(h_trial, target)
                    corr_trial = compute_corr(h_trial)
                    actual_new = float(np.max(corr_trial)) * dx
                    if actual_new < best_trial:
                        best_trial = actual_new
                        best_i = i
                        best_j = j
                        best_delta = max_d
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = compute_corr(h)
        else:

                corr_pert = compute_corr(h_pert)

                if delta_sc < 0 or np.log(rng.uniform()) > delta_sc / max(temp, 1e-10):

                    corr = corr_pert

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    alpha = _g_alpha
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def build_change_matrix(h, n):
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos = p + shifts
        neg = p - shifts
        mp = (pos >= 0) & (pos < n)
        mn = (neg >= 0) & (neg < n)
        change[p, mp] += (1.0 - h[pos[mp]])
        change[p, mn] -= h[neg[mn]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos = p + shifts
    neg = p - shifts
    mp = (pos >= 0) & (pos < n)
    mn = (neg >= 0) & (neg < n)
    change[p, mp] += (1.0 - h[pos[mp]])
    change[p, mn] -= h[neg[mn]]
    return change


def apply_structured_perturbation(h, n, rtype, rng, scale=0.04):
    h_new = h.copy()
    if rtype == 'gaussian':
        h_new += rng.randn(n) * scale
    elif rtype == 'fourier':
        freq = rng.randint(2, max(3, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        h_new += scale * np.sin(2 * np.pi * freq * x / n + phase)
        if rng.random() < 0.5:
            freq2 = rng.randint(2, max(3, n // 8))
            phase2 = rng.uniform(0, 2 * np.pi)
            h_new += 0.5 * scale * np.cos(2 * np.pi * freq2 * x / n + phase2)
    elif rtype == 'block':
        block_len = max(3, n // max(8, rng.randint(6, 12)))
        start = rng.randint(0, n - block_len)
        h_new[start:start + block_len] += rng.randn(block_len) * scale
    elif rtype == 'antisym':
        x = np.arange(n)
        mid = n // 2
        h_new += scale * np.sin(np.pi * (x - mid) / mid) if mid > 0 else 0
    return h_new

        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            weights = corr[top_k] / corr_top_sum

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        stuck_factor = min(consecutive_no_improve / max_no_improve, 1.0)
        base_ncand = int(n_cand_min + (n_cand_max - n_cand_min) * progress)
        current_ncand = min(int(base_ncand * (1.0 + 0.5 * stuck_factor)), n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star = int(np.argmax(corr))
        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                g_star = g[k_star]
                if g_star >= 0:
                    continue

                max_abs_g = np.max(np.abs(g))
                if max_abs_g < 1e-14:
                    continue
                if g_star > 0.5 * max_abs_g:
                    continue

                a_d = 0.0
                b_d = delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            corr = np.correlate(h, 1.0 - h, mode='full')
        else:
            if sweep % 5 == 0 and temp > 1e-7:
                rng = np.random.RandomState(sweep * 41 + 23)
                h_pert = h + rng.randn(n) * 0.005 * (temp / sa_temp + 0.1)
                h_pert = project_feasible(h_pert, target)
                corr_pert = np.correlate(h_pert, 1.0 - h_pert, mode='full')
                sc_pert = float(np.max(corr_pert)) * dx
                delta_sc = sc_pert - c5_val
                if delta_sc < 0 or np.log(rng.uniform(1e-10, 1.0)) > delta_sc / max(temp, 1e-10):
                    h = h_pert
                    corr = corr_pert
                    change = build_change_matrix(h, n)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    # Phase 1: Quick SLSQP at n0
    if tleft() > 80:
        deadline = time.time() + min(45, tleft() * 0.3)
        h_pol, sc = slsqp_polish(best_h, n0, target0,
                                  [80, 400, 2000, 8000],
                                  500, 1e-14, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD with SA at n0
    if tleft() > 250:
        tl = min(100, tleft() * 0.3)
        h_cd, c5_cd = cd_sa(best_h, n0, target0,
                             time_limit=tl, n_cand_min=15, n_cand_max=50,
                             n_active_shifts=4, sa_temp=0.003, sa_cool=0.999,
                             max_sweeps=20000, max_no_improve=200,
                             reheat_interval=60, reheat_scale=0.05)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP polish at n0
    if tleft() > 100:
        deadline = time.time() + min(40, tleft() * 0.25)
        h_pol, sc = slsqp_polish(best_h, n0, target0,
                                  [2000, 10000, 30000],
                                  800, 1e-14, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Interpolate to n=600
    if tleft() > 300:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1
            current_n = n1
            current_target = target1
        elif tleft() > 300:
            best_h = h1
            current_n = n1
            current_target = target1

        # Phase 5: CD at n=600
        if tleft() > 250:
            tl = min(150, tleft() * 0.4)
            h_cd1, c5_cd1 = cd_sa(best_h, n1, target1,
                                   time_limit=tl, n_cand_min=20, n_cand_max=55,
                                   n_active_shifts=5, sa_temp=0.002, sa_cool=0.9995,
                                   max_sweeps=30000, max_no_improve=300,
                                   reheat_interval=80, reheat_scale=0.04)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        # Phase 6: SLSQP at n=600
        if tleft() > 100:
            deadline = time.time() + min(55, tleft() * 0.3)
            h_pol1, sc1 = slsqp_polish(best_h, current_n, current_target,
                                        [500, 3000, 15000],
                                        800, 1e-14, deadline=deadline)
            if sc1 < best_score:
                best_score = sc1
                best_h = h_pol1

    # Phase 7: Interpolate to n=800
    if tleft() > 300:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
        current_n = n2
        current_target = target2

        # Phase 8: CD at n=800
        if tleft() > 200:
            tl = min(140, tleft() * 0.45)
            h_cd2, c5_cd2 = cd_sa(best_h, n2, target2,
                                   time_limit=tl, n_cand_min=25, n_cand_max=65,
                                   n_active_shifts=5, sa_temp=0.0015, sa_cool=0.9997,
                                   max_sweeps=25000, max_no_improve=350,
                                   reheat_interval=70, reheat_scale=0.035)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

        # Phase 9: SLSQP at n=800
        if tleft() > 80:
            deadline = time.time() + min(50, tleft() * 0.3)
            h_pol2, sc2 = slsqp_polish(best_h, current_n, current_target,
                                        [1000, 5000, 20000],
                                        700, 1e-14, deadline=deadline)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

    # Phase 10: Multi-start with structured perturbations
    if tleft() > 200:
        n_pert_start = len(best_h)
        pert_configs = [
            (0.03, 'gaussian'), (0.05, 'fourier'), (0.04, 'block'),
            (0.07, 'gaussian'), (0.06, 'fourier'), (0.08, 'antisym'),
            (0.10, 'gaussian'), (0.08, 'fourier')
        ]
        for pi, (scale, ptype) in enumerate(pert_configs):
            if tleft() < 50:
                break
            h_pert = apply_structured_perturbation(best_h, n_pert_start, ptype, rng, scale)
            h_pert += rng.randn(n_pert_start) * scale * 0.5
            h_pert = project_feasible(h_pert, current_target)

            deadline = time.time() + min(25, tleft() * 0.2)
            h_ref, sc_ref = slsqp_polish(h_pert, current_n, current_target,
                                          [500, 2000, 8000],
                                          400, 1e-14, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 11: Final CD round
    if tleft() > 100:
        tl = min(50, tleft() * 0.45)
        h_cd_f, c5_f = cd_sa(best_h, current_n, current_target,
                              time_limit=tl, n_cand_min=30, n_cand_max=70,
                              n_active_shifts=5, sa_temp=0.001, sa_cool=0.9999,
                              max_sweeps=15000, max_no_improve=200,
                              reheat_interval=50, reheat_scale=0.02)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    # Phase 12: Final SLSQP
    if tleft() > 35:
        deadline = time.time() + min(20, tleft() * 0.5)
        h_final, sc_final = slsqp_polish(best_h, current_n, current_target,
                                          [5000, 20000, 60000],
                                          700, 1e-15, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
