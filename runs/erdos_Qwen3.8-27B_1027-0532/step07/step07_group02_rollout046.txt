                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-12:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        return np.zeros(len(x))
    weights /= wsum
    n = len(x)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s in range(2 * n - 1):
        w = weights[s]
        if w < 1e-12:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_cd(h_init, n, target, time_limit, block_size=3, max_sweeps=5000, seed=77):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500:
            break
        start_idx = int(rng.randint(n - block_size + 1))
        block = slice(start_idx, start_idx + block_size)
        block_vals = h[block].copy()
        block_sum = np.sum(block_vals)
        base = h.copy()

        def block_obj(params):
            trial = base.copy()
            trial[block] = params
            return compute_c5(trial)

        res = minimize(block_obj, block_vals, method='Nelder-Mead',
                       options={'maxiter': 200, 'fatol': 1e-12})
        new_vals = np.clip(res.x, 0.0, 1.0)
        diff = block_sum - np.sum(new_vals)
        if abs(new_vals).sum() > 1e-15:
            new_vals += diff / block_size
            new_vals = np.clip(new_vals, 0.0, 1.0)
        h[block] = new_vals
        h = project_feasible(h, target)
        c5_after = compute_c5(h)
        if c5_after > c5_val + 1e-10:
            h[block] = block_vals
            h = project_feasible(h, target)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick pair CD to improve initial
    if tleft() > 200:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(120, tleft() * 0.4),
                              max_sweeps=30000,
                              n_cand_start=40, n_cand_end=130,
                              max_no_improve=300,
                              n_binsearch=20,
                              stochastic_frac=0.08, seed=42)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: Successive linearization (NEW approach)
    if tleft() > 250:
        h_sl, c5_sl = successive_linearization(best_h, n0, target0,
                                               time_limit=min(200, tleft() * 0.5),
                                               max_iter=400, seed=seed)
        if c5_sl < best_score:
            best_score = c5_sl
            best_h = h_sl
        # Polish after SL
        if tleft() > 80:
            h_pol, c5_pol = pair_cd(best_h, n0, target0,
                                    time_limit=min(60, tleft() * 0.3),
                                    max_sweeps=20000,
                                    n_cand_start=50, n_cand_end=150,
                                    max_no_improve=200,
                                    n_binsearch=20,
                                    stochastic_frac=0.05, seed=199)
            if c5_pol < best_score:
                best_score = c5_pol
                best_h = h_pol

    # Phase 3: More SL iterations from potentially better point
    if tleft() > 200:
        h_sl2, c5_sl2 = successive_linearization(best_h, n0, target0,
                                                 time_limit=min(150, tleft() * 0.5),
                                                 max_iter=300, seed=seed + 1)
        if c5_sl2 < best_score:
            best_score = c5_sl2
            best_h = h_sl2
        if tleft() > 60:
            h_pol2, c5_pol2 = pair_cd(best_h, n0, target0,
                                      time_limit=min(50, tleft() * 0.3),
                                      max_sweeps=15000,
                                      n_cand_start=50, n_cand_end=140,
                                      max_no_improve=200,
                                      n_binsearch=20,
                                      stochastic_frac=0.06, seed=251)
            if c5_pol2 < best_score:
                best_score = c5_pol2
                best_h = h_pol2

    # Phase 4: Try block CD for escaping local optima
    if tleft() > 100:
        h_bc, c5_bc = block_cd(best_h, n0, target0,
                               time_limit=min(60, tleft() * 0.5),
                               block_size=3, max_sweeps=3000, seed=333)
        if c5_bc < best_score:
            best_score = c5_bc
            best_h = h_bc

    # Phase 5: SLSQP polish with high alpha
    if tleft() > 80:
        h_sls, c5_sls = slsqp_optimize(best_h, n0, target0,
                                       [5000, 20000, 80000],
                                       2000, 1e-15)
        if c5_sls < best_score:
            best_score = c5_sls
            best_h = h_sls

    # Phase 6: Interpolate to higher resolution
    if tleft() > 250:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        if tleft() > 150:
            h_cd1, c5_cd1 = pair_cd(h1, n1, target1,
                                     time_limit=min(120, tleft() * 0.5),
                                     max_sweeps=25000,
                                     n_cand_start=30, n_cand_end=100,
                                     max_no_improve=300,
                                     n_binsearch=18,
                                     stochastic_frac=0.04, seed=411)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
        if tleft() > 100:
            h_sl1, c5_sl1 = successive_linearization(best_h, n1, target1,
                                                     time_limit=min(80, tleft() * 0.5),
                                                     max_iter=200, seed=500)
            if c5_sl1 < best_score:
                best_score = c5_sl1
                best_h = h_sl1
        if tleft() > 50:
            h_sls1, c5_sls1 = slsqp_optimize(best_h, n1, target1,
                                             [10000, 50000],
                                             1500, 1e-15)
            if c5_sls1 < best_score:
                best_score = c5_sls1
                best_h = h_sls1

    # Phase 7: Try n=800 if time allows
    if tleft() > 300:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 200:
            h_cd2, c5_cd2 = pair_cd(h2, n2, target2,
                                     time_limit=min(150, tleft() * 0.5),
                                     max_sweeps=20000,
                                     n_cand_start=25, n_cand_end=80,
                                     max_no_improve=250,
                                     n_binsearch=18,
                                     stochastic_frac=0.03, seed=577)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
        if tleft() > 100:
            h_sl2b, c5_sl2b = successive_linearization(best_h, n2, target2,
                                                       time_limit=min(80, tleft() * 0.5),
                                                       max_iter=150, seed=600)
            if c5_sl2b < best_score:
                best_score = c5_sl2b
                best_h = h_sl2b
        if tleft() > 50:
            h_sls2, c5_sls2 = slsqp_optimize(best_h, n2, target2,
                                             [20000],
                                             1500, 1e-15)
            if c5_sls2 < best_score:
                best_score = c5_sls2
                best_h = h_sls2

    # Phase 8: Final polish
    if tleft() > 60:
        h_final_cd, c5_final_cd = pair_cd(best_h, len(best_h), len(best_h) / 2.0,
                                          time_limit=min(40, tleft() * 0.6),
                                          max_sweeps=10000,
                                          n_cand_start=50, n_cand_end=150,
                                          max_no_improve=200,
                                          n_binsearch=20,
                                          stochastic_frac=0.05, seed=701)
        if c5_final_cd < best_score:
            best_score = c5_final_cd
            best_h = h_final_cd

    if tleft() > 30:
        h_final_sls, c5_final_sls = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                                   [100000],
                                                   2000, 1e-15)
        if c5_final_sls < best_score:
            best_score = c5_final_sls
            best_h = h_final_sls

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
