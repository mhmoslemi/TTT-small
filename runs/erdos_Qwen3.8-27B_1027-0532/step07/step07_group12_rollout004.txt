def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)

def gradient_for_shift(h, k, n):
    """Compute d(corr[k])/d(h) for a given shift k (in range -(n-1) to n-1)."""
    grad = np.zeros(n)
    idxs = np.arange(n)
    # (1-h[i+k]) term: valid when 0 <= i+k < n, i.e., -k <= i < n-k
    v1 = (idxs + k >= 0) & (idxs + k < n)
    if np.any(v1):
        grad[v1] += (1.0 - h[idxs[v1] + k])
    # -h[i-k] term: valid when 0 <= i-k < n, i.e., k <= i < n+k
    v2 = (idxs - k >= 0) & (idxs - k < n)
    if np.any(v2):
        grad[v2] -= h[idxs[v2] - k]
    return grad

def nesterov_prox(h_init, n, target, time_limit, max_steps=50000, n_active=3, base_eta=0.01):
    """Proximal gradient with Nesterov momentum."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    z = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    
    for t in range(1, max_steps + 1):
        if time.time() - start > time_limit:
            break
        
        # Compute correlation
        corr = np.correlate(z, 1.0 - z, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                break
        
        # Get top n_active shifts
        top_idx = np.argsort(corr)[-n_active:]
        top_val = np.max(corr[top_idx])
        # Softmax weights over top shifts
        alpha_w = 10.0
        w = np.exp(alpha_w * (corr[top_idx] - top_val))
        w = w / w.sum()
        
        # Compute combined gradient
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            grad += wi * gradient_for_shift(z, k, n)
        
        # Nesterov momentum
        momentum = (t - 1.0) / (t + 1.0)
        
        # Step size (decreasing)
        eta = base_eta / (1.0 + 0.01 * t)
        
        # Try a few step sizes
        improved = False
        for eta_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            eta_try = eta * eta_mult
            h_new = z - eta_try * grad * dx
            h_new = prox_project(h_new, target)
            h_new_z = h_new + momentum * (h_new - h_prev)
            h_new_z = prox_project(h_new_z, target)
            c5_try = compute_c5(h_new)
            if c5_try < c5_now:
                h_prev = h
                h = h_new
                z = h_new_z
                improved = True
                break
        
        if not improved and t > 1000:
            # Try without momentum
            for eta_mult in [0.1, 0.05, 0.02, 0.01]:
                eta_try = base_eta * eta_mult
                h_new = z - eta_try * grad * dx
                h_new = prox_project(h_new, target)
                c5_try = compute_c5(h_new)
                if c5_try < c5_now:
                    h_prev = h
                    h = h_new
                    z = h_new
                    improved = True
                    break
            if not improved:
                break
    
    return best_h, best_c5

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] + k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] - k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_optimize(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)

        w = np.exp(alpha_w * (corr[top_idx] - top_val))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            grad += wi * gradient_for_shift(z, k, n)

        momentum = (t - 1.0) / (t + 1.0)
        eta = base_eta / (1.0 + 0.005 * t)

        improved = False
        for eta_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            eta_try = eta * eta_mult
            h_new = z - eta_try * grad * dx
            h_new = prox_project(h_new, target)
            z_new = h_new + momentum * (h_new - h_prev)
            z_new = prox_project(z_new, target)
            c5_try = compute_c5(h_new)
            if c5_try < c5_now:
                h_prev = h
                h = h_new
                z = z_new
                improved = True
                break

        if not improved and t > 2000:
            for eta_mult in [0.01, 0.005, 0.002]:
                h_new = z - base_eta * eta_mult * grad * dx
                h_new = prox_project(h_new, target)
                c5_try = compute_c5(h_new)
                if c5_try < c5_now:
                    h_prev = h
                    h = h_new
                    z = h_new
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_c5

def fast_pair_cd(h_init, n, target, time_limit, n_cand, max_sweeps, n_shifts):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    ks_all = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 600:
                break

        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = gradient_for_shift(h, k_star, n)
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            top_val = float(np.max(corr[top_idx]))
            alpha_w = 30.0
            w = np.exp(alpha_w * (corr[top_idx] - top_val))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                grad += wi * gradient_for_shift(h, k, n)

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        cands_arr = all_cands.astype(int)

        sums1 = cands_arr[:, None] + ks_all[None, :]
        valid1 = (sums1 >= 0) & (sums1 < n)
        clamped1 = np.clip(sums1, 0, n - 1)
        term1 = np.where(valid1, 1.0 - h[clamped1], 0.0)

        sums2 = cands_arr[:, None] - ks_all[None, :]
        valid2 = (sums2 >= 0) & (sums2 < n)
        clamped2 = np.clip(sums2, 0, n - 1)
        term2 = np.where(valid2, -h[clamped2], 0.0)

        profiles = term1 + term2

        cand_pos = {int(c): i for i, c in enumerate(cands_arr)}
        high_list = [int(x) for x in high_set]
        low_list = [int(x) for x in low_set]

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_list:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[i]
            for j in low_list:
                if i == j:
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[j]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-12:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] + k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] - k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_optimize(h_start, n, target, alphas, maxiter, time_limit, penalty):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 50:
        hp, sc = lbfgs_optimize(best_h, n, target,
                                alphas=[100, 500, 2000],
                                maxiter=250, time_limit=min(35, tleft() * 0.3),
                                penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 80:
        hn, scn = nesterov_prox(best_h, n, target, time_limit=min(50, tleft() * 0.3),
                                max_steps=50000, n_active=5, base_eta=0.015)
        if scn < best_score:
            best_score = scn
            best_h = hn

    if tleft() > 150:
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=min(250, tleft() * 0.35),
                                  n_cand=20, max_sweeps=50000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 70:
        hp, sc = lbfgs_optimize(best_h, n, target,
                                alphas=[5000, 20000, 100000],
                                maxiter=250, time_limit=min(45, tleft() * 0.35),
                                penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    if tleft() > 100:
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=min(130, tleft() * 0.35),
                                  n_cand=25, max_sweeps=50000, n_shifts=3)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 60:
        hn, scn = nesterov_prox(best_h, n, target, time_limit=min(40, tleft() * 0.3),
                                max_steps=50000, n_active=10, base_eta=0.01)
        if scn < best_score:
            best_score = scn
            best_h = hn

    if tleft() > 50:
        rng = np.random.RandomState(seed)
        n_use = len(best_h)
        target_use = n_use / 2.0
        for attempt in range(3):
            if tleft() < 30:
                break
            noise_scale = 0.015 * (attempt + 1)
            h_pert = best_h + rng.randn(n_use) * noise_scale
            h_pert = project_feasible(h_pert, target_use)
            hp2, sc2 = lbfgs_optimize(h_pert, n_use, target_use,
                                       alphas=[2000, 20000],
                                       maxiter=150, time_limit=min(20, tleft() * 0.2),
                                       penalty=500.0)
            if sc2 < best_score:
                best_score = sc2
                best_h = hp2

    if tleft() > 180:
        n_new = 900
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = lbfgs_optimize(h_interp, n_new, target_new,
                                alphas=[100, 1000, 10000, 50000],
                                maxiter=200, time_limit=min(80, tleft() * 0.35),
                                penalty=100.0)
        if sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new
        if tleft() > 60:
            hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=min(50, tleft() * 0.4),
                                      n_cand=20, max_sweeps=30000, n_shifts=1)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = fast_pair_cd(best_h, n_use, target_use, time_limit=min(30, tleft() * 0.5),
                                  n_cand=15, max_sweeps=30000, n_shifts=1)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_optimize(best_h, n_use, target_use,
                                alphas=[500000, 2000000],
                                maxiter=100, time_limit=min(20, tleft() * 0.7),
                                penalty=5000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
