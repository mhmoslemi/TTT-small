def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def proj_constrained(z, target_sum):
    """Project z onto {h: 0 <= h <= 1, sum(h) = target_sum} via bisection on lambda."""
    n = len(z)
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(z - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) / 2, 0.0, 1.0)

def compute_grad_smoothed(h, p, n):
    """Compute gradient of smoothed objective."""
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_offset in range(2 * n - 1):
        w = p[k_offset]
        if w < 1e-12:
            continue
        k = k_offset - (n - 1)
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] += w * (-h[idxs[v2] - k])
    return grad

def prox_grad_fista(h_init, n, target_sum, alpha, n_steps, step_scale, time_limit, start_time):
    """Proximal gradient with FISTA acceleration."""
    h = h_init.copy().astype(float)
    y = h.copy()
    t = 1.0
    step_size = step_scale / (alpha * n)
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(y, 1.0 - y, mode='full')
        shifted = np.clip(alpha * (corr - corr.max()), -500, 0)
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones(2*n-1) / (2*n-1)
        else:
            p = p / p_sum
        
        grad = compute_grad_smoothed(y, p, n)
        
        z = y - step_size * grad
        h_new = proj_constrained(z, target_sum)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        y = np.clip(y, -0.5, 1.5)
        
        h = h_new
        t = t_new
    
    return h

def cd_polish(h_init, n, target_sum, time_limit, max_sweeps=10000, n_cand=50):
    """Coordinate descent polish."""
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, 2*n-1))
    for p_idx in range(n):
        pos_mask = (p_idx + shifts >= 0) & (p_idx + shifts < n)
        neg_mask = (p_idx - shifts >= 0) & (p_idx - shifts < n)
        change[p_idx, pos_mask] += (1.0 - h[p_idx + shifts[pos_mask]])
        change[p_idx, neg_mask] -= h[p_idx - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > 300:
            break
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target_sum)
            # Update change rows
            for p_idx in [best_i, best_j]:
                change[p_idx, :] = 0.0
                pos_mask = (p_idx + shifts >= 0) & (p_idx + shifts < n)
                neg_mask = (p_idx - shifts >= 0) & (p_idx - shifts < n)
                change[p_idx, pos_mask] += (1.0 - h[p_idx + shifts[pos_mask]])
                change[p_idx, neg_mask] -= h[p_idx - shifts[neg_mask]]
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def proj_constrained(z, target_sum):
    n = len(z)
    lo = z.min() - 2.0
    hi = z.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(z - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask])

def smooth_obj_func(x, alpha, k_top):
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(corr)[-k_top:]
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr[idx] - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_constraint(x, target):
    return np.sum(x) - target

def sum_constraint_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h_start, n, target, alphas, k_tops, maxiter_per_alpha, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for i, alpha in enumerate(alphas):
        if time.time() - start_time > time_limit:
            break
        k_top = k_tops[min(i, len(k_tops) - 1)]
        try:
            res = minimize(smooth_obj_func, h_cur, args=(alpha, k_top),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def multi_shift_cd(h_init, n, target, time_limit, max_sweeps=20000,
                   n_cand_start=25, n_cand_end=60, n_shifts=3,
                   max_no_improve=400, bs_iters=12):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-n_shifts:]
        c_max = corr[np.argmax(corr)]
        weights = np.exp(-3.0 * (c_max - corr[top_shifts]) / max(c_max * dx, 1e-10))
        weights = weights / np.sum(weights)

        grad = np.zeros(n)
        for wi, s_idx in enumerate(top_shifts):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += weights[wi] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= weights[wi] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bs_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            # No improving pair found, try a small random perturbation
            if consecutive_no_improve % 50 == 0 and consecutive_no_improve < max_no_improve:
                p1 = rng_global.randint(n)
                p2 = rng_global.randint(n)
                if p1 != p2:
                    d = 0.01 * rng_global.rand()
                    h[p1] -= d
                    h[p2] += d
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    update_change_row(change, p1, h, n, shifts)
                    update_change_row(change, p2, h, n, shifts)

    return best_h, best_c5

def sa_escape(h_init, n, target, time_limit, max_steps=15000,
              t_start=0.03, t_end=0.0002, n_cand=40):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    local_rng = np.random.RandomState(777)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 500:
            break

        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[local_rng.randint(len(high_set))])
        j = int(low_set[local_rng.randint(len(low_set))])
        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue

        delta = (delta_hi - delta_lo) * local_rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-12:
            if local_rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            cur_c5 = approx_new_c5

            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
