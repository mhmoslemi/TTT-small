        idx = np.argpartition(corr, -top_k)[-top_k:]
        corr_use = corr[idx]

def smooth_obj_and_grad(h, alpha, top_k=0):
    n = len(h)
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k > 0 and top_k < n_corr:
        idx = np.argpartition(corr, -top_k)[-top_k:]
        corr_use = corr.copy()
        # Zero out non-selected for weight computation
        mask_sel = np.zeros(n_corr, dtype=bool)
        mask_sel[idx] = True
        corr_use = np.where(mask_sel, corr, -np.inf)
    else:
        corr_use = corr
        mask_sel = np.ones(n_corr, dtype=bool)
        idx = np.arange(n_corr)
    
    c_max = np.max(corr_use[mask_sel])
    shifted = np.clip(alpha * (corr_use - c_max), -500.0, 0.0)

def slsqp_step(h, n, target, alpha, maxiter=200, top_k=0):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_obj_val(x, alpha, top_k)
    
    def obj_grad(x):
        _, g = smooth_obj_and_grad(x, alpha, top_k)
        return g
    
    def cons_fun(x):
        return np.sum(x) - target
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': cons_fun}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})

def projected_gradient(h_init, n, target, alpha, time_limit, 
                       max_iter=50000, top_k=0, tol=1e-16):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    g_prev = None
    x_prev = None
    eta = 1.0 / (alpha * 2)  # initial step size
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        obj, grad = smooth_obj_and_grad(h, alpha, top_k)
        
        # Remove mean component (tangent to the equality constraint)
        grad_tan = grad - np.mean(grad)
        
        # BB step size
        if g_prev is not None:
            s = h - x_prev
            y = grad_tan - g_prev
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            if yy > 1e-16 and sy > 1e-16:
                eta_new = sy / yy
                eta = 0.5 * eta + 0.5 * min(eta_new, 5.0 * eta)
        
        # Projected gradient step
        g_norm = np.linalg.norm(grad_tan)
        if g_norm < 1e-14:
            break
        
        # Backtracking
        step = eta
        for _ in range(30):
            h_try = h - step * grad_tan
            h_try = project_feasible(h_try, target)
            # Check improvement
            h_try_orig = h_try.copy()
            obj_try, _ = smooth_obj_and_grad(h_try_orig, alpha, top_k)
            if obj_try < obj - 1e-4 * step * g_norm * g_norm:
                break
            step *= 0.5
        
        x_prev = h.copy()
        g_prev = grad_tan.copy()
        h = h_try
        
        c5_cur = compute_c5(h)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 100.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(h, alpha):
    n = len(h)
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - mid
        m1 = idxs - k
        m2 = idxs + k
        mask1 = (m1 >= 0) & (m1 < n)
        mask2 = (m2 >= 0) & (m2 < n)
        grad[mask1] += w * (1.0 - h[m1[mask1]])
        grad[mask2] -= w * h[m2[mask2]]
    return grad

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha=400):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        def _obj(x):
            return smooth_obj_val(x, alpha)
        def _grad(x):
            return smooth_obj_grad(x, alpha)
        def _cons(x):
            return np.sum(x) - target
        try:
            res = minimize(_obj, h_cur, jac=_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _cons}],
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def pair_cd(h_init, n, target, time_limit, max_sweeps=100000,
            n_cand_start=30, n_cand_end=80, top_k_eval=50,
            max_no_improve=500, n_shifts_score=2):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    corr = np.correlate(h, 1.0 - h, mode='full')

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        top_shifts = np.argsort(corr)[-n_shifts_score:]
        k_star_idx = int(top_shifts[-1])

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 4)

        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            m1 = idxs - k
            m2 = idxs + k
            mask1 = (m1 >= 0) & (m1 < n)
            mask2 = (m2 >= 0) & (m2 < n)
            grad[mask1] += (1.0 - h[m1[mask1]])
            grad[mask2] -= h[m2[mask2]]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval][:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-12
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            p1, p2 = best_i, best_j
            corr = np.correlate(h, 1.0 - h, mode='full')
            for p in [p1, p2]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5

def pg_optimize(h_init, n, target, alpha, time_limit, max_iter=100000):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    g_prev = None
    x_prev = None
    eta = 1.0 / max(alpha, 1.0)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        grad = smooth_obj_grad(h, alpha)
        grad_tan = grad - np.mean(grad)
        g_norm = np.linalg.norm(grad_tan)
        if g_norm < 1e-13:
            break
        if g_prev is not None:
            s_vec = h - x_prev
            y_vec = grad_tan - g_prev
            sy = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            if yy > 1e-16 and sy > 1e-16:
                eta = 0.5 * eta + 0.5 * min(sy / yy, 3.0 * eta)
        step = eta
        for _ in range(25):
            h_try = project_feasible(h - step * grad_tan, target)
            if smooth_obj_val(h_try, alpha) < smooth_obj_val(h, alpha) - 1e-5 * step * g_norm**2:
                break
            step *= 0.5
        x_prev = h.copy()
        g_prev = grad_tan.copy()
        h = h_try
        c5_cur = compute_c5(h)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    sc_hi = compute_c5(h_hi)
    if sc_hi < best_score:
        best_score = sc_hi
        best_h = h_hi
    else:
        best_h = h_hi

    tl = _tl()
    if tl > 200:
        h_cd, c5_cd = pair_cd(h_hi, n_hi, target_hi,
                              time_limit=min(500, tl * 0.65),
                              max_sweeps=200000,
                              n_cand_start=30, n_cand_end=90,
                              top_k_eval=60, max_no_improve=800,
                              n_shifts_score=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    tl = _tl()
    if tl > 80:
        h_pg, c5_pg = pg_optimize(best_h, n_hi, target_hi, 500,
                                  time_limit=min(150, tl * 0.6),
                                  max_iter=50000)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    tl = _tl()
    if tl > 60:
        h_pol = slsqp_optimize(best_h, n_hi, target_hi, [2000, 10000], 400)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    tl = _tl()
    if tl > 150:
        h_cd2, c5_cd2 = pair_cd(best_h, n_hi, target_hi,
                                time_limit=min(200, tl * 0.7),
                                max_sweeps=150000,
                                n_cand_start=40, n_cand_end=100,
                                top_k_eval=70, max_no_improve=600,
                                n_shifts_score=3)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    tl = _tl()
    if tl > 200:
        n_hi2 = 3200
        target_hi2 = n_hi2 / 2.0
        h_hi2 = interpolate_to(best_h, n_hi2, target_hi2)
        sc2 = compute_c5(h_hi2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_hi2
        else:
            best_h = h_hi2

        tl = _tl()
        if tl > 120:
            h_cd3, c5_cd3 = pair_cd(h_hi2, n_hi2, target_hi2,
                                    time_limit=min(180, tl * 0.7),
                                    max_sweeps=100000,
                                    n_cand_start=30, n_cand_end=80,
                                    top_k_eval=50, max_no_improve=500,
                                    n_shifts_score=2)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

        tl = _tl()
        if tl > 50:
            h_pol2 = slsqp_optimize(best_h, n_hi2, target_hi2, [5000, 20000], 300)
            sc2 = compute_c5(h_pol2)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_pol2

    tl = _tl()
    if tl > 40:
        rng = np.random.RandomState(seed)
        for ns in [0.05, 0.1, 0.15]:
            tl = _tl()
            if tl < 30:
                break
            hp = project_feasible(best_h + rng.randn(len(best_h)) * ns,
                                   len(best_h) / 2.0)
            h_ref = slsqp_optimize(hp, len(hp), len(hp) / 2.0,
                                   [1000, 5000], 300)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    tl = _tl()
    if tl > 30:
        n_out = len(best_h)
        h_final = slsqp_optimize(best_h, n_out, n_out / 2.0, [10000], 200)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_out = len(best_h)
    final_h = project_feasible(best_h, n_out / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, n_out
