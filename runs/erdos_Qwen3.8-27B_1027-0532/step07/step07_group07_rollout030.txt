    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)

            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    mid = n - 1
    
    # Find top shifts
    top_indices = np.argsort(corr)[::-1][:top_k]
    
    # Compute weighted subgradient (smooth max approximation)
    # Use softmax weighting
    shifted = corr[top_indices] - corr[top_indices].max()
    weights = np.exp(shifted * 100)  # sharp softmax
    weights /= weights.sum()
    
    g = np.zeros(n)

def subgradient_c5_vec(h, top_k=5, alpha=100):
    """Compute a smoothed subgradient of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    mid = n - 1
    
    top_indices = np.argsort(corr)[::-1][:top_k]
    c_max = corr[top_indices].max()
    shifted = alpha * (corr[top_indices] - c_max)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    g = np.zeros(n)
    for w, idx in zip(weights, top_indices):
        k = int(idx) - mid
        # Vectorized: ∂C_k/∂h = (1-h shifted by k) - (h shifted by -k)
        # With boundary: terms are 0 where index is out of range
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if k >= 0:
            term1[:n-k] = (1.0 - h[k:])
            term2[k:] = h[:n-k]
        else:
            term1[-k:] = (1.0 - h[:n+k])
            term2[:-k] = h[-k:]
        g += w * (term1 - term2)
    
    return g * dx, float(np.max(corr)) * dx

_g_target = 0.0

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14

            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(x, 1.0 - x, mode='full')

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

            best_h = h_new.copy()
            y = h_new.copy()
            no_improve = 0
            momentum = min(momentum + 0.005, 0.95)
        else:
            y = h_new + momentum * (h_new - h)
            y = project_feasible(y, target)
            h = h_new
            no_improve += 1
            if no_improve > 5000:
                eta *= 0.5
                no_improve = 0
        eta *= eta_decay
    return best_h, best_c5

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=80, top_k_eval=50,
            max_no_improve=600, n_shifts_grad=2):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        k_star_idx = int(np.argmax(corr))
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]
        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            g_vec = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(10):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g_vec
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g_vec[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g_vec) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)
    return best_h, best_c5

def block_cd_slsqp(h_init, n, target, time_limit, block_size=8, n_blocks_per_iter=20, max_iters=2000):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    rng = np.random.RandomState(12345)
    no_improve = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if no_improve > 200:
            break
        for _ in range(n_blocks_per_iter):
            start_idx = rng.randint(0, max(1, n - block_size))
            block = slice(start_idx, start_idx + block_size)
            h_block = h[block].copy()
            rest_sum = np.sum(h) - np.sum(h_block)
            block_target = target - rest_sum
            block_target = max(block_size * 0.1, min(block_size * 0.9, block_target))
            def block_obj(x_b):
                h_try = h.copy()
                h_try[block] = x_b
                h_try = project_feasible(h_try, target)
                return smooth_obj(h_try, 500)
            def block_grad(x_b):
                h_try = h.copy()
                h_try[block] = x_b
                h_try = project_feasible(h_try, target)
                full_g = smooth_obj_grad(h_try, 500)
                return full_g[block]
            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': lambda x_b, st=block_target: np.sum(x_b) - st}]
            try:
                res = minimize(block_obj, h_block, jac=block_grad, method='SLSQP',
                              bounds=bounds_b, constraints=cons_b,
                              options={'maxiter': 50, 'ftol': 1e-12})
                h_new = h.copy()
                h_new[block] = res.x
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < best_c5 - 1e-15:
                    best_c5 = sc
                    best_h = h_new.copy()
                    h = h_new
                    no_improve = 0
                else:
                    no_improve += 1
            except Exception:
                pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multishift_cd(h_init, n, target, time_limit, max_sweeps=30000,
                  n_cand=50, top_k_eval=40, n_shifts=3, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        top_shifts = np.argsort(corr)[-n_shifts:]
        k_star_idx = int(top_shifts[-1])
        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx]
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]
        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            g_vec = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(10):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g_vec
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g_vec[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g_vec) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)
    return best_h, best_c5

def stochastic_perturb(h_init, n, target, time_limit, max_steps=20000):
    start = time.time()
    rng = np.random.RandomState(999)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break
        scale = 0.05 * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.choice([3, 5, 8, 12, 16]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 300:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
