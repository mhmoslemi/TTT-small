```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def subgradient_method(h_init, n, target, time_limit, alpha0=0.2, alpha_min=0.005):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    stale = 0
    momentum = np.zeros(n)

    for step in range(200000):
        if time.time() - start > time_limit:
            break
        if alpha < alpha_min:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
            alpha = min(alpha * 1.05, alpha0)
        else:
            stale += 1
            alpha *= 0.97
            if stale > 800:
                break

        c_max = float(np.max(corr))
        threshold = c_max - 0.1 * c_max
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        n_act = len(active)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = 1.0 / n_act
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        # Momentum
        momentum = 0.9 * momentum - alpha * grad
        h_new = h + momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def vectorized_cd(h_init, n, target, time_limit, max_sweeps=100000,
                  n_cand=40, max_no_improve=400, n_active_shifts=4,
                  reheat_interval=100, reheat_scale=0.03, bisection_iters=12):
    start = time.time()

    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    sweeps_since_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if sweeps_since_improve > reheat_interval and sweeps_since_improve % 40 == 0:
            rng = np.random.RandomState(sweep * 11 + 3)
            n_pert = max(5, n // 40)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 1e-5:
                h = h_pert
                change = compute_change_matrix(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')

                sweeps_since_improve = 0
                continue

        top_k = np.argsort(corr)[-n_active_shifts:]
        k_star = int(np.argmax(corr))

        grad = np.zeros(n)
        n_act = len(top_k)

            v2 = (idxs + k >= 0) & (idxs + k < n)
            w = 1.0 / n_act
            grad[v1] += w * (1.0 - h[idxs[v1] - k])

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand * (0.5 + 0.5 * progress))

                valid_mask = valid
            else:
                valid_mask = valid

            idx_valid = np.where(valid_mask)[0]

            a_arr = delta_lo[idx_valid].copy()
            b_arr = delta_hi[idx_valid].copy()

            for _ in range(bisection_iters):
                mid_arr = 0.5 * (a_arr + b_arr)

                f_mid = np.max(vals, axis=1)
                active = vals >= (f_mid[:, None] - 1e-14)
                f_prime = np.max(g_valid * active, axis=1)
                incr = f_prime > 1e-12
                a_arr = np.where(incr, mid_arr, a_arr)

            delta_star = 0.5 * (a_arr + b_arr)

            min_idx = int(np.argmin(approx_max))

                best_j = int(low_set[idx_valid[min_idx]])

            update_change_row(change, best_i, h, n, shifts)

def subgradient_method(h_init, n, target, time_limit, alpha0=0.15, alpha_min=0.005):

            break
        if alpha < alpha_min:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stale = 0
            alpha = min(alpha * 1.05, alpha0)
        else:
            stale += 1
            alpha *= 0.97
            if stale > 600:
                break

        c_max = float(np.max(corr))
        threshold = c_max - 0.08 * max(c_max, 0.01)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])

        grad = np.zeros(n)
        n_act = len(active)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)

            grad[v1] += w * (1.0 - h[idxs[v1] - k])

        h_new = h + momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                [50, 200, 1000, 5000, 20000, 80000],
                                1200, 1e-15, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    Phase 2: Vectorized CD at n=400
    if tleft() > 120:
        h_cd, c5_cd = vectorized_cd(best_h, current_n, current_target,
                                    time_limit=min(70, tleft() * 0.45),
                                    max_sweeps=80000,
                                    n_cand=45, max_no_improve=350,
                                    n_active_shifts=4,
                                    reheat_interval=80, reheat_scale=0.03,
                                    bisection_iters=12)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    Phase 3: SLSQP polish at n=400
    if tleft() > 50:
        deadline = time.time() + min(35, tleft() * 0.4)
        h_pol2, sc2 = slsqp_round(best_h, current_n, current_target,
                                  [10000, 50000, 200000],
                                  1500, 1e-15, deadline=deadline)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    Phase 4: Interpolate to n=600 and optimize
    if tleft() > 250:
        n1 = 600
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)

        if tleft() > 150:
            deadline = time.time() + min(40, tleft() * 0.3)
            h1_pol, sc1p = slsqp_round(h1, n1, target1,
                                       [200, 1000, 5000, 20000],
                                       1000, 1e-15, deadline=deadline)
            if sc1p < compute_c5(h1):
                h1 = h1_pol

        if tleft() > 120:
            h_cd1, c5_cd1 = vectorized_cd(h1, n1, target1,
                                           time_limit=min(80, tleft() * 0.4),
                                           max_sweeps=40000,
                                           n_cand=35, max_no_improve=250,
                                           n_active_shifts=4,
                                           reheat_interval=70, reheat_scale=0.03)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1
                current_n = n1
                current_target = target1
            elif c5_cd1 < compute_c5(h1):
                best_h = h_cd1
                current_n = n1
                current_target = target1

        if tleft() > 50:
            deadline = time.time() + min(30, tleft() * 0.4)
            h1_pol2, sc1p2 = slsqp_round(best_h, current_n, current_target,
                                         [5000, 20000, 80000],
                                         1000, 1e-15, deadline=deadline)
            if sc1p2 < best_score:
                best_score = sc1p2
                best_h = h1_pol2

    Phase 5: Interpolate to n=800 and optimize
    if tleft() > 200:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        if tleft() > 120:
            deadline = time.time() + min(35, tleft() * 0.3)
            h2_pol, sc2p = slsqp_round(h2, n2, target2,
                                       [500, 2000, 10000],
                                       800, 1e-15, deadline=deadline)
            if sc2p < compute_c5(h2):
                h2 = h2_pol

        if tleft() > 100:
            h_cd2, c5_cd2 = vectorized_cd(h2, n2, target2,
                                           time_limit=min(65, tleft() * 0.4),
                                           max_sweeps=25000,
                                           n_cand=30, max_no_improve=200,
                                           n_active_shifts=4,
                                           reheat_interval=60, reheat_scale=0.025)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                current_n = n2
                current_target = target2
            elif c5_cd2 < compute_c5(h2):
                best_h = h_cd2
                current_n = n2
                current_target = target2

        if tleft() > 50:
            deadline = time.time() + min(30, tleft() * 0.4)
            h2_pol2, sc2p2 = slsqp_round(best_h, current_n, current_target,
                                         [5000, 20000, 80000],
                                         1000, 1e-15, deadline=deadline)
            if sc2p2 < best_score:
                best_score = sc2p2
                best_h = h2_pol2

    Phase 6: Multi-start perturbation at current resolution
    if tleft() > 120:
        scales = [0.02, 0.04, 0.07, 0.10, 0.14, 0.20]
        for attempt in range(len(scales)):
            if tleft() < 35:
                break
            noise_scale = scales[attempt]
            h_pert = best_h + rng.randn(current_n) * noise_scale
            h_pert = project_feasible(h_pert, current_target)
            deadline = time.time() + min(18, tleft() * 0.25)
            h_ref, sc_ref = slsqp_round(h_pert, current_n, current_target,
                                        [500, 2000, 10000],
                                        700, 1e-15, deadline=deadline)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    Phase 7: Subgradient method
    if tleft() > 60:
        h_sg, c5_sg = subgradient_method(best_h, current_n, current_target,
                                          time_limit=min(35, tleft() * 0.5),
                                          alpha0=0.12, alpha_min=0.008)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    Phase 8: Final CD
    if tleft() > 50:
        h_cd_f, c5_f = vectorized_cd(best_h, current_n, current_target,
                                      time_limit=min(30, tleft() * 0.5),
                                      max_sweeps=20000,
                                      n_cand=40, max_no_improve=200,
                                      n_active_shifts=5,
                                      reheat_interval=50, reheat_scale=0.02)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_cd_f

    Phase 9: Final SLSQP
    if tleft() > 25:
        deadline = time.time() + min(15, tleft() * 0.6)
        h_final, sc_final = slsqp_round(best_h, current_n, current_target,
                                         [10000, 50000],
                                         1000, 1e-15, deadline=deadline)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
