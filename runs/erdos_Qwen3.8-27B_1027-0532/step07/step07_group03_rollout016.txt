def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break

            mask = h < 1.0 - 1e-14

            h = np.clip(h + step, 0.0, 1.0)

        h = np.clip(h, 0.0, 1.0)

    mid = n - 1
    k = k_star - mid

    idxs = np.arange(n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:

def corr_grad_multi(h, k_stars, weights):
    n = len(h)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for ks, w in zip(k_stars, weights):
        k = int(ks) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

    F = []

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_averaging(h_init, n, target, time_limit, max_iters=5000,
                          T0=100.0, cooling=0.999):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h_avg = h_init.copy().astype(float)
    h_curr = h_init.copy().astype(float)
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)
    T = T0
    count = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h_curr)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h_curr.copy()

        k_idx = int(np.argmax(corr))
        k = k_idx - mid
        g = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        g[v1] += (1.0 - h_curr[idxs[v1] - k])
        g[v2] -= h_curr[idxs[v2] + k]
        g -= np.mean(g)

        count += 1
        step = T / (count ** 0.7)
        step = min(step, 0.1)
        h_new = h_curr - step * g
        h_new = project_feasible(h_new, target)

        h_avg = (h_avg * count + h_new) / (count + 1)
        h_curr = h_new
        T *= cooling

    sc_avg = compute_c5(h_avg)
    if sc_avg < best_score:
        best_score = sc_avg
        best_h = h_avg.copy()

    return best_h, best_score

        h_s2, sc = slsqp_round(best_h, n0, target0,
                                [5000, 20000], 600, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_s2

    if tleft() > 70:
        rng = np.random.RandomState(seed)
        for ns in [0.03, 0.08]:
            if tleft() < 50:
                break
            h_p = best_h + rng.randn(n0) * ns
            h_p = project_feasible(h_p, target0)
            h_ref, sc_ref = cutting_plane_v2(h_p, n0, target0,
                                              time_limit=min(30, tleft() * 0.3),
                                              max_iters=2000, tol=1e-9)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 50:
        h_cd3, sc = fast_cd_v2(best_h, n0, target0,
                                time_limit=min(45, tleft() * 0.5),
                                max_sweeps=10000,
                                n_cand_start=40, n_cand_end=80,
                                max_no_improve=400)
        if sc < best_score:
            best_score = sc
            best_h = h_cd3

    if tleft() > 35:
        h_s3, sc = slsqp_round(best_h, n0, target0,
                                [10000, 50000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_s3

    if tleft() > 25:
        h_sg, sc = subgradient_averaging(best_h, n0, target0,
                                          time_limit=min(20, tleft() * 0.5),
                                          max_iters=3000, T0=50.0)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
