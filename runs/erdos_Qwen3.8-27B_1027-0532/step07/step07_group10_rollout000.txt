    h_bar = h.copy()  # stability center
    best_h = h.copy()
    best_c5 = compute_c5(h)

        corr = np.correlate(h, 1.0 - h, mode='full') * dx

            best_h = h.copy()

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [

            h_var >= 0,
            h_var <= 1
        ]

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_grad_single(h, k, n):
    """Compute gradient of f_k with respect to h."""
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g * (2.0 / n)


def smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = (idxs + k >= 0) & (idxs + k < n)
        wm = (idxs - k >= 0) & (idxs - k < n)
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, jac=smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def trust_region_method(h_init, n, target, time_limit, max_iter=50, 
                        mu_start=0.5, mu_end=0.01, n_active_max=30):
    """Bundle-type trust region method for min-max problem."""
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy()
    h_stab = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    mu = mu_start
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 15:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = np.max(corr)
        
        if c_max < best_c5 - 1e-12:
            best_c5 = c_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active shifts
        threshold = c_max - max(0.002, 0.01 * c_max)
        active_indices = np.where(corr >= threshold)[0]
        
        # Limit active set size
        if len(active_indices) > n_active_max:
            top_vals = corr[active_indices]
            order = np.argsort(top_vals)[::-1][:n_active_max]
            active_indices = active_indices[order]
        
        # Build CVXPY problem
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        cons_list = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1,
        ]
        
        for ki in active_indices:
            k = int(ki) - mid
            g = compute_grad_single(h, k, n)
            f_val = corr[int(ki)]
            # Linearization: f_k(h) ≈ f_k(h_cur) + g^T (h_new - h_cur)
            # Constraint: f_k(h_cur) + g^T (h_var - h) <= t_var
            cons_list.append(f_val + g @ (h_var - h) <= t_var)
        
        obj = cp.Minimize(t_var + (mu / 2.0) * cp.sum_squares(h_var - h_stab))
        prob = cp.Problem(obj, cons_list)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = project_feasible(np.array(h_var.value).flatten(), target)
        c5_new = compute_c5(h_new)
        
        if c5_new < c_max - 1e-10:
            h = h_new
            h_stab = h_new
            mu = max(mu * 0.6, mu_end)
            no_improve = 0
        else:
            h_stab = 0.7 * h_stab + 0.3 * h_new
            mu = min(mu * 1.5, 50.0)
    
    return best_h, best_c5


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change
