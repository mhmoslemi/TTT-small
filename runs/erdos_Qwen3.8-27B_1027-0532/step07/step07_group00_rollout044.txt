import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, alpha, top_k=50):
    n = len(h)
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    probs = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    top_idx = np.argsort(probs)[-top_k:]
    for s_idx in top_idx:
        w = probs[s_idx]
        if w < 1e-12:
            continue
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    smooth_val = c_max + np.log(np.sum(exp_s)) / alpha
    return smooth_val, grad, corr

def heavy_ball_opt(h_init, n, target, time_limit, alpha_init=30.0, alpha_end=3000.0,
                   max_iter=30000, step_init=0.003, momentum_init=0.85):
    start = time.time()
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    momentum = momentum_init
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break
        progress = min(float(it) / max(max_iter, 1), 1.0)
        alpha = alpha_init * (alpha_end / alpha_init) ** progress
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)
        _, grad, _ = smooth_obj_grad(y, alpha, top_k=60)
        grad -= np.mean(grad)
        h_prev = h.copy()
        h_new = y - step * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
            step = min(step * 1.02, 0.03)
        else:
            no_improve += 1
            step *= 0.97
            h_new = h
        h = h_new
        momentum = min(momentum + 0.0001, 0.95)
    return best_h, best_c5

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def slsqp_polish(h_init, n, target, alphas, maxiter_per_alpha=400, ftol=1e-14):
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_mutation(h_init, n, target, n_blocks=20):
    rng = np.random.RandomState(42)
    h = h_init.copy()
    block_size = n // n_blocks
    for _ in range(n_blocks // 2):
        b1 = rng.randint(0, n_blocks)
        b2 = rng.randint(0, n_blocks)
        if b1 == b2:
            continue
        s1, e1 = b1 * block_size, (b1 + 1) * block_size
        s2, e2 = b2 * block_size, (b2 + 1) * block_size
        h[s1:e1], h[s2:e2] = h[s2:e2].copy(), h[s1:e1].copy()
    return project_feasible(h, target)

        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.10]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, target_cur)
            h_hb2, sc2 = heavy_ball_opt(h_pert, n_cur, target_cur,
                                        time_limit=min(40, tleft() * 0.3),
                                        alpha_init=50, alpha_end=2000,
                                        max_iter=8000, step_init=0.004)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_hb2
            if tleft() > 60:
                h_cd2, sc_cd2 = pair_cd_enhanced(best_h, n_cur, target_cur,
                                                 time_limit=min(40, tleft() * 0.3),
                                                 max_sweeps=15000,
                                                 n_cand_start=25, n_cand_end=50,
                                                 max_no_improve=200,
                                                 weight_beta=5.0)
                if sc_cd2 < best_score:
                    best_score = sc_cd2
                    best_h = h_cd2
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for _ in range(3):
            if tleft() < 60:
                break
            h_mut = block_mutation(best_h, n_cur, target_cur, n_blocks=20)
            h_hb3, sc3 = heavy_ball_opt(h_mut, n_cur, target_cur,
                                        time_limit=min(30, tleft() * 0.3),
                                        alpha_init=50, alpha_end=2000,
                                        max_iter=6000, step_init=0.004)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_hb3
            if tleft() > 40:
                h_cd3, sc_cd3 = pair_cd_enhanced(best_h, n_cur, target_cur,
                                                 time_limit=min(30, tleft() * 0.3),
                                                 max_sweeps=10000,
                                                 n_cand_start=25, n_cand_end=50,
                                                 max_no_improve=150,
                                                 weight_beta=5.0)
                if sc_cd3 < best_score:
                    best_score = sc_cd3
                    best_h = h_cd3

    if tleft() > 250 and len(best_h) < 1200:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 200:
            h_hb4, sc4 = heavy_ball_opt(h2, n2, target2,
                                        time_limit=min(100, tleft() * 0.4),
                                        alpha_init=30, alpha_end=2000,
                                        max_iter=20000, step_init=0.003)
            if tleft() > 100:
                h_cd4, sc_cd4 = pair_cd_enhanced(h_hb4, n2, target2,
                                                 time_limit=min(100, tleft() * 0.4),
                                                 max_sweeps=20000,
                                                 n_cand_start=20, n_cand_end=50,
                                                 max_no_improve=250,
                                                 weight_beta=5.0)
                if sc_cd4 < best_score:
                    best_score = sc_cd4
                    best_h = h_cd4
                if tleft() > 50:
                    h_pol4, sc_pol4 = slsqp_polish(best_h, n2, target2,
                                                   [2000, 10000], maxiter_per_alpha=300)
                    if sc_pol4 < best_score:
                        best_score = sc_pol4
                        best_h = h_pol4

    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin_cd, sc_fin_cd = pair_cd_enhanced(best_h, n_cur, target_cur,
                                               time_limit=min(50, tleft() * 0.6),
                                               max_sweeps=10000,
                                               n_cand_start=30, n_cand_end=70,
                                               max_no_improve=150,
                                               weight_beta=8.0)
        if sc_fin_cd < best_score:
            best_score = sc_fin_cd
            best_h = h_fin_cd

    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final, sc_final = slsqp_polish(best_h, n_cur, target_cur,
                                         [10000, 50000], maxiter_per_alpha=300)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
