        elif consecutive_no_improve % 100 == 0 and consecutive_no_improve > 100:
            idxs_pert = rng.choice(n, size=max(1, n // 10), replace=False)
            h[idxs_pert] += rng.randn(len(idxs_pert)) * 0.03
            h = project_feasible(h, target)
            for p in idxs_pert:
                update_change_row(change, int(p), h, n, shifts)
    return best_h, best_c5

def nesterov_subgrad(h_init, n, target, time_limit, max_iter=500000,
                     step_init=0.01, momentum=0.9, seed=77):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    rng = np.random.RandomState(seed)
    h = h_init.copy().astype(float)
    y = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = step_init
    no_improve = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it % 100 == 0 and no_improve > 5000:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr)
        active = corr >= c_max - 1e-8 * max(1.0, c_max)
        n_active = int(np.sum(active))
        if n_active == 0:
            active = np.zeros(n_corr, dtype=bool)
            active[int(np.argmax(corr))] = True
        grad = np.zeros(n)
        for s in range(n_corr):
            if not active[s]:
                continue
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - y[idxs[v1] - k])
            grad[v2] -= y[idxs[v2] + k]
        grad /= n_active
        grad -= np.mean(grad)
        y_new = y - step * grad
        y_new = project_feasible(y_new, target)
        h_new = momentum * y_new + (1.0 - momentum) * h
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
            step = min(step * 1.05, 0.1)
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                step *= 0.95
        h, y = h_new, y_new
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
