import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_O_circ(h):
    n = len(h)
    H = np.fft.fft(h)
    ac = np.real(np.fft.ifft(H * np.conj(H)))
    return n / 2.0 - ac

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _g_n_global():
    return _gn_val

_gn_val = 0
_target_val = 0.0

def _cons_fun(x):
    return np.sum(x) - _target_val

def _cons_jac(x):
    return np.ones(_gn_val)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time):
    global _gn_val, _target_val
    _gn_val = n
    _target_val = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_lse, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def subgrad_phase(h_init, n, target, time_limit, start_time):
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_actual = compute_c5(h)
    h_prev = h.copy()
    step = 0.008
    no_improve = 0
    t = 0
    max_iters = 500000
    top_k = 3

    while t < max_iters:
        if time.time() - start_time > time_limit:
            break
        O = compute_O_circ(h)
        O_max = float(np.max(O))
        if O_max < np.max(compute_O_circ(best_h)) - 1e-12:
            no_improve = 0
            best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 8000:
                break
        top_shifts = np.argsort(O)[-top_k:]
        shifts_diff = O[top_shifts] - O_max
        weights = np.exp(shifts_diff * 20.0)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.ones(top_k) / top_k
        else:
            weights /= wsum
        grad = np.zeros(n)
        for w_k, k_idx in zip(weights, top_shifts):
            k = int(k_idx)
            if k > n // 2:
                k -= n
            grad += w_k * (1.0 - np.roll(h, k) - np.roll(h, -k))
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        beta = (t + 1.0) / (t + 3.0)
        h_y = h + beta * (h - h_prev)
        h_new = h_y - step * (grad / gnorm) * 0.3
        h_new = project_feasible(h_new, target)
        h_prev = h.copy()
        h = h_new
        t += 1
        if t % 500 == 0:
            step *= 0.995
            actual = compute_c5(h)
            if actual < best_actual - 1e-14:
                best_actual = actual
                best_h = h.copy()
    actual_best = compute_c5(best_h)
    if actual_best < best_actual:
        best_actual = actual_best
    return best_h, best_actual

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def pair_cd(h_init, n, target, time_limit, start_time,
            max_sweeps=20000, n_cand_start=15, n_cand_end=55,
            max_no_improve=300, n_top_shifts=3, bisection_iters=16):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_shifts].astype(float)
            wsum = weights.sum()
            if wsum < 1e-15:
                weights = np.ones(n_top_shifts) / n_top_shifts
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def stochastic_refine(h_init, n, target, time_limit, start_time, max_steps=30000, base_scale=0.02):
    rng = np.random.RandomState(42)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 5000:
            break
        scale = base_scale * max(0.03, 1.0 - step / float(max_steps))
        n_pert = int(rng.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        perm = rng.permutation(n)
        idxs = perm[:n_pert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 100:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0
    return best_h, best_c5

            best_h = h2

    remaining = budget_s - 60.0 - (time.time() - start_time)
    if remaining > 100:
        h3, sc3 = slsqp_round(best_h, n0, target0,
                              [500, 2000, 8000], 1000,
                              start_time + min(100, remaining - 20), start_time)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    remaining = budget_s - 60.0 - (time.time() - start_time)
    if remaining > 300:
        n1 = 2 * n0
        target1 = n1 / 2.0
        h_int = interpolate_to(best_h, n1, target1)
        h4, sc4 = pair_cd(h_int, n1, target1,
                          start_time + min(300, remaining - 30), start_time,
                          max_sweeps=30000, n_cand_start=12, n_cand_end=45,
                          max_no_improve=350, n_top_shifts=3, bisection_iters=14)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4
            n0 = n1
            target0 = target1

    remaining = budget_s - 60.0 - (time.time() - start_time)
    if remaining > 150:
        h5, sc5 = pair_cd(best_h, n0, target0,
                          start_time + min(150, remaining - 20), start_time,
                          max_sweeps=20000, n_cand_start=20, n_cand_end=60,
                          max_no_improve=300, n_top_shifts=5, bisection_iters=18)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    remaining = budget_s - 60.0 - (time.time() - start_time)
    if remaining > 80:
        h6, sc6 = slsqp_round(best_h, n0, target0,
                              [5000, 20000], 800,
                              start_time + min(80, remaining - 15), start_time)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    remaining = budget_s - 60.0 - (time.time() - start_time)
    if remaining > 100:
        h7, sc7 = stochastic_refine(best_h, n0, target0,
                                    start_time + min(100, remaining - 20), start_time,
                                    max_steps=30000, base_scale=0.015)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    remaining = budget_s - 60.0 - (time.time() - start_time)
    if remaining > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05]:
            remaining = budget_s - 60.0 - (time.time() - start_time)
            if remaining < 60:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_r, sc_r = pair_cd(h_pert, n0, target0,
                                start_time + min(60, remaining - 20), start_time,
                                max_sweeps=10000, n_cand_start=15, n_cand_end=40,
                                max_no_improve=200, n_top_shifts=3, bisection_iters=16)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r

    remaining = budget_s - 60.0 - (time.time() - start_time)
    if remaining > 60:
        h_fin, sc_fin = pair_cd(best_h, n0, target0,
                                start_time + min(60, remaining - 10), start_time,
                                max_sweeps=15000, n_cand_start=25, n_cand_end=70,
                                max_no_improve=150, n_top_shifts=1, bisection_iters=20)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 40:
        h_final, sc_final = slsqp_round(best_h, n0, target0,
                                        [10000, 50000], 600,
                                        start_time + min(40, remaining - 5), start_time)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
