def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():

            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[k_idx] * (1.0 - x[idxs[v1] - k])
        grad[v2] -= weights[k_idx] * x[idxs[v2] + k]
    return grad

    return np.sum(x) - _g_target

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def balancing_gradient_descent(h_init, n, target, time_limit, n_active=8,

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[0]]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        return max(0.0, budget_s - 25.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, best_n, best_target,
                                [200, 1000, 5000], 800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        h_cd, c5_cd = vectorized_cd(best_h, best_n, best_target,
                                    time_limit=min(80, tleft() * 0.5),
                                    n_cand=40, max_sweeps=20000,
                                    n_deltas=5, max_no_improve=400)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_bg, c5_bg = balancing_gradient_descent(best_h, best_n, best_target,
                                                  time_limit=min(40, tleft() * 0.5),
                                                  n_active=6, base_lr=0.02,
                                                  max_steps=30000)
        if c5_bg < best_score:
            best_score = c5_bg
            best_h = h_bg

    if tleft() > 50:
        h_pol2, sc2 = slsqp_round(best_h, best_n, best_target,
                                  [10000, 50000], 1500, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    n1 = 800
    if tleft() > 300:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 250:
            h1_pol, sc1p = slsqp_round(h1, n1, target1,
                                       [200, 1000, 5000], 800, 1e-15)
            if sc1p < sc1:
                h1 = h1_pol
                sc1 = sc1p

        if tleft() > 200:
            h1_cd, c5_1cd = vectorized_cd(h1, n1, target1,
                                          time_limit=min(150, tleft() * 0.5),
                                          n_cand=45, max_sweeps=30000,
                                          n_deltas=5, max_no_improve=500)
            if c5_1cd < best_score:
                best_score = c5_1cd
                best_h = h1_cd
                best_n = n1
                best_target = target1
            else:
                h1 = h1_cd if c5_1cd < sc1 else h1

        if tleft() > 100:
            h1_ms, c5_1ms = multi_shift_cd(h1 if c5_1cd < sc1 else h1,
                                           n1, target1,
                                           time_limit=min(80, tleft() * 0.4),
                                           n_cand=40, n_active=5,
                                           max_sweeps=15000, max_no_improve=300)
            if c5_1ms < best_score:
                best_score = c5_1ms
                best_h = h1_ms
                best_n = n1
                best_target = target1

        if tleft() > 80:
            h1_bg, c5_1bg = balancing_gradient_descent(
                h1 if c5_1cd < sc1 else h1, n1, target1,
                time_limit=min(50, tleft() * 0.4),
                n_active=8, base_lr=0.015, max_steps=30000)
            if c5_1bg < best_score:
                best_score = c5_1bg
                best_h = h1_bg
                best_n = n1
                best_target = target1

        if tleft() > 80:
            h1_pol2, sc1p2 = slsqp_round(best_h if best_n == n1 else h1,
                                         n1, target1,
                                         [10000, 50000, 100000], 2000, 1e-15)
            if sc1p2 < best_score:
                best_score = sc1p2
                best_h = h1_pol2
                best_n = n1
                best_target = target1

    n2 = 1200
    if tleft() > 350:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 280:
            h2_pol, sc2p = slsqp_round(h2, n2, target2,
                                       [200, 1000, 3000], 600, 1e-15)
            if sc2p < sc2:
                h2 = h2_pol
                sc2 = sc2p

        if tleft() > 200:
            h2_cd, c5_2cd = vectorized_cd(h2, n2, target2,
                                          time_limit=min(160, tleft() * 0.5),
                                          n_cand=35, max_sweeps=25000,
                                          n_deltas=5, max_no_improve=400)
            if c5_2cd < best_score:
                best_score = c5_2cd
                best_h = h2_cd
                best_n = n2
                best_target = target2

        if tleft() > 120:
            h2_ms, c5_2ms = multi_shift_cd(h2 if c5_2cd < sc2 else h2,
                                           n2, target2,
                                           time_limit=min(80, tleft() * 0.4),
                                           n_cand=35, n_active=4,
                                           max_sweeps=10000, max_no_improve=300)
            if c5_2ms < best_score:
                best_score = c5_2ms
                best_h = h2_ms
                best_n = n2
                best_target = target2

        if tleft() > 80:
            h2_bg, c5_2bg = balancing_gradient_descent(
                h2 if c5_2cd < sc2 else h2, n2, target2,
                time_limit=min(50, tleft() * 0.4),
                n_active=10, base_lr=0.01, max_steps=30000)
            if c5_2bg < best_score:
                best_score = c5_2bg
                best_h = h2_bg
                best_n = n2
                best_target = target2

        if tleft() > 80:
            h2_pol2, sc2p2 = slsqp_round(best_h if best_n == n2 else h2,
                                         n2, target2,
                                         [10000, 50000], 1500, 1e-15)
            if sc2p2 < best_score:
                best_score = sc2p2
                best_h = h2_pol2
                best_n = n2
                best_target = target2

    if tleft() > 100:
        h_final_cd, c5_final = vectorized_cd(best_h, best_n, best_target,
                                             time_limit=min(80, tleft() * 0.5),
                                             n_cand=50, max_sweeps=20000,
                                             n_deltas=6, max_no_improve=300)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if tleft() > 80:
        h_final_pol, sc_final = slsqp_round(best_h, best_n, best_target,
                                            [20000, 100000, 200000], 2000, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    if tleft() > 60:
        h_final_bg, c5_fb = balancing_gradient_descent(best_h, best_n, best_target,
                                                       time_limit=min(40, tleft() * 0.5),
                                                       n_active=12, base_lr=0.005,
                                                       max_steps=20000)
        if c5_fb < best_score:
            best_score = c5_fb
            best_h = h_final_bg

    if tleft() > 40:
        h_last_cd, c5_last = vectorized_cd(best_h, best_n, best_target,
                                           time_limit=min(20, tleft() * 0.5),
                                           n_cand=60, max_sweeps=5000,
                                           n_deltas=5, max_no_improve=100)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_last_cd

    if tleft() > 25:
        h_last_pol, sc_last = slsqp_round(best_h, best_n, best_target,
                                          [50000, 200000], 1000, 1e-15)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last_pol

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
