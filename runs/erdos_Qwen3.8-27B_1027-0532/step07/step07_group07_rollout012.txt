```python
def project_feasible(h, target_sum):
```

```python
def compute_smooth_grad(y, alpha, n, dx, n_corr, shifts):
    corr_y = np.correlate(y, 1.0 - y, mode='full')
    scaled = alpha * corr_y * dx
    scaled -= np.max(scaled)
    weights = np.exp(scaled)
    weights /= np.sum(weights)
    
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - y[idxs[v1] - k])
        grad[v2] -= w * y[idxs[v2] + k]
    return grad
```
