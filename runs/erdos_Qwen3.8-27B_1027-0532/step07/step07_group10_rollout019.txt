```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros(len(x) * 2 - 1)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = idxs + k < n
        wm = idxs - k >= 0
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)
```

```python
def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15, time_limit=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    t0 = time.time()
    for i, alpha in enumerate(alphas):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        _g_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, jac=smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def momentum_gd(h_init, n, target, time_limit, base_lr=0.01, mu=0.85,
                n_active=10, temp=15.0, max_steps=100000):
    """Active-set gradient descent with heavy-ball momentum."""
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = base_lr
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 8000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                lr *= 0.9
                if lr < 1e-8:
                    break
        
        # Softmax weights for active shifts
        c_max = float(np.max(corr))
        # Use temperature to control width of active set
        exp_vals = np.exp(-temp * (c_max - corr))
        wsum = np.sum(exp_vals)
        if wsum < 1e-300:
            weights = np.zeros(n_corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights = exp_vals / wsum
        
        # Compute gradient of weighted overlap
        grad = np.zeros(n)
        # Only process shifts with non-negligible weight
        active = np.where(weights > 1e-10)[0]
        for si in active:
            k = int(si) - mid
            w = weights[si]
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] + k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Heavy-ball momentum update
        v = mu * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < c5_val - 1e-15:
            h = h_new
        else:
            # Try smaller step
            h_try = h + 0.5 * v
            h_try = project_feasible(h_try, target)
            sc_try = compute_c5(h_try)
            if sc_try < c5_val - 1e-15:
                h = h_try
                v *= 0.5
            else:
                v *= mu  # dampen momentum
                lr *= 0.8
                if lr < 1e-10:
                    break
    
    return best_h, best_c5
```

```python
def fast_cd(h_init, n, target, time_limit, n_cand=50, max_sweeps=20000,
            max_no_improve=400):
    """Pair coordinate descent with exact line search."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        # Gradient for active shift
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        # Try pair swaps
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                
                g = change[j] - change[i]
                
                # Binary search for optimal delta
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                
                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-15:
                    best_trial_c5 = float(approx_max) * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    return best_h, best_c5
```

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))

def smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)

    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum
    n = len(x)

    idxs = np.arange(n)

        wm = (idxs - k) >= 0

        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
