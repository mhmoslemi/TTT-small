import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    grad = np.zeros(n)

            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

    """Frank-Wolfe step: compute subgradient and find s"""
    mid = n - 1
    idxs = np.arange(n)
    
    # Get top-k lags
    top_idx = np.argsort(corr)[-top_k:]
    cur_max = float(np.max(corr))
    # Soft weights over top-k
    alpha_w = 30.0
    w = np.exp(alpha_w * (corr[top_idx] - cur_max))
    w = w / w.sum()
    
    # Compute subgradient
    grad = np.zeros(n)
    for wi, si in zip(w, top_idx):
        k = int(si) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        if np.any(v1):
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        if np.any(v2):
            grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    
    # Frank-Wolfe: minimize s·grad subject to s∈[0,1]^n, sum(s)=target
    # This gives: s_i = 1 if grad_i < λ, 0 if grad_i > λ

    s = (grad < lam).astype(float)

    diff = target - s.sum()
    if abs(diff) > 1e-10:

        if len(boundary) == 0:

            dists = np.abs(grad - lam)
            boundary = np.argsort(dists)[:int(abs(diff)) + 1]
        if diff > 0:
            s[boundary[:int(diff)]] = 1.0
        elif diff < 0:
            s[boundary[:int(-diff)]] = 0.0
    return s, grad

                h = h_try
                improved = True
                break
        if not improved and step > 2000:

    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        top_idx = np.argsort(corr)[-top_lags:]
        alpha_w = 40.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)

        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        active_lags = top_idx
        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (lo + hi_d)
                    vals = corr[active_lags] + mid_d * g[active_lags]
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-12)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active_lags][active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr[active_lags] + delta_star * g[active_lags]))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def frank_wolfe_phase(h_init, n, target, time_limit, max_steps=50000, top_k=5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break

        top_idx = np.argsort(corr)[-top_k:]
        cur_max = float(np.max(corr))
        alpha_w = 25.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx

        sorted_grad = np.sort(grad)
        lam = float(sorted_grad[int(target - 1)])
        s = (grad < lam).astype(float)
        diff_s = target - s.sum()
        if abs(diff_s) > 1e-10:
            dists = np.abs(grad - lam)
            boundary = np.argsort(dists)[:int(abs(diff_s)) + 1]
            if diff_s > 0:
                s[boundary[:int(np.ceil(diff_s))]] = 1.0
            else:
                s[boundary[:int(np.ceil(-diff_s))]] = 0.0

        improved = False
        decay = 1.0 / (1.0 + 0.002 * step)
        for t_mult in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
            t = t_mult * decay
            h_try = (1.0 - t) * h + t * s
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved and step > 5000:
            break

    return best_h, best_c5

def perturb_refine(h_best, n, target, time_limit, seed, n_trials=5):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    bounds = [(0.0, 1.0)] * n

    for trial in range(n_trials):
        if time.time() - start > time_limit:
            break
        scale = 0.03 / (1.0 + 0.5 * trial)
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        for alpha in [5000, 50000]:
            if time.time() - start > time_limit:
                break
            try:
                res = minimize(lse_objective, h_pert, args=(alpha, target, 100.0),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 150, 'ftol': 1e-16, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_pert = h_new
            except Exception:
                pass

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def waterfill_phase(h_init, n, target, time_limit, max_sweeps=20000, top_lags=5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        top_idx = np.argsort(corr)[-top_lags:]
        top_corr = corr[top_idx]
        mean_top = float(np.mean(top_corr))
        spread = float(np.max(top_corr) - np.min(top_corr))
        if spread < 1e-10:
            break

        alpha_w = 100.0
        w = np.exp(alpha_w * (top_corr - float(np.max(top_corr))))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx

        decay = 1.0 / (1.0 + 0.003 * sweep)
        improved = False
        for lr_mult in [0.05, 0.02, 0.01, 0.005, 0.001]:
            lr = lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val - 1e-15:
                h = h_try
                improved = True
                break

        if not improved and sweep > 3000:
            break

    return best_h, best_c5
