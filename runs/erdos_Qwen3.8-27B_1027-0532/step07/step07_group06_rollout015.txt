def build_gradient_matrix(h, n):
    # Returns A of shape (2n-1, n) where A[m, :] is the gradient of f_k with k = m-(n-1)
    A = np.zeros((2*n-1, n))
    k_shifts = np.arange(-(n-1), n)
    for m, k in enumerate(k_shifts):
        if k >= 0:
            # First term: (1-h[j+k]) for j=0..n-1-k
            A[m, :n-k] = 1.0 - h[k:n]  # h[j+k] for j=0..n-1-k
            # Second term: -h[j-k] for j=k..n-1
            A[m, k:] -= h[:n-k]  # h[j-k] for j=k..n-1
        else:
            kp = -k
            # First term: (1-h[j-kp]) for j=kp..n-1
            A[m, kp:] = 1.0 - h[:n-kp]
            # Second term: -h[j+kp] for j=0..n-1-kp
            A[m, :n-kp] -= h[kp:n]
    return A

    corr = np.correlate(h, 1.0 - h, mode='full')

    n = len(h)

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14

            h = np.clip(h + diff/n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    """Compute all f_k values vectorized."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr  # shape (2n-1,)

def gradient_matrix(h, n):
    """Build the (2n-1) x n gradient matrix."""
    A = np.zeros((2*n-1, n))
    shifts = np.arange(-(n-1), n)
    for m, k in enumerate(shifts):
        # g_k[j] = (1-h[j+k])*[0≤j+k<n] - h[j-k]*[0≤j-k<n]
        # First term
        if k >= 0:
            A[m, :n-k] = 1.0 - h[k:]
        else:
            A[m, -k:] = 1.0 - h[:n+k]
        # Second term
        if k > 0:
            A[m, k:] -= h[:n-k]
        elif k < 0:
            A[m, :n+k] -= h[-k:]
    return A

    """Solve SCA QP iterations."""
    A = gradient_matrix(h_bar, n)
    current = h_bar.copy()
    for it in range(max_iter):
        corr = np.correlate(current, 1.0 - current, mode='full')
        f_k = corr  # f_k values
        c_vec = -f_k + A @ current  # RHS of linearized constraints
        
        # Build CVXPY problem
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        obj = cp.Minimize(t_var + (mu/2.0) * cp.sum_squares(h_var - current))

                h_var >= 0, h_var <= 1]

        h_new = project_feasible(h_new, target)

        old_c5 = compute_c5(current)
        
        if new_c5 < old_c5 - tol:
            current = h_new
        else:

    A = np.zeros((2 * n - 1, n))
    shifts = np.arange(-(n - 1), n)
    for m, k in enumerate(shifts):

            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(2 * n - 1)
                for q in range(2 * n - 1):
                    kk = q - mid
                    gp = 0.0
                    if 0 <= i + kk < n:
                        gp += (1.0 - h[i + kk])
                    if 0 <= i - kk < n:
                        gp -= h[i - kk]
                    gj = 0.0
                    if 0 <= j + kk < n:
                        gj += (1.0 - h[j + kk])
                    if 0 <= j - kk < n:
                        gj -= h[j - kk]
                    g[q] = gj - gp
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def stochastic_refine(h_init, n, target, time_limit, start_time, max_steps=15000, base_scale=0.03):
    rng = np.random.RandomState(12345)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 2500:
            break
        scale = base_scale * max(0.05, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        perm = rng.permutation(n)
        idxs = perm[:n_pert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 60:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0
    return best_h, best_c5

def make_block_pattern(n, target, n_blocks):
    h = np.zeros(n)
    block_len = n // n_blocks
    on_len = max(1, block_len // 2)
    for b in range(n_blocks):
        start = b * block_len
        end = min(start + on_len, n)
        h[start:end] = 1.0
    h = project_feasible(h, target)
    return h

def make_periodic_pattern(n, target, period):
    h = np.zeros(n)
    on = period // 2
    for i in range(n):
        if i % period < on:
            h[i] = 1.0
    h = project_feasible(h, target)
    return h
