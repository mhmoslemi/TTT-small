import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_topk(x, alpha, top_k):
    corr = np.correlate(x, 1.0 - x, mode='full')
    k = min(top_k, len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    top = corr[idx]
    m = np.max(top)
    return float(m + np.log(np.sum(np.exp(top - m))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_opt(h_start, n, target, alphas, maxiter, top_k=5):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj_topk, h_cur, args=(alpha, top_k),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def interp_to(src, n_dst, target_dst):
    n_src = len(src)
    if n_src == n_dst:
        return project_feasible(src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, src)
    return project_feasible(h_out, target_dst)


def downsample_to(src, n_dst, target_dst):
    n_src = len(src)
    if n_src == n_dst:
        return project_feasible(src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, src)
    return project_feasible(h_out, target_dst)


def pair_cd(h_init, n, target, time_limit, max_sweeps=30000,

    best_h = h.copy()

    step = 0

    for sweep in range(max_sweeps):

            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_act = min(n_active_shifts, n_corr)
        top_idx = np.argsort(corr)[-n_act:]
        w = corr[top_idx].copy()
        w = w - np.min(w) + 1e-10
        w = w / np.sum(w)

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4 + 1)

        freedom = np.minimum(h, 1.0 - h) + 1e-12
        score = np.abs(grad) * freedom
        sorted_idx = np.argsort(score)[::-1]
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)])

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue

                a_d = delta_min
                b_d = delta_max

                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-12
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-10:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)

                    continue

                approx_max = np.max(corr + delta_star * g)

                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def projected_grad_step(h_init, n, target, time_limit, max_iter=5000, lr_start=0.005, n_active=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    mom = np.zeros(n)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_act = min(n_active, n_corr)
        top_idx = np.argsort(corr)[-n_act:]
        max_corr = np.max(corr)
        tol = max(1e-10, 1e-8 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)

        progress = min(it / max_iter, 1.0)
        lr = lr_start * (1.0 - 0.9 * progress) ** 2
        beta = 0.8

        mom = beta * mom + (1.0 - beta) * grad
        h_new = project_feasible(h - lr * mom, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val - 1e-15:
            h = h_new
        else:
            mom *= 0.5

    return best_h, best_c5

def freq_perturb(h, n, target, rng, scale=0.03):

    h_hat[mask] += noise
    h_new = np.fft.irfft(h_hat, n)
    return project_feasible(h_new, target)

        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(150, tleft() * 0.5),
                              max_sweeps=20000,
                              n_cand_start=20, n_cand_end=70,
                              max_no_improve=400, n_active_shifts=4,
                              bisect_iters=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n1 = 800
    target1 = n1 / 2.0
    h1 = interp_to(best_h, n1, target1)

    if tleft() > 300:
        h_cd1, c5_cd1 = pair_cd(h1, n1, target1,
                                time_limit=min(200, tleft() * 0.5),
                                max_sweeps=15000,
                                n_cand_start=15, n_cand_end=60,
                                max_no_improve=500, n_active_shifts=4,
                                bisect_iters=13)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1
            n0 = n1
            target0 = target1
        else:
            pass

    if tleft() > 150:
        h_sg, c5_sg = projected_grad_step(best_h, n0, target0,
                                          time_limit=min(80, tleft() * 0.5),
                                          max_iter=8000, lr_start=0.004,
                                          n_active=4)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 100:
        h_pol2, sc2 = slsqp_opt(best_h, n0, target0,
                                [500, 3000, 10000], 600, top_k=4)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 150:
        h_bm, c5_bm = block_swap_cd(best_h, n0, target0,
                                    time_limit=min(100, tleft() * 0.5),
                                    max_steps=8000)
        if c5_bm < best_score:
            best_score = c5_bm
            best_h = h_bm

    if tleft() > 200:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.015, 0.04, 0.08]:
            if tleft() < 80:
                break
            h_pert = freq_perturb(best_h, n0, target0, rng, scale=noise_scale)
            h_ref, sc_ref = pair_cd(h_pert, n0, target0,
                                    time_limit=min(50, tleft() * 0.3),
                                    max_sweeps=5000,
                                    n_cand_start=20, n_cand_end=50,
                                    max_no_improve=300, n_active_shifts=3,
                                    bisect_iters=12)
            if tleft() > 30:
                h_ref, sc_ref = slsqp_opt(h_ref, n0, target0,
                                          [1000, 5000], 300, top_k=3)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 100:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interp_to(best_h, n2, target2)
        if tleft() > 80:
            h_cd2, c5_cd2 = pair_cd(h2, n2, target2,
                                    time_limit=min(60, tleft() * 0.5),
                                    max_sweeps=8000,
                                    n_cand_start=12, n_cand_end=40,
                                    max_no_improve=400, n_active_shifts=3,
                                    bisect_iters=12)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                n0 = n2
                target0 = target2
        if tleft() > 40:
            h_pol3, sc3 = slsqp_opt(best_h, n0, target0,
                                    [2000, 8000], 400, top_k=3)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3

    if tleft() > 60:
        h_cd_final, c5_final = pair_cd(best_h, n0, target0,
                                       time_limit=min(40, tleft() * 0.5),
                                       max_sweeps=10000,
                                       n_cand_start=30, n_cand_end=80,
                                       max_no_improve=300, n_active_shifts=5,
                                       bisect_iters=15)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 30:
        h_final, sc_final = slsqp_opt(best_h, n0, target0,
                                      [3000, 10000, 30000], 400, top_k=3)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
