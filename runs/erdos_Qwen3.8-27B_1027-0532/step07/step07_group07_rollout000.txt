import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _time_left(start_time, budget_s):
    return budget_s - 25.0 - (time.time() - start_time)

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-15):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    ones_vec = np.ones(n)
    for alpha in alphas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
        def grad(x):
            nn = len(x)
            n_corr = 2 * nn - 1
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            exp_s = np.exp(shifted)
            s = np.sum(exp_s)
            if s < 1e-300:
                return np.zeros(nn)
            weights = exp_s / s
            idxs = np.arange(nn)
            g = np.zeros(nn)
            for k_idx in range(n_corr):
                w = weights[k_idx]
                if w < 1e-10:
                    continue
                k = k_idx - (nn - 1)
                v1 = (idxs - k >= 0) & (idxs - k < nn)
                v2 = (idxs + k >= 0) & (idxs + k < nn)
                g[v1] += w * (1.0 - x[idxs[v1] - k])
                g[v2] -= w * x[idxs[v2] + k]
            return g
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: ones_vec}]
        try:
            res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def aug_lag_lbfgsb(h_start, n, target, alphas, maxiter, start_time, budget_s):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    ones_vec = np.ones(n)
    for alpha in alphas:
        if _time_left(start_time, budget_s) < 20:
            break
        lam = 0.0
        mu = 10.0
        for outer in range(8):
            if _time_left(start_time, budget_s) < 20:
                break
            def obj_aug(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
                obj_val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
                viol = np.sum(x) - target
                return obj_val + lam * viol + mu * viol * viol
            def grad_aug(x):
                nn = len(x)
                n_corr = 2 * nn - 1
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
                exp_s = np.exp(shifted)
                s = np.sum(exp_s)
                weights = exp_s / max(s, 1e-300)
                idxs = np.arange(nn)
                g = np.zeros(nn)
                for k_idx in range(n_corr):
                    w = weights[k_idx]
                    if w < 1e-10:
                        continue
                    k = k_idx - (nn - 1)
                    v1 = (idxs - k >= 0) & (idxs - k < nn)
                    v2 = (idxs + k >= 0) & (idxs + k < nn)
                    g[v1] += w * (1.0 - x[idxs[v1] - k])
                    g[v2] -= w * x[idxs[v2] + k]
                viol = np.sum(x) - target
                g += (lam + 2.0 * mu * viol) * ones_vec
                return g
            try:
                res = minimize(obj_aug, h_cur, jac=grad_aug, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
                h_new = res.x
                viol = np.sum(h_new) - target
                lam += mu * viol
                if abs(viol) < 1e-8:
                    mu *= 0.5
                else:
                    mu *= 2.0
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    cur_score = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, cur_score

def coord_descent(h_init, n, target, time_limit, max_sweeps, n_cand_start, n_cand_end,
                  top_k_eval, max_no_improve, n_shifts_grad, start_time, budget_s):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if _time_left(start_time, budget_s) < 40:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        n_cand = max(n_cand, 10)
        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]
        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            g = change[j] - change[i]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def stochastic_search(h_init, n, target, time_limit, start_time, budget_s, max_steps=20000):
    start = time.time()
    rng = np.random.RandomState(12345)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if _time_left(start_time, budget_s) < 40:
            break
        if no_improve > 2000:
            break
        scale = 0.03 * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.choice([2, 3, 5, 8, 10, 15]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 150:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0
    return best_h, best_c5

        if tl > 200:
            h_cd, c5_cd = coord_descent(best_h, n_cur, target_cur,
                                        time_limit=min(180, tl * 0.75),
                                        max_sweeps=50000, n_cand_start=25, n_cand_end=70,
                                        top_k_eval=80, max_no_improve=600,
                                        n_shifts_grad=3, start_time=start_time,
                                        budget_s=budget_s)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

        tl = _time_left(start_time, budget_s)
        if tl > 60:
            h_pol, sc = slsqp_optimize(best_h, n_cur, target_cur, [500, 2000, 10000], 500, 1e-15)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

        tl = _time_left(start_time, budget_s)
        if tl > 100:
            h_cd2, c5_cd2 = coord_descent(best_h, n_cur, target_cur,
                                          time_limit=min(90, tl * 0.8),
                                          max_sweeps=40000, n_cand_start=35, n_cand_end=90,
                                          top_k_eval=100, max_no_improve=700,
                                          n_shifts_grad=4, start_time=start_time,
                                          budget_s=budget_s)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    tl = _time_left(start_time, budget_s)
    if tl > 80:
        h_al, sc_al = aug_lag_lbfgsb(best_h, n_cur, target_cur, [2000, 8000, 20000], 300, start_time, budget_s)
        if sc_al < best_score:
            best_score = sc_al
            best_h = h_al

    tl = _time_left(start_time, budget_s)
    if tl > 60:
        h_ss, c5_ss = stochastic_search(best_h, n_cur, target_cur,
                                        time_limit=min(40, tl * 0.6),
                                        start_time=start_time, budget_s=budget_s)
        if c5_ss < best_score:
            best_score = c5_ss
            best_h = h_ss

    tl = _time_left(start_time, budget_s)
    if tl > 100:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.015, 0.04, 0.08]:
            tl = _time_left(start_time, budget_s)
            if tl < 50:
                break
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, target_cur)
            h_ref, sc_ref = slsqp_optimize(h_pert, n_cur, target_cur,
                                           [200, 1000, 5000], 400, 1e-15)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            tl = _time_left(start_time, budget_s)
            if tl < 30:
                break
            h_cd, c5_cd = coord_descent(h_ref, n_cur, target_cur,
                                        time_limit=min(20, tl * 0.4),
                                        max_sweeps=10000, n_cand_start=30, n_cand_end=60,
                                        top_k_eval=60, max_no_improve=200,
                                        n_shifts_grad=3, start_time=start_time,
                                        budget_s=budget_s)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    tl = _time_left(start_time, budget_s)
    if tl > 40:
        h_cd_final, c5_final = coord_descent(best_h, n_cur, target_cur,
                                             time_limit=min(30, tl * 0.7),
                                             max_sweeps=15000, n_cand_start=40, n_cand_end=100,
                                             top_k_eval=120, max_no_improve=200,
                                             n_shifts_grad=5, start_time=start_time,
                                             budget_s=budget_s)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    tl = _time_left(start_time, budget_s)
    if tl > 25:
        h_final, sc_final = slsqp_optimize(best_h, n_cur, target_cur, [5000, 20000], 400, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
