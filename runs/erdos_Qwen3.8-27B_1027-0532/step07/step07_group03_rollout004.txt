def get_subgradient(h, corr, threshold_frac=0.01):
    """Compute a subgradient using all near-worst shifts with exponential weights."""
    n = len(h)
    mid = n - 1
    c_max = np.max(corr)
    # Select shifts with corr within threshold_frac of max
    threshold = c_max - threshold_frac * (c_max - np.min(corr))
    active = corr >= threshold
    if active.sum() == 0:
        active = np.zeros(len(corr), dtype=bool)
        active[int(np.argmax(corr))] = True
    
    # Exponential weights
    gaps = c_max - corr[active]
    weights = np.exp(-50.0 * gaps / (c_max + 1e-15))
    weights /= weights.sum()
    
    active_idx = np.where(active)[0]
    idxs = np.arange(n)
    grad = np.zeros(n)

        if w < 1e-10:
            continue
        k = idx - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]

    b = delta_hi

import numpy as np
from scipy.optimize import linprog
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_subgradient(h, corr, threshold_frac=0.05):
    n = len(h)
    mid = n - 1
    c_max = np.max(corr)
    c_range = c_max - np.min(corr) + 1e-15
    threshold = c_max - threshold_frac * c_range
    active = corr >= threshold
    if not active.any():
        active = np.zeros(len(corr), dtype=bool)
        active[int(np.argmax(corr))] = True
    gaps = c_max - corr[active]
    scale = np.max(gaps) + 1e-15
    weights = np.exp(-20.0 * gaps / scale)
    weights /= weights.sum()
    active_idx = np.where(active)[0]
    idxs = np.arange(n)
    grad = np.zeros(n)
    for idx, w in zip(active_idx, weights):
        if w < 1e-10:
            continue
        k = idx - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

def bisection_line_search(corr, g_vec, a, b, n_iters=16):
    for _ in range(n_iters):
        mid = 0.5 * (a + b)
        vals = corr + mid * g_vec
        f_mid = np.max(vals)
        active_m = vals >= f_mid - 1e-13
        if active_m.any():
            f_prime = np.max(g_vec[active_m])
        else:
            f_prime = 0.0
        if f_prime > 1e-13:
            b = mid
        else:
            a = mid
    return 0.5 * (a + b)

def interceptive_cd_v2(h_init, n, target, time_limit,
                       n_cands_base=35, max_sweeps=50000, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    recompute_interval = 200

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h)
        else:
            corr = compute_corr(h)

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        c_max = np.max(corr)
        active_mask = corr >= c_max - 1e-10
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = max(10, int(n_cands_base * (0.6 + 0.4 * progress)))
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                max_deriv = np.max(g_vec[active_mask])
                if max_deriv > -1e-12 and max_deriv > 0:
                    if delta_lo >= -1e-12:
                        delta_lo = -1e-6
                delta_star = bisection_line_search(corr, g_vec, delta_lo, delta_hi)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i, :] = 0.0
            change[best_j, :] = 0.0
            pi, pj = best_i, best_j
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj + shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    return best_h, best_c5

def smooth_grad_descent(h_init, n, target, time_limit, alpha_start=500.0,
                        alpha_end=50000.0, max_iters=5000, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    h_curr = h_init.copy().astype(float)
    h_prev = h_curr.copy()
    best_h = h_curr.copy()
    best_score = compute_c5(h_curr)
    idxs = np.arange(n)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum)) / 2.0
        y = h_curr + ((t_k - 1.0) / t_k) * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)

        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(max_iters, 1))
        corr_y = compute_corr(y)
        c_max = np.max(corr_y)
        shifted = np.clip(alpha * (corr_y - c_max), -600.0, 0.0)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            p = np.zeros(len(corr_y))
            p[int(np.argmax(corr_y))] = 1.0
        else:
            p = p / p_sum

        grad = np.zeros(n)
        for m in range(len(corr_y)):
            if p[m] < 1e-15:
                continue
            k = m - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += p[m] * (1.0 - y[idxs[v1] + k])
            grad[v2] -= p[m] * y[idxs[v2] - k]
        grad /= alpha

        step = 0.05
        for attempt in range(15):
            h_new = y - step * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            cur_sc = compute_c5(h_curr)
            if sc < cur_sc + 1e-10:
                break
            step *= 0.5

        h_prev = h_curr.copy()
        h_curr = h_new
        sc = compute_c5(h_curr)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_curr.copy()

    return best_h, best_score

def cutting_plane_refine(h_init, n, target, time_limit, max_iters=5000, tol=1e-9):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_f = []
    cuts_h = []
    max_cuts = 100
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            cuts_g = cuts_g[-8:]
            cuts_f = cuts_f[-8:]
            cuts_h = cuts_h[-8:]
            no_improve = 0

        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])

        active_mask = corr >= corr[k_star] - 1e-6
        active_lags = np.where(active_mask)[0]
        if len(active_lags) == 1:
            g = get_subgradient(h, corr, 0.01)
        else:
            w = np.ones(len(active_lags)) / len(active_lags)
            idxs = np.arange(n)
            g = np.zeros(n)
            for k_s, wgt in zip(active_lags, w):
                k = int(k_s) - mid
                v1 = (idxs + k >= 0) & (idxs + k < n)
                v2 = (idxs - k >= 0) & (idxs - k < n)
                g[v1] += wgt * (1.0 - h[idxs[v1] + k])
                g[v2] -= wgt * h[idxs[v2] - k]

        cuts_g.append(g)
        cuts_f.append(f_val)
        cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break

    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_perturbations(h_base, n, target, rng):
    candidates = []
    candidates.append(h_base.copy())

    for freq in [1, 2, 3, 5]:
        x = np.linspace(0.0, 2.0, n)
        for amp in [0.02, 0.05, 0.1]:
            h_pert = h_base + amp * np.sin(np.pi * freq * x)
            h_pert = project_feasible(h_pert, target)
            candidates.append(h_pert)

    for center in [0.25, 0.5, 0.75]:
        x = np.linspace(0.0, 2.0, n)
        for sigma in [0.1, 0.2, 0.3]:
            bump = np.exp(-0.5 * ((x - center) / sigma) ** 2)
            for sign in [1, -1]:
                for amp in [0.03, 0.08]:
                    h_pert = h_base + sign * amp * bump
                    h_pert = project_feasible(h_pert, target)
                    candidates.append(h_pert)

    for ns in [0.01, 0.03, 0.05, 0.08]:
        h_pert = h_base + rng.randn(n) * ns
        h_pert = project_feasible(h_pert, target)
        candidates.append(h_pert)

    for power in [0.7, 1.3, 0.5, 1.5]:
        h_pert = np.power(np.clip(h_base, 1e-10, 1.0), power)
        h_pert = project_feasible(h_pert, target)
        candidates.append(h_pert)

    return candidates
