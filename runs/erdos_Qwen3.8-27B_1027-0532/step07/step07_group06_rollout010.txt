def frank_wolfe_step(h, n, target, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    idxs = np.arange(n)
    
    # Get top k shifts by correlation value
    top_indices = np.argsort(corr)[-top_k:]
    weights = corr[top_indices].astype(float)
    wsum = np.sum(weights)
    if wsum < 1e-15:
        weights = np.ones(top_k) / top_k
    else:
        weights = weights / wsum
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for s_idx, w in zip(top_indices, weights):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    
    # Frank-Wolfe: find h* that minimizes <grad, h*> subject to constraints
    # This is: put 1 where grad is smallest, 0 where largest, one fractional
    sorted_idx = np.argsort(grad)
    h_star = np.zeros(n)
    h_star[sorted_idx[:target]] = 1.0
    # Adjust for exact sum (target should be integer n/2)
    # Actually target = n/2 which is integer for even n
    
    # Step: h_new = (1-t)*h + t*h_star
    # Optimal t by line search
    best_t = 0.0
    best_c5 = compute_c5(h)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        h_try = (1-t)*h + t*h_star
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_t = t
    if best_t > 0:
        h_new = (1-best_t)*h + best_t*h_star
        h_new = project_feasible(h_new, target)
        return h_new, best_c5
    return h, best_c5

                     max_no_improve=300, n_top_shifts=3, bisection_iters=18):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    h_vec = h.flatten()
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h_vec[p + shifts[pos_mask]])
        change[p, neg_mask] -= h_vec[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Try escape: random perturbation
            rng = np.random.RandomState(sweep)
            h = h + rng.randn(n) * 0.02
            h = project_feasible(h, target)
            no_improve = 0
            continue
        
        # Compute gradient using top shifts
        top_indices = np.argsort(corr)[-n_top_shifts:]
        weights = corr[top_indices].astype(float)
        wsum = np.sum(weights)
        if wsum < 1e-15:
            weights = np.ones(n_top_shifts) / n_top_shifts
        else:
            weights = weights / wsum
        
        grad = np.zeros(n)
        for si, w in zip(top_indices, weights):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Adaptive candidate set
        progress = min(float(no_improve) / max_no_improve, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)
        ncand = max(ncand, 5)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        
        # Try all pairs
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            ci = change[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-14:
                    continue
                
                # g = change[j] - change[i] (change in corr when moving delta from i to j)
                g = change[j] - ci
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-14:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for row in (best_i, best_j):
                change[row, :] = 0.0
                pos_mask = (row + shifts >= 0) & (row + shifts < n)
                neg_mask = (row - shifts >= 0) & (row - shifts < n)
                change[row, pos_mask] += (1.0 - h[row + shifts[pos_mask]])
                change[row, neg_mask] -= h[row - shifts[neg_mask]]
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_target = 0.0
_g_n = 0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _sum_eq(x):
    return np.sum(x) - _g_target


def _sum_eq_jac(x):
    return np.ones(_g_n)


def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))


def _lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_round(h_start, n, target, p_list, maxiter_per_p, deadline):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for p in p_list:
        if time.time() > deadline:
            break
        try:
            res = minimize(_pnorm_obj, h_cur, args=(p,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_p, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def lse_round(h_start, n, target, alpha_list, maxiter_per_alpha, deadline):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for a in alpha_list:
        if time.time() > deadline:
            break
        try:
            res = minimize(_lse_obj, h_cur, args=(a,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def build_change(h, n):
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]
    return change


def update_change_row(change, row, h, n, shifts):
    change[row, :] = 0.0
    pm = (row + shifts >= 0) & (row + shifts < n)
    nm = (row - shifts >= 0) & (row - shifts < n)
    change[row, pm] += (1.0 - h[row + shifts[pm]])
    change[row, nm] -= h[row - shifts[nm]]


def pair_cd_main(h_init, n, target, deadline,
                 max_sweeps=60000, n_cand_lo=10, n_cand_hi=70,
                 max_no_improve=350, n_top=3, bisection_iters=18):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    h = project_feasible(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = build_change(h, n)
    escape_count = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            if escape_count < 15:
                rng = np.random.RandomState(1000 + sweep * 7)
                h = best_h + rng.randn(n) * 0.03
                h = project_feasible(h, target)
                change = build_change(h, n)
                escape_count += 1
            else:
                break
            no_improve = 0
            continue

        top_idx = np.argsort(corr)[-n_top:]
        wts = corr[top_idx].astype(float)
        wsum = np.sum(wts)
        if wsum < 1e-15:
            wts = np.ones(n_top) / n_top
        else:
            wts /= wsum

        grad = np.zeros(n)
        for si, w in zip(top_idx, wts):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        frac = float(no_improve) / max(1, max_no_improve)
        ncand = int(n_cand_lo + (n_cand_hi - n_cand_lo) * min(frac, 1.0))
        ncand = max(ncand, 5)
        ncand = min(ncand, n // 4)

        sidx = np.argsort(grad)
        low_set = sidx[:ncand]
        high_set = sidx[-ncand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_l = len(low_set)

            i = int(high_set[ii])
            ci = change[i]
            hi = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                hj = h[j]
                dlo = max(hi - 1.0, -hj)
                dhi = min(hi, 1.0 - hj)
                if dhi - dlo < 1e-14:
                    continue
                g = change[j] - ci
                a_d = dlo
                b_d = dhi
                for _ in range(bisection_iters):
                    md = 0.5 * (a_d + b_d)
                    vals = corr + md * g
                    fm = np.max(vals)
                    active = vals >= fm - 1e-14
                    fp = np.max(g[active]) if active.any() else 0.0
                    if fp > 1e-12:
                        b_d = md
                    else:
                        a_d = md
                ds = 0.5 * (a_d + b_d)
                if abs(ds) < 1e-14:
                    continue
                am = float(np.max(corr + ds * g)) * dx
                if am < best_trial:
                    best_trial = am
                    best_i = i
                    best_j = j
                    best_delta = ds

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def block_cd_local(h_init, n, target, deadline, block_size=4, max_iters=2000):

            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1

            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        best_start = 0
        best_score = np.inf
        step_max = max(1, (n - block_size) // 200)
        for s in range(0, n - block_size + 1, step_max):
            sc = np.sum(grad[s:s + block_size] ** 2)
            if sc < best_score:
                best_score = sc
                best_start = s

        bidx = np.arange(best_start, best_start + block_size)
        rest_sum = target - np.sum(h) + np.sum(h[bidx])

        def _block_fn(xb, _bidx=bidx, _h=h, _rest_sum=rest_sum, _target=target):
            ht = _h.copy()
            ht[_bidx] = np.clip(xb, 0.0, 1.0)
            ht = project_feasible(ht, _target)
            c = np.correlate(ht, 1.0 - ht, mode='full')
            cm = np.max(c)
            if cm <= 0:
                return 0.0
            sh = np.clip(800.0 * (c - cm), -500.0, 0.0)
            return float(cm + np.log(np.sum(np.exp(sh))) / 800.0)

            h = project_feasible(h, target)
            sc = compute_c5(h)
            if sc < c5_val - 1e-14:

        except Exception:
            pass

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    return project_feasible(h_out, target_dst)
