from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def enforce_symmetry(h):
    """Enforce h[i] + h[n-1-i] = 1"""
    n = len(h)
    half = n // 2
    for i in range(half):
        avg = (h[i] + (1.0 - h[n-1-i])) / 2.0
        h[i] = avg
        h[n-1-i] = 1.0 - avg
    if n % 2 == 1:
        h[n//2] = 0.5
    return h

def make_full(h_half, n):
    """Construct full h from half vector (assuming even n)"""
    h = np.empty(n)
    h[:n//2] = h_half
    h[n//2:] = 1.0 - h_half[::-1]
    return h

def smooth_max_grad(h_half, n, alpha):
    """
    Compute gradient of smoothed max objective w.r.t. h_half.
    h_half has length n//2, full h has length n.
    Returns (objective_value, gradient_wrt_half)
    """
    h = make_full(h_half, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    mid = n - 1
    
    # Smoothed max
    shifted = alpha * corr - alpha * np.max(corr)
    log_sum_exp = np.log(np.sum(np.exp(shifted)))
    obj = (np.max(corr) + log_sum_exp / alpha) * dx
    
    # Weights for each shift
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient of corr_s w.r.t. h_full[p]:
    # ∂corr_s/∂h_full[p] = (1-h_full[p-s]) * [valid] - h_full[p+s] * [valid]
    # where s = idx - mid
    
    idxs = np.arange(n)
    grad_full = np.zeros(n)
    
    for s_idx in range(n_corr):
        w = weights[s_idx]
        if w < 1e-15:
            continue
        s = s_idx - mid
        # h_full[p] appears with (1-h_full[p-s]) when p-s is valid
        v1 = (idxs - s >= 0) & (idxs - s < n)
        grad_full[v1] += w * (1.0 - h[idxs[v1] - s])
        # h_full[p] appears with -h_full[p+s] when p+s is valid
        v2 = (idxs + s >= 0) & (idxs + s < n)
        grad_full[v2] -= w * h[idxs[v2] + s]
    
    grad_full *= dx
    
    # Gradient w.r.t. h_half:

    # h_full[n-1-i] = 1 - h_half[i] for i < n//2

    half = n // 2
    grad_half = grad_full[:half] - grad_full[n-1-np.arange(half)]
    
    return obj, grad_half

        h_half = h_half + v
        h_half = np.clip(h_half, 0.0, 1.0)
    
    return best_h_half, best_c5

def coord_desc_symmetric(h_half, n, time_limit, max_passes=100):
    start = time.time()
    half = n // 2
    h_half = h_half.copy()
    best_h_half = h_half.copy()
    best_c5 = compute_c5(make_full_from_half(h_half))
    dx = 2.0 / n
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        
        improved = False
        # Random order
        rng = np.random.RandomState(p * 31 + 17)
        order = rng.permutation(half)
        
        for i in order:
            if time.time() - start > time_limit:
                break
            
            h_full = make_full_from_half(h_half)
            corr_base = np.correlate(h_full, 1.0 - h_full, mode='full')
            
            # Compute the effect of changing h_half[i]
            # Change in corr_s = 2*(t - h_half[i]) * a_s
            # where a_s = (1-h[i-s]) - h[i+s] (with validity)
            t0 = h_half[i]
            n_full = n
            s_vals = np.arange(-(n_full-1), n_full)
            a = np.zeros(len(s_vals))
            for s_idx, s in enumerate(s_vals):
                # (1-h[i-s]) valid if 0 <= i-s < n
                # -h[i+s] valid if 0 <= i+s < n
                val = 0.0
                if 0 <= i - s < n_full:
                    val += (1.0 - h_full[i - s])
                if 0 <= i + s < n_full:
                    val -= h_full[i + s]
                a[s_idx] = 2.0 * val
            
            # f(t) = max_s (corr_base[s] + (t - t0) * a[s]) * dx
            # This is convex in t. Find minimum by bisection on subgradient.
            lo = 0.0
            hi = 1.0
            
            for _ in range(30):
                mid = 0.5 * (lo + hi)
                vals = corr_base + (mid - t0) * a
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    # Subgradient interval: [min(a[active]), max(a[active])]
                    sg_lo = np.min(a[active])
                    sg_hi = np.max(a[active])
                    if sg_lo <= 0 <= sg_hi:
                        lo = hi = mid  # optimal
                        break
                    elif sg_lo > 0:
                        hi = mid
                    else:
                        lo = mid
                else:
                    break
            
            t_star = 0.5 * (lo + hi)
            if abs(t_star - t0) > 1e-10:
                h_half[i] = t_star
                h_full_new = make_full_from_half(h_half)
                c5_new = compute_c5(h_full_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h_half = h_half.copy()
                improved = True
        
        if not improved and p > 5:
            break
    
    return best_h_half, best_c5

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    return np.interp(x_new, x_old, h_src)
