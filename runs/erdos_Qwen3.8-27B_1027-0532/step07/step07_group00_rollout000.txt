import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_opt(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def active_set_subgrad(h_init, n, target, time_limit, top_k=10, 
                        step_scale=0.01, patience=500):
    """Active-set subgradient method with adaptive step size."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_c5_so_far = best_c5
    no_improve = 0
    
    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            break
        
        # Find active set (top-k shifts)
        top_idx = np.argsort(corr)[-top_k:]
        weights = np.zeros(n_corr)
        for idx in top_idx:
            w = (corr[idx] / max(corr[top_idx[0]], 1e-20)) ** 4
            weights[idx] = w
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        idxs = np.arange(n)
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-10:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        # Zero-mean constraint on gradient
        grad -= np.mean(grad)
        
        # Polyak step size
        gap = c5_val - best_c5_so_far
        if gap < 1e-10:
            gap = 1e-10
        g_norm_sq = np.sum(grad * grad)
        if g_norm_sq < 1e-20:
            break
        alpha = min(step_scale * gap / g_norm_sq, 0.1)
        
        h_try = h - alpha * grad
        h_try = project_feasible(h_try, target)
        
        # Check if improvement
        c5_try = compute_c5(h_try)
        if c5_try < c5_val:
            h = h_try
            if c5_try < best_c5_so_far:
                best_c5_so_far = c5_try
        else:
            # Try smaller step
            alpha *= 0.5
            h_try = h - alpha * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val:
                h = h_try
            else:
                pass  # No improvement, continue
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
