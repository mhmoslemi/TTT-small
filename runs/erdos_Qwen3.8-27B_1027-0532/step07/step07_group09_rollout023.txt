import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_eq_fun(x):
    return np.sum(x) - _g_target


def cons_eq_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def aggressive_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                       n_cand_low=15, n_cand_high=15, max_no_improve=500):
    """Aggressive pairwise CD with exact line search and adaptive candidate selection."""

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')

    for sweep in range(max_sweeps):

        if no_improve > max_no_improve:

        c5_scaled = c5_val * dx

            no_improve = 0
        else:

        # Find active shifts (those near the maximum)
        max_corr = np.max(corr)

        n_active = min(len(active), 10)
        top_idx = np.argsort(corr)[-n_active:]

        # Compute weighted gradient
        w = corr[top_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Adaptive candidate count
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_low + (n_cand_high - n_cand_low) * 0.3 * progress)
        ncand = max(ncand, 8)
        ncand = min(ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        for r in range(n_cands):
            G[r, :] = change[all_cands[r], :]

        # Try all pairs

        n_l = len(low_set)
        best_move_val = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = cand_map[i]
            h_i = h[i]
            Gi = G[hr]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = cand_map[j]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-13:
                    continue

                g = G[lr] - Gi

                # Bisection for optimal delta
                a_d = delta_min
                b_d = delta_max
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid < best_move_val:
                        a_d = mid_d
                    else:
                        # Check if we can still improve
                        active_m = vals >= f_mid - 1e-13
                        if active_m.any():
                            f_prime = np.max(g[active_m])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            a_d = mid_d
                        else:
                            b_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_move_val:
                    best_move_val = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

def triple_cd(h_init, n, target, time_limit, max_sweeps=20000):

            no_improve = 0
        else:
            no_improve += 1

            # Case 1: move from i to j and k (split)
            # Case 2: move from i,j to k (merge)
            case = rng.randint(2)
            i = rng.randint(n)
            j = rng.randint(n)
            k = rng.randint(n)
            if len(set([i, j, k])) < 3:
                continue

            if case == 0:
                # Move from i to j and k
                if h[i] < 1e-10:
                    continue
                # delta from i splits: alpha to j, (1-alpha) to k
                alpha = rng.uniform(0.2, 0.8)
                delta_max = min(h[i], (1.0 - h[j]) / alpha, (1.0 - h[k]) / (1.0 - alpha))
                if delta_max < 1e-10:
                    continue
                delta = rng.uniform(0.1, 0.9) * delta_max
                # Check if it helps
                g = -1.0 * change[i] + alpha * change[j] + (1.0 - alpha) * change[k]
                new_corr = corr + delta * g
                if np.max(new_corr) < c5_val - 1e-14:
                    h[i] -= delta
                    h[j] += alpha * delta
                    h[k] += (1.0 - alpha) * delta
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    update_change_row(change, i, h, n, shifts)
                    update_change_row(change, j, h, n, shifts)
                    update_change_row(change, k, h, n, shifts)
                    corr = np.correlate(h, 1.0 - h, mode='full')
            else:
                # Move from i and j to k
                if h[i] + h[j] < 1e-10 or h[k] >= 1.0 - 1e-10:
                    continue
                delta_max = min(h[i], h[j], 1.0 - h[k])
                if delta_max < 1e-10:
                    continue
                delta = rng.uniform(0.1, 0.9) * delta_max
                g = -1.0 * change[i] - 1.0 * change[j] + 2.0 * change[k]
                new_corr = corr + delta * g
                if np.max(new_corr) < c5_val - 1e-14:
                    h[i] -= delta
                    h[j] -= delta
                    h[k] += 2.0 * delta
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, target)
                    update_change_row(change, i, h, n, shifts)
                    update_change_row(change, j, h, n, shifts)
                    update_change_row(change, k, h, n, shifts)
                    corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5

            best_h = h.copy()

        # The autocorrelation of h is related to |FFT(h)|^2
        # We want to spread out the energy in frequency domain
        ft_h = np.fft.rfft(h)

        total_power = np.sum(power)

        # Gradient: direction that equalizes the frequency power spectrum
        # This is a heuristic - we want to reduce concentration in any frequency
        if total_power > 0:

            # For each frequency bin, if power is high, we want to reduce it
            # The derivative of |F(h)|^2 with respect to h is related to IFFT(F(h) * conj(phase))
            # Simplified: use the imaginary part to get a gradient direction
            phase = np.angle(ft_h)
            # Direction that reduces peak power
            max_power_idx = np.argmax(power[1:]) + 1  # skip DC
            if power[max_power_idx] > 2.0 * total_power / n:
                # Reduce this frequency component
                reduction = np.fft.irfft(np.zeros_like(ft_h), n=n)
                ft_h[max_power_idx] *= 0.99
                if max_power_idx < n - n // 2:
                    ft_h[n - max_power_idx] = np.conj(ft_h[max_power_idx]) if n - max_power_idx < len(ft_h) else ft_h[max_power_idx]
                reduction = np.fft.irfft(ft_h, n=n)
                freq_grad = h - reduction

        # Combine with standard gradient
        max_corr = np.max(corr)
        tol = max(1e-8 * max_corr, 1e-12)
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        for k_idx in active[:5]:

def compute_c5(h):
    n = len(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_change_matrix(h, n):

    change = np.zeros((n, n_corr))

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

            if abs(delta_star) < 1e-13:

            approx_max = np.max(corr + delta_star * g)
            if approx_max < best_move_val:
                best_move_val = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5


def directed_moves_cd(h_init, n, target, time_limit, max_sweeps=15000):
    """CD focusing on moves that directly reduce the worst shift."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        c5_val = float(np.max(corr))
        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        worst_idx = int(np.argmax(corr))
        worst_k = worst_idx - mid

        v1 = (idxs - worst_k >= 0) & (idxs - worst_k < n)
        v2 = (idxs + worst_k >= 0) & (idxs + worst_k < n)

        direct_grad = np.zeros(n)
        direct_grad[v1] = (1.0 - h[idxs[v1] - worst_k])
        direct_grad[v2] -= h[idxs[v2] + worst_k]
        direct_grad -= np.mean(direct_grad)

        n_cand = 25
        sorted_idx = np.argsort(direct_grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            G[r, :] = change[all_cands[r], :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_move_val = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = cand_map[i]
            h_i = h[i]
            Gi = G[hr]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = cand_map[j]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-13:
                    continue

                g = G[lr] - Gi

                a_d = delta_min
                b_d = delta_max
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid < best_move_val:
                        a_d = mid_d
                    else:
                        active_m = vals >= f_mid - 1e-13
                        if active_m.any():
                            f_prime = np.max(g[active_m])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            a_d = mid_d
                        else:
                            b_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_move_val:
                    best_move_val = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish at n=400
    if tleft() > 60:
        h_pol = slsqp_optimize(best_h, n0, target0,
                               [100, 500, 2000, 8000], 500, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Quick CD at n=400
    if tleft() > 80:
        h_cd0, c5_cd0 = efficient_pair_cd(best_h, n0, target0,
                                          time_limit=min(50, tleft() * 0.5),
                                          max_sweeps=15000,
                                          n_cand_low=10, n_cand_high=20,
                                          max_no_improve=200)
        if c5_cd0 < best_score:
            best_score = c5_cd0
            best_h = h_cd0

    # Phase 2: Move to n=600
    n1 = 600
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)
    sc1 = compute_c5(h1)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    if tleft() > 200:
        h_cd1, c5_cd1 = efficient_pair_cd(h1, n1, target1,
                                          time_limit=min(150, tleft() * 0.55),
                                          max_sweeps=30000,
                                          n_cand_low=12, n_cand_high=28,
                                          max_no_improve=350)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1

    if tleft() > 60:
        h_pol1 = slsqp_optimize(best_h, n1, target1,
                                [500, 2000, 8000], 700, 1e-15)
        sc = compute_c5(h_pol1)
        if sc < best_score:
            best_score = sc
            best_h = h_pol1

    if tleft() > 100:
        h_cd1b, c5_cd1b = directed_moves_cd(best_h, n1, target1,
                                            time_limit=min(70, tleft() * 0.5),
                                            max_sweeps=12000)
        if c5_cd1b < best_score:
            best_score = c5_cd1b
            best_h = h_cd1b

    # Phase 3: Move to n=800
    n2 = 800
    target2 = n2 / 2.0
    h2 = interpolate_to(best_h, n2, target2)
    sc2 = compute_c5(h2)
    if sc2 < best_score:
        best_score = sc2
        best_h = h2

    if tleft() > 250:
        h_cd2, c5_cd2 = efficient_pair_cd(best_h, n2, target2,
                                          time_limit=min(200, tleft() * 0.55),
                                          max_sweeps=25000,
                                          n_cand_low=15, n_cand_high=35,
                                          max_no_improve=400)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 80:
        h_pol2 = slsqp_optimize(best_h, n2, target2,
                                [1000, 5000, 20000], 800, 1e-15)
        sc = compute_c5(h_pol2)
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    if tleft() > 120:
        h_cd2b, c5_cd2b = directed_moves_cd(best_h, n2, target2,
                                            time_limit=min(90, tleft() * 0.5),
                                            max_sweeps=15000)
        if c5_cd2b < best_score:
            best_score = c5_cd2b
            best_h = h_cd2b

    # Phase 4: Wide search for escape
    if tleft() > 100:
        h_wide, c5_wide = wide_search_cd(best_h, n2, target2,
                                         time_limit=min(80, tleft() * 0.5),
                                         max_sweeps=15000,
                                         n_cand=35)
        if c5_wide < best_score:
            best_score = c5_wide
            best_h = h_wide

    # Phase 5: Try n=1000
    if tleft() > 200:
        n3 = 1000
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        sc3 = compute_c5(h3)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3
            n2 = n3
            target2 = target3

        if tleft() > 150:
            h_cd3, c5_cd3 = efficient_pair_cd(best_h, n2, target2,
                                              time_limit=min(120, tleft() * 0.55),
                                              max_sweeps=20000,
                                              n_cand_low=12, n_cand_high=25,
                                              max_no_improve=350)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

        if tleft() > 60:
            h_pol3 = slsqp_optimize(best_h, n2, target2,
                                    [2000, 8000, 30000], 600, 1e-15)
            sc = compute_c5(h_pol3)
            if sc < best_score:
                best_score = sc
                best_h = h_pol3

    # Phase 6: Perturbation diversification
    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for attempt in range(6):
            if tleft() < 40:
                break
            noise_scale = 0.008 + 0.012 * attempt
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, target2)
            h_ref, c5_ref = efficient_pair_cd(h_pert, n2, target2,
                                              time_limit=min(20, tleft() * 0.25),
                                              max_sweeps=5000,
                                              n_cand_low=12, n_cand_high=25,
                                              max_no_improve=150)
            if c5_ref < best_score:
                best_score = c5_ref
                best_h = h_ref

    # Phase 7: Final intensive CD
    if tleft() > 60:
        h_final_cd, c5_final = efficient_pair_cd(best_h, n2, target2,
                                                 time_limit=min(45, tleft() * 0.55),
                                                 max_sweeps=15000,
                                                 n_cand_low=20, n_cand_high=45,
                                                 max_no_improve=250)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    # Phase 8: Final SLSQP polish
    if tleft() > 30:
        h_final = slsqp_optimize(best_h, n2, target2,
                                 [5000, 20000, 80000], 500, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
