import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 5.0, 20.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
            except Exception:
                pass

    return best_h, best_c5

def pair_cd_v2(h_init, n, target, time_limit, n_cand=40, max_sweeps=50000,
               n_shifts=1, ns_iters=20, two_pass=True):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(777)
    stall_thresh = 300

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        if no_improve > stall_thresh:
            h_pert = best_h + rng.randn(n) * 0.03
            h_pert = project_feasible(h_pert, target)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5 + 0.002:
                h = h_pert
                no_improve = 0
                continue
            if no_improve > stall_thresh * 4:
                break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > stall_thresh * 5:
                break

        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = 40.0
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        n_rand = max(5, n_cand // 5)
        existing = set(int(x) for x in np.concatenate([high_set, low_set]))
        remaining = np.array([i for i in range(n) if i not in existing])
        if len(remaining) > 0:
            rand_set = remaining[rng.choice(len(remaining), size=min(n_rand, len(remaining)), replace=False)]
        else:
            rand_set = np.array([], dtype=int)
        all_cands = np.unique(np.concatenate([high_set, low_set, rand_set]))
        nc = len(all_cands)

        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        high_list = [int(x) for x in high_set]
        low_list = [int(x) for x in low_set]
        partial_scores = np.zeros(len(high_list) * len(low_list))
        pair_info = []

        for ii, i in enumerate(high_list):
            hi_val = float(h[i])
            if hi_val < 1e-12:
                pair_info.append((ii, ii * len(low_list), ii * len(low_list) + len(low_list), -1))
                continue
            ci = cand_pos[i]
            base = ii * len(low_list)
            for jj, j in enumerate(low_list):
                if i == j:
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[j]
                g = profiles[ci] - profiles[cj]
                mid_d = 0.5 * delta_max
                vals = corr + mid_d * g
                f_mid = float(np.max(vals))
                rough_improve = cur_max - f_mid
                idx = base + jj
                partial_scores[idx] = rough_improve
                pair_info.append((ii, idx, idx + 1, (i, j, delta_max, cj, ci)))

        # Get top pairs from rough screening
        valid_indices = np.where(partial_scores > 0)[0]
        if len(valid_indices) == 0:
            continue
        if len(valid_indices) > 80:
            top_pairs_idx = valid_indices[np.argsort(-partial_scores[valid_indices])[:80]]
        else:
            top_pairs_idx = valid_indices

        for pidx in top_pairs_idx:
            info = pair_info[int(pidx)]
            if len(info) < 5 or info[3] == -1:
                continue
            i, j, delta_max, cj, ci = info[3]
            g = profiles[ci] - profiles[cj]

            lo = 0.0
            hi_d = delta_max
            for _ in range(ns_iters):
                mid_d = 0.5 * (lo + hi_d)
                vals = corr + mid_d * g
                f_mid = float(np.max(vals))
                active = vals >= (f_mid - 1e-14)
                if not np.any(active):
                    break
                f_prime = float(np.max(g[active]))
                if f_prime > 1e-14:
                    hi_d = mid_d
                else:
                    lo = mid_d

            delta_star = 0.5 * (lo + hi_d)
            if delta_star < 1e-10:
                continue

            new_max = float(np.max(corr + delta_star * g))
            improve = cur_max - new_max
            if improve > best_improve:
                best_improve = improve
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def momentum_subgradient(h_init, n, target, time_limit, max_steps=50000, n_shifts=5, base_lr=0.04, beta=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0
    velocity = np.zeros(n)
    rng = np.random.RandomState(42)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= (2.0 / n) * np.sum(grad)

        velocity = beta * velocity + (1.0 - beta) * grad

        improved = False
        decay = 1.0 / (1.0 + 0.002 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * velocity
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved:
            velocity *= 0.5
            if step > 5000:
                break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
