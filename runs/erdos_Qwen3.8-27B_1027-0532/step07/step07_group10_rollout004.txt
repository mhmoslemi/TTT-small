                        continue
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_corr_max - 1e-15:
                    best_trial_corr_max = float(approx_max)
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def multi_active_cd(h_init, n, target, time_limit, n_cand=60, n_active=4,
                    line_search_iter=18):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(30000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 250:
            break

        top_shifts = np.argsort(corr)[-n_active:]
        weights = np.zeros(n_corr)
        for si in top_shifts:
            weights[si] = 1.0
        sw = np.sum(weights)
        weights /= sw

        grad = np.zeros(n)
        for si in top_shifts:
            k = int(si) - mid
            w = weights[si]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_trial_corr_max = c5_val / dx

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(line_search_iter):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    if f_mid + 1e-15 >= best_trial_corr_max:
                        b_d = mid_d
                        continue
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_corr_max - 1e-15:
                    best_trial_corr_max = float(approx_max)
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def gradient_descent_active(h_init, n, target, time_limit, max_steps=50000,
                            base_lr=0.05, n_active=6, lr_decay=0.9998):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 8000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.85
                if lr < 1e-7:
                    break

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[-1]]
        raw_w = np.exp(-100.0 * (c_max - corr[top_shifts]))
        weights = raw_w / np.sum(raw_w)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
        else:
            lr *= 0.7

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    rng = np.random.RandomState(seed)

    # Phase 1: SLSQP refinement at n=400 with adaptive alpha
    if tleft() > 200:
        alphas_1 = [20, 80, 300, 1000, 3000, 10000]
        h_pol, sc = slsqp_optimize(best_h, best_n, best_target,
                                   alphas_1, 600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: CD at n=400
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.5)
        h_cd, c5_cd = fast_cd(best_h, best_n, best_target,
                              time_limit=cd_time,
                              max_sweeps=30000,
                              n_cand_start=30, n_cand_end=100,
                              max_no_improve=400,
                              line_search_iter=20)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: GD at n=400
    if tleft() > 60:
        gd_time = min(60, tleft() * 0.5)
        h_gd, c5_gd = gradient_descent_active(best_h, best_n, best_target,
                                              time_limit=gd_time,
                                              max_steps=50000,
                                              base_lr=0.04, n_active=8)
        if c5_gd < best_score:
            best_score = c5_gd
            best_h = h_gd

    # Phase 4: Multi-active CD at n=400
    if tleft() > 60:
        macd_time = min(80, tleft() * 0.5)
        h_macd, c5_macd = multi_active_cd(best_h, best_n, best_target,
                                          time_limit=macd_time,
                                          n_cand=50, n_active=4,
                                          line_search_iter=18)
        if c5_macd < best_score:
            best_score = c5_macd
            best_h = h_macd

    # Phase 5: SLSQP polish at n=400 with very high alpha
    if tleft() > 50:
        h_pol2, sc2 = slsqp_optimize(best_h, best_n, best_target,
                                     [5000, 15000, 50000], 800, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 6: Upsample to n=800
    n1 = 800
    if tleft() > 300:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 250:
            h1_pol, sc1p = slsqp_optimize(h1, n1, target1,
                                          [50, 200, 800, 3000], 500, 1e-15)
            if sc1p < sc1:
                h1 = h1_pol
                sc1 = sc1p

        if tleft() > 180:
            h1_cd, c5_1cd = fast_cd(h1, n1, target1,
                                    time_limit=min(200, tleft() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_start=30, n_cand_end=90,
                                    max_no_improve=350,
                                    line_search_iter=18)
            if c5_1cd < best_score:
                best_score = c5_1cd
                best_h = h1_cd
                best_n = n1
                best_target = target1

        if tleft() > 80:
            h1_gd, c5_1gd = gradient_descent_active(h1, n1, target1,
                                                    time_limit=min(60, tleft() * 0.4),
                                                    max_steps=40000,
                                                    base_lr=0.03, n_active=8)
            if c5_1gd < best_score:
                best_score = c5_1gd
                best_h = h1_gd
                best_n = n1
                best_target = target1

        if tleft() > 60:
            h1_macd, c5_1macd = multi_active_cd(h1, n1, target1,
                                                time_limit=min(60, tleft() * 0.4),
                                                n_cand=60, n_active=5,
                                                line_search_iter=16)
            if c5_1macd < best_score:
                best_score = c5_1macd
                best_h = h1_macd
                best_n = n1
                best_target = target1

        if tleft() > 50:
            h1_pol2, sc1p2 = slsqp_optimize(best_h, best_n, best_target,
                                            [5000, 20000], 600, 1e-15)
            if sc1p2 < best_score:
                best_score = sc1p2
                best_h = h1_pol2

    # Phase 7: Try n=600 path as alternative
    n_alt = 600
    if tleft() > 200 and best_n == n0:
        target_alt = n_alt / 2.0
        h_alt = interpolate_to(best_h, n_alt, target_alt)

        if tleft() > 150:
            h_alt_pol, sc_alt = slsqp_optimize(h_alt, n_alt, target_alt,
                                               [100, 500, 2000], 400, 1e-15)
            if sc_alt < compute_c5(h_alt):
                h_alt = h_alt_pol

        if tleft() > 100:
            h_alt_cd, c5_alt = fast_cd(h_alt, n_alt, target_alt,
                                       time_limit=min(120, tleft() * 0.5),
                                       max_sweeps=20000,
                                       n_cand_start=25, n_cand_end=80,
                                       max_no_improve=300,
                                       line_search_iter=18)
            if c5_alt < best_score:
                best_score = c5_alt
                best_h = h_alt_cd
                best_n = n_alt
                best_target = target_alt

    # Phase 8: Final SLSQP polish
    if tleft() > 60:
        h_fin, sc_fin = slsqp_optimize(best_h, best_n, best_target,
                                       [10000, 30000, 100000], 500, 1e-16)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 9: Final CD
    if tleft() > 40:
        h_cd_fin, c5_fin = fast_cd(best_h, best_n, best_target,
                                   time_limit=min(50, tleft() * 0.6),
                                   max_sweeps=10000,
                                   n_cand_start=50, n_cand_end=120,
                                   max_no_improve=200,
                                   line_search_iter=22)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_cd_fin

    # Phase 10: Structured perturbation restarts
    if tleft() > 80:
        for attempt in range(3):
            if tleft() < 30:
                break
            # Different perturbation strategies
            if attempt == 0:
                # Small random perturb
                h_pert = best_h + rng.randn(best_n) * 0.03
            elif attempt == 1:
                # Block perturbation
                h_pert = best_h.copy()
                block_size = best_n // 10
                start_b = rng.randint(0, best_n - block_size)
                h_pert[start_b:start_b + block_size] += rng.randn(block_size) * 0.08
            else:
                # Shift-based perturbation
                shift_amt = rng.randint(1, 4)
                h_pert = np.roll(best_h, shift_amt)
                h_pert[:shift_amt] = 1.0 - best_h[-shift_amt:]
                h_pert[-shift_amt:] = 1.0 - best_h[:shift_amt]

            h_pert = project_feasible(h_pert, best_target)
            h_refined, sc_ref = slsqp_optimize(h_pert, best_n, best_target,
                                               [200, 1000, 5000], 400, 1e-14)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_refined

            if tleft() > 30:
                h_cd_ref, c5_ref = fast_cd(best_h, best_n, best_target,
                                           time_limit=min(25, tleft() * 0.4),
                                           max_sweeps=5000,
                                           n_cand_start=40, n_cand_end=100,
                                           max_no_improve=150,
                                           line_search_iter=18)
                if c5_ref < best_score:
                    best_score = c5_ref
                    best_h = h_cd_ref

    # Phase 11: Final micro-polish
    if tleft() > 30:
        h_last, sc_last = slsqp_optimize(best_h, best_n, best_target,
                                         [50000, 200000], 400, 1e-16)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    if tleft() > 20:
        h_cd_last, c5_last = fast_cd(best_h, best_n, best_target,
                                     time_limit=min(15, tleft() * 0.5),
                                     max_sweeps=5000,
                                     n_cand_start=60, n_cand_end=120,
                                     max_no_improve=100,
                                     line_search_iter=24)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_cd_last

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
