import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for p in range(n):
        pos_mask = (idxs - shifts >= 0) & (idxs - shifts < n)
        neg_mask = (idxs + shifts >= 0) & (idxs + shifts < n)
        change[p, pos_mask] += (1.0 - h[idxs[pos_mask] - shifts[pos_mask]])
        change[p, neg_mask] -= h[idxs[neg_mask] + shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    idxs = np.arange(n)
    change[p, :] = 0.0
    pos_mask = (idxs - shifts >= 0) & (idxs - shifts < n)
    neg_mask = (idxs + shifts >= 0) & (idxs + shifts < n)
    change[p, pos_mask] += (1.0 - h[idxs[pos_mask] - shifts[pos_mask]])
    change[p, neg_mask] -= h[idxs[neg_mask] + shifts[neg_mask]]

def block_coord_descent(h_init, n, target, time_limit, block_size=4, max_sweeps=20000,
                        n_low_blocks=6, n_high_blocks=6):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        n_blocks = n // block_size
        block_grads = np.zeros(n_blocks)
        for b in range(n_blocks):
            block_grads[b] = np.mean(grad[b * block_size:(b + 1) * block_size])

        sorted_blocks = np.argsort(block_grads)
        low_blocks = sorted_blocks[:n_low_blocks]
        high_blocks = sorted_blocks[-n_high_blocks:]

        best_trial = c5_val
        best_move = None

        for hb in high_blocks:
            for lb in low_blocks:
                if hb == lb:
                    continue
                hi_start = hb * block_size
                hi_end = (hb + 1) * block_size
                lo_start = lb * block_size
                lo_end = (lb + 1) * block_size

                max_move = min(np.sum(h[hi_start:hi_end]),
                               np.sum(1.0 - h[lo_start:lo_end]))
                if max_move < 1e-10:
                    continue

                for frac in [0.3, 0.5, 0.7, 1.0]:
                    delta = frac * max_move / block_size
                    h_try = h.copy()
                    h_try[hi_start:hi_end] -= delta
                    h_try[lo_start:lo_end] += delta
                    h_try = np.clip(h_try, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_trial - 1e-14:
                        best_trial = sc
                        best_move = (h_try.copy(), sc)

        if best_move is not None:
            h = best_move[0]
        else:
            no_improve += 1

    return best_h, best_c5


def proximal_gradient(h_init, n, target, time_limit, n_steps=5000, initial_step=0.01):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = initial_step
    prev_h = h.copy()

    alpha_seq = [20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    steps_per_alpha = max(n_steps // len(alpha_seq), 100)

    for alpha_idx, alpha in enumerate(alpha_seq):
        for step_i in range(steps_per_alpha):
            if time.time() - start > time_limit:
                return best_h, best_c5

            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            log_sum = np.log(np.sum(np.exp(shifted)))

            weights = np.exp(shifted)
            weights /= weights.sum()

            grad = np.zeros(n)
            top_positions = np.argsort(weights)[-min(10, n_corr_max(n)):]
            for s_idx in top_positions:
                k = int(s_idx) - mid
                w = weights[s_idx]
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
            grad -= np.mean(grad)

            h_new = h - step * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            if sc < compute_c5(h) - 1e-10:
                step = min(step * 1.2, 0.1)
                prev_h = h.copy()
                h = h_new
                if sc < best_c5 - 1e-13:
                    best_c5 = sc
                    best_h = h.copy()
            else:
                step *= 0.7

    return best_h, best_c5


def n_corr_max(n):
    return 2 * n - 1


def top_k_weights(corr, alpha, k_max=10):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-30:
        idx = np.argmax(corr)
        weights = np.zeros_like(corr)
        weights[idx] = 1.0
        return weights
    weights /= total
    return weights


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_refine(h_init, n, target, time_limit, n_starts=6, noise_scales=None):
    start = time.time()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    rng = np.random.RandomState(12345)

    if noise_scales is None:
        noise_scales = [0.03, 0.06, 0.1, 0.15, 0.2, 0.3]

    for r in range(n_starts):
        if time.time() - start > time_limit:
            break
        scale = noise_scales[r % len(noise_scales)]
        h_pert = h_init + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pert.copy()

    return best_h, best_c5
