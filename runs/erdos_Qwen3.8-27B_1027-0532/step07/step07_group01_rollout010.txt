```python
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds_lp = [(0, 1)] * n + [(None, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        
        if not res.success:
            break
        
        h_new = project_feasible(res.x[:n], target)
        t_new = res.x[-1] * dx
        
        sc = compute_c5(h_new)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new.copy()
        
        h = h_new
        
        # Check for violated constraints (non-active shifts with corr > t)
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        t_val = np.max(corr_new)
        violated = np.where(corr_new > t_val + 1e-10)[0]
        # Actually, we want shifts where corr > current t from the LP
        # The LP t is res.x[-1]
        lp_t = res.x[-1]
        violated = np.where(corr_new > lp_t + 1e-6)[0]
        violated = violated[~np.isin(violated, active_arr)]
        
        if len(violated) == 0:
            break
        
        # Add most violated to active set
        active = np.concatenate([active_arr, violated[:5]])
        active = np.unique(active)
        # Limit active set size
        if len(active) > 50:
            # Keep only the most important ones
            active_corr = corr_new[active]
            active = active[np.argsort(active_corr)[-50:]]
    
    return best_h, best_c5
```

Wait, there's an issue. The LP gives a lower bound on the true objective (since we're linearizing a non-convex function from below... actually no, the linearization is a first-order approximation, not a lower bound in general).

Actually, for the function corr(k) = sum_i h[i]*(1-h[i+k]), the gradient at h gives:
corr(k; h + d) = corr(k; h) + grad(k; h)^T d + O(||d||^2)

The quadratic term is -sum_i d[i]*d[i+k], which can be positive or negative. So the linearization is not a valid underestimator or overestimator.

However, the LP approach can still work as an iterative method: solve the LP, get a new point, re-linearize, repeat. This is essentially the "successive linear programming" method used in non-linear optimization.

The key question is whether it converges. In practice, for problems like this, it often works well, especially when combined with other methods.

Let me also think about using CVXPY, which would handle the LP more cleanly:

```python
import cvxpy as cp

def solve_lp(h, n, target, active_shifts):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    idxs = np.arange(n)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    
    for k_idx in active_shifts:
        k = int(k_idx) - (n - 1)
        # corr(k) linearized at h:
        # corr_lin = corr(k; h) + sum_i grad_i * (h_new[i] - h[i])
        grad = np.zeros(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
        
        corr_k = float(np.correlate(h, 1.0 - h, mode='full')[k_idx])
        linearized = corr_k - grad @ h + grad @ h_var
        
        constraints.append(linearized <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve()
    
    if h_var.value is not None:
        h_new = project_feasible(h_var.value, target)
        return h_new, float(t_var.value)
    return None, None
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_round(h_start, n, target, alphas, maxiter=600, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    
    for alpha in alphas:
        try:
            res = minimize(lambda x: obj(x, alpha), h_cur,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

def cd_optimize(h_init, n, target, time_limit, max_sweeps=50000,
                n_cand_start=20, n_cand_end=70, max_no_improve=600,
                n_bisect=10, n_active_shifts=3):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    rng = np.random.RandomState(12345)
    
    def comp_change_row(p):
        row = np.zeros(n_corr)
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        row[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        row[neg_mask] -= h[p - shifts[neg_mask]]
        return row
    
    change = np.array([comp_change_row(p) for p in range(n)])
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            h = best_h.copy()
            for scale in [0.03, 0.08, 0.15]:
                freqs = np.fft.rfftfreq(n)
                noise_f = rng.randn(len(freqs)) * np.exp(-2.0 * freqs / n)
                noise = np.fft.irfft(noise_f * n, n=n)
                h_try = best_h + noise * scale
                h_try = project_feasible(h_try, target)
                sc_try = compute_c5(h_try)
                if sc_try < best_c5 + 0.001:
                    h = h_try
                    break
            change = np.array([comp_change_row(p) for p in range(n)])
            no_improve = 0
            continue
        
        top_idx = np.argsort(corr)[-n_active_shifts:]
        top_sum = np.sum(corr[top_idx]) + 1e-30
        
        grad = np.zeros(n)
        for si in top_idx:
            w = corr[si] / top_sum
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, max(n // 4, 10))
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                g = change[j] - change[i]
                a_d, b_d = d_lo, d_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-14:
                    continue
                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i] = comp_change_row(best_i)
            change[best_j] = comp_change_row(best_j)
    
    return best_h, best_c5

def lp_linearize(h_init, n, target, time_limit, max_iters=80):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    active = np.where(corr >= c_max - 1e-8)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    active = active[:30]
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        grads = np.zeros((len(active), n))
        for ai, k_idx in enumerate(active):
            k = int(k_idx) - mid
            g = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v1] += (1.0 - h[idxs[v1] - k])
            g[v2] -= h[idxs[v2] + k]
            grads[ai] = g
        
        corr_cur = np.correlate(h, 1.0 - h, mode='full')
        b = corr_cur[active] - grads @ h
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((len(active), n + 1))
        for ai in range(len(active)):
            A_ub[ai, :n] = grads[ai]
            A_ub[ai, -1] = -1.0
        b_ub = -b + 1e-12
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds_lp = [(0, 1)] * n + [(-1e10, 1e10)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs')
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        h_new = project_feasible(res.x[:n], target)
        sc = compute_c5(h_new)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new.copy()
        
        h = h_new
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        t_lp = res.x[-1]
        
        violated = np.where(corr_new > t_lp + 1e-5)[0]
        violated = violated[~np.isin(violated, active)]
        
        if len(violated) == 0:
            break
        
        active = np.concatenate([active, violated[:10]])
        active = np.unique(active)
        active_corr = corr_new[active]
        if len(active) > 60:
            active = active[np.argsort(active_corr)[-60:]]
        active = np.sort(active)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)
```
