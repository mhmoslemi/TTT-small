            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(top_idx)
        grad -= np.mean(grad)

        progress = min(float(sweep) / 2000.0, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)

        low_set = sorted_idx[:ncand]

                j = int(low_set[jj])

                    continue

                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-13:
                    continue

                a = d_lo
                b = d_hi

                        f_prime = float(np.max(g[active_mask]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b = m
                    else:
                        a = m

                if abs(d_star) < 1e-13:
                    continue

                    best_j = j

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        h = np.clip(h, 0.0, 1.0)
    return h

    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 4000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.85
                if lr < 1e-7:
                    break

        top_idx = np.argsort(corr)[-n_active:]
        c_max = corr[top_idx[0]]
        weights = np.exp(-40.0 * (c_max - corr[top_idx]))
        ws = np.sum(weights)
        if ws > 0:
            weights /= ws
        else:
            weights = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(weights, top_idx):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-16:
            h = h_try
            cur_c5 = sc
        else:
            lr *= 0.5
            if lr < 1e-9:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 12.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    # Phase 1: Quick polish at original resolution
    if tleft() > 25:
        h_p, sc_p = slsqp_round(best_h, best_n, best_target,
                                [50, 200, 800], 400, 1e-14)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p

    if tleft() > 15:
        h_cd0, c5_0 = fast_cd(best_h, best_n, best_target,
                              time_limit=min(15, tleft() * 0.8),
                              n_cand_start=20, n_cand_end=45,
                              max_no_improve=150, n_active=4, n_bisect=12)
        if c5_0 < best_score:
            best_score = c5_0
            best_h = h_cd0

    # Phase 2: Main optimization at n=800
    n1 = 800
    if tleft() > 250:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 220:
            h1_p, sc1_p = slsqp_round(h1, n1, target1,
                                       [100, 400], 300, 1e-14)
            if sc1_p < sc1:
                h1 = h1_p
                sc1 = sc1_p

        if tleft() > 130:
            h1_cd, c5_1 = fast_cd(h1, n1, target1,
                                  time_limit=min(180, tleft() * 0.55),
                                  n_cand_start=25, n_cand_end=65,
                                  max_no_improve=300, n_active=5, n_bisect=13)
            if c5_1 < best_score:
                best_score = c5_1
                best_h = h1_cd
                best_n = n1
                best_target = target1
            h1 = best_h

        if tleft() > 80:
            h1_sg, c5_1sg = subgrad_method(h1, n1, target1,
                                           time_limit=min(50, tleft() * 0.5),
                                           base_lr=0.02, n_active=8)
            if c5_1sg < best_score:
                best_score = c5_1sg
                best_h = h1_sg
                best_n = n1
                best_target = target1
            h1 = best_h

        if tleft() > 60:
            h1_p2, sc1_p2 = slsqp_round(h1, n1, target1,
                                         [1500, 5000], 600, 1e-14)
            if sc1_p2 < best_score:
                best_score = sc1_p2
                best_h = h1_p2

        if tleft() > 100:
            h1_cd2, c5_1b = fast_cd(best_h, n1, best_target,
                                    time_limit=min(130, tleft() * 0.55),
                                    n_cand_start=30, n_cand_end=75,
                                    max_no_improve=250, n_active=6, n_bisect=13)
            if c5_1b < best_score:
                best_score = c5_1b
                best_h = h1_cd2
                best_n = n1
                best_target = best_target

        if tleft() > 50:
            h1_sg2, c5_1sg2 = subgrad_method(best_h, best_n, best_target,
                                             time_limit=min(40, tleft() * 0.5),
                                             base_lr=0.01, n_active=10)
            if c5_1sg2 < best_score:
                best_score = c5_1sg2
                best_h = h1_sg2

        if tleft() > 50:
            h1_p3, sc1_p3 = slsqp_round(best_h, best_n, best_target,
                                         [10000, 30000], 500, 1e-14)
            if sc1_p3 < best_score:
                best_score = sc1_p3
                best_h = h1_p3

    # Phase 3: Try n=1200
    n2 = 1200
    if tleft() > 350:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 280:
            h2_p, sc2_p = slsqp_round(h2, n2, target2,
                                       [100, 500], 250, 1e-14)
            if sc2_p < sc2:
                h2 = h2_p
                sc2 = sc2_p

        if tleft() > 200:
            h2_cd, c5_2 = fast_cd(h2, n2, target2,
                                  time_limit=min(250, tleft() * 0.5),
                                  n_cand_start=30, n_cand_end=70,
                                  max_no_improve=280, n_active=5, n_bisect=12)
            if c5_2 < best_score:
                best_score = c5_2
                best_h = h2_cd
                best_n = n2
                best_target = target2

        if tleft() > 80:
            h2_sg, c5_2sg = subgrad_method(h2, n2, target2,
                                           time_limit=min(60, tleft() * 0.5),
                                           base_lr=0.012, n_active=6)
            if c5_2sg < best_score:
                best_score = c5_2sg
                best_h = h2_sg
                best_n = n2
                best_target = target2

        if tleft() > 60:
            h2_p2, sc2_p2 = slsqp_round(best_h, best_n, best_target,
                                         [2000, 8000], 400, 1e-14)
            if sc2_p2 < best_score:
                best_score = sc2_p2
                best_h = h2_p2

        if tleft() > 80:
            h2_cd2, c5_2b = fast_cd(best_h, best_n, best_target,
                                    time_limit=min(100, tleft() * 0.5),
                                    n_cand_start=35, n_cand_end=80,
                                    max_no_improve=200, n_active=6, n_bisect=12)
            if c5_2b < best_score:
                best_score = c5_2b
                best_h = h2_cd2
                best_n = n2
                best_target = n2 / 2.0

    # Phase 4: Final refinement
    if tleft() > 50:
        h_fin, c5_fin = fast_cd(best_h, best_n, best_target,
                                time_limit=min(45, tleft() * 0.7),
                                n_cand_start=40, n_cand_end=85,
                                max_no_improve=180, n_active=8, n_bisect=13)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    if tleft() > 35:
        h_pol, sc_pol = slsqp_round(best_h, best_n, best_target,
                                     [5000, 20000], 400, 1e-14)
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    if tleft() > 25:
        rng = np.random.RandomState(seed)
        for scale in [0.02, 0.05, 0.10]:
            if tleft() < 12:
                break
            h_pert = best_h + rng.randn(best_n) * scale
            h_pert = project_feasible(h_pert, best_target)
            h_ref, sc = slsqp_round(h_pert, best_n, best_target,
                                     [500, 3000], 300, 1e-13)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 15:
        h_final_cd, c5_last = fast_cd(best_h, best_n, best_target,
                                      time_limit=min(12, tleft() * 0.7),
                                      n_cand_start=50, n_cand_end=100,
                                      max_no_improve=80, n_active=8, n_bisect=12)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_final_cd

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
