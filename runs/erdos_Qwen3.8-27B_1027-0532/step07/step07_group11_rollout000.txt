            bc = cv
            bh = h.copy()
            ni = 0
        else:
            ni += 1
            if ni > maxni:
                temp *= 0.9
                if temp < 1e-6:
                    break
        
        bs = block_sizes[np.random.randint(len(block_sizes))]
        if bs >= n:
            continue
        
        # Pick a random block
        start = np.random.randint(0, n - bs + 1)
        block = np.arange(start, start + bs)
        
        # Pick a destination block
        start2 = np.random.randint(0, n - bs + 1)
        block2 = np.arange(start2, start2 + bs)
        
        if np.any(np.isin(block, block2)):
            continue
        
        # Move: transfer mass from block to block2
        total = np.sum(h[block])
        capacity = np.sum(1.0 - h[block2])
        transfer = min(total, capacity) * np.random.uniform(0.3, 0.8)
        if transfer < 1e-10:
            continue
        
        h_try = h.copy()
        h_try[block] -= transfer / bs
        h_try[block2] += transfer / bs
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, tgt)
        
        sc_try = compute_c5(h_try)
        delta = sc_try - cv
        if delta < 0 or np.random.random() < np.exp(-delta / (temp * dx)):
            h = h_try
        temp *= decay
    
    return bh, bc

```python
def best_response(h0, n, tgt, tl, K=3, step=0.02):
    st = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    h = h0.copy()
    bh = h.copy()
    bc = compute_c5(h)
    eta = step
    
    while time.time() - st < tl:
        corr = np.correlate(h, 1.0 - h, mode='full')
        cv = float(np.max(corr)) * dx
        if cv < bc - 1e-14:
            bc = cv
            bh = h.copy()
            eta = min(eta * 1.02, 0.1)
        else:
            eta *= 0.95
        
        # Top K shifts
        tk = np.argsort(corr)[-K:]
        
        # For each top shift, find the best pair (i,j) to reduce it
        dir_sum = np.zeros(n)
        for si in tk:
            k = int(si) - mid
            # Gradient for this shift
            grad = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
            grad -= np.mean(grad)
            dir_sum += grad / K
        
        # Take a step
        h_try = h - eta * dir_sum
        h_try = project_feasible(h_try, tgt)
        sc = compute_c5(h_try)
        if sc < cv:
            h = h_try
        # else: reduce eta, already handled
    
    return bh, bc

def hybrid_es(h0, n, tgt, tl, P=15, max_gen=500):
    st = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(42)
    
    # Initialize population
    pop = [h0.copy() for _ in range(P)]
    scores = [compute_c5(pop[i]) for i in range(P)]
    
    # Diversify initial population
    for i in range(1, P):
        pop[i] = h0 + rng.randn(n) * 0.03 * (i / P)
        pop[i] = project_feasible(pop[i], tgt)
        scores[i] = compute_c5(pop[i])
    
    best_idx = int(np.argmin(scores))
    bh = pop[best_idx].copy()
    bc = scores[best_idx]
    
    for gen in range(max_gen):
        if time.time() - st > tl:
            break
        
        # Sort by score
        order = np.argsort(scores)
        elites = order[:P//2]
        
        # Generate offspring
        for i in range(P//2, P):
            # Parent selection: tournament
            p1 = pop[order[rng.randint(P//2)]]
            p2 = pop[order[rng.randint(P//2)]]
            
            # Crossover
            alpha = rng.uniform(0.3, 0.7)
            child = alpha * p1 + (1 - alpha) * p2
            
            # Gradient-guided mutation
            corr = np.correlate(child, 1 - child, mode='full')
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - child[idxs[v1] - k_star])
            grad[v2] -= child[idxs[v2] + k_star]
            grad -= np.mean(grad)
            
            child = child - 0.01 * grad + rng.randn(n) * 0.01
            child = project_feasible(child, tgt)
            
            pop[i] = child
            scores[i] = compute_c5(child)
        
        # Update best
        bi = int(np.argmin(scores))
        if scores[bi] < bc:
            bc = scores[bi]
            bh = pop[bi].copy()
        
        # Replace worst with best (elitism)
        pop[order[-1]] = pop[bi].copy()
        scores[order[-1]] = scores[bi]
    
    return bh, bc
```

One more optimization for CD: Instead of iterating over all (i,j) pairs in the nested loop, I can vectorize the inner loop. For a fixed i, the "gain" for all j is:
gain_j = max_k (corr_k + delta * (change[j,k] - change[i,k]))

This is a matrix operation: for all j simultaneously, compute the max of (corr + delta * (change[j] - change[i])) over k. But delta depends on j (it's the clamped value), so we can't fully vectorize.

However, we can use a first-order approximation: for small delta, the change in max is approximately delta * (change[j,k*] - change[i,k*]) where k* is the current argmax. This gives a fast screening step.

Let me implement this:
1. Compute k* = argmax(corr)
2. For each candidate (i,j), estimate the gain as delta * (change[j,k*] - change[i,k*])
3. Only do the full binary search for the top few (i,j) pairs by estimated gain
4. This reduces the number of full evaluations

This is a huge speedup! Instead of O(n_cand² * 16 * n_corr) per sweep, it's O(n_cand² + n_cand * 16 * n_corr).

For n_cand=50 and n_corr=1600 (n=800):
- Without: 50*50*16*1599 ≈ 64M
- With: 50*50 + 50*16*1599 ≈ 1.3M

    nc = 2 * n - 1
    ch, shifts = build_change(h0, n)
    idxs = np.arange(n)

            break

        cv = float(np.max(corr)) * dx

            bc = cv
            bh = h.copy()
            ni = 0
        else:
            ni += 1
        
        # Top-k shifts for gradient
        tk = np.argsort(corr)[-nk:]
        grad = np.zeros(n)
        for si in tk:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]

        grad -= np.mean(grad)

        prog = min(float(sw) / max(maxsw, 1), 1.0)
        ncc = int(nc_lo + (nc_hi - nc_lo) * prog)
        ncc = min(ncc, n // 3)

        lo = sidx[:ncc]

        k_star = int(np.argmax(corr))

        # gain ≈ delta_max * (ch[j, k_star] - ch[i, k_star])
        # where delta_max = min(h[i], 1-h[j])

        # For a given i, the best j maximizes:
        #   min(h[i], 1-h[j]) * (ch[j, k_star] - ch[i, k_star])
        # subject to ch[j, k_star] - ch[i, k_star] > 0

        bi = bj = -1

            gains = (cj_star - ci_star)  # shape (nc_lo,)
            dmax = np.minimum(h[i], 1.0 - h[lo])  # max possible transfer

            est_gain = np.where(gains > 0, est_gain, -np.inf)
            
            if np.all(est_gain <= -1e10):
                continue

            top_js = lo[np.argsort(-est_gain)[:5]]
            
            for j in top_js:
                j = int(j)
                if j == i:
                    continue
                dlo = max(h[i] - 1.0, -h[j])
                dhi = min(h[i], 1.0 - h[j])
                if dhi - dlo < 1e-12:
