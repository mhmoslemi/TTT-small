import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def pnorm_gradient(h, p):
    """Compute gradient of dx * (sum corr^p)^(1/p) with respect to h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    safe_corr = np.maximum(corr, 1e-30)
    base = np.sum(safe_corr**p) ** (1.0/p - 1.0)
    coeff = p * base * safe_corr**(p-1.0)
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        c_val = coeff[s + n - 1]
        if abs(c_val) < 1e-20:
            continue
        if s >= 0:
            lo1, hi1 = 0, n - s
        else:
            lo1, hi1 = -s, n
        if hi1 > lo1:
            grad[lo1:hi1] += c_val * (1.0 - h[lo1+s:hi1+s])
        
        if s >= 0:
            lo2, hi2 = s, n
        else:
            lo2, hi2 = 0, n + s
        if hi2 > lo2:
            grad[lo2:hi2] -= c_val * h[lo2-s:hi2-s]
    
    return dx * grad

def proj_grad_momentum(h_init, n, target, p, time_limit, 
                        lr_base=0.01, momentum=0.9, max_iter=50000):
    """Projected gradient descent with momentum on p-norm objective."""
    dx = 2.0 / n
    start = time.time()
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    consecutive_no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 2000:
            break
        
        grad = pnorm_gradient(h, p)
        
        # Backtracking line search for step size
        # Use a fixed learning rate with decay
        lr = lr_base / (1.0 + 0.001 * it)
        
        # Momentum step
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        # Evaluate

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        h = h_new
    
    return best_h, best_c5

Let me also consider: what if I use a **different parameterization**? Instead of optimizing h directly, what if I optimize the "deviation from 1/2"?

Let h = 1/2 + u, where sum(u) = 0.
Then corr[k] = Σ_i (1/2+u[i]) * (1/2-u[i+s]) = Σ_i [1/4 - (1/2)u[i+s] + (1/2)u[i] - u[i]u[i+s]]
= n/4 - (1/2)Σ_i u[i+s] + (1/2)Σ_i u[i] - Σ_i u[i]u[i+s]
= n/4 - (1/2)*0 + (1/2)*0 - Σ_i u[i]u[i+s]  (since sum(u)=0)
= n/4 - Σ_i u[i]u[i+s]

So corr[k] = n/4 - Σ_i u[i]*u[i+s]

And dx = 2/n, so dx * corr[k] = (2/n) * (n/4 - Σ_i u[i]*u[i+s]) = 1/2 - (2/n) * Σ_i u[i]*u[i+s]

We want to minimize max_k dx * corr[k] = max_k (1/2 - (2/n) * Σ_i u[i]*u[i+s])
= 1/2 - (2/n) * min_k Σ_i u[i]*u[i+s]

So minimizing max_k corr[k] is equivalent to maximizing min_k Σ_i u[i]*u[i+s].

This is a different perspective! We want the autocorrelation of u to be as "flat" as possible (or rather, the minimum to be as high as possible).

The constraint is: sum(u) = 0, and -1/2 ≤ u[i] ≤ 1/2.

This is an interesting reformulation. The objective becomes: maximize min_k autocorr_u[k]
where autocorr_u[k] = Σ_i u[i]*u[i+k].

This is like trying to make u have a flat autocorrelation (like a Zadoff-Chu sequence or a Barker code).

For binary sequences (u[i] ∈ {-1/2, 1/2}), this is exactly the problem of finding sequences with low autocorrelation sidelobes. The best known binary sequences have sidelobes around n/7 or so.

For continuous u[i] ∈ [-1/2, 1/2], we can do better.

This reformulation suggests that the optimal u should be "pseudo-noise" like - having a flat autocorrelation. This is a useful intuition but doesn't directly give an algorithm.

However, it does suggest an alternative optimization approach:
- Maximize the minimum of the autocorrelation vector of u
- Use a smooth approximation: maximize (min_k autocorr_u[k]) ≈ (1/q) * log(Σ_k exp(-q * autocorr_u[k]))
  (negative because we want to maximize the minimum)
  or: minimize (Σ_k exp(-q * autocorr_u[k]))^(1/q) (q-norm of exp(-q*autocorr))

This might give better gradient behavior than the original formulation.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / len(x)
    safe = np.maximum(corr, 1e-30)
    return dx * (np.sum(safe**p))**(1.0/p)

def pnorm_gradient_direct(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    safe_corr = np.maximum(corr, 1e-30)
    base = np.sum(safe_corr**p) ** (1.0/p - 1.0)
    coeff = np.float64(p) * base * safe_corr**(p-1.0)
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        c_val = coeff[s + n - 1]
        if abs(c_val) < 1e-25:
            continue
        if s >= 0:
            lo1, hi1 = 0, n - s
        else:
            lo1, hi1 = -s, n
        if hi1 > lo1:
            grad[lo1:hi1] += c_val * (1.0 - h[lo1+s:hi1+s])
        if s >= 0:
            lo2, hi2 = s, n
        else:
            lo2, hi2 = 0, n + s
        if hi2 > lo2:
            grad[lo2:hi2] -= c_val * h[lo2-s:hi2-s]
    return dx * grad

def proj_grad_momentum(h_init, n, target, p, time_limit, lr_init=0.005,
                       momentum=0.85, max_iter=100000, eval_every=5):
    dx = 2.0 / n
    start = time.time()
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    it = 0
    while it < max_iter:
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break
        grad = pnorm_gradient_direct(h, p)
        lr = lr_init / (1.0 + 0.0005 * it)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        h = h_new
        it += 1
        if it % eval_every == 0:
            c5_val = compute_c5(h)
            if c5_val < best_c5 - 1e-15:
                best_c5 = c5_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
    return best_h, best_c5

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def cd_refine(h_init, n, target, time_limit, max_sweeps=30000,
              n_cand_start=30, n_cand_end=80, max_no_improve=400,
              n_shifts=1):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > max_no_improve:
            break
        if n_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
            grad -= np.mean(grad)
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            c_max = corr[np.argmax(corr)]
            wts = np.exp(-5.0 * (c_max - corr[top_idx]) / max(c_max, 1e-10))
            wts /= np.sum(wts)
            grad = np.zeros(n)
            for wi, s_idx in enumerate(top_idx):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += wts[wi] * (1.0 - h[idxs[v1] - k])
                grad[v2] -= wts[wi] * h[idxs[v2] + k]
            grad -= np.mean(grad)
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 2)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = d_lo
                b_d = d_hi
                for _ in range(14):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_pri = np.max(g[active])
                    else:
                        f_pri = 0.0
                    if f_pri > 1e-13:
                        b_d = m_d
                    else:
                        a_d = m_d
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) < 1e-13:
                    continue
                approx = np.max(corr + d_star * g) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = d_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha,
                 time_limit, start_time):
    dx_val = 2.0 / n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > start_time + time_limit:
            break
        def obj_fn(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx_val
        def obj_jac(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            weights = np.exp(shifted) / np.sum(np.exp(shifted))
            n_local = len(x)
            mid = n_local - 1
            grad = np.zeros(n_local)
            for k_idx in range(2 * n_local - 1):
                wk = weights[k_idx]
                if wk < 1e-15:
                    continue
                k = k_idx - mid
                v1 = (np.arange(n_local) - k >= 0) & (np.arange(n_local) - k < n_local)
                v2 = (np.arange(n_local) + k >= 0) & (np.arange(n_local) + k < n_local)
                grad[v1] += wk * (1.0 - x[np.arange(n_local)[v1] - k])
                grad[v2] -= wk * x[np.arange(n_local)[v2] + k]
            return dx_val * grad
        def cons_fun(x):
            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)
        try:
            res = minimize(obj_fn, h_cur, jac=obj_jac, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
