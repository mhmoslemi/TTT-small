                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx_local
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h[best_i] = max(0.0, min(1.0, h[best_i]))
            h[best_j] = max(0.0, min(1.0, h[best_j]))
            if sweep % 200 == 0:
                h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    h = project_feasible(h, target)
    final_score = compute_c5(h)
    if final_score < best_c5:
        best_c5 = final_score
        best_h = h.copy()
    return best_h, best_c5

def subspace_optimize(h_init, n, target, time_limit, n_subsets, subset_size, start_time):
    global _g_h_full, _g_idx, _g_sub_target, _g_sub_alpha
    rng = np.random.RandomState(12345)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    _g_sub_alpha = 2000.0

    for s in range(n_subsets):
        if time.time() - start_time > time_limit:
            break
        idx = rng.choice(n, subset_size, replace=False)
        idx = np.sort(idx)
        h_sub_orig = h[idx].copy()
        _g_sub_target = float(np.sum(h_sub_orig))
        _g_h_full = h
        _g_idx = idx
        bounds_sub = [(0.0, 1.0)] * subset_size
        cons_sub = [{'type': 'eq', 'fun': _subspace_con_fun, 'jac': _subspace_con_jac}]
        try:
            res = minimize(_subspace_obj, h_sub_orig, method='SLSQP', bounds=bounds_sub,
                         constraints=cons_sub, options={'maxiter': 200, 'ftol': 1e-12})
            h_new = h.copy()
            h_new[idx] = res.x
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                h = h_new
        except Exception:
            pass
    return h, best_c5

def sa_block_optimize(h_init, n, target, time_limit, max_steps, t_start, t_end, n_cand):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(99)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 800:
            break

        progress = min(step / max(float(max_steps), 1), 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            consecutive_no_improve += 1
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            consecutive_no_improve += 1
            continue

        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            consecutive_no_improve += 1
            continue

        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h[i] = max(0.0, min(1.0, h[i]))
            h[j] = max(0.0, min(1.0, h[j]))
            if step % 100 == 0:
                h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

            if approx_new_c5 < best_c5 - 1e-14:
                best_c5 = approx_new_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

dx_local = 0.01

def run(seed=42, budget_s=1300, **kwargs):
    global dx_local
    run_start = time.time()

    def tleft():
        return budget_s - 15.0 - (time.time() - run_start)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Interpolate to n=1000 and do initial CD
    n1 = 1000
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)
    dx_local = 2.0 / n1

    if tleft() > 200:
        h_cd1, c5_1 = cd_optimize(h1, n1, target1, min(150, tleft() * 0.7),
                                   max_sweeps=30000, n_cand_start=40, n_cand_end=80,
                                   max_no_improve=300, n_shifts=1)
        if c5_1 < best_score:
            best_score = c5_1
            best_h = h_cd1

    # Phase 2: Subspace SLSQP at n=1000
    if tleft() > 80:
        h_sub, c5_sub = subspace_optimize(best_h, n1, target1,
                                           min(60, tleft() * 0.8), 40, 15, run_start)
        if c5_sub < best_score:
            best_score = c5_sub
            best_h = h_sub

    # Phase 3: More CD at n=1000 with multi-shift
    if tleft() > 150:
        h_cd2, c5_2 = cd_optimize(best_h, n1, target1, min(120, tleft() * 0.7),
                                   max_sweeps=20000, n_cand_start=50, n_cand_end=90,
                                   max_no_improve=250, n_shifts=3)
        if c5_2 < best_score:
            best_score = c5_2
            best_h = h_cd2

    # Phase 4: SA for exploration
    if tleft() > 80:
        h_sa, c5_sa = sa_block_optimize(best_h, n1, target1,
                                         min(70, tleft() * 0.8), 25000,
                                         0.03, 0.0002, 50)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 5: Interpolate to n=600 and SLSQP
    if tleft() > 120:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_pol, sc = slsqp_round(h2, n2, target2, [500, 2000, 10000],
                                 800, min(100, tleft() * 0.8), run_start)
        # Interpolate back to n=1000
        h_back = interpolate_to(h_pol, n1, target1)
        sc_back = compute_c5(h_back)
        if sc_back < best_score:
            best_score = sc_back
            best_h = h_back
        elif sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: More CD at n=1000
    if tleft() > 150:
        h_cd3, c5_3 = cd_optimize(best_h, n1, target1, min(130, tleft() * 0.7),
                                   max_sweeps=25000, n_cand_start=55, n_cand_end=100,
                                   max_no_improve=200, n_shifts=1)
        if c5_3 < best_score:
            best_score = c5_3
            best_h = h_cd3

    # Phase 7: Subspace SLSQP with larger subsets
    if tleft() > 80:
        h_sub2, c5_sub2 = subspace_optimize(best_h, n1, target1,
                                             min(60, tleft() * 0.8), 35, 20, run_start)
        if c5_sub2 < best_score:
            best_score = c5_sub2
            best_h = h_sub2

    # Phase 8: Try n=1200
    if tleft() > 200:
        n3 = 1200
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        dx_local = 2.0 / n3
        h_cd4, c5_4 = cd_optimize(h3, n3, target3, min(170, tleft() * 0.8),
                                   max_sweeps=20000, n_cand_start=35, n_cand_end=70,
                                   max_no_improve=300, n_shifts=1)
        if c5_4 < best_score:
            best_score = c5_4
            best_h = h_cd3 if False else h_cd4
            # Keep at n=1200 for further refinement
            n1 = n3
            target1 = target3

    # Phase 9: Final CD
    if tleft() > 80:
        h_cd5, c5_5 = cd_optimize(best_h, len(best_h), len(best_h)/2.0,
                                   min(70, tleft() * 0.7),
                                   max_sweeps=15000, n_cand_start=50, n_cand_end=90,
                                   max_no_improve=150, n_shifts=2)
        if c5_5 < best_score:
            best_score = c5_5
            best_h = h_cd5

    # Phase 10: Final subspace
    if tleft() > 40:
        nn = len(best_h)
        h_sub3, c5_sub3 = subspace_optimize(best_h, nn, nn/2.0,
                                             min(35, tleft() * 0.8), 20, 15, run_start)
        if c5_sub3 < best_score:
            best_score = c5_sub3
            best_h = h_sub3

    # Phase 11: Final SLSQP at lower resolution
    if tleft() > 50:
        n_final = min(600, len(best_h))
        target_final = n_final / 2.0
        h_f = interpolate_to(best_h, n_final, target_final)
        h_pol2, sc2 = slsqp_round(h_f, n_final, target_final, [5000, 20000],
                                   500, min(40, tleft() * 0.7), run_start)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            # But we want to return at higher n if possible
            # Interpolate back up
            h_up = interpolate_to(h_pol2, len(best_h), len(best_h)/2.0)
            sc_up = compute_c5(h_up)
            if sc_up < best_score:
                best_score = sc_up
                best_h = h_up
            else:
                best_score = sc2
                best_h = h_pol2

    # Final projection and evaluation
    nn = len(best_h)
    final_h = project_feasible(best_h, nn / 2.0)
    final_score = compute_c5(final_h)
    if final_score < best_score:
        best_score = final_score

    return final_h, best_score, nn
