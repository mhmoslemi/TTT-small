def compute_c5(h):

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    """Precompute the change in correlation for each coordinate."""
    n_corr = 2 * n - 1

    change = np.zeros((n, n_corr))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def cd_optimize(h_init, n, target, time_limit, start_time,

    consecutive_no_improve = 0

def momentum_gradient(h_init, n, target, time_limit, start_time,
                      alphas=None, k_tops=None):

        alphas = [50, 100, 200, 500, 1000, 3000, 8000]
    if k_tops is None:
        k_tops = [5, 10, 15, 20, 30, 50, 100]

    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for phase, (alpha, k_top) in enumerate(zip(alphas, k_tops)):
        if time.time() - start_time > time_limit:
            break

        lr = 2.0 / n
        momentum = 0.80 + 0.02 * phase
        step = 0
        max_steps = 2000
        patience = 200
        no_improve = 0

        while step < max_steps:
            if time.time() - start_time > time_limit:
                break

            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_val = float(np.max(corr)) * dx

            topk_idx = np.argsort(corr)[-k_top:]
            c_max = corr[topk_idx[0]]
            shifted = alpha * (corr[topk_idx] - c_max)
            shifted = np.clip(shifted, -500, 0)
            exp_vals = np.exp(shifted)
            weights = exp_vals / exp_vals.sum()

            grad = np.zeros(n)
            for wi, ki in enumerate(topk_idx):
                m = int(ki) - mid
                w = weights[wi]
                v1m = (idxs + m >= 0) & (idxs + m < n)
                if np.any(v1m):
                    grad[v1m] += w * (1.0 - h[idxs[v1m] + m])
                v2m = (idxs - m >= 0) & (idxs - m < n)
                if np.any(v2m):
                    grad[v2m] -= w * h[idxs[v2m] - m]

            grad -= np.mean(grad)

            v = momentum * v + lr * grad
            h_new = h - v
            h_new = project_feasible(h_new, target)

            c5_new = compute_c5(h_new)

            if c5_new > c5_val + 0.003:
                lr *= 0.7
                v *= 0.5
                step += 1
                continue

            h = h_new
            step += 1

            if c5_new < best_c5 - 1e-10:
                best_c5 = c5_new
                best_h = h.copy()

            else:
                no_improve += 1

            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

            if approx_new_c5 < best_c5 - 1e-14:
                best_c5 = approx_new_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def perturb_structured(h, n, target, rng, mode=0):
    if mode == 0:
        scale = 0.03
        h_pert = h + rng.randn(n) * scale
    elif mode == 1:
        freq = rng.randint(2, 10)
        phase = rng.rand() * 2 * np.pi
        x = np.linspace(0, 1, n)
        h_pert = h + 0.02 * np.sin(2 * np.pi * freq * x + phase)
    else:
        block_size = n // rng.randint(4, 12)
        n_blocks = n // block_size
        block_deltas = rng.randn(n_blocks) * 0.02
        h_pert = h.copy()
        for b in range(n_blocks):
            lo = b * block_size
            hi = lo + block_size
            h_pert[lo:hi] += block_deltas[b]
    return project_feasible(h_pert, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick CD on initial grid to warm up
    if tleft() > 60:
        h1, c51 = cd_optimize(best_h, n0, target0,
                              time_limit=min(40, tleft() * 0.4),
                              start_time=start_time,
                              n_cand_start=25, n_cand_end=50,
                              max_sweeps=8000, max_no_improve=150,
                              rng=rng)
        if c51 < best_score:
            best_score = c51
            best_h = h1

    # Phase 2: Interpolate to n=800 and CD
    if tleft() > 250:
        n1 = 800
        target1 = n1 / 2.0
        h_interp = interpolate_to(best_h, n1, target1)

        h2, c52 = cd_optimize(h_interp, n1, target1,
                              time_limit=min(100, tleft() * 0.35),
                              start_time=start_time,
                              n_cand_start=40, n_cand_end=75,
                              max_sweeps=12000, max_no_improve=250,
                              rng=rng)
        if c52 < best_score:
            best_score = c52
            best_h = h2

        # Momentum gradient refinement at n=800
        if tleft() > 80:
            h3, c53 = momentum_gradient(best_h, n1, target1,
                                        time_limit=min(50, tleft() * 0.3),
                                        start_time=start_time)
            if c53 < best_score:
                best_score = c53
                best_h = h3

    # Phase 3: Multi-active CD
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h4, c54 = multi_active_cd(best_h, n_cur, target_cur,
                                  time_limit=min(50, tleft() * 0.4),
                                  start_time=start_time,
                                  n_cand=45, n_shifts=4,
                                  max_sweeps=8000, max_no_improve=200)
        if c54 < best_score:
            best_score = c54
            best_h = h4

    # Phase 4: SA escape
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h5, c55 = sa_refine(best_h, n_cur, target_cur,
                            time_limit=min(50, tleft() * 0.4),
                            start_time=start_time,
                            max_steps=15000, t_start=0.02, t_end=0.0005,
                            n_cand=50, seed=seed + 1)
        if c55 < best_score:
            best_score = c55
            best_h = h5

    # Phase 5: Try n=1200
    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h_interp2 = interpolate_to(best_h, n2, target2)

        h6, c56 = cd_optimize(h_interp2, n2, target2,
                              time_limit=min(100, tleft() * 0.35),
                              start_time=start_time,
                              n_cand_start=35, n_cand_end=65,
                              max_sweeps=8000, max_no_improve=200,
                              rng=rng)
        if c56 < best_score:
            best_score = c56
            best_h = h6

        if tleft() > 60:
            h7, c57 = momentum_gradient(best_h, n2, target2,
                                        time_limit=min(40, tleft() * 0.3),
                                        start_time=start_time,
                                        alphas=[100, 300, 1000, 3000, 8000],
                                        k_tops=[10, 20, 40, 60, 100])
            if c57 < best_score:
                best_score = c57
                best_h = h7

    # Phase 6: Structured perturbations + CD
    if tleft() > 120:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for mode in range(3):
            if tleft() < 40:
                break
            h_pert = perturb_structured(best_h, n_cur, target_cur, rng, mode=mode)
            sc_pert = compute_c5(h_pert)
            h8, c58 = cd_optimize(h_pert, n_cur, target_cur,
                                  time_limit=min(30, tleft() * 0.25),
                                  start_time=start_time,
                                  n_cand_start=40, n_cand_end=70,
                                  max_sweeps=6000, max_no_improve=150,
                                  rng=rng)
            if c58 < best_score:
                best_score = c58
                best_h = h8

    # Phase 7: Final momentum + CD on best
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h9, c59 = momentum_gradient(best_h, n_cur, target_cur,
                                    time_limit=min(35, tleft() * 0.4),
                                    start_time=start_time,
                                    alphas=[200, 500, 2000, 8000, 20000],
                                    k_tops=[15, 30, 50, 80, 120])
        if c59 < best_score:
            best_score = c59
            best_h = h9

    # Phase 8: Final CD polish
    if tleft() > 25:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h10, c510 = cd_optimize(best_h, n_cur, target_cur,
                                time_limit=min(20, tleft() * 0.6),
                                start_time=start_time,
                                n_cand_start=50, n_cand_end=80,
                                max_sweeps=5000, max_no_improve=100,
                                rng=rng)
        if c510 < best_score:
            best_score = c510
            best_h = h10

    # Final SA burst if time remains
    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h11, c511 = sa_refine(best_h, n_cur, target_cur,
                              time_limit=min(20, tleft() * 0.5),
                              start_time=start_time,
                              max_steps=8000, t_start=0.01, t_end=0.0003,
                              n_cand=40, seed=seed + 42)
        if c511 < best_score:
            best_score = c511
            best_h = h11

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
