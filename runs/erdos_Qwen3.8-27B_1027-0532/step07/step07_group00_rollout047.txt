    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 0

    beta = 0.9  # momentum

        if time.time() - start > time_limit:
            break

        y = h + beta * v
        y = np.clip(y, 0.0, 1.0)
        
        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_val = float(np.max(corr)) * dx

            best_h = y.copy()

        top_idx = np.argsort(corr)[-top_k:]

    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def exhaustive_pair_cd(h_init, n, target, time_limit, n_cand=80, max_sweeps=15000,
                       max_no_improve=250, weight_beta=6.0, n_top_shifts=12):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        c_max = float(np.max(corr))
        top_idx = np.argsort(corr)[-n_top_shifts:]

        weights = np.exp(weight_beta * (corr[top_idx] - c_max) / max(c_max, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, s_idx in zip(weights, top_idx):
            if w < 1e-7:
                continue
            k = int(s_idx) - mid
            g = compute_grad_for_shift(h, k, n)
            grad += w * g
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        shifts_arr = np.arange(-(n - 1), n)

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                pos_j = (j + shifts_arr >= 0) & (j + shifts_arr < n)
                neg_j = (j - shifts_arr >= 0) & (j - shifts_arr < n)
                ch_j = np.zeros(n_corr)
                ch_j[pos_j] += (1.0 - h[j + shifts_arr[pos_j]])
                ch_j[neg_j] -= h[j - shifts_arr[neg_j]]

                pos_i = (i + shifts_arr >= 0) & (i + shifts_arr < n)
                neg_i = (i - shifts_arr >= 0) & (i - shifts_arr < n)
                ch_i = np.zeros(n_corr)
                ch_i[pos_i] += (1.0 - h[i + shifts_arr[pos_i]])
                ch_i[neg_i] -= h[i - shifts_arr[neg_i]]

                g = ch_j - ch_i
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-13
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def subgradient_optimize(h_init, n, target, time_limit, n_iter=50000, top_k=8):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.008
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_idx))
        grad -= np.mean(grad)

        direction = -grad
        c5_curr = float(np.max(corr)) * dx

        alpha = step
        improved = False
        for _ in range(12):
            h_try = h + alpha * direction
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_curr - 1e-15:
                h = h_try
                improved = True
                break
            alpha *= 0.5

        if improved:
            step = min(step * 1.03, 0.04)
        else:
            step *= 0.9

    return best_h, best_c5

def multi_point_cd(h_init, n, target, time_limit, n_moves=3, max_sweeps=10000,
                   max_no_improve=200, n_cand=50):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    shifts_arr = np.arange(-(n - 1), n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        c_max = float(np.max(corr))
        top_idx = np.argsort(corr)[-10:]
        weights = np.exp(5.0 * (corr[top_idx] - c_max) / max(c_max, 1e-10))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, s_idx in zip(weights, top_idx):
            if w < 1e-7:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        moves_made = 0
        for _m in range(n_moves):
            best_trial = c5_val
            best_i = -1
            best_j = -1
            best_delta = 0.0

            for ii in range(len(high_set)):
                i = int(high_set[ii])
                for jj in range(len(low_set)):
                    j = int(low_set[jj])
                    if i == j:
                        continue
                    delta_lo = max(h[i] - 1.0, -h[j])
                    delta_hi = min(h[i], 1.0 - h[j])
                    if delta_hi - delta_lo < 1e-12:
                        continue

                    pos_j = (j + shifts_arr >= 0) & (j + shifts_arr < n)
                    neg_j = (j - shifts_arr >= 0) & (j - shifts_arr < n)
                    ch_j = np.zeros(n_corr)
                    ch_j[pos_j] += (1.0 - h[j + shifts_arr[pos_j]])
                    ch_j[neg_j] -= h[j - shifts_arr[neg_j]]

                    pos_i = (i + shifts_arr >= 0) & (i + shifts_arr < n)
                    neg_i = (i - shifts_arr >= 0) & (i - shifts_arr < n)
                    ch_i = np.zeros(n_corr)
                    ch_i[pos_i] += (1.0 - h[i + shifts_arr[pos_i]])
                    ch_i[neg_i] -= h[i - shifts_arr[neg_i]]

                    g = ch_j - ch_i
                    a_d = delta_lo
                    b_d = delta_hi
                    for _ in range(18):
                        mid_d = 0.5 * (a_d + b_d)
                        vals = corr + mid_d * g
                        f_mid = np.max(vals)
                        active_mask = vals >= f_mid - 1e-13
                        if active_mask.any():
                            f_prime = np.max(g[active_mask])
                        else:
                            f_prime = 0.0
                        if f_prime > 1e-12:
                            b_d = mid_d
                        else:
                            a_d = mid_d

                    delta_star = 0.5 * (a_d + b_d)
                    if abs(delta_star) < 1e-12:
                        continue

                    approx_max = np.max(corr + delta_star * g) * dx
                    if approx_max < best_trial:
                        best_trial = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star

            if best_i >= 0:
                h[best_i] -= best_delta
                h[best_j] += best_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                if c5_val < best_c5 - 1e-14:
                    best_c5 = c5_val
                    best_h = h.copy()
                moves_made += 1

        if moves_made == 0:
            no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP polish at initial resolution
    if tleft() > 80:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                    [200, 1000, 5000, 20000], 400)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Subgradient optimization (novel approach)
    if tleft() > 200:
        h_sg, c5_sg = subgradient_optimize(best_h, n0, target0,
                                           time_limit=min(120, tleft() * 0.5),
                                           n_iter=40000, top_k=10)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 3: Exhaustive pair CD with larger candidate set
    if tleft() > 200:
        h_cd, c5_cd = exhaustive_pair_cd(best_h, n0, target0,
                                         time_limit=min(180, tleft() * 0.5),
                                         n_cand=90, max_sweeps=20000,
                                         max_no_improve=300, weight_beta=6.0,
                                         n_top_shifts=12)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Multi-point CD (apply multiple moves per sweep)
    if tleft() > 150:
        h_mp, c5_mp = multi_point_cd(best_h, n0, target0,
                                     time_limit=min(130, tleft() * 0.5),
                                     n_moves=3, max_sweeps=10000,
                                     max_no_improve=200, n_cand=60)
        if c5_mp < best_score:
            best_score = c5_mp
            best_h = h_mp

    # Phase 5: SLSQP with very high alpha
    if tleft() > 100:
        h_pol2, sc2 = slsqp_optimize(best_h, n0, target0,
                                      [50000, 200000], 500)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 6: Multi-start with diverse perturbations
    if tleft() > 200:
        rng = np.random.RandomState(seed)
        n_cur = len(best_h)
        target_cur = n_cur / 2.0

        for noise_scale in [0.005, 0.015, 0.04, 0.08]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, target_cur)

            h_ref, sc_ref = slsqp_optimize(h_pert, n_cur, target_cur,
                                            [500, 5000, 50000], 400)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

            if tleft() > 60:
                h_cd_ref, c5_cd_ref = exhaustive_pair_cd(best_h, n_cur, target_cur,
                                                         time_limit=min(50, tleft() * 0.4),
                                                         n_cand=70, max_sweeps=8000,
                                                         max_no_improve=150, weight_beta=7.0)
                if c5_cd_ref < best_score:
                    best_score = c5_cd_ref
                    best_h = h_cd_ref

    # Phase 7: Try coarser grid then refine (to escape local minima)
    if tleft() > 300:
        n_coarse = max(400, n0 // 2)
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)

        h_c_pol, sc_c = slsqp_optimize(h_coarse, n_coarse, target_coarse,
                                        [500, 5000, 50000], 300)
        if tleft() > 150:
            h_c_cd, c5_c = exhaustive_pair_cd(h_c_pol, n_coarse, target_coarse,
                                              time_limit=min(100, tleft() * 0.4),
                                              n_cand=60, max_sweeps=8000,
                                              max_no_improve=200, weight_beta=6.0)
            h_refine = interpolate_to(h_c_cd, n0, target0)
            h_r_pol, sc_r = slsqp_optimize(h_refine, n0, target0,
                                            [1000, 10000, 50000], 400)
            if tleft() > 80:
                h_r_cd, c5_r = exhaustive_pair_cd(h_r_pol, n0, target0,
                                                  time_limit=min(60, tleft() * 0.4),
                                                  n_cand=70, max_sweeps=6000,
                                                  max_no_improve=150, weight_beta=7.0)
                if c5_r < best_score:
                    best_score = c5_r
                    best_h = h_r_cd

    # Phase 8: Final subgradient polish
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sg2, c5_sg2 = subgradient_optimize(best_h, n_cur, target_cur,
                                             time_limit=min(60, tleft() * 0.5),
                                             n_iter=30000, top_k=8)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2

    # Phase 9: Final pair CD
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_fin_cd, c5_fin = exhaustive_pair_cd(best_h, n_cur, target_cur,
                                              time_limit=min(45, tleft() * 0.6),
                                              n_cand=100, max_sweeps=5000,
                                              max_no_improve=150, weight_beta=8.0)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin_cd

    # Phase 10: Final SLSQP
    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final, sc_final = slsqp_optimize(best_h, n_cur, target_cur,
                                            [100000, 500000], 300)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
