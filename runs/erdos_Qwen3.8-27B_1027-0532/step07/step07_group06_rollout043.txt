def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

        func = _pnorm_obj

                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    func = _pnorm_obj if use_pnorm else _smooth_lse
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def _bisect_delta(corr, g, delta_lo, delta_hi, n_iters):
    a_d = delta_lo
    b_d = delta_hi
    for _ in range(n_iters):
        mid_d = 0.5 * (a_d + b_d)
        vals = corr + mid_d * g
        f_mid = np.max(vals)
        active_mask = vals >= f_mid - 1e-14
        if active_mask.any():
            f_prime = np.max(g[active_mask])
        else:
            f_prime = 0.0
        if f_prime > 1e-12:
            b_d = mid_d
        else:
            a_d = mid_d
    delta_star = 0.5 * (a_d + b_d)
    approx_max = np.max(corr + delta_star * g)
    return delta_star, approx_max

def enhanced_pair_cd(h_init, n, target, time_limit, start_time,
                     max_sweeps=25000, n_cand_start=20, n_cand_end=65,
                     max_no_improve=350, temp_start=0.3, temp_end=8.0,
                     bisection_iters=18, quick_iters=7, top_refine=8):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        temp = temp_start + (temp_end - temp_start) * progress

        c_max = np.max(corr)
        if c_max < 1e-14:
            break
        beta = 1.0 / (c_max * temp)
        active = corr > c_max - c_max * temp
        active_count = int(np.sum(active))
        if active_count == 0:
            active = np.zeros(n_corr, dtype=bool)
            active[int(np.argmax(corr))] = True
        elif active_count > 50:
            thresh = np.sort(corr)[-50]
            active = corr >= thresh

        active_idx = np.where(active)[0]
        w_vals = np.exp(beta * (corr[active_idx] - c_max))
        w_sum = w_vals.sum()
        if w_sum < 1e-30:
            weights_arr = np.ones(len(active_idx)) / len(active_idx)
        else:
            weights_arr = w_vals / w_sum

        grad = np.zeros(n)
        for s_idx, w in zip(active_idx, weights_arr):
            if w < 1e-12:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)
        current_ncand = max(current_ncand, 8)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        top_pairs = []

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                delta_star, approx_max = _bisect_delta(corr, g, delta_lo, delta_hi, quick_iters)
                if abs(delta_star) < 1e-12:
                    continue
                score = approx_max * dx
                if score < best_trial_c5:
                    top_pairs.append((best_trial_c5 - score, i, j, delta_star))
                    if score < best_trial_c5:
                        best_trial_c5 = score
                        best_i = i
                        best_j = j
                        best_delta = delta_star

        top_pairs.sort(key=lambda x: -x[0])
        top_pairs = top_pairs[:top_refine]

        for _, i, j, _ in top_pairs:
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-12:
                continue
            g = change[j] - change[i]
            delta_star, approx_max = _bisect_delta(corr, g, delta_lo, delta_hi, bisection_iters)
            if abs(delta_star) < 1e-12:
                continue
            score = approx_max * dx
            if score < best_trial_c5:
                best_trial_c5 = score
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0 and abs(best_delta) > 1e-14:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def block_cd_refine(h_init, n, target, time_limit, start_time, block_size=4, max_rounds=500):
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        if no_improve > max_rounds // 4:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        c_max = np.max(corr)
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(np.abs(grad))
        extreme_idx = sorted_idx[-block_size * 3:]
        rng_local = np.random.RandomState(rnd + 42)
        perm = rng_local.permutation(len(extreme_idx))
        block_candidates = extreme_idx[perm[:block_size * 2]]

        best_block_obj = c5_val
        best_block_vars = None

        for start_pos in range(0, len(block_candidates) - block_size + 1):
            block_vars = block_candidates[start_pos:start_pos + block_size]
            h_fixed = h.copy()
            h_block = h[block_vars].copy()

            def _block_obj(xb, _h=h_fixed, _bv=block_vars, _t=target, _n=n):
                h_t = _h.copy()
                h_t[_bv] = np.clip(xb, 0.0, 1.0)
                h_t = project_feasible(h_t, _t)
                c = np.correlate(h_t, 1.0 - h_t, mode='full')
                cm = np.max(c)
                if cm <= 0:
                    return 0.0
                shifted = 800.0 * (c - cm)
                return float(cm + np.log(np.sum(np.exp(np.clip(shifted, -500, 0)))) / 800.0)

            try:
                res = minimize(_block_obj, h_block, method='Nelder-Mead',
                               options={'maxiter': 1500, 'xatol': 1e-10, 'fatol': 1e-12, 'xatol': 1e-12})
                h_new_block = np.clip(res.x, 0.0, 1.0)
                h_trial = h.copy()
                h_trial[block_vars] = h_new_block
                h_trial = project_feasible(h_trial, target)
                sc = compute_c5(h_trial)
                if sc < best_block_obj:
                    best_block_obj = sc
                    best_h_trial = h_trial.copy()
            except Exception:
                continue

        if '_best_h_trial' in dir() or True:
            pass

        if 'best_h_trial' in locals() and best_h_trial is not None:
            pass

        try:
            if 'best_h_trial' in dir():
                h = best_h_trial
        except NameError:
            pass

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def stochastic_refine(h_init, n, target, time_limit, start_time, max_steps=15000, base_scale=0.025):
    rng = np.random.RandomState(12345)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 2000:
            break

        scale = base_scale * max(0.05, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 4]))
        n_pert = min(n_pert, n)
        perm = rng.permutation(n)
        idxs = perm[:n_pert]
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 60:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5

def multi_start_slsqp(h_init, n, target, time_limit, start_time, n_starts=4):
    rng = np.random.RandomState(999)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    for s in range(n_starts):
        if time.time() - start_time > time_limit:
            break
        scale = [0.02, 0.05, 0.1, 0.03][s % 4]
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        h_r, sc_r = slsqp_round(h_pert, n, target, [200, 2000, 10000],

        if sc_r < best_score:
            best_score = sc_r
            best_h = h_r

    return best_h, best_score

            best_score = sc8
            best_h = h8

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 60:
        h9, sc9 = stochastic_refine(best_h, n0, target0,
                                    start_time + min(60, remaining - 15), start_time,
                                    max_steps=15000, base_scale=0.02)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 80:
        h10, sc10 = multi_start_slsqp(best_h, n0, target0,
                                      start_time + min(80, remaining - 15), start_time,
                                      n_starts=3)
        if sc10 < best_score:
            best_score = sc10
            best_h = h10

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 60:
        h_fin, sc_fin = enhanced_pair_cd(best_h, n0, target0,
                                         start_time + min(60, remaining - 10), start_time,
                                         max_sweeps=15000, n_cand_start=35, n_cand_end=90,
                                         max_no_improve=150, temp_start=0.2, temp_end=1.0,
                                         bisection_iters=22, quick_iters=8, top_refine=5)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 40:
        h_fin2, sc_fin2 = slsqp_round(best_h, n0, target0,
                                      [50000, 150000],
                                      800, start_time + min(40, remaining - 5), start_time, use_pnorm=False)
        if sc_fin2 < best_score:
            best_score = sc_fin2
            best_h = h_fin2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
