_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha, mu):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    penalty = mu * (np.sum(x) - _g_target) ** 2
    return val + penalty

    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change, shifts

    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def _smooth_obj(x, alpha, mu):

def lbfgsb_optimize(h_start, n, target, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    configs = [
        (100.0, 1.0, 100),
        (500.0, 10.0, 150),
        (2000.0, 100.0, 200),
        (10000.0, 1000.0, 300),
        (50000.0, 10000.0, 500),
    ]
    for alpha, mu, maxiter in configs:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:

    return h_cur, cur_score

def _build_change(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]
    return change, shifts

def _upd_row(change, p, h, shifts, n):
    change[p, :] = 0.0
    pm = (p + shifts >= 0) & (p + shifts < n)
    nm = (p - shifts >= 0) & (p - shifts < n)
    change[p, pm] += (1.0 - h[p + shifts[pm]])
    change[p, nm] -= h[p - shifts[nm]]

def cd_optimize(h_init, n, target, time_limit, n_cand=50, max_sweeps=20000,

    mid = n - 1

    no_imp = 0

        if time.time() - t0 > time_limit:
            break

        if c5_val < best_c5 - 1e-13:

            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp > max_no_improve:
            break

            hp = project_feasible(hp, target)

                change, shifts = _build_change(h, n)

            v2 = (idxs + k >= 0) & (idxs + k < n)

                    else:
                        a = md

    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    configs = [
        (100.0, 1.0, 100),
        (500.0, 10.0, 150),
        (2000.0, 100.0, 200),
        (10000.0, 1000.0, 300),
        (50000.0, 10000.0, 500),
    ]
    for alpha, mu, maxiter in configs:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def _build_change(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]
    return change, shifts


def _upd_row(change, p, h, shifts, n):
    change[p, :] = 0.0
    pm = (p + shifts >= 0) & (p + shifts < n)
    nm = (p - shifts >= 0) & (p - shifts < n)
    change[p, pm] += (1.0 - h[p + shifts[pm]])
    change[p, nm] -= h[p - shifts[nm]]


def cd_optimize(h_init, n, target, time_limit, n_cand=50, max_sweeps=20000,
                max_no_improve=500, n_active=5, reheat_int=80, reheat_scale=0.03,
                grad_thresh=0.3):
    t0 = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_imp = 0
    change, shifts = _build_change(h, n)

    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > max_no_improve:
            break

        if no_imp > reheat_int and no_imp % 20 == 0:
            rng = np.random.RandomState(sweep * 31 + 13)
            np_ = max(3, n // 50)
            pi_ = rng.choice(n, size=np_, replace=False)
            hp = h.copy()
            hp[pi_] += rng.randn(np_) * reheat_scale
            hp = project_feasible(hp, target)
            if compute_c5(hp) < c5_val + 1e-4:
                h = hp
                change, shifts = _build_change(h, n)
                no_imp = 0
                continue

        top_k = np.argsort(corr)[-n_active:]
        ws = corr[top_k].sum()
        if ws < 1e-10:
            wts = np.ones(n_active) / n_active
        else:
            wts = corr[top_k] / ws

        grad = np.zeros(n)
        for si, w in zip(top_k, wts):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= grad.mean()

        srt = np.argsort(grad)
        lo_s = srt[:n_cand]
        hi_s = srt[-n_cand:]

        ks = int(np.argmax(corr))
        best_trial = c5_val
        bm = None

        for ii in range(len(hi_s)):
            i = int(hi_s[ii])
            hi_val = h[i]
            for jj in range(len(lo_s)):
                j = int(lo_s[jj])
                if i == j:
                    continue
                lo_val = h[j]
                dh = min(hi_val, 1.0 - lo_val)
                dl = max(hi_val - 1.0, -lo_val)
                if dh - dl < 1e-12:
                    continue
                g = change[j] - change[i]
                gmax = np.max(np.abs(g))
                if gmax < 1e-15:
                    continue
                if g[ks] > grad_thresh * gmax:
                    continue
                a, b = dl, dh
                for _ in range(10):
                    md = 0.5 * (a + b)
                    vals = corr + md * g
                    fm = np.max(vals)
                    act = vals >= fm - 1e-12
                    fp = np.max(g[act]) if act.any() else 0.0
                    if fp > 1e-10:
                        b = md
                    else:
                        a = md
                ds = 0.5 * (a + b)
                if abs(ds) < 1e-12:
                    continue
                tr = np.max(corr + ds * g) * dx
                if tr < best_trial - 1e-15:
                    best_trial = tr
                    bm = (i, j, ds)

        if bm is not None:
            i, j, d = bm
            h[i] -= d
            h[j] += d
            np.clip(h, 0.0, 1.0, out=h)
            h = project_feasible(h, target)
            _upd_row(change, i, h, shifts, n)
            _upd_row(change, j, h, shifts, n)

        if sweep % 1000 == 0 and sweep > 0:
            change, shifts = _build_change(h, n)

    return best_h, best_c5


def cd_multistart(h_init, n, target, time_limit, n_restarts=3, n_cand=45,
                  max_sweeps=15000, max_no_improve=400, n_active=5,
                  reheat_int=60, reheat_scale=0.04, grad_thresh=0.2):
    t0 = time.time()
    best_h = h_init.copy()
    best_sc = compute_c5(h_init)
    rng = np.random.RandomState(12345)
    per_restart = time_limit / n_restarts
    for r in range(n_restarts):
        if time.time() - t0 > time_limit:
            break
        if r == 0:
            h_start = h_init.copy()
        else:
            noise = rng.randn(n) * reheat_scale * (1 + r * 0.5)
            h_start = h_init + noise
            h_start = project_feasible(h_start, target)
        remaining = min(per_restart, time_limit - (time.time() - t0))
        if remaining < 10:
            break
        hc, sc = cd_optimize(h_start, n, target, time_limit=remaining,
                             n_cand=n_cand, max_sweeps=max_sweeps,
                             max_no_improve=max_no_improve, n_active=n_active,
                             reheat_int=reheat_int, reheat_scale=reheat_scale,
                             grad_thresh=grad_thresh)
        if sc < best_sc:
            best_sc = sc
            best_h = hc
    return best_h, best_sc


def interp_to(h_src, n_dst, tgt):
    if len(h_src) == n_dst:
        return project_feasible(h_src.copy(), tgt)
    xn = np.linspace(0.0, 1.0, n_dst)
    xo = np.linspace(0.0, 1.0, len(h_src))
    return project_feasible(np.interp(xn, xo, h_src), tgt)
