```python
import numpy as np
from scipy.optimize import minimize
import time

_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _time_left(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _obj_func(x):
    return smooth_obj(x, _g_alpha)

def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_target, _g_alpha
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur
```

```python
def pair_cd_multishift_weighted(h_init, n, target, time_limit, max_sweeps,
                                 n_cand_start, n_cand_end, top_k_eval,
                                 max_no_improve, n_shifts_top, shift_weight_decay):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

            best_h = h.copy()

        weights = np.exp(shift_weight_decay * (corr[top_shift_indices] - c_max))

        n_low = len(low_set)
        n_high = len(high_set)
```

1. Compute corr
2. Find active set: all k where corr[k] ≥ corr_max - epsilon
3. Compute gradient = average of gradients for active k's
4. Take a step in the negative gradient direction
5. Project back to feasible set

```python
                          epsilon_active=1e-10, step_decay=0.999):

    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    step_size = 0.01
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                step_size *= 0.5
                h = best_h.copy()
                no_improve = 0
            if no_improve > 2000:
                break
        
        # Find active set
        c_max = np.max(corr)
        active = np.where(corr >= c_max - epsilon_active)[0]
        
        # Compute exact subgradient (average of active gradients)
        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(active))
        grad -= np.mean(grad)  # Project out the mean for the sum constraint
        
        # Take step
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, target)
        
        h = h_new
        step_size *= step_decay
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

# Global variables for SLSQP callbacks
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _time_left(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _obj_func(x):
    return smooth_obj(x, _g_alpha)

def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_target, _g_alpha
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new
        except Exception:

def pair_cd_weighted(h_init, n, target, time_limit, max_sweeps,
                     n_cand_start, n_cand_end, top_k_eval,
                     max_no_improve, n_shifts_top, shift_weight_temp):
    """Coordinate descent with multi-shift weighted scoring."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # Select top shifts for weighted scoring
        c_max = np.max(corr)
        # Use softmax-like weighting over top shifts
        top_n = min(n_shifts_top, n_corr)
        top_shift_indices = np.argsort(corr)[-top_n:]
        
        # Exponential weighting: shifts closer to max get more weight
        temp = shift_weight_temp
        exp_w = np.exp((corr[top_shift_indices] - c_max) / temp)
        weights = exp_w / exp_w.sum()
        
        # k_star is the worst shift
        k_star_idx = int(top_shift_indices[-1])
        k_star = k_star_idx - mid
        
        # Gradient for selecting high/low candidate sets
        grad = np.zeros(n)
        for si, s_idx in enumerate(top_shift_indices):
            w = weights[si]
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        n_cand = max(n_cand, 10)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        n_low = len(low_set)
        
        # Weighted change vectors
        # weighted_change[j] = sum_s w_s * change[low_set[j], s]
        weighted_cl = change[low_set, top_shift_indices].dot(weights)
        weighted_ch = change[high_set, top_shift_indices].dot(weights)
        
        score_mat = weighted_cl[None, :] - weighted_ch[:, None]
        
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for tf in top_flat:
            ii = int(tf // n_low)
            jj = int(tf % n_low)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue
            
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue
            
            g = change[j] - change[i]
            
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(12):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-13
                f_prime = np.max(g[active_mask])
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
```

```python
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _time_left(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _obj_func(x):
    return smooth_obj(x, _g_alpha)


def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_target, _g_alpha
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,

        else:

        grad = np.zeros(n)
        for si, s_idx in enumerate(top_shift_indices):

            k = int(s_idx) - mid

            grad[v2] -= w * h[idxs[v2] + k]

        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)
        n_cand = max(n_cand, 10)

        high_set = sorted_idx[-n_cand:]

        weighted_ch = change[high_set, top_shift_indices].dot(weights)

                else:
                    a_d = mid_d

                best_j = j

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

def subgrad_descent(h_init, n, target, time_limit, max_steps,
                    epsilon=1e-6, init_step=0.003, decay=0.999):

    h = h_init.copy().astype(float)
    best_h = h.copy()

    no_improve = 0
    step_size = init_step
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx if (dx := 2.0 / n) else 0

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                step_size *= 0.8
                if step_size < 1e-7:
                    break
            if no_improve > 800:
                break

        c_max = np.max(corr)
        active = np.where(corr >= c_max - epsilon)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(active))
        grad -= np.mean(grad)

        h_new = h - step_size * grad
        h_new = project_feasible(h_new, target)
        h = h_new
        step_size *= decay

    return best_h, best_c5

            best_score = sc
            best_h = h_ref

    tl = _time_left(start_time, budget_s)
    if tl > 50:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_cd_fin, c5_fin = pair_cd_weighted(best_h, n_cur, tgt_cur,
                                            time_limit=min(35, tl * 0.7),
                                            max_sweeps=20000,
                                            n_cand_start=50, n_cand_end=100,
                                            top_k_eval=80, max_no_improve=300,
                                            n_shifts_top=7, shift_weight_temp=5.0)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_cd_fin

    tl = _time_left(start_time, budget_s)
    if tl > 40:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        h_fin = slsqp_optimize(best_h, n_cur, tgt_cur,
                               [20000, 100000], 500, 1e-15)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
