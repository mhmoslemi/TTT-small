```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    idxs = np.arange(n)
    for p in range(n):
        # When h[p] increases by 1, corr changes by:
        # For lag k >= 0 (position mid+k in full corr):
        #   corr[mid+k] = sum_x h[x]*(1-h[x+k]) for valid x
        #   d/dh[p]: if p is in the "first" slot (x=p), contribution is (1-h[p+k])
        #            if p is in the "second" slot (x+k=p, i.e., x=p-k), contribution is -h[p-k]
        # For lag k < 0 (position mid+k in full corr, k is negative):
        #   Let l = -k > 0, corr[mid-l] = sum_x h[x+l]*(1-h[x]) for valid x
        #   Actually in numpy: full[k] = sum_x h[x] * g[x+k] where g = 1-h
        #   So full[mid+k] = sum_{x: 0<=x<n, 0<=x+k<n} h[x] * (1-h[x+k])
        #   d/dh[p]: p appears as x (contrib: 1-h[p+k] if 0<=p+k<n)
        #            p appears as x+k (contrib: -h[p-k] if 0<=p-k<n)
        
        for k in range(-(n-1), n):
            pos = mid + k
            # h[p] appears as first factor when x=p, need 0 <= p+k < n
            if 0 <= p + k < n:

            # h[p] appears as second factor when x+k=p, i.e., x=p-k, need 0 <= p-k < n
            if 0 <= p - k < n:
                change[p, pos] -= h[p - k]
    return change
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

        grad = np.zeros(n)
        for idx in range(n_active):
            w = weights[idx]
            pos = int(top_idx[idx])
            k = pos - mid
            idxs = np.arange(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        active_set = top_idx

        high_vals = np.array([np.max(change[hi, active_set]) for hi in high_set])
        low_vals = np.array([np.min(change[lo, active_set]) for lo in low_set])

        promising = high_vals - low_vals[np.newaxis, :]
        neg_mask = promising < 0.0

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        pair_list = []
        if neg_mask.sum() <= 300:
            for ii in range(n_h):
                for jj in range(n_l):
                    if neg_mask[ii, jj]:
                        pair_list.append((ii, jj))
        else:
            flat = neg_mask.flatten()
            if flat.sum() > 0:
                idxs_pos = np.where(flat)[0]
                shuffled = idxs_pos[rng.permutation(len(idxs_pos))]
                for pos in shuffled[:300]:
                    ii = pos // n_l
                    jj = pos % n_l
                    pair_list.append((ii, jj))

        rng_pairs = min(50, len(pair_list) // 2 + 1)
        for _ in range(rng_pairs):
            ii = int(rng.randint(n_h))
            jj = int(rng.randint(n_l))
            pair_list.append((ii, jj))

        for ii, jj in pair_list:
            i_val = int(high_set[ii])
            j_val = int(low_set[jj])
            if i_val == j_val:
                continue

            delta_hi = min(h[i_val], 1.0 - h[j_val])
            delta_lo = -min(h[j_val], 1.0 - h[i_val])

            if delta_hi < 1e-13 and delta_lo > -1e-13:
                continue

            g = change[j_val] - change[i_val]
            delta_star = bisection_delta(corr, g, delta_lo, delta_hi, bisection_iters)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial - 1e-15:
                best_trial = approx_max
                best_i = i_val
                best_j = j_val
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            actual_delta_i = h[best_i] - (h[best_i] + best_delta)
            d_i = -best_delta
            d_j = best_delta
            correct_change_row(change, best_i, d_i, n)
            correct_change_row(change, best_j, d_j, n)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5


def binary_local_search(h_init, n, target, time_limit, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    start = time.time()
    dx = 2.0 / n

    h_bin = (h_init >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_target = int(target)
    if n_ones > n_target:
        extras = np.where(h_bin == 1)[0]
        rng.shuffle(extras)
        for k in range(n_ones - n_target):
            h_bin[extras[k]] = 0.0
    elif n_ones < n_target:
        zeros = np.where(h_bin == 0)[0]
        rng.shuffle(zeros)
        for k in range(n_target - n_ones):
            h_bin[zeros[k]] = 1.0

    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h_bin.copy()

    n_corr = 2 * n - 1
    mid = n - 1
    no_improve = 0
    max_no_improve = 2000

    while time.time() - start < time_limit and no_improve < max_no_improve:
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_bin.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idx = np.argsort(corr)[-3:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(3):
            pos = int(top_idx[idx])
            k = pos - mid
            w = weights[idx]
            idxs = np.arange(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h_bin[idxs[v1] - k])
            grad[v2] -= w * h_bin[idxs[v2] + k])
        grad -= np.mean(grad)

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        n_try = min(100, len(ones) * len(zeros))
        found_improve = False
        best_swap = None
        best_swap_val = c5_val

        for _ in range(n_try):
            i_idx = int(rng.randint(len(ones)))
            j_idx = int(rng.randint(len(zeros)))
            i_val = int(ones[i_idx])
            j_val = int(zeros[j_idx])

            h_test = h_bin.copy()
            h_test[i_val] = 0.0
            h_test[j_val] = 1.0
            corr_test = np.correlate(h_test, 1.0 - h_test, mode='full')
            val = float(np.max(corr_test)) * dx
            if val < best_swap_val:
                best_swap_val = val
                best_swap = (i_val, j_val)

        if best_swap is not None:
            i_val, j_val = best_swap
            h_bin[i_val] = 0.0
            h_bin[j_val] = 1.0
            corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n0, target0, [100, 500], 400, 1e-13)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_cd, c5_cd = enhanced_cd(best_h, n0, target0,
                                   time_limit=min(120, tleft() * 0.35),
                                   max_sweeps=20000, n_active=5,
                                   bisection_iters=14, max_no_improve=300,
                                   n_cand_start=20, n_cand_end=50, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 500:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        best_h = h1
        best_score = compute_c5(h1)
        n0 = n1
        target0 = target1

        if tleft() > 400:
            h_cd1, c5_cd1 = enhanced_cd(h1, n1, target1,
                                         time_limit=min(200, tleft() * 0.4),
                                         max_sweeps=30000, n_active=5,
                                         bisection_iters=14, max_no_improve=400,
                                         n_cand_start=25, n_cand_end=60, rng=rng)
            if c5_cd1 < best_score:
                best_score = c5_cd1
                best_h = h_cd1

        if tleft() > 200:
            h_pol1 = slsqp_round(best_h, n0, target0, [500, 2000], 500, 1e-13)
            sc = compute_c5(h_pol1)
            if sc < best_score:
                best_score = sc
                best_h = h_pol1

        if tleft() > 150:
            h_cd2, c5_cd2 = enhanced_cd(best_h, n0, target0,
                                         time_limit=min(100, tleft() * 0.35),
                                         max_sweeps=20000, n_active=7,
                                         bisection_iters=14, max_no_improve=300,
                                         n_cand_start=30, n_cand_end=70, rng=rng)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    if tleft() > 200:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if compute_c5(h2) < best_score:
            best_score = compute_c5(h2)
            best_h = h2
            n0 = n2
            target0 = target2

            if tleft() > 100:
                h_cd3, c5_cd3 = enhanced_cd(best_h, n0, target0,
                                             time_limit=min(80, tleft() * 0.4),
                                             max_sweeps=15000, n_active=5,
                                             bisection_iters=12, max_no_improve=250,
                                             n_cand_start=30, n_cand_end=60, rng=rng)
                if c5_cd3 < best_score:
                    best_score = c5_cd3
                    best_h = h_cd3

    if tleft() > 100:
        for scale in [0.05, 0.10]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(n0) * scale
            h_pert = project_feasible(h_pert, target0)
            h_re, sc_re = enhanced_cd(h_pert, n0, target0,
                                       time_limit=min(60, tleft() * 0.3),
                                       max_sweeps=10000, n_active=5,
                                       bisection_iters=12, max_no_improve=200,
                                       n_cand_start=25, n_cand_end=55, rng=rng)
            if sc_re < best_score:
                best_score = sc_re
                best_h = h_re

    if tleft() > 60:
        h_cd_final, c5_final = enhanced_cd(best_h, n0, target0,
                                            time_limit=min(50, tleft() * 0.5),
                                            max_sweeps=15000, n_active=7,
                                            bisection_iters=14, max_no_improve=200,
                                            n_cand_start=35, n_cand_end=65, rng=rng)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 40:
        h_pol_final = slsqp_round(best_h, n0, target0, [5000, 20000], 400, 1e-13)
        sc = compute_c5(h_pol_final)
        if sc < best_score:
            best_score = sc
            best_h = h_pol_final

    if tleft() > 30:
        h_cd_last, c5_last = enhanced_cd(best_h, n0, target0,
                                          time_limit=min(20, tleft() * 0.5),
                                          max_sweeps=10000, n_active=5,
                                          bisection_iters=14, max_no_improve=150,
                                          n_cand_start=30, n_cand_end=50, rng=rng)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_cd_last

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
