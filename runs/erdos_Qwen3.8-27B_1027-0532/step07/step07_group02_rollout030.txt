```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def alternating_lp(h_init, n, target, n_iters=50, time_limit=60, solver='ECOS'):
    """Alternating LP optimization."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        
        # Build LP: min t s.t. A_k @ h * dx <= t, sum(h) = target, 0 <= h <= 1
        # where A_k[i] = g[i+k] for valid i
        # n_corr = 2n - 1 shifts
        n_corr = 2 * n - 1
        # Build constraint matrix
        A = np.zeros((n_corr, n))
        shifts = np.arange(-(n-1), n)
        for s_idx, d in enumerate(shifts):
            if d >= 0:
                # h[i] * g[i+d] for i = 0, ..., n-1-d
                valid = np.arange(n - d)
                A[s_idx, valid] = g[valid + d]
            else:
                # h[i] * g[i+d] for i = -d, ..., n-1
                valid = np.arange(-d, n)
                A[s_idx, valid] = g[valid + d]
        
        # Solve LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        for s_idx in range(n_corr):
            constraints.append(A[s_idx] @ h_var * dx <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver)
            if h_var.value is not None:
                h_new = project_feasible(h_var.value, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                h = h_new
            else:
                break
        except Exception:
            break
    
    return best_h, best_score
```

**Another idea: Spectral method**

The autocorrelation of h is related to the Fourier transform:
autocorr(k) = (1/2π) ∫ |H(ω)|² e^{iωk} dω

where H is the Fourier transform of h. We want to make autocorr(k) as flat as possible.

In the frequency domain, |H(ω)|² is the power spectrum. We want the power spectrum to be as "flat" as possible (white noise-like) while maintaining the constraint that h is a [0,1]-valued function with integral 1.

This suggests an iterative frequency-domain approach:
1. Compute the Fourier transform of h
2. Modify |H(ω)| to be more flat
3. Take the inverse Fourier transform
4. Project back to [0,1] and the integral constraint

This is similar to the Gerchberg-Saxton algorithm used in phase retrieval!

Let me think about this more concretely:
- We want autocorr(k) ≈ C for all k (constant)
- In frequency domain: |H(ω)|² ≈ C * 2π (flat power spectrum)
- So we want |H(ω)| to be as constant as possible

Algorithm:
1. Start with h
2. Compute H = FFT(h)
3. Set H_new[ω] = |H(ω)| * exp(i*arg(H(ω))) but with |H| modified to be flatter

4. h_new = IFFT(H_new).real
5. Project h_new to [0,1] with sum = target
6. Repeat

The "flattening" step can be done by:
|H_new(ω)| = |H(ω)|^p * C_p where p < 1 compresses the dynamic range

Or: |H_new(ω)| = avg(|H|) + λ*(|H(ω)| - avg(|H|)) with λ < 1

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        adj = diff / mask.sum()
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def build_shift_matrix(n):
    """Build the matrix A where A[s, i] = 1 if i+shift is valid."""
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    A = np.zeros((n_corr, n))
    for s, d in enumerate(shifts):
        if d >= 0:
            A[s, :n-d] = 1.0
        else:
            A[s, -d:] = 1.0
    return A

def alternating_lp(h_init, n, target, n_iters=20, time_limit=30):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        
        # LP: min t s.t. for all shifts d: sum_i h[i]*g[i+d]*dx <= t
        # n corr rows
        n_corr = 2*n - 1
        shifts = np.arange(-(n-1), n)
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
        
        for s, d in enumerate(shifts):
            if d >= 0:
                coeffs = g[d:]  # h[i] * g[i+d] for i=0..n-1-d
            else:
                coeffs = g[:n+d]  # h[i] * g[i+d] for i=-d..n-1
            valid_len = len(coeffs)
            row = np.zeros(n)
            if d >= 0:
                row[:valid_len] = coeffs
            else:
                row[-d:] = coeffs
            cons.append(cp.sum(row * h_var) * dx <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver='ECOS', verbose=False)
            if h_var.value is not None:
                h_new = project_feasible(np.asarray(h_var.value).flatten(), target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                h = h_new
            else:
                break
        except Exception:
            try:
                prob.solve(solver='SCS', verbose=False)
                if h_var.value is not None:
                    h_new = project_feasible(np.asarray(h_var.value).flatten(), target)
                    sc = compute_c5(h_new)
                    if sc < best_score:
                        best_score = sc
                        best_h = h_new.copy()
                    h = h_new
                else:
                    break
            except Exception:
                break
    return best_h, best_score
```

**Spectral flattening:**

The idea is to iteratively modify the frequency content of h to make the autocorrelation flatter.

For the linear autocorrelation:
r[k] = sum_{i=0}^{n-1-k} h[i] * h[i+k] for k ≥ 0
r[k] = r[-k] for k < 0 (by symmetry of autocorrelation)

Wait, that's only true if we define r[k] = sum_i h[i]h[i+k] symmetrically. The np.correlate(h, h, 'full') gives:
corr[s] = sum_i h[i] * h[i + s - (n-1)]

For the overlap:
overlap[s] = np.correlate(h, 1-h, 'full')[s] = sum_i h[i] * (1-h[i+s-(n-1)])
= sum_i h[i] - sum_i h[i]*h[i+s-(n-1)] (for valid i)

So overlap[s] = (sum of h over the valid range for shift s-(n-1)) - autocorr_linear[s]

The "sum of h over valid range" depends on the shift, so it's not simply 1.

For the spectral approach, I'll work with the circular autocorrelation (which IS the inverse FFT of |FFT(h)|²):

circular_autocorr[k] = sum_i h[i] * h[(i+k) mod n]

And I want this to be as flat as possible. Then I'll check the actual linear overlap.

Algorithm:
1. Compute H = FFT(h)
2. Compute power = |H|²
3. Modify power to be flatter: power_new = power^p * (mean(power))^(1-p) for p in (0,1)
   Or: power_new = mean(power) + alpha * (power - mean(power)) for alpha in (0,1)
4. H_new = sqrt(power_new) * exp(1j * angle(H))
5. h_new = IFFT(H_new).real
6. Project h_new to [0,1] with sum = target
7. Repeat

The parameter p (or alpha) controls how aggressively we flatten. Small p means more flattening.

This is similar to the "successive over-relaxation" method in phase retrieval.

Let me implement this:

```python
def spectral_flatten(h, n, target, n_iters=100, time_limit=60, p_start=0.5, p_end=0.8):
    start = time.time()
    best_h = h.copy()
    best_score = compute_c5(h)
    h = h.copy()
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        # Interpolate p
        p = p_start + (p_end - p_start) * (it / max(n_iters-1, 1))
        
        H = np.fft.fft(h)
        power = np.abs(H)**2
        mean_power = np.mean(power)
        
        if p < 1:
            power_new = (power / mean_power)**p * mean_power
        else:
            alpha = 1.0 / p
            power_new = mean_power + alpha * (power - mean_power)
        
        H_new = np.sqrt(power_new) * np.exp(1j * np.angle(H))
        h_new = np.fft.ifft(H_new).real
        h = project_feasible(h_new, target)
        
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    return best_h, best_score
```

```python
def subgradient_momentum(h_init, n, target, time_limit=60, n_iters=50000, momentum=0.9, lr_start=0.01, lr_end=0.001):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    v = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    consecutive_no_improve = 0
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        
        progress = it / max(n_iters, 1)
        lr = lr_start * (lr_end / lr_start)**progress
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))
        
        if c5_val < best_score * (n/2.0)**(-1) - 1e-12:  # need to be careful with dx
            pass  # handled below
        
        actual_c5 = c5_val * (2.0/n)
        if actual_c5 < best_score - 1e-12:
            best_score = actual_c5
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 5000:
            break
        
        # Subgradient at active shift
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)  # maintain sum constraint
        
        # Momentum update
        v = momentum * v - lr * grad
        h = h + v
        h = project_feasible(h, target)
    
    return best_h, best_score
```

```python
def fast_pair_cd(h_init, n, target, time_limit=60, n_cand=40, max_sweeps=100000, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    consecutive_no_improve = 0
    
    # Precompute change matrix (how corr changes when h[p] changes by 1)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        # Subgradient
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                # Binary search (fewer iterations)
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(15):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _build_lp_rhs(h, n):
    """Build the right-hand side matrix for the LP.
    Returns B where B[s, i] = (1-h[i+d]) * dx for valid i, for shift d corresponding to s."""
    g = 1.0 - h
    dx = 2.0 / n
    n_corr = 2 * n - 1
    B = np.zeros((n_corr, n))
    shifts = np.arange(-(n-1), n)
    for s, d in enumerate(shifts):
        if d >= 0:
            B[s, :n-d] = g[d:]
        else:
            B[s, -d:] = g[:n+d]
    B *= dx
    return B

def _alternating_lp(h_init, n, target, n_iters=30, time_limit=100):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    for _ in range(n_iters):
        if time.time() - start > time_limit:
            break
        B = _build_lp_rhs(h, n)
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [
            B @ h_var <= t_var,
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for solver in ['ECOS', 'SCS', 'CLARABEL']:
            try:
                prob.solve(solver=solver)
                if h_var.value is not None:
                    h_new = project_feasible(np.asarray(h_var.value).ravel(), target)
                    sc = compute_c5(h_new)

                        best_h = h_new.copy()
                    h = h_new
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
    return best_h, best_score
