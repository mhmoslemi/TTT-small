import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_weighted_grad(h, n, temp):
    """Compute gradient of softmax-weighted sum of correlations."""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    # softmax weights
    shifted = corr - np.max(corr)
    weights = np.exp(shifted / temp)
    weights /= weights.sum()
    # gradient
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(n_corr):
        if weights[s] < 1e-10:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += -weights[s] * dx * h[idxs[v1] - k]
        grad[v2] += -weights[s] * dx * h[idxs[v2] + k]
    # subtract mean for constraint
    grad -= np.mean(grad)
    return grad, corr, weights

def nesterov_subgrad(h_init, n, target, time_limit, temp_init=0.5, 
                      lr_init=0.01, momentum=0.9, temp_decay=0.995):
    """Nesterov accelerated subgradient method."""
    start = time.time()
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    temp = temp_init
    lr = lr_init
    stalled = 0
    
    while time.time() - start < time_limit:
        batch = 50
        for _ in range(batch):
            if time.time() - start > time_limit:
                break
            grad, corr, _ = softmax_weighted_grad(h, n, temp)
            # Nesterov lookahead
            y = h + momentum * v
            # Gradient at y (approximate by using h's gradient)
            v = momentum * v - lr * grad
            h_new = y + v  # This is the Nesterov step
            # Actually standard Nesterov: y = x + m*v, then g = grad(y), v_new = m*v - lr*g, x_new = y + v_new
            # Let me just do standard heavy ball with momentum
            h_new = h + v
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                stalled = 0
            else:
                stalled += 1
            h = h_new
            temp *= temp_decay
            lr *= 0.999
        
        if stalled > batch * 3:
            break
    
    return best_h, best_c5

            # Flip some 1s to 0s
            idx_1 = np.where(h == 1)[0]
            n_flip = ones - target

        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def binary_sa(h_init, n, target, time_limit, T_init=0.005, T_floor=0.0001,
              random_seed=42):
    start = time.time()
    rng = np.random.RandomState(random_seed)
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)

    h = (h_init > 0.5).astype(float)
    ones = int(np.sum(h))
    if ones > target:
        idx_1 = np.where(h == 1.0)[0]
        to_flip = rng.choice(idx_1, size=ones - int(target), replace=False)
        h[to_flip] = 0.0
    elif ones < target:
        idx_0 = np.where(h == 0.0)[0]
        to_flip = rng.choice(idx_0, size=int(target) - ones, replace=False)
        h[to_flip] = 1.0

    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    T = T_init
    n_accept = 0
    n_try = 0
    stall_count = 0

    while time.time() - start < time_limit:
        batch_size = 5000
        for _ in range(batch_size):
            if time.time() - start > time_limit:
                break
            idx_1 = np.where(h == 1.0)[0]
            idx_0 = np.where(h == 0.0)[0]
            if len(idx_1) == 0 or len(idx_0) == 0:
                break
            i = int(idx_1[rng.randint(len(idx_1))])
            j = int(idx_0[rng.randint(len(idx_0))])

            dc = change_row_diff = None
            g = None
            corr_new_max = np.max(corr - change[i, :] + change[j, :])
            c5_new = corr_new_max * dx

            if c5_new < best_c5 - 1e-15:
                h[i] = 0.0
                h[j] = 1.0
                corr = corr - change[i, :] + change[j, :]
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
                best_c5 = c5_new
                best_h = h.copy()
                stall_count = 0
                n_accept += 1
            else:
                delta = c5_new - vec_c5
                if rng.rand() < np.exp(-delta / T):
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = corr - change[i, :] + change[j, :]
                    update_change_row(change, i, h, n, shifts)
                    update_change_row(change, j, h, n, shifts)
                    n_accept += 1
            n_try += 1

        if n_try > 0:
            acc_rate = n_accept / n_try
            if acc_rate > 0.5:
                T *= 0.9
            elif acc_rate < 0.1:
                T = min(T * 1.3, T_init)
            T = max(T, T_floor)
        n_accept = 0
        n_try = 0

        if best_c5 < float(np.max(corr)) * dx - 1e-12:
            stall_count += 1
        else:
            stall_count = 0
        if stall_count > 50:
            break

    return best_h, best_c5


def continuous_refine(h_init, n, target, time_limit, n_cand=60, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(999)

    change = compute_change_matrix(h, n)

    for sweep in range(50000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            h = best_h + rng.randn(n) * 0.01
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0

    return best_h, best_c5


def multi_constraint_cd(h_init, n, target, time_limit, n_cand=50, n_active=8, max_no_improve=400):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(555)

    change = compute_change_matrix(h, n)

    for sweep in range(50000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break

        top_k = np.argsort(corr)[-n_active:]
        top_sum = np.sum(corr[top_k])
        if top_sum < 1e-15:
            top_sum = 1.0
        grad = np.zeros(n)
        for s_idx in top_k:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            h = best_h + rng.randn(n) * 0.015
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0

    return best_h, best_c5


def binary_sa_fast(h_init, n, target, time_limit, T_init=0.004, T_floor=0.00005, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    shifts = np.arange(-(n - 1), n)
    change = compute_change_matrix(h_init, n)

    h = (h_init > 0.5).astype(float)
    ones = int(np.sum(h))
    if ones != int(target):
        if ones > int(target):
            idx_1 = np.where(h == 1.0)[0]
            to_flip = rng.choice(idx_1, size=ones - int(target), replace=False)
            h[to_flip] = 0.0
        else:
            idx_0 = np.where(h == 0.0)[0]
            to_flip = rng.choice(idx_0, size=int(target) - ones, replace=False)
            h[to_flip] = 1.0
        change = compute_change_matrix(h, n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    T = T_init
    stall = 0
    n_batch = 0

    while time.time() - start < time_limit:
        for _ in range(3000):
            if time.time() - start > time_limit:
                break
            idx_1 = np.where(h == 1.0)[0]
            idx_0 = np.where(h == 0.0)[0]
            if len(idx_1) == 0 or len(idx_0) == 0:
                break
            i = int(idx_1[rng.randint(len(idx_1))])
            j = int(idx_0[rng.randint(len(idx_0))])
            g = -change[i, :] + change[j, :]
            new_corr_max = np.max(corr + g)
            c5_new = new_corr_max * dx
            c5_cur = float(np.max(corr)) * dx
            if c5_new < best_c5 - 1e-15:
                h[i] = 0.0
                h[j] = 1.0
                corr = corr + g
                update_change_row(change, i, h, n, shifts)
                update_change_row(change, j, h, n, shifts)
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                delta = c5_new - c5_cur
                if delta < 0 or rng.rand() < np.exp(-delta / T):
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = corr + g
                    update_change_row(change, i, h, n, shifts)
                    update_change_row(change, j, h, n, shifts)
        n_batch += 1
        if n_batch % 10 == 0:
            T = max(T * 0.97, T_floor)
        if stall > 80:
            h = (best_h > 0.5).astype(float)
            change = compute_change_matrix(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stall = 0

    return best_h, best_c5


def threshold_and_refine(h_cont, n, target, time_limit, seed=7):
    h_bin = (h_cont > 0.5).astype(float)
    ones = int(np.sum(h_bin))
    rng = np.random.RandomState(seed)
    if ones > int(target):
        idx_1 = np.where(h_bin == 1.0)[0]
        to_flip = rng.choice(idx_1, size=ones - int(target), replace=False)
        h_bin[to_flip] = 0.0
    elif ones < int(target):
        idx_0 = np.where(h_bin == 0.0)[0]
        to_flip = rng.choice(idx_0, size=int(target) - ones, replace=False)
        h_bin[to_flip] = 1.0
    return h_bin


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    current_n = n0
    current_target = target0

    if tleft() > 60:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [30, 100, 500, 2000, 10000, 50000],
                                600, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_bin = threshold_and_refine(best_h, current_n, current_target, 10, seed=seed)
        h_sa, c5_sa = binary_sa_fast(h_bin, current_n, current_target,
                                     time_limit=min(180, tleft() * 0.5),
                                     T_init=0.005, T_floor=0.00005,
                                     seed=seed + 1)
        h_cont, sc_cont = continuous_refine(h_sa, current_n, current_target,
                                            time_limit=min(80, tleft() * 0.3),
                                            n_cand=50, max_no_improve=300)
        if sc_cont < best_score:
            best_score = sc_cont
            best_h = h_cont

    if tleft() > 150:
        h_mc, c5_mc = multi_constraint_cd(best_h, current_n, current_target,
                                          time_limit=min(130, tleft() * 0.6),
                                          n_cand=55, n_active=10,
                                          max_no_improve=400)
        if c5_mc < best_score:
            best_score = c5_mc
            best_h = h_mc

    if tleft() > 80:
        h_pol, sc = slsqp_round(best_h, current_n, current_target,
                                [5000, 20000, 80000, 300000],
                                800, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        rng = np.random.RandomState(seed + 100)
        for attempt in range(4):
            if tleft() < 100:
                break
            noise_scales = [0.02, 0.05, 0.10, 0.20]
            h_pert = best_h + rng.randn(current_n) * noise_scales[attempt % 4]
            h_pert = project_feasible(h_pert, current_target)
            h_bin_p = threshold_and_refine(h_pert, current_n, current_target, 5,
                                           seed=seed + 200 + attempt * 7)
            h_sa_p, c5_sa_p = binary_sa_fast(h_bin_p, current_n, current_target,
                                             time_limit=min(60, tleft() * 0.3),
                                             T_init=0.004, T_floor=0.0001,
                                             seed=seed + 300 + attempt * 13)
            h_cr_p, sc_cr_p = continuous_refine(h_sa_p, current_n, current_target,
                                                time_limit=min(40, tleft() * 0.25),
                                                n_cand=50, max_no_improve=200)
            if sc_cr_p < best_score:
                best_score = sc_cr_p
                best_h = h_cr_p

    if tleft() > 100:
        h_mc2, c5_mc2 = multi_constraint_cd(best_h, current_n, current_target,
                                            time_limit=min(90, tleft() * 0.55),
                                            n_cand=65, n_active=12,
                                            max_no_improve=500)
        if c5_mc2 < best_score:
            best_score = c5_mc2
            best_h = h_mc2

    if tleft() > 60:
        h_bin2 = threshold_and_refine(best_h, current_n, current_target, 5, seed=seed + 500)
        h_sa2, c5_sa2 = binary_sa_fast(h_bin2, current_n, current_target,
                                       time_limit=min(50, tleft() * 0.5),
                                       T_init=0.003, T_floor=0.00005,
                                       seed=seed + 600)
        h_cr2, sc_cr2 = continuous_refine(h_sa2, current_n, current_target,
                                          time_limit=min(30, tleft() * 0.4),
                                          n_cand=60, max_no_improve=200)
        if sc_cr2 < best_score:
            best_score = sc_cr2
            best_h = h_cr2

    if tleft() > 40:
        h_pol_final, sc_final = slsqp_round(best_h, current_n, current_target,
                                            [10000, 50000, 200000, 500000],
                                            1000, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_pol_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
