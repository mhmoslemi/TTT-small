def update_A_row_fast(A, i, h, n):
    """After h changes at positions that affect A[i,:], recompute A[i,:]."""
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    A[i, :] = 0.0
    # h[i+s] for valid s:
    pos = i + shifts
    valid_pos = (pos >= 0) & (pos < n)
    A[i, valid_pos] = h[pos[valid_pos]]
    # h[i-s] for valid s:
    neg = i - shifts
    valid_neg = (neg >= 0) & (neg < n)
    A[i, valid_neg] += h[neg[valid_neg]]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    target = n // 2
    idx = np.argsort(-h)[:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def compute_A_all(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    A = np.zeros((n, n_corr))
    for s_idx in range(n_corr):
        s = s_idx - mid
        if s >= 0:
            if s < n:
                A[:n - s, s_idx] += h[s:]
            if s > 0:
                A[s:, s_idx] += h[:n - s]
        else:
            sp = -s
            if sp < n:
                A[sp:, s_idx] += h[:n - sp]
                A[:n - sp, s_idx] += h[sp:]
    return A

def update_A_row(A, i, h, n):
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    A[i, :] = 0.0
    pos = i + shifts
    valid_pos = (pos >= 0) & (pos < n)
    A[i, valid_pos] = h[pos[valid_pos]]
    neg = i - shifts
    valid_neg = (neg >= 0) & (neg < n)
    A[i, valid_neg] += h[neg[valid_neg]]

def pair_cd_continuous(h_init, n, target, time_limit, n_cand=50, max_sweeps=10000, weight_beta=5.0):
    """Weighted pair coordinate descent for continuous h."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_m = (p + shifts >= 0) & (p + shifts < n)
        neg_m = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_m] += (1.0 - h[p + shifts[pos_m]])
        change[p, neg_m] -= h[p - shifts[neg_m]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 300:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        c_max = float(np.max(corr))
        weights = np.exp(weight_beta * (corr - c_max) / max(c_max, 1e-10))
        
        # Compute gradient
        top_k = min(20, n_corr)
        top_idx = np.argsort(weights)[-top_k:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Select candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_move = None
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                a_d, b_d = d_lo, d_hi
                for _ in range(16):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g
                    f_m = np.max(vals)
                    act = vals >= f_m - 1e-14
                    f_p = np.max(g[act]) if act.any() else 0.0
                    if f_p > 1e-12:
                        b_d = m_d
                    else:
                        a_d = m_d
                
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) < 1e-12:
                    continue
                
                approx = np.max(corr + d_star * g) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_move = (i, j, d_star)
        
        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change for i and j
            for p in [i, j]:
                change[p, :] = 0.0
                pos_m = (p + shifts >= 0) & (p + shifts < n)
                neg_m = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_m] += (1.0 - h[p + shifts[pos_m]])
                change[p, neg_m] -= h[p - shifts[neg_m]]
    
    return best_h, best_c5

def slsqp_polish(h_init, n, target, time_limit):
    """SLSQP with smooth max approximation."""
    global _g_n, _g_target
    start = time.time()
    _g_n = n
    _g_target = target
    
    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    
    alphas = [100, 500, 2000, 10000]
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 300, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    target = n // 2
    idx = np.argsort(-h)[:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def compute_A_all(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    A = np.zeros((n, n_corr))
    for s_idx in range(n_corr):
        s = s_idx - mid
        if s >= 0:
            A[:n - s, s_idx] += h[s:]
            if s > 0:
                A[s:, s_idx] += h[:n - s]
        else:
            sp = -s
            A[sp:, s_idx] += h[:n - sp]
            A[:n - sp, s_idx] += h[sp:]
    return A

def update_A_row(A, i, h, n):
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    A[i, :] = 0.0
    pos = i + shifts
    vp = (pos >= 0) & (pos < n)
    A[i, vp] = h[pos[vp]]
    neg = i - shifts
    vn = (neg >= 0) & (neg < n)
    A[i, vn] += h[neg[vn]]

def binary_ls(h_init, n, time_limit, max_iter=20000, n_cand=60, chunk=12, tabu_len=30):

    dx = 2.0 / n
    n_corr = 2 * n - 1

    h = h_init.copy().astype(float)

    A = compute_A_all(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    no_improve = 0
    tabu = {}

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 400:
            break

        M = float(np.max(corr))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        score_r = A[ones] @ corr
        score_a = A[zeros] @ corr

        n_r = min(n_cand, len(ones))
        n_a = min(n_cand, len(zeros))
        top_r = ones[np.argsort(-score_r)[:n_r]]
        top_a = zeros[np.argsort(score_a)[:n_a]]

        Ar = A[top_r]
        Aa = A[top_a]

def pair_cd(h_init, n, target, time_limit, n_cand=40, max_sweeps=10000, weight_beta=5.0):

    best_h = h.copy()

        pm = (p + shifts >= 0) & (p + shifts < n)
        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]

                    vals = corr + m_d * g
                    f_m = np.max(vals)
                    act = vals >= f_m - 1e-14
                    f_p = np.max(g[act]) if act.any() else 0.0
                    if f_p > 1e-12:
                        b_d = m_d
                    else:
                        a_d = m_d
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) < 1e-12:
                    continue
                approx = float(np.max(corr + d_star * g)) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_move = (i, j, d_star)

        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [i, j]:
                change[p, :] = 0.0
                pm = (p + shifts >= 0) & (p + shifts < n)
                nm = (p - shifts >= 0) & (p - shifts < n)
                change[p, pm] += (1.0 - h[p + shifts[pm]])
                change[p, nm] -= h[p - shifts[nm]]

    return best_h, best_c5

def slsqp_polish(h_init, n, target, time_limit):
    global _g_n, _g_target
    start = time.time()
    _g_n = n
    _g_target = target

    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def cf(x):
        return np.sum(x) - target

    def cj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)

    for alpha in [200, 1000, 5000, 20000]:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_orig = np.array(initial_h_values, dtype=float)
    n0 = len(h_orig)
    target0 = n0 // 2

    best_h = project_feasible(h_orig, target0)
    best_score = compute_c5(best_h)

    h_bin = binarize(best_h, n0)
    s_bin = compute_c5(h_bin)
    if s_bin < best_score:
        best_score = s_bin
        best_h = h_bin

    if tleft() > 80:
        h_ls, s_ls = binary_ls(best_h, n0, min(70, tleft() * 0.7),
                                max_iter=15000, n_cand=60, chunk=12)
        if s_ls < best_score:
            best_score = s_ls
            best_h = h_ls

    rng = np.random.RandomState(seed)
    for restart in range(15):
        if tleft() < 250:
            break
        k_flip = 8 + restart * 4
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        if len(ones) < k_flip or len(zeros) < k_flip:
            break
        ff = rng.choice(ones, size=k_flip, replace=False)
        ft = rng.choice(zeros, size=k_flip, replace=False)
        h_pert[ff] = 0.0
        h_pert[ft] = 1.0
        h_try, s_try = binary_ls(h_pert, n0, min(35, tleft() * 0.25),
                                  max_iter=8000, n_cand=50, chunk=12)
        if s_try < best_score:
            best_score = s_try
            best_h = h_try

    if tleft() > 200:
        h_sa, s_sa = binary_sa(best_h, n0, min(150, tleft() * 0.5),
                                max_iter=25000, n_cand=35, T0=3.0, T1=0.005, chunk=10)
        if s_sa < best_score:
            best_score = s_sa
            best_h = h_sa

    if tleft() > 180:
        h_cd, s_cd = pair_cd(best_h, n0, target0, min(160, tleft() * 0.6),
                              n_cand=40, max_sweeps=8000, weight_beta=5.0)
        if s_cd < best_score:
            best_score = s_cd
            best_h = h_cd

    if tleft() > 100:
        h_sl, s_sl = slsqp_polish(best_h, n0, target0, min(90, tleft() * 0.7))
        if s_sl < best_score:
            best_score = s_sl
            best_h = h_sl

    if tleft() > 80:
        h_bin2 = binarize(best_h, n0)
        s_bin2 = compute_c5(h_bin2)
        start2 = best_h if compute_c5(best_h) <= s_bin2 else h_bin2
        h_ls2, s_ls2 = binary_ls(start2, n0, min(70, tleft() * 0.7),
                                  max_iter=10000, n_cand=50, chunk=12)
        if s_ls2 < best_score:
            best_score = s_ls2
            best_h = h_ls2

    if tleft() > 350:
        n1 = 1200
        target1 = n1 // 2
        h1 = interp_to(best_h, n1, target1)
        h1_bin = binarize(h1, n1)
        s1c = compute_c5(h1)
        s1b = compute_c5(h1_bin)
        h1_work = h1 if s1c <= s1b else h1_bin

        if tleft() > 300:
            h1_ls, s1_ls = binary_ls(h1_work, n1, min(100, tleft() * 0.4),
                                      max_iter=10000, n_cand=50, chunk=10)
            if s1_ls < best_score:
                best_score = s1_ls
                best_h = h1_ls

        if tleft() > 180:
            h1_cd, s1_cd = pair_cd(best_h, len(best_h), len(best_h) // 2,
                                    min(150, tleft() * 0.5),
                                    n_cand=35, max_sweeps=6000, weight_beta=5.0)
            if s1_cd < best_score:
                best_score = s1_cd
                best_h = h1_cd

        if tleft() > 80:
            h1_sl, s1_sl = slsqp_polish(best_h, len(best_h), len(best_h) // 2,
                                         min(70, tleft() * 0.7))
            if s1_sl < best_score:
                best_score = s1_sl
                best_h = h1_sl

    if tleft() > 50:
        n_cur = len(best_h)
        h_fin, s_fin = pair_cd(best_h, n_cur, n_cur // 2,
                                min(40, tleft() * 0.6),
                                n_cand=50, max_sweeps=4000, weight_beta=8.0)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin

    if tleft() > 25:
        n_cur = len(best_h)
        h_f2, s_f2 = slsqp_polish(best_h, n_cur, n_cur // 2,
                                   min(20, tleft() * 0.7))
        if s_f2 < best_score:
            best_score = s_f2
            best_h = h_f2

    final_h = project_feasible(best_h, len(best_h) // 2)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
