        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new

To minimize ∑h[i] * (1-h[i]) = n/2 - ∑h[i]², we want to maximize ∑h[i]². Since h[i] ∈ [0,1] and ∑h[i] = n/2, ∑h[i]² is maximized when h is as "binary" as possible (half 0s, half 1s), giving ∑h[i]² = n/2, and the zero-lag overlap is n/2 - n/2 = 0.

But for non-zero shifts, binary h might give high overlap. So there's a trade-off.

This confirms that the optimal solution is somewhere between "smooth" and "binary."

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, args=(),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                h_cur = h_new
        except Exception:

def spectral_shaping(h_init, target, n_iter, strength):
    n = len(h_init)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    freqs = np.arange(n // 2 + 1)
    for it in range(n_iter):
        H = np.fft.rfft(h)
        P = np.abs(H) ** 2
        P_total = P.sum()
        if P_total < 1e-30:
            break
        P_norm = P / P_total
        target_P = np.exp(-strength * freqs / max(n * 0.05, 1.0))
        target_P = target_P / target_P.sum()
        alpha_blend = 0.08 + 0.12 * (it / max(n_iter, 1))
        new_P = (1.0 - alpha_blend) * P_norm + alpha_blend * target_P
        new_P = new_P / new_P.sum() * P_total
        phases = np.angle(H)
        new_H = np.sqrt(new_P) * np.exp(1j * phases)
        new_h = np.fft.irfft(new_H, n=n)
        new_h = project_feasible(new_h.real, target)
        sc = compute_c5(new_h)
        if sc < best_c5 - 1e-15:
            best_c5 = sc
            best_h = new_h.copy()
        h = new_h
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
