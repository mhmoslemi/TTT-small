import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = idxs + k < n
        wm = idxs - k >= 0
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < len(shifts) - 1 - (n - 1 - p))
    n = len(h)
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                if i == j or h[i] <= 1e-12:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                delta = min(delta_max, 0.05)
                h_trial = h.copy()
                h_trial[i] -= delta
                h_trial[j] += delta
                h_trial = project_feasible(h_trial, target)
                sc = compute_c5(h_trial)
                if sc < best_trial - 1e-15:
                    best_trial = sc
                    best_update = (i, j, delta)

        if best_update is not None:
            i, j, delta = best_update
            h[i] -= delta
            h[j] += delta
            h = project_feasible(h, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    if tleft() > 180:
        h_pol, sc = nesterov_projected(best_h, best_n, best_target,
                                       time_limit=min(120, tleft() * 0.6),
                                       alpha_start=50.0, alpha_end=20000.0,
                                       alpha_steps=40, lr_start=0.08,
                                       lr_decay=0.9997, max_steps=30000)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 120:
        h_cd, c5_cd = fast_pair_cd(best_h, best_n, best_target,
                                   time_limit=min(80, tleft() * 0.5),
                                   max_sweeps=30000,
                                   n_cand_start=30, n_cand_end=80,
                                   max_no_improve=300)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol2, sc2 = slsqp_round(best_h, best_n, best_target,
                                  [2000, 10000, 50000], 1500, 1e-15)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    resolutions = [600, 800, 1000, 1200]
    for n_new in resolutions:
        if tleft() < 150:
            break
        target_new = n_new / 2.0
        h_new = interpolate_to(best_h, n_new, target_new)
        sc_new = compute_c5(h_new)
        time_for_stage = min(tleft() * 0.45, 200)

        if tleft() > 100:
            h_nm, sc_nm = nesterov_projected(h_new, n_new, target_new,
                                             time_limit=min(60, time_for_stage * 0.3),
                                             alpha_start=100.0, alpha_end=50000.0,
                                             alpha_steps=30, lr_start=0.04,
                                             lr_decay=0.9998, max_steps=20000)
            if sc_nm < sc_new:
                h_new = h_nm
                sc_new = sc_nm

        if tleft() > 80:
            h_cd, c5_cd = fast_pair_cd(h_new, n_new, target_new,
                                       time_limit=min(70, time_for_stage * 0.4),
                                       max_sweeps=30000,
                                       n_cand_start=30, n_cand_end=90,
                                       max_no_improve=350)
            if c5_cd < sc_new:
                h_new = h_cd
                sc_new = c5_cd

        if tleft() > 60:
            h_bl, c5_bl = block_cd(h_new, n_new, target_new,
                                   time_limit=min(30, time_for_stage * 0.15),
                                   block_size=5, n_cand=40, max_sweeps=5000,
                                   max_no_improve=200)
            if c5_bl < sc_new:
                h_new = h_bl
                sc_new = c5_bl

        if tleft() > 50:
            h_s, sc_s = slsqp_round(h_new, n_new, target_new,
                                    [5000, 20000], 1000, 1e-15)
            if sc_s < sc_new:
                h_new = h_s
                sc_new = sc_s

        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
            best_n = n_new
            best_target = target_new

    if tleft() > 80:
        h_cd_final, c5_cf = fast_pair_cd(best_h, best_n, best_target,
                                         time_limit=min(60, tleft() * 0.5),
                                         max_sweeps=40000,
                                         n_cand_start=50, n_cand_end=120,
                                         max_no_improve=250)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cd_final

    if tleft() > 60:
        h_nest, sc_nest = nesterov_projected(best_h, best_n, best_target,
                                             time_limit=min(40, tleft() * 0.4),
                                             alpha_start=5000.0, alpha_end=200000.0,
                                             alpha_steps=20, lr_start=0.02,
                                             lr_decay=0.9999, max_steps=20000)
        if sc_nest < best_score:
            best_score = sc_nest
            best_h = h_nest

    if tleft() > 40:
        h_pol3, sc3 = slsqp_round(best_h, best_n, best_target,
                                  [10000, 50000, 100000], 2000, 1e-15)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    if tleft() > 40:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.05, 0.08]:
            if tleft() < 25:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            for _ in range(3):
                if tleft() < 20:
                    break
                h_s, sc = slsqp_round(h_pert, best_n, best_target,
                                      [2000, 10000], 500, 1e-14)
                if sc < best_score:
                    best_score = sc
                    best_h = h_s
                    h_pert = h_s + rng.randn(best_n) * 0.02
                    h_pert = project_feasible(h_pert, best_target)
                else:
                    break

    if tleft() > 20:
        h_cd_last, c5_last = fast_pair_cd(best_h, best_n, best_target,
                                          time_limit=min(15, tleft() * 0.5),
                                          max_sweeps=10000,
                                          n_cand_start=60, n_cand_end=120,
                                          max_no_improve=100)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_cd_last

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
