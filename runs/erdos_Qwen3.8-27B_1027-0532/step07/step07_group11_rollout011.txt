    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return c_max + log_sum / alpha

def smooth_max_grad(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    return weights

def get_corr_gradient(h, weights):
    """Compute Σ_k w_k * ∂corr_k/∂h for all coordinates of h."""
    n = len(h)
    n_corr = 2 * n - 1
    mid = n - 1

        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    return exp_s / np.sum(exp_s)

def corr_gradient(h, weights):
    n = len(h)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s in range(n_corr):
        w = weights[s]
        if w < 1e-14:
            continue
        k = s - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad

def compute_change_col(h, p, n):
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    col = np.zeros(n_corr)
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    col[pos] += (1.0 - h[p + shifts[pos]])
    col[neg] -= h[p - shifts[neg]]
    return col

def slsqp_phase(h_init, n, target, alphas, time_limit, maxiter_base=400):

             'jac': lambda x: np.ones(n)}]

    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        elapsed = time.time() - start

        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max_val(corr, alpha)
        def grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            w = smooth_max_weights(corr, alpha)
            return corr_gradient(x, w)
        try:
            res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)

                    best_i = i
                    best_j = j
                    best_delta = delta_star
                elif temp > 1e-10 and approx_max < c5_val + 0.001:
                    accept_prob = np.exp(-(approx_max - c5_val) / (temp * dx))
                    if np.random.random() < accept_prob:
                        best_trial = approx_max
                        best_i = i
                        best_j = j
                        best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i, :] = compute_change_col(h, best_i, n)
            change[best_j, :] = compute_change_col(h, best_j, n)
            changes += 2
            if changes > n // 2:
                for p in range(n):
                    change[p, :] = compute_change_col(h, p, n)
                changes = 0
            temp *= anneal_decay
        else:
            temp = max(temp * 0.95, 1e-10)
    return best_h, best_c5

def subgradient_polish(h_init, n, target, time_limit, n_steps=20000, step_scale=0.003):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        top_k = np.argsort(corr)[-3:]
        p = np.exp(corr * 80.0)
        p /= p.sum()
        grad = np.zeros(n)
        for s_idx in top_k:
            k = int(s_idx) - mid
            w = p[s_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= grad.mean()
        h_try = project_feasible(h - eta * grad, target)
        sc = compute_c5(h_try)
        if sc < c5_val - 1e-15:
            h = h_try
            eta = min(eta * 1.02, 0.05)
        else:
            eta *= 0.9
            if eta < 1e-6:
                eta = step_scale * 0.5
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def constrained_feasibility(h_init, n, target, t_target, time_limit):
    start = time.time()
    corr = np.correlate(h_init, 1.0 - h_init, mode='full')
    thresh = t_target * n / 2.0
    active = np.where(corr > thresh * 0.7)[0]
    if len(active) == 0:
        return h_init.copy(), compute_c5(h_init)
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    for s_idx in active:
        k = s_idx - (n - 1)
        idxs = np.arange(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        def make_constraint(v1m, v2m, k_val, h_full):
            def c(x):
                val = np.zeros(2 * len(h_full) - 1)
                val[(v1m)] = x[v1m] * (1.0 - x[v1m + k_val] if k_val >= 0 else x[v1m - k_val])
                return 0.0
            return c
        for s in active:
            k = s - (n - 1)
            def make_ineq(s_val):
                k_val = s_val - (n - 1)
                idxs_l = np.arange(n)
                v1m = (idxs_l - k_val >= 0) & (idxs_l - k_val < n)
                v2m = (idxs_l + k_val >= 0) & (idxs_l + k_val < n)
                def c(x):
                    parts = []
                    if np.any(v1m):
                        pp = idxs_l[v1m]
                        if k_val >= 0:
                            parts.append(x[pp] * (1.0 - x[pp + k_val]))
                        else:
                            parts.append(x[pp] * (1.0 - x[pp - k_val]))
                    total = np.sum(parts) if parts else 0.0
                    return total - thresh
                def j(x):
                    jac = np.zeros(n)
                    if np.any(v1m):
                        pp = idxs_l[v1m]
                        if k_val >= 0:
                            jac[pp] += (1.0 - x[pp + k_val])
                            jac[pp + k_val] -= x[pp]
                        else:
                            jac[pp] += (1.0 - x[pp - k_val])
                            jac[pp - k_val] -= x[pp]
                    return jac
                return {'type': 'ineq', 'fun': lambda x, c_fn=c: -c_fn(x), 'jac': lambda x, j_fn=j: -j_fn(x)}
            cons.append({'type': 'ineq', 'fun': (lambda x, s_val=s: -(np.correlate(x, 1.0-x, mode='full')[s_val] - thresh)),
                         'jac': None})
    def obj(x):
        corr_x = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_val(corr_x, 500.0)
    try:
        res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                      constraints=cons[:min(len(active)+1, 20)],
                      options={'maxiter': 300, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            return h_new, sc
    except Exception:
        pass
    return h_cur, cur_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)
    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    if tleft() > 80:
        h_pol, sc = slsqp_phase(best_h, n0, target0, [50, 200, 800],
                                min(70, tleft() - 5), 400)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
    if tleft() > 100:
        h_sub, sc = subspace_optimize(best_h, n0, target0,
                                      min(80, tleft() - 5), sub_size=8, alpha=200, n_random=1)
        if sc < best_score:
            best_score = sc
            best_h = h_sub
    if tleft() > 400:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if tleft() > 300:
            h_cd, c5_cd = pair_cd_optimize(h1, n1, target1,
                                           min(250, tleft() * 0.5),
                                           n_cand=40, n_shifts=3,
                                           max_no_improve=500,
                                           anneal_temp=0.01, anneal_decay=0.9997)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
            elif sc1 < best_score:
                best_score = sc1
                best_h = h1
        if tleft() > 100:
            h_sub1, sc_s1 = subspace_optimize(best_h, len(best_h), len(best_h)/2.0,
                                              min(80, tleft() - 5), sub_size=10, alpha=300, n_random=2)
            if sc_s1 < best_score:
                best_score = sc_s1
                best_h = h_sub1
    if tleft() > 80:
        cur_n = len(best_h)
        h_pol, sc = slsqp_phase(best_h, cur_n, cur_n / 2.0,
                                [1000, 5000], min(70, tleft() - 5), 500)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
    if tleft() > 200:
        h_cd, c5_cd = pair_cd_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                       min(150, tleft() * 0.5),
                                       n_cand=60, n_shifts=4,
                                       max_no_improve=400,
                                       anneal_temp=0.005, anneal_decay=0.999)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
    if tleft() > 350:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 250:
            h_cd2, c5_cd2 = pair_cd_optimize(h2, n2, target2,
                                             min(200, tleft() * 0.5),
                                             n_cand=35, n_shifts=3,
                                             max_no_improve=400,
                                             anneal_temp=0.005, anneal_decay=0.9998)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
            elif compute_c5(h2) < best_score:
                best_score = compute_c5(h2)
                best_h = h2
    if tleft() > 80:
        cur_n = len(best_h)
        h_sg, c5_sg = subgradient_polish(best_h, cur_n, cur_n / 2.0,
                                         min(60, tleft() - 5), n_steps=15000, step_scale=0.002)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg
    if tleft() > 60:
        rng = np.random.RandomState(seed)
        cur_n = len(best_h)
        for ns in [0.03, 0.08, 0.15]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(cur_n) * ns
            h_pert = project_feasible(h_pert, cur_n / 2.0)
            h_ref, sc = slsqp_phase(h_pert, cur_n, cur_n / 2.0,
                                    [200, 800, 3000], min(30, tleft() - 5), 300)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
    if tleft() > 50:
        h_final, sc = pair_cd_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                       min(35, tleft() - 5),
                                       n_cand=80, n_shifts=5,
                                       max_no_improve=200,
                                       anneal_temp=0.0, anneal_decay=1.0)
        if sc < best_score:
            best_score = sc
            best_h = h_final
    if tleft() > 30:
        h_pol, sc = slsqp_phase(best_h, len(best_h), len(best_h) / 2.0,
                                [5000, 20000], min(25, tleft() - 5), 600)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
