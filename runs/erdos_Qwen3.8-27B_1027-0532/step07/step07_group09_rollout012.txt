def _obj_tophat(x, n_top):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    sorted_c = np.sort(corr)[::-1]
    k = min(n_top, len(corr))
    top_avg = np.mean(sorted_c[:k])
    shifted = np.clip(30.0 * (corr - cmax), -500.0, 0.0)
    lse = cmax + np.log(np.sum(np.exp(shifted))) / 30.0
    return 0.7 * top_avg + 0.3 * lse

def _cons_eq(x):
    return np.sum(x) - _gs_target

def _cons_eq_jac(x):
    return np.ones(_gs_n)

def slsqp_round(h_start, n, target, alphas, n_tops, maxiter, ftol=1e-14):
    global _gs_n, _gs_target
    _gs_n = n
    _gs_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_obj_lse, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    for nt in n_tops:
        try:
            res = minimize(_obj_tophat, h_cur, args=(nt,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd_enhanced(h_init, n, target, time_limit, max_sweeps=50000,
                     n_cand_min=30, n_cand_max=80, max_no_improve=500,
                     adapt_active=True):
    """Enhanced coordinate descent with adaptive active set and more candidates."""
    start_t = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive active set
        cmax = np.max(corr)
        if adapt_active:
            # Use all shifts within 95% of max
            tol = 0.05 * cmax + 1e-10
            active_idx = np.where(corr >= cmax - tol)[0]
            if len(active_idx) > 30:
                # Keep top 30
                active_idx = np.argsort(corr)[-30:]
        else:
            n_active = min(8, n_corr)
            active_idx = np.argsort(corr)[-n_active:]

        # Weighted gradient from active shifts
        w = corr[active_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(len(active_idx)) / len(active_idx)

        grad = np.zeros(n)
        for wi, si in zip(w, active_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Progressive candidate selection
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_min + (n_cand_max - n_cand_min) * progress)
        ncand = min(ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=np.int64)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        # Efficient pair search using the change matrix
        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = cand_map[i]
            h_i = h[i]
            if h_i < 1e-10:
                continue
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = cand_map[j]
                h_j = h[j]
                if h_j > 1.0 - 1e-10:
                    continue

                delta_max = min(h_i, 1.0 - h_j)
                if delta_max < 1e-12:
                    continue

                g = change[lr] - change[hr]
                
                # Binary search for best delta
                a_d = 0.0
                b_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_search(h_best, n, target, time_limit, seed=42):
    """Try multiple structurally different perturbations."""
    start_t = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    
    strategies = [
        ('gauss', 0.03),
        ('gauss', 0.07),
        ('block_swap', 0),
        ('shift', 0),
        ('compress', 0),
    ]
    
    for strat, scale in strategies:
        if time.time() - start_t > time_limit:
            break
        
        h_pert = best_h.copy()
        if strat == 'gauss':
            h_pert = best_h + rng.randn(n) * scale
        elif strat == 'block_swap':
            # Swap two random blocks
            bs = n // 10
            p1 = rng.randint(0, n - 2 * bs)
            p2 = p1 + bs + rng.randint(0, n - 2 * p1 - bs)
            if p2 + bs <= n:
                tmp = h_pert[p1:p1+bs].copy()
                h_pert[p1:p1+bs] = h_pert[p2:p2+bs]
                h_pert[p2:p2+bs] = tmp
        elif strat == 'shift':
            # Circular shift by a few points
            sh = rng.randint(2, 10)
            h_pert = np.roll(best_h, sh)
        elif strat == 'compress':
            # Compress towards center
            x = np.arange(n) / n
            x_new = 0.5 + 0.4 * (x - 0.5)
            h_pert = np.interp(x_new, x, best_h)
        
        h_pert = project_feasible(h_pert, target)
        
        # Quick SLSQP polish
        h_pol, sc = slsqp_round(h_pert, n, target, [500, 2000], [5, 10], 300, 1e-13)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_pol
    
    return best_h, best_c5


def subgradient_refine(h_init, n, target, time_limit, lr_start=0.005, decay=0.97):
    """Subgradient method with decay for final polishing."""
    start_t = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_start
    no_improve = 0
    
    while time.time() - start_t < time_limit:
        if no_improve > 2000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        c5_val = cmax * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            lr = min(lr_start, lr * 1.1)
        else:
            no_improve += 1
            lr *= decay
        
        # Use top-m active shifts
        tol = max(1e-8, 0.02 * cmax)
        active = np.where(corr >= cmax - tol)[0]
        
        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)
        
        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val - 1e-15:
            h = h_new
    
    return best_h, best_c5


def block_shift_search(h_init, n, target, time_limit, max_steps=30000):
    """Move contiguous blocks to reduce overlap."""
    start_t = time.time()
    rng = np.random.RandomState(12345)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    change = build_change_matrix(h, n)
    block_sizes = [1, 2, 3, 4, 5, 7, 10]
    
    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        if no_improve > 3000:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))
        
        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        # Try moving a block
        bs = block_sizes[step % len(block_sizes)]
        if n - bs < 2:
            continue
        
        # Find best source position (high gradient sum = want to decrease)
        block_grads = np.array([grad[i:i+bs].sum() for i in range(n - bs + 1)])
        # Source: most negative (want to decrease h here)
        src = int(np.argmin(block_grads))
        # Target: most positive (want to increase h here)
        dst = int(np.argmax(block_grads))
        
        if src == dst:
            continue
        
        src_block = h[src:src+bs]
        dst_block = h[dst:dst+bs]
        
        delta_max = min(src_block.max(), 1.0 - dst_block.min())
        if delta_max < 1e-12:
            continue
        
        # Compute change vector
        g = np.zeros(n_corr)
        for t in range(bs):
            g += change[dst + t, :] - change[src + t, :]
        
        # Binary search
        a_d = 0.0
        b_d = delta_max
        for _ in range(12):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr + mid_d * g
            f_mid = np.max(vals)
            active_mask = vals >= f_mid - 1e-14
            if active_mask.any():
                f_prime = np.max(g[active_mask])
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                b_d = mid_d
            else:
                a_d = mid_d
        
        delta_star = 0.5 * (a_d + b_d)
        if delta_star > 1e-12:
            approx_max = np.max(corr + delta_star * g)
            if approx_max < c5_val - 1e-14:
                h[src:src+bs] -= delta_star
                h[dst:dst+bs] += delta_star
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                for t in range(bs):
                    update_change_row(change, src + t, h, n, shifts)
                    update_change_row(change, dst + t, h, n, shifts)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_gs_n = 0
_gs_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _obj_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    return float(cmax + np.log(np.sum(np.exp(shifted))) / alpha)


def _obj_tophat(x, n_top):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    sorted_c = np.sort(corr)[::-1]
    k = min(n_top, len(corr))
    top_avg = np.mean(sorted_c[:k])
    shifted = np.clip(40.0 * (corr - cmax), -500.0, 0.0)
    lse = cmax + np.log(np.sum(np.exp(shifted))) / 40.0
    return 0.65 * top_avg + 0.35 * lse


def _obj_power(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    top_k = max(5, len(corr) // 8)
    sorted_c = np.sort(corr)[::-1]
    val = np.sum(sorted_c[:top_k] ** p) ** (1.0 / p)
    return val


def _cons_eq(x):
    return np.sum(x) - _gs_target


def _cons_eq_jac(x):
    return np.ones(_gs_n)


def slsqp_round(h_start, n, target, alphas, n_tops, p_vals, maxiter, ftol=1e-14):
    global _gs_n, _gs_target
    _gs_n = n
    _gs_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_obj_lse, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    for nt in n_tops:
        try:
            res = minimize(_obj_tophat, h_cur, args=(nt,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

        try:
            res = minimize(_obj_power, h_cur, args=(p,),

            sc = compute_c5(h_new)
            if sc < cur_score:

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def pair_cd_v2(h_init, n, target, time_limit, max_sweeps=50000,

    n_corr = 2 * n - 1

    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for sweep in range(max_sweeps):

            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if adapt_active:

            active_idx = np.where(corr >= cmax - tol)[0]

                active_idx = np.argsort(corr)[-25:]
        else:
            n_act = min(6, n_corr)
            active_idx = np.argsort(corr)[-n_act:]

        w = corr[active_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(len(active_idx)) / len(active_idx)

        grad = np.zeros(n)
        for wi, si in zip(w, active_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_min + (n_cand_max - n_cand_min) * progress)
        ncand = min(ncand, n // 3)

        best_j = -1

            hr = cand_map[i]
            h_i = h[i]
            if h_i < 1e-10:
                continue
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = cand_map[j]
                h_j = h[j]
                if h_j > 1.0 - 1e-10:
                    continue

                delta_max = min(h_i, 1.0 - h_j)
                if delta_max < 1e-12:
                    continue

                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:

                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def block_shift_search(h_init, n, target, time_limit, max_steps=30000):

    rng = np.random.RandomState(9999)
    dx = 2.0 / n

    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    change = build_change_matrix(h, n)
    block_sizes = [1, 2, 3, 4, 6, 8, 12]

    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        bs = block_sizes[step % len(block_sizes)]
        if n - bs < 2:
            continue

        n_pos = n - bs + 1
        block_grads = np.array([grad[i:i + bs].sum() for i in range(n_pos)])
        src = int(np.argmin(block_grads))
        dst = int(np.argmax(block_grads))

        if src == dst:
            continue

        src_block = h[src:src + bs]
        dst_block = h[dst:dst + bs]

        delta_max = min(src_block.max(), 1.0 - dst_block.min())
        if delta_max < 1e-12:
            continue

        g = np.zeros(n_corr)
        for t in range(bs):
            g += change[dst + t, :] - change[src + t, :]

        a_d = 0.0
        b_d = delta_max
        for _ in range(11):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr + mid_d * g
            f_mid = np.max(vals)
            active_mask = vals >= f_mid - 1e-14
            if active_mask.any():
                f_prime = np.max(g[active_mask])
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                b_d = mid_d
            else:
                a_d = mid_d

        delta_star = 0.5 * (a_d + b_d)
        if delta_star > 1e-12:
            approx_max = np.max(corr + delta_star * g)
            if approx_max < c5_val - 1e-14:
                h[src:src + bs] -= delta_star
                h[dst:dst + bs] += delta_star
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                for t in range(bs):
                    update_change_row(change, src + t, h, n, shifts)
                    update_change_row(change, dst + t, h, n, shifts)

    return best_h, best_c5


def multi_start_search(h_best, n, target, time_limit, seed=42):
    start_t = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)

    perturbations = [
        (0, 0.02), (0, 0.05), (0, 0.1),
        (1, 0), (2, 0), (3, 0.03)
    ]

    for ptype, scale in perturbations:
        if time.time() - start_t > time_limit:
            break

        h_pert = best_h.copy()
        if ptype == 0:
            h_pert = best_h + rng.randn(n) * scale
        elif ptype == 1:
            bs = max(3, n // 15)
            p1 = rng.randint(0, n - 2 * bs)
            p2 = p1 + bs + rng.randint(0, max(1, n - 2 * p1 - bs))
            if p2 + bs <= n:
                tmp = h_pert[p1:p1 + bs].copy()
                h_pert[p1:p1 + bs] = h_pert[p2:p2 + bs]
                h_pert[p2:p2 + bs] = tmp
        elif ptype == 2:
            sh = rng.randint(3, max(4, n // 20))
            h_pert = np.roll(best_h, sh)
        else:
            x = np.arange(n) / float(n)
            factor = 0.3 + 0.4 * (ptype - 2)
            x_new = 0.5 + factor * (x - 0.5)
            x_new = np.clip(x_new, 0.0, 1.0)

def subgradient_refine(h_init, n, target, time_limit, lr_start=0.004, decay=0.975):
    start_t = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_start
    no_improve = 0

    while time.time() - start_t < time_limit:
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        c5_val = cmax * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            lr = min(lr_start, lr * 1.05)
        else:
            no_improve += 1
            lr *= decay

        tol = max(1e-8, 0.03 * cmax)
        active = np.where(corr >= cmax - tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)

        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val - 1e-15:
            h = h_new

    return best_h, best_c5

            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
