from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-15) if diff > 0 else (h > 1e-15)
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)

def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1

        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]

def compute_change_matrix(h, n):

    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

                    f_mid = np.max(vals_d)
                    active = vals_d >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        else:
            break

    return best_h, best_c5

def gradient_descent_active(h_init, n, target, time_limit, max_steps=100000,
                            base_lr=0.02, n_active=8, lr_decay=0.9999):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.9
                if lr < 1e-6:
                    break

        top_shifts = np.argsort(corr)[-n_active:]
        c_max = corr[top_shifts[-1]]
        weights = np.exp(-50.0 * (c_max - corr[top_shifts]))
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for wi, si in zip(weights, top_shifts):
            if wi < 1e-10:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0)
            v2 = (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)

        sc = compute_c5(h_try)
        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
        else:
            lr *= 0.5
            if lr < 1e-8:
                break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
