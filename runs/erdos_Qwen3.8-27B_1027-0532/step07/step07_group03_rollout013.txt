        weights = np.exp(np.clip(exp_arg, -500.0, 0.0))
        sw = np.sum(weights)
        if sw < 1e-15:
            weights = np.ones(len(active_lags)) / len(active_lags)
        else:
            weights = weights / sw

        k_shifts = active_lags - mid
        g = multi_grad(h, k_shifts, weights, n)

        cuts_g.append(g)
        cuts_f.append(f_val)
        cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                continue
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new

            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < 1e-10:
                break
        except Exception:
            continue

    return best_h, best_score


def interceptive_cd(h_init, n, target, time_limit, max_sweeps=50000,
                    n_cand_start=25, n_cand_end=110, max_no_improve=800, bisect_steps=14):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consec_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consec_no_improve = 0
        else:
            consec_no_improve += 1

        if consec_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = corr_grad_at(h, k_star, n)
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisect_steps):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for px in (best_i, best_j):
                change[px, :] = 0.0
                pos_p = (px + shifts >= 0) & (px + shifts < n)
                neg_p = (px - shifts >= 0) & (px - shifts < n)
                change[px, pos_p] += (1.0 - h[px + shifts[pos_p]])
                change[px, neg_p] -= h[px - shifts[neg_p]]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def nesterov_agg(h_init, n, target, time_limit, n_iters=5000):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    h_cur = h_init.copy().astype(float)
    h_prev = h_cur.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    idxs = np.arange(n)
    beta = 0.9

    for it in range(n_iters):
        if time.time() - start > time_limit:
            break

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * beta)) / 2.0
        y = h_cur + ((t_k - 1.0) / t_k) * (h_cur - h_prev)
        y = np.clip(y, 0.0, 1.0)

        alpha = 100.0 + 500.0 * (it / max(n_iters, 1))
        corr_y = compute_corr(y)
        c_max = np.max(corr_y)
        shifted = np.clip(alpha * (corr_y - c_max), -500.0, 0.0)
        p = np.exp(shifted)
        p = p / np.sum(p)

        grad = np.zeros(n)
        for m in range(len(corr_y)):
            if p[m] < 1e-12:
                continue
            k = m - mid
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += p[m] * (1.0 - y[idxs[v1] + k])
            grad[v2] -= p[m] * y[idxs[v2] - k]
        grad /= alpha

        step_size = 0.05
        h_new = y - step_size * grad
        h_new = project_feasible(h_new, target)
        h_prev = h_cur.copy()
        h_cur = h_new
        sc = compute_c5(h_cur)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h_cur.copy()

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    remaining = lambda: budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0

    if remaining() > 180:
        h_sls, sc = slsqp_refine(best_h, n0, target0, [200, 1000, 5000], 700, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_sls

    if remaining() > 350:
        h_cp, sc_cp = cutting_plane(best_h, n0, target0,
                                     time_limit=min(350, remaining() * 0.45),
                                     max_iters=15000, max_cuts=350)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp

    if remaining() > 250:
        h_cd, sc_cd = interceptive_cd(best_h, n0, target0,
                                       time_limit=min(250, remaining() * 0.4),
                                       max_sweeps=40000,
                                       n_cand_start=25, n_cand_end=100,
                                       max_no_improve=600, bisect_steps=14)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if remaining() > 200:
        h_cp2, sc_cp2 = cutting_plane(best_h, n0, target0,
                                       time_limit=min(200, remaining() * 0.4),
                                       max_iters=12000, max_cuts=300)
        if sc_cp2 < best_score:
            best_score = sc_cp2
            best_h = h_cp2

    if remaining() > 50:
        h_nest, sc_nest = nesterov_agg(best_h, n0, target0,
                                       time_limit=min(50, remaining() * 0.4), n_iters=3000)
        if sc_nest < best_score:
            best_score = sc_nest
            best_h = h_nest

    if remaining() > 450:
        n800 = 800
        target800 = n800 / 2.0
        h800 = interpolate_to(best_h, n800, target800)
        sc800 = compute_c5(h800)

        if sc800 < best_score:
            best_score = sc800
            best_h = h800
            best_n = n800

        if remaining() > 350:
            h800_cp, sc800_cp = cutting_plane(best_h, best_n, best_n / 2.0,
                                               time_limit=min(300, remaining() * 0.45),
                                               max_iters=10000, max_cuts=250)
            if sc800_cp < best_score:
                best_score = sc800_cp
                best_h = h800_cp

        if remaining() > 200:
            h800_cd, sc800_cd = interceptive_cd(best_h, best_n, best_n / 2.0,
                                                 time_limit=min(180, remaining() * 0.45),
                                                 max_sweeps=20000,
                                                 n_cand_start=20, n_cand_end=60,
                                                 max_no_improve=500, bisect_steps=12)
            if sc800_cd < best_score:
                best_score = sc800_cd
                best_h = h800_cd

        if remaining() > 100:
            h800_cp2, sc800_cp2 = cutting_plane(best_h, best_n, best_n / 2.0,
                                                 time_limit=min(80, remaining() * 0.4),
                                                 max_iters=6000, max_cuts=200)
            if sc800_cp2 < best_score:
                best_score = sc800_cp2
                best_h = h800_cp2

    if remaining() > 80:
        nb = len(best_h)
        tb = nb / 2.0
        h_final_sls, sc_final = slsqp_refine(best_h, nb, tb,
                                              [5000, 20000], 600, 1e-15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_sls

    if remaining() > 60:
        nb = len(best_h)
        tb = nb / 2.0
        h_final_cd, sc_fc = interceptive_cd(best_h, nb, tb,
                                             time_limit=min(50, remaining() * 0.5),
                                             max_sweeps=8000,
                                             n_cand_start=30, n_cand_end=80,
                                             max_no_improve=300, bisect_steps=12)
        if sc_fc < best_score:
            best_score = sc_fc
            best_h = h_final_cd

    if remaining() > 60:
        nb = len(best_h)
        tb = nb / 2.0
        rng = np.random.RandomState(seed)
        for ns in (0.01, 0.03, 0.005):
            if remaining() < 40:
                break
            h_pert = best_h + rng.randn(nb) * ns
            h_pert = project_feasible(h_pert, tb)
            h_ref, sc_ref = cutting_plane(h_pert, nb, tb,
                                           time_limit=min(30, remaining() * 0.4),
                                           max_iters=3000, max_cuts=100)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if remaining() > 40:
        nb = len(best_h)
        tb = nb / 2.0
        h_last_cd, sc_last = interceptive_cd(best_h, nb, tb,
                                              time_limit=min(30, remaining() * 0.5),
                                              max_sweeps=5000,
                                              n_cand_start=40, n_cand_end=90,
                                              max_no_improve=200, bisect_steps=12)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last_cd

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
