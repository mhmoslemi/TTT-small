    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def thue_morse_n(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        h[i] = 1.0 if c % 2 == 0 else 0.0
    return project_feasible(h, n / 2.0)


def smooth_max_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_max_grad(corr, x, alpha):
    n = len(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    n_corr = len(corr)
    mid = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))

        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def stochastic_search(h_init, n, target, time_limit, max_steps=30000, seed_val=999):
    start = time.time()
    rng = np.random.RandomState(seed_val)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        progress = step / max_steps
        scale = 0.08 * (1.0 - progress) * max(0.3, 1.0 - progress)
        n_pert = int(rng.choice([2, 3, 5, 8, 15]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-15:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 300:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def gradient_flow(h_init, n, target, time_limit, max_iter=10000):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    idxs = np.arange(n)
    mid = n - 1
    alpha = 500.0
    step_size = 0.05
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 1000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        grad = smooth_max_grad(corr, h, alpha)
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        step = step_size / max(1.0, grad_norm)
        h_new = h - step * grad_norm * 0.1
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < cur_c5 - 1e-15:
            h = h_new
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                step_size *= 0.8

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_mu, _g_lam
    start_time = time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    def _time_left():
        return budget_s - 15.0 - (time.time() - start_time)

    # Phase 1: Quick L-BFGS-B polish at original resolution
    if _time_left() > 40:
        best_h, best_score = augmented_lbgfs(
            best_h, n0, target0, [100, 500, 2000, 10000],
            time_limit=min(35, _time_left() * 0.5), mu=50.0, maxiter=150)

    # Phase 2: Coordinate descent at original resolution
    if _time_left() > 50:
        best_h, best_score = pair_cd(
            best_h, n0, target0, time_limit=min(40, _time_left() * 0.5),
            max_sweeps=20000, n_cand=40, top_k_eval=50, n_shifts_grad=3)

    # Phase 3: Go to n=800
    if _time_left() > 180:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

        if _time_left() > 120:
            h1, sc1 = augmented_lbgfs(
                h1, n1, target1, [100, 500, 2000, 10000],
                time_limit=min(100, _time_left() * 0.5), mu=30.0, maxiter=200)
            if sc1 < best_score:
                best_score = sc1
                best_h = h1

        if _time_left() > 80:
            h1, sc1 = pair_cd(
                h1, n1, target1, time_limit=min(60, _time_left() * 0.5),
                max_sweeps=25000, n_cand=50, top_k_eval=50, n_shifts_grad=3)
            if sc1 < best_score:
                best_score = sc1
                best_h = h1

    # Phase 4: Try Thue-Morse at n=800
    if _time_left() > 80:
        n_tm = 800
        target_tm = n_tm / 2.0
        h_tm = thue_morse_n(n_tm)
        sc_tm = compute_c5(h_tm)
        if _time_left() > 60:
            h_tm, sc_tm = augmented_lbgfs(
                h_tm, n_tm, target_tm, [100, 500, 2000],
                time_limit=min(50, _time_left() * 0.5), mu=30.0, maxiter=150)
            if _time_left() > 20:
                h_tm, sc_tm = pair_cd(
                    h_tm, n_tm, target_tm, time_limit=min(20, _time_left() * 0.5),
                    max_sweeps=10000, n_cand=40, top_k_eval=40, n_shifts_grad=2)
            if sc_tm < best_score:
                best_score = sc_tm
                best_h = h_tm

    # Phase 5: Try higher resolution n=1200
    if _time_left() > 150:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

        if _time_left() > 100:
            h2, sc2 = augmented_lbgfs(
                h2, n2, target2, [100, 500, 2000],
                time_limit=min(80, _time_left() * 0.5), mu=20.0, maxiter=150)
            if sc2 < best_score:
                best_score = sc2
                best_h = h2

        if _time_left() > 50:
            h2, sc2 = pair_cd(
                h2, n2, target2, time_limit=min(40, _time_left() * 0.5),
                max_sweeps=15000, n_cand=40, top_k_eval=40, n_shifts_grad=2)
            if sc2 < best_score:
                best_score = sc2
                best_h = h2

    # Phase 6: Stochastic search for escape
    if _time_left() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_ss, sc_ss = stochastic_search(
            best_h, n_cur, target_cur,
            time_limit=min(40, _time_left() * 0.5), max_steps=20000)
        if sc_ss < best_score:
            best_score = sc_ss
            best_h = h_ss

    # Phase 7: Block coordinate descent
    if _time_left() > 50:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_bc, sc_bc = block_cd(
            best_h, n_cur, target_cur,
            time_limit=min(35, _time_left() * 0.5),
            max_sweeps=15000, block_size=4, top_k_eval=15)
        if sc_bc < best_score:
            best_score = sc_bc
            best_h = h_bc

    # Phase 8: Gradient flow refinement
    if _time_left() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_gf, sc_gf = gradient_flow(
            best_h, n_cur, target_cur,
            time_limit=min(20, _time_left() * 0.5), max_iter=5000)
        if sc_gf < best_score:
            best_score = sc_gf
            best_h = h_gf

    # Phase 9: Multi-restart with perturbation
    if _time_left() > 60:
        rng = np.random.RandomState(seed)
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for noise_scale in [0.03, 0.08, 0.15]:
            if _time_left() < 20:
                break
            h_pert = best_h + rng.randn(n_cur) * noise_scale
            h_pert = project_feasible(h_pert, target_cur)
            h_pert, sc = augmented_lbgfs(
                h_pert, n_cur, target_cur, [200, 1000, 5000],
                time_limit=min(15, _time_left() * 0.4), mu=30.0, maxiter=100)
            if sc < best_score:
                best_score = sc
                best_h = h_pert

    # Phase 10: Final coordinate descent + L-BFGS-B
    if _time_left() > 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        best_h, best_score = pair_cd(
            best_h, n_cur, target_cur, time_limit=min(12, _time_left() * 0.5),
            max_sweeps=8000, n_cand=40, top_k_eval=50, n_shifts_grad=3)

    if _time_left() > 15:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        best_h, best_score = augmented_lbgfs(
            best_h, n_cur, target_cur, [5000, 20000],
            time_limit=min(10, _time_left() * 0.7), mu=20.0, maxiter=100)

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
