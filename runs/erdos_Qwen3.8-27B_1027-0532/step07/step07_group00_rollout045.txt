import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alpha_list, maxiter, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alpha_list:
        try:
            res = minimize(smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def nesterov_gradient(h_init, n, target, time_limit, p=8, n_iter=10000,
                       step_init=0.004, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    no_improve = 0

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_k = min(25, n_corr)
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            w = corr[s_idx] ** (p - 1)
            if w < 1e-20:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        v = momentum * v + grad
        h_new = h - step * v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < c5_val - 1e-16:
            h = h_new
            step = min(step * 1.03, 0.04)
        else:
            step *= 0.88
            v = 0.5 * v

    return best_h, best_c5

                                   top_k_shifts=30,
                                   momentum_decay=0.85)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 5: Nesterov gradient descent
    if tleft() > 120:
        t_ng = min(100, tleft() * 0.3)
        h_ng, sc_ng = nesterov_gradient(best_h, n0, target0,
                                         time_limit=t_ng,
                                         p=8, n_iter=8000,
                                         step_init=0.003,
                                         momentum=0.92)
        if sc_ng < best_score:
            best_score = sc_ng
            best_h = h_ng

    # Phase 6: Block CD for structured moves
    if tleft() > 100:
        t_bc = min(80, tleft() * 0.3)
        h_bc, sc_bc = block_cd(best_h, n0, target0,
                                time_limit=t_bc,
                                max_sweeps=8000,
                                block_size=3,
                                max_no_improve=200,
                                top_k_shifts=20)
        if sc_bc < best_score:
            best_score = sc_bc
            best_h = h_bc

    # Phase 7: Annealed CD to escape local minima
    if tleft() > 120:
        t_an = min(100, tleft() * 0.35)
        h_an, sc_an = improved_cd(best_h, n0, target0,
                                   time_limit=t_an,
                                   max_sweeps=15000,
                                   n_cand_start=20, n_cand_end=50,
                                   max_no_improve=400,
                                   weight_beta=4.0,
                                   use_anneal=True,
                                   T0=0.0008,
                                   top_k_shifts=25)
        if sc_an < best_score:
            best_score = sc_an
            best_h = h_an

    # Phase 8: Multi-start with structured perturbations
    if tleft() > 150:
        rng = np.random.RandomState(seed)
        perturbations = [
            (0.02, "noise"),
            (0.05, "noise"),
            (0.10, "noise"),
        ]
        for noise_scale, ptype in perturbations:
            if tleft() < 80:
                break
            if ptype == "noise":
                h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)

            t_sub = min(40, tleft() * 0.2)
            h_r, sc_r = slsqp_optimize(h_pert, n0, target0,
                                        [200, 2000, 8000], 300, 1e-14)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r

            if tleft() > 40 and best_h is h_r:
                t_cd_sub = min(25, tleft() * 0.2)
                h_cd_r, sc_cd_r = improved_cd(best_h, n0, target0,
                                               time_limit=t_cd_sub,
                                               max_sweeps=6000,
                                               n_cand_start=15, n_cand_end=40,
                                               max_no_improve=150,
                                               weight_beta=5.0)
                if sc_cd_r < best_score:
                    best_score = sc_cd_r
                    best_h = h_cd_r

    # Phase 9: High-alpha SLSQP for final polish
    if tleft() > 60:
        h_pol3, sc3 = slsqp_optimize(best_h, n0, target0,
                                      [10000, 50000, 100000], 500, 1e-14)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    # Phase 10: Final CD polish with aggressive parameters
    if tleft() > 50:
        t_final = min(35, tleft() * 0.6)
        h_fin_cd, sc_fin = improved_cd(best_h, n0, target0,
                                        time_limit=t_final,
                                        max_sweeps=10000,
                                        n_cand_start=25, n_cand_end=65,
                                        max_no_improve=200,
                                        weight_beta=8.0,
                                        top_k_shifts=35,
                                        momentum_decay=0.9)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin_cd

    # Final SLSQP
    if tleft() > 20:
        h_fin, sc_final = slsqp_optimize(best_h, n0, target0,
                                          [50000, 100000], 300, 1e-14)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_fin

    final_h = project_feasible(best_h, n0 / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, n0
