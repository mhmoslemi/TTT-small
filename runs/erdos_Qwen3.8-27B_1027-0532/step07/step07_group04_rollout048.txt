import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_point(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def topk_gradient(h, K):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    top_idx = np.argsort(corr)[-K:]
    c_max = np.max(corr)
    weights = np.exp(-3.0 * (c_max - corr[top_idx]) / max(c_max, 1e-10))
    weights = weights / np.sum(weights)
    idxs = np.arange(n)
    grad = np.zeros(n)
    for wi, k_idx in enumerate(top_idx):
        k = int(k_idx) - mid
        w = weights[wi]
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]
    return grad, c_max

def projected_gradient_optimize(h_start, target, time_limit, K=5, momentum=0.85,
                                 shake_interval=400, shake_scale=0.02):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    step = 0.01
    for it in range(2000000):
        if time.time() - start > time_limit:
            break
        if it > 0 and it % shake_interval == 0:
            h = project_point(h + np.random.randn(n) * shake_scale, target)
            v = np.zeros(n)
            step = 0.01
            continue
        grad, _ = topk_gradient(h, K)
        grad -= np.mean(grad)
        v = momentum * v - step * grad
        h_new = project_point(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < compute_c5(h) + 1e-14:
            h = h_new
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.5
            if step < 1e-6:
                step = 0.01
                v = np.zeros(n)
    return best_h, best_c5

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def multi_cd(h_start, target, time_limit, n_cand=60, n_shifts=3, max_no_improve=300):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_start.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = compute_change_matrix(h, n)
    for sweep in range(100000):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        top_shifts = np.argsort(corr)[-n_shifts:]
        c_max = corr[np.argmax(corr)]
        weights = np.exp(-4.0 * (c_max - corr[top_shifts]) / max(c_max, 1e-10))
        weights = weights / np.sum(weights)
        grad = np.zeros(n)
        for wi, s_idx in enumerate(top_shifts):
            k = int(s_idx) - mid
            w = weights[wi]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                d_lo = max(h[i] - 1.0, -h[j])
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi - d_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d, b_d = d_lo, d_hi
                for _ in range(14):
                    m_d = 0.5 * (a_d + b_d)
                    vals = corr + m_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = m_d
                    else:
                        a_d = m_d
                d_star = 0.5 * (a_d + b_d)
                if abs(d_star) < 1e-13:
                    continue
                approx = np.max(corr + d_star * g) * dx
                if approx < best_trial:
                    best_trial = approx
                    best_i = i
                    best_j = j
                    best_delta = d_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_point(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    return best_h, best_c5

def sa_cd(h_start, target, time_limit, n_cand=50, t_start=0.03, t_end=0.001):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    rng = np.random.RandomState(7)
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_start.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    no_improve = 0
    change = compute_change_matrix(h, n)
    for step in range(500000):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break
        progress = min(step / 100000.0, 1.0)
        temp = t_start * (t_end / t_start) ** progress
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue
        d_lo = max(h[i] - 1.0, -h[j])
        d_hi = min(h[i], 1.0 - h[j])
        if d_hi - d_lo < 1e-13:
            continue
        delta = d_lo + (d_hi - d_lo) * rng.rand()
        if abs(delta) < 1e-13:
            continue
        g = change[j] - change[i]
        approx_new = np.max(corr + delta * g) * dx
        dc = approx_new - c5_val
        accept = dc < -1e-15
        if not accept and temp > 1e-12:
            if rng.rand() < np.exp(-max(dc, 0.0) / max(temp * dx, 1e-15)):
                accept = True
        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_point(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            cur_c5 = approx_new
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_c5

def slsqp_smooth(h_start, target, alphas, maxiter, time_limit, start_time):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
        def cons_fun(x):
            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_point(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_point(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_point(h_out, target_dst)
