**Better idea: "Active set" SLSQP**

The key observation is that at the optimum, only a few shifts are active (achieving the max). If we can identify these, we can solve a much simpler problem:

(This is a quadratic program in h!)

The procedure:
1. Evaluate all g_k(h) for current h
2. Identify top m shifts (m = 5-10) where g_k is close to max
3. Solve the quadratic program: minimize sum_{k in active} g_k(h) s.t. constraints
4. Go to step 1

For step 3, since sum g_k(h) = (constant) - h^T Q h where Q is positive semi-definite (it's a sum of autocorrelation matrices), we're maximizing h^T Q h, which is a non-convex QP. But we can solve it with SLSQP since the gradient is linear in h: ∇(h^T Q h) = 2Qh.

       = sum_{i: valid} h[i] - sum_{i: valid} h[i] * h[i+k]

The first sum is a linear function of h (but the range of i depends on k).
The second sum is a quadratic form.

For the quadratic part: sum_{k in active} sum_{i: valid for k} h[i]*h[i+k]
∂/∂h[j] = sum_k (contribution of h[j] as h[i] term + contribution as h[i+k] term)

This is just (Qh)[j] where Q is the matrix with Q[j, l] = number of active k such that |l-j| = |k| and both j and l are valid for lag k.

import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)

def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = _g_n
    mid = n - 1
    idxs = np.arange(n)

        if weights[k_idx] < 1e-14:

        hi1 = min(n, n - k)

        if k >= 0:

        else:
            lo2, hi2 = 0, n
        if hi2 > lo2:
            grad[lo2:hi2] -= weights[k_idx] * x[lo2 - k:hi2 - k]
    return grad

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def l2norm_grad(h, n):
    """Compute gradient of 0.5 * sum_k corr[k]^2."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    mid = n - 1
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = corr[k_idx]
        if abs(w) < 1e-12:
            continue
        k = k_idx - mid
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 + k:hi1 + k])
        if k >= 0:
            lo2, hi2 = k, n
        else:
            lo2, hi2 = 0, n
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 - k:hi2 - k]
    grad -= np.mean(grad)
    return grad

def l2norm_optimize(h_init, n, target, time_limit, max_steps=30000, base_lr=0.02):
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = base_lr
    no_improve = 0
    best_l2 = float(np.sum(np.correlate(h, 1.0 - h, mode='full')**2))
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break
        
        grad = l2norm_grad(h, n)
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        
        l2_try = float(np.sum(np.correlate(h_try, 1.0 - h_try, mode='full')**2))
        c5_try = compute_c5(h_try)
        
        if l2_try < best_l2 - 1e-8:
            h = h_try
            best_l2 = l2_try
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.1)
        else:
            no_improve += 1
            lr *= 0.8
            if lr < 1e-9:
                break
    
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, n_cand=50, max_sweeps=10000, max_no_improve=300):
    """Pair coordinate descent."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                
                g = change[j] - change[i]
                a_d, b_d = delta_lo, delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
