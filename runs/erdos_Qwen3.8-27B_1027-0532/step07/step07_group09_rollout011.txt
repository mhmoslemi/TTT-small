def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_change_matrix(h, n):
    """For each position p, compute how corr changes when h[p] increases by 1."""
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):

    neg_mask = (p - shifts >= 0) & (p - shifts < n)

def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _g_target,
             'jac': lambda x: np.ones(_g_n)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def find_optimal_delta(corr, g, a_d, b_d):
    if b_d - a_d < 1e-14:
        return 0.5 * (a_d + b_d)
    for _ in range(25):
        mid_d = 0.5 * (a_d + b_d)
        vals = corr + mid_d * g
        f_mid = np.max(vals)
        active_mask = vals >= f_mid - 1e-13
        if active_mask.any():
            f_prime = np.max(g[active_mask])
        else:
            f_prime = 0.0
        if f_prime > 1e-13:
            b_d = mid_d
        else:
            a_d = mid_d
    return 0.5 * (a_d + b_d)


def enhanced_cd(h_init, n, target, time_limit, max_sweeps=50000,
                n_cand_min=30, n_cand_max=100, max_no_improve=600,
                n_active_shifts=5, n_max_pairs=200, seed_cd=42,
                perturb_every=0, perturb_scale=0.05):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    rng = np.random.RandomState(seed_cd)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_act = min(n_active_shifts, n_corr)
        top_idx = np.argsort(corr)[-n_act:]
        w = corr[top_idx].copy()
        w_norm = w.sum()
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_act) / n_act

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        ncand = min(n_cand_max, n // 4)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = change[all_cands.astype(int), :]

        n_h = len(high_set)
        n_l = len(low_set)

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)])

        n_total_pairs = n_h * n_l
        if n_total_pairs > n_max_pairs:
            pair_indices = rng.choice(n_total_pairs, size=n_max_pairs, replace=False)
            pair_list = [(int(pi // n_l), int(pi % n_l)) for pi in pair_indices]
        else:
            pair_list = [(ii, jj) for ii in range(n_h) for jj in range(n_l)]

        for pi, pj in pair_list:
            i = int(high_set[pi])
            j = int(low_set[pj])
            hr = hi_rows[pi]
            lr = lo_rows[pj]
            h_i = h[i]
            h_j = h[j]

            delta_max = min(h_i, 1.0 - h_j)
            delta_min = max(h_i - 1.0, -h_j)
            if delta_max - delta_min < 1e-12:
                continue

            g = G[lr] - G[hr]
            delta_star = find_optimal_delta(corr, g, delta_min, delta_max)
            if abs(delta_star) < 1e-12:
                continue

            approx_max = float(np.max(corr + delta_star * g))
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) > 1e-14:
                free = (h > 1e-10) & (h < 1.0 - 1e-10)
                if free.any():
                    h[free] += diff / free.sum()
                    h = np.clip(h, 0.0, 1.0)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
        elif perturb_every > 0 and sweep % perturb_every == 0:
            n_pert = rng.randint(3, 8)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h[pert_idx] += rng.uniform(-perturb_scale, perturb_scale, size=n_pert)
            h = project_feasible(h, target)
            for pi_idx in pert_idx:
                update_change_row(change, int(pi_idx), h, n, shifts)

    return best_h, best_c5


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def gen_random_binary(n, target, rng):
    h = np.zeros(n)
    n_ones = int(round(target))
    positions = rng.choice(n, size=n_ones, replace=False)
    h[positions] = 1.0
    return project_feasible(h, target)


def gen_periodic_binary(n, target, rng):
    period = n // 2
    half_t = int(round(target / 2.0))
    h_period = np.zeros(period)
    positions = rng.choice(period, size=min(half_t, period), replace=False)
    h_period[positions] = 1.0
    h = np.tile(h_period, n // period)
    return project_feasible(h, target)


def gen_smooth_pattern(n, target, rng):
    freq = rng.uniform(2.0, 8.0)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 1, n)
    raw = np.sin(2 * np.pi * freq * x + phase)
    threshold = np.percentile(raw, 50)
    h = (raw > threshold).astype(float)
    return project_feasible(h, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 50:
        h_pol = slsqp_polish(best_h, n0, target0, [200, 1000], 300, 1e-13)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Multi-start at n0
    rng = np.random.RandomState(seed)
    n_starts = 4
    for s in range(n_starts):
        if tleft() < 80:
            break
        if s == 0:
            h_start = gen_random_binary(n0, target0, rng)
        elif s == 1:
            h_start = gen_periodic_binary(n0, target0, rng)
        elif s == 2:
            h_start = gen_smooth_pattern(n0, target0, rng)
        else:
            h_start = gen_random_binary(n0, target0, rng)

        alloc = min(60, tleft() * 0.2)
        h_cand, sc_cand = enhanced_cd(h_start, n0, target0,
                                      time_limit=alloc,
                                      max_sweeps=15000,
                                      n_cand_min=25, n_cand_max=70,
                                      max_no_improve=300,
                                      n_max_pairs=150,
                                      seed_cd=seed + s * 100)
        if sc_cand < best_score:
            best_score = sc_cand
            best_h = h_cand

    # Main CD on best at n0
    if tleft() > 150:
        h_cd, c5_cd = enhanced_cd(best_h, n0, target0,
                                  time_limit=min(120, tleft() * 0.35),
                                  max_sweeps=30000,
                                  n_cand_min=30, n_cand_max=90,
                                  max_no_improve=500,
                                  n_max_pairs=200,
                                  seed_cd=seed + 999)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Go to n1 = 600
    n1 = 600
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)
    n0 = n1
    target0 = target1

    if tleft() > 200:
        h_cd1, c5_cd1 = enhanced_cd(h1, n1, target1,
                                    time_limit=min(160, tleft() * 0.45),
                                    max_sweeps=25000,
                                    n_cand_min=25, n_cand_max=80,
                                    max_no_improve=400,
                                    n_max_pairs=200,
                                    seed_cd=seed + 777)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1

    # SLSQP polish
    if tleft() > 80:
        h_pol2 = slsqp_polish(best_h, n0, target0, [500, 2000, 5000], 400, 1e-14)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # More CD at n1
    if tleft() > 130:
        h_cd2, c5_cd2 = enhanced_cd(best_h, n0, target0,
                                    time_limit=min(100, tleft() * 0.45),
                                    max_sweeps=20000,
                                    n_cand_min=40, n_cand_max=100,
                                    max_no_improve=400,
                                    n_max_pairs=250,
                                    seed_cd=seed + 555)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Try n2 = 800
    if tleft() > 250:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_cd3, c5_cd3 = enhanced_cd(h2, n2, target2,
                                    time_limit=min(160, tleft() * 0.5),
                                    max_sweeps=20000,
                                    n_cand_min=25, n_cand_max=70,
                                    max_no_improve=350,
                                    n_max_pairs=200,
                                    seed_cd=seed + 333)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
            n0 = n2
            target0 = target2

    # Final SLSQP
    if tleft() > 70:
        h_final = slsqp_polish(best_h, n0, target0, [1000, 5000, 10000], 500, 1e-14)
        sc_final = compute_c5(h_final)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Final CD round
    if tleft() > 50:
        h_cd_final, c5_final = enhanced_cd(best_h, n0, target0,
                                           time_limit=min(35, tleft() * 0.6),
                                           max_sweeps=15000,
                                           n_cand_min=50, n_cand_max=120,
                                           max_no_improve=250,
                                           n_max_pairs=300,
                                           seed_cd=seed + 111)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
