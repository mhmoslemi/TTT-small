def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def obj_pnorm(x, p):

    pos = np.maximum(corr, 0.0)
    return float(np.sum(pos ** p) ** (1.0 / p))

def obj_lse(x, alpha):

def slsqp_round(h_start, n, target, params, use_pnorm, maxiter_per_param, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def pair_cd(h_init, n, target, time_limit, start_time,
            max_sweeps=30000, n_cand_start=15, n_cand_end=50,
            max_no_improve=300, n_top_shifts=3, bisection_iters=16,
            random_kick_every=800, kick_scale=0.02):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(12345)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = build_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:

            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if random_kick_every > 0 and sweep > 0 and sweep % random_kick_every == 0 and consecutive_no_improve > 50:
            kick_idx = rng.choice(n, size=5, replace=False)
            h[kick_idx] += rng.randn(5) * kick_scale
            h = project_feasible(h, target)
            for ki in kick_idx:
                update_change_row(change, int(ki), h, n, shifts)
            continue

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_idx].astype(float)
            wsum = np.sum(weights)
            if wsum < 1e-15:
                weights = np.ones(n_top_shifts) / n_top_shifts
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for s_idx, w in zip(top_idx, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _bt in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):

        h6, sc6 = slsqp_round(best_h, n0, target0, [4096, 16384, 65536], True, 1000, start_time + min(80, remaining - 15), start_time)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 100:
        h7, sc7 = pair_cd(best_h, n0, target0, start_time + min(100, remaining - 15), start_time,
                          max_sweeps=25000, n_cand_start=30, n_cand_end=80,
                          max_no_improve=200, n_top_shifts=1, bisection_iters=20,
                          random_kick_every=0, kick_scale=0.0)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 60:
        h8, sc8 = stochastic_refine(best_h, n0, target0, start_time + min(60, remaining - 15), start_time,
                                    max_steps=15000, base_scale=0.02)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    remaining = budget_s - 40.0 - (time.time() - start_time)
    if remaining > 60:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.01, 0.03, 0.06]:
            remaining = budget_s - 40.0 - (time.time() - start_time)
            if remaining < 50:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_r, sc_r = slsqp_round(h_pert, n0, target0,
                                    [64, 512, 4096], True, 600, start_time + min(40, remaining - 15), start_time)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 40:
        h_fin, sc_fin = slsqp_round(best_h, n0, target0,
                                    [16384, 65536], True, 800, start_time + min(40, remaining - 10), start_time)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    remaining = budget_s - 25.0 - (time.time() - start_time)
    if remaining > 30:
        h_fin2, sc_fin2 = pair_cd(best_h, n0, target0, start_time + min(30, remaining - 5), start_time,
                                  max_sweeps=10000, n_cand_start=40, n_cand_end=90,
                                  max_no_improve=100, n_top_shifts=1, bisection_iters=20,
                                  random_kick_every=0, kick_scale=0.0)
        if sc_fin2 < best_score:
            best_score = sc_fin2
            best_h = h_fin2

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
