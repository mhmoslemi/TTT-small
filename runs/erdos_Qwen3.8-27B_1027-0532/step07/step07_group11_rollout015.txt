def projected_gradient(h_init, n, target, time_limit, n_steps=100000):
    start = time.time()
    n_corr = 2 * n - 1
    idxs = np.arange(n)
    mid = n - 1
    
    h = h_init.copy().astype(float)
    h_avg = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        
        # Compute objective and gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx  # need dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Get top-K shifts with softmax weights
        alpha = 10.0 + 10.0 * (step / n_steps)  # increasing concentration
        shifted = np.clip(alpha * (corr - np.max(corr)), -500.0, 0.0)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        active = np.where(weights > 1e-6)[0]
        for k_idx in active:
            k = int(k_idx) - mid
            w = weights[k_idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Step size (polynomial decay)
        eta = 0.1 / (1.0 + 0.01 * step)
        
        # Projected step
        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        
        # Accept if improvement
        c5_try = compute_c5(h_try)
        if c5_try < c5_val:
            h = h_try
            h_avg = (1.0 - 1.0/(step+1)) * h_avg + (1.0/(step+1)) * h

            h_try2 = project_feasible(h_try2, target)
            c5_try2 = compute_c5(h_try2)

**Key realization**: The problem is a "minimax" problem. At the optimum, multiple constraints (multiple shifts) should be active. This suggests that a "dual" approach might work well:

2. The objective is the weighted sum: sum_k p_k * corr_k
3. This is a convex function in h (wait, is it? corr_k = sum_i h[i]*(1-h[i+k]) is NOT convex in h... it's the difference of a linear and a quadratic term)

Actually, corr_k = sum_i h[i] - sum_i h[i]*h[i+k]. The first term is linear, the second is quadratic (specifically, it's -h^T A_k h where A_k is a shift matrix). So corr_k is a concave function of h (negative of a positive semi-definite quadratic, plus linear).
