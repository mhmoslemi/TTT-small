                                       max_no_improve=500,
                                       weight_beta=5.0,
                                       use_full_grad=True,
                                       adaptive_beta=True,
                                       beta_min=3.0, beta_max=12.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP with high alpha
    if tleft() > 150:
        h_pol2, sc2 = slsqp_round(best_h, n0, target0,
                                  [10000, 50000, 200000],
                                  800, 1e-14)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    # Phase 4: Perturbed multi-start CD
    if tleft() > 200:
        h_ps, c5_ps = perturbed_cd(best_h, n0, target0,
                                   time_limit=min(180, tleft() * 0.5),
                                   n_restarts=5,
                                   noise_scales=[0.003, 0.008, 0.02, 0.05, 0.1],
                                   max_sweeps_per=8000)
        if c5_ps < best_score:
            best_score = c5_ps
            best_h = h_ps

    # Phase 5: Final CD with very high beta (focus on worst shifts)
    if tleft() > 120:
        h_cd2, c5_cd2 = weighted_pair_cd(best_h, n0, target0,
                                         time_limit=min(100, tleft() * 0.5),
                                         max_sweeps=20000,
                                         n_cand_start=25, n_cand_end=70,
                                         max_no_improve=300,
                                         weight_beta=10.0,
                                         adaptive_beta=False)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 6: Very high precision SLSQP
    if tleft() > 80:
        h_pol3, sc3 = slsqp_round(best_h, n0, target0,
                                  [100000, 500000],
                                  600, 1e-15)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    # Phase 7: Fine-grained CD polish
    if tleft() > 60:
        h_cd3, c5_cd3 = weighted_pair_cd(best_h, n0, target0,
                                         time_limit=min(50, tleft() * 0.6),
                                         max_sweeps=15000,
                                         n_cand_start=30, n_cand_end=80,
                                         max_no_improve=150,
                                         weight_beta=15.0)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 8: Final SLSQP
    if tleft() > 30:
        h_fin, sc_fin = slsqp_round(best_h, n0, target0,
                                    [500000, 1000000],
                                    400, 1e-15)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
