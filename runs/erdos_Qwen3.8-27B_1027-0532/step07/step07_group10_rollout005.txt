def compute_lp_data(h_t, n):
    """Compute the LP data for the linearized problem at h_t."""
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    # For each shift k, compute the linear coefficients a_k and constant b_k
    # such that S(k) - Q_lin(k) = a_k · h + b_k
    a_list = []
    b_list = []
    for k in range(-(n-1), n):
        # Valid i: 0 <= i < n and 0 <= i+k < n
        i_start = max(0, -k)
        i_end = min(n, n-k)  # exclusive
        a = np.zeros(n)
        b = 0.0
        for i in range(i_start, i_end):
            j = i + k
            # S(k) contributes +1 to h[i]
            a[i] += 1.0
            # -h_t[i]*h[i+k] contributes -h_t[i] to h[j]
            a[j] -= h_t[i]
            # -h_t[i+k]*h[i] contributes -h_t[j] to h[i]
            a[i] -= h_t[j]
            # +h_t[i]*h_t[i+k] is constant
            b += h_t[i] * h_t[j]
        a_list.append(a)
        b_list.append(b)
    return np.array(a_list), np.array(b_list), dx

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_linearize(h_t, n, solver='CLARABEL'):
    """Solve the LP linearization at h_t."""
    dx = 2.0 / n
    target = n / 2.0
    
    # Compute linear coefficients for each shift
    n_shifts = 2 * n - 1
    A = np.zeros((n_shifts, n))
    b = np.zeros(n_shifts)
    
    idxs = np.arange(n)
    for ki, k in enumerate(range(-(n-1), n)):
        i_start = max(0, -k)
        i_end = min(n, n - k)
        valid_i = np.arange(i_start, i_end)
        valid_j = valid_i + k
        # S(k) = sum of h[i] for valid i
        A[ki, valid_i] += 1.0
        # -h_t[i] * h[i+k]: coefficient of h[j] is -h_t[i] where j = i+k
        A[ki, valid_j] -= h_t[valid_i]
        # -h_t[i+k] * h[i]: coefficient of h[i] is -h_t[j]
        A[ki, valid_i] -= h_t[valid_j]
        # constant: h_t[i] * h_t[i+k]
        b[ki] = np.sum(h_t[valid_i] * h_t[valid_j])
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1,
    ]
    # A @ h + b <= t / dx
    constraints.append(A @ h_var + b <= t_var / dx)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except:
        try:
            prob.solve(solver='ECOS')
        except:
            return None
    
    if h_var.value is None:
        return None
    
    h_lp = np.clip(h_var.value, 0.0, 1.0)
    h_lp = project_feasible(h_lp, target)
    return h_lp

def lp_iteration(h_start, n, target, n_iters, time_limit, solver='CLARABEL'):
    """Iterate the LP linearization method."""
    start_time = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        
        h_lp = lp_linearize(h, n, solver)
        if h_lp is None:
            break
        
        # Try different relaxation parameters
        best_alpha_score = best_score
        best_alpha_h = best_h
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            h_try = (1 - alpha) * h + alpha * h_lp
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_alpha_score:
                best_alpha_score = sc
                best_alpha_h = h_try
        
        if best_alpha_score < best_score - 1e-15:
            best_score = best_alpha_score
            best_h = best_alpha_h
            h = best_alpha_h
        else:
            # Try the LP solution directly
            sc = compute_c5(h_lp)
            if sc < best_score:
                best_score = sc
                best_h = h_lp
                h = h_lp
            else:
                break
    
    return best_h, best_score

                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-15:

                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            
            # Update change matrix
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
            
            # Update correlation
            corr = np.correlate(h, 1.0 - h, mode='full')

```python
def compute_lp_matrices(h_t, n):
    """Compute A (2n-1 x n) and b (2n-1,) for the LP."""
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    A = np.zeros((n_shifts, n))
    b = np.zeros(n_shifts)
    
    for ki, s in enumerate(range(-(n-1), n)):
        i_start = max(0, -s)
        i_end = min(n, n - s)
        valid_i = np.arange(i_start, i_end)
        valid_j = valid_i + s  # these are the j = i+s values
        
        # S(s) part: +1 for valid i
        A[ki, valid_i] += 1.0
        
        # -h_t[i] * h[i+s] part: coefficient of h[j] where j = i+s is -h_t[i]
        A[ki, valid_j] -= h_t[valid_i]
        
        # -h_t[i+s] * h[i] part: coefficient of h[i] is -h_t[i+s] = -h_t[j]
        A[ki, valid_i] -= h_t[valid_j]
        
        # Constant: h_t[i] * h_t[i+s]
        b[ki] = np.sum(h_t[valid_i] * h_t[valid_j])
    
    return A, b
```

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_lp_matrices(h_t, n):
    n_shifts = 2 * n - 1
    A = np.zeros((n_shifts, n))
    b = np.zeros(n_shifts)
    for ki, s in enumerate(range(-(n - 1), n)):
        i_start = max(0, -s)
        i_end = min(n, n - s)
        valid_i = np.arange(i_start, i_end)
        valid_j = valid_i + s
        A[ki, valid_i] += 1.0
        A[ki, valid_j] -= h_t[valid_i]
        A[ki, valid_i] -= h_t[valid_j]
        b[ki] = np.sum(h_t[valid_i] * h_t[valid_j])
    return A, b


def solve_lp_at(h_t, n):
    target = n / 2.0
    dx = 2.0 / n
    A, b = compute_lp_matrices(h_t, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1,
        A @ h_var + b <= t_var / dx,
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solvers = ['CBC', 'GLPK', 'ECOS', 'PDLP', 'SCIP', 'CVXOPT']
    for solver in solvers:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None:
                return np.clip(h_var.value, 0.0, 1.0)
        except Exception:
            continue
    return None


def lp_iteration(h_start, n, target, n_iters, time_limit):
    start_time = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    no_improve = 0

        if time.time() - start_time > time_limit:
            break
        if no_improve > 8:
            break
        h_lp = solve_lp_at(h, n)
        if h_lp is None:
            break
        h_lp = project_feasible(h_lp, target)
        best_alpha_h = best_h
        best_alpha_score = best_score
        for alpha in [1.0, 0.85, 0.7, 0.5, 0.35, 0.2, 0.1]:

            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_alpha_score:

        if best_alpha_score < best_score - 1e-15:
            best_score = best_alpha_score
            best_h = best_alpha_h
            h = best_alpha_h
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_score
