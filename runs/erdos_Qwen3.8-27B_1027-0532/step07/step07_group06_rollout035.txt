```python
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    h = h.astype(float).copy()
    n = len(h)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_for_shift(h, k, n):
    """Compute gradient of corr_k w.r.t. h. k is the actual shift (can be negative)."""
    grad = np.zeros(n)
    if k >= 0:
        # corr_k = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k])
        valid = np.arange(n - k)
        grad[valid] += (1.0 - h[valid + k])
        if k > 0:
            valid2 = np.arange(k, n)
            grad[valid2] -= h[valid2 - k]
    else:
        kk = -k
        valid = np.arange(kk, n)
        grad[valid] += (1.0 - h[valid - kk])
        valid2 = np.arange(n - kk)
        grad[valid2] -= h[valid2 + kk]
    return grad

def lp_subproblem(h_star, n, active_idxs, target_sum, tol=1e-10):
    """Solve LP: min t s.t. linearized corr_k <= t for k in active_idxs, bounds, sum constraint."""
    shifts = np.arange(-(n-1), n)
    corr_star = compute_corr(h_star)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    
    for k_idx in active_idxs:
        k = int(shifts[k_idx])
        grad = compute_grad_for_shift(h_star, k, n)
        # Linearization: corr_star[k_idx] + grad . (h - h_star) <= t
        # grad . h <= t - corr_star[k_idx] + grad . h_star
        c0 = -corr_star[k_idx] + np.dot(grad, h_star)
        constraints.append(grad @ h_var <= t_var + c0)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL, tol_gap_abs=1e-12, tol_feas=1e-12)

            prob.solve(solver=cp.ECOS, max_iter=10000, abstol=1e-12, reltol=1e-12)
        except:
            prob.solve()
    
    if h_var.value is None:
        return h_star.copy(), corr_star.max()
    
    h_new = np.clip(h_var.value, 0.0, 1.0)
    return project_feasible(h_new, target_sum), float(t_var.value)

def cutting_plane_method(h_init, n, target_sum, time_limit, start_time, 
                          n_active_max=50, max_iters=200, tol=1e-12):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    shifts = np.arange(-(n-1), n)
    
    active_set = list(range(2*n-1))  # start with all
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Solve LP subproblem
        h_new, t_lp = lp_subproblem(h, n, active_set, target_sum)
        
        # Check actual objective
        actual_c5 = compute_c5(h_new)
        
        if actual_c5 < best_c5 - 1e-14:
            best_c5 = actual_c5
            best_h = h_new.copy()
        
        # Determine new active set
        corr = compute_corr(h_new)
        max_corr = corr.max()
        # Keep shifts where corr_k > max_corr - threshold
        threshold = max_corr - max(1e-6, 0.01 * max_corr)
        active_set = list(np.where(corr > threshold)[0])
        
        # Also always include the argmax
        k_star = int(np.argmax(corr))
        if k_star not in active_set:
            active_set.append(k_star)
        
        # Cap active set size
        if len(active_set) > n_active_max:
            # Keep the top n_active_max
            active_set = list(np.argsort(corr)[-n_active_max:])
        
        # Check convergence
        if abs(t_lp - max_corr) < tol:
            # LP solution is also optimal for original
            break
        
        h = h_new
    
    return best_h, best_c5
```

```python
def gradient_direction_lp(h, n, active_shifts_idx, target_sum, step_bound):
    """Find direction d that reduces max correlation.
    Returns d and t."""
    shifts = np.arange(-(n-1), n)
    n_var = n
    d = cp.Variable(n_var)
    t = cp.Variable()
    
    constraints = [
        cp.sum(d) == 0,
        d >= -h - 1e-10,
        d <= 1 - h + 1e-10,
    ]
    
    for k_idx in active_shifts_idx:
        k = int(shifts[k_idx])
        grad = compute_grad_for_shift(h, k, n)
        constraints.append(grad @ d <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, tol_gap_abs=1e-14, tol_feas=1e-14)
    except:
        prob.solve(solver=cp.ECOS)
    
    if d.value is None:
        return np.zeros(n), 0.0
    
    return d.value, float(t.value)
```

- Feasibility: alpha ∈ [0, alpha_max] where alpha_max = min over i with d_i>0 of (1-h_i)/d_i, and min over i with d_i<0 of -h_i/d_i
- Objective: we want max_k corr_k(h + alpha*d) to be minimized. For small alpha, this is approximately max_k (corr_k + alpha * G_k d). Since G_k d ≤ t < 0 for all k in S, the linear approximation decreases. But for larger alpha, the quadratic terms matter.

```python
import numpy as np
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    n = len(h)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    return h

def compute_grad_for_shift(h, k, n):
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            valid = np.arange(n - k)
            grad[valid] += (1.0 - h[valid + k])
        if k > 0:
            valid2 = np.arange(k, n)
            grad[valid2] -= h[valid2 - k]
    else:
        kk = -k
        if kk < n:
            valid = np.arange(kk, n)
            grad[valid] += (1.0 - h[valid - kk])
        valid2 = np.arange(n - kk)
        grad[valid2] -= h[valid2 + kk]
    return grad

def direction_lp(h, n, active_idxs, shifts):
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.sum(d) == 0,
        d >= -h - 1e-10,
        d <= 1.0 - h + 1e-10,
    ]
    for k_idx in active_idxs:
        k = int(shifts[k_idx])
        if k == 0:
            continue  # skip k=0 (corr_0 = sum h_i(1-h_i), gradient is 1-2h_i)
        grad = compute_grad_for_shift(h, k, n)
        constraints.append(grad @ d <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, tol_gap_abs=1e-14, tol_feas=1e-14)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=5000, abstol=1e-10, reltol=1e-10)
        except Exception:
            try:
                prob.solve()
            except Exception:
                return np.zeros(n), 0.0
    
    if d.value is None:
        return np.zeros(n), 0.0
    return np.asarray(d.value).flatten(), float(t.value)

def cg_method(h_init, n, target_sum, time_limit, start_time, max_iters=500, 
              n_active=30, cg_tol=1e-8):
    """Conditional gradient method with LP for direction finding."""
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    shifts = np.arange(-(n-1), n)
    dx = 2.0 / n
    mid = n - 1
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find active set: top shifts
        top_idxs = np.argsort(corr)[-n_active:]
        max_corr = corr.max()
        # Also include shifts within threshold of max
        thresh = max_corr - 0.05 * max_corr
        active_idxs = set(np.where(corr > thresh)[0].tolist())
        active_idxs.update(top_idxs.tolist())
        active_idxs = list(active_idxs)
        # Cap size
        if len(active_idxs) > 60:
            active_idxs = list(np.argsort(corr)[-60:])
        
        # Solve direction LP
        d, t_val = direction_lp(h, n, active_idxs, shifts)
        
        if t_val >= -cg_tol:
            # No improving direction found for this active set
            break
        
        # Line search for step size
        # Max feasible step
        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1e10
        if pos_mask.any():
            alpha_max = min(alpha_max, np.min((1.0 - h[pos_mask]) / d[pos_mask]))
        if neg_mask.any():
            alpha_max = min(alpha_max, np.min(-h[neg_mask] / d[neg_mask]))
        alpha_max = max(alpha_max, 0.01)
        
        # Try different step sizes
        best_alpha = 0.0
        best_alpha_c5 = c5_val
        for frac in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            alpha = frac * alpha_max
            h_trial = h + alpha * d
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = project_feasible(h_trial, target_sum)
            sc = compute_c5(h_trial)
            if sc < best_alpha_c5:
                best_alpha_c5 = sc
                best_alpha = alpha
        
        if best_alpha > 0:
            h = h + best_alpha * d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target_sum)
        else:
            break
    
    return best_h, best_c5
```

```python
import numpy as np
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    n = len(h)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    return h

def grad_for_m(h, m, n):
    s = m - (n - 1)
    grad = np.zeros(n)
    j_min1 = max(0, s)
    j_max1 = min(n, s + n)
    if j_max1 > j_min1:
        grad[j_min1:j_max1] += (1.0 - h[j_min1 - s:j_max1 - s])
    j_min2 = max(0, -s)
    j_max2 = min(n, n - s)
    if j_max2 > j_min2:
        grad[j_min2:j_max2] -= h[j_min2 + s:j_max2 + s]
    return grad

def direction_lp(h, n, active_ms):
    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.sum(d) == 0,
        d >= -h - 1e-10,
        d <= 1.0 - h + 1e-10,
    ]
    for m_idx in active_ms:
        grad = grad_for_m(h, int(m_idx), n)
        constraints.append(grad @ d <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, tol_gap_abs=1e-14, tol_feas=1e-14)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=5000, abstol=1e-10, reltol=1e-10)
        except Exception:
            try:
                prob.solve()
            except Exception:
                return np.zeros(n), 0.0
    if d.value is None:
        return np.zeros(n), 0.0
    return np.asarray(d.value).flatten(), float(t.value)

def cg_lp_method(h_init, n, target, time_limit, start_time, max_iters=300, n_active=25, cg_tol=1e-9):
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    no_improve = 0
    prev_d = np.zeros(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 80:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_corr = float(np.max(corr))
        thresh = max_corr - max(1e-6, 0.03 * max_corr)
        active_set = set(np.where(corr > thresh)[0].tolist())
        top_k = min(n_active, 2 * n - 1)
        top_idxs = np.argsort(corr)[-top_k:]
        active_set.update(top_idxs.tolist())
        active_ms = list(active_set)
        if len(active_ms) > 50:
            active_ms = list(np.argsort(corr)[-50:])

        d, t_val = direction_lp(h, n, active_ms)

        if t_val >= -cg_tol:
            reduced = list(np.argsort(corr)[-5:])
            d, t_val = direction_lp(h, n, reduced)
            if t_val >= -cg_tol:
                reduced2 = [int(np.argmax(corr))]
                d, t_val = direction_lp(h, n, reduced2)
                if t_val >= -cg_tol:
                    break

        d = d + 0.3 * prev_d
        prev_d = d

        pos_mask = d > 1e-12
        neg_mask = d < -1e-12
        alpha_max = 1e10
        if pos_mask.any():
            alpha_max = min(alpha_max, np.min((1.0 - h[pos_mask]) / d[pos_mask]))
        if neg_mask.any():
            alpha_max = min(alpha_max, np.min(-h[neg_mask] / d[neg_mask]))
        alpha_max = max(min(alpha_max, 0.5), 1e-6)

        best_alpha = 0.0
        best_alpha_c5 = c5_val
        fracs = np.array([1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05])
        for frac in fracs:
            alpha = frac * alpha_max
            h_trial = np.clip(h + alpha * d, 0.0, 1.0)
            h_trial = project_feasible(h_trial, target)
            sc = compute_c5(h_trial)
            if sc < best_alpha_c5:
                best_alpha_c5 = sc
                best_alpha = alpha

        if best_alpha > 1e-8:
            h = h + best_alpha * d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, start_time, max_sweeps=30000,
            n_cand_start=15, n_cand_end=60, max_no_improve=350, n_top=3, bisect_iters=16):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break

        top_shifts = np.argsort(corr)[-n_top:]
        weights = corr[top_shifts].astype(float)
        wsum = np.sum(weights)
        if wsum < 1e-15:
            weights = np.ones(n_top) / n_top
        else:
            weights = weights / wsum
        grad = np.zeros(n)
        for s_idx, w in zip(top_shifts, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = max(min(ncand, n // 3), 5)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                m_j = j
                m_i = i
                for m in range(n_corr):
                    s = m - mid
                    contrib_j = 0.0
                    if 0 <= m_j + s < n:
                        contrib_j += (1.0 - h[m_j + s])
                    if 0 <= m_j - s < n:
                        contrib_j -= h[m_j - s]
                    contrib_i = 0.0
                    if 0 <= m_i + s < n:
                        contrib_i += (1.0 - h[m_i + s])
                    if 0 <= m_i - s < n:
                        contrib_i -= h[m_i - s]
                    g[m] = contrib_j - contrib_i
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisect_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def smooth_lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    from scipy.optimize import minimize
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_lse_obj, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):

        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 60:
        h9, sc9 = slsqp_round(best_h, n0, target0,
                              [30000, 80000],
                              500, start_time + min(60, remaining - 10), start_time)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 40:
        rng = np.random.RandomState(seed)
        for ns in [0.02, 0.05]:
            remaining = budget_s - 30.0 - (time.time() - start_time)
            if remaining < 30:
                break
            h_pert = best_h + rng.randn(n0) * ns
            h_pert = project_feasible(h_pert, target0)
            h_r, sc_r = cg_lp_method(h_pert, n0, target0,
                                     start_time + min(30, remaining - 10), start_time,
                                     max_iters=80, n_active=15, cg_tol=1e-9)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
