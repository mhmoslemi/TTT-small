        for i in range(n_cuts):

            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,

                best_score = sc
                best_h = h_new.copy()
            h = h_new

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_corr_gradient(h, k_star):
    n = len(h)
    mid = n - 1
    k = k_star - mid
    grad = np.zeros(n)
    idxs = np.arange(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad


def improved_cd(h_init, n, target, time_limit, max_sweeps=50000,
                n_cand_start=30, n_cand_end=120, max_no_improve=1500,
                use_annealing=True, T_init=0.01, T_min=1e-7, seed_cd=42):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed_cd)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    T = T_init

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            T = max(T_min, T * 0.9)
        else:
            consecutive_no_improve += 1
            T = max(T_min, T * 0.9995)

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if use_annealing and T > T_min and best_i < 0 and consecutive_no_improve > 30:
            n_try = min(30, n // 8)
            for _ in range(n_try):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                delta = 0.02 * rng.choice([1.0, -1.0])
                if h[i] + delta < -1e-10 or h[i] + delta > 1.0 + 1e-10:
                    continue
                if h[j] - delta < -1e-10 or h[j] - delta > 1.0 + 1e-10:
                    continue
                g_vec = change[j] - change[i]
                approx_max = np.max(corr + delta * g_vec) * dx
                delta_c5 = approx_max - c5_val
                accept_prob = np.exp(-delta_c5 / (T * 1.0))
                if delta_c5 < 0 or accept_prob > rng.random():
                    best_i = i
                    best_j = j
                    best_delta = delta
                    best_trial_c5 = approx_max
                    break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i, :] = 0.0
            change[best_j, :] = 0.0
            pi = best_i
            pj = best_j
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    return best_h, best_c5


def symmetric_cd(h_init, n, target, time_limit, max_sweeps=20000,
                 n_cand_start=15, n_cand_end=50, max_no_improve=800):
    start = time.time()
    dx = 2.0 / n
    n_half = n // 2
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    n_corr = 2 * n - 1

    a = h_init[:n_half].copy().astype(float)

    def build_h_from_a(a_vec):
        h_vec = np.zeros(n)
        for i in range(n_half):
            h_vec[i] = a_vec[i]
            h_vec[n - 1 - i] = 1.0 - a_vec[i]
        if n % 2 == 1:
            h_vec[n_half] = 0.5
        return h_vec

    h = build_h_from_a(a)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change_full = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change_full[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change_full[p, neg_mask] -= h[p - shifts[neg_mask]]

    change_eff = np.zeros((n_half, n_corr))
    for i in range(n_half):
        change_eff[i] = change_full[i] - change_full[n - 1 - i]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad_full = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad_full[v1] += (1.0 - h[idxs[v1] - k_star])
        grad_full[v2] -= h[idxs[v2] + k_star]

        grad = np.zeros(n_half)
        for i in range(n_half):
            grad[i] = grad_full[i] - grad_full[n - 1 - i]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, max(n_half // 2, 5))

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_pair = None

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(0.0, a[i] - 1.0)
                delta_hi = min(1.0, a[i], 1.0 - a[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change_eff[j] - change_eff[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_pair = (i, j, delta_star)

        if best_pair is not None:
            i, j, delta = best_pair
            a[i] -= delta
            a[j] += delta
            a = np.clip(a, 0.0, 1.0)

            h = build_h_from_a(a)

            pos_i = (i + shifts >= 0) & (i + shifts < n)
            neg_i = (i - shifts >= 0) & (i - shifts < n)
            change_full[i, :] = 0.0
            change_full[i, pos_i] += (1.0 - h[i + shifts[pos_i]])
            change_full[i, neg_i] -= h[i - shifts[neg_i]]

            pi2 = n - 1 - i
            pos_i2 = (pi2 + shifts >= 0) & (pi2 + shifts < n)
            neg_i2 = (pi2 - shifts >= 0) & (pi2 - shifts < n)
            change_full[pi2, :] = 0.0
            change_full[pi2, pos_i2] += (1.0 - h[pi2 + shifts[pos_i2]])
            change_full[pi2, neg_i2] -= h[pi2 - shifts[neg_i2]]

            pos_j = (j + shifts >= 0) & (j + shifts < n)
            neg_j = (j - shifts >= 0) & (j - shifts < n)
            change_full[j, :] = 0.0
            change_full[j, pos_j] += (1.0 - h[j + shifts[pos_j]])
            change_full[j, neg_j] -= h[j - shifts[neg_j]]

            pj2 = n - 1 - j
            pos_j2 = (pj2 + shifts >= 0) & (pj2 + shifts < n)
            neg_j2 = (pj2 - shifts >= 0) & (pj2 - shifts < n)
            change_full[pj2, :] = 0.0
            change_full[pj2, pos_j2] += (1.0 - h[pj2 + shifts[pos_j2]])
            change_full[pj2, neg_j2] -= h[pj2 - shifts[neg_j2]]

            change_eff[i] = change_full[i] - change_full[n - 1 - i]
            change_eff[j] = change_full[j] - change_full[n - 1 - j]

    return best_h, best_c5


def multi_chain_cd(h_init, n, target, time_limit, n_chains=4,
                   n_cand_start=20, n_cand_end=80, max_no_improve=600, seed_mc=777):
    start = time.time()
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    rng = np.random.RandomState(seed_mc)

    for chain in range(n_chains):
        remaining = time_limit - (time.time() - start)
        if remaining < 40:
            break

        if chain == 0:
            h_chain = best_h.copy()
        else:
            noise_level = 0.02 + 0.015 * chain
            h_chain = best_h + rng.randn(n) * noise_level
            h_chain = project_feasible(h_chain, target)

        h_out, sc = improved_cd(h_chain, n, target,
                                time_limit=remaining * 0.75,
                                max_sweeps=25000,
                                n_cand_start=n_cand_start,
                                n_cand_end=n_cand_end,
                                max_no_improve=max_no_improve,
                                use_annealing=True,
                                T_init=0.008, T_min=1e-7,
                                seed_cd=seed_mc + chain * 100)
        if sc < best_score:
            best_score = sc
            best_h = h_out

    return best_h, best_score


def cutting_plane_method(h_init, n, target, time_limit, max_iters=10000, tol=1e-9):
    start = time.time()
    mid = n - 1
    dx = 2.0 / n

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_f = []
    cuts_h = []

    max_cuts = 200
    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            cuts_g = cuts_g[-10:]
            cuts_f = cuts_f[-10:]
            cuts_h = cuts_h[-10:]
            no_improve = 0

        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])

        g = compute_corr_gradient(h, k_star)

        cuts_g.append(g)
        cuts_f.append(f_val)
        cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            cuts_g = [cuts_g[0]] + cuts_g[-keep:]
            cuts_f = [cuts_f[0]] + cuts_f[-keep:]
            cuts_h = [cuts_h[0]] + cuts_h[-keep:]

        n_cuts = len(cuts_g)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(cuts_g[i] @ cuts_h[i]) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                break
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            t_val = float(res.x[n])
            gap = sc / dx - t_val
            if gap < tol:
                h = h_new
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h.copy()
                break

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            break

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_averaged_cd(h_init, n, target, time_limit, max_sweeps=30000,
                            n_cand_start=25, n_cand_end=100, max_no_improve=1000,
                            n_active=5):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        sorted_corr = np.argsort(corr)[::-1]
        active_shifts = sorted_corr[:min(n_active, n_corr)]

        grad = np.zeros(n)
        for ks in active_shifts:
            k = int(ks) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active_shifts)
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i, :] = 0.0
            change[best_j, :] = 0.0
            pi = best_i
            pj = best_j
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP to establish baseline
    if tleft() > 80:
        h_sls, sc = slsqp_round(best_h, n0, target0,
                                [50, 200, 1000], 400, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_sls

    # Phase 2: Main coordinate descent with annealing
    if tleft() > 280:
        h_cd, sc_cd = improved_cd(best_h, n0, target0,
                                  time_limit=min(250, tleft() * 0.65),
                                  max_sweeps=50000,
                                  n_cand_start=30, n_cand_end=110,
                                  max_no_improve=1500,
                                  use_annealing=True,
                                  T_init=0.01, T_min=1e-7,
                                  seed_cd=seed)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 3: Subgradient-averaged CD (different direction signal)
    if tleft() > 200:
        h_sg, sc_sg = subgradient_averaged_cd(best_h, n0, target0,
                                              time_limit=min(180, tleft() * 0.7),
                                              max_sweeps=30000,
                                              n_cand_start=25, n_cand_end=90,
                                              max_no_improve=1000,
                                              n_active=5)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 4: Symmetric coordinate descent
    if tleft() > 150:
        h_sym, sc_sym = symmetric_cd(best_h, n0, target0,
                                     time_limit=min(130, tleft() * 0.6),
                                     max_sweeps=20000,
                                     n_cand_start=15, n_cand_end=50,
                                     max_no_improve=600)
        if sc_sym < best_score:
            best_score = sc_sym
            best_h = h_sym

    # Phase 5: Multi-chain for diversity
    if tleft() > 200:
        h_mc, sc_mc = multi_chain_cd(best_h, n0, target0,
                                     time_limit=min(180, tleft() * 0.7),
                                     n_chains=3,
                                     n_cand_start=25, n_cand_end=80,
                                     max_no_improve=600,
                                     seed_mc=seed + 999)
        if sc_mc < best_score:
            best_score = sc_mc
            best_h = h_mc

    # Phase 6: Cutting plane for structural improvement
    if tleft() > 150:
        h_cp, sc_cp = cutting_plane_method(best_h, n0, target0,
                                           time_limit=min(130, tleft() * 0.6),
                                           max_iters=8000, tol=1e-10)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp

    # Phase 7: Try higher resolution
    if tleft() > 180:
        n1 = 600
        target1 = n1 / 2.0
        h600 = interpolate_to(best_h, n1, target1)
        if tleft() > 130:
            h600_cd, sc600 = improved_cd(h600, n1, target1,
                                          time_limit=min(110, tleft() * 0.5),
                                          max_sweeps=30000,
                                          n_cand_start=20, n_cand_end=60,
                                          max_no_improve=800,
                                          use_annealing=True,
                                          T_init=0.005, T_min=1e-7,
                                          seed_cd=seed + 2000)
            if sc600 < best_score:
                best_score = sc600
                best_h = h600_cd

    # Phase 8: CD on current best
    current_n = len(best_h)
    current_target = current_n / 2.0
    if tleft() > 100:
        h_cd2, sc_cd2 = improved_cd(best_h, current_n, current_target,
                                    time_limit=min(80, tleft() * 0.6),
                                    max_sweeps=30000,
                                    n_cand_start=40, n_cand_end=120,
                                    max_no_improve=1000,
                                    use_annealing=True,
                                    T_init=0.005, T_min=1e-7,
                                    seed_cd=seed + 5000)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 9: Final SLSQP polish
    if tleft() > 50:
        h_final_sls, sc_final = slsqp_round(best_h, len(best_h), len(best_h) / 2.0,
                                            [5000, 20000], 400, 1e-14)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_sls

    # Phase 10: One last CD round
    if tleft() > 40:
        h_cd3, sc_cd3 = improved_cd(best_h, len(best_h), len(best_h) / 2.0,
                                    time_limit=min(30, tleft() * 0.6),
                                    max_sweeps=15000,
                                    n_cand_start=50, n_cand_end=130,
                                    max_no_improve=300,
                                    use_annealing=False,
                                    T_init=0.0, T_min=0.0,
                                    seed_cd=seed + 9000)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
