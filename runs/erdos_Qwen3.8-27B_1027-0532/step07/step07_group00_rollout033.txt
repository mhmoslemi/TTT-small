import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

                n_cand=25, max_no_improve=600,
                weight_beta=5.0, bisection_iters=20, top_k_shifts=30):

    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    best_h = h.copy()

    n_src = len(h_src)

def subgradient_step(h, n, target, top_k=3):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    top_indices = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for s_idx in top_indices:
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    grad /= len(top_indices)
    grad -= np.mean(grad)
    return grad, corr, dx
