def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    sweep = 0

        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:

            if no_improve > 200:

            grad = np.zeros(n)

            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):

            if np.any(v2):

            alpha_w = 30.0

                k = int(si) - mid

                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]

        nc = len(all_cands)

            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):

            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        high_cand_pos = [cand_pos[int(i)] for i in high_set]
        low_cand_pos = [cand_pos[int(j)] for j in low_set]

            hi_val = float(h[all_cands[ci]])

                if ci == cj:

                hj_val = float(h[all_cands[cj]])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                
                g = profiles[ci] - profiles[cj]
                
                # Bisection
                lo = 0.0
                hi_d = delta_max
                for _ in range(bisection_iters):

                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))

                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                
                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

    n_high = len(high_pos)
    n_low = len(low_pos)
    n_corr = len(corr)
    cur_max = np.max(corr)
    
    # G[j_hi, j_lo, :] = profiles[high_pos[j_hi]] - profiles[low_pos[j_lo]]
    # Shape: (n_high, n_low, n_corr)
    G = profiles[high_pos][:, None, :] - profiles[low_pos][None, :, :]
    
    # delta_max for each pair
    hi_vals = h[all_cands[high_pos]]  # shape (n_high,)
    lo_vals = h[all_cands[low_pos]]  # shape (n_low,)
    delta_max = np.minimum(hi_vals[:, None], 1.0 - lo_vals[None, :])  # (n_high, n_low)

        increase = f_prime > 1e-14
        hi = np.where(increase, mid, hi)
        lo = np.where(increase, lo, mid)
        # Only update valid pairs
        hi = np.where(valid, hi, hi)
        lo = np.where(valid, lo, lo)
    
    delta_star = 0.5 * (lo + hi)
    delta_star = np.where(valid, delta_star, 0.0)
    
    # Compute improvement for each pair
    vals_final = corr[None, None, :] + delta_star[:, :, None] * G
    new_max = np.max(vals_final, axis=2)
    improve = (cur_max - new_max) * valid
    
    # Find best
    best_idx = np.unravel_index(np.argmax(improve), improve.shape)
    best_improve = float(improve[best_idx])
    
    if best_improve > 1e-13:
        best_i = int(all_cands[high_pos[best_idx[0]]])
        best_j = int(all_cands[low_pos[best_idx[1]]])
        best_delta = float(delta_star[best_idx])
    else:
        best_i = -1
        best_j = -1
        best_delta = 0.0
    
    return best_i, best_j, best_delta, best_improve

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def pair_cd_vectorized(h_init, n, target, time_limit, n_cand=40, n_shifts=1, bisection_iters=14):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    no_improve = 0
    sweep = 0

    while time.time() - start < time_limit:
        sweep += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break

        if n_shifts == 1:
            k_star = int(np.argmax(corr)) - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            if np.any(v1):
                grad[v1] += (1.0 - h[idxs[v1] - k_star])
            if np.any(v2):
                grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_idx = np.argsort(corr)[-n_shifts:]
            alpha_w = 40.0
            w = np.exp(alpha_w * (corr[top_idx] - cur_max))
            w = w / w.sum()
            grad = np.zeros(n)
            for wi, si in zip(w, top_idx):
                k = int(si) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                if np.any(v1):
                    grad[v1] += wi * (1.0 - h[idxs[v1] - k])
                if np.any(v2):
                    grad[v2] -= wi * h[idxs[v2] + k]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)

        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p

        high_pos = [cand_pos[int(i)] for i in high_set]
        low_pos = [cand_pos[int(j)] for j in low_set]

        hi_vals = h[all_cands[high_pos]]
        lo_vals = h[all_cands[low_pos]]
        delta_max_mat = np.minimum(hi_vals[:, None], 1.0 - lo_vals[None, :])

        valid = delta_max_mat > 1e-12
        if not np.any(valid):
            continue

        G = profiles[high_pos][:, None, :].astype(np.float64) - profiles[low_pos][None, :, :].astype(np.float64)

        lo = np.zeros((len(high_pos), len(low_pos)))
        hi = delta_max_mat.astype(np.float64).copy()

        for _ in range(bisection_iters):
            mid_d = 0.5 * (lo + hi)
            vals = corr[None, None, :] + mid_d[:, :, None] * G
            f_mid = np.max(vals, axis=2)
            active = vals >= (f_mid[:, :, None] - 1e-14)
            G_masked = np.where(active, G, -1e30)
            f_prime = np.max(G_masked, axis=2)
            increase = (f_prime > 1e-14) & valid
            hi = np.where(increase, mid_d, hi)
            lo = np.where(increase, lo, mid_d)

        delta_star = 0.5 * (lo + hi)
        delta_star = np.where(valid, delta_star, 0.0)
        delta_star = np.where(delta_star > 1e-12, delta_star, 0.0)

        vals_final = corr[None, None, :] + delta_star[:, :, None] * G
        new_max = np.max(vals_final, axis=2)
        improve = (cur_max - new_max) * valid

        flat_idx = np.argmax(improve)
        best_improve = float(improve.flat[flat_idx])

        if best_improve > 1e-13:
            bi, bj = np.unravel_index(flat_idx, improve.shape)
            best_i = int(all_cands[high_pos[bi]])
            best_j = int(all_cands[low_pos[bj]])
            best_delta = float(delta_star[bi, bj])
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            if sweep % 50 == 0:
                h = project_feasible(h, target)
        else:
            if no_improve > 50:
                rng = np.random.RandomState(sweep)
                n_perturb = max(5, n // 100)
                pert_idx = rng.choice(n, n_perturb, replace=False)
                h[pert_idx] += rng.randn(n_perturb) * 0.01
                h = project_feasible(h, target)
                no_improve = 0

    return best_h, best_c5

def lbfgs_phase(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_phase(h_init, n, target, time_limit, max_steps=100000, n_shifts=5, base_lr=0.05):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 20.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= (2.0 / n) * np.sum(grad)

        improved = False
        decay = 1.0 / (1.0 + 0.003 * step)
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr = base_lr * lr_mult * decay
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved and step > 3000:
            break

    return best_h, best_c5
