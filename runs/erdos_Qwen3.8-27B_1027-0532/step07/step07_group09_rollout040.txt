    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):

            h = np.clip(h + diff / n, 0.0, 1.0)
            break

        h = np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    k = min(_g_k, len(corr))
    top_k = np.sort(corr)[-k:]
    c_max = top_k[-1]

    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

                                     time_limit=min(50, tleft() * 0.4),
                                     lr_start=0.003, momentum0=0.85)
        if c5_ng < best_score:
            best_score = c5_ng
            best_h = h_ng

    # Phase 4: Block CD at 800
    if tleft() > 100:
        h_bc, c5_bc = block_cd(h1, n1, target1,
                                time_limit=min(70, tleft() * 0.5),
                                max_steps=12000, block_sizes=(2, 3, 4, 6))
        if c5_bc < best_score:
            best_score = c5_bc
            best_h = h_bc
            h1 = h_bc

    # Phase 5: Go to n=1200
    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        cur_n = n2
        cur_target = target2

        if tleft() > 200:
            alphas = [500, 2000, 8000, 30000]
            kvals = [n2 * 2 - 1, 60, 40, 25]
            h2 = slsqp_optimize(h2, n2, target2, alphas, kvals, 500, 1e-14)
            sc = compute_c5(h2)
            if sc < best_score:
                best_score = sc
                best_h = h2

        if tleft() > 120:
            h_cd2, c5_cd2 = pair_cd(h2, n2, target2,
                                     time_limit=min(80, tleft() * 0.5),
                                     max_sweeps=15000, max_no_improve=300)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                h2 = h_cd2

        if tleft() > 60:
            h_ng2, c5_ng2 = nesterov_grad(h2, n2, target2,
                                           time_limit=min(40, tleft() * 0.4),
                                           lr_start=0.002, momentum0=0.85)
            if c5_ng2 < best_score:
                best_score = c5_ng2
                best_h = h_ng2

    # Phase 6: Cycle back to 800 with improved high-res solution
    if tleft() > 150:
        h_back = interpolate_to(best_h, n1, target1)
        if tleft() > 100:
            alphas = [2000, 10000, 50000]
            kvals = [50, 30, 20]
            h_back = slsqp_optimize(h_back, n1, target1, alphas, kvals, 600, 1e-14)
            sc = compute_c5(h_back)
            if sc < best_score:
                best_score = sc
                best_h = h_back
                cur_n = n1
                cur_target = target1

        if tleft() > 60:
            h_cd3, c5_cd3 = pair_cd(h_back, n1, target1,
                                     time_limit=min(40, tleft() * 0.5),
                                     max_sweeps=10000, max_no_improve=250)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
                cur_n = n1
                cur_target = target1

    # Phase 7: Final SLSQP polish
    if tleft() > 50:
        alphas = [5000, 20000, 80000]
        kvals = [40, 25, 15]
        h_fin = slsqp_optimize(best_h, cur_n, cur_target, alphas, kvals, 800, 1e-14)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Phase 8: Perturbation restarts
    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for attempt in range(3):
            if tleft() < 40:
                break
            noise_scale = 0.015 * (attempt + 1)
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, cur_target)
            alphas = [1000, 5000, 20000]
            kvals = [50, 30, 20]
            h_ref = slsqp_optimize(h_pert, cur_n, cur_target, alphas, kvals, 400, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
            if tleft() > 30:
                h_cd_r, c5_cr = pair_cd(h_ref, cur_n, cur_target,
                                         time_limit=min(20, tleft() * 0.4),
                                         max_sweeps=6000, max_no_improve=200)
                if c5_cr < best_score:
                    best_score = c5_cr
                    best_h = h_cd_r

    # Phase 9: Final CD
    if tleft() > 30:
        h_cf, c5_cf = pair_cd(best_h, cur_n, cur_target,
                               time_limit=min(20, tleft() * 0.5),
                               max_sweeps=5000, max_no_improve=150)
        if c5_cf < best_score:
            best_score = c5_cf
            best_h = h_cf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
