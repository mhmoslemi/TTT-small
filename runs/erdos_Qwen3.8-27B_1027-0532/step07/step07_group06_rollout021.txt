    x_new = np.linspace(0.0, 1.0, n_dst)

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):

    change = np.zeros((n, 2*n-1))
    for p in range(n):
        ip = p + shifts
        valid_p = (ip >= 0) & (ip < n)

        change[p, valid_p] += (1.0 - h[ip[valid_p]])

                shifted = np.clip(p * (corr - c_max), -500.0, 0.0)

def compute_change_matrix(h, n):
    shifts = np.arange(-(n-1), n)
    change = np.zeros((n, 2*n-1))
    for p in range(n):
        ip = p + shifts
        valid_p = (ip >= 0) & (ip < n)
        im = p - shifts
        valid_m = (im >= 0) & (im < n)
        change[p, valid_p] = 1.0 - h[ip[valid_p]]
        change[p, valid_m] = change[p, valid_m] - h[im[valid_m]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    ip = p + shifts
    valid_p = (ip >= 0) & (ip < n)
    im = p - shifts
    valid_m = (im >= 0) & (im < n)
    change[p, valid_p] = 1.0 - h[ip[valid_p]]
    change[p, valid_m] = change[p, valid_m] - h[im[valid_m]]

def pair_cd_fast(h_init, n, target, time_limit, start_time,
                 max_sweeps=30000, n_cand_start=20, n_cand_end=50,
                 max_no_improve=250, n_top_shifts=3, bisection_iters=14):
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Compute weighted subgradient
        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_shifts].astype(float)
            wsum = np.sum(weights)
            if wsum < 1e-15:
                weights = np.ones(n_top_shifts) / float(n_top_shifts)
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        
        grad -= np.mean(grad)
        
        # Candidate selection with adaptive size
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)
        ncand = max(ncand, 8)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            hi = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                hj = h[j]
                delta_lo = max(hi - 1.0, -hj)
                delta_hi = min(hi, 1.0 - hj)
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = change[j] - change[i]
                
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = float(np.max(g[active_mask]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                
                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            np.clip(h, 0.0, 1.0, out=h)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)
    
    return best_h, best_c5

def stochastic_kick(h_best, n, target, scale, seed):
    rng = np.random.RandomState(seed)
    h_pert = h_best + rng.randn(n) * scale
    return project_feasible(h_pert, target)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break

        h = np.clip(h, 0.0, 1.0)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        ip = p + shifts
        valid_p = (ip >= 0) & (ip < n)
        im = p - shifts
        valid_m = (im >= 0) & (im < n)
        change[p, valid_p] = 1.0 - h[ip[valid_p]]
        change[p, valid_m] = change[p, valid_m] - h[im[valid_m]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    ip = p + shifts
    valid_p = (ip >= 0) & (ip < n)
    im = p - shifts
    valid_m = (im >= 0) & (im < n)
    change[p, valid_p] = 1.0 - h[ip[valid_p]]
    change[p, valid_m] = change[p, valid_m] - h[im[valid_m]]


def pair_cd_fast(h_init, n, target, time_limit, start_time,
                 max_sweeps=30000, n_cand_start=20, n_cand_end=50,
                 max_no_improve=250, n_top_shifts=3, bisection_iters=14):
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n - 1), n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            break

        one_minus_h = 1.0 - h

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += one_minus_h[idxs[v1] - k_star]
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_shifts].astype(float)
            wsum = np.sum(weights)
            if wsum < 1e-15:
                weights = np.ones(n_top_shifts) / float(n_top_shifts)
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * one_minus_h[idxs[v1] - k]
                grad[v2] -= w * h[idxs[v2] + k]

        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)
        ncand = max(ncand, 8)

        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_l = len(low_set)

                delta_hi = min(hi, 1.0 - hj)
