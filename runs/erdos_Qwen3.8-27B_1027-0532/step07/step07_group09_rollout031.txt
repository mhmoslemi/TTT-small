```python
import numpy as np
from scipy.optimize import minimize
import time

# Global variables for SLSQP callbacks
_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h, n, target, alphas, maxiter, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def efficient_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                      n_active_start=3, n_active_end=12,
                      n_cand_start=30, n_cand_end=80,
                      max_no_improve=500, rng_seed=123):
    start = time.time()
    rng = np.random.RandomState(rng_seed)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-15:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive active set size
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_active = int(n_active_start + (n_active_end - n_active_start) * progress)
        n_active = min(n_active, n_corr)
        
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        # Compute weighted gradient
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Candidate selection: top/bottom of gradient + some random
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        # Add some random candidates for diversity
        n_random = max(5, ncand // 5)
        if no_improve > 100:
            n_random = max(20, ncand // 2)
        rand_cands = rng.choice(n, size=n_random, replace=False)
        
        all_cands = np.unique(np.concatenate([low_set, high_set, rand_cands]))
        n_cands = len(all_cands)
        
        # Get change rows for candidates
        G = change[all_cands, :]
        
        n_hi = len(high_set)
        n_lo = len(low_set)
        
        # Map from grid index to candidate index
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        # Try pairs: high candidates (reduce) vs low candidates (increase)
        # Limit the number of pairs we try per sweep for speed
        max_pairs_per_sweep = 500
        
        pair_count = 0
        hi_indices = [int(high_set[ii]) for ii in range(n_hi)]
        lo_indices = [int(low_set[jj]) for jj in range(n_lo)]
        
        # Shuffle for diversity when stuck
        if no_improve > 200:
            rng.shuffle(hi_indices)
            rng.shuffle(lo_indices)
        else:
            # Sort by |gradient| for more targeted search
            hi_indices.sort(key=lambda x: -abs(grad[x]))
            lo_indices.sort(key=lambda x: -abs(grad[x]))

        for i in hi_indices:
            if pair_count >= max_pairs_per_sweep:
                break
            ci = cand_map[i]
            h_i = h[i]
            if h_i < 0.01:
                continue
            hr = G[ci, :]
            for j in lo_indices:
                if pair_count >= max_pairs_per_sweep:
                    break
                pair_count += 1
                cj = cand_map[j]
                h_j = h[j]
                if h_j > 0.99:
                    continue

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue

                g = G[cj, :] - hr

                # Binary search for optimal delta
                a_d = delta_min
                b_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def multi_start_cd(h_init, n, target, time_limit, n_starts, rng_seed=42):
    """Try multiple perturbed starts and keep the best."""
    start = time.time()
    rng = np.random.RandomState(rng_seed)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    time_per_start = (time_limit - 10) / n_starts
    
    for s in range(n_starts):
        if time.time() - start > time_limit - 15:
            break
        # Create perturbed start
        noise_scale = 0.03 + 0.02 * s
        h_pert = h_init + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        
        t_rem = time_per_start - (time.time() - start - s * time_per_start)
        if t_rem < 30:
            t_rem = 30
        
        h_cd, c5_cd = efficient_pair_cd(h_pert, n, target,
                                         time_limit=min(t_rem, 80),
                                         max_sweeps=10000,
                                         n_active_start=3, n_active_end=8,
                                         n_cand_start=30, n_cand_end=60,
                                         max_no_improve=300,
                                         rng_seed=rng_seed + s * 100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_cd(h_init, n, target, time_limit, max_steps=10000, block_sizes=(2, 3, 4, 5, 7)):

    start = time.time()
    rng = np.random.RandomState(999)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    best_h = h.copy()

    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 1500:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-15:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find the dominant shift
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        # Gradient for dominant shift
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        n_pos = n - bs + 1

        # Find source (low gradient sum) and destination (high gradient sum)
        block_gp = np.array([grad[i:i+bs].sum() for i in range(0, n_pos, max(1, n_pos // 200))])
        src_region = block_gp < block_gp.min() + 0.1 * (block_gp.max() - block_gp.min())
        dst_region = block_gp > block_gp.max() - 0.1 * (block_gp.max() - block_gp.min())
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter, ftol=1e-14):

        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def interpolate_to(h_src, n_dst, target_dst):

    return project_feasible(h_out, target_dst)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-15:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = min(5, 2 * n - 1)
        top_idx = np.argsort(corr)[-n_active:]
        grad = np.zeros(n)
        for si in top_idx:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= n_active
        grad -= np.mean(grad)

        bs = block_sizes[step % len(block_sizes)]
        n_pos = n - bs + 1

        n_sample = min(300, n_pos)
        sample_positions = rng.choice(n_pos, size=n_sample, replace=False)

        block_grads = np.array([grad[sp:sp + bs].sum() for sp in sample_positions])

        src_candidates = sample_positions[block_grads < np.percentile(block_grads, 20)]
        dst_candidates = sample_positions[block_grads > np.percentile(block_grads, 80)]

        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            continue

        best_move = None
        best_improve = 0.0

        n_try = min(len(src_candidates), len(dst_candidates), 30)
        src_pool = rng.choice(src_candidates, size=min(n_try, len(src_candidates)), replace=False)
        dst_pool = rng.choice(dst_candidates, size=min(n_try, len(dst_candidates)), replaceFalse if False else False)

        for sp in src_pool:
            for dp in dst_pool:
                if sp == dp:
                    continue
                src_block = h[sp:sp + bs]
                dst_block = h[dp:dp + bs]

                if src_block.mean() <= dst_block.mean():
                    continue

                delta_max = min(src_block.min(), 1.0 - dst_block.max())
                if delta_max <= 1e-12:
                    continue

                g = np.zeros(n_corr)
                for t in range(bs):
                    g += change[dp + t, :] - change[sp + t, :]

                a_d = 0.0
                b_d = delta_max
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                improve = c5_val - approx_max
                if improve > best_improve:
                    best_improve = improve
                    best_move = (sp, dp, delta_star)

        if best_move is not None:
            sp, dp, delta = best_move
            h[sp:sp + bs] -= delta
            h[dp:dp + bs] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for t in range(bs):
                update_change_row(change, sp + t, h, n, shifts)
                update_change_row(change, dp + t, h, n, shifts)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def _tleft():
        return budget_s - 45.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_score = compute_c5(best_h)

    if _tleft() > 60:

                                  [100, 500, 2000], 400, 1e-14)

            best_score = sc
            best_h = h_pol

        h_cd, c5_cd = efficient_pair_cd(best_h, n0, target0,
                                         time_limit=min(200, _tleft() * 0.6),

                                         n_cand_start=30, n_cand_end=80,
                                         max_no_improve=500, rng_seed=123,
                                         max_pairs=500)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if _tleft() > 80:
        h_pol, sc = slsqp_polish(best_h, n0, target0,
                                  [1000, 5000, 20000], 600, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if _tleft() > 120:
        h_bm, c5_bm = block_transfer_cd(best_h, n0, target0,

    if _tleft() > 350:

        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if _tleft() > 300:
```
