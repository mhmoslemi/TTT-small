def subgradient_multishift(h_init, n, target, time_limit, n_steps=20000, n_top=5, step_scale=0.004):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    eta = step_scale

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

        # Use softmax over top shifts
        top_k = np.argsort(corr)[-n_top:]
        weights = np.exp((corr[top_k] - corr[top_k].max()) * 100.0)
        weights /= weights.sum()

        grad = np.zeros(n)
        for si in range(n_top):
            k = int(top_k[si]) - (n - 1)
            w = weights[si]
            if k >= 0:
                if n - k > 0:
                    grad[0:n-k] += w * (1.0 - h[k:n])
                if k < n:
                    grad[k:n] -= w * h[0:n-k]
            else:
                ka = -k
                grad[ka:n] += w * (1.0 - h[0:n-ka])
                grad[0:n-ka] -= w * h[ka:n]

        grad -= np.mean(grad)

        h_try = h - eta * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                h_try2 = h - eta * factor * grad
                h_try2 = project_feasible(h_try2, target)
                sc2 = compute_c5(h_try2)
                if sc2 < c5_val:
                    h = h_try2
                    break
            else:
                eta *= 0.8

        # Adapt step size periodically
        if step % 1000 == 0 and step > 0:
            eta = max(step_scale * max(0.1, 1.0 - step / n_steps), 1e-5)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
