import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change


def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]


def cd_optimize(h_init, n, target, time_limit, max_sweeps=50000,
                n_active_start=2, n_active_end=5,
                n_cand_start=20, n_cand_end=60,
                max_no_improve=500, bisection_iters=16,
                max_pairs=200):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    corr = np.correlate(h, 1.0 - h, mode='full')
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_active = int(n_active_start + (n_active_end - n_active_start) * progress)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            if k >= 0:
                valid1 = np.arange(0, n - k)
                grad[valid1] += w * (1.0 - h[valid1 + k])
                valid2 = np.arange(k, n)
                grad[valid2] -= w * h[valid2 - k]
            else:
                kp = -k
                valid1 = np.arange(kp, n)
                grad[valid1] += w * (1.0 - h[valid1 - kp])
                valid2 = np.arange(0, n - kp)
                grad[valid2] -= w * h[valid2 + kp]

        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = float(np.max(corr))
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        if n_h * n_l <= max_pairs:
            pair_list = [(int(high_set[ii]), int(low_set[jj]))
                         for ii in range(n_h) for jj in range(n_l)
                         if int(high_set[ii]) != int(low_set[jj])]
        else:
            pair_list = []
            for frac in np.linspace(0.0, 1.0, min(max_pairs, 30)):
                hi = int(frac * (n_h - 1))
                lo = int(frac * (n_l - 1))
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pair_list.append((iv, jv))
            while len(pair_list) < max_pairs:
                ii = np.random.randint(n_h)
                jj = np.random.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    pair_list.append((iv, jv))

        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-14:
                continue

            g = change[j] - change[i]
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-13
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-13:
                    b_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if delta_star < 1e-14:
                continue

            approx_max = np.max(corr + delta_star * g)
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5


def block_cd(h_init, n, target, time_limit, max_sweeps=20000,
             n_active=3, block_size=4, bisection_iters=14):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    corr = np.correlate(h, 1.0 - h, mode='full')

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > 400:
                break

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            if k >= 0:
                grad[:n-k] += w * (1.0 - h[k:])
                grad[k:] -= w * h[:-k]
            else:
                kp = -k
                grad[kp:] += w * (1.0 - h[:-kp])
                grad[:n-kp] -= w * h[kp:]

        grad -= np.mean(grad)

        low_set = np.argsort(grad)[:block_size]
        high_set = np.argsort(grad)[-block_size:]

        best_trial = float(np.max(corr))
        best_move = None

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi < 1e-14:
                    continue
                g = np.zeros(2*n-1)
                if i >= 0:
                    lags1 = np.arange(0, i+1)
                    g[np.clip(mid + lags1, 0, 2*n-2)] += (1.0 - np.clip(h[np.maximum(i-lags1,0)],0,1))
                    lags2 = np.arange(-i, n-i)
                    g[np.clip(mid + lags2, 0, 2*n-2)] -= np.clip(h[np.clip(i+lags2,0,n-1)],0,1)
                if j >= 0:
                    lags1 = np.arange(0, j+1)
                    g[np.clip(mid + lags1, 0, 2*n-2)] -= (1.0 - np.clip(h[np.maximum(j-lags1,0)],0,1))
                    lags2 = np.arange(-j, n-j)
                    g[np.clip(mid + lags2, 0, 2*n-2)] += np.clip(h[np.clip(j+lags2,0,n-1)],0,1)

                a_d = 0.0
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-13
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-14:
                    continue
                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_move = (int(i), int(j), delta_star)

        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
