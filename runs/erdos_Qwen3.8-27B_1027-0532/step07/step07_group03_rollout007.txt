def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(corr, alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

        try:

def compute_corr_gradient_at(h, k):
    n = len(h)
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad

def cutting_plane(h_init, n, target, time_limit, max_iters=20000, max_cuts=300):

            keep = max_cuts // 2

            # Sort by f_val to keep the largest

            keep_set = set(sorted_idx[:keep])
            # Also keep most recent

                keep_set.add(i)
            new_g = [cuts_g[i] for i in keep_set]

            continue

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            for ak in active_lags:
                grad += corr_gradient_at(h, int(ak) - mid)
            grad /= len(active_lags)
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-14:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = np.max(vals)
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = np.max(g_vec[active_m])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if delta_star < 1e-14:
                    continue

                approx_max = np.max(corr + delta_star * g_vec) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0 and best_delta > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pi = best_i
            pj = best_j
            change[pi, :] = 0.0
            change[pj, :] = 0.0
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]
        else:
            consecutive_no_improve += 3

    return best_h, best_c5

def cutting_plane_opt(h_init, n, target, time_limit, max_iters=20000, max_cuts=250):
    start = time.time()
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)

    cuts_g = []
    cuts_f = []
    cuts_h = []

    no_improve = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr))

        if c5_val * (2.0 / n) < best_score - 1e-14:
            best_score = c5_val * (2.0 / n)
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 4000:
            cuts_g = []
            cuts_f = []
            cuts_h = []
            no_improve = 0

        k_star = int(np.argmax(corr))
        f_val = float(corr[k_star])

        active_mask = corr >= f_val - 1e-8
        active_lags = np.where(active_mask)[0]
        if len(active_lags) == 1:
            g = corr_gradient_at(h, k_star - mid)
        else:
            g = np.zeros(n)
            for ak in active_lags:
                g += corr_gradient_at(h, int(ak) - mid)
            g /= len(active_lags)

        cuts_g.append(g.copy())
        cuts_f.append(f_val)
        cuts_h.append(h.copy())

        if len(cuts_g) > max_cuts:
            keep = max_cuts // 2
            sorted_idx = sorted(range(len(cuts_g)), key=lambda x: cuts_f[x], reverse=True)
            keep_set = set(sorted_idx[:keep])
            for i in range(max(0, len(cuts_g) - max_cuts // 4), len(cuts_g)):
                keep_set.add(i)
            cuts_g = [cuts_g[i] for i in sorted(keep_set)]
            cuts_f = [cuts_f[i] for i in sorted(keep_set)]
            cuts_h = [cuts_h[i] for i in sorted(keep_set)]

        n_cuts = len(cuts_g)
        if n_cuts < 2:
            continue

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_cuts, n + 1))
        b_ub = np.zeros(n_cuts)
        for i in range(n_cuts):
            A_ub[i, :n] = cuts_g[i]
            A_ub[i, n] = -1.0
            b_ub[i] = float(np.dot(cuts_g[i], cuts_h[i])) - cuts_f[i]

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp,
                          method='highs')
            if not res.success:
                continue
            h_new = res.x[:n].copy()
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)

            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass

    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def fast_cd(h_init, n, target, time_limit, step_sizes=None):
    start = time.time()
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    if step_sizes is None:
        step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002]

    for it in range(200000):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * (2.0 / n)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        grad = corr_gradient_at(h, k_star)
        grad -= np.mean(grad)

        i = int(np.argmin(grad))
        j = int(np.argmax(grad))
        if i == j:
            no_improve += 100
            continue

        improved = False
        for step in step_sizes:
            trial_i = h[i] - step
            trial_j = h[j] + step
            if trial_i < -1e-14 or trial_j > 1.0 + 1e-14:
                continue
            trial_i = max(trial_i, 0.0)
            trial_j = min(trial_j, 1.0)
            h[i] = trial_i
            h[j] = trial_j
            h = project_feasible(h, target)
            new_c5 = compute_c5(h)
            if new_c5 < c5_val - 1e-15:
                improved = True
                break
            else:
                h[i] = h_init[i] if False else trial_i
                h[j] = trial_j
        if not improved:
            h[i] = h[i] + step  # revert last
            h[j] = h[j] - step
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            no_improve += 5

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 80:
        h_p, sc = slsqp_optimize(best_h, n0, target0, [100, 500, 2000], 300)
        if sc < best_score:
            best_score = sc
            best_h = h_p

    if tleft() > 180:
        h_cp, sc_cp = cutting_plane_opt(best_h, n0, target0,
                                         time_limit=min(150, tleft() * 0.55),
                                         max_iters=15000, max_cuts=200)
        if sc_cp < best_score:
            best_score = sc_cp
            best_h = h_cp

    if tleft() > 200:
        h_cd, sc_cd = interp_cd(best_h, n0, target0,
                                 time_limit=min(200, tleft() * 0.55),
                                 max_sweeps=40000,
                                 n_cand_start=15, n_cand_end=55,
                                 max_no_improve=500)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 150:
        n1 = 600
        target1 = n1 / 2.0
        h600 = interpolate_to(best_h, n1, target1)
        h600_cd, sc600 = interp_cd(h600, n1, target1,
                                     time_limit=min(120, tleft() * 0.5),
                                     max_sweeps=20000,
                                     n_cand_start=10, n_cand_end=35,
                                     max_no_improve=400)
        if sc600 < best_score:
            best_score = sc600
            best_h = h600_cd

    if len(best_h) != n0 and tleft() > 100:
        h_back = interpolate_to(best_h, n0, target0)
        sc_back = compute_c5(h_back)
        if sc_back < best_score:
            best_score = sc_back
            best_h = h_back

    if tleft() > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cd2, sc_cd2 = interp_cd(best_h, n_cur, target_cur,
                                   time_limit=min(100, tleft() * 0.5),
                                   max_sweeps=25000,
                                   n_cand_start=20, n_cand_end=70,
                                   max_no_improve=400)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cp2, sc_cp2 = cutting_plane_opt(best_h, n_cur, target_cur,
                                           time_limit=min(60, tleft() * 0.5),
                                           max_iters=8000, max_cuts=150)
        if sc_cp2 < best_score:
            best_score = sc_cp2
            best_h = h_cp2

    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sls, sc_sls = slsqp_optimize(best_h, n_cur, target_cur,
                                        [5000, 20000], 400)
        if sc_sls < best_score:
            best_score = sc_sls
            best_h = h_sls

    if tleft() > 50:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        rng = np.random.RandomState(seed)
        for attempt in range(3):
            if tleft() < 20:
                break
            h_pert = best_h + rng.randn(n_cur) * 0.01
            h_pert = project_feasible(h_pert, target_cur)
            h_ref, sc_ref = interp_cd(h_pert, n_cur, target_cur,
                                       time_limit=min(15, tleft() * 0.3),
                                       max_sweeps=8000,
                                       n_cand_start=20, n_cand_end=60,
                                       max_no_improve=200)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cd3, sc_cd3 = interp_cd(best_h, n_cur, target_cur,
                                   time_limit=min(25, tleft() * 0.5),
                                   max_sweeps=10000,
                                   n_cand_start=25, n_cand_end=80,
                                   max_no_improve=300)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    if tleft() > 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final, sc_final = slsqp_optimize(best_h, n_cur, target_cur,
                                            [10000, 50000], 300)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
