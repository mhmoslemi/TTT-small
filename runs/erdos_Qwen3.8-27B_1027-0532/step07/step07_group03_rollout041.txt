def compute_grad_single(h, k):
    n = len(h)
    idxs = np.arange(n)
    grad = np.zeros(n)
    v1 = (idxs + k >= 0) & (idxs + k < n)
    v2 = (idxs - k >= 0) & (idxs - k < n)
    grad[v1] += (1.0 - h[idxs[v1] + k])
    grad[v2] -= h[idxs[v2] - k]
    return grad

        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    return grad

        g = compute_grad_weighted(h, k_list, weights)
        g -= np.sum(g) / n * np.ones(n)

        momentum_v = 0.9 * momentum_v + 0.1 * g
        h_new = h - step * momentum_v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < c5_val:
            h = h_new
            step = min(step * 1.02, 0.05)
        else:
            step *= step_decay

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def targeted_perturbation(h_best, n, target, rng, time_limit, n_trials=8, pert_scale=0.03):
    start = time.time()
    best_h = h_best.copy()
    best_score = compute_c5(h_best)

    for trial in range(n_trials):
        if time.time() - start > time_limit:
            break
        h_pert = h_best + rng.randn(n) * pert_scale
        h_pert = project_feasible(h_pert, target)
        h_ref, sc_ref = enhanced_cutting_plane(h_pert, n, target,
                                                time_limit=min(30, (time_limit - (time.time() - start)) * 0.5),
                                                max_iters=3000, max_cuts=150, tol=1e-9)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    return best_h, best_score
