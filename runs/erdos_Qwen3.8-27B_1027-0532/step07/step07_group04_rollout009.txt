import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14,
                time_limit=None, start_time=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def multi_active_cd(h_init, n, target, time_limit, max_sweeps=30000,

                delta_lo = max(h[p] for p in i_block) - 1.0
                delta_lo = max(0.0, delta_lo)
                delta_lo = max(delta_lo, -min(h[p] for p in j_block))
                delta_hi = min(min(h[p] for p in i_block),
                               1.0 - max(h[p] for p in j_block))
                if delta_hi - delta_lo < 1e-12:
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_block_i = i_block
                    best_block_j = j_block
                    best_delta = delta_star

        if best_block_i is not None:
            for p in best_block_i:
                h[p] -= best_delta
            for p in best_block_j:
                h[p] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in best_block_i + best_block_j:
                update_change_row(change, p, h, n, shifts)

    return best_h, best_c5

def targeted_cd(h_init, n, target, time_limit, max_sweeps=20000,
                n_cand=50, max_no_improve=300):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        if k_star >= 0:
            region_lo = 0
            region_hi = n - k_star
        else:
            region_lo = -k_star
            region_hi = n

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        region_grad = grad.copy()
        outside_mask = np.zeros(n, dtype=bool)
        outside_mask[:region_lo] = True
        outside_mask[region_hi:] = True
        region_grad[outside_mask] -= 5.0 * abs(grad[outside_mask]).max()

        sorted_idx = np.argsort(region_grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(0.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def sa_structured(h_init, n, target, time_limit, max_steps=30000,
                  t_start=0.08, t_end=0.001, n_cand=50, seed=42):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(seed)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 800:
            break

        progress = min(step / max_steps, 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        mode = rng.rand()
        if mode < 0.7:
            i = int(high_set[rng.randint(len(high_set))])
            j = int(low_set[rng.randint(len(low_set))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                i = (i + 1) % n

        if i == j:
            continue

        delta_lo = max(0.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue

        step_scale = temperature * 2.0
        delta = delta_lo + (delta_hi - delta_lo) * rng.rand()
        delta = min(delta, step_scale + 0.01)
        delta = max(delta, 0.01)

        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-10:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)
            cur_c5 = approx_new_c5

            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def gradient_descent_smooth(h_init, n, target, time_limit, alpha_start=50,
                            alpha_end=50000, n_steps=2000, lr=0.002):
    dx = 2.0 / n
    start = time.time()
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)

    for step in range(n_steps):
        if time.time() - start > time_limit:
            break

        progress = min(step / n_steps, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr_step = lr * (1.0 - 0.5 * progress)

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        weights = weights / np.sum(weights)

        grad = np.zeros(n)
        for k_idx, w in enumerate(weights):
            if w < 1e-10:
                continue
            k = k_idx - (n - 1)
            idxs = np.arange(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]

        grad -= np.mean(grad)

        v = 0.9 * v - lr_step * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        sc = compute_c5(h_new)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5
