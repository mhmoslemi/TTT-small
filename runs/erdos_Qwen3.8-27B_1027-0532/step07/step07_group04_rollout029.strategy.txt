def project_feasible(h, *, target_sum, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-7:
        return h
    # Two-phase repair: adjust sum and clip
    diff = target_sum - current_sum
    while abs(diff) > 1e-3:
        adjustment = diff / n / 0.1  # Small perturbation
        h_center = np.clip(h + adjustment/2, 0, 1)
        h_edge = np.clip(h + adjustment, 1, 0)

def crossover(parent1, parent2, n):
    if len(parent1) != n or len(parent2) != n:
        raise ValueError("Parents must have length n")
    
    # Two-point crossover
    indices = np.arange(n)
    mid1 = np.random.randint(0, n//2)
    mid2 = np.random.randint(n//2, n)
    mask = indices < mid1 or indices > mid2
    child = np.zeros_like(parent1)
    child[mask] = parent1[mask]
    child[~mask] = parent2[~mask]
    return project_feasible(child, target_sum=n/2, n=n)

def mutate(individual, rate=0.1, sigma=0.03):
    n = len(individual)
    mask = np.random.rand(n) < rate
    num_changes = np.sum(mask)
    changes = np.random.normal(scale=sigma, size=num_changes)
    
    if num_changes > 0:
        indices = np.random.permutation(range(len(individual)))[mask]
        for idx, change in zip(indices, changes):
            individual[idx] += change
    return project_feasible(individual, target_sum=n/2, n=n)
