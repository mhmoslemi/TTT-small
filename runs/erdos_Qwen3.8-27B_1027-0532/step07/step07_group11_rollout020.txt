    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row_fast(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def batch_bisect(corr, g_batch, delta_lo, delta_hi, n_iters=16):
    """
    Batch bisection for minimizing max(corr + delta * g) over delta in [delta_lo, delta_hi].
    g_batch: (n_pairs, n_corr)
    delta_lo, delta_hi: (n_pairs,) arrays
    Returns: optimal delta for each pair, shape (n_pairs,)
    """
    lo = delta_lo.copy()
    hi = delta_hi.copy()
    for _ in range(n_iters):
        mid = 0.5 * (lo + hi)
        # Compute max(corr + mid * g) for each pair
        vals = corr[None, :] + mid[:, None] * g_batch
        max_val = vals.max(axis=1)
        # Get subgradient
        s_star = vals.argmax(axis=1)
        subgrad = g_batch[np.arange(g_batch.shape[0]), s_star]
        # Update bounds
        mask_dec = subgrad < -1e-14
        mask_inc = subgrad > 1e-14
        lo[mask_dec] = mid[mask_dec]
        hi[mask_inc] = mid[mask_inc]
    return 0.5 * (lo + hi)

def pair_cd_v2(h_init, n, target, time_limit, max_sweeps=50000,
               beta_start=10.0, beta_end=80.0,
               n_cand_start=60, n_cand_end=20,
               max_no_improve=800, n_bisect=14):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)
    changes_since_rebuild = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        # Adaptive beta
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        beta = beta_start + (beta_end - beta_start) * progress

        # Exponential shift weighting
        shifted_corr = corr - np.max(corr)
        w = np.exp(beta * shifted_corr)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(n_corr)
            w[np.argmax(corr)] = 1.0
        else:
            w /= w_sum

        # Weighted gradient
        grad = np.zeros(n)
        for s in range(n_corr):
            if w[s] < 1e-12:
                continue
            k = s - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w[s] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Adaptive candidate set
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = max(current_ncand, 10)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        # Build batch of candidate pairs
        n_h = len(high_set)
        n_l = len(low_set)
        
        # Limit total pairs for memory
        max_pairs = 5000
        if n_h * n_l > max_pairs:
            # Reduce by taking fewer from one side
            factor = max_pairs / (n_h * n_l)
            n_h_eff = max(int(n_h * np.sqrt(factor)), 5)
            n_l_eff = max(int(n_l * np.sqrt(factor)), 5)
            high_set = high_set[:n_h_eff]
            low_set = low_set[:n_l_eff]

        pairs_i = []
        pairs_j = []
        for i in high_set:
            for j in low_set:
                if i != j:
                    pairs_i.append(i)
                    pairs_j.append(j)
        
        if not pairs_i:
            continue
        
        pairs_i = np.array(pairs_i, dtype=int)
        pairs_j = np.array(pairs_j, dtype=int)
        n_pairs = len(pairs_i)

        # Get change vectors for this batch
        g_batch = change[pairs_j] - change[pairs_i]  # (n_pairs, n_corr)

        # Feasible delta range
        delta_lo = np.maximum(h[pairs_i] - 1.0, -h[pairs_j])
        delta_hi = np.minimum(h[pairs_i], 1.0 - h[pairs_j])

        # Filter out infeasible pairs
        feasible = delta_hi - delta_lo > 1e-12
        if not feasible.any():
            continue
        g_batch = g_batch[feasible]
        pairs_i = pairs_i[feasible]
        pairs_j = pairs_j[feasible]
        delta_lo = delta_lo[feasible]
        delta_hi = delta_hi[feasible]
        n_pairs = len(pairs_i)
        if n_pairs == 0:
            continue

        # Batch bisection
        opt_delta = batch_bisect(corr, g_batch, delta_lo, delta_hi, n_bisect=n_bisect)

        # Evaluate the best move
        vals = corr[None, :] + opt_delta[:, None] * g_batch
        new_maxes = vals.max(axis=1)
        new_c5 = (new_maxes * dx)
        
        best_idx = new_c5.argmin()
        best_new_c5 = new_c5[best_idx]
        best_delta = opt_delta[best_idx]

        if best_new_c5 < c5_val - 1e-14:
            i = int(pairs_i[best_idx])
            j = int(pairs_j[best_idx])
            h[i] -= best_delta
            h[j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row_fast(change, i, h, n, shifts)
            update_change_row_fast(change, j, h, n, shifts)
            changes_since_rebuild += 2
            if changes_since_rebuild > n // 3:
                change = compute_change_matrix(h, n)
                changes_since_rebuild = 0
        else:
            # Try a few random pairs as a fallback
            for _ in range(3):
                i = idxs[np.random.randint(n)]
                j = idxs[np.random.randint(n)]
                if i == j:
                    continue
                delta_lo_r = max(h[i] - 1.0, -h[j])
                delta_hi_r = min(h[i], 1.0 - h[j])
                if delta_hi_r - delta_lo_r < 1e-12:
                    continue
                # Try a few delta values
                for frac in [0.2, 0.5, 0.8]:
                    candidates = [delta_lo_r + frac * (delta_hi_r - delta_lo_r)]
                    for delta_r in candidates:
                        g_r = change[j] - change[i]
                        new_corr_r = corr + delta_r * g_r
                        sc_r = float(np.max(new_corr_r)) * dx

                            h[j] += delta_r
                            h = np.clip(h, 0.0, 1.0)
                            h = project_feasible(h, target)
                            update_change_row_fast(change, i, h, n, shifts)
                            update_change_row_fast(change, j, h, n, shifts)
                            changes_since_rebuild += 2
                            break

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_eq_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        s_vals = shifts
        pos = (p + s_vals >= 0) & (p + s_vals < n)
        neg = (p - s_vals >= 0) & (p - s_vals < n)
        change[p, pos] += (1.0 - h[p + s_vals[pos]])
        change[p, neg] -= h[p - s_vals[neg]]
    return change


def change_row_update(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos = (p + shifts >= 0) & (p + shifts < n)
    neg = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos] += (1.0 - h[p + shifts[pos]])
    change[p, neg] -= h[p - shifts[neg]]


def batch_bisect(corr, g_batch, d_lo, d_hi, n_iter=14):
    lo = d_lo.copy()
    hi = d_hi.copy()
    for _ in range(n_iter):
        mid = 0.5 * (lo + hi)
        vals = corr[None, :] + mid[:, None] * g_batch
        s_star = vals.argmax(axis=1)
        g_star = g_batch[np.arange(g_batch.shape[0]), s_star]
        dec = g_star < -1e-15
        inc = g_star > 1e-15
        lo[dec] = mid[dec]
        hi[inc] = mid[inc]
    return 0.5 * (lo + hi)

            best_h = h_c1

    if _time_remaining(t_start, budget_s) > 60:
        h_p1, sc_p1 = slsqp_optimize(best_h, n1, tgt1, [2000, 8000], 300, 1e-14)
        if sc_p1 < best_sc:
            best_sc = sc_p1
            best_h = h_p1

    n2 = 1600
    tgt2 = n2 / 2.0
    h2 = interp_up(best_h, n2, tgt2)
    sc2 = compute_c5(h2)
    if sc2 < best_sc:
        best_sc = sc2
        best_h = h2

    if _time_remaining(t_start, budget_s) > 200:
        h_c2, sc_c2 = cd_optimize(h2, n2, tgt2,
                                  time_limit=min(300.0, _time_remaining(t_start, budget_s) * 0.55),
                                  beta_start=5.0, beta_end=80.0,
                                  ncand_start=50, ncand_end=15,
                                  max_sweeps=40000, max_no_imp=400,
                                  n_bisect=14, anneal_t=0.005, anneal_decay=0.9997,
                                  seed=seed + 2)
        if sc_c2 < best_sc:
            best_sc = sc_c2
            best_h = h_c2

    if _time_remaining(t_start, budget_s) > 50:
        h_p2, sc_p2 = slsqp_optimize(best_h, n2, tgt2, [5000, 15000], 200, 1e-14)
        if sc_p2 < best_sc:
            best_sc = sc_p2
            best_h = h_p2

    if _time_remaining(t_start, budget_s) > 300:
        n3 = 2000
        tgt3 = n3 / 2.0
        h3 = interp_up(best_h, n3, tgt3)
        sc3 = compute_c5(h3)
        if sc3 < best_sc:
            best_sc = sc3
            best_h = h3

        if _time_remaining(t_start, budget_s) > 200:
            h_c3, sc_c3 = cd_optimize(h3, n3, tgt3,
                                      time_limit=min(250.0, _time_remaining(t_start, budget_s) * 0.55),
                                      beta_start=5.0, beta_end=80.0,
                                      ncand_start=40, ncand_end=12,
                                      max_sweeps=30000, max_no_imp=400,
                                      n_bisect=14, anneal_t=0.0, anneal_decay=1.0,
                                      seed=seed + 3)
            if sc_c3 < best_sc:
                best_sc = sc_c3
                best_h = h_c3

    n_best = len(best_h)
    tgt_best = n_best / 2.0

    if _time_remaining(t_start, budget_s) > 60:
        h_cf, sc_cf = cd_optimize(best_h, n_best, tgt_best,
                                  time_limit=min(50.0, _time_remaining(t_start, budget_s) * 0.6),
                                  beta_start=20.0, beta_end=100.0,
                                  ncand_start=30, ncand_end=15,
                                  max_sweeps=20000, max_no_imp=300,
                                  n_bisect=16, anneal_t=0.0, anneal_decay=1.0,
                                  seed=seed + 4)
        if sc_cf < best_sc:
            best_sc = sc_cf
            best_h = h_cf

    if _time_remaining(t_start, budget_s) > 40:
        h_pf, sc_pf = slsqp_optimize(best_h, n_best, tgt_best, [10000, 30000], 300, 1e-14)
        if sc_pf < best_sc:
            best_sc = sc_pf
            best_h = h_pf

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_sc = compute_c5(final_h)

    return final_h, final_sc, len(final_h)
