def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_topk(h, alpha, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk_vals = np.sort(corr)[-k:]
    c_max = topk_vals[-1]
    shifted = np.clip(alpha * (topk_vals - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def sa_with_topk(h_init, n, target, time_limit, max_steps=40000,

    dx = 2.0 / n
    start = time.time()

            v1 = (idxs - k >= 0) & (idxs - k < n)

            grad[v2] -= h[idxs[v2] + k]

        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def pmean_slsqp(h, n, target, p_values, time_limit, start_time, maxiter=2000):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for p in p_values:
        if time.time() - start_time > time_limit:
            break
        
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return float(np.mean(corr**p)) ** (1.0/p)
        
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, cur_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def smooth_topk_obj(x, alpha, k):
    corr = np.correlate(x, 1.0 - x, mode='full')
    topk_vals = np.sort(corr)[-k:]
    c_max = topk_vals[-1]
    shifted = np.clip(alpha * (topk_vals - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_topk(h, n, target, alphas, k_vals, maxiter, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

    for k_val in k_vals:
        for alpha in alphas:
            if time.time() - start_time > time_limit:
                break
            try:
                res = minimize(smooth_topk_obj, h_cur, args=(alpha, k_val),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    cur_score = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, cur_score

def pmean_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.mean(corr ** p)) ** (1.0 / p)

def slsqp_pmean(h, n, target, p_vals, maxiter, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

    for p in p_vals:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(pmean_obj, h_cur, args=(p,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def fast_cd(h_init, n, target, time_limit, max_sweeps=50000,
            n_cand_start=30, n_cand_end=100, max_no_improve=400,
            use_multi_shift=True, n_shifts=4, topk_weight=3.0):
    dx = 2.0 / n
    start = time.time()
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if use_multi_shift:
            top_idx = np.argsort(corr)[-n_shifts:]
            c_max = corr[top_idx[-1]]
            weights = np.exp(-topk_weight * (c_max - corr[top_idx]) / max(c_max, 1e-10))
            weights = weights / np.sum(weights)
            grad = np.zeros(n)
            for wi, s_idx in enumerate(top_idx):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += weights[wi] * (1.0 - h[idxs[v1] - k])
                grad[v2] -= weights[wi] * h[idxs[v2] + k]
        else:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5

def sa_topk(h_init, n, target, time_limit, max_steps=40000,
            t_start=0.03, t_end=0.0002, n_cand=60, topk=3,
            rng_seed=42):
    dx = 2.0 / n
    start = time.time()
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    rng = np.random.RandomState(rng_seed)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if consecutive_no_improve > 500:
            break

        progress = min(step / max(max_steps, 1), 1.0)
        temperature = t_start * (t_end / t_start) ** progress

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        top_k_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for s_idx in top_k_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= topk
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        i = int(high_set[rng.randint(len(high_set))])
        j = int(low_set[rng.randint(len(low_set))])
        if i == j:
            continue

        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue

        delta = (delta_hi - delta_lo) * rng.rand() + delta_lo
        if abs(delta) < 1e-12:
            continue

        g = change[j] - change[i]
        approx_new_c5 = np.max(corr + delta * g) * dx

        delta_c = approx_new_c5 - c5_val
        accept = False
        if delta_c < -1e-14:
            accept = True
        elif temperature > 1e-12:
            if rng.rand() < np.exp(-max(delta_c, 0.0) / max(temperature * dx, 1e-15)):
                accept = True

        if accept:
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

            if approx_new_c5 < best_c5 - 1e-14:
                best_c5 = approx_new_c5
                best_h = h.copy()
                consecutive_no_improve = 0
            else:
                consecutive_no_improve += 1
        else:
            consecutive_no_improve += 1

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_cd(h_init, n, target, time_limit, block_size=8, max_sweeps=5000,
             max_no_improve=200):
    dx = 2.0 / n
    start = time.time()
    idxs = np.arange(n)
    mid = n - 1
    shifts = np.arange(-(n - 1), n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        top_idx = np.argsort(corr)[-3:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(np.abs(grad))
        active_block = sorted_idx[-block_size:]

        h_block = h[active_block].copy()
        block_sum = np.sum(h_block)

        def block_obj(xb):
            h_trial = h.copy()
            h_trial[active_block] = xb
            return compute_c5(h_trial)

        bounds_block = [(max(0, h_block[bg] - h[active_block[bg]]),
                         min(1, h_block[bg] + (1 - h[active_block[bg]]))) for bg in range(len(active_block))]
        bounds_block = [(0.0, 1.0)] * len(active_block)

        def block_cons(xb):
            return np.sum(xb) - block_sum

        cons_block = [{'type': 'eq', 'fun': block_cons, 'jac': lambda x: np.ones(len(x))}]

        try:
            res = minimize(block_obj, h_block, method='SLSQP',
                           bounds=bounds_block, constraints=cons_block,
                           options={'maxiter': 200, 'ftol': 1e-12})
            h_new_block = np.clip(res.x, 0.0, 1.0)
            h_trial = h.copy()
            h_trial[active_block] = h_new_block
            h_trial = project_feasible(h_trial, target)
            sc = compute_c5(h_trial)
            if sc < c5_val - 1e-14:
                for bg in range(len(active_block)):
                    update_change_row(change, int(active_block[bg]), h_trial, n, shifts)
                h = h_trial
        except Exception:
            pass

    return best_h, best_c5

        h_cd2, sc_cd2 = fast_cd(h2, n2, target2,
                                 time_limit=min(100, tleft() * 0.45),
                                 max_sweeps=30000,
                                 n_cand_start=35, n_cand_end=95,
                                 max_no_improve=300,
                                 use_multi_shift=True, n_shifts=5, topk_weight=4.0)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2
            best_n = n2
            best_target = target2

        # SA on n=800
        if tleft() > 50:
            h_sa2, sc_sa2 = sa_topk(best_h, best_n, best_target,
                                     time_limit=min(45, tleft() * 0.5),
                                     max_steps=20000,
                                     t_start=0.02, t_end=0.0002,
                                     n_cand=55, topk=4, rng_seed=777)
            if sc_sa2 < best_score:
                best_score = sc_sa2
                best_h = h_sa2

        # CD after SA
        if tleft() > 30:
            h_cd2b, sc_cd2b = fast_cd(best_h, best_n, best_target,
                                       time_limit=min(25, tleft() * 0.6),
                                       max_sweeps=15000,
                                       n_cand_start=50, n_cand_end=95,
                                       max_no_improve=200,
                                       use_multi_shift=True, n_shifts=4, topk_weight=4.5)
            if sc_cd2b < best_score:
                best_score = sc_cd2b
                best_h = h_cd2b

    # Phase 7: Final SLSQP polish
    if tleft() > 40:
        h_pol3, sc_pol3 = slsqp_topk(best_h, best_n, best_target,
                                      [5000, 20000],
                                      [3, 5],
                                      600,
                                      time_limit=min(35, tleft() * 0.8),
                                      start_time=start_time)
        if sc_pol3 < best_score:
            best_score = sc_pol3
            best_h = h_pol3

    # Phase 8: Final rapid CD
    if tleft() > 25:
        h_cd_fin, sc_fin = fast_cd(best_h, best_n, best_target,
                                    time_limit=min(20, tleft() * 0.7),
                                    max_sweeps=15000,
                                    n_cand_start=50, n_cand_end=100,
                                    max_no_improve=150,
                                    use_multi_shift=True, n_shifts=5, topk_weight=5.0)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_cd_fin

    # Phase 9: Final SLSQP
    if tleft() > 15:
        h_pol4, sc_pol4 = slsqp_topk(best_h, best_n, best_target,
                                      [20000],
                                      [3],
                                      400,
                                      time_limit=min(12, tleft() * 0.8),
                                      start_time=start_time)
        if sc_pol4 < best_score:
            best_score = sc_pol4
            best_h = h_pol4

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
