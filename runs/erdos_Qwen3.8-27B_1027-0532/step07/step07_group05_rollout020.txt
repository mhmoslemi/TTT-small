import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    lags = np.arange(0, p + 1)
    change[p, mid + lags] += (1.0 - h[p - lags])
    lags2 = np.arange(-p, n - p)
    change[p, mid + lags2] -= h[p + lags2]

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):

    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        lags = np.arange(0, p + 1)
        change[p, mid + lags] += (1.0 - h[p - lags])
        lags2 = np.arange(-p, n - p)
        change[p, mid + lags2] -= h[p + lags2]
    return change

def update_change_row(change, p, h, n):

    change[p, mid + lags2] -= h[p + lags2]

    idxs = np.arange(n)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_active = int(n_active_start + (n_active_end - n_active_start) * progress)
        cand_frac = n_cand_frac_start + (n_cand_frac_end - n_cand_frac_start) * progress
        n_cand = max(5, int(n * cand_frac))
        n_cand = min(n_cand, n // 4)
        
        # Find top active lags
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Select candidates
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        
        # Generate pairs with stratified sampling
        if n_h * n_l <= max_pairs:
            pair_list = [(int(high_set[ii]), int(low_set[jj])) 
                        for ii in range(n_h) for jj in range(n_l)
                        if int(high_set[ii]) != int(low_set[jj])]
        else:
            rng_cd = np.random.RandomState(sweep % 100000)
            pair_list = []
            n_strata = 5
            per_stratum = max_pairs // n_strata
            for s in range(n_strata):
                frac = (s + 0.5) / n_strata
                center_h = int(frac * (n_h - 1))
                center_l = int(frac * (n_l - 1))
                spread_h = max(1, n_h // n_strata)
                spread_l = max(1, n_l // n_strata)
                for _ in range(per_stratum):
                    ii = max(0, min(n_h-1, center_h + rng_cd.randint(-spread_h, spread_h+1)))
                    jj = max(0, min(n_l-1, center_l + rng_cd.randint(-spread_l, spread_l+1)))
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pair_list.append((iv, jv))
        
        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-13:
                continue
            
            g = change[j] - change[i]
            
            # Quick check: is this direction promising?
            # f(0) = max(corr), f'(0+) = max(g[active])
            active0 = corr >= np.max(corr) - 1e-14
            if np.max(g[active0]) > 1e-10:
                continue  # Moving in this direction increases the max
            
            # Bisection
            a_d = 0.0
            b_d = delta_hi
            for _ in range(bisection_iters):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-14
                if active.any():
                    f_prime = np.max(g[active])
                else:
                    f_prime = 0.0
                if f_prime > 1e-12:
                    b_d = mid_d
                else:
                    a_d = mid_d
            
            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue
            
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
    
    return best_h, best_c5

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        if n_h * n_l <= max_pairs:
            pair_list = [(int(high_set[ii]), int(low_set[jj]))
                         for ii in range(n_h) for jj in range(n_l)
                         if int(high_set[ii]) != int(low_set[jj])]
        else:
            pair_list = []
            n_strata = 5
            per_stratum = max(1, max_pairs // n_strata)
            for s in range(n_strata):
                frac = (s + 0.5) / n_strata
                center_h = int(frac * (n_h - 1))
                center_l = int(frac * (n_l - 1))
                spread_h = max(1, n_h // (n_strata * 2))
                spread_l = max(1, n_l // (n_strata * 2))
                for _ in range(per_stratum):
                    ii = max(0, min(n_h - 1, center_h + rng.randint(-spread_h, spread_h + 1)))
                    jj = max(0, min(n_l - 1, center_l + rng.randint(-spread_l, spread_l + 1)))
                    iv = int(high_set[ii])
                    jv = int(low_set[jj])
                    if iv != jv:
                        pair_list.append((iv, jv))

        active_mask = corr >= np.max(corr) - 1e-12
        n_active_lags = int(np.sum(active_mask))

        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-13:
                continue

            g = change[j] - change[i]

            if np.max(g[active_mask]) > 1e-10:
                continue

            delta_star = bisection_delta(corr, g, delta_hi, bisection_iters)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5


def cd_with_blocks(h_init, n, target, time_limit, max_sweeps=30000,
                   max_no_improve=500, bisection_iters=12,
                   block_size=3, seed=42):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)
    rng = np.random.RandomState(seed + 1000)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        n_active = 3
        n_cand = max(8, int(n * 0.06))

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_update = None

        active_mask = corr >= np.max(corr) - 1e-12

        n_h = len(high_set)
        n_l = len(low_set)

        n_pair_evals = min(150, n_h * n_l)
        if n_h * n_l <= n_pair_evals:
            candidates = [(int(high_set[ii]), int(low_set[jj]))
                          for ii in range(n_h) for jj in range(n_l)
                          if int(high_set[ii]) != int(low_set[jj])]
        else:
            candidates = []
            for _ in range(n_pair_evals):
                ii = rng.randint(n_h)
                jj = rng.randint(n_l)
                iv = int(high_set[ii])
                jv = int(low_set[jj])
                if iv != jv:
                    candidates.append((iv, jv))

        for i, j in candidates:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-13:
                continue
            g = change[j] - change[i]
            if np.max(g[active_mask]) > 1e-10:
                continue
            delta_star = bisection_delta(corr, g, delta_hi, bisection_iters)
            if abs(delta_star) < 1e-13:
                continue
            approx_max = np.max(corr + delta_star * g) * dx
            if approx_max < best_trial:
                best_trial = approx_max
                best_update = ('pair', i, j, delta_star)

        if sweep % 5 == 0 and n >= 200:
            result = rng.randint(block_size)
            if result == 2:
                bl = rng.randint(20, n - 20)
                br = bl + block_size
                block = h[bl:br].copy()
                block_sum = np.sum(block)
                if block_sum > 1e-6 and block_sum < block_size - 1e-6:
                    perturb = rng.randn(block_size)
                    perturb -= np.mean(perturb)
                    step = 0.05
                    for trial in range(5):
                        new_block = block + step * perturb
                        new_block = np.clip(new_block, 0.0, 1.0)
                        new_block = project_feasible(new_block, block_sum)
                        h_trial = h.copy()
                        h_trial[bl:br] = new_block
                        corr_trial = np.correlate(h_trial, 1.0 - h_trial, mode='full')
                        trial_score = np.max(corr_trial) * dx
                        if trial_score < best_trial:
                            best_trial = trial_score
                            best_update = ('block', bl, br, new_block, step)
                        step *= 0.5

        if best_update is not None:
            if best_update[0] == 'pair':
                _, i, j, delta = best_update
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                update_change_row(change, i, h, n)
                update_change_row(change, j, h, n)
            else:
                _, bl, br, new_block, _ = best_update
                h[bl:br] = new_block
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, target)
                for p in range(bl, br):
                    update_change_row(change, p, h, n)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 30.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0
    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 50:
        h_cd, sc_cd = cd_optimized(best_h, n0, target0, time_limit=min(35, tleft() * 0.5),
                                   max_sweeps=12000, bisection_iters=12,
                                   max_no_improve=250, seed=seed)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    n1 = 800
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)

    if tleft() > 250:
        h_cd1, sc_cd1 = cd_optimized(h1, n1, target1, time_limit=min(300, tleft() * 0.45),
                                     max_sweeps=50000, bisection_iters=14,
                                     max_no_improve=700,
                                     n_active_start=2, n_active_end=4,
                                     n_cand_frac_start=0.03, n_cand_frac_end=0.07,
                                     max_pairs=180, seed=seed + 1)
        if sc_cd1 < best_score:
            best_score = sc_cd1
            best_h = h_cd1
        else:
            best_h = h1

    if tleft() > 120:
        h_pol = slsqp_round(best_h, n1, target1,
                            [200, 1000, 5000], 600, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 180:
        h_cd2, sc_cd2 = cd_with_blocks(best_h, n1, target1, time_limit=min(160, tleft() * 0.5),
                                       max_sweeps=25000, bisection_iters=13,
                                       max_no_improve=400, seed=seed + 10)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    if tleft() > 100:
        h_pol2 = slsqp_round(best_h, n1, target1,
                             [5000, 20000, 80000], 700, 1e-14)
        sc = compute_c5(h_pol2)
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    if tleft() > 200:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        h_cd3, sc_cd3 = cd_optimized(h2, n2, target2, time_limit=min(180, tleft() * 0.55),
                                     max_sweeps=35000, bisection_iters=14,
                                     max_no_improve=500,
                                     n_active_start=2, n_active_end=4,
                                     n_cand_frac_start=0.02, n_cand_frac_end=0.05,
                                     max_pairs=150, seed=seed + 20)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3
            n1 = n2
            target1 = target2

    if tleft() > 100:
        h_pol3 = slsqp_round(best_h, n1, target1,
                             [10000, 50000], 500, 1e-14)
        sc = compute_c5(h_pol3)
        if sc < best_score:
            best_score = sc
            best_h = h_pol3

    if tleft() > 60:
        h_cd_final, sc_final = cd_optimized(best_h, n1, target1,
                                            time_limit=min(50, tleft() * 0.6),
                                            max_sweeps=20000, bisection_iters=13,
                                            max_no_improve=300,
                                            n_active_start=3, n_active_end=5,
                                            n_cand_frac_start=0.05, n_cand_frac_end=0.09,
                                            max_pairs=200, seed=seed + 30)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_cd_final

    if tleft() > 40:
        h_last = slsqp_round(best_h, n1, target1,
                             [50000, 200000], 400, 1e-14)
        sc_last = compute_c5(h_last)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
