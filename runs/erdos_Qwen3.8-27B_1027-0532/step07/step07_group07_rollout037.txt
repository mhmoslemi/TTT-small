```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _obj_func(x):
    return smooth_obj(x, _g_alpha)

def _obj_grad(x):
    return smooth_obj_grad(x, _g_alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_alpha, _g_target
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_obj_func, h_cur, jac=_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=100000,
                 n_cand_start=30, n_cand_end=80, top_k_eval=60,
                 max_no_improve=1000, n_shifts_grad=3,
                 active_margin_frac=0.02):
    """Fast pair coordinate descent with active-set-based binary search."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    def update_change_row(p):
        change[p, :] = 0.0
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))
        c_max = corr[k_star_idx]

        # Active set: shifts within margin of max
        margin = max((c_max - float(np.min(corr))) * active_margin_frac, 1e-10)
        active_set = np.where(corr >= c_max - margin)[0]

        # Compute gradient from top shifts
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for tf in top_flat:
            ii = int(tf // n_tl)
            jj = int(tf % n_tl)
            i = int(high_set[ii])
            j = int(low_set[jj])
            if i == j:
                continue

            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-13:
                continue

            g = change[j] - change[i]

            # Binary search using active set only
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(10):
                mid_d = 0.5 * (a_d + b_d)
                vals_active = corr[active_set] + mid_d * g[active_set]
                f_mid = float(np.max(vals_active))
                if f_mid > c_max + 1e-13:
                    # Need to also check the max over ALL shifts
                    f_all = float(np.max(corr + mid_d * g))
                    if f_all > c_max + 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                else:
                    a_d = mid_d

            delta_star = 0.5 * (a_d + b_d)
            if abs(delta_star) < 1e-13:
                continue

            approx_max = float(np.max(corr + delta_star * g)) * dx
            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(best_i)
            update_change_row(best_j)

    return best_h, best_c5
```

```python
        lr = base_lr / (1.0 + 0.01 * t)

        if sc < c5_val + 0.001 * lr:  # Allow slight deterioration
            h = h_new
        else:
            v *= 0.5  # Dampen velocity on rejection
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad

def _sf(x):
    return smooth_obj(x, _g_alpha)

def _sg(x):
    return smooth_obj_grad(x, _g_alpha)

def _fc(x):
    return np.sum(x) - _g_target

def slsqp_optimize(h_start, target, alphas, maxiter):
    global _g_alpha, _g_target
    n = len(h_start)
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _fc}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_sf, h_cur, jac=_sg, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def vcd(h_init, n, target, time_limit, max_sweeps, n_cand_start, n_cand_end,
        top_k_eval, max_no_improve, n_shifts_grad, active_frac):
    """Vectorized pair coordinate descent with active-set acceleration."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos = (p + shifts >= 0) & (p + shifts < n)
        neg = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos] += (1.0 - h[p + shifts[pos]])
        change[p, neg] -= h[p - shifts[neg]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        c_max = float(np.max(corr))

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star_idx = int(np.argmax(corr))

        # Active set
        margin = max((c_max - float(np.min(corr))) * active_frac, 1e-10)
        active_set = np.where(corr >= c_max - margin)[0]

        # Gradient
        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, n // 3)

        top_shifts = np.argsort(corr)[-n_shifts_grad:]
        grad = np.zeros(n)
        for s_idx in top_shifts:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(top_shifts))
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        # Score matrix
        ch_h_star = change[high_set, k_star_idx]
        ch_l_star = change[low_set, k_star_idx]
        score_mat = ch_l_star[None, :] - ch_h_star[:, None]

        n_tl = len(low_set)
        flat_scores = score_mat.ravel()
        n_to_eval = min(top_k_eval, flat_scores.size)
        if n_to_eval <= 0:
            continue
        top_flat = np.argpartition(flat_scores, n_to_eval)[:n_to_eval]

        # Extract candidate pairs
        ii_list = np.array([int(tf // n_tl) for tf in top_flat])
        jj_list = np.array([int(tf % n_tl) for tf in top_flat])
        
        # Filter out i==j
        valid_mask = (high_set[ii_list] != low_set[jj_list])
        if not np.any(valid_mask):
            continue
        ii_list = ii_list[valid_mask]
        jj_list = jj_list[valid_mask]
        
        hi_idx = high_set[ii_list]
        lo_idx = low_set[jj_list]
        
        # Compute bounds
        a_vec = np.maximum(h[hi_idx] - 1.0, -h[lo_idx])
        b_vec = np.minimum(h[hi_idx], 1.0 - h[lo_idx])
        valid2 = (b_vec - a_vec) > 1e-13
        if not np.any(valid2):
            continue
        hi_idx = hi_idx[valid2]
        lo_idx = lo_idx[valid2]
        a_vec = a_vec[valid2]
        b_vec = b_vec[valid2]
        
        if len(hi_idx) == 0:
            continue

        # G matrix
        G = change[lo_idx] - change[hi_idx]
        
        # Vectorized binary search
        for _ in range(10):
            mid_d = 0.5 * (a_vec + b_vec)
            # Fast check on active set
            vals_active = corr[active_set][None, :] + np.outer(mid_d, G[:, active_set])
            f_mids = vals_active.max(axis=1)
            exceeds = f_mids > c_max + 1e-13
            # Verify with full set only for suspected exceeders
            if np.any(exceeds):
                full_vals = corr[None, :] + np.outer(mid_d[exceeds], G[exceeds])
                true_exceeds = full_vals.max(axis=1) > c_max + 1e-13
                final_exceeds = np.zeros(len(mid_d), dtype=bool)
                final_exceeds[exceeds] = true_exceeds
            else:
                final_exceeds = exceeds
            b_vec[final_exceeds] = mid_d[final_exceeds]
            a_vec[~final_exceeds] = mid_d[~final_exceeds]

        delta_star = 0.5 * (a_vec + b_vec)
        
        # Final evaluation
        final_max = (corr[None, :] + np.outer(delta_star, G)).max(axis=1) * dx
        
        best_trial_c5 = c5_val
        best_local = -1
        for m in range(len(delta_star)):
            if final_max[m] < best_trial_c5:
                best_trial_c5 = final_max[m]
                best_local = m
        
        if best_local >= 0:
            i = int(hi_idx[best_local])
            j = int(lo_idx[best_local])
            delta = float(delta_star[best_local])
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change rows
            for p in [i, j]:
                change[p, :] = 0.0
                pos = (p + shifts >= 0) & (p + shifts < n)
                neg = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos] += (1.0 - h[p + shifts[pos]])
                change[p, neg] -= h[p - shifts[neg]]

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def momentum_subgrad(h_init, n, target, time_limit, max_iter=30000,
                     base_lr=0.005, momentum=0.9, active_frac=0.03):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 3000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        c_max = float(np.max(corr))

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        margin = max((c_max - float(np.min(corr))) * active_frac, 1e-10)
        active = np.where(corr >= c_max - margin)[0]

        grad = np.zeros(n)
        for s_idx in active:
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= float(len(active))
        grad -= np.mean(grad)

        lr = base_lr / (1.0 + 0.005 * t)
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, target)
        sc = compute_c5(h_new)
        if sc <= c5_val + lr * 0.1:
            h = h_new
        else:
            v *= 0.5

    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(x, alpha):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / s
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - x[idxs[v1] - k])
        grad[v2] -= w * x[idxs[v2] + k]
    return grad


def _sf(x):
    return smooth_obj(x, _g_alpha)


def _sg(x):
    return smooth_obj_grad(x, _g_alpha)


def _fc(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, target, alphas, maxiter):
    global _g_alpha, _g_target
    n = len(h_start)
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _fc}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_sf, h_cur, jac=_sg, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = np.zeros((n, n_corr))

            going_up = sg > 1e-13
            b_vec[going_up] = mid_d[going_up]

        delta_star = 0.5 * (a_vec + b_vec)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def momentum_subgrad(h_init, n, target, time_limit, max_iter, base_lr, momentum, active_frac):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        c_max = float(np.max(corr))

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        margin = max((c_max - float(np.min(corr))) * active_frac, 1e-10)
        active = np.where(corr >= c_max - margin)[0]

        grad = np.zeros(n)
        for s_idx in active:
            k = int(s_idx) - mid
            v1m = (idxs - k >= 0) & (idxs - k < n)
            v2m = (idxs + k >= 0) & (idxs + k < n)
            grad[v1m] += (1.0 - h[idxs[v1m] - k])
            grad[v2m] -= h[idxs[v2m] + k]
        grad /= float(len(active))
        grad -= np.mean(grad)

        lr = base_lr / (1.0 + 0.003 * t)
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, target)
        sc = compute_c5(h_new)
        if sc <= c5_val + lr * 0.5:
            h = h_new
        else:
            v *= 0.5

    return best_h, best_c5

                            top_k_eval=120, max_no_improve=800,
                            n_shifts_grad=4, active_frac=0.015)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    tl = _tl()
    if tl > 60:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05, 0.10]:
            tl = _tl()
            if tl < 45:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref, c5_ref = vcd(h_pert, n0, target0,
                                time_limit=min(35, tl * 0.7),
                                max_sweeps=100000,
                                n_cand_start=30, n_cand_end=80,
                                top_k_eval=80, max_no_improve=500,
                                n_shifts_grad=3, active_frac=0.02)
            if c5_ref < best_score:
                best_score = c5_ref
                best_h = h_ref

    tl = _tl()
    if tl > 40:
        h_fin = slsqp_optimize(best_h, target0, [5000, 20000, 80000], 600)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    tl = _tl()
    if tl > 30:
        h_cd_fin, c5_fin = vcd(best_h, n0, target0,
                               time_limit=min(30, tl * 0.8),
                               max_sweeps=50000,
                               n_cand_start=50, n_cand_end=100,
                               top_k_eval=100, max_no_improve=300,
                               n_shifts_grad=4, active_frac=0.01)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_cd_fin

    final_h = project_feasible(best_h, float(len(best_h)) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
