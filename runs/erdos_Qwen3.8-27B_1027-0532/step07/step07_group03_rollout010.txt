    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    max_no_improve = 3000

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(10 + 90 * progress)
        current_ncand = min(current_ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g_vec = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(22):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_vec
                    f_mid = float(np.max(vals))
                    active_m = vals >= f_mid - 1e-14
                    if active_m.any():
                        f_prime = float(np.max(g_vec[active_m]))
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = float(np.max(corr + delta_star * g_vec)) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            pi = best_i
            pj = best_j
            change[pi, :] = 0.0
            change[pj, :] = 0.0
            pos_i = (pi + shifts >= 0) & (pi + shifts < n)
            neg_i = (pi - shifts >= 0) & (pi - shifts < n)
            change[pi, pos_i] += (1.0 - h[pi + shifts[pos_i]])
            change[pi, neg_i] -= h[pi - shifts[neg_i]]
            pos_j = (pj + shifts >= 0) & (pj + shifts < n)
            neg_j = (pj - shifts >= 0) & (pj - shifts < n)
            change[pj, pos_j] += (1.0 - h[pj + shifts[pos_j]])
            change[pj, neg_j] -= h[pj - shifts[neg_j]]

    return best_h, best_c5


def smooth_slsqp(h_init, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target
    _g_n = n
    _g_target = target

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, cur_score


def multi_round_cp(h_init, n, target, total_time, n_rounds=3):
    start = time.time()
    h_cur = h_init.copy()
    best = h_cur.copy()
    best_sc = compute_c5(h_cur)
    round_time = total_time / n_rounds

    for r in range(n_rounds):
        tl = min(round_time, total_time - (time.time() - start) - 5.0)
        if tl < 10:
            break
        tols = [1e-10, 1e-11, 1e-12][:n_rounds]
        tol = tols[min(r, len(tols) - 1)]
        h_new, sc = enhanced_cutting_plane(h_cur, n, target, time_limit=tl,
                                           max_iters=20000, tol=tol, max_cuts=300)
        if sc < best_sc:
            best_sc = sc
            best = h_new.copy()
            h_cur = h_new.copy()

    return best, best_sc


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init.copy(), target0)
    best_score = compute_c5(best_h)

    # Phase 1: Multi-round cutting plane
    if tleft() > 250:
        h1, sc1 = multi_round_cp(best_h, n0, target0,
                                  total_time=min(300, tleft() * 0.5), n_rounds=3)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: Refined coordinate descent
    if tleft() > 180:
        h2, sc2 = refined_cd(best_h, n0, target0,
                             time_limit=min(200, tleft() * 0.45),
                             max_sweeps=50000)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Another cutting plane round for polishing
    if tleft() > 150:
        h3, sc3 = enhanced_cutting_plane(best_h, n0, target0,
                                          time_limit=min(180, tleft() * 0.4),
                                          max_iters=20000, tol=1e-11, max_cuts=300)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: SLSQP with high alpha
    if tleft() > 100:
        h4, sc4 = smooth_slsqp(best_h, n0, target0,
                                alphas=[5000, 20000, 80000],
                                maxiter_per_alpha=600, ftol=1e-15)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Final CD pass
    if tleft() > 100:
        h5, sc5 = refined_cd(best_h, n0, target0,
                             time_limit=min(120, tleft() * 0.4),
                             max_sweeps=40000)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: Random perturbations + re-optimization
    if tleft() > 150:
        rng = np.random.RandomState(seed)
        for ns in [0.02, 0.04, 0.06, 0.10]:
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(n0) * ns
            h_pert = project_feasible(h_pert, target0)
            h_ref, sc_ref = enhanced_cutting_plane(h_pert, n0, target0,
                                                    time_limit=min(60, tleft() * 0.3),
                                                    max_iters=8000, tol=1e-10, max_cuts=200)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
            # Quick CD polish on the perturbed result
            if tleft() > 40:
                hcd, scd = refined_cd(h_ref, n0, target0,
                                       time_limit=min(30, tleft() * 0.2),
                                       max_sweeps=15000)
                if scd < best_score:
                    best_score = scd
                    best_h = hcd

    # Phase 7: Final cutting plane from best
    if tleft() > 60:
        h7, sc7 = enhanced_cutting_plane(best_h, n0, target0,
                                          time_limit=min(55, tleft() * 0.5),
                                          max_iters=10000, tol=1e-11, max_cuts=200)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 8: Final SLSQP + CD
    if tleft() > 30:
        h8, sc8 = smooth_slsqp(best_h, n0, target0,
                                alphas=[50000, 200000],
                                maxiter_per_alpha=400, ftol=1e-15)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    if tleft() > 20:
        h9, sc9 = refined_cd(best_h, n0, target0,
                             time_limit=min(15, tleft() * 0.5),
                             max_sweeps=8000)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
