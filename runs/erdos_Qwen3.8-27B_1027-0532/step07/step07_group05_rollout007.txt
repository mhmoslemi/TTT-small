                best_j = j
                best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)
    
    return best_h, best_c5

def slsqp_polish(h_start, n, target, alphas=[5000, 20000], maxiter=500):
    """SLSQP with smooth approximation."""
    _n = n
    _target = target
    
    def obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_fun(x):
        return np.sum(x) - _target
    
    def cons_jac(x):
        return np.ones(_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        try:
            res = minimize(obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur

def binary_ls(h_start, n, target, time_limit):
    """Binary local search: find good binary solution via random swaps."""
    start = time.time()
    dx = 2.0 / n
    rng = np.random.RandomState(12345)
    
    # Round to binary
    h_bin = h_start.copy()
    sorted_idx = np.argsort(-h_bin)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    best_score = np.max(corr) * dx
    best_h = h_bin.copy()
    
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    
    no_improve = 0
    max_no_improve = 200
    temp = 1.0  # simulated annealing temperature
    
    while time.time() - start < time_limit and no_improve < max_no_improve:
        temp *= 0.995
        n_trials = 200
        improved = False
        
        for _ in range(n_trials):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if i == j:
                continue
            
            # Try swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            new_corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
            new_score = np.max(new_corr) * dx
            
            if new_score < best_score:
                best_score = new_score
                best_h = h_bin.copy()
                improved = True
                no_improve = 0
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
            else:
                # Store old values to undo
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        
        if not improved:
            no_improve += 1
    
    return best_h, best_score

def subgradient_polish(h_start, n, target, time_limit, n_active=5):
    """Subgradient method with projection onto feasible set."""
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    step_size = 0.1
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(100000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = np.max(corr) * dx
        
        if c5_val < best_score:
            best_score = c5_val
            best_h = h.copy()
        
        # Subgradient: weighted average of gradients of top lags
        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum
        
        grad = np.zeros(n)
        idxs = np.arange(n)
        mid = n - 1
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        
        # Nesterov-like update
        v = momentum * v - step_size * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        if np.max(np.abs(h_new - h)) < 1e-10:
            step_size *= 0.95
        h = h_new
        step_size *= 0.9999  # slow decay
    
    return best_h, best_score
