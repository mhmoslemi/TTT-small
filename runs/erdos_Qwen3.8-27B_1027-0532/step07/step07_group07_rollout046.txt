def subgradient_descent(h_init, n, target, time_limit, max_iter=5000, 
                         n_top_shifts=5, lr_start=0.05, lr_end=0.005):
    """Subgradient method with projection for the max-of-quadratics problem."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find top shifts by correlation value
        top_indices = np.argsort(corr)[-n_top_shifts:]
        # Weight by how much each shift contributes to the max
        top_weights = np.zeros(2*n-1)
        c_max = np.max(corr)
        for idx in top_indices:
            top_weights[idx] = min(1.0, (corr[idx] / c_max) ** 2)
        if top_weights.sum() > 0:
            top_weights /= top_weights.sum()
        
        # Compute subgradient from top shifts
        idxs = np.arange(n)
        mid = n - 1
        grad = np.zeros(n)
        for s_idx in top_indices:
            w = top_weights[s_idx]
            if w < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        
        # Learning rate schedule
        progress = it / max(max_iter, 1)
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Take step
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        
        # Backtracking line search
        for back in range(8):
            t = lr * (0.5 ** back)
            h_try = h - t * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < c5_val:
                break
        
        h = h_try
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)

        if abs(diff) < 1e-15:
            break

            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):

        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    n = len(x)

            if approx_max < best_trial_c5:
                best_trial_c5 = approx_max
                best_i = i
                best_j = j
                best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, h, best_i, shifts, n)
            update_change_row(change, h, best_j, shifts, n)

    return best_h, best_c5


def subgradient_step(h, n, target, n_top=5, lr=0.03):
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    top_indices = np.argsort(corr)[-n_top:]
    weights = np.zeros(2 * n - 1)
    for idx in top_indices:
        weights[idx] = np.exp((corr[idx] - c_max) * 10.0)
    if weights.sum() < 1e-30:
        weights[top_indices[-1]] = 1.0
    else:
        weights /= weights.sum()

    grad = np.zeros(n)
    for s_idx in top_indices:
        w = weights[s_idx]
        if w < 1e-8:
            continue
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w * h[idxs[v2] + k]

    h_try = h - lr * grad
    h_try = project_feasible(h_try, target)
    return h_try


def subgradient_descent(h_init, n, target, time_limit, max_iter=3000,
                        n_top=5, lr_start=0.04, lr_end=0.002):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break

        progress = it / max(max_iter, 1)
        lr = lr_start * (lr_end / lr_start) ** progress

        c5_cur = compute_c5(h)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        h_try = subgradient_step(h, n, target, n_top=n_top, lr=lr)
        sc_try = compute_c5(h_try)

        if sc_try < c5_cur:
            h = h_try
        else:
            h_half = subgradient_step(h, n, target, n_top=n_top, lr=lr * 0.3)
            sc_half = compute_c5(h_half)
            if sc_half < c5_cur:
                h = h_half

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def fourier_init(n, n_modes, rng):
    x = np.linspace(0.0, 2.0, n)
    h = np.full(n, 0.5)
    for k in range(1, n_modes + 1):
        a = rng.randn() * (0.4 / (k + 1))
        b = rng.randn() * (0.4 / (k + 1))
        h += a * np.sin(np.pi * k * x) + b * np.cos(np.pi * k * x)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)
