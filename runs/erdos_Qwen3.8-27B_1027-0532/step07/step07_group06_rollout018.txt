from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    if use_pnorm:
        func = _pnorm_obj
    else:

            res = minimize(func, h_cur, args=(param,),

                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1

    change = np.zeros((n, n_corr))

        change[p, neg_mask] -= h[p - shifts[neg_mask]]

The idea: maintain a distribution over shifts {p_k}. The inner problem is min_h Σ_k p_k g_k(h) which is a quadratic program (since each g_k is quadratic in h). We can solve this with projected gradient or SLSQP. Then update p_k to put more weight on the currently worst shifts.

def dual_method(h_init, n, target, time_limit, start_time, 
                n_inner_iters=5, lr_dual=1.0):

    p = np.ones(n_corr) / n_corr  # uniform distribution over shifts
    
    for outer_iter in range(100):
        if time.time() - start_time > time_limit:
            break

        #      = sum_i h[i] * (sum_k p_k * |S_k|) - sum_k p_k * sum_i h[i]*h[i+k]
        
        # Actually, let me just use SLSQP on the weighted sum
        def weighted_obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return float(np.sum(p * corr))
        
        # Use a few SLSQP iterations
        try:
            res = minimize(weighted_obj, h, method='SLSQP',
                          bounds=[(0,1)]*n,
                          constraints=[{'type':'eq','fun':_cons_fun,'jac':_cons_jac}],
                          options={'maxiter': n_inner_iters, 'ftol': 1e-12})

        # Update dual: put more weight on worst shifts
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Exponential weight update

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_p = None  # for dual method

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_lse(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _pnorm_obj(x, p):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    if c_max <= 1e-15:
        return 0.0
    scaled = np.maximum(corr, 0.0) / c_max
    return float(c_max * (np.sum(scaled ** p)) ** (1.0 / p))

def _weighted_corr_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.sum(_g_p * corr))

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, time_limit, start_time, use_pnorm=False):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    if use_pnorm:
        func = _pnorm_obj
    else:
        func = _smooth_lse
    for param in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(func, h_cur, args=(param,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def dual_method(h_init, n, target, time_limit, start_time,
                n_outer=200, lr_dual=0.5, n_inner_evals=3):
    global _g_n, _g_target, _g_p
    _g_n = n
    _g_target = target
    n_corr = 2 * n - 1
    h = h_init.copy()
    p = np.ones(n_corr) / n_corr
    
    for outer_iter in range(n_outer):
        if time.time() - start_time > time_limit:
            break
        
        _g_p = p
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        
        try:
            res = minimize(_weighted_corr_obj, h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': n_inner_evals, 'ftol': 1e-13})
            h_new = project_feasible(res.x, target)
            h = h_new
        except Exception:
            pass
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        if c_max < 1e-15:
            break
        p_new = p * np.exp(lr_dual * corr / c_max)
        p = p_new / np.sum(p_new)
    
    return h, compute_c5(h)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p_idx in range(n):
        pos_mask = (p_idx + shifts >= 0) & (p_idx + shifts < n)
        neg_mask = (p_idx - shifts >= 0) & (p_idx - shifts < n)
        change[p_idx, pos_mask] += (1.0 - h[p_idx + shifts[pos_mask]])
        change[p_idx, neg_mask] -= h[p_idx - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]

def pair_cd_enhanced(h_init, n, target, time_limit, start_time,
                      max_sweeps=50000, n_cand_start=20, n_cand_end=100,
                      max_no_improve=500, n_top_shifts=3, bisection_iters=18,
                      annealing=False, T_start=0.005, T_end=0.0005):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    rng = np.random.RandomState(12345)
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        progress = float(sweep) / max(max_sweeps, 1)
        if annealing:
            T = T_start * (T_end / T_start) ** progress
        else:
            T = 0.0

        if n_top_shifts <= 1:
            k_star_idx = int(np.argmax(corr))
            k_star = k_star_idx - mid
            grad = np.zeros(n)
            v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
            v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
            grad[v2] -= h[idxs[v2] + k_star]
        else:
            top_shifts = np.argsort(corr)[-n_top_shifts:]
            weights = corr[top_shifts].astype(float)
            wsum = np.sum(weights)
            if wsum < 1e-15:
                weights = np.ones(n_top_shifts) / n_top_shifts
            else:
                weights = weights / wsum
            grad = np.zeros(n)
            for s_idx, w in zip(top_shifts, weights):
                k = int(s_idx) - mid
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h[idxs[v1] - k])
                grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 10)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                trial_c5 = approx_max * dx
                
                if trial_c5 < best_trial_c5:
                    best_trial_c5 = trial_c5
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                elif annealing and T > 1e-10:
                    delta_obj = (trial_c5 - c5_val) / T
                    if delta_obj < 500 and rng.rand() < np.exp(-delta_obj):
                        if trial_c5 < best_trial_c5:
                            best_trial_c5 = trial_c5
                            best_i = i
                            best_j = j
                            best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            sum_err = target - np.sum(h)
            if abs(sum_err) > 1e-13:
                h[best_i] += sum_err
                h = np.clip(h, 0.0, 1.0)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5
