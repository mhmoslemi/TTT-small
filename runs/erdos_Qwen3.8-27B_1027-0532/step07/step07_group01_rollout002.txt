               restart_every=200, restart_scale=0.03, fw_every=500):

    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(12345)
    
    change = compute_change_matrix(h, n)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > max_no_improve:
            break
        
        # Frank-Wolfe step periodically
        if sweep > 0 and sweep % fw_every == 0 and consecutive_no_improve > 100:
            # Compute smooth gradient
            alpha = 100.0
            _, grad = smooth_obj_and_grad(h, alpha)
            grad -= np.mean(grad)  # maintain sum constraint
            
            # Frank-Wolfe: find v in [0,1]^n with sum=v target that minimizes grad·v
            # This means v should be 1 where grad is smallest, 0 where largest
            sorted_idx = np.argsort(grad)
            v = np.zeros(n)
            # We need sum(v) = target
            # Set v[sorted_idx[:target]] = 1 if target is integer
            k_full = int(np.floor(target))
            frac = target - k_full
            v[sorted_idx[:k_full]] = 1.0
            if frac > 1e-10:
                v[sorted_idx[k_full]] = frac
            
            # Take a step toward v
            gamma = 0.1
            h_new = (1 - gamma) * h + gamma * v
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < best_c5:
                h = h_new
                best_c5 = sc_new
                best_h = h.copy()
                change = compute_change_matrix(h, n)
                consecutive_no_improve = 0
            else:
                # Try different gamma
                for gamma in [0.05, 0.2, 0.3, 0.5]:
                    h_new = (1 - gamma) * h + gamma * v
                    h_new = np.clip(h_new, 0.0, 1.0)
                    h_new = project_feasible(h_new, target)
                    sc_new = compute_c5(h_new)
                    if sc_new < c5_val:
                        h = h_new
                        change = compute_change_matrix(h, n)
                        break
        
        # Restart with noise if stuck
        if consecutive_no_improve > restart_every and consecutive_no_improve % 50 == 0:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0
        
        # Standard CD step
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            # Update change matrix

                col = m + n - 1
                update_change_column(change, col, h, n, m)
    
    return best_h, best_c5
