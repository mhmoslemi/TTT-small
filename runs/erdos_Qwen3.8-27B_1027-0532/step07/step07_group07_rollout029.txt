import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, alpha, penalty, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    penalty_term = penalty * (np.sum(x) - target) ** 2
    return float(obj + penalty_term)

def smooth_obj_grad(x, alpha, penalty, target):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        w = np.ones(n_corr) / n_corr
    else:
        w = exp_shifted / s
    
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += wk * (1.0 - x[idxs[v1] + k])
        if v2.any():
            grad[v2] -= wk * x[idxs[v2] - k]
    
    # Add penalty gradient
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbgfsb_optimize(h_start, target, alphas, penalty_schedule, maxiter=500, ftol=1e-14):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha, penalty in zip(alphas, penalty_schedule):
        def obj(x):
            return smooth_obj(x, alpha, penalty, target)
        def grad(x):
            return smooth_obj_grad(x, alpha, penalty, target)
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, cur_score

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_obj_penalty(x, alpha, penalty, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    base = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    pen = penalty * (np.sum(x) - target) ** 2
    return float(base + pen)

def smooth_obj_penalty_grad(x, alpha, penalty, target):
    n = len(x)
    n_corr = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s < 1e-300:
        w = np.ones(n_corr) / n_corr
    else:
        w = exp_shifted / s
    
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        k = k_idx - (n - 1)
        v1 = (idxs + k >= 0) & (idxs + k < n)
        if v1.any():
            grad[v1] += wk * (1.0 - x[idxs[v1] + k])
        v2 = (idxs - k >= 0) & (idxs - k < n)
        if v2.any():
            grad[v2] -= wk * x[idxs[v2] - k]
    
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad
