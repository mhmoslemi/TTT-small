        self.target = target
        self.alpha = 1000.0

        c_max = np.max(corr)
        shifted = np.clip(self.alpha * (corr - c_max), -500.0, 0.0)

        # d_corp/d_h[j] is the (j, p) entry of the change matrix
        # So grad = change_matrix.T @ weights... but we haven't computed the change matrix
        # Let's compute it on the fly

        # The gradient w.r.t. h[j] of corr[p] is:
        # (1-h[j+s]) for p = mid + s where j+s in [0,n-1]
        # - h[j-s] for p = mid + s where j-s in [0,n-1]
        
        n = self.n
        mid = n - 1

            if w < 1e-15:
                continue

            j_hi = min(n - 1, n - 1 - s)
            if j_lo <= j_hi:

                grad[j_lo:j_hi+1] += w * (1.0 - h[js])

            # d/dh[j] of (1-h[j-s]*h[j]) = -h[j-s]
            # Wait, corr[p] = sum_i h[i]*(1-h[i+s])
            # h[j] appears in second factor when i+s = j, i.e., i = j-s
            # d/dh[j] of -(h[j-s])term: the term is h[j-s]*(1-h[j]), d/dh[j] = -h[j-s]
            # Valid j: j-s in [0,n-1], so j in [s, s+n-1], intersect [0,n-1]
            j_lo2 = max(0, s)
            j_hi2 = min(n - 1, s + n - 1)
            if j_lo2 <= j_hi2:
                js2 = np.arange(j_lo2, j_hi2 + 1) - s
                grad[j_lo2:j_hi2+1] -= w * h[js2]
        
        return obj, grad

        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        sum_exp = np.sum(exp_vals)
        weights = exp_vals / sum_exp
    
    grad = np.zeros(n)
    for p in range(n_corr):
        w = weights[p]
        if w < 1e-10:
            continue
        s = p - mid
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1) + s
            grad[j_lo:j_hi+1] += w * (1.0 - x[js])
        j_lo2 = max(0, s)
        j_hi2 = min(n - 1, s + n - 1)
        if j_lo2 <= j_hi2:
            js2 = np.arange(j_lo2, j_hi2 + 1) - s
            grad[j_lo2:j_hi2+1] -= w * x[js2]
    return grad

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_with_grad(h_start, n, target, alpha, maxiter, ftol=1e-12):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_func, h_start, jac=grad_func,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-12):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        h_new = slsqp_with_grad(h_cur, n, target, alpha, maxiter_per_alpha, ftol)
        sc = compute_c5(h_new)
        if sc < cur_score:
            cur_score = sc
            h_cur = h_new
    return h_cur

    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

Now for the CD part, let me make it faster. The key optimization is to avoid recomputing the full correlation from scratch each iteration. Instead, maintain it incrementally using the change matrix.

But actually, for the CD, I don't need the full gradient - I just need to identify which coordinates to move. Let me use a simpler, faster approach:

1. Compute correlation (O(n²))
2. Find top-k active correlations
3. Compute gradient at those active correlations (O(n*k))
4. Select top candidates for decrease and increase
5. For each candidate pair, use a simple step formula (not bisection)
6. Update correlation incrementally

The step formula: for pair (i, j) with change vector g = change[j] - change[i]:
- delta_max = min(h[i], 1-h[j])
- Try delta = 0.1, 0.3, 0.5, 0.8, 1.0 (fraction of delta_max)
- Pick the one that gives the best new max

This is O(n_corr) per trial step, O(5 * n_corr) per pair, O(n_pairs * 5 * n_corr) per sweep.
For n=800, n_pairs=50: 50 * 5 * 1600 = 400K per sweep. Much faster than bisection!

And I can maintain the correlation incrementally:
corr_new = corr + delta * g

where g = change[j] - change[i] (precomputed in the change matrix).

Let me also add a "recompute" step every 100 sweeps to avoid numerical drift.

OK let me write the complete solution now. I'll aim for:
1. SLSQP at n=400 with gradient (fast)
2. Interpolate to n=800
3. Fast CD at n=800
4. SLSQP at n=800 with gradient (multiple alphas)
5. More CD
6. Final SLSQP

Let me also add a "restart from scratch" option: try a random initialization at n=800 to see if it leads to a better solution.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 1000.0
_g_cache_corr = None
_g_cache_w = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(x):
    n = _g_n
    alpha = _g_alpha
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    _g_cache_corr = corr
    _g_cache_w = exp_vals / sum_exp
    return float(c_max + np.log(sum_exp) / alpha)

def grad_func(x):
    global _g_cache_corr, _g_cache_w
    n = _g_n
    alpha = _g_alpha
    mid = n - 1
    n_corr = 2 * n - 1
    if _g_cache_corr is not None and len(_g_cache_corr) == n_corr:
        weights = _g_cache_w
    else:
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        sum_exp = np.sum(exp_vals)
        weights = exp_vals / sum_exp
    grad = np.zeros(n)
    active_p = np.where(weights > 1e-10)[0]
    for p in active_p:
        w = weights[p]
        s = p - mid
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi + 1) + s
            grad[j_lo:j_hi+1] += w * (1.0 - x[js])
        j_lo2 = max(0, s)
        j_hi2 = min(n - 1, s + n - 1)
        if j_lo2 <= j_hi2:
            js2 = np.arange(j_lo2, j_hi2 + 1) - s
            grad[j_lo2:j_hi2+1] -= w * x[js2]
    return grad

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_with_grad(h_start, n, target, alpha, maxiter, ftol=1e-12):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_func, h_start, jac=grad_func,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-12):
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        h_new = slsqp_with_grad(h_cur, n, target, alpha, maxiter_per_alpha, ftol)
        sc = compute_c5(h_new)
        if sc < cur_score:
            cur_score = sc
            h_cur = h_new
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    change = np.zeros((n, n_corr))
    for p in range(n):
        s1 = np.arange(-p, n - p)
        k1 = p + s1
        change[p, mid + s1] += (1.0 - h[k1])
        s2 = np.arange(p - (n - 1), p + 1)
        k2 = p - s2
        change[p, mid + s2] -= h[k2]
    return change

def update_change_row(change, p, h, n):
    mid = n - 1
    change[p, :] = 0.0
    s1 = np.arange(-p, n - p)
    k1 = p + s1
    change[p, mid + s1] += (1.0 - h[k1])
    s2 = np.arange(p - (n - 1), p + 1)
    k2 = p - s2
    change[p, mid + s2] -= h[k2]

def fast_cd(h_init, n, target, time_limit, max_sweeps=100000,
            n_active=3, n_cand_start=20, n_cand_end=60,
            max_no_improve=200, n_fractions=4):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    fractions = np.linspace(0.1, 1.0, n_fractions)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if sweep % 200 == 0 and sweep > 0:
            corr = np.correlate(h, 1.0 - h, mode='full')

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve > max_no_improve:
                break

        top_idx = np.argsort(corr)[-n_active:]
        w_sum = np.sum(corr[top_idx])
        if w_sum <= 0:
            w_sum = 1.0
        weights = corr[top_idx] / w_sum

        grad = np.zeros(n)
        for idx in range(n_active):
            k = int(top_idx[idx]) - mid
            w = weights[idx]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        n_cand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        n_cand = min(n_cand, max(8, n // 5))

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val / dx
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        max_pairs = 80
        if n_h * n_l <= max_pairs:
            pair_list = [(int(high_set[ii]), int(low_set[jj]))
                        for ii in range(n_h) for jj in range(n_l)
                        if int(high_set[ii]) != int(low_set[jj])]
        else:
            pair_list = []
            for frac_idx in range(min(max_pairs, n_h)):
                hi = int(frac_idx * (n_h - 1) / max(1, min(max_pairs, n_h) - 1)) if n_h > 1 else 0
                lo = frac_idx % n_l
                iv = int(high_set[hi])
                jv = int(low_set[lo])
                if iv != jv:
                    pair_list.append((iv, jv))

        top1 = int(np.argmax(corr))

        for i, j in pair_list:
            if i == j:
                continue
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi < 1e-12:
                continue
            g = change[j] - change[i]
            gt = g[top1]
            if gt >= -1e-14:
                continue
            for frac in fractions:
                d = frac * delta_hi
                new_max = np.max(corr + d * g)
                if new_max < best_trial:
                    best_trial = new_max
                    best_i = i
                    best_j = j
                    best_delta = d
                elif new_max > best_trial:
                    break

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            corr += best_delta * (change[best_j] - change[best_i])
            update_change_row(change, best_i, h, n)
            update_change_row(change, best_j, h, n)

    return best_h, best_c5
