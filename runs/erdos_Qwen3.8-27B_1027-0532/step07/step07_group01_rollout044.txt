            g = g_weighted
            g -= np.mean(g)  # project out the constant component (for the equality constraint)

        h_new = np.clip(h_new, 0.0, 1.0)

            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(x, alpha):
    n = len(x)
    mid = n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    for s_idx in range(len(corr)):
        if weights[s_idx] < 1e-12:
            continue
        k = s_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[s_idx] * (1.0 - x[idxs[v1] - k])
        grad[v2] -= weights[s_idx] * x[idxs[v2] + k]
    return grad

def cons_fun(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, time_limit=600):
    global _g_n, _g_target
    start = time.time()
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def nesterov_subgrad(h_init, n, target, time_limit, lr_init=0.05, lr_end=0.001,
                      beta=0.5, p_power=3.0, min_steps=100, seed=789):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)

    for t in range(min_steps, 100000):
        if time.time() - start > time_limit:
            break

        progress = (t - min_steps) / max(1, t - min_steps)
        lr = lr_init * (lr_end / lr_init) ** min(progress, 1.0)

        # Nesterov lookahead
        y = h + (t - 1) / (t + 2.0) * v
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)

        # Compute weighted subgradient using power-sum
        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr)
        if c_max < 1e-15:
            break
        weights = (corr / c_max) ** (p_power - 1)
        w_sum = np.sum(weights)
        if w_sum < 1e-15:
            break
        weights /= w_sum

        idxs = np.arange(n)
        mid = n - 1
        g = np.zeros(n)
        for s_idx in range(len(corr)):
            if weights[s_idx] < 1e-12:
                continue
            k = s_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v1] += weights[s_idx] * (1.0 - y[idxs[v1] - k])
            g[v2] -= weights[s_idx] * y[idxs[v2] + k]
        g -= np.mean(g)

        v = y - h  # momentum update
        h_new = y - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14, time_limit=120):
    global _g_n, _g_target
    start = time.time()
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(seed)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if consecutive_no_improve > restart_every and consecutive_no_improve % 40 == 0:
            h = best_h + rng.randn(n) * restart_scale
            h = project_feasible(h, target)
            change = compute_change_matrix(h, n)
            consecutive_no_improve = 0

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)

        effective_h = high_set
        effective_l = low_set
        if use_random_pairs and rng.random() < random_frac:
            effective_h = rng.choice(n, size=min(current_ncand, n), replace=False)
            effective_l = rng.choice(n, size=min(current_ncand, n), replace=False)

        for ii in range(len(effective_h)):
            i = int(effective_h[ii])
            for jj in range(len(effective_l)):
                j = int(effective_l[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def multi_shift_cd(h_init, n, target, time_limit, max_sweeps=20000, n_cand=50,
                   n_shifts=5, seed=456):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts_arr = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    rng = np.random.RandomState(seed)

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 350:
            break

        top_shifts = np.argsort(corr)[-n_shifts:]
        top_sum = np.sum(corr[top_shifts])
        if top_sum < 1e-15:
            top_sum = 1.0

        grad = np.zeros(n)
        for s_idx in top_shifts:
            w = corr[s_idx] / top_sum
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for ii in range(len(high_set)):
            i = int(high_set[ii])
            for jj in range(len(low_set)):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts_arr)
            update_change_row(change, best_j, h, n, shifts_arr)

    return best_h, best_c5


def nesterov_subgrad(h_init, n, target, time_limit, lr_init=0.03, lr_end=0.0005,
                     p_power=3.0, max_steps=20000, seed=789):
    start = time.time()
    h = h_init.copy().astype(float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1

    for t in range(1, max_steps):
        if time.time() - start > time_limit:
            break

        progress = t / max_steps
        lr = lr_init * (lr_end / lr_init) ** progress

        mom = (t - 1) / (t + 2.0) if t > 1 else 0.0
        y = h + mom * v
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)

        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = np.max(corr)
        if c_max < 1e-15:
            break
        weights = (corr / c_max) ** (p_power - 1)
        w_sum = np.sum(weights)
        if w_sum < 1e-15:
            break
        weights /= w_sum

        g = np.zeros(n)
        for s_idx in range(len(corr)):
            if weights[s_idx] < 1e-10:
                continue
            k = s_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v1] += weights[s_idx] * (1.0 - y[idxs[v1] - k])
            g[v2] -= weights[s_idx] * y[idxs[v2] + k]
        g -= np.mean(g)

        v = y - h
        h_new = y - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src + 1, endpoint=True)
    x_old_mid = x_old[:-1] + (x_old[1] - x_old[0]) / 2.0
    x_new = np.linspace(0.0, 2.0, n_dst + 1, endpoint=True)
    x_new_mid = x_new[:-1] + (x_new[1] - x_new[0]) / 2.0
    h_out = np.interp(x_new_mid, x_old_mid, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
