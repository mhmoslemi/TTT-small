def compute_change_row(p, h, n):
    """Compute the change in correlation when h[p] increases by 1."""
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    row = np.zeros(n_corr)
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    row[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    row[neg_mask] -= h[p - shifts[neg_mask]]
    return row

def pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
            n_cand_start=30, n_cand_end=80, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    # Precompute change matrix
    change = np.zeros((n, n_corr))
    for p in range(n):
        change[p] = compute_change_row(p, h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Use weighted active set (top 10)
        n_active = min(10, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        # Compute gradient
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Select candidates
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        # Build candidate map
        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        # Get G rows for candidates
        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            G[r, :] = change[all_cands[r], :]

        # Try all pairs (high, low)
        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val * dx
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[high_set[ii]] for ii in range(n_h)], dtype=int)
        lo_rows = np.array([cand_map[low_set[jj]] for jj in range(n_l)], dtype=int)

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue

                g = G[lr] - G[hr]

                # Binary search for best delta
                a_d = delta_min
                b_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta

            h = project_feasible(h, target)
            change[best_i] = compute_change_row(best_i, h, n)
            change[best_j] = compute_change_row(best_j, h, n)

    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _slsqp_constraint(x):
    return np.sum(x) - _g_target

def _slsqp_cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_constraint, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def compute_change_row(p, h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    row = np.zeros(n_corr)
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    row[pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    row[neg_mask] -= h[p - shifts[neg_mask]]
    return row

def pair_cd(h_init, n, target, time_limit, max_sweeps=30000,
            n_cand_min=30, n_cand_max=80, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    # Precompute change matrix
    change = np.zeros((n, n_corr))
    shifts = np.arange(-(n - 1), n)
    for p in range(n):
        change[p] = compute_change_row(p, h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Weighted active set
        n_active = min(10, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w = w / (w.sum() + 1e-30)

        # Gradient
        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        # Candidate selection
        progress = min(float(sweep) / max_sweeps, 1.0)
        ncand = int(n_cand_min + (n_cand_max - n_cand_min) * progress)
        ncand = min(ncand, n // 3)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        G = change[all_cands, :]  # Shape (n_cands, n_corr)

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val * dx
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[high_set[ii]] for ii in range(n_h)], dtype=int)
        lo_rows = np.array([cand_map[low_set[jj]] for jj in range(n_l)], dtype=int)

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue

                g = G[lr] - G[hr]

                # Binary search
                a_d = delta_min
                b_d = delta_max
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            change[best_i] = compute_change_row(best_i, h, n)
            change[best_j] = compute_change_row(best_j, h, n)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
