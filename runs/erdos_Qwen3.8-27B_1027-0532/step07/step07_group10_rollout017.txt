                approx_max = np.max(corr_copy + delta_star * g)
                if approx_max < best_trial:
                    best_trial = float(approx_max)
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def projected_subgrad(h_init, n, target, time_limit, n_active=5,
                      base_lr=0.01, lr_decay=0.99995, max_steps=100000,
                      stall_limit=2000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = base_lr
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > stall_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.95
                if lr < 1e-7:
                    break

        top_k = min(n_active, 2 * n - 1)
        top_shifts = np.argsort(corr)[-top_k:]
        c_max = corr[top_shifts[-1]]
        w = np.exp(-30.0 * (c_max - corr[top_shifts]))
        wsum = np.sum(w)
        if wsum < 1e-30:
            w = np.zeros(len(top_shifts))
            w[-1] = 1.0
        else:
            w /= wsum

        grad = np.zeros(n)
        for wi, si in zip(w, top_shifts):
            if wi < 1e-12:
                continue
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad -= np.mean(grad)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < compute_c5(h) - 1e-15:
            h = h_try
        else:
            lr *= 0.5
            if lr < 1e-8:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0 / n_dst + 2.0 / n_dst, n_dst)
    x_old = np.linspace(0.0, 2.0 / n_src + 2.0 / n_src, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def randomized_perturbation_optimize(h_init, n, target, time_limit, seed,
                                     n_active=3, n_cand=60):
    start = time.time()
    rng = np.random.RandomState(seed)
    h_best = h_init.copy()
    best_score = compute_c5(h_best)

    n_perturb = 5
    per_time = time_limit / n_perturb

    for p in range(n_perturb):
        if time.time() - start > time_limit:
            break
        remaining = min(per_time, time_limit - (time.time() - start))
        if remaining < 10:
            break

        scale = 0.03 * (1.0 + 0.5 * p)
        h_pert = h_best + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)

        sc = compute_c5(h_pert)
        if sc > best_score + 0.05:
            continue

        h_cd, sc_cd = multi_shift_cd(h_pert, n, target,
                                     time_limit=remaining * 0.6,
                                     n_active=n_active,
                                     n_cand_start=30, n_cand_end=n_cand,
                                     max_sweeps=20000, max_no_improve=200)
        if sc_cd < best_score:
            best_score = sc_cd
            h_best = h_cd

    return h_best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)
    best_n = n0
    best_target = target0

    if tleft() > 100:
        h_pol, sc = slsqp_round(best_h, best_n, best_target,
                                [100, 300, 1000, 3000], 800,
                                time_limit=min(120, tleft() * 0.4))
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_cd, c5_cd = multi_shift_cd(best_h, best_n, best_target,
                                     time_limit=min(200, tleft() * 0.5),
                                     n_active=4, n_cand_start=30, n_cand_end=100,
                                     max_sweeps=50000, max_no_improve=400)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_sg, c5_sg = projected_subgrad(best_h, best_n, best_target,
                                        time_limit=min(60, tleft() * 0.4),
                                        n_active=6, base_lr=0.015)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 60:
        h_pol2, sc2 = slsqp_round(best_h, best_n, best_target,
                                  [5000, 20000], 1000,
                                  time_limit=min(60, tleft() * 0.5))
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    n1 = 800
    if tleft() > 400:
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)

        if tleft() > 300:
            h1_pol, sc1p = slsqp_round(h1, n1, target1,
                                       [200, 800, 3000, 10000], 600,
                                       time_limit=min(150, tleft() * 0.4))
            if sc1p < sc1:
                h1 = h1_pol
                sc1 = sc1p

        if tleft() > 200:
            h1_cd, c5_1cd = multi_shift_cd(h1, n1, target1,
                                           time_limit=min(250, tleft() * 0.5),
                                           n_active=4, n_cand_start=40, n_cand_end=120,
                                           max_sweeps=80000, max_no_improve=500)
            if c5_1cd < best_score:
                best_score = c5_1cd
                best_h = h1_cd
                best_n = n1
                best_target = target1

        if tleft() > 100:
            h1_sg, c5_1sg = projected_subgrad(h1 if best_n != n1 else best_h,
                                              best_n, best_target,
                                              time_limit=min(80, tleft() * 0.4),
                                              n_active=8, base_lr=0.01)
            if c5_1sg < best_score:
                best_score = c5_1sg
                best_h = h1_sg

        if tleft() > 80:
            h1_pol2, sc1p2 = slsqp_round(best_h, best_n, best_target,
                                         [20000, 50000], 1000,
                                         time_limit=min(80, tleft() * 0.5))
            if sc1p2 < best_score:
                best_score = sc1p2
                best_h = h1_pol2

    n2 = 1200
    if tleft() > 350:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)

        if tleft() > 250:
            h2_pol, sc2p = slsqp_round(h2, n2, target2,
                                       [300, 1000, 5000], 500,
                                       time_limit=min(120, tleft() * 0.4))
            if sc2p < sc2:
                h2 = h2_pol
                sc2 = sc2p

        if tleft() > 150:
            h2_cd, c5_2cd = multi_shift_cd(h2, n2, target2,
                                           time_limit=min(200, tleft() * 0.5),
                                           n_active=3, n_cand_start=40, n_cand_end=100,
                                           max_sweeps=50000, max_no_improve=400)
            if c5_2cd < best_score:
                best_score = c5_2cd
                best_h = h2_cd
                best_n = n2
                best_target = target2

        if tleft() > 60:
            h2_pol2, sc2p2 = slsqp_round(best_h, best_n, best_target,
                                         [10000, 30000], 800,
                                         time_limit=min(60, tleft() * 0.5))
            if sc2p2 < best_score:
                best_score = sc2p2
                best_h = h2_pol2

    if tleft() > 100:
        h_rnd, sc_rnd = randomized_perturbation_optimize(
            best_h, best_n, best_target,
            time_limit=min(120, tleft() * 0.4),
            seed=seed, n_active=4, n_cand=80)
        if sc_rnd < best_score:
            best_score = sc_rnd
            best_h = h_rnd

    if tleft() > 80:
        h_cd_final, c5_final = multi_shift_cd(best_h, best_n, best_target,
                                              time_limit=min(100, tleft() * 0.5),
                                              n_active=5, n_cand_start=50, n_cand_end=130,
                                              max_sweeps=50000, max_no_improve=300)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 50:
        h_pol3, sc3 = slsqp_round(best_h, best_n, best_target,
                                  [30000, 100000], 1000,
                                  time_limit=min(50, tleft() * 0.6))
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3

    if tleft() > 30:
        h_cd_last, c5_last = multi_shift_cd(best_h, best_n, best_target,
                                            time_limit=min(25, tleft() * 0.6),
                                            n_active=4, n_cand_start=60, n_cand_end=120,
                                            max_sweeps=20000, max_no_improve=100)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_cd_last

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
