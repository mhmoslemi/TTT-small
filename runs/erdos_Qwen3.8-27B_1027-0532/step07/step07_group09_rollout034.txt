        if time.time() - start > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = _compute_corr(h)
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        n_active = min(8, n_corr)
        top_idx = np.argsort(corr)[-n_active:]
        w = corr[top_idx].copy()
        w_norm = np.sum(w)
        if w_norm > 0:
            w = w / w_norm
        else:
            w = np.ones(n_active) / n_active

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            lo1 = max(0, -k)
            hi1 = min(n, n - k)
            lo2 = max(0, k)
            hi2 = min(n, n + k)
            if lo1 < hi1:
                grad[lo1:hi1] += wi * (1.0 - h[lo1 + k:hi1 + k])
            if lo2 < hi2:
                grad[lo2:hi2] -= wi * h[lo2 - k:hi2 - k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_low]
        high_set = sorted_idx[-n_high:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[int(all_cands[r])] = r

        G = np.zeros((n_cands, n_corr))
        for r in range(n_cands):
            p = int(all_cands[r])
            G[r, :] = change[p, :]

        n_h = len(high_set)
        n_l = len(low_set)
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(n_h)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(n_l)])

        for ii in range(n_h):
            i = int(high_set[ii])
            hr = hi_rows[ii]
            h_i = h[i]
            for jj in range(n_l):
                j = int(low_set[jj])
                lr = lo_rows[jj]
                h_j = h[j]

                delta_max = min(h_i, 1.0 - h_j)
                delta_min = max(h_i - 1.0, -h_j)
                if delta_max - delta_min < 1e-12:
                    continue

                g = G[lr] - G[hr]
                a_d = delta_min
                b_d = delta_max
                for _ in range(12):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def projected_subgrad(h_init, n, target, time_limit, max_iter=50000,
                      active_tol=1e-5, t_start=0.005):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = t_start
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        corr = _compute_corr(h)
        c_max = np.max(corr)
        c5_val = c_max * dx

        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            t = min(t * 1.05, 0.02)
        else:
            no_improve += 1
            t = t * 0.98

        tol = max(1e-10, active_tol * c_max)
        active = np.where(corr >= c_max - tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            lo1 = max(0, -k)
            hi1 = min(n, n - k)
            lo2 = max(0, k)
            hi2 = min(n, n + k)
            if lo1 < hi1:
                grad[lo1:hi1] += (1.0 - h[lo1 + k:hi1 + k])
            if lo2 < hi2:
                grad[lo2:hi2] -= h[lo2 - k:hi2 - k]
        grad /= len(active)
        grad -= np.mean(grad)

        h_new = project_feasible(h - t * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val:
            h = h_new
        else:
            t = t * 0.7
            h_new = project_feasible(h - t * grad, target)
            if compute_c5(h_new) < c5_val:
                h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def peaking_transform(h, strength):
    n = len(h)
    target = n / 2.0
    h_centered = h - 0.5
    h_peaked = np.sign(h_centered) * np.power(np.abs(h_centered), 1.0 / strength)
    h_peaked = 0.5 + strength * h_peaked
    h_peaked = np.clip(h_peaked, 0.0, 1.0)
    return project_feasible(h_peaked, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 120:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [50, 200, 1000, 5000, 20000],
                                   300, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_cd, c5_cd = pair_cd(best_h, n0, target0,
                              time_limit=min(130, tleft() * 0.5),
                              max_sweeps=20000, n_low=12, n_high=12,
                              max_no_improve=400)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_sg, c5_sg = projected_subgrad(best_h, n0, target0,
                                        time_limit=min(60, tleft() * 0.5),
                                        max_iter=30000, active_tol=1e-5,
                                        t_start=0.004)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 60:
        h_pol, sc = slsqp_optimize(best_h, n0, target0,
                                   [2000, 10000, 50000],
                                   500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n1 = 800
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)

    if tleft() > 200:
        h_pol1, sc1 = slsqp_optimize(h1, n1, target1,
                                     [50, 200, 1000, 5000, 20000],
                                     300, 1e-16)
        sc1 = compute_c5(h_pol1)
        if sc1 < best_score:
            best_score = sc1
            best_h = h_pol1

    if tleft() > 200:
        h_cd1, c5_cd1 = pair_cd(best_h, n1, target1,
                                time_limit=min(180, tleft() * 0.5),
                                max_sweeps=15000, n_low=15, n_high=15,
                                max_no_improve=400)
        if c5_cd1 < best_score:
            best_score = c5_cd1
            best_h = h_cd1

    if tleft() > 100:
        h_sg1, c5_sg1 = projected_subgrad(best_h, n1, target1,
                                          time_limit=min(80, tleft() * 0.5),
                                          max_iter=30000, active_tol=1e-5,
                                          t_start=0.003)
        if c5_sg1 < best_score:
            best_score = c5_sg1
            best_h = h_sg1

    if tleft() > 80:
        h_pol2, sc2 = slsqp_optimize(best_h, n1, target1,
                                     [5000, 20000, 100000],
                                     500, 1e-16)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 120:
        h_cd2, c5_cd2 = pair_cd(best_h, n1, target1,
                                time_limit=min(100, tleft() * 0.5),
                                max_sweeps=15000, n_low=18, n_high=18,
                                max_no_improve=300)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 100:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.06, 0.1]:
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, target1)
            h_ref, sc_ref = slsqp_optimize(h_pert, n1, target1,
                                           [200, 1000, 5000],
                                           300, 1e-16)
            sc_ref = compute_c5(h_ref)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 80:
        h_peak = peaking_transform(best_h, 0.7)
        h_peak_opt, sc_peak = slsqp_optimize(h_peak, n1, target1,
                                             [500, 2000, 10000],
                                             300, 1e-16)
        sc_peak = compute_c5(h_peak_opt)
        if sc_peak < best_score:
            best_score = sc_peak
            best_h = h_peak_opt

    if tleft() > 60:
        h_final_cd, c5_final = pair_cd(best_h, n1, target1,
                                       time_limit=min(40, tleft() * 0.5),
                                       max_sweeps=10000, n_low=20, n_high=20,
                                       max_no_improve=200)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cd

    if tleft() > 40:
        h_final, sc_final = slsqp_optimize(best_h, n1, target1,
                                           [10000, 50000, 200000],
                                           400, 1e-16)
        sc_final = compute_c5(h_final)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
