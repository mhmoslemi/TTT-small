from scipy.optimize import minimize, linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def shift_gradient(h, k, n):
    """Compute gradient of corr at shift k with respect to h"""
    grad = np.zeros(n)
    idxs = np.arange(n)
    # h[i] appears in corr[k] as:
    # (1-h[i+k]) for the first factor, with index i
    # -h[i-k] for the second factor (when h[i] is the h[i+k] term)
    # More precisely:
    # corr at shift k = sum_i h[i]*(1-h[i+k])
    # d(corr)/d(h[j]) = (1-h[j+k]) if j+k < n, plus (-h[j-k]) if j-k >= 0
    
    valid_pos = (idxs + k >= 0) & (idxs + k < n)
    valid_neg = (idxs - k >= 0) & (idxs - k < n)
    grad[valid_pos] += (1.0 - h[idxs[valid_pos] + k])
    grad[valid_neg] -= h[idxs[valid_neg] - k]
    return grad

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def shift_gradient(h, k, n):
    idxs = np.arange(n)
    grad = np.zeros(n)
    vp = (idxs + k >= 0) & (idxs + k < n)
    vn = (idxs - k >= 0) & (idxs - k < n)
    grad[vp] += (1.0 - h[idxs[vp] + k])
    grad[vn] -= h[idxs[vn] - k]
    return grad

We want to minimize t = max_k corr(k) subject to constraints.

For each shift k, corr(k) is a function of h. The constraint corr(k) ≤ t must hold for all k.

In the SLP, we linearize corr(k) around the current h_old:
corr(k, h) ≈ corr(k, h_old) + ∇_h corr(k, h_old) · (h - h_old)

Let g_k = ∇_h corr(k, h_old) (the gradient vector).
Let c_k = corr(k, h_old) (the current value).

The linearized constraint is:
c_k + g_k · (h - h_old) ≤ t
⇔ g_k · h ≤ t - c_k + g_k · h_old
⇔ g_k · h - t ≤ -(c_k - g_k · h_old)
⇔ g_k · h - t ≤ -offset_k

where offset_k = c_k - g_k · h_old.

In the LP:
- Variables: x = [h_0, ..., h_{n-1}, t]
- Objective: minimize x[n] (= t)
- Inequality constraints: g_k · h - t ≤ -offset_k for each active k
  In matrix form: [g_k, -1] · x ≤ -offset_k
- Equality: sum(h) = target, i.e., [1,1,...,1,0] · x = target
- Bounds: 0 ≤ h_i ≤ 1, t ≥ 0

corr(k) = Σ_{i: valid} h[i] * (1 - h[i+k])

where "valid" means 0 ≤ i < n and 0 ≤ i+k < n.

∂corr(k)/∂h[j] = Σ_{i: valid} [∂(h[i](1-h[i+k]))/∂h[j]]

For a specific i:
∂(h[i](1-h[i+k]))/∂h[j] =

import numpy as np
from scipy.optimize import linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def shift_gradient(h, k, n):
    idxs = np.arange(n)
    grad = np.zeros(n)
    vp = (idxs + k >= 0) & (idxs + k < n)
    vn = (idxs - k >= 0) & (idxs - k < n)
    grad[vp] += (1.0 - h[idxs[vp] + k])
    grad[vn] -= h[idxs[vn] - k]
    return grad


def slp_optimize(h_init, n, target, dx, time_limit, max_iter=500):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    mid = n - 1
    n_corr = 2 * n - 1
    stall = 0
    n_active_frac = 0.3

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        prog = float(it) / max_iter
        n_active_frac = 0.3 + 0.5 * min(prog * 2, 1.0)
        n_active = max(10, int(n_corr * n_active_frac))

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_score - 1e-14:
            best_score = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break

        active_idx = np.argsort(corr)[-n_active:]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_ub_list = []
        b_ub_list = []

        for s in active_idx:
            s_int = int(s)
            k = s_int - mid
            grad = shift_gradient(h, k, n)
            corr_s = corr[s_int]
            offset = corr_s - np.dot(grad, h)
            row = np.zeros(n + 1)
            row[:n] = grad
            row[-1] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(-offset)

        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status != 0:
                break
            h_lp = res.x[:n].copy()
        except Exception:
            break

        h_lp = project_feasible(h_lp, target)
        lp_score = compute_c5(h_lp)

        if lp_score < c5_val - 1e-12:
            h = h_lp
        else:
            found = False
            for alpha in [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125]:
                h_try = (1.0 - alpha) * h + alpha * h_lp
                h_try = project_feasible(h_try, target)
                try_score = compute_c5(h_try)
                if try_score < c5_val - 1e-12:
                    h = h_try
                    found = True
                    break
            if not found:
                subgrad_step(h, n, target, dx)
                h_new_score = compute_c5(h)
                if h_new_score >= c5_val:
                    break

    return best_h, best_score


def subgrad_step(h, n, target, dx):
    idxs = np.arange(n)
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    threshold = c_max - 0.1 * max(c_max, 0.01)
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])

    grad = np.zeros(n)
    for s in active:
        k = int(s) - mid
        vp = (idxs + k >= 0) & (idxs + k < n)
        vn = (idxs - k >= 0) & (idxs - k < n)
        grad[vp] += (1.0 - h[idxs[vp] + k])
        grad[vn] -= h[idxs[vn] - k]
    grad /= len(active)

    step = 0.02
    h_new = h - step * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, target)
    h[:] = h_new


def cd_optimize(h_init, n, target, dx, time_limit, max_sweeps=50000,
                n_cand_start=20, n_cand_end=80, n_active_shifts=3,
                max_no_improve=400, reheat_interval=100, reheat_scale=0.04):
    start = time.time()
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    sweeps_since_improve = 0

    change = np.zeros((n, 2 * n - 1))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
            sweeps_since_improve = 0
        else:
            consecutive_no_improve += 1
            sweeps_since_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        if sweeps_since_improve > reheat_interval and sweeps_since_improve % 30 == 0:
            rng = np.random.RandomState(sweep * 13 + 7)
            n_pert = max(10, n // 30)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[pert_idx] += rng.randn(n_pert) * reheat_scale
            h_pert = project_feasible(h_pert, target)
            sc_pert = compute_c5(h_pert)
            if sc_pert < c5_val + 1e-5:
                h = h_pert
                for p in range(n):
                    pos_mask = (p + shifts >= 0) & (p + shifts < n)
                    neg_mask = (p - shifts >= 0) & (p - shifts < n)
                    change[p, :] = 0.0
                    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                    change[p, neg_mask] -= h[p - shifts[neg_mask]]
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                sweeps_since_improve = 0
                continue

        top_k = np.argsort(corr)[-n_active_shifts:]
        corr_top_sum = np.sum(corr[top_k])
        if corr_top_sum < 1e-10:
            weights = np.ones(n_active_shifts) / n_active_shifts
        else:
            weights = corr[top_k] / corr_top_sum

        grad = np.zeros(n)
        for s_idx, w in zip(top_k, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        k_star_idx = int(np.argmax(corr))
        n_h = len(high_set)
        n_l = len(low_set)

        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = change[j] - change[i]
                if g[k_star_idx] > 0.5 * np.max(np.abs(g)):
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(10):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    if n_src > n_dst:
        x_old = np.linspace(0.0, 1.0, n_src)
        x_new = np.linspace(0.0, 1.0, n_dst)
        h_out = np.interp(x_new, x_old, h_src)
    else:
        x_old = np.linspace(0.0, 1.0, n_src)
        x_new = np.linspace(0.0, 1.0, n_dst)
        h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_refine(h_best, n, target, dx, time_limit, seed, n_starts=3):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_score = compute_c5(h_best)

    for s in range(n_starts):
        if time.time() - start > time_limit:
            break
        remaining = time_limit - (time.time() - start)
        if remaining < 20:
            break
        noise_scale = 0.05 + 0.05 * s
        h_pert = h_best + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        slp_time = min(remaining * 0.4, 60)
        h_try, sc_try = slp_optimize(h_pert, n, target, dx, time_limit=slp_time, max_iter=200)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try

    return best_h, best_score
