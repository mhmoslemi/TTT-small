import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change

def binary_to_feasible(h, target):
    """Convert to binary with exactly target ones"""
    h = (h > 0.5).astype(float)
    current_ones = int(np.sum(h))
    if current_ones > target:
        # Need to flip some 1s to 0s
        indices = np.where(h == 1.0)[0]
        rng = np.random.RandomState(42)
        to_flip = rng.choice(indices, size=current_ones - target, replace=False)
        h[to_flip] = 0.0
    elif current_ones < target:
        indices = np.where(h == 0.0)[0]
        rng = np.random.RandomState(42)
        to_flip = rng.choice(indices, size=target - current_ones, replace=False)
        h[to_flip] = 1.0
    return h

def simulated_annealing_binary(h_init, n, target, time_limit, seed=42,
                                  T_init=50.0, T_min=0.01, cooling=0.999):
    start = time.time()
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    
    h = h_init.copy()
    # Ensure binary
    h = binary_to_feasible(h, target)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(corr)) * dx
    best_max = current_max
    best_h = h.copy()
    
    change = compute_change_matrix(h, n)
    
    ones_indices = np.where(h == 1.0)[0]
    zeros_indices = np.where(h == 0.0)[0]
    
    T = T_init
    no_improve = 0
    
    while time.time() - start < time_limit:
        if T < T_min:
            break
        
        # Pick random swap
        i = ones_indices[rng.randint(len(ones_indices))]
        j = zeros_indices[rng.randint(len(zeros_indices))]

        new_corr = corr + g
        new_max = float(np.max(new_corr)) * dx

                no_improve = 0
            else:
                no_improve += 1
        else:

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(x):
    return np.sum(x) - _g_target

def slsqp_refine(h_start, n, target, time_limit, alphas=(2000, 10000, 50000)):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 1500, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_c5 = compute_c5(best_h)

    rng = np.random.RandomState(seed)

    # Phase 1: Initial SLSQP polish on the parent
    if tleft() > 800:
        t_lim = min(60, tleft() * 0.1)
        t_deadline = start_time + (budget_s - 30.0) - t_lim
        h_pol, sc_pol = slsqp_refine(best_h, n, target, t_deadline, alphas=(1000, 5000))
        if sc_pol < best_c5:
            best_c5 = sc_pol
            best_h = h_pol.copy()

    # Phase 2: Binary SA from parent
    if tleft() > 600:
        h_bin = to_binary(best_h, int(target), rng)
        h_sa1, c5_sa1 = binary_sa(h_bin, n, int(target),
                                   time_limit=min(250, tleft() * 0.35),
                                   seed=1, T_init=0.1, T_min=0.0005,
                                   cooling=0.99990, max_no_improve=400000)
        if c5_sa1 < best_c5:
            best_c5 = c5_sa1
            best_h = h_sa1.copy()

    # Phase 3: Second SA run with different params
    if tleft() > 400:
        h_bin2 = to_binary(best_h, int(target), rng)
        h_sa2, c5_sa2 = binary_sa(h_bin2, n, int(target),
                                   time_limit=min(200, tleft() * 0.35),
                                   seed=2, T_init=0.05, T_min=0.0003,
                                   cooling=0.99995, max_no_improve=300000)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: Greedy polish
    if tleft() > 250:
        h_bin3 = to_binary(best_h, int(target), rng)
        h_g, c5_g = binary_greedy_polish(h_bin3, n, int(target),
                                          time_limit=min(150, tleft() * 0.4),
                                          seed=42)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 5: Random restarts with SA
    if tleft() > 350:
        for attempt in range(4):
            if tleft() < 180:
                break
            h_rand = np.zeros(n)
            chosen = rng.choice(n, size=int(target), replace=False)
            h_rand[chosen] = 1.0
            h_r, c5_r = binary_sa(h_rand, n, int(target),
                                   time_limit=min(120, tleft() * 0.35),
                                   seed=100 + attempt,
                                   T_init=0.15, T_min=0.001,
                                   cooling=0.99980, max_no_improve=250000)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: More SA from best
    if tleft() > 250:
        h_bin4 = to_binary(best_h, int(target), rng)
        h_sa3, c5_sa3 = binary_sa(h_bin4, n, int(target),
                                   time_limit=min(150, tleft() * 0.4),
                                   seed=3, T_init=0.03, T_min=0.0002,
                                   cooling=0.99998, max_no_improve=500000)
        if c5_sa3 < best_c5:
            best_c5 = c5_sa3
            best_h = h_sa3.copy()

    # Phase 7: Greedy polish again
    if tleft() > 150:
        h_bin5 = to_binary(best_h, int(target), rng)
        h_g2, c5_g2 = binary_greedy_polish(h_bin5, n, int(target),
                                            time_limit=min(100, tleft() * 0.5),
                                            seed=789)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    # Phase 8: Non-binary refinement
    if tleft() > 120:
        h_nb = best_h + rng.randn(n) * 0.05
        h_nb = project_feasible(h_nb, target)
        t_deadline = start_time + (budget_s - 50.0)
        h_ref, sc_ref = slsqp_refine(h_nb, n, target, t_deadline, alphas=(5000, 20000, 100000))
        if sc_ref < best_c5:
            best_c5 = sc_ref
            best_h = h_ref.copy()

    # Phase 9: Final binary polish
    if tleft() > 80:
        h_final_bin = to_binary(best_h, int(target), rng)
        h_fp, c5_fp = binary_greedy_polish(h_final_bin, n, int(target),
                                            time_limit=min(60, tleft() * 0.7),
                                            seed=1234)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
