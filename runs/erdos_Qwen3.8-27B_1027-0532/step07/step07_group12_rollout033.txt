import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_optimize(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass
    
    return best_h, best_c5

        best_j = -1

            if hi_val < 1e-12:

            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[cj] - profiles[ci]
                
                # Binary search for optimal delta
                lo = 0.0
                hi_d = delta_max
                for _ in range(18):

                    vals = corr + mid_d * g
                    f_mid = float(np.max(vals))
                    if f_mid > cur_max + 1e-14:
                        hi_d = mid_d
                    else:
                        lo = mid_d
                
                delta_star = lo
                if delta_star < 1e-10:
                    continue
                
                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve + 1e-15:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                    cj = cand_pos[int(j)]
                    g = profiles[cj] - profiles[ci]

Wait, there's a problem with the incremental update. After we change h, the profiles become stale because they depend on h. The profile for point p is:
profile[p, m] = (1-h[p+k]) - h[p-k] (with appropriate bounds)

After changing h[i] by -δ and h[j] by +δ, the profile for point p changes if:
- p+k == i or p+k == j (affects the (1-h[p+k]) term)
- p-k == i or p-k == j (affects the h[p-k] term)

So the profile update is:
For point p where p+k_{m} == i: profile[p, m] += δ (since h[i] decreased by δ, 1-h[i] increased by δ)
For point p where p+k_{m} == j: profile[p, m] -= δ (since h[j] increased by δ, 1-h[j] decreased by δ)
For point p where p-k_{m} == i: profile[p, m] += δ (since h[i] decreased by δ)
For point p where p-k_{m} == j: profile[p, m] -= δ (since h[j] increased by δ)

This is getting complicated. For simplicity, let me just recompute the profiles after each update. The cost is O(nc * n) which is small compared to the O(nc^2 * n) cost of the pair search.

But the binary search is the expensive part (18 iterations × O(n_corr) per pair). I can't easily vectorize the binary search because each pair has a different optimal delta.

However, I can vectorize the "feasibility check" part: for a given delta, which pairs are feasible (i.e., don't increase the max)?

Actually, let me think about this differently. The binary search finds the largest δ such that max_k (corr[k] + δ * D[k]) <= cur_max.

This is equivalent to finding the largest δ such that for all k:
corr[k] + δ * D[k] <= cur_max
i.e., δ * D[k] <= cur_max - corr[k] for all k
i.e., δ <= (cur_max - corr[k]) / D[k] for all k where D[k] > 0

So the optimal δ is:
δ* = min over k where D[k] > 0 of (cur_max - corr[k]) / D[k]
(also bounded by delta_max)

This is MUCH simpler than binary search! I can compute this directly:
delta_star = min(delta_max, min_{k: D[k]>0} (cur_max - corr[k]) / D[k])

If there's no k with D[k] > 0, then delta_star = delta_max (any delta works).

Wait, but this gives the LARGEST delta that keeps all correlations <= cur_max. But we also need to make sure the resulting point is still feasible (0 <= h <= 1), which is handled by delta_max.

Let me verify: if D[k] > 0, increasing δ increases corr[k], so we need corr[k] + δ*D[k] <= cur_max, giving δ <= (cur_max - corr[k])/D[k].
If D[k] <= 0, increasing δ doesn't increase corr[k], so no constraint from this k.

So: δ* = min(delta_max, min_{k: D[k] > 0} (cur_max - corr[k]) / D[k])

If the set {k: D[k] > 0} is empty, δ* = delta_max.

This is a HUGE optimization! Instead of 18 iterations of binary search (each O(n_corr)), we do a single O(n_corr) computation per pair.

def pair_cd_vectorized(h_init, n, target, time_limit, n_cand=30, max_sweeps=100000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy().astype(np.float64)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)
    mid = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                break
        
        # Gradient for candidate selection
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]
        
        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]
        all_cands = np.unique(np.concatenate([high_set, low_set]))
        nc = len(all_cands)
        if nc < 2:
            continue
        
        # Compute profiles (nc × n_corr)
        profiles = np.zeros((nc, n_corr))
        for ci in range(nc):
            p = int(all_cands[ci])
            ks = np.arange(-(n-1), n)
            idx1 = p + ks
            idx2 = p - ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1 + (n-1)] = (1.0 - h[idx1[valid1]])
            if np.any(valid2):
                profiles[ci, valid2 + (n-1)] -= h[idx2[valid2]]
        
        # Map from index to profile position
        cand_pos = np.full(n, -1, dtype=int)
        for p in range(nc):
            cand_pos[int(all_cands[p])] = p
        
        # Get profile indices for high and low sets
        high_prof_idx = cand_pos[high_set]
        low_prof_idx = cand_pos[low_set]
        
        # D[i,j] = profiles[low_prof_idx[j]] - profiles[high_prof_idx[i]]
        # Shape: (n_high, n_low, n_corr)
        # This might be large: 30 * 30 * 1200 = 1,080,000 for n=600
        # That's fine memory-wise (about 8 MB)
        
        n_high = len(high_set)
        n_low = len(low_set)
        
        # delta_max for each pair
        delta_maxes = np.minimum(h[high_set][:, np.newaxis], 1.0 - h[low_set][np.newaxis, :])
        # Shape: (n_high, n_low)
        
        # Filter out impossible pairs
        valid_pairs = delta_maxes > 1e-12
        if not np.any(valid_pairs):
            continue
        
        # Compute D for all pairs
        # D[i,j,:] = profiles[low_prof_idx[j],:] - profiles[high_prof_idx[i],:]
        D = profiles[low_prof_idx[np.newaxis, :, :]] - profiles[high_prof_idx[:, np.newaxis, :]]
        # Shape: (n_high, n_low, n_corr)
        
        # For each pair, find delta_star = min(delta_max, min_{k: D>0} (cur_max - corr)/D)
        # (cur_max - corr) is the same for all pairs, shape (n_corr,)
        slack = cur_max - corr  # shape (n_corr,), non-negative by definition of cur_max
        
        # Constraint: for D[i,j,k] > 0, we need delta <= slack[k] / D[i,j,k]
        # delta_star[i,j] = min(delta_maxes[i,j], min_k where D[i,j,k]>0 of slack[k]/D[i,j,k])
        
        # Where D > 0:
        D_pos = D > 0
        # slack / D where D > 0 (set to inf where D <= 0)
        with np.errstate(divide='ignore', invalid='ignore'):
            ratios = np.where(D_pos, slack[np.newaxis, np.newaxis, :] / D, np.inf)
        
        # min over k for each (i,j)
        min_ratio = np.min(ratios, axis=2)  # shape (n_high, n_low)
        
        # delta_star = min(delta_max, min_ratio), but only for valid pairs
        delta_star = np.minimum(delta_maxes, min_ratio)
        delta_star = np.where(valid_pairs, delta_star, 0.0)
        
        # For invalid cases (all D <= 0), min_ratio is inf, so delta_star = delta_max
        # That's correct: if D <= 0 everywhere, increasing delta doesn't increase any corr
        
        # Compute the improvement for each pair
        # new_max[i,j] = max_k (corr[k] + delta_star[i,j] * D[i,j,k])
        new_max = np.max(corr[np.newaxis, np.newaxis, :] + delta_star[:, :, np.newaxis] * D, axis=2)
        improvements = cur_max - new_max  # shape (n_high, n_low)
        
        # Find the best pair
        improvements = np.where(valid_pairs, improvements, -np.inf)
        flat_idx = np.argmax(improvements)
        i_best = flat_idx // n_low
        j_best = flat_idx % n_low
        
        best_improve = float(improvements[i_best, j_best])
        
        if best_improve < 1e-13:
            continue
        
        best_i = int(high_set[i_best])
        best_j = int(low_set[j_best])
        best_delta = float(delta_star[i_best, j_best])
        
        h[best_i] -= best_delta
        h[best_j] += best_delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = (c_max + log_sum / alpha) * dx
    obj += penalty * (np.sum(x) - target) ** 2
    return obj

def lse_gradient(x, alpha, target, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    total_e = np.sum(e)
    weights = e / total_e
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    active_mask = weights > 1e-12
    if np.any(active_mask):
        active_idx = np.where(active_mask)[0]
        for m in active_idx:
            k = int(m) - mid
            w = weights[int(m)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += w * (1.0 - x[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= w * x[idxs[v2] + k]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgs_optimize(h_start, n, target, alphas, maxiter, time_limit, penalty=100.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for pen_mult in [1.0, 10.0, 100.0]:
            if time.time() - start > time_limit:
                break
            pen = penalty * pen_mult
            try:
                res = minimize(lse_objective, h_cur, args=(alpha, target, pen),
                              jac=lse_gradient, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_cur = h_new
                if abs(np.sum(h_cur) - target) < 1e-8:
                    break
            except Exception:
                pass
    return best_h, best_c5

        D_pos = D > 0
        with np.errstate(divide='ignore', invalid='ignore'):
            ratios = np.where(D_pos, slack[np.newaxis, np.newaxis, :] / D, np.inf)
        min_ratio = np.min(ratios, axis=2)
        delta_star = np.minimum(delta_maxes, min_ratio)
        delta_star = np.where(valid_pairs, delta_star, 0.0)

        new_max = np.max(corr[np.newaxis, np.newaxis, :] + delta_star[:, :, np.newaxis] * D, axis=2)
        improvements = cur_max - new_max
        improvements = np.where(valid_pairs, improvements, -np.inf)

        flat_idx = np.argmax(improvements)
        i_best = flat_idx // n_low
        j_best = flat_idx % n_low
        best_improve = float(improvements[i_best, j_best])

        if best_improve < 1e-13:
            continue

        best_i = int(high_set[i_best])
        best_j = int(low_set[j_best])
        best_delta = float(delta_star[i_best, j_best])

        h[best_i] -= best_delta
        h[best_j] += best_delta

        if sweep % 200 == 0:
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, target)
    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 35.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Vectorized pair CD at initial resolution (main workhorse)
    if tleft() > 120:
        h1, s1 = pair_cd_vectorized(best_h, n, target, time_limit=min(110, tleft() * 0.55),
                                     n_cand=30, multi_shift=1)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: L-BFGS refinement
    if tleft() > 100:
        hp, sc = lbfgs_optimize(best_h, n, target, alphas=[500, 5000, 50000],
                                maxiter=300, time_limit=min(80, tleft() * 0.5), penalty=500.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 3: Multi-shift pair CD
    if tleft() > 100:
        h3, s3 = pair_cd_vectorized(best_h, n, target, time_limit=min(90, tleft() * 0.5),
                                     n_cand=35, multi_shift=3)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Upsample to 2x resolution
    n2 = n * 2
    if n2 <= 1600 and tleft() > 200:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        # Quick L-BFGS at new resolution
        if tleft() > 150:
            hp2, sc2 = lbfgs_optimize(h2, n2, target2, alphas=[200, 2000, 20000],
                                      maxiter=250, time_limit=min(70, tleft() * 0.3), penalty=500.0)
            if sc2 < compute_c5(h2):
                h2 = hp2
        # Vectorized pair CD at new resolution
        if tleft() > 120:
            h2b, s2b = pair_cd_vectorized(h2, n2, target2, time_limit=min(130, tleft() * 0.5),
                                           n_cand=30, multi_shift=1)
            if s2b < compute_c5(h2):
                h2 = h2b
        s2_final = compute_c5(h2)
        if s2_final < best_score:
            best_score = s2_final
            best_h = h2
            n = n2
            target = target2

    # Phase 5: Another round of pair CD (multi-shift)
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h5, s5 = pair_cd_vectorized(best_h, n_use, target_use, time_limit=min(90, tleft() * 0.5),
                                     n_cand=40, multi_shift=5)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Try another upsample
    n3 = len(best_h) * 2
    if n3 <= 3200 and tleft() > 250:
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        if tleft() > 150:
            hp3, sc3 = lbfgs_optimize(h3, n3, target3, alphas=[500, 5000, 50000],
                                      maxiter=200, time_limit=min(70, tleft() * 0.3), penalty=500.0)
            if sc3 < compute_c5(h3):
                h3 = hp3
        if tleft() > 150:
            h3b, s3b = pair_cd_vectorized(h3, n3, target3, time_limit=min(140, tleft() * 0.5),
                                           n_cand=30, multi_shift=1)
            if s3b < compute_c5(h3):
                h3 = h3b
        if tleft() > 80:
            h3c, s3c = pair_cd_vectorized(h3, n3, target3, time_limit=min(70, tleft() * 0.4),
                                           n_cand=35, multi_shift=3)
            if s3c < compute_c5(h3):
                h3 = h3c
        s3_final = compute_c5(h3)
        if s3_final < best_score:
            best_score = s3_final
            best_h = h3

    # Phase 7: Final L-BFGS polishing with very high alpha
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = lbfgs_optimize(best_h, n_use, target_use,
                                 alphas=[200000, 1000000], maxiter=200,
                                 time_limit=min(50, tleft() * 0.6), penalty=10000.0)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 8: Final vectorized pair CD
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h8, s8 = pair_cd_vectorized(best_h, n_use, target_use, time_limit=min(30, tleft() * 0.7),
                                     n_cand=25, multi_shift=1)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    # Phase 9: Random perturbation + quick recovery
    if tleft() > 80:
        rng = np.random.RandomState(seed)
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert = best_h + rng.randn(n_use) * 0.01
        h_pert = project_feasible(h_pert, target_use)
        hp, sc = lbfgs_optimize(h_pert, n_use, target_use,
                                 alphas=[5000, 50000], maxiter=150,
                                 time_limit=min(40, tleft() * 0.5), penalty=500.0)
        if sc < best_score and tleft() > 30:
            h9, s9 = pair_cd_vectorized(hp, n_use, target_use, time_limit=min(25, tleft() * 0.5),
                                         n_cand=25, multi_shift=1)
            if s9 < best_score:
                best_score = s9
                best_h = h9
        elif sc < best_score:
            best_score = sc
            best_h = hp

    # Final
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
