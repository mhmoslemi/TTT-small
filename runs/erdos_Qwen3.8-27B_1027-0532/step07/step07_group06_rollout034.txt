```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_C5(h, k_star):
    """Compute subgradient of C5 with respect to h when k_star is the active constraint."""
    n = len(h)
    idxs = np.arange(n)
    grad = np.zeros(n)
    # corr_k = sum_i h[i]*(1-h[i+k]) for valid i
    # d/dh[p] corr_k:
    #   If p+k is valid: contribution from i=p: (1-h[p+k])
    #   If p-k is valid: contribution from i=p-k: -h[p-k]
    v1 = (idxs + k_star >= 0) & (idxs + k_star < n)
    v2 = (idxs - k_star >= 0) & (idxs - k_star < n)
    grad[v1] += (1.0 - h[idxs[v1] + k_star])
    grad[v2] -= h[idxs[v2] - k_star]
    # Remove mean to handle the equality constraint
    grad -= np.mean(grad)
    return grad

def multi_shift_grad(h, corr, top_k):
    """Compute weighted gradient using top-k active shifts."""
    n = len(h)
    mid = n - 1
    top_indices = np.argsort(corr)[-top_k:]
    weights = corr[top_indices] ** 4  # Sharpen to focus on most active
    wsum = np.sum(weights)
    if wsum < 1e-15:
        weights = np.ones(top_k) / top_k
    else:
        weights = weights / wsum
    
    idxs = np.arange(n)
    grad = np.zeros(n)
    for s_idx, w in zip(top_indices, weights):
        k = int(s_idx) - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        grad[v2] -= w * h[idxs[v2] - k]
    grad -= np.mean(grad)
    return grad

def subgradient_optimize(h_init, n, target, time_limit, start_time, 
                         max_iter=50000, top_k=3, lr_init=0.01, cooling=0.999):
    """Subgradient descent with Polyak-like step size."""
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = lr_init
    no_improve = 0
    best_lr = lr
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0/n)
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
            best_lr = lr
        else:
            no_improve += 1
        
        if no_improve > 2000:
            break
        
        # Compute multi-shift gradient
        grad = multi_shift_grad(h, corr, top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        # Adaptive step size
        step = lr / grad_norm
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        new_c5 = compute_c5(h_new)
        
        if new_c5 < c5_val - 1e-15:
            h = h_new
            cur_c5 = new_c5
        else:
            # Try smaller step
            step = lr / (2 * grad_norm)
            h_new = h - step * grad
            h_new = project_feasible(h_new, target)
            new_c5 = compute_c5(h_new)
            if new_c5 < c5_val - 1e-15:
                h = h_new
                cur_c5 = new_c5
                lr *= 0.9
            else:
                lr *= 0.8
                if lr < 1e-8:
                    break
        
        # Cool down
        lr *= cooling + 0.001 * (c5_val - best_c5) * 100  # Increase lr when far from best
    
    return best_h, best_c5
```

```python
def project_box_eq(h, target):
    """Project h onto {x : 0≤x≤1, sum(x)=target}."""
    h = h.copy()
    if target <= 0 or target >= len(h):
        return np.clip(h, 0, 1)
    
    def proj(lo, hi):
        # Find gamma such that sum(clip(h - gamma, lo, hi)) = target
        for _ in range(60):
            mid = 0.5 * (lo + hi)
            val = np.sum(np.clip(h - mid, 0.0, 1.0))
            if val > target:
                lo = mid
            else:
                hi = mid
        return 0.5 * (lo + hi)
    
    gamma = proj(-1.0, 2.0)
    return np.clip(h - gamma, 0.0, 1.0)
```

```python
def compute_grad_lse(h, corr, alpha, min_weight=1e-10):
    n = len(h)
    mid = n - 1
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    total = np.sum(weights)
    if total < 1e-300:
        return np.zeros(n)
    weights = weights / total
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < min_weight:
            continue
        k = k_idx - mid
        if k >= 0:
            # grad[p] += w * (1-h[p-k]) for p in [k, n-1]
            grad[k:] += w * (1.0 - h[:n-k])
            # grad[p] -= w * h[p+k] for p in [0, n-1-k]
            grad[:n-k] -= w * h[k:]
        else:
            kk = -k
            # grad[p] += w * (1-h[p+kk]) for p in [0, n-1-kk]
            grad[:n-kk] += w * (1.0 - h[kk:])
            # grad[p] -= w * h[p-kk] for p in [kk, n-1]
            grad[kk:] -= w * h[:n-kk]
    
    # Remove mean for the equality constraint
    grad -= np.mean(grad)
    return grad
```

```python
            break
```

```python
def project_box_eq(h, target):
    """Project onto {x: 0<=x<=1, sum(x)=target}."""
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid

    return np.clip(h - gamma, 0.0, 1.0)

    """Compute gradient of LSE given pre-computed weights."""
    mid = n - 1
    grad = np.zeros(n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 100.0
_g_gamma = 0.0
_g_rho = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_box_eq(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    n = len(h)
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    gamma = 0.5 * (lo + hi)
    return np.clip(h - gamma, 0.0, 1.0)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _grad_from_weights(x, weights, n):
    mid = n - 1
    grad = np.zeros(n)
    for k_idx in range(len(weights)):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - mid
        if k >= 0:
            grad[k:] += w * (1.0 - x[:n - k])
            grad[:n - k] -= w * x[k:]
        else:
            kk = -k
            grad[:n - kk] += w * (1.0 - x[kk:])
            grad[kk:] -= w * x[:n - kk]
    grad -= np.mean(grad)
    return grad


def _lse_val_and_weights(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exps = np.exp(shifted)
    total = float(np.sum(exps))
    if total < 1e-300:
        weights = np.zeros(len(corr))
        weights[int(np.argmax(corr))] = 1.0
        lse_val = c_max
    else:
        weights = exps / total
        lse_val = c_max + np.log(total) / alpha
    return lse_val, weights


def _aug_obj(x):
    lse_val, _ = _lse_val_and_weights(x, _g_alpha)
    viol = np.sum(x) - _g_target
    return lse_val + _g_gamma * viol + 0.5 * _g_rho * viol * viol


def _aug_grad(x):
    _, weights = _lse_val_and_weights(x, _g_alpha)
    g = _grad_from_weights(x, weights, _g_n)
    viol = np.sum(x) - _g_target
    g += (_g_gamma + _g_rho * viol)
    return g


def aug_lbfgsb_single(h_init, n, target, alpha, max_outer, inner_iter, time_limit, start_time):
    global _g_n, _g_target, _g_alpha, _g_gamma, _g_rho
    _g_n = n
    _g_target = target
    _g_alpha = alpha
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    gamma = 0.0
    rho = 1.0
    for outer in range(max_outer):
        if time.time() - start_time > time_limit:
            break
        _g_gamma = gamma
        _g_rho = rho
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_aug_obj, h, jac=_aug_grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': inner_iter, 'ftol': 1e-18, 'maxls': 40})
            h = project_box_eq(res.x, target)
        except Exception:
            pass
        viol = np.sum(h) - target
        gamma += rho * viol
        if abs(viol) > 2.0:
            rho *= 4.0
        elif abs(viol) > 0.5:
            rho *= 2.0
        elif abs(viol) < 0.05:
            rho = max(0.5 * rho, 0.1)
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5


def aug_lbfgsb_multi(h_init, n, target, alphas, outer_per_alpha, inner_per_alpha, time_limit, start_time):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        if time.time() - start_time > time_limit:

        h_new, c5_new = aug_lbfgsb_single(h, n, target, alpha, outer_per_alpha, inner_per_alpha, time_limit, start_time)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        h = h_new
    return best_h, best_c5


def _slsqp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def _slsqp_cons(x):
    return np.sum(x) - _g_target


def _slsqp_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_init, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_slsqp_obj, h, args=(alpha,),
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac}],
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_box_eq(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            h = h_new
        except Exception:
            pass
    return best_h, best_c5


def pair_cd(h_init, n, target, time_limit, start_time,
            max_sweeps=50000, n_cand_start=15, n_cand_end=60,
            max_no_improve=500, n_top_shifts=3, bisection_iters=16):
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1
    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > max_no_improve:
            break
        top_shifts = np.argsort(corr)[-n_top_shifts:]
        weights = corr[top_shifts].astype(float)
        wsum = np.sum(weights)
        if wsum < 1e-15:
            weights = np.ones(n_top_shifts) / n_top_shifts
        else:
            weights = weights / wsum
        grad = np.zeros(n)
        for s_idx, w in zip(top_shifts, weights):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h[idxs[v1] - k])
            grad[v2] -= w * h[idxs[v2] + k]
        grad -= np.mean(grad)
        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 3)
        current_ncand = max(current_ncand, 5)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(bisection_iters):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-12:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_box_eq(h, target)
            for p in [best_i, best_j]:
                change[p, :] = 0.0
                pos_mask = (p + shifts >= 0) & (p + shifts < n)
                neg_mask = (p - shifts >= 0) & (p - shifts < n)
                change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
                change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_eq(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_eq(h_out, target_dst)

                                    start_time + min(60, remaining - 10), start_time,
                                    max_steps=20000, base_scale=0.02)
        if sc10 < best_score:
            best_score = sc10
            best_h = h10
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 40:
        rng = np.random.RandomState(seed)
        for ns in [0.02, 0.05]:
            remaining = budget_s - 30.0 - (time.time() - start_time)
            if remaining < 30:
                break
            h_pert = best_h + rng.randn(n0) * ns
            h_pert = project_box_eq(h_pert, target0)
            h_r, sc_r = aug_lbfgsb_multi(h_pert, n0, target0,
                                         [1000, 5000], 4, 200,
                                         start_time + min(30, remaining - 10), start_time)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r
    remaining = budget_s - 30.0 - (time.time() - start_time)
    if remaining > 40:
        h_fin, sc_fin = pair_cd(best_h, n0, target0,
                                start_time + min(50, remaining - 5), start_time,
                                max_sweeps=20000, n_cand_start=40, n_cand_end=90,
                                max_no_improve=150, n_top_shifts=1, bisection_iters=20)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin
    final_h = project_box_eq(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
```
