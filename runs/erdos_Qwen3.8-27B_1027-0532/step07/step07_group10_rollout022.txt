import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np

import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def slp_single_step(h, n, target, t_ub, solver=None):
    dx = 2.0 / n
    mid = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')

    h_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

    for k in range(n):
        if k == 0:
            grad_k = 1.0 - 2.0 * h
        else:
            grad_k = np.zeros(n)
            first = np.arange(0, n - k)
            second = np.arange(k, n)
            grad_k[first] = 1.0 - h[first + k]
            grad_k[second] += -h[second - k]
        rhs_k = t_ub / dx - corr[k + mid] + np.dot(grad_k, h)
        constraints.append(cp.sum(grad_k * h_var) <= rhs_k)

        if k > 0:
            grad_neg = np.zeros(n)
            first = np.arange(k, n)
            second = np.arange(0, n - k)
            grad_neg[first] = 1.0 - h[first - k]
            grad_neg[second] += -h[second + k]
            rhs_neg = t_ub / dx - corr[mid - k] + np.dot(grad_neg, h)
            constraints.append(cp.sum(grad_neg * h_var) <= rhs_neg)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    if solver is None:
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                return h, t_ub, False
    else:
        try:
            prob.solve(solver=solver)
        except Exception:
            return h, t_ub, False

    if h_var.value is None:
        return h, t_ub, False

    h_new = np.clip(h_var.value, 0.0, 1.0)
    h_new = project_feasible(h_new, target)
    t_new = compute_c5(h_new)
    return h_new, t_new, True


def slp_loop(h_init, n, target, time_limit, max_iters=300):
    start = time.time()
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_t = compute_c5(h)
    t_ub = best_t
    no_improve = 0
    damping = 1.0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        if no_improve > 15:
            damping = max(0.2, damping * 0.7)
        elif no_improve == 0:
            damping = min(1.0, damping * 1.2)

        h_raw, t_raw, ok = slp_single_step(h, n, target, t_ub)
        if not ok:
            no_improve += 1
            continue

        h_damped = damping * h_raw + (1.0 - damping) * h
        h_damped = project_feasible(h_damped, target)
        t_damped = compute_c5(h_damped)

        if t_damped < best_t - 1e-12:
            h = h_damped
            best_h = h.copy()
            best_t = t_damped

        else:
            h_try2 = project_feasible(h_raw, target)

            else:

def _smooth_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(c_max + log_sum / _g_alpha)


def _smooth_obj_grad(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -600.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(x)
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-12:
            continue
        k = k_idx - mid
        wp = (idxs + k < n) & (idxs + k >= 0)
        wm = (idxs - k >= 0) & (idxs - k < n)
        grad[wp] += weights[k_idx] * (1.0 - x[idxs[wp] + k])
        grad[wm] -= weights[k_idx] * x[idxs[wm] - k]
    return grad


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, jac=_smooth_obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] = (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=20, n_cand_end=80, max_no_improve=500):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0
    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        top_k = min(5, n_corr)
        top_shifts = np.argsort(corr)[-top_k:]
        k_star = int(top_shifts[-1]) - mid

        weights = np.zeros(n_corr)
        c_top = corr[top_shifts[0]]
        for si in top_shifts:
            weights[si] = np.exp(-20.0 * (c_top - corr[si]))
        sw = np.sum(weights)
        if sw > 0:
            weights /= sw
        else:
            weights[k_star + mid] = 1.0

        grad = np.zeros(n)
        for s_idx in range(n_corr):
            if weights[s_idx] < 1e-12:
                continue
            k = s_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += weights[s_idx] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= weights[s_idx] * h[idxs[v2] + k]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(max_sweeps, 1), 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        n_h = len(high_set)
        n_l = len(low_set)
        for ii in range(n_h):
            i = int(high_set[ii])
            for jj in range(n_l):
                j = int(low_set[jj])
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue

                g = change[j] - change[i]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-15
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-13:
                    continue

                approx_max = np.max(corr + delta_star * g)
                if approx_max < best_trial_c5 / dx - 1e-16:
                    best_trial_c5 = float(approx_max) * dx
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

    return best_h, best_c5


def gradient_descent_active(h_init, n, target, time_limit, max_steps=100000,

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def polyak_subgrad(h_init, n, target, time_limit, max_steps=200000):
    start = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    f_lower = best_c5 * 0.95
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 10000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        active = np.where(corr >= np.max(corr) - 1e-10)[0]
        grad = np.zeros(n)
        for si in active:
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= len(active)
        grad -= np.mean(grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        gap = max(c5_val - f_lower, 1e-10)
        lr = min(gap / (gnorm * gnorm), 0.1)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < c5_val:
            h = h_try
            if sc < f_lower + 1e-4:
                f_lower = max(sc - 0.01, 0.3)
        elif sc < c5_val + lr * 0.01:
            h = h_try

    return best_h, best_c5

        if tleft() > 250:
            h1_slp, sc1_slp = slp_loop(h1, n1, target1,
                                        time_limit=min(50, tleft() * 0.25), max_iters=150)
            if sc1_slp < compute_c5(h1):
                h1 = h1_slp

        # CD at n=800
        if tleft() > 200:
            h1_cd, c5_1cd = fast_pair_cd(h1, n1, target1,
                                          time_limit=min(150, tleft() * 0.55),
                                          max_sweeps=40000,
                                          n_cand_start=25, n_cand_end=90,
                                          max_no_improve=500)
            if c5_1cd < best_score:
                best_score = c5_1cd
                best_h = h1_cd
                best_n = n1
                best_target = target1
                h1 = h1_cd

        # SLSQP at n=800
        if tleft() > 80:
            h1_pol, sc1_pol = slsqp_round(h1, n1, target1,
                                           [500, 2000, 8000], 1000, 1e-15)
            if sc1_pol < best_score:
                best_score = sc1_pol
                best_h = h1_pol
                best_n = n1
                best_target = target1

        # Polyak at n=800
        if tleft() > 80:
            h1_ps, sc1_ps = polyak_subgrad(best_h, best_n, best_target,
                                            time_limit=min(50, tleft() * 0.4))
            if sc1_ps < best_score:
                best_score = sc1_ps
                best_h = h1_ps
                best_n = n1
                best_target = target1

    # Phase 6: Try n=600 as intermediate if n=800 path was not fully explored
    n2 = 600
    if tleft() > 200 and best_n == n0:
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

        if tleft() > 150:
            h2_slp, sc2_slp = slp_loop(h2, n2, target2,
                                        time_limit=min(30, tleft() * 0.2), max_iters=150)
            if sc2_slp < compute_c5(h2):
                h2 = h2_slp

        if tleft() > 100:
            h2_cd, c5_2cd = fast_pair_cd(h2, n2, target2,
                                          time_limit=min(80, tleft() * 0.5),
                                          max_sweeps=30000,
                                          n_cand_start=20, n_cand_end=80,
                                          max_no_improve=400)
            if c5_2cd < best_score:
                best_score = c5_2cd
                best_h = h2_cd
                best_n = n2
                best_target = target2

        if tleft() > 60:
            h2_pol, sc2_pol = slsqp_round(best_h, best_n, best_target,
                                           [500, 2000, 10000], 800, 1e-15)
            if sc2_pol < best_score:
                best_score = sc2_pol
                best_h = h2_pol

    # Phase 7: Final SLP + SLSQP polishing on best so far
    if tleft() > 80:
        h_fin_slp, sc_fin_slp = slp_loop(best_h, best_n, best_target,
                                          time_limit=min(40, tleft() * 0.4), max_iters=100)
        if sc_fin_slp < best_score:
            best_score = sc_fin_slp
            best_h = h_fin_slp

    if tleft() > 60:
        h_fin_cd, c5_fin_cd = fast_pair_cd(best_h, best_n, best_target,
                                            time_limit=min(50, tleft() * 0.5),
                                            max_sweeps=20000,
                                            n_cand_start=40, n_cand_end=100,
                                            max_no_improve=300)
        if c5_fin_cd < best_score:
            best_score = c5_fin_cd
            best_h = h_fin_cd

    if tleft() > 40:
        h_fin_pol, sc_fin_pol = slsqp_round(best_h, best_n, best_target,
                                             [5000, 20000, 50000], 1500, 1e-15)
        if sc_fin_pol < best_score:
            best_score = sc_fin_pol
            best_h = h_fin_pol

    # Phase 8: Random restarts to escape local optima
    if tleft() > 60:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.08, 0.15]:
            if tleft() < 25:
                break
            h_pert = best_h + rng.randn(best_n) * noise_scale
            h_pert = project_feasible(h_pert, best_target)
            h_ref, sc_ref = slp_loop(h_pert, best_n, best_target,
                                      time_limit=min(20, tleft() * 0.3), max_iters=50)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    # Phase 9: Final SLP squeeze
    if tleft() > 30:
        h_last, sc_last = slp_loop(best_h, best_n, best_target,
                                    time_limit=min(20, tleft() * 0.5), max_iters=50)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    # Phase 10: Final CD squeeze
    if tleft() > 20:
        h_last_cd, c5_last = fast_pair_cd(best_h, best_n, best_target,
                                           time_limit=min(12, tleft() * 0.5),
                                           max_sweeps=5000,
                                           n_cand_start=50, n_cand_end=100,
                                           max_no_improve=100)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_last_cd

    final_h = project_feasible(best_h, best_n / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, best_n
