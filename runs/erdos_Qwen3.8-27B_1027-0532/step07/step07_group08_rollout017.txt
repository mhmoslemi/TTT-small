from scipy.optimize import minimize
import time

# Global state for SLSQP callbacks
_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _eq_con(x):
    return np.sum(x) - _g_target

def _eq_jac(x):
    return np.ones(_g_n)

def slsqp_polish(h0, n, target, alphas, maxiter, deadline=None):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac}]
    h_cur = h0.copy()
    best = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            r = minimize(_smooth_obj, h_cur, args=(a,), method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(r.x, target)
            sc = compute_c5(h_new)
            if sc < best:
                best = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best

def compute_change_matrix(h, n):

    change = np.zeros((n, n_corr))

        nm = (p - shifts >= 0) & (p - shifts < n)
        change[p, pm] += (1.0 - h[p + shifts[pm]])
        change[p, nm] -= h[p - shifts[nm]]
    return change
