    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_vectorized_cd(h_init, n, target, time_limit, max_sweeps=20000,
                       n_cand_start=25, n_cand_end=45, max_no_improve=350,
                       n_bisect=6, active_tol_frac=5e-8):
    start_t = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        if no_improve > max_no_improve:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_corr = np.max(corr)
        tol = max(1e-12, active_tol_frac * max_corr)
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(len(active), 1)
        grad -= np.mean(grad)

        progress = min(float(sweep) / max(1, max_sweeps), 1.0)
        ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        ncand = min(ncand, n // 4)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:ncand]
        high_set = sorted_idx[-ncand:]

        all_cands = np.unique(np.concatenate([low_set, high_set]))
        n_cands = len(all_cands)
        cand_map = np.full(n, -1, dtype=int)
        for r in range(n_cands):
            cand_map[all_cands[r]] = r

        hi_rows = np.array([cand_map[int(high_set[ii])] for ii in range(ncand)])
        lo_rows = np.array([cand_map[int(low_set[jj])] for jj in range(ncand)])

        pair_count = ncand * ncand
        G = change[all_cands, :]

        pairs_i = np.repeat(hi_rows, ncand)
        pairs_j = np.tile(lo_rows, ncand)

        h_hi = h[all_cands[pairs_i]]
        h_lo = h[all_cands[pairs_j]]

        delta_max = np.minimum(h_hi, 1.0 - h_lo)
        delta_min = np.maximum(h_hi - 1.0, -h_lo)

        valid = (delta_max - delta_min) > 1e-12
        n_valid = valid.sum()
        if n_valid == 0:
            continue

        G_diff = G[pairs_j] - G[pairs_i]
        Gd = G_diff[valid]
        a_d = delta_min[valid].copy()
        b_d = delta_max[valid].copy()

        batch_size = max(200, min(n_valid, int(2e6 / n_corr)))
        best_trial = c5_val
        best_rel_idx = -1
        best_delta_star = 0.0

        for batch_start in range(0, n_valid, batch_size):
            batch_end = min(batch_start + batch_size, n_valid)
            a_b = a_d[batch_start:batch_end].copy()
            b_b = b_d[batch_start:batch_end].copy()
            Gb = Gd[batch_start:batch_end]

            for _ in range(n_bisect):
                mid_d = 0.5 * (a_b + b_b)
                vals = corr[None, :] + mid_d[:, None] * Gb
                f_mid = np.max(vals, axis=1)
                active_mask = (vals >= f_mid[:, None] - 1e-14)
                G_clipped = np.where(active_mask, Gb, -np.inf)
                f_prime = np.max(G_clipped, axis=1)
                inc = f_prime > 1e-12
                b_b[inc] = mid_d[inc]
                a_b[~inc] = mid_d[~inc]

            delta_star = 0.5 * (a_b + b_b)
            approx_max = np.max(corr[None, :] + delta_star[:, None] * Gb, axis=1)

            local_best = np.argmin(approx_max)
            if approx_max[local_best] < best_trial:
                best_trial = approx_max[local_best]
                best_rel_idx = batch_start + local_best
                best_delta_star = float(delta_star[local_best])

        if best_rel_idx >= 0:
            valid_indices = np.where(valid)[0]
            pair_idx = int(valid_indices[best_rel_idx])
            i = int(high_set[pair_idx // ncand])
            j = int(low_set[pair_idx % ncand])
            delta = best_delta_star

            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            update_change_row(change, i, h, n, shifts)
            update_change_row(change, j, h, n, shifts)

    return best_h, best_c5


def block_transfer_cd(h_init, n, target, time_limit, max_steps=15000,
                      block_sizes=(2, 3, 4, 6, 8, 12)):
    start_t = time.time()
    rng = np.random.RandomState(1234)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    mid = n - 1
    idxs = np.arange(n)

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    change = compute_change_matrix(h, n)

    for step in range(max_steps):
        if time.time() - start_t > time_limit:
            break
        if no_improve > 1500:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr))

        if c5_val * dx < best_c5 - 1e-14:
            best_c5 = c5_val * dx
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        bs = int(block_sizes[step % len(block_sizes)])
        if bs >= n:
            bs = n // 4
        n_pos = n - bs + 1
        if n_pos <= 1:
            continue

        n_search = min(80, n_pos)
        offsets = rng.permutation(n_pos)[:n_search]

        block_grads = np.array([grad[sp:sp + bs].sum() for sp in offsets])
        src_local = int(np.argmin(block_grads))
        dst_local = int(np.argmax(block_grads))

        if src_local == dst_local:
            continue

        src_start = int(offsets[src_local])
        dst_start = int(offsets[dst_local])

        src_block = h[src_start:src_start + bs]
        dst_block = h[dst_start:dst_start + bs]

        delta_max = min(src_block.min(), (1.0 - dst_block).min())
        delta_min = max(-src_block.min(), dst_block.min() - 1.0)

        if delta_max - delta_min < 1e-12:
            continue

        g = np.zeros(n_corr)
        for t in range(bs):
            g += change[dst_start + t, :] - change[src_start + t, :]

        a_d = delta_min
        b_d = delta_max
        for _ in range(8):
            mid_d = 0.5 * (a_d + b_d)
            vals = corr + mid_d * g
            f_mid = np.max(vals)
            active_mask = vals >= f_mid - 1e-14
            if active_mask.any():
                f_prime = np.max(g[active_mask])
            else:
                f_prime = 0.0
            if f_prime > 1e-12:
                b_d = mid_d
            else:
                a_d = mid_d

        delta_star = 0.5 * (a_d + b_d)
        if abs(delta_star) < 1e-12:
            continue

        approx_max = np.max(corr + delta_star * g)
        if approx_max < c5_val - 1e-12:
            h[src_start:src_start + bs] -= delta_star
            h[dst_start:dst_start + bs] += delta_star
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
            for t in range(bs):
                update_change_row(change, src_start + t, h, n, shifts)
                update_change_row(change, dst_start + t, h, n, shifts)

    return best_h, best_c5


def subgradient_refine(h_init, n, target, time_limit, max_iter=20000, lr_start=0.003):
    start_t = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        if no_improve > 2000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        max_corr = np.max(corr)
        tol = max(1e-10, 1e-8 * max_corr)
        active = np.where(corr >= max_corr - tol)[0]

        grad = np.zeros(n)
        for k_idx in active:
            k = int(k_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad /= max(len(active), 1)
        grad -= np.mean(grad)

        progress = min(it / max_iter, 1.0)
        lr = lr_start * (1.0 - 0.9 * progress) ** 2

        h_new = project_feasible(h - lr * grad, target)
        new_c5 = compute_c5(h_new)
        if new_c5 < c5_val - 1e-15:
            h = h_new

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 50:
        h_pol = slsqp_round(best_h, n0, target0, [200, 1000, 5000], 400, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    n1 = 800
    target1 = n1 / 2.0
    h1 = interpolate_to(best_h, n1, target1)
    sc1 = compute_c5(h1)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1
    n_cur = n1
    target_cur = target1

    if tleft() > 250:
        h_cd, c5_cd = fast_vectorized_cd(h1, n_cur, target_cur,
                                         time_limit=min(200, tleft() * 0.55),
                                         max_sweeps=20000,
                                         n_cand_start=25, n_cand_end=45,
                                         max_no_improve=400, n_bisect=6)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n_cur, target_cur,
                            [500, 2000, 10000], 600, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        h_cd2, c5_cd2 = fast_vectorized_cd(best_h, n_cur, target_cur,
                                           time_limit=min(170, tleft() * 0.6),
                                           max_sweeps=18000,
                                           n_cand_start=30, n_cand_end=50,
                                           max_no_improve=350, n_bisect=7)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    if tleft() > 100:
        h_bm, c5_bm = block_transfer_cd(best_h, n_cur, target_cur,
                                        time_limit=min(80, tleft() * 0.55),
                                        max_steps=12000)
        if c5_bm < best_score:
            best_score = c5_bm
            best_h = h_bm

    if tleft() > 80:
        h_pol2 = slsqp_round(best_h, n_cur, target_cur,
                             [2000, 10000, 30000], 800, 1e-15)
        sc2 = compute_c5(h_pol2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2

    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
            n_cur = n2
            target_cur = target2
        else:
            n_cur = n2
            target_cur = target2
            best_h = h2

        if tleft() > 180:
            h_cd3, c5_cd3 = fast_vectorized_cd(best_h, n_cur, target_cur,
                                               time_limit=min(150, tleft() * 0.6),
                                               max_sweeps=12000,
                                               n_cand_start=20, n_cand_end=40,
                                               max_no_improve=300, n_bisect=5)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

        if tleft() > 80:
            h_cd4, c5_cd4 = fast_vectorized_cd(best_h, n_cur, target_cur,
                                               time_limit=min(60, tleft() * 0.7),
                                               max_sweeps=8000,
                                               n_cand_start=35, n_cand_end=55,
                                               max_no_improve=250, n_bisect=6)
            if c5_cd4 < best_score:
                best_score = c5_cd4
                best_h = h_cd4

    if tleft() > 60:
        h_sg, c5_sg = subgradient_refine(best_h, n_cur, target_cur,
                                         time_limit=min(40, tleft() * 0.6),
                                         max_iter=10000, lr_start=0.002)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.06, 0.01]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, target_cur)
            h_ref = slsqp_round(h_pert, n_cur, target_cur,
                                [500, 2000, 8000], 400, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 60:
        h_cd_final, c5_final = fast_vectorized_cd(best_h, n_cur, target_cur,
                                                  time_limit=min(45, tleft() * 0.6),
                                                  max_sweeps=6000,
                                                  n_cand_start=35, n_cand_end=60,
                                                  max_no_improve=200, n_bisect=7)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 30:
        h_final = slsqp_round(best_h, n_cur, target_cur,
                              [5000, 20000], 400, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
