def pair_cd(h_init, n, target, time_limit, max_sweeps=50000,

            return float(np.max(corr))

        # Don't constrain the sum of the block (let it be free)

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0
_g_n = 0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            h = np.clip(h + diff / n, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def time_left(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)


def build_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, h, p, shifts, n):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]


def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)


def _slsqp_cons(x):
    return np.sum(x) - _g_target


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _g_alpha, _g_target, _g_n
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score

        h_cd, c5_cd = pair_cd(h_cur, n_cur, target_cur,
                              time_limit=min(160, tl * 0.75),
                              max_sweeps=100000,
                              n_cand_start=50, n_cand_end=160,
                              top_k_eval=120, n_shifts_grad=5,
                              max_no_improve=1000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
            h_cur = h_cd

    tl = time_left(start_time, budget_s)
    if tl > 60:
        h_pol, sc = slsqp_optimize(h_cur, n_cur, target_cur,
                                   [2000, 8000, 30000], 800, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
            h_cur = h_pol

    tl = time_left(start_time, budget_s)
    if tl > 100:
        h_cd2, c5_cd2 = pair_cd(h_cur, n_cur, target_cur,
                                time_limit=min(90, tl * 0.8),
                                max_sweeps=80000,
                                n_cand_start=70, n_cand_end=180,
                                top_k_eval=150, n_shifts_grad=6,
                                max_no_improve=800)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
            h_cur = h_cd2

    # Multi-shift CD variant
    tl = time_left(start_time, budget_s)
    if tl > 80:
        h_ms, c5_ms = multi_shift_cd(h_cur, n_cur, target_cur,
                                     time_limit=min(70, tl * 0.8),
                                     max_sweeps=60000, n_cand=100,
                                     top_k_eval=120, n_shifts=6,
                                     max_no_improve=600)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms
            h_cur = h_ms

    # Upsample to n=1200
    tl = time_left(start_time, budget_s)
    if tl > 150:
        n_cur = 1200
        target_cur = n_cur / 2.0
        h_cur = interpolate_to(h_cur, n_cur, target_cur)

        tl = time_left(start_time, budget_s)
        if tl > 100:
            h_cd3, c5_cd3 = pair_cd(h_cur, n_cur, target_cur,
                                    time_limit=min(90, tl * 0.8),
                                    max_sweeps=60000,
                                    n_cand_start=60, n_cand_end=140,
                                    top_k_eval=120, n_shifts_grad=5,
                                    max_no_improve=700)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3
                h_cur = h_cd3

        tl = time_left(start_time, budget_s)
        if tl > 50:
            h_pol3, sc3 = slsqp_optimize(h_cur, n_cur, target_cur,
                                         [5000, 20000, 80000], 600, 1e-14)
            if sc3 < best_score:
                best_score = sc3
                best_h = h_pol3
                h_cur = h_pol3

        tl = time_left(start_time, budget_s)
        if tl > 60:
            h_cd4, c5_cd4 = pair_cd(h_cur, n_cur, target_cur,
                                    time_limit=min(55, tl * 0.8),
                                    max_sweeps=40000,
                                    n_cand_start=80, n_cand_end=180,
                                    top_k_eval=150, n_shifts_grad=6,
                                    max_no_improve=500)
            if c5_cd4 < best_score:
                best_score = c5_cd4
                best_h = h_cd4
                h_cur = h_cd4

    # Sinusoidal refinement
    tl = time_left(start_time, budget_s)
    if tl > 50:
        h_sin, c5_sin = sinusoidal_refine(h_cur, n_cur, target_cur,
                                          time_limit=min(40, tl * 0.7),
                                          max_steps=8000)
        if c5_sin < best_score:
            best_score = c5_sin
            best_h = h_sin
            h_cur = h_sin

    # Try higher resolution if time allows
    tl = time_left(start_time, budget_s)
    if tl > 120:
        n_cur = 1600
        target_cur = n_cur / 2.0
        h_cur = interpolate_to(h_cur, n_cur, target_cur)

        tl = time_left(start_time, budget_s)
        if tl > 80:
            h_cd5, c5_cd5 = pair_cd(h_cur, n_cur, target_cur,
                                    time_limit=min(70, tl * 0.8),
                                    max_sweeps=40000,
                                    n_cand_start=60, n_cand_end=150,
                                    top_k_eval=120, n_shifts_grad=5,

            if c5_cd5 < best_score:
                best_score = c5_cd5
                best_h = h_cd5

        tl = time_left(start_time, budget_s)
        if tl > 40:
            h_pol5, sc5 = slsqp_optimize(h_cur, n_cur, target_cur,
                                         [5000, 20000], 400, 1e-14)
            if sc5 < best_score:
                best_score = sc5
                best_h = h_pol5
                h_cur = h_pol5

    # Final pair CD
    tl = time_left(start_time, budget_s)
    if tl > 40:
        n_fin = len(h_cur)
        t_fin = n_fin / 2.0
        h_cd6, c5_cd6 = pair_cd(h_cur, n_fin, t_fin,
                                time_limit=min(35, tl * 0.8),
                                max_sweeps=30000,
                                n_cand_start=80, n_cand_end=200,
                                top_k_eval=200, n_shifts_grad=6,
                                max_no_improve=300)
        if c5_cd6 < best_score:
            best_score = c5_cd6
            best_h = h_cd6

    # Final SLSQP
    tl = time_left(start_time, budget_s)
    if tl > 30:
        n_fin = len(best_h)
        t_fin = n_fin / 2.0
        h_pol6, sc6 = slsqp_optimize(best_h, n_fin, t_fin,
                                     [10000, 50000, 200000], 400, 1e-14)
        if sc6 < best_score:
            best_score = sc6
            best_h = h_pol6

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)
    return final_h, final_score, len(final_h)
